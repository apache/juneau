34  /**
35   * JMH micro-benchmark asserting the observability hot path in {@link RestOpInvoker}
36   * is zero-allocation when both {@link MetricsRecorder} and {@link TracerHook} resolve to their
37   * NoOp singletons.
38   *
39   * <p>
40   * Mirrors the observability block inside {@code RestOpInvoker.invoke(opSession, observable=true)}:
41   * </p>
42   * <pre class='bjava'>
43   * 	MetricsRecorder recorder = NoOpMetricsRecorder.INSTANCE;
44   * 	TracerHook tracer = NoOpTracerHook.INSTANCE;
45   * 	long startNanos = System.nanoTime();
46   * 	Scope scope = tracer.startSpan(<jk>null</jk>);           <jc>// NoOp ignores request arg</jc>
47   * 	<jk>try</jk> {
48   * 		scope.setStatusCode(200);
49   * 	} <jk>finally</jk> {
50   * 		scope.close();
51   * 		var elapsed = Duration.ofNanos(System.nanoTime() - startNanos);
52   * 		recorder.record(OP_NAME, HTTP_METHOD, URI_TEMPLATE, 200, elapsed, <jk>null</jk>);
53   * 	}
54   * </pre>
55   *
56   * <p>
57   * Expected result: {@code gc.alloc.rate.norm} ≤ 8 bytes/op (zero under typical HotSpot
58   * escape-analysis; the 8-byte epsilon absorbs JIT-on-warmup noise).
59   *
60   * <h5 class='section'>Running:</h5>
61   * <pre class='bconsole'>
62   * 	$ python3 scripts/microbench.py observability
63   *
64   * 	<jc># Or directly:</jc>
65   * 	$ mvn -pl juneau-integration-tests -Pmicrobench test-compile exec:java \
66   * 	      -Dexec.mainClass=org.apache.juneau.marshall.microbench.observability.ObservabilityNoopBenchmark
67   * </pre>
68   *
69   * @since 10.0.0
70   * @see MetricsRecorder
71   * @see TracerHook
72   * @see NoOpMetricsRecorder
73   * @see NoOpTracerHook
74   */
75  @State(org.openjdk.jmh.annotations.Scope.Benchmark)
76  @BenchmarkMode(Mode.AverageTime)
77  @OutputTimeUnit(TimeUnit.NANOSECONDS)
78  @Warmup(iterations = 5, time = 1, timeUnit = TimeUnit.SECONDS)
79  @Measurement(iterations = 5, time = 1, timeUnit = TimeUnit.SECONDS)
80  // Class-level @Fork(0) is overridden by OptionsBuilder.forks() in main(); kept at 1 so IDE
81  // runners that auto-discover benchmarks get proper fork isolation.
82  @Fork(1)
83  @SuppressWarnings({
84  	"java:S8692" // Benchmark output filename intentionally stamps the real run date; fixing the clock would collide results.
85  })
86  public class ObservabilityNoopBenchmark {
87  
88  	private static final String OP_NAME = "org.apache.juneau.rest.server.server.BenchResource.get()";
89  	private static final String HTTP_METHOD = "GET";
90  	private static final String URI_TEMPLATE = "/bench";
91  	private static final int STATUS_CODE = 200;
92  
93  	// Interface-typed to match the production RestOpInvoker call pattern (virtual dispatch,
94  	// no static devirtualisation by the benchmarking harness).
95  	private MetricsRecorder recorder;
96  	private TracerHook tracer;
97  
98  	/**
99  	 * Initialises the NoOp singleton references once per benchmark trial.
100 	 */
101 	@Setup(Level.Trial)
102 	public void setup() {
103 		recorder = NoOpMetricsRecorder.INSTANCE;
104 		tracer = NoOpTracerHook.INSTANCE;
105 	}
106 
107 	/**
108 	 * Exercises the observability hot path using NoOp singletons.
109 	 *
110 	 * <p>
111 	 * The {@link Blackhole} parameter prevents the JIT from discarding the scope reference after
112 	 * {@link Scope#close()} — keeping the virtual-dispatch chain alive for accurate allocation
113 	 * measurement.
114 	 *
115 	 * @param bh JMH blackhole to prevent dead-code elimination.
116 	 */
117 	@Benchmark
118 	public void observabilityNoopBlock(Blackhole bh) {
119 		long startNanos = System.nanoTime();
120 		// NoOpTracerHook.startSpan() ignores the request arg; null is safe on the NoOp path.
121 		@SuppressWarnings({
122 			"java:S2637"  // Suppression required for test context; see annotation for details.
123 		})
124 		var scope = tracer.startSpan((RestRequest) null);
125 		try {
126 			scope.setStatusCode(STATUS_CODE);
127 		} finally {
128 			scope.close();
129 			var elapsed = Duration.ofNanos(System.nanoTime() - startNanos);
130 			recorder.record(OP_NAME, HTTP_METHOD, URI_TEMPLATE, STATUS_CODE, elapsed, null, "", "");
131 		}
132 		bh.consume(scope);
133 	}
134 
135 	/**
136 	 * Benchmark entry point.
137 	 *
138 	 * <p>
139 	 * Runs the benchmark with the GC allocation profiler ({@code -prof gc}) and writes
140 	 * results to {@code juneau-integration-tests/jmh-results/observability-YYYY-MM-DD.json}.
141 	 * Use {@code python3 scripts/microbench.py observability} for the recommended invocation,
142 	 * which reads the JSON output and asserts the allocation threshold.
143 	 *
144 	 * @param args Unused; JMH options are hard-coded for reproducibility.
145 	 * @throws RunnerException If JMH fails to run the benchmark.
146 	 */
147 	public static void main(String[] args) throws RunnerException {
148 		var dateTag = LocalDate.now().toString();
149 		var resultDir = Paths.get("juneau-integration-tests", "jmh-results");
150 		try {
151 			Files.createDirectories(resultDir);
152 		} catch (Exception e) {
153 			// Best-effort; JMH will fail with a clear message if the path is unwritable.
154 		}
155 		var jsonPath = resultDir.resolve("observability-" + dateTag + ".json").toString();
156 
157 		var opt = new OptionsBuilder()
158 			.include(ObservabilityNoopBenchmark.class.getSimpleName())
159 			.warmupIterations(5)
160 			.warmupTime(TimeValue.seconds(1))
161 			.measurementIterations(5)
162 			.measurementTime(TimeValue.seconds(1))
163 			// forks(0): run in the same JVM as exec:java to avoid the forked process inheriting
164 			// an incomplete java.class.path (exec-maven-plugin loads via its own ClassLoader,
165 			// not via the system class path, so ForkedMain is unavailable to child processes).
166 			.forks(0)
167 			.addProfiler(GCProfiler.class)
168 			.resultFormat(ResultFormatType.JSON)
169 			.result(jsonPath)
170 			.build();
171 
172 		new Runner(opt).run();
173 	}
174 }
