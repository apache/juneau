17  package org.apache.juneau.microbench.runtime;
18  
19  import java.util.concurrent.*;
20  
21  import org.apache.juneau.marshall.json.*;
22  import org.apache.juneau.microbench.runtime.BenchBeans.*;
23  import org.openjdk.jmh.annotations.*;
24  
25  /**
26   * JMH single-shot benchmark for the first serialize and parse in a fresh JVM.
27   *
28   * <p>
29   * Each fork runs one operation with no warmup, so the score includes class loading and static initialization of
30   * the marshalling stack, building the {@code ClassMeta} graph and interpreted execution. This is the latency the
31   * first request of a newly started application sees.
32   *
33   * <h5 class='section'>Running:</h5>
34   * <pre class='bconsole'>
35   * 	$ python3 scripts/microbench.py runtime --include ColdStartBenchmark
36   * </pre>
37   *
38   * @since 10.0.0
39   */
40  @State(Scope.Benchmark)
41  @BenchmarkMode(Mode.SingleShotTime)
42  @OutputTimeUnit(TimeUnit.MILLISECONDS)
43  @Warmup(iterations = 0)
44  @Measurement(iterations = 1)
45  @Fork(10)
46  public class ColdStartBenchmark {
47  
48  	/**
49  	 * Serializes the order graph with {@link JsonSerializer#DEFAULT} for the first time in this JVM.
50  	 *
51  	 * @return The serialized text.
52  	 * @throws Exception If serialization fails.
53  	 */
54  	@Benchmark
55  	public String firstJsonSerialize() throws Exception {
56  		return JsonSerializer.DEFAULT.write(BenchBeans.order());
57  	}
58  
59  	/**
60  	 * Parses a JSON order graph with {@link JsonParser#DEFAULT} for the first time in this JVM.
61  	 *
62  	 * <p>
63  	 * The input is a literal so the serializer stays uninitialized.
64  	 *
65  	 * @return The parsed graph.
66  	 * @throws Exception If parsing fails.
67  	 */
68  	@Benchmark
69  	public Order firstJsonParse() throws Exception {
70  		return JsonParser.DEFAULT.read(SAMPLE_JSON, Order.class);
71  	}
72  
73  	private static final String SAMPLE_JSON = """
74  		{"id":"ORD-1","status":"PAID","created":"2026-10-19T14:30:15Z","shipBy":"2026-10-24","priority":true,"total":12.5,
75  		"customer":{"id":1,"name":"Jane","email":"jane@example.org","since":"2019-04-01",
76  		"addresses":[{"street":"1 Main Street","city":"Springfield","postalCode":"12345","country":"US","primary":true}],
77  		"preferences":{"language":"en"}},
78  		"items":[{"sku":"SKU-1","description":"Item","quantity":2,"unitPrice":6.25,"status":"PAID","tags":["a","b"]}],
79  		"history":["PENDING","PAID"],"attributes":{"k":"v"},"counters":{"c":1}}
80  		""";
81  }
