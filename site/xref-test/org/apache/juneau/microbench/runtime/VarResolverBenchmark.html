17  package org.apache.juneau.microbench.runtime;
18  
19  import java.util.*;
20  import java.util.concurrent.*;
21  
22  import org.apache.juneau.commons.svl.*;
23  import org.openjdk.jmh.annotations.*;
24  
25  /**
26   * JMH benchmark for resolving pre-compiled SVL templates.
27   *
28   * <p>
29   * Covers plain text, a {@code ${...}} var reference, scalar {@code #{...}} expressions and a {@code .![ ]} projection
30   * over a list of maps.  It is the reference for before/after comparisons of changes to the segment classes.
31   *
32   * <h5 class='section'>Running:</h5>
33   * <pre class='bconsole'>
34   * 	$ python3 scripts/microbench.py runtime --include 'VarResolverBenchmark.*'
35   * </pre>
36   *
37   * @since 10.0.0
38   */
39  @State(Scope.Benchmark)
40  @BenchmarkMode(Mode.Throughput)
41  @OutputTimeUnit(TimeUnit.MILLISECONDS)
42  @Warmup(iterations = 3, time = 1)
43  @Measurement(iterations = 5, time = 1)
44  @Fork(2)
45  public class VarResolverBenchmark {
46  
47  	/** Counter var: {@code ${C:x}} returns {@code "x"} prefixed with a constant. */
48  	public static class CountVar extends SimpleVar {
49  		/** Constructor. */
50  		public CountVar() { super("C"); }
51  		@Override public String resolve(VarResolverSession session, String key) { return "c-" + key; }
52  	}
53  
54  	/** Function {@code rows()} returning ten rows of two fields. */
55  	public static class Rows implements VarFunction {
56  		private static final List<Object> ROWS = rows();
57  
58  		@Override public String name() { return "rows"; }
59  		@Override public int minArity() { return 0; }
60  		@Override public int maxArity() { return 0; }
61  		@Override public String invoke(VarResolverSession session, List<Object> args) { return "rows"; }
62  		@Override public Object invokeValue(VarResolverSession session, List<Object> args) { return ROWS; }
63  
64  		private static List<Object> rows() {
65  			var l = new ArrayList<Object>();
66  			for (var i = 0; i < 10; i++)
67  				l.add(Map.of("id", "r" + i, "n", (long)i));
68  			return l;
69  		}
70  	}
71  
72  	private VarResolver resolver;
73  	private VarTemplate plain, varRef, mixed, arith, func, projection;
74  
75  	/** Builds the resolver and compiles the templates. */
76  	@Setup
77  	public void setup() {
78  		resolver = VarResolver.create().defaultFunctions().vars(CountVar.class).functions(Rows.class).build();
79  		plain = resolver.compile("Hello world, no markers here");
80  		varRef = resolver.compile("Value: ${C:abc}");
81  		mixed = resolver.compile("a ${C:x} b ${C:y} c");
82  		arith = resolver.compile("Sum: #{1 + 2}");
83  		func = resolver.compile("Name: #{upper('a')}");
84  		projection = resolver.compile("#{join(',', rows().![id + '-' + n])}");
85  	}
86  
87  	/** @return Resolved text. */
88  	@Benchmark public String plainText() { return plain.resolve(resolver.createSession()); }
89  
90  	/** @return Resolved text. */
91  	@Benchmark public String varReference() { return varRef.resolve(resolver.createSession()); }
92  
93  	/** @return Resolved text. */
94  	@Benchmark public String mixedVars() { return mixed.resolve(resolver.createSession()); }
95  
96  	/** @return Resolved text. */
97  	@Benchmark public String exprArithmetic() { return arith.resolve(resolver.createSession()); }
98  
99  	/** @return Resolved text. */
100 	@Benchmark public String exprFunction() { return func.resolve(resolver.createSession()); }
101 
102 	/** @return Resolved text. */
103 	@Benchmark public String exprProjection() { return projection.resolve(resolver.createSession()); }
104 }
