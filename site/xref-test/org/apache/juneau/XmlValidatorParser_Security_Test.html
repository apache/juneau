35  /**
36   * Exercises the independent StAX path used to validate serialized XML in integration tests.
37   */
38  class XmlValidatorParser_Security_Test extends TestBase {
39  
40  	private HttpServer server;
41  	private final AtomicInteger requests = new AtomicInteger();
42  	private String external;
43  
44  	@BeforeEach void startServer() throws Exception {
45  		server = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
46  		server.createContext("/external", exchange -> {
47  			requests.incrementAndGet();
48  			var body = "<!ENTITY injected 'FETCHED'>".getBytes(StandardCharsets.UTF_8);
49  			exchange.sendResponseHeaders(200, body.length);
50  			try (var stream = exchange.getResponseBody()) {
51  				stream.write(body);
52  			}
53  		});
54  		server.start();
55  		external = "http://127.0.0.1:" + server.getAddress().getPort() + "/external";
56  	}
57  
58  	@AfterEach void stopServer() {
59  		if (server != null)
60  			server.stop(0);
61  	}
62  
63  	private static String externalDocument(String vector, String uri) {
64  		return switch (vector) {
65  			case "general" -> "<!DOCTYPE A [<!ENTITY ext SYSTEM '" + uri + "'>]><A>&ext;</A>";
66  			case "parameter" -> "<!DOCTYPE A [<!ENTITY % ext SYSTEM '" + uri + "'>%ext;]><A>safe</A>";
67  			case "systemDtd" -> "<!DOCTYPE A SYSTEM '" + uri + "'><A>safe</A>";
68  			case "publicDtd" -> "<!DOCTYPE A PUBLIC '-//Juneau//DTD XXE Test//EN' '" + uri + "'><A>safe</A>";
69  			default -> throw new IllegalArgumentException(vector);
70  		};
71  	}
72  
73  	private static String expansionDocument() {
74  		// 100,000 leaf expansions exceed the JDK 17/21 default count, but stay small if protection regresses.
75  		var dtd = new StringBuilder("<!DOCTYPE A [<!ENTITY e0 'x'>");
76  		for (int i = 1; i <= 5; i++)
77  			dtd.append("<!ENTITY e").append(i).append(" '").append(("&e" + (i - 1) + ";").repeat(10)).append("'>");
78  		return dtd.append("]><A>&e5;</A>").toString();
79  	}
80  
81  	@Test void ordinaryXmlStillValidates() throws Exception {
82  		new XmlValidatorParser().validate(new StringReader("<A>safe &amp; sound</A>"));
83  	}
84  
85  	@ParameterizedTest
86  	@ValueSource(strings = {"general", "parameter", "systemDtd", "publicDtd"})
87  	void externalReferencesNeverFetch(String vector) throws Exception {
88  		try {
89  			new XmlValidatorParser().validate(new StringReader(externalDocument(vector, external)));
90  		} catch (XMLStreamException expected) {
91  			// Empty resolution or rejection is allowed, but fetching before rejection is not.
92  		}
93  		assertEquals(0, requests.get(), vector + " fetched an external resource");
94  	}
95  
96  	@Test void fileEntityIsNotExpanded(@TempDir Path directory) throws Exception {
97  		var secret = directory.resolve("secret.txt");
98  		Files.writeString(secret, "PRIVATE_FILE_MARKER");
99  		try {
100 			var reader = new XmlValidatorParser().getStaxReader(new StringReader(externalDocument("general", secret.toUri().toString())));
101 			try {
102 				while (reader.hasNext()) {
103 					reader.next();
104 					if (reader.hasText())
105 						assertFalse(reader.getText().contains("PRIVATE_FILE_MARKER"));
106 				}
107 			} finally {
108 				reader.close();
109 			}
110 		} catch (XMLStreamException expected) {
111 			for (Throwable cause = expected; cause != null; cause = cause.getCause())
112 				assertFalse(cause.toString().contains("PRIVATE_FILE_MARKER"));
113 		}
114 	}
115 
116 	@Test
117 	@Timeout(10)
118 	void exponentialEntityExpansionIsRejected() {
119 		assertThrows(XMLStreamException.class, () -> new XmlValidatorParser().validate(new StringReader(expansionDocument())));
120 	}
121 }
