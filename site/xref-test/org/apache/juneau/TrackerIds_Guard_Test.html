32  /**
33   * Build-wide guard: no work-tracker id may appear in any {@code src/main} or {@code src/test} tree of this repository.
34   *
35   * <p>
36   * Tracker ids are letter-plus-four-digit references into a private work board (a project prefix, a dash, a one-letter
37   * project code, then the number, with an optional lettered-child suffix) and a legacy form that used a different
38   * prefix.  They mean nothing to a reader of the public source and rot as soon as the board is reorganised, so code,
39   * tests, resources and in-code docs say what they mean instead of citing the item.
40   *
41   * <p>
42   * The scan lists files with {@code git ls-files --cached --others --exclude-standard} from this module's parent directory
43   * (so gitignored files such as {@code __pycache__} are never read), falling back to a directory walk when git is not
44   * available; it considers only files below a
45   * {@code src/main} or {@code src/test} directory, and skips build output ({@code target}), {@code node_modules} and
46   * {@code .git}.  Files are read as ISO-8859-1 so binary resources cannot raise a decoding error; the pattern is pure
47   * ASCII.  There are currently no historical-record files in these trees that need exempting.
48   */
49  class TrackerIds_Guard_Test extends TestBase {
50  
51  	/** The prefixes and shape are assembled from segments so this guard's own source is not a hit. */
52  	private static final Pattern TRACKER_ID = Pattern.compile("(?:" + String.join("|", "WO" + "RK", "TO" + "DO") + ")-[A-Z][0-9]{4}[a-z]?");
53  
54  	private static final Set<String> SKIPPED_DIRS = Set.of("target", "node_modules", ".git");
55  
56  	/** Files larger than this are not source text and are not scanned. */
57  	private static final long MAX_BYTES = 8L * 1024 * 1024;
58  
59  	@Test void a01_noTrackerIdsInSourceTrees() throws IOException {
60  		var root = Path.of("..").toAbsolutePath().normalize();
61  		assertTrue(Files.isDirectory(root.resolve("juneau-core")), () -> "Repository root not found at '" + root + "'; the test expects the module basedir as its working directory.");
62  
63  		var offenders = scan(root, candidateFiles(root));
64  		assertTrue(offenders.isEmpty(), () -> "Tracker ids found in source trees (state what the code does instead of citing a work item): " + offenders.stream().collect(Collectors.joining("\n  ", "\n  ", "")));
65  	}

67  	@Test void a02_gitListingSkipsIgnoredFilesButKeepsUntrackedOnes(@TempDir Path dir) throws Exception {
68  		if (! git(dir, "init", "-q"))
69  			return;  // git not available; the walk fallback is covered by a03
70  		var id = "WO" + "RK-J0001";
71  		Files.writeString(dir.resolve(".gitignore"), "ignored/\n");
72  		var ignored = Files.createDirectories(dir.resolve("ignored/src/main"));
73  		Files.writeString(ignored.resolve("A.pyc"), id);
74  		var untracked = Files.createDirectories(dir.resolve("m/src/test"));
75  		Files.writeString(untracked.resolve("B.txt"), id);
76  
77  		var listed = gitFiles(dir);
78  		assertNotNull(listed);
79  		assertTrue(listed.contains(dir.resolve("m/src/test/B.txt")), "untracked, non-ignored file must be listed");
80  		assertFalse(listed.contains(dir.resolve("ignored/src/main/A.pyc")), "ignored file must not be listed");
81  		assertEquals(1, scan(dir, listed).size());
82  	}

84  	@Test void a03_walkFallbackFindsIds(@TempDir Path dir) throws Exception {
85  		var src = Files.createDirectories(dir.resolve("m/src/main"));
86  		Files.writeString(src.resolve("C.txt"), "WO" + "RK-J0001");
87  		Files.writeString(Files.createDirectories(dir.resolve("m/target/src/main")).resolve("D.txt"), "WO" + "RK-J0001");
88  		var found = scan(dir, walkFiles(dir));
89  		assertEquals(1, found.size());
90  		assertTrue(found.get(0).startsWith("m/src/main/C.txt"), found.get(0));
91  	}

93  	/** Files to scan: tracked plus untracked-but-not-ignored when git is usable, otherwise a plain directory walk. */
94  	private static List<Path> candidateFiles(Path root) throws IOException {
95  		var listed = gitFiles(root);
96  		return listed != null ? listed : walkFiles(root);
97  	}
98  
99  	/** Returns the files git would track (cached plus untracked, excluding ignored), or null when git cannot answer. */
100 	private static List<Path> gitFiles(Path root) {
101 		try {
102 			var pb = new ProcessBuilder("git", "ls-files", "-z", "--cached", "--others", "--exclude-standard").directory(root.toFile()).redirectErrorStream(false);
103 			pb.redirectError(ProcessBuilder.Redirect.DISCARD);
104 			var proc = pb.start();
105 			var out = new String(proc.getInputStream().readAllBytes(), StandardCharsets.UTF_8);
106 			if (proc.waitFor() != 0)
107 				return null;
108 			var result = new ArrayList<Path>();
109 			for (var name : out.split("\0"))
110 				if (! name.isEmpty())
111 					result.add(root.resolve(name));
112 			return result;
113 		} catch (IOException e) {
114 			return null;
115 		} catch (InterruptedException e) {
116 			Thread.currentThread().interrupt();
117 			return null;
118 		}
119 	}
120 
121 	private static boolean git(Path dir, String...args) throws Exception {
122 		var cmd = new ArrayList<String>();
123 		cmd.add("git");
124 		cmd.addAll(List.of(args));
125 		try {
126 			var pb = new ProcessBuilder(cmd).directory(dir.toFile());
127 			pb.redirectErrorStream(true);
128 			var proc = pb.start();
129 			proc.getInputStream().readAllBytes();
130 			return proc.waitFor() == 0;
131 		} catch (IOException e) {
132 			return false;
133 		}
134 	}
135 
136 	private static List<Path> walkFiles(Path root) throws IOException {
137 		var files = new ArrayList<Path>();
138 		Files.walkFileTree(root, new SimpleFileVisitor<Path>() {
139 			@Override public FileVisitResult preVisitDirectory(Path dir, BasicFileAttributes attrs) {
140 				return SKIPPED_DIRS.contains(dir.getFileName().toString()) ? FileVisitResult.SKIP_SUBTREE : FileVisitResult.CONTINUE;
141 			}
142 			@Override public FileVisitResult visitFile(Path file, BasicFileAttributes attrs) {
143 				if (attrs.isRegularFile())
144 					files.add(file);
145 				return FileVisitResult.CONTINUE;
146 			}
147 		});
148 		return files;
149 	}
150 
151 	/** Scans the source-tree files among the candidates; returns sorted "relative-path (id)" entries. */
152 	private static List<String> scan(Path root, List<Path> candidates) throws IOException {
153 		var offenders = new ArrayList<String>();
154 		for (var file : candidates) {
155 			var rel = root.relativize(file);
156 			if (isInSourceTree(rel) && Files.isRegularFile(file) && Files.size(file) <= MAX_BYTES && ! hasSkippedDir(rel)) {
157 				var m = TRACKER_ID.matcher(Files.readString(file, StandardCharsets.ISO_8859_1));
158 				if (m.find())
159 					offenders.add(rel + " (" + m.group() + ")");
160 			}
161 		}
162 		offenders.sort(null);
163 		return offenders;
164 	}
165 
166 	private static boolean hasSkippedDir(Path relative) {
167 		for (var i = 0; i < relative.getNameCount() - 1; i++)
168 			if (SKIPPED_DIRS.contains(relative.getName(i).toString()))
169 				return true;
170 		return false;
171 	}
172 
173 	/** True when the repo-relative path has a {@code src/main} or {@code src/test} segment pair. */
174 	private static boolean isInSourceTree(Path relative) {
175 		for (var i = 0; i < relative.getNameCount() - 1; i++)
176 			if (relative.getName(i).toString().equals("src") && Set.of("main", "test").contains(relative.getName(i + 1).toString()))
177 				return true;
178 		return false;
179 	}
