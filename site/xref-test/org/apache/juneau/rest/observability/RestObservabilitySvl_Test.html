33  /**
34   * Tests SVL variable support in observability annotation attributes.
35   *
36   * <p>
37   * The resource-level {@code @Rest(observability)} attribute is resolved at {@code RestContext} build time
38   * (the constructor pre-warms the memoizer via the startup-fail check). Op-level attributes
39   * ({@code metricName}, {@code metricTags}) are resolved lazily on first request.
40   *
41   * <p>
42   * Note: {@code MockRestClient} caches the {@link org.apache.juneau.rest.server.server.RestContext} per resource class.
43   * Each scenario uses a distinct static inner class so the SVL state at build time is deterministic.
44   */
45  @ResourceLock(Resources.SYSTEM_PROPERTIES)
46  class RestObservabilitySvl_Test extends TestBase {
47  
48  	public static final class RecordingMetricsRecorder implements MetricsRecorder {
49  		public final List<Event> events = new CopyOnWriteArrayList<>();
50  
51  		@SuppressWarnings({
52  			"java:S6213" // 'record' mirrors the MetricsRecorder SPI method name; renaming would break the @Override contract.
53  		})
54  		@Override
55  		public void record(String opName, String httpMethod, String uriTemplate, int statusCode, Duration elapsed, Throwable error, String metricName, String metricTags) {
56  			events.add(new Event(metricName, metricTags));
57  		}
58  
59  		public Event last() { return events.get(events.size() - 1); }
60  		public void clear() { events.clear(); }
61  	}
62  
63  	public static final class Event {
64  		public final String metricName, metricTags;
65  		Event(String metricName, String metricTags) {
66  			this.metricName = metricName; this.metricTags = metricTags;
67  		}
68  	}
69  
70  	// -----------------------------------------------------------------------------------------------------------------
71  	// A: metricName and metricTags with literal values flow correctly to record().
72  	// -----------------------------------------------------------------------------------------------------------------
73  
74  	private static final RecordingMetricsRecorder A_REC = new RecordingMetricsRecorder();
75  
76  	@Rest
77  	public static class A extends RestServlet {
78  		private static final long serialVersionUID = 1L;
79  
80  		@org.apache.juneau.commons.inject.Bean
81  		public MetricsRecorder recorder() { return A_REC; }
82  
83  		@RestGet(path = "/op", metricName = "my.service.requests", metricTags = "svc=orders")
84  		public String op() { return "ok"; }
85  	}
86  
87  	private static final MockRestClient CA = MockRestClient.buildLax(A.class);
88  
89  	@Test void a01_literalMetricNameAndTags_resolvedCorrectly() throws Exception {
90  		A_REC.clear();
91  		CA.get("/op").run().assertStatus(200);
92  		assertEquals("my.service.requests", A_REC.last().metricName,
93  			"literal metricName must reach MetricsRecorder.record() unchanged");
94  		assertEquals("svc=orders", A_REC.last().metricTags,
95  			"literal metricTags must reach MetricsRecorder.record() unchanged");
96  	}
97  
98  	// -----------------------------------------------------------------------------------------------------------------
99  	// B: SVL variable in @Rest(observability) is resolved at RestContext build time.
100 	//    When the system property resolves to "false", metrics are suppressed for the entire resource.
101 	//    The static initializer sets the property before building the client and then clears it; // NOSONAR
102 	//    this works because the resource-level observability attribute IS resolved eagerly (the
103 	//    startup-fail check pre-warms it inside the RestContext constructor).
104 	// -----------------------------------------------------------------------------------------------------------------
105 
106 	private static final RecordingMetricsRecorder B_REC = new RecordingMetricsRecorder();
107 
108 	@Rest(observability = "$S{test.svl.obs.b,}")
109 	public static class B extends RestServlet {
110 		private static final long serialVersionUID = 1L;
111 
112 		@org.apache.juneau.commons.inject.Bean
113 		public MetricsRecorder recorder() { return B_REC; }
114 
115 		@RestGet("/ping")
116 		public String ping() { return "pong"; }
117 	}
118 
119 	// Build the client while the property is "false" so the context captures observability=false.
120 	private static final MockRestClient CB;
121 	static {
122 		System.setProperty("test.svl.obs.b", "false");
123 		CB = MockRestClient.buildLax(B.class);
124 		System.clearProperty("test.svl.obs.b");
125 	}
126 
127 	@Test void b01_svlObservability_resolvedAtBuildTime_suppressesMetrics() throws Exception {
128 		B_REC.clear();
129 		CB.get("/ping").run().assertStatus(200);
130 		assertEquals(0, B_REC.events.size(),
131 			"$S{} SVL variable resolving to 'false' at context build time must suppress observability block");
132 	}
133 }
