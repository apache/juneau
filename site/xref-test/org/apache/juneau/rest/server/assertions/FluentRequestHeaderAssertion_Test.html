22  import java.util.*;
23  
24  import org.apache.juneau.*;
25  import org.apache.juneau.rest.server.httppart.*;
26  import org.apache.juneau.test.assertions.*;
27  import org.junit.jupiter.api.*;

29  /**
30   * Validates {@link FluentRequestHeaderAssertion}.
31   *
32   * <p>
33   * Regression: the chained constructor used to hard-code {@code super(null, ...)}, discarding its {@code creator}
34   * argument, so config (custom failure message, output stream, etc.) set on the creating assertion was lost when a
35   * header assertion was built via the chained constructor.  The sibling query/form-param assertions correctly pass
36   * {@code super(creator, ...)}.
37   */
38  class FluentRequestHeaderAssertion_Test extends TestBase {
39  
40  	@Test void a01_creatorMsgPropagates() {
41  		var header = mock(RequestHeader.class);
42  		when(header.asString()).thenReturn(Optional.of("actual"));
43  
44  		// The custom failure message configured on the creating assertion must propagate to the header assertion
45  		// (it did not before the fix, because the chained ctor hard-coded super(null, ...)).
46  		var creator = new FluentStringAssertion<>("ignored", null).setMsg("CUSTOM {msg}");
47  		var h = new FluentRequestHeaderAssertion<>(creator, header, null);
48  
49  		var e = assertThrows(RuntimeException.class, () -> h.is("expected"));
50  		assertTrue(e.getMessage().startsWith("CUSTOM "), () -> "Message did not carry creator config: " + e.getMessage());
51  	}
52  
53  	@Test void a02_noCreatorUsesDefaultMsg() {
54  		var header = mock(RequestHeader.class);
55  		when(header.asString()).thenReturn(Optional.of("actual"));
56  
57  		var h = new FluentRequestHeaderAssertion<>(header, null);
58  
59  		var e = assertThrows(RuntimeException.class, () -> h.is("expected"));
60  		assertFalse(e.getMessage().startsWith("CUSTOM "));
61  	}
62  
63  	// A null RequestHeader used to NPE in the constructor (value.asString()) and in every
64  	// transform method (value.asBoolean(), etc.). Verify null is now tolerated end-to-end.
65  
66  	@Test void a03_nullValue_chainedConstructor_doesNotThrow() {
67  		var h = new FluentRequestHeaderAssertion<>(null, (RequestHeader)null, null);
68  		assertDoesNotThrow(h::isNull);
69  	}
70  
71  	@Test void a04_nullValue_directConstructor_doesNotThrow() {
72  		var h = new FluentRequestHeaderAssertion<>((RequestHeader)null, null);
73  		assertDoesNotThrow(h::isNull);
74  	}
75  
76  	@Test void a05_nullValue_transformMethodsDoNotThrowAndYieldNull() {
77  		var h = new FluentRequestHeaderAssertion<>((RequestHeader)null, null);
78  		assertDoesNotThrow(() -> h.asBoolean().isNull());
79  		assertDoesNotThrow(() -> h.asInteger().isNull());
80  		assertDoesNotThrow(() -> h.asLong().isNull());
81  		assertDoesNotThrow(() -> h.asZonedDateTime().isNull());
82  		assertDoesNotThrow(() -> h.as(String.class).isNull());
83  	}
84  }
