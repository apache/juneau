39  	//------------------------------------------------------------------------------------------------------------------
40  	// In-memory storage surfaces live bucket state after activity.
41  	//------------------------------------------------------------------------------------------------------------------
42  
43  	@Test void a01_inMemorySnapshotReportsTokensAndRemainingAfterPartialDrain() {
44  		var s = RateLimitGuard.Storage.inMemory();
45  		s.tryAcquire("k", 5, 1.0);
46  		s.tryAcquire("k", 5, 1.0);
47  		s.tryAcquire("k", 5, 1.0);
48  		var snapshot = s.snapshot();
49  		assertEquals(1, snapshot.size());
50  		var b = snapshot.get("k");
51  		assertNotNull(b);
52  		assertEquals("k", b.key());
53  		// Started full (5), three tokens consumed → at-most ~2 left, refill barely measurable in the test
54  		// window so the actual fraction sits in (2.0, 2.0 + ε].
55  		assertTrue(b.tokens() >= 2.0 && b.tokens() < 3.0,
56  			"expected tokens in [2.0, 3.0), got " + b.tokens());
57  		assertEquals((int) Math.floor(b.tokens()), b.remaining());
58  		assertFalse(b.throttled(), "bucket should not be throttled with 2+ tokens left");
59  		assertNotNull(b.lastRequest());
60  		// Wall-clock instant should be within a few seconds of now.
61  		var skewMillis = Math.abs(System.currentTimeMillis() - b.lastRequest().toEpochMilli());
62  		assertTrue(skewMillis < 5_000L, "lastRequest drift too large: " + skewMillis + "ms");
63  	}

66  	// Exhausted bucket flips the throttled flag.
67  	//------------------------------------------------------------------------------------------------------------------
68  
69  	@Test void a02_exhaustedBucketReportsThrottledTrueAndZeroRemaining() {
70  		var s = RateLimitGuard.Storage.inMemory();
71  		// burst of 1, drain it, then poke once more to fail (consumes the refill but stays empty).
72  		s.tryAcquire("k", 1, 0.01);
73  		s.tryAcquire("k", 1, 0.01);
74  		var b = s.snapshot().get("k");
75  		assertNotNull(b);
76  		assertTrue(b.tokens() < 1.0, "expected tokens < 1.0, got " + b.tokens());
77  		assertEquals(0, b.remaining());
78  		assertTrue(b.throttled(), "expected throttled=true on empty bucket");
79  	}

83  	//------------------------------------------------------------------------------------------------------------------
84  
85  	@Test void a03_multiKeySnapshotHasOneEntryPerKey() {
86  		var s = RateLimitGuard.Storage.inMemory();
87  		s.tryAcquire("alpha", 3, 1.0);
88  		s.tryAcquire("beta", 3, 1.0);
89  		s.tryAcquire("gamma", 3, 1.0);
90  		var snapshot = s.snapshot();
91  		assertEquals(3, snapshot.size());
92  		assertEquals(Set.of("alpha", "beta", "gamma"), snapshot.keySet());
93  		for (var e : snapshot.entrySet())
94  			assertEquals(e.getKey(), e.getValue().key());
95  	}

120 	//------------------------------------------------------------------------------------------------------------------
121 	// Backwards-compat: a custom Storage impl that does NOT override snapshot()
122 	// inherits the default which returns Map.of() without throwing.
123 	//------------------------------------------------------------------------------------------------------------------
124 
125 	/** Custom storage that only implements the mandatory SPI methods. */
126 	static final class C01_NoSnapshotStorage implements RateLimitGuard.Storage {
127 		@Override public Storage.AcquireResult tryAcquire(String key, int capacity, double permitsPerSecond) {
128 			return new Storage.AcquireResult(true, capacity - 1, 0L);
129 		}
130 		@Override public void evict(Duration ttl) { /* intentionally empty */ }
131 	}
132 
133 	@Test void c01_customStorageWithoutSnapshotOverrideReturnsEmptyMap() {
134 		RateLimitGuard.Storage s = new C01_NoSnapshotStorage();
135 		var snapshot = s.snapshot();
136 		assertNotNull(snapshot);
137 		assertTrue(snapshot.isEmpty());
138 		// Calling it twice should be safe.
139 		assertEquals(Map.of(), s.snapshot());
140 	}

142 	//------------------------------------------------------------------------------------------------------------------
143 	// RateLimitGuard.snapshot() convenience delegates to the configured storage.
144 	//------------------------------------------------------------------------------------------------------------------
145 
146 	@Test void d01_guardSnapshotDelegatesToStorage() {
147 		var g = RateLimitGuard.create()
148 			.permitsPerSecond(1)
149 			.burst(3)
150 			.build();
151 		assertSame(g.getStorage(), g.getStorage());  // sanity — same instance
152 		assertTrue(g.snapshot().isEmpty());
153 		g.getStorage().tryAcquire("k1", 3, 1.0);
154 		g.getStorage().tryAcquire("k2", 3, 1.0);
155 		var snapshot = g.snapshot();
156 		assertEquals(2, snapshot.size());
157 		assertEquals(Set.of("k1", "k2"), snapshot.keySet());
158 	}

164 	@Test void d02_guardConfigAccessorsReflectBuilderState() {
165 		var g = RateLimitGuard.create()
166 			.permitsPerMinute(60)
167 			.burst(42)
168 			.xForwardedForAware(true)
169 			.exemptPaths("/healthz", "/livez")
170 			.build();
171 		assertEquals(42, g.getCapacity());
172 		assertEquals(1.0, g.getPermitsPerSecond(), 1e-9);
173 		assertTrue(g.isXForwardedForAware());
174 		assertEquals(Set.of("/healthz", "/livez"), g.getExemptPaths());
175 		assertNotNull(g.getStorage());
