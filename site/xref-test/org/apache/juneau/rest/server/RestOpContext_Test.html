49  	static RestOpContext opOf(RestContext ctx, String methodName) {
50  		return ctx.getRestOperations().getOpContexts().stream()
51  			.filter(oc -> oc.getJavaMethod().getName().equals(methodName))
52  			.findFirst()
53  			.orElseThrow(() -> new AssertionError("No op context found for method " + methodName));
54  	}

56  	// The real failure detail from checkOpObservabilityBackendPresent() is buried a few levels down a
57  	// wrapped-ServletException cause chain by the time it surfaces from getRestOperations(); this walks
58  	// the chain looking for a given substring rather than asserting on the (wrapper) top-level message.
59  	static boolean causeChainContains(Throwable t, String substring) {
60  		for (Throwable c = t; c != null; c = c.getCause())
61  			if (c.getMessage() != null && c.getMessage().contains(substring))
62  				return true;
63  		return false;
64  	}
