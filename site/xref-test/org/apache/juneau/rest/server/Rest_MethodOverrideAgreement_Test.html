33  /**
34   * Pins that dispatch ({@link RestSession#getMethod()}) and the write-protection filters ({@link MethodSafety#effectiveMethod})
35   * agree: every method claim dispatch honours must make {@code effectiveMethod} unsafe when the claimed verb is unsafe.
36   * On a safe request line, dispatch must also stay safe: it either refuses the claim with {@code 405} or dispatches the
37   * line itself, a safe method, or an operation declared {@link RestOp#read() read=true}.
38   *
39   * <p>
40   * The filter side is called with the {@code HttpServletRequest}'s own request-line method, raw query and headers, not with
41   * the dispatched verb, which would make the check pass by construction.
42   *
43   * <p>
44   * Limitation: {@code MockServletRequest} builds parameters from the URI query only, so these rows cannot show a container
45   * surfacing a form-body {@code method=} on a {@code GET}.  Dispatch reads the claim from the raw query string only, so that
46   * risk is removed structurally rather than by a row.
47   */
48  @SuppressWarnings({
49  	"java:S5961" // High assertion count acceptable in comprehensive test
50  })
51  class Rest_MethodOverrideAgreement_Test extends TestBase {

57  	/** Reports the dispatched method and the method a write-protection filter would judge the same request on. */
58  	public static class Base {
59  		@RestOp(method="*", path="/*")
60  		public void any(RestRequest req, RestResponse res) {
61  			var hsr = req.getHttpServletRequest();
62  			res.setHeader("X-Result", req.getMethod() + "|" + MethodSafety.effectiveMethod(hsr.getMethod(), hsr.getQueryString(), hsr::getHeader));
63  		}
64  	}
65  
66  	/** Adds two read operations, which the explicit keys reach ahead of the {@code *} operation. */
67  	public static class ReadBase extends Base {
68  		@RestOp(method="VIEW", path="/*", read=true)
69  		public void view(RestRequest req, RestResponse res) {
70  			any(req, res);
71  		}
72  
73  		@RestOp(method="PARSE", path="/*", read=true)
74  		public void parse(RestRequest req, RestResponse res) {
75  			any(req, res);
76  		}
77  	}

116 	private static final Map<Class<?>,MockRestClient> CLIENTS = new ConcurrentHashMap<>();
117 
118 	private static MockRestClient client(Class<?> c) {
119 		return CLIENTS.computeIfAbsent(c, k -> MockRestClient.create(k).ignoreErrors().build());
120 	}

122 	/** The {@code X-Result} header: {@code dispatched|effective}. */
123 	private static String[] result(RestResponse_ r) {
124 		return r.header.split("\\|");
125 	}
126 
127 	private record RestResponse_(int status, String header, String allow, String content) {}
128 
129 	private static RestResponse_ call(Class<?> c, String line, String uri, String... headers) throws Exception {
130 		var req = client(c).request(line, uri);
131 		for (var i = 0; i < headers.length; i += 2)
132 			req = req.header(headers[i], headers[i + 1]);
133 		var r = req.run();
134 		return new RestResponse_(r.getStatusCode(), r.getStringHeader("X-Result").orElse(null), r.getStringHeader("Allow").orElse(null), r.getContent().asString());
135 	}

141 	static Stream<Arguments> agreementRows() {
142 		var configs = List.of(Star.class, OnlyPut.class, OnlyDelete.class, OnlyPatch.class, OnlyPost.class, OnlyFrob.class);
143 		var lines = List.of("GET", "HEAD", "OPTIONS", "TRACE");
144 		List<String> channels = list("q:method", "q:METHOD", "q:%6Dethod");
145 		MethodSafety.overrideHeaders().forEach(h -> channels.add("h:" + h));
146 		var values = List.of("DELETE", "PUT", "PATCH", "POST", "FROB", "delete", "VIEW", "view", "PARSE");
147 		var rows = new ArrayList<Arguments>();
148 		for (var c : configs)
149 			for (var l : lines)
150 				for (var ch : channels)
151 					for (var v : values)
152 						rows.add(Arguments.of(c, l, ch, v));
153 		return rows.stream();
154 	}
155 
156 	/** How a guarded-line row ended. */
157 	private enum Outcome { REFUSED, READ, OTHER }
158 
159 	/** Every line in the matrix is safe, so every row is checked against the rule as well as the filter. */
160 	private static Outcome check(Class<?> config, String line, String channel, String value) throws Exception {
161 		var r = channel.startsWith("q:") ? call(config, line, "/x?" + channel.substring(2) + "=" + value) : call(config, line, "/x", channel.substring(2), value);
162 		var row = config.getSimpleName() + " " + line + " " + channel + "=" + value;
163 		if (r.status == 405) {
164 			assertNull(r.header, row + ": refused, but an operation ran");
165 			assertNotNull(r.allow, row + ": 405 without Allow");
166 			assertFalse(r.allow.isEmpty(), row + ": 405 with an empty Allow");
167 			return Outcome.REFUSED;
168 		}
169 		assertNotNull(r.header, row + ": status " + r.status + " but nothing was dispatched");
170 		var x = result(r);
171 		var dispatched = x[0];
172 		var effective = x[1];
173 		var read = READS.contains(dispatched);
174 		assertTrue(dispatched.equals(line) || MethodSafety.isSafe(dispatched) || read, row + ": a safe line dispatched " + dispatched);
175 		if (! dispatched.equals(line) && ! MethodSafety.isSafe(dispatched))
176 			assertFalse(MethodSafety.isSafe(effective), row + ": dispatched " + dispatched + " but the filter judged " + effective);
177 		return read ? Outcome.READ : Outcome.OTHER;
178 	}

180 	@ParameterizedTest(name="{0} {1} {2}={3}")
181 	@MethodSource("agreementRows")
182 	void a01_dispatchedClaimIsUnsafeToTheFilter(Class<?> config, String line, String channel, String value) throws Exception {
183 		check(config, line, channel, value);
184 	}

186 	@Test void a02_overrideIsActuallyHonoured() throws Exception {
187 		// Guards a01 against being vacuous: the claims it generates really do change the dispatched method.  An unsafe
188 		// claim is honoured only on an unsafe line, where the line itself is what the filter judges.
189 		assertEquals("DELETE|POST", call(Star.class, "POST", "/x?method=DELETE").header);
190 		assertEquals("DELETE|POST", call(Star.class, "POST", "/x?METHOD=delete").header);
191 		for (var h : MethodSafety.overrideHeaders())
192 			assertEquals("DELETE|POST", call(Star.class, "POST", "/x", h, "DELETE").header, h);
193 		// On a safe line, a read is honoured.
194 		assertEquals("VIEW|VIEW", call(Star.class, "GET", "/x?method=VIEW").header);
195 	}

202 		// Headers on an enabled channel that normalise equal do not conflict; the honoured value is the first admitted one.

206 	@Test void a05_guardedLineMatrixNotVacuous() throws Exception {
207 		var counts = new EnumMap<Outcome,Integer>(Outcome.class);
208 		for (var row : agreementRows().toList()) {
209 			var a = row.get();
210 			counts.merge(check((Class<?>)a[0], (String)a[1], (String)a[2], (String)a[3]), 1, Integer::sum);
211 		}
212 		assertTrue(counts.getOrDefault(Outcome.REFUSED, 0) > 0, counts.toString());
213 		assertTrue(counts.getOrDefault(Outcome.READ, 0) > 0, counts.toString());
214 	}
