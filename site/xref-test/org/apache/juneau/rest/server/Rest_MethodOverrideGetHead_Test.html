42   * Pins the rule for method overrides on a safe request line: a {@code GET}, {@code HEAD}, {@code OPTIONS} or
43   * {@code TRACE} may be overridden only to another safe method or to an operation declared
44   * {@link RestOp#read() read=true}.  Anything else is refused with {@code 405} and an {@code Allow} header, before the
45   * operation, its hooks or its guards run.
46   */

252 	private static MockRestClient client(Class<?> c) {
253 		return CLIENTS.computeIfAbsent(c, k -> MockRestClient.create(k).ignoreErrors().build());
254 	}

256 	private record Result(int status, String op, String allow, String content) {}
257 
258 	private static Result call(Class<?> c, String line, String uri, String... headers) throws Exception {
259 		var req = client(c).request(line, uri);
260 		for (var i = 0; i < headers.length; i += 2)
261 			req = req.header(headers[i], headers[i + 1]);
262 		var r = req.run();
263 		return new Result(r.getStatusCode(), r.getStringHeader("X-Op").orElse(null), r.getStringHeader("Allow").orElse(null), r.getContent().asString());
264 	}
265 
266 	/** Sends a claim on one channel: {@code q} is the {@code method} parameter, anything else names a header. */
267 	private static Result claim(Class<?> c, String line, String channel, String value) throws Exception {
268 		return "q".equals(channel) ? call(c, line, "/x?method=" + value) : call(c, line, "/x", channel, value);
269 	}
270 
271 	private static void assertRefused(Result r, String row) {
272 		assertEquals(405, r.status, row);
273 		assertNull(r.op, row + ": an operation ran");
274 		assertNotNull(r.allow, row + ": no Allow header");
275 		assertEquals(0, WRITES.get(), row + ": a write ran");
276 		// Every unsafe claim has an operation at /x, so this is the new refusal, not findOperation's own 405.
277 		if (! row.startsWith("HEAD"))
278 			assertTrue(r.content.contains("Method override not allowed on a safe request method."), row + ": " + r.content);
279 	}
280 
281 	static Stream<Arguments> unsafeClaims() {
282 		var channels = new ArrayList<String>();
283 		channels.add("q");
284 		channels.addAll(MethodSafety.overrideHeaders());
285 		var values = List.of("DELETE", "delete", "Delete", "PATCH", "PUT", "POST", "FROB");
286 		var rows = new ArrayList<Arguments>();
287 		for (var ch : channels)
288 			for (var v : values)
289 				rows.add(Arguments.of(ch, v));
290 		return rows.stream();
291 	}

322 		// The classic mock client upper-cases the line, so these go straight to RestContext.execute.
323 		var ctx = new RestContext(new RestContext.Args(RawLine.class, null, null, RawLine::new, "", null, null, null, RestContext.ContextKind.ROOT)).postInit().postInitChildFirst();
324 		for (var line : List.of("get", "Get")) {
325 			RawLine.seen = null;
326 			var res = MockServletResponse.create();
327 			ctx.execute(ctx.getResource(), MockServletRequest.create(line, "/raw").servletPath("").contextPath(""), res);
328 			assertEquals(200, res.getStatus(), line);
329 			assertEquals(line, RawLine.seen, "the line did not arrive in its original case");
330 
331 			res = MockServletResponse.create();
332 			ctx.execute(ctx.getResource(), MockServletRequest.create(line, "/x?method=DELETE").servletPath("").contextPath(""), res);
333 			assertEquals(405, res.getStatus(), line);
334 			assertEquals(0, WRITES.get(), line);
335 		}
