54  	@Test void a01_managerGetSetClear() {
55  		// Manager with a null context resolves the JUL default backend.
56  		var m = new LoggersManager();
57  		m.setLevel(null, L_NAME, "FINE");
58  		assertEquals("FINE", m.getLevel(null, L_NAME));
59  		assertTrue(m.getLevels(null).containsKey(L_NAME));
60  		assertEquals("FINE", m.getLevels(null).get(L_NAME));
61  		// Blank clears the level (inherits) -> empty string.
62  		m.setLevel(null, L_NAME, "");
63  		assertEquals("", m.getLevel(null, L_NAME));
64  	}

102 	@Test void a08_julBackendDirect() {
103 		// JulLogBackend carries the JUL logic directly (the default backend).
104 		var b = JulLogBackend.INSTANCE;
105 		b.setLevel(L_NAME, "FINE");
106 		assertEquals("FINE", b.getLevel(L_NAME));
107 		assertTrue(b.getLevels().containsKey("ROOT"));
108 		b.setLevel(L_NAME, null);
109 		assertEquals("", b.getLevel(L_NAME));
110 	}

112 	@Test void a09_explicitBackendSelection() {
113 		// A LoggersSettings with an explicitly-declared backend drives that backend, not JUL.
114 		var calls = new java.util.ArrayList<String>();
115 		LogBackend fake = new LogBackend() {
116 			@Override public java.util.Map<String,String> getLevels() { calls.add("getLevels"); return java.util.Map.of("x", "DEBUG"); }
117 			@Override public String getLevel(String name) { calls.add("getLevel:" + name); return "DEBUG"; }
118 			@Override public void setLevel(String name, String level) { calls.add("setLevel:" + name + "=" + level); }
119 		};
120 		var s = LoggersSettings.create().backend(fake).build();
121 		assertSame(fake, s.getBackend());
122 		s.getBackend().setLevel("x", "DEBUG");
123 		assertEquals("DEBUG", s.getBackend().getLevel("x"));
124 		assertTrue(calls.contains("setLevel:x=DEBUG"));
125 	}
