32  /**
33   * Tests {@link AsyncResponseProcessor} — the new processor that unwraps
34   * {@link CompletableFuture} / {@link CompletionStage} return values from {@code @RestOp} methods
35   * and bridges them to the servlet container's {@link jakarta.servlet.AsyncContext} lifecycle.
36   *
37   * <p>
38   * In the {@code MockRestClient} test harness the underlying {@code MockServletRequest} reports
39   * {@code isAsyncSupported() == false}, so this processor exercises its synchronous-fallback path
40   * (block on the future with the configured timeout) which is exactly the path that test suites
41   * outside a real servlet container will take. The async {@code AsyncContext}-driven path is
42   * covered by integration tests against real containers; here we verify the unwrap, error
43   * propagation, timeout, and bare-{@code Future} rejection contracts.
44   */
45  class AsyncResponseProcessor_Test extends TestBase {

348 	//    thenApply still surfaces as ExecutionException in the sync fallback.

351 	@Rest
352 	public static class K {
353 		@RestGet("/chained")
354 		public CompletionStage<String> chained() {
355 			return CompletableFuture.supplyAsync(() -> "step1")
356 				.thenApply(s -> {
357 					throw new IllegalStateException("downstream-failure");
358 				});
359 		}
360 
361 		@RestGet("/withCompletionException")
362 		public CompletableFuture<String> withCompletionException() {
363 			var f = new CompletableFuture<String>();
364 			// Wrap directly in CompletionException — exercises the unwrap/cause chain.
365 			f.completeExceptionally(new CompletionException(new IllegalArgumentException("inner")));
366 			return f;
367 		}
368 	}
