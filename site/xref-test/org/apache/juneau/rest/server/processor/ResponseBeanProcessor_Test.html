29  /**
30   * Tests {@link ResponseBeanProcessor} — the processor that handles {@link Response @Response}-annotated
31   * return values, dispatching status code, response headers ({@link Header @Header}), and body content
32   * ({@link Content @Content}) into the underlying {@code RestResponse}.
33   *
34   * <p>
35   * These tests build minimal {@code @Response} bean shapes inline and exercise them through
36   * {@link MockRestClient}, covering: default status / explicit {@link StatusCode @StatusCode} /
37   * {@link StatusCode @StatusCode}-via-method, {@link Header @Header}-annotated single-value methods,
38   * {@link Header @Header}-with-name-{@code "*"} methods returning Map / Collection / Array /
39   * {@code HttpHeader} / {@code HttpPart}, {@link Content @Content}-annotated body methods including
40   * {@code OutputStream}/{@code Writer} parameter overloads, response-bean-as-{@link Throwable}, and
41   * pass-through (NEXT) for non-response output.
42   */
43  @SuppressWarnings({
44  	"serial"  // serialVersionUID not required for test classes.
45  })
46  class ResponseBeanProcessor_Test extends TestBase {
47

49  	// A: @Response bean — explicit @StatusCode on class, @Content method, plain @Header methods.
50  	// -----------------------------------------------------------------------------------------------------------------
51  
52  	@Response @StatusCode(201)
53  	public static class A1 {
54  		@Header("X-Foo")
55  		public String getFoo() { return "foo-value"; }
56  
57  		@Header("X-Bar")
58  		public int getBar() { return 42; }
59  
60  		@Content
61  		public String getContent() { return "a1-body"; }
62  	}
63  
64  	@Rest
65  	public static class A {
66  		@RestGet("/a1")
67  		public A1 a1() { return new A1(); }
68  	}

322 	public static class F1Plain {
323 		@Override public String toString() { return "f1-plain"; }
324 	}
325 
326 	@Rest
327 	public static class F {
328 		@RestGet("/f1")
329 		@Response
330 		public F1Plain f1() { return new F1Plain(); }
331 
332 		// Non-response output: plain string → ResponseBeanProcessor returns NEXT immediately.
333 		@RestGet("/f2")
334 		public String f2() { return "plain-string"; }
335 
336 		// Null output → ResponseBeanProcessor returns NEXT immediately.
337 		@RestGet("/f3")
338 		public Object f3() { return null; }
339 	}

348 		CF.get("/f2").run().assertStatus(200).assertContent("plain-string");
