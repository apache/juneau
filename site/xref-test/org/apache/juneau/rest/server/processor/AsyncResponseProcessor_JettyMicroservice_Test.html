257 	public static class AsyncCaptureServlet extends RestServlet {
258 		private static final long serialVersionUID = 1L;
259 
260 		/** Test-controlled gate future for the delayed endpoint; completed by the test to trigger completion-path emit. */
261 		static volatile CompletableFuture<String> gate;
262 
263 		/** Test-controlled gate future for the ad-hoc-log endpoint (async-carry tripwire). */
264 		static volatile CompletableFuture<String> adhocGate;
265 
266 		/** Counted down by {@code @RestEndCall} on the request thread — i.e. AFTER {@code RestSession.finish()} returns. */
267 		static volatile CountDownLatch finishLatch;
268 
269 		/**
270 		 * When non-<jk>null</jk>, the async handlers open a {@link LogContext} scope carrying {@code corrId=carryValue} on
271 		 * the request thread. Async-carry v1 leaves request-id/{@code LogContext}-scope ownership to a future
272 		 * {@code RequestIdFilter}, so these Phase-6 async-carry tests establish their own request-thread scope instead of
273 		 * relying on any framework-opened scope. The scope stays open across the async hand-off (so
274 		 * {@code RestDebugPipeline.captureAsyncSnapshot(...)} snapshots it) and is closed in {@link #endCall()} on the same
275 		 * request thread — no thread-pool leak.
276 		 */
277 		static volatile String carryValue;
278 
279 		/** The request-thread {@link LogContext} scope opened from {@link #carryValue}; closed in {@link #endCall()}. */
280 		static volatile LogContext.Scope carryScope;
281 
282 		private void openCarryScopeIfRequested() {
283 			var cv = carryValue;
284 			if (cv != null)
285 				carryScope = RichLogger.context().with("corrId", cv);
286 		}
287 
288 		@RestGet("/delayed")
289 		public CompletableFuture<String> delayed(RestResponse res) {
290 			openCarryScopeIfRequested();
291 			var g = new CompletableFuture<String>();
292 			gate = g;
293 			// The header + content-type are set on the COMPLETION thread (inside thenApply), so a record that captured
294 			// them proves rendering happened after async completion, not during synchronous finish().
295 			return g.thenApply(body -> {
296 				res.setContentType("text/plain");
297 				res.setHeader("X-Late-Header", "late-header-value");
298 				return body;
299 			});
300 		}

320 		@RestGet("/adhoc")
321 		public CompletableFuture<String> adhoc(RestRequest req, RestResponse res) {
322 			openCarryScopeIfRequested();
323 			var g = new CompletableFuture<String>();
324 			adhocGate = g;
325 			// The ad-hoc app log runs inside the completion callback (the COMPLETION thread), where the request-thread
326 			// LogContext scope has already closed — documented async-carry edge.
327 			return g.thenApply(body -> {
328 				res.setContentType("text/plain");
329 				req.fine("adhoc-completion-log");
330 				return body;
331 			});
332 		}
333 
334 		@RestEndCall
335 		public void endCall() {
336 			// Close the request-thread carry scope here — on the request thread, AFTER captureAsyncSnapshot(...) has
337 			// already snapshotted it during response processing — so the thread-local is restored and never leaks into a
338 			// pooled Jetty worker.
339 			var sc = carryScope;
340 			if (sc != null) {
341 				sc.close();
342 				carryScope = null;
343 			}
344 			var l = finishLatch;
345 			if (l != null)
346 				l.countDown();
347 		}
348 	}

360 	/** JUL handler that records only the events for a single op logger (attached with {@code useParentHandlers=false}). */
361 	private static final class CollectingHandler extends Handler {
362 		private final List<LogRecord> records = Collections.synchronizedList(new ArrayList<>());
363 
364 		@Override public void publish(LogRecord rec) {
365 			// LogRecord's own "record" identifier is a restricted identifier since records were added; use "rec" instead.
366 			if (isLoggable(rec))
367 				records.add(rec);
368 		}
369 
370 		@Override public void flush() {
371 			// No buffering to flush — records are added directly to the in-memory list.
372 		}
373 		@Override public void close() {
374 			// Nothing to release — no external resources are held by this in-memory handler.
375 		}
376 
377 		List<LogRecord> forLogger(String name) {
378 			synchronized (records) {
379 				return records.stream().filter(x -> eq(name, x.getLoggerName())).toList();
380 			}
381 		}
382 	}

384 	/** Snapshot/restore of a JUL logger's mutable level/handler state so tests never leak elevated levels. */
385 	private static final class LoggerState {
386 		private final Logger logger;
387 		private final Level level;
388 		private final boolean useParentHandlers;
389 		private final Handler[] handlers;
390 
391 		LoggerState(Logger logger) {
392 			this.logger = logger;
393 			level = logger.getLevel();
394 			useParentHandlers = logger.getUseParentHandlers();
395 			handlers = logger.getHandlers();
396 		}
397 
398 		void restore() {
399 			for (var h : logger.getHandlers())
400 				logger.removeHandler(h);
401 			for (var h : handlers)
402 				logger.addHandler(h);
403 			logger.setUseParentHandlers(useParentHandlers);
404 			logger.setLevel(level);
405 		}
406 	}
407 
408 	private static CollectingHandler attach(Logger logger, Level tier) {
409 		for (var h : logger.getHandlers())
410 			logger.removeHandler(h);
411 		logger.setUseParentHandlers(false);
412 		logger.setLevel(tier);
413 		var handler = new CollectingHandler();
414 		handler.setLevel(Level.INFO);
415 		logger.addHandler(handler);
416 		return handler;
417 	}

428 	/**
429 	 * Polls {@code handler.forLogger(loggerName)} until at least {@code minCount} records have been captured, or a
430 	 * 5-second deadline elapses.
431 	 *
432 	 * <p>
433 	 * {@code AsyncResponseProcessor.finalizeAsync(...)} emits the completion-path record AFTER {@code res.flushBuffer()}
434 	 * has already written the response to the client (by design — see that method's javadoc: "Completion-path emission
435 	 * runs AFTER the body/headers are written"). A test that calls a blocking HTTP client and then immediately inspects
436 	 * the collecting handler is racing the server-side emission against the client's own wake-up from the socket read.
437 	 * That race is normally unobservable (the emit is the very next statement on the same thread that just flushed), but
438 	 * widens under load on the async-timeout path specifically: there, the emitting thread is the container's own
439 	 * async-timeout dispatch thread (invoked via {@code AsyncListener.onTimeout}) rather than a thread whose only
440 	 * remaining job is to emit-then-return, so it can be preempted between the flush and the emit. Bounded polling
441 	 * (instead of loosening the assertion) keeps the "exactly one record" contract intact while tolerating that
442 	 * scheduling gap; the fast, overwhelmingly common case (record already present) pays no extra latency.
443 	 *
444 	 * @param handler The collecting handler to poll.
445 	 * @param loggerName The logger name to filter on.
446 	 * @param minCount The minimum record count to wait for.
447 	 * @return The current record list for {@code loggerName} (may have fewer than {@code minCount} entries if the
448 	 * 	deadline elapsed).
449 	 * @throws InterruptedException If interrupted while waiting.
450 	 */
451 	private static List<LogRecord> awaitRecords(CollectingHandler handler, String loggerName, int minCount) throws InterruptedException {
452 		var deadline = System.nanoTime() + TimeUnit.SECONDS.toNanos(5);
453 		List<LogRecord> recs;
454 		while ((recs = handler.forLogger(loggerName)).size() < minCount && System.nanoTime() < deadline)
455 			Thread.sleep(10);
456 		return recs;
457 	}

577 	// H05/H06 (async-carry): the async-completion debug record carries the request-thread's LogContext across the
578 	// async hop (RestDebugSnapshot carry + emit pre-seed), while ad-hoc app logging on the completion thread does NOT
579 	// (the documented v1 thread-confined boundary).
580 	//
581 	// Async-carry v1 does NOT open any framework LogContext scope itself (request-id / scope ownership is deferred to a
582 	// future RequestIdFilter, opened after @RestStartCall). These tests therefore establish their own request-thread scope
583 	// in the handler (carrying a generic "corrId"), left open across the async hand-off and closed in @RestEndCall — which
584 	// is exactly the shape that future filter will produce — to exercise the async-carry mechanism in isolation.
585 	// -----------------------------------------------------------------------------------------------------------------
586 
587 	@Test void h05_asyncCompletionRecord_carriesRequestThreadContextField() throws Exception {
588 		var opLogger = Logger.getLogger(AsyncCaptureServlet.class.getName() + ".delayed");
589 		var state = new LoggerState(opLogger);
590 		try {
591 			var handler = attach(opLogger, Level.FINE);
592 			AsyncCaptureServlet.gate = null;
593 			AsyncCaptureServlet.carryValue = "async-corr-1";  // handler opens a request-thread LogContext scope from this
594 			AsyncCaptureServlet.finishLatch = new CountDownLatch(1);
595 
596 			var respFuture = HTTP.sendAsync(
597 				HttpRequest.newBuilder()
598 					.uri(URI.create(capFixture.getRootUrl() + "/asyncCap/delayed"))
599 					.timeout(Duration.ofSeconds(15))
600 					.GET()
601 					.build(),
602 				BodyHandlers.ofString());
603 
604 			// The request thread has returned from finish() and @RestEndCall closed the carry scope — so the completion
605 			// record can only carry corrId if it was snapshotted on the request thread and re-established at emit (carry).
606 			assertTrue(AsyncCaptureServlet.finishLatch.await(10, TimeUnit.SECONDS),
607 				"request thread did not reach @RestEndCall");
608 			assertNotNull(AsyncCaptureServlet.gate, "gate future should have been published by the handler");
609 			AsyncCaptureServlet.gate.complete("delayed-body-value");
610 
611 			var resp = respFuture.get(15, TimeUnit.SECONDS);
612 			assertEquals(200, resp.statusCode(), "body: " + resp.body());
613 
614 			var recs = handler.forLogger(opLogger.getName());
615 			assertSize(() -> "exactly one completion-path record expected", 1, recs);
616 			var rec = recs.get(0);
617 			assertEquals("async-corr-1", LogRecordContext.of(rec).get("corrId"),
618 				"the async completion record must carry the LogContext captured on the request thread (design §8.2 carry)");
619 
620 			// Deferred (Phase 10) async gate: the always-on requestId scope (opened in RestSession's constructor) is
621 			// carried across the async hop via the same 364 snapshot, AND the completion-thread rendered message
622 			// carries the [requestId=<id>] prefix (statusLine reading session.getRequestId()) — proving 364's carry
623 			// and 402's statusLine read compose on the completion thread, where the live LogContext is empty.
624 			var rid = LogRecordContext.of(rec).get("requestId");
625 			assertNotNull(rid, "the async completion record must also carry the always-on framework requestId");
626 			assertTrue(rec.getMessage().contains("[requestId=" + rid + "] "),
627 				"the async completion rendered message must carry the requestId prefix: " + rec.getMessage());
628 		} finally {
629 			AsyncCaptureServlet.carryValue = null;
630 			AsyncCaptureServlet.finishLatch = null;
631 			var g = AsyncCaptureServlet.gate;
632 			if (g != null)
633 				g.complete("cleanup");
634 			AsyncCaptureServlet.gate = null;
635 			state.restore();
636 		}
637 	}
