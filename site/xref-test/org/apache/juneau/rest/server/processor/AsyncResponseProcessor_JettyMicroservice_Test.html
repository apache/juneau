39  /**
40   * Real-Jetty deployment-parity coverage for {@link AsyncResponseProcessor}.
41   *
42   * <p>
43   * The {@code MockServletRequest} test harness reports {@code isAsyncSupported() == false}, so the
44   * sibling unit-test class {@code AsyncResponseProcessor_Test} only exercises the synchronous-fallback
45   * path. This class boots a real Jetty microservice on an ephemeral port to drive the async-dispatch
46   * branch ({@code req.startAsync()}, the {@code AsyncListener} callbacks, the {@code finalizeAsync}
47   * pipeline, and the {@code unwrap} static helper) so that the {@code AsyncContext}-driven path is
48   * actually exercised at coverage time.
49   *
50   * @since 10.0.0
51   */
52  @org.apache.juneau.testing.JettyMicroserviceTest
53  class AsyncResponseProcessor_JettyMicroservice_Test extends TestBase {
54  
55  	// -----------------------------------------------------------------------------------------------------------------
56  	// Resource — exercises success, failed-future, timeout, completion-stage, and bare-Future rejection on the
57  	// real async-dispatch path.
58  	// -----------------------------------------------------------------------------------------------------------------
59  
60  	@Rest(paths="/async/*", asyncTimeoutMillis="200")
61  	public static class AsyncTestServlet extends RestServlet {
62  		private static final long serialVersionUID = 1L;
63  
64  		@RestGet("/ok")
65  		public CompletableFuture<String> ok() {
66  			return CompletableFuture.supplyAsync(() -> "async-ok");
67  		}
68  
69  		@RestGet("/okStage")
70  		public CompletionStage<String> okStage() {
71  			return CompletableFuture.supplyAsync(() -> "async-stage-ok");
72  		}
73  
74  		@RestGet("/failed")
75  		public CompletableFuture<String> failed() {
76  			return CompletableFuture.supplyAsync(() -> {
77  				throw new IllegalStateException("kaboom-async");
78  			});
79  		}
80  
81  		@RestGet("/failedDirect")
82  		public CompletableFuture<String> failedDirect() {
83  			var f = new CompletableFuture<String>();
84  			f.completeExceptionally(new IllegalStateException("direct-failure"));
85  			return f;
86  		}
87  
88  		@RestGet("/timeout")
89  		public CompletableFuture<String> timeout() {
90  			return new CompletableFuture<>();  // Never completes — fires the AsyncContext timeout.
91  		}
92  
93  		@RestGet("/bareFuture")
94  		public Future<String> bareFuture() {
95  			// FutureTask is not a CompletionStage — must be rejected with a 500.
96  			return new FutureTask<>(() -> "should-not-block");
97  		}
98  
99  		@RestGet("/sync")
100 		public String sync() {
101 			return "still-sync";
102 		}
103 
104 		@RestGet("/failedExecutionException")
105 		public CompletableFuture<String> failedExecutionException() {
106 			// Complete with an ExecutionException directly (not the usual CompletionException)
107 			// so that unwrap(...) hits its ExecutionException-stripping branch.
108 			var f = new CompletableFuture<String>();
109 			f.completeExceptionally(new ExecutionException("wrapped", new IllegalStateException("inner-cause")));
110 			return f;
111 		}
112 	}

124 	private static final HttpClient HTTP = HttpClient.newBuilder()
125 		.connectTimeout(Duration.ofSeconds(5))
126 		.followRedirects(HttpClient.Redirect.NEVER)
127 		.build();

129 	private static HttpResponse<String> get(String path) throws Exception {
130 		var req = HttpRequest.newBuilder()
131 			.uri(URI.create(fixture.getRootUrl() + path))
132 			.timeout(Duration.ofSeconds(15))
133 			.GET()
134 			.build();
135 		return HTTP.send(req, BodyHandlers.ofString());
136 	}

138 	// -----------------------------------------------------------------------------------------------------------------
139 	// A: 200 responses — happy-path async (CompletableFuture, CompletionStage) and the synchronous pass-through
140 	// (backward-compat smoke test that non-future content is returned unchanged on the real container).
141 	// -----------------------------------------------------------------------------------------------------------------
142 
143 	@ParameterizedTest
144 	@CsvSource({
145 		"/async/ok,      async-ok",        // CompletableFuture unwrapped
146 		"/async/okStage, async-stage-ok",  // CompletionStage unwrapped
147 		"/async/sync,    still-sync"        // synchronous handler passed through unchanged
148 	})
149 	void a01_successResponses_return200WithBody(String path, String expected) throws Exception {
150 		var resp = get(path);
151 		assertEquals(200, resp.statusCode(), "body: " + resp.body());
152 		assertTrue(resp.body().contains(expected), "body: " + resp.body());
153 	}
154 
155 	// -----------------------------------------------------------------------------------------------------------------
156 	// B: Failed-future paths — exercise the unwrap(...) static helper for CompletionException-wrapped causes
157 	// (supplyAsync wraps the thrown exception in CompletionException) AND the direct-completeExceptionally case.
158 	// -----------------------------------------------------------------------------------------------------------------
159 
160 	@ParameterizedTest
161 	@ValueSource(strings = {
162 		// supplyAsync → exception thrown in supplier wrapped in CompletionException — unwrap strips that layer.
163 		"/async/failed",
164 		// completeExceptionally with a plain throwable — unwrap's "no CompletionException wrapping" fall-through.
165 		"/async/failedDirect",
166 		// completeExceptionally(ExecutionException) — unwrap strips the ExecutionException wrapper (line 297-298).
167 		"/async/failedExecutionException",
168 		// Bare Future that is not a CompletionStage — rejected outright (category D).
169 		"/async/bareFuture"
170 	})
171 	void b01_errorResponses_return500(String path) throws Exception {
172 		var resp = get(path);
173 		assertEquals(500, resp.statusCode(), "body: " + resp.body());
174 	}
175 
176 	// -----------------------------------------------------------------------------------------------------------------
177 	// C: Async timeout — the AsyncContext fires onTimeout() after the configured 200ms ceiling.
178 	// -----------------------------------------------------------------------------------------------------------------
179 
180 	@Test void c01_neverCompleting_returns504() throws Exception {
181 		var before = System.currentTimeMillis();
182 		var resp = get("/async/timeout");
183 		var elapsed = System.currentTimeMillis() - before;
184 		assertEquals(504, resp.statusCode(), "body: " + resp.body());
185 		assertTrue(elapsed < 5_000, "200ms timeout should fire well before the 5s client timeout — actual " + elapsed + "ms");
186 	}

189 	// Resource G — async-completion executor configured exercises the whenCompleteAsync branch
190 	// (line 237 true, line 238 invocation).
191 	// -----------------------------------------------------------------------------------------------------------------
192 
193 	@Rest(paths="/asyncExec/*", asyncTimeoutMillis="200", asyncCompletionExecutor="execPool")
194 	public static class AsyncExecServlet extends RestServlet {
195 		private static final long serialVersionUID = 1L;
196 
197 		// The bean is declared on the resource class itself so the resource's bean store
198 		// resolves the executor name at startup.
199 		@Bean(name="execPool")
200 		public Executor execPool() {
201 			return Executors.newSingleThreadExecutor(r -> {
202 				var t = new Thread(r, "async-exec-pool");
203 				t.setDaemon(true);
204 				return t;
205 			});
206 		}
207 
208 		@RestGet("/value")
209 		public CompletableFuture<String> value() {
210 			return CompletableFuture.supplyAsync(() -> "exec-pool-async");
211 		}
212 	}

224 	@Test void g01_completionExecutor_routesCallback() throws Exception {
225 		// asyncCompletionExecutor configured — exercises whenCompleteAsync(callback, executor)
226 		// path (line 237 true / line 238).
227 		var port = execFixture.getPort();
228 		var req = HttpRequest.newBuilder()
229 			.uri(URI.create("http://localhost:" + port + "/asyncExec/value"))
230 			.timeout(Duration.ofSeconds(15))
231 			.GET()
232 			.build();
233 		var resp = HTTP.send(req, BodyHandlers.ofString());
234 		assertEquals(200, resp.statusCode(), "body: " + resp.body());
235 		assertTrue(resp.body().contains("exec-pool-async"), "body: " + resp.body());
236 	}
