17  package org.apache.juneau.rest.server;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.apache.juneau.*;
22  import org.apache.juneau.rest.mock.classic.*;
23  import org.apache.juneau.rest.server.logger.*;
24  import org.apache.juneau.rest.server.servlet.*;
25  import org.junit.jupiter.api.*;
26  
27  /**
28   * Phase 6 &mdash; proves a {@code @Rest(lazyChildren="true", childrenDefs=@Child(...))} child receives its
29   * host-seeded settings on first-request materialization, not just when eagerly built. Mirrors
30   * {@link LazyChildren_Test}'s style/package (same package so it lives alongside the base lazy-children suite).
31   */
32  class LazyChildren_ChildDefs_Test extends TestBase {
33  
34  	public static class SeedLogger extends CallLogger {
35  		public SeedLogger(org.apache.juneau.commons.inject.BeanStore bs) { super(bs); }
36  	}
37  
38  	@Rest(path="/lazychild")
39  	public static class LazyChildResource {
40  		@RestGet("/ping") public String ping() { return "lazy-pong"; }
41  	}
42  
43  	@Rest(
44  		path="/root",
45  		lazyChildren="true",
46  		childrenDefs=@Child(type=LazyChildResource.class, callLogger=SeedLogger.class)
47  	)
48  	public static class LazySeedParent extends BasicRestServletGroup {
49  		private static final long serialVersionUID = 1L;
50  	}
51  
52  	@Test void a01_seedAppliesAfterLazyMaterialization() throws Exception {
53  		var parent = new LazySeedParent();
54  		var client = MockRestClient.createLax(parent).build();
55  		var rc = parent.getContext();
56  
57  		assertTrue(rc.isLazyChildren());
58  
59  		// Before first request: registered for routing but not yet materialized.
60  		var entries = rc.getRestChildren().getLazyEntries();
61  		var entry = entries.get("lazychild");
62  		assertNotNull(entry, "Lazy entry key should be 'lazychild'");
63  		assertFalse(entry.isMaterialized(), "Should not be materialized before first request");
64  
65  		// First request triggers materialization.
66  		client.get("/lazychild/ping").run().assertStatus(200).assertContent("lazy-pong");
67  		assertTrue(entry.isMaterialized(), "Should be materialized after first request");
68  
69  		// The materialized context must carry the host's seeded callLogger -- proving the ResolvedChild
70  		// (including its @Child seed) survived deferred construction, not just eager construction.
71  		// (Materialized lazy children live on the LazyChildEntry itself, not in RestChildren.asMap() --
72  		// that map holds only eagerly-built children.)
73  		var childCtx = entry.materialized;
74  		assertNotNull(childCtx);
75  		assertInstanceOf(SeedLogger.class, childCtx.getCallLogger(),
76  			"Lazily-materialized child must receive the host's seeded SeedLogger, same as an eager child would");
77  	}
78  }
