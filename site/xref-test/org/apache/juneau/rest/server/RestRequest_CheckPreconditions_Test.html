25  /**
26   * Tests for {@link RestRequest#checkPreconditions(RestResponse)} — RFC 7232 conditional-request handling.
27   *
28   * <p>
29   * The test resource exposes endpoints that set well-known {@code ETag} and {@code Last-Modified} values on the
30   * response and then short-circuits on {@code checkPreconditions(res)}: if the returned {@code Optional} is non-empty
31   * the helper exception is thrown (yielding 304 / 412); otherwise the handler returns 200 with body {@code "OK"}.
32   */
33  class RestRequest_CheckPreconditions_Test extends TestBase {
34  
35  	// Fixed reference values used across tests.
36  	private static final String ETAG_STRONG = "\"v1\"";
37  	private static final String ETAG_WEAK = "W/\"v1\"";
38  	private static final String LM = "Fri, 22 May 2026 00:00:00 GMT";
39  	private static final String LM_EARLIER = "Thu, 21 May 2026 00:00:00 GMT";
40  	private static final String LM_LATER = "Sat, 23 May 2026 00:00:00 GMT";
41  
42  	@Rest
43  	public static class R {
44  		@RestGet("/strong")
45  		public String strong(RestRequest req, RestResponse res) {
46  			res.eTag(ETAG_STRONG).lastModified(Instant.parse("2026-05-22T00:00:00Z"));
47  			req.checkPreconditions(res).ifPresent(e -> { throw e; });
48  			return "OK";
49  		}
50  
51  		@RestGet("/weak")
52  		public String weak(RestRequest req, RestResponse res) {
53  			res.eTag(ETAG_WEAK).lastModified(Instant.parse("2026-05-22T00:00:00Z"));
54  			req.checkPreconditions(res).ifPresent(e -> { throw e; });
55  			return "OK";
56  		}
57  
58  		@RestGet("/lmOnly")
59  		public String lmOnly(RestRequest req, RestResponse res) {
60  			res.lastModified(Instant.parse("2026-05-22T00:00:00Z"));
61  			req.checkPreconditions(res).ifPresent(e -> { throw e; });
62  			return "OK";
63  		}
64  
65  		@RestGet("/etagOnly")
66  		public String etagOnly(RestRequest req, RestResponse res) {
67  			res.eTag(ETAG_STRONG);
68  			req.checkPreconditions(res).ifPresent(e -> { throw e; });
69  			return "OK";
70  		}
71  
72  		@RestGet("/bare")
73  		public String bare(RestRequest req, RestResponse res) {
74  			req.checkPreconditions(res).ifPresent(e -> { throw e; });
75  			return "OK";
76  		}
77  
78  		// Non-safe method to verify If-Modified-Since is ignored.
79  		@SuppressWarnings({
80  			"java:S4144" // Intentionally mirrors strong() to verify If-Modified-Since is ignored on a non-safe method.
81  		})
82  		@RestPost("/postStrong")
83  		public String postStrong(RestRequest req, RestResponse res) {
84  			res.eTag(ETAG_STRONG).lastModified(Instant.parse("2026-05-22T00:00:00Z"));
85  			req.checkPreconditions(res).ifPresent(e -> { throw e; });
86  			return "OK";
87  		}
88  	}

262 	//-----------------------------------------------------------------------------------------------------------------
263 	// F: Combinations of all four conditional headers — 16-cell precedence matrix.
264 	//
265 	// Layout: (M=If-Match, U=If-Unmodified-Since, N=If-None-Match, S=If-Modified-Since)
266 	//   - present-with-pass and present-with-fail are exercised in B/C/D/E.
267 	//   - here we exercise the cross-products that depend on RFC 7232 §6 ordering.
268 	//-----------------------------------------------------------------------------------------------------------------
269 
270 	@Test void f01_allFourPass_returns200() throws Exception {
271 		var c = MockRestClient.buildLax(R.class);
272 		c.get("/strong")
273 			.header("If-Match", ETAG_STRONG)
274 			.header("If-Unmodified-Since", LM_LATER)
275 			.header("If-None-Match", "\"v2\"")
276 			.header("If-Modified-Since", LM_EARLIER)
277 			.run().assertStatus(200);
278 	}
279 
280 	@Test void f02_ifMatchFails_returns412_evenWithNoneMatchPass() throws Exception {
281 		// If-Match short-circuits at step 1; If-None-Match steps are not consulted.
282 		var c = MockRestClient.buildLax(R.class);
283 		c.get("/strong")
284 			.header("If-Match", "\"v2\"")
285 			.header("If-None-Match", ETAG_STRONG)  // would also 304 standalone
286 			.run().assertStatus(412);
287 	}
288 
289 	@Test void f03_ifMatchPasses_thenIfNoneMatchTriggers304() throws Exception {
290 		// If-Match passes (step 1), If-Unmodified-Since skipped (step 2), If-None-Match triggers (step 3).
291 		var c = MockRestClient.buildLax(R.class);
292 		c.get("/strong")
293 			.header("If-Match", ETAG_STRONG)
294 			.header("If-None-Match", ETAG_STRONG)
295 			.run().assertStatus(304);
296 	}
297 
298 	@Test void f04_unmodifiedSincePassed_thenModifiedSinceTriggers304() throws Exception {
299 		// If-Match absent → If-Unmodified-Since checked (passes). If-None-Match absent → If-Modified-Since
300 		// triggers 304.
301 		var c = MockRestClient.buildLax(R.class);
302 		c.get("/lmOnly")
303 			.header("If-Unmodified-Since", LM_LATER)
304 			.header("If-Modified-Since", LM)
305 			.run().assertStatus(304);
306 	}
307 
308 	@Test void f05_unmodifiedSinceFails_412_beforeModifiedSinceEvaluated() throws Exception {

310 		var c = MockRestClient.buildLax(R.class);

313 			.header("If-Modified-Since", LM_LATER)  // would 304 standalone
314 			.run().assertStatus(412);
