27  /**
28   * Tests for {@link MethodSafety} and {@link Mutating @Mutating}: the safe-method set, the per-operation rule, and
29   * the startup failure a real {@link Rest @Rest} resource gets when it declares a mutating operation behind a safe
30   * method.
31   *
32   * <p>
33   * The group-c tests construct real {@link RestContext} instances, because the claim being made is that the
34   * application does not start &mdash; not that a static method throws when called directly.  They use
35   * {@code eagerInit} so the failure lands in the constructor, which is what "fail at boot" means.
36   *
37   * @since 10.0.0
38   */

40  	"unused" // Fixture methods and classes are reached only reflectively or through annotations, so they appear unused

103 	@Test void a05_theBoundaryAndTheBootCheckAgreeOnEveryMethod() {
104 		// The point of the delegation. If these two ever disagree, the boot check would clear a method the
105 		// boundary treats as a write (or worse, the reverse) and both controls would be quietly wrong. Asserted
106 		// over the safe set and a spread of others rather than trusting that one delegates to the other today.
107 		for (var m : new String[] { "GET", "HEAD", "OPTIONS", "TRACE", "POST", "PUT", "PATCH", "DELETE", "QUERY", "", "*" })
108 			assertEquals(MethodSafety.isSafe(m), ! LoopbackBoundary.isStateChanging(m), "disagreement on '" + m + "'");
109 		assertEquals(MethodSafety.isSafe(null), ! LoopbackBoundary.isStateChanging(null), "disagreement on null");
110 	}

259 	/** A header lookup that knows no headers. */
260 	static final UnaryOperator<String> NO_HEADERS = x -> null;
261 
262 	/** A header lookup holding exactly one header, matched case-insensitively as a servlet container would. */
263 	static UnaryOperator<String> header(String name, String value) {
264 		return x -> name.equalsIgnoreCase(x) ? value : null;
265 	}

625 	/** The framework wraps init failures, so assertions match against the whole cause chain. */
626 	static String messageChain(Throwable t) {
627 		var sb = new StringBuilder();
628 		for (var x = t; x != null; x = x.getCause())
629 			sb.append(x.getMessage()).append(" | ");
630 		return sb.toString();
631 	}
