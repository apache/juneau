17  package org.apache.juneau.rest.server;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.io.*;
22  import java.util.*;
23  
24  import org.apache.juneau.http.response.*;
25  import org.apache.juneau.rest.server.processor.*;
26  import org.junit.jupiter.api.*;
27  
28  /**
29   * Verifies the mixin-declared {@link Rest#mergeResponseProcessorsIntoHost() @Rest(mergeResponseProcessorsIntoHost=true)}
30   * opt-in: a mixin that declares it on its own class has its {@code responseProcessors} folded into the host's own chain
31   * under a plain {@link Rest#mixins() @Rest(mixins=...)} reference, while a mixin that does NOT declare it keeps today's
32   * isolated scoping (the non-silent contract).
33   *
34   * <p>
35   * This is the mixin-side, response-processor-scoped counterpart to
36   * {@code MixinInheritance_ResponseProcessors_Test} (a02/a04) in {@code juneau-integration-tests}, exercised here at the
37   * {@link RestContext} level without a mock client so it runs inside {@code juneau-rest-server}'s own test suite.
38   *
39   * @since 10.0.0
40   */
41  class MixinResponseProcessorFold_Test extends org.apache.juneau.TestBase {
42  
43  	//-----------------------------------------------------------------------------------------------------------
44  	// Fixtures
45  	//-----------------------------------------------------------------------------------------------------------
46  
47  	public static class HostRp1 implements ResponseProcessor {
48  		@Override public int process(RestOpSession s) throws IOException, NotAcceptable, BasicHttpException { return NEXT; }
49  	}
50  
51  	public static class MixinRp1 implements ResponseProcessor {
52  		@Override public int process(RestOpSession s) throws IOException, NotAcceptable, BasicHttpException { return NEXT; }
53  	}
54  
55  	/** Opted-in mixin: declares the response-processor fold on its own class. */
56  	@Rest(responseProcessors={MixinRp1.class}, mergeResponseProcessorsIntoHost=true)
57  	public static class M_OptedIn {
58  		@RestGet(path="/my") public String my() { return "my"; }
59  	}
60  
61  	/** Non-opted-in mixin: same response processor, but no fold opt-in (regression guard for the non-silent contract). */
62  	@Rest(responseProcessors={MixinRp1.class})
63  	public static class M_NotOptedIn {
64  		@RestGet(path="/my") public String my() { return "my"; }
65  	}
66  
67  	@Rest(responseProcessors={HostRp1.class}, mixins={M_OptedIn.class})
68  	public static class HostWithOptedInMixin {
69  		@RestGet(path="/h") public String h() { return "h"; }
70  	}
71  
72  	@Rest(responseProcessors={HostRp1.class}, mixins={M_NotOptedIn.class})
73  	public static class HostWithNonOptedInMixin {
74  		@RestGet(path="/h") public String h() { return "h"; }
75  	}
76  
77  	static RestContext.Args argsOf(Class<?> resourceClass, java.util.function.Supplier<?> supplier) {
78  		return new RestContext.Args(resourceClass, null, null, supplier, null, null, null, null, null, null);
79  	}
80  
81  	private static List<Class<?>> classesOf(ResponseProcessor[] rps) {
82  		var l = new ArrayList<Class<?>>(rps.length);
83  		for (var rp : rps) l.add(rp.getClass());
84  		return l;
85  	}
86  
87  	//-----------------------------------------------------------------------------------------------------------
88  	// a - opt-in folds the mixin's response processor into the host chain
89  	//-----------------------------------------------------------------------------------------------------------
90  
91  	@Test void a01_optedInMixin_foldsResponseProcessorIntoHostChain() throws Exception {
92  		var hostCtx = new RestContext(argsOf(HostWithOptedInMixin.class, HostWithOptedInMixin::new));
93  		var hostRps = classesOf(hostCtx.getResponseProcessors());
94  
95  		assertTrue(hostRps.contains(MixinRp1.class),
96  			"Host must fold in the opted-in mixin's MixinRp1 under a plain mixins= reference");
97  		assertTrue(hostRps.contains(HostRp1.class),
98  			"Host must still register its own HostRp1");
99  
100 		var mixinCtx = hostCtx.getMixinContexts().get(M_OptedIn.class);
101 		assertNotNull(mixinCtx);
102 		assertTrue(classesOf(mixinCtx.getResponseProcessors()).contains(MixinRp1.class),
103 			"Mixin endpoint must still have its own MixinRp1");
104 	}
105 
106 	//-----------------------------------------------------------------------------------------------------------
107 	// b - non-opted-in mixin keeps today's isolated scoping (non-silent contract)
108 	//-----------------------------------------------------------------------------------------------------------
109 
110 	@Test void b01_nonOptedInMixin_keepsIsolatedScoping() throws Exception {
111 		var hostCtx = new RestContext(argsOf(HostWithNonOptedInMixin.class, HostWithNonOptedInMixin::new));
112 		var hostRps = classesOf(hostCtx.getResponseProcessors());
113 
114 		assertFalse(hostRps.contains(MixinRp1.class),
115 			"Host must NOT have MixinRp1 — a mixin without the opt-in stays scoped to its own endpoints");
116 		assertTrue(hostRps.contains(HostRp1.class),
117 			"Host must still register its own HostRp1");
118 
119 		var mixinCtx = hostCtx.getMixinContexts().get(M_NotOptedIn.class);
120 		assertNotNull(mixinCtx);
121 		assertTrue(classesOf(mixinCtx.getResponseProcessors()).contains(MixinRp1.class),
122 			"Mixin endpoint must still have its own MixinRp1 via its own @Rest(responseProcessors=)");
123 	}
124 
125 	//-----------------------------------------------------------------------------------------------------------
126 	// c - ordering / precedence: the host's own processor precedes the folded mixin processor
127 	//-----------------------------------------------------------------------------------------------------------
128 
129 	@Test void c01_foldedProcessorAppendedAfterHostChain() throws Exception {
130 		var hostCtx = new RestContext(argsOf(HostWithOptedInMixin.class, HostWithOptedInMixin::new));
131 		var hostRps = classesOf(hostCtx.getResponseProcessors());
132 
133 		var hostIdx = hostRps.indexOf(HostRp1.class);
134 		var mixinIdx = hostRps.indexOf(MixinRp1.class);
135 		assertTrue(hostIdx >= 0 && mixinIdx >= 0, "Both processors must be present");
136 		assertTrue(hostIdx < mixinIdx,
137 			"Host's own HostRp1 must precede the folded MixinRp1 (append semantics)");
138 	}
139 }
