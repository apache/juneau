20  import static org.junit.jupiter.api.Assertions.*;

22  import java.nio.charset.*;

24  import org.apache.juneau.*;
25  import org.junit.jupiter.api.*;

27  /**
28   * Coverage for op-level annotation overrides on:
29   * <ul>
30   *   <li>{@link RestOpContext#getDefaultCharset()} — exercises the {@code v.isPresent()} branch in
31   *       {@code findDefaultCharset()}.
32   *   <li>{@link RestOpContext#getMaxInput()} — exercises the {@code v.isPresent()} branch in
33   *       {@code findMaxInput()}.
34   *   <li>{@link RestOpContext#createSession} — exercises the explicit-debug branch in
35   *       {@code findDebugEnablement()} via {@code @RestGet(debug="true")}.
36   * </ul>
37   *
38   * <p>These cases were not covered by {@code NoInherit_Test}, which only exercises class-level
39   * {@code @Rest} fallbacks and the {@code noInherit} skip path.
40   */
41  class RestOpContext_OpLevelOverrides_Test extends TestBase {
42  
43  	private static RestContext build(Class<?> c) throws Exception {
44  		var o = c.getDeclaredConstructor().newInstance();
45  		return new RestContext(new RestContext.Args(c, null, null, () -> o, "", null, null, null, RestContext.ContextKind.ROOT)).postInit().postInitChildFirst();
46  	}
47  
48  	private static RestOpContext op(RestContext ctx) {
49  		return ctx.getRestOperations().getOpContexts().get(0);
50  	}
51  
52  	@Rest(defaultCharset = "UTF-8")
53  	public static class A {
54  		@RestGet(defaultCharset = "ISO-8859-1")
55  		public void get() { /* intentionally empty */ }
56  	}
57  
58  	@Test void a01_opLevelDefaultCharset_overridesClassLevel() throws Exception {
59  		var ctx = build(A.class);
60  		assertEquals(StandardCharsets.ISO_8859_1, op(ctx).getDefaultCharset(),
61  			"@RestGet(defaultCharset=...) should override class-level @Rest(defaultCharset=...)");
62  	}
63  
64  	@Rest(maxInput = "1M")
65  	public static class B {
66  		@RestOp(method = "get", maxInput = "9M")
67  		public void get() { /* intentionally empty */ }
68  	}
69  
70  	@Test void b01_opLevelMaxInput_overridesClassLevel() throws Exception {
71  		var ctx = build(B.class);
72  		assertEquals(parseLongWithSuffix("9M"), op(ctx).getMaxInput(),
73  			"@RestOp(maxInput=...) should override class-level @Rest(maxInput=...)");
74  	}
75  
76  	@Rest
77  	public static class C {
78  		@RestGet(debug="true")
79  		public void get() { /* intentionally empty */ }
80  	}
81  
82  	@Test void c01_opLevelDebug_buildsSuccessfully() throws Exception {
83  		// Building the context invokes findDebugEnablement() in the constructor; a non-empty
84  		// debug attribute exercises the "v.isPresent()" branch in that finder.
85  		var ctx = build(C.class);
86  		assertNotNull(op(ctx));
87  	}
88  }
