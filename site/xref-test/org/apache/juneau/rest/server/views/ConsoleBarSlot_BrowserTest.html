17  package org.apache.juneau.rest.server.views;
18  
19  import static org.apache.juneau.rest.server.views.ConsoleBrowserFixture.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.util.*;
23  
24  import org.apache.juneau.*;
25  import org.junit.jupiter.api.*;
26  import org.junit.jupiter.api.condition.*;
27  
28  /**
29   * Bar slots hydrate through {@code JuneauConsole.chrome}: once after mount (a document scan), and again when
30   * {@code juneau-views.js} inserts a row-detail panel and calls {@code enhanceChromeInPanel}. Without the shell the
31   * slot stays as served and views logs a console.error.
32   */
33  @EnabledIfSystemProperty(named=ConsoleBrowserFixture.GATE, matches="true",
34  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -f juneau-rest/juneau-rest-server-views/pom.xml test`")
35  class ConsoleBarSlot_BrowserTest extends TestBase {
36  
37  	private static final String SEG = """
38  		{"contractVersion":"1","title":"Bar","cards":[{"id":"jc-seg-1","type":"html","template":"jc-seg-1","bare":true}]}
39  		""";
40  
41  	private static final String SEG_TEMPLATE = """
42  		<template data-card="jc-seg-1"><div data-juneau-bar-slot="b1"><span data-juneau-badge="pending">0</span></div>
43  		<script type="application/json" id="juneau-bar:b1">{"contractVersion":"1","badges":{"pending":7}}</script></template>
44  		""";
45  
46  	private static final String INSERT_PANEL = """
47  		<script src="/views/juneau-views.js"></script>
48  		<script>
49  		window.addEventListener('load', function () {
50  			var panel = document.createElement('div');
51  			panel.className = 'juneau-view-detail-panel';
52  			panel.innerHTML = '<div data-juneau-bar-slot="b2"><span data-juneau-badge="n" data-juneau-badge-max="9">0</span></div>'
53  				+ '<script type="application/json" id="juneau-bar:b2">{"contractVersion":"1","badges":{"n":12}}<\\/script>';
54  			document.body.appendChild(panel);
55  			window.__probe = { enhanced: window.JuneauViews.init.enhanceChromeInPanel(panel) };
56  		});
57  		</script>
58  		""";
59  
60  	private static final String BADGE_B2 = "[data-juneau-bar-slot=\"b2\"] [data-juneau-badge=\"n\"]";
61  
62  	private static Map<String,Map<String,Object>> report;
63  
64  	@BeforeAll
65  	static void probe() throws Exception {
66  		var viewsOnly = "<!DOCTYPE html><html><head><meta charset=\"utf-8\"><link rel=\"icon\" href=\"data:,\"></head><body>\n"
67  			+ INSERT_PANEL + "\n</body></html>";
68  		report = ConsoleBrowserFixture.create("bar-slot")
69  			.page("scan", ORIGIN + "/scan", contractPage(SEG, SEG_TEMPLATE),
70  				"badge", "[data-juneau-bar-slot=\"b1\"] [data-juneau-badge=\"pending\"]")
71  			.page("insert", ORIGIN + "/insert", contractPage("{\"contractVersion\":\"1\",\"title\":\"Bar\",\"cards\":[]}", "", "", INSERT_PANEL),
72  				"badge", BADGE_B2)
73  			.page("no-shell", ORIGIN + "/no-shell", viewsOnly, "badge", BADGE_B2)
74  			.run();
75  	}
76  
77  	@Test void a01_mountScansTheDocument() {
78  		var r = report.get("scan");
79  		assertNoShellErrors(r);
80  		assertEquals("7", queries(r).get("badge"));
81  	}
82  
83  	@Test void a02_insertedPanel_hydratesThroughEnhanceChromeInPanel() {
84  		var r = report.get("insert");
85  		assertNoShellErrors(r);
86  		assertEquals(Boolean.TRUE, ConsoleBrowserFixture.probe(r).get("enhanced"));
87  		assertEquals("9+", queries(r).get("badge"), "count 12 is clamped by data-juneau-badge-max=9");
88  	}
89  
90  	@Test void a03_withoutTheShell_slotStaysAndViewsLogs() {
91  		var r = report.get("no-shell");
92  		assertEquals(Boolean.FALSE, ConsoleBrowserFixture.probe(r).get("enhanced"));
93  		assertEquals("0", queries(r).get("badge"));
94  		assertTrue(((List<?>)r.get("consoleErrors")).contains("[juneau-views] bar slot not hydrated: juneau-console.js is not loaded"),
95  			() -> r.get("consoleErrors").toString());
96  	}
97  }
