28   * Node behavioral checks for the job-log card in a row's detail panel: the {@code job-log} populator and

34  	private static Map<?,?> report() {
35  		var r = RegionsHarness.reportWith("job-log-region.cjs", ViewsMixin.HELPERS_JS_RESOURCE, ConsoleChromeMixin.JOBLOG_JS_RESOURCE);
36  		assumeTrue(r != null, "node (or job-log-region.cjs) not available - JS layer skipped");
37  		return r;
38  	}
39  
40  	private static Map<?,?> part(String key) {
41  		var s = (Map<?,?>)report().get(key);
42  		assertNotNull(s, () -> "section '" + key + "' missing from the harness output");
43  		assertNull(s.get("crash"), () -> "section '" + key + "' crashed: " + s.get("crash"));
44  		return s;
45  	}
46  
47  	private static Map<?,?> m(Map<?,?> m, String key) { return (Map<?,?>)m.get(key); }
48  	private static int n(Map<?,?> m, String key) { return ((Number)m.get(key)).intValue(); }

104 	@Test void r05_presets_ownRowPropertyAtSubmit_inheritedAndObjectRefused_nullSent() {
105 		var r = part("sources");
106 		assertEquals(Map.of("species", "cat"), m(r, "own").get("sent"));
107 		assertEquals(Map.of("species", "dog"), m(r, "atSubmit").get("sent"));
108 		for (var k : List.of("proto", "ctor", "toStr", "object")) {
109 			var c = m(r, k);
110 			assertNull(c.get("sent"), k);
111 			assertEquals(1, n(c, "e110"), k);
112 			assertTrue(String.valueOf(c.get("banner")).startsWith("E-JS-110: "), k);
113 		}
114 		assertEquals(Map.of("p", "d"), m(r, "protoDefault").get("sent"));
115 		assertEquals(Map.of("o", 7), m(r, "objectDefault").get("sent"));
116 		var nul = new LinkedHashMap<String,Object>();
117 		nul.put("n", null);
118 		assertEquals(nul, m(r, "nul").get("sent"));
119 		// A visible preset is a prefill; a value it cannot use leaves the input empty, with no error.
120 		assertEquals(List.of("cat", ""), m(r, "prefill").get("values"));
121 		assertEquals(0, n(m(r, "prefill"), "e110"));
