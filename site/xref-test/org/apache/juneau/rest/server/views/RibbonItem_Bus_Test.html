17  package org.apache.juneau.rest.server.views;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.*;
24  import org.apache.juneau.marshall.marshaller.*;
25  import org.junit.jupiter.api.*;
26  
27  class RibbonItem_Bus_Test extends TestBase {
28  
29  	@Test void a01_publish() {
30  		var i = RibbonItem.publish("Focus east", "app.region-picked", Map.of("region", "east"));
31  		assertEquals("{type:'publish',title:'Focus east',topic:'app.region-picked',payload:{region:'east'}}", Json5.DEFAULT.write(i.toMap()));
32  		assertEquals("{type:'publish',title:'Go live',topic:'app.go-live'}",
33  			Json5.DEFAULT.write(RibbonItem.publish("Go live", "app.go-live", null).toMap()));
34  	}
35  
36  	@Test void a02_target_onRefresh() {
37  		assertEquals("{type:'refresh',title:'Refresh tasks',target:'tasks'}",
38  			Json5.DEFAULT.write(RibbonItem.refresh().title("Refresh tasks").target("tasks").toMap()));
39  	}
40  
41  	@Test void a03_target_onEveryTargetableType() {
42  		for (var i : List.of(RibbonItem.collapseAll(), RibbonItem.pausePolling(),
43  				RibbonItem.option("mine").param("owner").value("me"),
44  				RibbonItem.optionGroup("phase", RibbonItem.option("all").column("phase").value("$eq(x)"))))
45  			assertEquals("tasks", i.target("tasks").toMap().getString("target"));
46  	}
47  
48  	@Test void b01_target_notAccepted() {
49  		var target2 = RibbonItem.dialog("add").target("tasks");
50  		var e = assertThrows(IllegalArgumentException.class, target2::toMap);
51  		assertEquals("RibbonItem dialog 'add' does not accept 'target'.", e.getMessage());
52  	}
53  
54  	@Test void b02_target_onGroupMember_notAccepted() {
55  		var g = RibbonItem.optionGroup("phase", RibbonItem.option("all").column("phase").value("$eq(x)").target("tasks"));
56  		var e = assertThrows(IllegalArgumentException.class, g::toMap);
57  		assertEquals("RibbonItem optionGroup 'phase' member 'all' does not accept 'target'.", e.getMessage());
58  	}
59  
60  	@Test void b03_target_badId() {
61  		var target3 = RibbonItem.refresh();
62  		var e = assertThrows(IllegalArgumentException.class, () -> target3.target("1x"));
63  		assertEquals("RibbonItem target '1x' must match ^[A-Za-z][A-Za-z0-9_-]{0,63}$.", e.getMessage());
64  	}
65  
66  	@Test void b04_publish_needsTitleAndTopic() {
67  		assertEquals("RibbonItem publish requires a title.",
68  			assertThrows(IllegalArgumentException.class, () -> RibbonItem.publish("", "app.x", null)).getMessage());
69  		assertEquals("RibbonItem publish 'T' requires a topic.",
70  			assertThrows(IllegalArgumentException.class, () -> RibbonItem.publish("T", " ", null)).getMessage());
71  	}
72  
73  	@Test void b05_publish_rejectsTarget() {
74  		var target4 = RibbonItem.publish("T", "app.x", null).target("tasks");
75  		var e = assertThrows(IllegalArgumentException.class, target4::toMap);
76  		assertTrue(e.getMessage().endsWith("does not accept 'target'."), e.getMessage());
77  	}
78  }
