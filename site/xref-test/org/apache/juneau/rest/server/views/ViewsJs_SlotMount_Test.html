39  	private static Map<?,?> report() {
40  		var r = RegionsHarness.report("slot-mount.cjs");
41  		assumeTrue(r != null, "node not available or slot-mount.cjs not found - behavioral layer skipped");
42  		return r;
43  	}
44  
45  	private static void assertAllTrue(Map<?,?> r, String... keys) {
46  		for (var k : keys)
47  			assertEquals(true, r.get(k), () -> k + " -> " + r.get(k) + " in " + r);
48  	}
49  
50  	private static String viewsJs() throws IOException {
51  		try (var in = ViewsMixin.class.getResourceAsStream(ViewsMixin.VIEWS_JS_RESOURCE)) {
52  			assertNotNull(in, () -> "missing classpath resource: " + ViewsMixin.VIEWS_JS_RESOURCE);
53  			return new String(in.readAllBytes(), UTF_8);
54  		}
55  	}

57  	@Test void a01_sourceShape_slotInitIsNotSidecarPath() throws Exception {
58  		var body = viewsJs();
59  		assertTrue(body.contains("JUNEAU_SLOT_CONTRACT_VERSION = \"1\""), body);
60  		assertTrue(body.contains("NS.SLOT_CONTRACT_VERSION = JUNEAU_SLOT_CONTRACT_VERSION"), body);
61  		assertTrue(body.contains("initTableFromDef: initTableFromDef"), body);
62  		assertTrue(body.contains("mountTableSlot: mountTableSlot"), body);
63  		assertTrue(body.contains("return Promise.resolve(initTableFromDef(table, viewDef, extras))"), body);
64  		assertTrue(body.contains("slot.closest(\"[data-juneau-csrf]\")"), body);
65  		assertFalse(body.contains("data-ssc-csrf"), "slot CSRF copy must not read data-ssc-csrf-*");
66  		assertTrue(body.contains("dataset.juneauRegionDeclared"), body);
67  		assertTrue(body.contains("dataset.juneauRegionType"), body);
68  		var paintStart = body.indexOf("function paintSlotTable(");
69  		assertTrue(paintStart >= 0, body);
70  		var paintEnd = body.indexOf("function copyCsrfOntoTable(", paintStart);
71  		assertTrue(paintEnd > paintStart, body);
72  		var paintFn = body.substring(paintStart, paintEnd);
73  		assertFalse(paintFn.contains("initTable(table)"), "slot paint must not load a VIEW_META sidecar");
74  		assertFalse(paintFn.contains("data-juneau-region-meta"), paintFn);
75  		var buildStart = body.indexOf("function buildDetailTemplate(");
76  		assertTrue(buildStart >= 0, body);
77  		var buildEnd = body.indexOf("\n\tfunction buildDetailHeader(", buildStart);
78  		assertTrue(buildEnd > buildStart, body);
79  		var buildFn = body.substring(buildStart, buildEnd);
80  		assertTrue(buildFn.contains("const dest = tpl.content"), buildFn);
81  		assertTrue(buildFn.contains("dest.appendChild"), buildFn);
82  		assertFalse(buildFn.contains("tpl.appendChild"),
83  			() -> "detail chrome must go into tpl.content, the fragment expand clones: " + buildFn);
84  	}

174 	@Test void b17_quickStatPainters_useCssomNotStyleAttribute() throws Exception {
175 		var body = viewsJs();
176 		var start = body.indexOf("function paintStatBar(");
177 		var end = body.indexOf("function buildDetailTemplate(", start);
178 		assertTrue(start >= 0 && end > start, body);
179 		var fn = body.substring(start, end);
180 		assertTrue(fn.contains("fill.style.width = pct + \"%\""), fn);
181 		assertTrue(fn.contains("fill.style.flexGrow = String(count)"), fn);
182 		assertFalse(fn.contains("setAttribute(\"style\""), "painters must size through CSSOM, not a style attribute");
183 		assertFalse(fn.contains("cssText"), fn);
184 		assertFalse(fn.contains("inner" + "HTML"), "painters must not use a raw HTML-sink assignment");
185 	}
