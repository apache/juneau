32  /**
33   * The wiring layer of the ribbon parity contract: boots a REAL DataTable (jQuery + DataTables from the js-tests
34   * npm install) with the real shipped scripts and checks that the shared {@code ribbon-corpus.json} semantics reach
35   * the wire in server mode and the DOM in client mode.
36   *
37   * <p>
38   * The pure layers are covered by {@link RibbonCorpus_Test} (Java) and {@link RibbonCorpus_Parity_Test} (JS); this
39   * class proves the two callers - {@code buildServerAjax} and {@code installClientRibbonFilter} - actually feed them.
40   * Nothing touches the network: the page and both data endpoints are answered by the prober's request routing.
41   *
42   * <p>
43   * Disabled unless the {@value ConfigPersistence_BrowserTest#GATE} system property is set, which only the module's
44   * opt-in {@code js-tests} Maven profile does.
45   */

59  	@BeforeAll
60  	static void probe() throws Exception {
61  		var dir = Path.of(requiredProperty("juneau.jsTests.dir"));
62  		var harness = Path.of(requiredProperty("juneau.jsTests.harness")).getParent().resolve("ribbon-parity-browser.cjs");
63  		var nodeModules = dir.resolve("node_modules");
64  
65  		// Server table: corpus case jrm-dropped-only, persisted ON, so the FIRST request is already filtered.
66  		// Client table: corpus case foundry-review-phase with a default member, over the corpus rows.
67  		var serverDef = "{id:'srv',contractVersion:'5',dataMode:'server',dataUrl:'http://corpus.test/server-rows',"
68  			+ "columns:[{data:'id',title:'Id',searchable:false},{data:'status',title:'Status'}],"
69  			+ "ribbon:[{type:'option',id:'dropped-only',title:'Dropped only',column:'status',value:'$eq(DROPPED)',persist:true}]}";
70  		var clientDef = "{id:'cli',contractVersion:'5',dataMode:'client',dataUrl:'http://corpus.test/client-rows',"
71  			+ "columns:[{data:'id',title:'Id'},{data:'status',title:'Status'},{data:'stream',title:'Stream'},"
72  			+ "{data:'phase',title:'Phase'},{data:'isNew',title:'New'}],"
73  			+ "ribbon:[{type:'optionGroup',id:'reviewStateBucket',default:'reviewed',options:["
74  			+ "{id:'all',title:'All'},"
75  			+ "{id:'waiting',title:'Waiting',column:'phase',value:'$in(Waiting,\"Partially reviewed\")'},"
76  			+ "{id:'reviewed',title:'Reviewed',column:'phase',value:'$in(\"Ready to push\",\"Committed, not pushed\",Completed)'}]}]}";
77  
78  		var fixture = "<!DOCTYPE html><html><head><meta charset=\"utf-8\"></head><body>\n"
79  			+ "<table id=\"srv\" data-juneau-view=\"srv\"></table>\n<table id=\"cli\" data-juneau-view=\"cli\"></table>\n"
80  			+ "<script>localStorage.setItem('juneau.view.srv.ribbon.dropped-only','true');</script>\n"
81  			+ script(Files.readString(nodeModules.resolve("jquery/dist/jquery.min.js")))
82  			+ script(Files.readString(nodeModules.resolve("datatables.net/js/dataTables.min.js")))
83  			+ script(resource(ViewsMixin.DATATABLES_JS_RESOURCE))
84  			+ script(resource(ViewsMixin.RENDERS_JS_RESOURCE))
85  			+ script(resource(ViewsMixin.VIEWS_JS_RESOURCE))
86  			+ script(resource(ViewsMixin.URLSTATE_JS_RESOURCE))
87  			+ script(resource(ViewsMixin.SEARCH_JS_RESOURCE))
88  			+ script(resource(ViewsMixin.PAGESTATE_JS_RESOURCE))
89  			+ script(resource(ViewsMixin.CONFIG_JS_RESOURCE))
90  			+ script(resource(ViewsMixin.RIBBON_JS_RESOURCE))
91  			+ script("JuneauViews.init.initTableFromDef(document.getElementById('srv')," + serverDef + ");\n"
92  				+ "JuneauViews.init.initTableFromDef(document.getElementById('cli')," + clientDef + ");\n"
93  				+ "window.__corpusReady = true;")
94  			+ "</body></html>";
95  		var fixtureFile = Files.createDirectories(dir.resolve("fixtures")).resolve("ribbon-parity.html");
96  		Files.write(fixtureFile, fixture.getBytes(UTF_8));
97  
98  		report = Json.to(run(dir, harness, fixtureFile), Map.class);
99  	}

101 	private static String script(String body) {

105 	private static String requiredProperty(String name) {
106 		var v = System.getProperty(name);
107 		assertNotNull(v, () -> "-D" + name + " not set; the js-tests profile is responsible for providing it");
108 		return v;
109 	}
110 
111 	/** Runs the prober, failing with its stderr attached (its exit code alone is not a diagnosis). */
112 	private static String run(Path dir, Path harness, Path fixture) throws Exception {
113 		var cmd = List.of(System.getProperty("juneau.jsTests.node", "node"), harness.toString(), fixture.toString());
114 		var stdout = dir.resolve("ribbon-parity-stdout.json");
115 		var stderr = dir.resolve("ribbon-parity-stderr.txt");
116 		var pb = new ProcessBuilder(cmd).redirectOutput(stdout.toFile()).redirectError(stderr.toFile());
117 		pb.environment().put("NODE_PATH", dir.resolve("node_modules").toString());
118 		pb.environment().put("PLAYWRIGHT_BROWSERS_PATH", requiredProperty("juneau.jsTests.browsers"));
119 		pb.environment().put("CORPUS_ROWS", Json.of(RibbonCorpus_Test.corpus().rows));
120 
121 		var p = pb.start();
122 		if (! p.waitFor(3, TimeUnit.MINUTES)) {
123 			p.destroyForcibly();
124 			fail("prober did not finish within 3m; stderr:\n" + quietRead(stderr));
125 		}
126 		assertEquals(0, p.exitValue(), () -> "prober exited non-zero; stderr:\n" + quietRead(stderr));
127 		return Files.readString(stdout);
128 	}
129 
130 	private static String quietRead(Path p) {
131 		try {
132 			return Files.readString(p);
133 		} catch (IOException e) {
134 			return "<unreadable: " + e + ">";
135 		}
136 	}
