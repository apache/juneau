30  /**
31   * The row-selection contract: {@link WritePermit}, {@link SelectionDef}, and {@link BulkMutateDef} &mdash; the API-shape half
32   * of the separability guarantee.
33   *
34   * <p>
35   * These are NOT prose assertions: {@link BulkMutateDef#create(WritePermit, SelectionDef)} is the ONLY way to obtain
36   * a {@link BulkMutateDef} (a private constructor + reflection below prove there is no other, narrower factory), and
37   * both parameters are non-null-checked at that single call site &mdash; so a bulk-mutate opt-in that omits a
38   * {@link WritePermit} cannot compile against any DIFFERENT, permit-less overload (there isn't one) and cannot
39   * construct one by passing {@code null} either (it throws).
40   */

156 	/**
157 	 * The API-SHAPE proof, not a prose claim: {@code create(...)} is the ONLY public factory on
158 	 * {@link BulkMutateDef} and its signature is EXACTLY {@code (WritePermit, SelectionDef)} - there is no
159 	 * narrower overload a caller could reach for that skips the permit. Combined with c01 (a null permit throws)
160 	 * and c05/c06 below (no public constructor, no setter for the permit/selection fields), this proves a
161 	 * bulk-mutate opt-in cannot be constructed without supplying a real {@link WritePermit} - by shape, not by
162 	 * javadoc sentence.
163 	 */
164 	@Test void c05_onlyOneCreateOverload_andItsSignatureRequiresWritePermitFirst() {
165 		var createMethods = Arrays.stream(BulkMutateDef.class.getMethods())
166 			.filter(m -> eq(m.getName(), "create"))
167 			.toList();
168 		assertEquals(1, createMethods.size(), () -> "expected exactly one create(...) overload: " + createMethods);
169 		var m = createMethods.get(0);
170 		assertArrayEquals(new Class<?>[] { WritePermit.class, SelectionDef.class }, m.getParameterTypes(),
171 			() -> "BulkMutateDef.create(...) must require (WritePermit, SelectionDef), in that order: " + m);
172 	}

179 	/**
180 	 * The permit/selection fields backing separability are NOT public mutable fields either - a caller cannot
181 	 * bypass the constructor-time requirement by building a bare instance (impossible; no public ctor) and later
182 	 * poking a permit in via reflection-free code.
183 	 */
184 	@Test void c07_permitAndSelectionAreNotPublicSettableFields() {
185 		for (var f : BulkMutateDef.class.getDeclaredFields()) {
186 			if (f.getType() == WritePermit.class || f.getType() == SelectionDef.class)
187 				assertFalse(Modifier.isPublic(f.getModifiers()), () -> "must not be a public field: " + f);
188 		}
189 	}
