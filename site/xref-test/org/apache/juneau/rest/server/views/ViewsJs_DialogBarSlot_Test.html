34  /**
35   * Node-sandbox coverage for the dialog bar slot painter in {@code juneau-views.js} ({@code dialog-bar-slot.cjs}): a bar
36   * badge's {@code data-juneau-badge-tone} attribute is set only for a status-tone wire token, and nothing else.
37   *
38   * <p>
39   * Gated on {@code node} being on {@code PATH} (skipped otherwise - no {@code -Pjs-tests} required).
40   */
41  class ViewsJs_DialogBarSlot_Test extends TestBase {
42  
43  	private static Map<?,?> report;
44  
45  	@BeforeAll
46  	static void probeIfNodeAvailable() throws Exception {
47  		if (!nodeAvailable())
48  			return;
49  		var harness = locateHarness();
50  		if (harness == null)
51  			return;
52  		var views = Files.createTempFile("juneau-views-", ".js");
53  		var console = Files.createTempFile("juneau-console-", ".js");
54  		var renders = Files.createTempFile("juneau-renders-", ".js");
55  		try {
56  			Files.writeString(views, asset(ViewsMixin.VIEWS_JS_RESOURCE, ViewsMixin.class), UTF_8);
57  			Files.writeString(console, asset(ConsoleChromeMixin.CONSOLE_JS_RESOURCE, ConsoleChromeMixin.class), UTF_8);
58  			Files.writeString(renders, asset(ViewsMixin.RENDERS_JS_RESOURCE, ViewsMixin.class), UTF_8);
59  			report = Json.to(runNode(harness, views, console, renders), Map.class);
60  		} finally {
61  			Files.deleteIfExists(views);
62  			Files.deleteIfExists(console);
63  			Files.deleteIfExists(renders);
64  		}
65  	}
66  
67  	private static String asset(String name, Class<?> anchor) throws IOException {
68  		try (var in = anchor.getResourceAsStream(name)) {
69  			assertNotNull(in, () -> "missing classpath resource: " + name);
70  			return new String(in.readAllBytes(), UTF_8);
71  		}
72  	}
73  
74  	private static boolean nodeAvailable() {
75  		try {
76  			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
77  			if (!p.waitFor(5, TimeUnit.SECONDS)) {
78  				p.destroyForcibly();
79  				return false;
80  			}
81  			return p.exitValue() == 0;
82  		} catch (Exception e) {
83  			return false;
84  		}
85  	}
86  
87  	private static Path locateHarness() {
88  		var basedir = System.getProperty("basedir");
89  		if (basedir != null) {
90  			var p = Path.of(basedir, "src/test/js/dialog-bar-slot.cjs");
91  			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
92  		}
93  		for (var rel : List.of(
94  			"src/test/js/dialog-bar-slot.cjs",
95  			"juneau-rest/juneau-rest-server-views/src/test/js/dialog-bar-slot.cjs"
96  		)) {
97  			var p = Path.of(rel);
98  			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
99  		}
100 		return null;
101 	}
102 
103 	private static String runNode(Path harness, Path views, Path console, Path renders) throws Exception {
104 		var stdout = Files.createTempFile("dialog-bar-stdout-", ".json");
105 		var stderr = Files.createTempFile("dialog-bar-stderr-", ".txt");
106 		try {
107 			var p = new ProcessBuilder(List.of("node", harness.toString(), views.toString(), console.toString(), renders.toString()))
108 				.redirectOutput(stdout.toFile())
109 				.redirectError(stderr.toFile())
110 				.start();
111 			if (!p.waitFor(30, TimeUnit.SECONDS)) {
112 				p.destroyForcibly();
113 				fail("dialog-bar-slot.cjs did not finish within 30s; stderr:\n" + quietRead(stderr));
114 			}
115 			if (p.exitValue() != 0)
116 				fail("dialog-bar-slot.cjs exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr)
117 					+ "\nstdout:\n" + quietRead(stdout));
118 			return Files.readString(stdout, UTF_8);
119 		} finally {
120 			Files.deleteIfExists(stdout);
121 			Files.deleteIfExists(stderr);
122 		}
123 	}
124 
125 	private static String quietRead(Path p) {
126 		try { return Files.readString(p, UTF_8); }
127 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
128 	}
129 
130 	private static Map<?,?> r() {
131 		assumeTrue(report != null, "node not available or dialog-bar-slot.cjs not found - dialog bar slot harness skipped");
132 		return report;
133 	}

138 	}

147 		assertBean(r, "tone_bogus,tone_retiredWarn,tone_retiredAccent,tone_absent", "<null>,<null>,<null>,<null>");
148 	}
