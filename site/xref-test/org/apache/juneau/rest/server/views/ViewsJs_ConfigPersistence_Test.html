50  		try (var in = ViewsMixin.class.getResourceAsStream(CONFIG_JS_RESOURCE)) {
51  			assertNotNull(in, () -> "missing classpath resource: " + CONFIG_JS_RESOURCE);
52  			return new String(in.readAllBytes(), UTF_8);

56  	/**
57  	 * Extracts a named function/method's body via BALANCED brace matching (from the first {@code {} after
58  	 * {@code signature} to its true matching close), skipping braces inside string/template literals and
59  	 * comments.  A plain {@code indexOf("\n\t}", start)} (the sibling {@code ViewsJs_RowActions_Test}/
60  	 * {@code ViewsMixin_Serving_Test} helper) only works for functions indented exactly one tab; several of this
61  	 * file's functions/methods are nested two or three tabs deep inside a provider factory, where that shortcut
62  	 * would overshoot into unrelated trailing code.
63  	 */
64  	private static String functionBody(String body, String signature) {
65  		var start = body.indexOf(signature);
66  		assertTrue(start >= 0, () -> "'" + signature + "' not found:\n" + body);
67  		var i = body.indexOf('{', start);
68  		assertTrue(i >= 0, () -> "'" + signature + "' has no opening brace:\n" + body);
69  		var depth = 0;
70  		var j = i;
71  		for (; j < body.length(); j++) {
72  			var c = body.charAt(j);
73  			if (c == '{') {
74  				depth++;
75  			} else if (c == '}') {
76  				depth--;
77  				if (depth == 0) { j++; break; }
78  			} else if (c == '"' || c == '\'' || c == '`') {
79  				var quote = c;
80  				j++;
81  				while (j < body.length() && body.charAt(j) != quote) { if (body.charAt(j) == '\\') j++; j++; }
82  			} else if (c == '/' && j + 1 < body.length() && body.charAt(j + 1) == '/') {
83  				while (j < body.length() && body.charAt(j) != '\n') j++;
84  			} else if (c == '/' && j + 1 < body.length() && body.charAt(j + 1) == '*') {
85  				j = body.indexOf("*/", j) + 1;
86  			}
87  		}
88  		return body.substring(start, j);
89  	}
90  
91  	/**
92  	 * Extracts an ENTIRE top-level factory function's body (from `function <name>(` to the next `// ====` section
93  	 * divider) - needed so a nested object-literal method signature is searched within the intended
94  	 * factory only.
95  	 */
96  	private static String factoryBody(String body, String factoryName) {
97  		var start = body.indexOf("function " + factoryName + "(");
98  		assertTrue(start >= 0, () -> "'" + factoryName + "' not found:\n" + body);
99  		var end = body.indexOf("\n\t// ====", start);
100 		return body.substring(start, end < 0 ? body.length() : end);
101 	}
