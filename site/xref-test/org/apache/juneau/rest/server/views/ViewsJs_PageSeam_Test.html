17  package org.apache.juneau.rest.server.views;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.apache.juneau.*;
22  import org.apache.juneau.rest.mock.classic.*;
23  import org.apache.juneau.rest.server.*;
24  import org.junit.jupiter.api.*;
25  
26  /**
27   * Lazy-init seam test for {@code juneau-views.js}.
28   *
29   * <p>
30   * Asserts on the served script text: {@code initTable} must be reachable off the public {@code NS.init} namespace
31   * (previously private), and {@code initAll} must skip tables scoped under a {@code [data-juneau-page]} ancestor so
32   * a host can lazy-init via {@code initTable}/{@code initTableFromDef} (HTML-slot {@code regions.mount} uses the
33   * latter).  These text assertions are always-on tripwires for the seam's <i>shape</i>.
34   */
35  @SuppressWarnings({
36  	"resource" // Closeable test fixtures held in static fields; lifecycle managed by the test/framework, not a real leak.
37  })
38  class ViewsJs_PageSeam_Test extends TestBase {
39  
40  	@Rest(mixins=ViewsMixin.class)
41  	public static class WithMixin extends org.apache.juneau.rest.server.servlet.BasicRestServlet {
42  		private static final long serialVersionUID = 1L;
43  	}
44  
45  	private static final MockRestClient c = MockRestClient.buildLax(WithMixin.class);
46  
47  	@Test void a01_initTableIsPubliclyExposedOnNsInit() throws Exception {
48  		var body = c.get(ViewsMixin.VIEWS_JS_PATH).run().assertStatus(200).getContent().asString();
49  		assertTrue(body.contains("initTable: initTable"), body);
50  	}
51  
52  	@Test void a02_initAllSkipsTablesScopedUnderAJuneauPageShell() throws Exception {
53  		var body = c.get(ViewsMixin.VIEWS_JS_PATH).run().assertStatus(200).getContent().asString();
54  		var initAllStart = body.indexOf("function initAll(");
55  		assertTrue(initAllStart >= 0, () -> "initAll not found:\n" + body);
56  		var initAllEnd = body.indexOf("\n\t}", initAllStart);
57  		var initAllBody = body.substring(initAllStart, initAllEnd < 0 ? body.length() : initAllEnd);
58  		assertTrue(initAllBody.contains("data-juneau-page"), initAllBody);
59  	}
60  
61  	@Test void a03_initTableRemainsIdempotentGuardIntact() throws Exception {
62  		// Regression: the seam must not remove the existing isDataTable idempotent re-entry guard.
63  		var body = c.get(ViewsMixin.VIEWS_JS_PATH).run().assertStatus(200).getContent().asString();
64  		assertTrue(body.contains("isDataTable(table)"), body);
65  	}
66  }
