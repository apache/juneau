17  package org.apache.juneau.rest.server.views;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.*;
24  import org.apache.juneau.marshall.marshaller.*;
25  import org.junit.jupiter.api.*;
26  
27  /** Unit tests for the {@link VisibilityRule} builder, its {@code test(rules, map)} evaluator, and its wire shape. */
28  class VisibilityRule_Test extends TestBase {
29  
30  	@Test void a01_eqSerializesFieldOpValue() {
31  		var r = VisibilityRule.when("status").eq("open");
32  		assertEquals("status", r.field);
33  		assertEquals("eq", r.op);
34  		assertEquals("open", r.value);
35  	}
36  
37  	@Test void a02_presentAndAbsentOmitValue() {
38  		assertNull(VisibilityRule.when("name").present().value);
39  		assertNull(VisibilityRule.when("name").absent().value);
40  	}
41  
42  	@Test void a03_inWrapsVarargsAsAList() {
43  		assertEquals(List.of("open", "acked"), VisibilityRule.when("status").in("open", "acked").value);
44  	}
45  
46  	@Test void a04_whenRejectsBlankField() {
47  		assertThrows(IllegalArgumentException.class, () -> VisibilityRule.when(" "));
48  		assertThrows(IllegalArgumentException.class, () -> VisibilityRule.when(null));
49  	}
50  
51  	@Test void a05_eqRejectsNullValue() {
52  		var target = VisibilityRule.when("status");
53  		assertThrows(IllegalArgumentException.class, () -> target.eq(null));
54  		var target2 = VisibilityRule.when("status");
55  		assertThrows(IllegalArgumentException.class, () -> target2.contains(null));
56  	}
57  
58  	@Test void a06_testMatchesAndRule() {
59  		var rules = List.of(VisibilityRule.when("status").eq("open"), VisibilityRule.when("priority").in("p1", "p2"));
60  		assertTrue(VisibilityRule.test(rules, Map.of("status", "open", "priority", "p1")));
61  		assertFalse(VisibilityRule.test(rules, Map.of("status", "open", "priority", "p3")));
62  	}
63  
64  	@Test void a07_testFailsClosedOnMissingField() {
65  		assertFalse(VisibilityRule.test(List.of(VisibilityRule.when("status").ne("open")), Map.of()));
66  	}
67  
68  	@Test void a08_absentMatchesMissingAndNull() {
69  		var rules = List.of(VisibilityRule.when("name").absent());
70  		assertTrue(VisibilityRule.test(rules, Map.of()));
71  		var m = new HashMap<String,Object>();
72  		m.put("name", null);
73  		assertTrue(VisibilityRule.test(rules, m));
74  		assertFalse(VisibilityRule.test(rules, Map.of("name", "a")));
75  	}
76  
77  	@Test void a09_dottedPathResolvesNestedFacts() {
78  		var rules = List.of(VisibilityRule.when("facts.viewer.roles").contains("admin"));
79  		assertTrue(VisibilityRule.test(rules, Map.of("facts", Map.of("viewer", Map.of("roles", List.of("admin"))))));
80  	}
81  
82  	@Test void a10_emptyRuleListAlwaysMatches() {
83  		assertTrue(VisibilityRule.test(List.of(), Map.of()));
84  		assertTrue(VisibilityRule.test(null, Map.of()));
85  	}
86  
87  	@Test void a11_unknownOpThrows() {
88  		var r = VisibilityRule.when("status").eq("open");
89  		r.op = "bogus";
90  		Map<String,Object> arg = Map.of("status", "open");
91  		var rules = List.of(r);
92  		var ex = assertThrows(IllegalArgumentException.class, () -> VisibilityRule.test(rules, arg));
93  		assertTrue(ex.getMessage().contains("'bogus'"), ex.getMessage());
94  	}
95  
96  	@Test void a12_toMapOmitsValueWhenAbsent() {
97  		assertEquals(Map.of("field", "n", "op", "present"), VisibilityRule.when("n").present().toMap());
98  		assertEquals(Map.of("field", "s", "op", "in", "value", List.of("a")), VisibilityRule.when("s").in("a").toMap());
99  	}
100 
101 	@Test void b01_rowActionSerializesVisibleWhen() {
102 		var a = RowAction.create("escalate").visibleWhen(VisibilityRule.when("status").eq("open"));
103 		var json = Json.of(a);
104 		assertTrue(json.contains("visibleWhen"), json);
105 		assertTrue(json.contains("\"op\":\"eq\""), json);
106 	}
107 
108 	@Test void b02_rowActionOmitsVisibleWhenWhenUnset() {
109 		assertFalse(Json.of(RowAction.create("x")).contains("visibleWhen"));
110 	}
111 
112 	@Test void b03_regionAndFieldCarryVisibleWhenInContract() {
113 		var rule = VisibilityRule.when("facts.admin").eq(true);
114 		var region = RegionDef.create("r").visibleWhen(rule)
115 			.fields(RegionDef.Field.of("a").visibleWhen(rule), RegionDef.Field.of("b"));
116 		var m = region.toContractMap();
117 		assertEquals(List.of(rule.toMap()), m.get("visibleWhen"));
118 		var fields = (List<?>)m.get("fields");
119 		assertEquals(List.of(rule.toMap()), ((Map<?,?>)fields.get(0)).get("visibleWhen"));
120 		assertFalse(((Map<?,?>)fields.get(1)).containsKey("visibleWhen"));
121 	}
122 
123 	@Test void b04_enabledWhenRejectsCollectionOps() {
124 		assertThrows(IllegalArgumentException.class,
125 			() -> org.apache.juneau.rest.server.views.RowActionEnabledRule.of("s", org.apache.juneau.rest.server.widgets.Op.IN, "x", "r"));
126 	}
127 }
