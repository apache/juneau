17  package org.apache.juneau.rest.server.views;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  import static org.junit.jupiter.api.Assumptions.*;
21  
22  import java.util.*;
23  
24  import org.apache.juneau.*;
25  import org.junit.jupiter.api.*;
26  
27  /** Node behavioral checks for the {@code edit} renderer and the inline cell editor it drives. */
28  class ViewsJs_InlineEdit_Test extends TestBase {
29  
30  	private static Map<?,?> report;
31  
32  	private static Map<?,?> report() {
33  		if (report == null)
34  			report = NodeHarness.report("edit-renderer.cjs", ViewsMixin.RENDERS_JS_RESOURCE, ViewsMixin.VIEWS_JS_RESOURCE);
35  		assumeTrue(report != null, "node (or edit-renderer.cjs) not available - behavioral layer skipped");
36  		return report;
37  	}
38  
39  	private static Map<?,?> map(String key) {
40  		return (Map<?,?>)report().get(key);
41  	}
42  
43  	@Test void a01_rendererMarkupIsEscapedAndCarriesTheWiring() {
44  		assertEquals(Boolean.TRUE, report().get("registered"));
45  		for (var e : map("markup").entrySet())
46  			assertEquals(Boolean.TRUE, e.getValue(), () -> "markup." + e.getKey());
47  		assertEquals(Boolean.TRUE, report().get("noActionPlain"));
48  		assertEquals(Boolean.TRUE, report().get("hostileActionEscaped"));
49  		assertEquals(Boolean.TRUE, report().get("wrapClass"));
50  	}
51  
52  	@Test void b01_visibleWhenHidesTheMissingActionAndTheFailingRow() {
53  		assertEquals(Boolean.TRUE, report().get("unknownActionHides"));
54  		assertEquals(Boolean.TRUE, report().get("visibleWhenHides"));
55  	}
56  
57  	@Test void b02_enabledWhenDisablesWithTheReasonAndNeverOpens() {
58  		var d = map("enabledWhenDisables");
59  		assertEquals(Boolean.TRUE, d.get("disabled"));
60  		assertEquals("Closed rows cannot be reassigned", d.get("title"));
61  		assertEquals(Boolean.TRUE, report().get("disabledNoEditor"));
62  	}
63  
64  	@Test void c01_enterCommitsTheEditedColumnThroughTheAction() {
65  		assertEquals(Boolean.TRUE, report().get("opened"));
66  		assertEquals("old", report().get("prefill"));
67  		var c = map("commit");
68  		assertEquals(1, ((Number)c.get("fetches")).intValue());
69  		assertEquals("/x/7", c.get("url"));
70  		assertEquals("assign", c.get("action"));
71  		assertEquals("7", c.get("targetId"));
72  		assertEquals(Map.of("owner", "new"), c.get("fields"));
73  		assertEquals(Boolean.TRUE, report().get("closedAfterSuccess"));
74  		assertEquals("new", report().get("shownValue"));
75  		assertEquals(1, ((Number)report().get("reloadedAfterSuccess")).intValue());
76  	}
77  
78  	@Test void c02_escapeAndUnchangedValueSendNothing() {
79  		var e = map("escape");
80  		assertEquals(0, ((Number)e.get("fetches")).intValue());
81  		assertEquals(Boolean.TRUE, e.get("closed"));
82  		assertEquals("old", e.get("restored"));
83  		assertEquals(Boolean.TRUE, e.get("pencilBack"));
84  		var u = map("unchanged");
85  		assertEquals(0, ((Number)u.get("fetches")).intValue());
86  		assertEquals(Boolean.TRUE, u.get("closed"));
87  	}
88  
89  	@Test void c03_blurCommits() {
90  		var b = map("blurCommit");
91  		assertEquals(1, ((Number)b.get("fetches")).intValue());
92  		assertEquals(Map.of("owner", "blurred"), b.get("fields"));
93  	}
94  
95  	@Test void c04_commitOnAConfirmActionNeverPrompts() {
96  		var n = map("confirmActionNoPrompt");
97  		assertEquals(0, ((Number)n.get("enterDialogs")).intValue(), "Enter must not open a confirm dialog");
98  		assertEquals(0, ((Number)n.get("blurDialogs")).intValue(), "blur must not open a confirm dialog");
99  		assertEquals(0, ((Number)n.get("afterDialogs")).intValue());
100 		assertEquals(2, ((Number)n.get("fetches")).intValue());
101 		assertEquals(List.of(Map.of("owner", "enter-new"), Map.of("owner", "blur-new")), n.get("fields"));
102 	}
103 
104 	@Test void d01_failureKeepsTheEditorOpenWithTheErrorInline() {
105 		var f = map("failure");
106 		assertEquals(Boolean.TRUE, f.get("stillOpen"));
107 		assertEquals("Not allowed", f.get("error"));
108 		assertEquals(Boolean.TRUE, f.get("errorVisible"));
109 		assertEquals("alert", f.get("role"));
110 		assertEquals(Boolean.TRUE, f.get("enabled"));
111 		assertEquals(0, ((Number)f.get("reloads")).intValue());
112 		assertEquals(1, ((Number)report().get("failureBlurNoResubmit")).intValue(), "blur after a failure must not resubmit");
113 	}
114 
115 	@Test void d02_transportRefusalAlsoSurfacesInline() {
116 		var r = map("refusal");
117 		assertEquals(Boolean.TRUE, r.get("stillOpen"));
118 		assertEquals(Boolean.TRUE, r.get("hasMessage"));
119 	}
120 
121 	@Test void e01_selectInputListsTheOptionsAndPrefills() {
122 		var s = map("select");
123 		assertEquals("SELECT", s.get("tag"));
124 		assertEquals(List.of("x", "y", "z"), s.get("options"));
125 		assertEquals("y", s.get("value"));
126 	}
127 
128 	@Test void e02_aNestedTablesPencilIsNotTheParentsToHandle() {
129 		assertEquals(Boolean.TRUE, report().get("nestedIgnored"));
130 	}
131 
132 	@Test void f01_aSavedEditsMessageShowsAsAStatusToastWhenThereIsNoDetailPanel() {
133 		var m = map("messageNoPanel");
134 		assertEquals(Boolean.TRUE, m.get("closed"));
135 		assertEquals(List.of(Map.of("role", "status", "text", "Staged 'Rex'")), m.get("toasts"));
136 	}
137 
138 	@Test void f02_aSavedEditsMessageStaysInTheDetailPanelWhenThereIsOne() {
139 		var m = map("messageWithPanel");
140 		assertEquals("Staged 'Rex'", m.get("panelText"));
141 		assertEquals(List.of(), m.get("toasts"));
142 	}
143 
144 	@Test void f03_aSavedEditWithNoMessageShowsNoToast() {
145 		assertEquals(0, ((Number)report().get("noMessageNoToast")).intValue());
146 	}
147 }
