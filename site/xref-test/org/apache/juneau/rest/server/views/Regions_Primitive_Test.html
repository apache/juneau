27  /**
28   * Always-on coverage for the {@code populate(ctx, container)} primitive in {@code juneau-regions.js}.
29   *
30   * <p>
31   * The primitive is the one member of the region contract that every populator sees on every invocation, so its
32   * surface is pinned as a GOLDEN rather than spot-checked: the whole {@code ctx} key set, the whole {@code declared}
33   * key set and the whole {@code ids} key set are compared against literals, and a member that changes shape breaks
34   * this test rather than surfacing later as an author-visible contract break.  The behavioral half runs the real
35   * runtime source under a DOM shim and a controllable clock (see {@code src/test/js/regions-primitive.cjs}).
36   */
37  class Regions_Primitive_Test extends TestBase {

42  		return r;
43  	}

45  	private static void assertAllTrue(Map<?,?> r, String... keys) {
46  		for (var k : keys)
47  			assertEquals(true, r.get(k), () -> k + " -> " + r.get(k) + " in " + r);
48  	}

55  	 * The AbortController baseline is tested EXACTLY ONCE, at load.  A per-call feature test is what makes a degraded
56  	 * mode possible, and the design's answer to a missing AbortController is a refusal rather than a degraded mode.
57  	 */
58  	@Test void a01_sourceShape_abortControllerTestedOnceAtLoad() throws Exception {
59  		var body = RegionsHarness.regionsJs();
60  		var occurrences = body.split("typeof AbortController", -1).length - 1;
61  		assertEquals(1, occurrences, "the AbortController baseline must be feature-tested exactly once, at load");
62  		assertTrue(body.contains("const HAS_ABORT_CONTROLLER"), body);
63  	}

65  	/**
66  	 * {@code fork} is attached with {@code Object.defineProperty} on the INSTANCE and the region runtime never touches
67  	 * {@code AbortSignal.prototype}.  The lazy way to implement {@code fork()} is a prototype patch, which mutates a
68  	 * platform type for every script on the page including ones that have never heard of Juneau.
69  	 */
70  	@Test void a02_sourceShape_forkIsAnInstancePropertyNeverAPrototypePatch() throws Exception {
71  		var body = RegionsHarness.regionsJs();
72  		assertTrue(body.contains("Object.defineProperty(signal, \"fork\""), body);
73  		// The prose above the attachment says "NEVER on AbortSignal.prototype", so the scan is for a MEMBER ACCESS on
74  		// that prototype rather than for the phrase.
75  		assertFalse(body.contains("AbortSignal.prototype."), "the region runtime must never patch AbortSignal.prototype");
76  		assertFalse(body.contains("AbortSignal.prototype["), "the region runtime must never patch AbortSignal.prototype");
77  	}

79  	/**
80  	 * The barrier deadline is a clamped constant with no setter and no configuration read: it is never resolved from
81  	 * an attribute, a global or a descriptor field, because an author-settable barrier deadline is a way to configure
82  	 * a page-wide freeze.
83  	 */
84  	@Test void a03_sourceShape_barrierDeadlineIsNotConfigurable() throws Exception {
85  		var body = RegionsHarness.regionsJs();
86  		assertTrue(body.contains("const BARRIER_DEADLINE_MS = 2000;"), body);
87  		// Exactly one assignment in the whole file - the declaration - so there is no setter and no reassignment...
88  		assertEquals(1, body.split("BARRIER_DEADLINE_MS = ", -1).length - 1,
89  			"BARRIER_DEADLINE_MS must be assigned exactly once, at its declaration");
90  		// ...and the deadline is never resolved from configuration: no attribute read, no global, no descriptor field.
91  		for (var configPath : List.of(
92  			"data-juneau-region-barrier", "data-juneau-barrier", "JUNEAU_BARRIER", "barrierMs", "barrierDeadline"))
93  			assertFalse(body.contains(configPath), "the barrier deadline must not be readable from " + configPath);
94  	}

98  	 * of the other.  Conflating them was the defect this pair of constants exists to prevent: a shared ceiling
99  	 * reports a cycle for legitimate non-cyclic traffic.
100 	 */
101 	@Test void a04_sourceShape_theTwoCapsAreIndependentConstants() throws Exception {
102 		var body = RegionsHarness.regionsJs();

104 		assertTrue(body.contains("const REPLAY_BUFFER_CAP = 256;"), body);
105 		assertFalse(body.contains("REPLAY_BUFFER_CAP = BUS_QUEUE_DEPTH_CAP"), body);
106 		assertFalse(body.contains("BUS_QUEUE_DEPTH_CAP * "), body);
107 	}
