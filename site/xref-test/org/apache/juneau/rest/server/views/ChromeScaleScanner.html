24  /**
25   * Test-only, <b>private</b> stylesheet scanner enforcing the shared chrome scale contract on
26   * {@code juneau-views.css}: the named dimensional and type steps every chrome surface spends.
27   *
28   * <p>
29   * Not shipped and deliberately not public API, mirroring the discipline of the {@link RawContentSinkScanner}
30   * test-only guard this class is modeled on and sits beside.
31   *
32   * <h5 class='section'>What it enforces</h5>
33   * <ol>
34   * 	<li><b>Declaration.</b> Every step in {@link #scale()} is declared exactly once in the file. A step declared
35   * 		twice has two values in flight; a step declared zero times cannot be spent.
36   * 	<li><b>No literal that duplicates a step.</b> On the properties a chrome surface actually sizes with, a
37   * 		literal whose value equals a declared step is a violation &mdash; if a value equals a step, spend the
38   * 		step. This is the half that turns the contract's own "never write a literal that duplicates a step" rule
39   * 		from advice into a build failure.
40   * </ol>
41   *
42   * <h5 class='section'>Why the second half is narrower than it first looks, and deliberately so</h5>
43   * <p>
44   * The prohibition binds <b>computed</b> values, not written ones, and an {@code em} value can only be computed
45   * once you know its declaration's font-size context. That context is <i>not</i> uniform in this file &mdash; it
46   * runs from a UA/system control font through 10.2px, 12px, 16px and up &mdash; so a checker that assumed one
47   * context would reproduce, in test form, exactly the defect this contract exists to close. Two consequences,
48   * both intentional:
49   * <ul>
50   * 	<li><b>{@code em} values are never evaluated here.</b> Only absolute {@code px}/{@code rem} literals are
51   * 		compared. In particular a form control that sets no {@code font-size} of its own resolves its {@code em}
52   * 		against a UA/system font that varies by browser and platform, so a computation against one engine's
53   * 		value would report a violation on that engine and pass on every other. This scanner must never do that.
54   * 	<li><b>{@code rem} is normalised to px at 16px per rem.</b> That is safe in a way {@code em} is not: nothing
55   * 		in this file, and nothing in the console chrome that themes it, sets a {@code font-size} on
56   * 		{@code html}/{@code body}, so the root context is the browser default in both a themeless and a themed
57   * 		render.
58   * </ul>
59   * <p>
60   * <b>{@code font-size}'s {@code rem} reachability ({@code the font-size rem-reachability carve-out}).</b> {@code font-size}'s three
61   * chrome-scale steps ({@code --jc-chrome-font-size-1/2/3}) are declared in {@code rem}, not {@code px} - the
62   * only family in {@link #scale()} that is. That is a different question from the one above: this class's own
63   * literal-duplicate matcher has admitted both {@code px} and {@code rem} on every property it checks since its
64   * original implementation (a {@code gap} literal needs the same {@code rem} reach {@link #RECORDED_LITERALS}
65   * already relies on it for), so a {@code rem}-valued {@code font-size} duplicate was never structurally invisible
66   * to <i>this</i> class the way it was to the separate, one-time manual membership audit that named the
67   * {@code em}-only rule when the shared chrome-scale contract was first designed.
68   * {@link #findDuplicatedStep} still gives {@code font-size} its own named, dedicated
69   * {@link #FONT_SIZE_ABSOLUTE} matcher rather than leaning on the general {@link #ABSOLUTE} one, mirroring the
70   * {@code line-height}-scoped {@link #BARE_NUMBER} carve-out below and making the reachability an explicit,
71   * auditable, per-property fact rather than an incidental side effect - see
72   * {@code ChromeScale_ContractScan_Test.b12}/{@code b13} for the proof, in both directions, that this fix does
73   * not disturb the locked exclusion list.
74   *
75   * <h5 class='section'>Provisional steps are excluded from the second half</h5>
76   * <p>
77   * A step whose value is still provisional has not been rendered on a views surface yet. Binding the file's
78   * existing declarations to it would tie unrelated surfaces to a number the surface that owns it is going to
79   * re-decide &mdash; which is the "re-value a step out from under the other surfaces" failure, arrived at from
80   * the other direction. Provisional steps are still checked for {@link #scale() declaration}.
81   *
82   * <h5 class='section'>Why there is a recorded-exception table rather than a clean sheet</h5>
83   * <p>
84   * {@link #recordedLiterals()} lists every literal that stays a literal, each with a reason. It exists so the
85   * check is a <b>ratchet</b>: the recorded set is what the file looks like today, and anything new is a failure.
86   * An entry is not an excuse &mdash; a reader can see precisely which declarations have not been routed onto the
87   * ladder and why. The table is asserted to be non-vacuous by the accompanying test: an entry matching nothing is
88   * itself a failure, so the list cannot rot into a silent allow-everything.
89   */

199 	/** {@code width} is only a scale property on an SVG glyph; elsewhere it is a content measure. */
200 	private static final String SVG_SELECTOR_SUFFIX = "svg";
201 
202 	/** Matches one {@code selector { ... }} rule block. */
203 	private static final Pattern RULE = Pattern.compile("([^{}]+)\\{([^{}]*)\\}");
204 
205 	/** Matches a {@code var(...)} reference, including any fallback, so a fallback is not read as a bare literal. */
206 	private static final Pattern VAR_REF = Pattern.compile("var\\([^()]*(?:\\([^()]*\\)[^()]*)*\\)");
207 
208 	/**
209 	 * Matches an absolute {@code px} or {@code rem} length/size token. General on purpose: unlike
210 	 * {@code line-height}'s unitless step (see {@link #BARE_NUMBER}), several properties this contract checks
211 	 * carry a legitimate {@code rem} literal that must stay reachable regardless of which family it duplicates -
212 	 * for example {@code gap}'s card-grid density declaration ({@link #RECORDED_LITERALS}), not just
213 	 * {@code font-size}. See {@link #FONT_SIZE_ABSOLUTE} for the one property with its own dedicated,
214 	 * additionally-documented reference to this same {@code rem} domain.
215 	 */
216 	private static final Pattern ABSOLUTE = Pattern.compile("(?<![\\w.-])(\\d*\\.?\\d+)(px|rem)(?![\\w-])");
217 
218 	/**
219 	 * The {@code font-size}-scoped counterpart to {@link #ABSOLUTE}, used only for the {@code font-size}
220 	 * property (see {@link #findDuplicatedStep}). Declared as its own named pattern - identical in what it
221 	 * matches to {@link #ABSOLUTE} today - so that {@code font-size}'s {@code rem} reachability is an explicit,
222 	 * auditable carve-out in its own right (mirroring the {@code line-height}-scoped {@link #BARE_NUMBER}
223 	 * carve-out) rather than an incidental side effect of the general matcher, closing the unit-domain mismatch
224 	 * named by {@code the font-size rem-reachability carve-out}: {@code font-size}'s three chrome-scale steps
225 	 * ({@code --jc-chrome-font-size-1/2/3}) are declared in {@code rem}, and a {@code rem}-valued
226 	 * {@code font-size} literal that duplicates one must never go unmatched the way a tenth un-tokenized site
227 	 * once did.
228 	 */
229 	private static final Pattern FONT_SIZE_ABSOLUTE = Pattern.compile("(?<![\\w.-])(\\d*\\.?\\d+)(px|rem)(?![\\w-])");
230 
231 	/**
232 	 * Matches a bare, unitless decimal. {@code line-height}'s scale step ({@code --jc-chrome-line-height},
233 	 * {@code 1.2}) is the only step value in the ladder with no {@code px}/{@code rem} unit, so this pattern is
234 	 * used only for the {@code line-height} property (see {@link #findDuplicatedStep}) - it must never widen
235 	 * matching for any other property.
236 	 */
237 	private static final Pattern BARE_NUMBER = Pattern.compile("(?<![\\w.-])(\\d*\\.?\\d+)(?![\\w.%-])");

249 	/**
250 	 * Reports any step not declared exactly once in the stylesheet.
251 	 *
252 	 * @param css The stylesheet text.
253 	 * @return One message per step that is missing or declared more than once; empty when every step is sound.
254 	 */
255 	static List<String> checkDeclaredExactlyOnce(String css) {
256 		var code = stripComments(css);
257 		var out = new ArrayList<String>();
258 		for (var s : SCALE) {
259 			var m = Pattern.compile(Pattern.quote(s.token()) + "\\s*:\\s*([^;}]+)").matcher(code);
260 			var found = new ArrayList<String>();
261 			while (m.find())
262 				found.add(m.group(1).trim());
263 			if (found.size() != 1) {
264 				out.add(s.token() + ": expected exactly one declaration, found " + found.size() + " " + found);
265 			} else if (!found.get(0).equals(s.value())) {
266 				out.add(s.token() + ": declared as '" + found.get(0) + "', contract says '" + s.value() + "'");
267 			}
268 		}
269 		return out;
270 	}

273 	 * Scans for literals that duplicate a declared step on a property the scale covers.
274 	 *
275 	 * @param css The stylesheet text.
276 	 * @return The violations, the declarations actually examined, and which recorded exceptions were matched.
277 	 */
278 	static Result scan(String css) {
279 		return scan(css, SCALE);

282 	/**
283 	 * Scans against an explicit step list rather than {@link #SCALE}.
284 	 *
285 	 * <p>
286 	 * A test-only seam: {@link #SCALE} is {@code private static final} and cannot carry a synthetic step (nor
287 	 * should it - {@code a01}/{@code a02} require every declared step to be real and declared exactly once in
288 	 * the shipped stylesheet). This overload lets a test exercise the provisional-exemption branch against a
289 	 * scale of its own choosing, without widening {@link #SCALE} itself.
290 	 *
291 	 * @param css The stylesheet text.
292 	 * @param scale The step list to check against, in place of {@link #SCALE}.
293 	 * @return The violations, the declarations actually examined, and which recorded exceptions were matched.
294 	 */
295 	static Result scan(String css, List<Step> scale) {
296 		var violations = new ArrayList<String>();
297 		var checked = new ArrayList<Decl>();
298 		var matched = new LinkedHashSet<RecordedLiteral>();
299 		var code = stripComments(css);
300 
301 		var rules = RULE.matcher(code);
302 		while (rules.find()) {
303 			var selector = normalise(rules.group(1));
304 			if (selector.startsWith("@") || selector.isEmpty())
305 				continue;
306 			for (var d : declarations(selector, rules.group(2))) {
307 				var families = CHECKED_PROPERTIES.get(d.property());
308 				if (families == null)
309 					continue;
310 				if (eq(d.property(), "width") && !isSvgSelector(selector))
311 					continue;
312 				checked.add(d);
313 				var hit = findDuplicatedStep(d, families, scale);
314 				if (hit == null)
315 					continue;
316 				var recorded = findRecord(d);
317 				if (recorded != null) {
318 					matched.add(recorded);
319 					continue;
320 				}
321 				violations.add(d.selector() + " { " + d.property() + ": " + d.value() + " } duplicates the step "
322 					+ hit.token() + " (" + hit.value() + "). If a value equals a step, spend the step - or add a "
323 					+ "recorded exception with the reason it cannot.");
324 			}
325 		}
326 		return new Result(violations, checked, matched);
327 	}

329 	/**
330 	 * The confirmed step of an eligible family whose value the declaration writes out as a literal.
331 	 *
332 	 * <p>
333 	 * Role-named families are searched before the spacing ladder, so a 31px {@code height} is reported against
334 	 * {@code --jc-chrome-control-height} rather than falling through to the spacing ladder (which has no 31px
335 	 * step to collide with in the first place - unlike this step's former 32px value, which happened to
336 	 * numerically coincide with {@code --jc-space-6}). Both are violations either way; naming the apter one is
337 	 * what makes the message actionable.
338 	 */
339 	private static Step findDuplicatedStep(Decl d, Set<Family> families, List<Step> scale) {
340 		// line-height's step value is unitless, so it needs bare-number matching instead of the general
341 		// px/rem literal matcher - scoped to this one property so no other property's matching widens.
342 		var lineHeight = eq(d.property(), "line-height");
343 		// font-size's three steps are declared in rem, not px. The general ABSOLUTE matcher already admits
344 		// rem (other properties, e.g. gap, rely on that too), but font-size gets its own named matcher here so
345 		// its rem reachability is an explicit, font-size-scoped carve-out (the font-size rem-reachability carve-out) rather than an
346 		// incidental side effect of the shared one - mirroring the line-height carve-out above in shape.
347 		var fontSize = eq(d.property(), "font-size");
348 		var literals = lineHeight ? bareNumberValues(d.value())
349 			: fontSize ? fontSizeAbsoluteValues(d.value())
350 			: absoluteValues(d.value());
351 		if (literals.isEmpty())
352 			return null;
353 		for (var roleNamed : List.of(true, false)) {
354 			for (var s : scale) {
355 				if (!s.confirmed() || !families.contains(s.family()) || (s.family() == Family.SPACE) == roleNamed)
356 					continue;
357 				var stepValue = lineHeight ? bareValue(s.value()) : fontSize ? fontSizeToPx(s.value()) : toPx(s.value());
358 				if (stepValue != null && literals.contains(stepValue))
359 					return s;
360 			}
361 		}
362 		return null;
363 	}

365 	private static RecordedLiteral findRecord(Decl d) {
366 		for (var r : RECORDED_LITERALS)
367 			if (d.selector().contains(r.selector()) && r.property().equals(d.property()) && r.value().equals(d.value()))
368 				return r;
369 		return null;
370 	}

372 	/** True if any comma-separated selector in the group targets an {@code svg} element. */
373 	private static boolean isSvgSelector(String selector) {
374 		for (var s : selector.split(","))
375 			if (s.trim().endsWith(SVG_SELECTOR_SUFFIX))
376 				return true;
377 		return false;
378 	}

380 	/**
381 	 * Every absolute value in a declaration, normalised to px. {@code var(...)} expressions are removed first, so
382 	 * a token's own fallback (e.g. {@code var(--jc-card-gap, 1rem)}) is not mistaken for a bare literal - the
383 	 * declaration is already spending a token, which is the thing the rule asks for.
384 	 */
385 	private static Set<Double> absoluteValues(String value) {
386 		var out = new LinkedHashSet<Double>();
387 		var m = ABSOLUTE.matcher(VAR_REF.matcher(value).replaceAll(" "));
388 		while (m.find())
389 			out.add(px(Double.parseDouble(m.group(1)), m.group(2)));
390 		return out;
391 	}
392 
393 	private static Double toPx(String value) {
394 		var m = ABSOLUTE.matcher(value);
395 		return m.find() ? px(Double.parseDouble(m.group(1)), m.group(2)) : null;
396 	}

399 	 * The {@code font-size}-scoped counterpart to {@link #absoluteValues}: also admits {@code rem}. Only ever
400 	 * called for the {@code font-size} property (see {@link #findDuplicatedStep}).
401 	 */
402 	private static Set<Double> fontSizeAbsoluteValues(String value) {
403 		var out = new LinkedHashSet<Double>();
404 		var m = FONT_SIZE_ABSOLUTE.matcher(VAR_REF.matcher(value).replaceAll(" "));
405 		while (m.find())
406 			out.add(px(Double.parseDouble(m.group(1)), m.group(2)));
407 		return out;
408 	}

410 	/** The {@code font-size}-scoped counterpart to {@link #toPx}: also admits {@code rem}. */
411 	private static Double fontSizeToPx(String value) {
412 		var m = FONT_SIZE_ABSOLUTE.matcher(value);
413 		return m.find() ? px(Double.parseDouble(m.group(1)), m.group(2)) : null;
414 	}

416 	/**
417 	 * Every bare unitless number in a declaration's value, {@code var(...)} fallbacks excluded. Only ever called
418 	 * for the {@code line-height} property (see {@link #findDuplicatedStep}).
419 	 */
420 	private static Set<Double> bareNumberValues(String value) {
421 		var out = new LinkedHashSet<Double>();
422 		var m = BARE_NUMBER.matcher(VAR_REF.matcher(value).replaceAll(" "));
423 		while (m.find())

430 		var m = BARE_NUMBER.matcher(value);
431 		return m.find() ? Double.parseDouble(m.group(1)) : null;

434 	/** Normalises to px. {@code rem} resolves at the browser default because nothing here sets a root font-size. */
435 	private static double px(double n, String unit) {
436 		return eq(unit, "rem") ? n * 16.0 : n;
437 	}

439 	private static List<Decl> declarations(String selector, String body) {
440 		var out = new ArrayList<Decl>();
441 		for (var raw : body.split(";")) {
442 			var colon = raw.indexOf(':');
443 			if (colon < 0)
444 				continue;
445 			var property = raw.substring(0, colon).trim().toLowerCase(Locale.ROOT);
446 			var value = normalise(raw.substring(colon + 1));
447 			if (property.isEmpty() || property.startsWith("--") || value.isEmpty())
448 				continue;
449 			out.add(new Decl(selector, property, value));
450 		}
451 		return out;
452 	}

454 	private static String normalise(String s) {
455 		return s.replaceAll("\\s+", " ").trim();
456 	}

458 	/** Blanks out comment content so a value mentioned in a comment is not read as a declaration. */
459 	private static String stripComments(String s) {
460 		var out = new StringBuilder(s.length());
461 		var i = 0;
462 		while (i < s.length()) {
463 			if (s.charAt(i) == '/' && i + 1 < s.length() && s.charAt(i + 1) == '*') {
464 				i += 2;
465 				while (i + 1 < s.length() && !(s.charAt(i) == '*' && s.charAt(i + 1) == '/')) {
466 					if (s.charAt(i) == '\n')
467 						out.append('\n');
468 					i++;
469 				}
470 				i = Math.min(s.length(), i + 2);
471 				continue;
472 			}
473 			out.append(s.charAt(i));
474 			i++;
475 		}
476 		return out.toString();
477 	}
