19  import static java.nio.charset.StandardCharsets.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.io.*;
23  import java.nio.file.*;
24  import java.util.*;
25  import java.util.concurrent.*;
26  
27  import org.apache.juneau.marshall.marshaller.*;

29  /**
30   * Generic Node plumbing for the plain-Node behavioral harnesses under {@code src/test/js}.
31   *
32   * <p>
33   * Each harness is run as {@code node <harness> <asset-file>...}, where every asset is a classpath resource copied
34   * to a temp file, and must print exactly one JSON object to stdout.  Reports are cached per harness-and-assets.
35   * A machine without Node (or without the harness on any known path) yields {@code null}, which callers turn into a
36   * skip.
37   */
38  final class NodeHarness {
39  
40  	private static final Map<String,Map<?,?>> CACHE = new ConcurrentHashMap<>();
41  	private static final Set<String> UNAVAILABLE = ConcurrentHashMap.newKeySet();
42  
43  	private NodeHarness() {}
44  
45  	/**
46  	 * Runs {@code harnessName} against the given classpath resources (resolved relative to {@link ViewsMixin}) and
47  	 * returns its parsed report, or {@code null} when it cannot run on this machine.
48  	 */
49  	static Map<?,?> report(String harnessName, String...resources) {
50  		return report(Map.of(), harnessName, resources);
51  	}
52  
53  	/**
54  	 * As {@link #report(String, String...)}, with extra environment variables for the node process (for example
55  	 * {@code LC_ALL} and {@code TZ}, which the chart fn outputs depend on).
56  	 */
57  	static Map<?,?> report(Map<String,String> env, String harnessName, String...resources) {
58  		var key = harnessName + Arrays.toString(resources) + new TreeMap<>(env);
59  		if (UNAVAILABLE.contains(key))
60  			return null;
61  		var cached = CACHE.get(key);
62  		if (cached != null)
63  			return cached;
64  		try {
65  			var harness = nodeAvailable() ? locate(harnessName) : null;
66  			if (harness == null) {
67  				UNAVAILABLE.add(key);
68  				return null;
69  			}
70  			var report = Json.to(run(harness, harnessName, env, resources), Map.class);
71  			CACHE.put(key, report);
72  			return report;
73  		} catch (Exception e) {
74  			throw new AssertionError("could not run " + harnessName + ": " + e, e);
75  		}
76  	}
77  
78  	private static String run(Path harness, String harnessName, Map<String,String> env, String[] resources) throws Exception {
79  		var temps = new ArrayList<Path>();
80  		var stdout = Files.createTempFile("node-harness-stdout-", ".json");
81  		var stderr = Files.createTempFile("node-harness-stderr-", ".txt");
82  		try {
83  			var args = new ArrayList<String>(List.of("node", harness.toString()));
84  			for (var r : resources) {
85  				var f = Files.createTempFile("node-harness-asset-", ".js");
86  				temps.add(f);
87  				try (var in = ViewsMixin.class.getResourceAsStream(r)) {
88  					assertNotNull(in, () -> "missing classpath resource: " + r);
89  					Files.writeString(f, new String(in.readAllBytes(), UTF_8), UTF_8);
90  				}
91  				args.add(f.toString());
92  			}
93  			var pb = new ProcessBuilder(args).redirectOutput(stdout.toFile()).redirectError(stderr.toFile());
94  			pb.environment().putAll(env);
95  			var p = pb.start();
96  			if (!p.waitFor(60, TimeUnit.SECONDS)) {
97  				p.destroyForcibly();
98  				fail(harnessName + " did not finish within 60s; stderr:\n" + quietRead(stderr));
99  			}
100 			if (p.exitValue() != 0)
101 				fail(harnessName + " exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr) + "\nstdout:\n" + quietRead(stdout));
102 			return Files.readString(stdout, UTF_8);
103 		} finally {
104 			for (var f : temps)
105 				Files.deleteIfExists(f);
106 			Files.deleteIfExists(stdout);
107 			Files.deleteIfExists(stderr);
108 		}
109 	}
110 
111 	private static boolean nodeAvailable() {
112 		try {
113 			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
114 			if (!p.waitFor(5, TimeUnit.SECONDS)) {
115 				p.destroyForcibly();
116 				return false;
117 			}
118 			return p.exitValue() == 0;
119 		} catch (Exception e) {
120 			return false;
121 		}
122 	}
123 
124 	private static Path locate(String harnessName) {
125 		var basedir = System.getProperty("basedir");
126 		if (basedir != null) {
127 			var p = Path.of(basedir, "src/test/js/" + harnessName);
128 			if (Files.isRegularFile(p))
129 				return p.toAbsolutePath().normalize();
130 		}
131 		for (var rel : List.of("src/test/js/" + harnessName, "juneau-rest/juneau-rest-server-views/src/test/js/" + harnessName)) {
132 			var p = Path.of(rel);
133 			if (Files.isRegularFile(p))
134 				return p.toAbsolutePath().normalize();
135 		}
136 		return null;
137 	}
138 
139 	private static String quietRead(Path p) {
140 		try { return Files.readString(p, UTF_8); }
141 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
142 	}
143 }
