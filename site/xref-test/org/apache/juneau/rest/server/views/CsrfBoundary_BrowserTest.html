44  /**
45   * A {@link CsrfBoundary} behind a real {@link RequestBoundaryFilter}, attacked and used from a real headless browser.
46   *
47   * <p>
48   * The application is a Juneau {@link RestServlet} in a Jetty servlet container, with the boundary filter mapped at
49   * {@code /*} in front of it.  Its pages embed the token the filter publishes, the way a page renderer does, and load
50   * the real {@code juneau-views.js} and {@code juneau-console.js}.  A second Jetty server on another port plays the
51   * attacker: its pages make the browser send forms and fetches to the application.  Every attack is made by the
52   * browser itself, so the {@code Origin} and {@code Sec-Fetch-*} headers the boundary judges are the ones a browser
53   * really sends.
54   *
55   * <p>
56   * Both servers listen on {@code 127.0.0.1}, so the attacker is a different origin (another port) but the same site:
57   * the application's session cookie still rides on the attacker's requests, exactly as it would for a sibling
58   * application on the same host.  The resources the attacks target are plain counters that need no session, so the
59   * only thing standing between an attack and a changed counter is the boundary.  A recording filter in front of the
60   * boundary notes every request and the answer it got, so an attack the browser never sent cannot pass as refused.
61   *
62   * <p>
63   * Loopback addresses are one site, so those attacks arrive as {@code same-site}.  For a {@code cross-site} attack the
64   * browser is told to resolve two registrable names, {@value #SITE_HOST} and {@value #EVIL_HOST}, to {@code 127.0.0.1}:
65   * a third Jetty server, behind a boundary that answers only to {@value #SITE_HOST}, is the application, and the
66   * attacker's server is reached as {@value #EVIL_HOST}.  Nothing leaves the machine.  A browser sends
67   * {@code Sec-Fetch-Site} only to a trustworthy origin, so that browser treats those two origins, and no others, as
68   * secure, as it would over https.  Without that switch the browser sends no {@code Sec-Fetch-*} headers to a plain
69   * http host name, and the headless shell a bare Playwright launch starts ignores it, so these rows need the full
70   * Chromium Playwright installs (channel {@code chromium}).  If only the headless shell is installed, the prober fails
71   * at launch and the whole class fails.
72   *
73   * <h5 class='section'>Off by default:</h5>
74   * <p>
75   * Disabled unless the {@value #GATE} system property is set, which only the module's opt-in {@code js-tests} Maven
76   * profile does.  The prober is {@code csrf-boundary.cjs}, next to the profile's harness.
77   */
78  @EnabledIfSystemProperty(named=CsrfBoundary_BrowserTest.GATE, matches="true",
79  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -f juneau-rest/juneau-rest-server-views/pom.xml test`")
80  @SuppressWarnings({
81  	"resource" // Servlet response writers belong to the container, and each connector belongs to the server it is added to; none is closed here.
82  })

110 	/** Records each request and its answer; registered ahead of the boundary filter. */
111 	static final class Recorder implements Filter {
112 		@Override /* Filter */
113 		public void doFilter(ServletRequest req, ServletResponse res, FilterChain chain) throws IOException, ServletException {
114 			try {
115 				chain.doFilter(req, res);
116 			} finally {
117 				var q = (HttpServletRequest)req;
118 				var s = (HttpServletResponse)res;
119 				var cors = s.getHeaderNames().stream().anyMatch(n -> n.toLowerCase(Locale.ROOT).startsWith("access-control-allow-"));
120 				SEEN.add(new Seen(q.getMethod(), q.getRequestURI(), q.getHeader("Host"), q.getHeader("Origin"),
121 					q.getHeader("Sec-Fetch-Site"), s.getStatus(), s.getHeader(RequestBoundaryFilter.REJECTION_HEADER), cors));
122 			}
123 		}
124 	}

216 	/** The attacker's pages, each aimed at its own counter on the application. */
217 	static final class AttackServlet extends HttpServlet {
218 		private static final long serialVersionUID = 1L;
219 		private final String app;
220 		private final String site;
221 		AttackServlet(String app, String site) {
222 			this.app = app;
223 			this.site = site;
224 		}
225 		@Override protected void doGet(HttpServletRequest req, HttpServletResponse res) throws IOException {
226 			var body = switch (String.valueOf(req.getPathInfo())) {
227 				// A form that submits itself as soon as the page loads.
228 				case "/form" -> "<form id=\"f\" method=\"post\" action=\"" + app + "/count/a03\"><input name=\"x\" value=\"1\"></form>"
229 					+ "<script>document.getElementById('f').submit();</script>";
230 				// The same form, aimed at the application on another site.
231 				case "/cross-site-form" -> "<form id=\"f\" method=\"post\" action=\"" + site + "/count/a12\"><input name=\"x\" value=\"1\"></form>"
232 					+ "<script>document.getElementById('f').submit();</script>";
233 				// A fire-and-forget fetch: the response is opaque to the attacker, but the request is still sent.
234 				case "/no-cors" -> "<script>fetch('" + app + "/count/a04', { method: 'POST', mode: 'no-cors', credentials: 'include',"
235 					+ " headers: { 'Content-Type': 'text/plain' }, body: 'x=1' })"
236 					+ ".then(function (r) { window.attackDone = 'sent:' + r.type; }, function (e) { window.attackDone = 'failed:' + e; });</script>";
237 				// A sandboxed frame has an opaque origin, so its form is sent with 'Origin: null'.
238 				case "/sandboxed" -> "<iframe sandbox=\"allow-scripts allow-forms\" srcdoc=\"<form method='post' action='" + app
239 					+ "/count/a05'><input name='x' value='1'></form><script>document.forms[0].submit();</script>\"></iframe>";
240 				// A JSON write needs a preflight; the application never answers one with CORS permission.
241 				case "/json" -> "<script>fetch('" + app + "/count/a06', { method: 'POST', credentials: 'include',"
242 					+ " headers: { 'Content-Type': 'application/json' }, body: '{}' })"
243 					+ ".then(function (r) { window.attackDone = 'answered:' + r.status; }, function (e) { window.attackDone = 'failed:' + e; });</script>";
244 				default -> null;
245 			};
246 			if (body == null) {
247 				res.sendError(404);
248 				return;
249 			}
250 			res.setContentType("text/html;charset=utf-8");
251 			res.getWriter().write("<!DOCTYPE html><html><head><meta charset=\"utf-8\"><link rel=\"icon\" href=\"data:,\"></head><body>"
252 				+ body + "</body></html>");
253 		}
254 	}

274 	/** A server with one connector on {@code 127.0.0.1}, already listening, so its port is known before its handlers. */
275 	private static ServerConnector listen(Server server) throws Exception {
276 		var connector = new ServerConnector(server);
277 		connector.setHost("127.0.0.1");
278 		connector.setPort(0);
279 		server.addConnector(connector);
280 		connector.open();
281 		SERVERS.add(server);
282 		return connector;
283 	}

285 	@BeforeAll
286 	static void startAndProbe() throws Exception {
287 		var app = new Server();
288 		appPort = listen(app).getLocalPort();
289 		appOrigin = "http://127.0.0.1:" + appPort;
290 		var appCtx = new ServletContextHandler(ServletContextHandler.SESSIONS);
291 		appCtx.setContextPath("/");
292 		var request = EnumSet.of(DispatcherType.REQUEST);
293 		appCtx.addFilter(new FilterHolder(new Recorder()), "/*", request);
294 		appCtx.addFilter(new FilterHolder(new RequestBoundaryFilter(CsrfBoundary.create().build())), "/*", request);
295 		appCtx.addServlet(new ServletHolder(new AssetServlet()), "/assets/*");
296 		appCtx.addServlet(new ServletHolder(new AppResource()), "/*");
297 		app.setHandler(appCtx);
298 		app.start();
299 
300 		// The same application on its public host name, behind a boundary that answers to that name only.
301 		var site = new Server();
302 		sitePort = listen(site).getLocalPort();
303 		siteOrigin = "http://" + SITE_HOST + ":" + sitePort;
304 		var siteCtx = new ServletContextHandler(ServletContextHandler.SESSIONS);
305 		siteCtx.setContextPath("/");
306 		siteCtx.addFilter(new FilterHolder(new Recorder()), "/*", request);
307 		siteCtx.addFilter(new FilterHolder(new RequestBoundaryFilter(CsrfBoundary.create().allowedHosts(SITE_HOST).build())), "/*", request);
308 		siteCtx.addServlet(new ServletHolder(new AssetServlet()), "/assets/*");
309 		siteCtx.addServlet(new ServletHolder(new AppResource()), "/*");
310 		site.setHandler(siteCtx);
311 		site.start();
312 
313 		var attacker = new Server();
314 		var attackPort = listen(attacker).getLocalPort();
315 		attackOrigin = "http://127.0.0.1:" + attackPort;
316 		crossSiteAttackOrigin = "http://" + EVIL_HOST + ":" + attackPort;
317 		var attackCtx = new ServletContextHandler(ServletContextHandler.NO_SESSIONS);
318 		attackCtx.setContextPath("/");
319 		attackCtx.addServlet(new ServletHolder(new AttackServlet(appOrigin, siteOrigin)), "/attack/*");
320 		attacker.setHandler(attackCtx);
321 		attacker.start();
322 
323 		// The contrast for the host-name row: a loopback boundary that answers only to its own 127.0.0.1 authority.
324 		var loopback = new Server();
325 		var loopbackPort = listen(loopback).getLocalPort();
326 		var loopbackCtx = new ServletContextHandler(ServletContextHandler.NO_SESSIONS);
327 		loopbackCtx.setContextPath("/");
328 		loopbackCtx.addFilter(new FilterHolder(new LoopbackBoundaryFilter(
329 			LoopbackBoundary.create().authority("127.0.0.1", loopbackPort).token(SynchronizerToken.generate()).build())), "/*", request);
330 		loopbackCtx.addServlet(new ServletHolder(new PlainServlet()), "/*");
331 		loopback.setHandler(loopbackCtx);
332 		loopback.start();
333 
334 		report = Json.to(runProber(appOrigin, "http://localhost:" + appPort, attackOrigin, "http://localhost:" + loopbackPort,
335 			siteOrigin, crossSiteAttackOrigin), Map.class);
336 	}

358 	/** Runs the prober, failing with its stderr attached (its exit code alone is not a diagnosis). */
359 	private static String runProber(String...args) throws Exception {
360 		var dir = Path.of(requiredProperty("juneau.jsTests.dir"));
361 		var harness = Path.of(requiredProperty("juneau.jsTests.harness")).getParent().resolve("csrf-boundary.cjs");
362 		var cmd = new ArrayList<>(List.of(System.getProperty("juneau.jsTests.node", "node"), harness.toString()));
363 		cmd.addAll(List.of(args));
364 		var stdout = dir.resolve("csrf-boundary-stdout.json");
365 		var stderr = dir.resolve("csrf-boundary-stderr.txt");
366 		var pb = new ProcessBuilder(cmd).redirectOutput(stdout.toFile()).redirectError(stderr.toFile());
367 		pb.environment().put("NODE_PATH", dir.resolve("node_modules").toString());
368 		pb.environment().put("PLAYWRIGHT_BROWSERS_PATH", requiredProperty("juneau.jsTests.browsers"));
369 
370 		var p = pb.start();
371 		if (! p.waitFor(3, TimeUnit.MINUTES)) {
372 			p.destroyForcibly();
373 			fail("prober did not finish within 3m; stderr:\n" + quietRead(stderr));
374 		}
375 		assertEquals(0, p.exitValue(), () -> "prober exited non-zero; stderr:\n" + quietRead(stderr));
376 		return Files.readString(stdout);
377 	}
