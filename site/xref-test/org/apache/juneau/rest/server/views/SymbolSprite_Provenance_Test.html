30  /**
31   * The licensing-drift guard for {@code juneau-symbols.svg}: asserts that the shipped artwork is still the artwork
32   * the provenance manifest ({@code juneau-symbols-provenance.md}) approved, and that the sprite still honours the
33   * contracts the manifest declares.
34   *
35   * <h5 class='section'>Why this exists at all</h5>
36   * <p>
37   * Because origin and fingerprint are facts about today, and the failure they guard is silent. Some glyphs are
38   * Juneau-original; others are console-sprite counterparts cleared for artwork copy (operator ruling 2026-09-24) and
39   * recorded as {@code original-artwork}. The cheapest way for an unreviewed paste to land again is for
40   * someone to overwrite a path during an unrelated polish pass. Nothing else in the build would notice: the artwork
41   * renders, the ids are unchanged, and {@link SymbolSprite_StemIds_Test} pins <i>names</i> rather than paths. This
42   * test makes such a paste fail until the manifest row is deliberately edited, which turns an invisible act into a
43   * reviewable one.
44   *
45   * <h5 class='section'>Deliberately separate from the stem-id guard</h5>
46   * <p>
47   * {@link SymbolSprite_StemIds_Test} guards correctness &mdash; a rename blanks the Support Console's toolbar,
48   * because an app overrides these glyphs by stem name through the page-level override sprite (see
49   * {@code juneau-icons.js}), and an unresolved stem draws nothing rather than throwing. This one guards licensing drift. They fail
50   * for unrelated reasons and read as unrelated diagnostics, so they are two tests rather than one.
51   *
52   * <h5 class='section'>What it asserts</h5>
53   * <ul>
54   * 	<li>The manifest and the sprite describe the same glyph set, in the same order.
55   * 	<li>Every glyph's fingerprint still matches its approved value.
56   * 	<li>Every glyph's declared origin is either {@code juneau-original} or {@code original-artwork} &mdash; the only
57   * 		origins the manifest authorises. (This test does not read {@code NOTICE}; keeping its attribution in step
58   * 		with the {@code original-artwork} rows is a manual, reviewed act.)
59   * 	<li>The three contracts the manifest states as rules: {@code viewBox="0 0 24 24"} on every glyph, paint only
60   * 		ever {@code none}, {@code currentColor} or a themable {@code var(--x, currentColor)}, and an explicit
61   * 		{@code stroke-width} wherever a stroke is painted.
62   * </ul>
63   *
64   * <h5 class='section'>What it deliberately does not assert</h5>
65   * <p>
66   * Anything about how the artwork <i>looks</i>. Rasterised review is
67   * {@link SymbolSprite_Render_BrowserTest}'s job and it needs a browser, so it sits behind the {@code js-tests}
68   * profile. This test is cheap, needs nothing, and therefore runs on every build &mdash; which is the right split,
69   * because a provenance guard that only runs under an opt-in profile guards nothing on the default gate.
70   */

77  	private static final String MANIFEST_IS_THE_REVIEW =
78  		"If the artwork change is intended, update the row in " + SymbolProvenanceScanner.MANIFEST
79  			+ " - and read that file's `Authoring rules` section first, because editing the row is the reviewed act"
80  			+ " that this test exists to force.";

82  	private static final String REQUIRED_VIEWBOX = "0 0 24 24";
83  	private static final Set<String> ALLOWED_ORIGINS = Set.of("juneau-original", "original-artwork");

85  	private static String sprite() throws Exception {
86  		var s = SymbolProvenanceScanner.read(SymbolProvenanceScanner.SPRITE);
87  		assertNotNull(s, "could not read " + SymbolProvenanceScanner.SPRITE + " from the module's own resources");
88  		return s;
89  	}

92  		var s = SymbolProvenanceScanner.read(SymbolProvenanceScanner.MANIFEST);
93  		assertNotNull(s, "could not read " + SymbolProvenanceScanner.MANIFEST + " from the module's own resources");
94  		return s;
95  	}

97  	/**
98  	 * Appends a redundant {@code Z} to the first path of a document - a byte change that is valid in any path.
99  	 *
100 	 * <p>
101 	 * The needle carries its leading space deliberately: {@code id="} contains {@code d="}, so searching for the
102 	 * bare attribute name finds every glyph's <i>id</i> first and mutates the stem name instead of the artwork.
103 	 */
104 	private static String mutateFirstPath(String svg) {
105 		var at = svg.indexOf(" d=\"");
106 		assertTrue(at >= 0, "no path data to mutate; this file's anti-vacuous checks cannot run");
107 		var close = svg.indexOf('"', at + 4);
108 		assertTrue(close > at, "unterminated path data");
109 		return svg.substring(0, close) + "Z" + svg.substring(close);
110 	}

116 	@Test void a01_manifestCoversExactlyTheShippedGlyphs() throws Exception {
117 		var shipped = new ArrayList<>(SymbolProvenanceScanner.symbols(sprite()).keySet());
118 		var declared = new ArrayList<>(SymbolProvenanceScanner.manifestFingerprints(manifest()).keySet());
119 		assertNotEmpty(() -> "no <symbol> elements found in the sprite - this test would be vacuous", shipped);
120 		assertList(() -> "the manifest and the sprite describe different glyph sets (or the same set in a different order)."
121 				+ " A glyph shipped without a manifest row is unpinned artwork; a row without a glyph is a stale"
122 				+ " approval. " + MANIFEST_IS_THE_REVIEW, declared, shipped.toArray());
123 	}

125 	@Test void a02_everyGlyphMatchesItsApprovedFingerprint() throws Exception {
126 		var shipped = SymbolProvenanceScanner.symbols(sprite());
127 		var approved = SymbolProvenanceScanner.manifestFingerprints(manifest());
128 		var drifted = new ArrayList<String>();
129 		shipped.forEach((stem, element) -> {
130 			var actual = SymbolProvenanceScanner.fingerprint(element);
131 			if (!actual.equals(approved.get(stem)))
132 				drifted.add(stem + ": manifest=" + approved.get(stem) + " actual=" + actual);
133 		});
134 		assertEmpty(() -> "the artwork of these glyphs no longer matches what the manifest approved. "
135 			+ MANIFEST_IS_THE_REVIEW, drifted);
136 	}

138 	@Test void a03_everyGlyphDeclaresAnAllowedOrigin() throws Exception {
139 		// Origins are either Juneau-original or artwork cleared for copy.  Any other token would
140 		// leave LICENSE/NOTICE and the dual-hat rule out of sync with the sprite.
141 		var origins = SymbolProvenanceScanner.manifestOrigins(manifest());
142 		assertNotEmpty(() -> "no manifest rows parsed - this test would be vacuous", origins);
143 		var foreign = origins.entrySet().stream()
144 			.filter(e -> !ALLOWED_ORIGINS.contains(e.getValue()))
145 			.map(Object::toString)
146 			.toList();
147 		assertEmpty(() -> "a glyph declares an origin outside " + ALLOWED_ORIGINS + ". Update the manifest only after the artwork"
148 				+ " clearance and NOTICE attribution match.", foreign);
149 	}

151 	/**
152 	 * The a04 check as a function of its inputs: every document-family stem must be shipped and declared
153 	 * {@code original-artwork}.  Returns one problem string per violation (empty when the family is intact).
154 	 */
155 	private static List<String> documentFamilyProblems(Map<String,String> shipped, Map<String,String> origins) {
156 		var problems = new ArrayList<String>();
157 		for (var stem : SymbolProvenanceScanner.FRAMED_FAMILY) {
158 			if (shipped.get(stem) == null)
159 				problems.add("the document family member " + stem + " is missing from the sprite");
160 			if (neq(origins.get(stem), "original-artwork"))
161 				problems.add("document-family stem " + stem + " is expected to be original-artwork after the artwork replacement"
162 					+ " but is " + origins.get(stem));
163 		}
164 		return problems;
165 	}

175 		var offModulus = new ArrayList<String>();
176 		SymbolProvenanceScanner.symbols(sprite()).forEach((stem, element) -> {
177 			var vb = SymbolProvenanceScanner.viewBox(element);
178 			if (neq(vb, REQUIRED_VIEWBOX))
179 				offModulus.add(stem + ": viewBox=\"" + vb + "\"");
180 		});
181 		assertEmpty(() -> "a glyph declares a viewBox other than \"" + REQUIRED_VIEWBOX + "\". juneau-icons.js hard-codes the"
182 				+ " host <svg> at that modulus, so any other one interposes a scale factor between the author's"
183 				+ " coordinates and the pixel grid and the glyph's strokes stop landing on pixel boundaries.", offModulus);
184 	}

209 	@Test void b01_aOneCharacterArtworkChangeMovesTheFingerprint() throws Exception {
210 		var real = sprite();
211 		var mutated = mutateFirstPath(real);
212 		var before = SymbolProvenanceScanner.symbols(real);
213 		var after = SymbolProvenanceScanner.symbols(mutated);
214 		assertList(() -> "the mutation must not change the glyph set", after.keySet(), before.keySet().toArray());
215 		var stem = before.keySet().iterator().next();
216 		assertNotEquals(SymbolProvenanceScanner.fingerprint(before.get(stem)),
217 			SymbolProvenanceScanner.fingerprint(after.get(stem)),
218 			"a single added path command left the fingerprint unchanged, so a02 is hashing something other than the"
219 				+ " artwork - a normalised string, or a regex capturing only the opening tag, both look like this");
220 	}

222 	@Test void b02_aViewBoxChangeMovesTheFingerprint() throws Exception {
223 		// The gap this closes was explicit in the item: the stem-id guard never reads viewBox, so before this test
224 		// a glyph arriving at a foreign modulus passed every guard in the tree silently.
225 		var real = sprite();
226 		var mutated = real.replace("viewBox=\"" + REQUIRED_VIEWBOX + "\"", "viewBox=\"0 0 16 16\"");
227 		assertNotEquals(real, mutated, "the sprite no longer declares the host viewBox, so this check cannot run");
228 		var before = SymbolProvenanceScanner.symbols(real);
229 		var after = SymbolProvenanceScanner.symbols(mutated);
230 		var stem = before.keySet().iterator().next();
231 		assertNotEquals(SymbolProvenanceScanner.fingerprint(before.get(stem)),
232 			SymbolProvenanceScanner.fingerprint(after.get(stem)),
233 			"a viewBox change left the fingerprint unchanged, so the hash is not covering the opening tag");
234 	}

236 	@Test void b03_documentFamilyOriginAssertCanFail() throws Exception {
237 		// a04 pins original-artwork on the framed-family stems.  Run the SAME check against a mutated origins map and a
238 		// mutated sprite and require it to report the breach; a vacuous check (or a vacuous origins map) would not.
239 		var shipped = SymbolProvenanceScanner.symbols(sprite());
240 		var origins = SymbolProvenanceScanner.manifestOrigins(manifest());
241 		assertNotEmpty(() -> "precondition: manifest origins parse", origins);
242 		assertEmpty(() -> "precondition: the unmutated inputs must pass the check", documentFamilyProblems(shipped, origins));
243 
244 		var first = SymbolProvenanceScanner.FRAMED_FAMILY.get(0);
245 		var drifted = new LinkedHashMap<>(origins);
246 		drifted.put(first, "juneau-original");
247 		assertList(documentFamilyProblems(shipped, drifted),
248 			"document-family stem " + first + " is expected to be original-artwork after the artwork replacement but is juneau-original");
249 
250 		var missing = new LinkedHashMap<>(shipped);
251 		missing.remove(first);
252 		assertList(documentFamilyProblems(missing, origins), "the document family member " + first + " is missing from the sprite");
253 	}

255 	@Test void a08_shippedResourcesNameNoReferenceConsole() throws Exception {
256 		// The shipped sprites, manifests, scripts and stylesheets are Apache artifacts: provenance is recorded by origin
257 		// token (juneau-original / original-artwork), never by naming the console the artwork was cleared from.  The
258 		// bracketed letters keep this pattern from matching its own source text.
259 		var leak = Pattern.compile("\\bI[R]S|\\bi[r]s[-_.]");
260 		var root = RawContentSinkScanner.locateModuleRoot();
261 		assertNotNull(root, "could not locate the module's own resources");
262 		var dir = root.resolve(SymbolProvenanceScanner.RESOURCE_DIR);
263 		var offenders = new ArrayList<String>();
264 		try (var files = Files.list(dir)) {
265 			for (var f : files.filter(Files::isRegularFile).sorted().toList()) {
266 				var name = f.getFileName().toString();
267 				if (name.matches(".*\\.(md|svg|css|js)")) {
268 					var m = leak.matcher(Files.readString(f));
269 					if (m.find())
270 						offenders.add(name);
271 				}
272 			}
273 		}
274 		assertEmpty(() -> "these shipped resources name the source console; describe the reason instead", offenders);
275 		// Non-vacuous: the scan really lists the provenance manifest it is supposed to be covering.
276 		assertTrue(Files.isRegularFile(dir.resolve(SymbolProvenanceScanner.MANIFEST)), "scanned directory does not hold the manifest");
277 	}

289 		var real = sprite();
290 		assertNotEmpty(() -> "an injected literal colour went undetected, so a06 cannot fail",
291 			SymbolProvenanceScanner.offContractPaints(real.replace("fill=\"none\"", "fill=\"#1589EE\"")));
292 		assertNotEmpty(() -> "a literal colour hidden in a var() fallback went undetected, so a06 cannot fail",
293 			SymbolProvenanceScanner.offContractPaints("<symbol id=\"juneau-sym-x\"><path fill=\"var(--jc-x, #1589EE)\" d=\"M0 0\"/></symbol>"));
294 		assertEmpty(() -> "a themable var(--x, currentColor) paint must be accepted",
295 			SymbolProvenanceScanner.offContractPaints("<symbol id=\"juneau-sym-x\"><path fill=\"var(--jc-x, currentColor)\" d=\"M0 0\"/></symbol>"));
296 
297 		var stripped = real.replaceAll("\\s+stroke-width=\"[^\"]*\"", "");
298 		assertNotEquals(real, stripped, "the sprite declares no stroke-width at all, so this check cannot run");
299 		var breaches = SymbolProvenanceScanner.symbols(stripped).values().stream()
300 			.flatMap(e -> SymbolProvenanceScanner.strokedWithoutWidth(e).stream()).toList();
301 		assertNotEmpty(() -> "removing every stroke-width went undetected, so a07 cannot fail", breaches);
302 	}

304 	@Test void b06_theStrokeWidthScanSeesTheRealStrokedElements() throws Exception {
305 		// a07 asserts an empty list, which an element pattern that matched nothing would satisfy trivially.  This
306 		// pins that the sprite really does paint strokes and that they really are being examined.
307 		var stripped = SymbolProvenanceScanner.read(SymbolProvenanceScanner.SPRITE)

313 				+ " stroke-width - the element pattern has stopped matching the remaining stroked artwork");
