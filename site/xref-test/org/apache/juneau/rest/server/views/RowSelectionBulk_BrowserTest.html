32  /**
33   * The row-selection + bulk-mutation half of the module's <b>JavaScript-execution harness</b>:
34   * runs the REAL served {@code juneau-views.js} in a real headless browser and asserts, as a user would experience
35   * it, that:
36   * <ul>
37   * 	<li>per-row selection and select-all toggle a live selection set keyed by the STABLE row id (never a DOM index);
38   * 	<li>selection is persistent by default (a draw that removes a row keeps its id selected) and pruned per draw only
39   * 		under the {@code page} scope; the select-all header is tri-state and skips {@code selectableWhen}-disabled rows;
40   * 	<li>row-selection and bulk-mutation are two INDEPENDENT opt-ins - a selection-only (e.g. export) table never
41   * 		carries the bulk marker, even though both are declared via the same {@code hasSelection}/{@code hasBulk}
42   * 		DOM-attribute mechanism;
43   * 	<li>a bulk action goes through a mandatory confirm dialog and then runs either as N independent per-row writes
44   * 		(at most four in flight) or, for {@code aggregate} mode, as one POST of {@code {ids, idempotencyKey}} answered
45   * 		by a {@code BulkResult}; either way the run ends in one summary toast, succeeded and not-found ids leave the
46   * 		selection and failed ids stay selected;
47   * 	<li>the bulk toolbar sits in the right cluster, is hidden at a zero count, and the independently-versioned bulk
48   * 		sidecar is read/contract-checked correctly at runtime (R2).
49   * </ul>
50   *
51   * <h5 class='section'>Why this exists (beyond the always-on source-shape test):</h5>
52   * <p>
53   * {@link ViewsJs_Selection_Test} proves the shipped script <i>contains</i> the selection/bulk logic in the right
54   * shape; it cannot prove a checkbox click actually updates the live selection set, that a draw actually prunes an
55   * off-screen id, or that a bulk action actually issues N independent requests with N independent settled outcomes.
56   * This canary drives the real runtime in Chromium against a stubbed {@code fetch}, so those user-visible facts are
57   * measured rather than inferred.
58   *
59   * <h5 class='section'>Off by default:</h5>
60   * <p>
61   * Disabled unless the {@value #GATE} system property is set, which only the module's opt-in {@code js-tests} Maven
62   * profile does.  It reuses that profile's provisioned Node + Playwright browser and derives its own prober
63   * ({@code row-selection-bulk.cjs}) from the profile's {@code juneau.jsTests.harness} directory, so no pom change is
64   * needed.
65   *
66   * <h5 class='section'>See Also:</h5>
67   * <ul>
68   * 	<li class='jc'>{@link ModalResult_BrowserTest} &mdash; the sibling declarative-modal canary this reuses the settle
69   * 		path from.
70   * </ul>
71   */

84  	private static String resource(String path) throws IOException {
85  		try (var in = ViewsMixin.class.getResourceAsStream(path)) {
86  			assertNotNull(in, () -> "missing classpath resource: " + path);
87  			return new String(in.readAllBytes(), UTF_8);
88  		}
89  	}

91  	@BeforeAll
92  	static void probe() throws Exception {
93  		var dir = Path.of(requiredProperty("juneau.jsTests.dir"));
94  		var harness = Path.of(requiredProperty("juneau.jsTests.harness")).getParent().resolve("row-selection-bulk.cjs");
95  
96  		var fixture = "<!DOCTYPE html><html><head><meta charset=\"utf-8\"></head><body>\n<script>\n"
97  			+ resource(ViewsMixin.RENDERS_JS_RESOURCE) + "\n</script>\n<script>\n"
98  			+ resource(ViewsMixin.VIEWS_JS_RESOURCE)
99  			+ "\n</script></body></html>";
100 		var fixtureFile = Files.createDirectories(dir.resolve("fixtures")).resolve("row-selection-bulk.html");
101 		Files.write(fixtureFile, fixture.getBytes(UTF_8));
102 
103 		report = Json.to(run(dir, harness, fixtureFile), Map.class);
104 	}

106 	private static String requiredProperty(String name) {
107 		var v = System.getProperty(name);
108 		assertNotNull(v, () -> "-D" + name + " not set; the js-tests profile is responsible for providing it");
109 		return v;
110 	}

112 	private static String run(Path dir, Path harness, Path fixture) throws Exception {
113 		var cmd = List.of(System.getProperty("juneau.jsTests.node", "node"), harness.toString(), fixture.toString());
114 		var stdout = dir.resolve("row-selection-bulk-stdout.json");
115 		var stderr = dir.resolve("row-selection-bulk-stderr.txt");
116 		var pb = new ProcessBuilder(cmd).redirectOutput(stdout.toFile()).redirectError(stderr.toFile());
117 		pb.environment().put("NODE_PATH", dir.resolve("node_modules").toString());
118 		pb.environment().put("PLAYWRIGHT_BROWSERS_PATH", requiredProperty("juneau.jsTests.browsers"));
119 
120 		var p = pb.start();
121 		if (!p.waitFor(3, TimeUnit.MINUTES)) {
122 			p.destroyForcibly();
123 			fail("prober did not finish within 3m; stderr:\n" + quietRead(stderr));
124 		}
125 		assertEquals(0, p.exitValue(), () -> "prober exited non-zero; stderr:\n" + quietRead(stderr));
126 		return Files.readString(stdout);
127 	}
128 
129 	private static String quietRead(Path p) {
130 		try {
131 			return Files.readString(p);
132 		} catch (IOException e) {
133 			return "<unreadable: " + e + ">";
134 		}
135 	}
