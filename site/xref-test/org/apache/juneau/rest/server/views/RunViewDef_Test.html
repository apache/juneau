17  package org.apache.juneau.rest.server.views;
18  
19  import static org.apache.juneau.BasicTestUtils.*;
20  import static org.apache.juneau.test.bct.BctAssertions.*;
21  import static org.junit.jupiter.api.Assertions.*;
22  
23  import java.util.*;
24  
25  import org.apache.juneau.*;
26  import org.junit.jupiter.api.*;
27  
28  class RunViewDef_Test extends TestBase {
29  
30  	private static List<String> keysOf(Map<String,Object> m) {
31  		return new ArrayList<>(m.keySet());
32  	}
33  
34  	@Test void a01_defaultsToMap() {
35  		var d = RunViewDef.create("run").eventsUrl("/juneau-run-view/42/events");
36  		assertEquals(List.of("eventsUrl", "refreshMs", "poll", "compact"), keysOf(d.toMap()));
37  		assertEquals(2000L, d.toMap().get("refreshMs"));
38  		assertEquals(true, d.toMap().get("poll"));
39  		assertEquals(false, d.toMap().get("compact"));
40  	}
41  
42  	@Test void a02_refreshClampedToFloor() {
43  		assertEquals(1000L, RunViewDef.create("r").eventsUrl("/x").refreshMs(10).toMap().get("refreshMs"));
44  	}
45  
46  	@Test void a03_allKeysInOrder() {
47  		var d = RunViewDef.create("r").eventsUrl("/x").title("T").rawHref("#raw-O{offset}");
48  		assertEquals(RunViewDef.KEYS, keysOf(d.toMap()));
49  	}
50  
51  	@Test void b01_eventsUrlMustBeSameOriginPath() {
52  		assertThrowsWithMessage(IllegalArgumentException.class, "eventsUrl", () -> RunViewDef.create("r").eventsUrl("https://evil.example/x").validate());
53  		assertThrowsWithMessage(IllegalArgumentException.class, "eventsUrl", () -> RunViewDef.create("r").eventsUrl("/a/../b").validate());
54  	}
55  
56  	@Test void b02_idTemplateOnlyOnRowDetail() {
57  		assertThrowsWithMessage(IllegalArgumentException.class, "{id}", () -> RunViewDef.create("r").eventsUrl("/runs/{id}/events").validate());
58  		assertDoesNotThrow(() -> RunViewDef.create("r").type(RegionDef.TYPE_ROW_DETAIL).eventsUrl("/runs/{id}/events").validate());
59  	}
60  
61  	@Test void b03_eventsUrlRequiredUnlessPollFalse() {
62  		assertThrowsWithMessage(IllegalArgumentException.class, "eventsUrl is required", () -> RunViewDef.create("r").validate());
63  		assertDoesNotThrow(() -> RunViewDef.create("r").poll(false).validate());
64  		assertThrowsWithMessage(IllegalArgumentException.class, "eventsUrl", () -> RunViewDef.create("r").poll(false).eventsUrl("//evil").validate());
65  	}
66  
67  	@Test void b04_rawHrefTemplate() {
68  		assertDoesNotThrow(() -> RunViewDef.create("r").eventsUrl("/x").rawHref("#raw-O{offset}").validate());
69  		assertDoesNotThrow(() -> RunViewDef.create("r").eventsUrl("/x").rawHref("#term-O{offset}").validate());
70  		assertThrowsWithMessage(IllegalArgumentException.class, "rawHref", () -> RunViewDef.create("r").eventsUrl("/x").rawHref("#raw-L").validate());
71  		assertThrowsWithMessage(IllegalArgumentException.class, "exactly one {offset}", () -> RunViewDef.create("r").eventsUrl("/x").rawHref("#a{line}{offset}").validate());
72  	}
73  
74  	@Test void b05_typeAndIdAndRefresh() {
75  		assertThrowsWithMessage(IllegalArgumentException.class, "type", () -> RunViewDef.create("r").eventsUrl("/x").type("bogus").validate());
76  		assertThrowsWithMessage(IllegalArgumentException.class, "id", () -> RunViewDef.create(" ").eventsUrl("/x").validate());
77  		assertThrowsWithMessage(IllegalArgumentException.class, "refreshMs", () -> RunViewDef.create("r").eventsUrl("/x").refreshMs(0).validate());
78  	}
79  
80  	@Test void c01_fromMapRejectsUnknownKey() {
81  		assertThrowsWithMessage(IllegalArgumentException.class, "unknown", () -> RunViewDef.fromMap("r", Map.of("bogus", 1)));
82  		assertThrowsWithMessage(IllegalArgumentException.class, "unknown", () -> RunViewDef.fromMap("r", Map.of("fwLabels", "x")));
83  	}
84  
85  	@Test void c02_roundTrip() {
86  		var d = RunViewDef.create("r").eventsUrl("/x").refreshMs(5000).poll(true).compact(true).title("T").rawHref("#raw-O{offset}");
87  		assertEquals(d.toMap(), RunViewDef.fromMap("r", d.toMap()).toMap());
88  	}
89  
90  	@Test void c03_fromMapTypeChecks() {
91  		assertThrowsWithMessage(IllegalArgumentException.class, "must be a string", () -> RunViewDef.fromMap("r", Map.of("eventsUrl", 1)));
92  		assertThrowsWithMessage(IllegalArgumentException.class, "must be an integer", () -> RunViewDef.fromMap("r", Map.of("refreshMs", "x")));
93  		assertThrowsWithMessage(IllegalArgumentException.class, "must be a boolean", () -> RunViewDef.fromMap("r", Map.of("poll", "x")));
94  		assertEquals(3000L, RunViewDef.fromMap("r", Map.of("refreshMs", 3000.0)).refreshMs);
95  	}
96  
97  	@Test void d01_forMixin() {
98  		assertEquals("/app/juneau-run-view/{id}/events", RunViewDef.forMixin("r", "/app", "{id}").toMap().get("eventsUrl"));
99  		assertEquals("/app/juneau-run-view/r1/events", RunViewDef.forMixin("r", "/app/", "r1").toMap().get("eventsUrl"));
100 		assertThrowsWithMessage(IllegalArgumentException.class, "runIdTemplate", () -> RunViewDef.forMixin("r", "/app", "a b"));
101 	}
102 
103 	@Test void d02_forMixinMatchesRoutes() {
104 		var m = RunViewDef.forMixin("x", "/petstore", "job-1").toMap();
105 		assertEquals("/petstore" + RunViewMixin.EVENTS_PATH.replace("{runId}", "job-1"), m.get("eventsUrl"));
106 	}
107 
108 	@Test void d03_toRegionValidates() {
109 		var r = RunViewDef.create("run").eventsUrl("/x").title("T").rawHref("#raw-O{offset}").toRegion();
110 		assertBean(r, "id,populate", "run,run-view");
111 		assertEquals(List.of(RunViewDef.POPULATOR), List.copyOf(r.allowedPopulators));
112 	}
113 }
