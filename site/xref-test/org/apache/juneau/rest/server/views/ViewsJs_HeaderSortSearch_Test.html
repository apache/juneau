44  	private static String functionBody(String body, String signature) {
45  		var start = body.indexOf(signature);
46  		assertTrue(start >= 0, () -> "'" + signature + "' not found:\n" + body);
47  		var end = body.indexOf("\n\t}", start);
48  		return body.substring(start, end < 0 ? body.length() : end);
49  	}

189 		var body = cWithMixin.get(ViewsMixin.VIEWS_CSS_PATH).run().assertStatus(200).getContent().asString();

192 			"DT ::before/::after content must be none !important so dataTables.dataTables.css cannot win:\n" + excerptSort(body));
193 		assertFalse(body.contains(":not(:has(svg)):before"),
194 			":not(:has(svg)) chevrons lost to DT content:▲ when an SVG is present:\n" + excerptSort(body));
195 		assertFalse(hasContentValue(body, "▲"), "DT2 unicode up-triangle must not remain as content");
196 		assertFalse(hasContentValue(body, "▼"), "DT2 unicode down-triangle must not remain as content");
197 		assertFalse(hasContentValue(body, "✓"), body);
198 		assertFalse(hasContentValue(body, "✔"), body);
199 		assertFalse(body.contains("content: \"\\25"), body);
200 	}

202 	private static String excerptSort(String css) {
203 		var start = css.indexOf("span.dt-column-order");
204 		return start < 0 ? css : css.substring(start, Math.min(css.length(), start + 1800));
205 	}
206 
207 	private static boolean hasContentValue(String css, String glyph) {
208 		return css.contains("content: \"" + glyph + "\"") || css.contains("content: '" + glyph + "'");
209 	}
