79  	@BeforeAll
80  	static void probeIfNodeAvailable() throws Exception {
81  		if (!nodeAvailable())
82  			return;
83  		var harness = locateHarness();
84  		if (harness == null)
85  			return;
86  		var rendersFile = Files.createTempFile("juneau-renders-", ".js");
87  		try {
88  			Files.writeString(rendersFile, rendersJs(), UTF_8);
89  			report = Json.to(runNode(harness, rendersFile), Map.class);
90  		} finally {
91  			Files.deleteIfExists(rendersFile);
92  		}
93  	}

95  	private static boolean nodeAvailable() {
96  		try {
97  			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
98  			if (!p.waitFor(5, TimeUnit.SECONDS)) {
99  				p.destroyForcibly();
100 				return false;
101 			}
102 			return p.exitValue() == 0;
103 		} catch (Exception e) {
104 			return false;
105 		}
106 	}

108 	private static Path locateHarness() {
109 		var basedir = System.getProperty("basedir");
110 		if (basedir != null) {

112 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
113 		}

116 			"juneau-rest/juneau-rest-server-views/src/test/js/renders.cjs"

119 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
120 		}
121 		return null;

124 	private static String runNode(Path harness, Path rendersJs) throws Exception {
125 		var stdout = Files.createTempFile("renders-stdout-", ".json");
126 		var stderr = Files.createTempFile("renders-stderr-", ".txt");
127 		try {
128 			var pb = new ProcessBuilder(List.of("node", harness.toString(), rendersJs.toString()))
129 				.redirectOutput(stdout.toFile())
130 				.redirectError(stderr.toFile());
131 			pb.environment().put("TZ", LOCAL_TZ);
132 			var p = pb.start();
133 			if (!p.waitFor(30, TimeUnit.SECONDS)) {
134 				p.destroyForcibly();
135 				fail("renders.cjs did not finish within 30s; stderr:\n" + quietRead(stderr));
136 			}
137 			if (p.exitValue() != 0)
138 				fail("renders.cjs exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr)
139 					+ "\nstdout:\n" + quietRead(stdout));
140 			return Files.readString(stdout, UTF_8);
141 		} finally {
142 			Files.deleteIfExists(stdout);
143 			Files.deleteIfExists(stderr);
144 		}
145 	}

147 	private static String quietRead(Path p) {
148 		try { return Files.readString(p, UTF_8); }
149 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
150 	}

462 	/**
463 	 * Pins the fix for {@code javascript:S5852} in {@code normalizeTagToken}.  The original trim,
464 	 * {@code .replace(/^-+|-+$/g, "")}, retried its {@code -+$} alternative at every offset inside a dash run,
465 	 * which is quadratic as soon as the run is bracketed by non-dashes - a shape an all-dash probe cannot expose,
466 	 * because {@code ^-+} swallows an all-dash string in one bite.  A 160,000-dash cell value took ~8.8s before
467 	 * the fix and ~0.1ms after, so the threshold here is three orders of magnitude clear of both.
468 	 */
469 	@Test void e03_normalizeTagToken_isLinearAgainstABracketedDashRun() {
470 		var r = report();
471 		assertEquals(true, r.get("tagToken_adversarialUnchanged"));   // nothing to trim: both ends are non-dashes
472 		var ms = ((Number) r.get("tagToken_adversarialMs")).longValue();
473 		assertTrue(ms < 2000, () -> "normalizeTagToken took " + ms + "ms on a 160,000-dash value; expected linear");
474 	}
