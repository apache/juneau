30  /**
31   * Pins that every icon name the chrome scripts ask the registry for is registered in {@code juneau-icons.js},
32   * every registered stem exists in {@code juneau-symbols.svg}, and every shipped symbol is reachable by a name
33   * (one shipped set, no second inline/CSS path).
34   *
35   * <h5 class='section'>Example:</h5>
36   * <p class='bjava'>
37   * 	<jc>// A new call site that asks for an unregistered name fails a01 and names the missing icon:</jc>
38   * 	<jc>//   resolveIcon("my_new_icon")  -&gt;  a01 expected [] but was [my_new_icon]</jc>
39   * </p>
40   */
41  class ViewsJs_IconCallSites_Test extends TestBase {
42  
43  	private static final String BASE = "/org/apache/juneau/views/";
44  	private static final Pattern REG = Pattern.compile("\\breg\\(\"([^\"]+)\",\\s*\"([^\"]+)\"");
45  	private static final Pattern CALL = Pattern.compile("resolveIcon\\??\\.?\\(\\s*\"([A-Za-z0-9_-]+)\"");
46  	private static final Pattern PILL = Pattern.compile("pagingPillButton\\(\"[^\"]*\",\\s*\"([A-Za-z0-9_-]+)\"");
47  	private static final Pattern DEFAULTS = Pattern.compile("DEFAULT_ICONS = \\{([^}]*)\\}");
48  	private static final Pattern DEFAULT_VALUE = Pattern.compile(":\\s*\"([A-Za-z0-9_-]+)\"");
49  	private static final Pattern SYMBOL = Pattern.compile("<symbol\\s+id=\"juneau-sym-([^\"]+)\"");
50  
51  	private static String res(String name) throws IOException {
52  		try (var in = ViewsMixin.class.getResourceAsStream(BASE + name)) {
53  			assertNotNull(in, () -> "missing resource " + name);
54  			return new String(in.readAllBytes(), UTF_8);
55  		}
56  	}
57  
58  	private static Set<String> all(Pattern p, String text, int group) {
59  		var out = new TreeSet<String>();
60  		var m = p.matcher(text);
61  		while (m.find())
62  			out.add(m.group(group));
63  		return out;
64  	}
65  
66  	@Test void a01_everyCallSiteNameIsRegistered() throws Exception {
67  		var iconsJs = res("juneau-icons.js");
68  		var registered = all(REG, iconsJs, 1);
69  		var used = new TreeSet<String>();
70  		for (var f : new String[]{"juneau-views.js", "juneau-ribbon.js", "juneau-config.js", "juneau-helpers.js", "juneau-regions.js"}) {
71  			var src = res(f);
72  			used.addAll(all(CALL, src, 1));
73  			used.addAll(all(PILL, src, 1));
74  		}
75  		var defaults = DEFAULTS.matcher(res("juneau-ribbon.js"));
76  		assertTrue(defaults.find(),
77  			"the DEFAULT_ICONS table in juneau-ribbon.js no longer matches " + DEFAULTS.pattern()
78  				+ " - update the pattern so the ribbon's default icon names stay scanned");
79  		var defaultNames = all(DEFAULT_VALUE, defaults.group(1), 1);
80  		assertNotEmpty(() -> "DEFAULT_ICONS matched but no icon names were extracted from it", defaultNames);
81  		used.addAll(defaultNames);
82  		var missing = new TreeSet<>(used);
83  		missing.removeAll(registered);
84  		assertList(missing.stream().toList());
85  	}
86  
87  	@Test void a02_everyRegisteredStemExistsInTheSprite() throws Exception {
88  		var stems = all(REG, res("juneau-icons.js"), 2);
89  		var shipped = all(SYMBOL, res("juneau-symbols.svg"), 1);
90  		var missing = new TreeSet<>(stems);
91  		missing.removeAll(shipped);
92  		assertList(missing.stream().toList());
93  	}
94  
95  	@Test void a03_everyShippedSymbolHasAName() throws Exception {
96  		var stems = all(REG, res("juneau-icons.js"), 2);
97  		var shipped = all(SYMBOL, res("juneau-symbols.svg"), 1);
98  		var unnamed = new TreeSet<>(shipped);
99  		unnamed.removeAll(stems);
100 		assertList(unnamed.stream().toList());
101 	}

106 	/** Returns the body of {@code function name(...)} up to the next line that is a lone tab-indented closing brace. */
107 	private static String body(String src, String signature) {
108 		var start = src.indexOf(signature);
109 		assertTrue(start >= 0, () -> "function not found: " + signature);
110 		var end = src.indexOf("\n\t}\n", start);
111 		return src.substring(start, end);
112 	}

114 	@Test void a04_toolbarButtonHasNoTextFallbackAndKeepsItsAccessibleName() throws Exception {
115 		var fn = body(res("juneau-views.js"), "function toolbarButton(");
116 		assertFalse(fn.contains("textContent"), "toolbarButton must not fall back to label text");
117 		assertTrue(fn.contains("stampChromeTip(b, label)"), "toolbarButton must still stamp aria-label / tooltip");
118 	}

120 	@Test void a05_actionTriggerHasNoEllipsisFallbackAndKeepsItsAriaLabel() throws Exception {
121 		var src = res("juneau-views.js");
122 		var fn = body(src, "function actionTriggerMarkup(");
123 		assertFalse(fn.contains("u22EF") || fn.contains("\u22ef") || fn.contains("\u22EF"), "no ellipsis text fallback");
124 		assertTrue(fn.contains("resolveIcon(\"more_vert\")"), "trigger glyph resolves through the registry");
125 		assertTrue(fn.contains("aria-label=\"Row actions\""), "trigger keeps its accessible name");
126 	}

128 	@Test void a06_sortHeaderHasNoExpandMoreStandIn() throws Exception {
129 		var src = res("juneau-views.js");
130 		var at = src.indexOf("resolveIcon?.(\"sort\")");
131 		assertTrue(at >= 0, "sort header resolves the sort glyph");
132 		var window = src.substring(at, at + 200);
133 		assertFalse(window.contains("expand_more"), "no expand_more stand-in after the sort lookup");
134 		assertFalse(window.contains("||"), "no second resolution path after the sort lookup");
135 	}

137 	@Test void a07_rowExpandChevronsHaveNoUnicodeTriangleFallback() throws Exception {
138 		var fn = body(res("juneau-views.js"), "function detailsControlCellMarkup(");
139 		assertFalse(fn.contains("25B8") || fn.contains("25BE"), "no unicode triangle fallback");
140 		assertList(all(CALL, fn, 1).stream().toList(), "chevron_right", "expand_more");
141 	}

144 		var src = res("juneau-views.js");
145 		var at = src.indexOf("resolveIcon?.(\"search\")");
146 		assertTrue(at >= 0, "column search resolves the search glyph");
147 		assertFalse(src.substring(at, at + 120).contains("||"), "no second path after the search lookup");
