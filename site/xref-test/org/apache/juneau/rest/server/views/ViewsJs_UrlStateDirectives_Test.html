27  /** Node behavioral checks for the {@code ?state=} directive registry, the {@code window}/{@code ids}/{@code day} built-ins, and their live wire. */
28  class ViewsJs_UrlStateDirectives_Test extends TestBase {
29  
30  	private static Map<?,?> report() {
31  		var r = NodeHarness.report("urlstate-directives.cjs", ViewsMixin.RENDERS_JS_RESOURCE, ViewsMixin.VIEWS_JS_RESOURCE, ViewsMixin.URLSTATE_JS_RESOURCE);
32  		assumeTrue(r != null, "node (or urlstate-directives.cjs) not available - behavioral layer skipped");
33  		return r;
34  	}
35  
36  	@Test void d01_registerDirectiveRejectsReservedAndMalformedNames() {
37  		var r = report();
38  		assertEquals(true, r.get("registerReserved"));
39  		assertEquals(true, r.get("registerBadName"));
40  		assertEquals(true, r.get("registerOk"));
41  	}
42  
43  	@Test void d02_registerDirectiveRejectsDuplicateNames() {
44  		assertEquals(true, report().get("registerDuplicate"));
45  	}
46  
47  	@Test void d03_registeredDirectiveRoundTripsThroughExt() {
48  		var r = report();
49  		assertEquals("abc", r.get("extRoundTrip"));
50  		assertEquals("probe(abc)", r.get("encoded"));
51  	}
52  
53  	@Test void d04_unregisteredDirectiveStillDropped() {
54  		assertNull(report().get("stillDroppedUnregistered"));
55  	}
56  
57  	@Test void d05_isEmptyStateChecksExt() {
58  		var r = report();
59  		assertEquals(false, r.get("notEmptyBecauseOfExtOnly"));
60  		assertEquals(true, r.get("emptyWithNoExt"));
61  		assertEquals(true, r.get("emptyWithBlankExt"));
62  	}
63  
64  	@Test void d06_codecThrowOnDecodeIsEJS68AndDirectiveDropped() {
65  		var r = report();
66  		assertEquals("E-JS-68", r.get("throwingCodecLogged"));
67  		assertNull(r.get("throwingCodecValue"));
68  		assertEquals("a", r.get("throwingCodecKeepsTab"));
69  	}
70  
71  	@Test void d07_encodeArgPercentEncodesTheReservedChars() {
72  		assertEquals("a%3Bb%28c%29d%2Ce%3Df%25", report().get("encodeArgSample"));
73  	}
74  
75  	@Test void d08_windowBuiltin_roundTripsStartAndEnd() {
76  		var r = report();
77  		assertEquals("2026-10-01T00:00:00Z", r.get("windowStart"));
78  		assertEquals("2026-10-02T00:00:00Z", r.get("windowEnd"));
79  	}
80  
81  	@Test void d09_windowBuiltin_toParamsRenamesByConfig() {
82  		var r = report();
83  		assertEquals("2026-10-01T00:00:00Z", r.get("windowParamFrom"));
84  		assertEquals("2026-10-02T00:00:00Z", r.get("windowParamTo"));
85  		assertEquals("{\"from\":\"x\"}", r.get("windowOpenEnded"));
86  	}
87  
88  	@Test void d10_idsBuiltin_trimsDedupesAndCaps() {
89  		var r = report();
90  		assertEquals("a,b", r.get("idsNormalized"));
91  		assertEquals(true, r.get("idsCappedAtMax"));
92  		assertEquals("Showing the first 2 ids", r.get("idsCapMessage"));
93  		assertNull(r.get("idsNoMessageWhenNotCapped"));
94  		assertEquals("{\"idList\":\"a,b\"}", r.get("idsParam"));
95  	}
96  
97  	@Test void d11_idsBuiltin_parseThenSerializeIsIdempotent() {
98  		var r = report();
99  		assertEquals(true, r.get("idsIdempotent"));
100 		assertEquals(true, r.get("idsSpecialCharsRoundTrip"));
101 	}
102 
103 	@Test void d12_dayBuiltin_acceptsOnlyARealCalendarDate() {
104 		var r = report();
105 		assertEquals("2026-10-08", r.get("dayParsed"));
106 		assertEquals("day(2026-10-08)", r.get("dayRoundTrip"));
107 		assertEquals("{\"on\":\"2026-10-08\"}", r.get("dayParam"));
108 		assertEquals("{\"day\":\"2026-10-08\"}", r.get("dayDefaultParam"));
109 		assertEquals("null|null|null|null|null|null|null", r.get("dayMalformed"));
110 		assertEquals("", r.get("dayMalformedLeavesNoState"));
111 		// Years 0-99 are real dates (Date.UTC would shift them into the 1900s); 0100 is not a leap year.
112 		assertEquals("0000-01-01|0099-12-31|null", r.get("dayEarlyYears"));
113 	}
114 
115 	@Test void d13_dayBuiltin_requestCarriesTheParamNameFromTheViewConfig() {
116 		var r = report();
117 		assertEquals(true, r.get("setDayOk"));
118 		assertEquals("{\"idList\":\"z\",\"on\":\"2026-10-08\"}", r.get("dayRequestParams"));
119 		assertEquals(1, ((Number)r.get("dayReloaded")).intValue());
120 	}
121 
122 	@Test void e01_liveWire_seedsRequestParamsFromTheAddressBar() {
123 		var r = report();
124 		assertEquals("[\"p\",\"q\"]", r.get("seededFromUrl"));
125 		assertEquals("{\"idList\":\"p,q\"}", r.get("requestParamsSeeded"));
126 		assertEquals(1, ((Number)r.get("popstateListener")).intValue());
127 	}
128 
129 	@Test void e02_liveWire_setDirectiveUpdatesBarParamsAndRefetches() {
130 		var r = report();
131 		assertEquals(true, r.get("setDirectiveOk"));
132 		assertEquals(1, ((Number)r.get("setDirectiveReloaded")).intValue());
133 		var bar = (String)r.get("addressBar");
134 		assertTrue(bar.contains("window(start=2026-10-01T00:00:00Z;end=2026-10-02T00:00:00Z)"), bar);
135 		assertTrue(bar.contains("ids(p,q)"), bar);
136 		var params = (String)r.get("requestParamsAfterSet");
137 		assertTrue(params.contains("\"from\":\"2026-10-01T00:00:00Z\"") && params.contains("\"to\":\"2026-10-02T00:00:00Z\""), params);
138 		assertEquals("{\"idList\":\"p,q\"}", r.get("requestParamsAfterClear"));
139 		assertFalse(((String)r.get("addressBarAfterClear")).contains("window("));
140 		assertEquals(true, r.get("setDirectiveUnknown"));
141 	}
142 
143 	@Test void e03_liveWire_popstateReappliesAndRefetches() {
144 		var r = report();
145 		assertEquals("[\"z\"]", r.get("popstateIds"));
146 		assertEquals(1, ((Number)r.get("popstateReloaded")).intValue());
147 	}
