17  package org.apache.juneau.rest.server.views;
18  
19  import static org.apache.juneau.rest.server.views.ConsoleBrowserFixture.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.util.*;
23  
24  import org.apache.juneau.*;
25  import org.apache.juneau.marshall.marshaller.Json;
26  import org.junit.jupiter.api.*;
27  import org.junit.jupiter.api.condition.*;
28  
29  /**
30   * A {@code TableSpec}-built table card mounts the View Settings gear in Chromium when {@code columnConfig(true)} is set
31   * and has no gear when it is unset. {@code PageSpec.table(...)} passes {@link TableSpec#toCardBody()} through unchanged,
32   * so the page contract here is the parity corpus case with its table body replaced by the {@code TableSpec} output.
33   */
34  @EnabledIfSystemProperty(named=ConsoleBrowserFixture.GATE, matches="true",
35  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -f juneau-rest/juneau-rest-server-views/pom.xml test`")
36  class TableSpecColumnConfig_BrowserTest extends TestBase {
37  
38  	private static final String PROBE = """
39  		<script>
40  		window.__fetches = 0;
41  		(function () {
42  		  var open = XMLHttpRequest.prototype.open;
43  		  XMLHttpRequest.prototype.open = function (m, u) { if (String(u).indexOf("/rest/slo/data") >= 0) window.__fetches++; return open.apply(this, arguments); };
44  		  var f = window.fetch;
45  		  window.fetch = function (u) { if (String(u && u.url || u).indexOf("/rest/slo/data") >= 0) window.__fetches++; return f.apply(this, arguments); };
46  		})();
47  		window.__probe = {
48  		  get fetches() { return window.__fetches; },
49  		  get gears() { return document.querySelectorAll(".juneau-config-chooser-btn").length; },
50  		  get ths() { return Array.from(document.querySelectorAll("#slo-body table thead th")).map(function (e) { return e.textContent.trim(); }); }
51  		};
52  		</script>
53  		""";
54  
55  	private static final String ROWS = """
56  		[{"id":"1","pod":"p1","cause":"disk"}]
57  		""";
58  
59  	private static Map<String,Map<String,Object>> report;
60  
61  	private static String page(TableSpec table, boolean shellFirst) throws Exception {
62  		return page(table, shellFirst, viewsPack());
63  	}
64  
65  	@SuppressWarnings({
66  		"unchecked" // Casts a parsed JSON value to its known Map/List shape.
67  	})
68  	private static String page(TableSpec table, boolean shellFirst, String pack) throws Exception {
69  		var corpus = (Map<String,Object>)Json.to(resource("/pagespec-corpus/03-table-every-feature.json"), Map.class);
70  		var contract = (Map<String,Object>)corpus.get("contract");
71  		var card = (Map<String,Object>)((List<?>)contract.get("cards")).get(0);
72  		card.put("table", table.toCardBody());
73  		// shellFirst is the order a real page emits (shell, then the toolkit pack); the other loads the pack first.
74  		return shellFirst
75  			? contractPage(Json.of(contract), "", PROBE, pack)
76  			: contractPage(Json.of(contract), "", pack + PROBE, "");
77  	}
78  
79  	@BeforeAll
80  	static void probe() throws Exception {
81  		var gear = TableSpec.create("slo").dataUrl("/rest/slo/data").columns(Column.create("pod"), Column.create("cause").label("Cause").defaultVisible(false)).columnConfig(true);
82  		var plain = TableSpec.create("slo").dataUrl("/rest/slo/data").columns(Column.create("pod"));
83  		report = ConsoleBrowserFixture.create("tablespec-column-config")
84  			.asset("/rest/slo/data", ROWS, "application/json")
85  			.page("gear", ORIGIN + "/gear", page(gear, false))
86  			.page("gearShellFirst", ORIGIN + "/gear-sf", page(gear, true))
87  			.page("noConfigJs", ORIGIN + "/no-config", page(gear, true, viewsPack().replace("<script>\n" + resource(ViewsMixin.CONFIG_JS_RESOURCE) + "\n</script>\n", "")))
88  			.page("plain", ORIGIN + "/plain", page(plain, false))
89  			.waitFor("#slo-body tbody tr")
90  			.run();
91  	}
92  
93  	@Test void a01_columnConfigSet_mountsGear() {
94  		assertNoShellErrors(report.get("gear"));
95  		assertEquals(1, ((Number)ConsoleBrowserFixture.probe(report.get("gear")).get("gears")).intValue(), () -> report.get("gear").toString());
96  	}
97  
98  	@Test void a03_columnConfigSet_shellFirst_mountsGearAndHidesDefaultInvisibleColumn() {
99  		var r = report.get("gearShellFirst");
100 		assertNoShellErrors(r);
101 		var probe = ConsoleBrowserFixture.probe(r);
102 		assertEquals(1, ((Number)probe.get("gears")).intValue(), r::toString);
103 		assertEquals(List.of("pod"), probe.get("ths"), r::toString);
104 	}
105 
106 	@Test void a05_shellFirst_fetchesRowsOnce() {
107 		var r = report.get("gearShellFirst");
108 		assertEquals(1, ((Number)ConsoleBrowserFixture.probe(r).get("fetches")).intValue(), r::toString);
109 	}
110 
111 	@Test void a06_viewsPackWithoutConfigJs_stillRendersEveryColumn() {
112 		var r = report.get("noConfigJs");
113 		assertNoShellErrors(r);
114 		var probe = ConsoleBrowserFixture.probe(r);
115 		assertEquals(0, ((Number)probe.get("gears")).intValue(), r::toString);
116 		assertEquals(List.of("pod", "Cause"), probe.get("ths"), r::toString);
117 	}
118 
119 	@Test void a04_columnConfigSet_packFirst_hidesDefaultInvisibleColumn() {
120 		var r = report.get("gear");
121 		assertEquals(List.of("pod"), ConsoleBrowserFixture.probe(r).get("ths"), r::toString);
122 	}
123 
124 	@Test void a02_columnConfigUnset_hasNoGear() {
125 		assertNoShellErrors(report.get("plain"));
126 		assertEquals(0, ((Number)ConsoleBrowserFixture.probe(report.get("plain")).get("gears")).intValue(), () -> report.get("plain").toString());
127 	}
128 }
