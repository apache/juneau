28  /**
29   * Behavioral coverage for the stock SSE source.  {@code bus-bridge-sse.cjs} drives
30   * {@code JuneauViews.bus.sources.sse} with a scripted {@code fetch}, hand-fed response streams and a fake clock, and
31   * reports the session POST, the stream GET, chunked frame parsing, the HTTP status table, session-body checks,
32   * CSRF, the same-origin rule, the watchdog, per-message errors and detach.  Gated on {@code node} being on
33   * {@code PATH}.
34   */

37  	private static final String SID = "a".repeat(64);

39  	private static Map<String,Object> report;
40  
41  	@BeforeAll
42  	static void runHarness() {
43  		report = BusHarness.run("bus-bridge-sse.cjs", "/org/apache/juneau/views/juneau-bus.js");
44  	}

46  	@SuppressWarnings({
47  		"unchecked" // Casts a parsed JSON value to its known Map/List shape.
48  	})
49  	private static Map<String,Object> section(String... path) {
50  		assumeTrue(report != null, "node not on PATH");
51  		Map<String,Object> m = report;
52  		for (var p : path) {
53  			var next = (Map<String,Object>) m.get(p);
54  			assertNotNull(next, "report has no '" + String.join(".", path) + "': " + report);
55  			m = next;
56  		}
57  		return m;
58  	}
59  
60  	private static void assertJson(String expected, Object actual) {
61  		try {
62  			assertEquals(Json.to(expected, Object.class), actual, () -> String.valueOf(actual));
63  		} catch (Exception e) {
64  			fail("bad expected JSON: " + expected + ": " + e);
65  		}
66  	}

183 	@SuppressWarnings({
184 		"unchecked" // Casts a parsed JSON value to its known Map/List shape.
185 	})
186 	void a11_perMessageProblemsKeepTheStreamUp() {
187 		var r = section("perMessage");
188 		assertJson("[\"E-JS-58\",\"E-JS-55\",\"E-JS-59\",\"E-JS-54\"]", r.get("codes"));
189 		var m = (List<String>) r.get("messages");
190 		// The E-JS-58 detail is the engine's JSON.parse text, so only its prefix is pinned.
191 		assertTrue(m.get(0).startsWith("bridge 's': bad frame (not-json): "), m::toString);
192 		assertEquals(List.of(
193 			"bridge 's': upstream 'app.up' was refused by the server: bus:upstream-denied not granted",
194 			"bridge 's': server error bus:boom: kaput",
195 			"bridge 's' delivered 'app.secret', which it was not granted; dropped"
196 		), m.subList(1, 4));
197 		assertJson("[false,false,false,false]", r.get("banners"));
198 		assertEquals("open", r.get("state"));
199 		// 'event: other' and default 'message' events are not bus frames: app.a keeps its resync value.
200 		assertJson("{\"n\":1}", r.get("a"));
201 		assertNull(r.get("secret"));
202 		assertEquals(false, r.get("cancelled"));
203 	}
