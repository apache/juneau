17  package org.apache.juneau.rest.server.views;
18  
19  import java.util.*;
20  
21  /**
22   * Test-only bean for {@code ribbon-corpus.json}: the shared ribbon parity contract read by {@link RibbonCorpus_Test}
23   * (Java layer) and {@code ribbon-corpus.cjs} (JS layer).  Not public API.
24   */
25  public class RibbonCorpus {
26  
27  	/** Corpus schema version. */
28  	public int version;
29  
30  	/** Rows shared by every case. */
31  	public List<CorpusRow> rows;
32  
33  	/** Default DataTables column array (a case may override it). */
34  	public List<CorpusColumn> columns;
35  
36  	/** The cases. */
37  	public List<Case> cases;
38  
39  	/** One DataTables column as the corpus declares it. */
40  	public static class CorpusColumn {
41  		/** Row property, or null for a synthetic column. */
42  		public String data;
43  		/** Column name. */
44  		public String name;
45  		/** Whether the column takes a search. */
46  		public boolean searchable = true;
47  	}
48  
49  	/** One corpus case. */
50  	public static class Case {
51  		/** Stable case name. */
52  		public String name;
53  		/** Optional per-case column override. */
54  		public List<CorpusColumn> columns;
55  		/** viewDef.ribbon as the sidecar carries it (untyped; only JS interprets it). */
56  		public List<Map<String,Object>> ribbon;
57  		/** Active state (absent for the default cases). */
58  		public Map<String,Object> state;
59  		/** Stored state for the default cases. */
60  		public Map<String,String> persisted;
61  		/** dtIndex to user-typed $-expression. */
62  		public Map<String,String> userSearch;
63  		/** Expected outputs. */
64  		public Expect expect;
65  
66  		@Override /* Object */
67  		public String toString() { return name; }
68  	}
69  
70  	/** Expected outputs of one case. */
71  	public static class Expect {
72  		/** JS: derived state (default cases only). */
73  		public Map<String,Object> state;
74  		/** JS: ribbonColumnSearches output. */
75  		public Map<String,String> columnSearches;
76  		/** JS: ribbonQueryParams output. */
77  		public Map<String,String> queryParams;
78  		/** JS and Java: per-column values after the merge (optional). */
79  		public Map<String,String> merged;
80  		/** Java: BeanQuery.getSearch(); null means no search. */
81  		public String search;
82  		/** Java and JS: matching row ids in input order; null skips the rows layer. */
83  		public List<Integer> rowIds;
84  		/** JS: substring a console.error must contain (optional). */
85  		public String consoleError;
86  	}
87  }
