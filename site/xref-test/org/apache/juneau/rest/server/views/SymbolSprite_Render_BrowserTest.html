33  /**
34   * Rasterises every glyph in {@code juneau-symbols.svg} at its real render sizes and writes a set of review sheets
35   * plus a per-glyph metric report under {@code target/symbol-sprite/}.
36   *
37   * <h5 class='section'>Why a rasteriser and not a source assertion:</h5>
38   * <p>
39   * {@code juneau-icons.js} paints an icon by injecting a host {@code <svg viewBox="0 0 24 24">} carrying a single
40   * {@code <use href="#juneau-sym-{stem}"/>}, and it deliberately does <b>not</b> fall back when a stem fails to
41   * resolve - the host element is simply left empty. So a mangled, mis-scaled or entirely unresolvable glyph
42   * renders as <b>nothing, silently</b>, with no error anywhere. Reading the sprite's ids catches a rename;
43   * fingerprinting its bytes catches a paste; neither can tell a drawing from a blank square. Only a rasteriser
44   * can, and that is what {@link #a01_everyStemRendersWithInkAtBothRenderSizes()} asserts.
45   *
46   * <h5 class='section'>The other half is a report, not an assertion, and that is deliberate:</h5>
47   * <p>
48   * These glyphs are painted at the stylesheet's ribbon and paging-pill sizes. At that size a stroke either lands on the
49   * pixel grid or smears across it, and no source-level review can see which. The sheets and metrics written here
50   * exist so a human can compare a redrawn set against its predecessor <i>as pixels</i>, and - the part per-glyph
51   * review structurally cannot do - see the four document glyphs beside each other, where family drift lives.
52   * <p>
53   * <b>Do not add a threshold assertion on any of the four per-glyph metrics above (ink/solid/gradient/mush).</b>
54   * Antialiasing coverage is a property of the Chromium build doing the rasterising. The {@code js-tests} profile
55   * pins that build, which makes these numbers <i>reproducible</i> today but not <i>stable</i> across a future pin
56   * bump - so a numeric gate on one of them would convert a routine Playwright version bump into a false artwork
57   * regression, and the reflex fix for a false regression is to loosen the number until it passes, which leaves a
58   * gate that asserts nothing. The numbers are for a human to read.
59   * <p>
60   * <b>The one exception is {@link #c01_fourLockedGlyphsStayMutuallyDistinguishable()}</b> ({@code
61   * the locked-scope glyph distinguishability pin}), which thresholds the prober's {@code adjacencyDiffs} - the mean per-pixel luminance
62   * difference <i>between two glyphs rasterised in the same run, same size, same Chromium build</i>, not either
63   * glyph's absolute coverage. A pin bump moves both sides of that comparison together, so the pairwise diff does
64   * not carry the non-reproducibility problem the paragraph above describes; it is safe to threshold precisely
65   * because it is relative, not absolute. Before this class, `cancel`, `columns`, `edit` and `settings` had zero
66   * glyph-specific automated coverage - only the generic non-vacuous ink check every stem gets.
67   *
68   * <h5 class='section'>What is real and what is restated:</h5>
69   * <p>
70   * The sprite, the stylesheet and the icon registry are all read from the <b>real</b> classpath resources
71   * {@link ViewsMixin} serves, and the host markup for every glyph comes from {@code juneau-icons.js}'s own
72   * {@code resolveIcon()} at runtime rather than being re-spelled here - every stem happens to be registered under
73   * its own name, which is what makes that possible. The render sizes are read out of
74   * {@code juneau-views.css}'s {@code --jc-chrome-glyph-size} / {@code -small} custom properties rather than typed
75   * as literals, because those values are expected to move. The one restatement is the set of CSS class names used
76   * to build the two UI-context strips ({@code juneau-view-ribbon-btn}, the paging-pill classes, and
77   * {@code juneau-sym-flip-x}); those live in the prober, because building them from the real emitters would mean
78   * loading the whole view runtime to look at eight buttons.
79   *
80   * <h5 class='section'>Off by default:</h5>
81   * <p>
82   * Disabled unless the {@value #GATE} system property is set, which only the module's opt-in {@code js-tests}
83   * Maven profile does. Run it with
84   * {@code mvn -Pjs-tests -f juneau-rest/juneau-rest-server-views/pom.xml test}. See
85   * {@code src/test/js/symbol-sprite-render.cjs} for the prober this drives.
86   *
87   * <h5 class='section'>See Also:</h5>
88   * <ul>
89   * 	<li class='jc'>{@link SymbolSprite_StemIds_Test} &mdash; pins the stem id set, always on.
90   * 	<li class='jc'>{@link SymbolSprite_Provenance_Test} &mdash; pins the approved artwork bytes, always on.
91   * </ul>
92   */

106 	private static final Pattern GLYPH_SIZE =
107 		Pattern.compile("--jc-chrome-glyph-size:\\s*(\\d+(?:\\.\\d+)?)px");
108 	private static final Pattern GLYPH_SIZE_SMALL =
109 		Pattern.compile("--jc-chrome-glyph-size-small:\\s*(\\d+(?:\\.\\d+)?)px");
110 	private static final Pattern SYMBOL_ID = Pattern.compile("<symbol\\s+id=\"juneau-sym-([^\"]+)\"");

118 	private static String resource(String path) throws IOException {
119 		try (var in = ViewsMixin.class.getResourceAsStream(path)) {
120 			assertNotNull(in, () -> "missing classpath resource: " + path);
121 			return new String(in.readAllBytes(), UTF_8);
122 		}
123 	}

125 	/** Every stem in the sprite, in document order - i.e. the key sheet's grid order. */
126 	private static List<String> stems(String sprite) {
127 		var out = new ArrayList<String>();
128 		var m = SYMBOL_ID.matcher(sprite);
129 		while (m.find())
130 			out.add(m.group(1));
131 		assertFalse(out.isEmpty(), "no stems found in the sprite");
132 		return out;
133 	}

135 	/** A render size read out of the real stylesheet rather than typed here, because these values move. */
136 	private static double cssPx(Pattern p, String css) {
137 		var m = p.matcher(css);
138 		assertTrue(m.find(), () -> "no match for " + p.pattern() + " in juneau-views.css; the render sizes are"
139 			+ " read from the stylesheet on purpose - if the custom property was renamed, follow it here");
140 		return Double.parseDouble(m.group(1));
141 	}

143 	/**
144 	 * Matches the prober's {@code 'ink' + size} JSON key so assertions follow the stylesheet, not a typed
145 	 * {@code 16px}.
146 	 */
147 	private static String inkKey(double px) {
148 		return "ink" + (px == Math.rint(px) ? Long.toString((long) Math.rint(px)) : Double.toString(px));
149 	}
150 
151 	private static String pxLabel(double px) {
152 		return (px == Math.rint(px) ? Long.toString((long) Math.rint(px)) : Double.toString(px)) + "px";
153 	}

155 	/**
156 	 * The fixture: the real stylesheet, the real sprite, and the real icon registry.
157 	 *
158 	 * <p>
159 	 * The sprite is parked in an inert {@code <script type="text/plain">} and handed to the registry through a
160 	 * stubbed {@code fetch}, so {@code loadSymbolSprite()}'s real parse-and-inject path runs. A {@code file://}
161 	 * page cannot fetch a sibling file, and inlining the sprite directly would skip that path entirely and log a
162 	 * console error the prober would then have to learn to ignore.
163 	 */
164 	private static String fixture(String css, String sprite, String iconsJs) {
165 		return "<!DOCTYPE html><html><head><meta charset=\"utf-8\"><style>\n"
166 			+ css
167 			+ "\n</style></head><body style=\"margin:0;background:#ffffff;color:#16325c\">\n"
168 			+ "<script type=\"text/plain\" id=\"sprite-xml\">" + sprite + "</script>\n"
169 			+ "<script>\n"
170 			+ "window.fetch = function () {\n"
171 			+ "  var xml = document.getElementById('sprite-xml').textContent;\n"
172 			+ "  return Promise.resolve({ ok: true, text: function () { return Promise.resolve(xml); } });\n"
173 			+ "};\n"
174 			+ "</script>\n"
175 			+ "<script>\n" + iconsJs + "\n</script>\n"
176 			+ "<div id=\"stage\"></div>\n"
177 			+ "</body></html>";
178 	}

181 	static void probe() throws Exception {
182 		var dir = Path.of(requiredProperty("juneau.jsTests.dir"));
183 		var harness = Path.of(requiredProperty("juneau.jsTests.harness")).getParent()
184 			.resolve("symbol-sprite-render.cjs");
185 
186 		var sprite = resource(ViewsMixin.SYMBOLS_SVG_RESOURCE);
187 		var css = resource(ViewsMixin.VIEWS_CSS_RESOURCE);
188 		var iconsJs = resource(ViewsMixin.ICONS_JS_RESOURCE);
189 
190 		var fixtureFile = Files.createDirectories(dir.resolve("fixtures")).resolve("symbol-sprite.html");
191 		Files.write(fixtureFile, fixture(css, sprite, iconsJs).getBytes(UTF_8));
192 
193 		// Beside target/js rather than inside it: target/js is the profile's node/browser scratch space and is
194 		// what a developer deletes to force a re-provision, while these sheets are the deliverable.
195 		outputDir = Files.createDirectories(dir.getParent().resolve("symbol-sprite"));
196 
197 		ribbonPx = cssPx(GLYPH_SIZE, css);
198 		pillPx = cssPx(GLYPH_SIZE_SMALL, css);
199 
200 		var request = Json.of(Map.of(
201 			"outputDir", outputDir.toString(),
202 			"stems", stems(sprite),
203 			"bogusStem", BOGUS_STEM,
204 			"sizes", List.of(pillPx, ribbonPx, 24.0),
205 			"ribbonSize", ribbonPx,
206 			"pillSize", pillPx,
207 			// The four-glyph document family plus the one glyph the redraw gave a new meaning, whose ribbon-size
208 			// distinguishability against both settings and spreadsheet is a named review check.
209 			"family", List.of("csv", "pdf", "spreadsheet", "copy"),
210 			// the glyph distinguishability pin: every pairwise combination of the four locked-scope glyphs (cancel, columns,
211 			// edit, settings), plus the pre-existing columns-spreadsheet review-only pair (columns-settings is
212 			// reused rather than duplicated). See c01_fourLockedGlyphsStayMutuallyDistinguishable() below.
213 			"adjacencies", List.of(
214 				List.of("columns", "spreadsheet"),
215 				List.of("cancel", "columns"),
216 				List.of("cancel", "edit"),
217 				List.of("cancel", "settings"),
218 				List.of("columns", "edit"),
219 				List.of("columns", "settings"),
220 				List.of("edit", "settings"))));
221 
222 		report = Json.to(run(dir, harness, fixtureFile, request), Map.class);
223 	}

231 	/** Runs the prober, failing with its stderr attached (its exit code alone is not a diagnosis). */
232 	private static String run(Path dir, Path harness, Path fixture, String request) throws Exception {
233 		var cmd = List.of(System.getProperty("juneau.jsTests.node", "node"), harness.toString(),
234 			fixture.toString(), request);
235 		var stdout = dir.resolve("symbol-sprite-render-stdout.json");
236 		var stderr = dir.resolve("symbol-sprite-render-stderr.txt");
237 		var pb = new ProcessBuilder(cmd).redirectOutput(stdout.toFile()).redirectError(stderr.toFile());
238 		pb.environment().put("NODE_PATH", dir.resolve("node_modules").toString());
239 		pb.environment().put("PLAYWRIGHT_BROWSERS_PATH", requiredProperty("juneau.jsTests.browsers"));
240 
241 		var p = pb.start();
242 		if (!p.waitFor(5, TimeUnit.MINUTES)) {
243 			p.destroyForcibly();
244 			fail("prober did not finish within 5m; stderr:\n" + quietRead(stderr));
245 		}
246 		assertEquals(0, p.exitValue(), () -> "prober exited non-zero; stderr:\n" + quietRead(stderr));
247 		return Files.readString(stdout);
248 	}

250 	private static String quietRead(Path p) {
251 		try {
252 			return Files.readString(p);
253 		} catch (IOException e) {
254 			return "<unreadable: " + e + ">";
255 		}

258 	/** {@code report.glyphs[stem]["ink12"]} and friends, as a number. */
259 	private static double metric(String stem, String key) {
260 		var glyphs = (Map<?,?>) report.get("glyphs");
261 		assertNotNull(glyphs, () -> "prober report has no glyphs section: " + report);
262 		var g = (Map<?,?>) glyphs.get(stem);
263 		assertNotNull(g, () -> "prober report has no entry for stem " + stem + ": " + glyphs.keySet());
264 		var v = (Number) g.get(key);
265 		assertNotNull(v, () -> "prober report has no " + key + " for stem " + stem + ": " + g);
266 		return v.doubleValue();
267 	}

275 		assertNotNull(stems, () -> "prober report has no stem list: " + report);

282 			assertTrue(metric(stem, inkKey(pillPx)) > 0,
283 				() -> stem + " rasterised to an empty " + pxLabel(pillPx) + " box");
284 		}

300 	/** The exact locked scope - these four stems, and no others. */
301 	private static final List<String> FOUR_GLYPHS = List.of("cancel", "columns", "edit", "settings");
302 
303 	/**
304 	 * Minimum acceptable mean per-pixel luminance diff (0..255 scale; see the prober's {@code PIXEL_DIFF}, in
305 	 * {@code symbol-sprite-render.cjs}) between any two of {@link #FOUR_GLYPHS}, at either render size.
306 	 *
307 	 * <p>
308 	 * Chosen against the six pairwise diffs measured on {@code HEAD} at filing time, which ranged
309 	 * <b>30.66 - 50.45</b> (the tightest pair was {@code columns} vs {@code settings} at the 12px paging-pill
310 	 * size). This floor sits well under half the smallest of those - loose enough that ordinary antialiasing
311 	 * jitter from a future Playwright/Chromium pin bump should not trip it (both sides of the comparison move
312 	 * together across a pin bump; see the class javadoc), but tight enough that two glyphs actually drifting
313 	 * into each other visually - the real risk this check exists for, as in the earlier scare where
314 	 * {@code columns} briefly appeared to have been left as a copy of the {@code settings} cog - would have to
315 	 * land far closer to identical than any plausible accidental redraw before this goes green.
316 	 */
317 	private static final double MIN_PAIR_PIXEL_DIFF = 15.0;
318 
319 	/** Looks up the prober's {@code adjacencyDiffs} entry for an unordered glyph pair. */
320 	private static Map<?,?> adjacencyDiff(String a, String b) {
321 		var diffs = (List<?>) report.get("adjacencyDiffs");
322 		assertNotNull(diffs, () -> "prober report has no adjacencyDiffs section: " + report);
323 		for (var d : diffs) {
324 			var m = (Map<?,?>) d;
325 			var names = (List<?>) m.get("names");
326 			if (names != null && names.size() == 2 && names.contains(a) && names.contains(b))
327 				return m;
328 		}
329 		return null;
330 	}

332 	@Test void c01_fourLockedGlyphsStayMutuallyDistinguishable() {
333 		for (var i = 0; i < FOUR_GLYPHS.size(); i++) {
334 			for (var j = i + 1; j < FOUR_GLYPHS.size(); j++) {
335 				var a = FOUR_GLYPHS.get(i);
336 				var b = FOUR_GLYPHS.get(j);
337 				var pair = adjacencyDiff(a, b);
338 				assertNotNull(pair, () -> "no adjacencyDiffs entry for " + a + " vs " + b + "; the prober's"
339 					+ " adjacencies request list must carry every pairwise combination of " + FOUR_GLYPHS);
340 				var ribbon = ((Number) pair.get("pixelDiffRibbon")).doubleValue();
341 				var small = ((Number) pair.get("pixelDiffSmall")).doubleValue();
342 				assertTrue(ribbon >= MIN_PAIR_PIXEL_DIFF, () -> a + " and " + b + " are only " + ribbon
343 					+ " luminance levels apart at the ribbon (" + pxLabel(ribbonPx) + ") render size - below the "
344 					+ MIN_PAIR_PIXEL_DIFF + " distinguishability floor, i.e. they risk reading as the same glyph");
345 				assertTrue(small >= MIN_PAIR_PIXEL_DIFF, () -> a + " and " + b + " are only " + small
346 					+ " luminance levels apart at the paging-pill (" + pxLabel(pillPx) + ") render size - below the "
347 					+ MIN_PAIR_PIXEL_DIFF + " distinguishability floor");
348 			}
349 		}
350 	}
