28  /**
29   * Always-on coverage for the JuneauViews.bus wraps in juneau-views.js (probe group, async job).
30   *
31   * <p>
32   * The load-bearing fact: {@code probe:<group>} is published ONLY on a user-driven selection change.  The default
33   * selection at enhance time, an imperative {@code select()}, a no-op re-click and a status repaint publish nothing,
34   * so a subscriber (a page's probe listener) can never mistake a page load for a user pick.
35   */

59  	@Test void a01_probePublish_isInsideEmit_only() throws IOException {
60  		var js = BusHarness.source(ViewsMixin.VIEWS_JS_RESOURCE);
61  		var emitStart = js.indexOf("function emit(id) {");
62  		assertTrue(emitStart > 0, "enhanceProbeGroup's emit(id) not found");
63  		var emitEnd = js.indexOf("\n\t\t}\n", emitStart);
64  		var emit = js.substring(emitStart, emitEnd);
65  		assertTrue(emit.contains("publishProbeSelection(group, id)"), emit);
66  		// One definition plus one call site: nothing else publishes probe:.
67  		assertEquals(2, js.split("publishProbeSelection\\(", -1).length - 1, "publishProbeSelection must be defined once and called once");
68  		assertTrue(js.contains("if (changed && notify) emit(id);"), "selectId's user-driven gate must be unchanged");
69  	}
70  
71  	@Test void a02_jobPublish_wiredIntoStartJobStream() throws IOException {
72  		var js = BusHarness.source(ViewsMixin.VIEWS_JS_RESOURCE);
73  		var s = js.indexOf("function startJobStream(started, action, table, tr, ctx, opts) {");
74  		assertTrue(s > 0, "startJobStream not found");
75  		var body = js.substring(s, js.indexOf("\n\t}\n", s));
76  		assertTrue(body.contains("publishJobEvent(started, \"start\")"), body);
77  		assertTrue(body.contains("publishJobEvent(started, \"progress\", e.data)"), body);
78  		assertTrue(body.contains("if (first) publishJobEvent(started, \"result\", result)"), body);
79  		assertTrue(body.indexOf("finish(result,") < body.indexOf("if (first) publishJobEvent"), "the result must publish after finish()");
80  		assertTrue(js.contains("if (kind === \"result\") JOB_OWNERS.delete(jobId)"), "a terminal result must release the owner entry");
81  		assertFalse(body.contains("addEventListener(\"error\", function () {\n\t\t\tpublishJobEvent"), "a stream error must not publish a job state");
82  	}

206 	/** JSON numbers come back as Integer/Long/Double depending on the parser; compare them as ints. */
207 	private static Object normalize(Object o) {
208 		if (o instanceof Map<?,?> m) {
209 			var out = new LinkedHashMap<String,Object>();
210 			m.forEach((k, v) -> out.put((String)k, normalize(v)));
211 			return out;
212 		}
213 		if (o instanceof Number n && n.doubleValue() == n.intValue())
214 			return n.intValue();
215 		return o;
216 	}
