27  /**
28   * Always-on coverage for the region MESSAGE BUS in {@code juneau-regions.js}: broadcast by default, opt-in targeted
29   * delivery, opaque payloads, per-subscriber error isolation, subscription-ordered synchronous delivery with
30   * emit-during-delivery queued rather than nested, unsubscribe on teardown, and the negative half of the
31   * emit-ownership invariant.  The chrome half (selection, detail, redraw) moved to the page bus's topics and is
32   * covered with the datatables publishers.

45  	private static void assertAllTrue(Map<?,?> r, String... keys) {
46  		for (var k : keys)
47  			assertEquals(true, r.get(k), () -> k + " -> " + r.get(k) + " in " + r);
48  	}

54  	/**
55  	 * THE EMIT-OWNERSHIP INVARIANT, as a source scan: the region runtime installs NO delegated listener over
56  	 * author-painted content.  Chrome auto-emits because the framework owns the chrome; content emits explicitly
57  	 * because the author owns the content.  A "helpful" click delegate over a region container is exactly how this
58  	 * invariant gets broken later, and it would break it silently - so the negative is pinned here as well as
59  	 * behaviorally.
60  	 */
61  	@Test void a01_sourceShape_theRuntimeBindsNoDelegateOverAuthorContent() throws Exception {
62  		var body = RegionsHarness.regionsJs();
63  		assertFalse(body.contains("addEventListener(\"click\""),
64  			"the region runtime must not bind a click listener - content emits explicitly");
65  		assertFalse(body.contains("addEventListener(\"change\""), body);
66  		assertFalse(body.contains("addEventListener(\"input\""), body);
67  		assertFalse(body.contains("addEventListener(\"submit\""), body);
68  		// The whole listener budget is TWO, and neither is bound on a region's contents: the parent->child abort
69  		// linkage is on an AbortSignal, and the barrier's backstop is on the document.  Pinning the COUNT is what
70  		// makes a third listener - which is how a content delegate would arrive - fail here rather than pass because
71  		// it used an event name this test did not think to name above.
72  		assertEquals(2, body.split("addEventListener\\(", -1).length - 1,
73  			"the region runtime's only listeners are the parent->child abort linkage and the barrier backstop");
74  		assertTrue(body.contains("parent.addEventListener(\"abort\""), body);
75  		assertTrue(body.contains("window.document.addEventListener(\"DOMContentLoaded\""), body);
76  	}

78  	/**
79  	 * Payload OPACITY as a source scan, to complement the behavioral trap: the framework neither clones, serializes
80  	 * nor routes on the payload, so a payload carrying a DOM node or a class instance survives delivery unchanged and
81  	 * the payload's shape is a contract between the emitting and receiving regions rather than with the framework.
82  	 */
83  	@Test void a02_sourceShape_thePayloadIsNeverClonedOrInspected() throws Exception {
84  		var body = RegionsHarness.regionsJs();
85  		for (var forbidden : List.of("structuredClone", "JSON.parse(JSON.stringify", "Object.assign({}, msg",
86  				"msg.kind ===", "message.kind ==="))
87  			assertFalse(body.contains(forbidden), "the bus must not " + forbidden + " a payload");
88  	}

158 	/**
159 	 * An emit DURING delivery is QUEUED, not nested - asserted by CALL-ORDER RECORDING rather than by
160 	 * absence-of-crash, because a nested implementation does not crash: it just re-enters a subscriber mid-paint, and
161 	 * the only visible difference is the order the handlers ran in.  The nested signature would interleave the second
162 	 * message's handlers inside the first's; the queued signature completes the first fan-out entirely first.
163 	 */
164 	@Test void b23_emitDuringDeliveryIsQueuedNotNested() {
165 		var r = report();
166 		assertEquals(List.of("enter:b:m1", "exit:b:m1", "enter:c:m1", "exit:c:m1",
167 			"enter:c:m2", "exit:c:m2", "enter:driver:m2", "exit:driver:m2"), r.get("t23_trace"), r::toString);
168 		assertAllTrue(r, "t23_notNested", "t23_m1FanOutCompletedFirst");
169 		assertEquals(List.of("m1", "m2"), r.get("t23_cGot"), r::toString);
170 	}

172 	/**
173 	 * An emit cycle terminates the drain and errors loudly, PRINTING THE CYCLE - the shortest repeating
174 	 * {@code from -> to} suffix - rather than an arbitrary pair of keys.  Asserted with a THREE-region ring as well
175 	 * as a two-region one, because the two-region case cannot tell "prints the cycle" apart from "prints the last two
176 	 * hops", which is the mistake worth catching: a wrong cycle print sends the reader to the wrong region.
177 	 */
178 	@Test void b24_anEmitCycleIsTerminatedAndPrintsTheActualCycle() {

183 		assertEquals(3, ((Number)r.get("t24_3_hopCount")).intValue(),

227 	/**
228 	 * OUT-OF-ORDER SETTLE, the flagship shape for update-in-place regions.  Three rapid messages with responses
229 	 * returning in the order 3, 1, 2: the container holds the LAST REQUESTED probe, every superseded fetch's signal
230 	 * is aborted, a superseded response can no longer settle at all, and no {@code AbortError} ever surfaces as a
231 	 * region error - the swallow the design points at as the established idiom.
232 	 */
233 	@Test void b50_outOfOrderSettleLeavesTheLastRequestedResponsePainted() {
234 		var r = report();
235 		assertAllTrue(r, "t50_twoFetches", "t50_firstSignalAborted", "t50_secondSignalLive",
236 			"t50_secondSettled", "t50_staleFirstCannotSettle", "t50_noRegionError");
237 		assertEquals("probe-2", r.get("t50_containerHoldsSecond"), "a stale response must not overwrite a fresh one");
238 		assertEquals("probe-5", r.get("t50a_holdsLast"), r::toString);
239 		assertEquals(List.of(true, true, false), r.get("t50a_earlierSignalsAborted"), r::toString);
240 		assertEquals(List.of(false, false), r.get("t50a_supersededCannotSettle"), r::toString);
241 		assertEquals(true, r.get("t50a_noAbortErrorSurfaced"), r::toString);
242 	}
