27  /**
28   * Always-on behavioral coverage for {@code juneau-helpers.js}.
29   *
30   * <p>
31   * The paint library is exercised under a DOM shim with NO {@code juneau-regions.js} loaded (see {@code
32   * src/test/js/helpers-harness.cjs}), matching the contract that every helper is testable with no region
33   * at all.  The harness script computes facts only; every assertion against those facts lives here so a failure
34   * names the exact behavior that broke.
35   *
36   * <p>
37   * Test 29 (the module-shape purity scan) is a separate class, {@link Helpers_Purity_Test}, because it is a static
38   * source-text scan rather than a behavioral one and belongs with the other source-shape checks.
39   */
40  class Helpers_Test extends TestBase {
41  
42  	private static Map<?,?> report() {
43  		var r = HelpersHarness.report("helpers.cjs");
44  		assumeTrue(r != null, "node not available or helpers.cjs not found - behavioral layer skipped");
45  		return r;
46  	}
47  
48  	private static void assertAllTrue(Map<?,?> r, String... keys) {
49  		for (var k : keys)
50  			assertEquals(true, r.get(k), () -> k + " -> " + r.get(k) + " in " + r);
51  	}
