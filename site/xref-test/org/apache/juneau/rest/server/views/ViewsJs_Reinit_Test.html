62  	/** Balanced-brace extractor (nested functions / object literals would fool a plain {@code \\n\\t}} scan). */
63  	private static String functionBody(String body, String signature) {
64  		var start = body.indexOf(signature);
65  		assertTrue(start >= 0, () -> "'" + signature + "' not found:\n" + body);
66  		var i = body.indexOf('{', start);
67  		assertTrue(i >= 0, () -> "'" + signature + "' has no opening brace:\n" + body);
68  		var depth = 0;
69  		var j = i;
70  		for (; j < body.length(); j++) {
71  			var c = body.charAt(j);
72  			if (c == '{') {
73  				depth++;
74  			} else if (c == '}') {
75  				depth--;
76  				if (depth == 0) { j++; break; }
77  			} else if (c == '"' || c == '\'' || c == '`') {
78  				var quote = c;
79  				j++;
80  				while (j < body.length() && body.charAt(j) != quote) { if (body.charAt(j) == '\\') j++; j++; }
81  			} else if (c == '/' && j + 1 < body.length() && body.charAt(j + 1) == '/') {
82  				while (j < body.length() && body.charAt(j) != '\n') j++;
83  			} else if (c == '/' && j + 1 < body.length() && body.charAt(j + 1) == '*') {
84  				j = body.indexOf("*/", j) + 1;
85  			}
86  		}
87  		return body.substring(start, j);
88  	}

234 	@BeforeAll
235 	static void probeIfNodeAvailable() throws Exception {
236 		if (!nodeAvailable())
237 			return;
238 		var harness = locateHarness();
239 		if (harness == null)
240 			return;
241 		var configFile = Files.createTempFile("juneau-config-", ".js");
242 		var viewsFile = Files.createTempFile("juneau-views-", ".js");
243 		try {
244 			Files.writeString(configFile, configJs(), UTF_8);
245 			Files.writeString(viewsFile, viewsJs(), UTF_8);
246 			report = Json.to(runNode(harness, configFile, viewsFile), Map.class);
247 		} finally {
248 			Files.deleteIfExists(configFile);
249 			Files.deleteIfExists(viewsFile);
250 		}
251 	}

253 	private static boolean nodeAvailable() {
254 		try {
255 			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
256 			if (!p.waitFor(5, TimeUnit.SECONDS)) {
257 				p.destroyForcibly();
258 				return false;
259 			}
260 			return p.exitValue() == 0;
261 		} catch (Exception e) {
262 			return false;
263 		}
264 	}

266 	private static Path locateHarness() {
267 		var basedir = System.getProperty("basedir");
268 		if (basedir != null) {
269 			var p = Path.of(basedir, "src/test/js/config-reinit.cjs");
270 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
271 		}
272 		for (var rel : List.of(
273 			"src/test/js/config-reinit.cjs",
274 			"juneau-rest/juneau-rest-server-views/src/test/js/config-reinit.cjs"
275 		)) {
276 			var p = Path.of(rel);
277 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
278 		}
279 		return null;
280 	}

282 	private static String runNode(Path harness, Path configJs, Path viewsJs) throws Exception {
283 		var stdout = Files.createTempFile("config-reinit-stdout-", ".json");
284 		var stderr = Files.createTempFile("config-reinit-stderr-", ".txt");
285 		try {
286 			var pb = new ProcessBuilder(List.of(
287 					"node", harness.toString(), configJs.toString(), viewsJs.toString()))
288 				.redirectOutput(stdout.toFile())
289 				.redirectError(stderr.toFile());
290 			var p = pb.start();
291 			if (!p.waitFor(30, TimeUnit.SECONDS)) {
292 				p.destroyForcibly();
293 				fail("config-reinit.cjs did not finish within 30s; stderr:\n" + quietRead(stderr));
294 			}
295 			if (p.exitValue() != 0)
296 				fail("config-reinit.cjs exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr)
297 					+ "\nstdout:\n" + quietRead(stdout));
298 			return Files.readString(stdout, UTF_8);
299 		} finally {
300 			Files.deleteIfExists(stdout);
301 			Files.deleteIfExists(stderr);
302 		}
303 	}

305 	private static String quietRead(Path p) {
306 		try { return Files.readString(p, UTF_8); }
307 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
308 	}
