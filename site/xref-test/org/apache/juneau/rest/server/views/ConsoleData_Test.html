28  /**
29   * The data-source loader in {@code juneau-data.js} under the real bus: config and lookup, URLs and fetch, the parsers,
30   * the columnar builder, chunk lists, the cache, DataSource, Derived and boot.  Every check lives in {@code data.cjs};
31   * each section returns {@code {check: true | {actual, expected}}}.  Runs when node is on PATH.
32   */
33  class ConsoleData_Test extends TestBase {
34  
35  	private static Map<?,?> report;
36  
37  	private static synchronized Map<?,?> report() {
38  		if (report == null)
39  			report = NodeHarness.report("data.cjs", ViewsMixin.BUS_JS_RESOURCE, ViewsMixin.DATA_JS_RESOURCE);
40  		assumeTrue(report != null, "node (or data.cjs) not available - JS layer skipped");
41  		return report;
42  	}
43  
44  	// Every check in the section is true; a failure prints each failed check's {actual, expected}.
45  	private static void assertSection(String key) {
46  		var s = (Map<?,?>)report().get(key);
47  		assertNotNull(s, () -> "section '" + key + "' missing from the harness output");
48  		assertNull(s.get("$error"), () -> "section '" + key + "' crashed: " + s.get("$error"));
49  		assertFalse(s.isEmpty(), () -> "section '" + key + "' has no checks");
50  		var failed = s.entrySet().stream().filter(e -> ! Boolean.TRUE.equals(e.getValue()))
51  			.collect(Collectors.toMap(e -> String.valueOf(e.getKey()), e -> String.valueOf(e.getValue()), (a, b) -> a,
52  			TreeMap::new));
53  		assertEquals(Map.of(), failed, () -> "section '" + key + "' failed checks: " + failed);
54  	}
55  
56  	@Test void a01_config_messagesAndLookup() { assertSection("a_config"); }
57  	@Test void a02_urls_tokensQueryAndFetch() { assertSection("c_urls"); }
58  	@Test void a03_parsers_csvNdjsonJson() { assertSection("d_parsers"); }
59  	@Test void a04_frames_builderConcatAndAggregateContract() { assertSection("e_frames"); }
60  	@Test void a05_chunks_unitsAndOpenChunk() { assertSection("f_chunks"); }
61  	@Test void a06_cache_idbMemoryEpochAndFallback() { assertSection("g_cache"); }
62  	@Test void a07_source_generationsCacheAndAuto() { assertSection("h_source"); }
63  	@Test void a08_derived_transformsAndMirroring() { assertSection("i_derived"); }
64  	@Test void a09_boot_sourcesDerivedAndPartial() { assertSection("j_boot"); }
65  }
