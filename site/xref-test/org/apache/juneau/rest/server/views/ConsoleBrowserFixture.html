89  		var s = servedChromeCss;

91  			try (var c = MockRestClient.create(ChromeHost.class); var r = c.get(ConsoleChromeMixin.CHROME_CSS_PATH).run()) {

97  		return s;

247 	private ConsoleBrowserFixture add(String caseName, String url, String html, boolean computed, String... queries) {
248 		assertTrue(url.startsWith(ORIGIN + "/"), () -> "case URLs must be under " + ORIGIN + ": " + url);
249 		assertEquals(0, queries.length % 2, "queries are key/selector pairs");
250 		var path = url.substring(ORIGIN.length());
251 		var q = path.indexOf('?');
252 		Map<String,Object> c = map(
253 			"name", caseName,
254 			"url", url,
255 			"pages", Map.of(q < 0 ? path : path.substring(0, q), Map.of("type", "text/html", "body", html))
256 		);
257 		var qm = new LinkedHashMap<String,String>();
258 		for (var i = 0; i < queries.length; i += 2)
259 			qm.put(queries[i], queries[i + 1]);
260 		c.put("queries", qm);
261 		c.put("computed", computed);
262 		cases.add(c);
263 		return this;
264 	}

266 	/** Runs the harness and returns the report, keyed by case name. */
267 	Map<String,Map<String,Object>> run() throws Exception {
268 		var dir = Path.of(requiredProperty("juneau.jsTests.dir"));
269 		var harness = Path.of(requiredProperty("juneau.jsTests.harness")).getParent().resolve("console-shell.cjs");
270 		Map<String,Object> spec = map("assets", assets, "sequences", sequences, "cases", cases);
271 		var file = Files.createDirectories(dir.resolve("fixtures")).resolve("console-" + name + ".json");
272 		Files.writeString(file, Json.of(spec), UTF_8);
273 
274 		var cmd = List.of(System.getProperty("juneau.jsTests.node", "node"), harness.toString(), file.toString());
275 		var stdout = dir.resolve("console-" + name + "-stdout.json");
276 		var stderr = dir.resolve("console-" + name + "-stderr.txt");
277 		var pb = new ProcessBuilder(cmd).redirectOutput(stdout.toFile()).redirectError(stderr.toFile());
278 		pb.environment().put("NODE_PATH", dir.resolve("node_modules").toString());
279 		pb.environment().put("PLAYWRIGHT_BROWSERS_PATH", requiredProperty("juneau.jsTests.browsers"));
280 		var p = pb.start();
281 		if (! p.waitFor(3, TimeUnit.MINUTES)) {
282 			p.destroyForcibly();
283 			fail("prober did not finish within 3m; stderr:\n" + quietRead(stderr));
284 		}
285 		assertEquals(0, p.exitValue(), () -> "prober exited non-zero; stderr:\n" + quietRead(stderr));
286 		return (Map<String,Map<String,Object>>)Json.to(Files.readString(stdout), Map.class);
287 	}

297 	/**
298 	 * Like {@link #contractPage(String,String,String,String)}, with {@code htmlAttrs} written into the {@code <html>} tag
299 	 * (for example {@code data-jc-scheme-toggle}), {@link ConsoleChromeMixin#SCHEME_SCRIPT} first in the head, and, when
300 	 * {@code darkLink} is set, the dark layer linked between the chrome CSS and the palette.
301 	 */
302 	static String contractPage(String contractJson, String templates, String beforeShell, String afterShell, String htmlAttrs, boolean darkLink) {
303 		return "<!DOCTYPE html>\n<html lang=\"en\"" + (htmlAttrs.isEmpty() ? "" : " " + htmlAttrs) + "><head><meta charset=\"utf-8\">"
304 			+ (htmlAttrs.isEmpty() && ! darkLink ? "" : "<script>" + ConsoleChromeMixin.SCHEME_SCRIPT + "</script>")
305 			+ "<link rel=\"icon\" href=\"data:,\">\n"
306 			+ "<link rel=\"stylesheet\" href=\"" + CHROME_CSS + "\">\n"
307 			+ (darkLink ? "<link rel=\"stylesheet\" href=\"" + DARK_CSS + "\">\n" : "")
308 			+ "<link rel=\"stylesheet\" href=\"" + THEME_CSS + "\">\n"
309 			+ "<link rel=\"stylesheet\" href=\"" + VIEWS_CSS + "\">\n"
310 			+ "</head><body>\n"
311 			+ templates + "\n"
312 			+ "<script type=\"application/json\" id=\"juneau-page\">" + contractJson + "</script>\n"
313 			+ beforeShell + "\n"
314 			+ "<script src=\"" + SHELL + "\"></script>\n"
315 			+ afterShell + "\n"
316 			+ "</body></html>\n";
317 	}

327 	/** Like {@link #viewsPack()}, and without {@code juneau-bus.js} when {@code withBus} is false (for the missing-bus case). */
328 	static String viewsPack(boolean withBus) throws IOException {
329 		var nodeModules = Path.of(requiredProperty("juneau.jsTests.dir")).resolve("node_modules");
330 		var sb = new StringBuilder();
331 		sb.append("<script>\n").append(Files.readString(nodeModules.resolve("jquery/dist/jquery.min.js"))).append("\n</script>\n");
332 		sb.append("<script>\n").append(Files.readString(nodeModules.resolve("datatables.net/js/dataTables.min.js"))).append("\n</script>\n");
333 		for (var r : List.of(ViewsMixin.BUS_JS_RESOURCE, ViewsMixin.RENDERS_JS_RESOURCE, ViewsMixin.ICONS_JS_RESOURCE, ViewsMixin.SEARCH_JS_RESOURCE,
334 				ViewsMixin.PAGESTATE_JS_RESOURCE, ViewsMixin.URLSTATE_JS_RESOURCE, ViewsMixin.RIBBON_JS_RESOURCE,
335 				ViewsMixin.DATATABLES_JS_RESOURCE, ViewsMixin.VIEWS_JS_RESOURCE, ViewsMixin.CONFIG_JS_RESOURCE,
336 				ViewsMixin.REGIONS_JS_RESOURCE, ViewsMixin.HELPERS_JS_RESOURCE))
337 			if (withBus || ! r.equals(ViewsMixin.BUS_JS_RESOURCE))
338 				sb.append("<script>\n").append(resource(r)).append("\n</script>\n");
339 		return sb.toString();
340 	}
