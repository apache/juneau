24  /**
25   * Source-shape purity scan for {@code juneau-helpers.js}.
26   *
27   * <p>
28   * The invariants for the helper library:
29   * <ul>
30   * <li>Pure data -&gt; DOM: no {@code fetch}, no message-bus reference, no page-global mutable state.
31   * <li>The one amendment: a helper MAY hold per-instance state confined to the node it returned (tabStrip's
32   * fill-once bookkeeping), living in a closure captured by that node's own event listeners - never in a
33   * module-level {@code Map}/{@code Set}/{@code WeakMap}/counter.
34   * <li>No HTML-string sink, ever (R16), except {@code icon()}, which paints TRUSTED icon-registry sprite markup
35   * only (checked structurally in {@link ShippedJs_SinkScan_Test}, not re-checked here).
36   * <li>Themable by class, not by inline style - the one exception being {@code fieldGrid}'s {@code columns} option,
37   * which is a CSS custom property, not a {@code grid-template-columns} inline style.
38   * <li>{@code toast()} is the only helper allowed to write under {@code document.body} (one ephemeral
39   * {@code .jc-toast} node, no module registry).
40   * </ul>
41   *
42   * <p>
43   * This is a static source-text scan (like {@code ShippedJs_SinkScan_Test}'s HTML-sink checks), not a
44   * behavioral one - it is what lets an accidental module-level counter or a stray {@code WeakMap} fail this suite
45   * even on a code path the behavioral harness in {@link Helpers_Test} happens not to reach.
46   */
47  class Helpers_Purity_Test extends TestBase {
48  
49  	private static String source() throws Exception {
50  		return HelpersHarness.helpersJs();
51  	}
52  
53  	// =================================================================================================================
54  	// No page-global mutable state: no WeakMap/WeakSet/Map/Set at module scope, no assignment to a NS/window/
55  	// document/globalThis member other than the single `NS.helpers = {...}` export block.
56  	// =================================================================================================================
57  
58  	@Test void a01_noWeakMapOrWeakSet() throws Exception {
59  		var src = source();
60  		// Checks for actual CONSTRUCTION, not the bare identifier: the file's own header comment names
61  		// WeakMap/WeakSet as the thing this invariant forbids (see the doc comment above this class), so a
62  		// bare `contains("WeakMap")` would fail on that comment text alone rather than on real usage.
63  		assertFalse(src.contains("new WeakMap("), "juneau-helpers.js must not use WeakMap for per-instance state - "
64  			+ "fill-once bookkeeping must live in a closure, not a module-level identity map: " + src);
65  		assertFalse(src.contains("new WeakSet("), "juneau-helpers.js must not use WeakSet: " + src);
66  	}
67  
68  	/**
69  	 * No module-level counter/registry: the ONLY {@code new Map()}/{@code new Set()} construction allowed anywhere
70  	 * in the file is none at all - {@code tabStrip}'s uniqueness needs are met with a per-call random suffix
71  	 * (see {@code a03}), and every other helper's bookkeeping (the field-action bar, the filter-builder's
72  	 * predicate list) is a plain array or object living in that call's own closure, never shared across calls.
73  	 */
74  	@Test void a02_noModuleLevelMapOrSetConstruction() throws Exception {
75  		var src = source();
76  		assertFalse(src.contains("new Map("), "no module-level registry: " + src);
77  		assertFalse(src.contains("new Set("), "no module-level registry: " + src);
78  	}
79  
80  	/** {@code tabStrip} mints its per-instance DOM id namespace from a per-call random suffix, not a module counter. */
81  	@Test void a03_tabStripInstanceId_isPerCallRandomNotAModuleCounter() throws Exception {
82  		var src = source();
83  		assertTrue(src.contains("Math.random().toString(36)"), src);
84  		// A module-level counter would show up as a `let`/`var` declaration OUTSIDE any function, incremented on
85  		// every call.  The file's only top-level mutable bindings are `NS` and `I` (both assigned once, from
86  		// juneau-views.js's own already-published globals) - neither is a counter or a registry.
87  		assertFalse(src.contains("Seq = 0") || src.contains("Seq=0"), "no module-level sequence counter: " + src);
88  	}
89  
90  	/**
91  	 * The ONLY assignment to an {@code NS.*} member anywhere in the file is the single, one-time export block at
92  	 * the bottom ({@code NS.helpers = {...}}).  {@code fieldGrid}'s scoped-renderer-override mechanism calls the
93  	 * existing public {@code NS.registerRenderer}/{@code NS.resolveRenderer} FUNCTIONS (a call, not a member
94  	 * assignment) to swap and restore a registry entry within a single synchronous call - so it never appears as
95  	 * an {@code NS.<identifier> =} pattern.
96  	 */
97  	@Test void a04_onlyOneNsMemberAssignment_theHelpersExport() throws Exception {
98  		var src = source();
99  		var assignments = src.split("\\bNS\\.[A-Za-z_$][\\w$]*\\s*=(?!=)", -1).length - 1;
100 		assertEquals(1, assignments, "juneau-helpers.js must assign exactly one NS.* member (NS.helpers) - found "
101 			+ assignments + " occurrences of `NS.<member> =`:\n" + src);
102 		assertTrue(src.contains("NS.helpers = {"), src);
103 	}
104 
105 	/** {@code fieldGrid}'s renderer-override swap goes through the EXISTING public functions, never a direct write. */
106 	@Test void a05_scopedRendererOverride_usesPublicFunctionsNotDirectRegistryWrite() throws Exception {
107 		var src = source();
108 		assertTrue(src.contains("NS.registerRenderer(renderId, override);"), src);
109 		assertTrue(src.contains("NS.registerRenderer(renderId, prior);"), src);
110 		assertFalse(src.contains("NS._renderers"), "must go through the public registerRenderer/resolveRenderer "
111 			+ "functions, never touch the private _renderers registry directly: " + src);
112 	}
113 
114 	/**
115 	 * {@code window.*} appears in exactly two spots: the one-time namespace bootstrap
116 	 * ({@code var NS = window.JuneauViews = window.JuneauViews || {};}) and {@code icon()}'s read-only lookup of
117 	 * the icon registry ({@code window.JuneauViews && window.JuneauViews.icons}).  Neither is a page-global
118 	 * WRITE this file performs on its own account - the bootstrap assigns the same {@code JuneauViews} object
119 	 * juneau-views.js already created (a no-op if it exists), and the icon lookup only reads.  No OTHER
120 	 * {@code window.<member> =} assignment, and no {@code globalThis}/{@code document.<member> =} assignment,
121 	 * appears anywhere in the file.
122 	 */
123 	@Test void a06_noAssignmentToWindowGlobalThisOrDocumentMembers() throws Exception {
124 		var src = source();
125 		var windowAssignments = new java.util.regex.Matcher[]{
126 			java.util.regex.Pattern.compile("window\\.[A-Za-z_$][\\w$]*\\s*=(?!=)").matcher(src)};
127 		var count = 0;
128 		while (windowAssignments[0].find())
129 			count++;
130 		assertEquals(1, count, "the only window.<member> = assignment allowed is the namespace bootstrap "
131 			+ "(`window.JuneauViews = window.JuneauViews || {}`); found " + count + ":\n" + src);
132 		assertFalse(src.contains("globalThis"), src);
133 		assertFalse(src.matches("(?s).*\\bdocument\\.[A-Za-z_$][\\w$]*\\s*=(?!=).*"),
134 			"document.<member> = ASSIGNMENT is forbidden (document.createElement/createTextNode CALLS are fine and "
135 				+ "are not this pattern - it requires an `=` after the member name): " + src);
136 	}
137 
138 	/**
139 	 * {@code toast()} is the only helper allowed to write under {@code document.body}.  The hit is the
140 	 * documented exception (one ephemeral {@code .jc-toast} found by class query, not a module registry);
141 	 * a second {@code document.body.appendChild} anywhere else is a free-for-all and must fail this scan.
142 	 */
143 	@Test void a07_toast_isTheOnlyDocumentBodyAppend() throws Exception {
144 		var src = source();
145 		assertTrue(src.contains("document.querySelector(\".jc-toast\")"), src);
146 		var idx = 0;
147 		var hits = 0;
148 		while ((idx = src.indexOf("document.body.appendChild", idx)) >= 0) {
149 			hits++;
150 			idx++;
151 		}
152 		assertEquals(1, hits, "document.body.appendChild must appear exactly once (inside toast): " + hits);
153 		var toastFn = src.indexOf("function toast(");
154 		var appendAt = src.indexOf("document.body.appendChild");
155 		var nextFn = src.indexOf("\n\tfunction ", toastFn + 1);
156 		assertTrue(toastFn >= 0 && appendAt > toastFn && appendAt < nextFn,
157 			"document.body.appendChild must sit inside toast(), not a sibling helper");
158 	}
