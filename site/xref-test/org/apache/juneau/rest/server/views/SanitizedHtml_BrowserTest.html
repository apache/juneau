32  /**
33   * Opt-in Chromium XSS canary for {@link FieldFormat#SANITIZED_HTML}.
34   *
35   * <p>
36   * {@link ViewsJs_RowDetail_Test}'s {@code b17*} battery already proves the copier's behavior against this
37   * module's own regex-fixture {@code DOMParser} shim ({@code row-detail.cjs}) &mdash; a hand-rolled test parser
38   * that does not decode entities and does not do implicit tag nesting/foster-parenting the way a real browser
39   * HTML parser does (see the fidelity note at the top of that file's SANITIZED_HTML section). This class proves
40   * the SAME never-executes guarantee against a REAL browser HTML parser and a real DOM: a {@code <script>} and
41   * an {@code <img onerror>} payload must not produce an executable node, and benign markup (a {@code <b>} and a
42   * {@code <table>}) must still survive as real elements.
43   *
44   * <p>
45   * Disabled unless the {@value #GATE} system property is set (the module's {@code js-tests} Maven profile).
46   */
47  @EnabledIfSystemProperty(named=SanitizedHtml_BrowserTest.GATE, matches="true",
48  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -f juneau-rest/juneau-rest-server-views/pom.xml test`")
49  class SanitizedHtml_BrowserTest extends TestBase {

51  	/** System property the {@code js-tests} profile sets to enable this class. */
52  	static final String GATE = "juneau.jsTests";

54  	private static Map<?,?> report;

56  	private static String resource(String path) throws IOException {
57  		try (var in = ViewsMixin.class.getResourceAsStream(path)) {
58  			assertNotNull(in, () -> "missing classpath resource: " + path);
59  			return new String(in.readAllBytes(), UTF_8);
60  		}
61  	}
62  
63  	@BeforeAll
64  	static void probe() throws Exception {
65  		var dir = Path.of(requiredProperty("juneau.jsTests.dir"));
66  		var harness = Path.of(requiredProperty("juneau.jsTests.harness")).getParent().resolve("sanitized-html-browser.cjs");
67  
68  		var fixture = "<!DOCTYPE html><html><head><meta charset=\"utf-8\"></head><body>\n<script>\n"
69  			+ resource(ViewsMixin.VIEWS_JS_RESOURCE)
70  			+ "\n</script></body></html>";
71  		var fixtureFile = Files.createDirectories(dir.resolve("fixtures")).resolve("sanitized-html.html");
72  		Files.write(fixtureFile, fixture.getBytes(UTF_8));
73  
74  		report = Json.to(run(dir, harness, fixtureFile), Map.class);
75  	}
76  
77  	private static String requiredProperty(String name) {
78  		var v = System.getProperty(name);
79  		assertNotNull(v, () -> "-D" + name + " not set; the js-tests profile is responsible for providing it");
80  		return v;
81  	}
82  
83  	private static String run(Path dir, Path harness, Path fixture) throws Exception {
84  		var cmd = List.of(System.getProperty("juneau.jsTests.node", "node"), harness.toString(), fixture.toString());
85  		var stdout = dir.resolve("sanitized-html-stdout.json");
86  		var stderr = dir.resolve("sanitized-html-stderr.txt");
87  		var pb = new ProcessBuilder(cmd).redirectOutput(stdout.toFile()).redirectError(stderr.toFile());
88  		pb.environment().put("NODE_PATH", dir.resolve("node_modules").toString());
89  		pb.environment().put("PLAYWRIGHT_BROWSERS_PATH", requiredProperty("juneau.jsTests.browsers"));
90  
91  		var p = pb.start();
92  		if (!p.waitFor(3, TimeUnit.MINUTES)) {
93  			p.destroyForcibly();
94  			fail("prober did not finish within 3m; stderr:\n" + quietRead(stderr));
95  		}
96  		assertEquals(0, p.exitValue(), () -> "prober exited non-zero; stderr:\n" + quietRead(stderr));
97  		return Files.readString(stdout);
98  	}
99  
100 	private static String quietRead(Path p) {
101 		try {
102 			return Files.readString(p);
103 		} catch (IOException e) {
104 			return "<unreadable: " + e + ">";
105 		}
106 	}
