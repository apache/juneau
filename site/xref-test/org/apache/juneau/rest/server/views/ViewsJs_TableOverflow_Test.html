63  	@Test void a01_helpersExportedAndWired() throws Exception {
64  		var body = viewsJs();
65  		assertTrue(body.contains("ensureTableScroll: ensureTableScroll"), "missing NS export ensureTableScroll");
66  		assertTrue(body.contains("unwrapTableScroll: unwrapTableScroll"), "missing NS export unwrapTableScroll");
67  		// Wired from constructTable (the wrap is applied after DataTable boot + toolbar).
68  		assertTrue(body.contains("ensureTableScroll(table"), "ensureTableScroll must be called from constructTable");
69  		// Paired unwrap in teardownTable so a destroy()+reconstruct cannot nest the toolbar inside the box.
70  		assertTrue(body.contains("unwrapTableScroll(table"), "teardownTable must unwrap the DT1 scroll box");
71  	}

73  	@Test void a02_dt2SkipGuardPrecedesWrapClassSkip() throws Exception {
74  		var body = viewsJs();
75  		var dt2Guard = body.indexOf("closest(\".dt-layout-cell\")");
76  		// The wrap-class skip uses the TABLE_SCROLL_CLASS constant ("." + TABLE_SCROLL_CLASS), not a literal.
77  		var wrapGuard = body.indexOf("closest(\".\" + TABLE_SCROLL_CLASS)");
78  		assertTrue(dt2Guard >= 0, "ensureTableScroll must no-op on DT2 via table.closest('.dt-layout-cell')");
79  		assertTrue(wrapGuard >= 0, "ensureTableScroll must skip an already-wrapped table (closest('.'+TABLE_SCROLL_CLASS))");
80  		assertTrue(dt2Guard < wrapGuard,
81  			"the DT2 (.dt-layout-cell) skip-guard must fire BEFORE the .juneau-view-table-scroll skip");
82  	}

90  	@BeforeAll
91  	static void probeIfNodeAvailable() throws Exception {
92  		if (!nodeAvailable())
93  			return;
94  		var harness = locateHarness();
95  		if (harness == null)
96  			return;
97  		var viewsFile = Files.createTempFile("juneau-views-", ".js");
98  		var rendersFile = Files.createTempFile("juneau-renders-", ".js");
99  		try {
100 			Files.writeString(viewsFile, viewsJs(), UTF_8);
101 			Files.writeString(rendersFile, rendersJs(), UTF_8);
102 			report = Json.to(runNode(harness, rendersFile, viewsFile), Map.class);
103 		} finally {
104 			Files.deleteIfExists(viewsFile);
105 			Files.deleteIfExists(rendersFile);
106 		}
107 	}

109 	private static boolean nodeAvailable() {
110 		try {
111 			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
112 			if (!p.waitFor(5, TimeUnit.SECONDS)) {
113 				p.destroyForcibly();
114 				return false;
115 			}
116 			return p.exitValue() == 0;
117 		} catch (Exception e) {
118 			return false;
119 		}
120 	}

122 	private static Path locateHarness() {
123 		var basedir = System.getProperty("basedir");
124 		if (basedir != null) {
125 			var p = Path.of(basedir, "src/test/js/table-overflow.cjs");
126 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
127 		}
128 		for (var rel : List.of(
129 			"src/test/js/table-overflow.cjs",
130 			"juneau-rest/juneau-rest-server-views/src/test/js/table-overflow.cjs"
131 		)) {
132 			var p = Path.of(rel);
133 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
134 		}
135 		return null;
136 	}

138 	private static String runNode(Path harness, Path rendersJs, Path viewsJs) throws Exception {
139 		var stdout = Files.createTempFile("table-overflow-stdout-", ".json");
140 		var stderr = Files.createTempFile("table-overflow-stderr-", ".txt");
141 		try {
142 			var pb = new ProcessBuilder(List.of("node", harness.toString(), rendersJs.toString(), viewsJs.toString()))
143 				.redirectOutput(stdout.toFile())
144 				.redirectError(stderr.toFile());
145 			var p = pb.start();
146 			if (!p.waitFor(30, TimeUnit.SECONDS)) {
147 				p.destroyForcibly();
148 				fail("table-overflow.cjs did not finish within 30s; stderr:\n" + quietRead(stderr));
149 			}
150 			if (p.exitValue() != 0)
151 				fail("table-overflow.cjs exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr)
152 					+ "\nstdout:\n" + quietRead(stdout));
153 			return Files.readString(stdout, UTF_8);
154 		} finally {
155 			Files.deleteIfExists(stdout);
156 			Files.deleteIfExists(stderr);
157 		}
158 	}
159 
160 	private static String quietRead(Path p) {
161 		try { return Files.readString(p, UTF_8); }
162 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
163 	}

236 	/**
237 	 * The clip/ellipsis opt-out has to reach the cell to mean anything: the named emitters (progress / pill / tag /
238 	 * linked) put {@code juneau-cell-wrap} on their column's className, which is what stamps it on the {@code <td>}.
239 	 * Prose renderers stay on the clip default, and an author's own column class is preserved rather than replaced.
240 	 */
241 	@Test void b09_namedEmittersPutTheOptOutOnTheCell() {
242 		var r = report();
243 		assertEquals(true, r.get("wrap_pill"), "the `pill` column must carry juneau-cell-wrap");
244 		assertEquals(true, r.get("wrap_progress"), "the `progress` column must carry juneau-cell-wrap");
245 		assertEquals(true, r.get("wrap_tag"), "the `tag` column must carry juneau-cell-wrap");
246 		assertEquals(true, r.get("wrap_linked"), "the `linked` column must carry juneau-cell-wrap");
247 		assertEquals(true, r.get("wrap_truncateStaysClipped"),
248 			"`truncate` is unchanged and stays on the clip default - it must not opt out");
249 		assertEquals(true, r.get("wrap_dateStaysClipped"), "a prose renderer must stay on the clip default");
250 		assertEquals(true, r.get("wrap_authorClassPreserved"),
251 			"a renderer class must be APPENDED to the author's column class, never replace it");
