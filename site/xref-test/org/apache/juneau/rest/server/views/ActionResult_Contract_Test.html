27  /**
28   * Golden-fixture contract test for the typed {@code ActionResult} wire format (the write-path
29   * half of the row-action contract).
30   *
31   * <p>
32   * This pins the third independently-versioned wire contract of the toolkit (alongside {@code VIEW_META} and
33   * {@code PAGE_META}): its own {@link ActionResult#CONTRACT_VERSION}, its frozen top-level key order, the reserved
34   * async outcome set, and the omit-when-unset rule for every optional field.  Serialization uses the same canonical
35   * compact JSON marshaller ({@link Json#of(Object)}) the runtime reads.
36   */
37  class ActionResult_Contract_Test extends TestBase {
38  
39  	//------------------------------------------------------------------------------------------------------------------
40  	// The frozen ActionResult contract
41  	//------------------------------------------------------------------------------------------------------------------
42  
43  	@Test void a01_success_serializesToFrozenShape() {
44  		var row = new LinkedHashMap<String,Object>();
45  		row.put("id", "INC-1");
46  		row.put("status", "acknowledged");
47  		var json = Json.of(ActionResult.success(row));
48  
49  		var expected = Json.to("{\"contractVersion\":\"1\",\"outcome\":\"success\",\"row\":{\"id\":\"INC-1\",\"status\":\"acknowledged\"}}", Map.class);
50  		var actual = Json.to(json, Map.class);
51  		assertEquals(expected, actual, json);
52  	}
53  
54  	@Test void a02_contractVersion_isItsOwnValue_notAliasedToViewMeta() {
55  		// this is a SEPARATE contract from VIEW_META, so its version must not track ViewsMixin.CONTRACT_VERSION.
56  		assertEquals("1", ActionResult.CONTRACT_VERSION);
57  		assertNotEquals(ViewsMixin.CONTRACT_VERSION, ActionResult.CONTRACT_VERSION,
58  			"ActionResult.CONTRACT_VERSION must be its own value, never aliased to VIEW_META");
59  	}
60  
61  	@Test void a03_topLevelKeyOrder_success() {
62  		Map<?,?> actual = Json.to(Json.of(ActionResult.success(Map.of("id", "x"))), Map.class);
63  		assertEquals(List.of("contractVersion", "outcome", "row"), new ArrayList<>(actual.keySet()));
64  	}
65  
66  	@Test void a04_topLevelKeyOrder_fullResult_matchesBeanTypeOrder() {
67  		// contractVersion,outcome,replay,refusalCode,message,row,resultForm (the @BeanType-pinned order).
68  		var r = ActionResult.success(Map.of("id", "x")).replay(true).message("done");
69  		Map<?,?> actual = Json.to(Json.of(r), Map.class);
70  		assertEquals(List.of("contractVersion", "outcome", "replay", "message", "row"), new ArrayList<>(actual.keySet()));
71  	}
72  
73  	//------------------------------------------------------------------------------------------------------------------
74  	// resultForm - the additive receipt pointer
75  	//------------------------------------------------------------------------------------------------------------------
76  
77  	@Test void a05_resultForm_isAppendedLast_neverInterleaved() {
78  		// APPENDED, not interleaved: a reader positioned on the frozen prefix keeps reading the same keys in the
79  		// same order, which is what makes this additive rather than a contract change.
80  		var r = ActionResult.success(Map.of("id", "x")).replay(true).message("done").resultForm("/x/receipt");
81  		Map<?,?> actual = Json.to(Json.of(r), Map.class);
82  		assertEquals(List.of("contractVersion", "outcome", "replay", "message", "row", "resultForm"),
83  			new ArrayList<>(actual.keySet()));
84  	}
85  
86  	@Test void a06_resultForm_omittedWhenUnset() {
87  		for (var r : List.of(ActionResult.success(null), ActionResult.failure(), ActionResult.unknown())) {
88  			var json = Json.of(r);
89  			assertFalse(json.contains("resultForm"), () -> "unset resultForm leaked:\n" + json);
90  		}
91  		assertFalse(Json.of(ActionResult.success(null).resultForm(null)).contains("resultForm"));
92  	}
93  
94  	@Test void a07_resultForm_doesNotBumpTheContractVersion() {
95  		// The no-bump is the whole additive claim: an older runtime that has never heard of resultForm keeps
96  		// accepting these payloads and simply ignores the key.
97  		assertEquals("1", ActionResult.CONTRACT_VERSION);
98  		var json = Json.of(ActionResult.success(null).resultForm("/x/receipt"));
99  		assertTrue(json.contains("\"contractVersion\":\"1\""), json);
100 		assertTrue(json.contains("\"resultForm\":\"/x/receipt\""), json);
101 	}
102 
103 	@Test void a08_resultForm_isFluentAndReadable() {
104 		assertEquals("/x/receipt", ActionResult.success(null).resultForm("/x/receipt").resultForm);
105 	}

111 	@Test void b01_syncOutcomeWireTokens() {
112 		assertEquals("success", ActionResult.Outcome.SUCCESS.wire());
113 		assertEquals("failure", ActionResult.Outcome.FAILURE.wire());
114 		assertEquals("refusal", ActionResult.Outcome.REFUSAL.wire());
115 		assertEquals("unknown", ActionResult.Outcome.UNKNOWN.wire());
116 	}
117 
118 	@Test void b02_reservedAsyncOutcomes_areFrozenNow() {
119 		// Reserved from day one so the async-job feature cannot force a second result-contract bump.
120 		assertEquals("cancelled", ActionResult.Outcome.CANCELLED.wire());
121 		assertEquals("cancelled-after-effect", ActionResult.Outcome.CANCELLED_AFTER_EFFECT.wire());
122 		assertEquals(
123 			List.of("SUCCESS", "FAILURE", "REFUSAL", "UNKNOWN", "CANCELLED", "CANCELLED_AFTER_EFFECT"),
124 			Arrays.stream(ActionResult.Outcome.values()).map(Enum::name).toList());
125 	}

131 	@Test void c01_refusal_carriesOpaqueCode_noRowNoReplay() {
132 		var json = Json.of(ActionResult.refusal("write-guard:not-armed"));
133 		var actual = Json.to(json, Map.class);
134 		assertEquals(Json.to("{\"contractVersion\":\"1\",\"outcome\":\"refusal\",\"refusalCode\":\"write-guard:not-armed\"}", Map.class), actual, json);
135 	}
136 
137 	@Test void c02_refusal_rendersAnyNamespace() {
138 		// The runtime renders ANY code visibly; the type accepts framework codes and consumer app:/write-guard: codes alike.
139 		assertTrue(Json.of(ActionResult.refusal("CSRF_TOKEN_MISSING")).contains("\"refusalCode\":\"CSRF_TOKEN_MISSING\""));
140 		assertTrue(Json.of(ActionResult.refusal("app:custom-thing")).contains("\"refusalCode\":\"app:custom-thing\""));
141 	}
142 
143 	@Test void c03_refusal_blankCodeThrows() {
144 		var e = assertThrows(IllegalArgumentException.class, () -> ActionResult.refusal("  "));
145 		assertTrue(e.getMessage().contains("blank"), e::getMessage);
146 		assertThrows(IllegalArgumentException.class, () -> ActionResult.refusal(null));
147 	}

153 	@Test void d01_failure_omitsRowReplayRefusalMessage() {
154 		var json = Json.of(ActionResult.failure());
155 		assertEquals(Json.to("{\"contractVersion\":\"1\",\"outcome\":\"failure\"}", Map.class), Json.to(json, Map.class), json);
156 		for (var k : List.of("row", "replay", "refusalCode", "message"))
157 			assertFalse(json.contains("\"" + k + "\""), () -> "unset field leaked: " + k + "\n" + json);
158 	}
159 
160 	@Test void d02_unknown_isHonestTerminalState() {
161 		var json = Json.of(ActionResult.unknown());
162 		assertEquals(Json.to("{\"contractVersion\":\"1\",\"outcome\":\"unknown\"}", Map.class), Json.to(json, Map.class), json);
163 	}

165 	@Test void d03_replayFalseOrNull_isOmitted() {
166 		assertFalse(Json.of(ActionResult.success(Map.of("id", "x")).replay(false)).contains("\"replay\""));
167 		assertFalse(Json.of(ActionResult.success(Map.of("id", "x")).replay(null)).contains("\"replay\""));
168 	}
169 
170 	@Test void d04_replayTrue_isEmitted() {
171 		assertTrue(Json.of(ActionResult.success(Map.of("id", "x")).replay(true)).contains("\"replay\":true"));
172 	}
173 
174 	@Test void d05_successWithNullRow_omitsRow() {
175 		// A redraw/navigate success carries no row.
176 		var json = Json.of(ActionResult.success(null));
177 		assertEquals(Json.to("{\"contractVersion\":\"1\",\"outcome\":\"success\"}", Map.class), Json.to(json, Map.class), json);
178 	}
