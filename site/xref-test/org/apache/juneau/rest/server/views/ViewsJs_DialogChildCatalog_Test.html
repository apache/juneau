32  /**
33   * The dialog-scoped child-action catalog: {@code ModalDef.childActions} lets a
34   * {@code type="action"} input inside one dialog's form open a STACKED step that is not a row action, and therefore
35   * appears in no row's action menu and in no ribbon.
36   *
37   * <p>Two layers:
38   * <ul class='spaced-list'>
39   * 	<li><b>Default-gate source pins</b> (a-group): the catalog's scoping is structural rather than a filter some
40   * 		menu builder has to remember, the catalog reaches the click site as a threaded parameter rather than as
41   * 		framework state on a DOM node, and the resolution ORDER puts the row-action check first so a served
42   * 		payload can never shadow - and thereby bypass the gating of - a declared row action.
43   * 	<li><b>A behavioral harness</b> ({@code dialog-child-catalog.cjs}) driving real clicks through both
44   * 		row-painting paths.  Runs when {@code node} is on {@code PATH}.
45   * </ul>
46   */
47  class ViewsJs_DialogChildCatalog_Test extends TestBase {

59  	private static String functionBody(String body, String signature) {
60  		var start = body.indexOf(signature);
61  		assertTrue(start >= 0, () -> "'" + signature + "' not found in juneau-views.js");

63  		return body.substring(start, end < 0 ? body.length() : end);
64  	}

93  	/**
94  	 * The catalog reaches the click site as a THREADED PARAMETER.
95  	 *
96  	 * <p>Deliberately not stamped on the dialog element - framework state on a DOM node is forgeable from the
97  	 * document - and deliberately not looked up from {@code topLayer()} at click time, which would assume the
98  	 * button's own dialog is topmost.  That is true today and is exactly the kind of assumption a later stacked
99  	 * surface breaks silently.
100 	 */
101 	@Test void a02_theCatalogIsThreadedNotStampedOnTheDom() throws Exception {

103 		// One trailing optional parameter on each of the six functions in the chain, so every existing positional
104 		// call site keeps working unchanged.
105 		assertTrue(views.contains("function appendDialogForm(dialog, form, table, tr, ctx, seq, childCatalog) {"), views);
106 		assertTrue(views.contains("function appendSectionedDialogForm(dialog, form, table, tr, ctx, seq, childCatalog) {"), views);
107 		assertTrue(views.contains("function appendDialogFormRow(host, dialog, f, table, tr, ctx, seq, childCatalog) {"), views);
108 		assertTrue(views.contains("function paintFormControl(row, f, id, table, tr, ctx, childCatalog) {"), views);
109 		assertTrue(views.contains("function buildActionFormControl(f, table, tr, ctx, row, childCatalog) {"), views);
110 		assertTrue(views.contains("function openFormActionDialog(actionId, table, tr, ctx, childCatalog) {"), views);

122 	/**
123 	 * The resolution ORDER, which is the whole security property.
124 	 *
125 	 * <p>The existing row-action check runs FIRST and unchanged, {@code enabledWhen} included, so a served payload
126 	 * can never shadow a declared row action.  A collision with a NON-dialog row action is a fail-closed miss and
127 	 * the child catalog is not consulted at all - rescuing it there would be the same shadowing hole through a
128 	 * narrower door.  Only an id that resolves to nothing today reaches the catalog.
129 	 */
130 	@Test void a03_theRowActionCheckRunsFirstAndTheCatalogIsTheFallback() throws Exception {
131 		var views = viewsJs();
132 		for (var fn : List.of("\tfunction buildActionFormControl(", "\tfunction openFormActionDialog(")) {
133 			var body = functionBody(views, fn);
134 			var rowCheck = body.indexOf("dialogActionIsOpenable(ctx, actionId)");
135 			var shadow = body.indexOf("rowActionIdExists(ctx, actionId)");
136 			var child = body.indexOf("childActionById(childCatalog, actionId)");
137 			assertTrue(rowCheck >= 0, () -> fn + " must still run the row-action check:\n" + body);
138 			assertTrue(shadow >= 0, () -> fn + " must refuse a non-dialog row-action collision:\n" + body);
139 			assertTrue(child >= 0, () -> fn + " must fall back to the dialog catalog:\n" + body);
140 			assertTrue(rowCheck < shadow && shadow < child,
141 				() -> fn + " must check the row catalog, then the shadow case, then the child catalog:\n" + body);
142 			// The gate on the row-action branch was NOT relaxed to accommodate the new fallback.
143 			assertTrue(body.contains("firstFailingRowActionRule(target, rowDataForTr(ctx, tr))")
144 				|| body.contains("firstFailingRowActionRule(target,"), body);
145 		}
146 		// rowActionIdExists is deliberately BROADER than dialogActionIsOpenable - it is what makes the
147 		// non-dialog collision refusable.
148 		var exists = functionBody(views, "\tfunction rowActionIdExists(");
149 		assertTrue(exists.contains("ctx?.viewDef?.rowActions"), exists);
150 		assertFalse(exists.contains("isDialogAction("),
151 			() -> "rowActionIdExists must see NON-dialog row actions too, or a gated direct-submit action can be "
152 				+ "shadowed:\n" + exists);
153 	}

168 	/**
169 	 * The draft cap is measured on the ENCODED value.
170 	 *
171 	 * <p>Percent-encoding is up to 3x expansion per byte, so a cap applied to the raw JSON would let a ~2KB draft
172 	 * become a ~6KB query and blow a server or proxy request-line limit AFTER the client had already told the
173 	 * operator it was fine - a refusal arriving as an opaque transport error instead of a visible, actionable one.
174 	 */
175 	@Test void a05_theDraftCapIsMeasuredOnTheEncodedValue() throws Exception {
176 		var views = viewsJs();
177 		assertTrue(views.contains("const MAX_DRAFT_QUERY_BYTES = 2048;"), views);
178 		var fn = functionBody(views, "\tfunction withDraftQuery(");
179 		var encode = fn.indexOf("encodeURIComponent(JSON.stringify(drafts");
180 		var measure = fn.indexOf("draftQueryByteLength(encoded) > MAX_DRAFT_QUERY_BYTES");
181 		assertTrue(encode >= 0 && measure >= 0 && encode < measure,
182 			() -> "the cap must be applied to the ENCODED value, not the raw JSON:\n" + fn);
183 		// It returns a COPY rather than mutating the descriptor it was handed.
184 		assertTrue(fn.contains("const copy = {};"), fn);
185 		// Over the cap it returns null so the caller refuses VISIBLY - never a silently-truncated (and therefore
186 		// silently-wrong) prefill.
187 		assertTrue(fn.contains("return null;"), fn);
188 		assertTrue(functionBody(views, "\tfunction renderDialogDraftRefusal(")
189 			.contains("MAX_DRAFT_QUERY_BYTES"), views);
190 	}

209 	@BeforeAll
210 	static void probeIfNodeAvailable() throws Exception {
211 		if (!nodeAvailable())
212 			return;
213 		var harness = locateHarness();
214 		if (harness == null)
215 			return;
216 		var rendersFile = Files.createTempFile("juneau-renders-", ".js");
217 		var viewsFile = Files.createTempFile("juneau-views-", ".js");
218 		try {
219 			Files.writeString(rendersFile, rendersJs(), UTF_8);
220 			Files.writeString(viewsFile, viewsJs(), UTF_8);
221 			report = Json.to(runNode(harness, rendersFile, viewsFile), Map.class);
222 		} finally {
223 			Files.deleteIfExists(rendersFile);
224 			Files.deleteIfExists(viewsFile);
225 		}
226 	}

228 	private static boolean nodeAvailable() {
229 		try {
230 			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
231 			if (!p.waitFor(5, TimeUnit.SECONDS)) {
232 				p.destroyForcibly();
233 				return false;
234 			}
235 			return p.exitValue() == 0;
236 		} catch (Exception e) {
237 			return false;
238 		}

257 	private static String runNode(Path harness, Path rendersJs, Path viewsJs) throws Exception {
258 		var stdout = Files.createTempFile("dialog-child-catalog-stdout-", ".json");
259 		var stderr = Files.createTempFile("dialog-child-catalog-stderr-", ".txt");
260 		try {
261 			var pb = new ProcessBuilder(List.of("node", harness.toString(), rendersJs.toString(), viewsJs.toString()))
262 				.redirectOutput(stdout.toFile())
263 				.redirectError(stderr.toFile());
264 			var p = pb.start();
265 			if (!p.waitFor(30, TimeUnit.SECONDS)) {
266 				p.destroyForcibly();
267 				fail("dialog-child-catalog.cjs did not finish within 30s; stderr:\n" + quietRead(stderr));
268 			}
269 			if (p.exitValue() != 0)
270 				fail("dialog-child-catalog.cjs exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr)
271 					+ "\nstdout:\n" + quietRead(stdout));
272 			return Files.readString(stdout, UTF_8);
273 		} finally {
274 			Files.deleteIfExists(stdout);
275 			Files.deleteIfExists(stderr);
276 		}
277 	}

279 	private static String quietRead(Path p) {
280 		try { return Files.readString(p, UTF_8); }
281 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
282 	}
