242 	@Test void j02_toContractMap_fullDescriptor_roundTripsThroughJson() throws Exception {
243 		var r = RegionDef.create("posture").type(RegionDef.TYPE_CARD_BODY).populate("default")
244 			.dataUrl("/rest/gacks/42/diagnose").params(Map.of("scope", "recent")).renderer("field-grid")
245 			.titleFields("severity").refreshMs(15_000)
246 			.fields(
247 				RegionDef.Field.of("severity").label("Severity").render("pill")
248 					.renderMeta(Map.of("tone", "warning")),
249 				RegionDef.Field.of("host").label("Host").href("servlet:/hosts/{host}"),
250 				RegionDef.Field.of("summary").label("Summary").format(FieldFormat.MARKDOWN)
251 					.span(FieldSpan.FULL),
252 				RegionDef.Field.of("actions").label("").actions("ack", "resolve"));
253 		r.validate();
254 
255 		@SuppressWarnings({
256 			"unchecked" // The (Map<String,Object>) cast of the JSON round-trip of the contract map is safe because toContractMap() is a string-keyed map.
257 		})
258 		var parsed = (Map<String,Object>) Json.to(Json.of(r.toContractMap()), Map.class);
259 		// fields[]: 0=severity (pill render + renderMeta), 1=host (href), 2=summary (markdown format, full span),
260 		// 3=actions (ack/resolve).
261 		assertBean(parsed,
262 			"contractVersion,id,type,populate,dataUrl,params,renderer,lazy,refreshMs,titleFields,"
263 				+ "fields{length,0{data,render,renderMeta},1{href},2{format,span},3{actions}}",
264 			"1,posture,card-body,default,/rest/gacks/42/diagnose,{scope=recent},field-grid,false,15000,[severity],"
265 				+ "{4,{severity,pill,{tone=warning}},{servlet:/hosts/{host}},{markdown,full},{[ack,resolve]}}");
266 	}
