31  /**
32   * Always-on source-shape coverage that every name in {@code JuneauViews.HOOK_NAMESPACES} has a frozen assignment,
33   * plus a Node behavioral proof (via {@code hooks-api.cjs}) that every registered namespace exists and is frozen, that
34   * the registry lists exactly the frozen namespaces the loaded scripts define, and that the original eight namespaces
35   * expose the right member shapes.
36   */

38  	"resource" // Closeable test fixture held in a static field; lifecycle managed by the test/framework, not a real leak.

49  	private static String viewsJs() throws Exception {
50  		return c.get(ViewsMixin.VIEWS_JS_PATH).run().assertStatus(200).getContent().asString();
51  	}

57  	@Test void a01_everyRegisteredNamespaceIsFrozenAndPlacedAfterNsInit() throws Exception {
58  		var body = viewsJs();
59  		var anchor = body.indexOf("renderAsyncStatus: renderAsyncStatus");
60  		assertTrue(anchor >= 0, "NS.init closing anchor not found");
61  		var m = Pattern.compile("NS\\.HOOK_NAMESPACES = Object\\.freeze\\(\\[([^\\]]*)\\]\\)").matcher(body);
62  		assertTrue(m.find(), "NS.HOOK_NAMESPACES = Object.freeze([...]) not found");
63  		var names = new ArrayList<String>();
64  		var q = Pattern.compile("\"(\\w+)\"").matcher(m.group(1));
65  		while (q.find())
66  			names.add(q.group(1));
67  		assertTrue(names.containsAll(List.of("actions", "rowActions", "dialogs", "probeSelection", "tables", "status",
68  			"csrf", "rules", "details", "html")), () -> "registry lost a known namespace: " + names);
69  		for (var name : names) {
70  			var ns = "NS." + name + " = Object.freeze(";
71  			assertTrue(body.indexOf(ns) > anchor, () -> ns + " not found after the NS.init anchor");
72  		}
73  	}
74  
75  	@Test void a02_rowActionsRunResolvesContextBeforeDispatching() throws Exception {
76  		var body = viewsJs();
77  		var start = body.indexOf("NS.rowActions = Object.freeze(");
78  		var fragment = body.substring(start, body.indexOf("NS.dialogs = Object.freeze(", start));
79  		assertTrue(fragment.contains("hookContextOf"), fragment);
80  		assertTrue(fragment.contains("isDialogAction"), fragment);
81  		assertTrue(fragment.contains("E-JS-74"), fragment);
82  		assertTrue(fragment.contains("E-JS-70"), fragment);
83  	}

89  	private static Map<?,?> report() {
90  		var r = NodeHarness.report("hooks-api.cjs", ViewsMixin.VIEWS_JS_RESOURCE, ViewsMixin.RENDERS_JS_RESOURCE);
91  		assumeTrue(r != null, "node not available or hooks-api.cjs not found - behavioral layer skipped");
92  		return r;
93  	}

96  		"unchecked" // Casts a parsed JSON value to its known Map/List shape.
97  	})
98  	private static Map<String,Object> ns(String name) {
99  		return (Map<String,Object>)report().get(name);
100 	}

102 	@Test void b01_everyNamespaceExistsAndIsFrozen() {
103 		var registered = (Map<?,?>)report().get("registered");
104 		assertFalse(registered.isEmpty(), "JuneauViews.HOOK_NAMESPACES is missing or empty");
105 		for (var name : registered.keySet()) {
106 			var shape = (Map<?,?>)registered.get(name);
107 			assertEquals(Boolean.TRUE, shape.get("exists"), name.toString());
108 			assertEquals(Boolean.TRUE, shape.get("frozen"), name.toString());
109 		}
110 	}
111 
112 	@Test void b01a_registryIsFrozenAndListsEveryHookNamespace() {
113 		var r = report();
114 		assertEquals(Boolean.TRUE, r.get("hookNamespacesFrozen"), "HOOK_NAMESPACES must be a frozen array");
115 		assertEquals(new HashSet<>(List.of("actions", "rowActions", "dialogs", "probeSelection", "tables", "status",
116 			"csrf", "rules", "details", "html", "jobs")), new HashSet<>((Collection<?>)r.get("hookNamespaces")));
117 		assertEquals(new HashSet<>((Collection<?>)r.get("hookNamespaces")),
118 			new HashSet<>((Collection<?>)r.get("frozenNamespaces")),
119 			"every frozen top-level JuneauViews namespace must be listed in HOOK_NAMESPACES, and vice versa");
120 	}
