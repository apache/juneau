29  /**
30   * Shared Node plumbing for the {@code juneau-regions.js} behavioral harnesses.
31   *
32   * <p>
33   * The region runtime's behavioral coverage is split across harness scripts by SUBJECT - the populate primitive,
34   * the message bus, the initial-broadcast barrier, the declarative default populator - and each one is driven by its
35   * own test class so a failure names the layer that broke.  Splitting the driver plumbing out of those classes is not
36   * just deduplication: it means every harness runs the SAME loader against the SAME assets in the same order, so a
37   * "passes in one class, fails in another" result can only come from the harness under test.  The declarative default's
38   * harness alone also loads {@code juneau-helpers.js} (via {@link #reportWithHelpers}), because the reserved default
39   * paints through {@code ctx.helpers[...]} - the other three harnesses deliberately do not carry that dependency.
40   *
41   * <p>
42   * Each harness is spawned once per JVM and its JSON report cached, because the barrier harness alone builds
43   * twenty-odd independent pages.
44   */
45  final class RegionsHarness {
46  
47  	private static final Map<String,Map<?,?>> CACHE = new ConcurrentHashMap<>();
48  	private static final Set<String> FAILED = ConcurrentHashMap.newKeySet();
49  
50  	private RegionsHarness() {}
51  
52  	/**
53  	 * Runs the named harness and returns its report, or null when this machine cannot run it (no Node, or the script
54  	 * is not on any of the paths a Maven or IDE working directory produces).  Callers turn a null into a skip.
55  	 */
56  	static Map<?,?> report(String harnessName) {
57  		return reportImpl(harnessName, null);
58  	}
59  
60  	/**
61  	 * Runs the named harness with {@code juneau-helpers.js} ALSO loaded (a fourth argv path), for the declarative
62  	 * default's own harness - the reserved default paints via {@code ctx.helpers[...]}, so its harness needs the
63  	 * helper library the three original harnesses (primitive/bus/barrier) deliberately do not load.
64  	 */
65  	static Map<?,?> reportWithHelpers(String harnessName) {
66  		return reportImpl(harnessName, List.of(ViewsMixin.HELPERS_JS_RESOURCE));
67  	}
68  
69  	/**
70  	 * Runs the named harness with extra classpath resources written to temp files and appended to its argv, in the
71  	 * given order, after the renders/views/regions trio.  Each temp file keeps its resource's extension, so a
72  	 * {@code .json} resource stays {@code .json}.
73  	 */
74  	static Map<?,?> reportWith(String harnessName, String... extraResources) {
75  		return reportImpl(harnessName, List.of(extraResources));
76  	}
77  
78  	/**
79  	 * Runs the named harness with {@code juneau-bus.js} ALSO passed (a fourth argv path), for the page-bus harness:
80  	 * {@code ctx.publish} / {@code ctx.subscribe} delegate to {@code JuneauViews.bus}, and the harness hands the path
81  	 * to {@code load(..., {busJsPath})}, which loads it FIRST - the order the "views" toolkit pack emits it.
82  	 */
83  	static Map<?,?> reportWithBus(String harnessName) {
84  		return reportImpl(harnessName, List.of(ViewsMixin.BUS_JS_RESOURCE));
85  	}
86  
87  	private static Map<?,?> reportImpl(String harnessName, List<String> extras) {
88  		var cacheKey = (extras == null || extras.isEmpty() ? "" : String.join(",", extras) + ":") + harnessName;
89  		if (FAILED.contains(cacheKey))
90  			return null;
91  		var cached = CACHE.get(cacheKey);
92  		if (cached != null)
93  			return cached;
94  		try {
95  			if (!nodeAvailable())
96  				return markUnavailable(cacheKey);
97  			var harness = locate(harnessName);
98  			if (harness == null)
99  				return markUnavailable(cacheKey);
100 			var report = Json.to(run(harness, harnessName, extras == null ? List.of() : extras), Map.class);
101 			CACHE.put(cacheKey, report);
102 			return report;
103 		} catch (Exception e) {
104 			// A harness that cannot be READ is a skip; a harness that RAN and failed has already called fail().
105 			throw new AssertionError("could not run " + harnessName + ": " + e, e);
106 		}
107 	}
108 
109 	private static Map<?,?> markUnavailable(String harnessName) {
110 		FAILED.add(harnessName);
111 		return null;
112 	}
113 
114 	private static String run(Path harness, String harnessName, List<String> extras) throws Exception {
115 		var renders = Files.createTempFile("juneau-renders-", ".js");
116 		var views = Files.createTempFile("juneau-views-", ".js");
117 		var regions = Files.createTempFile("juneau-regions-", ".js");
118 		var stdout = Files.createTempFile("regions-stdout-", ".json");
119 		var stderr = Files.createTempFile("regions-stderr-", ".txt");
120 		var cleanup = new ArrayList<Path>(List.of(renders, views, regions, stdout, stderr));
121 		try {
122 			Files.writeString(renders, asset(ViewsMixin.RENDERS_JS_RESOURCE), UTF_8);
123 			Files.writeString(views, asset(ViewsMixin.VIEWS_JS_RESOURCE), UTF_8);
124 			Files.writeString(regions, asset(ViewsMixin.REGIONS_JS_RESOURCE), UTF_8);
125 			var args = new ArrayList<String>(List.of(
126 				"node", harness.toString(), renders.toString(), views.toString(), regions.toString()));
127 			for (var res : extras) {
128 				var dot = res.lastIndexOf('.');
129 				var extra = Files.createTempFile("juneau-extra-", dot < 0 ? ".js" : res.substring(dot));
130 				cleanup.add(extra);
131 				Files.writeString(extra, asset(res), UTF_8);
132 				args.add(extra.toString());
133 			}
134 			var p = new ProcessBuilder(args)
135 				.redirectOutput(stdout.toFile())
136 				.redirectError(stderr.toFile())
137 				.start();
138 			if (!p.waitFor(60, TimeUnit.SECONDS)) {
139 				p.destroyForcibly();
140 				fail(harnessName + " did not finish within 60s; stderr:\n" + quietRead(stderr));
141 			}
142 			if (p.exitValue() != 0)
143 				fail(harnessName + " exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr)
144 					+ "\nstdout:\n" + quietRead(stdout));
145 			return Files.readString(stdout, UTF_8);
146 		} finally {
147 			for (var f : cleanup)
148 				Files.deleteIfExists(f);
149 		}
150 	}
151 
152 	private static String asset(String resource) throws IOException {
153 		try (var in = ViewsMixin.class.getResourceAsStream(resource)) {
154 			assertNotNull(in, resource);
155 			return new String(in.readAllBytes(), UTF_8);
156 		}
157 	}
158 
159 	private static boolean nodeAvailable() {
160 		try {
161 			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
162 			if (!p.waitFor(5, TimeUnit.SECONDS)) {
163 				p.destroyForcibly();
164 				return false;
165 			}
166 			return p.exitValue() == 0;
167 		} catch (Exception e) {
168 			return false;
169 		}
170 	}
171 
172 	private static Path locate(String harnessName) {
173 		var basedir = System.getProperty("basedir");
174 		if (basedir != null) {
175 			var p = Path.of(basedir, "src/test/js/" + harnessName);
176 			if (Files.isRegularFile(p))
177 				return p.toAbsolutePath().normalize();
178 		}
179 		for (var rel : List.of(
180 			"src/test/js/" + harnessName,
181 			"juneau-rest/juneau-rest-server-views/src/test/js/" + harnessName
182 		)) {
183 			var p = Path.of(rel);
184 			if (Files.isRegularFile(p))
185 				return p.toAbsolutePath().normalize();
186 		}
187 		return null;
188 	}
189 
190 	private static String quietRead(Path p) {
191 		try { return Files.readString(p, UTF_8); }
192 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
193 	}
194 
195 	/** Reads the region runtime's source, for the source-shape assertions each driver opens with. */
196 	static String regionsJs() throws IOException {
197 		return asset(ViewsMixin.REGIONS_JS_RESOURCE);
198 	}
