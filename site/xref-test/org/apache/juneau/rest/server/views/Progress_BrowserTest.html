32  /**
33   * Opt-in Chromium canary for the {@code progress} renderer.  Always-on Node-vm coverage lives in
34   * {@link ViewsJs_Renders_Test}; this class injects {@code display()} HTML into a real DOM.
35   *
36   * <p>
37   * Disabled unless the {@value #GATE} system property is set (the module's {@code js-tests} Maven profile).
38   */
39  @EnabledIfSystemProperty(named=Progress_BrowserTest.GATE, matches="true",
40  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -f juneau-rest/juneau-rest-server-views/pom.xml test`")
41  class Progress_BrowserTest extends TestBase {
42  
43  	/** System property the {@code js-tests} profile sets to enable this class. */
44  	static final String GATE = "juneau.jsTests";
45  
46  	private static Map<?,?> report;
47  
48  	private static String resource(String path) throws IOException {
49  		try (var in = ViewsMixin.class.getResourceAsStream(path)) {
50  			assertNotNull(in, () -> "missing classpath resource: " + path);
51  			return new String(in.readAllBytes(), UTF_8);
52  		}
53  	}
54  
55  	@BeforeAll
56  	static void probe() throws Exception {
57  		var dir = Path.of(requiredProperty("juneau.jsTests.dir"));
58  		var harness = Path.of(requiredProperty("juneau.jsTests.harness")).getParent().resolve("progress-browser.cjs");
59  
60  		var fixture = "<!DOCTYPE html><html><head><meta charset=\"utf-8\"></head><body>\n<script>\n"
61  			+ resource(ViewsMixin.RENDERS_JS_RESOURCE)
62  			+ "\n</script></body></html>";
63  		var fixtureFile = Files.createDirectories(dir.resolve("fixtures")).resolve("progress.html");
64  		Files.write(fixtureFile, fixture.getBytes(UTF_8));
65  
66  		report = Json.to(run(dir, harness, fixtureFile), Map.class);
67  	}
68  
69  	private static String requiredProperty(String name) {
70  		var v = System.getProperty(name);
71  		assertNotNull(v, () -> "-D" + name + " not set; the js-tests profile is responsible for providing it");
72  		return v;
73  	}
74  
75  	private static String run(Path dir, Path harness, Path fixture) throws Exception {
76  		var cmd = List.of(System.getProperty("juneau.jsTests.node", "node"), harness.toString(), fixture.toString());
77  		var stdout = dir.resolve("progress-stdout.json");
78  		var stderr = dir.resolve("progress-stderr.txt");
79  		var pb = new ProcessBuilder(cmd).redirectOutput(stdout.toFile()).redirectError(stderr.toFile());
80  		pb.environment().put("NODE_PATH", dir.resolve("node_modules").toString());
81  		pb.environment().put("PLAYWRIGHT_BROWSERS_PATH", requiredProperty("juneau.jsTests.browsers"));
82  
83  		var p = pb.start();
84  		if (!p.waitFor(3, TimeUnit.MINUTES)) {
85  			p.destroyForcibly();
86  			fail("prober did not finish within 3m; stderr:\n" + quietRead(stderr));
87  		}
88  		assertEquals(0, p.exitValue(), () -> "prober exited non-zero; stderr:\n" + quietRead(stderr));
89  		return Files.readString(stdout);
90  	}
91  
92  	private static String quietRead(Path p) {
93  		try {
94  			return Files.readString(p);
95  		} catch (IOException e) {
96  			return "<unreadable: " + e + ">";
97  		}
98  	}

103 	}

116 		assertEquals(Boolean.TRUE, report.get("overLabel"), () -> report.toString());
117 	}
