29  /**
30   * Shared Node plumbing for the {@code juneau-bus.js} harnesses ({@code bus-*.cjs}).
31   *
32   * <p>
33   * Same contract as {@link RegionsHarness}, but the asset list is the caller's: each classpath resource is copied to a
34   * temp file and passed as one argv path, in order, so a harness that also needs {@code juneau-regions.js} or the
35   * console shell names it explicitly.  The JSON report is cached per (harness, assets) for the JVM.
36   *
37   * <h5 class='section'>Example:</h5>
38   * <p class='bjava'>
39   * 	<jk>var</jk> <jv>report</jv> = BusHarness.<jsm>run</jsm>(<js>"bus-core.cjs"</js>, ViewsMixin.<jsf>BUS_JS_RESOURCE</jsf>);
40   * 	<jsm>assumeTrue</jsm>(<jv>report</jv> != <jk>null</jk>, <js>"node not on PATH"</js>);
41   * </p>
42   */
43  final class BusHarness {
44  
45  	private static final Map<String,Map<String,Object>> CACHE = new ConcurrentHashMap<>();
46  	private static final Set<String> UNAVAILABLE = ConcurrentHashMap.newKeySet();
47  
48  	private BusHarness() {}
49  
50  	/**
51  	 * Runs {@code src/test/js/<cjsFileName>} with the given classpath resources as argv paths and returns its JSON
52  	 * report, or null when this machine cannot run it (no Node, or the script is not on any path a Maven or IDE
53  	 * working directory produces).  A harness that RAN and failed is an assertion failure, never a null.
54  	 */
55  	@SuppressWarnings("unchecked")
56  	static Map<String,Object> run(String cjsFileName, String... classpathResources) {
57  		var key = cjsFileName + ":" + String.join(",", classpathResources);
58  		if (UNAVAILABLE.contains(key))
59  			return null;
60  		var cached = CACHE.get(key);
61  		if (cached != null)
62  			return cached;
63  		try {
64  			if (!nodeAvailable()) {
65  				UNAVAILABLE.add(key);
66  				return null;
67  			}
68  			var harness = locate(cjsFileName);
69  			if (harness == null) {
70  				UNAVAILABLE.add(key);
71  				return null;
72  			}
73  			var report = (Map<String,Object>) Json.to(exec(harness, cjsFileName, classpathResources), Map.class);
74  			CACHE.put(key, report);
75  			return report;
76  		} catch (Exception e) {
77  			throw new AssertionError("could not run " + cjsFileName + ": " + e, e);
78  		}
79  	}
80  
81  	/** Reads a classpath asset's source, for the source-shape assertions a driver opens with. */
82  	static String source(String resource) throws IOException {
83  		try (var in = ViewsMixin.class.getResourceAsStream(resource)) {
84  			assertNotNull(in, resource);
85  			return new String(in.readAllBytes(), UTF_8);
86  		}
87  	}
88  
89  	private static String exec(Path harness, String name, String[] resources) throws Exception {
90  		var temps = new ArrayList<Path>();
91  		var stdout = Files.createTempFile("bus-stdout-", ".json");
92  		var stderr = Files.createTempFile("bus-stderr-", ".txt");
93  		try {
94  			var args = new ArrayList<String>(List.of("node", harness.toString()));
95  			for (var r : resources) {
96  				var base = r.substring(r.lastIndexOf('/') + 1).replace(".js", "");
97  				var f = Files.createTempFile(base + "-", ".js");
98  				temps.add(f);
99  				Files.writeString(f, source(r), UTF_8);
100 				args.add(f.toString());
101 			}
102 			var p = new ProcessBuilder(args).redirectOutput(stdout.toFile()).redirectError(stderr.toFile()).start();
103 			if (!p.waitFor(60, TimeUnit.SECONDS)) {
104 				p.destroyForcibly();
105 				fail(name + " did not finish within 60s; stderr:\n" + quietRead(stderr));
106 			}
107 			if (p.exitValue() != 0)
108 				fail(name + " exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr) + "\nstdout:\n" + quietRead(stdout));
109 			return Files.readString(stdout, UTF_8);
110 		} finally {
111 			temps.add(stdout);
112 			temps.add(stderr);
113 			for (var f : temps)
114 				Files.deleteIfExists(f);
115 		}
116 	}
117 
118 	private static boolean nodeAvailable() {
119 		try {
120 			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
121 			if (!p.waitFor(5, TimeUnit.SECONDS)) {
122 				p.destroyForcibly();
123 				return false;
124 			}
125 			return p.exitValue() == 0;
126 		} catch (Exception e) {
127 			return false;
128 		}
129 	}
130 
131 	private static Path locate(String name) {
132 		var basedir = System.getProperty("basedir");
133 		if (basedir != null) {
134 			var p = Path.of(basedir, "src/test/js/" + name);
135 			if (Files.isRegularFile(p))
136 				return p.toAbsolutePath().normalize();
137 		}
138 		for (var rel : List.of("src/test/js/" + name, "juneau-rest/juneau-rest-server-views/src/test/js/" + name)) {
139 			var p = Path.of(rel);
140 			if (Files.isRegularFile(p))
141 				return p.toAbsolutePath().normalize();
142 		}
143 		return null;
144 	}
145 
146 	private static String quietRead(Path p) {
147 		try { return Files.readString(p, UTF_8); }
148 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
149 	}
