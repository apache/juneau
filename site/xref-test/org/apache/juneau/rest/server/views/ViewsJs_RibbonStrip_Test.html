32  /**
33   * Always-on coverage for the SHARED ribbon-format strip builder.
34   *
35   * <p>The strip used to exist only inside the row-detail expander.  It is now {@code buildRibbonStrip(items, opts)} -
36   * given an ordered {@code [{id, label, pane}]} it wires {@code role=tab} / {@code aria-selected} /
37   * {@code aria-controls} / {@code role=tabpanel} / a roving tabindex, moves selection on Left/Right/Home/End, and
38   * fires {@code onActivate} once per activation.  After the detail-strip deletion, the remaining caller is
39   * {@code appendSectionedDialogForm} (a sectioned dialog form); {@code helpers.tabStrip} wraps the same builder.
40   */

57  	/** The body of a top-level {@code function name(} up to the next top-level {@code \n\tfunction } / {@code \n\t/**}. */
58  	private static String bodyOf(String src, String fn) {
59  		var start = src.indexOf("\tfunction " + fn + "(");
60  		assertTrue(start >= 0, () -> "no top-level function " + fn + " in juneau-views.js");
61  		var end = src.indexOf("\n\t}\n", start);
62  		assertTrue(end > start, () -> "unterminated function " + fn);
63  		return src.substring(start, end);
64  	}

128 	private static int count(String haystack, String needle) {
129 		var n = 0;
130 		for (var i = haystack.indexOf(needle); i >= 0; i = haystack.indexOf(needle, i + needle.length()))
131 			n++;
132 		return n;
133 	}

142 	static void probeIfNodeAvailable() throws Exception {
143 		if (!nodeAvailable())
144 			return;
145 		var harness = locateHarness();
146 		if (harness == null)
147 			return;
148 		var viewsFile = Files.createTempFile("juneau-views-", ".js");
149 		var rendersFile = Files.createTempFile("juneau-renders-", ".js");
150 		try {
151 			Files.writeString(viewsFile, viewsJs(), UTF_8);
152 			Files.writeString(rendersFile, rendersJs(), UTF_8);
153 			report = Json.to(runNode(harness, rendersFile, viewsFile), Map.class);
154 		} finally {
155 			Files.deleteIfExists(viewsFile);
156 			Files.deleteIfExists(rendersFile);
157 		}
158 	}

160 	private static boolean nodeAvailable() {
161 		try {
162 			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
163 			if (!p.waitFor(5, TimeUnit.SECONDS)) {
164 				p.destroyForcibly();
165 				return false;
166 			}
167 			return p.exitValue() == 0;
168 		} catch (Exception e) {
169 			return false;
170 		}
171 	}

173 	private static Path locateHarness() {

175 		if (basedir != null) {

177 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
178 		}
179 		for (var rel : List.of(

181 			"juneau-rest/juneau-rest-server-views/src/test/js/ribbon-strip.cjs"
182 		)) {
183 			var p = Path.of(rel);
184 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
185 		}
186 		return null;
187 	}

189 	private static String runNode(Path harness, Path rendersJs, Path viewsJs) throws Exception {
190 		var stdout = Files.createTempFile("ribbon-strip-stdout-", ".json");
191 		var stderr = Files.createTempFile("ribbon-strip-stderr-", ".txt");
192 		try {
193 			var pb = new ProcessBuilder(List.of("node", harness.toString(), rendersJs.toString(), viewsJs.toString()))
194 				.redirectOutput(stdout.toFile())
195 				.redirectError(stderr.toFile());
196 			var p = pb.start();
197 			if (!p.waitFor(30, TimeUnit.SECONDS)) {
198 				p.destroyForcibly();
199 				fail("ribbon-strip.cjs did not finish within 30s; stderr:\n" + quietRead(stderr));
200 			}
201 			if (p.exitValue() != 0)
202 				fail("ribbon-strip.cjs exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr)
203 					+ "\nstdout:\n" + quietRead(stdout));
204 			return Files.readString(stdout, UTF_8);
205 		} finally {
206 			Files.deleteIfExists(stdout);
207 			Files.deleteIfExists(stderr);
208 		}
209 	}

211 	private static String quietRead(Path p) {
212 		try { return Files.readString(p, UTF_8); }
213 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
214 	}
