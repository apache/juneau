30   * {@code JuneauViews.bus.wiring.attachBridges}.  {@code bus-bridge-ws.cjs} drives
31   * {@code JuneauViews.bus.sources.websocket} with a fake {@code WebSocket}, a scripted session {@code fetch} and a fake

46  	@SuppressWarnings({
47  		"unchecked" // Casts a parsed JSON value to its known Map/List shape.
48  	})
49  	private static Map<String,Object> section(String... path) {
50  		assumeTrue(report != null, "node not on PATH");
51  		Map<String,Object> m = report;
52  		for (var p : path) {
53  			var next = (Map<String,Object>) m.get(p);
54  			assertNotNull(next, "report has no '" + String.join(".", path) + "': " + report);
55  			m = next;
56  		}
57  		return m;
58  	}

60  	private static void assertJson(String expected, Object actual) {
61  		try {
62  			assertEquals(Json.to(expected, Object.class), actual, () -> String.valueOf(actual));
63  		} catch (Exception e) {
64  			fail("bad expected JSON: " + expected + ": " + e);
65  		}
66  	}

68  	/** One row of the close-code table: what bridge:w looked like after the server closed with {@code code}. */
69  	private static void assertClose(String code, String state, int nextRetryMs, String error, String reported, String message,
70  			int posts, int sockets) {
71  		var r = section("close", "c" + code);
72  		var why = "close " + code + ": " + r;
73  		assertEquals(state, r.get("state"), why);
74  		assertEquals(nextRetryMs, ((Number) r.get("nextRetryMs")).intValue(), why);
75  		assertEquals(error, r.get("error"), why);
76  		assertEquals(reported == null ? List.of() : List.of(reported), r.get("codes"), why);
77  		assertEquals(message, r.get("message"), why);
78  		assertEquals(posts, ((Number) r.get("posts")).intValue(), why);
79  		assertEquals(sockets, ((Number) r.get("sockets")).intValue(), why);
80  	}
