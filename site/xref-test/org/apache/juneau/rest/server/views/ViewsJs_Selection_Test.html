53  	private static String functionBody(String body, String signature) {
54  		var start = body.indexOf(signature);
55  		assertTrue(start >= 0, () -> "'" + signature + "' not found:\n" + body);
56  		var end = body.indexOf("\n\t}", start);
57  		return body.substring(start, end < 0 ? body.length() : end);
58  	}
