20  import static org.junit.jupiter.api.Assumptions.*;

27  /** Drives {@code run-view-env-check.cjs}: the run-view test environment's own decorations. */
28  class ViewsJs_RunViewEnv_Test extends TestBase {
29  
30  	static final String VECTORS = "/org/apache/juneau/rest/server/views/run-view-vectors.json";
31  
32  	static Map<?,?> report(String harness) {
33  		var r = RegionsHarness.reportWith(harness,
34  			ViewsMixin.HELPERS_JS_RESOURCE, ViewsMixin.RUN_VIEW_JS_RESOURCE, VECTORS);
35  		assumeTrue(r != null, "node not available or " + harness + " not found - skipped");
36  		return r;
37  	}
38  
39  	static void assertAllTrue(Map<?,?> r, String... keys) {
40  		for (var k : keys)
41  			assertEquals(true, r.get(k), () -> k + " -> " + r.get(k) + " in " + r);
42  	}
43  
44  	private static Map<?,?> r() { return report("run-view-env-check.cjs"); }
45  
46  	@Test void a01_fragments() { assertAllTrue(r(), "t01_fragments"); }
47  	@Test void a02_geometry() { assertAllTrue(r(), "t02_geometry"); }
48  	@Test void a03_scrollIntoView() { assertAllTrue(r(), "t03_scrollIntoView"); }
49  	@Test void a04_scrollEvents() { assertAllTrue(r(), "t04_scrollEvents"); }
50  	@Test void a05_windowState() { assertAllTrue(r(), "t05_windowState"); }
51  	@Test void a06_fakeDate() { assertAllTrue(r(), "t06_fakeDate"); }
52  	@Test void a07_loaded() { assertAllTrue(r(), "t07_loaded"); }
53  }
