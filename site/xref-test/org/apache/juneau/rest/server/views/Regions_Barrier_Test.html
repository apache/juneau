27  /**
28   * Always-on coverage for the page-scoped INITIAL-BROADCAST BARRIER and its bounded replay window.
29   *
30   * <p>
31   * The barrier exists for one shape: a control region that emits during its own {@code reason:"initial"} populate,
32   * before the regions that care have subscribed.  Buffering that emit is the easy half.  The hard half is RELEASING
33   * the buffer, and the failure the design rates worse than the bug it replaces is a barrier that never lifts - a page
34   * where all subsequent user-driven traffic is frozen for the life of the tab, with a clean console.  That is why
35   * three separate non-settlement modes each have their own test, why the arming protocol has a loud backstop, and why
36   * several assertions here are against the CLOCK rather than against delivery order: for a hidden enrolled card,
37   * delivery is correct under both the right and the wrong reading and only the latency differs, so an order-based
38   * assertion passes either way.
39   *
40   * <p>
41   * Behavioral half: {@code src/test/js/regions-barrier.cjs}.
42   */
43  class Regions_Barrier_Test extends TestBase {
44  
45  	private static Map<?,?> report() {
46  		var r = RegionsHarness.report("regions-barrier.cjs");
47  		assumeTrue(r != null, "node not available or regions-barrier.cjs not found - behavioral layer skipped");
48  		return r;
49  	}
50  
51  	private static void assertAllTrue(Map<?,?> r, String... keys) {
52  		for (var k : keys)
53  			assertEquals(true, r.get(k), () -> k + " -> " + r.get(k) + " in " + r);
54  	}
55  
56  	private static void assertNothingReceived(Map<?,?> r, String key) {
57  		assertEquals(List.of(), r.get(key), () -> key + " must have received nothing, got " + r.get(key));
58  	}
59  
60  	/** Every reading in {@code key} is the given clock time - the assertion that catches a needless page-wide wait. */
61  	private static void assertDeliveredAt(Map<?,?> r, String key, int expectedMs) {
62  		var readings = (List<?>)r.get(key);
63  		assertFalse(readings.isEmpty(), () -> key + " recorded no delivery at all");
64  		for (var at : readings)
65  			assertEquals(expectedMs, ((Number)at).intValue(),
66  				() -> key + " delivered at " + readings + ", expected every delivery at " + expectedMs + "ms");
67  	}

73  	/**
74  	 * A control region that emits during its own {@code reason:"initial"} populate AND INITIALIZES FIRST still reaches
75  	 * three chart regions that subscribe during their own later initial populates - including one whose populate
76  	 * returns a Promise settling after the control's, which is what proves the barrier waits on ASYNC populates and
77  	 * not merely on synchronous return.  A region activated LATER receives nothing retroactively: the design declines
78  	 * replay-on-subscribe for a region that was never a barrier member.
79  	 */
80  	@Test void b49_aControlEmittingDuringItsOwnInitialPopulateReachesLaterSubscribers() {
81  		var r = report();
82  		assertEquals(List.of(List.of(), List.of(), List.of()), r.get("t49a_beforeAsyncSettle"),
83  			"nothing may be delivered while an initial populate is still in flight");
84  		assertEquals(List.of(List.of("M1"), List.of("M1"), List.of("M1")), r.get("t49a_afterAsyncSettle"), r::toString);
85  		assertDeliveredAt(r, "t49a_drainedAtZeroNotDeadline", 0);
86  		assertAllTrue(r, "t49a_noDeadlineWarning", "t49a_controlDidNotReceiveItsOwn");
87  		assertNothingReceived(r, "t49b_lateActivatedGetsNothing");
88  	}

105 	 * Mode 2: a REJECTED initial populate releases the barrier.  Worth a separate test from the synchronous throw

108 	 */

153 	/**
154 	 * THE TWO-HOST CASE.  A page carrying both a card grid and a tabbed page shell, whose two runtimes boot
155 	 * independently and hold no reference to each other: the barrier is PAGE-scoped, so the grid's control card still
156 	 * reaches a chart in the page shell's initial panel.  Under a host-scoped barrier this fails, and it fails
157 	 * INTERMITTENTLY - on boot order - which is the worst way for it to fail.  Late activation gets nothing whether
158 	 * the late region is a naked tab body or a card in a non-initial panel, so that rule is about activation and not
159 	 * about which runtime enrolled the region.
160 	 */
161 	@Test void b49e_theBarrierIsPageScopedAcrossTwoIndependentRuntimes() {
162 		var r = report();
163 		assertNothingReceived(r, "t49e_notDrainedAfterFirstRuntime");
164 		assertEquals(List.of("M1"), r.get("t49e_i_gridChart"), r::toString);
165 		assertEquals(List.of("M1"), r.get("t49e_i_panelChart"),
166 			"a page-scoped barrier reaches the other runtime's regions");
167 		assertDeliveredAt(r, "t49e_i_drainedAtZero", 0);
168 		assertNothingReceived(r, "t49e_ii_lazyTab");
169 		assertNothingReceived(r, "t49e_iii_lazyCard");
170 	}

172 	/**
173 	 * THE ARMING PROTOCOL.  (i) An EMPTY outstanding set is not enough - the barrier holds until every registered
174 	 * runtime has declared its enrolment walk complete, because "no regions outstanding" and "no regions enrolled
175 	 * yet" are indistinguishable from the barrier's side.  (ii) A page with NO regions pays nothing: nothing arms, no
176 	 * timer is scheduled, no document listener is added.  (iii) A registered runtime that NEVER reports has the
177 	 * barrier lift one macrotask after DOMContentLoaded with exactly one error naming it - it degrades LOUDLY, which
178 	 * is the inverse of the failure mode a clock-only design was rejected for.
179 	 */
180 	@Test void b49f_theArmingProtocolHoldsOnAnEmptySetAndHasALoudBackstop() {
181 		var r = report();
182 		assertNothingReceived(r, "t49f_i_heldOnEmptyOutstanding");
183 		assertEquals(List.of("M1"), r.get("t49f_i_liftedOnSecondReport"), r::toString);
184 		assertDeliveredAt(r, "t49f_i_liftedAtZero", 0);
185 
186 		assertAllTrue(r, "t49f_ii_noTimers", "t49f_ii_noDocumentListeners", "t49f_ii_assetLoadedButUnarmed");
187 
188 		assertNothingReceived(r, "t49f_iii_heldBeforeMacrotask");
189 		assertEquals(List.of("M1"), r.get("t49f_iii_liftedAtMacrotask"), r::toString);
190 		assertDeliveredAt(r, "t49f_iii_liftedAtZeroNotDeadline", 0);
191 		assertAllTrue(r, "t49f_iii_oneErrorNamingRuntime", "t49f_iii_errorMentionsReady");
192 	}

198 	/**
199 	 * THE REPLAY WINDOW DELIVERS.  The healthy regions receive the broadcast at the deadline - no page-wide wait -
200 	 * and the slow region's LATER {@code ctx.on} receives the same messages, in the same order, exactly once.  The
201 	 * assertion that distinguishes a bounded replay window from a retained cache is the last one: a subsequent LIVE
202 	 * emit arrives after the replayed ones and is not duplicated.
203 	 */
204 	@Test void b49g_aReleasedRegionsLaterSubscribeReceivesWhatItMissed() {
205 		var r = report();
206 		assertNothingReceived(r, "t49g_heldBeforeDeadline");
207 		assertEquals(List.of("M1"), r.get("t49g_healthyGotItAtDeadline"), r::toString);
208 		assertDeliveredAt(r, "t49g_healthyAt", 2000);
209 		assertNothingReceived(r, "t49g_slowNothingYet");
210 		assertEquals(List.of("M1"), r.get("t49g_slowReplayed"), r::toString);
211 		assertEquals(true, r.get("t49g_slowReplayedOnce"), r::toString);
212 		assertEquals(List.of("M1", "M2"), r.get("t49g_slowAfterLiveEmit"),
213 			"a live emit lands after the replayed ones and is not duplicated");
214 		assertEquals(List.of("M1", "M2"), r.get("t49g_healthyAfterLiveEmit"), r::toString);
215 	}
216 
217 	/**
218 	 * THE WINDOW CLOSES, AND IS NOT A RETAINED CACHE.  Four negatives, because each is a way the replay could grow
219 	 * into the retained-cache option the design rejected: a region enrolled after the lift gets nothing; a released
220 	 * region that subscribes after its OWN populate settles gets nothing; the buffer is discarded once the last
221 	 * window closes; and the replay reads NO payload property - which is what keeps payload opacity true on the
222 	 * second delivery path, and precisely the objection the retained-cache option failed.
223 	 */
224 	@Test void b49h_theReplayWindowClosesAndIsNotARetainedCache() {
225 		var r = report();
226 		assertNothingReceived(r, "t49h_i_lateEnrolledGetsNothing");
227 		assertNothingReceived(r, "t49h_ii_afterOwnSettleGetsNothing");
228 		assertNothingReceived(r, "t49h_iii_freshRegionGetsNothing");
229 		assertAllTrue(r, "t49h_iv_replayReadsNothing", "t49h_iv_replayedIdentityPreserved");
230 		assertEquals(List.of("M1"), r.get("t49h_iv_replayed"), r::toString);
231 	}
232 
233 	/**
234 	 * A REPLAY IS A FAN-OUT: a replayed handler that emits has its emission QUEUED like any other, and a second
235 	 * released region whose window is still open receives it.  Run as two arms of the same page - one where both
236 	 * regions are deadline-released and one where both settle before the deadline - and the expected sequences are
237 	 * DERIVED by requiring the two arms to agree, rather than asserted as literals.  That is what keeps the replay
238 	 * path from drifting out of agreement with ordinary delivery: a literal would let both drift together.
239 	 */
240 	@Test void b49i_replayIsAFanOutAndAgreesWithOrdinaryDelivery() {
241 		var r = report();
242 		assertEquals(List.of("A-had-M2-at-emit-return:false"), r.get("t49i_i_emitWasQueuedNotLive"),
243 			"an emit from a replayed handler must be queued, not delivered live inside the replay");
244 		assertEquals(List.of("M1", "M2"), r.get("t49i_ii_cReceivedBoth"),
245 			"a still-open second window receives what the replay emits");
246 		assertEquals(true, r.get("t49i_iii_identicalSequences"),
247 			() -> "released arm " + r.get("t49i_iii_released") + " != control arm " + r.get("t49i_iii_control"));
248 		assertNothingReceived(r, "t49i_iv_lateEnrolled");
249 		assertNothingReceived(r, "t49i_iv_hiddenAtBoot");
250 		assertEquals(true, r.get("t49i_noCycleError"), r::toString);
251 	}
252 
253 	/**
254 	 * THE LAST-WINDOW CLOSE, with a genuinely SYNCHRONOUS settle - a thenable whose {@code then} runs the framework's
255 	 * continuation in the caller's own turn, so a released region's close condition becomes true PARTWAY THROUGH a
256 	 * live fan-out.  Resolving an ordinary promise from inside a subscriber cannot reach this case at all: resolving
257 	 * queues a microtask while the fan-out is synchronous, so the continuation runs after the drain has finished and
258 	 * the assertion silently tests the ordinary case instead.
259 	 *
260 	 * <p>
261 	 * Membership is snapshotted at the fan-out's START, so the message IS in the buffer that region replays - had the
262 	 * close taken effect immediately it would have replayed nothing - and the buffer's contents depend on message
263 	 * order and member-open-time rather than on the order subscribers happen to run in, which is what the two
264 	 * subscription-order arms check.
265 	 */
266 	@Test void b49l_aWindowThatClosesMidFanOutStillCountedForThatFanOut() {
267 		var r = report();
268 		assertEquals(List.of("M3"), r.get("t49l_i_cReplayedM3"), r::toString);
269 		assertEquals(true, r.get("t49l_i_exactlyOnce"), r::toString);
270 		assertEquals(true, r.get("t49l_ii_orderIndependent"),
271 			() -> "the two subscription-order arms disagreed: " + r.get("t49l_ii_arms"));
272 		assertEquals(List.of("M3", "M4"), r.get("t49l_iii_cGotM4Live"), r::toString);
273 		assertEquals(true, r.get("t49l_iv_bufferDiscarded"),
274 			"the buffer must be discarded once the last window closes");
275 	}

281 	/**
282 	 * ENROLLED-BUT-HIDDEN IS NOT A BARRIER MEMBER: enrolment and barrier membership are different axes, and getting
283 	 * them confused means every tabbed card page waits the full deadline before the visible card gets its initial
284 	 * broadcasts.  The first assertion is the one with teeth and it is against the CLOCK, because delivery is correct
285 	 * under both readings and only the latency differs.  The hidden region IS enrolled - it has a handle and a
286 	 * reachable teardown - it has simply had no populate at all, which is WHY it is not outstanding rather than a
287 	 * special case; and activating it later produces exactly one {@code reason:"activate"} populate that receives
288 	 * nothing retroactively.
289 	 */
290 	@Test void b49j_anEnrolledButHiddenRegionIsNotABarrierMember() {
291 		var r = report();

294 		assertEquals(0, ((Number)r.get("t49j_i_clockAtDelivery")).intValue(),
295 			"a hidden card must not delay the drain to the deadline");
296 		assertEquals(true, r.get("t49j_i_noDeadlineWarning"), r::toString);
297 
298 		assertAllTrue(r, "t49j_ii_enrolled", "t49j_ii_teardownReachable");
299 		assertEquals("hiddenCard", r.get("t49j_ii_hasKey"));
300 		assertEquals("idle", r.get("t49j_iii_state"));

308 	/**
309 	 * THE REPLAY BUFFER'S OVERFLOW IS NOT A CYCLE ERROR.  257 legitimate, non-cyclic, user-driven broadcasts, each
310 	 * from a different emitter at the top of its own turn, so no fan-out is ever nested.
311 	 *
312 	 * <p>
313 	 * (i) No cycle error, at any point.  Under a single shared cap this page raises one at append 65 - a false
314 	 * diagnostic naming a defect that does not exist, which is worse than silence.  (ii) A region that settles at
315 	 * append 200 replays exactly the first 200 in order, exactly once, so two regions on the same page reach two
316 	 * different outcomes.  (iii) At append 257 the buffer is ABANDONED WHOLE and loudly: the still-open window
317 	 * receives NO replay rather than a truncated one, because a run of messages with a hole is the outcome the design
318 	 * rates worse than none - the region cannot detect it - and exactly one error names the affected key, the cap, the
319 	 * exactly-once failure, and the fact that this is not a cycle.  (iv) The two caps are independent in BOTH
320 	 * directions: this page overflowed its buffer without a cycle error, and a genuinely nesting page raises the cycle
321 	 * error without overflowing a buffer.  Under one shared constant one of those two must fail.
322 	 */
323 	@Test void b49k_theTwoCapsAreIndependentAndOverflowIsNotReportedAsACycle() {
324 		var r = report();
325 		assertEquals(true, r.get("t49k_bothReleased"), r::toString);
326 
327 		assertEquals(true, r.get("t49k_i_noCycleError"),
328 			"257 top-of-turn broadcasts contain no cycle and must not be reported as one");
329 
330 		assertEquals(200, ((Number)r.get("t49k_ii_dReplayCount")).intValue(), r::toString);
331 		assertEquals("m1", r.get("t49k_ii_dReplayFirst"));
332 		assertEquals("m200", r.get("t49k_ii_dReplayLast"));
333 		assertEquals(true, r.get("t49k_ii_dInOrder"), r::toString);
334 
335 		assertAllTrue(r, "t49k_iii_exactlyOneOverflowError", "t49k_iii_namesKey", "t49k_iii_namesCap",
336 			"t49k_iii_namesExactlyOnce", "t49k_iii_saysNotACycle", "t49k_iii_namesOnlyTheOpenWindow",
337 			"t49k_iii_noPartialReplay");
338 		assertNothingReceived(r, "t49k_iii_cReceivedNoReplay");
339 
340 		assertAllTrue(r, "t49k_iv_bufferOverflowedWithoutCycleError", "t49k_iv_capsDiffer",
341 			"t49k_iv_nestingRaisesCycleError", "t49k_iv_nestingRaisedNoOverflowError");
