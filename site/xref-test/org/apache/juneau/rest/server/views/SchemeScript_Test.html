33  /**
34   * The pre-paint color-scheme script ({@link ConsoleChromeMixin#SCHEME_SCRIPT}, the exact constant) over its full
35   * precedence matrix: a stored choice wins only on a page that enables the toggle, then the server pin, then the OS;
36   * anything else is light.  Runs in plain Node ({@code scheme-script.cjs}); skipped when Node is not available.
37   */

102 	private static Map<String,String> expected() {
103 		var m = new LinkedHashMap<String,String>();
104 		for (var line : EXPECTED.strip().split("\n")) {
105 			var p = line.strip().split("\\s+");
106 			assertEquals(6, p.length, () -> "bad expected row: " + line);
107 			m.put("toggle=" + p[0] + " stored=" + p[1] + " pin=" + p[2] + " os=" + p[3], p[5]);
108 		}
109 		return m;
110 	}

112 	private static Map<String,Map<String,Object>> run(String script) throws Exception {
113 		var harness = locateHarness();
114 		assumeTrue(harness != null && nodeAvailable(), "node not available or scheme-script.cjs not found");
115 		var src = Files.createTempFile("scheme-script-", ".js");
116 		var stdout = Files.createTempFile("scheme-script-stdout-", ".json");
117 		var stderr = Files.createTempFile("scheme-script-stderr-", ".txt");
118 		try {
119 			Files.writeString(src, script, UTF_8);
120 			var p = new ProcessBuilder(List.of(System.getProperty("juneau.jsTests.node", "node"), harness.toString(), src.toString()))
121 				.redirectOutput(stdout.toFile()).redirectError(stderr.toFile()).start();
122 			if (! p.waitFor(60, TimeUnit.SECONDS)) {
123 				p.destroyForcibly();
124 				fail("scheme-script.cjs did not finish within 60s; stderr:\n" + quietRead(stderr));
125 			}
126 			assertEquals(0, p.exitValue(), () -> "scheme-script.cjs exited non-zero; stderr:\n" + quietRead(stderr));
127 			return (Map<String,Map<String,Object>>)Json.to(Files.readString(stdout, UTF_8), Map.class);
128 		} finally {
129 			Files.deleteIfExists(src);
130 			Files.deleteIfExists(stdout);
131 			Files.deleteIfExists(stderr);
132 		}
133 	}

135 	@Test void a01_matrix() throws Exception {
136 		var rows = run(ConsoleChromeMixin.SCHEME_SCRIPT).get("rows");
137 		var exp = expected();
138 		assertEquals(48, exp.size());
139 		assertEquals(exp.keySet(), rows.keySet());
140 		var wrong = new ArrayList<String>();
141 		exp.forEach((k, v) -> {
142 			if (! v.equals(rows.get(k)))
143 				wrong.add(k + ": expected " + v + ", got " + rows.get(k));
144 		});
145 		assertEquals(List.of(), wrong);
146 	}

148 	@Test void a02_failureRows() throws Exception {
149 		var f = run(ConsoleChromeMixin.SCHEME_SCRIPT).get("failures");
150 		assertEquals("dark", f.get("storageThrowsFallsToPin"), f::toString);
151 		assertEquals("dark", f.get("storageThrowsFallsToOs"), f::toString);
152 		assertEquals("light", f.get("matchMediaMissing"), f::toString);
153 		assertEquals("dark", f.get("matchMediaMissingPinned"), f::toString);
154 	}

156 	/** Without the toggle guard, a stored choice leaks onto pages that never offered the control. */
157 	@Test void a03_guardMatters() throws Exception {
158 		assertTrue(ConsoleChromeMixin.SCHEME_SCRIPT.contains(TOGGLE_GUARD), "guard text moved; update TOGGLE_GUARD");
159 		var rows = run(ConsoleChromeMixin.SCHEME_SCRIPT.replace(TOGGLE_GUARD, "")).get("rows");
160 		var exp = expected();
161 		var differ = exp.keySet().stream().filter(k -> ! exp.get(k).equals(rows.get(k))).toList();
162 		assertFalse(differ.isEmpty(), "removing the toggle guard changed no row");
163 		assertTrue(differ.contains("toggle=no stored=dark pin=none os=light"), differ::toString);
164 		assertTrue(differ.stream().allMatch(k -> k.startsWith("toggle=no ")), differ::toString);
165 	}

167 	private static Path locateHarness() {
168 		var basedir = System.getProperty("basedir");
169 		if (basedir != null) {
170 			var p = Path.of(basedir, "src/test/js/scheme-script.cjs");
171 			if (Files.isRegularFile(p))
172 				return p.toAbsolutePath().normalize();
173 		}
174 		for (var rel : List.of("src/test/js/scheme-script.cjs", "juneau-rest/juneau-rest-server-views/src/test/js/scheme-script.cjs")) {
175 			var p = Path.of(rel);
176 			if (Files.isRegularFile(p))
177 				return p.toAbsolutePath().normalize();
178 		}
179 		return null;
180 	}

182 	private static boolean nodeAvailable() {
183 		try {
184 			var p = new ProcessBuilder(System.getProperty("juneau.jsTests.node", "node"), "--version").redirectErrorStream(true).start();
185 			if (! p.waitFor(5, TimeUnit.SECONDS)) {
186 				p.destroyForcibly();
187 				return false;
188 			}
189 			return p.exitValue() == 0;
190 		} catch (IOException e) {
191 			return false;
192 		} catch (InterruptedException e) {
193 			Thread.currentThread().interrupt();
194 			return false;
195 		}
196 	}
197 
198 	private static String quietRead(Path p) {
199 		try {
200 			return Files.readString(p, UTF_8);
201 		} catch (IOException e) {
202 			return "<unreadable: " + e + ">";
203 		}
