32  /**
33   * The in-dialog result <b>receipt</b>: a dialog that opted in with
34   * {@code ModalDef.keepOpenOnSubmit} keeps its layer across the submit, and the runtime paints the write's result
35   * into that already-open dialog instead of closing it and painting a row banner.
36   *
37   * <p>Two layers, the same split {@code ViewsJs_RibbonDialog_Test} uses:
38   * <ul class='spaced-list'>
39   * 	<li><b>Default-gate source pins</b> (a-group): the <b>coupling guard</b> that keeps
40   * 		{@code submitActionDialog} free of receipt coupling, the <b>enforcement by absence</b> that makes the receipt
41   * 		non-submittable, the hold's one-settle discipline, and the narrow allowlist of sites permitted to tear the
42   * 		dialog stack down automatically.
43   * 	<li><b>A behavioral harness</b> ({@code dialog-receipt.cjs}) walking all nine submit terminals and checking
44   * 		the disposition each settles to.  The interesting failures here - a stuck busy modal, a Confirm left
45   * 		disabled on a retryable refusal, a receipt painted over a still-populated form, one outcome rendered twice
46   * 		- are DOM-shaped and invisible to a source pin.  Runs when {@code node} is on {@code PATH}.
47   * </ul>
48   */
49  class ViewsJs_DialogReceipt_Test extends TestBase {

61  	private static String functionBody(String body, String signature) {
62  		var start = body.indexOf(signature);
63  		assertTrue(start >= 0, () -> "'" + signature + "' not found in juneau-views.js");
64  		var end = body.indexOf("\n\t}", start);
65  		return body.substring(start, end < 0 ? body.length() : end);
66  	}

68  	private static int countOf(String body, String needle) {
69  		var n = 0;
70  		for (var i = body.indexOf(needle); i >= 0; i = body.indexOf(needle, i + needle.length()))
71  			n++;
72  		return n;
73  	}

79  	/**
80  	 * The COUPLING GUARD, and the single most load-bearing pin in this file.
81  	 *
82  	 * <p>The receipt needs to know, at settle time, which dialog to paint into - and the obvious way to arrange
83  	 * that is a callback parameter on the submit.  {@code submitActionDialog} is the one function EVERY
84  	 * {@code present=dialog} submit goes through, and the row-less dialog work put its {@code targetId} precedence rule and that
85  	 * rule's tripwire there precisely because it is the only path.  Threading a receipt parameter through it would
86  	 * re-open a reviewed function to add a feature that has nothing to do with what it decides.  The coupling
87  	 * therefore runs through a ctx-scoped hold registered at CLICK time instead, which leaves this function
88  	 * with no receipt coupling.
89  	 */
90  	@Test void a01_submitActionDialogHasNoReceiptCoupling() throws Exception {
91  		var fn = functionBody(viewsJs(), "\tfunction submitActionDialog(");
92  		// The targetId precedence rule and its opt-in gate, quoted verbatim.
93  		assertTrue(fn.contains("const targetId = (modal?.selfTargeted && modal?.idempotencyKey != null)"), fn);
94  		assertTrue(fn.contains(": (tr?.dataset?.juneauRowId ?? null);"), fn);
95  		// Nothing about the receipt reached it: no hold, no callback, no result-form parameter.
96  		for (var forbidden : List.of("_resultHold", "beginDialogResultHold", "settleDialogResultHold",
97  				"resultForm", "keepOpenOnSubmit", "onResult", "callback")) {
98  			assertFalse(fn.contains(forbidden),
99  				() -> "submitActionDialog must have no receipt coupling - it grew '" + forbidden + "':\n" + fn);
100 		}
101 		// Its only trailing parameter beyond the original signature is the rowActions.run caller's extra - nothing receipt-shaped.
102 		assertTrue(viewsJs().contains("function submitActionDialog(modal, action, table, tr, ctx, fields, callerExtra) {"), fn);
103 	}

154 	 *
155 	 * <p>The submit graph has nine terminals; they are mutually-exclusive paths through one {@code submitRowAction}
156 	 * call, so clearing before dispatching means a second settle on the same hold finds nothing. That ordering is

161 		var read = fn.indexOf("const hold = ctx?._resultHold");

163 		var live = fn.indexOf("heldLayerIsLive(ctx, hold)");
164 		assertTrue(read >= 0 && clear >= 0 && live >= 0, fn);
165 		assertTrue(read < clear && clear < live,
166 			() -> "the hold must be READ, then CLEARED, then dispatched - or a terminal can settle twice:\n" + fn);

169 	/**
170 	 * Every terminal in the submit graph hands off exactly once, and the exemption is per-disposition.
171 	 *
172 	 * <p>{@code closeCommitted} returns {@code false} on purpose: the dialog is gone, so the row / ribbon banner is
173 	 * the only surviving signal and suppressing it would leave a committed write with no visible confirmation
174 	 * anywhere.  Every other disposition returns {@code true}, because the dialog is still on screen and painting
175 	 * the row banner too would render one outcome twice, once behind a backdrop.
176 	 */
177 	@Test void a06_allNineTerminalsHandOffExactlyOnce() throws Exception {
178 		var views = viewsJs();
179 		var submit = functionBody(views, "\tfunction submitRowAction(");
180 		var settle = functionBody(views, "\tfunction settleActionResponse(");
181 		// Client refusal + network reject.
182 		assertEquals(2, countOf(submit, "settleDialogResultHold(ctx, "),
183 			() -> "submitRowAction has exactly two terminals:\n" + submit);
184 		// The six response terminals, six hand-offs, one each.
185 		assertEquals(6, countOf(settle, "settleDialogResultHold(ctx, "),
186 			() -> "settleActionResponse has exactly six terminals:\n" + settle);
187 		// The body-read failure terminal is shared by BOTH readBodyText chains, so it is one function reached from two
188 		// `.catch` arms.
189 		assertEquals(2, countOf(settle, "settleBodyReadFailure(action, table, tr, ctx)"),
190 			() -> "both body-read chains must route to the body-read terminal, or a held dialog hangs:\n" + settle);
191 		assertEquals(1, countOf(functionBody(views, "\tfunction settleBodyReadFailure("),
192 			"settleDialogResultHold(ctx, "), views);
193 		// The dispositions actually used, and no others.
194 		for (var d : List.of("\"retryable\"", "\"terminal\"", "\"closeCommitted\"", "\"receipt\""))
195 			assertTrue(views.contains("settleDialogResultHold(ctx, " + d + ")")
196 				|| views.contains("settleDialogResultHold(ctx, disposition)"), d);
197 		var hold = functionBody(views, "\tfunction settleDialogResultHold(");
198 		assertTrue(hold.contains("if (disposition === \"closeCommitted\")"), hold);
199 		assertTrue(hold.contains("closeActionDialog(ctx);\n\t\t\treturn false;"),
200 			() -> "closeCommitted must return FALSE so the caller's row painter still runs:\n" + hold);
201 	}

203 	/**
204 	 * The receipt terminal FETCHES before it settles.
205 	 *
206 	 * <p>Settling first would clear {@code ctx._resultHold} before the GET resolved, so every arm inside
207 	 * {@code fetchResultForm} would take the layer-gone escape and a committed write with a perfectly good receipt
208 	 * payload would paint no receipt at all - silently.
209 	 */
210 	@Test void a07_theReceiptTerminalFetchesBeforeItSettles() throws Exception {
211 		var settle = functionBody(viewsJs(), "\tfunction settleActionResponse(");
212 		var fetch = settle.indexOf("fetchResultForm(result.resultForm");
213 		assertTrue(fetch >= 0, settle);
214 		// The receipt branch returns immediately after the fetch: it must not also fall through to a settle.
215 		var tail = settle.substring(fetch);
216 		var nextSettle = tail.indexOf("settleDialogResultHold(");
217 		var ret = tail.indexOf("return;");
218 		assertTrue(ret >= 0 && (nextSettle < 0 || ret < nextSettle),
219 			() -> "the receipt branch must fetch and RETURN, never fetch and then settle:\n" + tail);
220 		// And the fetch is gated on a HELD success carrying a resultForm - never on a failure, refusal or unknown,
221 		// so no result host can appear for a write that did not commit.
222 		assertTrue(settle.contains("if (held && outcome === \"success\" && ! isBlankToken(result.resultForm))"),
223 			settle);
224 	}

226 	/**
227 	 * The automatic-teardown allowlist.
228 	 *
229 	 * <p>Exactly three NEW sites may call {@code closeActionDialog(} - the {@code closeCommitted} branch, the
230 	 * receipt's Close-button wiring, and the shared Close-only row helper - plus the pre-existing grandfathered
231 	 * {@code teardownTable} caller.  Every OTHER terminal path must end with the dialog OPEN and delegate its exit
232 	 * to a Close button the operator presses, because a terminal that closes the stack behind the operator's back
233 	 * destroys the message it was about to show them.
234 	 */
235 	@Test void a08_noTerminalPathTearsTheStackDownAutomatically() throws Exception {
236 		var views = viewsJs();
237 		// The BAN, stated as absences on the automatic paths: none of these may close the stack, because a
238 		// terminal that closes it behind the operator's back destroys the message it was about to show them.
239 		for (var fn : List.of("\tfunction settleBodyReadFailure(", "\tfunction fetchResultForm(",
240 				"\tfunction renderHeldResultNotice(", "\tfunction renderResultFormIgnored(",
241 				"\tfunction beginDialogResultHold(", "\tfunction removeDialogActionsRow(",
242 				"\tfunction teardownCommittedWriteBody(", "\tfunction restoreHeldChildActions(")) {
243 			var body = functionBody(views, fn);
244 			assertFalse(body.contains("closeActionDialog("),
245 				() -> fn + " must not tear the stack down automatically:\n" + body);
246 		}
247 		// The three permitted NEW sites.  Two of them are CLICK LISTENERS - a Close button the operator presses is
248 		// that operator's exit, not an automatic teardown - and the third is the one disposition whose write is
249 		// known-committed and whose dialog is therefore meant to go.
250 		assertTrue(functionBody(views, "\tfunction appendDialogCloseOnlyRow(")
251 			.contains("close.addEventListener(\"click\", function () { closeActionDialog(ctx); });"), views);
252 		var receipt = functionBody(views, "\tfunction paintDialogReceipt(");
253 		assertTrue(receipt.contains("close.addEventListener(\"click\", function () { closeActionDialog(ctx); });"),
254 			receipt);
255 		assertEquals(1, countOf(receipt, "closeActionDialog("),
256 			() -> "the receipt's ONLY teardown is its Close button's listener:\n" + receipt);
257 		var settle = functionBody(views, "\tfunction settleDialogResultHold(");
258 		assertEquals(1, countOf(settle, "closeActionDialog("),
259 			() -> "exactly one disposition may close the stack:\n" + settle);
260 		var branch = settle.indexOf("if (disposition === \"closeCommitted\")");
261 		assertTrue(branch >= 0 && branch < settle.indexOf("closeActionDialog("),
262 			() -> "the one teardown must sit inside the closeCommitted branch:\n" + settle);
263 	}

365 	private static String runNode(Path harness, Path rendersJs, Path viewsJs) throws Exception {
366 		var stdout = Files.createTempFile("dialog-receipt-stdout-", ".json");
367 		var stderr = Files.createTempFile("dialog-receipt-stderr-", ".txt");
368 		try {
369 			var pb = new ProcessBuilder(List.of("node", harness.toString(), rendersJs.toString(), viewsJs.toString()))
370 				.redirectOutput(stdout.toFile())
371 				.redirectError(stderr.toFile());
372 			var p = pb.start();
373 			if (!p.waitFor(30, TimeUnit.SECONDS)) {
374 				p.destroyForcibly();
375 				fail("dialog-receipt.cjs did not finish within 30s; stderr:\n" + quietRead(stderr));
376 			}
377 			if (p.exitValue() != 0)
378 				fail("dialog-receipt.cjs exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr)
379 					+ "\nstdout:\n" + quietRead(stdout));
380 			return Files.readString(stdout, UTF_8);
381 		} finally {
382 			Files.deleteIfExists(stdout);
383 			Files.deleteIfExists(stderr);
384 		}
385 	}

387 	private static String quietRead(Path p) {
388 		try { return Files.readString(p, UTF_8); }
389 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
390 	}
