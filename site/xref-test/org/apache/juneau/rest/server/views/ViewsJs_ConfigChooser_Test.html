54  	private static String functionBody(String body, String signature) {
55  		var start = body.indexOf(signature);
56  		assertTrue(start >= 0, () -> "'" + signature + "' not found:\n" + body);
57  		var i = body.indexOf('{', start);
58  		assertTrue(i >= 0, () -> "'" + signature + "' has no opening brace:\n" + body);
59  		var depth = 0;
60  		var j = i;
61  		for (; j < body.length(); j++) {
62  			var c = body.charAt(j);
63  			if (c == '{') {
64  				depth++;
65  			} else if (c == '}') {
66  				depth--;
67  				if (depth == 0) { j++; break; }
68  			} else if (c == '"' || c == '\'' || c == '`') {
69  				var quote = c;
70  				j++;
71  				while (j < body.length() && body.charAt(j) != quote) { if (body.charAt(j) == '\\') j++; j++; }
72  			} else if (c == '/' && j + 1 < body.length() && body.charAt(j + 1) == '/') {
73  				while (j < body.length() && body.charAt(j) != '\n') j++;
74  			} else if (c == '/' && j + 1 < body.length() && body.charAt(j + 1) == '*') {
75  				j = body.indexOf("*/", j) + 1;
76  			}
77  		}
78  		return body.substring(start, j);
79  	}

220 	@BeforeAll
221 	static void probeIfNodeAvailable() throws Exception {
222 		if (!nodeAvailable())
223 			return;
224 		var harness = locateHarness();
225 		if (harness == null)
226 			return;
227 		var configFile = Files.createTempFile("juneau-config-", ".js");
228 		try {
229 			Files.writeString(configFile, configJs(), UTF_8);
230 			report = Json.to(runNode(harness, configFile), Map.class);
231 		} finally {
232 			Files.deleteIfExists(configFile);
233 		}
234 	}

236 	private static boolean nodeAvailable() {
237 		try {
238 			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
239 			if (!p.waitFor(5, TimeUnit.SECONDS)) {
240 				p.destroyForcibly();
241 				return false;
242 			}
243 			return p.exitValue() == 0;
244 		} catch (Exception e) {
245 			return false;
246 		}
247 	}

249 	private static Path locateHarness() {
250 		var basedir = System.getProperty("basedir");
251 		if (basedir != null) {
252 			var p = Path.of(basedir, "src/test/js/config-chooser.cjs");
253 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
254 		}
255 		for (var rel : List.of(
256 			"src/test/js/config-chooser.cjs",
257 			"juneau-rest/juneau-rest-server-views/src/test/js/config-chooser.cjs"
258 		)) {
259 			var p = Path.of(rel);
260 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
261 		}
262 		return null;
263 	}

265 	private static String runNode(Path harness, Path configJs) throws Exception {
266 		var stdout = Files.createTempFile("config-chooser-stdout-", ".json");
267 		var stderr = Files.createTempFile("config-chooser-stderr-", ".txt");
268 		try {
269 			var pb = new ProcessBuilder(List.of("node", harness.toString(), configJs.toString()))
270 				.redirectOutput(stdout.toFile())
271 				.redirectError(stderr.toFile());
272 			var p = pb.start();
273 			if (!p.waitFor(30, TimeUnit.SECONDS)) {
274 				p.destroyForcibly();
275 				fail("config-chooser.cjs did not finish within 30s; stderr:\n" + quietRead(stderr));
276 			}
277 			if (p.exitValue() != 0)
278 				fail("config-chooser.cjs exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr)
279 					+ "\nstdout:\n" + quietRead(stdout));
280 			return Files.readString(stdout, UTF_8);
281 		} finally {
282 			Files.deleteIfExists(stdout);
283 			Files.deleteIfExists(stderr);
284 		}
285 	}

287 	private static String quietRead(Path p) {
288 		try { return Files.readString(p, UTF_8); }
289 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
290 	}
