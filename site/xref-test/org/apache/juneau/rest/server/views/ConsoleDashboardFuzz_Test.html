19  import static org.junit.jupiter.api.Assertions.*;
20  import static org.junit.jupiter.api.Assumptions.*;

22  import java.util.*;

24  import org.apache.juneau.*;

28  /**
29   * The seeded share-link fuzz over {@code juneau-dashboard.js} without a DOM, plus the hand-written edge cases for the
30   * banned keys, depth and cycle rejection, and the payload and fragment size caps.  Runs when node is on PATH.
31   */
32  class ConsoleDashboardFuzz_Test extends TestBase {
33  
34  	private static Map<?,?> part(String key) {
35  		var r = NodeHarness.report("dashboard-fuzz.cjs", ConsoleChromeMixin.DASHBOARD_JS_RESOURCE);
36  		assumeTrue(r != null, "node (or dashboard-fuzz.cjs) not available - JS layer skipped");
37  		var s = (Map<?,?>)r.get(key);
38  		assertNotNull(s, () -> "section '" + key + "' missing from the harness output");
39  		assertNull(s.get("crash"), () -> "section '" + key + "' crashed: " + s.get("crash"));
40  		return s;
41  	}
42  
43  	private static final String NOT_PLAIN = "E-JS-104: share link rejected: decoder 'app.d' returned a value that is not plain JSON; using saved state or defaults";
44  
45  	@Test void a01_fuzz_shareLinkCodecAndRestore() {
46  		var r = part("fuzz");
47  		assertEquals(List.of(), r.get("failures"), "seed 0x5eed0694; replay the case number shown");
48  		assertEquals(2400, ((Number)r.get("cases")).intValue());
49  		assertEquals(List.of("decoder", "link", "null", "saved"), r.get("sources"));
50  		var per = (Map<?,?>)r.get("perCategory");
51  		assertEquals(12, per.size());
52  		for (var v : per.values())
53  			assertEquals(200, ((Number)v).intValue());
54  		assertEquals(true, r.get("polluted"));
55  	}
56  
57  	@Test void a02_bannedKeys_rejectedEverywhere() {
58  		var e = part("edges");
59  		assertEquals(List.of("the link carries the key '__proto__'", "the link carries the key 'constructor'", "the link carries the key 'prototype'"), e.get("linkBanned"));
60  		for (var o : (List<?>)e.get("savedBanned")) {
61  			assertNull(((Map<?,?>)o).get("source"));
62  			assertEquals(true, ((Map<?,?>)o).get("deleteSaved"));
63  		}
64  		var dec = (List<?>)e.get("decoderBanned");
65  		assertEquals(3, dec.size());
66  		for (var o : dec) {
67  			assertNull(((Map<?,?>)o).get("source"));
68  			assertEquals(List.of(NOT_PLAIN), ((Map<?,?>)o).get("warns"));
69  		}
70  		assertEquals(true, e.get("polluted"));
71  	}
72  
73  	@Test void a03_depthAndCycles_rejected() {
74  		var e = part("edges");
75  		for (var k : List.of("cycle", "cycleArray", "deepDecoder")) {
76  			var m = (Map<?,?>)e.get(k);
77  			assertNull(m.get("source"), k);
78  			assertEquals(List.of(NOT_PLAIN), m.get("warns"), k);
79  		}
80  		assertEquals("decoder:app.d", ((Map<?,?>)e.get("shallowDecoder")).get("source"));
81  		assertEquals(List.of(), ((Map<?,?>)e.get("shallowDecoder")).get("warns"));
82  		assertNull(((Map<?,?>)e.get("deepSaved")).get("source"));
83  		assertEquals(true, ((Map<?,?>)e.get("deepSaved")).get("deleteSaved"));
84  		assertEquals("the link is nested more than 64 deep", e.get("deepLink"));
85  		assertEquals(List.of("t"), e.get("shallowLink"));
86  	}
87  
88  	@Test void a04_payloadCap_countsBytesNotCharacters() {
89  		var c = (Map<?,?>)part("edges").get("cap");
90  		for (var k : List.of("atBytes", "threeBytes", "fourBytes"))
91  			assertEquals(4096, ((Number)c.get(k)).intValue(), k);
92  		assertEquals(4097, ((Number)c.get("overBytes")).intValue());
93  		// Fewer than 4096 characters but more than 4096 bytes.
94  		assertTrue(((Number)c.get("charsLength")).intValue() < 4096);
95  		assertTrue(((Number)c.get("charsBytes")).intValue() > 4096);
96  		assertEquals(List.of("filter:q"), c.get("at"));
97  		assertEquals(List.of("filter:q"), c.get("three"));
98  		assertEquals(List.of("filter:q"), c.get("four"));
99  		assertEquals(List.of("filter:r"), c.get("over"), "the over-cap payload is dropped, the other topic kept");
100 		assertEquals(List.of("filter:r"), c.get("chars"));
101 		assertEquals(true, c.get("atRoundTrip"));
102 		assertEquals(true, c.get("overCheck"));
103 		assertEquals(List.of("filter:q", "filter:r"), c.get("twoSmall"));
104 	}
105 
106 	@Test void a05_fragmentCap_nullOverEightKiB() {
107 		var e = part("edges");
108 		assertEquals(true, ((Map<?,?>)e.get("cap")).get("tooBigIsNull"));
109 		assertEquals("the link is not base64url", ((Map<?,?>)e.get("fragmentCap")).get("at"));
110 		assertEquals("the link is over 8192 characters", ((Map<?,?>)e.get("fragmentCap")).get("over"));
111 	}
