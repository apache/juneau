33  /**
34   * Behavioral coverage for the general page-state store.  A Node harness ({@code page-state.cjs}) loads
35   * {@code juneau-pagestate.js} standalone into the DOM shim and reports: two tables on one page keep separate per-table
36   * keys, a page-level namespace is independent, a JSON value round-trips through the default store, a replaced store
37   * implementation is actually used (and the default store is bypassed), and a blocked store degrades to a silent no-op
38   * rather than throwing.  Gated on {@code node} being on {@code PATH}.
39   */
40  class ViewsJs_PageState_Test extends TestBase {
41  
42  	private static String resource(String name) throws IOException {
43  		try (var in = ViewsMixin.class.getResourceAsStream(name)) {
44  			assertNotNull(in, name);
45  			return new String(in.readAllBytes(), UTF_8);
46  		}
47  	}
48  
49  	private static Map<?,?> report;
50  
51  	@BeforeAll
52  	static void probeIfNodeAvailable() throws Exception {
53  		if (!nodeAvailable())
54  			return;
55  		var harness = locateHarness();
56  		if (harness == null)
57  			return;
58  		var pageStateFile = Files.createTempFile("juneau-pagestate-", ".js");
59  		try {
60  			Files.writeString(pageStateFile, resource(ViewsMixin.PAGESTATE_JS_RESOURCE), UTF_8);
61  			report = Json.to(runNode(harness, pageStateFile), Map.class);
62  		} finally {
63  			Files.deleteIfExists(pageStateFile);
64  		}
65  	}
66  
67  	private static boolean nodeAvailable() {
68  		try {
69  			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
70  			if (!p.waitFor(5, TimeUnit.SECONDS)) {
71  				p.destroyForcibly();
72  				return false;
73  			}
74  			return p.exitValue() == 0;
75  		} catch (Exception e) {
76  			return false;
77  		}
78  	}
79  
80  	private static Path locateHarness() {
81  		var basedir = System.getProperty("basedir");
82  		if (basedir != null) {
83  			var p = Path.of(basedir, "src/test/js/page-state.cjs");
84  			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
85  		}
86  		for (var rel : List.of(
87  			"src/test/js/page-state.cjs",
88  			"juneau-rest/juneau-rest-server-views/src/test/js/page-state.cjs"
89  		)) {
90  			var p = Path.of(rel);
91  			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
92  		}
93  		return null;
94  	}
95  
96  	private static String runNode(Path harness, Path pageStateJs) throws Exception {
97  		var stdout = Files.createTempFile("page-state-stdout-", ".json");
98  		var stderr = Files.createTempFile("page-state-stderr-", ".txt");
99  		try {
100 			var pb = new ProcessBuilder(List.of("node", harness.toString(), pageStateJs.toString()))
101 				.redirectOutput(stdout.toFile())
102 				.redirectError(stderr.toFile());
103 			var p = pb.start();
104 			if (!p.waitFor(30, TimeUnit.SECONDS)) {
105 				p.destroyForcibly();
106 				fail("page-state.cjs did not finish within 30s; stderr:\n" + quietRead(stderr));
107 			}
108 			if (p.exitValue() != 0)
109 				fail("page-state.cjs exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr)
110 					+ "\nstdout:\n" + quietRead(stdout));
111 			return Files.readString(stdout, UTF_8);
112 		} finally {
113 			Files.deleteIfExists(stdout);
114 			Files.deleteIfExists(stderr);
115 		}
116 	}
117 
118 	private static String quietRead(Path p) {
119 		try { return Files.readString(p, UTF_8); }
120 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
121 	}
122 
123 	private static Map<?,?> report() {
124 		assumeTrue(report != null, "node not available or page-state.cjs not found — behavioral layer skipped");
125 		return report;
126 	}
127 
128 	@Test void a01_moduleAttachesPageStateScope() {
129 		var r = report();
130 		assertEquals(true, r.get("hasPageState"), r::toString);
131 	}
132 
133 	@Test void a02_twoTablesKeepSeparateKeys() {
134 		var r = report();
135 		// Same value NAME under two distinct table keys must not clobber.  Each
136 		// scope lands under its own fully-qualified key; a page-level namespace is independent again.
137 		assertBean(r, "table1Cols,table2Cols,storeKeys,pageTab",
138 			"[name,status],[id],[juneau.pagestate.page.nav.tab,juneau.pagestate.table.releases.visibleCols,juneau.pagestate.table.users.visibleCols],setup");
139 	}
140 
141 	@Test void a03_jsonValueRoundTripsAndMissingKeyIsNull() {
142 		var r = report();
143 		assertMap((Map<?,?>) r.get("roundTrip"), "rows=compact", "wrap=false");
144 		assertNull(r.get("missingKey"), r::toString);
145 	}
146 
147 	@Test void a04_replacedStoreIsUsed() {
148 		var r = report();
149 		// A store installed via useStore() actually receives writes and serves reads...and the default
150 		// localStorage is bypassed once replaced.
151 		assertBean(r, "replacedStoreRecorded,replacedStoreValue,defaultStoreUntouchedByReplaced", "true,3,true");
152 	}
153 
154 	@Test void a05_blockedStoreIsSilentNoOp() {
155 		var r = report();
156 		// A store whose calls throw (private mode/quota) degrades to a no-op: no throw, and get reads back null.
157 		assertEquals(true, r.get("blockedNoThrow"), r::toString);
158 		assertNull(r.get("blockedGet"), r::toString);
159 	}
