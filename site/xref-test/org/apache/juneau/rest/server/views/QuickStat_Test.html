69  	@Test void a08_meter_toMap_shape() {
70  		var m = QuickStat.meter("inventory", "Inventory", 12, 40).text("12 of 40").statusTone(StatusTone.SUCCESS).toMap();
71  		assertEquals(List.of("id", "type", "label", "value", "max", "text", "tone"), new ArrayList<>(m.keySet()));

76  		assertEquals("success", m.get("tone"));

87  	@Test void a10_segmented_toMap_shape() {
88  		var m = QuickStat.segmented("orders", "Orders").segment("Placed", 3).segment("Approved", 5, StatusTone.INFO)
89  			.segment("Delivered", 9, StatusTone.SUCCESS).toMap();
90  		assertEquals("segmented", m.get("type"));
91  		assertFalse(m.containsKey("tone"));
92  		var segs = (List<?>)m.get("segments");
93  		assertEquals(3, segs.size());
94  		var s0 = (Map<?,?>)segs.get(0);
95  		assertEquals("Placed", s0.get("label"));
96  		assertEquals(3L, s0.get("count"));
97  		assertFalse(s0.containsKey("tone"));
98  		assertEquals("info", ((Map<?,?>)segs.get(1)).get("tone"));
99  	}

219 	@Test void a30_tone_and_statusTone_sameToken_lastWins() {
220 		assertEquals("info", QuickStat.of("a", "A", 1).tone(QuickStat.Tone.ERROR).statusTone(StatusTone.INFO).toMap().get("tone"));
221 		assertEquals("warning", QuickStat.of("a", "A", 1).statusTone(StatusTone.INFO).tone(QuickStat.Tone.WARNING).toMap().get("tone"));
222 		for (var t : QuickStat.Tone.values()) {
223 			var st = StatusTone.valueOf(t.name());
224 			assertEquals(QuickStat.of("a", "A", 1).tone(t).toMap().get("tone"), QuickStat.of("a", "A", 1).statusTone(st).toMap().get("tone"));
225 		}
