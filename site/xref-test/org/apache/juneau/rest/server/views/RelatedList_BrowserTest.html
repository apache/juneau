40  /**
41   * A nested related list (a {@code ViewTable} in a row's detail panel) in a real browser against a real JDK
42   * {@link HttpServer}: it pages, sorts and runs a row action, each as a request the server actually received.
43   *
44   * <p>
45   * Row 1 owns 30 events, so the nested table has a second page.  The prober pages forward, sorts the {@code what}
46   * column descending, and runs the {@code ack} row action from the nested row's menu; the server records each
47   * request and the assertions read those records, not just the DOM.
48   *
49   * <p>
50   * Disabled unless the {@value #GATE} system property is set, which only the module's opt-in {@code js-tests} Maven
51   * profile does: {@code mvn -Pjs-tests -f juneau-rest/juneau-rest-server-views/pom.xml test}.
52   */

54  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -f juneau-rest/juneau-rest-server-views/pom.xml test`")
55  class RelatedList_BrowserTest extends TestBase {

93  		var l = new ArrayList<Event>();

112 	@BeforeEach
113 	void startServer() throws IOException {
114 		server = HttpServer.create(new InetSocketAddress(InetAddress.getLoopbackAddress(), 0), 0);
115 		port = server.getAddress().getPort();
116 		var rootContext = InMemoryBeanQueryContext.create(Row.class).columns("id", "name").build();
117 		var eventContext = InMemoryBeanQueryContext.create(Event.class).columns("parentId", "what").build();
118 
119 		server.createContext("/query", ex -> {
120 			try {
121 				if (neq(ex.getRequestMethod(), "POST")) {
122 					ex.sendResponseHeaders(405, -1);
123 					return;
124 				}
125 				String body;
126 				try (var in = ex.getRequestBody()) {
127 					body = new String(in.readAllBytes(), UTF_8);
128 				}
129 				String json;
130 				try (var s = rootContext.getSession(rootRows())) {
131 					json = Json.of(DataTablesQuery.run(Json.to(body, DataTablesRequest.class), s));
132 				}
133 				writeJson(ex, json);
134 			} catch (Exception e) {
135 				sendFailure(ex, e);
136 			}
137 		});
138 		// A distinct "/query/" context (longest-prefix match) serves the row-detail GET envelope at /query/{id};
139 		// an empty-fields 200 is all expandDetailRow (juneau-views.js) requires to proceed to the nested table.
140 		server.createContext("/query/", ex -> writeJson(ex, "{\"contractVersion\":\"1\",\"fields\":{}}"));
141 		server.createContext("/nested", ex -> {
142 			try {
143 				if (neq(ex.getRequestMethod(), "POST")) {
144 					ex.sendResponseHeaders(405, -1);
145 					return;
146 				}
147 				var query = ex.getRequestURI().getRawQuery();
148 				nestedQueries.add(String.valueOf(query));
149 				var parentId = queryParam(query, "parentId");
150 				var scoped = allEvents().stream().filter(e -> String.valueOf(e.getParentId()).equals(parentId)).toList();
151 				String body;
152 				try (var in = ex.getRequestBody()) {
153 					body = new String(in.readAllBytes(), UTF_8);
154 				}
155 				nestedBodies.add(body);
156 				String json;
157 				try (var s = eventContext.getSession(scoped)) {
158 					json = Json.of(DataTablesQuery.run(Json.to(body, DataTablesRequest.class), s));
159 				}
160 				writeJson(ex, json);
161 			} catch (Exception e) {
162 				sendFailure(ex, e);
163 			}
164 		});
165 		server.createContext("/ack/", ex -> {
166 			try (var in = ex.getRequestBody()) {
167 				acks.add(ex.getRequestMethod() + " " + ex.getRequestURI().getPath() + " csrf="
168 					+ ex.getRequestHeaders().getFirst("X-CSRF-Token") + " body=" + new String(in.readAllBytes(), UTF_8));
169 			}
170 			writeJson(ex, "{\"outcome\":\"success\",\"message\":\"Acknowledged\"}");
171 		});
172 		server.createContext("/", ex -> {
173 			var bytes = fixturePage().getBytes(UTF_8);
174 			ex.getResponseHeaders().add("Content-Type", "text/html;charset=utf-8");
175 			ex.sendResponseHeaders(200, bytes.length);
176 			try (var os = ex.getResponseBody()) {
177 				os.write(bytes);
178 			}
179 		});
180 		server.start();
181 	}

188 	private static String queryParam(String query, String name) {
189 		if (query == null)
190 			return null;
191 		for (var pair : query.split("&")) {
192 			var i = pair.indexOf('=');
193 			if (i >= 0 && URLDecoder.decode(pair.substring(0, i), UTF_8).equals(name))
194 				return URLDecoder.decode(pair.substring(i + 1), UTF_8);
195 		}
196 		return null;
197 	}

199 	/**
200 	 * Answers a handler failure with a 500 carrying the stack trace, instead of throwing (which makes the JDK server
201 	 * drop the connection with no response) - so the prober's network diagnostics show why a draw failed.
202 	 */
203 	private static void sendFailure(HttpExchange ex, Exception e) throws IOException {
204 		var sw = new StringWriter();
205 		e.printStackTrace(new PrintWriter(sw));
206 		System.err.println(sw);
207 		var bytes = sw.toString().getBytes(UTF_8);
208 		ex.getResponseHeaders().add("Content-Type", "text/plain;charset=utf-8");
209 		ex.sendResponseHeaders(500, bytes.length);
210 		try (var os = ex.getResponseBody()) {
211 			os.write(bytes);
212 		}
213 	}

215 	private static void writeJson(HttpExchange ex, String json) throws IOException {
216 		var bytes = json.getBytes(UTF_8);
217 		ex.getResponseHeaders().add("Content-Type", "application/json");
218 		ex.sendResponseHeaders(200, bytes.length);
219 		try (var os = ex.getResponseBody()) {
220 			os.write(bytes);
221 		}
222 	}

224 	private static String resource(Class<?> anchor, String path) throws IOException {
225 		try (var in = anchor.getResourceAsStream(path)) {
226 			assertNotNull(in, () -> "missing classpath resource: " + path + " (relative to " + anchor.getName() + ")");
227 			return new String(in.readAllBytes(), UTF_8);
228 		}
229 	}

231 	private static String fixturePage() throws IOException {
232 		var nodeModules = Path.of(requiredProperty("juneau.jsTests.dir")).resolve("node_modules");
233 		var jquery = Files.readString(nodeModules.resolve("jquery/dist/jquery.min.js"));
234 		var dataTablesJs = Files.readString(nodeModules.resolve("datatables.net/js/dataTables.min.js"));
235 		var datatablesGlue = resource(DataTablesMixin.class, "juneau-datatables.js");
236 		var viewsJs = resource(ViewsMixin.class, ViewsMixin.VIEWS_JS_RESOURCE);
237 
238 		var rootMeta = "{\"contractVersion\":\"5\",\"id\":\"root\",\"dataMode\":\"server\",\"dataUrl\":\"/query\","
239 			+ "\"columns\":[{\"data\":\"id\",\"title\":\"ID\"},{\"data\":\"name\",\"title\":\"Name\"}]}";
240 		var nestedMeta = "{\"contractVersion\":\"5\",\"id\":\"events\",\"dataMode\":\"server\",\"dataUrl\":\"/nested\","
241 			+ "\"columns\":[{\"data\":\"what\",\"title\":\"What\"}],"
242 			+ "\"rowActions\":[{\"id\":\"ack\",\"label\":\"Acknowledge\",\"endpoint\":\"/ack/{what}\",\"method\":\"POST\","
243 			+ "\"onSuccess\":\"redraw\"}]}";
244 
245 		// The nested sidecar carries type="application/json" so the browser never executes it as script once the
246 		// row-detail template is cloned into the live document.
247 		return "<!DOCTYPE html><html><head><meta charset=\"utf-8\"></head><body>\n"
248 			+ "<table data-juneau-view=\"root\" id=\"root\" data-juneau-csrf=\"tok-9\"></table>\n"
249 			+ "<script type=\"application/json\" id=\"juneau-view:root\">" + rootMeta + "</script>\n"
250 			+ "<template data-juneau-row-detail=\"1\" data-juneau-detail-url=\"/query/{id}\">\n"
251 			+ "  <div class=\"juneau-view-detail-nested\" data-juneau-nested=\"1\" data-juneau-nested-contract=\"2\""
252 			+ " data-juneau-nested-scope-param=\"parentId\">\n"
253 			+ "    <table data-juneau-view=\"events\" data-juneau-csrf=\"tok-9\"></table>\n"
254 			+ "    <script type=\"application/json\" data-juneau-nested-meta=\"events\">" + nestedMeta + "</script>\n"
255 			+ "  </div>\n"
256 			+ "</template>\n"
257 			+ "<script>\n" + jquery + "\n</script>\n"
258 			+ "<script>\n" + dataTablesJs + "\n</script>\n"
259 			+ "<script>\n" + datatablesGlue + "\n</script>\n"
260 			+ "<script>\n" + resource(ViewsMixin.class, ViewsMixin.RENDERS_JS_RESOURCE) + "\n</script>\n"
261 			+ "<script>\n" + viewsJs + "\n</script>\n"
262 			+ "</body></html>";
263 	}

271 	private Map<?,?> runHarness() throws Exception {
272 		var dir = Path.of(requiredProperty("juneau.jsTests.dir"));
273 		var harness = Path.of(requiredProperty("juneau.jsTests.harness")).getParent().resolve("related-list.cjs");
274 		var baseUrl = "http://127.0.0.1:" + port + "/";
275 
276 		var cmd = List.of(System.getProperty("juneau.jsTests.node", "node"), harness.toString(), baseUrl);
277 		var stdout = dir.resolve("related-list-stdout.json");
278 		var stderr = dir.resolve("related-list-stderr.txt");
279 		var pb = new ProcessBuilder(cmd).redirectOutput(stdout.toFile()).redirectError(stderr.toFile());
280 		pb.environment().put("NODE_PATH", dir.resolve("node_modules").toString());
281 		pb.environment().put("PLAYWRIGHT_BROWSERS_PATH", requiredProperty("juneau.jsTests.browsers"));
282 
283 		var p = pb.start();
284 		if (! p.waitFor(3, TimeUnit.MINUTES)) {
285 			p.destroyForcibly();
286 			fail("prober did not finish within 3m; stderr:\n" + quietRead(stderr));
287 		}
288 		assertEquals(0, p.exitValue(), () -> "prober exited non-zero; stderr:\n" + quietRead(stderr));
289 		return Json.to(Files.readString(stdout), Map.class);
290 	}

292 	private static String quietRead(Path p) {
293 		try {
294 			return Files.readString(p);
295 		} catch (IOException e) {
296 			return "<unreadable: " + e + ">";
297 		}
298 	}
