39  	@Rest(mixins=ViewsMixin.class)
40  	public static class WithMixin extends BasicRestServlet {
41  		private static final long serialVersionUID = 1L;
42  		@RestGet(path="/items") public String items() { return "items"; }
43  	}
44  
45  	private static final MockRestClient c = MockRestClient.buildLax(WithMixin.class);
46  
47  	private static String viewsJs() throws Exception {
48  		return c.get(ViewsMixin.VIEWS_JS_PATH).run().assertStatus(200).getContent().asString();
49  	}
50  
51  	private static String fn(String body, String signature) {
52  		var start = body.indexOf(signature);
53  		assertTrue(start >= 0, () -> signature + " not found:\n" + body);
54  		var end = body.indexOf("\n\t}", start);
55  		return body.substring(start, end < 0 ? body.length() : end);
56  	}

251 	@Test void h01_appendDialogFormUsesCreateElementNeverInnerHtml() throws Exception {
252 		var body = viewsJs();
253 		// The row scaffolding (label + help/error siblings) is built via createElement + textContent, then each
254 		// control is delegated to paintFormControl.  The client never consumes form.template as markup.  The row
255 		// painter is its own function so a sectioned form paints byte-identical rows into a per-section pane; the
256 		// markup-free requirement covers the entry point, the row painter AND the sectioned painter.
257 		var row = fn(body, "function appendDialogFormRow(");
258 		assertTrue(row.contains("createElement(\"label\")"), row);
259 		assertTrue(row.contains("paintFormControl("), row);
260 		assertTrue(row.contains(".textContent"), row);
261 		for (var name : new String[]{
262 			"function appendDialogForm(", "function appendDialogFormRow(", "function appendSectionedDialogForm("
263 		}) {
264 			var append = fn(body, name);
265 			assertFalse(append.contains(".innerHTML"), () -> "FormDef paint must never use innerHTML:\n" + append);
266 			assertFalse(append.contains("insertAdjacentHTML"), append);
267 			assertFalse(append.contains("form.template"), () -> "client must not consume form.template as markup:\n" + append);
268 		}
269 		// The native controls themselves are created (never markup) in the per-control dispatcher, with prefills via
270 		// .value / .checked.
271 		var paint = fn(body, "function paintFormControl(");
272 		assertTrue(paint.contains("createElement(\"input\")"), paint);
273 		assertTrue(paint.contains("createElement(\"textarea\")"), paint);
274 		assertTrue(fn(body, "function buildSelectFormControl(").contains("createElement(\"select\")"), paint);
275 		assertTrue(paint.contains(".value"), paint);
276 		assertTrue(fn(body, "function buildCheckboxFormControl(").contains(".checked"), paint);
277 		assertFalse(paint.contains(".innerHTML"), () -> "paintFormControl must never use innerHTML:\n" + paint);
278 		assertFalse(paint.contains("insertAdjacentHTML"), paint);
279 	}

311 		assertTrue(row.contains("isTypedFormInputType(type)"), row);
312 		assertTrue(fn(body, "function appendSectionedDialogForm(").contains("appendDialogFormRow(pane, dialog,"),
