29  /**
30   * <b>The populator name is never interpolated into a selector or into
31   * markup</b>.
32   *
33   * <p>
34   * The populator name is the one piece of a region descriptor that arrives as a <i>string off a DOM attribute</i>
35   * ({@code data-juneau-region-populate}) and is then used to look something up. That combination is what makes it
36   * worth a test of its own: a name concatenated into a {@code querySelector} argument is a selector-injection, and a
37   * name concatenated into markup is an XSS. It must only ever be a <b>key</b>.
38   *
39   * <p>
40   * Server-side the name is additionally allowlisted at startup (already covered by
41   * {@code RegionDef_Test.b03}), so on a page Juneau emitted the name is a known-good token. This test covers the
42   * client side, which cannot assume that: the runtime reads the attribute off whatever DOM it is handed.
43   *
44   * <h5 class='section'>The three places the name legitimately goes, established by reading every use</h5>
45   * <ol>
46   * 	<li>{@code registry[name]} &mdash; an own-property lookup, guarded by {@code Object.hasOwn}.
47   * 	<li>An <b>error message string</b>, when no populator is registered under it. This is the interesting one: the
48   * 		name genuinely does reach the DOM here, and the test asserts it arrives as <b>text</b>. The path is
49   * 		{@code resolvePopulator} &rarr; {@code setRegionState(region,"error",message)} &rarr;
50   * 		{@code renderAsyncStatus} &rarr; {@code status.textContent = message}.
51   * 	<li>A {@code console.error} argument, which is not a DOM sink.
52   * </ol>
53   *
54   * <p>
55   * The absence of any HTML sink anywhere in {@code juneau-regions.js} is enforced separately and more broadly by
56   * {@link ShippedJs_SinkScan_Test}, which this item added the region runtime to; that
57   * scan is the universal, and this class pins the specific flow.
58   */
59  class Regions_PopulatorNameSafety_Test extends TestBase {

66  	}

75  	/** Strips {@code //} and block comments so prose about the name is neither a hit nor a miss. */
76  	private static String stripComments(String s) {
77  		var out = new StringBuilder(s.length());
78  		var i = 0;
79  		while (i < s.length()) {
80  			var c = s.charAt(i);
81  			if (c == '/' && i + 1 < s.length() && s.charAt(i + 1) == '/') {
82  				while (i < s.length() && s.charAt(i) != '\n') i++;
83  			} else if (c == '/' && i + 1 < s.length() && s.charAt(i + 1) == '*') {
84  				var end = s.indexOf("*/", i + 2);
85  				i = end < 0 ? s.length() : end + 2;
86  				out.append('\n');
87  			} else {
88  				out.append(c);
89  				i++;
90  			}
91  		}
92  		return out.toString();
93  	}

120 	/**
121 	 * Every {@code querySelector}/{@code querySelectorAll}/{@code closest}/{@code matches} argument in the region
122 	 * runtime must be a <b>constant</b> selector &mdash; a string literal, or a literal built from the module's own
123 	 * attribute-name constants. What must never appear is the populator name, in any of its spellings.
124 	 */
125 	@Test void b01_noSelectorArgumentMentionsThePopulatorName() throws Exception {
126 		var js = stripComments(regionsJs());
127 		var p = Pattern.compile("(?:querySelectorAll|querySelector|closest|matches)\\s*\\(([^;]{0,400}?)\\)\\s*[;.,)\\]]");
128 		var m = p.matcher(js);
129 		var offenders = new ArrayList<String>();
130 		var checked = 0;
131 		while (m.find()) {
132 			checked++;
133 			var arg = m.group(1);
134 			if (arg.contains("populateName") || arg.contains("populateFn")
135 					|| arg.contains("REGION_POPULATE_ATTR") && arg.contains("+"))
136 				offenders.add(arg.trim());
137 		}
138 		assertTrue(checked > 0, "the selector scan matched nothing, so it proves nothing");
139 		assertTrue(offenders.isEmpty(),
140 			() -> "the populator name reached a selector argument - a name off a DOM attribute concatenated into a "
141 				+ "selector is a selector-injection.  The name must only ever be a registry KEY:\n  "
142 				+ String.join("\n  ", offenders));
143 	}

145 	@Test void b02_theNameIsResolvedByOwnPropertyLookupOnly() throws Exception {
146 		var js = stripComments(regionsJs());
147 		// Object.hasOwn is what stops a name like "toString" / "constructor" / "valueOf" resolving to something off
148 		// Object.prototype and being CALLED as a populator.  A bare `registry[name]` truthiness check would.
149 		assertTrue(js.contains("Object.hasOwn(registry, name) ? registry[name] : null"),
150 			"populator resolution must stay an own-property lookup: a bare registry[name] check would let a name "
151 				+ "like 'toString' or 'constructor' resolve off Object.prototype and get invoked as a populator");
152 	}

170 	@Test void c02_theStatusRendererPaintsMessagesAsTextNotMarkup() throws Exception {
171 		var js = stripComments(viewsJs());
172 		var from = js.substring(js.indexOf("function renderAsyncStatus"));
173 		final var body = from.substring(0, from.indexOf("\n\tfunction ", 1));
174 		assertTrue(body.contains("status.textContent = message"),
175 			() -> "renderAsyncStatus must paint the error message with textContent.  This is the single sink the "
176 				+ "populator name reaches, so if it ever becomes innerHTML the name becomes an XSS vector:\n" + body);
177 		assertFalse(body.contains("innerHTML"),
178 			() -> "renderAsyncStatus must contain no innerHTML at all:\n" + body);
179 		assertTrue(body.contains("document.createTextNode(message"),
180 			() -> "the loading branch must also append the message as a TEXT NODE:\n" + body);
181 	}
