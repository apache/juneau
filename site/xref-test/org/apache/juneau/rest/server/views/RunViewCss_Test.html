46  	@BeforeAll static void load() throws IOException {
47  		try (var in = ViewsMixin.class.getResourceAsStream(ViewsMixin.VIEWS_CSS_RESOURCE)) {
48  			assertNotNull(in, "missing " + ViewsMixin.VIEWS_CSS_RESOURCE);
49  			css = new String(in.readAllBytes(), UTF_8).replaceAll("(?s)/\\*.*?\\*/", "");
50  		}
51  		var start = css.indexOf(MARKER);
52  		assertTrue(start > 0, "run-view section missing");
53  		before = css.substring(0, start);
54  		section = css.substring(start);
55  		rules = parse(section);
56  		shared = parse(before);
57  	}

59  	/** Selector (each comma-separated selector, whitespace-normalised) to its declarations, later rules overriding. */
60  	private static Map<String,Map<String,String>> parse(String text) {
61  		var out = new LinkedHashMap<String,Map<String,String>>();
62  		var m = Pattern.compile("([^{}]++)\\{([^{}]*+)\\}").matcher(text);
63  		while (m.find()) {
64  			var decls = new LinkedHashMap<String,String>();
65  			for (var d : m.group(2).split(";")) {
66  				var i = d.indexOf(':');
67  				if (i > 0)
68  					decls.put(d.substring(0, i).trim(), d.substring(i + 1).trim().replaceAll("\\s+", " "));
69  			}
70  			for (var s : m.group(1).split(","))
71  				out.computeIfAbsent(s.trim().replaceAll("\\s+", " "), k -> new LinkedHashMap<>()).putAll(decls);
72  		}
73  		return out;
74  	}
75  
76  	private static Map<String,String> rule(String selector) {
77  		var r = rules.get(selector);
78  		assertNotNull(r, () -> "no rule for '" + selector + "'");
79  		return r;
80  	}
81  
82  	private static void assertShared(String selector, String property, String value) {
83  		var r = shared.get(selector);
84  		assertNotNull(r, () -> "no shared rule for '" + selector + "'");
85  		assertEquals(value, r.get(property), () -> selector + " { " + property + " } in " + r);
86  	}
87  
88  	private static void assertDecl(String selector, String property, String value) {
89  		assertEquals(value, rule(selector).get(property), () -> selector + " { " + property + " } in " + rule(selector));
90  	}
