103 	private static final String CSP_META = "<meta http-equiv=\"Content-Security-Policy\" content=\"style-src 'self'\">";
104 
105 	private static Map<String,Map<String,Object>> report;
106 
107 	@BeforeAll
108 	static void run() throws Exception {
109 		var views = viewsPack();
110 		var light = contractPage(PAGE, "", "", views);
111 		var dark = contractPage(PAGE, "", "", views, "", true);
112 		var csp = light.replace("<meta charset=\"utf-8\">", "<meta charset=\"utf-8\">" + CSP_META);
113 		assertNotEquals(light, csp, "the CSP meta tag must be injected");
114 		var ready = "[data-juneau-stat=\"g1\"]";
115 		report = ConsoleBrowserFixture.create("quickstats")
116 			.page("light", ORIGIN + "/light", light).waitFor(ready).colorScheme("light").eval(PROBE)
117 			.page("dark", ORIGIN + "/dark", dark).waitFor(ready).colorScheme("dark").eval(PROBE)
118 			.page("csp", ORIGIN + "/csp", csp).waitFor(ready).colorScheme("light").eval(PROBE)
119 			.page("forced", ORIGIN + "/forced", light).waitFor(ready).colorScheme("light").forcedColors("active").eval(PROBE)
120 			.run();
121 	}

128 		return ((Number)o).doubleValue();

138 	private static void assertFillRatio(Map<String,Object> p) {
139 		var tolerance = 1.0 / num(p.get("trackWidth"));
140 		assertEquals(0.30, num(p.get("fillRatio")), tolerance + 0.005, p::toString);
141 	}
142 
143 	private static void assertOrderedWidths(Map<String,Object> p) {
144 		var w = list(p.get("segWidths"));
145 		assertSize(p::toString, 3, w);
146 		assertTrue(num(w.get(0)) < num(w.get(1)) && num(w.get(1)) < num(w.get(2)), () -> "3:5:9 widths: " + w);
147 	}

157 	@Test void a02_light_paint() {
158 		var p = p("light");
159 		assertEquals(p.get("tokenBg"), p.get("trackBg"), p::toString);
160 		assertEquals(p.get("tokenBg"), p.get("segbarBg"), p::toString);
161 		assertFillRatio(p);
162 		var bgs = list(p.get("segBgs"));
163 		assertSize(3, bgs);
164 		for (var bg : bgs) {
165 			assertNotEquals("rgba(0, 0, 0, 0)", bg, p::toString);
166 			assertNotEquals(p.get("segbarBg"), bg, p::toString);
167 		}
168 		assertSize(() -> "segment colours must be pairwise distinct: " + bgs, 3, new HashSet<>(bgs));
169 		assertOrderedWidths(p);
170 		assertTrue(String.valueOf(p.get("stripe")).contains("repeating-linear-gradient"), p::toString);
171 		assertTrue(String.valueOf(p.get("warnStripe")).contains("repeating-linear-gradient"), p::toString);
172 		assertTrue(String.valueOf(p.get("segStripe")).contains("repeating-linear-gradient"), p::toString);
173 		assertEquals("none", p.get("okStripe"), p::toString);
174 		assertEquals("none", p.get("okSegStripe"), p::toString);
175 		assertNotEquals("none", p.get("overShadow"), () -> "the over-limit cue must paint: " + p);
176 		assertEquals("solid", p.get("errOutline"), p::toString);
177 		assertEquals("none", p.get("okOutline"), p::toString);
178 		var swatches = list(p.get("swatchBgs"));
179 		assertSize(p::toString, 3, swatches);
180 		for (var bg : swatches)
181 			assertNotEquals("rgba(0, 0, 0, 0)", bg, () -> "swatch must be filled: " + p);
182 	}

184 	@Test void a03_dark_paint() {
185 		var p = p("dark");
186 		var l = p("light");
187 		assertEquals(p.get("tokenBg"), p.get("trackBg"), p::toString);
188 		assertEquals(p.get("tokenBg"), p.get("segbarBg"), p::toString);
189 		assertNotEquals(l.get("trackBg"), p.get("trackBg"), () -> "the dark layer must be live; light=" + l.get("trackBg") + " dark=" + p.get("trackBg"));
190 		assertNotEquals("rgba(0, 0, 0, 0)", p.get("fillBg"), p::toString);
191 		assertNotEquals(p.get("trackBg"), p.get("fillBg"), p::toString);
192 		for (var bg : list(p.get("segBgs"))) {
193 			assertNotEquals("rgba(0, 0, 0, 0)", bg, p::toString);
194 			assertNotEquals(p.get("segbarBg"), bg, p::toString);
195 		}

205 	@Test void a05_forcedColors() {
206 		var p = p("forced");
207 		assertEquals(p.get("canvasText"), p.get("fillBg"), p::toString);
208 		for (var bg : list(p.get("segBgs")))
209 			assertEquals(p.get("canvasText"), bg, p::toString);
210 		assertNotEquals("0px", p.get("trackBorder"), p::toString);
211 		assertEquals(p.get("canvasBg"), p.get("trackBg"), () -> "the track must be on Canvas: " + p);
212 		assertEquals(p.get("canvasBg"), p.get("segbarBg"), () -> "the segment bar must be on Canvas: " + p);
213 		for (var bg : list(p.get("swatchBgs")))
214 			assertEquals(p.get("canvasText"), bg, p::toString);
215 		assertEquals("none", p.get("adjust"), p::toString);
216 		assertNotEquals("none", p.get("overShadow"), () -> "the over-limit cue must survive forced colours: " + p);
217 		// Warning and error keep a stripe in system colours, so they differ from a plain success fill.
218 		for (var key : List.of("warnStripe", "stripe", "segStripe")) {
219 			var stripe = String.valueOf(p.get(key));
220 			assertTrue(stripe.contains("repeating-linear-gradient"), () -> key + ": " + p);
221 			assertTrue(stripe.contains(String.valueOf(p.get("canvasBg"))), () -> key + " must be drawn in the Canvas system colour: " + p);
222 		}
223 		assertEquals("none", p.get("okStripe"), p::toString);
224 		assertNotEquals(p.get("okStripe"), p.get("warnStripe"), p::toString);
225 		assertEquals("none", p.get("okSegStripe"), p::toString);
226 	}
