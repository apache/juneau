27  /**
28   * The console shell on the page bus: {@code card:<id>}
29   * lifecycle and ownership, implicit claims, {@code cmd:<id>} ops, declared roles and {@code params}, buffered
30   * {@code ctx.publish}/{@code ctx.subscribe}, the error painter, and E-JS-46.  Drives {@code console-shell-bus.cjs}
31   * against the real {@code juneau-bus.js} and {@code juneau-console.js}; gated on {@code node} being on {@code PATH}.
32   *
33   * <p>
34   * Lives in this module, not console-ui, for the same reason as {@code ConsoleCards_Shell_Test}: every Node harness
35   * that runs the shell lives here, and {@link BusHarness} is package-private.
36   */
37  class ConsoleBus_Shell_Test extends TestBase {

39  	private static final String CONSOLE_JS = "/org/apache/juneau/console/juneau-console.js";
40  
41  	private static Map<String,Object> report;
42  
43  	@BeforeAll
44  	static void runHarness() {
45  		report = BusHarness.run("console-shell-bus.cjs", ViewsMixin.BUS_JS_RESOURCE, CONSOLE_JS);
46  	}
47  
48  	@SuppressWarnings("unchecked")
49  	private static Map<String,Object> scenario(String name) {
50  		assumeTrue(report != null, "node not on PATH");
51  		assertNull(report.get("harnessError"), () -> String.valueOf(report.get("harnessError")));
52  		var s = (Map<String,Object>) report.get(name);
53  		assertNotNull(s, name);
54  		return s;
55  	}
56  
57  	@SuppressWarnings("unchecked")
58  	private static Map<String,Object> map(Map<String,Object> m, String key) {
59  		return (Map<String,Object>) m.get(key);
60  	}

62  	@Test void a00_sourceShape() throws Exception {
63  		var src = BusHarness.source(CONSOLE_JS);
64  		var validate = src.indexOf("paintWiringProblems(doc, bus.wiring.validate(contract, cardTypeTable()));");
65  		var anchor = src.indexOf("// ---- bus: bridges attach here ----");
66  		var wire = src.indexOf("wireCards(doc, bus, contract);");
67  		assertTrue(validate > 0 && validate < anchor && anchor < wire, "validate, then the bridge anchor, then wireCards");
68  		assertEquals(anchor, src.lastIndexOf("// ---- bus: bridges attach here ----"), "exactly one anchor");
69  		assertFalse(src.contains("extras.bus"), "handlers declare roles/ops/implicit directly");
70  	}
