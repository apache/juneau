34  	private static final Map<String,Object> REF = Map.of("jobId", "j1", "streamUrl", "/jobs/j1/stream", "cancelUrl", "/jobs/j1/cancel");
35  	private static final String NOT_FOUND = "Job not found; it may have finished or the server restarted.";
36  	private static final String ATTACH_FAILED = "Could not attach to the job's progress stream; the server may be busy. Reload the page to try again.";
37  	private static final String INTERRUPTED = "The progress stream was interrupted; the job may still be running. Reload the page to reattach.";

39  	private static Map<?,?> report() {
40  		var r = NodeHarness.report("job-log-card.cjs", ConsoleChromeMixin.CONSOLE_JS_RESOURCE, ConsoleChromeMixin.JOBLOG_JS_RESOURCE);
41  		assumeTrue(r != null, "node (or job-log-card.cjs) not available - JS layer skipped");
42  		return r;
43  	}
44  
45  	private static Map<?,?> part(String key) {
46  		var s = (Map<?,?>)report().get(key);
47  		assertNotNull(s, () -> "section '" + key + "' missing from the harness output");
48  		assertNull(s.get("crash"), () -> "section '" + key + "' crashed: " + s.get("crash"));
49  		return s;
50  	}
51  
52  	private static Map<?,?> state(String key) {
53  		return (Map<?,?>)part(key).get("state");
54  	}
