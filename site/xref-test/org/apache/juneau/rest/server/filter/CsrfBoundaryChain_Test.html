34  /**
35   * End-to-end behaviour of {@link RequestBoundaryFilter} applying a {@link CsrfBoundary}, judged by what reaches the
36   * {@link FilterChain}.
37   *
38   * <p>
39   * The chain is a Mockito mock standing in for everything behind the filter, so "never reaches the resource" is
40   * observable as "the chain was never touched".  A mock REST client cannot show that: it dispatches straight to the REST
41   * context and never runs a servlet filter, so a test built on it passes with no filter installed.
42   *
43   * <p>
44   * Requests are built with {@link CsrfBoundary_Test}'s fixture, which stubs {@code getHeaders(name)} alongside
45   * {@code getHeader(name)}; without that every request would look like it had no {@code Host} at all.
46   *
47   * @since 10.0.0
48   */
49  class CsrfBoundaryChain_Test extends TestBase {
50  
51  	private static final byte[] KEY = new byte[32];
52  	static {
53  		for (var i = 0; i < KEY.length; i++)
54  			KEY[i] = (byte)(0x30 + i);
55  	}
56  
57  	private static final CsrfBoundary B = CsrfBoundary.create().tokens(CsrfTokens.perSession().key(KEY)).build();
58  
59  	private static final String HOST = "app.example:8080";
60  	private static final String ORIGIN = "http://app.example:8080";
61  	private static final String EVIL = "http://evil.example";
62  	private static final String JSON = "application/json";
63  
64  	/** Builds the request, keeping the attributes the filter sets on it so the chain can read them back. */
65  	private static HttpServletRequest build(R r) {
66  		var req = r.build();
67  		var attrs = new HashMap<String,Object>();
68  		doAnswer(x -> attrs.put(x.getArgument(0), x.getArgument(1))).when(req).setAttribute(anyString(), any());
69  		when(req.getAttribute(anyString())).thenAnswer(x -> attrs.get(x.getArgument(0)));
70  		return req;
71  	}
72  
73  	/** A response that keeps the headers it is given the way a container does, and swallows the body. */
74  	@SuppressWarnings("resource") // The stub output stream is returned from the mocked response and holds nothing to close.
75  	private static HttpServletResponse response(Map<String,String> headers) throws IOException {
76  		var res = mock(HttpServletResponse.class);
77  		doAnswer(x -> headers.put(x.getArgument(0), x.getArgument(1))).when(res).setHeader(anyString(), anyString());
78  		doAnswer(x -> { headers.clear(); return null; }).when(res).reset();
79  		when(res.getOutputStream()).thenReturn(new ServletOutputStream() {
80  			@Override public void write(int b) { /* The body is not under test here. */ }
81  			@Override public boolean isReady() { return true; }
82  			@Override public void setWriteListener(WriteListener l) { /* Not used by the filter. */ }
83  		});
84  		return res;
85  	}
86  
87  	/** Runs the filter over the request; returns the chain so the caller can see whether it was reached. */
88  	private static FilterChain run(HttpServletRequest req) throws Exception {
89  		var chain = mock(FilterChain.class);
90  		new RequestBoundaryFilter(B).doFilter(req, response(new TreeMap<>(String.CASE_INSENSITIVE_ORDER)), chain);
91  		return chain;
92  	}
93  
94  	/** A masked token for a page served to a request in the given session, as the filter would publish it. */
95  	private static String pageToken(String session) throws Exception {
96  		var req = build(new R().method("GET").h("Host", HOST).session(session));
97  		var chain = run(req);
98  		verify(chain).doFilter(eq(req), any());
99  		return (String)req.getAttribute(RequestBoundaryFilter.TOKEN_ATTRIBUTE);
100 	}
101 
102 	private static R tokenOf(R r, String token) {
103 		return r.h(B.csrfHeader(), token);
104 	}

110 	@Test void a01_rejectedRequestNeverReachesChain() throws Exception {
111 		var token = pageToken("s1");
112 
113 		var refused = List.of(
114 			tokenOf(r("POST", "http", HOST, EVIL, "Content-Type", JSON), token),        // another site's page
115 			r("POST", "http", HOST, ORIGIN, "Content-Type", JSON),                       // the own page, no token
116 			r("GET", "http", HOST, null).query("method=DELETE")                         // a write disguised as a read
117 		);
118 		for (var r : refused)
119 			verifyNoInteractions(run(build(r)));
120 
121 		var allowed = List.of(
122 			tokenOf(r("POST", "http", HOST, ORIGIN, "Content-Type", JSON), token),
123 			tokenOf(r("POST", "http", HOST, ORIGIN, "Content-Type", JSON), pageToken("s1")),
124 			tokenOf(r("GET", "http", HOST, ORIGIN).query("method=DELETE"), token)
125 		);
126 		for (var r : allowed) {
127 			var req = build(r);
128 			verify(run(req), times(1)).doFilter(eq(req), any());
129 		}
130 	}

132 	@Test void a02_rejectionCarriesNoStoreAndRequestBoundaryHeader() throws Exception {
133 		var headers = new TreeMap<String,String>(String.CASE_INSENSITIVE_ORDER);
134 		var res = response(headers);
135 		var chain = mock(FilterChain.class);
136 		new RequestBoundaryFilter(B).doFilter(build(r("POST", "http", HOST, EVIL, "Content-Type", JSON)), res, chain);
137 		verifyNoInteractions(chain);
138 		verify(res).setStatus(403);
139 		assertEquals("ORIGIN_MISMATCH", headers.get(RequestBoundaryFilter.REJECTION_HEADER));
140 		assertEquals("no-store", headers.get("Cache-Control"));
141 	}

143 	@Test void a03_tokenPublishedOnAllowedSafeRequest() throws Exception {
144 		var req = build(r("GET", "http", HOST, null));
145 		var seen = new HashMap<String,Object>();
146 		var chain = mock(FilterChain.class);
147 		doAnswer(x -> {
148 			var q = (HttpServletRequest)x.getArgument(0);
149 			for (var a : List.of(RequestBoundaryFilter.TOKEN_ATTRIBUTE, RequestBoundaryFilter.TOKEN_SUPPLIER_ATTRIBUTE,
150 					RequestBoundaryFilter.HEADER_ATTRIBUTE))
151 				seen.put(a, q.getAttribute(a));
152 			return null;
153 		}).when(chain).doFilter(any(), any());
154 		new RequestBoundaryFilter(B).doFilter(req, response(new HashMap<>()), chain);
155 
156 		var token = assertInstanceOf(String.class, seen.get(RequestBoundaryFilter.TOKEN_ATTRIBUTE));
157 		assertTrue(token.matches("[0-9a-f]{128}"), token);
158 		var supplier = assertInstanceOf(Supplier.class, seen.get(RequestBoundaryFilter.TOKEN_SUPPLIER_ATTRIBUTE));
159 		var supplied = assertInstanceOf(String.class, supplier.get());
160 		assertTrue(supplied.matches("[0-9a-f]{128}"), supplied);
161 		assertEquals(B.csrfHeader(), seen.get(RequestBoundaryFilter.HEADER_ATTRIBUTE));
162 	}

164 	@Test void a05_sessionMintingObserved() throws Exception {
165 		// Only getSession(true) is observable here; the cookie it leads to is written by a real container.
166 		var navigate = build(r("GET", "http", HOST, null, "Sec-Fetch-Mode", "navigate").noSession());
167 		run(navigate);
168 		verify(navigate).getSession(true);
169 
170 		var html = build(r("GET", "http", HOST, null, "Accept", "text/html,application/xhtml+xml").noSession());
171 		run(html);
172 		verify(html).getSession(true);
173 
174 		var any = build(r("GET", "http", HOST, null, "Accept", "*/*").noSession());
175 		verify(run(any)).doFilter(eq(any), any());
176 		verify(any, never()).getSession(true);
177 	}

179 	@Test void a06_perSessionTokenWorksEndToEnd() throws Exception {
180 		var mine = pageToken("s1");
181 		var theirs = pageToken("s2");
182 
183 		var ok = build(tokenOf(r("POST", "http", HOST, ORIGIN, "Content-Type", JSON).session("s1"), mine));
184 		verify(run(ok)).doFilter(eq(ok), any());
185 
186 		var headers = new TreeMap<String,String>(String.CASE_INSENSITIVE_ORDER);
187 		var chain = mock(FilterChain.class);
188 		new RequestBoundaryFilter(B).doFilter(
189 			build(tokenOf(r("POST", "http", HOST, ORIGIN, "Content-Type", JSON).session("s1"), theirs)), response(headers), chain);
190 		verifyNoInteractions(chain);
191 		assertEquals("CSRF_TOKEN_MISMATCH", headers.get(RequestBoundaryFilter.REJECTION_HEADER));
192 	}

194 	@Test void a07_mutatedTokenOnSameSessionFails() throws Exception {
195 		var token = pageToken("s1");
196 		var last = token.charAt(token.length() - 1);
197 		var mutated = token.substring(0, token.length() - 1) + (last == '0' ? '1' : '0');
198 
199 		var headers = new TreeMap<String,String>(String.CASE_INSENSITIVE_ORDER);
200 		var chain = mock(FilterChain.class);
201 		new RequestBoundaryFilter(B).doFilter(
202 			build(tokenOf(r("POST", "http", HOST, ORIGIN, "Content-Type", JSON).session("s1"), mutated)), response(headers), chain);
203 		verifyNoInteractions(chain);
204 		assertEquals("CSRF_TOKEN_MISMATCH", headers.get(RequestBoundaryFilter.REJECTION_HEADER));
205 
206 		var ok = build(tokenOf(r("POST", "http", HOST, ORIGIN, "Content-Type", JSON).session("s1"), token));
207 		verify(run(ok)).doFilter(eq(ok), any());
208 	}
