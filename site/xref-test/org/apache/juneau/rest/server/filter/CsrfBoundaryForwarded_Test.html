53  	/** A same-origin JSON write as seen by a backend behind a proxy that rewrites {@code Host}. */
54  	private static R post(CsrfBoundary b, String origin) {
55  		return new R().method("POST").scheme("http").h("Host", "backend:8080").h("Origin", origin)
56  			.h("Content-Type", "application/json").h(b.csrfHeader(), b.tokenFor(new R().build()));
57  	}
58  
59  	private static R viaXf(CsrfBoundary b, String origin, String host, String proto) {
60  		return post(b, origin).h("X-Forwarded-Host", host).h("X-Forwarded-Proto", proto);
61  	}
62  
63  	private static void allowed(CsrfBoundary b, R r) {
64  		var x = b.check(r.build());
65  		assertTrue(x.isAllowed(), () -> "expected allowed, got: " + x);
66  	}
67  
68  	private static void refused(CsrfBoundary b, R r, BoundaryReason reason) {
69  		var x = b.check(r.build());
70  		assertFalse(x.isAllowed(), () -> "expected " + reason + ", got allowed");
71  		assertEquals(reason, x.reason(), x::toString);
72  		assertEquals(reason == HOST_NOT_ALLOWED ? 421 : 403, x.status(), x::toString);
73  	}
74  
75  	private static Map<String,List<String>> map(String...nameValue) {
76  		var m = new LinkedHashMap<String,List<String>>();
77  		for (var i = 0; i < nameValue.length; i += 2)
78  			m.computeIfAbsent(nameValue[i], k -> new ArrayList<>()).add(nameValue[i + 1]);
79  		return m;
80  	}
81  
82  	private static BoundaryResult hs(CsrfBoundary b, String scheme, String host, String...headers) {
83  		return b.checkHandshake(scheme, new HandshakeHosts(host == null ? List.of() : List.of(host), false), map(headers));
84  	}
85  
86  	private static void hsAllowed(CsrfBoundary b, String scheme, String host, String...headers) {
87  		var x = hs(b, scheme, host, headers);
88  		assertTrue(x.isAllowed(), () -> "expected allowed, got: " + x);
89  	}
90  
91  	private static void hsRefused(CsrfBoundary b, BoundaryReason reason, String scheme, String host, String...headers) {
92  		var x = hs(b, scheme, host, headers);
93  		assertFalse(x.isAllowed(), () -> "expected " + reason + ", got allowed");
94  		assertEquals(reason, x.reason(), x::toString);
95  	}

149 		var b = xf().build();
150 		// Default n=1 takes the rightmost entry; the client's prefix is ignored.
151 		allowed(b, viaXf(b, PUBLIC, "evil.example, app.example", "http, https"));
152 		allowed(b, post(b, PUBLIC).h("X-Forwarded-Host", "evil.example").h("X-Forwarded-Host", "app.example")
153 			.h("X-Forwarded-Proto", "http, https"));
154 		// Alignment: proto is taken at the same index as host.
155 		allowed(b, viaXf(b, "https://b", "a, b", "http, https"));
156 		refused(b, viaXf(b, PUBLIC, "a, app.example", "https"), FORWARDED_HOPS_MISMATCH);

295 	@Test void c03_hostHeaderFingerprintLeftOfSelected() {
296 		var b = xf().build();
297 		// Host equals the client-writable left entry: the container trusted a hop it must not have.
298 		refused(b, new R().method("POST").scheme("http").h("Host", "left.example").h("Origin", PUBLIC)
299 			.h("Content-Type", "application/json").h(b.csrfHeader(), b.tokenFor(new R().build()))
300 			.h("X-Forwarded-Host", "left.example, app.example").h("X-Forwarded-Proto", "http, https"),
301 			FORWARDED_CONTAINER_MISMATCH);
302 		// The transport name is replaced.
303 		allowed(b, viaXf(b, PUBLIC, "left.example, app.example", "http, https"));
304 		// A match to the right of the selected entry (an inner hop or the transport) is replaced, never refused.
305 		var two = xf().trustedHops(2).build();
306 		allowed(two, viaXf(two, PUBLIC, "app.example, backend", "https, http"));
307 		// Availability-only edge: a client can get its own request refused.
308 		refused(b, viaXf(b, PUBLIC, "backend, app.example", "http, https").remove("Host").h("Host", "backend"),
309 			FORWARDED_CONTAINER_MISMATCH);
310 		// Same host, different port: the TLS-terminating transport, replaced.
311 		allowed(b, post(b, PUBLIC).remove("Host").h("Host", "app.example:8080").h("X-Forwarded-Host", "app.example")
312 			.h("X-Forwarded-Proto", "https"));
313 	}
