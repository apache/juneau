38  /**
39   * Tests for {@link RequestBoundaryFilter} applying a {@link CsrfBoundary}: what an allowed request publishes for the
40   * page renderer, when a session is created so the page can carry a session-scoped token, and what a refusal looks
41   * like.
42   *
43   * <p>
44   * Every test drives {@link RequestBoundaryFilter#doFilter(ServletRequest, ServletResponse, FilterChain)} itself, with
45   * Mockito servlet mocks and a mocked {@link FilterChain}.  A mock REST client would not do: it dispatches straight to
46   * the REST context and never runs a servlet filter, so a test built on it passes with no filter installed.
47   *
48   * <p>
49   * Requests are built with {@link CsrfBoundary_Test}'s fixture, which stubs {@code getHeaders(name)} alongside
50   * {@code getHeader(name)}.  A request built with {@link R#noSession()} answers {@code getSession(true)} with a new
51   * session, so a test can observe whether one was created.
52   *
53   * @since 10.0.0
54   */
55  @SuppressWarnings({
56  	"java:S5976" // Each minting rule is its own named test so a failure names the rule it broke.
57  })
58  class CsrfBoundaryFilter_Test extends TestBase {

86  	/** A same-origin JSON write carrying a valid token for {@code b} in the fixture's session. */
87  	private static R write(CsrfBoundary b) {
88  		var r = r("POST", "http", HOST, ORIGIN, "Content-Type", JSON);
89  		return r.h(b.csrfHeader(), b.tokenFor(new R().build()));

92  	/** Builds the request, keeping the attributes the filter sets on it so a test can read them back. */
93  	private static HttpServletRequest build(R r) {
94  		var req = r.build();
95  		var attrs = new HashMap<String,Object>();
96  		doAnswer(x -> attrs.put(x.getArgument(0), x.getArgument(1))).when(req).setAttribute(anyString(), any());
97  		when(req.getAttribute(anyString())).thenAnswer(x -> attrs.get(x.getArgument(0)));
98  		return req;
99  	}

101 	/** Captures what the filter wrote to the response body. */
102 	private static final class CapturingOutputStream extends ServletOutputStream {
103 		final ByteArrayOutputStream baos = new ByteArrayOutputStream();
104 		@Override public void write(int b) { baos.write(b); }
105 		@Override public boolean isReady() { return true; }
106 		@Override public void setWriteListener(WriteListener l) { /* not used by this filter */ }
107 		String captured() { return baos.toString(StandardCharsets.UTF_8); }
108 	}

110 	/** A response paired with the body the filter wrote to it and the headers it ended up with. */
111 	private record Capture(HttpServletResponse res, CapturingOutputStream out, Map<String,List<String>> headers) {
112 		String body() { return out.captured(); }
113 		List<String> header(String name) { return headers.getOrDefault(name, List.of()); }
114 	}

116 	/**
117 	 * A response that keeps its headers the way a container does: {@code setHeader} replaces, {@code addHeader}
118 	 * appends, {@code reset} clears.  Its writer is unusable, so a refusal must be written as bytes.
119 	 */
120 	@SuppressWarnings({
121 		"resource" // The capture wraps an in-memory buffer and the mock's getOutputStream() acquires nothing; there is no resource to release.
122 	})
123 	private static Capture capturing() throws IOException {
124 		var out = new CapturingOutputStream();
125 		var r = mock(HttpServletResponse.class);
126 		when(r.getOutputStream()).thenReturn(out);
127 		when(r.getWriter()).thenThrow(new IllegalStateException("No writer: a refusal must not depend on one."));
128 		var headers = new TreeMap<String,List<String>>(String.CASE_INSENSITIVE_ORDER);
129 		doAnswer(x -> headers.put(x.getArgument(0), new ArrayList<>(List.of((String)x.getArgument(1))))).when(r).setHeader(anyString(), anyString());
130 		doAnswer(x -> headers.computeIfAbsent(x.getArgument(0), k -> new ArrayList<>()).add(x.getArgument(1))).when(r).addHeader(anyString(), anyString());
131 		doAnswer(x -> { headers.clear(); return null; }).when(r).reset();
132 		return new Capture(r, out, headers);
133 	}
134 
135 	/** Runs the filter over a request and returns the response it wrote. */
136 	private static Capture run(RequestBoundaryFilter f, HttpServletRequest req, FilterChain chain) throws Exception {
137 		var c = capturing();
138 		f.doFilter(req, c.res(), chain);
139 		return c;
140 	}

142 	private static boolean verifies(String masked, HttpServletRequest req) {
143 		return TOKENS.verify(masked, TOKENS.expected(req, false));
144 	}

146 	private static void assertMinted(CsrfBoundary b, R r) throws Exception {
147 		var req = build(r);
148 		var chain = mock(FilterChain.class);
149 		run(new RequestBoundaryFilter(b), req, chain);
150 		verify(chain).doFilter(eq(req), any());
151 		verify(req).getSession(true);
152 	}

154 	private static void assertNotMinted(CsrfBoundary b, R r) throws Exception {
155 		var req = build(r);
156 		run(new RequestBoundaryFilter(b), req, mock(FilterChain.class));
157 		verify(req, never()).getSession(true);
158 	}

261 	/** A boundary and a request it refuses for exactly the given reason. */
262 	private record Refusal(CsrfBoundary boundary, R request, int status) {}
263 
264 	private static Refusal refusedFor(BoundaryReason reason) {
265 		var b = boundary().build();
266 		return switch (reason) {
267 			case HOST_MISSING -> new Refusal(b, r("POST", "http", null, ORIGIN, "Content-Type", JSON), 400);
268 			case HOST_MALFORMED -> new Refusal(b, r("POST", "http", "app.example:8080/x", ORIGIN, "Content-Type", JSON), 400);
269 			case HOST_NOT_ALLOWED -> new Refusal(boundary().allowedHosts("other.example").build(), r("GET", "http", HOST, null), 421);
270 			case ORIGIN_MISSING -> new Refusal(boundary().nonBrowser(NonBrowser.REJECT).build(), r("DELETE", "http", HOST, null), 403);
271 			case ORIGIN_NULL -> new Refusal(b, r("POST", "http", HOST, "null", "Content-Type", JSON), 403);
272 			case ORIGIN_MISMATCH -> new Refusal(b, r("POST", "http", HOST, EVIL, "Content-Type", JSON), 403);
273 			case FETCH_SITE_NOT_SAME_ORIGIN -> new Refusal(b, write(b).h("Sec-Fetch-Site", "cross-site"), 403);
274 			case SIMPLE_REQUEST_WITHOUT_ORIGIN -> new Refusal(b,
275 				r("POST", "http", HOST, null, "Content-Type", "application/x-www-form-urlencoded"), 403);
276 			case CONTENT_TYPE_NOT_JSON -> {
277 				var j = boundary().requireJson(true).build();
278 				yield new Refusal(j, r("POST", "http", HOST, ORIGIN, "Content-Type", "text/plain")
279 					.h(j.csrfHeader(), j.tokenFor(new R().build())), 415);
280 			}
281 			case CSRF_TOKEN_MISSING -> new Refusal(b, r("POST", "http", HOST, ORIGIN, "Content-Type", JSON), 403);
282 			case CSRF_TOKEN_MISMATCH -> new Refusal(b, r("POST", "http", HOST, ORIGIN, "Content-Type", JSON)
283 				.h(b.csrfHeader(), "0".repeat(128)), 403);
284 			case FORWARDED_HOST_MISSING -> forwarded(x -> x, r("POST", "http", HOST, ORIGIN, "Content-Type", JSON));
285 			case FORWARDED_PROTO_MISSING -> forwarded(x -> x,
286 				r("POST", "http", HOST, ORIGIN, "Content-Type", JSON).h("X-Forwarded-Host", "app.example"));
287 			case FORWARDED_HOST_MALFORMED -> forwarded(x -> x,
288 				r("POST", "http", HOST, ORIGIN, "Content-Type", JSON).h("X-Forwarded-Host", "a@b").h("X-Forwarded-Proto", "http"));
289 			case FORWARDED_PROTO_MALFORMED -> forwarded(x -> x,
290 				r("POST", "http", HOST, ORIGIN, "Content-Type", JSON).h("X-Forwarded-Host", "app.example").h("X-Forwarded-Proto", "ws"));
291 			case FORWARDED_HOPS_INSUFFICIENT -> forwarded(x -> x.trustedHops(2),
292 				r("POST", "http", HOST, ORIGIN, "Content-Type", JSON).h("X-Forwarded-Host", "app.example").h("X-Forwarded-Proto", "http"));
293 			case FORWARDED_HOPS_MISMATCH -> forwarded(x -> x,
294 				r("POST", "http", HOST, ORIGIN, "Content-Type", JSON).h("X-Forwarded-Host", "a, b").h("X-Forwarded-Proto", "http"));
295 			case FORWARDED_CONTAINER_MISMATCH -> forwarded(x -> x.originSource(OriginSource.CONTAINER),
296 				r("POST", "http", HOST, ORIGIN, "Content-Type", JSON).h("X-Forwarded-Host", "app.example").h("X-Forwarded-Proto", "https"));
297 			case FORWARDED_PEER_NOT_TRUSTED -> forwarded(x -> x.trustedProxies("10.0.0.0/8"),
298 				r("POST", "http", HOST, ORIGIN, "Content-Type", JSON).h("X-Forwarded-Host", "app.example").h("X-Forwarded-Proto", "https"));
299 			default -> throw new IllegalArgumentException("CsrfBoundary does not emit " + reason);
300 		};
301 	}

303 	/** A trusting boundary (after {@code tweak}) and the request it refuses; every forwarded refusal is a 403. */
304 	private static Refusal forwarded(java.util.function.UnaryOperator<CsrfBoundary.Builder> tweak, R request) {
305 		var builder = boundary().trustedForwarding(CsrfBoundary.ForwardedSource.X_FORWARDED);
306 		return new Refusal(tweak.apply(builder).build(), request, 403);
307 	}

309 	@ParameterizedTest
310 	@EnumSource(value=BoundaryReason.class, names={
311 		"HOST_MISSING", "HOST_MALFORMED", "HOST_NOT_ALLOWED", "ORIGIN_MISSING", "ORIGIN_NULL", "ORIGIN_MISMATCH",
312 		"FETCH_SITE_NOT_SAME_ORIGIN", "SIMPLE_REQUEST_WITHOUT_ORIGIN", "CONTENT_TYPE_NOT_JSON", "CSRF_TOKEN_MISSING",
313 		"CSRF_TOKEN_MISMATCH", "FORWARDED_HOST_MISSING", "FORWARDED_PROTO_MISSING", "FORWARDED_HOST_MALFORMED",
314 		"FORWARDED_PROTO_MALFORMED", "FORWARDED_HOPS_INSUFFICIENT", "FORWARDED_HOPS_MISMATCH", "FORWARDED_CONTAINER_MISMATCH",
315 		"FORWARDED_PEER_NOT_TRUSTED"
316 	})
317 	void a10_rejectionNoStoreAndNoCacheOnAllReasons(BoundaryReason reason) throws Exception {
318 		var x = refusedFor(reason);
319 		var req = build(x.request().h("Sec-Fetch-Mode", "navigate"));
320 		var c = run(new RequestBoundaryFilter(x.boundary()), req, mock(FilterChain.class));
321 		verify(c.res()).setStatus(x.status());
322 		assertEquals(List.of(reason.name()), c.header(RequestBoundaryFilter.REJECTION_HEADER));
323 		assertEquals(List.of("no-store"), c.header("Cache-Control"));
324 		assertEquals(List.of("close"), c.header("Connection"), "an unread request body must not leave the connection reusable");
325 		for (var h : List.of("Expires", "ETag", "Last-Modified", "Set-Cookie"))
326 			assertEquals(List.of(), c.header(h), h);
327 		verify(req, never()).getSession(true);
328 	}
