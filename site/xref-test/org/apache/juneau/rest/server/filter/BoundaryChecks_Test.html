203 	@Test void p04_headerValues() {
204 		var r = mock(HttpServletRequest.class);
205 		when(r.getHeaders("Origin")).thenAnswer(i -> Collections.enumeration(List.of("a", "b")));
206 		when(r.getHeader("Origin")).thenReturn("a");
207 		assertEquals(List.of("a", "b"), BoundaryChecks.headerValues(r, "Origin"));
208 		// A container that will not enumerate a header returns null; the single value is then all there is.
209 		when(r.getHeaders("Host")).thenReturn(null);
210 		when(r.getHeader("Host")).thenReturn("h");
211 		assertEquals(List.of("h"), BoundaryChecks.headerValues(r, "Host"));
212 		assertEquals(List.of(), BoundaryChecks.headerValues(r, "Absent"));
213 	}

224 	// p06) hostValues / isSynthesisedHost: HTTP/2 :authority stands in for a missing Host
225 	//-----------------------------------------------------------------------------------------------------------------
226 
227 	private static HttpServletRequest h(String protocol, String name, int port, String...hosts) {
228 		var r = mock(HttpServletRequest.class);
229 		when(r.getProtocol()).thenReturn(protocol);
230 		when(r.getServerName()).thenReturn(name);
231 		when(r.getServerPort()).thenReturn(port);
232 		when(r.getHeaders("Host")).thenAnswer(i -> Collections.enumeration(List.of(hosts)));
233 		when(r.getHeader("Host")).thenReturn(hosts.length == 0 ? null : hosts[0]);
234 		return r;
235 	}

237 	@Test void p06_http2WithoutHostSynthesises() {
238 		var r = h("HTTP/2.0", "app.example", 8080);
239 		assertEquals(List.of("app.example:8080"), BoundaryChecks.hostValues(r));
240 		assertTrue(BoundaryChecks.isSynthesisedHost(r));
241 	}
242 
243 	@Test void p07_otherProtocolsDoNotSynthesise() {
244 		for (var protocol : new String[]{"HTTP/1.1", "HTTP/1.0", "HTTP/2", "http/2.0", "HTTP/3.0", "", null}) {
245 			var r = h(protocol, "app.example", 8080);
246 			assertEquals(List.of(), BoundaryChecks.hostValues(r), String.valueOf(protocol));
247 			assertFalse(BoundaryChecks.isSynthesisedHost(r), String.valueOf(protocol));
248 		}
249 	}

251 	@Test void p08_realHostWinsAndStaysRepeated() {
252 		var r = h("HTTP/2.0", "app.example", 8080, "other.example");
253 		assertEquals(List.of("other.example"), BoundaryChecks.hostValues(r));
254 		assertFalse(BoundaryChecks.isSynthesisedHost(r));
255 		var two = h("HTTP/2.0", "app.example", 8080, "a", "b");
256 		assertEquals(List.of("a", "b"), BoundaryChecks.hostValues(two));
257 		assertFalse(BoundaryChecks.isSynthesisedHost(two));
258 	}

260 	@Test void p09_portsAndNames() {
261 		assertEquals(List.of("app.example"), BoundaryChecks.hostValues(h("HTTP/2.0", "app.example", 0)));
262 		assertEquals(List.of("app.example"), BoundaryChecks.hostValues(h("HTTP/2.0", "app.example", -1)));
263 		assertEquals(List.of("[::1]:80"), BoundaryChecks.hostValues(h("HTTP/2.0", "::1", 80)));
264 		assertEquals(List.of("[::1]:80"), BoundaryChecks.hostValues(h("HTTP/2.0", "[::1]", 80)));
265 		assertEquals(List.of(), BoundaryChecks.hostValues(h("HTTP/2.0", null, 80)));
266 		assertEquals(List.of(), BoundaryChecks.hostValues(h("HTTP/2.0", "", 80)));
267 		assertEquals(List.of(), BoundaryChecks.hostValues(h("HTTP/2.0", "  ", 80)));
268 		assertFalse(BoundaryChecks.isSynthesisedHost(h("HTTP/2.0", null, 80)));
