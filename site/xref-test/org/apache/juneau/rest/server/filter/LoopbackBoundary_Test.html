30  /**
31   * Tests for {@link LoopbackBoundary}.
32   *
33   * <p>
34   * The negative cases carry the weight here.  A boundary with only positive tests passes just as happily when a
35   * check has been accidentally removed, so every rule below is asserted from both directions: the legitimate
36   * request must succeed, <i>and</i> a request violating exactly that one rule must be refused with the specific
37   * reason and status that rule is supposed to produce.
38   *
39   * @since 10.0.0
40   */
41  class LoopbackBoundary_Test extends TestBase {
42  
43  	private static final String AUTHORITY = "127.0.0.1:8790";
44  	private static final String ORIGIN = "http://127.0.0.1:8790";
45  	private static final SynchronizerToken TOKEN = SynchronizerToken.of("the-real-token");
46  
47  	private static final LoopbackBoundary BOUNDARY = LoopbackBoundary.create()
48  		.authority(AUTHORITY)
49  		.token(TOKEN)
50  		.build();
51  
52  	/**
53  	 * A fully legitimate write request: correct {@code Host}, correct {@code Origin}, same-origin fetch metadata,
54  	 * JSON content type, and this server's token.  Each negative test below is this request with exactly one
55  	 * header spoiled, so a failure isolates the rule that broke.
56  	 */
57  	private static Map<String,String> goodWriteHeaders() {
58  		var m = new LinkedHashMap<String,String>();
59  		m.put("Host", AUTHORITY);
60  		m.put("Origin", ORIGIN);
61  		m.put("Sec-Fetch-Site", "same-origin");
62  		m.put("X-Csrf-Token", TOKEN.value());
63  		return m;
64  	}
65  
66  	private static HttpServletRequest req(String method, Map<String,String> headers) {
67  		return req(method, headers, "application/json");
68  	}
69  
70  	private static HttpServletRequest req(String method, Map<String,String> headers, String contentType) {
71  		var r = mock(HttpServletRequest.class);
72  		when(r.getMethod()).thenReturn(method);
73  		when(r.getContentType()).thenReturn(contentType);
74  		headers.forEach((k, v) -> when(r.getHeader(k)).thenReturn(v));
75  		return r;
76  	}
77  
78  	/** The legitimate write request, optionally with one header replaced or (on a null value) removed. */
79  	private static HttpServletRequest write(String spoiledHeader, String value) {
80  		var h = goodWriteHeaders();
81  		if (spoiledHeader != null) {
82  			if (value == null)
83  				h.remove(spoiledHeader);
84  			else
85  				h.put(spoiledHeader, value);
86  		}
87  		return req("POST", h);
88  	}
89  
90  	private static void assertRejected(BoundaryResult r, BoundaryReason expectedReason, int expectedStatus) {
91  		assertFalse(r.isAllowed(), () -> "expected a rejection, got: " + r);
92  		assertEquals(expectedReason, r.reason());
93  		assertEquals(expectedStatus, r.status());
94  		assertNotNull(r.message(), "a rejection must carry a message; a silent refusal is not diagnosable");
95  		assertFalse(r.message().isBlank());
96  	}
