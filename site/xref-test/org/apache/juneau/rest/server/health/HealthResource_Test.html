19  import static java.util.EnumSet.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.*;
24  import org.apache.juneau.commons.inject.*;
25  import org.apache.juneau.rest.mock.classic.*;
26  import org.apache.juneau.rest.server.*;
27  import org.apache.juneau.rest.server.servlet.*;
28  import org.junit.jupiter.api.*;

30  /**
31   * Validates the {@link HealthResource} child-resource flavor mounted via {@code @Rest(children=...)}
32   * under a host.
33   *
34   * <p>
35   * The child mounts at the subtree {@code /health} (ops {@code /healthz}, {@code /readyz},
36   * {@code /livez}) and delegates to a shared {@link HealthAggregator} worker &mdash; the same logic
37   * the {@link HealthServlet} servlet and {@link HealthMixin} mixin flavors use. Indicators are
38   * resolved from the child's bean store. This mirrors {@link HealthServlet_Test} but exercises the
39   * routed-child deployment. Cases:
40   * <ul>
41   * 	<li>{@code /health/healthz} returns {@code 503} when any component is down.
42   * 	<li>{@code /health/livez} and {@code /health/readyz} filter indicators by probe type.
43   * </ul>
44   *
45   * @since 10.0.0
46   */
47  class HealthResource_Test extends TestBase {
48  
49  	/**
50  	 * Child subclass registering indicator beans into the child's bean store.
51  	 *
52  	 * <p>
53  	 * {@link HealthResource} now extends {@link BasicRestResource}, so the
54  	 * {@link HealthAggregator.HealthResponse} bean is serialized out of the box via the
55  	 * {@code BasicUniversalConfig} serializer set &mdash; no explicit {@code serializers=...} is needed
56  	 * (parity with the servlet flavor's {@link BasicRestServlet} base).
57  	 */
58  	@Rest(path="/health")
59  	public static class HealthChild extends HealthResource {
60  		@Bean(name="db")
61  		public HealthIndicator dbIndicator() {
62  			return new HealthIndicator() {
63  				@Override public Health check() { return Health.up("db").detail("ok", true).build(); }
64  				@Override public EnumSet<HealthProbe> probes() { return of(HealthProbe.LIVE); }
65  			};
66  		}
67  
68  		@Bean(name="cache")
69  		public HealthIndicator cacheIndicator() {
70  			return new HealthIndicator() {
71  				@Override public Health check() { return Health.down("cache", new IllegalStateException("offline")).build(); }
72  				@Override public EnumSet<HealthProbe> probes() { return of(HealthProbe.READY); }
73  			};
74  		}
75  	}
76  
77  	@Rest(children={HealthChild.class})
78  	public static class A extends BasicRestServlet {
79  		private static final long serialVersionUID = 1L;
80  	}
81  
82  	private static final MockRestClient c = MockRestClient.buildLax(A.class);
83  
84  	@Test void a01_healthzReturns503WhenAnyComponentDown() throws Exception {
85  		c.get("/health/healthz").accept("application/json").run()
86  			.assertStatus(503)
87  			.assertContent().asString().isContains("\"status\":\"DOWN\"");
88  	}
89  
90  	@Test void a02_livezAndReadyzFilterByProbeType() throws Exception {
91  		c.get("/health/livez").accept("application/json").run().assertStatus(200);
92  		c.get("/health/readyz").accept("application/json").run().assertStatus(503);
93  	}
94  }
