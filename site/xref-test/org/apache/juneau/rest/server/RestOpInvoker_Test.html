42  /**
43   * JaCoCo coverage tests for {@link RestOpInvoker}.
44   *
45   * <p>
46   * Targets the dispatch and error-classification logic in {@code invoke(...)} / {@code invokeOp(...)}:
47   * <ul>
48   *   <li>Happy path with observability on (default) and off ({@code @Rest(observability="false")}).</li>
49   *   <li>Per-op observability opt-out via {@code @RestOp(observability="false")}.</li>
50   *   <li>Error classification: {@link BasicHttpException} subclass re-throw, generic {@link RuntimeException}
51   *       wrap to 500, checked exception wrap to 500.</li>
52   *   <li>Parameter resolution failure paths: {@link BasicHttpException} re-throw, other exception wrapped
53   *       as {@link BadRequest}.</li>
54   *   <li>{@link CompletionStage} return values that defer observability close to {@code whenComplete}.</li>
55   *   <li>{@code Boolean} {@code Debug} request-attribute branches (TRUE / FALSE / null).</li>
56   *   <li>{@link MetricsRecorder} / {@link TracerHook} happy and error events through the public hooks.</li>
57   *   <li>Per-method execution stats — call counter increments on success and on exception.</li>
58   * </ul>
59   *
60   * <p>
61   * Coverage gaps intentionally not exercised here:
62   * <ul>
63   *   <li>The Java-21+ virtual-thread dispatch block in {@link RestOpInvoker#invokeOp(RestOpSession)}
64   *       (lines that submit to {@code vtExec} and unwrap {@link ExecutionException} /
65   *       {@link InterruptedException}) — covered by {@link VirtualThreadDispatch_Test}, requires Java 21
66   *       at runtime and is therefore unreachable on the project's Java 17 source/test runtime.</li>
67   *   <li>The {@code IllegalAccessException} / {@code IllegalArgumentException} catch in
68   *       {@link MethodInvoker#invoke(Object, Object...)} — the framework constructs a matching
69   *       {@code RestOpInvoker} for every {@code @RestOp} method, so reflective access is always permitted
70   *       and the args array always matches the method signature. Forcing this path requires a degenerate
71   *       reflective setup outside the public API.</li>
72   *   <li>The defensive catch in {@link RestOpInvoker#resolveUriTemplate(RestOpSession)} (lines 277-280) —
73   *       requires {@code RestOpContext.getPathPattern()} to throw, which the framework's path-matcher
74   *       wiring rules out for a successfully-built operation.</li>
75   *   <li>The {@code (cause instanceof Error)} re-throw in the VT executor branch — same Java-21+ block
76   *       as above.</li>
77   * </ul>
78   */

284 	/** Custom resolver — throws a generic RuntimeException (NOT a BasicHttpException) so the parameter-
285 	 *  resolution catch in {@link RestOpInvoker#invoke(RestOpSession)} routes through the
286 	 *  generic-exception branch rather than the BasicHttpException re-throw branch. */
287 	public static class FailingArg implements org.apache.juneau.rest.server.arg.RestOpArg {
288 		public static FailingArg create(org.apache.juneau.commons.reflect.ParameterInfo pi) {
289 			if (pi.isType(FailingMarker.class))
290 				return new FailingArg();
291 			return null;
292 		}
293 		@Override
294 		public Object resolve(RestOpSession opSession) {
295 			throw new IllegalStateException("simulated arg-resolver failure");
296 		}
297 	}
