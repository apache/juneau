45  	private static org.apache.juneau.bean.swagger.Swagger getSwagger(Object resource) throws Exception {
46  		var rc = new RestContext(new RestContext.Args(resource.getClass(), null, null, () -> resource, "", null, null, null, RestContext.ContextKind.ROOT));
47  		var roc = new RestOpContext(BasicSwaggerProviderSession_Coverage_Test.class.getMethod("testMethod"), rc);
48  		var call = RestSession.create(rc).resource(resource).req(new MockServletRequest()).res(new MockServletResponse()).build();
49  		var req = roc.createRequest(call);
50  		var ip = rc.getSwaggerProvider();
51  		return ip.getSwagger(rc, req.getLocale());
52  	}
