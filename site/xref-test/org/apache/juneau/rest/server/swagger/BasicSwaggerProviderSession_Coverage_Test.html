42  	private static org.apache.juneau.bean.swagger.Swagger getSwagger(Object resource) throws Exception {
43  		var rc = new RestContext(new RestContext.Args(resource.getClass(), null, null, () -> resource, "", null, null, null, RestContext.ContextKind.ROOT));
44  		var roc = new RestOpContext(BasicSwaggerProviderSession_Coverage_Test.class.getMethod("testMethod"), rc);
45  		var call = RestSession.create(rc).resource(resource).req(new MockServletRequest()).res(new MockServletResponse()).build();
46  		var req = roc.createRequest(call);
47  		var ip = rc.getSwaggerProvider();
48  		return ip.getSwagger(rc, req.getLocale());
49  	}
