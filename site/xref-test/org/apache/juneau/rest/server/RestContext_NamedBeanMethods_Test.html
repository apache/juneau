30  /**
31   * Tests that several {@link Bean @Bean} methods of the same return type and different names on one {@link Rest @Rest}
32   * class each register their own result, that overrides run once, that non-public methods are honored, and that
33   * named framework-typed methods alias the framework's single per-context instance.
34   *
35   * @since 10.0.0
36   */
37  @SuppressWarnings({
38  	"java:S1144", // Unused private methods are the fixtures: they are invoked reflectively via @Bean.
39  	"resource", // getBeanStore() returns the context's own (already-owned) BeanStore; the test doesn't own it.
40  	"unused" // Private @Bean fixture methods and their injected parameters (e.g. u(Missing m)) are only invoked reflectively
41  })
42  class RestContext_NamedBeanMethods_Test extends org.apache.juneau.TestBase {
43  
44  	static RestContext.Args argsOf(Class<?> resourceClass, java.util.function.Supplier<?> supplier) {
45  		return new RestContext.Args(resourceClass, null, null, supplier, null, null, null, null, null, null);
46  	}
47  
48  	/** Invocation counts by fixture method label. */
49  	static final Map<String,Integer> CALLS = new HashMap<>();
50  
51  	static void called(String label) { CALLS.merge(label, 1, Integer::sum); }
52  
53  	static int calls(String label) { return CALLS.getOrDefault(label, 0); }
54  
55  	@BeforeEach void reset() { CALLS.clear(); }
56  
57  	public static class Widget {
58  		private final String name;
59  		public Widget(String name) { this.name = name; }
60  		public String getName() { return name; }
61  	}
62  
63  	public static class Missing {}
64  
65  	private static <T> T bean(RestContext ctx, Class<T> type, String name) {
66  		return ctx.getBeanStore().getBean(type, name).orElse(null);
67  	}
68  
69  	private static List<String> keys(RestContext ctx, Class<?> type) {
70  		return new ArrayList<>(new TreeSet<>(ctx.getBeanStore().getBeansOfType(type).keySet()));
71  	}

74  	// a - each named method registers its own result
75  	//-----------------------------------------------------------------------------------------------------------
76  
77  	@Rest
78  	public static class Fix_Named {
79  		@Bean(name="a") public Widget a() { called("a"); return new Widget("A"); }
80  		@Bean(name="b") public Widget b() { called("b"); return new Widget("B"); }
81  		@Bean(name="c") public Widget c() { called("c"); return new Widget("C"); }
82  		@Bean public Widget plain() { called("plain"); return new Widget("P"); }
83  		@Bean("v") public Widget v() { called("v"); return new Widget("V"); }
84  	}
85  
86  	private static RestContext named() throws Exception {
87  		return new RestContext(argsOf(Fix_Named.class, Fix_Named::new));
88  	}
89  
90  	@Test void a01_eachNameResolvesToItsOwnMethod() throws Exception {
91  		var ctx = named();
92  		assertBean(bean(ctx, Widget.class, "a"), "name", "A");
93  		assertBean(bean(ctx, Widget.class, "b"), "name", "B");
94  		assertBean(bean(ctx, Widget.class, "c"), "name", "C");
95  		assertBean(ctx.getBeanStore().getBean(Widget.class).orElse(null), "name", "P");
96  		assertList(keys(ctx, Widget.class), "", "a", "b", "c", "v");
97  	}
98  
99  	@Test void a02_eachMethodInvokedOnce() throws Exception {
100 		named();
101 		assertEquals(List.of(1, 1, 1, 1, 1), List.of(calls("a"), calls("b"), calls("c"), calls("plain"), calls("v")));
102 	}
103 
104 	@Test void a03_valueAliasNamesTheBean() throws Exception {
105 		assertBean(bean(named(), Widget.class, "v"), "name", "V");
106 	}

242 	@Rest(path="/enc")
243 	public static class Fix_EncChild {}
244 
245 	@Rest(children={Fix_EncChild.class})
246 	public static class Fix_NamedFramework {
247 		@Bean(name="encoders")
248 		public EncoderSet encoders() { called("encoders"); return EncoderSet.create(new BasicBeanStore()).build(); }
249 	}
250 
251 	@Test void a12_namedFrameworkTypeInvokedOnceAndAliased() throws Exception {
252 		var ctx = new RestContext(argsOf(Fix_NamedFramework.class, Fix_NamedFramework::new));
253 		ctx.getEncoders();
254 		var child = ctx.getRestChildren().asMap().get("enc");
255 		child.getEncoders();
256 		assertEquals(1, calls("encoders"));
257 		assertSame(ctx.getBeanStore().getBean(EncoderSet.class).orElse(null), bean(ctx, EncoderSet.class, "encoders"));
258 		assertNull(bean(child, EncoderSet.class, "encoders"));
259 	}
