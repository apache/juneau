58  	//------------------------------------------------------------------------------------------------------------------
59  	// a01: HappyPath — annotation enables lazy; routing works immediately; first request triggers materialization; // NOSONAR
60  	//      subsequent requests reuse the context.
61  	//------------------------------------------------------------------------------------------------------------------
62  
63  	@Rest(path = "/root", children = { AlphaResource.class }, lazyChildren = "true")
64  	public static class A_Parent extends BasicRestServletGroup {
65  		private static final long serialVersionUID = 1L;
66  	}
67  
68  	@Test
69  	void a01_happyPath_lazyChildMaterializesOnFirstRequest() throws Exception {
70  		var parent = new A_Parent();
71  		var client = MockRestClient.createLax(parent).build();
72  		var rc = parent.getContext();
73  
74  		// lazyChildren=true is honoured.
75  		assertTrue(rc.isLazyChildren());
76  
77  		// Before first request: lazy entry is registered but NOT materialized.
78  		var entries = rc.getRestChildren().getLazyEntries();
79  		assertFalse(entries.isEmpty(), "Lazy entry should be registered at startup");
80  		var entry = entries.get("alpha");
81  		assertNotNull(entry, "Lazy entry key should be 'alpha'");
82  		assertFalse(entry.isMaterialized(), "Should not be materialized before first request");
83  
84  		// First request materializes the child.
85  		client.get("/alpha/ping").run().assertStatus(200).assertContent("alpha-pong");
86  
87  		// After first request: materialized.
88  		assertTrue(entry.isMaterialized(), "Should be materialized after first request");
89  
90  		// Subsequent requests continue to succeed (reuse materialized context).
91  		client.get("/alpha/ping").run().assertStatus(200).assertContent("alpha-pong");
92  		client.get("/alpha/ping").run().assertStatus(200).assertContent("alpha-pong");
93  	}

96  	// a02: Concurrent first-requests — only one materialization runs.
97  	//------------------------------------------------------------------------------------------------------------------
98  
99  	static final AtomicInteger A02_INIT_COUNT = new AtomicInteger(0);
100 
101 	@Rest(path = "/counted")
102 	public static class CountedResource {
103 		public CountedResource() {
104 			A02_INIT_COUNT.incrementAndGet();
105 		}
106 
107 		@RestGet("/ping")
108 		public String ping() {
109 			return "counted-pong";
110 		}
111 	}
112 
113 	@Rest(path = "/root", children = { CountedResource.class }, lazyChildren = "true")
114 	public static class B_Parent extends BasicRestServletGroup {
115 		private static final long serialVersionUID = 1L;
116 	}
117 
118 	@Test
119 	void a02_concurrent_exactlyOneMaterialization() throws Exception {
120 		A02_INIT_COUNT.set(0);
121 		var client = MockRestClient.createLax(new B_Parent()).build();
122 
123 		int concurrency = 8;
124 		var barrier = new CyclicBarrier(concurrency);
125 		var latch = new CountDownLatch(concurrency);
126 		var errors = new AtomicReference<Throwable>();
127 		var pool = Executors.newFixedThreadPool(concurrency);
128 
129 		for (int i = 0; i < concurrency; i++) {
130 			pool.submit(() -> {
131 				try {
132 					barrier.await();  // all threads start simultaneously
133 					client.get("/counted/ping").run().assertStatus(200).assertContent("counted-pong");
134 				} catch (Throwable t) {
135 					errors.compareAndSet(null, t);
136 				} finally {
137 					latch.countDown();
138 				}
139 			});
140 		}
141 
142 		latch.await(10, TimeUnit.SECONDS);
143 		pool.shutdown();
144 		assertNull(errors.get(), "Concurrent requests must not throw");
145 		assertEquals(1, A02_INIT_COUNT.get(), "CountedResource constructor must be called exactly once");
146 	}

173 		A03_INIT_COUNT.set(0);

195 	@Rest(path = "/root", children = { AlphaResource.class })
196 	public static class D_EagerParent extends BasicRestServletGroup {
197 		private static final long serialVersionUID = 1L;
198 	}
199 
200 	@Test
201 	void a04_builderOverride_lazyChildInitTrueOverridesEagerDefault() {
202 		// Use Builder.lazyChildInit(true) to force lazy on a resource that has no annotation.
203 		// The Builder is package-private, so this test is in the org.apache.juneau.rest.server.server package.
204 		var d = new D_EagerParent();
205 		MockRestClient.createLax(d).build();
206 		var rc = d.getContext();
207 
208 		// Default is eager; verify annotation-free behavior.
209 		assertFalse(rc.isLazyChildren(), "No annotation => eager (default)");
210 
211 		// Separately verify that the builder knob can be set and read.
212 		var args = new RestContext.Args(D_EagerParent.class, null, null, () -> d, "", null, null, null, RestContext.ContextKind.ROOT);
213 		var builder = new RestContext.Builder(args);
214 		assertNull(builder.lazyChildInit, "Default builder knob should be null");
215 		builder.lazyChildInit(true);
216 		assertTrue(builder.lazyChildInit, "Builder knob must be set to true after lazyChildInit(true)");
217 		builder.lazyChildInit(false);
218 		assertFalse(builder.lazyChildInit, "Builder knob must be set to false after lazyChildInit(false)");
219 	}

222 	// a05: Lifecycle shutdown — materialized lazy child sees destroy(); never-invoked child is skipped.
223 	//------------------------------------------------------------------------------------------------------------------
224 
225 	static final AtomicInteger A05_DESTROY_COUNT = new AtomicInteger(0);
226 
227 	@Rest(path = "/lifecycle-child")
228 	public static class LifecycleChildResource {
229 		@RestGet("/ping")
230 		public String ping() {
231 			return "lifecycle-pong";
232 		}
233 
234 		@RestDestroy
235 		public void onDestroy() {
236 			A05_DESTROY_COUNT.incrementAndGet();
237 		}
238 	}
239 
240 	@Rest(path = "/never-called")
241 	public static class NeverCalledLifecycle {
242 		@RestGet("/ping")
243 		public String ping() {
244 			return "never";
245 		}
246 
247 		@RestDestroy
248 		public void onDestroy() {
249 			// should NOT be called
250 			A05_DESTROY_COUNT.incrementAndGet();
251 		}
252 	}
253 
254 	@Rest(path = "/root", children = { LifecycleChildResource.class, NeverCalledLifecycle.class }, lazyChildren = "true")
255 	public static class E_Parent extends BasicRestServletGroup {
256 		private static final long serialVersionUID = 1L;
257 	}
258 
259 	@Test
260 	void a05_lifecycleShutdown_materializedChildSeesDestroy_neverInvokedSkipped() throws Exception {
261 		A05_DESTROY_COUNT.set(0);
262 		var parent = new E_Parent();
263 		var client = MockRestClient.createLax(parent).build();
264 
265 		// Trigger materialization of lifecycle-child only.
266 		client.get("/lifecycle-child/ping").run().assertStatus(200);
267 
268 		var rc = parent.getContext();
269 		var lazyEntries = rc.getRestChildren().getLazyEntries();
270 		assertTrue(lazyEntries.get("lifecycle-child").isMaterialized(), "lifecycle-child should be materialized");
271 		assertFalse(lazyEntries.get("never-called").isMaterialized(), "never-called should NOT be materialized");
272 
273 		// Destroy the parent — only materialized child's @RestDestroy should fire.
274 		rc.destroy();
275 
276 		assertEquals(1, A05_DESTROY_COUNT.get(),
277 			"@RestDestroy must fire exactly once (only on the materialized child)");
278 	}
