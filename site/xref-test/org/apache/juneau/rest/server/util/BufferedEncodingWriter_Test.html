28  /**
29   * Tests for {@link BufferedEncodingWriter}: output must match <c>OutputStreamWriter</c> byte for byte.
30   */
31  class BufferedEncodingWriter_Test {
32  
33  	private static byte[] reference(String text, String cs) throws IOException {
34  		var bos = new ByteArrayOutputStream();
35  		try (var w = new OutputStreamWriter(bos, cs)) {
36  			w.write(text);
37  		}
38  		return bos.toByteArray();
39  	}
40  
41  	private static byte[] actual(String text, String cs, int chunk) throws IOException {
42  		var bos = new ByteArrayOutputStream();
43  		try (var w = new BufferedEncodingWriter(bos, cs)) {
44  			var chars = text.toCharArray();
45  			for (var i = 0; i < chars.length; i += chunk)
46  				w.write(chars, i, Math.min(chunk, chars.length - i));
47  		}
48  		return bos.toByteArray();
49  	}
50  
51  	private static String sample(int len, long seed) {
52  		var r = new Random(seed);
53  		var sb = new StringBuilder();
54  		while (sb.length() < len) {
55  			switch (r.nextInt(4)) {
56  				case 0 -> sb.append((char)('a' + r.nextInt(26)));
57  				case 1 -> sb.append((char)(0x80 + r.nextInt(0x700)));
58  				case 2 -> sb.appendCodePoint(0x1F300 + r.nextInt(0x300));
59  				default -> sb.append((char)(0x4E00 + r.nextInt(0x100)));
60  			}
61  		}
62  		return sb.toString();
63  	}
64  
65  	@Test void a01_matchesOutputStreamWriter_acrossSizesAndChunks() throws IOException {
66  		for (var cs : new String[]{"UTF-8", "UTF-16", "ISO-8859-1", "US-ASCII"})
67  			for (var len : new int[]{0, 1, 5, 255, 256, 257, 8191, 8192, 8193, 20000, 70000})
68  				for (var chunk : new int[]{1, 7, 300, 8192, 100000}) {
69  					var text = sample(len, len * 31L + chunk);
70  					assertArrayEquals(reference(text, cs), actual(text, cs, chunk), cs + " len=" + len + " chunk=" + chunk);
71  				}
72  	}
73  
74  	@Test void a02_stringWritesMatch() throws IOException {
75  		var text = sample(30000, 7);
76  		var bos = new ByteArrayOutputStream();
77  		try (var w = new BufferedEncodingWriter(bos, "UTF-8")) {
78  			for (var i = 0; i < text.length(); i += 1000)
79  				w.write(text, i, Math.min(1000, text.length() - i));
80  			w.write('!');
81  		}
82  		assertArrayEquals(reference(text + "!", "UTF-8"), bos.toByteArray());
83  	}
84  
85  	@Test void a03_surrogatePairSplitAcrossWritesAndFlush() throws IOException {
86  		var bos = new ByteArrayOutputStream();
87  		var pair = "😀";
88  		try (var w = new BufferedEncodingWriter(bos, "UTF-8")) {
89  			w.write(pair.charAt(0));
90  			w.flush();
91  			assertSize(0, bos);
92  			w.write(pair.charAt(1));
93  		}
94  		assertArrayEquals(pair.getBytes(StandardCharsets.UTF_8), bos.toByteArray());
95  	}
96  
97  	@Test void a04_loneSurrogateReplacedOnClose() throws IOException {
98  		assertArrayEquals(reference("a\uD83D", "UTF-8"), actual("a\uD83D", "UTF-8", 10));
99  	}
100 
101 	@Test void a05_flushWritesPendingBytesAndFlushesStream() throws IOException {
102 		var flushed = new boolean[1];
103 		var bos = new ByteArrayOutputStream() {
104 			@Override public void flush() { flushed[0] = true; }
105 		};
106 		try (var w = new BufferedEncodingWriter(bos, "UTF-8")) {
107 			w.write("hi");
108 			assertSize(0, bos);
109 			w.flush();
110 			assertEquals("hi", bos.toString(StandardCharsets.UTF_8));
111 			assertTrue(flushed[0]);
112 		}
113 	}
114 
115 	@Test void a06_unsupportedEncoding() {
116 		var bos = new ByteArrayOutputStream();
117 		assertThrows(UnsupportedEncodingException.class, () -> new BufferedEncodingWriter(bos, "no-such-charset"));
118 		assertThrows(UnsupportedEncodingException.class, () -> new BufferedEncodingWriter(bos, "bad name!"));
119 		assertThrows(NullPointerException.class, () -> new BufferedEncodingWriter(bos, null));
120 	}

122 	@Test void a07_writeAfterCloseThrows() throws IOException {
123 		try (var w = new BufferedEncodingWriter(new ByteArrayOutputStream(), "UTF-8")) {
124 			w.close();
125 			w.close();
126 			assertThrows(IOException.class, () -> w.write("x"));
127 			assertThrows(IOException.class, w::flush);
128 		}
129 	}

131 	@Test void a08_closeClosesStream() throws IOException {
132 		var closed = new boolean[1];
133 		var bos = new ByteArrayOutputStream() {
134 			@Override public void close() { closed[0] = true; }
135 		};
136 		new BufferedEncodingWriter(bos, "UTF-8").close();
137 		assertTrue(closed[0]);
138 	}

140 	@Test void a09_badBounds() throws IOException {
141 		try (var w = new BufferedEncodingWriter(new ByteArrayOutputStream(), "UTF-8")) {
142 			assertThrows(IndexOutOfBoundsException.class, () -> w.write(new char[2], 1, 5));
143 			assertThrows(IndexOutOfBoundsException.class, () -> w.write("ab", -1, 1));
144 		}
145 	}
146 
147 	@Test void a10_printWriterWrapper() throws IOException {
148 		var bos = new ByteArrayOutputStream();
149 		var pw = new FinishablePrintWriter(bos, "UTF-8", false);
150 		pw.print("café");
151 		pw.flush();
152 		assertArrayEquals("café".getBytes(StandardCharsets.UTF_8), bos.toByteArray());
153 		assertThrows(UnsupportedEncodingException.class, () -> new FinishablePrintWriter(bos, "bogus", false));
154 	}
