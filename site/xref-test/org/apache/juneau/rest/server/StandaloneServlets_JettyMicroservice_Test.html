37  /**
38   * Real-Jetty deployment-parity assertion for the standalone {@code Basic*Servlet} companions.
39   *
40   * <p>
41   * Boots a {@link org.apache.juneau.marshall.microservice.Microservice Microservice} on an ephemeral port and
42   * mounts three standalone servlets as siblings &mdash; {@link VersionServlet} at
43   * {@code /version/*}, {@link StaticFilesServlet} at {@code /static/*}, and
44   * {@link AdminServlet} at {@code /admin/*} &mdash; relying on the {@code JettyServerComponent}
45   * auto-discovery loop to read each servlet's {@code @Rest(paths)} via
46   * {@code RestContext.resolveTopLevelPaths(...)}. This is the cross-runtime parity check that the
47   * sibling top-level mount + op-at-{@code /*} shape resolves correctly through a real servlet
48   * container (which {@code MockRestClient} cannot fully model for {@code paths()}-based mounts).
49   *
50   * @since 10.0.0
51   */
52  @org.apache.juneau.testing.JettyMicroserviceTest
53  class StandaloneServlets_JettyMicroservice_Test extends TestBase {
54  
55  	@Configuration
56  	public static class Config {
57  		@Bean(name="versionServlet") public Servlet versionServlet() { return new VersionServlet(); }
58  		@Bean(name="staticFilesServlet") public Servlet staticFilesServlet() { return new StaticFilesServlet(); }
59  		@Bean(name="adminServlet") public Servlet adminServlet() { return new AdminServlet(); }
60  	}
61  
62  	@RegisterExtension
63  	static MicroserviceTestFixture fixture = MicroserviceTestFixture.create()
64  		.configurations(Config.class);
65  
66  	private static final HttpClient HTTP = HttpClient.newBuilder()
67  		.connectTimeout(Duration.ofSeconds(5))
68  		.followRedirects(HttpClient.Redirect.NEVER)
69  		.build();
70  
71  	private static HttpResponse<String> get(String path) throws Exception {
72  		var req = HttpRequest.newBuilder()
73  			.uri(URI.create(fixture.getRootUrl() + path))
74  			.timeout(Duration.ofSeconds(10))
75  			.GET()
76  			.build();
77  		return HTTP.send(req, BodyHandlers.ofString());
78  	}
79  
80  	//-----------------------------------------------------------------------------------------------------------------
81  	// Version standalone servlet at /version/* with op /* (verifies the null-pathInfo root case).
82  	//-----------------------------------------------------------------------------------------------------------------
83  
84  	@Test void a01_versionServlet() throws Exception {
85  		var resp = get("/version");
86  		assertEquals(200, resp.statusCode(), "body: " + resp.body());
87  		assertTrue(resp.body().contains("\"javaVersion\""), "javaVersion key present: " + resp.body());
88  		var ct = resp.headers().firstValue("Content-Type").orElse("");
89  		assertTrue(ct.startsWith("application/json"), "Content-Type was: " + ct);
90  	}
91  
92  	//-----------------------------------------------------------------------------------------------------------------
93  	// Static-files standalone servlet at /static/* with op /*.
94  	//-----------------------------------------------------------------------------------------------------------------
95  
96  	@Test void b01_staticFilesServlet() throws Exception {
97  		var resp = get("/static/javadoc.css");
98  		assertEquals(200, resp.statusCode(), "body: " + resp.body());
99  		assertTrue(resp.body().contains("Licensed to the Apache Software Foundation"),
100 			"static file body should contain the Apache license header");
101 		var ct = resp.headers().firstValue("Content-Type").orElse("");
102 		assertTrue(ct.startsWith("text/css"), "Content-Type was: " + ct);
103 	}
104 
105 	@Test void b02_staticMissingReturns404() throws Exception {
106 		var resp = get("/static/does-not-exist.css");
107 		assertEquals(404, resp.statusCode());
108 	}
109 
110 	//-----------------------------------------------------------------------------------------------------------------
111 	// Admin standalone servlet at /admin/* — deny-all guard returns 403 until an auth chain is wired.
112 	//-----------------------------------------------------------------------------------------------------------------
113 
114 	@Test void c01_adminThreadsDeniedByDefault() throws Exception {
115 		var resp = get("/admin/threads");
116 		assertEquals(403, resp.statusCode(), "deny-all guard should forbid; body: " + resp.body());
117 	}
118 }
