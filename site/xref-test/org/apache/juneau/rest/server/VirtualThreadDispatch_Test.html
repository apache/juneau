31  /**
32   * Tests virtual-thread dispatch — opt-in via
33   * {@link Rest#virtualThreads() @Rest(virtualThreads=true)} or {@link RestOp#virtualThreads()}, with
34   * graceful degradation on Java 17/18/19/20 (one-shot {@code WARNING} log + caller-thread fallback).
35   *
36   * <p>
37   * Tests in section <b>A</b> are guarded by {@link DisabledOnJre} for any JVM older than 21 — the
38   * underlying API ({@code Thread.currentThread().isVirtual()}) doesn't exist before then. Tests in
39   * section <b>B</b> run on every JVM and verify the graceful-degradation contract: enabling the flag
40   * on a non-supporting JVM must not break the resource and must emit a one-shot warning.
41   *
42   * <h5 class='section'>Java 17 reflective compile-time strategy</h5>
43   * <p>
44   * The virtual-thread executor in {@link RestContext} is built reflectively
45   * ({@code Executors.class.getMethod("newVirtualThreadPerTaskExecutor")}) so the framework compiles
46   * cleanly on Java 17 even though the API only exists at runtime on Java 21+. The runtime check
47   * ({@code Runtime.version().feature() &gt;= 21}) gates the reflective lookup; on Java 17 we never
48   * touch the missing method and the {@code Executor} memoizer returns {@code null} after logging the
49   * warning. {@link RestOpInvoker} then falls through to the standard caller-thread dispatch path —
50   * exactly the existing behavior.
51   */
52  @SuppressWarnings({
53  	"java:S4144" // Per-fixture handlers share the same thread-introspection body but probe distinct virtual-thread scenarios across separate @Rest resources.
54  })
55  class VirtualThreadDispatch_Test extends TestBase {

57  	// -----------------------------------------------------------------------------------------------------------------
58  	// A: Java 21+ — @Rest(virtualThreads=true) actually dispatches handler invocation on a virtual thread.
59  	// -----------------------------------------------------------------------------------------------------------------
60  
61  	@Rest(virtualThreads = "true", serializers = JsonSerializer.class)
62  	public static class A {
63  		@RestGet("/which")
64  		public ThreadInfo which() {
65  			Thread t = Thread.currentThread();
66  			return new ThreadInfo(t.getName(), reflectIsVirtual(t));
67  		}
68  
69  		@RestGet("/asyncWhich")
70  		public CompletableFuture<ThreadInfo> asyncWhich() {
71  			Thread t = Thread.currentThread();
72  			return CompletableFuture.completedFuture(new ThreadInfo(t.getName(), reflectIsVirtual(t)));
73  		}
74  	}
75  
76  	/**
77  	 * Reflective {@code Thread.isVirtual()} so this test class compiles on the project's Java 17 source level.
78  	 * On Java < 21 the method does not exist and we return {@code false} — which is also the correct behavior
79  	 * since virtual threads cannot exist on those JVMs.
80  	 */
81  	static boolean reflectIsVirtual(Thread t) {
82  		try {
83  			var m = Thread.class.getMethod("isVirtual");
84  			return (Boolean) m.invoke(t);
85  		} catch (Exception e) {
86  			return false;
87  		}
88  	}
89  
90  	public static final class ThreadInfo {
91  		public final String threadName;
92  		public final boolean virtual;
93  		public ThreadInfo(String threadName, boolean virtual) { this.threadName = threadName; this.virtual = virtual; }
94  	}

96  	private static final MockRestClient CA = MockRestClient.buildLax(A.class);

98  	@Test
99  	@EnabledForJreRange(min = JAVA_21)
100 	void a01_java21Plus_virtualThreadDispatch() throws Exception {
101 		CA.get("/which").run().assertStatus(200).assertContent().isContains("\"virtual\":true");
102 	}
103 
104 	@Test
105 	@EnabledForJreRange(min = JAVA_21)
106 	void a02_java21Plus_virtualThreadDispatch_combinedWithCompletableFuture() throws Exception {
107 		// The handler runs on a virtual thread, returns a CompletableFuture, and the response comes back unwrapped.
108 		// This is the high-throughput pattern: VT + CompletableFuture together.
109 		CA.get("/asyncWhich").run().assertStatus(200).assertContent().isContains("\"virtual\":true");
110 	}

134 	@Test
135 	@DisabledForJreRange(min = JAVA_21)
136 	void b02_java17_logsWarningOnce() throws Exception {
137 		// Capture the WARNING emitted by RestContext when @Rest(virtualThreads=true) is configured on Java < 21.
138 		var captured = new StringBuilder();
139 		var logger = Logger.getLogger(RestContext.class.getName() + ".async");
140 		var handler = new Handler() {
141 			@Override public void publish(LogRecord r) {
142 				if (r.getLevel() == Level.WARNING)
143 					captured.append(r.getMessage());
144 			}
145 			@Override public void flush() { /* intentionally empty */ }
146 			@Override public void close() { /* intentionally empty */ }
147 		};
148 		logger.addHandler(handler);
149 		try {
150 			MockRestClient.buildLax(BWarning.class).get("/x").run().assertStatus(200);
151 			assertTrue(captured.toString().contains("virtual-thread") || captured.toString().contains("virtualThreads"),
152 				"expected warning about virtual threads on Java <21, captured: " + captured);
153 		} finally {
154 			logger.removeHandler(handler);
155 		}
156 	}
