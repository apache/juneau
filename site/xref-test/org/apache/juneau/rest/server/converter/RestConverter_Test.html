30  	//------------------------------------------------------------------------------------------------------------------
31  	// A - Traversable converter
32  	//------------------------------------------------------------------------------------------------------------------
33  
34  	@Rest(converters=Traversable.class)
35  	public static class A implements BasicUniversalConfig {
36  		@RestOp(path="/*")
37  		public Map<String,Object> a() {
38  			var m = new LinkedHashMap<String,Object>();
39  			m.put("a", "1");
40  			m.put("b", Map.of("c", "2"));
41  			return m;
42  		}
43  	}
44  
45  	@Test void a01_traversableFullObject() throws Exception {
46  		var c = MockRestClient.buildJson(A.class);
47  		c.get("/").run().assertContent("{\"a\":\"1\",\"b\":{\"c\":\"2\"}}");
48  	}
49  
50  	@Test void a02_traversableSubPath() throws Exception {
51  		var c = MockRestClient.buildJson(A.class);
52  		c.get("/b").run().assertContent("{\"c\":\"2\"}");
53  	}
54  
55  	@Test void a03_traversableLeafValue() throws Exception {
56  		var c = MockRestClient.buildJson(A.class);
57  		c.get("/b/c").run().assertContent("\"2\"");
58  	}
59  
60  	@Test void a04_traversableNullResponse() throws Exception {
61  		var c = MockRestClient.buildJson(A2.class);
62  		c.get("/").run().assertContent("null");
63  	}

77  	@Rest(converters=Queryable.class)
78  	public static class B implements BasicUniversalConfig {
79  		@RestOp(path="/")
80  		public List<Map<String,Object>> b() {
81  			return List.of(
82  				Map.of("name", "Alice", "age", 30),
83  				Map.of("name", "Bob", "age", 25),
84  				Map.of("name", "Charlie", "age", 35)
85  			);
86  		}
87  	}
88  
89  	@Test void b01_queryableNoParams() throws Exception {
90  		var c = MockRestClient.buildJson(B.class);
91  		// Just verify we get a valid array with 3 elements; ordering of Map.of() keys not guaranteed
92  		c.get("/").run().assertContent().isContains("Alice","Bob","Charlie");
93  	}
94  
95  	@Test void b02_queryableViewParam() throws Exception {
96  		var c = MockRestClient.buildJson(B.class);
97  		c.get("/?v=name").run().assertContent().isContains("Alice","Bob","Charlie");
98  		c.get("/?v=name").run().assertContent().isNotContains("age");
99  	}

154 	// D - Multiple converters on same resource
155 	//------------------------------------------------------------------------------------------------------------------
156 
157 	@Rest(converters={Traversable.class, Queryable.class})
158 	public static class D implements BasicUniversalConfig {
159 		@RestOp(path="/*")
160 		public Map<String,Object> d() {
161 			var m = new LinkedHashMap<String,Object>();
162 			m.put("items", List.of(
163 				Map.of("name", "Alice", "age", 30),
164 				Map.of("name", "Bob", "age", 25)
165 			));
166 			return m;
167 		}
168 	}
169 
170 	@Test void d01_multipleConvertersTraverseThenQuery() throws Exception {
171 		var c = MockRestClient.buildJson(D.class);
172 		c.get("/items?v=name").run().assertContent().isContains("Alice","Bob");
173 		c.get("/items?v=name").run().assertContent().isNotContains("age");
174 	}
175 
176 	@Test void d02_multipleConvertersTraverseOnly() throws Exception {
177 		var c = MockRestClient.buildJson(D.class);
178 		c.get("/items").run().assertContent().isContains("Alice","Bob","age");
179 	}
