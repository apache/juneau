41  	@Test void a01_globalSearch_synthesizesOrOfContains() {
42  		var req = new DataTablesRequest().setColumns(cols()).setSearch(new Search().setValue("bob"));
43  		var q = DataTablesQuery.toBeanQuery(req, ctx("name", SearchType.TEXT, "age", SearchType.TEXT));
44  		assertEquals("$or(name=$contains(\"bob\"),age=$contains(\"bob\"))", q.getSearch());
45  	}

48  		var columns = cols();
49  		columns.get(0).setSearch(new Search().setValue("$eq(Alice)"));
50  		var req = new DataTablesRequest().setColumns(columns).setSearch(new Search().setValue("x"));
51  		var q = DataTablesQuery.toBeanQuery(req, ctx("name", SearchType.TEXT, "age", SearchType.TEXT));
52  		assertEquals("name=$eq(Alice),$or(name=$contains(\"x\"),age=$contains(\"x\"))", q.getSearch());
53  	}

73  	/** Builds a {@link BeanQueryContext} declaring the given {@code column, SearchType, column, SearchType, ...} pairs. */
74  	private static BeanQueryContext<?> ctx(Object...pairs) {
75  		var b = InMemoryBeanQueryContext.<Map<String,Object>>create();
76  		for (var i = 0; i < pairs.length; i += 2)
77  			b.column((String) pairs[i], (SearchType) pairs[i + 1]);
78  		return b.build();
79  	}

137 	@Test void b10_mixedColumns_onlyParsingTypesContribute_unwrapsSingleSurvivor() {
138 		var columns = List.of(new Column().setData("name"), new Column().setData("age"), new Column().setData("active"));
139 		var req = new DataTablesRequest().setColumns(columns).setSearch(new Search().setValue("bob"));
140 		var q = DataTablesQuery.toBeanQuery(req, ctx("name", SearchType.TEXT, "age", SearchType.NUMERIC, "active", SearchType.BOOLEAN));
141 		// "bob" parses as neither NUMERIC nor BOOLEAN, so only the TEXT column survives -> unwrapped, no $or(...)
142 		assertEquals("name=$contains(\"bob\")", q.getSearch());

145 	public static class Rec {
146 		private final String name;
147 		private final int age;
148 
149 		Rec(String name, int age) {
150 			this.name = name;
151 			this.age = age;
152 		}
153 
154 		public String getName() { return name; }
155 		public int getAge() { return age; }
156 
157 		@Override public String toString() { return name; }
158 	}
159 
160 	private static final List<Rec> ROWS = List.of(
161 		new Rec("Alice", 30),
162 		new Rec("Bob", 25),
163 		new Rec("Carol", 40),
164 		new Rec("Dave", 17)
165 	);
166 
167 	private static InMemoryBeanQueryContext<Rec> context() {
168 		return InMemoryBeanQueryContext.create(Rec.class).build();  // name: TEXT, age: NUMERIC (inferred).
169 	}
170 
171 	private static List<String> names(DataTablesResults<Rec> r) {
172 		return r.getData().stream().map(Rec::getName).toList();
173 	}
174 
175 	@Test
176 	void c01_globalSearchRunsThroughARealSession() {
177 		var columns = new ArrayList<>(List.of(new Column().setData("name"), new Column().setData("age")));
178 		var req = new DataTablesRequest().setLength(10).setColumns(columns).setSearch(new Search().setValue("car"));
179 		DataTablesResults<Rec> r;
180 		try (var s = context().getSession(ROWS)) {
181 			r = DataTablesQuery.run(req, s);
182 		}
183 		// "car" matches Carol by name ($contains) and nothing by age (not numeric) — the per-type global $or (gap
184 		// 5/D9) resolves to one eligible column here, but it still goes through FilterEvaluator end to end.
185 		assertList(names(r), "Carol");
186 		assertBean(r, "recordsTotal,recordsFiltered", "4,1");
187 	}
