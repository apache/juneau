17  package org.apache.juneau.rest.server.datatables;
18  
19  import static org.apache.juneau.test.bct.BctAssertions.*;
20  
21  import org.apache.juneau.*;
22  import org.apache.juneau.rest.mock.classic.*;
23  import org.apache.juneau.rest.server.*;
24  import org.apache.juneau.rest.server.config.*;
25  import org.apache.juneau.rest.server.staticfile.*;
26  import org.junit.jupiter.api.*;
27  
28  /**
29   * MockRest tests of {@link WebJarsMixin}.  They live here because {@code juneau-rest-server} has no mock-client
30   * test dependency, and this module has both the mock client and the real WebJars on its classpath.
31   */
32  @SuppressWarnings({
33  	"resource" // MockRestClient.close() is a no-op (no real OS resource).
34  })
35  class WebJarsMixin_Test extends TestBase {
36  
37  	@Rest(mixins=WebJarsMixin.class)
38  	public static class Host implements BasicUniversalConfig {}
39  
40  	private static final MockRestClient C = MockRestClient.buildLax(Host.class);
41  
42  	@Test void a01_knownAsset_200_contentType_immutableCaching() throws Exception {
43  		var r = C.get("/webjars/fixture/1.0/Upper.JS").run().assertStatus(200);
44  		assertContains("text/javascript", r.getHeader("Content-Type").asString().orElse("(none)"));
45  		assertString("public, max-age=31536000, immutable", r.getHeader("Cache-Control").asString().orElse("(none)"));
46  		assertContains("window.fixture", r.getContent().asString());
47  	}
48  
49  	@Test void a02_realWebJar_isServed() throws Exception {
50  		var r = C.get("/webjars/jquery/3.7.1/jquery.min.js").run().assertStatus(200);
51  		assertContains("text/javascript", r.getHeader("Content-Type").asString().orElse("(none)"));
52  		C.get("/webjars/datatables.net-dt/2.3.8/css/dataTables.dataTables.min.css").run().assertStatus(200);
53  	}
54  
55  	// The empty-segment case ("a//b") is covered by WebJarsMixin_PathCheck_Test: the REST layer collapses "//" before the mixin sees it.
56  	@Test void a03_traversalAndEncoding_404() throws Exception {
57  		for (var p : new String[] {
58  				"a/../../x", "%2e%2e/x", "fixture/%2e%2e/fixture/1.0/Upper.JS", "fixture%2f1.0%2fUpper.JS",
59  				"fixture%5c1.0%5cUpper.JS", "./fixture/1.0/Upper.JS" })
60  			C.get("/webjars/" + p).run().assertStatus(404);
61  	}
62  
63  	@Test void a04_disallowedExtension_directory_missing_404() throws Exception {
64  		for (var p : new String[] { "fixture/1.0/readme.txt", "fixture/1.0", "fixture", "fixture/1.0/missing.js" })
65  			C.get("/webjars/" + p).run().assertStatus(404);
66  	}
67  
68  	@Test void a05_head_headersWithoutBody() throws Exception {
69  		var r = C.head("/webjars/fixture/1.0/Upper.JS").run().assertStatus(200);
70  		assertContains("text/javascript", r.getHeader("Content-Type").asString().orElse("(none)"));
71  		assertEmpty(r.getContent().asString());
72  	}
73  }
