17  package org.apache.juneau.rest.server.datatables;
18  
19  import static org.apache.juneau.test.bct.BctAssertions.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.util.*;
23  
24  import org.junit.jupiter.api.*;
25  
26  /**
27   * Tests {@link DataTablesResults}: the {@code long} counts and the {@link DataTablesResults#error(int, String)} factory.
28   */
29  class DataTablesResults_Test {
30  
31  	@Nested class A_settersAndGetters {
32  
33  		@Test void a01_drawRoundTrips() {
34  			assertEquals(7, DataTablesResults.<String>create().setDraw(7).getDraw());
35  		}
36  
37  		@Test void a02_recordsTotalIsLong() {
38  			assertEquals(5_000_000_000L, DataTablesResults.<String>create().setRecordsTotal(5_000_000_000L).getRecordsTotal());
39  		}
40  
41  		@Test void a03_recordsFilteredIsLong() {
42  			assertEquals(3_000_000_000L, DataTablesResults.<String>create().setRecordsFiltered(3_000_000_000L).getRecordsFiltered());
43  		}
44  
45  		@Test void a04_dataRoundTrips() {
46  			assertList(DataTablesResults.<String>create().setData(List.of("a", "b")).getData(), "a", "b");
47  		}
48  
49  		@Test void a05_errorRoundTrips() {
50  			assertEquals("boom", DataTablesResults.<String>create().setError("boom").getError());
51  		}
52  
53  		@Test void a06_defaultsAreZeroAndNull() {
54  			assertBean(DataTablesResults.<String>create(), "draw,recordsTotal,recordsFiltered,data,error", "0,0,0,<null>,<null>");
55  		}
56  	}
57  
58  	@Nested class B_errorFactory {
59  
60  		@Test void b01_errorFactorySetsDrawAndMessage() {
61  			DataTablesResults<Object> r = DataTablesResults.error(9, "Unknown column 'x'.");
62  			assertBean(r, "draw,error", "9,Unknown column 'x'.");
63  		}
64  
65  		@Test void b02_errorFactoryLeavesCountsAndDataAtDefaults() {
66  			DataTablesResults<Object> r = DataTablesResults.error(1, "boom");
67  			assertBean(r, "recordsTotal,recordsFiltered,data", "0,0,<null>");
68  		}
69  	}
70  }
