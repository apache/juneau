42  	/**
43  	 * Row bean exercising every column-title source:
44  	 * <ul>
45  	 * 	<li>{@code order} - public field, no {@code @Schema} - humanized name.
46  	 * 	<li>{@code code} - public field, {@code @Schema(title)} - schema title (field path).
47  	 * 	<li>{@code name} - getter, {@code @Schema(title)} - schema title (getter path).
48  	 * 	<li>{@code notes} - getter, {@code @Schema(title="")} - empty schema title falls back to humanized name.
49  	 * 	<li>{@code releaseDate} - getter, no {@code @Schema} - humanized (camel-case split); value left null.
50  	 * 	<li>{@code secret} - write-only (setter only) - skipped (not readable).
51  	 * </ul>
52  	 */
53  	public static class Row {
54  		public String order = "o";
55  		@Schema(title="Ship Code") public String code = "c";
56  		private String releaseDate;
57  		@Schema(title="Full Name") public String getName() { return "Alice"; }
58  		@Schema(title="") public String getNotes() { return "n"; }
59  		public String getReleaseDate() { return releaseDate; }
60  		public void setSecret(@SuppressWarnings("unused") String value) { /* write-only */ }
61  	}
62  
63  	private static Map<String,String> byData(List<Map<String,Object>> cols) {
64  		var m = new LinkedHashMap<String,String>();
65  		cols.forEach(c -> m.put((String)c.get("data"), (String)c.get("title")));
66  		return m;
67  	}

73  	@Test void a01_columnTitleSources() {
74  		var cols = DataTablesColumns.of(Row.class);
75  		var titles = byData(cols);
76  		assertEquals("Order", titles.get("order"), titles::toString);          // humanized public field
77  		assertEquals("Ship Code", titles.get("code"), titles::toString);       // @Schema on field
78  		assertEquals("Full Name", titles.get("name"), titles::toString);       // @Schema on getter
79  		assertEquals("Notes", titles.get("notes"), titles::toString);          // empty @Schema title -> humanized
80  		assertEquals("Release Date", titles.get("releaseDate"), titles::toString); // camel-case humanization
81  		assertFalse(titles.containsKey("secret"), titles::toString);           // write-only skipped
82  	}

102 	@Test void a04_humanizeEdgeCases() {
103 		assertEquals("Name", DataTablesColumns.humanize("name"));
104 		assertEquals("First Name", DataTablesColumns.humanize("firstName"));
105 		assertEquals("", DataTablesColumns.humanize(""));
106 		// Consecutive capitals: only a lower->upper boundary inserts a space, so an upper char preceded by a
107 		// non-lowercase char (the second 'B'/'C' below) must NOT split.
108 		assertEquals("A BC", DataTablesColumns.humanize("aBC"));
109 	}

134 	@Test void b02_mapRowsWithExplicitColumnsAndNullCell() {
135 		var columns = List.of(
136 			col("name", "Name"),
137 			col("age", "Age")
138 		);
139 		var rows = new ArrayList<Map<String,Object>>();
140 		rows.add(new LinkedHashMap<>(Map.of("name", "Bob", "age", 40)));
141 		var missing = new LinkedHashMap<String,Object>();
142 		missing.put("name", "Carol");   // "age" intentionally absent -> null cell
143 		rows.add(missing);
144 		var html = HtmlSerializer.DEFAULT.toString(DataTablesTable.of("people", rows, columns));
145 		assertTrue(html.contains("Bob") && html.contains("Carol"), html);
146 		assertTrue(html.contains(">40<") || html.contains("40"), html);
147 	}

154 	// MarshallingContext overloads (of(ctx,id,rows,rowType) and of(ctx,id,rows,columns)) render the same content.
155 	@Test void b04_marshallingContextOverloads() {
156 		var fromType = HtmlSerializer.DEFAULT.toString(DataTablesTable.of(MarshallingContext.DEFAULT, "releases", List.of(new Row()), Row.class));
157 		assertTrue(fromType.contains("Alice") && fromType.contains("Ship Code"), fromType);
158 
159 		var rows = new ArrayList<Map<String,Object>>();
160 		rows.add(new LinkedHashMap<>(Map.of("name", "Bob")));
161 		var fromCols = HtmlSerializer.DEFAULT.toString(DataTablesTable.of(MarshallingContext.DEFAULT, "people", rows, List.of(col("name", "Name"))));
162 		assertTrue(fromCols.contains("Bob"), fromCols);
163 	}

179 	@Test void c01_glueScriptServed() throws Exception {
180 		var c = MockRestClient.build(GlueHost.class);
181 		var resp = c.get("/juneau-datatables.js").run().assertStatus(200);
182 		var body = resp.getContent().asString();
183 		assertTrue(body.contains("DataTable"), body);
184 		assertTrue(body.contains("data-juneau-datatable"), body);
185 		assertTrue(resp.getHeader("Content-Type").asString().orElse("").contains("javascript"),
186 			resp.getHeader("Content-Type").asString().orElse("(none)"));
187 		// Second request exercises the memoized (already-cached) fast path of the double-checked-locking glue-script reader.
188 		var body2 = c.get("/juneau-datatables.js").run().assertStatus(200).getContent().asString();
189 		assertEquals(body, body2, body2);
190 	}
