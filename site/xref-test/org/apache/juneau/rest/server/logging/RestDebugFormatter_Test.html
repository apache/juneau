17  package org.apache.juneau.rest.server.logging;

19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.junit.jupiter.api.*;
22  
23  /**
24   * Tests that the {@link RestDebugFormatter} interface default methods are additive-safe — a bare implementation
25   * supplying only {@link RestDebugFormatter#formatBasic(org.apache.juneau.rest.server.RestRequest, org.apache.juneau.rest.server.RestResponse) formatBasic}
26   * inherits empty defaults for the additive tiers.
27   *
28   * @since 10.0.0
29   */
30  class RestDebugFormatter_Test {
31  
32  	@Test void a01_defaultHeaders_isEmpty() {
33  		RestDebugFormatter f = (req, res) -> "basic";
34  		assertEquals("", f.formatHeaders(null, null));
35  	}
36  
37  	@Test void a02_defaultBody_isEmpty() {
38  		RestDebugFormatter f = (req, res) -> "basic";
39  		assertEquals("", f.formatBody(null, null));
40  	}
41  
42  	@Test void a03_defaultBodyCap_is8k() {
43  		RestDebugFormatter f = (req, res) -> "basic";
44  		assertEquals(8 * 1024, f.bodyCap());
45  	}
46  
47  	@Test void a04_basic_returnsSuppliedValue() {
48  		RestDebugFormatter f = (req, res) -> "basic";
49  		assertEquals("basic", f.formatBasic(null, null));
50  	}
51  }
