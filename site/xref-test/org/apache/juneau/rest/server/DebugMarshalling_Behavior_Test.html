29  /**
30   * Proves REST-driven serializer/parser debug <i>behavior</i> is decoupled from the JUL logger level and is instead
31   * controlled by the cascading {@code debugMarshalling} setting.
32   *
33   * <p>
34   * Two session-level {@code .debug()} observables are exercised (recursion&rarr;500 is intentionally <b>not</b>
35   * treated as a valid session-debug observable, per the Option-A design correction):
36   * <ul>
37   * 	<li><b>Serializer traversal-stack prefix</b> &mdash; when marshalling debug is engaged, a getter-invocation failure
38   * 		message is prefixed with the debug traversal stack (e.g. {@code " > [0] root:..."}) and the session collects it
39   * 		as a warning ({@code "Warnings occurred in session:"}).  That prefix/warning is absent when debug is off.
40   * 	<li><b>Parser buffered-input retention</b> &mdash; when marshalling debug is engaged, a malformed-parse failure
41   * 		retains and quotes the buffered request content (e.g. <code>"1: {name:"</code>).  When debug is off the message
42   * 		instead reads {@code "Use BEAN_debug setting to display content."}.
43   * </ul>
44   *
45   * <p>
46   * The decoupling gates {@code b04}/{@code b08} are deliberately <b>red on unmodified {@code main}</b>: on {@code main}
47   * the seven REST marshalling sites keyed {@code req.isDebug()}, so raising the resolved op logger to {@code FINE}
48   * engaged the debug observables above.  These tests assert the post-decoupling behavior (logger has no effect), which
49   * fails against that old coupled behavior.  Verified red-on-main via the revert-and-probe technique.
50   *
51   * <p>
52   * The op logger for a plain op is {@code <resourceClass>.<method>}; raising {@code Logger.getLogger(<resourceClass>)}
53   * controls {@code isDebug()} for its ops.  {@code renderResponseStackTraces} is enabled so the failure messages
54   * (and thus the observables above) surface in the response body.
55   *
56   * @since 10.0.0
57   */
58  class DebugMarshalling_Behavior_Test extends TestBase {
59  
60  	/** Bean whose getter throws — drives {@code SerializerSession.onBeanGetterException}. */
61  	public static class Boom {
62  		public String getOk() { return "ok"; }
63  		public String getBoom() { throw new RuntimeException("BOOMDETAIL"); }
64  	}
65  
66  	/** Simple target bean for malformed-parse cases. */
67  	public static class Bean { public String name; }
68  
69  	/** Malformed JSON body (truncated after the key) — parses to a {@link org.apache.juneau.marshall.parser.ParseException}. */
70  	private static final String MALFORMED = "{name:";
71  
72  	@FunctionalInterface private interface ThrowingRunnable { void run() throws Exception; }
73  
74  	private static void withLoggerLevel(Class<?> resourceClass, Level level, ThrowingRunnable action) throws Exception {
75  		var logger = Logger.getLogger(resourceClass.getName());
76  		var prev = logger.getLevel();
77  		logger.setLevel(level);
78  		try {
79  			action.run();
80  		} finally {
81  			logger.setLevel(prev);
82  		}
83  	}

86  	// Fixtures.
87  	// -----------------------------------------------------------------------------------------
88  
89  	/** No resource-level debugMarshalling; one op opts in, one stays default (for both serialize and parse). */
90  	@Rest(serializers=JsonSerializer.class, parsers=JsonParser.class, renderResponseStackTraces="true")
91  	public static class R_Default {
92  		@RestGet(path="/serOff") public Boom serOff() { return new Boom(); }
93  		@RestGet(path="/serOn", debugMarshalling="true") public Boom serOn() { return new Boom(); }
94  		@RestPut(path="/parseOff") public String parseOff(@Content Bean b) { return "ok"; }
95  		@RestPut(path="/parseOn", debugMarshalling="true") public String parseOn(@Content Bean b) { return "ok"; }
96  	}
97  
98  	/** Resource-level debugMarshalling=true; ops leave the annotation blank (inherit). */
99  	@Rest(debugMarshalling="true", serializers=JsonSerializer.class, parsers=JsonParser.class, renderResponseStackTraces="true")
100 	public static class R_ResourceTrue {
101 		@RestGet(path="/serInherit") public Boom serInherit() { return new Boom(); }
102 		@RestPut(path="/parseInherit") public String parseInherit(@Content Bean b) { return "ok"; }
103 	}
104 
105 	/** JSON serializer with {@code Context.debug} enabled directly on the serializer instance. */
106 	public static class DebugJson extends JsonSerializer {
107 		public DebugJson(JsonSerializer.Builder<?> b) { super(enableDebug(b)); }
108 		private static JsonSerializer.Builder<?> enableDebug(JsonSerializer.Builder<?> b) { b.debug(); return b; }
109 	}
110 
111 	/** debugMarshalling OFF, but the registered serializer has Context.debug enabled directly. */
112 	@Rest(serializers=DebugJson.class, renderResponseStackTraces="true")
113 	public static class R_DirectContextDebug {
114 		@RestGet(path="/ser") public Boom ser() { return new Boom(); }
115 	}

117 	private static final MockRestClient cDefault = MockRestClient.buildLax(R_Default.class);
118 	private static final MockRestClient cResourceTrue = MockRestClient.buildLax(R_ResourceTrue.class);
119 
120 	// -----------------------------------------------------------------------------------------
121 	// Serializer observable — traversal-stack-prefix on a getter-invocation failure.
122 	// -----------------------------------------------------------------------------------------
123 
124 	@Test void b01_serializer_opTrueAddsTraversalStackPrefix() throws Exception {
125 		// Op-level debugMarshalling=true → session debug → warning collected with " > [0] ..." traversal-stack prefix.
126 		cDefault.get("/serOn").accept("application/json").run()
127 			.assertStatus(500)
128 			.assertContent().isContains("> [0]", "Warnings occurred in session", "Could not call getValue");
129 	}
130 
131 	@Test void b02_serializer_offOmitsTraversalStackPrefix() throws Exception {
132 		// Default (off) → the getter failure still throws (500) but with NO traversal-stack prefix and no session warning.
133 		cDefault.get("/serOff").accept("application/json").run()
134 			.assertStatus(500)
135 			.assertContent().isContains("Could not call getValue")
136 			.assertContent().isNotContains("> [0]", "Warnings occurred in session");
137 	}
138 
139 	@Test void b03_serializer_resourceTrueInheritedByBlankOp() throws Exception {
140 		// Resource-level true is inherited by a blank op → traversal-stack prefix engaged.
141 		cResourceTrue.get("/serInherit").accept("application/json").run()
142 			.assertStatus(500)
143 			.assertContent().isContains("> [0]", "Warnings occurred in session");
144 	}
145 
146 	@Test void b04_serializer_loggerDoesNotEngageMarshallingDebug() throws Exception {
147 		// RED on main: FINE → isDebug()==true → traversal-stack prefix engaged.  Post-decoupling: no prefix.
148 		withLoggerLevel(R_Default.class, Level.FINE, () ->
149 			cDefault.get("/serOff").accept("application/json").run()
150 				.assertStatus(500)
151 				.assertContent().isContains("Could not call getValue")
152 				.assertContent().isNotContains("> [0]", "Warnings occurred in session"));
153 	}
154 
155 	// -----------------------------------------------------------------------------------------
156 	// Parser observable — buffered-input retention on a malformed-parse failure.
157 	// -----------------------------------------------------------------------------------------
158 
159 	@Test void b05_parser_opTrueRetainsBufferedInput() throws Exception {
160 		// Op-level debugMarshalling=true → parser pipe buffers input → malformed-parse failure quotes it.
161 		cDefault.put("/parseOn", MALFORMED, APPLICATION_JSON).run()
162 			.assertStatus(400)
163 			.assertContent().isContains(MALFORMED)
164 			.assertContent().isNotContains("Use BEAN_debug setting to display content.");
165 	}
166 
167 	@Test void b06_parser_offShowsBufferHint() throws Exception {
168 		// Default (off) → input not buffered → failure shows the "Use BEAN_debug setting" hint instead of the content.
169 		cDefault.put("/parseOff", MALFORMED, APPLICATION_JSON).run()
170 			.assertStatus(400)
171 			.assertContent().isContains("Use BEAN_debug setting to display content.");
172 	}
173 
174 	@Test void b07_parser_resourceTrueInheritedByBlankOp() throws Exception {
175 		// Resource-level true is inherited by a blank op → buffered input retained.
176 		cResourceTrue.put("/parseInherit", MALFORMED, APPLICATION_JSON).run()
177 			.assertStatus(400)
178 			.assertContent().isContains(MALFORMED)
179 			.assertContent().isNotContains("Use BEAN_debug setting to display content.");
180 	}
181 
182 	@Test void b08_parser_loggerDoesNotEngageMarshallingDebug() throws Exception {
183 		// RED on main: FINE → isDebug()==true → buffered input retained.  Post-decoupling: buffer hint (no content).
184 		withLoggerLevel(R_Default.class, Level.FINE, () ->
185 			cDefault.put("/parseOff", MALFORMED, APPLICATION_JSON).run()
186 				.assertStatus(400)
187 				.assertContent().isContains("Use BEAN_debug setting to display content.")
188 				.assertContent().isNotContains("1: " + MALFORMED));
189 	}

191 	// -----------------------------------------------------------------------------------------
192 	// Context-vs-session distinction (NOT the REST-layer observable) — a directly-set Context.debug on the
193 	// serializer is a Context-level setting; the REST site passes .debug(null) when debugMarshalling is off, which
194 	// must NOT clobber that pre-existing Context.debug.  Documents why recursion→500 (a Context-level detectRecursions
195 	// behavior) cannot be used as the session-debug REST observable.
196 	// -----------------------------------------------------------------------------------------
197 
198 	@Test void b09_directContextDebugPreservedWhenRestMarshallingOff() throws Exception {
199 		// debugMarshalling is OFF → REST site passes .debug(null); the serializer's own Context.debug=true survives,
200 		// so the traversal-stack prefix still appears — proving null pass-through does not override Context-level debug.
201 		MockRestClient.buildLax(R_DirectContextDebug.class).get("/ser").accept("application/json").run()
202 			.assertStatus(500)
203 			.assertContent().isContains("> [0]", "Warnings occurred in session");
204 	}
