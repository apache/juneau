44  /**
45   * Tests for the {@code content} URL-parameter body-override method restriction in {@link RestRequest}, exercised
46   * via the package-private {@link RestRequest#isContentParamMethod(String)} helper.
47   *
48   * <p>
49   * Also covers the RFC 7232 conditional-request comparison helpers ({@code matchesAnyStrong}, {@code matchesAnyWeak},
50   * {@code parseHttpDate}, {@code notModified}, {@code preconditionFailed}) at the unit level. These back
51   * {@link RestRequest#checkPreconditions(RestResponse)}, which is already exercised end-to-end via
52   * {@code MockRestClient} by {@code juneau-integration-tests}' {@code RestRequest_CheckPreconditions_Test} --
53   * but this module cannot depend on {@code MockRestClient} (see {@code RestArgResolvers_Test}'s note), so the few
54   * sub-branch combinations that test doesn't happen to hit (e.g. a non-wildcard {@code If-Match} against a
55   * resource with no {@code ETag} at all) are covered here directly against the pure comparison logic instead.
56   *
57   * <p>
58   * The {@code f} group below constructs real {@link RestRequest} instances directly via its package-private
59   * constructor (this test class shares the {@code org.apache.juneau.rest.server} package), wiring minimal
60   * Mockito-mocked {@link RestOpContext}/{@link RestContext}/{@link RestSession} dependencies plus a handful of
61   * real lightweight leaf collaborators ({@link HttpHeaderList}, {@link HttpPartList}, {@link NamedAttributeMap},
62   * {@link EncoderSet}, {@link ParserSet}, {@link MarshallingContext}, {@link OpenApiParser#DEFAULT},
63   * {@link OpenApiSerializer#DEFAULT}) -- this lets the instance-level getters/setters be exercised directly,
64   * without needing {@code MockRestClient} or a full servlet container.
65   *
66   * @since 10.0.0
67   */

201 	private static HttpServletRequest mockServletRequest(String method, Map<String,String> headers) throws Exception {
202 		var r = mock(HttpServletRequest.class);
203 		when(r.getMethod()).thenReturn(method);
204 		when(r.getRequestURI()).thenReturn("/foo");
205 		when(r.getScheme()).thenReturn("http");
206 		when(r.getServerName()).thenReturn("example.com");
207 		when(r.getServerPort()).thenReturn(80);
208 		when(r.getProtocol()).thenReturn("HTTP/1.1");
209 		when(r.getLocale()).thenReturn(Locale.US);
210 		when(r.getServletPath()).thenReturn("/foo");
211 		when(r.getContextPath()).thenReturn("");
212 		when(r.getPathInfo()).thenReturn(null);
213 		when(r.getHeaderNames()).thenReturn(Collections.enumeration(headers.keySet()));
214 		when(r.getInputStream()).thenAnswer(inv -> new org.apache.juneau.rest.server.util.BoundedServletInputStream(new byte[0]));
215 		for (var e : headers.entrySet()) {
216 			when(r.getHeader(e.getKey())).thenReturn(e.getValue());
217 			when(r.getHeaders(e.getKey())).thenReturn(Collections.enumeration(List.of(e.getValue())));
218 		}
219 		// Backing store so setAttribute/getAttribute/getAttributeNames behave statefully, as
220 		// RequestAttributes (and thus RestRequest#setAttribute/getAttribute) delegate straight
221 		// through to the underlying HttpServletRequest.
222 		var attrs = new LinkedHashMap<String,Object>();
223 		doAnswer(inv -> {
224 			attrs.put(inv.getArgument(0), inv.getArgument(1));
225 			return null;
226 		}).when(r).setAttribute(any(), any());
227 		when(r.getAttribute(any())).thenAnswer(inv -> attrs.get((String)inv.getArgument(0)));
228 		when(r.getAttributeNames()).thenAnswer(inv -> Collections.enumeration(attrs.keySet()));
229 		return r;
230 	}
