29  /**
30   * Integration tests for {@link RrpcServlet}.
31   */
32  @SuppressWarnings({
33  	"serial"  // serialVersionUID not required for test classes.
34  })
35  class RrpcServlet_Test extends TestBase {
36  
37  	// =========================================================================
38  	// Test interface and implementation
39  	// =========================================================================
40  
41  	public interface TestInterface {
42  		String echo(String input);
43  		int add(int a, int b);
44  		void doNothing();
45  		String throwException() throws Exception;
46  	}
47  
48  	public static class TestInterfaceImpl implements TestInterface {
49  		@Override public String echo(String input) { return input; }
50  		@Override public int add(int a, int b) { return a + b; }
51  		@Override public void doNothing() { /* Intentionally empty: exercises RPC invocation of a void no-op interface method. */ }
52  		@Override public String throwException() throws Exception { throw new RuntimeException("test error"); }
53  	}
54  
55  	// =========================================================================
56  	// Test servlet
57  	// =========================================================================
58  
59  	@Rest
60  	public static class TestRrpcServlet extends RrpcServlet {
61  		private final Map<Class<?>,Object> services = Map.of(TestInterface.class, new TestInterfaceImpl());
62  
63  		@Override
64  		protected Map<Class<?>,Object> getServiceMap() {
65  			return services;
66  		}
67  	}
68  
69  	// =========================================================================
70  	// A — Servlet initialization and interface listing
71  	// =========================================================================
72  
73  	@Test
74  	void a01_getInterfaces_returnsRegisteredInterface() throws Exception {
75  		try (var client = MockRestClient.create(TestRrpcServlet.class)) {
76  			try (var response = client.get("/").run()) {
77  				assertEquals(200, response.getStatusCode());
78  				var body = response.getBodyAsString();
79  				assertTrue(body.contains("TestInterface"), "Expected 'TestInterface' in response: " + body);
80  			}
81  		}
82  	}

90  		try (var client = MockRestClient.create(TestRrpcServlet.class)) {

136 	@Test
137 	void c01_invokeMethod_noArgs() throws Exception {
138 		try (var client = MockRestClient.create(TestRrpcServlet.class)) {
139 			var iface = TestInterface.class.getName();
140 			try (var response = client.post("/" + iface + "/" + urlEncode("doNothing/"))
141 					.header("Content-Type", "application/json")
142 					.bodyString("[]")
143 					.run()) {
144 				assertEquals(200, response.getStatusCode());
145 			}
146 		}

148 
149 	@Test
150 	void c02_invokeMethod_withArgs_returnsResult() throws Exception {
151 		try (var client = MockRestClient.create(TestRrpcServlet.class)) {
152 			var iface = TestInterface.class.getName();
153 			try (var response = client.post("/" + iface + "/" + urlEncode("add/(int,int)"))
154 					.header("Content-Type", "application/json")
155 					.bodyString("[3,4]")
156 					.run()) {
157 				assertEquals(200, response.getStatusCode());
158 				var body = response.getBodyAsString();
159 				assertTrue(body.contains("7"), "Expected '7' in response: " + body);
160 			}
161 		}
162 	}
163 
164 	@Test
165 	void c03_listMethods_returnsMethodList() throws Exception {
166 		try (var client = MockRestClient.create(TestRrpcServlet.class)) {
167 			var iface = TestInterface.class.getName();
168 			try (var response = client.get("/" + iface).run()) {
169 				assertEquals(200, response.getStatusCode());
170 				var body = response.getBodyAsString();
171 				assertTrue(body.contains("echo"), "Expected 'echo' in response: " + body);
172 				assertTrue(body.contains("add"), "Expected 'add' in response: " + body);
173 			}
174 		}
175 	}
