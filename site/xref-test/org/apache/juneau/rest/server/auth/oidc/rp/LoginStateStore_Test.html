37  /**
38   * Security gates for the injectable {@link LoginStateStore} SPI: injected-store wiring, fail-closed store
39   * outages, consume-time redirect re-validation, the framework expiry ceiling (enforced before the token
40   * exchange), {@link PendingLogin} redaction, and the default impl's concurrent single-use race.
41   *
42   * <p>
43   * These are the substance ("red-on-broken") gates for the login-state-store security hardening: each is constructed so a no-op setter,
44   * a mis-ordered check, or a class-only assertion would fail it.  The recording fake plus distinctive
45   * hardcoded values ensure the injected instance (not the hidden default store) is actually exercised.
46   *
47   * @since 10.0.0
48   */
49  @SuppressWarnings({
50  	"java:S1130", // Test methods declare throws Exception for checked exceptions MockServletResponse may propagate; declarations are intentionally broad.
51  	"java:S5778", // assertThrows lambdas with chained calls; intermediate invocations do not throw in practice.
52  	"java:S5976", // Similar-shaped gates assert distinct security outcomes; parameterizing would obscure intent.
53  	"java:S8692", // Nimbus oauth2-oidc-sdk exposes no clock hook on the ID-token path; the login-state expiry re-check IS clock-injectable and is tested deterministically.
54  	"resource" // Closeable StubIdp fixture; lifecycle managed by @BeforeEach/@AfterEach, not a real leak.
55  })

91  	private OidcRelyingParty rp(LoginStateStore loginStateStore) {
92  		return OidcRelyingParty.create()
93  			.metadata(idp.metadata(AUTHZ, END_SESSION))
94  			.clientId(CID)
95  			.clientSecret("client-secret")
96  			.redirectUri(REDIRECT_URI)
97  			.scope("openid", "profile")
98  			.sessionStore(InMemorySessionStore.create())
99  			.jwkSet(publicJwks(key))
100 			.loginStateStore(loginStateStore)
101 			.clock(CLOCK)
102 			.build();
103 	}

105 	private String startLoginGetState(OidcRelyingParty rp) throws Exception {
106 		var req = MockServletRequest.create("GET", "/auth/login");
107 		var res = MockServletResponse.create();
108 		rp.startLogin(req, res);
109 		var loc = res.getHeader("Location");
110 		assertNotNull(loc, "startLogin must redirect to the IdP");
111 		return queryParam(loc, "state");
112 	}

115 		var req = MockServletRequest.create("GET", REDIRECT_URI + "?code=abc123&state=" + state);
116 		var res = MockServletResponse.create();
117 		rp.completeLogin(req, res);
118 		return res;
119 	}

134 	static final class FakeStore implements LoginStateStore {
135 		final IntegerHolder storeCount = IntegerHolder.create();
136 		final IntegerHolder consumeCount = IntegerHolder.create();
137 		volatile String lastStoreState;
138 		volatile String lastConsumeState;
139 		volatile Optional<PendingLogin> lastConsumeReturned = Optional.empty();
140 
141 		RuntimeException storeThrows;
142 		RuntimeException consumeThrows;
143 		Optional<PendingLogin> consumeResult = Optional.empty();
144 
145 		@Override public void store(String state, PendingLogin p) {
146 			storeCount.increment();
147 			lastStoreState = state;
148 			if (storeThrows != null)
149 				throw storeThrows;
150 		}
151 
152 		@Override public Optional<PendingLogin> consume(String state) {
153 			consumeCount.increment();
154 			lastConsumeState = state;
155 			if (consumeThrows != null)
156 				throw consumeThrows;
157 			lastConsumeReturned = consumeResult;
158 			return consumeResult;
159 		}
160 	}

166 	@Test void g01_injectedStore_honoredOnBothLegs() throws Exception {
167 		var fake = new FakeStore();
168 		fake.consumeResult = Optional.of(pending(FAKE_REDIRECT, NOW, NOW.plus(Duration.ofMinutes(5))));
169 		var rp = rp(fake);
170 
171 		var state = startLoginGetState(rp);
172 		assertTrue(fake.storeCount.is(1), "store must be invoked once on the injected instance");
173 		assertEquals(state, fake.lastStoreState, "store must receive the generated state");
174 
175 		signIdTokenWith(FAKE_NONCE);  // sign with the fake's distinctive nonce; a mismatch fails validation
176 		var res = completeLogin(rp, state);
177 
178 		assertTrue(fake.consumeCount.is(1), "consume must be invoked once on the injected instance");
179 		assertEquals(state, fake.lastConsumeState, "consume must receive the callback state");
180 		assertNotNull(cookieValue(res.getHeader("Set-Cookie")), "login must succeed (session cookie set)");
181 		assertEquals(FAKE_REDIRECT, res.getHeader("Location"), "Location must be the fake's distinctive redirectTarget");
182 		// The verifier exchanged with the IdP is the fake's distinctive one (record-only capture on the stub).
183 		assertEquals(FAKE_VERIFIER, idp.lastCodeVerifier, "the fake's distinctive code_verifier must be exchanged");
184 		// Login succeeding with an ID token signed by FAKE_NONCE proves the fake's nonce was the one validated.
185 	}

191 	@Test void g03_consumeMiss_onInjectedStore_rejected() throws Exception {
192 		var fake = new FakeStore();
193 		fake.consumeResult = Optional.empty();  // injected consume misses
194 		var rp = rp(fake);
195 
196 		// startLogin FIRST: a hidden default store would hold a live entry and login would succeed if the fake
197 		// were ignored.  So a no-op setter is caught two ways below (consume count stays 0; no miss).
198 		var state = startLoginGetState(rp);
199 		assertTrue(fake.storeCount.is(1));
200 		assertEquals(state, fake.lastStoreState);
201 
202 		signIdTokenWith(FAKE_NONCE);
203 		var ex = assertThrows(AuthenticationException.class, () -> completeLogin(rp, state));
204 
205 		assertTrue(fake.consumeCount.is(1), "the injected consume must be the CSRF/replay decision point");
206 		assertEquals(state, fake.lastConsumeState);
207 		assertTrue(ex.getMessage().contains("missing, expired, or already used"),
208 			"must reject with the consume-miss message, got: " + ex.getMessage());
209 		assertEquals(0, idp.tokenHits.get(), "a consume miss must not reach the token exchange");
210 	}

273 	@Test void g07_overTtlRecord_rejected_beforeExchange() throws Exception {
274 		// Past its minted expiresAt (now-1min) but well within MAX_TTL of createdAt (only 6 min old): a
275 		// MAX_TTL-only implementation would accept it, so this pins the minted-expiresAt re-check.
276 		var overTtl = pending(FAKE_REDIRECT, NOW.minus(Duration.ofMinutes(6)), NOW.minus(Duration.ofMinutes(1)));
277 		var fake = new FakeStore();
278 		fake.consumeResult = Optional.of(overTtl);
279 		var rp = rp(fake);
280 
281 		var state = startLoginGetState(rp);
282 		signIdTokenWith(FAKE_NONCE);  // would-succeed setup: the ONLY thing between success and rejection is the expiry re-check
283 		var ex = assertThrows(AuthenticationException.class, () -> completeLogin(rp, state));
284 
285 		// (2) message-specific, not class-only — distinct from the post-exchange throws.
286 		assertTrue(ex.getMessage().contains("missing, expired, or already used"),
287 			"over-age record must be rejected with the expiry/miss message, got: " + ex.getMessage());
288 		// (3) token endpoint never hit — proves the re-check runs BEFORE codeFlow().exchange.
289 		assertEquals(0, idp.tokenHits.get(), "the expiry re-check must run before the token exchange (code not burned)");
290 		// consume-miss disambiguation — the present over-TTL record WAS consumed once, then rejected for age.
291 		assertTrue(fake.consumeCount.is(1), "the injected consume must have been invoked once");
292 		assertEquals(state, fake.lastConsumeState);
293 		assertTrue(fake.lastConsumeReturned.isPresent(), "the record was present (not a consume miss)");
294 		assertEquals(NOW.minus(Duration.ofMinutes(1)), fake.lastConsumeReturned.get().expiresAt(),
295 			"the rejected record is the over-TTL one the fake returned");
296 	}

344 	// Default impl atomic single-use survives a concurrent-consume race (in-process only).
345 	//-----------------------------------------------------------------------------------------------------------------
346 
347 	@Test void g09_defaultStore_concurrentConsume_isSingleUse() throws Exception {
348 		var store = new InMemoryLoginStateStore(Duration.ofMinutes(5), 1000, CLOCK);
349 		var pool = Executors.newFixedThreadPool(2);
350 		try {
351 			for (var i = 0; i < 300; i++) {
352 				var stateKey = "state-" + i;
353 				store.store(stateKey, pending(null, NOW, NOW.plus(Duration.ofMinutes(5))));
354 				var present = new AtomicInteger();
355 				var start = new CountDownLatch(1);
356 				Callable<Void> task = () -> {
357 					start.await();
358 					if (store.consume(stateKey).isPresent())
359 						present.incrementAndGet();
360 					return null;
361 				};
362 				var f1 = pool.submit(task);
363 				var f2 = pool.submit(task);
364 				start.countDown();
365 				f1.get();
366 				f2.get();
367 				assertEquals(1, present.get(), "exactly one concurrent consume may win the single-use race");
368 			}
369 		} finally {
370 			pool.shutdownNow();
371 		}
