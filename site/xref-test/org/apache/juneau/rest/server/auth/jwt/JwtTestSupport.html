17  package org.apache.juneau.rest.server.auth.jwt;
18  
19  import java.time.*;
20  import java.util.*;
21  
22  import com.nimbusds.jose.*;
23  import com.nimbusds.jose.crypto.*;
24  import com.nimbusds.jose.jwk.*;
25  import com.nimbusds.jose.jwk.gen.*;
26  import com.nimbusds.jose.jwk.source.*;
27  import com.nimbusds.jose.proc.*;
28  import com.nimbusds.jwt.*;

30  /**
31   * Static helpers for {@code JwtTokenValidator_*_Test} files. Generates RSA/EC/HMAC keys,
32   * signs JWTs, and wraps a fixed {@link JWKSet} as a {@link JWKSource}.
33   *
34   * <p>
35   * Lives in the test tree so production code never sees these helpers; lives in {@code juneau-integration-tests}
36   * rather than {@code juneau-rest-server-auth-jwt/src/test} to match the centralized-tests convention
37   * (every other juneau-rest module has empty {@code src/test/}; tests live here).
38   */
39  final class JwtTestSupport {
40  
41  	static final String DEFAULT_ISSUER = "https://issuer.example.com/";
42  	static final String DEFAULT_AUDIENCE = "https://api.example.com";
43  
44  	private JwtTestSupport() {}
45  
46  	/** Generate a fresh RSA-2048 JWK with the given kid. */
47  	static RSAKey generateRsa(String kid) throws JOSEException {
48  		return new RSAKeyGenerator(2048)
49  			.keyID(kid)
50  			.algorithm(JWSAlgorithm.RS256)
51  			.generate();
52  	}
53  
54  	/** Generate a fresh EC P-256 JWK with the given kid. */
55  	static ECKey generateEc(String kid) throws JOSEException {
56  		return new ECKeyGenerator(Curve.P_256)
57  			.keyID(kid)
58  			.algorithm(JWSAlgorithm.ES256)
59  			.generate();
60  	}
61  
62  	/** Generate a fresh symmetric HMAC key (256 bits) with the given kid. */
63  	static OctetSequenceKey generateHmac(String kid) throws JOSEException {
64  		return new OctetSequenceKeyGenerator(256)
65  			.keyID(kid)
66  			.algorithm(JWSAlgorithm.HS256)
67  			.generate();
68  	}
69  
70  	/** Wrap a JWK as a fixed (no-network) JWKSource that publishes only the public material. */
71  	static JWKSource<SecurityContext> fixed(JWK... keys) {
72  		var set = new JWKSet(Arrays.asList(keys)).toPublicJWKSet();
73  		return new ImmutableJWKSet<>(set);
74  	}
75  
76  	/** Wrap a list of JWKs (including secret material — for HMAC) as a fixed JWKSource. */
77  	static JWKSource<SecurityContext> fixedRaw(JWK... keys) {
78  		return new ImmutableJWKSet<>(new JWKSet(Arrays.asList(keys)));
79  	}
80  
81  	/** Build a default claims set with all mandatory claims populated. */
82  	static JWTClaimsSet.Builder defaultClaims(Clock clock) {
83  		var now = clock.instant();
84  		return new JWTClaimsSet.Builder()
85  			.subject("alice")
86  			.issuer(DEFAULT_ISSUER)
87  			.audience(DEFAULT_AUDIENCE)
88  			.issueTime(Date.from(now))
89  			.notBeforeTime(Date.from(now))
90  			.expirationTime(Date.from(now.plus(Duration.ofMinutes(5))));
91  	}
92  
93  	/** Sign a claims set with an RSA key (RS256). */
94  	static String signRsa(RSAKey key, JWTClaimsSet claims) throws JOSEException {
95  		var header = new JWSHeader.Builder(JWSAlgorithm.RS256).keyID(key.getKeyID()).build();
96  		var jwt = new SignedJWT(header, claims);
97  		jwt.sign(new RSASSASigner(key));
98  		return jwt.serialize();
99  	}
100 
101 	/** Sign a claims set with an EC key (ES256). */
102 	static String signEc(ECKey key, JWTClaimsSet claims) throws JOSEException {
103 		var header = new JWSHeader.Builder(JWSAlgorithm.ES256).keyID(key.getKeyID()).build();
104 		var jwt = new SignedJWT(header, claims);
105 		jwt.sign(new ECDSASigner(key));
106 		return jwt.serialize();
107 	}
108 
109 	/** Sign a claims set with an HMAC key (HS256). */
110 	static String signHmac(OctetSequenceKey key, JWTClaimsSet claims) throws JOSEException {
111 		var header = new JWSHeader.Builder(JWSAlgorithm.HS256).keyID(key.getKeyID()).build();
112 		var jwt = new SignedJWT(header, claims);
113 		jwt.sign(new MACSigner(key.toByteArray()));
114 		return jwt.serialize();
115 	}
116 
117 	/**
118 	 * Build an HS256-signed JWT using the supplied raw secret bytes. Lets tests construct
119 	 * the algorithm-confusion attack token (HS256 signed with an RSA public key as the
120 	 * HMAC secret).
121 	 */
122 	static String signHmacRaw(byte[] secret, String kid, JWTClaimsSet claims) throws JOSEException {
123 		// MACSigner requires at least 32 bytes of secret for HS256. Real-world algorithm-confusion
124 		// attack tokens commonly use the PEM encoding of an RSA public key, which is >32 bytes; pad if needed.
125 		var bytes = secret;
126 		if (bytes.length < 32) {
127 			var padded = new byte[32];
128 			System.arraycopy(bytes, 0, padded, 0, bytes.length);
129 			bytes = padded;
130 		}
131 		var header = new JWSHeader.Builder(JWSAlgorithm.HS256).keyID(kid).build();
132 		var jwt = new SignedJWT(header, claims);
133 		jwt.sign(new MACSigner(bytes));
134 		return jwt.serialize();
135 	}
136 
137 	/**
138 	 * Build a parseable {@code alg: none} JWT (unsigned). Nimbus' {@code PlainJWT} forbids
139 	 * setting a signed-style header, so we hand-assemble the three Base64URL segments.
140 	 */
141 	static String unsignedToken(JWTClaimsSet claims) {
142 		var header = new com.nimbusds.jose.PlainHeader.Builder().build();
143 		var jwt = new com.nimbusds.jwt.PlainJWT(header, claims);
144 		return jwt.serialize();
145 	}
146 }
