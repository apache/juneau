19  import static org.apache.juneau.commons.utils.Shorts.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.time.*;
23  import java.util.*;
24  
25  import org.apache.juneau.*;
26  import org.apache.juneau.rest.server.auth.*;
27  import org.junit.jupiter.api.*;

29  /**
30   * Tests for {@link InMemorySessionStore} &mdash; server-side indexing + back-channel revocation.
31   *
32   * @since 10.0.0
33   */
34  class InMemorySessionStore_Test extends TestBase {
35  
36  	// Fixed time seam: the store checks expiry against its Clock, so the session base and the
37  	// store clock are both pinned to a constant instead of reading the system clock (java:S8692).
38  	private static final Instant NOW = Instant.parse("2026-01-01T00:00:00Z");
39  	private static final Clock CLOCK = Clock.fixed(NOW, ZoneOffset.UTC);
40  
41  	private static InMemorySessionStore store() {
42  		return new InMemorySessionStore(100, CLOCK);
43  	}
44  
45  	private static OidcSession session(String id, String subject, String sid) {
46  		return session(id, subject, sid, NOW);
47  	}
48  
49  	private static OidcSession session(String id, String subject, String sid, Instant base) {
50  		return new OidcSession(id, subject, o(sid),
51  			new ClaimsPrincipal(subject, Map.of("sub", subject)),
52  			Set.of("user"), oe(), base, base.plus(Duration.ofHours(8)));
53  	}
54  
55  	@Test void a01_createThenLookup_roundTrips() {
56  		var store = store();
57  		var cookie = store.createSessionCookieValue(session("id-1", "alice", "sess-1"));
58  		assertEquals("id-1", cookie);
59  		assertTrue(store.lookup("id-1").isPresent());
60  		assertEquals("alice", store.lookup("id-1").get().subject());
61  	}
62  
63  	@Test void a02_lookupUnknown_isEmpty() {
64  		assertTrue(InMemorySessionStore.create().lookup("nope").isEmpty());
65  	}
66  
67  	@Test void a03_invalidate_removesSession() {
68  		var store = store();
69  		store.createSessionCookieValue(session("id-1", "alice", "sess-1"));
70  		store.invalidate("id-1");
71  		assertTrue(store.lookup("id-1").isEmpty());
72  	}
73  
74  	@Test void b01_supportsServerSideRevocation() {
75  		assertTrue(InMemorySessionStore.create().supportsServerSideRevocation());
76  	}
77  
78  	@Test void b02_invalidateBySessionId_targetsSingleSession() {
79  		var store = store();
80  		store.createSessionCookieValue(session("id-1", "alice", "sess-1"));
81  		store.createSessionCookieValue(session("id-2", "alice", "sess-2"));
82  		assertEquals(1, store.invalidateBySessionId("sess-1"));
83  		assertTrue(store.lookup("id-1").isEmpty());
84  		assertTrue(store.lookup("id-2").isPresent());
85  	}
86  
87  	@Test void b03_invalidateBySubject_targetsAllOfSubject() {
88  		var store = store();
89  		store.createSessionCookieValue(session("id-1", "alice", "sess-1"));
90  		store.createSessionCookieValue(session("id-2", "alice", "sess-2"));
91  		store.createSessionCookieValue(session("id-3", "bob", "sess-3"));
92  		assertEquals(2, store.invalidateBySubject("alice"));
93  		assertTrue(store.lookup("id-1").isEmpty());
94  		assertTrue(store.lookup("id-2").isEmpty());
95  		assertTrue(store.lookup("id-3").isPresent());
96  	}
97  
98  	@Test void b04_invalidateByUnknownSubjectOrSid_returnsZero() {
99  		var store = store();
100 		assertEquals(0, store.invalidateBySubject("ghost"));
101 		assertEquals(0, store.invalidateBySessionId("ghost"));
102 	}
103 
104 	@Test void c01_expiredSession_missedAndRemoved() {
105 		var base = Instant.parse("2026-01-01T00:00:00Z");
106 		var clock = new EphemeralStore_Test.MutableClock(base);
107 		var store = new InMemorySessionStore(100, clock);
108 		store.createSessionCookieValue(session("id-1", "alice", "sess-1", base));
109 		clock.advance(Duration.ofHours(9));
110 		assertTrue(store.lookup("id-1").isEmpty());
111 		assertEquals(0, store.size());
112 	}
113 
114 	@Test void d01_sizeCap_evictsEldestAndUnindexes() {
115 		var store = new InMemorySessionStore(2, CLOCK);
116 		store.createSessionCookieValue(session("id-1", "alice", "sess-1"));
117 		store.createSessionCookieValue(session("id-2", "bob", "sess-2"));
118 		store.createSessionCookieValue(session("id-3", "carol", "sess-3"));  // evicts id-1
119 		assertEquals(2, store.size());
120 		assertTrue(store.lookup("id-1").isEmpty());
121 		assertEquals(0, store.invalidateBySubject("alice"), "evicted session must also be unindexed");
122 	}
123 }
