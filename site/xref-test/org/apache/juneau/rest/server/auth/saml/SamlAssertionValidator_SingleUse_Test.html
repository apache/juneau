17  package org.apache.juneau.rest.server.auth.saml;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.time.*;
22  import java.util.*;
23  
24  import org.apache.juneau.*;
25  import org.apache.juneau.commons.concurrent.*;
26  import org.apache.juneau.rest.server.auth.*;
27  import org.junit.jupiter.api.*;
28  
29  /**
30   * Tests for the one-time-use (replay) enforcement in {@link SamlAssertionValidator} &mdash; a validly-signed
31   * assertion succeeds once and is rejected on any subsequent presentation, and the check is fail-closed.
32   *
33   * @since 10.0.0
34   */
35  class SamlAssertionValidator_SingleUse_Test extends TestBase {
36  
37  	private static final String ISSUER = "https://idp.example.com";
38  	private static final String AUDIENCE = "https://sp.example.com";
39  	private static final Instant NOW = Instant.parse("2026-01-01T00:00:00Z");
40  	private static final Clock CLOCK = Clock.fixed(NOW, ZoneOffset.UTC);
41  
42  	private static SamlAssertionValidator.Builder base(org.opensaml.security.credential.Credential cred) {
43  		return SamlAssertionValidator.create()
44  			.spEntityId(AUDIENCE)
45  			.expectedIssuer(ISSUER)
46  			.signingCredential(cred)
47  			.clock(CLOCK);
48  	}
49  
50  	@Test void a01_firstPresentationSucceeds_secondRejected() throws Exception {
51  		var cred = SamlTestSupport.credential(SamlTestSupport.generateRsaKeyPair());
52  		var xml = SamlTestSupport.buildSignedResponse(cred, ISSUER, AUDIENCE, "alice",
53  			NOW.minusSeconds(60), NOW.plusSeconds(300), Map.of());
54  		var validator = base(cred).build();
55  
56  		assertEquals("alice", validator.validate(xml).getName());
57  		assertThrows(AuthenticationException.class, () -> validator.validate(xml));
58  	}
59  
60  	@Test void a02_separateValidatorInstances_haveIndependentCaches() throws Exception {
61  		var cred = SamlTestSupport.credential(SamlTestSupport.generateRsaKeyPair());
62  		var xml = SamlTestSupport.buildSignedResponse(cred, ISSUER, AUDIENCE, "alice",
63  			NOW.minusSeconds(60), NOW.plusSeconds(300), Map.of());
64  
65  		// The default in-memory cache is per-instance, so a fresh validator accepts the same assertion.
66  		assertEquals("alice", base(cred).build().validate(xml).getName());
67  		assertEquals("alice", base(cred).build().validate(xml).getName());
68  	}
69  
70  	@Test void a03_injectedSharedCache_rejectsAcrossValidators() throws Exception {
71  		var cred = SamlTestSupport.credential(SamlTestSupport.generateRsaKeyPair());
72  		var xml = SamlTestSupport.buildSignedResponse(cred, ISSUER, AUDIENCE, "alice",
73  			NOW.minusSeconds(60), NOW.plusSeconds(300), Map.of());
74  		var shared = new InMemoryReplayCache();
75  
76  		assertEquals("alice", base(cred).replayCache(shared).build().validate(xml).getName());
77  		// A different validator instance sharing the same cache sees the assertion as already consumed.
78  		var v2 = base(cred).replayCache(shared).build();
79  		assertThrows(AuthenticationException.class, () -> v2.validate(xml));
80  	}
81  
82  	@Test void b01_cacheUnavailable_failsClosed() throws Exception {
83  		var cred = SamlTestSupport.credential(SamlTestSupport.generateRsaKeyPair());
84  		var xml = SamlTestSupport.buildSignedResponse(cred, ISSUER, AUDIENCE, "alice",
85  			NOW.minusSeconds(60), NOW.plusSeconds(300), Map.of());
86  		// A cache that cannot answer (throws) must cause the assertion to be rejected, not accepted.
87  		ReplayCache throwing = (id, expiresAtMs) -> { throw new IllegalStateException("store down"); };
88  		var validator = base(cred).replayCache(throwing).build();
89  		assertThrows(AuthenticationException.class, () -> validator.validate(xml));
90  	}
91  
92  	@Test void b02_distinctAssertions_bothAccepted() throws Exception {
93  		var cred = SamlTestSupport.credential(SamlTestSupport.generateRsaKeyPair());
94  		var validator = base(cred).build();
95  		// Two independently-minted assertions carry distinct IDs, so neither is a replay of the other.
96  		var xml1 = SamlTestSupport.buildSignedResponse(cred, ISSUER, AUDIENCE, "alice",
97  			NOW.minusSeconds(60), NOW.plusSeconds(300), Map.of());
98  		var xml2 = SamlTestSupport.buildSignedResponse(cred, ISSUER, AUDIENCE, "bob",
99  			NOW.minusSeconds(60), NOW.plusSeconds(300), Map.of());
100 		assertEquals("alice", validator.validate(xml1).getName());
101 		assertEquals("bob", validator.validate(xml2).getName());
102 	}
103 }
