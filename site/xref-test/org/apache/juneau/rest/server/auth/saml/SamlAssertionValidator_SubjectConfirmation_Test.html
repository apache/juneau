39  	private static final String ISSUER = "https://idp.example.com";
40  	private static final String AUDIENCE = "https://sp.example.com";
41  	private static final String ACS = "https://sp.example.com/saml/acs";
42  	private static final Instant NOW = Instant.parse("2026-01-01T00:00:00Z");
43  	private static final Clock CLOCK = Clock.fixed(NOW, ZoneOffset.UTC);
44  	private static final Instant NBF = NOW.minusSeconds(60);
45  	private static final Instant NOA = NOW.plusSeconds(300);
46  
47  	private static SamlAssertionValidator.Builder base(Credential cred) {
48  		return SamlAssertionValidator.create().recipient(SamlTestSupport.ACS).allowIdpInitiated()
49  			.spEntityId(AUDIENCE)
50  			.expectedIssuer(ISSUER)
51  			.signingCredential(cred)
52  			.clock(CLOCK);
53  	}
54  
55  	private static String signedWithSubject(Credential cred, Subject sub) throws Exception {
56  		var assertion = SamlTestSupport.buildMinimalAssertionWithSubject(ISSUER, AUDIENCE, sub, NBF, NOA);
57  		return SamlTestSupport.signAndBuildResponse((BasicCredential) cred, ISSUER, assertion);
58  	}

274 	@Test void f08_customAuthnRequestStore_usedAndSingleUse() throws Exception {
275 		var cred = SamlTestSupport.credential(SamlTestSupport.generateRsaKeyPair());
276 		var remembered = new ArrayList<String>();
277 		var consumed = new ArrayList<String>();
278 		var ids = new HashSet<String>();
279 		AuthnRequestStore store = new AuthnRequestStore() {
280 			@Override public void remember(String id, long expiresAtMs, long nowMs) {
281 				remembered.add(id);
282 				ids.add(id);
283 			}
284 			@Override public boolean consume(String id, long nowMs) {
285 				consumed.add(id);
286 				return ids.remove(id);
287 			}
288 		};
289 		var v = base(cred).authnRequestStore(store).build();
290 		v.rememberAuthnRequest("_req-10");
291 		assertEquals(List.of("_req-10"), remembered);
292 		var xml1 = signedWithSubject(cred, SamlTestSupport.subjectWithConfirmations("alice", SamlTestSupport.bearerConfirmation(ACS, null, NOA, "_req-10", null)));
293 		assertEquals("alice", v.validate(xml1).getName());
294 		var xml2 = signedWithSubject(cred, SamlTestSupport.subjectWithConfirmations("alice", SamlTestSupport.bearerConfirmation(ACS, null, NOA, "_req-10", null)));
295 		assertThrows(AuthenticationException.class, () -> v.validate(xml2));
296 		assertEquals(List.of("_req-10", "_req-10"), consumed);
297 	}

305 		var store = new InMemoryAuthnRequestStore(2);
306 		store.remember("a", 1000, 0);
307 		store.remember("b", 50, 0);
308 		assertThrows(IllegalStateException.class, () -> store.remember("c", 1000, 10));
309 		store.remember("c", 1000, 60);
310 		assertFalse(store.consume("b", 60));
311 		assertTrue(store.consume("a", 60));
312 		assertFalse(store.consume("a", 60));
313 		assertThrows(IllegalArgumentException.class, () -> new InMemoryAuthnRequestStore(0));
314 	}

352 		var cred = SamlTestSupport.credential(SamlTestSupport.generateRsaKeyPair());
353 		var assertion = SamlTestSupport.buildMinimalAssertion(ISSUER, AUDIENCE, "alice", NBF, NOA);
354 		SamlTestSupport.addAttributeStatement(assertion, "iss", "https://evil.example.com");
355 		SamlTestSupport.addAttributeStatement(assertion, "aud", "other");
356 		SamlTestSupport.addAttributeStatement(assertion, "issuerType", "JWT");
357 		SamlTestSupport.addAttributeStatement(assertion, "dept", "eng");
358 		var xml = SamlTestSupport.signAndBuildResponse((BasicCredential) cred, ISSUER, assertion);
359 		var p = (ClaimsPrincipal) base(cred).build().validate(xml);
360 		assertEquals(ISSUER, p.getClaims().get("iss"));
361 		assertEquals(AUDIENCE, p.getClaims().get("aud"));
362 		assertEquals("SAML", p.getClaims().get("issuerType"));
363 		assertEquals("eng", p.getClaims().get("dept"));
