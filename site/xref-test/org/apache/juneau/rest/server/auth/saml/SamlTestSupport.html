73  	static Issuer issuer(String value) {
74  		var b = (SAMLObjectBuilder<Issuer>) bf().getBuilder(Issuer.DEFAULT_ELEMENT_NAME);
75  		var i = b.buildObject();
76  		i.setValue(value);
77  		return i;

80  	/** Builds a minimal Conditions element with audience restriction but no NotBefore/NotOnOrAfter. */
81  	static Conditions conditions(String audience, Instant notBefore, Instant notOnOrAfter) {
82  		var bf = bf();
83  		var condBuilder = (SAMLObjectBuilder<Conditions>) bf.getBuilder(Conditions.DEFAULT_ELEMENT_NAME);
84  		var arBuilder   = (SAMLObjectBuilder<AudienceRestriction>) bf.getBuilder(AudienceRestriction.DEFAULT_ELEMENT_NAME);
85  		var audBuilder  = (SAMLObjectBuilder<Audience>) bf.getBuilder(Audience.DEFAULT_ELEMENT_NAME);
86  		var conditions = condBuilder.buildObject();
87  		if (notBefore != null)
88  			conditions.setNotBefore(notBefore);
89  		if (notOnOrAfter != null)
90  			conditions.setNotOnOrAfter(notOnOrAfter);
91  		var ar = arBuilder.buildObject();
92  		var aud = audBuilder.buildObject();
93  		aud.setURI(audience);
94  		ar.getAudiences().add(aud);
95  		conditions.getAudienceRestrictions().add(ar);
96  		return conditions;
97  	}

99  	/** Builds a Subject with the given NameID value, or null subject if subjectName is null. */
100 	static Subject subject(String subjectName) {
101 		if (subjectName == null)
102 			return null;
103 		var bf = bf();
104 		var subBuilder = (SAMLObjectBuilder<Subject>) bf.getBuilder(Subject.DEFAULT_ELEMENT_NAME);
105 		var nidBuilder = (SAMLObjectBuilder<NameID>)  bf.getBuilder(NameID.DEFAULT_ELEMENT_NAME);
106 		var sub = subBuilder.buildObject();
107 		var nid = nidBuilder.buildObject();
108 		nid.setValue(subjectName);
109 		nid.setFormat(NameIDType.UNSPECIFIED);
110 		sub.setNameID(nid);
111 		return sub;
112 	}

114 	/** Builds a bearer {@code <SubjectConfirmation>} carrying a {@code <SubjectConfirmationData>}. */
115 	static SubjectConfirmation bearerConfirmation(String recipient, Instant notBefore, Instant notOnOrAfter,
116 			String inResponseTo, String address) {
117 		var bf = bf();
118 		var scb = (SAMLObjectBuilder<SubjectConfirmation>) bf.getBuilder(SubjectConfirmation.DEFAULT_ELEMENT_NAME);
119 		var scdb = (SAMLObjectBuilder<SubjectConfirmationData>) bf.getBuilder(SubjectConfirmationData.DEFAULT_ELEMENT_NAME);
120 		var sc = scb.buildObject();
121 		sc.setMethod(SubjectConfirmation.METHOD_BEARER);
122 		var scd = scdb.buildObject();
123 		if (recipient != null)
124 			scd.setRecipient(recipient);
125 		if (notBefore != null)
126 			scd.setNotBefore(notBefore);
127 		if (notOnOrAfter != null)
128 			scd.setNotOnOrAfter(notOnOrAfter);
129 		if (inResponseTo != null)
130 			scd.setInResponseTo(inResponseTo);
131 		if (address != null)
132 			scd.setAddress(address);
133 		sc.setSubjectConfirmationData(scd);
134 		return sc;
135 	}

165 	/**
166 	 * Builds an Assertion, signs it, wraps it in a Response, and serializes.
167 	 * responseIssuer may be null to omit the response-level {@code <Issuer>} element.
168 	 * The assertion is taken as-is; callers configure subject/conditions/attributes before passing in.
169 	 */
170 	static String signAndBuildResponse(BasicCredential signingCred, String responseIssuer, Assertion assertion) throws Exception {
171 		var bf = bf();
172 		var responseBuilder = (SAMLObjectBuilder<Response>) bf.getBuilder(Response.DEFAULT_ELEMENT_NAME);
173 		var response = responseBuilder.buildObject();
174 		response.setID("_resp-" + System.nanoTime());
175 		response.setIssueInstant(ISSUE_INSTANT);
176 		response.setVersion(org.opensaml.saml.common.SAMLVersion.VERSION_20);
177 		if (responseIssuer != null)
178 			response.setIssuer(buildIssuer(bf, responseIssuer));
179 		response.setStatus(buildStatus(bf));
180 
181 		var sig = new SignatureBuilder().buildObject();
182 		sig.setSigningCredential(signingCred);
183 		sig.setSignatureAlgorithm(SignatureConstants.ALGO_ID_SIGNATURE_RSA_SHA256);
184 		sig.setCanonicalizationAlgorithm(SignatureConstants.ALGO_ID_C14N_EXCL_OMIT_COMMENTS);
185 		assertion.setSignature(sig);
186 
187 		response.getAssertions().add(assertion);
188 
189 		MarshallerFactory mf = XMLObjectProviderRegistrySupport.getMarshallerFactory();
190 		var dom = mf.getMarshaller(response).marshall(response);
191 		Signer.signObject(sig);
192 
193 		return serialize(dom);
194 	}

269 	/**
270 	 * Appends an AttributeStatement with a single Attribute carrying the given values to an existing Assertion.
271 	 * Pass {@code name=null} to produce a nameless attribute (tests the null-name-skip branch in buildClaims).
272 	 */
273 	static void addAttributeStatement(Assertion assertion, String name, String... values) {
274 		var bf = bf();
275 		var asBuilder = (SAMLObjectBuilder<AttributeStatement>) bf.getBuilder(AttributeStatement.DEFAULT_ELEMENT_NAME);
276 		var attrBuilder = (SAMLObjectBuilder<Attribute>) bf.getBuilder(Attribute.DEFAULT_ELEMENT_NAME);
277 		var stringBuilder = (org.opensaml.core.xml.schema.impl.XSStringBuilder)
278 			bf.getBuilder(org.opensaml.core.xml.schema.XSString.TYPE_NAME);
279 		var stmt = asBuilder.buildObject();
280 		var attr = attrBuilder.buildObject();
281 		attr.setName(name);
282 		for (var v : values) {
283 			var av = stringBuilder.buildObject(AttributeValue.DEFAULT_ELEMENT_NAME,
284 				org.opensaml.core.xml.schema.XSString.TYPE_NAME);
285 			av.setValue(v);
286 			attr.getAttributeValues().add(av);
287 		}
288 		stmt.getAttributes().add(attr);
289 		assertion.getAttributeStatements().add(stmt);
290 	}

292 	private static String serialize(org.w3c.dom.Element dom) throws Exception {
293 		var sw = new StringWriter();
294 		var tf = TransformerFactory.newInstance();
295 		var transformer = tf.newTransformer();
296 		transformer.setOutputProperty(OutputKeys.OMIT_XML_DECLARATION, "yes");
297 		transformer.transform(new DOMSource(dom), new StreamResult(sw));
298 		return sw.toString();
299 	}

313 	/**
314 	 * Builds a minimal signed {@code <samlp:Response>} carrying a single {@code <Assertion>} with the
315 	 * supplied subject, issuer, audience, and notBefore/notOnOrAfter window.
316 	 */
317 	static String buildSignedResponse(
318 			BasicCredential signingCred,
319 			String issuer,
320 			String audience,
321 			String subjectName,
322 			Instant notBefore,
323 			Instant notOnOrAfter,
324 			java.util.Map<String,String> attributes) throws Exception {
325 
326 		var bf = XMLObjectProviderRegistrySupport.getBuilderFactory();
327 		var responseBuilder = (SAMLObjectBuilder<Response>) bf.getBuilder(Response.DEFAULT_ELEMENT_NAME);
328 		var response = responseBuilder.buildObject();
329 		response.setID("_resp-" + System.nanoTime());
330 		response.setIssueInstant(ISSUE_INSTANT);
331 		response.setVersion(org.opensaml.saml.common.SAMLVersion.VERSION_20);
332 
333 		response.setIssuer(buildIssuer(bf, issuer));
334 		response.setStatus(buildStatus(bf));
335 
336 		var assertion = buildAssertion(bf, issuer, audience, subjectName, notBefore, notOnOrAfter, attributes);
337 
338 		// Build a Signature object and attach to the assertion.
339 		var sig = new SignatureBuilder().buildObject();
340 		sig.setSigningCredential(signingCred);
341 		sig.setSignatureAlgorithm(SignatureConstants.ALGO_ID_SIGNATURE_RSA_SHA256);
342 		sig.setCanonicalizationAlgorithm(SignatureConstants.ALGO_ID_C14N_EXCL_OMIT_COMMENTS);
343 		assertion.setSignature(sig);
344 
345 		response.getAssertions().add(assertion);
346 
347 		// Marshal first so the Signature element is bound to the DOM; then sign.
348 		MarshallerFactory mf = XMLObjectProviderRegistrySupport.getMarshallerFactory();
349 		var mar = mf.getMarshaller(response);
350 		var dom = mar.marshall(response);
351 		Signer.signObject(sig);
352 
353 		// Serialize to XML string.
354 		var sw = new StringWriter();
355 		var tf = TransformerFactory.newInstance();
356 		var transformer = tf.newTransformer();
357 		transformer.setOutputProperty(OutputKeys.OMIT_XML_DECLARATION, "yes");
358 		transformer.transform(new DOMSource(dom), new StreamResult(sw));
359 		return sw.toString();
360 	}

379 	private static Assertion buildAssertion(
380 			org.opensaml.core.xml.XMLObjectBuilderFactory bf,
381 			String issuer, String audience, String subjectName,
382 			Instant notBefore, Instant notOnOrAfter,
383 			java.util.Map<String,String> attributes) {
384 		var ab = (SAMLObjectBuilder<Assertion>) bf.getBuilder(Assertion.DEFAULT_ELEMENT_NAME);
385 		var assertion = ab.buildObject();
386 		assertion.setID("_a-" + System.nanoTime());
387 		assertion.setIssueInstant(ISSUE_INSTANT);
388 		assertion.setVersion(org.opensaml.saml.common.SAMLVersion.VERSION_20);
389 		assertion.setIssuer(buildIssuer(bf, issuer));
390 
391 		var subBuilder = (SAMLObjectBuilder<Subject>) bf.getBuilder(Subject.DEFAULT_ELEMENT_NAME);
392 		var nidBuilder = (SAMLObjectBuilder<NameID>) bf.getBuilder(NameID.DEFAULT_ELEMENT_NAME);
393 		var sub = subBuilder.buildObject();
394 		var nid = nidBuilder.buildObject();
395 		nid.setValue(subjectName);
396 		nid.setFormat(NameIDType.UNSPECIFIED);
397 		sub.setNameID(nid);
398 		assertion.setSubject(sub);
399 
400 		var condBuilder = (SAMLObjectBuilder<Conditions>) bf.getBuilder(Conditions.DEFAULT_ELEMENT_NAME);
401 		var arBuilder = (SAMLObjectBuilder<AudienceRestriction>) bf.getBuilder(AudienceRestriction.DEFAULT_ELEMENT_NAME);
402 		var audBuilder = (SAMLObjectBuilder<Audience>) bf.getBuilder(Audience.DEFAULT_ELEMENT_NAME);
403 		var conditions = condBuilder.buildObject();
404 		conditions.setNotBefore(notBefore);
405 		conditions.setNotOnOrAfter(notOnOrAfter);
406 		var ar = arBuilder.buildObject();
407 		var aud = audBuilder.buildObject();
408 		aud.setURI(audience);
409 		ar.getAudiences().add(aud);
410 		conditions.getAudienceRestrictions().add(ar);
411 		assertion.setConditions(conditions);
412 
413 		if (attributes != null && !attributes.isEmpty()) {
414 			var asBuilder = (SAMLObjectBuilder<AttributeStatement>) bf.getBuilder(AttributeStatement.DEFAULT_ELEMENT_NAME);
415 			var attrBuilder = (SAMLObjectBuilder<Attribute>) bf.getBuilder(Attribute.DEFAULT_ELEMENT_NAME);
416 			var stringBuilder = (org.opensaml.core.xml.schema.impl.XSStringBuilder)
417 				bf.getBuilder(org.opensaml.core.xml.schema.XSString.TYPE_NAME);
418 			var stmt = asBuilder.buildObject();
419 			for (var e : attributes.entrySet()) {
420 				var attr = attrBuilder.buildObject();
421 				attr.setName(e.getKey());
422 				var av = stringBuilder.buildObject(AttributeValue.DEFAULT_ELEMENT_NAME,
423 					org.opensaml.core.xml.schema.XSString.TYPE_NAME);
424 				av.setValue(e.getValue());
425 				attr.getAttributeValues().add(av);
426 				stmt.getAttributes().add(attr);
427 			}
428 			assertion.getAttributeStatements().add(stmt);
429 		}
430 		return assertion;
431 	}
