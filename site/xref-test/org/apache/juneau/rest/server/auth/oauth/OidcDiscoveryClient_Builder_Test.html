17  package org.apache.juneau.rest.server.auth.oauth;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.net.*;
22  import java.util.*;
23  
24  import org.apache.juneau.*;
25  import org.apache.juneau.rest.auth.oauth.oidc.*;
26  import org.junit.jupiter.api.*;
27  
28  /**
29   * Builder tests for {@link OidcDiscoveryClient} and the {@link OidcMetadata} record's defensive copies.
30   *
31   * @since 10.0.0
32   */
33  @SuppressWarnings({
34  	"java:S5778"  // assertThrows lambdas with chained calls; intermediate invocations do not throw in practice
35  })
36  class OidcDiscoveryClient_Builder_Test extends TestBase {
37  
38  	@Test void a01_requireIssuer() {
39  		assertThrows(IllegalStateException.class, () -> OidcDiscoveryClient.create().build());
40  	}
41  
42  	@Test void a02_build_happyPath() {
43  		var c = OidcDiscoveryClient.create()
44  			.issuer(URI.create("https://idp.example.com"))
45  			.build();
46  		assertEquals(URI.create("https://idp.example.com"), c.getIssuer());
47  	}
48  
49  	@Test void b01_metadata_record_defensiveCopy() {
50  		var supportedScopes = new HashSet<>(Set.of("openid", "profile"));
51  		var extras = new HashMap<String,Object>();
52  		extras.put("foo", "bar");
53  		var md = new OidcMetadata(
54  			URI.create("https://idp.example.com"),
55  			URI.create("https://idp.example.com/token"),
56  			null, null, null, null, null,
57  			supportedScopes, extras);
58  		assertEquals(Set.of("openid", "profile"), md.supportedScopes());
59  		assertEquals(Map.of("foo", "bar"), md.extras());
60  		assertThrows(UnsupportedOperationException.class, () -> md.supportedScopes().add("z"));
61  		assertThrows(UnsupportedOperationException.class, () -> md.extras().put("a", "b"));
62  	}
63  }
