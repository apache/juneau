50  	/** Creates a mock request for the given URI path. Context path is empty. */
51  	private static HttpServletRequest req(String path) {
52  		var req = mock(HttpServletRequest.class);
53  		when(req.getRequestURI()).thenReturn(path);
54  		when(req.getContextPath()).thenReturn("");
55  		return req;
56  	}
57  
58  	/** Captures the request passed to chain.doFilter() so tests can assert on it. */
59  	private static class CapturingChain implements FilterChain {
60  		ServletRequest captured;
61  
62  		@Override
63  		public void doFilter(ServletRequest req, ServletResponse resp) {
64  			captured = req;
65  		}
66  	}
67  
68  	/** Writable response that captures status and body. */
69  	private static HttpServletResponse capturingResponse() throws IOException {
70  		var resp = mock(HttpServletResponse.class);
71  		var sw = new StringWriter();
72  		when(resp.getWriter()).thenReturn(new PrintWriter(sw));
73  		return resp;
74  	}
75  
76  	/** An AuthFilter that always returns the given AuthResult. */
77  	private static AuthFilter succeeds(Principal principal, String... roles) {
78  		return new AuthFilter() {
79  			@Override public Optional<AuthResult> authenticate(HttpServletRequest req) {
80  				return o(AuthResult.of(principal, roles));
81  			}
82  		};
83  	}
84  
85  	/** An AuthFilter that always returns empty (doesn't apply). */
86  	private static AuthFilter empty() {
87  		return new AuthFilter() {
88  			@Override public Optional<AuthResult> authenticate(HttpServletRequest req) {
89  				return oe();
90  			}
91  		};
92  	}
93  
94  	/** An AuthFilter that always throws AuthenticationException with the given message. */
95  	private static AuthFilter fails(String message) {
96  		return new AuthFilter() {
97  			@Override public Optional<AuthResult> authenticate(HttpServletRequest req) throws AuthenticationException {
98  				throw new AuthenticationException(message).wwwAuthenticate("Bearer realm=\"test\"");
99  			}
100 		};
101 	}

107 	@Test void a01_noMatchingFilters_passThroughUnchanged() throws Exception {
108 		var original = req("/public/page");
109 		var chain = AuthFilterChain.create(null)
110 			.append(succeeds(ALICE), "/api/*")
111 			.build();
112 		var capturing = new CapturingChain();
113 		chain.doFilter(original, capturingResponse(), capturing);
114 		// No filter matched /public/page — the original request passed through
115 		assertSame(original, capturing.captured);
116 	}

118 	@Test void a02_singleFilterSuccess_wrapsRequest() throws Exception {
119 		var chain = AuthFilterChain.create(null)
120 			.append(succeeds(ALICE, "user"))
121 			.build();
122 		var capturing = new CapturingChain();
123 		chain.doFilter(req("/anything"), capturingResponse(), capturing);
124 		assertInstanceOf(AuthenticatedRequestWrapper.class, capturing.captured);
125 		var w = (AuthenticatedRequestWrapper) capturing.captured;
126 		assertSame(ALICE, w.getUserPrincipal());
127 		assertTrue(w.isUserInRole("user"));
128 	}

130 	@Test void a03_patternMatch_onlyMatchingFilterRuns() throws Exception {
131 		var called = new AtomicBoolean(false);
132 		AuthFilter shouldNotRun = new AuthFilter() {
133 			@Override public Optional<AuthResult> authenticate(HttpServletRequest req) {
134 				called.set(true);
135 				return oe();
136 			}
137 		};
138 		var chain = AuthFilterChain.create(null)
139 			.append(shouldNotRun, "/api/*")
140 			.append(succeeds(ALICE, "user"), "/public/*")
141 			.build();
142 		var capturing = new CapturingChain();
143 		chain.doFilter(req("/public/page"), capturingResponse(), capturing);
144 		assertFalse(called.get(), "Filter for /api/* must not be called for /public/page");
145 		assertInstanceOf(AuthenticatedRequestWrapper.class, capturing.captured);
146 	}

149 		var chain = AuthFilterChain.create(null)
150 			.append(succeeds(ALICE, "user"))
151 			.append(succeeds(BOB, "admin"))
152 			.build();
153 		var capturing = new CapturingChain();

161 		var chain = AuthFilterChain.create(null)

163 			.append(succeeds(BOB, "admin", "billing"))

165 		var capturing = new CapturingChain();

247 	@Test void a12_wwwAuthenticateAggregatedOnAllFailure() throws Exception {
248 		var resp = mock(HttpServletResponse.class);
249 		var sw = new StringWriter();
250 		when(resp.getWriter()).thenReturn(new PrintWriter(sw));
251 
252 		AuthFilter f1 = new AuthFilter() {
253 			@Override public Optional<AuthResult> authenticate(HttpServletRequest req) throws AuthenticationException {
254 				throw new AuthenticationException("msg1").wwwAuthenticate("Bearer realm=\"r1\"");
255 			}
256 		};
257 		AuthFilter f2 = new AuthFilter() {
258 			@Override public Optional<AuthResult> authenticate(HttpServletRequest req) throws AuthenticationException {
259 				throw new AuthenticationException("msg2").wwwAuthenticate("ApiKey realm=\"r2\"");
260 			}
261 		};
262 
263 		var chain = AuthFilterChain.create(null).append(f1).append(f2).build();
264 		var capturing = new CapturingChain();
265 		chain.doFilter(req("/"), resp, capturing);
266 		verify(resp).setStatus(401);
267 		// Both WWW-Authenticate challenges must appear in the aggregated header
268 		var wwwCaptor = org.mockito.ArgumentCaptor.forClass(String.class);
269 		verify(resp).setHeader(eq("WWW-Authenticate"), wwwCaptor.capture());
270 		var headerValue = wwwCaptor.getValue();
271 		assertTrue(headerValue.contains("Bearer realm=\"r1\""), "Missing Bearer challenge: " + headerValue);
272 		assertTrue(headerValue.contains("ApiKey realm=\"r2\""), "Missing ApiKey challenge: " + headerValue);
273 	}
