45  	private static SamlAssertionValidator validator(java.util.function.Function<String,Principal> impl) throws Exception {
46  		var kp = KeyPairGenerator.getInstance("RSA");
47  		kp.initialize(2048);
48  		var pair = kp.generateKeyPair();
49  		return new SamlAssertionValidator(
50  				SamlAssertionValidator.create()
51  					.spEntityId("https://sp.example.com")
52  					.expectedIssuer("https://idp.example.com")
53  					.signingCredential(new BasicCredential(pair.getPublic(), pair.getPrivate()))) {
54  			@Override public Principal validate(String xml) {
55  				return impl.apply(xml);
56  			}
57  		};
58  	}

60  	private static HttpServletRequest req(String samlResponse) {
61  		var r = mock(HttpServletRequest.class);
62  		when(r.getPathInfo()).thenReturn("/saml/acs");
63  		when(r.getServletPath()).thenReturn("/saml/acs");
64  		when(r.getParameter("SAMLResponse")).thenReturn(samlResponse);
65  		return r;
66  	}

68  	/** Raw-DEFLATE (nowrap) + base64, matching the SAML REDIRECT binding encoding. */
69  	private static String deflateBase64(byte[] raw) {
70  		var deflater = new Deflater(Deflater.BEST_COMPRESSION, true);
71  		deflater.setInput(raw);
72  		deflater.finish();
73  		var out = new java.io.ByteArrayOutputStream();
74  		var buf = new byte[8192];
75  		while (!deflater.finished()) {
76  			int n = deflater.deflate(buf);
77  			out.write(buf, 0, n);
78  		}
79  		deflater.end();
80  		return Base64.getEncoder().encodeToString(out.toByteArray());
81  	}

83  	@Test void a01_highlyCompressiblePayload_exceedsDefaultCap_rejected() throws Exception {
84  		// 2 MiB of zeros deflates to a few KB; inflating exceeds the 1 MiB default cap → rejected.
85  		var bomb = new byte[2 * 1024 * 1024];
86  		var b64 = deflateBase64(bomb);
87  		var f = SamlAuthFilter.create().binding(SamlBinding.REDIRECT)
88  			.validator(validator(x -> new ClaimsPrincipal("x", Map.of())))
89  			.build();
90  		assertThrows(AuthenticationException.class, () -> f.authenticate(req(b64)));
91  	}
92  
93  	@Test void a02_customLowByteCap_rejectsModeratePayload() throws Exception {
94  		// 64 KiB of zeros inflates well past a 4 KiB cap → rejected.
95  		var payload = new byte[64 * 1024];
96  		var b64 = deflateBase64(payload);
97  		var f = SamlAuthFilter.create().binding(SamlBinding.REDIRECT)
98  			.maxInflatedBytes(4 * 1024)
99  			.validator(validator(x -> new ClaimsPrincipal("x", Map.of())))
100 			.build();
101 		assertThrows(AuthenticationException.class, () -> f.authenticate(req(b64)));
102 	}
103 
104 	@Test void a03_ratioGuard_rejectsEvenUnderByteCap() throws Exception {
105 		// 256 KiB of zeros deflates to well under 2 KiB; a ratio cap of 5 trips long before the byte cap.
106 		var payload = new byte[256 * 1024];
107 		var b64 = deflateBase64(payload);
108 		var f = SamlAuthFilter.create().binding(SamlBinding.REDIRECT)
109 			.maxInflatedBytes(0)   // disable absolute cap; rely solely on ratio guard
110 			.maxInflateRatio(5)
111 			.validator(validator(x -> new ClaimsPrincipal("x", Map.of())))
112 			.build();
113 		assertThrows(AuthenticationException.class, () -> f.authenticate(req(b64)));
114 	}
115 
116 	@Test void b01_legitimateSmallPayload_stillDecodes() throws Exception {
117 		var xml = "<samlp:Response>ok</samlp:Response>";
118 		var b64 = deflateBase64(xml.getBytes(StandardCharsets.UTF_8));
119 		var received = new String[1];
120 		var f = SamlAuthFilter.create().binding(SamlBinding.REDIRECT)
121 			.validator(validator(x -> { received[0] = x; return new ClaimsPrincipal("alice", Map.of()); }))
122 			.build();
123 		var result = f.authenticate(req(b64));
124 		assertTrue(result.isPresent());
125 		assertEquals(xml, received[0]);
126 	}
127 
128 	@Test void b02_payloadAtCustomCapBoundary_accepted() throws Exception {
129 		// A small payload comfortably under a generous custom cap is accepted.
130 		var xml = "<r>" + "a".repeat(500) + "</r>";
131 		var b64 = deflateBase64(xml.getBytes(StandardCharsets.UTF_8));
132 		var received = new String[1];
133 		var f = SamlAuthFilter.create().binding(SamlBinding.REDIRECT)
134 			.maxInflatedBytes(64 * 1024)
135 			.maxInflateRatio(0)  // disable ratio guard for this boundary check
136 			.validator(validator(x -> { received[0] = x; return new ClaimsPrincipal("alice", Map.of()); }))
137 			.build();
138 		var result = f.authenticate(req(b64));
139 		assertTrue(result.isPresent());
140 		assertEquals(xml, received[0]);
141 	}
