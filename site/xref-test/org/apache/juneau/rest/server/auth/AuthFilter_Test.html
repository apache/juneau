37   * (success / empty / failure), {@code init}/{@code destroy} no-ops, and the static

44  	private static final Principal ALICE = () -> "alice";
45  
46  	/**
47  	 * Concrete subclass driven by a function so individual tests can program the
48  	 * three-state behaviour (empty / present / throw).
49  	 */
50  	private static final class TestFilter extends AuthFilter {
51  		interface AuthFn {
52  			Optional<AuthResult> apply(HttpServletRequest req) throws AuthenticationException;
53  		}
54  		private final AuthFn fn;
55  		TestFilter(AuthFn fn) { this.fn = fn; }
56  		@Override public Optional<AuthResult> authenticate(HttpServletRequest req) throws AuthenticationException {
57  			return fn.apply(req);
58  		}
59  	}
60  
61  	/** Inline {@link FilterChain} that records whether/with-what it was invoked. */
62  	private static final class RecordingChain implements FilterChain {
63  		ServletRequest req;
64  		ServletResponse resp;
65  		int calls;
66  		IOException ioToThrow;
67  		ServletException seToThrow;
68  		@Override public void doFilter(ServletRequest req, ServletResponse resp) throws IOException, ServletException {
69  			this.req = req;
70  			this.resp = resp;
71  			this.calls++;
72  			if (ioToThrow != null) throw ioToThrow;
73  			if (seToThrow != null) throw seToThrow;
74  		}
75  	}

104 	@Test void b01_doFilter_authPresent_wrapsRequest_andCallsChain() throws Exception {
105 		var f = new TestFilter(req -> o(AuthResult.of(ALICE, "user")));
106 		var req = MockServletRequest.create("GET", "/x");
107 		var resp = MockServletResponse.create();
108 		var chain = new RecordingChain();
109 
110 		f.doFilter(req, resp, chain);
111 
112 		assertEquals(1, chain.calls);
113 		// Request passed downstream must be the AuthenticatedRequestWrapper, not the raw mock.
114 		assertInstanceOf(AuthenticatedRequestWrapper.class, chain.req);
115 		var wrapper = (AuthenticatedRequestWrapper) chain.req;
116 		assertSame(ALICE, wrapper.getUserPrincipal());
117 		assertTrue(wrapper.isUserInRole("user"));
118 		// Response object is forwarded unchanged.
119 		assertSame(resp, chain.resp);
120 		// No 401 written.
121 		assertEquals(0, resp.getStatus());
122 	}

147 	@Test void d01_doFilter_authException_writes401WithChallenge() throws Exception {
148 		var f = new TestFilter(req -> {
149 			throw new AuthenticationException("bad token").wwwAuthenticate("Bearer realm=\"api\"");
150 		});
151 		var req = MockServletRequest.create("GET", "/x");
152 		var resp = MockServletResponse.create();
153 		var chain = new RecordingChain();
154 
155 		f.doFilter(req, resp, chain);
156 
157 		// chain must NOT be invoked when authentication fails
158 		assertEquals(0, chain.calls);
159 		assertEquals(HttpServletResponse.SC_UNAUTHORIZED, resp.getStatus());
160 		assertEquals("Bearer realm=\"api\"", resp.getHeader("WWW-Authenticate"));
161 		// Note: the writer.write(msg) line is exercised here (msg is non-null), but
162 		// MockServletResponse#getWriter() returns a fresh PrintWriter each call wrapped around an
163 		// OutputStreamWriter; sendChallenge does not flush the writer, so bytes never reach the
164 		// underlying baos and getContent() stays empty. Coverage of the branch is what matters.
165 	}

198 	@Test void d04_sendChallenge_writesGenericBody_notExceptionMessage() throws Exception {
199 		var resp = mock(HttpServletResponse.class);
200 		var sw = new StringWriter();
201 		when(resp.getWriter()).thenReturn(new PrintWriter(sw));
202 		// Sensitive detail that must NOT be echoed back to the client.
203 		var e = new AuthenticationException("token signature invalid: secret-key-id=abc123")
204 			.wwwAuthenticate("Bearer realm=\"api\"");
205 
206 		AuthFilter.sendChallenge(resp, e);
207 
208 		verify(resp).setStatus(HttpServletResponse.SC_UNAUTHORIZED);
209 		verify(resp).setHeader("WWW-Authenticate", "Bearer realm=\"api\"");
210 		var body = sw.toString();
211 		assertEquals("Unauthorized", body);
212 		assertFalse(body.contains("secret-key-id"), "Auth-failure body must not leak the exception message");
213 	}
214 
215 	@Test void d05_sendChallenge_wrappedCauseNotLeaked() throws Exception {
216 		var resp = mock(HttpServletResponse.class);
217 		var sw = new StringWriter();
218 		when(resp.getWriter()).thenReturn(new PrintWriter(sw));
219 		// A custom validator can wrap an internal exception whose message must not reach the client.
220 		var e = new AuthenticationException(new IllegalStateException("jdbc://internal-host:5432 unreachable"));
221 
222 		AuthFilter.sendChallenge(resp, e);
223 
224 		verify(resp).setStatus(HttpServletResponse.SC_UNAUTHORIZED);
225 		var body = sw.toString();
226 		assertEquals("Unauthorized", body);
227 		assertFalse(body.contains("internal-host"), "Wrapped cause message must not leak to the client");
228 	}
