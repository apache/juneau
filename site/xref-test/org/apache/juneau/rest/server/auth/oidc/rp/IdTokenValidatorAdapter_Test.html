17  package org.apache.juneau.rest.server.auth.oidc.rp;
18  
19  import static org.apache.juneau.rest.server.auth.oidc.rp.OidcTestSupport.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.net.*;
23  import java.time.*;
24  
25  import org.apache.juneau.*;
26  import org.apache.juneau.rest.server.auth.*;
27  import org.junit.jupiter.api.*;
28  
29  import com.nimbusds.jose.*;
30  import com.nimbusds.jose.jwk.*;
31  
32  /**
33   * Tests for {@link IdTokenValidatorAdapter} &mdash; OIDC ID-token signature + claim validation via Nimbus.
34   *
35   * @since 10.0.0
36   */
37  @SuppressWarnings({
38  	"java:S5778"  // assertThrows lambdas with chained calls; intermediate invocations do not throw in practice
39  })
40  class IdTokenValidatorAdapter_Test extends TestBase {
41  
42  	private static final String ISS = "https://stub-idp.example.com";
43  	private static final String CID = "web-app";
44  
45  	private RSAKey key;
46  	private Instant now;
47  
48  	@SuppressWarnings({
49  		"java:S8692" // Nimbus oauth2-oidc-sdk 11.37.2 exposes no clock hook on the ID-token path (IDTokenClaimsVerifier reads new Date() internally), so injecting a clock there would require cloning Nimbus internals; tokens are minted at real "now" and validated against the real system clock.
50  	})
51  	@BeforeEach void setup() throws Exception {
52  		key = generateRsa("k1");
53  		now = Instant.now();  // validation uses the real system clock, so anchor tokens to "now"
54  	}
55  
56  	private IdTokenValidatorAdapter validator() {
57  		return IdTokenValidatorAdapter.create().issuer(ISS).clientId(CID).jwkSet(publicJwks(key)).build();
58  	}
59  
60  	@Test void a01_validToken_projectsClaims() throws Exception {
61  		var jwt = signIdToken(key, ISS, CID, "alice", "sess-1", "nonce-1", now, Duration.ofMinutes(5), null);
62  		var p = validator().validate(jwt, "nonce-1");
63  		assertEquals("alice", p.getName());
64  		assertEquals("sess-1", p.getClaim("sid", String.class).orElse(null));
65  	}
66  
67  	@Test void b01_wrongNonce_rejected() throws Exception {
68  		var jwt = signIdToken(key, ISS, CID, "alice", null, "nonce-1", now, Duration.ofMinutes(5), null);
69  		assertThrows(AuthenticationException.class, () -> validator().validate(jwt, "different-nonce"));
70  	}
71  
72  	@Test void b02_wrongIssuer_rejected() throws Exception {
73  		var jwt = signIdToken(key, "https://evil.example.com", CID, "alice", null, "n", now, Duration.ofMinutes(5), null);
74  		assertThrows(AuthenticationException.class, () -> validator().validate(jwt, "n"));
75  	}
76  
77  	@Test void b03_wrongAudience_rejected() throws Exception {
78  		var jwt = signIdToken(key, ISS, "some-other-client", "alice", null, "n", now, Duration.ofMinutes(5), null);
79  		assertThrows(AuthenticationException.class, () -> validator().validate(jwt, "n"));
80  	}
81  
82  	@Test void b04_expiredToken_rejected() throws Exception {
83  		var past = now.minus(Duration.ofHours(1));
84  		var jwt = signIdToken(key, ISS, CID, "alice", null, "n", past, Duration.ofMinutes(5), null);
85  		assertThrows(AuthenticationException.class, () -> validator().validate(jwt, "n"));
86  	}
87  
88  	@Test void b05_wrongSigningKey_rejected() throws Exception {
89  		var attacker = generateRsa("k1");  // same kid, different key material
90  		var jwt = signIdToken(attacker, ISS, CID, "alice", null, "n", now, Duration.ofMinutes(5), null);
91  		assertThrows(AuthenticationException.class, () -> validator().validate(jwt, "n"));
92  	}
93  
94  	@Test void b06_garbageToken_rejected() {
95  		assertThrows(AuthenticationException.class, () -> validator().validate("not-a-jwt", "n"));
96  	}
97  
98  	@Test void c01_noneAlgorithm_rejectedAtBuild() {
99  		assertThrows(IllegalArgumentException.class,
100 			() -> IdTokenValidatorAdapter.create().algorithms(new JWSAlgorithm("none")));
101 	}
102 
103 	@Test void c02_requiresExactlyOneJwkSource() {
104 		assertThrows(IllegalStateException.class,
105 			() -> IdTokenValidatorAdapter.create().issuer(ISS).clientId(CID).build());
106 		assertThrows(IllegalStateException.class,
107 			() -> IdTokenValidatorAdapter.create().issuer(ISS).clientId(CID)
108 				.jwkSet(publicJwks(key)).jwksUri(URI.create("https://idp/jwks")).build());
109 	}
110 
111 	@Test void c03_requiresIssuerAndClientId() {
112 		assertThrows(IllegalStateException.class,
113 			() -> IdTokenValidatorAdapter.create().clientId(CID).jwkSet(publicJwks(key)).build());
114 		assertThrows(IllegalStateException.class,
115 			() -> IdTokenValidatorAdapter.create().issuer(ISS).jwkSet(publicJwks(key)).build());
116 	}
117 }
