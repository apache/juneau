17  package org.apache.juneau.rest.server.auth;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.security.*;
22  import java.util.*;
23  
24  import org.apache.juneau.*;
25  import org.junit.jupiter.api.*;
26  
27  /**
28   * Tests for {@link AuthResult} factory methods, getters, and defensive-copy semantics.
29   *
30   * @since 10.0.0
31   */
32  @SuppressWarnings({
33  	"java:S5778"  // assertThrows lambdas with chained calls; intermediate invocations do not throw in practice
34  })
35  class AuthResult_Test extends TestBase {
36  
37  	private static final Principal ALICE = () -> "alice";
38  
39  	@Test void a01_ofVarargs_principalAndRoles() {
40  		var a = AuthResult.of(ALICE, "user", "admin");
41  		assertSame(ALICE, a.getPrincipal());
42  		assertEquals(Set.of("user", "admin"), a.getRoles());
43  	}
44  
45  	@Test void a02_ofVarargs_noRoles() {
46  		var a = AuthResult.of(ALICE);
47  		assertSame(ALICE, a.getPrincipal());
48  		assertTrue(a.getRoles().isEmpty());
49  	}
50  
51  	@Test void a03_ofSet_principalAndRoles() {
52  		var roles = new HashSet<>(Set.of("r1", "r2"));
53  		var a = AuthResult.of(ALICE, roles);
54  		assertSame(ALICE, a.getPrincipal());
55  		assertEquals(Set.of("r1", "r2"), a.getRoles());
56  	}
57  
58  	@Test void a04_ofSet_null() {
59  		var a = AuthResult.of(ALICE, (Set<String>) null);
60  		assertTrue(a.getRoles().isEmpty());
61  	}
62  
63  	@Test void a05_rolesAreDefensiveCopy() {
64  		var mutable = new HashSet<>(Set.of("user"));
65  		var a = AuthResult.of(ALICE, mutable);
66  		mutable.add("admin");
67  		// Mutation of source set must not affect the result
68  		assertEquals(Set.of("user"), a.getRoles());
69  	}
70  
71  	@Test void a06_rolesAreUnmodifiable() {
72  		var a = AuthResult.of(ALICE, "user");
73  		assertThrows(UnsupportedOperationException.class, () -> a.getRoles().add("admin"));
74  	}
75  
76  	@Test void a07_nullPrincipalThrows() {
77  		assertThrows(Exception.class, () -> AuthResult.of(null, "user"));
78  	}
79  }
