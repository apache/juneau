30  /**
31   * Security-critical tests for {@link JwtTokenValidator}: algorithm-confusion attacks, the
32   * "none" algorithm, and the HS256 opt-in path.
33   *
34   * @since 10.0.0
35   */
36  @SuppressWarnings({
37  	"java:S5778"  // assertThrows lambdas with chained calls; intermediate invocations do not throw in practice
38  })
39  class JwtTokenValidator_Security_Test extends TestBase {
40  
41  	private static final Clock CLOCK = Clock.fixed(Instant.parse("2026-01-01T00:00:00Z"), ZoneOffset.UTC);
42  
43  	@Test void a01_noneAlgorithm_isPermanentlyRejected() throws Exception {
44  		var rsa = generateRsa("kid-1");
45  		var unsigned = unsignedToken(defaultClaims(CLOCK).build());
46  		var validator = JwtTokenValidator.create()
47  			.issuer(DEFAULT_ISSUER)
48  			.audience(DEFAULT_AUDIENCE)
49  			.jwkSource(fixed(rsa))
50  			.clock(CLOCK)
51  			.build();
52  		var ex = assertThrows(AuthenticationException.class, () -> validator.validate(unsigned));
53  		assertTrue(ex.getMessage().toLowerCase().contains("alg") || ex.getMessage().toLowerCase().contains("sign"));
54  	}
55  
56  	@Test void a02_noneInBuilderAllowlist_isRefused() {
57  		// JWSAlgorithm has no static `NONE` constant typed as JWSAlgorithm (nimbus exposes it on the
58  		// shared Algorithm parent, returned as Algorithm).  Re-creating it by name lets us assert
59  		// the builder refuses the deny-listed value via the JWSAlgorithm path callers actually use.
60  		var none = new JWSAlgorithm("none");
61  		assertThrows(IllegalArgumentException.class, () ->
62  			JwtTokenValidator.create()
63  				.issuer(DEFAULT_ISSUER)
64  				.audience(DEFAULT_AUDIENCE)
65  				.algorithms(none)
66  		);
67  	}
68  
69  	@Test void b01_algorithmConfusion_rejected() throws Exception {
70  		// Classic algorithm-confusion attack:
71  		//  - Validator is configured for RS256 only, with an RSA public key.
72  		//  - Attacker grabs the (public) RSA modulus, encodes it, and uses it as the HS256 HMAC secret.
73  		//  - Attacker signs a JWT with HS256 using that secret and sends it.
74  		//  - If the validator naively re-used the same key for any algorithm, it would pass.
75  		// JwtTokenValidator must reject because HS256 is not in the allowlist.
76  		var rsa = generateRsa("kid-1");
77  		var fakeSecret = rsa.toPublicJWK().toJSONString().getBytes();
78  		var token = signHmacRaw(fakeSecret, "kid-1", defaultClaims(CLOCK).build());
79  
80  		var validator = JwtTokenValidator.create()
81  			.issuer(DEFAULT_ISSUER)
82  			.audience(DEFAULT_AUDIENCE)
83  			.jwkSource(fixed(rsa))
84  			.clock(CLOCK)
85  			.build();
86  
87  		var ex = assertThrows(AuthenticationException.class, () -> validator.validate(token));
88  		assertTrue(ex.getMessage().toLowerCase().contains("algorithm"));
89  	}
90  
91  	@Test void c01_hs256OptIn_isAccepted_whenAllowlisted() throws Exception {
92  		var hmac = generateHmac("kid-hmac");
93  		var token = signHmac(hmac, defaultClaims(CLOCK).build());
94  		var validator = JwtTokenValidator.create()
95  			.issuer(DEFAULT_ISSUER)
96  			.audience(DEFAULT_AUDIENCE)
97  			.algorithms(JWSAlgorithm.HS256)
98  			.jwkSource(fixedRaw(hmac))
99  			.clock(CLOCK)
100 			.build();
101 		var principal = validator.validate(token);
102 		assertEquals("alice", principal.getName());
103 	}
104 
105 	@Test void c02_hs256_rejectedByDefault() throws Exception {
106 		var hmac = generateHmac("kid-hmac");
107 		var token = signHmac(hmac, defaultClaims(CLOCK).build());
108 		var validator = JwtTokenValidator.create()
109 			.issuer(DEFAULT_ISSUER)
110 			.audience(DEFAULT_AUDIENCE)
111 			.jwkSource(fixedRaw(hmac))
112 			.clock(CLOCK)
113 			.build();
114 		var ex = assertThrows(AuthenticationException.class, () -> validator.validate(token));
115 		assertTrue(ex.getMessage().toLowerCase().contains("allowlist") || ex.getMessage().toLowerCase().contains("not"));
116 	}
117 
118 	@Test void d01_garbageToken_rejected() throws Exception {
119 		var rsa = generateRsa("kid-1");
120 		var validator = JwtTokenValidator.create()
121 			.issuer(DEFAULT_ISSUER)
122 			.audience(DEFAULT_AUDIENCE)
123 			.jwkSource(fixed(rsa))
124 			.clock(CLOCK)
125 			.build();
126 		assertThrows(AuthenticationException.class, () -> validator.validate("not.a.jwt"));
127 	}
128 
129 	@Test void d02_emptyToken_rejected() throws Exception {
130 		var rsa = generateRsa("kid-1");
131 		var validator = JwtTokenValidator.create()
132 			.issuer(DEFAULT_ISSUER)
133 			.audience(DEFAULT_AUDIENCE)
134 			.jwkSource(fixed(rsa))
135 			.clock(CLOCK)
136 			.build();
137 		assertThrows(AuthenticationException.class, () -> validator.validate(""));
138 	}
