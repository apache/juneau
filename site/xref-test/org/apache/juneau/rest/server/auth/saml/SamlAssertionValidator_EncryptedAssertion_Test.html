17  package org.apache.juneau.rest.server.auth.saml;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.apache.juneau.*;
22  import org.apache.juneau.rest.server.auth.*;
23  import org.junit.jupiter.api.*;
24  
25  /**
26   * Encrypted-assertion handling tests for {@link SamlAssertionValidator} (per OQA Q5).
27   *
28   * <p>
29   * The full encrypt-decrypt-validate round-trip needs an explicit OpenSAML encrypter pipeline (which is
30   * non-trivial); for v1 we cover the two negative paths called out by the OQA decision:
31   *
32   * <ol>
33   * 	<li><b>Missing decryption credential</b> &mdash; when an envelope carries {@code <EncryptedAssertion>}
34   * 		but the validator has no {@code decryptionCredential(...)} configured, validation fails fast with
35   * 		an {@code AuthenticationException} bearing the {@code decryption_required} challenge.
36   * 	<li><b>Decrypt fails &mdash; then validation fails</b> &mdash; when the wrong decryption key is
37   * 		configured, validation fails with {@code decryption_failed} rather than silently producing a bogus
38   * 		principal.
39   * </ol>
40   *
41   * <p>
42   * The full happy-path is covered indirectly by {@link SamlAssertionValidator_HappyPath_Test} using
43   * unencrypted assertions; encrypt-path full coverage is deferred to integration testing against a real IdP
44   * once the integration-test harness lands (separate task).
45   *
46   * @since 10.0.0
47   */
48  class SamlAssertionValidator_EncryptedAssertion_Test extends TestBase {
49  
50  	private static final String ISSUER = "https://idp.example.com";
51  	private static final String AUDIENCE = "https://sp.example.com";
52  
53  	private static final String ENCRYPTED_ENVELOPE_TEMPLATE =
54  		"<samlp:Response xmlns:samlp=\"urn:oasis:names:tc:SAML:2.0:protocol\""
55  		+ " xmlns:saml=\"urn:oasis:names:tc:SAML:2.0:assertion\""
56  		+ " xmlns:xenc=\"http://www.w3.org/2001/04/xmlenc#\""
57  		+ " ID=\"_r1\" Version=\"2.0\" IssueInstant=\"2030-01-01T00:00:00Z\">"
58  		+ "  <saml:Issuer>" + ISSUER + "</saml:Issuer>"
59  		+ "  <samlp:Status><samlp:StatusCode Value=\"urn:oasis:names:tc:SAML:2.0:status:Success\"/></samlp:Status>"
60  		+ "  <saml:EncryptedAssertion>"
61  		+ "    <xenc:EncryptedData Type=\"http://www.w3.org/2001/04/xmlenc#Element\">"
62  		+ "      <xenc:EncryptionMethod Algorithm=\"http://www.w3.org/2009/xmlenc11#aes128-gcm\"/>"
63  		+ "      <xenc:CipherData><xenc:CipherValue>SGVsbG8=</xenc:CipherValue></xenc:CipherData>"
64  		+ "    </xenc:EncryptedData>"
65  		+ "  </saml:EncryptedAssertion>"
66  		+ "</samlp:Response>";
67  
68  	@Test void a01_missingDecryptionCredential_rejected() throws Exception {
69  		var cred = SamlTestSupport.credential(SamlTestSupport.generateRsaKeyPair());
70  		var validator = SamlAssertionValidator.create()
71  			.spEntityId(AUDIENCE)
72  			.expectedIssuer(ISSUER)
73  			.signingCredential(cred)
74  			.build();
75  		var ex = assertThrows(AuthenticationException.class, () -> validator.validate(ENCRYPTED_ENVELOPE_TEMPLATE));
76  		var hdr = ex.getHeaders().stream()
77  			.filter(h -> "WWW-Authenticate".equalsIgnoreCase(h.getName()))
78  			.findFirst().orElseThrow();
79  		assertTrue(hdr.getValue().contains("decryption_required"),
80  			"Expected WWW-Authenticate to contain 'decryption_required', got: " + hdr.getValue());
81  	}
82  
83  	@Test void a02_wrongDecryptionCredential_rejected() throws Exception {
84  		var signing = SamlTestSupport.credential(SamlTestSupport.generateRsaKeyPair());
85  		var decryption = SamlTestSupport.credential(SamlTestSupport.generateRsaKeyPair());
86  		var validator = SamlAssertionValidator.create()
87  			.spEntityId(AUDIENCE)
88  			.expectedIssuer(ISSUER)
89  			.signingCredential(signing)
90  			.decryptionCredential(decryption)
91  			.build();
92  		var ex = assertThrows(AuthenticationException.class, () -> validator.validate(ENCRYPTED_ENVELOPE_TEMPLATE));
93  		var hdr = ex.getHeaders().stream()
94  			.filter(h -> "WWW-Authenticate".equalsIgnoreCase(h.getName()))
95  			.findFirst().orElseThrow();
96  		assertTrue(hdr.getValue().contains("decryption_failed"),
97  			"Expected WWW-Authenticate to contain 'decryption_failed', got: " + hdr.getValue());
98  	}
99  }
