17  package org.apache.juneau.rest.server.auth.oauth;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.io.*;
22  import java.net.*;
23  import java.nio.charset.*;
24  
25  import org.apache.juneau.*;
26  import org.apache.juneau.rest.server.auth.*;
27  import org.junit.jupiter.api.*;
28  
29  import com.sun.net.httpserver.*;
30  
31  /**
32   * Tests that {@link OAuthIntrospectionValidator} binds introspected tokens to an expected audience/resource
33   * when one is configured via {@link OAuthIntrospectionValidator.Builder#audience(String...)} /
34   * {@link OAuthIntrospectionValidator.Builder#resource(String...)}, and leaves today's active/scope-only
35   * behavior unchanged when no audience is configured.
36   *
37   * @since 10.0.0
38   */
39  class OAuthIntrospectionValidator_Audience_Test extends TestBase {
40  
41  	private HttpServer server;
42  	private volatile String nextResponse;
43  
44  	@BeforeEach
45  	void startServer() throws IOException {
46  		server = HttpServer.create(new InetSocketAddress(InetAddress.getLoopbackAddress(), 0), 0);
47  		server.createContext("/introspect", ex -> {
48  			var body = nextResponse.getBytes(StandardCharsets.UTF_8);
49  			ex.getResponseHeaders().add("Content-Type", "application/json");
50  			ex.sendResponseHeaders(200, body.length);
51  			try (var os = ex.getResponseBody()) {
52  				os.write(body);
53  			}
54  		});
55  		server.start();
56  	}
57  
58  	@AfterEach
59  	void stopServer() {
60  		server.stop(0);
61  	}
62  
63  	private URI endpoint() {
64  		return URI.create("http://127.0.0.1:" + server.getAddress().getPort() + "/introspect");
65  	}
66  
67  	private OAuthIntrospectionValidator.Builder validatorBuilder() {
68  		return OAuthIntrospectionValidator.create()
69  			.introspectionEndpoint(endpoint())
70  			.clientId("client")
71  			.clientSecret("secret");
72  	}
73  
74  	// -----------------------------------------------------------------------------------------------------------------
75  	// A: expected audience configured — mismatch rejected even though active=true and scopes match.
76  	// -----------------------------------------------------------------------------------------------------------------
77  
78  	@Test void a01_audienceConfigured_mismatchedSingleAud_rejected() {
79  		nextResponse = "{\"active\":true,\"sub\":\"alice\",\"scope\":\"read\",\"aud\":\"https://other.example\"}";
80  		var v = validatorBuilder().audience("https://api.example").requiredScopes("read").build();
81  		var ex = assertThrows(AuthenticationException.class, () -> v.validate("tok-wrong-aud"));
82  		assertTrue(ex.getMessage().toLowerCase().contains("audience"), "Expected an audience-mismatch message, got: " + ex.getMessage());
83  	}
84  
85  	@Test void a02_audienceConfigured_noAudInResponse_rejected() {
86  		// Token is active with matching scopes but the AS returned no aud claim at all.
87  		nextResponse = "{\"active\":true,\"sub\":\"alice\",\"scope\":\"read\"}";
88  		var v = validatorBuilder().audience("https://api.example").build();
89  		var ex = assertThrows(AuthenticationException.class, () -> v.validate("tok-no-aud"));
90  		assertTrue(ex.getMessage().toLowerCase().contains("audience"), "Expected an audience-mismatch message, got: " + ex.getMessage());
91  	}
92  
93  	// -----------------------------------------------------------------------------------------------------------------
94  	// B: expected audience configured — matching audience accepted.
95  	// -----------------------------------------------------------------------------------------------------------------
96  
97  	@Test void b01_audienceConfigured_matchingSingleAud_accepted() throws Exception {
98  		nextResponse = "{\"active\":true,\"sub\":\"alice\",\"scope\":\"read\",\"aud\":\"https://api.example\"}";
99  		var v = validatorBuilder().audience("https://api.example").build();
100 		var p = v.validate("tok-right-aud");
101 		assertEquals("alice", p.getName());
102 	}
103 
104 	@Test void b02_multiValuedAud_containingExpectedValue_accepted() throws Exception {
105 		// aud as a JSON array with multiple values; the expected value is one of several.
106 		nextResponse = "{\"active\":true,\"sub\":\"alice\",\"scope\":\"read\",\"aud\":[\"https://other.example\",\"https://api.example\"]}";
107 		var v = validatorBuilder().audience("https://api.example").build();
108 		var p = v.validate("tok-multi-aud");
109 		assertEquals("alice", p.getName());
110 	}
111 
112 	@Test void b03_resourceAlias_matchesAudClaim() throws Exception {
113 		// resource(...) is an alias for audience(...) — validated against the same aud claim.
114 		nextResponse = "{\"active\":true,\"sub\":\"alice\",\"scope\":\"read\",\"aud\":\"https://api.example\"}";
115 		var v = validatorBuilder().resource("https://api.example").build();
116 		var p = v.validate("tok-resource-alias");
117 		assertEquals("alice", p.getName());
118 	}
119 
120 	// -----------------------------------------------------------------------------------------------------------------
121 	// C: no audience configured — today's active/scope-only behavior is unchanged (regression guard).
122 	// -----------------------------------------------------------------------------------------------------------------
123 
124 	@Test void c01_noAudienceConfigured_mismatchedAudStillAccepted() throws Exception {
125 		// No audience bind configured — an "aud" for a completely different RS must NOT cause rejection.
126 		nextResponse = "{\"active\":true,\"sub\":\"alice\",\"scope\":\"read\",\"aud\":\"https://other.example\"}";
127 		var v = validatorBuilder().build();
128 		var p = v.validate("tok-no-bind");
129 		assertEquals("alice", p.getName());
130 	}
131 
132 	@Test void c02_noAudienceConfigured_noAudClaimAtAll_stillAccepted() throws Exception {
133 		nextResponse = "{\"active\":true,\"sub\":\"alice\",\"scope\":\"read\"}";
134 		var v = validatorBuilder().build();
135 		var p = v.validate("tok-no-bind-no-aud");
136 		assertEquals("alice", p.getName());
137 	}
138 }
