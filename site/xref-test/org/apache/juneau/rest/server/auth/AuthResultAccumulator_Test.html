17  package org.apache.juneau.rest.server.auth;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.security.*;
22  import java.util.*;
23  
24  import org.apache.juneau.*;
25  import org.junit.jupiter.api.*;
26  
27  /**
28   * Tests for {@link AuthResultAccumulator} fold semantics ({@code ADD} union, {@code REPLACE} reset, roles-only).
29   *
30   * @since 10.0.0
31   */
32  class AuthResultAccumulator_Test extends TestBase {
33  
34  	private static final Principal ALICE = () -> "alice";
35  	private static final Principal BOB = () -> "bob";
36  
37  	@Test void a01_addUnionsRoles_firstPrincipalWins() {
38  		var acc = new AuthResultAccumulator();
39  		acc.add(AuthResult.of(ALICE, "r1"));
40  		acc.add(AuthResult.of(BOB, "r2"));      // ADD: principal stays alice, roles union
41  		var r = acc.result().orElseThrow();
42  		assertSame(ALICE, r.getPrincipal());
43  		assertEquals(Set.of("r1", "r2"), r.getRoles());
44  	}
45  
46  	@Test void a02_addNullPrincipalKeepsExisting() {
47  		var acc = new AuthResultAccumulator();
48  		acc.add(AuthResult.of(ALICE, "r1"));
49  		acc.add(AuthResult.ofRoles("r2"));     // null principal contributes only roles
50  		var r = acc.result().orElseThrow();
51  		assertSame(ALICE, r.getPrincipal());
52  		assertEquals(Set.of("r1", "r2"), r.getRoles());
53  	}
54  
55  	@Test void a03_replaceResetsAccumulator() {
56  		var acc = new AuthResultAccumulator();
57  		acc.add(AuthResult.of(ALICE, "r1"));
58  		acc.add(AuthResult.replacing(BOB, "r2"));  // discard alice + r1
59  		var r = acc.result().orElseThrow();
60  		assertSame(BOB, r.getPrincipal());
61  		assertEquals(Set.of("r2"), r.getRoles());
62  	}
63  
64  	@Test void a04_emptyWhenNothingAdded() {
65  		assertTrue(new AuthResultAccumulator().result().isEmpty());
66  	}
67  
68  	@Test void a05_rolesOnlyNoPrincipalYieldsEmpty() {
69  		var acc = new AuthResultAccumulator();
70  		acc.add(AuthResult.ofRoles("r1"));   // roles but never a principal
71  		assertTrue(acc.result().isEmpty());  // no identity => not a successful auth
72  	}
73  
74  	@Test void a06_nullAddIgnored() {
75  		var acc = new AuthResultAccumulator();
76  		acc.add(null);
77  		assertTrue(acc.result().isEmpty());
78  	}
79  
80  	@Test void a07_rolesOnlyThenPrincipalAdopts() {
81  		var acc = new AuthResultAccumulator();
82  		acc.add(AuthResult.ofRoles("r1"));        // null principal, role r1
83  		acc.add(AuthResult.of(ALICE, "r2"));      // adopts alice, unions r2
84  		var r = acc.result().orElseThrow();
85  		assertSame(ALICE, r.getPrincipal());
86  		assertEquals(Set.of("r1", "r2"), r.getRoles());
87  	}
88  
89  	@Test void a08_replaceAfterMultipleAdds() {
90  		var acc = new AuthResultAccumulator();
91  		acc.add(AuthResult.of(ALICE, "r1"));
92  		acc.add(AuthResult.ofRoles("r2"));
93  		acc.add(AuthResult.replacing(BOB));   // discard everything, no roles
94  		var r = acc.result().orElseThrow();
95  		assertSame(BOB, r.getPrincipal());
96  		assertTrue(r.getRoles().isEmpty());
97  	}
98  }
