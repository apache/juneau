19  import static org.apache.juneau.commons.utils.Shorts.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.time.*;
23  import java.util.*;
24  
25  import org.apache.juneau.*;
26  import org.apache.juneau.rest.server.auth.*;
27  import org.junit.jupiter.api.*;

29  /**
30   * Tests for {@link SignedCookieSessionStore} &mdash; stateless HMAC-signed cookie round-trip, tamper
31   * rejection, expiry, and the size cap.
32   *
33   * @since 10.0.0
34   */
35  @SuppressWarnings({
36  	"java:S5778"  // assertThrows lambdas with chained calls; intermediate invocations do not throw in practice
37  })
38  class SignedCookieSessionStore_Test extends TestBase {
39  
40  	private static final String KEY = "0123456789abcdef0123456789abcdef";  // 32 bytes
41  
42  	// Fixed time seam: session timestamps are anchored to a constant and the round-trip store is
43  	// driven by a matching fixed Clock, instead of reading the system clock (java:S8692).
44  	private static final Instant NOW = Instant.parse("2026-01-01T00:00:00Z");
45  	private static final Clock CLOCK = Clock.fixed(NOW, ZoneOffset.UTC);
46  
47  	private static OidcSession session(Instant now, String sid) {
48  		return new OidcSession("id-1", "alice", o(sid),
49  			new ClaimsPrincipal("alice", Map.of("sub", "alice", "email", "alice@example.com")),
50  			new LinkedHashSet<>(List.of("user", "admin")),
51  			oe(), now, now.plus(Duration.ofHours(8)));
52  	}
53  
54  	@Test void a01_roundTrip_recoversClaimsAndRoles() {
55  		var store = SignedCookieSessionStore.create().signingKey(KEY).clock(CLOCK).build();
56  		var now = NOW;
57  		var cookie = store.createSessionCookieValue(session(now, "sess-1"));
58  		var s = store.lookup(cookie);
59  		assertTrue(s.isPresent());
60  		assertEquals("alice", s.get().subject());
61  		assertEquals("alice", s.get().principal().getName());
62  		assertEquals(Set.of("user", "admin"), s.get().roles());
63  		assertEquals("sess-1", s.get().sid().orElse(null));
64  		assertEquals("alice@example.com", s.get().principal().getClaim("email", String.class).orElse(null));
65  		assertTrue(s.get().token().isEmpty(), "cookie store must NOT retain raw tokens");
66  	}
67  
68  	@Test void b01_tamperedCookie_isRejected() {
69  		var store = SignedCookieSessionStore.create().signingKey(KEY).build();
70  		var cookie = store.createSessionCookieValue(session(NOW, null));
71  		var tampered = cookie.substring(0, cookie.length() - 3) + "AAA";
72  		assertTrue(store.lookup(tampered).isEmpty());
73  	}
74  
75  	@Test void b02_wrongKey_isRejected() {
76  		var signer = SignedCookieSessionStore.create().signingKey(KEY).build();
77  		var cookie = signer.createSessionCookieValue(session(NOW, null));
78  		var other = SignedCookieSessionStore.create().signingKey("ffffffffffffffffffffffffffffffff").build();
79  		assertTrue(other.lookup(cookie).isEmpty());
80  	}
81  
82  	@Test void b03_garbageCookie_isRejected() {
83  		var store = SignedCookieSessionStore.create().signingKey(KEY).build();
84  		assertTrue(store.lookup("not-a-jwt").isEmpty());
85  	}
86  
87  	@Test void c01_expiredCookie_isRejected() {
88  		var base = Instant.parse("2026-01-01T00:00:00Z");
89  		var clock = new EphemeralStore_Test.MutableClock(base);
90  		var store = SignedCookieSessionStore.create().signingKey(KEY).clock(clock).build();
91  		var cookie = store.createSessionCookieValue(session(base, null));
92  		clock.advance(Duration.ofHours(9));
93  		assertTrue(store.lookup(cookie).isEmpty());
94  	}
95  
96  	@Test void d01_doesNotSupportServerSideRevocation() {
97  		var store = SignedCookieSessionStore.create().signingKey(KEY).build();
98  		assertFalse(store.supportsServerSideRevocation());
99  		assertThrows(UnsupportedOperationException.class, () -> store.invalidateBySubject("alice"));
100 		assertThrows(UnsupportedOperationException.class, () -> store.invalidateBySessionId("sess-1"));
101 	}
102 
103 	@Test void e01_overCapPayload_throws() {
104 		var store = SignedCookieSessionStore.create().signingKey(KEY).maxCookieBytes(128).build();
105 		assertThrows(IllegalStateException.class,
106 			() -> store.createSessionCookieValue(session(NOW, "sess-1")));
107 	}
108 
109 	@Test void f01_shortKey_rejected() {
110 		assertThrows(IllegalArgumentException.class, () -> SignedCookieSessionStore.create().signingKey("too-short"));
111 	}
112 
113 	@Test void f02_missingKey_rejected() {
114 		assertThrows(IllegalStateException.class, () -> SignedCookieSessionStore.create().build());
115 	}
116 }
