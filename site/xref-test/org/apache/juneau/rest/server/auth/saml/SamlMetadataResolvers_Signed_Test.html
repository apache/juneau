41  /**
42   * Tests for {@link SamlMetadataResolvers#url(String, java.security.cert.X509Certificate)}: pinned-certificate
43   * metadata signature validation, plus fetch failures (interrupt, non-XML body, truncated body).
44   *
45   * @since 10.0.0
46   */
47  class SamlMetadataResolvers_Signed_Test extends TestBase {
48  
49  	private static final String UNSIGNED_METADATA =
50  		"<?xml version=\"1.0\" encoding=\"UTF-8\"?>"
51  		+ "<EntityDescriptor xmlns=\"urn:oasis:names:tc:SAML:2.0:metadata\" ID=\"_md1\" entityID=\"https://idp.example.com/idp\">"
52  		+ "<IDPSSODescriptor protocolSupportEnumeration=\"urn:oasis:names:tc:SAML:2.0:protocol\">"
53  		+ "<SingleSignOnService Binding=\"urn:oasis:names:tc:SAML:2.0:bindings:HTTP-Redirect\" Location=\"https://idp.example.com/sso\"/>"
54  		+ "</IDPSSODescriptor></EntityDescriptor>";
55  
56  	@FunctionalInterface
57  	private interface UrlAction {
58  		void run(String url) throws Exception;
59  	}
60  
61  	private static void withBody(byte[] body, UrlAction action) throws Exception {
62  		withServer(exchange -> {
63  			exchange.sendResponseHeaders(200, body.length);
64  			try (var os = exchange.getResponseBody()) {
65  				os.write(body);
66  			}
67  		}, action);
68  	}
69  
70  	private static void withServer(HttpHandler handler, UrlAction action) throws Exception {
71  		var server = HttpServer.create(new InetSocketAddress(InetAddress.getLoopbackAddress(), 0), 0);
72  		server.createContext("/metadata", handler);
73  		server.start();
74  		try {
75  			action.run("http://127.0.0.1:" + server.getAddress().getPort() + "/metadata");
76  		} finally {
77  			server.stop(0);
78  		}
79  	}
80  
81  	/** Signs {@link #UNSIGNED_METADATA} with the test key and returns the serialized document. */
82  	private static String signedMetadata() throws Exception {
83  		OpenSamlBootstrap.ensureInitialized(); // the registry lookups below need it, whatever order the tests run in
84  		var doc = SamlXml.newDocumentBuilder().parse(new ByteArrayInputStream(UNSIGNED_METADATA.getBytes(StandardCharsets.UTF_8)));
85  		var root = doc.getDocumentElement();
86  		var ed = (EntityDescriptor) XMLObjectProviderRegistrySupport.getUnmarshallerFactory().getUnmarshaller(root).unmarshall(root);
87  		var sig = new SignatureBuilder().buildObject();
88  		sig.setSigningCredential(SamlTestKeys.credential());
89  		sig.setSignatureAlgorithm(SignatureConstants.ALGO_ID_SIGNATURE_RSA_SHA256);
90  		sig.setCanonicalizationAlgorithm(SignatureConstants.ALGO_ID_C14N_EXCL_OMIT_COMMENTS);
91  		ed.setSignature(sig);
92  		var dom = XMLObjectProviderRegistrySupport.getMarshallerFactory().getMarshaller(ed).marshall(ed);
93  		Signer.signObject(sig);
94  		var sw = new StringWriter();
95  		var transformer = SecureTransformerFactory.newInstance().newTransformer();
96  		transformer.setOutputProperty(OutputKeys.OMIT_XML_DECLARATION, "yes");
97  		transformer.transform(new DOMSource(dom), new StreamResult(sw));
98  		return sw.toString();
99  	}

105 	@Test void a01_signedMetadata_pinnedCertMatches_resolves() throws Exception {
106 		var cert = SamlTestKeys.certificate();
107 		var body = signedMetadata().getBytes(StandardCharsets.UTF_8);
108 		withBody(body, url -> assertNotNull(SamlMetadataResolvers.url(url, cert)));
109 	}
110 
111 	@Test void a02_unsignedMetadata_pinnedCert_rejected() throws Exception {
112 		var cert = SamlTestKeys.certificate();
113 		withBody(UNSIGNED_METADATA.getBytes(StandardCharsets.UTF_8), url -> {
114 			// Positive control: the same document resolves when no certificate is pinned, so the pin is what rejects it.
115 			assertNotNull(SamlMetadataResolvers.url(url));
116 			var ex = assertThrows(IOException.class, () -> SamlMetadataResolvers.url(url, cert));
117 			assertTrue(ex.getMessage().contains("Failed to initialize DOMMetadataResolver"), ex.getMessage());
118 			assertInstanceOf(ComponentInitializationException.class, ex.getCause()); // not an XML parse failure
119 		});
120 	}
121 
122 	@Test void a03_signedMetadata_wrongPinnedCert_rejected() throws Exception {
123 		// Same document, but the pin is a different certificate (the signer's key does not match it).
124 		var other = SamlTestKeys.otherCertificate();
125 		var body = signedMetadata().getBytes(StandardCharsets.UTF_8);
126 		withBody(body, url -> {
127 			var ex = assertThrows(IOException.class, () -> SamlMetadataResolvers.url(url, other));
128 			// A signature-filter rejection, not a plain fetch failure.
129 			assertTrue(ex.getMessage().contains("Failed to initialize DOMMetadataResolver"), ex.getMessage());
130 			assertInstanceOf(ComponentInitializationException.class, ex.getCause()); // not an XML parse failure
131 		});
132 	}

138 	@Test void b01_bodyNotXml_throwsIOException() throws Exception {
139 		withBody("this is not xml".getBytes(StandardCharsets.UTF_8), url -> assertThrows(IOException.class, () -> SamlMetadataResolvers.url(url)));
140 	}
141 
142 	@Test void b02_interruptedBeforeFetch_throwsIOExceptionAndKeepsInterruptFlag() throws Exception {
143 		withBody(UNSIGNED_METADATA.getBytes(StandardCharsets.UTF_8), url -> {
144 			Thread.currentThread().interrupt();
145 			try {
146 				var ex = assertThrows(IOException.class, () -> SamlMetadataResolvers.url(url));
147 				assertTrue(ex.getMessage().contains("Interrupted"), ex.getMessage());
148 				assertTrue(Thread.currentThread().isInterrupted());
149 			} finally {
150 				Thread.interrupted(); // clear the flag so later tests are unaffected
151 			}
152 		});
153 	}
154 
155 	@Test void b03_connectionClosedMidBody_throwsIOException() throws Exception {
156 		withServer(exchange -> {
157 			exchange.sendResponseHeaders(200, 100_000); // promise far more bytes than are sent
158 			try (var os = exchange.getResponseBody()) {
159 				os.write(new byte[16]);
160 				os.flush();
161 			}
162 			exchange.close();
163 		}, url -> assertThrows(IOException.class, () -> SamlMetadataResolvers.url(url)));
164 	}
