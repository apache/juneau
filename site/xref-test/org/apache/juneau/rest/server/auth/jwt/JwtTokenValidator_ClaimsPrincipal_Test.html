17  package org.apache.juneau.rest.server.auth.jwt;

19  import static org.apache.juneau.rest.server.auth.jwt.JwtTestSupport.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.time.*;
23  
24  import org.apache.juneau.*;
25  import org.apache.juneau.rest.server.auth.*;
26  import org.junit.jupiter.api.*;
27  
28  /**
29   * Verifies that a successful {@link JwtTokenValidator#validate(String) validation} produces a
30   * {@link ClaimsPrincipal} whose claims are accessible via {@code getClaim(String, Class)}.
31   *
32   * @since 10.0.0
33   */
34  class JwtTokenValidator_ClaimsPrincipal_Test extends TestBase {
35  
36  	private static final Clock CLOCK = Clock.fixed(Instant.parse("2026-01-01T00:00:00Z"), ZoneOffset.UTC);
37  
38  	@Test void a01_claimsPrincipalExposesStandardClaims() throws Exception {
39  		var rsa = generateRsa("kid-1");
40  		var claims = defaultClaims(CLOCK)
41  			.claim("scope", "read write")
42  			.claim("groups", new String[]{"admins", "billing"})
43  			.build();
44  		var token = signRsa(rsa, claims);
45  		var validator = JwtTokenValidator.create()
46  			.issuer(DEFAULT_ISSUER)
47  			.audience(DEFAULT_AUDIENCE)
48  			.jwkSource(fixed(rsa))
49  			.clock(CLOCK)
50  			.build();
51  		var principal = validator.validate(token);
52  		assertInstanceOf(ClaimsPrincipal.class, principal);
53  		var cp = (ClaimsPrincipal) principal;
54  
55  		assertEquals("alice", cp.getName());
56  		assertEquals(DEFAULT_ISSUER, cp.getClaim("iss", String.class).orElse(null));
57  		assertEquals("read write", cp.getClaim("scope", String.class).orElse(null));
58  		assertNotNull(cp.getClaim("exp", Object.class).orElse(null));
59  		assertNotNull(cp.getClaim("nbf", Object.class).orElse(null));
60  	}
61  
62  	@Test void a02_missingClaimReturnsEmpty() throws Exception {
63  		var rsa = generateRsa("kid-1");
64  		var token = signRsa(rsa, defaultClaims(CLOCK).build());
65  		var validator = JwtTokenValidator.create()
66  			.issuer(DEFAULT_ISSUER)
67  			.audience(DEFAULT_AUDIENCE)
68  			.jwkSource(fixed(rsa))
69  			.clock(CLOCK)
70  			.build();
71  		var cp = (ClaimsPrincipal) validator.validate(token);
72  		assertTrue(cp.getClaim("not_in_token", String.class).isEmpty());
73  	}
74  }
