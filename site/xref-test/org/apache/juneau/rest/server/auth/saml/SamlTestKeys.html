91  	/** Returns the parsed certificate. */
92  	static X509Certificate certificate() throws Exception {
93  		var bytes = Base64.getDecoder().decode(CERT_BASE64);
94  		return (X509Certificate) CertificateFactory.getInstance("X.509").generateCertificate(new ByteArrayInputStream(bytes));
95  	}
96  
97  	/** Returns the second, unrelated certificate. */
98  	static X509Certificate otherCertificate() throws Exception {
99  		var bytes = Base64.getDecoder().decode(OTHER_CERT_BASE64);
100 		return (X509Certificate) CertificateFactory.getInstance("X.509").generateCertificate(new ByteArrayInputStream(bytes));
101 	}
102 
103 	/** Returns a signing/decryption credential holding the certificate's key pair. */
104 	static BasicX509Credential credential() throws Exception {
105 		var cert = certificate();
106 		var key = KeyFactory.getInstance("RSA").generatePrivate(new PKCS8EncodedKeySpec(Base64.getDecoder().decode(KEY_BASE64)));
107 		var c = new BasicX509Credential(cert, key);
108 		c.setUsageType(UsageType.UNSPECIFIED);
109 		return c;
110 	}
