39  /**
40   * End-to-end integration test verifying the composition of {@link AuthFilterChain} with
41   * {@link RoleBasedRestGuard} and {@link Auth @Auth Principal} op parameters.
42   *
43   * <h5 class='topic'>Note on MockRest and servlet filters</h5>
44   *
45   * <p>
46   * {@code MockRest} / {@code MockRestClient} invokes the Juneau {@code RestServlet} directly without going through
47   * the Jetty servlet container, so servlet filters are <b>not</b> executed in that path.  Full end-to-end filter +
48   * servlet testing requires a live Jetty container (e.g. a {@code JettyMicroserviceTest}), which is deliberately
49   * outside the fast unit-test tier.
50   *
51   * <p>
52   * This test class verifies the integration in two ways:
53   * <ol>
54   * 	<li><b>Filter-layer assertions</b>: directly drives {@link AuthFilterChain#doFilter} and checks that the
55   * 		resulting {@link AuthenticatedRequestWrapper} carries the correct principal, roles, and attributes.
56   * 	<li><b>Juneau-layer assertions</b>: uses {@code MockRestClient} to verify that the Juneau stack
57   * 		({@code RoleBasedRestGuard}, {@code @Auth Principal}) responds correctly when a request arrives with
58   * 		credentials already stashed by an upstream filter — simulated here via {@link BearerTokenGuard} (the
59   * 		op-level guard), which uses the same {@link RestServerConstants#PRINCIPAL_ATTR} stash key.
60   * </ol>
61   *
62   * @since 10.0.0
63   */
64  class AuthFilterChain_GuardIntegration_Test extends TestBase {

71  		"unused"  // Unused in this context; kept for API consistency or future use.
72  	})
73  	private static final Principal ALICE = () -> "alice";
74  	@SuppressWarnings({
75  		"unused"  // Unused in this context; kept for API consistency or future use.
76  	})
77  	private static final Principal BOB = () -> "bob";
78  
79  	/** TokenValidator: "bearer-user" → ClaimsPrincipal(alice, roles=[user]) */
80  	private static final TokenValidator BEARER_VALIDATOR = token -> {
81  		if ("bearer-user".equals(token))
82  			return new ClaimsPrincipal("alice", Map.of("roles", List.of("user"), "sub", "alice"));
83  		if ("bearer-admin".equals(token))
84  			return new ClaimsPrincipal("alice", Map.of("roles", List.of("admin"), "sub", "alice"));
85  		throw new AuthenticationException("Unknown bearer token").wwwAuthenticate("Bearer realm=\"api\"");
86  	};
87  
88  	/** ApiKeyStore: "apikey-admin" → ClaimsPrincipal(bob, roles=[admin]) */
89  	private static final ApiKeyStore KEY_STORE = key -> {
90  		if ("apikey-admin".equals(key))
91  			return o(new ClaimsPrincipal("bob", Map.of("roles", List.of("admin"), "sub", "bob")));
92  		return oe();
93  	};
94  
95  	private static AuthFilterChain buildChain() {
96  		return AuthFilterChain.create(null)
97  			.append(BearerTokenAuthFilter.create().validator(BEARER_VALIDATOR).build())
98  			.append(ApiKeyAuthFilter.create().store(KEY_STORE).build())
99  			.build();
100 	}

102 	// -----------------------------------------------------------------------------------------
103 	// Helper: drives chain.doFilter and returns the captured wrapped request (or null when 401).
104 	// -----------------------------------------------------------------------------------------
105 
106 	private static class Result {
107 		final ServletRequest captured;
108 		final int status;
109 
110 		Result(ServletRequest captured, int status) {
111 			this.captured = captured;
112 			this.status = status;
113 		}
114 	}
115 
116 	private static Result runChain(AuthFilterChain chain, String authHeader, String apiKeyHeader) throws Exception {
117 		var req = mock(HttpServletRequest.class);
118 		when(req.getRequestURI()).thenReturn("/api/me");
119 		when(req.getContextPath()).thenReturn("");
120 		when(req.getHeader("Authorization")).thenReturn(authHeader);
121 		when(req.getHeader("X-API-Key")).thenReturn(apiKeyHeader);
122 		when(req.getCookies()).thenReturn(null);
123 
124 		var resp = mock(HttpServletResponse.class);
125 		var sw = new StringWriter();
126 		when(resp.getWriter()).thenReturn(new PrintWriter(sw));
127 
128 		var captured = new Object[1];
129 		var status = new int[]{0};
130 		FilterChain captureChain = (r, s) -> captured[0] = r;
131 		doAnswer(inv -> { status[0] = (int) inv.getArguments()[0]; return null; }).when(resp).setStatus(anyInt());
132 
133 		chain.doFilter(req, resp, captureChain);
134 		return new Result((ServletRequest) captured[0], status[0]);
135 	}

141 	@Test void a01_validBearerToken_principalAndRoleCorrect() throws Exception {
142 		var r = runChain(buildChain(), "Bearer bearer-user", null);
143 		assertNotNull(r.captured, "Request must be forwarded on success");
144 		assertInstanceOf(AuthenticatedRequestWrapper.class, r.captured);
145 		var w = (AuthenticatedRequestWrapper) r.captured;
146 		assertEquals("alice", w.getUserPrincipal().getName());
147 		assertTrue(w.isUserInRole("user"));
148 		assertFalse(w.isUserInRole("admin"));
149 	}
150 
151 	@Test void a02_validApiKey_principalAndRoleCorrect() throws Exception {
152 		var r = runChain(buildChain(), null, "apikey-admin");
153 		assertNotNull(r.captured);
154 		var w = (AuthenticatedRequestWrapper) r.captured;
155 		assertEquals("bob", w.getUserPrincipal().getName());
156 		assertTrue(w.isUserInRole("admin"));
157 		assertFalse(w.isUserInRole("user"));
158 	}
159 
160 	@Test void a03_bothCredentials_bearerPrincipalWins_rolesUnion() throws Exception {
161 		// Both bearer-user (user role) and apikey-admin (admin role) present.
162 		// Bearer is registered first → alice wins for principal; roles = union.
163 		var r = runChain(buildChain(), "Bearer bearer-user", "apikey-admin");
164 		assertNotNull(r.captured);
165 		var w = (AuthenticatedRequestWrapper) r.captured;
166 		assertEquals("alice", w.getUserPrincipal().getName());
167 		// Union: user (from bearer) + admin (from api-key)
168 		assertTrue(w.isUserInRole("user"));
169 		assertTrue(w.isUserInRole("admin"));
170 	}
171 
172 	@Test void a04_noCredentials_passThroughUnchanged() throws Exception {
173 		// Both filters return empty — no credentials at all — pass through.
174 		var r = runChain(buildChain(), null, null);
175 		// Captured request is the original mock (not an AuthenticatedRequestWrapper)
176 		assertNotNull(r.captured, "Request must pass through");
177 		assertFalse(r.captured instanceof AuthenticatedRequestWrapper,
178 			"No credentials: must pass through without wrapping");
179 	}

181 	@Test void a05_invalidBearerToken_returns401() throws Exception {
182 		var r = runChain(buildChain(), "Bearer BAD", null);
183 		assertNull(r.captured, "Request must NOT be forwarded on 401");
184 		assertEquals(HttpServletResponse.SC_UNAUTHORIZED, r.status);

187 	@Test void a06_principalAttrStashedForAuthArgResolver() throws Exception {
188 		var r = runChain(buildChain(), "Bearer bearer-user", null);
189 		var w = (AuthenticatedRequestWrapper) r.captured;
190 		// @Auth arg resolver reads PRINCIPAL_ATTR
191 		var attr = w.getAttribute(RestServerConstants.PRINCIPAL_ATTR);
192 		assertNotNull(attr);
193 		assertInstanceOf(Principal.class, attr);
194 		assertEquals("alice", ((Principal) attr).getName());
195 	}

200 	// Note: MockRest does not run servlet filters, so AuthenticatedRequestWrapper is not in the
201 	// request chain here.  These tests use BearerTokenGuard as a stand-in to
202 	// pre-populate RestServerConstants.PRINCIPAL_ATTR, verifying that the Juneau stack reads
203 	// the same attribute key that AuthenticatedRequestWrapper writes.  Full
204 	// RoleBasedRestGuard + filter-produced isUserInRole() integration requires a live Jetty
205 	// container test.

208 	private static final TokenValidator MOCK_VALIDATOR = token -> {
209 		if ("user-tok".equals(token))
210 			return new ClaimsPrincipal("alice", Map.of("roles", List.of("user"), "sub", "alice"));
211 		throw new AuthenticationException("bad token").wwwAuthenticate("Bearer realm=\"test\"");
212 	};

214 	@Rest
215 	public static class GuardedResource extends RestServlet {
216 		private static final long serialVersionUID = 1L;
217 
218 		@Bean
219 		public RestGuardList guards(BeanStore bs) {
220 			return RestGuardList.create(bs)
221 				.append(BearerTokenGuard.create().validator(MOCK_VALIDATOR).build())
222 				.build();
223 		}
224 
225 		@RestGet(path="/me")
226 		public String me(@Auth Principal p) {
227 			return p == null ? "null" : p.getName();
228 		}
229 	}

231 	private static final MockRestClient GUARDED = MockRestClient.buildLax(GuardedResource.class);
232 
233 	@Test void b01_juneauLayer_validToken_authArgInjected() throws Exception {
234 		// BearerTokenGuard sets PRINCIPAL_ATTR; @Auth reads the same key.
235 		// This confirms AuthenticatedRequestWrapper.getAttribute(PRINCIPAL_ATTR) and the @Auth
236 		// resolver share the same stash contract.
237 		GUARDED.get("/me").header("Authorization", "Bearer user-tok")
238 			.run()
239 			.assertStatus(200)
240 			.assertContent().asString().is("alice");
241 	}
