47  	private static final String ISSUER = "https://idp.example.com";
48  	private static final String AUDIENCE = "https://sp.example.com";
49  	private static final Instant NOW = Instant.parse("2026-01-01T00:00:00Z");
50  	private static final Clock CLOCK = Clock.fixed(NOW, ZoneOffset.UTC);
51  	private static final Instant NBF = NOW.minusSeconds(60);
52  	private static final Instant NOA = NOW.plusSeconds(300);
53  
54  	private static SamlAssertionValidator validator(BasicCredential cred) {
55  		return SamlAssertionValidator.create()
56  			.spEntityId(AUDIENCE)
57  			.expectedIssuer(ISSUER)
58  			.signingCredential(cred)
59  			.clock(CLOCK)
60  			.build();
61  	}
