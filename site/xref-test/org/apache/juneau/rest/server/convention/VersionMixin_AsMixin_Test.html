30  /**
31   * Validates {@link VersionMixin} mounted as a mixin via {@code @Rest(mixins=...)} on a
32   * vanilla {@link RestServlet}.
33   *
34   * <p>
35   * Cases:
36   * <ul>
37   * 	<li>{@code /version} returns the JSON map (single SVL-configurable default mount).
38   * 	<li>{@code Content-Type: application/json}.
39   * 	<li>Importer's {@code @Bean VersionMixin} factory drives the entries map (manifest
40   * 		read, programmatic entries, custom Manifest).
41   * 	<li>Missing manifest gracefully resolves to {@code (unknown)}.
42   * </ul>
43   *
44   * <p>
45   * Per the multi-path collapse, the historical {@code /info} and {@code /about}
46   * mount aliases (formerly a triple default on a single op) are now reached via
47   * {@code -Djuneau.version.path=info} (or {@code about}); that behavior is covered by
48   * {@code VersionMixin_SvlPathOverride_Test}.
49   *
50   * @since 10.0.0
51   */
52  class VersionMixin_AsMixin_Test extends TestBase {
53  
54  	@Rest(mixins=VersionMixin.class)
55  	public static class A extends RestServlet {
56  		private static final long serialVersionUID = 1L;
57  		@RestGet(path="/items") public String items() { return "items"; }
58  	}
59  
60  	private static final MockRestClient ca = MockRestClient.buildLax(A.class);
61  
62  	@Test void a01_versionEndpointServed() throws Exception {
63  		ca.get("/version")
64  			.run()
65  			.assertStatus(200)
66  			.assertHeader("Content-Type").isContains("application/json");
67  	}
68  
69  	@Test void a02_legacyAliasesNotMountedByDefault() throws Exception {
70  		// /info and /about are no longer multi-path defaults. They only mount when
71  		// the deployer overrides juneau.version.path. Default-build hosts get 404 here.
72  		ca.get("/info").run().assertStatus(404);
73  		ca.get("/about").run().assertStatus(404);
74  	}
75  
76  	@Test void a03_defaultPayloadHasJavaVersionAtMinimum() throws Exception {
77  		var body = ca.get("/version").run().assertStatus(200).getContent().asString();
78  		var parsed = JsonParser.DEFAULT.read(body, Map.class);
79  		Assertions.assertNotNull(parsed.get("javaVersion"), "javaVersion present");
80  	}
81  
82  	@Test void a04_hostEndpointStillReachable() throws Exception {
83  		ca.get("/items").run().assertStatus(200).assertContent().asString().isContains("items");
84  	}
85  
86  	/** Host providing a programmatic version map via @Bean factory. */
87  	@Rest(mixins=VersionMixin.class)
88  	public static class B extends RestServlet {
89  		private static final long serialVersionUID = 1L;
90  		@Bean public VersionMixin version() {
91  			return VersionMixin.create()
92  				.entry("name", "my-app")
93  				.entry("version", "1.2.3")
94  				.entry("gitCommit", "abc123")
95  				.entry("gitBranch", "main")
96  				.entry("buildTime", "2026-05-24T18:00:00Z")
97  				.fromJavaVersion()
98  				.build();
99  		}
100 	}
101 
102 	private static final MockRestClient cb = MockRestClient.buildLax(B.class);
103 
104 	@Test void b01_programmaticEntriesSurface() throws Exception {
105 		var body = cb.get("/version").run().assertStatus(200).getContent().asString();
106 		var parsed = JsonParser.DEFAULT.read(body, Map.class);
107 		Assertions.assertEquals("my-app", parsed.get("name"));
108 		Assertions.assertEquals("1.2.3", parsed.get("version"));
109 		Assertions.assertEquals("abc123", parsed.get("gitCommit"));
110 		Assertions.assertEquals("main", parsed.get("gitBranch"));
111 		Assertions.assertEquals("2026-05-24T18:00:00Z", parsed.get("buildTime"));
112 		Assertions.assertNotNull(parsed.get("javaVersion"));
113 	}
114 
115 	/** Host providing a synthetic Manifest via fromManifest(Manifest). */
116 	@Rest(mixins=VersionMixin.class)
117 	public static class C extends RestServlet {
118 		private static final long serialVersionUID = 1L;
119 		@Bean public VersionMixin version() {
120 			var manifest = new Manifest();
121 			var attrs = manifest.getMainAttributes();
122 			attrs.putValue("Manifest-Version", "1.0");
123 			attrs.putValue("Implementation-Title", "synthetic-app");
124 			attrs.putValue("Implementation-Version", "9.9.9");
125 			attrs.putValue("Implementation-Vendor", "Acme");
126 			attrs.putValue("Build-Jdk", "21.0.0");
127 			return VersionMixin.create().fromManifest(manifest).build();
128 		}
129 	}
130 
131 	private static final MockRestClient cc = MockRestClient.buildLax(C.class);
132 
133 	@Test void c01_syntheticManifestSurfaces() throws Exception {
134 		var body = cc.get("/version").run().assertStatus(200).getContent().asString();
135 		var parsed = JsonParser.DEFAULT.read(body, Map.class);
136 		Assertions.assertEquals("synthetic-app", parsed.get("name"));
137 		Assertions.assertEquals("9.9.9", parsed.get("version"));
138 		Assertions.assertEquals("Acme", parsed.get("vendor"));
139 		Assertions.assertEquals("21.0.0", parsed.get("javaVersion"));
140 	}
141 
142 	@Test void d01_missingManifestGracefulFallback() {
143 		var v = VersionMixin.create()
144 			.fromManifest(new java.net.URLClassLoader(new java.net.URL[0], null))
145 			.build();
146 		Assertions.assertEquals(VersionMixin.UNKNOWN, v.getInfoMap().get("name"),
147 			"missing manifest must yield (unknown) name");
148 		Assertions.assertEquals(VersionMixin.UNKNOWN, v.getInfoMap().get("version"),
149 			"missing manifest must yield (unknown) version");
150 	}
