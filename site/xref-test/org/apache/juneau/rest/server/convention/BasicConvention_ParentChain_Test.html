28  /**
29   * Verifies that all four convention-pack mixins compose cleanly on a single host with no path
30   * collisions and that each mixin's {@code RestContext} is registered.
31   *
32   * <p>
33   * Setup: a single {@link RestServlet} host mounts all four convention mixins
34   * (favicon, SEO, version, well-known) and registers builder-configured {@code @Bean}
35   * factories for those that need explicit configuration.
36   *
37   * <p>
38   * Acceptance:
39   * <ul>
40   * 	<li>All four mixins appear in {@code RestContext.getMixinContexts()}.
41   * 	<li>Every convention path resolves (favicon.ico, robots.txt, sitemap.xml, version, info,
42   * 		about, .well-known/security.txt).
43   * 	<li>Host's own {@code /items} endpoint is unaffected.
44   * </ul>
45   *
46   * @since 10.0.0
47   */
48  class BasicConvention_ParentChain_Test extends TestBase {
49  
50  	@Rest(mixins={
51  		FaviconMixin.class,
52  		SeoMixin.class,
53  		VersionMixin.class,
54  		WellKnownMixin.class
55  	})
56  	public static class A extends RestServlet {
57  		private static final long serialVersionUID = 1L;
58  
59  		@RestGet(path="/items") public String items() { return "items"; }
60  
61  		@Bean public SeoMixin seo() {
62  			return SeoMixin.create()
63  				.robotsAllow("*", "/")
64  				.sitemapEntry("https://example.com/items")
65  				.build();
66  		}
67  
68  		@Bean public VersionMixin version() {
69  			return VersionMixin.create()
70  				.entry("name", "convention-pack")
71  				.entry("version", "10.0.0")
72  				.fromJavaVersion()
73  				.build();
74  		}
75  
76  		@Bean public WellKnownMixin wellKnown() {
77  			return WellKnownMixin.create()
78  				.securityTxt("Contact: security@example.com\nExpires: 2027-01-01T00:00:00Z\n")
79  				.build();
80  		}
81  	}
82  
83  	private static final MockRestClient c = MockRestClient.buildLax(A.class);
84  
85  	@Test void a01_allFourMixinContextsRegistered() {
86  		MockRestClient.buildLax(A.class);
87  		var hostCtx = RestContext.getGlobalRegistry().get(A.class);
88  		var ctxs = hostCtx.getMixinContexts();
89  
90  		assertNotNull(ctxs.get(FaviconMixin.class), "Favicon mixin context registered");
91  		assertNotNull(ctxs.get(SeoMixin.class), "SEO mixin context registered");
92  		assertNotNull(ctxs.get(VersionMixin.class), "Version mixin context registered");
93  		assertNotNull(ctxs.get(WellKnownMixin.class), "Well-known mixin context registered");
94  		assertEquals(4, ctxs.size(),
95  			"Expected exactly four mixin contexts; got: " + ctxs.keySet());
96  	}
97  
98  	@Test void a02_faviconResolves() throws Exception {
99  		c.get("/favicon.ico").run().assertStatus(200)
100 			.assertHeader("Content-Type").is("image/x-icon");
101 	}
102 
103 	@Test void a03_robotsTxtResolves() throws Exception {
104 		c.get("/robots.txt").run().assertStatus(200)
105 			.assertContent().asString().isContains("Allow: /");
106 	}
107 
108 	@Test void a04_sitemapXmlResolves() throws Exception {
109 		c.get("/sitemap.xml").run().assertStatus(200)
110 			.assertContent().asString().isContains("<loc>https://example.com/items</loc>");
111 	}
112 
113 	@Test void a05_versionEndpointResolves() throws Exception {
114 		c.get("/version").run().assertStatus(200)
115 			.assertContent().asString().isContains("\"name\": \"convention-pack\"");
116 	}
117 
118 	@Test void a06_infoLegacyAliasNotMountedByDefault() throws Exception {
119 		// /info is no longer a multi-path default. Migration to that secondary
120 		// alias is covered by VersionMixin_SvlPathOverride_Test#a02.
121 		c.get("/info").run().assertStatus(404);
122 	}
123 
124 	@Test void a07_aboutLegacyAliasNotMountedByDefault() throws Exception {
125 		// /about is no longer a multi-path default.
126 		c.get("/about").run().assertStatus(404);
127 	}
128 
129 	@Test void a08_securityTxtResolves() throws Exception {
130 		c.get("/.well-known/security.txt").run().assertStatus(200)
131 			.assertContent().asString().isContains("Contact: security@example.com");
132 	}
133 
134 	@Test void a09_hostEndpointStillReachable() throws Exception {
135 		c.get("/items").run().assertStatus(200)
136 			.assertContent().asString().isContains("items");
137 	}
138 }
