17  package org.apache.juneau.rest.server;
18  
19  import org.apache.juneau.*;
20  import org.apache.juneau.rest.server.servlet.*;
21  import org.apache.juneau.rest.mock.classic.*;
22  import org.junit.jupiter.api.*;
23  
24  /**
25   * Tests the default refusal of cross-site requests that use unsafe methods.
26   */
27  class Rest_CrossSite_Test extends TestBase {
28  
29  	@Rest
30  	public static class A extends BasicRestServlet {
31  		@RestPost(path="/x") public String post() { return "ok"; }
32  		@RestGet(path="/x") public String get() { return "ok"; }
33  	}
34  
35  	@Rest(allowCrossSite="true")
36  	public static class B extends BasicRestServlet {
37  		@RestPost(path="/x") public String post() { return "ok"; }
38  	}
39  
40  	@Rest(allowedOrigins="https://app.example,http://pub.example")
41  	public static class C extends BasicRestServlet {
42  		@RestPost(path="/x") public String post() { return "ok"; }
43  	}
44  
45  	private static final MockRestClient c = MockRestClient.buildLax(C.class);
46  	private static final MockRestClient a = MockRestClient.buildLax(A.class);
47  	private static final MockRestClient b = MockRestClient.buildLax(B.class);
48  
49  	@Test void a01_noFetchHeaders_passes() throws Exception {
50  		a.post("/x", "").run().assertStatus(200);
51  	}
52  
53  	@Test void a02_secFetchSiteCrossSite_refused() throws Exception {
54  		a.post("/x", "").header("Sec-Fetch-Site", "cross-site").run().assertStatus(403);
55  	}
56  
57  	@Test void a03_secFetchSiteSameOrigin_passes() throws Exception {
58  		a.post("/x", "").header("Sec-Fetch-Site", "same-origin").run().assertStatus(200);
59  	}
60  
61  	@Test void a04_originMismatch_refused() throws Exception {
62  		a.post("/x", "").header("Host", "example.com").header("Origin", "https://other.example").run().assertStatus(403);
63  	}
64  
65  	@Test void a05_originMatchesHost_passes() throws Exception {
66  		a.post("/x", "").header("Host", "example.com").header("Origin", "https://example.com").run().assertStatus(200);
67  		a.post("/x", "").header("Host", "example.com:8080").header("Origin", "http://example.com:8080").run().assertStatus(200);
68  	}
69  
70  	@Test void a06_safeMethod_notChecked() throws Exception {
71  		a.get("/x").header("Sec-Fetch-Site", "cross-site").run().assertStatus(200);
72  	}
73  
74  	@Test void a07_optOut_allowsCrossSite() throws Exception {
75  		b.post("/x", "").header("Sec-Fetch-Site", "cross-site").run().assertStatus(200);
76  	}
77  
78  	@Test void a08_listedOrigin_passes() throws Exception {
79  		c.post("/x", "").header("Host", "internal").header("Origin", "https://app.example").run().assertStatus(200);
80  	}
81  
82  	@Test void a09_unlistedOrigin_refused() throws Exception {
83  		c.post("/x", "").header("Host", "internal").header("Origin", "https://evil.example").run().assertStatus(403);
84  	}
85  
86  	@Test void a10_unlistedForwardedHost_refused() throws Exception {
87  		c.post("/x", "").header("Host", "internal").header("X-Forwarded-Host", "evil.example").header("Origin", "https://evil.example").run().assertStatus(403);
88  		a.post("/x", "").header("Host", "internal").header("X-Forwarded-Host", "evil.example").header("Origin", "https://evil.example").run().assertStatus(403);
89  	}
90  
91  	@Test void a11_listedForwardedHost_passes() throws Exception {
92  		c.post("/x", "").header("Host", "internal").header("X-Forwarded-Host", "pub.example").header("Origin", "https://pub.example").run().assertStatus(200);
93  	}
94  }
