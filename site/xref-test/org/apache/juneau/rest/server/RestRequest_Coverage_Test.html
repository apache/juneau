154 	@Rest
155 	public static class C {
156 		@RestGet("/auth")
157 		public String auth(RestRequest req) { return req.getAuthorityPath(); }
158 
159 		@RestGet("/ctx")
160 		public String ctx(RestRequest req) { return req.getContextPath(); }
161 
162 		@RestGet("/sp")
163 		public String sp(RestRequest req) { return req.getServletPath(); }
164 
165 		@RestGet("/line")
166 		public String line(RestRequest req) {
167 			var rl = req.getRequestLine();
168 			return rl.getMethod() + " " + rl.getUri() + " HTTP/" + rl.getProtocolVersion().major() + "." + rl.getProtocolVersion().minor();
169 		}
170 
171 		@RestGet("/proto")
172 		public String proto(RestRequest req) {
173 			var v = req.getProtocolVersion();
174 			return v.protocol() + "/" + v.major() + "." + v.minor();
175 		}
176 
177 		@RestGet("/uri")
178 		public String uri(RestRequest req) {
179 			return req.getUri(true, java.util.Map.of("extra", "x")).toString();
180 		}
181 
182 		@RestGet("/uriNoQuery")
183 		public String uriNoQuery(RestRequest req) {
184 			return req.getUri(false, null).toString();
185 		}
186 
187 		@RestGet("/uriCtx")
188 		public String uriCtx(RestRequest req) {
189 			var u = req.getUriContext();
190 			return u.authority + "|" + u.contextRoot + "|" + u.servletPath;
191 		}
192 
193 		@RestGet("/uriRes")
194 		public String uriRes(RestRequest req) {
195 			req.getUriResolver();
196 			req.getUriResolver(org.apache.juneau.marshall.UriResolution.ROOT_RELATIVE, org.apache.juneau.marshall.UriRelativity.RESOURCE);
197 			return "ok";
198 		}
199 	}
