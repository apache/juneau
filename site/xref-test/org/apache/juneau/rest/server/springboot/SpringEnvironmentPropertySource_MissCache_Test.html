50  	/** A map source that counts containsProperty calls per name. */
51  	static final class CountingSource extends MapPropertySource {
52  		private final Map<String,Integer> calls = new ConcurrentHashMap<>();
53  
54  		CountingSource(String name, Map<String,Object> map) {
55  			super(name, map);
56  		}
57  
58  		@Override
59  		public boolean containsProperty(String name) {
60  			calls.merge(name, 1, Integer::sum);
61  			return super.containsProperty(name);
62  		}
63  
64  		int calls(String name) {
65  			return calls.getOrDefault(name, 0);
66  		}
67  	}
68  
69  	private static CountingSource counting() {
70  		return new CountingSource("c", new HashMap<>());
71  	}
72  
73  	private static StandardEnvironment env(org.springframework.core.env.PropertySource<?> first) {
74  		var env = new StandardEnvironment();
75  		env.getPropertySources().addFirst(first);
76  		return env;
77  	}
78  
79  	private static MapPropertySource holding(String name) {
80  		return new MapPropertySource(name, Map.<String,Object>of(K, "v"));
81  	}
82  
83  	private static String value(PropertyLookupResult r) {
84  		return r.value().orElse(null);
85  	}
86  
87  	private static void assertSeenAfter(Consumer<MutablePropertySources> edit) {
88  		var env = env(counting());
89  		var src = new SpringEnvironmentPropertySource(env);
90  		assertFalse(src.get(K).isPresent());
91  		edit.accept(env.getPropertySources());
92  		assertEquals("v", value(src.get(K)));
93  	}

173 		var c = new CountingSource("c", new HashMap<>(Map.of("Present.Key", "p")));
174 		var src = new SpringEnvironmentPropertySource(env(c));
175 		var hot = List.of("Hot.A", "Hot.B", "Hot.C");
176 		hot.forEach(k -> assertFalse(src.get(k).isPresent()));
177 		for (var i = 0; i < SpringEnvironmentPropertySource.MISS_CACHE_MAX_ENTRIES + 100; i++)
178 			assertFalse(src.get("Flood.Key" + i).isPresent());
179 		assertEquals(SpringEnvironmentPropertySource.MISS_CACHE_MAX_ENTRIES, src.missCount());
180 		for (var k : hot) {
181 			assertFalse(src.get(k).isPresent());
182 			assertEquals(1, c.calls(k), "Hot key " + k + " must still be cached after the flood");
183 		}
184 		assertEquals("p", value(src.get("Present.Key")));
185 		var beyond = "Beyond.Cap";
186 		assertFalse(src.get(beyond).isPresent());
187 		assertFalse(src.get(beyond).isPresent());
188 		assertEquals(2, c.calls(beyond));

261 	@Test void a11_concurrentAddFirst_isSeenByLookupsStartedAfter() throws Exception {
262 		var env = env(counting());
263 		var src = new SpringEnvironmentPropertySource(env);
264 		var keys = List.of(K, "Other.A", "Other.B", "my.app.key");
265 		var added = new AtomicBoolean();
266 		var failures = new ConcurrentLinkedQueue<String>();
267 		var start = new CountDownLatch(1);
268 		var pool = Executors.newFixedThreadPool(8);
269 		try {
270 			var futures = new ArrayList<Future<?>>();
271 			for (var t = 0; t < 8; t++)
272 				futures.add(pool.submit(() -> {
273 					start.await();
274 					for (var i = 0; i < 20_000; i++) {
275 						var after = added.get();
276 						var r = src.get(keys.get(i % keys.size()));
277 						if (after && i % keys.size() == 0 && ! r.isPresent())
278 							failures.add("lookup " + i + " started after addFirst returned but missed");
279 					}
280 					return null;
281 				}));
282 			start.countDown();
283 			Thread.sleep(5);
284 			env.getPropertySources().addFirst(holding("added"));
285 			added.set(true);
286 			for (var f : futures)
287 				f.get(60, TimeUnit.SECONDS);
288 		} finally {
289 			pool.shutdownNow();
290 		}
291 		assertTrue(failures.isEmpty(), () -> String.join("\n", failures));
292 		assertEquals("v", value(src.get(K)));
293 	}
