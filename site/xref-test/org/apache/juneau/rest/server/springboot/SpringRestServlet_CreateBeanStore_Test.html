13  package org.apache.juneau.rest.server.springboot;
14  
15  import static org.apache.juneau.test.bct.BctAssertions.*;
16  
17  import java.util.*;
18  
19  import org.apache.juneau.*;
20  import org.apache.juneau.commons.inject.*;
21  import org.junit.jupiter.api.*;
22  import org.springframework.context.annotation.AnnotationConfigApplicationContext;
23  
24  /**
25   * Tests {@link SpringRestServlet#createBeanStore(Optional)} when the Spring {@code appContext} field is not injected.
26   *
27   * @since 10.0.0
28   */
29  @SuppressWarnings({
30  	"resource" // Stores and contexts here are test-local; no real resources are held.
31  })
32  class SpringRestServlet_CreateBeanStore_Test extends TestBase {
33  
34  	public static class SpringOnly {
35  		public String getOrigin() { return "spring"; }
36  	}
37  
38  	public static class Servlet extends BasicSpringRestServlet {
39  		private static final long serialVersionUID = 1L;
40  	}
41  
42  	@Test void a01_fieldNotInjected_returnsStore() {
43  		var store = new Servlet().createBeanStore(Optional.empty());
44  		assertBean(store, "class", "SpringBeanStore");
45  		Assertions.assertNull(store.getBean(SpringOnly.class).orElse(null));
46  	}
47  
48  	@Test void a02_fieldNotInjected_fallsBackToParent() {
49  		var parent = new BasicBeanStore(null);
50  		parent.addBean(SpringOnly.class, new SpringOnly());
51  		var store = new Servlet().createBeanStore(Optional.of(parent));
52  		assertBean(store.getBean(SpringOnly.class).orElse(null), "origin", "spring");
53  	}
54  
55  	@Test void a03_emptyOptionalInjected_returnsStore() throws Exception {
56  		var servlet = new Servlet();
57  		var f = SpringRestServlet.class.getDeclaredField("appContext");
58  		f.setAccessible(true);
59  		f.set(servlet, Optional.empty());
60  		assertBean(servlet.createBeanStore(Optional.empty()), "class", "SpringBeanStore");
61  	}
62  
63  	@Test void a04_realContext_resolvesSpringBean() throws Exception {
64  		try (var ctx = new AnnotationConfigApplicationContext()) {
65  			ctx.registerBean(SpringOnly.class);
66  			ctx.refresh();
67  			var servlet = new Servlet();
68  			var f = SpringRestServlet.class.getDeclaredField("appContext");
69  			f.setAccessible(true);
70  			f.set(servlet, Optional.of(ctx));
71  			assertBean(servlet.createBeanStore(Optional.empty()).getBean(SpringOnly.class).orElse(null), "origin", "spring");
72  		}
73  	}
74  }
