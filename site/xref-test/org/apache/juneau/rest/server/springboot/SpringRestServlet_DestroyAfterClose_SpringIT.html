17  import java.net.*;

20  import java.util.*;
21  import java.util.concurrent.atomic.*;
22  import java.util.logging.*;

24  import org.apache.juneau.rest.server.*;
25  import org.junit.jupiter.api.*;
26  import org.junit.jupiter.api.condition.*;

28  import org.springframework.boot.builder.*;
29  import org.springframework.boot.web.servlet.*;
30  import org.springframework.context.annotation.*;

32  /**
33   * Closes a Spring Boot application context running under embedded Tomcat and checks that the container's servlet
34   * destroy (which happens after the context closed) neither throws nor logs a warning.
35   *
36   * <p>
37   * Binds a socket, so it is off by default.  Run it with {@code -Djuneau.busIt=true}, outside the sandbox.
38   *
39   * @since 10.0.0
40   */
41  @EnabledIfSystemProperty(named="juneau.busIt", matches="true",
42  	disabledReason="binds a socket; run with -Dtest='SpringRestServlet_DestroyAfterClose_SpringIT' -Djuneau.busIt=true")
43  @SuppressWarnings({
44  	"resource" // The JDK HttpClient is not a resource; the test closes the Spring context itself.
45  })
46  class SpringRestServlet_DestroyAfterClose_SpringIT {
47  
48  	static final AtomicInteger DESTROYED = new AtomicInteger();
49  
50  	@Rest(path="/child")
51  	public static class Child {
52  		@RestGet(path="/x") public String x() { return "x"; }
53  	}
54  
55  	@Rest(paths="/root/*", children={Child.class})
56  	public static class Root extends BasicSpringRestServletGroup {
57  		private static final long serialVersionUID = 1L;
58  
59  		@Override /* Overridden from RestServlet */
60  		public synchronized void destroy() {
61  			DESTROYED.incrementAndGet();
62  			super.destroy();
63  		}
64  	}
65  
66  	@Configuration
67  	@EnableAutoConfiguration
68  	public static class App {
69  		@Bean public Root root() { return new Root(); }
70  		@Bean public Child child() { return new Child(); }
71  		@Bean public ServletRegistrationBean<Root> rootRegistration(Root root) { return JuneauServletRegistrations.forServlet(root, null); }
72  	}
73  
74  	private static class Capture extends Handler {
75  		final List<LogRecord> records = new ArrayList<>();
76  		@Override public void publish(LogRecord r) { records.add(r); }
77  		@Override public void flush() { /* no-op */ }
78  		@Override public void close() { /* no-op */ }
79  	}
80  
81  	@Test void a01_closeContext_containerDestroysServletQuietly() throws Exception {
82  		DESTROYED.set(0);
83  		var capture = new Capture();
84  		var root = Logger.getLogger("");
85  		root.addHandler(capture);
86  		try {
87  			var ctx = new SpringApplicationBuilder(App.class).properties("server.port=0").run();
88  			var port = ctx.getEnvironment().getProperty("local.server.port");
89  			var res = HttpClient.newHttpClient().send(HttpRequest.newBuilder(URI.create("http://127.0.0.1:" + port + "/root/child/x")).build(), BodyHandlers.ofString());
90  			assertEquals(200, res.statusCode());
91  			ctx.close();
92  		} finally {
93  			root.removeHandler(capture);
94  		}
95  		assertEquals(1, DESTROYED.get(), "servlet destroy should have run once");
96  		var bad = capture.records.stream().filter(r -> r.getLevel().intValue() >= Level.WARNING.intValue()).toList();
97  		assertTrue(bad.isEmpty(), () -> "Unexpected log records: " + bad.stream().map(r -> r.getMessage() + " " + r.getThrown()).toList());
98  	}
99  }
