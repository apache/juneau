77  	@Test void d01_installsSingleResetResistantPropagator_andSurvivesReset() {
78  		var cfg = new JuneauRestLoggingAutoConfiguration();
79  		var context = new LoggerContext();
80  		var jul = Logger.getLogger("todo368.phase3.d01");
81  		var prev = jul.getLevel();
82  		try {
83  			jul.setLevel(null);
84  			cfg.install(context, false);
85  			cfg.install(context, false);
86  
87  			var propagators = context.getCopyOfListenerList().stream()
88  				.filter(LevelChangePropagator.class::isInstance)
89  				.map(LevelChangePropagator.class::cast)
90  				.toList();
91  			assertSize(() -> "installer must be idempotent via instanceof check", 1, propagators);
92  			assertTrue(propagators.get(0).isResetResistant(), "installed listener must survive LoggerContext.reset()");
93  
94  			context.getLogger("todo368.phase3.d01").setLevel(Level.DEBUG);
95  			assertEquals(java.util.logging.Level.FINE, jul.getLevel(), "DEBUG should propagate to JUL FINE");
96  
97  			context.reset();
98  			context.getLogger("todo368.phase3.d01").setLevel(Level.TRACE);
99  			assertEquals(java.util.logging.Level.FINEST, jul.getLevel(),
100 				"propagation should remain active after reset");
101 		} finally {
102 			jul.setLevel(prev);
103 			context.stop();
104 		}
105 	}

108 	@SuppressWarnings({
109 		"java:S9357" // Handler is not a SAM (publish/flush/close); a lambda cannot replace this marker fixture.
110 	})
111 	void d02_defaultResetJulFalse_preservesExistingJulState() {
112 		var cfg = new JuneauRestLoggingAutoConfiguration();
113 		var context = new LoggerContext();
114 		var jul = Logger.getLogger("todo368.phase3.d02");
115 		var prevLevel = jul.getLevel();
116 		var prevUseParentHandlers = jul.getUseParentHandlers();
117 		var prevHandlers = jul.getHandlers();
118 		var marker = new Handler() {
119 			@Override public void publish(java.util.logging.LogRecord rec) {
120 				// No-op: this handler only marks its presence to verify it survives cfg.install(...).
121 			}
122 			@Override public void flush() {
123 				// No-op: marker handler; nothing to flush.
124 			}
125 			@Override public void close() {
126 				// No-op: marker handler; nothing to close.
127 			}
128 		};
129 		try {
130 			for (var h : prevHandlers)
131 				jul.removeHandler(h);
132 			jul.addHandler(marker);
133 			jul.setUseParentHandlers(false);
134 			jul.setLevel(java.util.logging.Level.WARNING);
135 
136 			cfg.install(context, false);
137 
138 			assertEquals(java.util.logging.Level.WARNING, jul.getLevel(),
139 				"reset-jul defaults to false, so explicit JUL levels stay intact");
140 			assertThat(jul.getHandlers()).contains(marker);
141 		} finally {
142 			jul.removeHandler(marker);
143 			for (var h : prevHandlers)
144 				jul.addHandler(h);
145 			jul.setUseParentHandlers(prevUseParentHandlers);
146 			jul.setLevel(prevLevel);
147 			context.stop();
148 		}
149 	}

171 	/** A {@link LoggerContext} whose listener-list accessor reports a propagator that is not in the real listener list. */
172 	static class ReportsPropagatorContext extends LoggerContext {
173 		int added;
174 
175 		@Override public void addListener(ch.qos.logback.classic.spi.LoggerContextListener listener) {
176 			added++;
177 			super.addListener(listener);
178 		}
179 
180 		@Override public java.util.List<ch.qos.logback.classic.spi.LoggerContextListener> getCopyOfListenerList() {
181 			return java.util.List.of(new JuneauResetResistantLevelChangePropagator());
182 		}
183 	}
184 
185 	private static long propagatorCount(LoggerContext context) {
186 		return context.getCopyOfListenerList().stream().filter(LevelChangePropagator.class::isInstance).count();
187 	}

189 	@Test void e01_otherListenerPresent_stillInstallsSinglePropagator() {
190 		var cfg = new JuneauRestLoggingAutoConfiguration();
191 		var context = new LoggerContext();
192 		try {
193 			context.addListener(org.mockito.Mockito.mock(ch.qos.logback.classic.spi.LoggerContextListener.class));
194 
195 			cfg.install(context, false);
196 			cfg.install(context, false);
197 
198 			assertEquals(1, propagatorCount(context));
199 			assertSize(2, context.getCopyOfListenerList());
200 		} finally {
201 			context.stop();
202 		}
203 	}
