33  /**
34   * Tests for {@link DebugConfig} and {@link DebugConfig.Builder}.
35   *
36   * <p>Exercises:
37   * <ul>
38   *   <li>Static factory and builder defaults.
39   *   <li>Builder fluent setters (defaultFormat / defaultLevel / conditional / defaultCacheBodies / rule).
40   *   <li>{@link DebugConfig#resolve(RestContext, HttpServletRequest)} branches:
41   *       null context, request "Debug" attribute, conditional predicate match,
42   *       no enablement signal at all.
43   *   <li>{@link DebugConfig#resolve(RestOpContext, HttpServletRequest)} method-level branch.
44   *   <li>{@code defaultCacheBodies} interaction with the resolved enablement.
45   *   <li>{@code beanStore()} accessor and bean-store-based constructor.
46   *   <li>BasicTextFormat resolution from the bean store at builder-construction time.
47   * </ul>
48   *
49   * <p>{@link DebugConfig.Builder#rule(String, Consumer)} stores the
50   * configured rule on the builder; the produced {@code DebugConfig} exposes it via
51   * {@link DebugConfig#getRuleFor(String)}.
52   */

59  	@Rest
60  	public static class R {
61  		@RestGet(path = "/x") public String x() { return "x"; }

63  
64  	private static DebugConfig.Builder newBuilder() {
65  		return DebugConfig.create(BasicBeanStore.INSTANCE);
66  	}
67  
68  	private static RestContext newRestContext() throws Exception {
69  		var resource = new R();
70  		return new RestContext(new RestContext.Args(R.class, null, null, () -> resource, "", null, null, null, RestContext.ContextKind.ROOT))
71  			.postInit().postInitChildFirst();
72  	}
73  
74  	private static RestOpContext opCtx(RestContext rc, String methodName) throws Exception {
75  		var m = R.class.getMethod(methodName);
76  		return new RestOpContext(m, rc);
77  	}

177 	@Test void b08_rule_returnsThisAndAcceptsConfigurer() {
178 		// rule(target, consumer) stores the built rule under the target key and is
179 		// retrievable via DebugConfig.getRuleFor(target).
180 		var consumed = new boolean[]{false};
181 		var b = newBuilder();
182 		var ret = b.rule("com.example.Foo", rb -> {
183 			consumed[0] = true;
184 			rb.always().level(Level.FINE);
185 		});
186 		assertSame(b, ret);
187 		assertTrue(consumed[0], "rule(target, consumer) must invoke the consumer");
188 		var cfg = b.build();
189 		var rule = cfg.getRuleFor("com.example.Foo");
190 		assertNotNull(rule, "rule must be retrievable via getRuleFor");
191 		assertEquals(Level.FINE, rule.getLevel());
192 		assertNull(cfg.getRuleFor("com.example.Bar"), "no rule should be returned for unconfigured target");
193 		// Regression: getRuleFor(null) NPE'd on the null-hostile Map.copyOf(...) map despite the "or null" @return.
194 		assertNull(cfg.getRuleFor(null), "null target must return null, not NPE");
195 	}

241 	@Test void c04_resolve_restContext_debugAttributeTrue_enables() throws Exception {
242 		// The internal resolver checks req.getAttribute("Debug") before falling through
243 		// to the conditional predicate.
244 		var ctx = newRestContext();
245 		var cfg = newBuilder().conditional(r -> false).build();
246 		var req = MockServletRequest.create("GET", "/x");
247 		req.attribute("Debug", Boolean.TRUE);
248 		var res = cfg.resolve(ctx, req);
249 		assertTrue(res.enabled(), "Debug attribute=true short-circuits before predicate");
250 	}
