30  /**
31   * Coverage tests for {@link MenuItemWidget}.
32   *
33   * <p>Most tests dispatch through a tiny {@link MockRestClient}-backed REST resource so the handler is given
34   * a real {@link RestRequest} / {@link RestResponse} pair. The handler invokes the widget's
35   * {@code getHtml/getScript/getStyle} methods and writes the result back to the client where the test
36   * asserts on the body. This exercises the branches in {@link MenuItemWidget#getHtml(RestRequest,RestResponse)}
37   * including the {@code pre}/{@code post} script paths, {@link Reader} / {@link CharSequence} / object
38   * content branches, and the private {@code getId} attribute increment.
39   */
40  @SuppressWarnings({
41  	"resource"  // Closeable resources in tests are intentionally unassigned; closing is handled by test infrastructure.
42  })
43  class MenuItemWidget_Test extends TestBase {

49  	/** Returns a CharSequence as content; no pre/post scripts. */
50  	public static class StringContentWidget extends MenuItemWidget {
51  		@Override public String getLabel(RestRequest req, RestResponse res) { return "string-label"; }
52  		@Override public Object getContent(RestRequest req, RestResponse res) { return "<p>plain string content</p>"; }
53  	}

99  	/** Returns a Reader that throws IOException on read - exercises the IOException catch branch. */
100 	public static class ThrowingReaderWidget extends MenuItemWidget {
101 		@Override public String getLabel(RestRequest req, RestResponse res) { return "throwing-reader"; }
102 		@Override public Object getContent(RestRequest req, RestResponse res) {
103 			return new Reader() {
104 				@Override public int read(char[] cbuf, int off, int len) throws IOException { throw new IOException("boom"); }
105 				@Override public void close() { /* no-op */ }
106 			};
107 		}
108 	}

165 		// Calls getHtml against a Reader that throws IOException; expects the call to throw a RuntimeException.
166 		// We capture the exception class+message and return as a string so the test can assert on it.
167 		@RestGet("/throwingReader/html")
168 		public String throwingReaderHtml(RestRequest req, RestResponse res) {
169 			try {
170 				new ThrowingReaderWidget().getHtml(req, res);
171 				return "NO_EXCEPTION";
172 			} catch (RuntimeException e) {
173 				return "EX:" + cns(e) + ":" + (e.getMessage() == null ? "" : e.getMessage());
174 			}
175 		}
176 
177 		// Calls getHtml against a bean whose getter throws; expects the call to throw a RuntimeException.
178 		@RestGet("/throwingBean/html")
179 		public String throwingBeanHtml(RestRequest req, RestResponse res) {
180 			try {
181 				new ThrowingBeanWidget().getHtml(req, res);
182 				return "NO_EXCEPTION";
183 			} catch (RuntimeException e) {
184 				return "EX:" + cns(e);
185 			}
186 		}
187 
188 		// Calls getHtml twice on the same request; verifies the LastMenuItemId attribute is preserved & incremented.
189 		@RestGet("/twice/html")
190 		public String twiceHtml(RestRequest req, RestResponse res) {
191 			var w = new BothShowScriptsWidget();
192 			var first = w.getHtml(req, res);
193 			var second = w.getHtml(req, res);
194 			return "FIRST:" + first + "|SECOND:" + second;
195 		}
196 
197 		// Forces the bean serialization branch and toggles isDebug = true.
198 		@RestGet("/bean/debug/html")
199 		public String beanDebugHtml(RestRequest req, RestResponse res) {
200 			req.setAttribute("Debug", true);
201 			return new BeanContentWidget().getHtml(req, res);
202 		}

204 		@RestGet("/script")
205 		public String script(RestRequest req, RestResponse res) {

208 		}

210 		@RestGet("/style")
211 		public String style(RestRequest req, RestResponse res) {
212 			var s = new StringContentWidget().getStyle(req, res);
213 			return s == null ? "<NULL>" : s;
214 		}
215 
216 		@RestGet("/default/script")
217 		public String defaultScript(RestRequest req, RestResponse res) {
218 			var w = new StringContentWidget();
219 			var pre = w.getBeforeShowScript(req, res);
220 			var post = w.getAfterShowScript(req, res);
221 			return "pre=" + (pre == null ? "null" : pre) + ";post=" + (post == null ? "null" : post);
222 		}

233 	@Test void a01_html_stringContent_rendersLabelAndContent() throws Exception {
234 		var html = client().get("/string/html").run().assertStatus(200).getContent().asString();
235 		assertContains(html, "menu-item");

239 		assertContains(html, "<p>plain string content</p>");
240 		// No pre/post-show script function should be emitted.
241 		assertFalse(html.contains("onPreShow"), () -> "Expected no pre-show script in:\n" + html);
242 		assertFalse(html.contains("onPostShow"), () -> "Expected no post-show script in:\n" + html);
243 	}

301 		var html = client().get("/both/html").run().assertStatus(200).getContent().asString();
