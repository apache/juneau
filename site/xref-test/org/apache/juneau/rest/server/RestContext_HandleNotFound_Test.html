19  import static jakarta.servlet.http.HttpServletResponse.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  import static org.mockito.Mockito.*;
22  
23  import java.util.function.*;
24  
25  import org.apache.juneau.http.response.*;
26  import org.junit.jupiter.api.*;
27  
28  import jakarta.servlet.*;

30  /**
31   * Regression tests for {@link RestContext#handleNotFound(RestSession)}.
32   *
33   * <p>
34   * Reaching {@code handleNotFound} means no operation matched the request, so the outcome must always be a 404 for any
35   * non-error status.  The status carried into this method mirrors {@code HttpServletResponse.getStatus()}: under a real
36   * servlet container that defaults to {@code 200}; {@link org.apache.juneau.rest.mock.MockServletResponse} used to
37   * default to a coincidental {@code 0}, which hid a straight {@code else}-throws-500 branch from the mock-based suite
38   * while it turned every unmatched path into an HTTP 500 in production.
39   * These tests stub the session status directly so both the container-default {@code 200} and an unset {@code 0}
40   * are exercised, without depending on {@code juneau-rest-mock} (which is intentionally absent from this
41   * module's test scope to avoid a Maven reactor cycle).
42   *
43   * @since 10.0.0
44   */
45  @SuppressWarnings({
46  	"java:S5778" // assertThrows lambda may invoke helpers that also throw; splitting would obscure the LNN case.
47  })
48  class RestContext_HandleNotFound_Test extends org.apache.juneau.TestBase {
49  
50  	@Rest
51  	static class A_Bare {}
52  
53  	static RestContext.Args argsOf(Class<?> resourceClass, Supplier<?> supplier) {
54  		return new RestContext.Args(resourceClass, null, null, supplier, null, null, null, null, null, null);
55  	}
56  
57  	private static RestContext bareContext() throws Exception {
58  		return new RestContext(argsOf(A_Bare.class, A_Bare::new));
59  	}
60  
61  	private static RestSession sessionWithStatus(int status) {
62  		var session = mock(RestSession.class);
63  		when(session.getStatus()).thenReturn(status);
64  		when(session.getPathInfo()).thenReturn("/missing");
65  		when(session.getMethod()).thenReturn("GET");
66  		return session;
67  	}
68  
69  	//-----------------------------------------------------------------------------------------------------------
70  	// a - A still-successful or unset status resolves to a clean 404, never a 500.
71  	//-----------------------------------------------------------------------------------------------------------
72  
73  	@Test void a01_successStatus_realContainerDefault200_throwsNotFound() {
74  		assertThrows(NotFound.class, () -> bareContext().handleNotFound(sessionWithStatus(SC_OK)));
75  	}
76  
77  	@Test void a02_unsetStatus_mockDefault0_throwsNotFound() {
78  		assertThrows(NotFound.class, () -> bareContext().handleNotFound(sessionWithStatus(0)));
79  	}
80  
81  	//-----------------------------------------------------------------------------------------------------------
82  	// b - The recognized status codes keep their existing, distinct exceptions.
83  	//-----------------------------------------------------------------------------------------------------------
84  
85  	@Test void b01_notFoundStatus_throwsNotFound() {
86  		assertThrows(NotFound.class, () -> bareContext().handleNotFound(sessionWithStatus(SC_NOT_FOUND)));
87  	}
88  
89  	@Test void b02_preconditionFailedStatus_throwsPreconditionFailed() {
90  		assertThrows(PreconditionFailed.class, () -> bareContext().handleNotFound(sessionWithStatus(SC_PRECONDITION_FAILED)));
91  	}
92  
93  	@Test void b03_methodNotAllowedStatus_throwsMethodNotAllowed() {
94  		assertThrows(MethodNotAllowed.class, () -> bareContext().handleNotFound(sessionWithStatus(SC_METHOD_NOT_ALLOWED)));
95  	}
96  
97  	//-----------------------------------------------------------------------------------------------------------
98  	// c - A genuinely unexpected error status still surfaces the "invalid method response" 500.
99  	//-----------------------------------------------------------------------------------------------------------
100 
101 	@Test void c01_unexpectedErrorStatus_throwsServletException() {
102 		var e = assertThrows(ServletException.class, () -> bareContext().handleNotFound(sessionWithStatus(SC_INTERNAL_SERVER_ERROR)));
103 		assertTrue(e.getMessage().contains("Invalid method response"));
104 	}
105 
106 	//-----------------------------------------------------------------------------------------------------------
107 	// d - Another client-error status left by a start call is kept, not turned into a 500.
108 	//-----------------------------------------------------------------------------------------------------------
109 
110 	@Test void d01_otherClientErrorStatus_keepsStatus() {
111 		for (var rc : new int[]{SC_BAD_REQUEST, SC_UNAUTHORIZED, SC_FORBIDDEN, SC_CONFLICT}) {
112 			var e = assertThrows(org.apache.juneau.http.response.BasicHttpException.class, () -> bareContext().handleNotFound(sessionWithStatus(rc)));
113 			assertEquals(rc, e.getStatusCode());
114 		}
115 	}
116 
117 	@Test void d02_serverErrorStatus_throwsServletException() {
118 		var e = assertThrows(ServletException.class, () -> bareContext().handleNotFound(sessionWithStatus(SC_SERVICE_UNAVAILABLE)));
119 		assertTrue(e.getMessage().contains("Invalid method response"));
120 	}
121 }
