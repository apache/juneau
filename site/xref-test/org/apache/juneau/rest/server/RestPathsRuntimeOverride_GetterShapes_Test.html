27  /**
28   * Validates the permissive {@link Object}-typed return contract for the {@code getPaths()} virtual
29   * getter on {@link RestServlet} / {@link RestResource}.
30   *
31   * <p>
32   * Since 10.0.0 the {@code getPaths()} signature returns {@link Object}, accepting any of:
33   * <ul>
34   * 	<li>{@code null} &mdash; no override (falls through to {@code @Rest(paths=...)} annotation default).
35   * 	<li>{@link String} &mdash; single path or comma-delimited list.
36   * 	<li>{@code String[]} &mdash; each element may itself be a comma-delimited list.
37   * 	<li>{@link Collection}, {@link List}, {@link Set}, {@link Iterable}, {@link java.util.stream.Stream Stream}.
38   * 	<li>Nested mixes of the above (recursively flattened via
39   * 		{@link org.apache.juneau.commons.utils.CollectionUtils#accumulate(Object...) CollectionUtils.accumulate}).
40   * 	<li>Primitive arrays (covered free by the flattening helper).
41   * 	<li>Any other type &mdash; coerced via {@link String#valueOf(Object)} (loose, permissive contract).
42   * </ul>
43   *
44   * <p>
45   * Each flattened leaf flows through the same per-element pipeline used for {@code @Rest(paths=...)}
46   * annotation elements: SVL substitution via the bean-store-backed
47   * {@link org.apache.juneau.commons.svl.VarResolverSession VarResolverSession}, then comma-split on the
48   * post-SVL value (each piece trimmed, empty pieces dropped).
49   *
50   * <p>
51   * Backward-compat with the prior {@code String[]}-typed signature is covered by
52   * {@link RestPathsRuntimeOverride_Getter_Test} &mdash; this test focuses on the new shapes the
53   * permissive contract enables.
54   *
55   * @since 10.0.0
56   */
57  class RestPathsRuntimeOverride_GetterShapes_Test extends TestBase {
58

60  	// a01 — getPaths() returns null → falls back to @Rest(paths) annotation default
61  	//-----------------------------------------------------------------------------------------------------------------
62  
63  	@Rest(paths={"/from-annotation"})
64  	public static class A01_ReturnsNull extends RestServlet {
65  		private static final long serialVersionUID = 1L;
66  		@Override public Object getPaths() { return null; }
67  	}
68  
69  	@Test
70  	void a01_returnsNull_fallsBackToAnnotation() throws Exception {
71  		var args = new RestContext.Args(A01_ReturnsNull.class, null, null, A01_ReturnsNull::new, "", null, null, null, RestContext.ContextKind.ROOT);
72  		var ctx = new RestContext(args).postInit().postInitChildFirst();
73  
74  		assertArrayEquals(new String[]{"/from-annotation"}, ctx.getPaths(),
75  			"null return from getPaths() should fall through to the @Rest(paths) annotation default");
76  	}

239 	// a10 — getPaths() returns List of one SVL String with an env-var default containing commas.
240 	//       The post-SVL value contains commas; comma-split happens on the SVL-resolved string, not on
241 	//       the raw element.  This matches the @Rest(paths={"$E{NAME,/x,/y}"}) annotation behavior
242 	//       documented in RestPathsRuntimeOverride_SVL_Test#e01.
243 	//-----------------------------------------------------------------------------------------------------------------
244 
245 	@Rest(paths={"/from-annotation"})
246 	public static class A10_ReturnsListWithSvlCdl extends RestServlet {
247 		private static final long serialVersionUID = 1L;
248 		@Override public Object getPaths() { return List.of("$E{REST_PATHS_TEST_VAR2,/x,/y}"); }
249 	}
250 
251 	@Test
252 	void a10_returnsListWithSvlAndCdl() throws Exception {
253 		// $E{NAME,a,b,c} concatenates everything after the first comma as the default — SVL returns
254 		// "/x,/y" here, then the post-SVL comma-split produces two mount paths.
255 		var args = new RestContext.Args(A10_ReturnsListWithSvlCdl.class, null, null, A10_ReturnsListWithSvlCdl::new, "", null, null, null, RestContext.ContextKind.ROOT);
256 		var ctx = new RestContext(args).postInit().postInitChildFirst();
257 
258 		assertArrayEquals(new String[]{"/x", "/y"}, ctx.getPaths(),
259 			"SVL with embedded commas in the default should resolve first, then the post-SVL value comma-splits — comma-split must NOT fire on the raw $E{...} element");
260 	}

319 	//-----------------------------------------------------------------------------------------------------------------
320 	// a14 — getPaths() returns an Integer → coerced via String.valueOf() (loose, permissive contract).
321 	//
322 	// Design decision: rather than throwing on non-String leaves, the contract is to use
323 	// String.valueOf(leaf).  Integer.valueOf(42) becomes the literal mount path "42" — caller's
324 	// responsibility to return paths that look like paths.  This matches the symmetry the
325 	// normalizePaths() adapter establishes for non-String leaves (Path, URI, etc.) and avoids an
326 	// instanceof ladder in the adapter.
327 	//-----------------------------------------------------------------------------------------------------------------
328 
329 	@Rest
330 	public static class A14_ReturnsInteger extends RestServlet {
331 		private static final long serialVersionUID = 1L;
332 		@Override public Object getPaths() { return Integer.valueOf(42); }
333 	}
334 
335 	@Test
336 	void a14_returnsInteger_treatedAsString() throws Exception {
337 		var args = new RestContext.Args(A14_ReturnsInteger.class, null, null, A14_ReturnsInteger::new, "", null, null, null, RestContext.ContextKind.ROOT);
338 		var ctx = new RestContext(args).postInit().postInitChildFirst();
339 
340 		assertArrayEquals(new String[]{"42"}, ctx.getPaths(),
341 			"A non-String leaf should be coerced via String.valueOf() — Integer 42 mounts on \"42\" (loose, permissive contract)");
342 	}
