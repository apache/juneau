19  import static java.nio.charset.StandardCharsets.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.util.*;
23  
24  import org.apache.juneau.*;
25  import org.apache.juneau.marshall.collections.*;

29  /**
30   * Pins {@link BusFrames} against the shared {@code bus-frames-corpus.json}.  The JS codec
31   * ({@code JuneauViews.bus.frames}) is pinned against the same file by {@code bus-frames.cjs}.
32   */
33  class BusFrames_Corpus_Test extends TestBase {
34  
35  	private static JsonMap corpus;
36  
37  	@BeforeAll static void load() throws Exception {
38  		try (var in = BusFrames_Corpus_Test.class.getResourceAsStream("/bus-frames-corpus.json")) {
39  			assertNotNull(in, "bus-frames-corpus.json not on the test classpath");
40  			corpus = JsonMap.ofString(new String(in.readAllBytes(), UTF_8), JsonParser.DEFAULT);
41  		}
42  	}
43  
44  	@SuppressWarnings("unchecked")
45  	private static List<Map<String,Object>> section(String name) {
46  		var list = (List<Map<String,Object>>)(List<?>) corpus.getList(name);
47  		assertFalse(list.isEmpty(), name);
48  		return list;
49  	}
50  
51  	@SuppressWarnings("unchecked")
52  	private static Map<String,Object> frame(Map<String,Object> c) {
53  		return (Map<String,Object>) c.get("frame");
54  	}
55  
56  	/** The corpus parser yields {@code Integer}s; {@code decode} yields {@code Long} sequence numbers.  Compare them as longs. */
57  	private static Map<String,Object> norm(Map<String,Object> frame) {
58  		var m = new LinkedHashMap<>(frame);
59  		if (m.get("seq") instanceof Number n)
60  			m.put("seq", n.longValue());
61  		return m;
62  	}
63  
64  	@Test void a00_version() {
65  		assertEquals(1, ((Number) corpus.get("version")).intValue());
66  	}
67  
68  	@Test void a01_encode() {
69  		for (var c : section("encode"))
70  			assertEquals(c.get("text"), BusFrames.encode(frame(c)), String.valueOf(c.get("name")));
71  	}
72  
73  	@Test void a02_decode() throws Exception {
74  		for (var c : section("decode"))
75  			assertEquals(norm(frame(c)), norm(BusFrames.decode((String) c.get("text"))), String.valueOf(c.get("name")));
76  	}
77  
78  	@Test void a03_bad() {
79  		for (var c : section("bad")) {
80  			var name = String.valueOf(c.get("name"));
81  			var e = assertThrows(BusRefusal.class, () -> BusFrames.decode((String) c.get("text")), name);
82  			assertEquals("bus:bad-frame", e.code(), name);
83  			assertEquals(400, e.status(), name);
84  		}
85  	}
86  
87  	@Test void a04_encodeThenDecodeRoundTrips() throws Exception {
88  		for (var c : section("encode"))
89  			assertEquals(norm(frame(c)), norm(BusFrames.decode(BusFrames.encode(frame(c)))), String.valueOf(c.get("name")));
90  	}
91  }
