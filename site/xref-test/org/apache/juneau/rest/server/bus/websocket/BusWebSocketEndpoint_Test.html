39  /**
40   * The per-connection endpoint, against a real {@link ServerBus} (sessions from the real session POST): it closes
41   * refused handshakes, connects an adapted {@link Session} (resync first), forwards inbound text to the bus, and closes
42   * with 1009 on oversize.  Only the jakarta.websocket interfaces are mocked.
43   */
44  @SuppressWarnings({
45  	"resource" // The shared MockRestClient lives for the whole class; each test's bus is closed in @AfterEach.

53  		@Override public ServerBus serverBus() { return bus; }
54  	}

56  	private static final MockRestClient C = MockRestClient.buildLax(R.class);
57  	private static final Principal JB = () -> "jb";
58  	private static final int MAX_BYTES = 256;
59  	private static final String CANCEL = "{\"v\":1,\"type\":\"pub\",\"topic\":\"ops.cancel-all\",\"payload\":{\"reason\":\"r1\"}}";

72  	@BeforeEach void setUp() throws Exception {
73  		bus = ServerBus.create(BusPolicy.create()
74  			.downstream("ops.jobs", true)
75  			.upstream("ops.cancel-all", (s, t, p) -> handled.add(p.get("reason")))
76  			.allowedOrigins("https://ops.example.com")
77  			.maxFrameBytes(MAX_BYTES)
78  			.maxQueuedFrames(2)
79  			.build());
80  		bus.enableWebSocket(BusWebSockets.PREFIX);
81  		R.bus = bus;
82  		var id = BusFixture.open(C, JB, "websocket", List.of("ops.jobs"), List.of("ops.cancel-all"));
83  		busSession = bus.session(id, JB).orElseThrow();
84  		ws = mock(Session.class);
85  		remote = mock(RemoteEndpoint.Async.class);
86  		config = mock(EndpointConfig.class);
87  		props = new HashMap<>();
88  		when(ws.getAsyncRemote()).thenReturn(remote);
89  		when(config.getUserProperties()).thenReturn(props);
90  		doAnswer(i -> {
91  			sent.add(i.getArgument(0));
92  			if (autoComplete)
93  				i.<SendHandler>getArgument(1).onResult(new SendResult(ws));
94  			return null;
95  		}).when(remote).sendText(anyString(), any(SendHandler.class));
96  	}

102 	private BusWebSocketEndpoint endpoint() {
103 		return new BusWebSocketEndpoint(bus, closer::add);

106 	private CloseReason closedWith() throws IOException {
107 		var c = ArgumentCaptor.forClass(CloseReason.class);
108 		verify(ws).close(c.capture());

112 	private void open(BusWebSocketEndpoint e) {

219 	@Test void c02_aSlowConsumerIsClosedWith4429_onAnotherThread() throws Exception {
220 		autoComplete = false;
221 		open(endpoint());                                      // resync-begin in flight, resync-end queued (1 of 2)
222 		bus.publish("ops.jobs", JsonMap.of("n", 1));           // queued (2 of 2)
223 		bus.publish("ops.jobs", JsonMap.of("n", 2));           // full: the bus detaches the sink while holding its lock
224 		verify(ws, never()).close(any(CloseReason.class));
225 		assertFalse(busSession.isConnected());
226 		var r = closer.poll();
227 		assertNotNull(r, "the close must have been handed off");
228 		r.run();
229 		var c = closedWith();
230 		assertEquals(4429, c.getCloseCode().getCode());
231 		assertEquals("bus:slow-consumer", c.getReasonPhrase());
232 	}

274 	@Test void d05_errorLogsOnlyTheExceptionClass() {
275 		var records = new ArrayList<java.util.logging.LogRecord>();
276 		var log = java.util.logging.Logger.getLogger(BusWebSocketEndpoint.class.getName());
277 		var handler = new java.util.logging.Handler() {
278 			@Override public void publish(java.util.logging.LogRecord r) { records.add(r); }
279 			@Override public void flush() { /* nothing buffered */ }
280 			@Override public void close() { /* nothing to release */ }
281 		};
282 		var old = log.getLevel();
283 		log.setLevel(java.util.logging.Level.ALL);
284 		log.addHandler(handler);
285 		try {
286 			var e = endpoint();
287 			open(e);
288 			e.onError(ws, new IOException("secret-session-id"));
289 		} finally {
290 			log.removeHandler(handler);
291 			log.setLevel(old);
292 		}
293 		assertFalse(records.isEmpty());
294 		for (var r : records) {
295 			assertNull(r.getThrown());
296 			var text = java.text.MessageFormat.format(r.getMessage(), r.getParameters());
297 			assertFalse(text.contains("secret-session-id"), text);
298 		}
299 		assertTrue(records.stream().anyMatch(r -> r.getParameters() != null && r.getParameters()[0].equals(IOException.class.getName())));
300 	}

306 	@Test void e01_theMessageHandlerIsRegisteredAfterTheBusConnects() {
307 		var handlerAtFirstSend = new boolean[] {false};
308 		var added = new boolean[] {false};
309 		doAnswer(i -> { added[0] = true; return null; }).when(ws).addMessageHandler(eq(String.class), ArgumentMatchers.<MessageHandler.Whole<String>>any());
310 		doAnswer(i -> {
311 			if (sent.isEmpty())
312 				handlerAtFirstSend[0] = added[0];
313 			sent.add(i.getArgument(0));
314 			i.<SendHandler>getArgument(1).onResult(new SendResult(ws));
315 			return null;
316 		}).when(remote).sendText(anyString(), any(SendHandler.class));
317 		open(endpoint());
318 		assertFalse(handlerAtFirstSend[0], "the resync went out before any inbound frame could be handled");
319 		assertTrue(added[0]);
320 	}

322 	@Test void e02_aFailedConnectClosesWith1011_andRegistersNoHandler() throws Exception {
323 		props.put(BusWebSocketConfigurator.SESSION_PROPERTY, mock(BusSession.class));
324 		var records = new ArrayList<java.util.logging.LogRecord>();
325 		var log = java.util.logging.Logger.getLogger(BusWebSocketEndpoint.class.getName());
326 		var handler = new java.util.logging.Handler() {
327 			@Override public void publish(java.util.logging.LogRecord r) { records.add(r); }
328 			@Override public void flush() { /* nothing buffered */ }
329 			@Override public void close() { /* nothing to release */ }
330 		};
331 		var useParent = log.getUseParentHandlers();
332 		log.setUseParentHandlers(false);  // the expected warning stays out of the test output
333 		log.addHandler(handler);
334 		try {
335 			endpoint().onOpen(ws, config);
336 		} finally {
337 			log.removeHandler(handler);
338 			log.setUseParentHandlers(useParent);
339 		}
340 		assertEquals(1, records.size());
341 		assertEquals(java.util.logging.Level.WARNING, records.get(0).getLevel());
342 		var r = closedWith();
343 		assertEquals(1011, r.getCloseCode().getCode());
344 		assertEquals("bus:connect-failed", r.getReasonPhrase());
345 		verify(ws, never()).addMessageHandler(eq(String.class), ArgumentMatchers.<MessageHandler.Whole<String>>any());
346 	}
