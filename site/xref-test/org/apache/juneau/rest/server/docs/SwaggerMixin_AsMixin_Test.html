17  package org.apache.juneau.rest.server.docs;
18  
19  import org.apache.juneau.*;
20  import org.apache.juneau.rest.mock.classic.*;
21  import org.apache.juneau.rest.server.*;
22  import org.apache.juneau.rest.server.servlet.*;
23  import org.junit.jupiter.api.*;
24  
25  /**
26   * Validates {@link SwaggerMixin} mounted as a mixin via {@code @Rest(mixins=...)} on a
27   * vanilla {@link RestServlet}.
28   *
29   * <p>
30   * Acceptance:
31   * <ul>
32   * 	<li>{@code GET /api} (no Accept) returns the Swagger document — this mixin uses Juneau's
33   * 		standard content negotiation, so the serializer-default media type wins (HTML by default).
34   * 	<li>{@code Accept: application/json} returns the Swagger v2 spec as JSON.
35   * 	<li>{@code Accept: text/html} renders the Swagger-UI HTML view.
36   * 	<li>The host's own endpoints are unaffected.
37   * </ul>
38   */
39  class SwaggerMixin_AsMixin_Test extends TestBase {
40  
41  	@Rest(mixins=SwaggerMixin.class)
42  	public static class A extends BasicRestServlet {
43  		private static final long serialVersionUID = 1L;
44  		@RestGet(path="/items") public String items() { return "items"; }
45  	}
46  
47  	private static final MockRestClient c = MockRestClient.buildLax(A.class);
48  
49  	@Test void a01_apiServesSwaggerSpec_jsonAccept() throws Exception {
50  		c.get("/api")
51  			.accept("application/json")
52  			.run()
53  			.assertStatus(200)
54  			.assertContent().asString().isContains("\"swagger\":\"2.0\"");
55  	}
56  
57  	@Test void a02_apiServesSwaggerUi_htmlAccept() throws Exception {
58  		c.get("/api")
59  			.accept("text/html")
60  			.run()
61  			.assertStatus(200)
62  			.assertContent().asString().isContains("<html");
63  	}
64  
65  	@Test void a03_hostEndpointStillReachable() throws Exception {
66  		c.get("/items")
67  			.accept("application/json")
68  			.run()
69  			.assertStatus(200)
70  			.assertContent().asString().isContains("items");
71  	}
72  }
