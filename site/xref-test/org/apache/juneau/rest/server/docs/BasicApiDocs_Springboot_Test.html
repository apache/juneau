36  /**
37   * Real-Spring-Boot deployment-parity assertion for the api-docs mixin pack.
38   *
39   * <p>
40   * Boots a full Spring Boot context with embedded Tomcat on a random port, registers a
41   * {@link BasicSpringRestServlet}-based host wired up with the four-mixin pack
42   * ({@link SwaggerUiMixin} + {@link RedocMixin} &mdash; transitively pulling in
43   * {@link SwaggerMixin} + {@link OpenApiMixin}) via {@link ServletRegistrationBean},
44   * and hits the six canonical api-docs URLs over real HTTP via {@link HttpClient}.
45   *
46   * <p>
47   * Each URL is validated against the shared
48   * {@link BasicApiDocsTestFixtures#sixUrlAssertions() fixture} so any drift between this Spring
49   * Boot path and the {@code MockRest} baseline ({@link BasicApiDocs_TransitiveDedupe_Test}) or the
50   * real-Jetty path ({@link BasicApiDocs_JettyMicroservice_Test}) is caught immediately.
51   *
52   * <p>
53   * <b>What this catches that {@code MockRest} and the Jetty path cannot:</b>
54   * <ul>
55   * 	<li>Spring's bean store adapter ({@link SpringBeanStore}) resolving the Juneau
56   * 		{@code OpenApiProvider} / {@code SwaggerProvider} beans through
57   * 		{@link org.springframework.context.ApplicationContext#getBeanProvider(Class)
58   * 		ApplicationContext.getBeanProvider(...)} end-to-end.
59   * 	<li>The {@code SpringRestServlet} {@code @Autowired ApplicationContext} field being populated
60   * 		by Spring before {@link jakarta.servlet.Servlet#init(jakarta.servlet.ServletConfig)
61   * 		Servlet.init()} runs.
62   * 	<li>Real embedded-Tomcat {@code Accept} negotiation across the docs endpoints &mdash; including
63   * 		the {@code defaultAccept="text/html"} hook on UI mixins.
64   * </ul>
65   *
66   * <p>
67   * Uses the JDK's built-in {@link HttpClient} rather than Spring's {@code TestRestTemplate} (which
68   * was removed/restructured in Spring Boot 4.0). Same client used by
69   * {@link BasicApiDocs_JettyMicroservice_Test} so the wire-level call shape is byte-identical
70   * across both real-container tests.
71   *
72   * @since 10.0.0
73   */
74  @org.apache.juneau.testing.SpringbootTest
75  @SpringBootTest(classes = BasicApiDocs_Springboot_Test.TestApp.class,
76  	webEnvironment = WebEnvironment.RANDOM_PORT)
77  @DirtiesContext(classMode = DirtiesContext.ClassMode.AFTER_CLASS)
78  class BasicApiDocs_Springboot_Test {
79  
80  	/**
81  	 * Minimal Spring Boot application that registers a single {@link Host} servlet at {@code /*}.
82  	 *
83  	 * <p>
84  	 * Stays narrow on purpose: no extra autoconfiguration, no global filters, no security &mdash;
85  	 * we want to exercise the Juneau mixin walk + serializer pipeline through Spring's bean store,
86  	 * nothing else.
87  	 */
88  	@SpringBootConfiguration
89  	@EnableAutoConfiguration
90  	public static class TestApp {
91  
92  		@Bean
93  		public Host hostServlet() {
94  			return new Host();
95  		}
96  
97  		@Bean
98  		public ServletRegistrationBean<Host> hostRegistration(Host servlet) {
99  			return new ServletRegistrationBean<>(servlet, "/*");
100 		}
101 	}
102 
103 	/**
104 	 * Test host: subclass of {@link BasicSpringRestServlet} carrying the four-mixin pack.
105 	 *
106 	 * <p>
107 	 * {@link BasicSpringRestServlet} itself declares only an empty {@code @Rest} (no mixins) so
108 	 * this subclass adds them explicitly. {@code @Rest} is {@link java.lang.annotation.Inherited
109 	 * Inherited}, so this child annotation aggregates with the parent's via the mixin walk
110 	 * (parent-first, child-appended; {@code LinkedHashSet} dedupes by class identity).
111 	 */
112 	@Rest(mixins = {SwaggerUiMixin.class, RedocMixin.class})
113 	public static class Host extends BasicSpringRestServlet {
114 		private static final long serialVersionUID = 1L;
115 	}
116 
117 	@LocalServerPort
118 	int port;
119 
120 	private static final HttpClient HTTP = HttpClient.newBuilder()
121 		.connectTimeout(Duration.ofSeconds(5))
122 		.followRedirects(HttpClient.Redirect.NEVER)
123 		.build();
124 
125 	private BasicApiDocsTestFixtures.Response get(String path, String accept) {
126 		try {
127 			var reqBuilder = HttpRequest.newBuilder()
128 				.uri(URI.create("http://localhost:" + port + path))
129 				.timeout(Duration.ofSeconds(10))
130 				.GET();
131 			if (accept != null)
132 				reqBuilder.header("Accept", accept);
133 			var resp = HTTP.send(reqBuilder.build(), BodyHandlers.ofString());
134 			var ct = resp.headers().firstValue("Content-Type").orElse(null);
135 			return new BasicApiDocsTestFixtures.Response(resp.statusCode(), ct, resp.body());
136 		} catch (Exception e) {
137 			throw new IllegalStateException("HTTP call failed for " + path + " (Accept=" + accept + ")", e);
138 		}
139 	}
140 
141 	@Test void a01_allSixUrlsServeExpectedShape() {
142 		BasicApiDocsTestFixtures.assertAllSixUrls(this::get);
143 	}
144 }
