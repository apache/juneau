29  /**
30   * Verifies transitive mixin resolution dedupes correctly when a host composes the two UI mixins
31   * together.
32   *
33   * <p>
34   * {@code @Rest(mixins={SwaggerUiMixin.class, RedocMixin.class})} should pull in
35   * {@code SwaggerMixin} (via {@code SwaggerUiMixin}) and
36   * {@code OpenApiMixin} (via {@code RedocMixin}) exactly once each &mdash; not twice,
37   * even though the user could equivalently have listed all four mixins explicitly.
38   *
39   * <p>
40   * Acceptance:
41   * <ul>
42   * 	<li>The four api-docs mixin classes are present in the host's {@code mixinContexts} map, alongside the
43   * 		four residual op-mixins ({@code ErrorMixin}, {@code HtdocMixin}, {@code StatsMixin},
44   * 		{@code FaviconMixin}) contributed by {@code BasicRestServlet}.
45   * 	<li>Each mixin appears exactly once (no duplicates) &mdash; eight in total.
46   * 	<li>All six api-docs URL paths resolve correctly.
47   * </ul>
48   */
49  class BasicApiDocs_TransitiveDedupe_Test extends TestBase {
50  
51  	@Rest(mixins={SwaggerUiMixin.class, RedocMixin.class})
52  	public static class A extends BasicRestServlet {
53  		private static final long serialVersionUID = 1L;
54  	}
55  
56  	private static final MockRestClient c = MockRestClient.buildLax(A.class);
57  
58  	@Test void a01_allMixinContextsPresent() {
59  		MockRestClient.buildLax(A.class);
60  		var hostCtx = RestContext.getGlobalRegistry().get(A.class);
61  		var contexts = hostCtx.getMixinContexts();
62  
63  		// api-docs pack — deduped from A's explicit SwaggerUiMixin/RedocMixin + BasicRestServlet's own list
64  		// plus the transitive SwaggerMixin/OpenApiMixin pulls (each present exactly once).
65  		assertNotNull(contexts.get(SwaggerUiMixin.class));
66  		assertNotNull(contexts.get(SwaggerMixin.class));
67  		assertNotNull(contexts.get(RedocMixin.class));
68  		assertNotNull(contexts.get(OpenApiMixin.class));
69  		// Residual op-mixins contributed by BasicRestServlet.
70  		assertNotNull(contexts.get(ErrorMixin.class));
71  		assertNotNull(contexts.get(HtdocMixin.class));
72  		assertNotNull(contexts.get(StatsMixin.class));
73  		assertNotNull(contexts.get(FaviconMixin.class));
74  
75  		assertEquals(8, contexts.size(),
76  			"Expected exactly eight mixin contexts (no duplicates); got: " + contexts.keySet());
77  	}
78  
79  	@Test void a02_apiPathReachable() throws Exception {
80  		c.get("/api").accept("application/json").run().assertStatus(200)
81  			.assertContent().asString().isContains("\"swagger\":\"2.0\"");
82  	}
83  
84  	@Test void a03_swaggerPathReachable() throws Exception {
85  		c.get("/swagger").run().assertStatus(200).assertContent().asString().isContains("<html");
86  	}
87  
88  	@Test void a04_openapiPathReachable() throws Exception {
89  		c.get("/openapi").accept("application/json").run().assertStatus(200)
90  			.assertContent().asString().isContains("\"openapi\"");
91  	}
92  
93  	@Test void a05_openapiJsonPathReachable() throws Exception {
94  		c.get("/openapi.json").run().assertStatus(200)
95  			.assertContent().asString().isContains("\"openapi\"");
96  	}
97  
98  	@Test void a06_openapiYamlPathReachable() throws Exception {
99  		c.get("/openapi.yaml").run().assertStatus(200)
100 			.assertContent().asString().isContains("openapi:");
101 	}
102 
103 	@Test void a07_redocPathReachable() throws Exception {
104 		c.get("/redoc").run().assertStatus(200).assertContent().asString().isContains("<html");
105 	}
106 }
