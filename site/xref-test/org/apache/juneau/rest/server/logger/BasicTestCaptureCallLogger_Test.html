32  /**
33   * Tests for {@link BasicTestCaptureCallLogger} — the in-memory capturing test logger.
34   *
35   * @since 10.0.0
36   */
37  class BasicTestCaptureCallLogger_Test extends TestBase {
38  
39  	private static HttpServletRequest req(Map<String,Object> attrs) {
40  		var r = mock(HttpServletRequest.class);
41  		when(r.getMethod()).thenReturn("GET");
42  		when(r.getRequestURI()).thenReturn("/foo");
43  		when(r.getHeaderNames()).thenReturn(Collections.enumeration(List.of()));
44  		for (var e : attrs.entrySet())
45  			when(r.getAttribute(e.getKey())).thenReturn(e.getValue());
46  		return r;
47  	}
48  
49  	private static HttpServletResponse res(int status) {
50  		var r = mock(HttpServletResponse.class);
51  		when(r.getStatus()).thenReturn(status);
52  		when(r.getHeaderNames()).thenReturn(Set.of());
53  		return r;
54  	}

56  	// -----------------------------------------------------------------------------------------
57  	// a — construction
58  	// -----------------------------------------------------------------------------------------
59  
60  	@Test void a01_construct_defaultConstructor() {
61  		assertNotNull(new BasicTestCaptureCallLogger());
62  	}
63  
64  	@SuppressWarnings("resource") // BasicBeanStore is a short-lived in-memory test fixture backed by a Map; nothing external to leak.
65  	@Test void a02_construct_withBeanStore() {
66  		assertNotNull(new BasicTestCaptureCallLogger(new BasicBeanStore()));
67  	}
68  
69  	// -----------------------------------------------------------------------------------------
70  	// b — getters/assertions with nothing logged yet
71  	// -----------------------------------------------------------------------------------------
72  
73  	@Test void b01_noMessageLogged_gettersReturnNull() {
74  		var l = new BasicTestCaptureCallLogger();
75  		assertNull(l.getMessage());
76  		assertNull(l.getLevel());
77  		assertNull(l.getThrown());
78  		assertNull(l.getMessageAndReset());
79  	}
80  
81  	// -----------------------------------------------------------------------------------------
82  	// c — log(): normal (non-debug) rules
83  	// -----------------------------------------------------------------------------------------
84  
85  	@Test void c01_status500_logsWithSevereAndHeaderDetail() {
86  		var l = new BasicTestCaptureCallLogger();
87  		l.log(req(Map.of()), res(500));
88  		assertNotNull(l.getMessage());
89  		assertEquals(Level.SEVERE, l.getLevel());
90  	}
91  
92  	@Test void c02_status400_logsWithWarningAndStatusLineDetail() {
93  		var l = new BasicTestCaptureCallLogger();
94  		l.log(req(Map.of()), res(400));
95  		assertNotNull(l.getMessage());
96  		assertEquals(Level.WARNING, l.getLevel());
97  	}
98  
99  	@Test void c03_status200_notLogged() {
100 		var l = new BasicTestCaptureCallLogger();
101 		l.log(req(Map.of()), res(200));
102 		assertNull(l.getMessage());
103 	}
104 
105 	// -----------------------------------------------------------------------------------------
106 	// d — log(): debug rules
107 	// -----------------------------------------------------------------------------------------
108 
109 	@Test void d01_debugEnabled_logsWithSevereAndEntityDetail() {
110 		var debugConfig = DebugConfig.create(BasicBeanStore.INSTANCE).defaultLevel(Level.SEVERE).build();
111 		var l = new BasicTestCaptureCallLogger();
112 		l.log(req(Map.of("DebugConfig", debugConfig, "Debug", Boolean.TRUE)), res(200));
113 		assertNotNull(l.getMessage());
114 		assertEquals(Level.SEVERE, l.getLevel());
115 	}

121 	@Test void e01_assertMessage_containsExpectedText() {
122 		var l = new BasicTestCaptureCallLogger();
123 		l.log(req(Map.of()), res(500));
124 		l.assertMessage().isContains("HTTP GET /foo");
125 	}
126 
127 	@Test void e02_assertMessageAndReset_clearsAfterwards() {
128 		var l = new BasicTestCaptureCallLogger();
129 		l.log(req(Map.of()), res(500));
130 		l.assertMessageAndReset().isContains("HTTP GET /foo");
131 		assertNull(l.getMessage());
132 	}
133 
134 	@Test void e03_assertThrown_withLoggedException() {
135 		var l = new BasicTestCaptureCallLogger();
136 		l.log(req(Map.of()), res(500));
137 		l.assertThrown().isNull();
138 	}
139 
140 	@Test void e04_reset_returnsThisAndClearsMessage() {
141 		var l = new BasicTestCaptureCallLogger();
142 		l.log(req(Map.of()), res(500));
143 		assertSame(l, l.reset());
144 		assertNull(l.getMessage());
145 	}
