57  	/** {@link AsyncContext} that records what the processor does to it. */
58  	static final class FakeAsyncContext implements AsyncContext {
59  		final List<AsyncListener> listeners = new ArrayList<>();
60  		final AtomicInteger completes = new AtomicInteger();
61  		volatile long timeout = -1;
62  		boolean completeThrows;
63  
64  		@Override public ServletRequest getRequest() { return null; }
65  		@Override public ServletResponse getResponse() { return null; }
66  		@Override public boolean hasOriginalRequestAndResponse() { return true; }
67  		@Override public void dispatch() { /* unused */ }
68  		@Override public void dispatch(String path) { /* unused */ }
69  		@Override public void dispatch(ServletContext context, String path) { /* unused */ }
70  		@Override public void complete() {
71  			completes.incrementAndGet();
72  			if (completeThrows)
73  				throw new IllegalStateException("already completed");
74  		}
75  		@Override public void start(Runnable run) { /* unused */ }
76  		@Override public void addListener(AsyncListener listener) { listeners.add(listener); }
77  		@Override public void addListener(AsyncListener listener, ServletRequest req, ServletResponse res) { listeners.add(listener); }
78  		@Override public <T extends AsyncListener> T createListener(Class<T> clazz) { return null; }
79  		@Override public void setTimeout(long ms) { timeout = ms; }
80  		@Override public long getTimeout() { return timeout; }
81  	}

99  		boolean committed;
100 		boolean failFlush;
101 		boolean failWrite;
102 		boolean sendErrorThrows;
103 		int sendErrors;
104 
105 		@Override public boolean isCommitted() { return committed; }
106 		@Override public void flushBuffer() throws IOException {
107 			if (failFlush)
108 				throw new IOException("flush failed");
109 		}
110 		@Override public void sendError(int sc) throws IOException {
111 			sendErrors++;
112 			if (sendErrorThrows)
113 				throw new IOException("sendError failed");
114 			super.sendError(sc);
115 		}
116 		@Override public ServletOutputStream getOutputStream() throws IOException {
117 			if (! failWrite)
118 				return super.getOutputStream();
119 			return new ServletOutputStream() {
120 				@Override public boolean isReady() { return true; }
121 				@Override public void setWriteListener(WriteListener l) { /* unused */ }
122 				@Override public void write(int b) throws IOException { throw new IOException("client gone"); }
123 			};
124 		}

127 	/** Publisher that records its subscription and hands the subscriber to a callback. */
128 	static final class ScriptedPublisher<T> implements Flow.Publisher<T> {
129 		final AtomicBoolean cancelled = new AtomicBoolean();
130 		private final java.util.function.Consumer<Flow.Subscriber<? super T>> script;
131 		ScriptedPublisher(java.util.function.Consumer<Flow.Subscriber<? super T>> script) { this.script = script; }
132 
133 		@Override public void subscribe(Flow.Subscriber<? super T> sub) {
134 			sub.onSubscribe(new Flow.Subscription() {
135 				@Override public void request(long n) { /* script drives emission */ }
136 				@Override public void cancel() { cancelled.set(true); }
137 			});
138 			script.accept(sub);
139 		}
140 	}

142 	/** Emits one event on the first request, then throws the supplied exception from the second request. */
143 	static Flow.Publisher<SseEvent> failingSecondRequest(RuntimeException failure) {
144 		return sub -> sub.onSubscribe(new Flow.Subscription() {
145 			private int requests;
146 			@Override public void request(long n) {
147 				if (++requests == 1)
148 					sub.onNext(new SseEvent("tick", "first"));
149 				else
150 					throw failure;
151 			}
152 			@Override public void cancel() { /* nothing to release */ }
153 		});
154 	}

245 	private static final Logger LOG = Logger.getLogger(ReactiveResponseProcessor.class.getName());
246 	private final List<LogRecord> records = new ArrayList<>();
247 	private final Handler handler = new Handler() {
248 		@Override public void publish(LogRecord r) { records.add(r); }
249 		@Override public void flush() { /* nothing buffered */ }
250 		@Override public void close() { /* nothing to release */ }
251 	};
252 	private Level savedLevel;
253 
254 	@BeforeEach void captureLog() {
255 		savedLevel = LOG.getLevel();
256 		LOG.setLevel(Level.ALL);
257 		handler.setLevel(Level.ALL);
258 		LOG.addHandler(handler);
259 	}
260 
261 	@AfterEach void releaseLog() {
262 		LOG.removeHandler(handler);
263 		LOG.setLevel(savedLevel);
264 	}

266 	/** Message of the first captured record containing the text, or null. */
267 	private String logged(String text) {
268 		return records.stream().map(r -> r.getMessage() == null ? "" : r.getMessage()).filter(m -> m.contains(text)).findFirst().orElse(null);
269 	}

300 	@Test void a01_inlinePublisher_streamsThenCompletesAsyncContextOnce() throws Exception {
301 		var req = request("/list");
302 		var res = new TestResponse();
303 		run(req, res);
304 
305 		assertEquals("event: tick\ndata: one\n\nevent: tick\ndata: two\n\n", body(res));
306 		assertEquals(0L, req.async.timeout, "stream must not use a container timeout");
307 		assertSize(1, req.async.listeners);
308 		assertEquals(1, req.async.completes.get());
309 		assertEquals(0, res.sendErrors);
310 
311 		// The container's own onComplete notification is a gated no-op.
312 		req.async.listeners.get(0).onComplete(event(req, null));
313 		req.async.listeners.get(0).onStartAsync(event(req, null));
314 		assertEquals(1, req.async.completes.get());
315 	}
