57  	/**
58  	 * Trampolined synchronous {@link Flow.Publisher Flow.Publisher} that replays a
59  	 * fixed list on subscribe, honoring {@code request(n)} without recursing on re-request (so the
60  	 * one-at-a-time streaming subscriber is exercised deterministically on a single thread).
61  	 */
62  	static final class ListPublisher<T> implements Flow.Publisher<T> {
63  		private final List<T> items;
64  		ListPublisher(List<T> items) { this.items = items; }
65  
66  		@SafeVarargs
67  		static <T> ListPublisher<T> of(T... items) { return new ListPublisher<>(List.of(items)); }
68  
69  		@Override public void subscribe(Flow.Subscriber<? super T> sub) {
70  			sub.onSubscribe(new Flow.Subscription() {
71  				private int idx;
72  				private final AtomicLong demand = new AtomicLong();
73  				private boolean draining;
74  				private volatile boolean cancelled;
75  
76  				@Override public void request(long n) {
77  					if (cancelled)
78  						return;
79  					demand.addAndGet(n);
80  					if (draining)
81  						return;
82  					draining = true;
83  					try {
84  						while (! cancelled && demand.get() > 0 && idx < items.size()) {
85  							demand.decrementAndGet();
86  							sub.onNext(items.get(idx++));
87  						}
88  						if (! cancelled && idx >= items.size())
89  							sub.onComplete();
90  					} finally {
91  						draining = false;
92  					}
93  				}
94  
95  				@Override public void cancel() { cancelled = true; }
96  			});
97  		}
98  	}

100 	/** Publisher that emits one element then errors. */
101 	static final class ErrorPublisher<T> implements Flow.Publisher<T> {
102 		private final T first;
103 		private final boolean emitFirst;
104 		ErrorPublisher(T first, boolean emitFirst) { this.first = first; this.emitFirst = emitFirst; }
105 
106 		@Override public void subscribe(Flow.Subscriber<? super T> sub) {
107 			sub.onSubscribe(new Flow.Subscription() {
108 				private boolean done;
109 				@Override public void request(long n) {
110 					if (done)
111 						return;
112 					done = true;
113 					if (emitFirst)
114 						sub.onNext(first);
115 					sub.onError(new RuntimeException("boom"));
116 				}
117 				@Override public void cancel() { done = true; }
118 			});
119 		}
120 	}

700 	static final class SilentPublisher<T> implements Flow.Publisher<T> {
701 		final AtomicBoolean cancelled = new AtomicBoolean();
702 		final boolean interruptOnSubscribe;
703 		SilentPublisher(boolean interruptOnSubscribe) { this.interruptOnSubscribe = interruptOnSubscribe; }
704 
705 		@Override public void subscribe(Flow.Subscriber<? super T> sub) {
706 			sub.onSubscribe(new Flow.Subscription() {
707 				@Override public void request(long n) { /* never emits */ }
708 				@Override public void cancel() { cancelled.set(true); }
709 			});
710 			if (interruptOnSubscribe)
711 				Thread.currentThread().interrupt();
712 		}
713 	}

776 		public String getValue() { throw new IllegalArgumentException("cannot serialize"); }
777 	}

779 	/** Emits the supplied first element, then one more only if the subscriber keeps requesting. */
780 	static final class TrackingPublisher<T> implements Flow.Publisher<T> {
781 		final AtomicBoolean cancelled = new AtomicBoolean();
782 		final AtomicInteger emitted = new AtomicInteger();
783 		private final T first;
784 		private final boolean failSecondRequest;
785 		TrackingPublisher(T first, boolean failSecondRequest) { this.first = first; this.failSecondRequest = failSecondRequest; }
786 
787 		@Override public void subscribe(Flow.Subscriber<? super T> sub) {
788 			sub.onSubscribe(new Flow.Subscription() {
789 				private int requests;
790 				@Override public void request(long n) {
791 					requests++;
792 					if (failSecondRequest && requests == 2)
793 						throw new IllegalArgumentException("bad request");
794 					if (requests == 1) {
795 						emitted.incrementAndGet();
796 						sub.onNext(first);
797 					}
798 				}
799 				@Override public void cancel() { cancelled.set(true); }
800 			});
801 		}
802 	}
