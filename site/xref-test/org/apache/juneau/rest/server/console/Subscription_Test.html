17  package org.apache.juneau.rest.server.console;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.apache.juneau.*;
22  import org.apache.juneau.marshall.marshaller.*;
23  import org.apache.juneau.rest.server.console.Subscription.*;
24  import org.junit.jupiter.api.*;
25  
26  class Subscription_Test extends TestBase {
27  
28  	@Test void a01_masterDetail_specExample() {
29  		var s = Subscription.to(Topics.selection("changes")).as("params")
30  			.map("changeId", "ids.0").whenEmpty(WhenEmpty.CLEAR).emptyText("Select a change to see its tasks.");
31  		assertEquals("{topic:'selection:changes',as:'params',map:{changeId:'ids.0'},whenEmpty:'clear',emptyText:'Select a change to see its tasks.'}",
32  			Json5.DEFAULT.write(s.toMap()));
33  	}
34  
35  	@Test void a02_minimal_omitsOptionalKeys() {
36  		assertEquals("{topic:'filter:changes',as:'filter'}",
37  			Json5.DEFAULT.write(Subscription.to(Topics.filter("changes")).as("filter").toMap()));
38  	}
39  
40  	@Test void a03_mapIsRepeatable_andKeepsOrder() {
41  		var s = Subscription.to("app.region-picked").as("filter").map("columns.region", "region").map("columns.zone", "zone.0");
42  		assertEquals("{topic:'app.region-picked',as:'filter',map:{'columns.region':'region','columns.zone':'zone.0'}}",
43  			Json5.DEFAULT.write(s.toMap()));
44  	}
45  
46  	@Test void a04_keep() {
47  		assertEquals("keep", Subscription.to("cmd:x").as("refresh").whenEmpty(WhenEmpty.KEEP).toMap().getString("whenEmpty"));
48  	}
49  
50  	@Test void b01_badTopic_isE40() {
51  		var e = assertThrows(IllegalArgumentException.class, () -> Subscription.to("selection:"));
52  		assertEquals("invalid topic 'selection:': expected family[:key] (see the topic syntax)", e.getMessage());
53  		// Subscriptions are concrete: no wildcard.
54  		e = assertThrows(IllegalArgumentException.class, () -> Subscription.to("ssc.alert:*"));
55  		assertEquals("invalid topic 'ssc.alert:*': expected family[:key] (see the topic syntax)", e.getMessage());
56  		assertThrows(IllegalArgumentException.class, () -> Subscription.to(null));
57  	}
58  
59  	@Test void b02_badRole() {
60  		var target2 = Subscription.to("cmd:x");
61  		var e = assertThrows(IllegalArgumentException.class, () -> target2.as("Refresh"));
62  		assertEquals("subscription to 'cmd:x': role 'Refresh' must match ^[a-z][a-z0-9-]{0,31}$", e.getMessage());
63  	}
64  
65  	@Test void b03_badMapPath() {
66  		var target3 = Subscription.to("cmd:x").as("params");
67  		var e = assertThrows(IllegalArgumentException.class, () -> target3.map("id", "ids..0"));
68  		assertEquals("subscription to 'cmd:x': map path 'ids..0' must be a dotted path such as 'ids.0'", e.getMessage());
69  		var target4 = Subscription.to("cmd:x").as("params");
70  		e = assertThrows(IllegalArgumentException.class, () -> target4.map("", "ids.0"));
71  		assertEquals("subscription to 'cmd:x': map target '' must be a dotted name", e.getMessage());
72  	}
73  
74  	@Test void b04_toMapWithoutRole() {
75  		var target5 = Subscription.to("cmd:x");
76  		var e = assertThrows(IllegalArgumentException.class, target5::toMap);
77  		assertEquals("subscription to 'cmd:x' needs as(role)", e.getMessage());
78  	}
79  }
