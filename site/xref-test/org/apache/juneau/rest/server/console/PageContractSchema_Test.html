58  	/** The card-level oneOf message cannot name the field, so prove the param is the cause: only its value differs from a valid page. */
59  	@ParameterizedTest
60  	@ValueSource(strings = {"blank", "badchars", "whitespace", "toolong"})
61  	void a01b_dayParamIsTheOnlyDifference(String kind) throws Exception {
62  		var bad = fixture("invalid-urlstate-day-param-" + kind + ".json");
63  		var fixed = bad.replaceAll("\"param\":\"[^\"]*\"", "\"param\":\"on\"");
64  		assertNotEquals(bad, fixed);
65  		assertFalse(PageContractSchema.get().validate(bad).isEmpty());
66  		assertEquals(List.of(), PageContractSchema.get().validate(fixed));
67  	}

69  	/** The builder refuses these options (TableSpec.urlState(String, Map)); the wire contract must refuse them too, and only the bad fragment differs from a valid page. */
70  	@ParameterizedTest
71  	@CsvSource(delimiter = '|', value = {
72  		"ids-max-zero|\"max\":0|\"max\":1",
73  		"ids-max-negative|\"max\":-1|\"max\":1",
74  		"ids-max-fraction|\"max\":1.5|\"max\":2",
75  		"ids-max-string|\"max\":\"50\"|\"max\":50",
76  		"ids-param-badchars|p&x=1|pid",
77  		"window-param-badchars|a b|ab",
78  		"window-key|\"middle\"|\"start\"",
79  		"custom|\"bogus\"|\"ids\""})
80  	void a01c_urlStateConfigOptionsAreRefused(String kind, String badFragment, String goodFragment) throws Exception {
81  		var bad = fixture("invalid-urlstate-config-" + kind + ".json");
82  		var fixed = bad.replace(badFragment, goodFragment);
83  		assertNotEquals(bad, fixed, kind);
84  		assertFalse(PageContractSchema.get().validate(bad).isEmpty(), kind);
85  		assertEquals(List.of(), PageContractSchema.get().validate(fixed), kind);
86  	}

177 	@Test void a03_r4_danglingTemplate() throws Exception {
178 		var c = fixture("valid-example.json");
179 		assertEquals(List.of(), PageContractSchema.get().validate(c, Set.of("header.banner", "jc-seg-1")));
180 		var errors = PageContractSchema.get().validate(c, Set.of("header.banner"));
181 		assertEquals(List.of("R-4: template 'jc-seg-1' is referenced but not present"), errors);
182 	}
