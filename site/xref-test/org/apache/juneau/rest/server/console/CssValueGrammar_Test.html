26  /**
27   * Tests for {@code CssValueGrammar}'s allowlist-grammar accept/reject sweep, tested through
28   * {@link Theme.Builder#token(String, String)}.
29   *
30   * <p>
31   * {@code token(...)} is no longer the <i>only</i> production call site for {@code CssValueGrammar}: with the
32   * {@code var(--jc-name)} Theme-layer reference feature, {@code Theme.Builder.build()} re-validates each resolved
33   * literal too (defense-in-depth). {@code var()} itself is never a grammar value shape &mdash; it is recognized and
34   * resolved one layer above the grammar &mdash; so the {@code var()} reject vectors below still fail here exactly as
35   * any other out-of-grammar value: a malformed {@code var(...)} (fallback, non-{@code --jc-} target, empty, or
36   * nested inside a gradient) never matches the anchored reference recognizer and falls straight through to grammar
37   * rejection at {@code token()} time.
38   *
39   * <p>
40   * The GREEN bypass-vector sweep below must beat every known bypass vector, not just a literal {@code url(}.
41   */
42  class CssValueGrammar_Test extends TestBase {
43  
44  	//-----------------------------------------------------------------------------------------------------------------
45  	// GREEN: the bypass-vector sweep - each MUST throw.
46  	//-----------------------------------------------------------------------------------------------------------------
47  
48  	@ParameterizedTest
49  	@ValueSource(strings = {
50  		"url(https://evil/x)",
51  		"URL(https://evil)",
52  		"URl(x)",
53  		"url (https://evil/x)",       // space
54  		"url\t(https://evil)",        // tab (C0)
55  		"url\n(https://evil)",        // newline (C0)
56  		"url/**/(https://evil)",      // comment
57  		"  url(x)",                   // leading whitespace
58  		"linear-gradient(url(evil))", // nested url() inside an otherwise-shaped gradient - grammar-layer reject
59  		// The "none"-interaction vectors below (folded in from the former a12, merged to close out S4144 - these
60  		// share a01's exact shape and target, so splitting them into a second method was pure duplication):
61  		"url(none)",                       // "none" as a url() argument must not smuggle the url() production through
62  		"URL(none.png)",
63  		"url\t(none)",                     // tab (C0) between url and paren
64  		"none url(evil)",                  // "none" prefix must not whitelist a trailing url()
65  		"linear-gradient(none, url(evil))",
66  	})
67  	void a01_bypassVectorSweep_throws(String payload) {
68  		var b = Theme.create("x");
69  		assertThrows(IllegalArgumentException.class, () -> b.token("--jc-page-bg", payload));
70  	}

72  	@Test void a02_cssHexEscapedUrl_throws() {
73  		// CSS-hex-escaped letters spelling "url(" - the grammar has no production admitting a literal backslash,
74  		// so these die in the allowlist grammar (not the normalization url-regex, which does not match "\75rl(").
75  		var b = Theme.create("x");
76  		assertThrows(IllegalArgumentException.class, () -> b.token("--jc-page-bg", "\\75rl(x)"));
77  		assertThrows(IllegalArgumentException.class, () -> b.token("--jc-page-bg", "url\\3C(x)"));
78  	}

81  		var b = Theme.create("x");
82  		assertThrows(IllegalArgumentException.class, () -> b.token("--jc-accent", "#1589EE\u0000"));
83  		assertThrows(IllegalArgumentException.class, () -> b.token("--jc-accent", "\u0007red"));
84  	}

89  		assertThrows(IllegalArgumentException.class, () -> b.token("--jc-accent", "red\u0085"));

92  	//-----------------------------------------------------------------------------------------------------------------
93  	// Prove the C0/newline cases throw at NORMALIZATION (before the grammar), not merely because the belt
94  	// regex (?i)url\s*\( also happens to match whitespace. A payload the grammar would ACCEPT if the C0 reject
95  	// were removed, but that still must throw because of the control character alone.
96  	//-----------------------------------------------------------------------------------------------------------------
97  
98  	@Test void a05_controlCharacterInOtherwiseValidGrammar_throwsAtNormalization_notGrammar() {
99  		// rgb(...)'s functional-color charset includes whitespace; String.trim() does not strip an INTERIOR
100 		// newline. This value contains no "url(" anywhere, so if the C0/C1 normalization reject were deleted,
101 		// the (?i)url\s*\( belt would NOT catch it either, and the grammar would ACCEPT it (whitespace is in the
102 		// rgb() charset). It must still throw - and for a DIFFERENT reason (control character) than an
103 		// out-of-grammar value.
104 		var b = Theme.create("x");
105 		var ex = assertThrows(IllegalArgumentException.class,
106 			() -> b.token("--jc-accent", "rgb(21,\n137,238)"));
107 		assertTrue(ex.getMessage().contains("control character"), () -> "unexpected message: " + ex.getMessage());
108 	}
109 
110 	@Test void a06_outOfGrammarValue_throwsWithDistinctMessage_notControlCharacter() {
111 		var b = Theme.create("x");
112 		var ex = assertThrows(IllegalArgumentException.class,
113 			() -> b.token("--jc-accent", "not-a-css-value!!!"));
114 		assertFalse(ex.getMessage().contains("control character"), () -> "unexpected message: " + ex.getMessage());
115 	}

154 	// (background-image: none), but admitting it must NOT reopen the url() ban - "none" is an orthogonal branch of

160 		var theme = Theme.create("x").token("--jc-page-bg", "none").build();

162 	}

169 	//-----------------------------------------------------------------------------------------------------------------
170 	// GREEN: malformed var(...) forms are NOT the anchored var(--jc-name) reference, so they never reach the
171 	// reference recognizer and REJECT at the grammar layer at token() time, exactly like any other non-shape value.
172 	// A well-formed var(--jc-name) reference is deliberately absent here - it is ACCEPTED (deferred) at token() and
173 	// is exercised by Theme_VarReferences_Test instead; it must NOT be added to a07's positive (verbatim) sweep.
174 	//-----------------------------------------------------------------------------------------------------------------
175 
176 	@ParameterizedTest
177 	@ValueSource(strings = {
178 		"var(--jc-x, #fff)",                    // fallback branch - comma inside parens, never matches the recognizer
179 		"VAR(--jc-x, red)",                     // case-insensitive fallback - still a fallback, still rejected
180 		"var(--not-jc-x)",                      // target is not a --jc- name
181 		"var()",                                // no argument
182 		"var(--jc-x) #fff",                     // reference is not the ENTIRE value (recognizer is fully anchored)
183 		"linear-gradient(var(--jc-a), #fff)",   // nested var() - grammar-layer reject (var not in the nested allowlist)
184 	})
185 	void a12_malformedOrNestedVar_rejectedAtGrammarLayer(String payload) {
186 		var b = Theme.create("x");
187 		assertThrows(IllegalArgumentException.class, () -> b.token("--jc-accent", payload));
188 	}
