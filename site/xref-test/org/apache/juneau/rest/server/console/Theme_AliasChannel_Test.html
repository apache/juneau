28  /**
29   * The alias channel on {@link Theme.Builder}: alias-name shape, the anchored {@code var(--jc-name)} target
30   * recognizer, the one-channel-per-name rule, and {@link Theme#getAliases()}'s ordering and immutability.
31   */
32  class Theme_AliasChannel_Test extends TestBase {
33  
34  	private static Theme.Builder builder() {
35  		return Theme.create("corporate").token("--jc-accent", "#b45309");
36  	}
37  
38  	private static String aliasTargetOf(String target) {
39  		return builder().alias("--jc-tab-bar-bg", target).build().getAliases().get("--jc-tab-bar-bg");
40  	}

43  	// a) Alias name shape
44  	//-----------------------------------------------------------------------------------------------------------------

46  	@ParameterizedTest
47  	@NullSource
48  	@ValueSource(strings = {
49  		"",                // empty
50  		"--jc-",           // no name part
51  		"jc-accent",       // missing the -- prefix
52  		"--JC-accent",     // uppercase prefix
53  		"--jc-Accent",     // uppercase name
54  		"--jc-foo;--bar",  // full-string guard: "--jc-foo" matches only as a LEADING substring
55  		"--jc-foo bar",    // whitespace
56  		"--other-accent",  // outside the --jc- namespace
57  	})
58  	void a01_aliasName_invalidShapes_rejected(String name) {
59  		var b = builder();
60  		assertThrows(IllegalArgumentException.class, () -> b.alias(name, "var(--jc-accent)"));
61  	}

63  	/** The guard fires on the DECLARED name only: aliasing a theme token <i>to</i> a reserved ladder step is legal. */
64  	@Test void a02_aliasingToAReservedChromeStep_isLegal() {
65  		var t = builder().alias("--jc-tab-bar-height", "var(--jc-chrome-control-height)").build();
66  		assertEquals("var(--jc-chrome-control-height)", t.getAliases().get("--jc-tab-bar-height"));
67  	}

73  	@Test void b01_nameDeclaredByBothChannels_rejectedAtBuild_inEitherDeclarationOrder() {
74  		var aliasFirst = Theme.create("both").alias("--jc-accent", "var(--jc-link)").token("--jc-accent", "#b45309");
75  		var e1 = assertThrows(IllegalArgumentException.class, aliasFirst::build);
76  		assertTrue(e1.getMessage().contains("declares '--jc-accent' as both"), e1::getMessage);
77  		var tokenFirst = Theme.create("both").token("--jc-accent", "#b45309").alias("--jc-accent", "var(--jc-link)");
78  		var e2 = assertThrows(IllegalArgumentException.class, tokenFirst::build);
79  		assertTrue(e2.getMessage().contains("declares '--jc-accent' as both"), e2::getMessage);
80  	}

85  
86  	/**
87  	 * Insertion order is what makes the served block byte-stable. The names are deliberately neither alphabetical
88  	 * nor reverse-alphabetical, so a sorted or hash-ordered map fails rather than accidentally passing.
89  	 */
90  	@Test void c01_getAliases_isInsertionOrdered_andImmutable() {
91  		var t = builder()
92  			.alias("--jc-z-last", "var(--jc-accent)")
93  			.alias("--jc-a-first", "var(--jc-link)")
94  			.alias("--jc-m-middle", "var(--jc-text)")
95  			.build();
96  		assertEquals(List.of("--jc-z-last", "--jc-a-first", "--jc-m-middle"), new ArrayList<>(t.getAliases().keySet()));
97  		var aliases = t.getAliases();
98  		assertThrows(UnsupportedOperationException.class, () -> aliases.put("--jc-x", "var(--jc-accent)"));
99  	}

107 	@Test void c03_alias_doesNotLeakIntoTheLeafMap() {

117 	@Test void d01_aliasTargets_acceptedShapes_storeTheNormalizedReference() {
118 		assertEquals("var(--jc-accent)", aliasTargetOf("var(--jc-accent)"));
119 		assertEquals("VAR(--jc-accent)", aliasTargetOf("VAR(--jc-accent)"), "the recognizer is case-insensitive on the function name");
120 		assertEquals("var( --jc-accent )", aliasTargetOf("var( --jc-accent )"), "internal whitespace inside the parens is legal");
121 		assertEquals("var(--jc-accent)", aliasTargetOf("  var(--jc-accent)  "), "the shared normalization belt trims");
122 		assertEquals("var(--jc-accent)", aliasTargetOf("var/**/(--jc-accent)"), "the shared normalization belt strips comments");
123 	}

125 	@ParameterizedTest
126 	@NullSource
127 	@ValueSource(strings = {
128 		"",                                    // not a reference
129 		"var(--jc-x, #fff)",                   // the fallback form must FAIL, never silently fall back
130 		"var(--jc-a) var(--jc-b)",             // two references
131 		"linear-gradient(var(--jc-a), #fff)",  // a reference embedded in a larger value
132 		"#ffffff",                             // a literal belongs in token(), not here
133 		"1.2rem",
134 		"red",
135 		"url(https://evil)",
136 		"url (https://evil)",
137 		"url/**/(https://evil)",
138 		"var(--jc-a);color:red",               // declaration breakout
139 		"var(--jc-a)}html{color:red",          // block breakout
140 		"var(--jc-a)/*",                       // unterminated comment survives the strip and must not match
141 		"var(--jc-A)",                         // uppercase in the target name
142 		"var(--other-a)",                      // outside the --jc- namespace
143 		"var(--jc-a",                          // unbalanced
144 		"ｖar(--jc-a)",                    // fullwidth 'v' homoglyph
145 		"var‎(--jc-a)",                   // a bidi/format control between the name and the paren
146 	})
147 	void d02_aliasTargets_rejectedShapes(String target) {
148 		var b = builder();
149 		assertThrows(IllegalArgumentException.class, () -> b.alias("--jc-tab-bar-bg", target));
150 	}

152 	/**
153 	 * Control characters are rejected by the shared belt on the RAW value, before any trim - which is what kills
154 	 * the {@code url\t(} / {@code url\n(} CSS-hex reconstruction vector for this channel too.
155 	 */
156 	@Test void d03_aliasTargets_withControlCharacters_rejected() {
157 		var b = builder();
158 		assertThrows(IllegalArgumentException.class, () -> b.alias("--jc-tab-bar-bg", "var(--jc-a\t)"));
159 		assertThrows(IllegalArgumentException.class, () -> b.alias("--jc-tab-bar-bg", "var(--jc-a)\n"));
160 		assertThrows(IllegalArgumentException.class, () -> b.alias("--jc-tab-bar-bg", "\u0000var(--jc-a)"));
161 		assertThrows(IllegalArgumentException.class, () -> b.alias("--jc-tab-bar-bg", "var(--jc-a)\u007F"));
162 	}

164 	/**
165 	 * The shared-recognizer pin: for every candidate, alias() and token()'s reference resolution agree on whether
166 	 * it is a {@code var()} reference. A duplicated regex fixed in one copy only would be a silent,
167 	 * security-relevant divergence. The token side is observed through resolution: a recognized reference is
168 	 * resolved to {@link Theme#OPEN}'s {@code --jc-accent} literal by {@code build()}.
169 	 */
170 	@ParameterizedTest
171 	@ValueSource(strings = {
172 		"var(--jc-accent)",                 // accepted by both
173 		"VAR(--jc-accent)",
174 		"var( --jc-accent )",
175 		"var/**/(--jc-accent)",
176 		"var(--jc-accent, #fff)",           // rejected by both
177 		"var(--jc-accent) var(--jc-link)",
178 		"#ffffff",
179 		"1.2rem",
180 		"red",
181 	})
182 	void d04_aliasAndTokenChannels_agreeOnWhatAReferenceIs(String value) {
183 		assertEquals(tokenTreatsAsReference(value), aliasAccepts(value),
184 			() -> "the two channels disagree about '" + value + "' - the recognizer has been duplicated and has drifted");
185 	}
186 
187 	private static boolean aliasAccepts(String value) {
188 		try {
189 			Theme.create("probe").alias("--jc-tab-bar-bg", value);
190 			return true;
191 		} catch (IllegalArgumentException e) {
192 			return false;
193 		}
194 	}
195 
196 	private static boolean tokenTreatsAsReference(String value) {
197 		try {
198 			return Theme.OPEN.getTokens().get("--jc-accent")
199 				.equals(Theme.create("probe").token("--jc-tab-bar-bg", value).build().getTokens().get("--jc-tab-bar-bg"));
200 		} catch (IllegalArgumentException e) {
201 			return false;
202 		}
203 	}
