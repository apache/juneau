28  /**
29   * Fails when {@code juneau-page.schema.json} uses a JSON Schema keyword outside the subset that the in-house
30   * {@link SchemaSubsetValidator} implements (C1-D2).
31   */
32  class PageContractSchema_KeywordSubset_Test extends TestBase {
33  
34  	@Test void a01_schemaUsesOnlySupportedKeywords() throws Exception {
35  		var schema = JsonMap.ofString(PageContractSchema.get().schemaJson());
36  		var unsupported = new TreeSet<String>();
37  		walk(schema, unsupported, false);
38  		assertTrue(unsupported.isEmpty(), () -> "unsupported keywords: " + unsupported);
39  	}
40  
41  	@Test void a02_subsetIsTheDocumentedOne() {
42  		assertEquals(
43  			SchemaSubsetValidator.KEYWORDS,
44  			Set.of("$schema", "$id", "$defs", "$ref", "title", "description", "default", "type", "const", "enum",
45  				"required", "properties", "additionalProperties", "propertyNames", "items", "pattern", "minLength",
46  				"maxLength", "minItems", "minimum", "allOf", "anyOf", "oneOf", "not"));
47  	}
48  
49  	// Walks schema objects. Inside "properties" and "$defs" the keys are names, not keywords.
50  	private static void walk(Object node, Set<String> unsupported, boolean namesLevel) {
51  		if (node instanceof Map<?,?> m) {
52  			for (var e : m.entrySet()) {
53  				var k = (String)e.getKey();
54  				if (! namesLevel && ! SchemaSubsetValidator.KEYWORDS.contains(k))
55  					unsupported.add(k);
56  				var childIsNames = ! namesLevel && (eqa(k, "properties", "$defs"));
57  				walk(e.getValue(), unsupported, childIsNames);
58  			}
59  		} else if (node instanceof List<?> l) {
60  			for (var x : l)
61  				walk(x, unsupported, false);
62  		}
63  	}
64  }
