17  package org.apache.juneau.rest.server.console;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.junit.jupiter.api.*;
22  
23  class CardSource_Test {
24  
25  	@Test void hasJsonBody_sniffsFirstNonWhitespaceChar() {
26  		var jsonSrc = CardSource.create("kpi", "k").body("  \n {value:3}").build();
27  		assertTrue(jsonSrc.hasJsonBody());
28  		var markupSrc = CardSource.create("html", "h").body("<p>hi</p>").build();
29  		assertFalse(markupSrc.hasJsonBody());
30  		var emptySrc = CardSource.create("html", "h").build();
31  		assertFalse(emptySrc.hasJsonBody());
32  	}
33  
34  	@Test void json_parsesJson5AndCaches() {
35  		var src = CardSource.create("kpi", "k").body("{value:3, label:'N'}").build();
36  		var m1 = src.json();
37  		var m2 = src.json();
38  		assertSame(m1, m2); // cached, not re-parsed
39  		assertEquals(3, m1.get("value"));
40  		assertEquals("N", m1.get("label"));
41  	}
42  
43  	@Test void json_onMalformedJson5_throwsE24() {
44  		var src = CardSource.create("kpi", "k").body("{value: [}").build();
45  		var ex = assertThrows(IllegalArgumentException.class, src::json);
46  		// E-24 keeps the C1 'Card JSON5 is invalid: ' substring pinned by PetstoreFailLoud_Test a04 (Auto-decision 8).
47  		assertTrue(ex.getMessage().startsWith("<@card id='k'> Card JSON5 is invalid: "), ex.getMessage());
48  	}
49  
50  	@Test void json_onMarkupBody_throwsE25Shaped() {
51  		var src = CardSource.create("kpi", "k").body("<p>not json</p>").build();
52  		var ex = assertThrows(IllegalArgumentException.class, src::json);
53  		assertEquals("<@card id='k'> type='kpi' requires a JSON5 object body; got '<p>not json</p>'.", ex.getMessage());
54  	}
55  
56  	@Test void bodyMap_isTreatedAsAlreadyParsedJson() {
57  		var src = CardSource.create("kpi", "k").bodyMap(java.util.Map.of("value", 3)).build();
58  		assertTrue(src.hasJsonBody());
59  		assertEquals(3, src.json().get("value"));
60  	}
61  
62  	@Test void captureTemplate_callsSinkAndReturnsId() {
63  		var captured = new String[2];
64  		var src = CardSource.create("html", "h")
65  			.body("<p>hi</p>")
66  			.templateSink((id, markup) -> { captured[0] = id; captured[1] = markup; })
67  			.build();
68  		assertEquals("h", src.captureTemplate());
69  		assertEquals("h", captured[0]);
70  		assertEquals("<p>hi</p>", captured[1]);
71  	}
72  
73  	@Test void captureTemplate_withNoSink_throws() {
74  		var src = CardSource.create("html", "h").body("<p>hi</p>").build();
75  		assertThrows(IllegalArgumentException.class, src::captureTemplate);
76  	}
77  
78  	@Test void error_prefixesCardId() {
79  		var src = CardSource.create("kpi", "k").build();
80  		var ex = src.error("needs '%s'.", "value");
81  		assertEquals("<@card id='k'> needs 'value'.", ex.getMessage());
82  	}
83  
84  	@Test void create_rejectsBlankTypeOrId() {
85  		assertThrows(IllegalArgumentException.class, () -> CardSource.create("", "k"));
86  		assertThrows(IllegalArgumentException.class, () -> CardSource.create("kpi", " "));
87  	}
88  }
