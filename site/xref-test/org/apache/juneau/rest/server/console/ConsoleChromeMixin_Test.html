669 	/** WCAG 2.x contrast ratio between two {@code "#rrggbb"} literals: {@code (lighter+0.05)/(darker+0.05)}. */
670 	private static double contrastRatio(String hex1, String hex2) {
671 		var l1 = relativeLuminance(hex1);
672 		var l2 = relativeLuminance(hex2);
673 		return (Math.max(l1, l2) + 0.05) / (Math.min(l1, l2) + 0.05);
674 	}
675 
676 	/** WCAG 2.x relative luminance of a {@code "#rrggbb"} literal. */
677 	private static double relativeLuminance(String hex) {
678 		var rgb = Integer.parseInt(hex.substring(1), 16);
679 		var r = srgbChannelToLinear((rgb >> 16) & 0xFF);
680 		var g = srgbChannelToLinear((rgb >> 8) & 0xFF);
681 		var b = srgbChannelToLinear(rgb & 0xFF);
682 		return 0.2126 * r + 0.7152 * g + 0.0722 * b;
683 	}
684 
685 	private static double srgbChannelToLinear(int value8Bit) {
686 		var c = value8Bit / 255.0;
687 		return c <= 0.03928 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4);
688 	}
