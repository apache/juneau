28  /**
29   * {@link ConsoleChromeMixin#overrideBlock(Theme)}: the single emitter for a theme's override block. Leaves are
30   * escaped and emitted first in declaration order, then aliases verbatim. An empty theme emits nothing.
31   */
32  class ConsoleChromeMixin_OverrideBlock_Test extends TestBase {
33  
34  	private static final Theme EMISSION = Theme.create("corporate")
35  		.token("--jc-accent", "#b45309")
36  		.token("--jc-link", "#0b6bcb")
37  		.alias("--jc-tab-bar-bg", "var(--jc-card-bg)")
38  		.alias("--jc-tab-selected-bg", "var(--jc-white)")
39  		.build();
40  
41  	private static List<String> declaredNames(String block) {
42  		var names = new ArrayList<String>();
43  		var m = Pattern.compile("(--[a-zA-Z0-9-]++)\\s*:").matcher(block);
44  		while (m.find())
45  			names.add(m.group(1));
46  		return names;
47  	}
48  
49  	//-----------------------------------------------------------------------------------------------------------------
50  	// a) Shape and order
51  	//-----------------------------------------------------------------------------------------------------------------
52  
53  	@Test void a01_block_isOneHtmlRootBlock() {
54  		var block = ConsoleChromeMixin.overrideBlock(EMISSION);
55  		assertTrue(block.startsWith("html:root{"), block);
56  		assertTrue(block.endsWith("}"), block);
57  		assertEquals(1, block.split("html:root\\{", -1).length - 1, block);
58  	}
59  
60  	@Test void a02_block_carriesLeavesThenAliases_inDeclarationOrder() {
61  		assertList(declaredNames(ConsoleChromeMixin.overrideBlock(EMISSION)),
62  			"--jc-accent", "--jc-link", "--jc-tab-bar-bg", "--jc-tab-selected-bg");
63  	}
64  
65  	/**
66  	 * The two var() paths must never be unified: a var() given to token() is resolved before the wire (a05); an
67  	 * alias survives as a reference. If either test has to change to make the other pass, the change is wrong.
68  	 */
69  	@Test void a03_alias_survivesAsAReference() {
70  		var block = ConsoleChromeMixin.overrideBlock(EMISSION);
71  		assertContainsAll(block, "--jc-tab-bar-bg:var(--jc-card-bg);", "--jc-tab-selected-bg:var(--jc-white);");
72  	}
73  
74  	/** Every declaration is a --jc- name; the digit-admitting class makes sure names like --evil-1 are inspected. */
75  	@Test void a04_block_declaresOnlyJcNames() {
76  		var names = declaredNames(ConsoleChromeMixin.overrideBlock(EMISSION));
77  		assertEquals(4, names.size(), names::toString);
78  		for (var n : names)
79  			assertTrue(n.startsWith("--jc-"), n);
80  	}
81  
82  	@Test void a05_leafVarReference_isResolvedToItsLiteral() {
83  		var t = Theme.create("derived").token("--jc-danger", "#c23934").token("--jc-tag-red-text", "var(--jc-danger)").build();
84  		var block = ConsoleChromeMixin.overrideBlock(t);
85  		assertTrue(block.contains("--jc-tag-red-text:#c23934;"), block);
86  		assertFalse(block.contains("--jc-tag-red-text:var("), block);
87  	}
88  
89  	//-----------------------------------------------------------------------------------------------------------------
90  	// b) Escaping
91  	//-----------------------------------------------------------------------------------------------------------------
92  
93  	/**
94  	 * The escaper-wiring gate. {@code 'My;Font'} is grammar-accepted, but a raw {@code ;} would end the
95  	 * declaration early. This is the one value shape where the escaper is not a no-op.
96  	 */
97  	@Test void b01_leafWithSemicolon_isEscaped() {
98  		var block = ConsoleChromeMixin.overrideBlock(Theme.create("semi").token("--jc-font", "'My;Font', sans-serif").build());
99  		assertFalse(block.contains("'My;Font'"), block);
100 		assertTrue(block.contains("\\3B "), block);
101 	}
102 
103 	@Test void b02_valuePositives_surviveByteForByte() {
104 		var block = ConsoleChromeMixin.overrideBlock(Theme.create("positives")
105 			.token("--jc-accent", "#e91e63")
106 			.token("--jc-page-bg", "linear-gradient(180deg, #aabbcc 0%, #112233 100%)")
107 			.token("--jc-font", "'Helvetica Neue', Inter, sans-serif")
108 			.build());
109 		assertContainsAll(block, "--jc-accent:#e91e63;", "--jc-page-bg:linear-gradient(180deg, #aabbcc 0%, #112233 100%);",
110 			"--jc-font:'Helvetica Neue', Inter, sans-serif;");
111 	}
112 
113 	@Test void b03_block_passesChromeCssScanner() {
114 		var block = ConsoleChromeMixin.overrideBlock(EMISSION);
115 		assertEquals(List.of(), ChromeCssScanner.scan(block), block);
116 	}
117 
118 	@Test void b04_redTriad_isThemeable() {
119 		var block = ConsoleChromeMixin.overrideBlock(Theme.create("red-override")
120 			.token("--jc-tag-red-bg", "#ffe0e2")
121 			.token("--jc-tag-red-text", "#5a0f14")
122 			.token("--jc-tag-red-border", "#f0b3b7")
123 			.build());
124 		assertContainsAll(block, "--jc-tag-red-bg:#ffe0e2;", "--jc-tag-red-text:#5a0f14;", "--jc-tag-red-border:#f0b3b7;");
125 	}
