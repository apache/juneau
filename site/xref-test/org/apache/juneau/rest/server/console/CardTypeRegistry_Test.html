117 		// The registry never sees attributes: a base key in the body is E-23 here; subscribes/publishes duplicates are
118 		// rejected by <@card> after promotion (see CardDirective_Bus_Test).
119 		var r = CardTypeRegistry.standard();
120 		var src = CardSource.create("kpi", "k").title("T").body("{title:'x', subscribes:'a'}").build();
121 		var ex = assertThrows(IllegalArgumentException.class, () -> r.toCard(src));
122 		assertEquals("<@card id='k'> body key 'title' is reserved; set it as an attribute.", ex.getMessage());

168 	/** Minimal handler whose fragment never carries `visibleWhen` itself — exercises toCard's promotion. */
169 	private static final class TableLikeCardType implements CardTypeHandler {
170 		@Override public String type() { return "tablelike"; }
171 		@Override public JsonMap toFragment(CardSource source) {
172 			var body = source.json();
173 			var table = new JsonMap();
174 			table.putAll(body);
175 			table.remove("visibleWhen");
176 			var frag = new JsonMap();
177 			frag.put("table", table);
178 			return frag;
179 		}
180 	}

182 	private static final class TestCardType implements CardTypeHandler {
183 		private final String type;
184 		TestCardType(String type) { this.type = type; }
185 		@Override public String type() { return type; }
186 		@Override public JsonMap toFragment(CardSource source) { return new JsonMap(); }
187 	}
