17  package org.apache.juneau.rest.server.console;
18  
19  import static org.apache.juneau.test.bct.BctAssertions.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import org.apache.juneau.*;
23  import org.apache.juneau.marshall.marshaller.*;
24  import org.apache.juneau.rest.server.console.TopicDecl.*;
25  import org.junit.jupiter.api.*;
26  
27  class TopicDecl_Test extends TestBase {
28  
29  	@Test void a01_pageLevel() {
30  		assertEquals("{topic:'app.region-picked',retain:true,publisher:'script'}",
31  			Json5.DEFAULT.write(TopicDecl.of("app.region-picked").retain(true).publisher(Publisher.SCRIPT).toMap()));
32  		assertEquals("{topic:'ssc.alert:*',retain:false,publisher:'server'}",
33  			Json5.DEFAULT.write(TopicDecl.of("ssc.alert:*").retain(false).publisher(Publisher.SERVER).toMap()));
34  		assertEquals("ribbon", TopicDecl.of("app.go-live").retain(false).publisher(Publisher.RIBBON).toMap().getString("publisher"));
35  	}
36  
37  	@Test void a02_jobWildcard_onlyWithServer() {
38  		assertEquals("{topic:'job:*',retain:true,publisher:'server'}",
39  			Json5.DEFAULT.write(TopicDecl.of("job:*").retain(true).publisher(Publisher.SERVER).toMap()));
40  		var target = TopicDecl.of("job:*").retain(true).publisher(Publisher.SCRIPT);
41  		var e = assertThrows(IllegalArgumentException.class, target::toMap);
42  		assertEquals("'job' is a framework family; framework topics are implicit and may not be declared", e.getMessage());
43  	}
44  
45  	@Test void a03_publication() {
46  		assertEquals("{topic:'ssc.focus',retain:true}", Json5.DEFAULT.write(TopicDecl.of("ssc.focus").retain(true).toPublicationMap()));
47  		var target2 = TopicDecl.of("ssc.focus").retain(true).publisher(Publisher.SCRIPT);
48  		var e = assertThrows(IllegalArgumentException.class,
49  			target2::toPublicationMap);
50  		assertEquals("topic 'ssc.focus': publisher is page-level only; a card's publishes may not set it", e.getMessage());
51  		var target3 = TopicDecl.of("job:*").retain(true);
52  		e = assertThrows(IllegalArgumentException.class,
53  			target3::toPublicationMap);
54  		assertEquals("'job' is a framework family; framework topics are implicit and may not be declared", e.getMessage());
55  	}
56  
57  	@Test void b01_frameworkFamily_isE45() {
58  		var e = assertThrows(IllegalArgumentException.class, () -> TopicDecl.of("selection:changes"));
59  		assertEquals("'selection' is a framework family; framework topics are implicit and may not be declared", e.getMessage());
60  		e = assertThrows(IllegalArgumentException.class, () -> TopicDecl.of("job:j-91"));
61  		assertEquals("'job' is a framework family; framework topics are implicit and may not be declared", e.getMessage());
62  	}
63  
64  	@Test void b02_unnamespaced_isE45() {
65  		var e = assertThrows(IllegalArgumentException.class, () -> TopicDecl.of("focus"));
66  		assertEquals("custom topic family 'focus' must be namespaced (e.g. 'app.focus')", e.getMessage());
67  	}
68  
69  	@Test void b03_badSyntax_isE40() {
70  		var e = assertThrows(IllegalArgumentException.class, () -> TopicDecl.of("App.Focus"));
71  		assertEquals("invalid topic 'App.Focus': expected family[:key] (see the topic syntax)", e.getMessage());
72  		assertThrows(IllegalArgumentException.class, () -> TopicDecl.of(null));
73  	}
74  
75  	@Test void b04_retainRequired() {
76  		var target4 = TopicDecl.of("ssc.focus").publisher(Publisher.SCRIPT);
77  		var e = assertThrows(IllegalArgumentException.class, target4::toMap);
78  		assertEquals("topic 'ssc.focus' is declared with retain=unset here; call retain(true) or retain(false)", e.getMessage());
79  		var target5 = TopicDecl.of("ssc.focus");
80  		assertThrows(IllegalArgumentException.class, target5::toPublicationMap);
81  	}
82  
83  	@Test void b05_pageLevelNeedsPublisher() {
84  		var target6 = TopicDecl.of("ssc.focus").retain(true);
85  		var e = assertThrows(IllegalArgumentException.class, target6::toMap);
86  		assertEquals("topic 'ssc.focus': a page-level topic needs publisher(SCRIPT|SERVER|RIBBON)", e.getMessage());
87  	}
88  
89  	@Test void d01_data_pageScriptRetained() {
90  		assertBean(TopicDecl.of("data:x").retain(true).publisher(TopicDecl.Publisher.SCRIPT).toMap(), "topic,retain,publisher", "data:x,true,script");
91  	}
92  
93  	@Test void d02_data_rejectsOtherShapes() {
94  		var target7 = TopicDecl.of("data:x").retain(false).publisher(TopicDecl.Publisher.SCRIPT);
95  		assertThrows(IllegalArgumentException.class, target7::toMap);
96  		var target8 = TopicDecl.of("data:x").retain(true).publisher(TopicDecl.Publisher.SERVER);
97  		assertThrows(IllegalArgumentException.class, target8::toMap);
98  		var target9 = TopicDecl.of("data:x").retain(true);
99  		assertThrows(IllegalArgumentException.class, target9::toPublicationMap);
100 		assertThrows(IllegalArgumentException.class, () -> TopicDecl.of("data:*"));
101 	}
102 
103 	@Test void d03_zoomLegend_cardPublicationRetained() {
104 		assertBean(TopicDecl.of("zoom:t").retain(true).toPublicationMap(), "topic,retain", "zoom:t,true");
105 		assertBean(TopicDecl.of("legend:t").retain(true).toPublicationMap(), "topic,retain", "legend:t,true");
106 		var target10 = TopicDecl.of("zoom:t").retain(false);
107 		assertThrows(IllegalArgumentException.class, target10::toPublicationMap);
108 		var target11 = TopicDecl.of("zoom:t").retain(true).publisher(TopicDecl.Publisher.SCRIPT);
109 		assertThrows(IllegalArgumentException.class, target11::toMap);
110 	}
111 
112 	@Test void d04_otherFramework_stillE45() {
113 		var e = assertThrows(IllegalArgumentException.class, () -> TopicDecl.of("filter:t"));
114 		assertContains("'filter' is a framework family", e.getMessage());
115 	}
