28  /**
29   * Pins the inner-main well, card fill-width, and outer table-chrome contracts in {@code chrome.css}.
30   */
31  class ChromeCss_MainInner_Test extends TestBase {
32  
33  	private static String readChromeCss() throws IOException {
34  		try (var in = ChromeCss_MainInner_Test.class.getResourceAsStream("/org/apache/juneau/console/chrome.css")) {
35  			assertNotNull(in, "missing classpath resource: /org/apache/juneau/console/chrome.css");
36  			return new String(in.readAllBytes(), UTF_8);
37  		}
38  	}
39  
40  	private static String flat() throws IOException {
41  		return readChromeCss().replaceAll("\\s+", " ");
42  	}
43  
44  	@Test void a01_jcMain_paintsMainBg_gutter_radius_noBorder() throws Exception {
45  		var c = flat();
46  		assertTrue(c.contains(".jc-main {"), ".jc-main rule missing");
47  		var start = c.indexOf(".jc-main {");
48  		var end = c.indexOf('}', start);
49  		var body = c.substring(start, end);
50  		assertContainsAll(body, "background-color: var(--jc-main-bg)", "margin: var(--jc-space-3) var(--jc-space-3) 0",
51  			"border-radius: var(--jc-radius)", "padding: 20px 24px 40px", "overflow: auto", "min-height: 0");
52  		assertFalse(body.contains("margin: var(--jc-space-3);"),
53  			"four-sided margin on .jc-main clips at scroll end: " + body);
54  		assertFalse(body.contains("max-width:"), "Q4: no max-width on .jc-main: " + body);
55  		assertFalse(body.contains("border:"), "Q4: no extra border on .jc-main: " + body);
56  	}
57  
58  	@Test void a01b_body_keepsBottomGutterAfterScroll() throws Exception {
59  		var c = flat();
60  		assertContainsAll(c, "padding-bottom: var(--jc-space-3)", "min-height: 100vh", "overflow: hidden", "display: flex", "flex-direction: column");
61  	}
62  
63  	@Test void a01c_jcFooter_modestMutedCentered() throws Exception {
64  		var c = flat();
65  		assertTrue(c.contains(".jc-page-footer"), "public footer class is .jc-page-footer: " + c);
66  		assertTrue(c.contains(".jc-footer") && c.contains("body::after"),
67  			"footer HTML class and mixin body::after share one rule: " + c);
68  		var sel = ".jc-page-footer, .jc-footer, body::after {";
69  		var start = c.indexOf(sel);
70  		assertTrue(start >= 0, "expected grouped footer rule, css:\n" + c);
71  		var end = c.indexOf('}', start);
72  		var body = c.substring(start, end);
73  		assertTrue(body.contains("color: var(--jc-text-muted)"), body);
74  		assertTrue(body.contains("padding: var(--jc-space-3) var(--jc-space-3) 0"), body);
75  		assertTrue(body.contains("text-align: center"), body);
76  		assertFalse(body.contains("padding: 24px 0 32px"), "footer padding must stay tighter than a typical app strip: " + body);
77  	}
78  
79  	@Test void a04_pageHeader_fillsWellAsTitleCard() throws Exception {
80  		var c = flat();
81  		assertTrue(c.contains(".jc-page-header {"), ".jc-page-header rule missing");
82  		var start = c.indexOf(".jc-page-header {");
83  		var end = c.indexOf('}', start);
84  		var body = c.substring(start, end);
85  		assertTrue(body.contains("width: 100%"), "title card must fill the well: " + body);
86  		assertFalse(body.contains("margin: 0 auto"), "no leftover 1180px-era centering: " + body);
87  		assertTrue(body.contains("background-color: var(--jc-card-bg)"), body);
88  		assertTrue(body.contains("box-shadow: var(--jc-card-shadow)"),
89  			"title card spends --jc-card-shadow: " + body);
90  		assertTrue(body.contains("padding: var(--jc-card-padding)"),
91  			"title card spends --jc-card-padding: " + body);
92  		assertTrue(body.contains("padding-top: var(--jc-space-2)"),
93  			"title card top inset is --jc-space-2 (8px), not content-card 16px: " + body);
94  		assertFalse(body.contains("padding: 18px 20px"),
95  			"off-scale 18px 20px title-card inset must not remain: " + body);
96  		assertTrue(body.contains("gap: 0"), "title/sub stack with h1 margin, not flex gap: " + body);
97  		assertFalse(body.contains("gap: 10px"), "10px flex gap between title and description: " + body);
98  	}

100 	@Test void a05_pageHeader_titleAndSubAreTight() throws Exception {
101 		var css = readChromeCss();
102 		assertTrue(css.contains(".jc-page-header h1"), css);
103 		var h1Start = css.indexOf(".jc-page-header h1 {");
104 		assertTrue(h1Start >= 0, css);
105 		var h1 = css.substring(h1Start, css.indexOf('}', h1Start));
106 		assertTrue(h1.contains("margin: 0 0 var(--jc-space-1);"), h1);
107 		assertTrue(h1.contains("line-height: 1.2;"), h1);
108 		var subStart = css.indexOf(".jc-page-header .jc-page-sub {");
109 		assertTrue(subStart >= 0, css);
110 		var sub = css.substring(subStart, css.indexOf('}', subStart));
111 		assertTrue(sub.contains("margin: 0;"), sub);
112 		assertFalse(sub.contains("margin-top: 2px"), sub);
113 	}
114 
115 	@Test void a02_cardAndPageHeader_haveNo1180MaxWidth() throws Exception {
116 		var css = readChromeCss();
117 		assertFalse(css.contains("max-width: 1180px"), "Q5: drop max-width 1180px on .jc-card / .jc-page-header");
118 	}

120 	@Test void a03_tableChrome_transparentUnpadded_outerOnly() throws Exception {
121 		var c = flat();
122 		var sel = ".jc-card:has([data-juneau-layout=\"wide\"]):not(.jc-card .jc-card):not(.jc-card-bare .jc-card)";
123 		assertTrue(c.contains(sel), "missing table-chrome selector");
124 		var start = c.indexOf(sel);
125 		var brace = c.indexOf('{', start);
126 		var end = c.indexOf('}', brace);
127 		var body = c.substring(brace, end);
128 		assertContainsAll(body, "background-color: transparent", "padding: 0", "box-shadow: none");
129 	}
