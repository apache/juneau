17  package org.apache.juneau.rest.server.console;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.*;
24  import org.junit.jupiter.api.*;
25  
26  /**
27   * The {@link Theme} dark-accent channel: {@link Theme.Builder#dark(String, String)}, {@link Theme#getDarkTokens()},
28   * and the stock themes' dark accent values.
29   */
30  class Theme_DarkChannel_Test extends TestBase {
31  
32  	private static Map<String,String> map(String... kv) {
33  		var m = new LinkedHashMap<String,String>();
34  		for (var i = 0; i < kv.length; i += 2)
35  			m.put(kv[i], kv[i + 1]);
36  		return m;
37  	}
38  
39  	@Test void a01_builderDark() {
40  		var theme = Theme.create("x").token("--jc-accent", "#123456").dark("--jc-link", "#6cb4ff").build();
41  		assertEquals(map("--jc-link", "#6cb4ff"), theme.getDarkTokens());
42  		var target = theme.getDarkTokens();
43  		assertThrows(UnsupportedOperationException.class, () -> target.put("--jc-focus", "#fff"));
44  	}
45  
46  	@Test void a02_darkRejectsVar() {
47  		var b = Theme.create("x");
48  		assertThrows(IllegalArgumentException.class, () -> b.dark("--jc-link", "var(--jc-accent)"));
49  	}
50  
51  	@Test void a03_darkReservedRejected() {
52  		var b = Theme.create("x");
53  		assertThrows(IllegalArgumentException.class, () -> b.dark("--jc-chrome-bg-x", "#000000"));
54  		// --jc-chrome-bg is a Theme token today, so the reserved guard allows it.
55  		assertEquals(map("--jc-chrome-bg", "#7a1a05"), b.dark("--jc-chrome-bg", "#7a1a05").build().getDarkTokens());
56  	}
57  
58  	@Test void a04_darkGrammar() {
59  		var b = Theme.create("x");
60  		assertThrows(IllegalArgumentException.class, () -> b.dark("--jc-link", "red; }"));
61  		assertThrows(IllegalArgumentException.class, () -> b.dark("--jc-link", "url(http://x/y)"));
62  		assertThrows(IllegalArgumentException.class, () -> b.dark("link", "#fff"));
63  	}
64  
65  	@Test void a05_stockValues() {
66  		var open = map(
67  			"--jc-link", "#6cb4ff",
68  			"--jc-focus", "#4aa3ff",
69  			"--jc-accent-selected", "#4aa3ff",
70  			"--jc-accent-wash", "rgba(21,137,238,0.22)");
71  		assertEquals(open, Theme.OPEN.getDarkTokens());
72  		assertEquals(open, Theme.GRAY.getDarkTokens());
73  		assertEquals(map(
74  			"--jc-link", "#d9a65c",
75  			"--jc-focus", "#a9772f",
76  			"--jc-accent-wash", "rgba(169,119,47,0.25)"), Theme.LIGHT_BROWN.getDarkTokens());
77  		assertEquals(map(
78  			"--jc-link", "#f27a9a",
79  			"--jc-accent", "#e0557a",
80  			"--jc-focus", "#e0557a",
81  			"--jc-accent-selected", "#e0557a",
82  			"--jc-accent-wash", "rgba(224,85,122,0.22)"), Theme.LIGHT_RED.getDarkTokens());
83  		assertEquals(map(
84  			"--jc-link", "#ff8a70",
85  			"--jc-chrome-bg", "#7a1a05",
86  			"--jc-accent-wash", "rgba(255,86,48,0.22)"), Theme.RED.getDarkTokens());
87  	}
88  
89  	@Test void a06_deriveFromCarriesDark() {
90  		// GRAY is derived from OPEN and declares no dark accents of its own, so it inherits OPEN's.
91  		assertFalse(Theme.OPEN.getDarkTokens().isEmpty());
92  		assertEquals(new ArrayList<>(Theme.OPEN.getDarkTokens().entrySet()), new ArrayList<>(Theme.GRAY.getDarkTokens().entrySet()));
93  	}
94  
95  	@Test void a07_darkNeverLeaksIntoLightTokens() {
96  		var theme = Theme.create("x").token("--jc-link", "#111111").dark("--jc-link", "#eeeeee").build();
97  		assertEquals("#111111", theme.getTokens().get("--jc-link"));
98  		assertEquals("#eeeeee", theme.getDarkTokens().get("--jc-link"));
99  	}
100 }
