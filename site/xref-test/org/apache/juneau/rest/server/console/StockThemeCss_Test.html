28  /**
29   * Drift guard for the shipped {@code juneau-theme-<name>.css} stock theme stylesheets the FTL {@code <@theme>}
30   * directive links: each stylesheet's {@code html:root{}} token block must be <b>provably identical</b> (same tokens,
31   * same order, same values) to its {@link Theme} constant, and the {@code @media not print} dark block that follows it must be
32   * identical to {@link Theme#getDarkTokens()}, so a change to a {@link Theme} that is not mirrored
33   * into its stylesheet (or vice-versa) fails the build rather than silently serving a stale palette.
34   *
35   * @since 10.0.0
36   */
37  class StockThemeCss_Test extends TestBase {
38  
39  	/** The selector that opens a stylesheet's dark accent block. */
40  	private static final String DARK_SELECTOR = "html:root[data-jc-scheme=\"dark\"]";
41  
42  	/**
43  	 * Parses a stylesheet into its two token blocks: the {@code html:root{ ... }} light block, then the
44  	 * {@code @media not print{ html:root[data-jc-scheme="dark"]{ ... } }} dark block that must follow it.
45  	 *
46  	 * @return A two-element list: the light tokens, then the dark tokens, each insertion-ordered.
47  	 */
48  	private static List<LinkedHashMap<String,String>> parse(String rawCss) {
49  		// Strip block comments first - the drift-guard header mentions "html:root{}", whose braces would
50  		// otherwise fool the brace-locating below.
51  		var css = rawCss.replaceAll("(?s)/\\*.*?\\*/", "");
52  		var media = css.indexOf("@media not print");
53  		assertTrue(media > 0, () -> "no '@media not print' dark block after the light block:\n" + css);
54  		var light = css.substring(0, media);
55  		var open = light.indexOf('{');
56  		var close = light.lastIndexOf('}');
57  		assertTrue(open >= 0 && close > open, () -> "no html:root{} block:\n" + light);
58  		assertEquals("html:root", light.substring(0, open).trim(), () -> "light block must come first:\n" + light);
59  		var dark = css.substring(media);
60  		var sel = dark.indexOf(DARK_SELECTOR);
61  		assertTrue(sel > 0, () -> "no " + DARK_SELECTOR + " rule in the @media block:\n" + dark);
62  		var dOpen = dark.indexOf('{', sel);
63  		var dClose = dark.indexOf('}', dOpen);
64  		assertTrue(dOpen > sel && dClose > dOpen, () -> "unterminated dark block:\n" + dark);
65  		return List.of(
66  			declarations(light.substring(open + 1, close)),
67  			declarations(dark.substring(dOpen + 1, dClose)));
68  	}
69  
70  	private static LinkedHashMap<String,String> declarations(String body) {
71  		var out = new LinkedHashMap<String,String>();
72  		for (var decl : body.split(";")) {
73  			var d = decl.trim();
74  			if (d.isEmpty())
75  				continue;
76  			var colon = d.indexOf(':');
77  			assertTrue(colon > 0, () -> "malformed declaration: '" + d + "'");
78  			out.put(d.substring(0, colon).trim(), d.substring(colon + 1).trim());
79  		}
80  		return out;
81  	}
82  
83  	private static String read(String name) throws IOException {
84  		var resource = "/org/apache/juneau/console/juneau-theme-" + name + ".css";
85  		try (var in = StockThemeCss_Test.class.getResourceAsStream(resource)) {
86  			assertNotNull(in, () -> "missing classpath resource: " + resource);
87  			return new String(in.readAllBytes(), UTF_8);
88  		}
89  	}
90  
91  	private void assertStockCssMatchesTheme(String name, Theme theme) throws IOException {
92  		var blocks = parse(read(name));
93  		var light = blocks.get(0);
94  		var dark = blocks.get(1);
95  		// Same key set, same iteration order, same values as the Theme constant, in each block.
96  		assertEquals(new ArrayList<>(light.keySet()), new ArrayList<>(theme.getTokens().keySet()),
97  			() -> "token order/keys drift in juneau-theme-" + name + ".css");
98  		assertEquals(light, theme.getTokens(),
99  			() -> "token value drift in juneau-theme-" + name + ".css");
100 		assertEquals(new ArrayList<>(dark.keySet()), new ArrayList<>(theme.getDarkTokens().keySet()),
101 			() -> "dark token order/keys drift in juneau-theme-" + name + ".css");
102 		assertEquals(dark, theme.getDarkTokens(),
103 			() -> "dark token value drift in juneau-theme-" + name + ".css");
104 	}

112 	@Test void a06_builtinNamesMatchThemeConstants() {
113 		assertEquals(
114 			ConsoleChromeMixin.BUILTIN_THEME_NAMES,
115 			List.of("open", "light-red", "light-brown", "red", "gray"));
116 	}
117 
118 	@Test void a07_stockThemes_doNotDeclareHeaderOrNavBg() throws Exception {
119 		for (var name : ConsoleChromeMixin.BUILTIN_THEME_NAMES) {
120 			for (var parsed : parse(read(name))) {
121 				assertFalse(parsed.containsKey("--jc-header-bg"),
122 					() -> "juneau-theme-" + name + ".css must not declare --jc-header-bg");
123 				assertFalse(parsed.containsKey("--jc-nav-bg"),
124 					() -> "juneau-theme-" + name + ".css must not declare --jc-nav-bg");
125 			}
126 		}
127 	}

129 	@Test void a08_stockThemes_doNotClaimToBeGenerated() throws Exception {
130 		for (var name : ConsoleChromeMixin.BUILTIN_THEME_NAMES)
131 			assertFalse(read(name).toLowerCase().contains("generated"),
132 				() -> "juneau-theme-" + name + ".css is hand-maintained and must not claim to be generated");
133 	}
