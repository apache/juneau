17  package org.apache.juneau.rest.server.console.test;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.apache.juneau.*;
22  import org.junit.jupiter.api.*;
23  
24  class PageContractAssert_Bus_Test extends TestBase {
25  
26  	private static String page(String contractJson) {
27  		return "<html><body><script type=\"application/json\" id=\"juneau-page\">" + contractJson + "</script></body></html>";
28  	}
29  
30  	private static final String WIRED = page("""
31  		{"contractVersion":"1","title":"t","nav":[],"activeNav":[],
32  		 "topics":[{"topic":"app.region-picked","retain":true,"publisher":"script"},
33  		           {"topic":"ops.jobs","retain":true,"publisher":"server"}],
34  		 "bridges":[{"id":"ops","transport":"sse","session":"/s","downstream":["ops.jobs"]}],
35  		 "cards":[{"id":"a","type":"html","template":"a","publishes":[{"topic":"ssc.focus","retain":true}]},
36  		          {"id":"b","type":"html","template":"b","subscribes":[{"topic":"ssc.focus","as":"refresh"},{"topic":"app.region-picked","as":"refresh"}]}]}
37  		""");
38  
39  	@Test void a01_positive() {
40  		PageContractAssert.assertPage(WIRED)
41  			.publishes("a", "ssc.focus")
42  			.subscribes("b", "ssc.focus")
43  			.subscribes("b", "app.region-picked", "refresh")
44  			.declaresTopic("ops.jobs")
45  			.hasBridge("ops")
46  			.hasNoWiringErrors();
47  	}
48  
49  	@Test void b01_negative_messages() {
50  		var a = PageContractAssert.assertPage(WIRED);
51  		assertEquals("card 'a' does not subscribe to 'ssc.focus'; subscribes: []",
52  			assertThrows(AssertionError.class, () -> a.subscribes("a", "ssc.focus")).getMessage());
53  		assertEquals("card 'b' subscribes to 'ssc.focus' as [refresh], not 'filter'",
54  			assertThrows(AssertionError.class, () -> a.subscribes("b", "ssc.focus", "filter")).getMessage());
55  		assertEquals("card 'b' does not publish 'ssc.focus'; publishes: []",
56  			assertThrows(AssertionError.class, () -> a.publishes("b", "ssc.focus")).getMessage());
57  		assertEquals("no topics entry 'app.nope'; topics: [app.region-picked, ops.jobs]",
58  			assertThrows(AssertionError.class, () -> a.declaresTopic("app.nope")).getMessage());
59  		assertEquals("no bridge 'nope'; bridges: [ops]",
60  			assertThrows(AssertionError.class, () -> a.hasBridge("nope")).getMessage());
61  	}
62  
63  	@Test void b02_wiringErrors_listEveryProblem() {
64  		var broken = page("""
65  			{"contractVersion":"1","title":"t","nav":[],"activeNav":[],
66  			 "cards":[{"id":"b","type":"html","template":"b","subscribes":[{"topic":"ssc.focus","as":"refresh"}]}]}
67  			""");
68  		var target = PageContractAssert.assertPage(broken);
69  		var e = assertThrows(AssertionError.class, target::hasNoWiringErrors);
70  		assertTrue(e.getMessage().startsWith("page wiring has errors:\nE-41 card 'b' subscribes to 'ssc.focus' but nothing publishes it"), e.getMessage());
71  	}
72  
73  	@Test void a02_twoRolesOnOneTopic() {
74  		var two = page("""
75  			{"contractVersion":"1","title":"t","nav":[],"activeNav":[],
76  			 "cards":[{"id":"a","type":"html","template":"a","publishes":[{"topic":"ssc.focus","retain":true}]},
77  			          {"id":"b","type":"html","template":"b","subscribes":[{"topic":"ssc.focus","as":"refresh"},{"topic":"ssc.focus","as":"filter"}]}]}
78  			""");
79  		PageContractAssert.assertPage(two).subscribes("b", "ssc.focus", "filter").subscribes("b", "ssc.focus", "refresh");
80  	}
81  }
