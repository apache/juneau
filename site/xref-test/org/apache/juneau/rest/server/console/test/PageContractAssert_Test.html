28  	// The contract as the server writes it ('<' escaped), plus its templates. One nested <template>
29  	// inside a segment proves the depth-aware template scan.
30  	static final String HTML = "<!DOCTYPE html><html><head><title>t</title></head><body data-juneau-csrf=\"x\">"
31  		+ "<script type=\"application/json\" id=\"juneau-page\">"
32  		+ "{\"contractVersion\":\"1\",\"title\":\"Support Enablement Console\",\"theme\":{\"name\":\"open\"},"
33  		+ "\"header\":{\"title\":\"Support Enablement Console\",\"chrome\":true,\"slots\":{\"banner\":\"header.banner\"}},"
34  		+ "\"footer\":{\"text\":\"Sandbox Support Console\"},"
35  		+ "\"nav\":[{\"id\":\"home\",\"label\":\"Home\",\"href\":\"/home\",\"children\":["
36  		+ "{\"id\":\"about\",\"label\":\"About\",\"href\":\"/home/about\"},{\"id\":\"setup\",\"label\":\"Setup\",\"href\":\"/home/setup\"}]},"
37  		+ "{\"id\":\"slo\",\"label\":\"SLO\",\"href\":\"/slo\"}],"
38  		+ "\"activeNav\":[\"home\",\"setup\"],"
39  		+ "\"cards\":[{\"id\":\"jc-seg-1\",\"type\":\"html\",\"template\":\"jc-seg-1\",\"bare\":true},"
40  		+ "{\"id\":\"releases\",\"type\":\"datatables\",\"table\":{\"dataUrl\":\"/rest/releases/data\","
41  		+ "\"columns\":[{\"key\":\"name\",\"label\":\"Name\"}]}}]}"
42  		+ "</script>"
43  		+ "<template data-slot=\"header.banner\"><div class=\"demo\">\\u003cdemo</div></template>"
44  		+ "<template data-card=\"jc-seg-1\"><p id=\"ssc-table-slot\">x</p><template id=\"inner\"><i>i</i></template><b>after</b></template>"
45  		+ "<script src=\"/juneau-console/juneau-console.js\"></script></body></html>";
46  
47  	@Test void a01_happyPath() {
48  		PageContractAssert.assertPage(HTML)
49  			.isValid()
50  			.hasContractVersion("1")
51  			.hasTitle("Support Enablement Console")
52  			.hasTheme("open")
53  			.hasActiveNav("home", "setup")
54  			.hasNavPath("home", "about")
55  			.hasNavHref("home/setup", "/home/setup")
56  			.hasNavHref("slo", "/slo")
57  			.hasNavChildren("home", "about", "setup")
58  			.hasHeaderTitle("Support Enablement Console")
59  			.hasHeaderSlot("banner")
60  			.hasFooterText("Sandbox Support Console")
61  			.hasCard("releases", "datatables")
62  			.hasCardOrder("jc-seg-1", "releases")
63  			.templateContains("jc-seg-1", "id=\"ssc-table-slot\"")
64  			.templateContains("jc-seg-1", "<b>after</b>");
65  		assertEquals("open", PageContractAssert.assertPage(HTML).contract().getMap("theme").getString("name"));
66  	}

68  	@Test void a02_failuresAreReadable() {
69  		var a = PageContractAssert.assertPage(HTML);
70  		assertMessage(() -> a.hasActiveNav("slo"), "activeNav: expected [slo] but was [home, setup]");
71  		assertMessage(() -> a.hasNavPath("home", "nope"), "nav path [home, nope] not found (failed at 'nope')");
72  		assertMessage(a::hasNavPath, "hasNavPath() requires at least one path segment");
73  		assertMessage(() -> a.hasNavHref("slo", "/x"), "nav 'slo' href: expected '/x' but was '/slo'");
74  		assertMessage(() -> a.hasNavChildren("home", "about"), "nav 'home' children: expected [about] but was [about, setup]");
75  		assertMessage(() -> a.hasFooterText("x"), "footer.text: expected 'x' but was 'Sandbox Support Console'");
76  		assertMessage(() -> a.hasFooterSlot("content"), "footer.slots has no 'content'");
77  		assertMessage(() -> a.hasHeaderSlot("brand"), "header.slots has no 'brand'");
78  		assertMessage(() -> a.hasCard("releases", "html"), "card 'releases' type: expected 'html' but was 'datatables'");
79  		assertMessage(() -> a.hasCard("nope", "html"), "no card 'nope'; cards: [jc-seg-1, releases]");
80  		assertMessage(() -> a.hasCardOrder("releases", "jc-seg-1"), "card order: expected [releases, jc-seg-1] but was [jc-seg-1, releases]");
81  		assertMessage(() -> a.templateContains("jc-seg-1", "zzz"), "template 'jc-seg-1' does not contain 'zzz'");
82  		assertMessage(() -> a.template("nope"), "no <template data-card|data-slot=\"nope\">; templates: [header.banner, jc-seg-1]");
83  		assertMessage(a::hasNoActiveNav, "activeNav: expected [] but was [home, setup]");
84  	}

108 	private static void assertMessage(Runnable r, String expected) {
109 		var e = assertThrows(AssertionError.class, r::run);
110 		assertEquals(expected, e.getMessage());
111 	}

123 			"{\"id\":\"releases\",\"type\":\"datatables\",\"table\":{\"dataUrl\":\"/rest/releases/data\","
124 				+ "\"columns\":[{\"key\":\"name\",\"label\":\"Name\"}]}}",

144 	@Test void a11_isTwinOf_sameContractAndStrippedTemplates() {
145 		var a = PageContractAssert.assertPage(HTML);
146 		assertSame(a, a.isTwinOf(PageContractAssert.assertPage(HTML)));
147 		// Whitespace around a template body is not part of the contract.
148 		var padded = HTML.replace("<template data-slot=\"header.banner\">", "<template data-slot=\"header.banner\">\n  ")
149 			.replace("demo</div></template>", "demo</div>\n</template>");
150 		a.isTwinOf(PageContractAssert.assertPage(padded));
151 	}
