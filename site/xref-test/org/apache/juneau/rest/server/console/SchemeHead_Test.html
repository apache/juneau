31  /**
32   * The colour-scheme head pieces on {@link ConsoleChromeMixin}: the byte-constant pre-paint script and its CSP hash,
33   * the dark override block, and the {@code schemeHead} helper for pages that don't render through the console
34   * directive.
35   */
36  @SuppressWarnings({
37  	"resource" // MockRestClient/RestResponse lifecycles are owned by the fluent chain; nothing long-lived leaks.
38  })
39  class SchemeHead_Test extends TestBase {
40  
41  	/** The pre-paint script's inner text, pasted verbatim from the design. */
42  	private static final String EXPECTED_SCRIPT = """
43  		(function(d){var e=d.documentElement,k="juneau-console:scheme",s=null;
44  		if(e.hasAttribute("data-jc-scheme-toggle")){try{s=localStorage.getItem(k)}catch(x){}}
45  		if(s!=="light"&&s!=="dark"){s=e.getAttribute("data-jc-scheme-pin");}
46  		if(s!=="light"&&s!=="dark"){try{s=matchMedia("(prefers-color-scheme: dark)").matches?"dark":"light"}catch(x){s="light"}}
47  		e.setAttribute("data-jc-scheme",s);})(document);""";
48  
49  	private static final String SEP = "\n--8<--\n";
50  
51  	@Rest
52  	public static class Host extends BasicRestServlet {
53  		private static final long serialVersionUID = 1L;
54  
55  		@RestGet(path="/head", produces="text/plain")
56  		public String head(RestRequest req) {
57  			var nonce = req.getQueryParam("nonce").asString().orElse(null);
58  			if (nonce != null)
59  				req.setAttribute(RestRequest.CSP_NONCE_ATTR, nonce);
60  			var pin = req.getQueryParam("pin").asString().orElse(null);
61  			var toggle = "true".equals(req.getQueryParam("toggle").asString().orElse(null));
62  			var h = ConsoleChromeMixin.schemeHead(req, pin, toggle);
63  			return h.htmlAttributes() + SEP + h.head() + SEP + ConsoleChromeMixin.themeAssetUrl(req, "dark");
64  		}
65  	}
66  
67  	private static String[] head(String query) throws Exception {
68  		var body = MockRestClient.buildLax(Host.class).get("/head?" + query).header("Accept", "text/plain").run().assertStatus(200).getContent().asString();
69  		var parts = body.split(SEP, -1);
70  		assertEquals(3, parts.length, () -> body);
71  		return parts;
72  	}
73  
74  	@Test void a01_scriptBytes() {
75  		assertEquals(EXPECTED_SCRIPT, ConsoleChromeMixin.SCHEME_SCRIPT);
76  	}
77  
78  	@Test void a02_hash() throws Exception {
79  		var digest = MessageDigest.getInstance("SHA-256").digest(EXPECTED_SCRIPT.getBytes(StandardCharsets.UTF_8));
80  		assertEquals(ConsoleChromeMixin.SCHEME_SCRIPT_CSP_HASH, "'sha256-" + Base64.getEncoder().encodeToString(digest) + "'");
81  	}
82  
83  	@Test void a03_overrideBlock() {
84  		var tokens = new LinkedHashMap<String,String>();
85  		tokens.put("--jc-link", "#6cb4ff");
86  		assertEquals("@media not print{html:root[data-jc-scheme=\"dark\"]{--jc-link:#6cb4ff;--jc-focus:var(--jc-accent);}}",
87  			ConsoleChromeMixin.schemeOverrideBlock(tokens, Map.of("--jc-focus", "var(--jc-accent)")));
88  		assertEquals("", ConsoleChromeMixin.schemeOverrideBlock(Map.of(), Map.of()));
89  	}
90  
91  	@Test void a03b_overrideBlock_leafIsEscaped() {
92  		var css = ConsoleChromeMixin.schemeOverrideBlock(Map.of("--jc-font", "'My;Font'"), Map.of());
93  		assertTrue(css.contains("\\3B "), css);
94  		assertFalse(css.contains("'My;Font'"), css);
95  	}
96  
97  	@Test void a03c_overrideBlock_rejectsNonLiteralLeaf() {
98  		var arg = Map.of("--jc-link", "var(--jc-accent)");
99  		Map<String,String> none = Map.of();
100 		var ex = assertThrows(IllegalArgumentException.class,
101 			() -> ConsoleChromeMixin.schemeOverrideBlock(arg, none));
102 		assertTrue(ex.getMessage().contains("dark values must be literals"), ex::getMessage);
103 	}
104 
105 	@Test void a03d_overrideBlock_rejectsBadAliasAndReservedName() {
106 		var arg2 = Map.of("--jc-focus", "#fff;}body{");
107 		Map<String,String> none = Map.of();
108 		assertThrows(IllegalArgumentException.class,
109 			() -> ConsoleChromeMixin.schemeOverrideBlock(none, arg2));
110 		var arg3 = Map.of("--jc-chrome-foo", "#fff");
111 		var ex = assertThrows(IllegalArgumentException.class,
112 			() -> ConsoleChromeMixin.schemeOverrideBlock(arg3, none));
113 		assertTrue(ex.getMessage().contains("Cannot declare reserved token '--jc-chrome-foo'"), ex::getMessage);
114 		var arg4 = Map.of("--jc-x", "#fff");
115 		var arg5 = Map.of("--jc-x", "var(--jc-accent)");
116 		ex = assertThrows(IllegalArgumentException.class,
117 			() -> ConsoleChromeMixin.schemeOverrideBlock(arg4, arg5));
118 		assertTrue(ex.getMessage().contains("both a leaf token and an alias"), ex::getMessage);
119 	}
120 
121 	@Test void a04_schemeHead() throws Exception {
122 		var p = head("pin=dark&toggle=true");
123 		assertEquals("data-jc-scheme=\"dark\" data-jc-scheme-pin=\"dark\" data-jc-scheme-toggle", p[0]);
124 		assertEquals("<script>" + EXPECTED_SCRIPT + "</script>\n<link rel=\"stylesheet\" href=\"" + p[2] + "\">\n", p[1]);
125 		assertTrue(p[2].contains("juneau-theme-dark.css?v="), () -> p[2]);
126 
127 		assertEquals("data-jc-scheme=\"light\" data-jc-scheme-pin=\"light\"", head("pin=light&toggle=false")[0]);
128 		assertEquals("data-jc-scheme-toggle", head("pin=auto&toggle=true")[0]);
129 
130 		var auto = head("pin=auto&toggle=false");
131 		assertEquals("", auto[0]);
132 		assertTrue(auto[1].startsWith("<script>" + EXPECTED_SCRIPT + "</script>\n"), () -> auto[1]);
133 
134 		assertThrows(IllegalArgumentException.class, () -> ConsoleChromeMixin.schemeHead(null, "dim", false));
135 		assertThrows(IllegalArgumentException.class, () -> ConsoleChromeMixin.schemeHead(null, null, false));
136 	}
137 
138 	@Test void a05_schemeHead_stampsRequestNonce() throws Exception {
139 		var p = head("pin=auto&toggle=false&nonce=abc");
140 		assertTrue(p[1].startsWith("<script nonce=\"abc\">" + EXPECTED_SCRIPT + "</script>\n"), () -> p[1]);
141 	}
