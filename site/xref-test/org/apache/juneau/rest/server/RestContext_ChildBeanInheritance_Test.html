20  import static org.junit.jupiter.api.Assertions.*;

26  /**
27   * Tests that {@link Rest#children() @Rest(children)} sub-resources inherit the {@link Bean @Bean} method results and
28   * {@link Bean @Bean} fields declared on their group resource, transitively through every nesting level, while
29   * framework-managed beans and the children's own beans stay isolated.
30   *
31   * @since 10.0.0
32   */
33  @SuppressWarnings({
34  	"resource" // getBeanStore() returns the context's own (already-owned) BeanStore; the test doesn't own it.
35  })
36  class RestContext_ChildBeanInheritance_Test extends org.apache.juneau.TestBase {
37  
38  	static RestContext.Args argsOf(Class<?> resourceClass, java.util.function.Supplier<?> supplier) {
39  		return new RestContext.Args(resourceClass, null, null, supplier, null, null, null, null, null, null);
40  	}
41  
42  	public static class Greeting {
43  		private final String text;
44  		public Greeting(String text) { this.text = text; }
45  		public String getText() { return text; }
46  	}
47  
48  	public static class Counter {
49  		private final String origin;
50  		public Counter(String origin) { this.origin = origin; }
51  		public String getOrigin() { return origin; }
52  	}
53  
54  	public static class Shout {
55  		private final String text;
56  		public Shout(String text) { this.text = text; }
57  		public String getText() { return text; }
58  	}
59  
60  	/** Captures beans injected into a child's {@code @RestInit} hook. */
61  	public static class Captured {
62  		public Greeting greeting;
63  		public Counter counter;
64  		public Greeting getGreeting() { return greeting; }
65  		public Counter getCounter() { return counter; }
66  	}
67  
68  	@Rest(path="/grandchild")
69  	public static class GrandChild {}
70  
71  	@Rest(path="/child", children={GrandChild.class})
72  	public static class Child {
73  		static final Captured CAPTURED = new Captured();
74  
75  		// A child-level @Bean method that composes a bean inherited from the group.
76  		@Bean public Shout shout(Greeting greeting) { return new Shout(greeting.getText().toUpperCase()); }
77  
78  		@RestInit public void init(Greeting greeting, Counter counter) {
79  			CAPTURED.greeting = greeting;
80  			CAPTURED.counter = counter;
81  		}
82  	}
83  
84  	@Rest(path="/sibling")
85  	public static class Sibling {
86  		@Bean public Counter counter() { return new Counter("sibling"); }
87  	}
88  
89  	@Rest(path="/override")
90  	public static class OverridingChild {
91  		@Bean public Greeting greeting() { return new Greeting("override"); }
92  	}
93  
94  	@Rest(children={Child.class, Sibling.class, OverridingChild.class})
95  	public static class Group {
96  		static final MethodExecStore GROUP_EXEC_STORE = MethodExecStore.create().build();
97  
98  		@Bean public Counter counter = new Counter("field");
99  
100 		@Bean(name="named") public Greeting namedGreeting = new Greeting("named");
101 
102 		@Bean public Greeting greeting() { return new Greeting("group"); }
103 
104 		// Framework-managed type: stays local to the group.
105 		@Bean public MethodExecStore methodExecStore() { return GROUP_EXEC_STORE; }
106 	}
107 
108 	private static RestContext group() throws Exception {
109 		return new RestContext(argsOf(Group.class, Group::new));
110 	}
111 
112 	private static RestContext child(RestContext parent, String path) {
113 		return parent.getRestChildren().asMap().get(path);
114 	}
115 
116 	private static <T> T bean(RestContext ctx, Class<T> type) {
117 		return ctx.getBeanStore().getBean(type).orElse(null);
118 	}
119 
120 	//-----------------------------------------------------------------------------------------------------------
121 	// a - group beans reach children
122 	//-----------------------------------------------------------------------------------------------------------
123 
124 	@Test void a01_childSeesGroupBeanMethodResult() throws Exception {
125 		var a = child(group(), "child");
126 		assertBean(bean(a, Greeting.class), "text", "group");
127 	}
128 
129 	@Test void a02_childSeesGroupBeanField() throws Exception {
130 		var a = child(group(), "child");
131 		assertBean(bean(a, Counter.class), "origin", "field");
132 		assertBean(a.getBeanStore().getBean(Greeting.class, "named").orElse(null), "text", "named");
133 	}
134 
135 	@Test void a03_childBeanMethodAndRestInitResolveGroupBeans() throws Exception {
136 		var a = child(group(), "child");
137 		assertBean(bean(a, Shout.class), "text", "GROUP");
138 		assertBean(Child.CAPTURED, "greeting{text},counter{origin}", "{group},{field}");
139 	}
140 
141 	@Test void a04_grandchildReachesRoot() throws Exception {
142 		var a = child(child(group(), "child"), "grandchild");
143 		assertBean(bean(a, Greeting.class), "text", "group");
144 		assertBean(bean(a, Counter.class), "origin", "field");
145 		// The intermediate child's own @Bean results are inherited too.
146 		assertBean(bean(a, Shout.class), "text", "GROUP");
147 	}
148 
149 	@Test void a05_sameInstanceSharedWithGroup() throws Exception {
150 		var a = group();
151 		assertSame(bean(a, Greeting.class), bean(child(a, "child"), Greeting.class));
152 		assertSame(bean(a, Counter.class), bean(child(a, "child"), Counter.class));
153 	}
154 
155 	//-----------------------------------------------------------------------------------------------------------
156 	// b - isolation
157 	//-----------------------------------------------------------------------------------------------------------
158 
159 	@Test void b01_childOwnBeanWinsOverGroupBean() throws Exception {
160 		var a = group();
161 		assertBean(bean(child(a, "override"), Greeting.class), "text", "override");
162 		assertBean(bean(a, Greeting.class), "text", "group");
163 	}
164 
165 	@Test void b02_childBeansDoNotLeakToGroupOrSiblings() throws Exception {
166 		var a = group();
167 		assertBean(bean(child(a, "sibling"), Counter.class), "origin", "sibling");
168 		assertBean(bean(child(a, "child"), Counter.class), "origin", "field");
169 		assertNull(bean(a, Shout.class));
170 		assertNull(bean(child(a, "sibling"), Shout.class));
171 	}
172 
173 	@Test void b03_groupFrameworkBeansStayLocal() throws Exception {
174 		var a = group();
175 		assertSame(Group.GROUP_EXEC_STORE, bean(a, MethodExecStore.class));
176 		assertNotSame(Group.GROUP_EXEC_STORE, bean(child(a, "child"), MethodExecStore.class));
177 		assertNotSame(Group.GROUP_EXEC_STORE, child(a, "child").getMethodExecStore());
178 		assertNotSame(a.getRestChildren(), child(a, "child").getRestChildren());
179 	}
