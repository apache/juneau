85  		@RestGet(path="/mutate")
86  		public String mutate(RequestQueryParamList q) {
87  			q.add("added1", "v1");
88  			q.add(HttpStringPart.of("added2", "v2"), HttpStringPart.of("added3", "v3"));
89  			q.set("orig", "newVal");
90  			q.set(HttpStringPart.of("setHttpPart", "setVal"));
91  			var beforeRemove = q.size();
92  			q.remove("removeMe");
93  			var afterRemove = q.size();
94  			return "added1=" + q.get("added1").asString().orElse("?")
95  				+ ",added2=" + q.get("added2").asString().orElse("?")
96  				+ ",added3=" + q.get("added3").asString().orElse("?")
97  				+ ",orig=" + q.get("orig").asString().orElse("?")
98  				+ ",setHttpPart=" + q.get("setHttpPart").asString().orElse("?")
99  				+ ",sizeBefore=" + beforeRemove
100 				+ ",sizeAfter=" + afterRemove;
101 		}

129 		// Exercises caseSensitive(boolean) and getNames().
130 		@RestGet(path="/caseSensitive")
131 		public String caseSensitive(RequestQueryParamList q) {
132 			q.caseSensitive(true);
133 			q.add("UPPER", "u1");
134 			q.add("upper", "u2");
135 			var sensCount = q.getAll("UPPER").size();
136 			q.caseSensitive(false);
137 			var insensCount = q.getAll("UPPER").size();
138 			return "sens=" + sensCount + ",insens=" + insensCount + ",namesContainsUpper=" + q.getNames().contains("upper");
139 		}

141 		// Exercises getSorted().
142 		@RestGet(path="/sorted")
143 		public String sorted(RequestQueryParamList q) {
144 			q.caseSensitive(false);
145 			var ci = q.getSorted().toList().stream().map(RequestQueryParam::getName).toList();
146 			q.caseSensitive(true);
147 			var cs = q.getSorted().toList().stream().map(RequestQueryParam::getName).toList();
148 			return "ci=" + ci + ",cs=" + cs;
149 		}

151 		// Exercises set(HttpPart...) and equals/hashCode and toString.
152 		@SuppressWarnings({
153 			"unlikely-arg-type" // Intentionally tests equals() returns false for a non-compatible argument type.
154 		})
155 		@RestGet(path="/equality")
156 		public String equality(RequestQueryParamList q) {
157 			var c1 = q.copy();
158 			var c2 = q.copy();
159 			q.set(HttpStringPart.of("a", "modified"));
160 			var s = q.toString();
161 			var hc = c1.hashCode() == c2.hashCode();
162 			return "eqSelf=" + c1.equals(c1)
163 				+ ",eqOther=" + c1.equals(c2)
164 				+ ",eqDiff=" + c1.equals(q)
165 				+ ",eqString=" + c1.equals("not a list")
166 				+ ",hashEq=" + hc
167 				+ ",toStringNotNull=" + (s != null && !s.isEmpty());
168 		}

396 			} catch (IllegalArgumentException e) {
397 				return "iae:" + e.getMessage().contains("headers");
398 			}
399 		}

403 				q.add((String)null, "v");
404 				return "no-throw";
405 			} catch (IllegalArgumentException e) {
406 				return "iae:" + e.getMessage().contains("name");
407 			}
408 		}

410 		public String containsAnyNullThrows(RequestQueryParamList q) {
411 			try {
412 				q.containsAny((String[])null);

414 			} catch (IllegalArgumentException e) {
415 				return "iae:" + e.getMessage().contains("names");
416 			}
417 		}
