17  package org.apache.juneau.rest.server.staticfile;
18  
19  import org.apache.juneau.*;
20  import org.apache.juneau.rest.mock.classic.*;
21  import org.apache.juneau.rest.server.*;
22  import org.apache.juneau.rest.server.servlet.*;
23  import org.junit.jupiter.api.*;
24  
25  /**
26   * Validates {@link StaticFilesMixin} mounted as a mixin via {@code @Rest(mixins=...)} on a
27   * vanilla {@link RestServlet}.
28   *
29   * <p>
30   * Acceptance:
31   * <ul>
32   * 	<li>{@code GET /static/javadoc.css} returns the file (single SVL-configurable default mount).
33   * 	<li>{@code GET} on a missing path returns 404.
34   * 	<li>{@code HEAD /static/...} mirrors GET headers with empty body.
35   * 	<li>The host's own endpoints are unaffected by the mixin.
36   * </ul>
37   *
38   * <p>
39   * The classpath resource {@code htdocs/javadoc.css} ships with {@code juneau-rest-server} and is
40   * therefore visible to {@code juneau-integration-tests}'s test classpath via the
41   * {@link BasicStaticFiles} default constructor's recursive {@code cp(...,"htdocs",true)} walk.
42   * The default {@link BasicStaticFiles} classpath search root still walks both {@code static/}
43   * and {@code htdocs/} directories &mdash; only the URL-side mount is now single. Per the
44   * multi-path collapse, the {@code /htdocs/*} mount alias (formerly a dual default)
45   * is now reached via {@code -Djuneau.staticfiles.path=htdocs}; that behavior is covered by
46   * {@code StaticFilesMixin_SvlPathOverride_Test}.
47   *
48   * @since 10.0.0
49   */
50  class StaticFilesMixin_AsMixin_Test extends TestBase {
51  
52  	@Rest(mixins=StaticFilesMixin.class)
53  	public static class A extends BasicRestServlet {
54  		private static final long serialVersionUID = 1L;
55  		@RestGet(path="/items") public String items() { return "items"; }
56  	}
57  
58  	private static final MockRestClient c = MockRestClient.buildLax(A.class);
59  
60  	@Test void a01_staticPathServesClasspathFile() throws Exception {
61  		c.get("/static/javadoc.css")
62  			.run()
63  			.assertStatus(200)
64  			.assertContent().asString().isContains("Licensed to the Apache Software Foundation");
65  	}
66  
67  	@Test void a02_missingFileReturns404() throws Exception {
68  		c.get("/static/does-not-exist.css")
69  			.run()
70  			.assertStatus(404);
71  	}
72  
73  	@Test void a03_hostEndpointStillReachable() throws Exception {
74  		c.get("/items")
75  			.accept("application/json")
76  			.run()
77  			.assertStatus(200)
78  			.assertContent().asString().isContains("items");
79  	}
80  
81  	@Test void a04_headStaticReturnsHeadersWithEmptyBody() throws Exception {
82  		// HEAD must mirror GET's Content-Type and Content-Length while suppressing the body.
83  		var get = c.get("/static/javadoc.css").run();
84  		get.assertStatus(200);
85  		var getContentType = get.getHeader("Content-Type").asString().orElse("");
86  		var getContentLength = get.getHeader("Content-Length").asString().orElse("");
87  
88  		var head = c.head("/static/javadoc.css").run();
89  		head.assertStatus(200);
90  		head.assertContent().is("");
91  
92  		// Header parity (RFC 7231 §4.3.2).
93  		var headContentType = head.getHeader("Content-Type").asString().orElse("");
94  		var headContentLength = head.getHeader("Content-Length").asString().orElse("");
95  		if (!getContentType.isEmpty())
96  			head.assertHeader("Content-Type").is(getContentType);
97  		if (!getContentLength.isEmpty())
98  			head.assertHeader("Content-Length").is(getContentLength);
99  		assert headContentType != null && headContentLength != null;
100 	}
101 
102 	@Test void a05_headMissingFileReturns404() throws Exception {
103 		c.head("/static/does-not-exist.css")
104 			.run()
105 			.assertStatus(404);
106 	}
107 }
