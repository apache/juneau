36  /**
37   * Real-Jetty deployment-parity assertion for {@link StaticFilesMixin}.
38   *
39   * <p>
40   * Boots a {@link org.apache.juneau.marshall.microservice.Microservice Microservice} backed by
41   * {@link org.apache.juneau.marshall.microservice.jetty.JettyConfiguration JettyConfiguration} on an
42   * ephemeral port via {@link MicroserviceTestFixture}, mounts a {@link BasicRestServlet} host with
43   * the static-files mixin, and hits the {@code /static/*} URL over real HTTP. The {@code /htdocs/*} mount has been removed from the mixin's default URL surface
44   * (single path per op), but the {@code Host} class extends {@link BasicRestServlet}
45   * which still owns the legacy {@code /htdocs/*} via {@code BasicRestOperations#getHtdoc}; the
46   * second test below verifies that legacy mount still serves end-to-end.
47   *
48   * <p>
49   * Catches things {@code MockRest} can't:
50   * <ul>
51   * 	<li><b>Real {@code Content-Type} negotiation.</b> The Jetty pipeline serves the file with the
52   * 		correct MIME type derived from the file extension (e.g. {@code text/css; charset=UTF-8}).
53   * 	<li><b>Real HEAD-via-GET contract.</b> The Servlet container sees the HEAD method and the
54   * 		response writer suppresses the body via
55   * 		{@link org.apache.juneau.rest.server.server.processor.HttpResourceProcessor}.
56   * 	<li><b>Real classpath resource resolution under a JAR.</b> {@code BasicStaticFiles} walks the
57   * 		classloader for {@code htdocs/javadoc.css} (contributed by {@code juneau-rest-server}'s
58   * 		main resources) so the URL routing flows end-to-end through the real network stack.
59   * </ul>
60   *
61   * @since 10.0.0
62   */
63  @org.apache.juneau.testing.JettyMicroserviceTest
64  class StaticFilesMixin_JettyMicroservice_Test extends TestBase {
65  
66  	/** Test host: vanilla {@link BasicRestServlet} subclass with the static-files mixin attached. */
67  	@Rest(mixins=StaticFilesMixin.class)
68  	public static class Host extends BasicRestServlet {
69  		private static final long serialVersionUID = 1L;
70  	}
71  
72  	/** {@code @Configuration} contributing the {@link Host} servlet for auto-mount. */
73  	@Configuration
74  	public static class HostConfig {
75  		@Bean
76  		public Servlet hostServlet() {
77  			return new Host();
78  		}
79  	}
80  
81  	@RegisterExtension
82  	static MicroserviceTestFixture fixture = MicroserviceTestFixture.create()
83  		.configurations(HostConfig.class);
84  
85  	private static final HttpClient HTTP = HttpClient.newBuilder()
86  		.connectTimeout(Duration.ofSeconds(5))
87  		.followRedirects(HttpClient.Redirect.NEVER)
88  		.build();
89  
90  	private static HttpResponse<String> get(String path) throws Exception {
91  		var req = HttpRequest.newBuilder()
92  			.uri(URI.create(fixture.getRootUrl() + path))
93  			.timeout(Duration.ofSeconds(10))
94  			.GET()
95  			.build();
96  		return HTTP.send(req, BodyHandlers.ofString());
97  	}
98  
99  	private static HttpResponse<String> head(String path) throws Exception {
100 		var req = HttpRequest.newBuilder()
101 			.uri(URI.create(fixture.getRootUrl() + path))
102 			.timeout(Duration.ofSeconds(10))
103 			.method("HEAD", HttpRequest.BodyPublishers.noBody())
104 			.build();
105 		return HTTP.send(req, BodyHandlers.ofString());
106 	}
107 
108 	@Test void a01_getStaticFileOverRealHttp() throws Exception {
109 		var resp = get("/static/javadoc.css");
110 		assertEquals(200, resp.statusCode());
111 		assertTrue(resp.body().contains("Licensed to the Apache Software Foundation"),
112 			"Body should contain the Apache license header");
113 		var contentType = resp.headers().firstValue("Content-Type").orElse("");
114 		assertTrue(contentType.startsWith("text/css"),
115 			"Content-Type should be text/css but was: " + contentType);
116 		var cacheControl = resp.headers().firstValue("Cache-Control").orElse("");
117 		assertEquals("max-age=86400, public", cacheControl,
118 			"Cache-Control should preserve the BasicStaticFiles default");
119 	}
120 
121 	@Test void a02_getHtdocsFileViaLegacyMount() throws Exception {
122 		// /htdocs/* is no longer a multi-path default on the mixin, but
123 		// BasicRestServlet still owns the legacy /htdocs/* via BasicRestOperations#getHtdoc.
124 		var resp = get("/htdocs/javadoc.css");
125 		assertEquals(200, resp.statusCode());
126 		assertTrue(resp.body().contains("Licensed to the Apache Software Foundation"),
127 			"Body should contain the Apache license header");
128 	}
129 
130 	@Test void a03_missingFileReturns404OverRealHttp() throws Exception {
131 		var resp = get("/static/does-not-exist.css");
132 		assertEquals(404, resp.statusCode());
133 	}
134 
135 	@Test void a04_headOverRealHttpReturnsHeadersWithEmptyBody() throws Exception {
136 		// HEAD: identical headers to GET, empty body. Compare against the prior GET result.
137 		var getResp = get("/static/javadoc.css");
138 		var headResp = head("/static/javadoc.css");
139 
140 		assertEquals(200, headResp.statusCode());
141 		assertEquals("", headResp.body(), "HEAD body must be empty");
142 
143 		var getCt = getResp.headers().firstValue("Content-Type").orElse("");
144 		var headCt = headResp.headers().firstValue("Content-Type").orElse("");
145 		assertEquals(getCt, headCt, "Content-Type must match GET");
146 
147 		var getCacheControl = getResp.headers().firstValue("Cache-Control").orElse("");
148 		var headCacheControl = headResp.headers().firstValue("Cache-Control").orElse("");
149 		assertEquals(getCacheControl, headCacheControl, "Cache-Control must match GET");
150 	}
151 }
