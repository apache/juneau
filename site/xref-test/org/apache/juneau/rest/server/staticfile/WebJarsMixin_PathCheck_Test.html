17  package org.apache.juneau.rest.server.staticfile;
18  
19  import static org.apache.juneau.bct.BctAssertions.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import org.apache.juneau.*;
23  import org.junit.jupiter.api.*;
24  
25  class WebJarsMixin_PathCheck_Test extends TestBase {
26  
27  	@Test void a01_canonicalPath_passes() {
28  		assertString("jquery/3.7.1/jquery.min.js", WebJarsMixin.checkedPath("jquery/3.7.1/jquery.min.js"));
29  	}
30  
31  	@Test void a02_badPaths_areRejected() {
32  		for (var raw : new String[] {
33  				null, "", "/jquery/x.js", "a/../../x", "%2e%2e/x", "a/%2e%2e/b", "a/./b.js", "./a.js",
34  				"a//b.js", "a/", "a%2fb.js", "a%2Fb.js", "a%5cb.js", "a\\b.js", "a/%2e/b.js", "a%252fb.js", "a/%zz.js" })
35  			assertNull(WebJarsMixin.checkedPath(raw), () -> "accepted: " + raw);
36  	}
37  
38  	@Test void a03_encodedButSafe_isDecoded() {
39  		assertString("a b/c+d.js", WebJarsMixin.checkedPath("a%20b/c+d.js"));
40  	}
41  
42  	@Test void b01_contentType_byExtension_ignoringCase() {
43  		assertString("text/javascript;charset=utf-8", WebJarsMixin.contentType("a/B.JS"));
44  		assertString("text/css;charset=utf-8", WebJarsMixin.contentType("a/b.css"));
45  		assertString("application/json", WebJarsMixin.contentType("a/b.min.js.map"));
46  		assertString("font/woff2", WebJarsMixin.contentType("a/f.woff2"));
47  		assertString("font/woff", WebJarsMixin.contentType("a/f.woff"));
48  		assertString("font/ttf", WebJarsMixin.contentType("a/f.ttf"));
49  		assertString("image/svg+xml", WebJarsMixin.contentType("a/i.svg"));
50  		assertString("image/png", WebJarsMixin.contentType("a/i.png"));
51  	}
52  
53  	@Test void b02_contentType_otherExtensions_areNull() {
54  		for (var p : new String[] { "a/README.md", "a/LICENSE", "a/b.txt", "a/b.class", "a/1.0", "a/b.properties" })
55  			assertNull(WebJarsMixin.contentType(p), () -> p);
56  	}
57  }
