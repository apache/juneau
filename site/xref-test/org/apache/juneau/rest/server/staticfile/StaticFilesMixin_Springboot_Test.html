38  /**
39   * Real-Spring-Boot deployment-parity assertion for {@link StaticFilesMixin}.
40   *
41   * <p>
42   * Boots a full Spring Boot context with embedded Tomcat on a random port, registers a
43   * {@link BasicSpringRestServlet}-based host with the static-files mixin via
44   * {@link ServletRegistrationBean}, and hits the {@code /static/*} and {@code /htdocs/*} URLs over
45   * real HTTP.
46   *
47   * <p>
48   * Catches things {@code MockRest} and the {@link StaticFilesMixin_JettyMicroservice_Test
49   * Jetty parity test} cannot:
50   * <ul>
51   * 	<li>Spring's bean store adapter ({@code SpringBeanStore}) resolving Juneau {@code StaticFiles}
52   * 		beans through {@link org.springframework.context.ApplicationContext#getBeanProvider(Class)
53   * 		ApplicationContext.getBeanProvider(...)} end-to-end.
54   * 	<li>Real embedded-Tomcat {@code Content-Type} negotiation for the {@code text/css} response on
55   * 		a CSS file served from a classpath JAR.
56   * 	<li>The {@code SpringRestServlet} {@code @Autowired ApplicationContext} field being populated
57   * 		by Spring before {@link jakarta.servlet.Servlet#init(jakarta.servlet.ServletConfig)
58   * 		Servlet.init()} runs &mdash; without which the mixin's {@code RestContext.getStaticFiles()}
59   * 		lookup would fail.
60   * </ul>
61   *
62   * @since 10.0.0
63   */
64  @org.apache.juneau.testing.SpringbootTest
65  @SpringBootTest(classes = StaticFilesMixin_Springboot_Test.TestApp.class,
66  	webEnvironment = WebEnvironment.RANDOM_PORT)
67  @DirtiesContext(classMode = DirtiesContext.ClassMode.AFTER_CLASS)
68  class StaticFilesMixin_Springboot_Test {
69  
70  	/**
71  	 * Minimal Spring Boot application that registers a single {@link Host} servlet at {@code /*}.
72  	 */
73  	@SpringBootConfiguration
74  	@EnableAutoConfiguration
75  	public static class TestApp {
76  
77  		@Bean
78  		public Host hostServlet() {
79  			return new Host();
80  		}
81  
82  		@Bean
83  		public ServletRegistrationBean<Host> hostRegistration(Host servlet) {
84  			return new ServletRegistrationBean<>(servlet, "/*");
85  		}
86  	}
87  
88  	/**
89  	 * Test host: subclass of {@link BasicSpringRestServlet} carrying the static-files mixin.
90  	 */
91  	@Rest(mixins = StaticFilesMixin.class)
92  	public static class Host extends BasicSpringRestServlet {
93  		private static final long serialVersionUID = 1L;
94  	}
95  
96  	@LocalServerPort
97  	int port;
98  
99  	private static final HttpClient HTTP = HttpClient.newBuilder()
100 		.connectTimeout(Duration.ofSeconds(5))
101 		.followRedirects(HttpClient.Redirect.NEVER)
102 		.build();
103 
104 	private HttpResponse<String> get(String path) throws Exception {
105 		var req = HttpRequest.newBuilder()
106 			.uri(URI.create("http://localhost:" + port + path))
107 			.timeout(Duration.ofSeconds(10))
108 			.GET()
109 			.build();
110 		return HTTP.send(req, BodyHandlers.ofString());
111 	}
112 
113 	private HttpResponse<String> head(String path) throws Exception {
114 		var req = HttpRequest.newBuilder()
115 			.uri(URI.create("http://localhost:" + port + path))
116 			.timeout(Duration.ofSeconds(10))
117 			.method("HEAD", HttpRequest.BodyPublishers.noBody())
118 			.build();
119 		return HTTP.send(req, BodyHandlers.ofString());
120 	}
121 
122 	@Test void a01_staticPathServesFileUnderSpringBoot() throws Exception {
123 		var resp = get("/static/javadoc.css");
124 		assertEquals(200, resp.statusCode());
125 		assertTrue(resp.body().contains("Licensed to the Apache Software Foundation"),
126 			"Body should contain the Apache license header");
127 		var ct = resp.headers().firstValue("Content-Type").orElse("");
128 		assertTrue(ct.startsWith("text/css"),
129 			"Content-Type should be text/css but was: " + ct);
130 	}
131 
132 	@Test void a02_htdocsPathServesFileViaLegacyMount() throws Exception {
133 		// /htdocs/* is no longer a multi-path default on the mixin, but
134 		// BasicSpringRestServlet still owns the legacy /htdocs/* via BasicRestOperations#getHtdoc.
135 		var resp = get("/htdocs/javadoc.css");
136 		assertEquals(200, resp.statusCode());
137 		assertTrue(resp.body().contains("Licensed to the Apache Software Foundation"),
138 			"Body should contain the Apache license header");
139 	}
140 
141 	@Test void a03_missingFileReturns404UnderSpringBoot() throws Exception {
142 		var resp = get("/static/does-not-exist.css");
143 		assertEquals(404, resp.statusCode());
144 	}
145 
146 	@Test void a04_headProbeUnderSpringBoot() throws Exception {
147 		var resp = head("/static/javadoc.css");
148 		assertEquals(200, resp.statusCode());
149 		assertEquals("", resp.body(), "HEAD body must be empty");
150 	}
151 
152 	@Test void a05_cacheControlPreservedUnderSpringBoot() throws Exception {
153 		var resp = get("/static/javadoc.css");
154 		assertEquals(200, resp.statusCode());
155 		assertEquals("max-age=86400, public",
156 			resp.headers().firstValue("Cache-Control").orElse(""));
157 	}
158 }
