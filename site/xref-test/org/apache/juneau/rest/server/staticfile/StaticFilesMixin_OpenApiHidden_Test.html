17  package org.apache.juneau.rest.server.staticfile;
18  
19  import org.apache.juneau.*;
20  import org.apache.juneau.rest.mock.classic.*;
21  import org.apache.juneau.rest.server.*;
22  import org.apache.juneau.rest.server.docs.*;
23  import org.apache.juneau.rest.server.servlet.*;
24  import org.apache.juneau.rest.server.swagger.*;
25  import org.junit.jupiter.api.*;
26  
27  /**
28   * Validates that the static-files mixin's greedy {@code /static/*} handler is excluded from the
29   * generated Swagger/OpenAPI spec, per {@link OpSwagger#ignore() @OpSwagger(ignore=true)} on
30   * {@link StaticFilesMixin#getStaticFile} and the matching HEAD handler.
31   *
32   * <p>
33   * The host below mounts both {@link StaticFilesMixin} (static-file serving) and
34   * {@link OpenApiMixin} (OpenAPI generator) as mixins on a vanilla {@link RestServlet}
35   * (NOT {@link BasicRestServlet}) so the legacy
36   * {@code BasicRestOperations.getHtdoc(...)} method does not pollute the spec independently
37   * of what the mixin emits. The generated spec must list the host's own {@code /items} endpoint
38   * but NOT the static-file mounts contributed by the mixin.
39   *
40   * @since 10.0.0
41   */
42  class StaticFilesMixin_OpenApiHidden_Test extends TestBase {
43  
44  	/**
45  	 * Host extends vanilla {@link RestServlet} (NOT {@link BasicRestServlet}) so the legacy
46  	 * {@code getHtdoc(...)} from {@code BasicRestOperations} does not pollute the spec
47  	 * independently of the mixin. The spec generator needs a {@code SwaggerProvider} bean to
48  	 * generate; we wire it explicitly via {@link Rest#swaggerProvider}.
49  	 */
50  	@Rest(
51  		mixins={StaticFilesMixin.class, OpenApiMixin.class},
52  		swaggerProvider=BasicSwaggerProvider.class
53  	)
54  	public static class A extends RestServlet {
55  		private static final long serialVersionUID = 1L;
56  		@RestGet(path="/items") public String items() { return "items"; }
57  	}
58  
59  	private static final MockRestClient c = MockRestClient.buildLax(A.class);
60  
61  	@Test void a01_openapiSpecExcludesStaticPaths() throws Exception {
62  		// /openapi.json is format-pinned to JSON regardless of Accept (per OpenApiMixin); // NOSONAR
63  		// avoids the need to register JSON serializers on the vanilla RestServlet host.
64  		var spec = c.get("/openapi.json")
65  			.run()
66  			.assertStatus(200)
67  			.getContent().asString();
68  
69  		// Sanity: host's own endpoint must be present.
70  		assertContains(spec, "/items");
71  		// Static-file mounts must NOT be present.
72  		assertNotContains(spec, "/static");
73  		assertNotContains(spec, "/htdocs");
74  	}
75  
76  	@Test void a02_staticFilesStillServedDespiteHiddenFromSpec() throws Exception {
77  		c.get("/static/javadoc.css")
78  			.run()
79  			.assertStatus(200);
80  		// /htdocs/* is no longer a multi-path default; migration covered by
81  		// StaticFilesMixin_SvlPathOverride_Test#a02.
82  		c.get("/htdocs/javadoc.css")
83  			.run()
84  			.assertStatus(404);
85  	}
86  
87  	/**
88  	 * Host that exercises the LEGACY static-file path: a {@link BasicRestServlet} subclass with
89  	 * the api-docs mixin pack inherited and no explicit {@link StaticFilesMixin}. The
90  	 * legacy {@code getHtdoc(...)} method on {@code BasicRestOperations} now carries
91  	 * {@code @OpSwagger(ignore=true)} so {@code /htdocs/*} must NOT appear in the generated spec
92  	 * even when the new mixin is not mounted.
93  	 */
94  	@Rest
95  	public static class B extends BasicRestServlet {
96  		private static final long serialVersionUID = 1L;
97  		@RestGet(path="/items") public String items() { return "items"; }
98  	}
99  
100 	private static final MockRestClient cb = MockRestClient.buildLax(B.class);
101 
102 	@Test void b01_legacyGetHtdocAlsoHiddenFromSpec() throws Exception {
103 		var spec = cb.get("/openapi.json")
104 			.run()
105 			.assertStatus(200)
106 			.getContent().asString();
107 
108 		// Sanity: host's own endpoint must be present.
109 		assertContains(spec, "/items");
110 		// Legacy /htdocs/* must NOT be present after the @OpSwagger(ignore=true) cleanup.
111 		assertNotContains(spec, "/htdocs");
112 	}
113 
114 	@Test void b02_legacyHtdocStillServedDespiteHiddenFromSpec() throws Exception {
115 		// Legacy getHtdoc(...) handler still serves the file even though it's hidden from spec.
116 		cb.get("/htdocs/javadoc.css")
117 			.run()
118 			.assertStatus(200);
119 	}
120 
121 	private static void assertContains(String s, String needle) {
122 		if (!s.contains(needle))
123 			throw new AssertionError("Expected to contain '" + needle + "' but did not. Body: " + s);
124 	}
125 
126 	private static void assertNotContains(String s, String needle) {
127 		if (s.contains(needle))
128 			throw new AssertionError("Expected NOT to contain '" + needle + "' but did. Body excerpt: "
129 				+ s.substring(Math.max(0, s.indexOf(needle) - 50), Math.min(s.length(), s.indexOf(needle) + 100)));
130 	}
131 }
