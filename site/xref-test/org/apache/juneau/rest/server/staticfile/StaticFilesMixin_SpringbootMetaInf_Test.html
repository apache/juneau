39  /**
40   * Spring Boot {@code META-INF/resources/} bridge test for {@link StaticFilesMixin}.
41   *
42   * <p>
43   * Spring Boot's embedded Tomcat / Jetty auto-serves files placed under the conventional
44   * {@code META-INF/resources/} classpath location at the application root (so
45   * {@code META-INF/resources/foo.txt} is served at {@code http://host/foo.txt}). A Juneau service
46   * deployed under Spring Boot may want to expose the same resources through the mixin's
47   * {@code /static/*} or {@code /htdocs/*} mounts &mdash; e.g. to apply the mixin's
48   * {@code Cache-Control} headers, or to share asset paths between Spring's and Juneau's
49   * static-file handlers.
50   *
51   * <p>
52   * This test pins the bridge: the importer registers a custom {@code @Bean StaticFiles} that
53   * adds a {@code cp(Host.class, "/META-INF/resources", true)} classpath search root, and the
54   * mixin then serves {@code GET /static/spring-fixture.txt} returning the content of
55   * {@code juneau-integration-tests/src/test/resources/META-INF/resources/spring-fixture.txt}.
56   *
57   * <p>
58   * Companion to {@link StaticFilesMixin_Springboot_Test} which exercises the default
59   * classpath ({@code static/} + {@code htdocs/}) under the same Spring Boot wiring.
60   *
61   * @since 10.0.0
62   */
63  @org.apache.juneau.testing.SpringbootTest
64  @SpringBootTest(classes = StaticFilesMixin_SpringbootMetaInf_Test.TestApp.class,
65  	webEnvironment = WebEnvironment.RANDOM_PORT)
66  @DirtiesContext(classMode = DirtiesContext.ClassMode.AFTER_CLASS)
67  class StaticFilesMixin_SpringbootMetaInf_Test {
68  
69  	@SpringBootConfiguration
70  	@EnableAutoConfiguration
71  	public static class TestApp {
72  
73  		@Bean
74  		public Host hostServlet() {
75  			return new Host();
76  		}
77  
78  		@Bean
79  		public ServletRegistrationBean<Host> hostRegistration(Host servlet) {
80  			return new ServletRegistrationBean<>(servlet, "/*");
81  		}
82  	}
83  
84  	/**
85  	 * Test host: subclass of {@link BasicSpringRestServlet} carrying the static-files mixin AND
86  	 * an importer-supplied {@code StaticFiles} factory that adds {@code /META-INF/resources/} to
87  	 * the classpath search list.
88  	 */
89  	@Rest(mixins = StaticFilesMixin.class)
90  	public static class Host extends BasicSpringRestServlet {
91  		private static final long serialVersionUID = 1L;
92  
93  		@org.apache.juneau.commons.inject.Bean
94  		public StaticFiles staticFiles(BeanStore bs) {
95  			return BasicStaticFiles
96  				.create(bs)
97  				.cp(Host.class, "/META-INF/resources", true)
98  				.build();
99  		}
100 	}
101 
102 	@LocalServerPort
103 	int port;
104 
105 	private static final HttpClient HTTP = HttpClient.newBuilder()
106 		.connectTimeout(Duration.ofSeconds(5))
107 		.followRedirects(HttpClient.Redirect.NEVER)
108 		.build();
109 
110 	private HttpResponse<String> get(String path) throws Exception {
111 		var req = HttpRequest.newBuilder()
112 			.uri(URI.create("http://localhost:" + port + path))
113 			.timeout(Duration.ofSeconds(10))
114 			.GET()
115 			.build();
116 		return HTTP.send(req, BodyHandlers.ofString());
117 	}
118 
119 	@Test void a01_metaInfResourcesServedViaMixinStaticMount() throws Exception {
120 		var resp = get("/static/spring-fixture.txt");
121 		assertEquals(200, resp.statusCode());
122 		assertTrue(resp.body().contains("spring-boot meta-inf fixture"),
123 			"Body should contain the META-INF/resources fixture content, was: " + resp.body());
124 	}
125 
126 	@Test void a02_metaInfResourcesServedViaLegacyHtdocsMount() throws Exception {
127 		// /htdocs/* is no longer a multi-path default on the mixin, but
128 		// BasicSpringRestServlet still owns the legacy /htdocs/* via BasicRestOperations#getHtdoc.
129 		// The same META-INF/resources file is reachable through the legacy mount.
130 		var resp = get("/htdocs/spring-fixture.txt");
131 		assertEquals(200, resp.statusCode());
132 		assertTrue(resp.body().contains("spring-boot meta-inf fixture"),
133 			"Body should contain the META-INF/resources fixture content, was: " + resp.body());
134 	}
135 
136 	@Test void a03_missingMetaInfResourceReturns404() throws Exception {
137 		// Files not present in META-INF/resources/ still 404 cleanly even with the extra search root.
138 		var resp = get("/static/no-such-meta-inf-file.txt");
139 		assertEquals(404, resp.statusCode());
140 	}
141 }
