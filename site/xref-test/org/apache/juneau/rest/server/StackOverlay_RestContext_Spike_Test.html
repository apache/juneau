25  /**
26   * Validates that wedging a {@link StackOverlay} into the
27   * {@code overridingParent} slot of a {@link BasicBeanStore} preserves the existing
28   * resolution semantics that {@code RestContext_Precedence_Test} exercises.
29   *
30   * <p>
31   * The framework {@link RestContext} integration of {@link StackOverlay} happens in later
32   * phases.  This spike exercises the composition shape directly: a {@link BasicBeanStore}
33   * whose {@code overridingParent} is a {@link StackOverlay} should behave identically to
34   * one whose {@code overridingParent} is a single {@link BasicBeanStore} (as in the existing
35   * Spring bridge), while supporting push/pop layering on top.
36   *
37   * <p>
38   * <b>Go/no-go gate:</b> if any of the assertions below fail, the composition path is
39   * non-trivial and the design should fall back to OQ4 option (a) — replace the slot with
40   * a stack — per the plan's Phase 0 fallback gate.
41   */
42  class StackOverlay_RestContext_Spike_Test extends TestBase {
43  
44  	//-----------------------------------------------------------------------------------------------------------------
45  	// Marker beans
46  	//-----------------------------------------------------------------------------------------------------------------
47  
48  	private static final class Svc {
49  		final String tag;
50  		Svc(String tag) { this.tag = tag; }
51  		@Override public String toString() { return "Svc[" + tag + "]"; }
52  	}
53  
54  	//-----------------------------------------------------------------------------------------------------------------
55  	// a — empty StackOverlay in overridingParent slot behaves like null overriding parent
56  	//-----------------------------------------------------------------------------------------------------------------
57  
58  	@Test
59  	void a01_emptyOverlay_doesNotShadowLocalEntries() {
60  		var local = new Svc("local");
61  		var overlay = new StackOverlay();
62  		var store = new BasicBeanStore(null, overlay).addBean(Svc.class, local);
63  
64  		// With an empty stack overlay, lookup falls through to local entries — the same behavior
65  		// as if overridingParent were null.
66  		assertSame(local, store.getBean(Svc.class).get());
67  	}
68  
69  	@Test
70  	void a02_emptyOverlay_fallsThroughToParent() {
71  		var fromParent = new Svc("from-parent");
72  		var parent = new BasicBeanStore().addBean(Svc.class, fromParent);
73  		var overlay = new StackOverlay();
74  		var store = new BasicBeanStore(parent, overlay);
75  
76  		// Empty stack overlay should fall through to local entries (none here) and then to parent.
77  		assertSame(fromParent, store.getBean(Svc.class).get());
78  	}
79  
80  	//-----------------------------------------------------------------------------------------------------------------
81  	// b — pushed frame wins over local entries (the whole point of overridingParent precedence)
82  	//-----------------------------------------------------------------------------------------------------------------
83  
84  	@Test
85  	void b01_pushedFrame_winsOverLocalEntry() {
86  		var localBean = new Svc("local");
87  		var overrideBean = new Svc("override");
88  
89  		var overlay = new StackOverlay();
90  		var store = new BasicBeanStore(null, overlay).addBean(Svc.class, localBean);
91  
92  		overlay.push(new BasicBeanStore().addBean(Svc.class, overrideBean));
93  		assertSame(overrideBean, store.getBean(Svc.class).get(),
94  			"Pushed override frame must win over the resource's local entry");
95  
96  		overlay.pop();
97  		assertSame(localBean, store.getBean(Svc.class).get(),
98  			"After popping, local entry should be visible again");
99  	}
100 
101 	@Test
102 	void b02_pushedFrame_winsOverParentToo() {
103 		var parentBean = new Svc("from-parent");
104 		var overrideBean = new Svc("override");
105 
106 		var parent = new BasicBeanStore().addBean(Svc.class, parentBean);
107 		var overlay = new StackOverlay();
108 		var store = new BasicBeanStore(parent, overlay);
109 
110 		overlay.push(new BasicBeanStore().addBean(Svc.class, overrideBean));
111 		assertSame(overrideBean, store.getBean(Svc.class).get());
112 
113 		overlay.pop();
114 		assertSame(parentBean, store.getBean(Svc.class).get());
115 	}
116 
117 	@Test
118 	void b03_pushedFrame_winsOverDefaultSupplier() {
119 		var defaultBean = new Svc("default");
120 		var overrideBean = new Svc("override");
121 
122 		var overlay = new StackOverlay();
123 		var store = new BasicBeanStore(null, overlay).addDefaultSupplier(Svc.class, () -> defaultBean);
124 
125 		overlay.push(new BasicBeanStore().addBean(Svc.class, overrideBean));
126 		assertSame(overrideBean, store.getBean(Svc.class).get());
127 
128 		overlay.pop();
129 		assertSame(defaultBean, store.getBean(Svc.class).get(), "Default supplier resolves after popping");
130 	}

134 	//-----------------------------------------------------------------------------------------------------------------
135 
136 	@Test
137 	void c01_topFrame_winsOverBottomFrame_inOverridingParentSlot() {
138 		var bottom = new Svc("bottom");
139 		var top = new Svc("top");
140 		var local = new Svc("local");
141 
142 		var overlay = new StackOverlay();
143 		var store = new BasicBeanStore(null, overlay).addBean(Svc.class, local);
144 
145 		overlay.push(new BasicBeanStore().addBean(Svc.class, bottom));
146 		overlay.push(new BasicBeanStore().addBean(Svc.class, top));
147 
148 		assertSame(top, store.getBean(Svc.class).get());
149 
150 		overlay.pop();
151 		assertSame(bottom, store.getBean(Svc.class).get());
152 
153 		overlay.pop();
154 		assertSame(local, store.getBean(Svc.class).get());
155 	}

157 	//-----------------------------------------------------------------------------------------------------------------
158 	// d — toString lifecycle / cycle safety on the production-shaped composition
159 	//-----------------------------------------------------------------------------------------------------------------
160 
161 	@SuppressWarnings({
162 		"java:S125"  // Explanatory prose describing the fixture shape; not dead code.
163 	})
164 	@Test
165 	void d01_toString_doesNotStackOverflow_onProductionShapedComposition() {
166 		// Production shape: outer BasicBeanStore holds StackOverlay in its overridingParent slot;
167 		// pushed frames may themselves be BasicBeanStores whose parent is the outer store
168 		// (e.g. when the Mode OVERLAY test fixture chains a TestBeanStore onto the live SUT's store).
169 		var overlay = new StackOverlay();
170 		var outer = new BasicBeanStore(null, overlay);
171 		var frame = new BasicBeanStore(outer);
172 		overlay.push(frame);
173 
174 		assertDoesNotThrow(outer::toString, "BasicBeanStore.toString must not stack-overflow on cyclic StackOverlay composition");
175 		assertDoesNotThrow(overlay::toString, "StackOverlay.toString must not stack-overflow on cyclic frame chain");
176 	}

178 	//-----------------------------------------------------------------------------------------------------------------
179 	// e — getBeansOfType across all four resolution tiers
180 	//-----------------------------------------------------------------------------------------------------------------
181 
182 	@Test
183 	void e01_getBeansOfType_includesAllTiers_overlayWins() {
184 		var overlay = new StackOverlay();
185 		var local = new Svc("local");
186 		var fromOverlay = new Svc("from-overlay");
187 		var fromDefault = new Svc("from-default");
188 
189 		var store = new BasicBeanStore(null, overlay)
190 			.addBean(Svc.class, local, "local-only")
191 			.addBean(Svc.class, new Svc("local-shadowed"), "shared")
192 			.addDefaultSupplier(Svc.class, () -> fromDefault, "default-only");
193 		overlay.push(new BasicBeanStore().addBean(Svc.class, fromOverlay, "shared"));
194 
195 		var all = store.getBeansOfType(Svc.class);
196 		assertSame(local, all.get("local-only"));
197 		assertSame(fromOverlay, all.get("shared"), "Overlay frame should shadow local entry on shared name");
198 		assertSame(fromDefault, all.get("default-only"));
199 	}
