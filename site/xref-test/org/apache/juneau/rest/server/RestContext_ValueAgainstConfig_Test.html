35  /**
36   * Acceptance tests for per-{@code RestContext} {@code @Value} resolution against
37   * {@code @Rest(config=...)} {@link org.apache.juneau.marshall.config.Config}s.
38   *
39   * <p>
40   * The test writes its config fixtures to the cwd in {@link BeforeAll @BeforeAll} (because the
41   * default {@link org.apache.juneau.marshall.config.store.FileStore#DEFAULT FileStore.DEFAULT} resolves
42   * names against the cwd; with Maven Surefire that is the module directory). The cwd-resident
43   * files are removed in {@link AfterAll @AfterAll}. Names use a fixed {@code valcfg-} prefix so
44   * they are easy to spot and clean up by hand should a test crash partway through.
45   *
46   * <p>
47   * Each resource class uses a unique name so {@code MockRestClient}'s static {@code RestContext}
48   * cache does not return a stale {@code RestContext} from an unrelated test class.
49   */
50  @SuppressWarnings({
51  	"serial" // Test resources extend BasicRestServlet which is serializable; not relevant.
52  })
53  class RestContext_ValueAgainstConfig_Test extends TestBase {

59  	private static final String CFG_A = "valcfg-rcvac-a.cfg";
60  	private static final String CFG_B = "valcfg-rcvac-b.cfg";
61  	private static final String CFG_PARENT = "valcfg-rcvac-parent.cfg";
62  	private static final String CFG_CHILD = "valcfg-rcvac-child.cfg";
63  	private static final String CFG_OVERRIDE = "valcfg-rcvac-override.cfg";
64  	private static final String CFG_ASYNC = "valcfg-rcvac-async.cfg";

66  	@BeforeAll
67  	static void writeFixtures() throws IOException {
68  		write(CFG_A,
69  			"api.key = secret-A",
70  			"api.url = https://a.example.org/",
71  			"foo = A",
72  			"[section]",
73  			"nested = nested-A");
74  		write(CFG_B,
75  			"api.key = secret-B",
76  			"foo = B");
77  		write(CFG_PARENT,
78  			"parent.only = parent-value",
79  			"shared = parent-shared");
80  		write(CFG_CHILD,
81  			"child.only = child-value",
82  			"shared = child-shared");
83  		write(CFG_OVERRIDE,
84  			"override.key = from-config");
85  		write(CFG_ASYNC,
86  			"async.greeting = hello-from-config");
87  	}
88  
89  	@AfterAll
90  	static void removeFixtures() throws IOException {
91  		for (var n : List.of(CFG_A, CFG_B, CFG_PARENT, CFG_CHILD, CFG_OVERRIDE, CFG_ASYNC))
92  			Files.deleteIfExists(Path.of(n));
93  	}

95  	private static void write(String name, String... lines) throws IOException {
96  		Files.writeString(Path.of(name), String.join(System.lineSeparator(), lines) + System.lineSeparator(),
97  			StandardCharsets.UTF_8);
98  	}

202 	// collision, parent fills the gaps.
203 	//------------------------------------------------------------------------------------------------------------------
204 
205 	@Rest(config=CFG_PARENT)
206 	public static class InheritanceParent extends BasicRestServlet {
207 		@Value("${shared}")
208 		String shared;
209 
210 		@Value("${parent.only}")
211 		String parentOnly;
212 
213 		@Value("${child.only:absent-on-parent}")
214 		String childOnly;
215 	}
216 
217 	@Rest(config=CFG_CHILD)
218 	public static class InheritanceChild extends InheritanceParent {}
219 
220 	@Test void a05_inheritance_child_wins_parent_fills_gaps() {
221 		var rc = build(InheritanceChild.class);
222 		var bean = (InheritanceChild) rc.getResource();
223 		// Child wins on collision: shared resolves to child's value.
224 		assertEquals("child-shared", bean.shared);
225 		// Parent fills the gap: parent.only is only in parent.cfg, child sees it.
226 		assertEquals("parent-value", bean.parentOnly);
227 		// Child has its own key:
228 		assertEquals("child-value", bean.childOnly);
229 	}
230 
231 	@Test void a05b_parent_alone_resolves_parent_keys() {
232 		var rc = build(InheritanceParent.class);
233 		var bean = (InheritanceParent) rc.getResource();
234 		assertEquals("parent-shared", bean.shared);
235 		assertEquals("parent-value", bean.parentOnly);
236 		// child.only is absent from parent.cfg → falls through to the default branch.
237 		assertEquals("absent-on-parent", bean.childOnly);
238 	}

241 	// Acceptance #6 — Settings.setLocal()/setGlobal() overrides still win over resource Config.
242 	//------------------------------------------------------------------------------------------------------------------
243 
244 	// MockRestClient caches RestContext per class, so a @Value String field is captured ONCE at
245 	// resource-bean construction time. To exercise the "override wins later" path we use a
246 	// Supplier<String> @Value — it re-evaluates on each .get() call against a fresh
247 	// VarResolverSession that carries the resource's BeanStore as a session bean.
248 	@Rest(config=CFG_OVERRIDE)
249 	public static class OverrideResource extends BasicRestServlet {
250 		@Value("${override.key}")
251 		Supplier<String> value;
252 	}
253 
254 	@Test void a06_setGlobal_overrides_resource_config() {
255 		var rc = build(OverrideResource.class);
256 		var bean = (OverrideResource) rc.getResource();
257 		// No override active → resource Config value wins.
258 		assertEquals("from-config", bean.value.get());
259 
260 		Settings.get().setGlobal("override.key", "from-settings-global");
261 		try {
262 			// Override active → setGlobal must beat resource Config.
263 			assertEquals("from-settings-global", bean.value.get(),
264 				"Settings.setGlobal must win over resource @Rest(config) values.");
265 		} finally {
266 			Settings.get().unsetGlobal("override.key");
267 		}
268 		// Override gone → resource Config value visible again (Supplier re-evaluates).
269 		assertEquals("from-config", bean.value.get());
270 	}
271 
272 	@Test void a06b_setLocal_overrides_resource_config() {
273 		var rc = build(OverrideResource.class);
274 		var bean = (OverrideResource) rc.getResource();
275 
276 		Settings.get().setLocal("override.key", "from-settings-local");
277 		try {
278 			assertEquals("from-settings-local", bean.value.get(),
279 				"Settings.setLocal must win over resource @Rest(config) values.");
280 		} finally {
281 			Settings.get().unsetLocal("override.key");
282 		}
283 		assertEquals("from-config", bean.value.get());
284 	}

287 	// Acceptance #7 — Async / virtual-thread composition. CompletableFuture<String> @RestOp method
288 	// body reads a @Value field; assertion holds whether the future resolves on the dispatch thread
289 	// or a worker thread.
290 	//------------------------------------------------------------------------------------------------------------------
291 
292 	@Rest(config=CFG_ASYNC)
293 	public static class AsyncResource extends BasicRestServlet {
294 		@Value("${async.greeting}")
295 		String greeting;
296 
297 		@Value("${async.greeting}")
298 		Supplier<String> greetingSupplier;
299 	}
300 
301 	@Test void a07_async_composition_value_visible() throws Exception {
302 		var rc = build(AsyncResource.class);
303 		var bean = (AsyncResource) rc.getResource();
304 		// The field value is captured at injection time — verify directly on the bean.
305 		assertEquals("hello-from-config", bean.greeting);
306 		// The Supplier<String> re-evaluates on every .get() call.
307 		assertEquals("hello-from-config", bean.greetingSupplier.get());
308 
309 		// Re-evaluation from a worker thread (CompletableFuture) must see the same value because
310 		// the BeanStore reference captured by the Supplier participates regardless of which thread
311 		// calls .get() — i.e., the resolution is not pinned to the dispatch thread.
312 		var worker = CompletableFuture.supplyAsync(bean.greetingSupplier::get).get(5, TimeUnit.SECONDS);
313 		assertEquals("hello-from-config", worker);
314 	}
315 
316 	@Rest(config=CFG_ASYNC, virtualThreads="true")
317 	public static class AsyncVirtualThreadsResource extends BasicRestServlet {
318 		@Value("${async.greeting}")
319 		String greeting;
320 
321 		@Value("${async.greeting}")
322 		Supplier<String> greetingSupplier;
323 	}
324 
325 	@Test void a07b_virtualThreads_value_visible() throws Exception {
326 		var rc = build(AsyncVirtualThreadsResource.class);
327 		var bean = (AsyncVirtualThreadsResource) rc.getResource();
328 		// Field injection captured at construction time (the BeanStore had the rest.config source
329 		// in place by then, so this verifies the construction-time path holds when the resource is
330 		// configured with virtualThreads="true").
331 		assertEquals("hello-from-config", bean.greeting);
332 		// Re-evaluation from a worker thread continues to see the value (BeanStore-by-reference
333 		// capture in ValueResolver's supplier path, not pinned to the dispatch thread).
334 		var worker = CompletableFuture.supplyAsync(bean.greetingSupplier::get).get(5, TimeUnit.SECONDS);
335 		assertEquals("hello-from-config", worker);
336 	}

349 	@Test void a08b_no_config_no_registration() {
350 		// A @Rest resource with no config= attribute and no @Bean Config method should leave the
351 		// "rest.config" slot empty (rawConfig.get() returns an empty Config that the bridge skips).
352 		var rc = build(NoConfigResource.class);
353 		var src = rc.getBeanStore().getBean(PropertySource.class, "rest.config").orElse(null);
354 		// An empty Config still has the bridge registered (since rawConfig.get() is non-null), but
355 		// querying any unknown key yields missing. Either no bean registered OR a bean that returns
356 		// missing for an arbitrary key is acceptable; assert the latter.
357 		if (src != null)
358 			assertFalse(src.get("definitely-not-here").isPresent());
359 	}

362 	// Helper — build a RestContext and look it up in the global registry so the test can read
363 	// back the resource bean and the BeanStore directly.
364 	//------------------------------------------------------------------------------------------------------------------
365 
366 	private static RestContext build(Class<?> resourceClass) {
367 		// MockRestClient.build() instantiates a RestContext for the resource class (and caches it
368 		// statically); look the result up in the global registry to read the injected fields.
369 		MockRestClient.build(resourceClass);
370 		var rc = RestContext.getGlobalRegistry().get(resourceClass);
371 		assertNotNull(rc, "RestContext for " + resourceClass.getSimpleName() + " not in REGISTRY after build.");
372 		return rc;
373 	}
