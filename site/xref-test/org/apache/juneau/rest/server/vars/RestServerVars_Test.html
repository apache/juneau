39  /**
40   * Branch-coverage unit tests for the SVL variable classes in {@code org.apache.juneau.rest.server.vars}.
41   *
42   * <p>
43   * Each variable class exposes a {@code canResolve(VarResolverSession)} method whose two branches
44   * (bean present / bean absent) can be exercised without a live HTTP request by supplying or
45   * withholding the required bean(s) in a {@link BasicBeanStore}-backed {@link VarResolverSession}.
46   *
47   * @since 10.0.0
48   */
49  @SuppressWarnings("resource") // 'store' (BasicBeanStore/WritableBeanStore) instances are short-lived in-memory test fixtures backed by a Map; nothing external to leak.
50  class RestServerVars_Test extends TestBase {
51  
52  	// -----------------------------------------------------------------------------------------
53  	// Helper: session factory
54  	// -----------------------------------------------------------------------------------------
55  
56  	private static VarResolverSession sessionWith(Object... beans) {
57  		var store = new BasicBeanStore();
58  		for (var b : beans)
59  			addToStore(store, b);
60  		return VarResolver.DEFAULT.createSession(store);
61  	}
62  
63  	@SuppressWarnings({
64  		"unchecked" // Generic cast is safe: b.getClass() matches b's runtime type
65  	})
66  	private static <T> void addToStore(WritableBeanStore store, T b) {
67  		store.addBean((Class<T>) b.getClass(), b);
68  	}
69  
70  	private static VarResolverSession emptySession() {
71  		return VarResolver.DEFAULT.createSession(new BasicBeanStore());
72  	}

74  	// -----------------------------------------------------------------------------------------
75  	// Helper: deep-stub RestRequest with pre-seeded stubs for tricky chains
76  	//
77  	// Uses RETURNS_DEEP_STUBS so arbitrary method chains produce mocks rather than null.
78  	// Two extra stubs prevent NPE in chains Mockito cannot handle automatically:
79  	//   • req.getContext() → ctx (RETURNS_DEEP_STUBS) with getResourceClass() → String.class
80  	//   • req.getAttributes().get(any()) → attr with asString() → Optional.empty()
81  	// -----------------------------------------------------------------------------------------
82  
83  	private static RestRequest deepStubReq() {
84  		var req = mock(RestRequest.class, RETURNS_DEEP_STUBS);
85  		var ctx = mock(RestContext.class, RETURNS_DEEP_STUBS);
86  		doReturn(String.class).when(ctx).getResourceClass();
87  		when(req.getContext()).thenReturn(ctx);
88  		var attr = mock(RequestAttribute.class);
89  		when(attr.asString()).thenReturn(oe());
90  		when(req.getAttributes().get(any())).thenReturn(attr);
91  		return req;
92  	}
