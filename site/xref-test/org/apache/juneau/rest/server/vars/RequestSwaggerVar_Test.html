376 	// f - Nested-variable handling: allowNested()=false, allowRecurse()=false. Putting another var
377 	//     inside a $RS{...} body causes the resolver to NOT recursively resolve it (returned literal).
378 	//-----------------------------------------------------------------------------------------------------------------
379 
380 	@Test void f01_allowNested_isFalse_innerVarLeftLiteral() throws Exception {
381 		// Inner $RQ{x} would normally resolve to "title", but allowNested()==false means the body
382 		// is passed verbatim as the key.  "$RQ{x}" is not a recognized key -> null -> "".
383 		clientA.get("/raw").queryData("expr", "$RS{$RQ{x}}").queryData("x", "title").run()
384 			.assertStatus(200)
385 			.assertContent("");
386 	}
387 
388 	@Test void f02_allowRecurse_isFalse_returnedValueNotReResolved() throws Exception {
389 		// Use a class whose swagger title contains a "$" — when this value resolves the resolver
390 		// will check allowRecurse() to decide whether to re-resolve. allowRecurse()==false so the
391 		// raw value is returned. This test mainly forces the allowRecurse() override to execute.
392 		var client = MockRestClient.build(F.class);
393 		client.get("/").run()
394 			.assertStatus(200)
395 			.assertContent().asString().isContains("dollar-sign-here");
396 	}
