21  import org.apache.juneau.*;

27  /**
28   * Validates {@link NavigationMixin} mounted as a mixin via {@code @Rest(mixins=...)} on a vanilla
29   * {@link RestServlet} host.
30   *
31   * <p>
32   * The navigation op lives on the mixin's (empty) sub-context but must enumerate the HOST's child resources
33   * via {@link RestMixin#getHostContext()}. These tests confirm the {@code GET /} navigation page lists the
34   * host's children rather than the mixin's own (empty) children.
35   *
36   * @since 10.0.0
37   */
38  class NavigationMixin_AsMixin_Test extends TestBase {
39  
40  	@Rest(path="/c1", title="Child One") public static class Child1 extends BasicRestServlet { private static final long serialVersionUID = 1L; @RestGet(path="/") public String x() { return "c1"; } }
41  	@Rest(path="/c2", title="Child Two") public static class Child2 extends BasicRestServlet { private static final long serialVersionUID = 1L; @RestGet(path="/") public String x() { return "c2"; } }
42  
43  	// Host extends BasicRestServlet (for serializers) and composes NavigationMixin directly, rather than
44  	// inheriting it from a Basic*Group base, to prove the mixin enumerates the HOST's children.
45  	@Rest(mixins=NavigationMixin.class, children={Child1.class, Child2.class})
46  	public static class A extends BasicRestServlet {
47  		private static final long serialVersionUID = 1L;
48  	}
49  
50  	private static final MockRestClient a = MockRestClient.buildLax(A.class);
51  
52  	@Test void a01_navigationListsHostChildren() throws Exception {
53  		var body = a.get("/").accept("application/json").run().assertStatus(200).getContent().asString();
54  		assertTrue(body.contains("c1"), "Navigation page must list host child 'c1'; got: " + body);
55  		assertTrue(body.contains("c2"), "Navigation page must list host child 'c2'; got: " + body);
56  	}
57  
58  	@Test void a02_navigationNotEmpty() throws Exception {
59  		var body = a.get("/").accept("application/json").run().assertStatus(200).getContent().asString();
60  		assertNotEquals("[]", body.trim(), "Navigation page must not enumerate the mixin's own (empty) children.");
61  	}
62  }
