17  package org.apache.juneau.rest.server.ops;
18  
19  import org.apache.juneau.*;
20  import org.apache.juneau.commons.inject.*;
21  import org.apache.juneau.rest.mock.classic.*;
22  import org.apache.juneau.rest.server.*;
23  import org.apache.juneau.rest.server.guard.*;
24  import org.apache.juneau.rest.server.servlet.*;
25  import org.junit.jupiter.api.*;
26  
27  /**
28   * Validates the {@link AdminResource} child-resource flavor mounted via {@code @Rest(children=...)}
29   * under a host.
30   *
31   * <p>
32   * The child mounts at the subtree {@code /admin} (ops at the bare endpoint names) and delegates to a
33   * shared {@link AdminMixin} worker. Cases:
34   * <ul>
35   * 	<li>Default-deny: the inherited {@link org.apache.juneau.rest.server.server.guard.DenyAllGuard DenyAllGuard}
36   * 		returns {@code 403 Forbidden} on {@code /admin/threads}.
37   * 	<li>The host's own endpoints are unaffected by the mounted child.
38   * 	<li>With an allow-all {@code @Bean RestGuardList} on the child, {@code /admin/threads} serves.
39   * </ul>
40   *
41   * @since 10.0.0
42   */
43  class AdminResource_Test extends TestBase {
44  
45  	// -----------------------------------------------------------------------------------------
46  	// Default-deny child.
47  	// -----------------------------------------------------------------------------------------
48  
49  	@Rest(children={AdminResource.class})
50  	public static class P extends RestServlet {
51  		private static final long serialVersionUID = 1L;
52  		@RestGet(path="/items") public String items() { return "items"; }
53  	}
54  
55  	private static final MockRestClient cp = MockRestClient.buildLax(P.class);
56  
57  	@Test void a01_threadsDeniedByDefault() throws Exception {
58  		cp.get("/admin/threads").run().assertStatus(403);
59  	}
60  
61  	@Test void a02_hostEndpointStillReachable() throws Exception {
62  		cp.get("/items").run().assertStatus(200).assertContent().asString().isContains("items");
63  	}
64  
65  	// -----------------------------------------------------------------------------------------
66  	// Allow-all child (guard replaced via @Bean RestGuardList).
67  	// -----------------------------------------------------------------------------------------
68  
69  	/** Child subclass that replaces the inherited deny-all guard with an allow-all chain. */
70  	public static class AdminChild extends AdminResource {
71  		@Bean public RestGuardList guards(BeanStore bs) {
72  			return RestGuardList.create(bs).build();
73  		}
74  	}
75  
76  	@Rest(children={AdminChild.class})
77  	public static class Q extends RestServlet {
78  		private static final long serialVersionUID = 1L;
79  	}
80  
81  	private static final MockRestClient cq = MockRestClient.buildLax(Q.class);
82  
83  	@Test void b01_threadsServesWhenGuardReplaced() throws Exception {
84  		cq.get("/admin/threads")
85  			.run()
86  			.assertStatus(200)
87  			.assertHeader("Content-Type").isContains("application/json");
88  	}
89  }
