24  import org.apache.juneau.rest.server.*;

30  /**
31   * End-to-end integration smoke test for the
32   * {@link AdminMixin}-with-DenyAllGuard-default ↔ {@link BearerTokenGuard}
33   * cross-cut. Confirms that swapping
34   * {@link org.apache.juneau.rest.server.server.guard.DenyAllGuard DenyAllGuard} for an AuthN guard is a
35   * zero-mixin-source-change drop-in.
36   *
37   * <p>
38   * Each case sits on a host servlet that mounts {@link AdminMixin} as a mixin and supplies
39   * its own {@code @Bean RestGuardList} containing a {@link BearerTokenGuard} backed by a
40   * fixed-token validator. The framework's {@code @Bean RestGuardList} override REPLACES the
41   * annotation-derived deny-all on the mixin, so the admin paths gate on the bearer token instead.
42   *
43   * @since 10.0.0
44   */
45  class AdminMixin_AuthIntegration_Test extends TestBase {
46  
47  	private static final Principal ADMIN = () -> "admin";
48  
49  	/** Accepts only "secret-token"; throws for anything else. */
50  	private static final TokenValidator FIXED_TOKEN = token -> {
51  		if ("secret-token".equals(token))
52  			return ADMIN;
53  		throw new AuthenticationException("Invalid token");
54  	};
55  
56  	@Rest(mixins=AdminMixin.class)
57  	public static class A extends RestServlet {
58  		private static final long serialVersionUID = 1L;
59  
60  		@Bean
61  		public RestGuardList guards(BeanStore bs) {
62  			return RestGuardList.create(bs)
63  				.append(BearerTokenGuard.create().realm("admin").validator(FIXED_TOKEN).build())
64  				.build();
65  		}
66  
67  		@RestGet(path="/items") public String items() { return "items"; }
68  	}
69  
70  	private static final MockRestClient ca = MockRestClient.buildLax(A.class);
71  
72  	// -----------------------------------------------------------------------------------------
73  	// Missing token: 401 with bearer challenge - proves the AuthN guard is now gating /admin/*
74  	// instead of the DenyAllGuard which would have returned 403.
75  	// -----------------------------------------------------------------------------------------
76  
77  	@Test void a01_missingTokenReturns401OnAdminThreads() throws Exception {
78  		ca.get("/admin/threads")
79  			.run()
80  			.assertStatus(401)
81  			.assertHeader("WWW-Authenticate").is("Bearer realm=\"admin\"");
82  	}
83  
84  	@Test void a02_missingTokenReturns401OnAdminHeap() throws Exception {
85  		ca.get("/admin/heap")
86  			.run()
87  			.assertStatus(401)
88  			.assertHeader("WWW-Authenticate").is("Bearer realm=\"admin\"");
89  	}
90  
91  	@Test void a03_missingTokenReturns401OnCacheFlush() throws Exception {
92  		ca.post("/admin/cache/flush", "")
93  			.run()
94  			.assertStatus(401);
95  	}
96  
97  	// -----------------------------------------------------------------------------------------
98  	// Invalid token: validator throws -> 401, NOT 403. Confirms the swap.
99  	// -----------------------------------------------------------------------------------------
100 
101 	@Test void b01_invalidTokenReturns401() throws Exception {
102 		ca.get("/admin/threads").header("Authorization", "Bearer wrong-token")
103 			.run()
104 			.assertStatus(401)
105 			.assertHeader("WWW-Authenticate").is("Bearer realm=\"admin\"")
106 			.assertContent().asString().isContains("Invalid token");
107 	}
108 
109 	// -----------------------------------------------------------------------------------------
110 	// Valid token: admin paths serve.
111 	// -----------------------------------------------------------------------------------------
112 
113 	@Test void c01_validTokenServesThreads() throws Exception {
114 		ca.get("/admin/threads").header("Authorization", "Bearer secret-token")
115 			.run()
116 			.assertStatus(200)
117 			.assertHeader("Content-Type").isContains("application/json");
118 	}
119 
120 	@Test void c02_validTokenServesHeap() throws Exception {
121 		ca.get("/admin/heap").header("Authorization", "Bearer secret-token")
122 			.run()
123 			.assertStatus(200)
124 			.assertHeader("Content-Type").isContains("application/json");
125 	}
126 
127 	@Test void c03_validTokenServesCacheFlush() throws Exception {
128 		ca.post("/admin/cache/flush", "").header("Authorization", "Bearer secret-token")
129 			.run()
130 			.assertStatus(200);
131 	}
132 
133 	@Test void c04_validTokenServesHostEndpoint() throws Exception {
134 		ca.get("/items").header("Authorization", "Bearer secret-token")
135 			.run()
136 			.assertStatus(200)
137 			.assertContent().asString().isContains("items");
138 	}
139 }
