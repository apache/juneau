29  /**
30   * Validates {@link EchoMixin} mounted as a mixin via {@code @Rest(mixins=...)} on a vanilla
31   * {@link RestServlet}.
32   *
33   * <p>
34   * Cases covered:
35   * <ul>
36   * 	<li>Default deny &mdash; no {@code @Rest(debug)} on the host returns {@code 404} from
37   * 		{@code /echo/*} so the endpoint's existence isn't disclosed.
38   * 	<li>{@code @Rest(debug=@Debug("always"))} unlocks the endpoint and returns the full echo payload.
39   * 	<li>{@code @Rest(debug=@Debug("conditional"))} requires the {@code Debug: true} request header.
40   * 	<li>Sensitive headers ({@code Authorization}, {@code Cookie}) are redacted by default.
41   * 	<li>Importer's {@code @Bean EchoMixin} factory drives the body cap and redact list.
42   * 	<li>Body capture truncates correctly when the inbound body exceeds the configured cap.
43   * 	<li>Path remainder, query string, and query params are surfaced.
44   * 	<li>The handler dispatches on {@code @RestOp(method="*")} &mdash; POST and PUT both work.
45   * 	<li>The host's own endpoints remain reachable.
46   * </ul>
47   *
48   * @since 10.0.0
49   */
50  @SuppressWarnings({
51  	"java:S5778"  // assertThrows lambdas with chained calls; intermediate invocations do not throw in practice
52  })

78  	/** Host with debug always-on so the echo endpoint serves. */
79  	@Rest(mixins=EchoMixin.class, debug=@Debug("always"))
80  	public static class B extends RestServlet {
81  		private static final long serialVersionUID = 1L;
82  		@RestGet(path="/items") public String items() { return "items"; }
83  	}

85  	private static final MockRestClient cb = MockRestClient.buildLax(B.class);

87  	@Test void b01_echoServesFullPayload() throws Exception {
88  		var body = cb.get("/echo/foo/bar?x=1&y=hello")
89  			.run()
90  			.assertStatus(200)
91  			.assertHeader("Content-Type").isContains("application/json")
92  			.getContent().asString();
93  		var parsed = JsonParser.DEFAULT.read(body, Map.class);
94  		Assertions.assertEquals("GET", parsed.get("method"));
95  		Assertions.assertTrue(((String) parsed.get("path")).endsWith("/echo/foo/bar"),
96  			"path should end with /echo/foo/bar; got: " + parsed.get("path"));
97  		Assertions.assertEquals("x=1&y=hello", parsed.get("queryString"));
98  		Assertions.assertEquals("foo/bar", parsed.get("pathRemainder"));
99  		Assertions.assertEquals(Boolean.FALSE, parsed.get("truncated"));
100 		var qp = (Map<?,?>) parsed.get("queryParams");
101 		Assertions.assertEquals("1", qp.get("x"));
102 		Assertions.assertEquals("hello", qp.get("y"));
103 	}

171 	/** Host with a custom redact list and a tight body cap via @Bean factory. */
172 	@Rest(mixins=EchoMixin.class, debug=@Debug("always"))
173 	public static class D extends RestServlet {
174 		private static final long serialVersionUID = 1L;
175 		@Bean public EchoMixin echo() {
176 			return EchoMixin.create()
177 				.bodyLimit(8L)
178 				.redactHeader("X-Internal-Trace")
179 				.build();
180 		}
181 	}
182 
183 	private static final MockRestClient cd = MockRestClient.buildLax(D.class);
184 
185 	@Test void d01_customRedactHeaderHonored() throws Exception {
186 		var body = cd.get("/echo/")
187 			.header("X-Internal-Trace", "abc-trace-id")
188 			.run()
189 			.assertStatus(200)
190 			.getContent().asString();
191 		Assertions.assertFalse(body.contains("abc-trace-id"),
192 			"Custom-redacted header value must NOT be reflected back; body was: " + body);
193 		Assertions.assertTrue(body.contains(EchoMixin.REDACTED));
194 	}
195 
196 	@Test void d02_authorizationStillRedactedAfterAddingCustom() throws Exception {
197 		var body = cd.get("/echo/")
198 			.header("Authorization", "Bearer secret-token")
199 			.run()
200 			.assertStatus(200)
201 			.getContent().asString();
202 		Assertions.assertFalse(body.contains("secret-token"),
203 			"Built-in default redactions must remain in place when redactHeader(...) is called; body was: "
204 				+ body);
205 	}

207 	@Test void d03_bodyCapTruncates() throws Exception {
208 		var body = cd.post("/echo/", "0123456789ABCDEF")
209 			.run()
210 			.assertStatus(200)
211 			.getContent().asString();
212 		var parsed = JsonParser.DEFAULT.read(body, Map.class);
213 		Assertions.assertEquals(Boolean.TRUE, parsed.get("truncated"),
214 			"Body must be flagged truncated when it exceeds the cap; body was: " + body);
215 		Assertions.assertEquals(8L, ((Number) parsed.get("contentLength")).longValue());
216 		Assertions.assertEquals("01234567", parsed.get("content"));
217 	}

221 	public static class G extends RestServlet {

230 	@Test void g01_zeroBodyLimitTruncatesNonEmptyBody() throws Exception {
231 		var body = cg.post("/echo/", "ANY")
232 			.run()
233 			.assertStatus(200)
234 			.getContent().asString();

241 	/** Host with a redactedHeaders(...) replace-list that disables built-in defaults. */
242 	@Rest(mixins=EchoMixin.class, debug=@Debug("always"))
243 	public static class E extends RestServlet {
244 		private static final long serialVersionUID = 1L;
245 		@Bean public EchoMixin echo() {
246 			return EchoMixin.create()
247 				.redactedHeaders("X-Custom-Only")
248 				.build();
249 		}
250 	}

252 	private static final MockRestClient ce = MockRestClient.buildLax(E.class);
253 
254 	@Test void e01_replaceListDropsBuiltInRedactions() throws Exception {
255 		var body = ce.get("/echo/")
256 			.header("Authorization", "Bearer not-redacted-here")
257 			.run()
258 			.assertStatus(200)
259 			.getContent().asString();

261 			"redactedHeaders(...) replace-list should drop built-in defaults; body was: " + body);
262 	}

264 	@Test void e02_replaceListRedactsCustomHeader() throws Exception {
265 		var body = ce.get("/echo/")
266 			.header("X-Custom-Only", "secret-custom")

268 			.assertStatus(200)
269 			.getContent().asString();
