31  /**
32   * Validates that the {@link RouteIndexMixin} route-index op is <b>content-negotiated</b> &mdash; it returns a
33   * {@link org.apache.juneau.rest.server.server.beans.RouteDescriptions} POJO rather than writing a fixed format, so the host's
34   * configured serializers render HTML / JSON / XML based on the request {@code Accept} header.
35   *
36   * <p>
37   * Mirrors the {@code NavigationMixin.getChildren(...)} pattern (see {@link NavigationMixin_AsMixin_Test}): the host
38   * provides serializers (here via {@link BasicUniversalConfig}), an {@code Accept: text/html} request yields a
39   * browsable page with the route {@linkplain org.apache.juneau.rest.server.server.beans.RouteDescription#getPath() path} rendered as
40   * a clickable link, while {@code application/json} / {@code text/xml} clients receive the same entries in their
41   * requested format.
42   *
43   * @since 10.0.0
44   */
45  class RouteIndexMixin_ContentNegotiation_Test extends TestBase {
46  
47  	@Rest(mixins=RouteIndexMixin.class)
48  	public static class A extends RestServlet implements BasicUniversalConfig {
49  		private static final long serialVersionUID = 1L;
50  		@RestGet(path="/items", summary="List items") public String items() { return "items"; }
51  		@RestPost(path="/items", summary="Create item") public String create() { return "created"; }
52  	}
53  
54  	private static final MockRestClient a = MockRestClient.buildLax(A.class);
55  
56  	@Test void a01_htmlYieldsBrowsablePageWithLinks() throws Exception {
57  		var body = a.get("/options")
58  			.accept("text/html")
59  			.run()
60  			.assertStatus(200)
61  			.assertHeader("Content-Type").isContains("text/html")
62  			.getContent().asString();
63  		// Rendered as an HTML table whose path cells are clickable links.
64  		assertTrue(body.contains("<table"), "HTML output should render a table; got: " + body);
65  		assertTrue(body.contains("href=") && body.contains("/items"),
66  			"HTML output should render the route path as a clickable link; got: " + body);
67  	}
68  
69  	@Test void a02_jsonYieldsStrictJson() throws Exception {
70  		var body = a.get("/options")
71  			.accept("application/json")
72  			.run()
73  			.assertStatus(200)
74  			.assertHeader("Content-Type").isContains("application/json")
75  			.getContent().asString();
76  		// Must be strict, parseable JSON (not Json5) preserving the path/methods/summary/description/deprecated shape.
77  		var parsed = JsonParser.DEFAULT.read(body, List.class);
78  		assertFalse(parsed.isEmpty(), "JSON route index should not be empty");
79  		assertTrue(body.contains("\"path\""), "JSON output should carry the 'path' property; got: " + body);
80  		assertTrue(body.contains("/items"), "JSON output should list /items; got: " + body);
81  	}
82  
83  	@Test void a03_xmlYieldsXml() throws Exception {
84  		var body = a.get("/options")
85  			.accept("text/xml")
86  			.run()
87  			.assertStatus(200)
88  			.assertHeader("Content-Type").isContains("xml")
89  			.getContent().asString();
90  		assertTrue(body.contains("/items"), "XML output should list /items; got: " + body);
91  	}
92  
93  	@Test void a04_sameEntriesAcrossFormats() throws Exception {
94  		// All three negotiated formats must surface the same routes (both /items ops here).
95  		var json = a.get("/options").accept("application/json").run().assertStatus(200).getContent().asString();
96  		var html = a.get("/options").accept("text/html").run().assertStatus(200).getContent().asString();
97  		var xml = a.get("/options").accept("text/xml").run().assertStatus(200).getContent().asString();
98  		for (var format : List.of(json, html, xml))
99  			assertTrue(format.contains("/items"), "Every negotiated format should list /items; got: " + format);
100 	}
101 }
