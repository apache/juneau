20  import org.apache.juneau.marshall.html.*;
21  import org.apache.juneau.rest.mock.classic.*;
22  import org.apache.juneau.rest.server.*;

27  /**
28   * Validates that a mixin op inherits the host resource's class-level {@link HtmlDocConfig @HtmlDocConfig}
29   * page decoration.
30   *
31   * <p>
32   * A mixin's {@code @RestOp} methods bind to a per-mixin {@link RestContext} sub-context whose
33   * {@code getResourceClass()} is the mixin class, not the host. Without inheritance the mixin's HTML page
34   * renders with default (empty) decoration even when the host declares class-level {@code @HtmlDocConfig}.
35   * These tests cover host-only config, mixin-only config (still wins), and host+mixin-both precedence /
36   * list-merge semantics, plus the {@link StatsMixin#getStats} per-method regression guard.
37   *
38   * @since 10.0.0
39   */
40  @SuppressWarnings({
41  	"serial"  // serialVersionUID not required for test classes.
42  })
43  class MixinHtmlDocInheritance_Test extends TestBase {
44  
45  	// A page-less mixin op: relies entirely on the host's class-level @HtmlDocConfig.
46  	@Rest
47  	public static class A01_HostOnlyMixin extends RestMixin {
48  		@RestGet(path="/m")
49  		public Object m() { return "OK"; }
50  	}
51  
52  	// Host declares class-level decoration; mixin op declares none.
53  	@Rest(mixins=A01_HostOnlyMixin.class)
54  	@HtmlDocConfig(navlinks={"home: servlet:/"}, aside={"hostAside"})
55  	public static class A01_HostOnly extends BasicRestServlet implements BasicJsonHtmlConfig {}
56  
57  	@Test void a01_hostOnlyConfigInheritedByMixinOp() throws Exception {
58  		var c = MockRestClient.build(A01_HostOnly.class);
59  		var r = c.get("/m").accept("text/html").run().assertStatus(200).getContent().asString();
60  		assertContains(r, "<aside>hostAside</aside>");
61  		assertContains(r, "home");
62  	}
63  
64  	// Mixin op carries its own per-method config; host declares none.
65  	@Rest
66  	public static class A02_MixinOnlyMixin extends RestMixin {
67  		@RestGet(path="/m")
68  		@HtmlDocConfig(aside={"mixinAside"})
69  		public Object m() { return "OK"; }
70  	}
71  
72  	@Rest(mixins=A02_MixinOnlyMixin.class)
73  	public static class A02_MixinOnly extends BasicRestServlet implements BasicJsonHtmlConfig {}
74  
75  	@Test void a02_mixinOnlyConfigStillWins() throws Exception {
76  		var c = MockRestClient.build(A02_MixinOnly.class);
77  		var r = c.get("/m").accept("text/html").run().assertStatus(200).getContent().asString();
78  		assertContains(r, "<aside>mixinAside</aside>");
79  	}
80  
81  	// Both host and mixin-op declare config: covers method-level INHERIT merge and method-level override.
82  	@Rest
83  	public static class A03_BothMixin extends RestMixin {
84  		// Method INHERIT pulls in the host's class-level aside, then appends the mixin's own value.
85  		@RestGet(path="/merge")
86  		@HtmlDocConfig(aside={"INHERIT","mixinAside"})
87  		public Object merge() { return "OK"; }
88  
89  		// Method declares aside without INHERIT: overrides the host's class-level aside entirely.
90  		@RestGet(path="/override")
91  		@HtmlDocConfig(aside={"overrideAside"})
92  		public Object override() { return "OK"; }
93  	}
94  
95  	@Rest(mixins=A03_BothMixin.class)
96  	@HtmlDocConfig(aside={"hostAside"})
97  	public static class A03_Both extends BasicRestServlet implements BasicJsonHtmlConfig {}
98  
99  	@Test void a03_hostAndMixinMergeSemantics() throws Exception {
100 		var c = MockRestClient.build(A03_Both.class);
101 		var merge = c.get("/merge").accept("text/html").run().assertStatus(200).getContent().asString();
102 		assertContains(merge, "<aside>hostAside mixinAside</aside>");
103 		var override = c.get("/override").accept("text/html").run().assertStatus(200).getContent().asString();
104 		assertContains(override, "<aside>overrideAside</aside>");
105 	}
106 
107 	// Mixin-class-level config overrides host-class config (precedence: mixin-class > host-class).
108 	@Rest
109 	@HtmlDocConfig(aside={"mixinClassAside"})
110 	public static class A04_MixinClassMixin extends RestMixin {
111 		@RestGet(path="/m")
112 		public Object m() { return "OK"; }
113 	}
114 
115 	@Rest(mixins=A04_MixinClassMixin.class)
116 	@HtmlDocConfig(aside={"hostAside"})
117 	public static class A04_MixinClassWins extends BasicRestServlet implements BasicJsonHtmlConfig {}
118 
119 	@Test void a04_mixinClassOverridesHostClass() throws Exception {
120 		var c = MockRestClient.build(A04_MixinClassWins.class);
121 		var r = c.get("/m").accept("text/html").run().assertStatus(200).getContent().asString();
122 		assertContains(r, "<aside>mixinClassAside</aside>");
123 	}
124 
125 	// Opt-out: @Rest(noInherit) naming the host config annotation type suppresses inheritance.
126 	@Rest(noInherit={"HtmlDocConfig"})
127 	public static class A05_NoInheritMixin extends RestMixin {
128 		@RestGet(path="/m")
129 		public Object m() { return "OK"; }
130 	}
131 
132 	@Rest(mixins=A05_NoInheritMixin.class)
133 	@HtmlDocConfig(aside={"hostAside"})
134 	public static class A05_NoInherit extends BasicRestServlet implements BasicJsonHtmlConfig {}
135 
136 	@Test void a05_noInheritSuppressesHostConfig() throws Exception {
137 		var c = MockRestClient.build(A05_NoInherit.class);
138 		var r = c.get("/m").accept("text/html").run().assertStatus(200).getContent().asString();
139 		assertNotContains(r, "hostAside");
140 	}
141 
142 	// Regression guard: StatsMixin's per-method @HtmlDocConfig (navlinks back/json, aside=NONE) still renders,
143 	// and is not clobbered by the host's class-level decoration.
144 	@Rest(mixins=StatsMixin.class)
145 	@HtmlDocConfig(navlinks={"home: servlet:/"}, aside={"hostAside"})
146 	public static class A06_StatsHost extends BasicRestServlet implements BasicJsonHtmlConfig {}
147 
148 	@Test void a06_statsMixinPerMethodConfigStillRenders() throws Exception {
149 		var c = MockRestClient.build(A06_StatsHost.class);
150 		var r = c.get("/stats").accept("text/html").run().assertStatus(200).getContent().asString();
151 		// StatsMixin.getStats declares its own navlinks (back/json) at rank=10 and aside=NONE, so the host's
152 		// aside content must not appear on the /stats page.
153 		assertContains(r, "back");
154 		assertNotContains(r, "hostAside");
155 	}
156 
157 	private static void assertContains(String s, String needle) {
158 		if (!s.contains(needle))
159 			throw new AssertionError("Expected to contain '" + needle + "' but did not. Body: " + s);
160 	}
161 
162 	private static void assertNotContains(String s, String needle) {
163 		if (s.contains(needle))
164 			throw new AssertionError("Expected NOT to contain '" + needle + "' but did. Body excerpt: "
165 				+ s.substring(Math.max(0, s.indexOf(needle) - 50), Math.min(s.length(), s.indexOf(needle) + 100)));
166 	}
