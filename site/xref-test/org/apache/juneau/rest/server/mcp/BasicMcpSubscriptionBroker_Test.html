64  	/**
65  	 * Blocks (via a bounded poll loop, not a fixed sleep) until the given thread has actually parked in a
66  	 * blocking call, so tests that exercise {@code close()}/{@code unregister()} racing a blocked
67  	 * {@code take()} are deterministic rather than timing-dependent.
68  	 */
69  	@SuppressWarnings({
70  		"java:S2925" // The sleep is the poll interval of this bounded-deadline poll loop, not a fixed wait-and-hope delay.
71  	})
72  	private static void awaitParked(Thread thread) throws InterruptedException {
73  		var deadline = System.nanoTime() + TimeUnit.SECONDS.toNanos(2);
74  		while (thread.getState() != Thread.State.WAITING && thread.getState() != Thread.State.TIMED_WAITING) {
75  			if (System.nanoTime() > deadline)
76  				fail("thread did not reach a parked state in time (state=" + thread.getState() + ")");
77  			Thread.sleep(5);
78  		}
79  	}
80  
81  	private static McpChangeEvent pollNoWait(McpSubscription sub) throws Exception {
82  		var holder = new AtomicReference<McpChangeEvent>();
83  		var thread = new Thread(() -> {
84  			try {
85  				holder.set(sub.take());
86  			} catch (InterruptedException e) {
87  				Thread.currentThread().interrupt();
88  			}
89  		});
90  		thread.start();
91  		thread.join(200);
92  		if (thread.isAlive()) {
93  			thread.interrupt();
94  			thread.join(200);
95  			return null;
96  		}
97  		return holder.get();
98  	}

104 				var everything = broker.register("s3", new McpSubscriptionFilter(true, true, true, Set.of("file:///a.txt")))) {

133 		var broker = new BasicMcpSubscriptionBroker(4);
134 		try (var sub = broker.register("s1", new McpSubscriptionFilter(true, true, true, Set.of()))) {
135 			var received = new AtomicReference<McpChangeEvent>();
136 			var started = new CountDownLatch(1);
137 			var thread = new Thread(() -> {
138 				started.countDown();
139 				try {
140 					received.set(sub.take());
141 				} catch (InterruptedException e) {
142 					Thread.currentThread().interrupt();
143 				}
144 			});
145 			thread.start();
146 			assertTrue(started.await(2, TimeUnit.SECONDS));
147 			awaitParked(thread);
148 			broker.resourcesListChanged();
149 			thread.join(2000);
150 			assertFalse(thread.isAlive());
151 			assertEquals(McpChangeKind.RESOURCES_LIST_CHANGED, received.get().kind());

155 	@Test void a08_close_interruptsBlockedTake() throws Exception {
156 		var broker = new BasicMcpSubscriptionBroker(4);
157 		// broker.unregister("s1") below closes 'sub' as part of the behavior under test; the
158 		// try-with-resources close() at the end of this block is a harmless idempotent no-op (see
159 		// BasicMcpSubscription.close()) that satisfies the resource-leak analysis without disturbing
160 		// the interrupt-timing assertions.
161 		try (var sub = broker.register("s1", new McpSubscriptionFilter(true, true, true, Set.of()))) {
162 			var interrupted = new AtomicBoolean(false);
163 			var started = new CountDownLatch(1);
164 			var thread = new Thread(() -> {
165 				started.countDown();
166 				try {
167 					sub.take();
168 				} catch (InterruptedException e) {
169 					interrupted.set(true);
170 				}
171 			});
172 			thread.start();
173 			assertTrue(started.await(2, TimeUnit.SECONDS));
174 			awaitParked(thread);
175 			broker.unregister("s1");
176 			thread.join(2000);
177 			assertFalse(thread.isAlive());
178 			assertTrue(interrupted.get());
179 			assertEquals(0, broker.activeCount());
180 		}
181 	}

238 	@Test void a14_registerIfUnder_concurrentRace_neverExceedsCapAndRejectsExactlyTheRightCount() throws Exception {
239 		final var max = 10;
240 		final var seeded = max - 1;
241 		var broker = new BasicMcpSubscriptionBroker(4);
242 		for (var i = 0; i < seeded; i++)
243 			broker.registerIfUnder(max, "seed-" + i, new McpSubscriptionFilter(true, true, true, Set.of()));
244 		assertEquals(seeded, broker.activeCount());
245 
246 		final var threadCount = 50;
247 		var ready = new CountDownLatch(threadCount);
248 		var start = new CountDownLatch(1);
249 		var done = new CountDownLatch(threadCount);
250 		var accepted = new AtomicInteger();
251 		var rejected = new AtomicInteger();
252 		var threads = new ArrayList<Thread>();
253 		for (var i = 0; i < threadCount; i++) {
254 			final var id = "racer-" + i;
255 			var t = new Thread(() -> {
256 				ready.countDown();
257 				try {
258 					start.await();
259 				} catch (InterruptedException e) {
260 					Thread.currentThread().interrupt();
261 					return;
262 				}
263 				var result = broker.registerIfUnder(max, id, new McpSubscriptionFilter(true, true, true, Set.of()));
264 				(result.isPresent() ? accepted : rejected).incrementAndGet();
265 				done.countDown();
266 			});
267 			threads.add(t);
268 			t.start();
269 		}
270 		assertTrue(ready.await(2, TimeUnit.SECONDS), "threads did not all reach the start gate in time");
271 		start.countDown();
272 		assertTrue(done.await(5, TimeUnit.SECONDS), "threads did not all finish in time");
273 		for (var t : threads)
274 			t.join(1000);
275 
276 		assertEquals(max, broker.activeCount(), "active count must never exceed the cap");
277 		assertEquals(1, accepted.get(), "exactly one racer should have been admitted to fill the last slot");
278 		assertEquals(threadCount - 1, rejected.get(), "every other racer must be rejected, not silently over-admitted");
279 	}
