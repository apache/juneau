37  	/**
38  	 * Polls until no live thread named {@code name} remains (or the timeout elapses), used to confirm the
39  	 * worker/heartbeat threads (whose executors/handles are private to {@code Pump}) actually terminate
40  	 * rather than merely being asked to.
41  	 */
42  	private static boolean awaitNoThreadNamed(String name, Duration timeout) throws InterruptedException {
43  		return awaitTrue(() -> Thread.getAllStackTraces().keySet().stream().noneMatch(t -> t.getName().equals(name)), timeout);
44  	}

46  	/** Polls {@code condition} until it is {@code true} (or the timeout elapses), for state with no completion signal. */
47  	@SuppressWarnings({
48  		"java:S2925" // The sleep is the poll interval of this bounded-deadline poll loop, not a fixed wait-and-hope delay.
49  	})
50  	private static boolean awaitTrue(java.util.function.BooleanSupplier condition, Duration timeout) throws InterruptedException {
51  		var deadline = System.nanoTime() + timeout.toNanos();
52  		do {
53  			if (condition.getAsBoolean())
54  				return true;
55  			Thread.sleep(10);
56  		} while (System.nanoTime() < deadline);
57  		return false;
58  	}

60  	/** Manual demand control: {@link #nextWithin} issues exactly one {@code request(1)} then waits. */
61  	private static final class RecordingSubscriber implements Flow.Subscriber<SseEvent> {
62  		final BlockingQueue<SseEvent> events = new LinkedBlockingQueue<>();
63  		final CountDownLatch completedOrErrored = new CountDownLatch(1);
64  		volatile Throwable error;
65  		Flow.Subscription subscription;
66  
67  		@Override public void onSubscribe(Flow.Subscription s) { subscription = s; }
68  		@Override public void onNext(SseEvent item) { events.add(item); }
69  		@Override public void onComplete() { completedOrErrored.countDown(); }
70  		@Override public void onError(Throwable t) { error = t; completedOrErrored.countDown(); }
71  
72  		SseEvent nextWithin(Duration timeout) throws InterruptedException {
73  			subscription.request(1);
74  			var event = events.poll(timeout.toMillis(), TimeUnit.MILLISECONDS);
75  			assertNotNull(event, "expected a frame within " + timeout);
76  			return event;
77  		}
78  	}

80  	/**
81  	 * Real blocking queue + close()-interrupts-take() semantics, matching the {@code McpSubscription} contract.
82  	 *
83  	 * <p>
84  	 * Guards the "check closed / register taker" step in {@link #take()} and the "flip closed / read taker +
85  	 * interrupt" step in {@link #close()} with the same {@code lock}, exactly like {@code BasicMcpSubscription}
86  	 * (Phase 1): without this, a {@code close()} that lands between two {@code take()} calls (i.e. while
87  	 * {@code taker} is transiently {@code null}) would silently skip the interrupt and the pump thread would
88  	 * block forever in the next {@code queue.take()} with no further events ever offered — a lost-interrupt
89  	 * hang, not a rare flake, since nothing thereafter can wake it.
90  	 */
91  	private static final class FakeSubscription implements McpSubscription {
92  		private final BlockingQueue<McpChangeEvent> queue = new LinkedBlockingQueue<>();
93  		private final AtomicBoolean closed = new AtomicBoolean();
94  		private final Object lock = new Object();
95  		private volatile Thread taker;
96  
97  		void offer(McpChangeEvent event) { queue.offer(event); }
98  
99  		@Override public String getId() { return "99"; }
100 		@Override public McpSubscriptionFilter getFilter() { return new McpSubscriptionFilter(true, true, true, Set.of("file:///a")); }
101 
102 		@Override
103 		public McpChangeEvent take() throws InterruptedException {
104 			synchronized (lock) {
105 				if (closed.get())
106 					throw new InterruptedException("subscription closed");
107 				taker = Thread.currentThread();
108 			}
109 			try {
110 				return queue.take();
111 			} finally {
112 				taker = null;
113 			}
114 		}
115 
116 		@Override public boolean isClosed() { return closed.get(); }
117 
118 		@Override
119 		public void close() {
120 			if (closed.compareAndSet(false, true)) {
121 				synchronized (lock) {
122 					var t = taker;
123 					if (t != null)
124 						t.interrupt();
125 				}
126 			}
127 		}
128 	}

242 			"the heartbeat executor's thread must not survive an onError termination");
243 		assertTrue(awaitNoThreadNamed("mcp-subscriptions-listen-321", Duration.ofSeconds(2)),
244 			"the worker thread must not survive an onError termination");

247 	// C2: cancel() racing request()'s startup block must never leave the heartbeat executor running. Races
248 	// many iterations via a shared start latch to probe for the interleaving; the fix makes this hold
249 	// deterministically for every ordering, so this also serves as a permanent regression guard.
250 	//
251 	// I2 coverage gap: uses a positive idleTimeoutMs (large enough it can never itself fire during this
252 	// test) so idleWatchdogExecutor is also constructed inside request()'s startupLock section here,
253 	// exercising it under the exact same startup race the design exists for - the sibling race test below
254 	// keeps idleTimeoutMs at 0 (watchdog never constructed) so both configurations remain covered.
255 	@Test
256 	void cancelRacingRequestDuringStartupNeverLeaksTheHeartbeatExecutor() throws Exception {
257 		for (var i = 0; i < 50; i++) {
258 			var listenId = "race-" + i;
259 			var subscription = new FakeSubscription();
260 			var publisher = new SubscriptionsListenPublisher(listenId, new SubscriptionFilter(), subscription, 5L, 60_000L);
261 			var subscriber = new RecordingSubscriber();
262 			publisher.subscribe(subscriber);
263 
264 			var start = new CountDownLatch(1);
265 			var requester = new Thread(() -> {
266 				await(start);
267 				subscriber.subscription.request(1);
268 			});
269 			var canceler = new Thread(() -> {
270 				await(start);
271 				subscriber.subscription.cancel();
272 			});
273 			requester.start();
274 			canceler.start();
275 			start.countDown();
276 			requester.join(2000);
277 			canceler.join(2000);
278 
279 			assertTrue(awaitNoThreadNamed("mcp-subscriptions-heartbeat-" + listenId, Duration.ofSeconds(2)),
280 				"iteration " + i + ": heartbeat executor thread must not survive a cancel()/request() race");
281 			assertTrue(awaitNoThreadNamed("mcp-subscriptions-idle-" + listenId, Duration.ofSeconds(2)),
282 				"iteration " + i + ": idle watchdog executor thread must not survive a cancel()/request() race");
283 			assertTrue(awaitNoThreadNamed("mcp-subscriptions-listen-" + listenId, Duration.ofSeconds(2)),
284 				"iteration " + i + ": worker thread must not survive a cancel()/request() race");
285 		}
286 	}
