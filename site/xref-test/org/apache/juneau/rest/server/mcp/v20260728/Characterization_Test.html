251 	 * Deterministic, allocation-trivial {@link TracerHook}: renders the exact remote-parent value the
252 	 * real composite carrier resolved (already {@code params._meta}-over-HTTP-header precedence,
253 	 * applied by {@code McpRevision}'s {@code TraceContextExtractor}), or a fixed root {@code traceparent}
254 	 * literal when the carrier recognizes no remote parent at all. Every rendered value is therefore
255 	 * produced by executing this hook against the real carrier, never hand-invented per fixture.
256 	 */
257 	private static final class DeterministicTracer implements TracerHook {
258 		static final TracerHook INSTANCE = new DeterministicTracer();
259 		static final String ROOT_TRACEPARENT = "00-00000000000000000000000000000c-000000000000000c-01";
260 
261 		@Override public Scope startSpan(RestRequest request) {
262 			return startSpan(request, null, TraceOperation.DEFAULT);
263 		}
264 
265 		@Override public Scope startSpan(RestRequest request, TraceContextCarrier carrier, TraceOperation operation) {
266 			var traceparent = carrier == null ? null : carrier.get(RequestMeta.KEY_TRACEPARENT);
267 			request.setAttribute(TraceContextResponseProcessor.ATTR_TRACEPARENT, traceparent != null ? traceparent : ROOT_TRACEPARENT);
268 			var tracestate = carrier == null ? null : carrier.get(RequestMeta.KEY_TRACESTATE);
269 			if (tracestate != null)
270 				request.setAttribute(TraceContextResponseProcessor.ATTR_TRACESTATE, tracestate);
271 			var baggage = carrier == null ? null : carrier.get(RequestMeta.KEY_BAGGAGE);
272 			if (baggage != null)
273 				request.setAttribute(TraceContextResponseProcessor.ATTR_BAGGAGE, baggage);
274 			return new Scope() {
275 				@Override public void setStatusCode(int statusCode) { /* no fixture asserts on a recorded status code */ }
276 				@Override public void setError(Throwable error) { /* no fixture asserts on a recorded error */ }
277 				@Override public void close() { /* nothing to release; this scope only writes request attributes above */ }
278 			};
279 		}

369 	/**
370 	 * Test-only deterministic {@link RequestStateCodec}: AES-256-GCM with a <b>hardcoded</b> key and nonce.
371 	 *
372 	 * <p>
373 	 * Unlike the production {@link AeadRequestStateCodec} (random per-process key + random per-seal nonce), this
374 	 * codec produces identical ciphertext for identical {@code (state, aad)} inputs and can unseal a token sealed by
375 	 * any other instance &mdash; exactly the reproducibility a committed characterization fixture needs. It is still
376 	 * genuinely tamper-evident (the GCM auth tag rejects the {@code MRTR-tampered-request-state} corruption) and
377 	 * honors the AAD binding. Fixed-nonce AES-GCM is catastrophically insecure for real traffic; it is used here
378 	 * <em>solely</em> because determinism, not confidentiality, is the goal in a fixture.
379 	 */
380 	static final class FixedKeyGcmCodec implements RequestStateCodec {
381 		private static final SecretKey KEY = new SecretKeySpec(new byte[32], "AES");
382 		private static final byte[] NONCE = new byte[12];
383 		private static final Base64.Encoder B64 = Base64.getUrlEncoder().withoutPadding();
384 
385 		@Override public String seal(McpRequestState state, String aad, Principal principal) {
386 			try {
387 				var cipher = Cipher.getInstance("AES/GCM/NoPadding");
388 				cipher.init(Cipher.ENCRYPT_MODE, KEY, new GCMParameterSpec(128, NONCE));
389 				cipher.updateAAD(aad.getBytes(StandardCharsets.UTF_8));
390 				var ciphertext = cipher.doFinal(Json.of(state).getBytes(StandardCharsets.UTF_8));
391 				return B64.encodeToString(NONCE) + "." + B64.encodeToString(ciphertext);
392 			} catch (Exception e) {
393 				throw new RuntimeException("Fixture codec seal failed", e);
394 			}
395 		}
396 
397 		@Override public Optional<McpRequestState> unseal(String token, String aad, Principal principal) {
398 			try {
399 				var parts = token.split("\\.", 2);
400 				var nonce = Base64.getUrlDecoder().decode(parts[0]);
401 				var ciphertext = Base64.getUrlDecoder().decode(parts[1]);
402 				var cipher = Cipher.getInstance("AES/GCM/NoPadding");
403 				cipher.init(Cipher.DECRYPT_MODE, KEY, new GCMParameterSpec(128, nonce));
404 				cipher.updateAAD(aad.getBytes(StandardCharsets.UTF_8));
405 				return Optional.of(Json.to(new String(cipher.doFinal(ciphertext), StandardCharsets.UTF_8), McpRequestState.class));
406 			} catch (@SuppressWarnings("unused") Exception e) {
407 				return Optional.empty();
408 			}
409 		}
410 	}

481 	/** Placeholder in an {@code MRTR-*} resume fixture's committed request body, replaced at replay time. */
482 	private static final String TOKEN_PLACEHOLDER = "__REQUEST_STATE__";

484 	// Fixed expiry constants embedded in a sealed requestState so a RESUME fixture's token is reproducible
485 	// (never "now + ttl"): far-future for a still-valid token, a fixed past instant for the expired fixture.
486 	private static final long FAR_FUTURE_MS = 32503680000000L; // ~ year 3000
487 	private static final long PAST_MS = 1000L;
488 
489 	// Mirror of McpRevision#aad(method, target): the sealed AAD binds the operation target (tool name / resource
490 	// uri) in addition to method+version, so every fixture token below must seal under the same target its
491 	// resume request names or the GCM tag check fails on unseal.
492 	private static String aad(String method, String target) {
493 		return method + '\u0000' + McpProtocol.VERSION_2026_07_28 + '\u0000' + (target == null ? "" : target);
494 	}

496 	/**
497 	 * The canonical {@code argumentsHash} for every RESUME-family fixture below. Every {@code ask}/{@code confirm}
498 	 * resume request in this harness (see the {@code *.request.json} fixtures under
499 	 * {@code MRTR-resume-*}/{@code MRTR-expired-*}/{@code MRTR-max-rounds-*}/{@code MRTR-tampered-*}/
500 	 * {@code ELICIT-resume-*}) carries no {@code arguments} member at all &mdash; and the original PAUSE-triggering
501 	 * call in {@code MRTR-input-required-response.request.json} sends an explicit empty {@code "arguments":{}}
502 	 * &mdash; so every sealed fixture token here must embed the same canonical-empty-object hash that
503 	 * {@code McpRevision#resolveMrtrContext} recomputes from the RESUME request's (absent) {@code arguments} on
504 	 * every replay. Delegates to the real package-private {@link McpRevision#argumentsHash(Map)} (this test class
505 	 * shares its package) rather than re-deriving the RFC-8785/SHA-256/base64url pipeline by hand, so the fixture
506 	 * hash can never drift from what the dispatcher actually computes.
507 	 */
508 	private static final String EMPTY_ARGS_HASH = McpRevision.argumentsHash(Map.of());
509 
510 	/**
511 	 * The {@code requestState} the harness seals (with {@link FixedKeyGcmCodec}, matching {@link F_Mrtr}'s codec)
512 	 * for each RESUME-family fixture. Substituted into the committed request in place of {@link #TOKEN_PLACEHOLDER}
513 	 * so the token stays framework-owned/opaque while the request pins the surrounding wire shape.
514 	 *
515 	 * <p>
516 	 * {@code MRTR-resume-complete} and {@code MRTR-resume-input-required-again} have byte-identical committed
517 	 * request files by design: the behavioral difference between the two fixtures lives entirely inside the sealed
518 	 * opaque token's continuation value ({@code "complete-me"} vs {@code "pause-again"}), not in anything visible
519 	 * on the wire request body.
520 	 *
521 	 * <p>
522 	 * The plan's original "commit the corrupted requestState as a frozen literal" constraint for
523 	 * {@code MRTR-tampered-request-state} was consciously superseded: because {@link FixedKeyGcmCodec} makes
524 	 * {@code seal(...)} fully deterministic (fixed key, fixed nonce), regenerating the tampered token on every run
525 	 * via {@code tamper(codec.seal(...))} is equally reproducible and avoids hand-maintaining opaque ciphertext.
526 	 *
527 	 * <p>
528 	 * Every branch below passes {@link #EMPTY_ARGS_HASH} as {@code argumentsHash} (see its javadoc: every resume
529 	 * request in this fixture set carries no {@code arguments}, so this is always the value
530 	 * {@code McpRevision#resolveMrtrContext}'s always-on argument-binding check recomputes and compares against).
531 	 * {@code jti} is a stable, per-branch literal purely for readability: none of these fixtures wire a
532 	 * {@link ReplayCache} into {@link F_Mrtr}/{@link F_Elicit}'s {@code McpMrtrConfig}, so the dispatcher's
533 	 * opt-in replay check never runs and the exact {@code jti} value is otherwise inert.
534 	 */
535 	private static String mrtrToken(String fixture) {
536 		var codec = new FixedKeyGcmCodec();
537 		return switch (fixture) {
538 			case "MRTR-resume-complete" -> codec.seal(new McpRequestState("complete-me", "tools/call", 1, FAR_FUTURE_MS, "jti-resume-complete", EMPTY_ARGS_HASH), aad("tools/call", "ask"));
539 			case "MRTR-resume-input-required-again" -> codec.seal(new McpRequestState("pause-again", "tools/call", 1, FAR_FUTURE_MS, "jti-resume-pause-again", EMPTY_ARGS_HASH), aad("tools/call", "ask"));
540 			case "MRTR-expired-request-state" -> codec.seal(new McpRequestState("cont-1", "tools/call", 1, PAST_MS, "jti-expired", EMPTY_ARGS_HASH), aad("tools/call", "ask"));
541 			case "MRTR-max-rounds-exceeded" -> codec.seal(new McpRequestState("cont-1", "tools/call", McpMrtrConfig.DEFAULT_MAX_ROUNDS, FAR_FUTURE_MS, "jti-max-rounds", EMPTY_ARGS_HASH), aad("tools/call", "ask"));
542 			case "MRTR-tampered-request-state" -> tamper(codec.seal(new McpRequestState("cont-1", "tools/call", 1, FAR_FUTURE_MS, "jti-tampered", EMPTY_ARGS_HASH), aad("tools/call", "ask")));
543 			case "ELICIT-resume-accept-complete", "ELICIT-resume-decline", "ELICIT-resume-cancel" ->
544 				codec.seal(new McpRequestState("cont-1", "tools/call", 1, FAR_FUTURE_MS, "jti-elicit-resume", EMPTY_ARGS_HASH), aad("tools/call", "confirm"));
545 			default -> throw new IllegalArgumentException("No MRTR token mapping for fixture: " + fixture);
546 		};
547 	}

549 	/** Flips one ciphertext byte of a valid token so its GCM tag fails verification (the tamper fixture). */
550 	private static String tamper(String token) {
551 		var parts = token.split("\\.", 2);
552 		var ciphertext = Base64.getUrlDecoder().decode(parts[1]);
553 		ciphertext[0] ^= 1;
554 		return parts[0] + "." + Base64.getUrlEncoder().withoutPadding().encodeToString(ciphertext);
555 	}

558 	 * Redacts the opaque {@code requestState} token value in a response body to a fixed placeholder before
559 	 * comparison. The token is per-process-nondeterministic ciphertext (production uses a random key and embeds a
560 	 * wall-clock expiry), so pinning its exact bytes is neither possible nor meaningful; the fixture pins its
561 	 * presence and the surrounding wire shape instead.
562 	 */
563 	private static String normalizeRequestState(String responseBody) {
564 		return responseBody.replaceAll("\"requestState\":\"[^\"]+\"", "\"requestState\":\"<sealed>\"");
565 	}

609 	private String replayHttp(String fixture) throws Exception {
610 		var requestBody = Files.readString(DIR.resolve(fixture + ".request.json")).strip();
611 		if (requestBody.contains(TOKEN_PLACEHOLDER))
612 			requestBody = requestBody.replace(TOKEN_PLACEHOLDER, mrtrToken(fixture));
613 		var headers = loadHeaders(fixture);
614 		var client = MockRestClient.create(servletFor(fixture)).json()
615 			.contentType("application/json").accept("application/json").ignoreErrors().build();
616 		var req = client.post("/").contentString(requestBody);
617 		headers.forEach(req::header);
618 		var res = req.run();
619 		assertEquals(200, res.getStatusCode(), () -> fixture + ": HTTP status changed");
620 		return res.getContent().asString();
621 	}

629 		for (var name : props.stringPropertyNames())

631 		return m;
