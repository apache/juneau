36  /**
37   * Coverage for post-handler {@code structuredContent} validation in {@link McpRevision#callTool}: a
38   * tool's returned structured result is checked after the handler returns and before wire conversion,
39   * raising {@code -32603} (internal error) rather than the {@code -32602} (invalid params) used for
40   * input failures. This revision additionally requires the structured result to be a JSON object
41   * (the {@code 2025-06-18} wire {@code CallToolResult.structuredContent} is object-shaped), checked
42   * before the shared depth/node/deadline bounds.
43   */
44  class McpStructuredOutputSafety_Test {
45  
46  	private final BeanStore ctx = new BasicBeanStore();
47  
48  	// -------- fixtures ---------
49  
50  	private static McpServerConfig configReturning(Object value) {
51  		return new McpServerConfig().addTool(new McpToolHandler() {
52  			@Override public McpToolSpec descriptor() { return new McpToolSpec().setName("out"); }
53  			@Override public McpToolOutcome call(Map<String,Object> arguments, BeanStore ctx) {
54  				return McpToolOutcome.text("ok").setStructuredContent(value);
55  			}
56  		});
57  	}
58  
59  	private static JsonRpcRequest req(Object id, String method, Object params) {
60  		return new JsonRpcRequest().setJsonrpc(McpProtocol.JSON_RPC_2_0).setId(id).setMethod(method).setParams(params);
61  	}
62  
63  	private JsonRpcResponse send(McpServerConfig config, JsonRpcRequest r) {
64  		var result = new McpRevision(null).dispatch(new McpExchange(r, n -> null), config, ctx);
65  		return result instanceof McpResponseResult mrr ? mrr.response() : null;
66  	}
67  
68  	private static JsonRpcRequest requestWithId(Object id) {
69  		return req(id, McpMethods.TOOLS_CALL, JsonMap.of("name", "out", "arguments", JsonMap.of()));
70  	}
71  
72  	private static JsonRpcRequest notificationWithoutId() {
73  		return requestWithId(null);
74  	}
75  
76  	/** Builds a map nested {@code depth} levels deep. */
77  	private static Map<String,Object> nest(int depth) {
78  		Map<String,Object> node = new LinkedHashMap<>();
79  		for (var i = 1; i < depth; i++) {
80  			var parent = new LinkedHashMap<String,Object>();
81  			parent.put("child", node);
82  			node = parent;
83  		}
84  		return node;
85  	}
86  
87  	/** Builds a list nested {@code depth} levels deep. */
88  	private static List<Object> nestList(int depth) {
89  		List<Object> node = new ArrayList<>();
90  		for (var i = 1; i < depth; i++) {
91  			var parent = new ArrayList<Object>();
92  			parent.add(node);
93  			node = parent;
94  		}
95  		return node;
96  	}
97  
98  	private static Map<String,Object> flat(int nodeCount) {
99  		var m = new LinkedHashMap<String,Object>();
100 		for (var i = 0; i < nodeCount - 1; i++)  // root map (1) + (nodeCount-1) scalar entries == nodeCount nodes
101 			m.put("k" + i, i);
102 		return m;
103 	}
104 
105 	// -------- failure cases: structuredContent violating the shared bounds ---------
106 
107 	private void assertStructuredOutputRejected(Object value) {
108 		var a = send(configReturning(value), requestWithId(1));
109 		assertEquals(-32603, a.getError().getCode());
110 		assertContains("Tool structuredContent", a.getError().getMessage());
111 		assertFalse(a.getError().getMessage().contains("input"));
112 		assertNull(send(configReturning(value), notificationWithoutId()));
113 	}
114 
115 	@Test
116 	void a01_depth65Object_rejected() {
117 		assertStructuredOutputRejected(nest(65));
118 	}
119 
120 	@Test
121 	void a02_depth65Array_rejected() {
122 		// Not a Map at all: this revision's object-shape guard rejects it before the depth check runs.
123 		assertStructuredOutputRejected(nestList(65));
124 	}
125 
126 	@Test
127 	void a03_nodes10001_rejected() {
128 		assertStructuredOutputRejected(flat(10_001));
129 	}
130 
131 	@Test
132 	void a04_nonJsonLeaf_rejected() {
133 		assertStructuredOutputRejected(JsonMap.of("x", new Object()));
134 	}
135 
136 	// -------- revision-specific object-shape guard ---------
137 
138 	@Test
139 	void a05_arrayStructuredContent_rejected_notObject() {
140 		var a = send(configReturning(JsonList.of(1, 2)), requestWithId(1));
141 		assertEquals(-32603, a.getError().getCode());
142 		assertContains("2025-06-18", a.getError().getMessage());
143 		assertContains("JSON object", a.getError().getMessage());
144 	}
145 
146 	@Test
147 	void a06_stringStructuredContent_rejected_notObject() {
148 		var a = send(configReturning("text"), requestWithId(1));
149 		assertEquals(-32603, a.getError().getCode());
150 		assertContains("2025-06-18", a.getError().getMessage());
151 		assertContains("JSON object", a.getError().getMessage());
152 	}
153 
154 	// -------- accepted cases: structuredContent within bounds ---------
155 
156 	@Test
157 	void b01_normalObject_accepted() {
158 		var resp = send(configReturning(JsonMap.of("a", 1, "b", JsonMap.of("c", 2))), requestWithId(1));
159 		assertNull(resp.getError());
160 		assertInstanceOf(CallToolResult.class, resp.getResult());
161 	}
162 
163 	@Test
164 	void b02_sharedDiamond_acyclic_terminatesCleanly() {
165 		// A diamond: two parent keys point at the same shared sub-map (no cycle back to an ancestor).
166 		// Shared references are pruned by identity, so this cannot inflate the node count or loop.
167 		var shared = JsonMap.of("s", 1);
168 		var root = JsonMap.of("left", shared, "right", shared);
169 		assertTimeoutPreemptively(java.time.Duration.ofSeconds(2), () -> {
170 			var resp = send(configReturning(root), requestWithId(1));
171 			assertNull(resp.getError());
172 			assertInstanceOf(CallToolResult.class, resp.getResult());
173 		});
174 	}
