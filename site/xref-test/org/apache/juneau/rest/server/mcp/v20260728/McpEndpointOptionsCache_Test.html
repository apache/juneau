17  package org.apache.juneau.rest.server.mcp.v20260728;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.apache.juneau.rest.server.mcp.McpServerConfig;
22  import org.junit.jupiter.api.Test;
23  
24  /**
25   * Regression coverage: {@link McpEndpointOptionsCache} must key by reference
26   * <b>identity</b>, never by an endpoint's own {@code equals()}/{@code hashCode()}.
27   */
28  class McpEndpointOptionsCache_Test {
29  
30  	/**
31  	 * A resource "fixture" that overrides {@code equals}/{@code hashCode} the way a record, a Lombok
32  	 * {@code @Data} class, or any other value-ish endpoint type might: every instance of this class
33  	 * compares equal to every other instance of this class, and every instance hashes identically.
34  	 *
35  	 * <p>
36  	 * A {@code WeakHashMap}-backed cache (keyed by {@code equals()}/{@code hashCode()}) collapses distinct
37  	 * instances of this class onto ONE cache entry, so a second instance would silently observe the FIRST
38  	 * instance's memoized {@link McpOptions} (and thus its MRTR AES key and subscription broker) &mdash;
39  	 * exactly the JVM-wide sharing regression this cache's reference-identity keying eliminates.
40  	 */
41  	public static class EqualsOverridingEndpoint implements McpEndpoint {
42  		@Override public McpServerConfig getMcpConfig() { return new McpServerConfig(); }
43  		@Override public boolean equals(Object o) { return o instanceof EqualsOverridingEndpoint; }
44  		@Override public int hashCode() { return 42; }
45  	}
46  
47  	@Test void a01_fixtureReallyDoesCollapseUnderEqualsHashCode() {
48  		var e1 = new EqualsOverridingEndpoint();
49  		var e2 = new EqualsOverridingEndpoint();
50  		assertEquals(e1, e2, "fixture sanity check: two distinct instances must compare equal");
51  		assertEquals(e1.hashCode(), e2.hashCode(), "fixture sanity check: two distinct instances must hash identically");
52  		assertNotSame(e1, e2, "fixture sanity check: they must still be distinct object instances");
53  	}
54  
55  	/**
56  	 * The core regression: two distinct {@link EqualsOverridingEndpoint} instances &mdash; which compare
57  	 * {@code equals()} and hash identically &mdash; must still resolve to two distinct memoized
58  	 * {@link McpOptions}. This is the assertion that FAILS (two instances collapse onto one shared
59  	 * {@link McpOptions}) against a {@code WeakHashMap}-backed cache keyed by {@code equals()}/
60  	 * {@code hashCode()}, and PASSES against the identity-keyed {@link McpEndpointOptionsCache}.
61  	 */
62  	@Test void a02_distinctInstances_resolveToDistinctMcpOptions_despiteEqualsCollapsing() {
63  		var e1 = new EqualsOverridingEndpoint();
64  		var e2 = new EqualsOverridingEndpoint();
65  		var options1 = McpEndpointOptionsCache.resolve(e1);
66  		var options2 = McpEndpointOptionsCache.resolve(e2);
67  		assertNotSame(options1, options2,
68  			"two distinct endpoint instances must resolve to distinct McpOptions even though equals()/hashCode() "
69  			+ "collapse them, or the identity-key contract has regressed back to equals()-keying");
70  	}
71  
72  	/** Same regression, surfaced through the mixin's own {@code subscriptionBroker()} seam. */
73  	@Test void a03_distinctInstances_resolveToDistinctBrokers_despiteEqualsCollapsing() {
74  		var e1 = new EqualsOverridingEndpoint();
75  		var e2 = new EqualsOverridingEndpoint();
76  		assertNotSame(e1.subscriptionBroker(), e2.subscriptionBroker(),
77  			"two distinct endpoint instances must NOT share a subscription broker even though equals()/hashCode() "
78  			+ "collapse them");
79  	}
80  
81  	/** Same regression, surfaced through the mixin's own {@code revision()} seam (MRTR codec/AES key). */
82  	@Test void a04_distinctInstances_resolveToDistinctMrtrCodecs_despiteEqualsCollapsing() {
83  		var e1 = new EqualsOverridingEndpoint();
84  		var e2 = new EqualsOverridingEndpoint();
85  		var rev1 = (McpRevision)e1.revision();
86  		var rev2 = (McpRevision)e2.revision();
87  		assertNotSame(rev1.mrtrConfig(), rev2.mrtrConfig(),
88  			"two distinct endpoint instances must NOT share an MRTR config even though equals()/hashCode() collapse them");
89  		assertNotSame(rev1.mrtrConfig().getCodec(), rev2.mrtrConfig().getCodec(),
90  			"two distinct endpoint instances must NOT share an MRTR AES key even though equals()/hashCode() collapse them");
91  	}
92  
93  	@Test void a05_resolve_isStillMemoizedPerInstance() {
94  		var e1 = new EqualsOverridingEndpoint();
95  		assertSame(McpEndpointOptionsCache.resolve(e1), McpEndpointOptionsCache.resolve(e1),
96  			"repeated resolution of the SAME instance must still return the identical memoized McpOptions");
97  	}
98  }
