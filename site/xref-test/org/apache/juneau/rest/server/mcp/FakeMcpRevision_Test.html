32  /**
33   * The architecture acceptance test for the revision-neutral core.
34   *
35   * <p>
36   * A revision implementation written entirely from scratch, inside the core module's own test tree,
37   * with no access to any revision-specific module, must be able to serve a full MCP request through
38   * both integration paths. If this ever stops compiling or stops passing, some revision-specific
39   * assumption has leaked into the core — which is precisely the regression this test exists to catch.
40   * Keep it permanently.
41   */
42  @SuppressWarnings({
43  	"resource" // MockRestClient is a Closeable test helper; lifetime is bounded by the test method.
44  })
45  class FakeMcpRevision_Test extends TestBase {
46  
47  	/** Records what it received and returns a canned response. */
48  	static class FakeRevision implements McpRevision {
49  		McpExchange lastExchange;
50  		McpServerConfig lastConfig;
51  		BeanStore lastCtx;
52  		int calls;
53  
54  		@Override
55  		public String protocolVersion() {
56  			return "0000-00-00";
57  		}
58  
59  		@Override
60  		public McpDispatchResult dispatch(McpExchange exchange, McpServerConfig config, BeanStore ctx) {
61  			calls++;
62  			lastExchange = exchange;
63  			lastConfig = config;
64  			lastCtx = ctx;
65  			return new McpResponseResult(JsonRpcResponse.ok(exchange.request().getId(), Map.of("fake", protocolVersion())));
66  		}
67  
68  		@Override
69  		public int errorCode(McpErrorKind kind) {
70  			return -1;
71  		}
72  	}
73  
74  	static final FakeRevision SERVLET_REVISION = new FakeRevision();
75  	static final FakeRevision MIXIN_REVISION = new FakeRevision();
76  
77  	@Rest(serializers = JsonSerializer.class, parsers = JsonParser.class, defaultAccept = "application/json")
78  	public static class A extends AbstractMcpRestServlet {
79  		private static final long serialVersionUID = 1L;
80  		@Override protected McpServerConfig createMcpConfig() { return new McpServerConfig().setInstructions("fake"); }
81  		@Override protected McpRevision revision() { return SERVLET_REVISION; }
82  	}
83  
84  	@Rest(path = "/api", serializers = JsonSerializer.class, parsers = JsonParser.class, defaultAccept = "application/json")
85  	public static class B extends BasicRestServlet implements McpEndpointMixin {
86  		private static final long serialVersionUID = 1L;
87  		@Override public McpServerConfig getMcpConfig() { return new McpServerConfig().setInstructions("fake-mixin"); }
88  		@Override public McpRevision revision() { return MIXIN_REVISION; }
89  	}
90  
91  	@Test
92  	void a01_servletPath_reachesTheFakeRevision() throws Exception {
93  		var c = MockRestClient.create(A.class).json().contentType("application/json").accept("application/json").build();
94  		var res = c.post("/").contentString("{\"jsonrpc\":\"2.0\",\"id\":9,\"method\":\"anything\"}").run()
95  			.assertStatus(200).getContent().asString();
96  		assertTrue(res.contains("0000-00-00"), res);
97  		assertEquals(1, SERVLET_REVISION.calls);
98  		assertEquals("anything", SERVLET_REVISION.lastExchange.request().getMethod());
99  		assertEquals("fake", SERVLET_REVISION.lastConfig.getInstructions());
100 		assertTrue(SERVLET_REVISION.lastCtx.getBean(RestRequest.class).isPresent());
101 	}
102 
103 	@Test
104 	void a02_mixinPath_reachesTheFakeRevision() throws Exception {
105 		var c = MockRestClient.create(B.class).json().contentType("application/json").accept("application/json").build();
106 		var res = c.post("/mcp").contentString("{\"jsonrpc\":\"2.0\",\"id\":9,\"method\":\"anything\"}").run()
107 			.assertStatus(200).getContent().asString();
108 		assertTrue(res.contains("0000-00-00"), res);
109 		assertEquals(1, MIXIN_REVISION.calls);
110 		assertEquals("fake-mixin", MIXIN_REVISION.lastConfig.getInstructions());
111 	}
112 
113 	@Test
114 	void a03_headerAccessorReachesTheHttpHeaders() throws Exception {
115 		var c = MockRestClient.create(A.class).json().contentType("application/json").accept("application/json").build();
116 		c.post("/").contentString("{\"jsonrpc\":\"2.0\",\"id\":1,\"method\":\"anything\"}").header("Mcp-Method", "tools/call").run().assertStatus(200);
117 		assertEquals("tools/call", SERVLET_REVISION.lastExchange.header("Mcp-Method"));
118 		assertNull(SERVLET_REVISION.lastExchange.header("Absent-Header"));
119 	}
120 
121 	@Test
122 	void a04_coreTestTreeSeesNoRevisionModule() {
123 		assertThrows(ClassNotFoundException.class,
124 			() -> Class.forName("org.apache.juneau.rest.server.mcp.v20250618.McpRevision"),
125 			"the core module's test classpath must not contain any revision adapter");
126 		assertThrows(ClassNotFoundException.class,
127 			() -> Class.forName("org.apache.juneau.bean.mcp.v20250618.Tool"),
128 			"the core module's test classpath must not contain any revision's wire beans");
129 	}
