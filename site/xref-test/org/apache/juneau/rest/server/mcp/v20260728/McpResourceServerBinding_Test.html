59  	 * Fake validator: {@code good} = correct audience + required scope; {@code noscope} = correct audience, wrong scope;
60  	 * {@code wrongaud} = wrong audience; anything else throws.
61  	 */
62  	private static final TokenValidator VALIDATOR = token -> switch (token) {
63  		case "good" -> new ClaimsPrincipal("alice", Map.of("aud", RESOURCE, "scope", "mcp.read mcp.write"));
64  		// Cross-identity fixture: a DIFFERENT authenticated principal (bob) with the same audience + baseline
65  		// scope, so it clears the gate but binds a different requestState identity than 'good' (alice).
66  		case "good2" -> new ClaimsPrincipal("bob", Map.of("aud", RESOURCE, "scope", "mcp.read mcp.write"));
67  		// Real iss|sub claim fixtures: 'good'/'good2' above carry no iss/sub claims, so the requestState
68  		// binding they exercise end-to-end is only the getName() fallback. 'good3'/'good4' carry the SAME subject
69  		// (alice) but a DIFFERENT issuer, so they exercise the actual, primary iss|sub claim-read path (settled
70  		// decision 1) end-to-end, not the fallback.
71  		case "good3" -> new ClaimsPrincipal("alice", Map.of("aud", RESOURCE, "scope", "mcp.read mcp.write", "iss", "https://idp-a", "sub", "alice"));
72  		case "good4" -> new ClaimsPrincipal("alice", Map.of("aud", RESOURCE, "scope", "mcp.read mcp.write", "iss", "https://idp-b", "sub", "alice"));
73  		case "noscope" -> new ClaimsPrincipal("alice", Map.of("aud", RESOURCE, "scope", "other"));
74  		case "wrongaud" -> new ClaimsPrincipal("alice", Map.of("aud", "http://evil.example.com", "scope", "mcp.read"));
75  		// SEP-2350 step-up fixtures: baseline mcp.read + the per-operation tools.exec (exact and hierarchical grants).
76  		case "opok" -> new ClaimsPrincipal("alice", Map.of("aud", RESOURCE, "scope", "mcp.read tools.exec"));
77  		case "broad" -> new ClaimsPrincipal("alice", Map.of("aud", RESOURCE, "scope", "mcp.read tools"));
78  		// H1 baseline-hierarchy fixture: granted ONLY the broad ancestor "mcp", which satisfies the baseline
79  		// "mcp.read" hierarchically (never exact-string).
80  		case "basehier" -> new ClaimsPrincipal("alice", Map.of("aud", RESOURCE, "scope", "mcp"));
81  		default -> throw new AuthenticationException("bad token");
82  	};

342 	// F4 (READY-312f): the F2-authenticated principal is threaded into the RequestStateCodec seal/unseal seam so
343 	// a hardened codec can bind the requestState to who requested it.  A capturing codec records the principal it
344 	// receives; an end-to-end pause (seal) then resume (unseal), both under the same bearer token, proves the
345 	// authenticated principal reaches the codec at BOTH points.
346 	// ---------------------------------------------------------------------------------------------
347 
348 	static final class J_CapturingCodec implements RequestStateCodec {
349 		final RequestStateCodec delegate = new AeadRequestStateCodec();
350 		final AtomicReference<String> sealPrincipal = new AtomicReference<>("<unset>");
351 		final AtomicReference<String> unsealPrincipal = new AtomicReference<>("<unset>");
352 
353 		@Override public String seal(McpRequestState state, String aad, Principal principal) {
354 			sealPrincipal.set(principal == null ? null : principal.getName());
355 			return delegate.seal(state, aad, principal);
356 		}
357 
358 		@Override public Optional<McpRequestState> unseal(String token, String aad, Principal principal) {
359 			unsealPrincipal.set(principal == null ? null : principal.getName());
360 			return delegate.unseal(token, aad, principal);
361 		}
362 	}
