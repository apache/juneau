65  		return JsonMap.of(

67  			RequestMeta.KEY_CLIENT_INFO, JsonMap.of("name", "fixture-client", "version", "1.0"),

71  	private static JsonRpcRequest req(Object id, String method, Map<String,Object> params, boolean elicitation) {
72  		var p = params == null ? new JsonMap() : new JsonMap(params);
73  		p.put("_meta", validMeta(elicitation));
74  		return new JsonRpcRequest().setJsonrpc(McpProtocol.JSON_RPC_2_0).setId(id).setMethod(method).setParams(p);
75  	}

77  	private static Map<String,String> hdrs(String method, String name) {
78  		var m = new LinkedHashMap<String,String>();
79  		m.put("Mcp-Method", method);
80  		m.put("Mcp-Name", name);
81  		return m;
82  	}

84  	private JsonRpcResponse send(McpRevision rev, McpServerConfig config, JsonRpcRequest r, Map<String,String> headers) {
85  		var result = rev.dispatch(new McpExchange(r, headers::get), config, ctx);
86  		return result instanceof McpResponseResult mrr ? mrr.response() : null;
87  	}

97  	// Mirror of McpRevision#aad(method, target): the sealed AAD now binds the operation target (tool/prompt
98  	// name or resource uri) in addition to method+version, so every hand-sealed fixture below must seal under
99  	// the same target it will be resumed against or the GCM tag check fails on unseal.
100 	private static String aad(String method, String target) {
101 		return method + '\u0000' + "2026-07-28" + '\u0000' + (target == null ? "" : target);
102 	}

744 		assertEquals("QyWM_3g_5wNtikMDP4MK38YOwDc4JHNUisdCuIgpJ3c", McpRevision.argumentsHash(JsonMap.of("b", 2, "a", 1)));

747 		assertEquals("RBNvo1WzZ4oRRq0W9-hknpT7T8If536DEMBg9hyq_4o", NO_ARGS_HASH);
