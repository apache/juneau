66  	private static JsonRpcRequest req(Object id, String method, Object params) {
67  		var p = params instanceof Map<?,?> m ? new JsonMap(m) : new JsonMap();
68  		p.put("_meta", validMeta());
69  		return new JsonRpcRequest().setJsonrpc(McpProtocol.JSON_RPC_2_0).setId(id).setMethod(method).setParams(p);
70  	}

72  	private static Map<String,String> hdrs(String method, String name) {
73  		var m = new LinkedHashMap<String,String>();
74  		m.put("Mcp-Method", method);
75  		m.put("Mcp-Name", name);
76  		return m;
77  	}

79  	private JsonRpcResponse send(McpServerConfig config, JsonRpcRequest r, Map<String,String> headers) {
80  		var result = new McpRevision(null).dispatch(new McpExchange(r, headers::get), config, ctx);
81  		return result instanceof McpResponseResult mrr ? mrr.response() : null;
82  	}

84  	private static Map<String,Object> nest(int depth) {
85  		Map<String,Object> node = new LinkedHashMap<>();
86  		for (var i = 1; i < depth; i++) {
87  			var parent = new LinkedHashMap<String,Object>();
88  			parent.put("child", node);
89  			node = parent;
90  		}
91  		return node;
92  	}
93  
94  	private static Map<String,Object> flat(int nodeCount) {
95  		var m = new LinkedHashMap<String,Object>();
96  		for (var i = 0; i < nodeCount - 1; i++)  // root map (1) + (nodeCount-1) scalar entries == nodeCount nodes
97  			m.put("k" + i, i);
98  		return m;
99  	}

109 	@Test
110 	void a02_externalRef_serializesListsAndDispatchesWithoutFetch() {
111 		// An unresolvable host: if any code path tried to dereference this $ref, DNS resolution would throw.
112 		var schema = McpSchema.of(JsonMap.of("$ref", "https://never.invalid/schema.json"));
113 		var config = new McpServerConfig().addTool(tool("refTool", schema));
114 
115 		// serialize: the $ref survives verbatim as opaque data.
116 		var wireTool = McpWire.toWire(new McpToolSpec().setName("refTool").setInputSchema(schema));
117 		assertContains("https://never.invalid/schema.json", Json.of(wireTool));
118 
119 		// list: dispatch tools/list and confirm the $ref round-trips.
120 		var listed = (ListToolsResult) send(config, req(1, "tools/list", null), hdrs("tools/list", "")).getResult();
121 		assertContains("https://never.invalid/schema.json", Json.of(listed));
122 
123 		// dispatch: tools/call runs input validation against the $ref schema and completes without a fetch.
124 		var params = JsonMap.of("name", "refTool", "arguments", JsonMap.of("anything", 1));
125 		var resp = send(config, req(2, "tools/call", params), hdrs("tools/call", "refTool"));
126 		assertNull(resp.getError());
127 		assertInstanceOf(CallToolResult.class, resp.getResult());
128 	}

164 	@Test
165 	void d01_adversarialValidation_terminatesWithinDeadline() {
166 		// A catastrophically-backtracking pattern applied to a non-matching input hangs the validator.
167 		var schema = McpSchema.of(JsonMap.of(
168 			"type", "object",
169 			"properties", JsonMap.of("s", JsonMap.of("type", "string", "pattern", "^(.*a){25}$"))));
170 		var args = new LinkedHashMap<String,Object>();
171 		args.put("s", "aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa!");  // 40 'a's then a non-matching char
172 
173 		var start = System.nanoTime();
174 		var e = assertThrows(McpException.class, () -> McpSchemaSafety.validateInput(schema, args));
175 		var elapsedMs = (System.nanoTime() - start) / 1_000_000;
176 
177 		assertEquals(-32602, e.getCode());
178 		assertContains("exceeded " + McpSchemaSafety.MAX_VALIDATION_MILLIS + " ms", e.getMessage());
179 		assertTrue(elapsedMs < McpSchemaSafety.MAX_VALIDATION_MILLIS + 5000, () -> "validation did not terminate promptly: elapsed=" + elapsedMs + "ms");
180 	}

182 	@SuppressWarnings({
183 		"java:S2925" // Thread.sleep here simulates the deterministic scheduling latency under test, not a wait-and-hope synchronization delay.
184 	})
185 	@Test
186 	void d02_schedulingLatency_notCountedAgainstComputeBudget() {
187 		// Exercises McpSchemaSafety.awaitBounded() directly (rather than saturating the shared
188 		// VALIDATION_POOL, which other tests in this class can leave with a permanently-stuck thread since a
189 		// catastrophically-backtracking regex match is not interruptible) with a task-local executor that
190 		// deliberately delays counting down `started` well past MAX_VALIDATION_MILLIS before doing its
191 		// (instantaneous) "work". If scheduling latency were - the regression this guards against - counted
192 		// against the compute budget, awaitBounded() would throw a timeout error despite the task's own
193 		// work costing nothing; with the root-cause fix, only compute time (measured from `started`) counts.
194 		var schedulingDelayMillis = 3 * McpSchemaSafety.MAX_VALIDATION_MILLIS;
195 		assertTrue(schedulingDelayMillis < McpSchemaSafety.MAX_SCHEDULING_MILLIS, "test fixture assumption");
196 
197 		var started = new CountDownLatch(1);
198 		var startedAtNanos = new AtomicLong();
199 		var executor = Executors.newSingleThreadExecutor();
200 		try {
201 			var future = executor.submit(() -> {
202 				Thread.sleep(schedulingDelayMillis);  // simulates thread-pool queueing/scheduling delay
203 				startedAtNanos.set(System.nanoTime());
204 				started.countDown();
205 				return null;  // the "validation" work itself is instantaneous
206 			});
207 
208 			assertDoesNotThrow(() -> McpSchemaSafety.awaitBounded(
209 				future, started, startedAtNanos, TimeUnit.MILLISECONDS.toNanos(McpSchemaSafety.MAX_VALIDATION_MILLIS)));
210 		} finally {
211 			executor.shutdownNow();
212 		}
213 	}

246 	private static File repoRoot() {
247 		try {
248 			var pb = new ProcessBuilder("git", "rev-parse", "--show-toplevel");
249 			pb.directory(new File(System.getProperty("user.dir")));
250 			var p = pb.start();
251 			String out;
252 			try (var r = new BufferedReader(new InputStreamReader(p.getInputStream()))) {
253 				out = r.readLine();
254 			}
255 			return p.waitFor() == 0 && out != null ? new File(out.trim()) : null;
256 		} catch (IOException | InterruptedException e) {
257 			return null;
258 		}
