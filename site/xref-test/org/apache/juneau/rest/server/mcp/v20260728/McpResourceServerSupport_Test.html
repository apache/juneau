164 	// Router-matcher path decision (host implements-McpEndpoint case).  The gate reuses the router's OWN
165 	// UrlPathMatcher for the /mcp dispatch op, so it agrees with the router on every path edge case.  These pin the
166 	// matcher decisions the gate delegates to: trailing slash (router-tolerant) matches, a doubled-slash prefix does
167 	// not (the router 404s it too, so there is no gap to bypass).
168 	// ---------------------------------------------------------------------------------------------
169 
170 	private static final UrlPathMatcher MCP_MATCHER = UrlPathMatcher.of("/mcp");
171 
172 	@Test void e01_matcher_exactPath_matches() {
173 		assertNotNull(MCP_MATCHER.match(UrlPath.of("/mcp")));
174 	}
175 
176 	@Test void e02_matcher_trailingSlash_matches() {
177 		// POST /mcp/ still routes to the MCP handler, so the reused matcher must match it too (B1).
178 		assertNotNull(MCP_MATCHER.match(UrlPath.of("/mcp/")));
179 	}
180 
181 	@Test void e03_matcher_doubledSlashPrefix_agreesWithRouter() {
182 		// //mcp parses to parts ["","mcp"]; the router does not route it to /mcp, and neither does the reused
183 		// matcher — so gate and router agree (both 404), leaving no bypass gap.
184 		assertNull(MCP_MATCHER.match(UrlPath.of("//mcp")));
185 	}
186 
187 	@Test void e04_matcher_innerDoubledSlash_agreesWithRouter() {
188 		// /mcp// parses to parts ["mcp","",""] (two trailing empties); the router does not route it to /mcp, and
189 		// neither does the reused matcher — so gate and router agree (both 404), leaving no bypass gap.
190 		assertNull(MCP_MATCHER.match(UrlPath.of("/mcp//")));
191 	}

254 	@Test void f10_enforce_insufficientThrows403WithScopedChallenge() {
255 		var c = cfg().addOperationScope("tools/call", "delete_repo", "repo.delete");
256 		var grantedScopes = Set.of("mcp.read");
257 		var ctx = op("tools/call", "delete_repo");
258 		var e = assertThrows(Forbidden.class,
259 			() -> McpResourceServerSupport.enforceOperationScopes(c, grantedScopes, ctx));
260 		assertEquals(403, e.getStatusCode());
261 		var challenge = e.getHeaders().stream()
262 			.filter(h -> "WWW-Authenticate".equalsIgnoreCase(h.getName()))
263 			.map(HttpHeader::getValue)
264 			.findFirst().orElseThrow();
265 		assertTrue(challenge.contains("error=\"insufficient_scope\""), challenge);
266 		assertTrue(challenge.contains("scope=\"repo.delete\""), challenge);
267 	}
