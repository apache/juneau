17  package org.apache.juneau.rest.server.mcp.v20250618;
18  
19  
20  import org.apache.juneau.*;
21  import org.apache.juneau.marshall.json.*;
22  import org.apache.juneau.rest.mock.classic.*;
23  import org.apache.juneau.rest.server.*;
24  import org.apache.juneau.rest.server.auth.*;
25  import org.apache.juneau.rest.server.mcp.*;
26  import org.junit.jupiter.api.*;
27  
28  /**
29   * Tests for the admission checks applied to MCP POST requests.
30   */
31  @SuppressWarnings({
32  	"resource" // Test helpers return Closeables; Eclipse JDT @Owning warning is by design.
33  })
34  class McpRequestPolicy_Test extends TestBase {
35  
36  	private static final String PING = "{\"jsonrpc\":\"2.0\",\"id\":1,\"method\":\"ping\"}";
37  
38  	@Rest(serializers = JsonSerializer.class, parsers = JsonParser.class, defaultAccept = "application/json", allowedOrigins = "https://app.example.com")
39  	public static class Open extends McpRestServlet {
40  		private static final long serialVersionUID = 1L;
41  
42  		@Override
43  		protected McpServerConfig createMcpConfig() {
44  			return new McpServerConfig().setName("t").setVersion("1");
45  		}
46  	}
47  
48  	@Rest(serializers = JsonSerializer.class, parsers = JsonParser.class, defaultAccept = "application/json")
49  	public static class Public extends McpRestServlet {
50  		private static final long serialVersionUID = 1L;
51  
52  		@Override
53  		protected McpServerConfig createMcpConfig() {
54  			return new McpServerConfig().setName("t").setVersion("1").allowUnauthenticated(true);
55  		}
56  	}
57  
58  	@Rest(serializers = JsonSerializer.class, parsers = JsonParser.class, defaultAccept = "application/json")
59  	public static class Secured extends McpRestServlet {
60  		private static final long serialVersionUID = 1L;
61  
62  		@Override
63  		protected McpServerConfig createMcpConfig() {
64  			return new McpServerConfig().setName("t").setVersion("1").setTokenValidator(t -> {
65  				if (! "good".equals(t))
66  					throw new AuthenticationException("bad");
67  				return () -> "user";
68  			});
69  		}
70  	}
71  
72  	private static MockRestClient client(Class<?> c) {
73  		return MockRestClient.create(c).accept("application/json").build();
74  	}
75  
76  	@Test void a01_wrongContentType_is415() throws Exception {
77  		client(Open.class).post("/", PING).contentType("text/plain").ignoreErrors().run().assertStatus(415);
78  	}
79  
80  	@Test void a02_sameHostOrigin_allowed() throws Exception {
81  		client(Open.class).post("/", PING).contentType("application/json").header("Host", "localhost:8080").header("Origin", "http://localhost:8080").ignoreErrors().run().assertStatus(200);
82  	}
83  
84  	@Test void a03_foreignOrigin_isRefused() throws Exception {
85  		client(Open.class).post("/", PING).contentType("application/json").header("Host", "localhost:8080").header("Origin", "http://other.example.com").ignoreErrors().run().assertStatus(403);
86  	}
87  
88  	@Test void a04_originListedOnRest_allowed() throws Exception {
89  		client(Open.class).post("/", PING).contentType("application/json").header("Host", "localhost:8080").header("Origin", "https://app.example.com").ignoreErrors().run().assertStatus(200);
90  	}
91  
92  	@Test void a05_noOrigin_allowed() throws Exception {
93  		client(Open.class).post("/", PING).contentType("application/json").ignoreErrors().run().assertStatus(200);
94  	}
95  
96  	@Test void a06_unauthenticated_foreignHost_is403() throws Exception {
97  		client(Open.class).post("/", PING).contentType("application/json").header("Host", "rebound.example.com").ignoreErrors().run().assertStatus(403);
98  	}
99  
100 	@Test void a07_allowUnauthenticated_servesAnyHost() throws Exception {
101 		client(Public.class).post("/", PING).contentType("application/json").header("Host", "rebound.example.com").ignoreErrors().run().assertStatus(200);
102 	}
103 
104 	@Test void a08_tokenValidator_requiresBearer() throws Exception {
105 		client(Secured.class).post("/", PING).contentType("application/json").ignoreErrors().run().assertStatus(401);
106 		client(Secured.class).post("/", PING).contentType("application/json").header("Authorization", "Bearer bad").ignoreErrors().run().assertStatus(401);
107 		client(Secured.class).post("/", PING).contentType("application/json").header("Authorization", "Bearer good").ignoreErrors().run().assertStatus(200);
108 	}
109 }
