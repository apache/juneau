55  		return exchangeFor(req, "text/event-stream");

58  	// C1: subscriptions/listen must negotiate SSE via the Accept header before registering with the broker.
59  	// Every other test in this class goes through this same helper with a valid "text/event-stream" Accept
60  	// value, so a real regression in the gate itself would fail broadly, not just the dedicated C1 tests below.
61  	private static McpExchange exchangeFor(JsonRpcRequest req, String accept) {
62  		return new McpExchange(req, n -> switch (n) {
63  			case "Mcp-Method" -> McpMethods.SUBSCRIPTIONS_LISTEN;
64  			case "Mcp-Name" -> "";
65  			case "Accept" -> accept;
66  			default -> null;
67  		});
68  	}

78  	// C1: a subscriptions/listen request that never negotiated SSE (Accept missing, or some other media type)
79  	// must be rejected with a JSON-RPC error BEFORE any broker registration happens - otherwise it falls into
80  	// ReactiveResponseProcessor's BUFFER shape (infinite demand, response never flushed), leaking the broker
81  	// slot + pump thread + heartbeat executor forever once a real socket is involved.
82  	@Test
83  	void missingAcceptHeaderIsRejectedBeforeRegisteringWithTheBroker() {
84  		var caps = new ServerCapabilities().setResources(new ResourceCapability().setSubscribe(true));
85  		var revision = new McpRevision(caps);
86  		var broker = new BasicMcpSubscriptionBroker(1024);
87  		var config = new McpServerConfig();
88  		var ctx = ctxWith(broker, new McpSubscriptionsConfig());
89  
90  		var result = revision.dispatch(exchangeFor(listenRequest(7, JsonMap.of()), null), config, ctx);
91  
92  		assertTrue(result instanceof McpResponseResult, "expected a McpResponseResult error, got: " + result);
93  		var resp = ((McpResponseResult) result).response();
94  		assertNotNull(resp.getError());
95  		assertEquals(McpRevision.CODE_INVALID_REQUEST, resp.getError().getCode());
96  		assertEquals(0, broker.activeCount(), "a rejected non-SSE request must not consume a broker slot");
97  	}

218 	@Test
219 	void concurrentListenRequestsNeverExceedTheCap_evenAgainstASharedBroker() throws Exception {
220 		final var max = 10;
221 		final var seeded = max - 1;
222 		var caps = new ServerCapabilities().setResources(new ResourceCapability().setSubscribe(true));
223 		var revision = new McpRevision(caps);
224 		var broker = new BasicMcpSubscriptionBroker(1024);
225 		for (var i = 0; i < seeded; i++)
226 			broker.register("seed-" + i, new McpSubscriptionFilter(false, false, false, Set.of()));
227 		var config = new McpServerConfig();
228 		var ctx = ctxWith(broker, new McpSubscriptionsConfig().setMaxConcurrentSubscriptions(max));
229 
230 		final var threadCount = 50;
231 		var ready = new CountDownLatch(threadCount);
232 		var start = new CountDownLatch(1);
233 		var done = new CountDownLatch(threadCount);
234 		var accepted = new AtomicInteger();
235 		var rejected = new AtomicInteger();
236 		var threads = new ArrayList<Thread>();
237 		for (var i = 0; i < threadCount; i++) {
238 			final var id = 1000 + i;
239 			var t = new Thread(() -> {
240 				ready.countDown();
241 				try {
242 					start.await();
243 				} catch (InterruptedException e) {
244 					Thread.currentThread().interrupt();
245 					return;
246 				}
247 				var result = revision.dispatch(exchangeFor(listenRequest(id, JsonMap.of())), config, ctx);
248 				(result instanceof McpStreamResult ? accepted : rejected).incrementAndGet();
249 				done.countDown();
250 			});
251 			threads.add(t);
252 			t.start();
253 		}
254 		assertTrue(ready.await(2, TimeUnit.SECONDS), "threads did not all reach the start gate in time");
255 		start.countDown();
256 		assertTrue(done.await(5, TimeUnit.SECONDS), "threads did not all finish in time");
257 		for (var t : threads)
258 			t.join(1000);
259 
260 		assertEquals(max, broker.activeCount(), "active count must never exceed the cap");
261 		assertEquals(1, accepted.get(), "exactly one racer should have been admitted to fill the last slot");
262 		assertEquals(threadCount - 1, rejected.get(), "every other racer must be rejected, not silently over-admitted");
263 	}

265 	/** Minimal capturing {@code Flow.Subscriber} used only to inspect a stream's own ack frame. */
266 	private static final class CapturingSubscriber implements Flow.Subscriber<SseEvent> {
267 		final BlockingQueue<SseEvent> events = new LinkedBlockingQueue<>();
268 
269 		@Override public void onSubscribe(Flow.Subscription s) { s.request(1); }
270 		@Override public void onNext(SseEvent item) { events.add(item); }
271 		@Override public void onComplete() { /* unused */ }
272 		@Override public void onError(Throwable t) { /* unused */ }
273 	}

275 	// C3: the internal broker registry key must be decoupled from the client-facing JSON-RPC id, so two
276 	// concurrent listens sharing the same client id never evict each other - while each stream still echoes
277 	// the CLIENT's original id (not the internal registry key) in its own frames.
278 	@Test
279 	void concurrentListenRequestsWithTheSameClientIdDoNotEvictEachOther() throws Exception {
280 		var caps = new ServerCapabilities().setResources(new ResourceCapability().setSubscribe(true));
281 		var revision = new McpRevision(caps);
282 		var broker = new BasicMcpSubscriptionBroker(1024);
283 		var config = new McpServerConfig();
284 		var ctx = ctxWith(broker, new McpSubscriptionsConfig());
285 
286 		var result1 = revision.dispatch(exchangeFor(listenRequest(7, JsonMap.of())), config, ctx);
287 		var result2 = revision.dispatch(exchangeFor(listenRequest(7, JsonMap.of())), config, ctx);
288 
289 		assertTrue(result1 instanceof McpStreamResult, "expected a McpStreamResult, got: " + result1);
290 		assertTrue(result2 instanceof McpStreamResult, "expected a McpStreamResult, got: " + result2);
291 		var publisher1 = ((McpStreamResult) result1).stream();
292 		var publisher2 = ((McpStreamResult) result2).stream();
293 		assertTrue(publisher1 instanceof SubscriptionsListenPublisher, "expected a SubscriptionsListenPublisher, got: " + publisher1);
294 		assertTrue(publisher2 instanceof SubscriptionsListenPublisher, "expected a SubscriptionsListenPublisher, got: " + publisher2);
295 		assertEquals(2, broker.activeCount(),
296 			"two distinct streams sharing the same client id must both remain registered, not evict each other");
297 
298 		var sub1 = new CapturingSubscriber();
299 		((SubscriptionsListenPublisher) publisher1).subscribe(sub1);
300 		var ack1 = sub1.events.poll(2, TimeUnit.SECONDS);
301 		assertNotNull(ack1, "expected the first stream's ack frame");
302 		assertTrue(ack1.getData().contains("\"" + RequestMeta.KEY_SUBSCRIPTION_ID + "\":7"),
303 			"the first stream must still echo the CLIENT's original id in its ack frame, not the internal registry key");
304 
305 		var sub2 = new CapturingSubscriber();
306 		((SubscriptionsListenPublisher) publisher2).subscribe(sub2);
307 		var ack2 = sub2.events.poll(2, TimeUnit.SECONDS);
308 		assertNotNull(ack2, "expected the second stream's ack frame");
309 		assertTrue(ack2.getData().contains("\"" + RequestMeta.KEY_SUBSCRIPTION_ID + "\":7"),
310 			"the second stream must still echo the CLIENT's original id in its ack frame, not the internal registry key");
311 	}
