55  	private static McpToolHandler tool(String name, Function<Map<String,Object>,McpToolOutcome> fn) {
56  		return new McpToolHandler() {
57  			@Override public McpToolSpec descriptor() { return new McpToolSpec().setName(name); }
58  			@Override public McpToolOutcome call(Map<String,Object> arguments, BeanStore ctx) { return fn.apply(arguments); }
59  		};
60  	}

83  	/** Merges {@code _meta} into a copy of {@code baseParams} (an object params member, or {@code null}). */
84  	private static Object withMeta(Object baseParams, Object meta) {
85  		var p = baseParams instanceof Map<?,?> m ? new JsonMap(m) : new JsonMap();
86  		p.put("_meta", meta);
87  		return p;
88  	}
89  
90  	private static JsonRpcRequest req(Object id, String method, Object params) {
91  		return new JsonRpcRequest().setJsonrpc(McpProtocol.JSON_RPC_2_0).setId(id).setMethod(method).setParams(params);
92  	}
93  
94  	private static Map<String,String> hdrs(String method, String name) {
95  		var m = new LinkedHashMap<String,String>();
96  		if (method != null)
97  			m.put("Mcp-Method", method);
98  		if (name != null)
99  			m.put("Mcp-Name", name);
100 		return m;
101 	}

103 	private JsonRpcResponse send(McpServerConfig config, JsonRpcRequest r, Map<String,String> headers) {
104 		var result = new McpRevision(null).dispatch(new McpExchange(r, headers::get), config, ctx);
105 		return result instanceof McpResponseResult mrr ? mrr.response() : null;
106 	}
