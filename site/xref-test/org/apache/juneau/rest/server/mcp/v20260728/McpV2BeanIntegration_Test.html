42   * ({@code $comment}/{@code $defs}/{@code $id}/{@code $ref}/{@code $schema}), both {@link Tool}

47  	private static final JsonSerializer MCP_JSON =
48  		JsonSerializer.create()
49  			.addBeanTypes()
50  			.typePropertyName(CompletionReference.class, "type")
51  			.build();
52  
53  	private static final JsonParser MCP_JSON_PARSER =
54  		JsonParser.create()
55  			.typePropertyName(CompletionReference.class, "type")
56  			.build();
57  
58  	private static void assertJsonRoundTrip(Object bean, Class<?> type) {
59  		var j1 = MCP_JSON.write(bean);
60  		var copy = MCP_JSON_PARSER.read(j1, type);
61  		var j2 = MCP_JSON.write(copy);
62  		assertEquals(j1, j2, () -> "Round-trip JSON mismatch for " + type.getName() + ": " + j1);
63  	}

126 	@Test void b01_cacheScopeAndFiveCarriers_roundTrip() {
127 		assertArrayEquals(new McpCacheScope[] { McpCacheScope.PUBLIC, McpCacheScope.PRIVATE }, McpCacheScope.values());
128 		assertEquals("\"public\"", Json.of(McpCacheScope.PUBLIC));
129 		assertEquals("\"private\"", Json.of(McpCacheScope.PRIVATE));
130 		for (var value : List.<CacheableResult<?>>of(new ListToolsResult(), new ListPromptsResult(),
131 			new ListResourcesResult(), new ListResourceTemplatesResult(), new ReadResourceResult())) {
132 			assertFalse(Json.of(value).contains("ttlMs"));
133 			assertFalse(Json.of(value).contains("cacheScope"));
134 			value.setTtlMs(0).setCacheScope(McpCacheScope.PRIVATE);
135 			assertTrue(Json.of(value).contains("\"ttlMs\":0"));
136 			assertTrue(Json.of(value).contains("\"cacheScope\":\"private\""));
137 			var copy = (CacheableResult<?>)JsonParser.DEFAULT.read(Json.of(value), value.getClass());
138 			assertEquals(0, copy.getTtlMs());
139 			assertEquals(McpCacheScope.PRIVATE, copy.getCacheScope());
140 		}
141 	}

143 	@Test void b02_resourceTemplateAndList_publicContract() {
144 		var template = new ResourceTemplate().setUriTemplate("file:///{name}").setName("n")
145 			.setTitle("t").setDescription("d").setMimeType("text/plain");
146 		var result = new ListResourceTemplatesResult().setResourceTemplates(template)
147 			.setNextCursor("1").setTtlMs(5).setCacheScope(McpCacheScope.PUBLIC);
148 		var copy = JsonParser.DEFAULT.read(Json.of(result), ListResourceTemplatesResult.class);
149 		assertEquals("file:///{name}", copy.getResourceTemplates().get(0).getUriTemplate());
150 		assertEquals("n", copy.getResourceTemplates().get(0).getName());
151 		assertEquals("t", copy.getResourceTemplates().get(0).getTitle());
152 		assertEquals("d", copy.getResourceTemplates().get(0).getDescription());
153 		assertEquals("text/plain", copy.getResourceTemplates().get(0).getMimeType());
154 		assertEquals("1", copy.getNextCursor());
155 		var templates = copy.getResourceTemplates();
156 		assertThrows(UnsupportedOperationException.class, () -> templates.add(template));
157 		assertEquals("resources/templates/list", McpMethods.RESOURCES_TEMPLATES_LIST);
158 	}

160 	@Test void b03_resourceTemplateAndListResult_cacheBearingRoundTrip_reassertsC2Precedence() {
161 		// Reasserts, from the consumed artifact, that C2's cache-bearing carrier contract still holds for
162 		// the C4 template-list result: ttlMs/cacheScope round-trip identically to every other CacheableResult.
163 		var template = new ResourceTemplate().setUriTemplate("file:///{name}").setName("n").setMimeType("text/plain");
164 		var result = new ListResourceTemplatesResult().setResourceTemplates(template)
165 			.setNextCursor("2").setTtlMs(60000).setCacheScope(McpCacheScope.PRIVATE);
166 		var json = Json.of(result);
167 		assertTrue(json.contains("\"ttlMs\":60000"));
168 		assertTrue(json.contains("\"cacheScope\":\"private\""));
169 		var copy = JsonParser.DEFAULT.read(json, ListResourceTemplatesResult.class);
170 		assertEquals(60000, copy.getTtlMs());
171 		assertEquals(McpCacheScope.PRIVATE, copy.getCacheScope());
172 		assertEquals("file:///{name}", copy.getResourceTemplates().get(0).getUriTemplate());
173 		assertEquals("2", copy.getNextCursor());
174 	}
