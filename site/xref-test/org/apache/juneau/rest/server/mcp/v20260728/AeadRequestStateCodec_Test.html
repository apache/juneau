283 	/**
284 	 * Pins the principal-bound-AAD contract: the caller's authenticated identity is folded into the AEAD's
285 	 * authenticated data at seal, so a token minted for one principal cannot be resumed by another. A real
286 	 * {@link AeadRequestStateCodec} seal uses a random per-seal nonce (see the class Javadoc), so two {@code seal}
287 	 * calls are never byte-identical even with everything else held fixed; this proves the binding invariant directly
288 	 * on ONE sealed token: it unseals successfully under the sealing principal, but fails ({@link Optional#empty()})
289 	 * under a completely different principal and under a <jk>null</jk> (anonymous) principal alike.
290 	 *
291 	 * <p>
292 	 * This is the inversion of the READY-312f F4 tripwire (formerly {@code a15_principalIsNotYetBoundSo...}): before
293 	 * the principal binding landed, all three unseals succeeded; binding the identity into the AAD is exactly the
294 	 * change that flips the {@code bob} / <jk>null</jk> cases to failure.
295 	 */
296 	@Test void a15_principalIsBoundSoTokenRejectsDifferentPrincipal() {
297 		var a = new AeadRequestStateCodec();
298 		var state = new McpRequestState("continuation-value", "tools/call", 1, 123456789L, "jti-1", "args-hash-1");
299 		Principal alice = () -> "alice";
300 		Principal bob = () -> "bob";
301 		var token = a.seal(state, AAD, alice);
302 		var underSamePrincipal = a.unseal(token, AAD, alice);
303 		var underDifferentPrincipal = a.unseal(token, AAD, bob);
304 		var underNullPrincipal = a.unseal(token, AAD, null);
305 		assertTrue(underSamePrincipal.isPresent(), "round trip under the sealing principal must succeed");
306 		assertEquals(state, underSamePrincipal.get());
307 		assertTrue(underDifferentPrincipal.isEmpty(),
308 			"principal is bound to the AAD, so a different principal must fail GCM tag verification");
309 		assertTrue(underNullPrincipal.isEmpty(),
310 			"principal is bound to the AAD, so a null (anonymous) principal must not unseal an authenticated-sealed token");
311 	}

314 	 * Pins the anonymous-caller policy (settled decision 2 &mdash; fail-closed with a fixed sentinel): a
315 	 * <jk>null</jk> principal binds to a constant sentinel identity rather than "skip binding", so anonymous&harr;
316 	 * anonymous round-trips while anonymous&harr;authenticated is rejected in BOTH directions.
317 	 */
318 	@Test void a16_anonymousSentinelRoundTripsButNeverCrossesAuthenticatedBoundary() {
319 		var a = new AeadRequestStateCodec();
320 		var state = new McpRequestState("continuation-value", "tools/call", 1, 123456789L, "jti-1", "args-hash-1");
321 		Principal alice = () -> "alice";
322 
323 		// Anonymous -> anonymous round-trips (the null sentinel is deterministic).
324 		var anonToken = a.seal(state, AAD, null);
325 		var anonUnderNull = a.unseal(anonToken, AAD, null);
326 		assertTrue(anonUnderNull.isPresent(), "anonymous seal must unseal under a null (anonymous) principal");
327 		assertEquals(state, anonUnderNull.get());
328 
329 		// Anonymous-sealed token must NOT unseal under an authenticated principal.
330 		assertTrue(a.unseal(anonToken, AAD, alice).isEmpty(),
331 			"an anonymous-sealed token must not be resumable by an authenticated principal");

341 	 * Pins cross-IdP collision safety at the codec level (settled decision 1 &mdash; the bound identity is
342 	 * {@code iss|sub}, not a bare {@code sub}): two {@link ClaimsPrincipal}s sharing the same {@code sub} but issued by
343 	 * different {@code iss} are distinct identities, so a token sealed under one does not unseal under the other. Also
344 	 * confirms the exact-match case still round-trips.
345 	 */
346 	@Test void a17_claimsPrincipalIssuerScopingPreventsCrossIdpResume() {
347 		var a = new AeadRequestStateCodec();
348 		var state = new McpRequestState("continuation-value", "tools/call", 1, 123456789L, "jti-1", "args-hash-1");
349 		Principal idpA = new ClaimsPrincipal("user-1", Map.of("iss", "https://idp-a.example.com", "sub", "user-1"));
350 		Principal idpB = new ClaimsPrincipal("user-1", Map.of("iss", "https://idp-b.example.com", "sub", "user-1"));
351 
352 		var token = a.seal(state, AAD, idpA);
353 		var sameIdp = a.unseal(token, AAD, idpA);
354 		assertTrue(sameIdp.isPresent(), "same iss+sub must round-trip");
355 		assertEquals(state, sameIdp.get());
356 		assertTrue(a.unseal(token, AAD, idpB).isEmpty(),
357 			"same sub but a different iss is a different identity (cross-IdP collision safety), so unseal must fail");
358 	}
