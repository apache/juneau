51  	private static Object nestedMeta() {
52  		return JsonMap.of(
53  			RequestMeta.KEY_PROTOCOL_VERSION, "2026-07-28",
54  			RequestMeta.KEY_CLIENT_INFO, JsonMap.of("name", "c", "version", "1"),
55  			RequestMeta.KEY_CLIENT_CAPABILITIES, JsonMap.of());
56  	}

58  	@Test
59  	void a01_requestMeta_opaqueMap_roundTrips() {
60  		var meta = McpWire.requestMeta(JsonMap.of("_meta", nestedMeta()));
61  		assertEquals("2026-07-28", meta.getProtocolVersion());
62  		assertEquals("c", meta.getClientInfo().getName());
63  		assertEquals("1", meta.getClientInfo().getVersion());
64  		assertNotNull(meta.getClientCapabilities());

276 	@Test
277 	void f03_resourceTemplate_mapsAllFieldsBothWaysAndNull() {
278 		var neutral = new McpResourceTemplateSpec().setUriTemplate("file:///{name}").setName("n")
279 			.setTitle("t").setDescription("d").setMimeType("text/plain");
280 		var wire = McpWire.toWire(neutral);
281 		assertEquals("file:///{name}", wire.getUriTemplate());
282 		assertEquals("n", wire.getName());
283 		assertEquals("t", wire.getTitle());
284 		assertEquals("d", wire.getDescription());
285 		assertEquals("text/plain", wire.getMimeType());
286 		var copy = McpWire.toNeutral(wire);
287 		assertEquals(neutral.getUriTemplate(), copy.getUriTemplate());
288 		assertEquals(neutral.getName(), copy.getName());
289 		assertEquals(neutral.getTitle(), copy.getTitle());
290 		assertEquals(neutral.getDescription(), copy.getDescription());
291 		assertEquals(neutral.getMimeType(), copy.getMimeType());
292 		assertNull(McpWire.toWire((McpResourceTemplateSpec) null));
293 		assertNull(McpWire.toNeutral((ResourceTemplate) null));
294 	}
