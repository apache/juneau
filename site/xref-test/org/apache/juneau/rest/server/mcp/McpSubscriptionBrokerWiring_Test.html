31  /**
32   * Coverage for the neutral opt-in subscription-broker hook (Phase 3, spec §"Neutral core" BeanStore-injection
33   * requirement): default {@code null} means no broker beans are added; an override wires
34   * {@link McpSubscriptionBroker} and {@link McpSubscriptions} into the request-scoped {@link BeanStore} as the
35   * same instance, on both the servlet and mixin entrypoints.
36   */
37  @SuppressWarnings({
38  	"resource" // MockRestClient is a Closeable test helper; lifetime is bounded by the test method.
39  })
40  class McpSubscriptionBrokerWiring_Test {
41  
42  	static class RecordingRevision implements McpRevision {
43  		BeanStore lastCtx;
44  		@Override public String protocolVersion() { return "0000-00-00"; }
45  		@Override public McpDispatchResult dispatch(McpExchange exchange, McpServerConfig config, BeanStore ctx) {
46  			lastCtx = ctx;
47  			return new McpResponseResult(JsonRpcResponse.ok(exchange.request().getId(), Map.of("ok", true)));
48  		}
49  		@Override public int errorCode(McpErrorKind kind) { return -1; }
50  	}
51  
52  	static final RecordingRevision NO_BROKER_SERVLET_REVISION = new RecordingRevision();
53  	static final RecordingRevision BROKER_SERVLET_REVISION = new RecordingRevision();
54  	static final RecordingRevision NO_BROKER_MIXIN_REVISION = new RecordingRevision();
55  	static final RecordingRevision BROKER_MIXIN_REVISION = new RecordingRevision();
56  	static final McpSubscriptionBroker SERVLET_BROKER = new BasicMcpSubscriptionBroker(4);
57  	static final McpSubscriptionBroker MIXIN_BROKER = new BasicMcpSubscriptionBroker(4);
58  
59  	@Rest(serializers = JsonSerializer.class, parsers = JsonParser.class, defaultAccept = "application/json")
60  	public static class NoBrokerServlet extends AbstractMcpRestServlet {
61  		private static final long serialVersionUID = 1L;
62  		@Override protected McpServerConfig createMcpConfig() { return new McpServerConfig(); }
63  		@Override protected McpRevision revision() { return NO_BROKER_SERVLET_REVISION; }
64  	}
65  
66  	@Rest(serializers = JsonSerializer.class, parsers = JsonParser.class, defaultAccept = "application/json")
67  	public static class BrokerServlet extends AbstractMcpRestServlet {
68  		private static final long serialVersionUID = 1L;
69  		@Override protected McpServerConfig createMcpConfig() { return new McpServerConfig(); }
70  		@Override protected McpRevision revision() { return BROKER_SERVLET_REVISION; }
71  		@Override protected McpSubscriptionBroker getSubscriptionBroker() { return SERVLET_BROKER; }
72  	}
73  
74  	@Rest(path = "/api", serializers = JsonSerializer.class, parsers = JsonParser.class, defaultAccept = "application/json")
75  	public static class NoBrokerMixin extends BasicRestServlet implements McpEndpointMixin {
76  		private static final long serialVersionUID = 1L;
77  		@Override public McpServerConfig getMcpConfig() { return new McpServerConfig(); }
78  		@Override public McpRevision revision() { return NO_BROKER_MIXIN_REVISION; }
79  	}
80  
81  	@Rest(path = "/api", serializers = JsonSerializer.class, parsers = JsonParser.class, defaultAccept = "application/json")
82  	public static class BrokerMixin extends BasicRestServlet implements McpEndpointMixin {
83  		private static final long serialVersionUID = 1L;
84  		@Override public McpServerConfig getMcpConfig() { return new McpServerConfig(); }
85  		@Override public McpRevision revision() { return BROKER_MIXIN_REVISION; }
86  		@Override public McpSubscriptionBroker subscriptionBroker() { return MIXIN_BROKER; }
87  	}
88  
89  	private static MockRestClient client(Class<?> c) {
90  		return MockRestClient.create(c).json().contentType("application/json").accept("application/json").build();
91  	}
92  
93  	@Test void a01_defaultServletHook_addsNoBrokerBeans() throws Exception {
94  		client(NoBrokerServlet.class).post("/")
95  			.contentString("{\"jsonrpc\":\"2.0\",\"id\":1,\"method\":\"anything\"}").run().assertStatus(200);
96  		assertFalse(NO_BROKER_SERVLET_REVISION.lastCtx.hasBean(McpSubscriptionBroker.class));
97  		assertFalse(NO_BROKER_SERVLET_REVISION.lastCtx.hasBean(McpSubscriptions.class));
98  	}
99  
100 	@Test void a02_overriddenServletHook_addsBothBeanAliasesAsTheSameInstance() throws Exception {
101 		client(BrokerServlet.class).post("/")
102 			.contentString("{\"jsonrpc\":\"2.0\",\"id\":1,\"method\":\"anything\"}").run().assertStatus(200);
103 		var ctx = BROKER_SERVLET_REVISION.lastCtx;
104 		assertSame(SERVLET_BROKER, ctx.getBean(McpSubscriptionBroker.class).orElse(null));
105 		assertSame(SERVLET_BROKER, ctx.getBean(McpSubscriptions.class).orElse(null));
106 	}
107 
108 	@Test void a03_defaultMixinHook_addsNoBrokerBeans() throws Exception {
109 		client(NoBrokerMixin.class).post("/mcp")
110 			.contentString("{\"jsonrpc\":\"2.0\",\"id\":1,\"method\":\"anything\"}").run().assertStatus(200);
111 		assertFalse(NO_BROKER_MIXIN_REVISION.lastCtx.hasBean(McpSubscriptionBroker.class));
112 		assertFalse(NO_BROKER_MIXIN_REVISION.lastCtx.hasBean(McpSubscriptions.class));
113 	}
114 
115 	@Test void a04_overriddenMixinHook_addsBothBeanAliasesAsTheSameInstance() throws Exception {
116 		client(BrokerMixin.class).post("/mcp")
117 			.contentString("{\"jsonrpc\":\"2.0\",\"id\":1,\"method\":\"anything\"}").run().assertStatus(200);
118 		var ctx = BROKER_MIXIN_REVISION.lastCtx;
119 		assertSame(MIXIN_BROKER, ctx.getBean(McpSubscriptionBroker.class).orElse(null));
120 		assertSame(MIXIN_BROKER, ctx.getBean(McpSubscriptions.class).orElse(null));
121 	}
122 
123 	@Test void a05_servletDefaultHook_returnsNull() {
124 		assertNull(new NoBrokerServlet().getSubscriptionBroker());
125 	}
126 
127 	@Test void a06_mixinDefaultHook_returnsNull() {
128 		assertNull(new NoBrokerMixin().subscriptionBroker());
129 	}
130 }
