73  	static final class RecordingScope implements Scope {
74  		final List<String> events = Collections.synchronizedList(new ArrayList<>());
75  		volatile Integer lastErrorCode;
76  		volatile String lastErrorMessage;
77  
78  		@Override /* Scope */ public void setStatusCode(int statusCode) { events.add("status:" + statusCode); }
79  		@Override /* Scope */ public void setError(Throwable error) { events.add("error:" + error.getClass().getSimpleName()); }
80  		@Override /* Scope */ public void close() { events.add("close"); }
81  
82  		@Override /* Scope */
83  		public void recordRpcError(int code, String message) {
84  			lastErrorCode = code;
85  			lastErrorMessage = message;
86  			events.add("rpcError:" + code);
87  		}
88  	}

90  	/**
91  	 * Records the carrier/operation an active {@link TraceContextExtractor} handed to
92  	 * {@link #startSpan(RestRequest, TraceContextCarrier, TraceOperation)}, and stashes rendered W3C
93  	 * fields as request attributes exactly the way a real bridge (e.g. {@code OtelTracerHook}) does at
94  	 * span-start time &mdash; honoring an explicitly recognized parent from the carrier, or falling back
95  	 * to a fixed "freshly started root span" value when none was recognized.
96  	 */
97  	static final class RecordingTracer implements TracerHook {
98  		static final String ROOT_TRACEPARENT = "00-00000000000000000000000000000a-000000000000000a-01";
99  
100 		volatile TraceContextCarrier lastCarrier;
101 		volatile TraceOperation lastOperation;
102 		volatile RecordingScope lastScope;
103 
104 		void reset() {
105 			lastCarrier = null;
106 			lastOperation = null;
107 			lastScope = null;
108 		}
109 
110 		@Override /* TracerHook */
111 		public Scope startSpan(RestRequest request) {
112 			return startSpan(request, null, TraceOperation.DEFAULT);
113 		}
114 
115 		@Override /* TracerHook */
116 		public Scope startSpan(RestRequest request, TraceContextCarrier carrier, TraceOperation operation) {
117 			lastCarrier = carrier;
118 			lastOperation = operation;
119 			var scope = new RecordingScope();
120 			lastScope = scope;
121 			var traceparent = carrier == null ? null : carrier.get(RequestMeta.KEY_TRACEPARENT);
122 			request.setAttribute(TraceContextResponseProcessor.ATTR_TRACEPARENT, traceparent != null ? traceparent : ROOT_TRACEPARENT);
123 			var tracestate = carrier == null ? null : carrier.get(RequestMeta.KEY_TRACESTATE);
124 			if (tracestate != null)
125 				request.setAttribute(TraceContextResponseProcessor.ATTR_TRACESTATE, tracestate);
126 			var baggage = carrier == null ? null : carrier.get(RequestMeta.KEY_BAGGAGE);
127 			if (baggage != null)
128 				request.setAttribute(TraceContextResponseProcessor.ATTR_BAGGAGE, baggage);
129 			return scope;
130 		}
131 	}

144 	private static Object withMeta(Object baseParams, Map<String,Object> extraMeta) {
145 		var p = baseParams instanceof Map<?,?> m ? new JsonMap(m) : new JsonMap();
146 		var meta = new JsonMap(validMeta());
147 		if (extraMeta != null)
148 			meta.putAll(extraMeta);
149 		p.put("_meta", meta);
150 		return p;
151 	}
