28  /**
29   * Coverage for the {@code 2025-06-18} dual-role schema-capability check.
30   *
31   * <p>
32   * This revision's wire schema bean supports exactly six keywords and requires an object root. A
33   * neutral {@link McpSchema} can carry anything, so a config whose input OR output schema uses a
34   * keyword this revision cannot express — or whose root is not an object — is rejected on first
35   * dispatch rather than silently dropping the keyword on the wire.
36   */
37  class McpSchemaCheck_Test {
38  
39  	private final BeanStore ctx = new BasicBeanStore();
40  
41  	private static McpToolHandler tool(String name, JsonMap input, JsonMap output) {
42  		return new McpToolHandler() {
43  			@Override public McpToolSpec descriptor() {
44  				return new McpToolSpec().setName(name)
45  					.setInputSchema(input == null ? null : McpSchema.of(input))
46  					.setOutputSchema(output == null ? null : McpSchema.of(output));
47  			}
48  			@Override public McpToolOutcome call(Map<String,Object> arguments, BeanStore ctx2) {
49  				return new McpToolOutcome();
50  			}
51  		};
52  	}
53  
54  	private static McpExchange ping() {
55  		return new McpExchange(new org.apache.juneau.bean.jsonrpc.JsonRpcRequest().setId(1).setMethod("ping"), n -> null);
56  	}
57  
58  	/** Builds a map nested {@code depth} levels deep through supported keys. */
59  	private static JsonMap nested(int depth) {
60  		var root = new JsonMap();
61  		var cur = root;
62  		for (var i = 1; i < depth; i++) {
63  			var next = new JsonMap();
64  			cur.put("x", next);
65  			cur = next;
66  		}
67  		return root;
68  	}

75  	void a01_unsupportedTopLevelKeyword_isRejectedNamingToolAndKeyword() {
76  		var config = new McpServerConfig().addTool(tool("risky", JsonMap.of("type", "object", "oneOf", List.of()), null));
77  		var revision = new McpRevision(null);
78  		var exchange = ping();
79  		var e = assertThrows(IllegalArgumentException.class,
80  			() -> revision.dispatch(exchange, config, ctx));
81  		assertTrue(e.getMessage().contains("risky"), e.getMessage());
82  		assertTrue(e.getMessage().contains("inputSchema"), e.getMessage());
83  		assertTrue(e.getMessage().contains("oneOf"), e.getMessage());
84  		assertTrue(e.getMessage().contains("2025-06-18"), e.getMessage());
85  	}

223 	void d01_cyclicGraph_terminatesCleanly() {
224 		var root = JsonMap.of("type", "object");
225 		var props = new JsonMap();
226 		root.put("properties", props);
227 		props.put("x", root); // cycle back to root through supported keywords only
228 		var config = new McpServerConfig().addTool(tool("ok", JsonMap.of("type", "object"), root));
229 		assertTimeoutPreemptively(java.time.Duration.ofSeconds(2),
230 			() -> assertNotNull(new McpRevision(null).dispatch(ping(), config, ctx)));
231 	}
