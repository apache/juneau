184 	 * Backs every {@code COMPLETE-*} fixture: a prompt whose two declared arguments each carry a
185 	 * deterministic completer (one ignoring context, one consuming it), and a resource-template variable
186 	 * completer whose current-value branches deterministically reproduce the small and the 101-value/capped
187 	 * cases from one registration.
188 	 */
189 	@Rest(serializers = JsonSerializer.class, parsers = JsonParser.class, defaultAccept = "application/json")
190 	public static class F_Complete extends McpRestServlet {
191 		private static final long serialVersionUID = 1L;
192 		@Override protected McpServerConfig createMcpConfig() {
193 			var greet = new McpPromptSpec().setName("greet").setDescription("pd")
194 				.setArguments(List.of(
195 					new McpPromptArgument().setName("who")
196 						.setCompleter((request, ctx) -> new McpCompletionResult().setValues(List.of("Bob", "Alice", "Bob"))),
197 					new McpPromptArgument().setName("style")
198 						.setCompleter((request, ctx) -> {
199 							var greeting = request.getContextArguments().getOrDefault("greeting", "");
200 							return new McpCompletionResult().setValues(List.of(greeting + " Alice", greeting + " Bob"));
201 						})));
202 			return new McpServerConfig()
203 				.addPrompt(new McpPromptHandler() {
204 					@Override public McpPromptSpec descriptor() { return greet; }
205 					@Override public McpPromptOutcome get(Map<String,Object> arguments, BeanStore ctx) { return new McpPromptOutcome(); }
206 				})
207 				.addResourceTemplate(new McpResourceTemplateHandler() {
208 					@Override public McpResourceTemplateSpec descriptor() {
209 						return new McpResourceTemplateSpec().setUriTemplate("file:///{name}").setName("simple").setMimeType("text/plain");
210 					}
211 					@Override public McpResourceOutcome read(String uri, Map<String,String> variables, BeanStore ctx) { return null; }
212 					@Override public McpCompleter completer(String variableName) {
213 						if (! "name".equals(variableName))
214 							return null;
215 						return (request, ctx) -> {
216 							if ("cap".equals(request.getValue())) {
217 								var values = new ArrayList<String>();
218 								for (var i = 0; i < 101; i++)
219 									values.add("item" + i);
220 								return new McpCompletionResult().setValues(values).setTotal(101);
221 							}
222 							return new McpCompletionResult().setValues(List.of("alpha", "beta"));
223 						};
224 					}
225 				});
226 		}
227 	}

263 	private static Class<?> servletFor(String fixture) {
264 		var prefix = fixture.substring(0, fixture.indexOf('-'));
265 		return switch (prefix) {
266 			case "EMPTY" -> F_Empty.class;
267 			case "FULL" -> F_Full.class;
268 			case "CAPS" -> F_Caps.class;
269 			case "PAGED" -> F_Paged.class;
270 			case "THROW" -> F_Throw.class;
271 			case "STRUCTURED" -> F_Structured.class;
272 			case "TEMPLATE" -> F_Template.class;
273 			case "COMPLETE" -> F_Complete.class;
274 			default -> throw new IllegalArgumentException("Unknown fixture prefix: " + prefix);
275 		};
276 	}

290 	@ParameterizedTest
291 	@MethodSource("fixtures")
292 	void a01_wireIsUnchanged(String fixture) throws Exception {
293 		var requestBody = Files.readString(DIR.resolve(fixture + ".request.json")).strip();
294 		var client = MockRestClient.create(servletFor(fixture)).json()
295 			.contentType("application/json").accept("application/json").ignoreErrors().build();
296 		var req = client.post("/").contentString(requestBody);
297 		loadHeaders(fixture).forEach(req::header);
298 		var res = req.run();
299 		assertEquals(200, res.getStatusCode(), () -> fixture + ": HTTP status changed");
300 		var actual = res.getContent().asString();
301 		var expected = DIR.resolve(fixture + ".response.json");
302 		if (WRITE) {
303 			Files.writeString(expected, actual);
304 			return;
305 		}
306 		assertEquals(Files.readString(expected), actual,
307 			() -> fixture + ": WIRE FORMAT CHANGED. Do not update the fixture — fix the code.");
308 	}

310 	private static Map<String,String> loadHeaders(String fixture) throws IOException {
311 		var m = new LinkedHashMap<String,String>();
312 		var path = DIR.resolve(fixture + ".headers.properties");
313 		if (! Files.exists(path))
314 			return m;
315 		var props = new Properties();
316 		try (var in = Files.newBufferedReader(path)) {
317 			props.load(in);
318 		}
319 		for (var name : props.stringPropertyNames())
320 			m.put(name, props.getProperty(name));
321 		return m;
322 	}
