84  			var req = new McpCompletionRequest().setContextArguments(null);

87  		}

89  		@Test void b04_contextIsImmutable() {
90  			var req = new McpCompletionRequest().setContextArguments(Map.of("a", "1"));
91  			var contextArguments = req.getContextArguments();
92  			assertThrows(UnsupportedOperationException.class, () -> contextArguments.put("b", "2"));
93  		}
94  
95  		@Test void b05_contextPreservesInsertionOrderIndependentOfSourceMap() {
96  			var source = new LinkedHashMap<String,String>();
97  			source.put("z", "1");
98  			source.put("a", "2");
99  			source.put("m", "3");
100 			var req = new McpCompletionRequest().setContextArguments(source);
101 			assertEquals(List.of("z", "a", "m"), List.copyOf(req.getContextArguments().keySet()));
102 			// Mutating the caller's source map afterward must not affect the stored copy.
103 			source.put("extra", "4");
104 			assertEquals(3, req.getContextArguments().size());
105 		}

154 			var raw = new McpCompletionResult().setValues(List.of("a", "a", "b", "a"));
155 			var n = McpCompletionResult.normalize(raw);
156 			assertEquals(List.of("a", "a", "b", "a"), n.values());

159 		@Test void d04_hundredValuesSurviveUnchanged() {
160 			var values = IntStream.range(0, 100).mapToObj(i -> "v" + i).toList();
161 			var raw = new McpCompletionResult().setValues(values).setHasMore(false);
162 			var n = McpCompletionResult.normalize(raw);
163 			assertEquals(100, n.values().size());
164 			assertEquals(values, n.values());
165 			assertEquals(Boolean.FALSE, n.hasMore());
166 		}
167 
168 		@Test void d05_hundredAndOneValuesTruncateToFirstHundredAndForceHasMore() {
169 			var values = IntStream.range(0, 101).mapToObj(i -> "v" + i).toList();
170 			var raw = new McpCompletionResult().setValues(values).setHasMore(false);
171 			var n = McpCompletionResult.normalize(raw);
172 			assertEquals(100, n.values().size());
173 			assertEquals(values.subList(0, 100), n.values());
174 			assertEquals(Boolean.TRUE, n.hasMore());
175 		}
176 
177 		@Test void d06_normalizedValuesAreImmutable() {
178 			var raw = new McpCompletionResult().setValues(new ArrayList<>(List.of("a", "b")));
179 			var n = McpCompletionResult.normalize(raw);
180 			var values = n.values();
181 			assertThrows(UnsupportedOperationException.class, () -> values.add("c"));
182 		}

255 	@Nested class G_neutrality {
256 
257 		private static final List<Class<?>> DOMAIN_TYPES =
258 			List.of(McpCompleter.class, McpCompletionRef.class, McpCompletionRequest.class, McpCompletionResult.class);
259 
260 		@Test void g01_noDatedImportInSourceFiles() throws IOException {
261 			var srcDir = Paths.get("src/main/java/org/apache/juneau/rest/server/mcp");
262 			for (var type : DOMAIN_TYPES) {
263 				var file = srcDir.resolve(type.getSimpleName() + ".java");
264 				assertTrue(Files.exists(file), () -> "Missing source file: " + file);
265 				var text = Files.readString(file);
266 				assertFalse(text.contains("org.apache.juneau.bean.mcp.v20"),
267 					() -> type.getSimpleName() + " must not import a dated MCP bean package");
268 				assertFalse(text.contains("org.apache.juneau.rest.server.mcp.v20"),
269 					() -> type.getSimpleName() + " must not import a dated MCP adapter package");
270 			}
271 		}
272 
273 		@Test void g02_noDatedTypeReachableViaReflection() {
274 			for (var type : DOMAIN_TYPES) {
275 				for (var member : allReferencedTypes(type)) {
276 					var name = member.getName();
277 					assertFalse(name.contains(".bean.mcp.v20"), () -> type + " reaches dated bean type " + name);
278 					assertFalse(name.contains(".rest.server.mcp.v20"), () -> type + " reaches dated adapter type " + name);
279 				}
280 			}
281 		}
282 
283 		private static List<Class<?>> allReferencedTypes(Class<?> type) {
284 			var out = new ArrayList<Class<?>>();
285 			for (var field : type.getDeclaredFields())
286 				out.add(field.getType());
287 			for (var method : type.getDeclaredMethods()) {
288 				out.add(method.getReturnType());
289 				out.addAll(List.of(method.getParameterTypes()));
290 			}
291 			for (var ctor : type.getDeclaredConstructors())
292 				out.addAll(List.of(ctor.getParameterTypes()));
293 			return out;
294 		}
