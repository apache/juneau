40  
41  		@Test void a01_simpleScalar() {
42  			var a = McpUriTemplateMatcher.compile("file:///{name}");
43  			assertTrue(a.isReverseMatchable());
44  			assertEquals(Map.of("name", "report"), a.match("file:///report"));
45  			assertNull(a.match("file:///one/two"));
46  		}
47  
48  		@Test void a02_reservedScalar() {
49  			var a = McpUriTemplateMatcher.compile("file:///{+name}");
50  			assertTrue(a.isReverseMatchable());
51  			assertEquals(Map.of("name", "one/two"), a.match("file:///one/two"));
52  		}
53  
54  		@Test void a03_fragmentScalar() {
55  			var a = McpUriTemplateMatcher.compile("file:///doc{#name}");
56  			assertTrue(a.isReverseMatchable());
57  			assertEquals(Map.of("name", "sec/tion"), a.match("file:///doc#sec/tion"));
58  			assertNull(a.match("file:///docsec"), "fragment operator requires the leading '#'");
59  		}
60  
61  		@Test void a04_pathSegmentsSingleAndMultiple() {
62  			var a = McpUriTemplateMatcher.compile("r{/x}");
63  			assertTrue(a.isReverseMatchable());
64  			assertEquals(Map.of("x", "foo"), a.match("r/foo"));
65  
66  			var b = McpUriTemplateMatcher.compile("r{/x,y}");
67  			assertTrue(b.isReverseMatchable());
68  			assertEquals(Map.of("x", "foo", "y", "bar"), b.match("r/foo/bar"));
69  			assertEquals(List.of("x", "y"), List.copyOf(b.match("r/foo/bar").keySet()));
70  		}
71  
72  		@Test void a05_labelSegmentsSingleAndMultiple() {
73  			var a = McpUriTemplateMatcher.compile("r{.x}");
74  			assertTrue(a.isReverseMatchable());
75  			assertEquals(Map.of("x", "val"), a.match("r.val"));
76  
77  			var b = McpUriTemplateMatcher.compile("r{.x,y}");
78  			assertTrue(b.isReverseMatchable());
79  			assertEquals(Map.of("x", "foo", "y", "bar"), b.match("r.foo.bar"));
80  		}
81  
82  		@Test void a06_queryStartSingleAndMultiple() {
83  			var a = McpUriTemplateMatcher.compile("r{?x}");
84  			assertTrue(a.isReverseMatchable());
85  			assertEquals(Map.of("x", "1"), a.match("r?x=1"));
86  
87  			var b = McpUriTemplateMatcher.compile("r{?x,y}");
88  			assertTrue(b.isReverseMatchable());
89  			assertEquals(Map.of("x", "1", "y", "2"), b.match("r?x=1&y=2"));
90  		}
91  
92  		@Test void a07_queryContinuationSingleAndMultiple() {
93  			var a = McpUriTemplateMatcher.compile("r?a=1{&x}");
94  			assertTrue(a.isReverseMatchable());
95  			assertEquals(Map.of("x", "1"), a.match("r?a=1&x=1"));
96  
97  			var b = McpUriTemplateMatcher.compile("r?a=1{&x,y}");
98  			assertTrue(b.isReverseMatchable());
99  			assertEquals(Map.of("x", "1", "y", "2"), b.match("r?a=1&x=1&y=2"));
100 		}
101 
102 		@Test void a08_queryValuesMayBeEmpty() {
103 			var a = McpUriTemplateMatcher.compile("r{?x}");
104 			assertEquals(Map.of("x", ""), a.match("r?x="));
105 		}
106 
107 		@Test void a09_pureLiteralTemplateMatchesOnlyItself() {
108 			var a = McpUriTemplateMatcher.compile("file:///static/logo.png");
109 			assertTrue(a.isReverseMatchable());
110 			assertEquals(Map.of(), a.match("file:///static/logo.png"));
111 			assertNull(a.match("file:///static/other.png"));
112 		}

129 			assertEquals(Map.of("name", "one/two"), plusName.match("file:///one/two"));

132 		@Test void b02_literalPrefixAndSuffixAroundCapture() {
133 			var a = McpUriTemplateMatcher.compile("file:///docs/{name}.txt");
134 			assertEquals(Map.of("name", "report"), a.match("file:///docs/report.txt"));
135 			assertNull(a.match("file:///docs/report.md"));
136 			assertNull(a.match("file:///other/report.txt"));
137 		}

139 		@Test void b03_longestReservedCaptureThatStillLetsSuffixMatch() {
140 			var a = McpUriTemplateMatcher.compile("{+path}.json");
141 			// The literal suffix ".json" also occurs inside the reserved capture's candidate text; the
142 			// longest capture that still lets ".json" match afterward wins.

305 
306 		@Test void f01_explode() {
307 			for (var t : List.of("file:///{var*}", "r{/path*}", "r{?tags*}")) {
308 				var a = McpUriTemplateMatcher.compile(t);
309 				assertFalse(a.isReverseMatchable(), t);
310 			}
311 		}
312 
313 		@Test void f02_prefix() {
314 			var a = McpUriTemplateMatcher.compile("file:///{var:3}");
315 			assertFalse(a.isReverseMatchable());
316 			assertNull(a.match("abc"));
317 		}
318 
319 		@Test void f03_matrix() {
320 			var a = McpUriTemplateMatcher.compile("r{;x}");
321 			assertFalse(a.isReverseMatchable());
322 			assertNull(a.match("r;x=1"));
323 		}
324 
325 		@Test void f04_multiVariableSimpleReservedFragment() {
326 			assertFalse(McpUriTemplateMatcher.compile("file:///{x,y}").isReverseMatchable());
327 			assertFalse(McpUriTemplateMatcher.compile("file:///{+x,y}").isReverseMatchable());
328 			assertFalse(McpUriTemplateMatcher.compile("file:///doc{#x,y}").isReverseMatchable());
329 		}
330 
331 		@Test void f05_adjacentExpressionsAreAmbiguous() {
332 			var a = McpUriTemplateMatcher.compile("file:///{a}{b}");
333 			assertFalse(a.isReverseMatchable());
334 			assertEquals(List.of("a", "b"), a.variableNames());
335 		}
336 
337 		@Test void f06_variablesOfNonMatchableTemplateRemainDeclared() {
338 			var a = McpUriTemplateMatcher.compile("r{?tags*}");
339 			assertEquals(List.of("tags"), a.variableNames());
340 			assertNull(a.match("r?tags=a&tags=b"));
341 		}
342 
343 		@Test void f07_queryFormsWithOmittedReorderedDuplicateOrExtraKeysDoNotMatch() {
344 			var a = McpUriTemplateMatcher.compile("r{?x,y}");
345 			assertTrue(a.isReverseMatchable(), "the query expression itself remains reverse-matchable");
346 			assertNull(a.match("r?x=1"), "omitted declared key");
347 			assertNull(a.match("r?y=2&x=1"), "reordered keys");
348 			assertNull(a.match("r?x=1&x=2&y=3"), "duplicate key");
349 			assertNull(a.match("r?x=1&y=2&z=3"), "extra key");
350 			assertNull(a.match("r?x&y=2"), "key-only parameter, missing '='");
351 		}

419 
420 		@Test void h01_longLiteralMatchIsFast() {
421 			var literal = "x".repeat(200_000);
422 			var a = McpUriTemplateMatcher.compile("file:///" + literal);
423 			assertTimeoutPreemptively(Duration.ofSeconds(2), () -> assertEquals(Map.of(), a.match("file:///" + literal)));
424 		}
425 
426 		@Test void h02_longReservedCaptureIsFast() {
427 			var value = "a".repeat(200_000);
428 			var a = McpUriTemplateMatcher.compile("file:///{+name}");
429 			assertTimeoutPreemptively(Duration.ofSeconds(2),
430 				() -> assertEquals(Map.of("name", value), a.match("file:///" + value)));
431 		}
432 
433 		@Test void h03_boundedCaptureWithDistantLiteralSuffixIsFast() {
434 			var value = "a".repeat(200_000);
435 			var a = McpUriTemplateMatcher.compile("{name}END");
436 			assertTimeoutPreemptively(Duration.ofSeconds(2), () -> assertEquals(Map.of("name", value), a.match(value + "END")));
437 		}
438 
439 		@Test void h04_repeatingAdjacentLiteralPatternIsFast() {
440 			var a = McpUriTemplateMatcher.compile("{+a}AB");
441 			var uri = "AB".repeat(100_000);
442 			assertTimeoutPreemptively(Duration.ofSeconds(2), () -> assertNotNull(a.match(uri)));
443 		}
444 
445 		@Test void h05_noMatchOnLongAdversarialInputIsFast() {
446 			var a = McpUriTemplateMatcher.compile("file:///{name}/fixed");
447 			var uri = "file:///" + "a".repeat(200_000);
448 			assertTimeoutPreemptively(Duration.ofSeconds(2), () -> assertNull(a.match(uri)));
449 		}
