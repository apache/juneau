35  	@Test void a01_toolSpecNeutralToWireMapsBothSchemas() {
36  		var a = McpSchema.of(JsonMap.of("type", "object"));
37  		var b = McpSchema.of(JsonMap.of("type", "object", "properties", JsonMap.of("x", JsonMap.of("type", "integer"))));
38  		var c = McpWire.toWire(new McpToolSpec().setName("t").setDescription("d").setInputSchema(a).setOutputSchema(b));
39  		assertEquals("t", c.getName());
40  		// Compare as parsed maps: the neutral JsonMap preserves insertion order while the v1 JsonSchema bean
41  		// serializes properties alphabetically, so an order-sensitive string compare would spuriously fail.
42  		assertEquals(Json.to(Json.of(a.toJsonMap()), JsonMap.class), Json.to(Json.of(c.getInputSchema()), JsonMap.class));
43  		assertEquals(Json.to(Json.of(b.toJsonMap()), JsonMap.class), Json.to(Json.of(c.getOutputSchema()), JsonMap.class));
44  	}

148 	@Test void e02_contentBlocks_allKindsBothDirections() {
149 		var text = McpWire.toWire(McpContentBlock.text("t"));
150 		assertEquals("t", ((TextContent) text).getText());
151 		var image = (ImageContent) McpWire.toWire(McpContentBlock.image("ZGF0YQ==", "image/png"));
152 		assertEquals("ZGF0YQ==", image.getData());
153 		assertEquals("image/png", image.getMimeType());
154 		var embedded = (EmbeddedResourceContent) McpWire.toWire(McpContentBlock.resource(McpResourceContents.text("u:1", "text/plain", "body")));
155 		assertEquals("u:1", ((TextResourceContents) embedded.getResource()).getUri());
156 
157 		assertEquals("t", McpWire.toNeutral(text).text());
158 		var image2 = McpWire.toNeutral(image);
159 		assertEquals(McpContentBlock.Kind.IMAGE, image2.kind());
160 		assertEquals("ZGF0YQ==", image2.data());
161 		assertEquals("image/png", image2.mimeType());
162 		var embedded2 = McpWire.toNeutral(embedded);
163 		assertEquals(McpContentBlock.Kind.RESOURCE, embedded2.kind());
164 		assertEquals("body", embedded2.resource().text());
165 	}

175

179 		var blob2 = McpWire.toNeutral(blob);
180 		assertEquals(McpResourceContents.Kind.BLOB, blob2.kind());
181 		assertEquals("AAEC", blob2.blob());
182 		assertEquals("application/octet-stream", blob2.mimeType());

201 		var bare = McpWire.toWire(new McpPromptSpec().setName("p"));
202 		assertNull(bare.getArguments());
203 		var full = McpWire.toWire(new McpPromptSpec().setName("p").setDescription("d")
204 			.setArguments(List.of(new McpPromptArgument().setName("a").setRequired(true))));
205 		assertEquals(1, full.getArguments().size());
206 		assertEquals("a", full.getArguments().get(0).getName());
207 
208 		assertNull(McpWire.toNeutral(new Prompt().setName("p")).getArguments());
209 		var neutral = McpWire.toNeutral(new Prompt().setName("p").setArguments(List.of(new PromptArgument().setName("a"))));
210 		assertEquals("a", neutral.getArguments().get(0).getName());
211 
212 		assertNull(McpWire.toWire(new McpPromptOutcome().setDescription("d")).getMessages());
213 		assertNull(McpWire.toNeutral(new GetPromptResult().setDescription("d")).getMessages());
214 		var msg = new McpPromptMessage().setRole(McpRole.ASSISTANT).setContent(McpContentBlock.text("hi"));
215 		var wire = McpWire.toWire(new McpPromptOutcome().setMessages(List.of(msg)));
216 		assertEquals(Role.ASSISTANT, wire.getMessages().get(0).getRole());
217 		var back = McpWire.toNeutral(wire);
218 		assertEquals(McpRole.ASSISTANT, back.getMessages().get(0).getRole());
219 		assertEquals("hi", back.getMessages().get(0).getContent().text());
