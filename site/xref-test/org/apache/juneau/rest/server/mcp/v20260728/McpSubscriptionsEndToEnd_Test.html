42  /**
43   * Proves Phase 1 (neutral broker) and Phase 3 (BeanStore wiring) compose correctly through a REAL v2
44   * {@link McpRevision} dispatch (not a fake): a tool handler registered on an ordinary
45   * {@link McpServerConfig} can fetch the injected {@link McpSubscriptionBroker}/{@link McpSubscriptions}
46   * beans from its {@link BeanStore} argument, and a change published by one handler invocation reaches a
47   * subscription registered by an earlier invocation sharing the same request-scoped {@link BeanStore}.
48   */
49  class McpSubscriptionsEndToEnd_Test {
50  
51  	private static Object validMeta() {
52  		return JsonMap.of(
53  			RequestMeta.KEY_PROTOCOL_VERSION, "2026-07-28",
54  			RequestMeta.KEY_CLIENT_CAPABILITIES, JsonMap.of());
55  	}
56  
57  	private static JsonRpcRequest toolCallRequest(String toolName, Map<String,Object> arguments) {
58  		var params = JsonMap.of("name", toolName, "arguments", arguments);
59  		params.put("_meta", validMeta());
60  		return new JsonRpcRequest().setJsonrpc(McpProtocol.JSON_RPC_2_0).setId(1).setMethod("tools/call")
61  			.setParams(params);
62  	}
63  
64  	private static Map<String,String> toolCallHeaders(String toolName) {
65  		return Map.of("Mcp-Method", "tools/call", "Mcp-Name", toolName);
66  	}
67  
68  	@SuppressWarnings({
69  		"resource" // Each .addBean(...) link returns 'this' (a Closeable) already owned by the enclosing try-with-resources; Eclipse JDT @Owning warning on the intermediate chain links is by design.
70  	})
71  	@Test void a01_toolHandler_publishReachesSubscriptionRegisteredThroughTheSameRequestScopedBroker() throws Exception {
72  		var broker = new BasicMcpSubscriptionBroker(4);
73  		var capturedSub = new AtomicReference<McpSubscription>();
74  
75  		var subscribeTool = new McpToolHandler() {
76  			@Override public McpToolSpec descriptor() { return new McpToolSpec().setName("subscribe"); }
77  			@Override public McpToolOutcome call(Map<String,Object> arguments, BeanStore ctx) {
78  				var b = ctx.getBean(McpSubscriptionBroker.class)
79  					.orElseThrow(() -> new IllegalStateException("McpSubscriptionBroker bean missing from request BeanStore"));
80  				var filter = new McpSubscriptionFilter(false, false, false, Set.of(String.valueOf(arguments.get("uri"))));
81  				capturedSub.set(b.register("sub-1", filter));
82  				return McpToolOutcome.text("subscribed");
83  			}
84  		};
85  		var touchTool = new McpToolHandler() {
86  			@Override public McpToolSpec descriptor() { return new McpToolSpec().setName("touch"); }
87  			@Override public McpToolOutcome call(Map<String,Object> arguments, BeanStore ctx) {
88  				var subs = ctx.getBean(McpSubscriptions.class)
89  					.orElseThrow(() -> new IllegalStateException("McpSubscriptions bean missing from request BeanStore"));
90  				subs.resourceUpdated(String.valueOf(arguments.get("uri")));
91  				return McpToolOutcome.text("published");
92  			}
93  		};
94  
95  		var config = new McpServerConfig().setName("test").setVersion("1.0.0")
96  			.addTool(subscribeTool).addTool(touchTool);
97  		var rev = new McpRevision(null, new McpCacheConfig(), null);
98  		try (var ctx = new BasicBeanStore()
99  				.addBean(McpSubscriptionBroker.class, broker)
100 				.addBean(McpSubscriptions.class, broker)) {
101 
102 			var subscribeResult = rev.dispatch(
103 				new McpExchange(toolCallRequest("subscribe", Map.of("uri", "file:///a.txt")), toolCallHeaders("subscribe")::get),
104 				config, ctx);
105 			assertInstanceOf(McpResponseResult.class, subscribeResult);
106 			assertNull(((McpResponseResult) subscribeResult).response().getError());
107 			var sub = capturedSub.get();
108 			assertNotNull(sub);
109 			assertEquals(1, broker.activeCount());
110 			try {
111 				var touchResult = rev.dispatch(
112 					new McpExchange(toolCallRequest("touch", Map.of("uri", "file:///a.txt")), toolCallHeaders("touch")::get),
113 					config, ctx);
114 				assertInstanceOf(McpResponseResult.class, touchResult);
115 				assertNull(((McpResponseResult) touchResult).response().getError());
116 
117 				var event = sub.take();
118 				assertEquals(McpChangeKind.RESOURCE_UPDATED, event.kind());
119 				assertEquals("file:///a.txt", event.resourceUri());
120 			} finally {
121 				sub.close();
122 			}
123 		}
124 	}
125 }
