48  /**
49   * Real-Spring-Boot proof for the MCP-on-Spring-Boot recipe: that {@link SpringMcpRestServlet} actually
50   * bridges the Spring {@code ApplicationContext} into an MCP tool handler's per-request {@code BeanStore},
51   * versus the plain servlet-subclass path ({@link McpRestServlet}) which does not.
52   *
53   * <p>
54   * Before this test, nothing anywhere exercised MCP + Spring Boot together: the recipe was documented
55   * (see {@code SpringMcpRestServlet}'s javadoc and the {@code juneau-docs} MCP recipes topic) but never
56   * proven against a real embedded-Tomcat Spring Boot context.
57   *
58   * <p>
59   * Boots a single Spring Boot context with embedded Tomcat on a random port (mirroring
60   * {@code BasicApiDocs_Springboot_Test}'s idiom) registering <b>two</b> independent MCP servlets side by
61   * side, both backed by the very same {@code ApplicationContext} and the very same {@code @Bean
62   * MySpringService} singleton:
63   * <ul>
64   * 	<li><b>Positive case</b> ({@link SpringMcpResource}, {@code /mcp-spring/mcp}): extends
65   * 		{@link SpringMcpRestServlet}. Its {@code whoAmI} tool handler calls
66   * 		{@code BeanStore.getBean(MySpringService.class)} and must resolve the real Spring-managed
67   * 		singleton &mdash; proven by echoing back a per-instance random marker that exactly matches the
68   * 		same instance {@code @Autowired} into this test class.
69   * 	<li><b>Negative/contrast case</b> ({@link PlainMcpResource}, {@code /mcp-plain/mcp}): extends
70   * 		{@link McpRestServlet} (the plain servlet-subclass path, no Spring base class). The identical
71   * 		{@code whoAmI} tool handler's {@code BeanStore.getBean(MySpringService.class)} call must come
72   * 		back empty, documenting the recipe's known boundary: subclassing {@link McpRestServlet} does not
73   * 		carry Spring DI, even when deployed inside the same Spring Boot app as a servlet that does.
74   * </ul>
75   *
76   * @since 10.0.0
77   */
78  @org.apache.juneau.testing.SpringbootTest
79  @SpringBootTest(classes = McpSpringBootIntegration_Test.TestApp.class, webEnvironment = WebEnvironment.RANDOM_PORT)
80  @DirtiesContext(classMode = DirtiesContext.ClassMode.AFTER_CLASS)
81  class McpSpringBootIntegration_Test {
82  
83  	/**
84  	 * A Spring-managed singleton service. Generates a random per-instance marker at construction so any
85  	 * two distinct instances are trivially distinguishable: a tool handler that echoes back a marker
86  	 * matching {@link #mySpringService}'s (the instance this test class itself has {@code @Autowired})
87  	 * can only be doing so because it resolved that exact same Spring-container-managed instance, not a
88  	 * coincidentally-equal one it constructed itself.
89  	 */
90  	static class MySpringService {
91  		private final String marker = "spring-bean-" + UUID.randomUUID();
92  		String getMarker() { return marker; }
93  	}
94  
95  	/** Sentinel returned by {@link #whoAmITool()} when {@code BeanStore.getBean(MySpringService.class)} finds nothing. */
96  	private static final String NO_BEAN = "NO_SPRING_BEAN_RESOLVED";
97  
98  	/**
99  	 * Shared tool logic registered identically on both {@link SpringMcpResource} and
100 	 * {@link PlainMcpResource} below: resolve {@link MySpringService} through the per-request
101 	 * {@link BeanStore} and echo back its marker, or {@link #NO_BEAN} if it could not be resolved. Using
102 	 * one shared handler implementation for both fixtures ensures the only variable between the two test
103 	 * cases is which servlet base class is doing the resolving, not any difference in the handler itself.
104 	 */
105 	private static McpToolHandler whoAmITool() {
106 		return McpToolHandler.of(
107 			new McpToolSpec().setName("whoAmI").setDescription("Resolves MySpringService via BeanStore.getBean(...)"),
108 			(arguments, ctx) -> McpToolOutcome.text(ctx.getBean(MySpringService.class).map(MySpringService::getMarker).orElse(NO_BEAN)));
109 	}
110 
111 	// -------- positive case: the MCP-on-Spring-Boot recipe ---------
112 
113 	/**
114 	 * The recipe under test. {@link SpringMcpRestServlet} extends {@link BasicSpringRestServlet}, whose
115 	 * {@code SpringRestServlet} ancestor has an {@code @Autowired ApplicationContext} field and a
116 	 * {@code @Bean createBeanStore(...)} factory method that wraps it in a {@code SpringBeanStore}; that
117 	 * store becomes this resource's {@code RestContext} bean store, and the MCP endpoint mixin's
118 	 * {@code handleMcpRequest} wraps a fresh per-request {@code BasicBeanStore} around it before invoking
119 	 * the tool handler &mdash; so {@code BeanStore.getBean(MySpringService.class)} chains all the way
120 	 * through to {@code ApplicationContext.getBeanProvider(MySpringService.class)}.
121 	 */
122 	@Rest(serializers = JsonSerializer.class, parsers = JsonParser.class, defaultAccept = "application/json")
123 	public static class SpringMcpResource extends SpringMcpRestServlet {
124 		private static final long serialVersionUID = 1L;
125 		@Override public McpServerConfig getMcpConfig() {
126 			return new McpServerConfig().addTool(whoAmITool());
127 		}
128 	}
129 
130 	// -------- negative/contrast case: the servlet-subclass path does NOT carry Spring DI ---------
131 
132 	/**
133 	 * Contrast fixture documenting the recipe's known boundary. {@link McpRestServlet} is a sibling
134 	 * {@code RestServlet} subclass to {@link BasicSpringRestServlet} (both ultimately extend
135 	 * {@code RestServlet}), <b>not</b> a subclass of it &mdash; so it never gets the
136 	 * {@code @Autowired ApplicationContext} field {@code SpringRestServlet} injects, and never publishes
137 	 * a {@code SpringBeanStore}-backed {@code createBeanStore()} bean. Its {@code RestContext} bean store
138 	 * is therefore a plain, Spring-oblivious store even though this servlet is registered in the very
139 	 * same Spring Boot context as {@link SpringMcpResource} (same {@code ApplicationContext}, same
140 	 * {@code @Bean MySpringService}). Its {@code whoAmI} tool handler's
141 	 * {@code BeanStore.getBean(MySpringService.class)} call therefore cannot see that bean.
142 	 */
143 	@Rest(serializers = JsonSerializer.class, parsers = JsonParser.class, defaultAccept = "application/json")
144 	public static class PlainMcpResource extends McpRestServlet {
145 		private static final long serialVersionUID = 1L;
146 		@Override protected McpServerConfig createMcpConfig() {
147 			return new McpServerConfig().addTool(whoAmITool());
148 		}
149 	}

153 	 * container URL mappings, plus the single shared {@code @Bean MySpringService} both endpoints (try
154 	 * to) resolve.

197 	private static Object validMeta() {
198 		return JsonMap.of(
199 			RequestMeta.KEY_PROTOCOL_VERSION, "2026-07-28",
200 			RequestMeta.KEY_CLIENT_INFO, JsonMap.of("name", "fixture-client", "version", "1.0"),
201 			RequestMeta.KEY_CLIENT_CAPABILITIES, JsonMap.of());
202 	}
203 
204 	private static String toolCallBody(String toolName) {
205 		var params = JsonMap.of("name", toolName, "arguments", JsonMap.of(), "_meta", validMeta());
206 		return Json.of(new JsonRpcRequest().setJsonrpc(McpProtocol.JSON_RPC_2_0).setId(1).setMethod("tools/call").setParams(params));
207 	}
208 
209 	/**
210 	 * Drives a real {@code tools/call} for the {@code whoAmI} tool over real HTTP against {@code path},
211 	 * and returns the text content of the (successful) result.
212 	 */
213 	private String callWhoAmI(String path) throws Exception {
214 		var req = HttpRequest.newBuilder()
215 			.uri(URI.create("http://localhost:" + port + path))
216 			.timeout(Duration.ofSeconds(10))
217 			.header("Content-Type", "application/json")
218 			.header("Accept", "application/json")
219 			.header("Mcp-Method", "tools/call")
220 			.header("Mcp-Name", "whoAmI")
221 			.POST(BodyPublishers.ofString(toolCallBody("whoAmI")))
222 			.build();
223 		var resp = HTTP.send(req, BodyHandlers.ofString());
224 		assertEquals(200, resp.statusCode(), () -> "Unexpected status; body=" + resp.body());
225 		var envelope = Json.to(resp.body(), JsonMap.class);
226 		assertNull(envelope.get("error"), () -> "Unexpected JSON-RPC error: " + envelope.get("error"));
227 		var result = (Map<?,?>) envelope.get("result");
228 		var content = (List<?>) result.get("content");
229 		var first = (Map<?,?>) content.get(0);
230 		return (String) first.get("text");
231 	}

233 	@Test
234 	void a01_springMcpRestServlet_toolHandlerResolvesRealSpringManagedBean() throws Exception {
235 		// SpringMcpRestServlet exposes the MCP endpoint via the McpEndpoint mixin's handleMcpRequest,
236 		// which is annotated @RestPost(path = "/mcp") -- relative to this resource's own (default,
237 		// root) @Rest path -- so the wire URL is basePath + "/mcp".
238 		var marker = callWhoAmI("/mcp-spring/mcp");
239 
240 		assertNotEquals(NO_BEAN, marker, "Positive case: the Spring-DI-aware endpoint must resolve MySpringService");
241 		assertEquals(mySpringService.getMarker(), marker,
242 			"The MCP tool handler must resolve the SAME Spring-managed singleton this test class has @Autowired -- "
243 				+ "proving BeanStore.getBean(...) really flowed through to Spring's ApplicationContext, not just "
244 				+ "some independently-constructed lookalike");
245 	}
246 
247 	@Test
248 	void b01_plainServletSubclassPath_doesNotCarrySpringDi() throws Exception {
249 		// Contrast/documentation case: same Spring Boot app, same ApplicationContext, same @Bean
250 		// MySpringService -- but this endpoint extends McpRestServlet (not SpringMcpRestServlet /
251 		// BasicSpringRestServlet), so its RestContext BeanStore was never bridged to Spring. This is a
252 		// known, documented boundary of the recipe, not a bug: BeanStore.getBean(MySpringService.class)
253 		// must come back empty here, in contrast to a01 above.
254 		// McpRestServlet's operation is declared @RestPost(path = "/") -- its servlet root, not "/mcp"
255 		// -- unlike the mixin path exercised in a01 above, so the wire URL is just basePath itself.
256 		var marker = callWhoAmI("/mcp-plain");
257 
258 		assertEquals(NO_BEAN, marker,
259 			"extends McpRestServlet (the plain servlet-subclass path, no Spring bridge) must NOT resolve the "
260 				+ "Spring-managed bean -- this is the recipe's documented boundary, not a regression");
261 	}
