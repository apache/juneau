19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.marshall.json.*;
24  import org.apache.juneau.rest.mock.classic.MockRestClient;
25  import org.apache.juneau.rest.server.*;
26  import org.apache.juneau.rest.server.mcp.McpServerConfig;
27  import org.apache.juneau.rest.server.mcp.McpSubscriptionBroker;
28  import org.apache.juneau.rest.server.mcp.McpSubscriptionFilter;
29  import org.apache.juneau.rest.server.servlet.*;
30  import org.junit.jupiter.api.*;
31  
32  @SuppressWarnings({
33  	"resource" // MockRestClient is a Closeable test helper; lifetime is bounded by the test method.
34  })
35  class McpSubscriptionBrokerMixin_Test {
36  
37  	@Rest(path = "/api", serializers = JsonSerializer.class, parsers = JsonParser.class, defaultAccept = "application/json")
38  	public static class A extends BasicRestServlet implements McpEndpoint {
39  		private static final long serialVersionUID = 1L;
40  		@Override public McpServerConfig getMcpConfig() { return new McpServerConfig(); }
41  	}
42  
43  	@Test void a01_mixinDefault_returnsNonNullBroker() {
44  		assertNotNull(new A().subscriptionBroker());
45  	}
46  
47  	@Test void a02_mixinDefault_isStableAcrossCallsOnSameInstance() {
48  		var a = new A();
49  		assertSame(a.subscriptionBroker(), a.subscriptionBroker(), "the mixin default broker must be memoized per-binding");
50  	}
51  
52  	/**
53  	 * Regression: the pre-consolidation mixin default accidentally shared its broker JVM-wide
54  	 * ({@code SharedSubscriptionBroker}) across every distinct endpoint instance. Post-consolidation, the
55  	 * broker is derived per-binding (memoized via {@link McpEndpointOptionsCache}): two distinct endpoint
56  	 * instances must resolve to two distinct brokers, proving the accidental JVM-wide sharing is gone.
57  	 */
58  	@Test void a03_mixinDefault_distinctAcrossDistinctEndpointInstances() {
59  		var broker1 = new A().subscriptionBroker();
60  		var broker2 = new A().subscriptionBroker();
61  		assertNotSame(broker1, broker2, "two distinct endpoint instances must NOT share the same broker (no JVM-wide sharing)");
62  	}
63  
64  	@Test void a04_mixinDefault_subscriptionRegisteredOnOneInstanceNotVisibleOnAnother() {
65  		McpSubscriptionBroker broker1 = new A().subscriptionBroker();
66  		McpSubscriptionBroker broker2 = new A().subscriptionBroker();
67  		var before = broker2.activeCount();
68  		var sub = broker1.register("mixin-isolated-test-s1", new McpSubscriptionFilter(true, true, true, Set.of()));
69  		assertEquals(before, broker2.activeCount(), "a subscription registered via one mixin instance's "
70  			+ "broker must NOT be visible via another instance's broker, since each binding now derives its own");
71  		sub.close();
72  	}
73  
74  	@Test void a05_mixinDefault_subscriptionsConfigIsStableAcrossCallsOnSameInstance() {
75  		var a = new A();
76  		var config1 = a.mcpOptionsBean().getSubscriptions();
77  		var config2 = a.mcpOptionsBean().getSubscriptions();
78  		assertSame(config1, config2, "McpOptions (and its nested subscriptions config) is memoized per-binding via "
79  			+ "the mcpOptionsBean()/McpEndpointOptionsCache seam, so repeated access on the same endpoint instance "
80  			+ "returns the identical instance, even though the raw getMcpOptions() override point may return a "
81  			+ "fresh instance per call");
82  	}
83  
84  	@Test void a06_mcpOptionsBean_isAnnotatedAndMemoizedPerInstance() throws Exception {
85  		var a = new A();
86  		var m = McpEndpoint.class.getMethod("mcpOptionsBean");
87  		assertTrue(m.isAnnotationPresent(org.apache.juneau.commons.inject.Bean.class),
88  			"mcpOptionsBean() must be @Bean-annotated so the RestContext bean store discovers it, "
89  			+ "the same mechanism already used by mcpTraceContextExtractor()");
90  		assertSame(a.mcpOptionsBean(), a.mcpOptionsBean(),
91  			"mcpOptionsBean() must return the SAME memoized McpOptions on every call for a given endpoint instance");
92  		assertNotSame(a.mcpOptionsBean(), new A().mcpOptionsBean(),
93  			"a distinct endpoint instance must resolve to a distinct memoized McpOptions");
94  	}
95  
96  	/**
97  	 * IMPORTANT-5 code-review regression: {@code a06} above only checks the {@code @Bean} annotation
98  	 * reflectively, which would NOT catch a future break in interface-default {@code @Bean} discovery (for
99  	 * example, if the {@code RestContext} bean-collection walk stopped considering default methods on mixin
100 	 * interfaces). This test proves the wiring end-to-end through a real {@code RestContext}: the resource's
101 	 * bean store must resolve {@link McpOptions} to the exact SAME instance {@link McpEndpoint#mcpOptionsBean()}
102 	 * returns, so {@code subscriptions/listen} dispatch (which reads {@code McpOptions} from the bean store,
103 	 * not by calling {@code mcpOptionsBean()} directly) never silently falls back to defaults.
104 	 */
105 	@Test void a07_mcpOptionsBean_isPublishedInRestContextBeanStore_sameInstanceAsMemoized() {
106 		var a = new A();
107 		MockRestClient.create(a).json().contentType("application/json").accept("application/json").build();
108 		var fromBeanStore = a.getContext().getBeanStore().getBean(McpOptions.class)
109 			.orElseThrow(() -> new AssertionError("McpOptions bean not found in RestContext's bean store"));
110 		assertSame(a.mcpOptionsBean(), fromBeanStore,
111 			"the RestContext bean store's published McpOptions must be the SAME instance as the mixin's memoized "
112 			+ "mcpOptionsBean(), proving the @Bean interface-default discovery wires the memoized instance end-to-end");
113 	}
114 }
