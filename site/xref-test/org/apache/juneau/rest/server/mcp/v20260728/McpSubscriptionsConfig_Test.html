17  package org.apache.juneau.rest.server.mcp.v20260728;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.junit.jupiter.api.Test;
22  
23  class McpSubscriptionsConfig_Test {
24  
25  	@Test void a01_defaultsMatchDocumentedConstants() {
26  		var a = new McpSubscriptionsConfig();
27  		assertEquals(256, McpSubscriptionsConfig.DEFAULT_MAX_CONCURRENT_SUBSCRIPTIONS);
28  		assertEquals(1024, McpSubscriptionsConfig.DEFAULT_QUEUE_SIZE);
29  		assertEquals(15_000L, McpSubscriptionsConfig.DEFAULT_HEARTBEAT_INTERVAL_MS);
30  		assertEquals(0L, McpSubscriptionsConfig.DEFAULT_IDLE_TIMEOUT_MS);
31  		assertEquals(McpSubscriptionsConfig.DEFAULT_MAX_CONCURRENT_SUBSCRIPTIONS, a.getMaxConcurrentSubscriptions());
32  		assertEquals(McpSubscriptionsConfig.DEFAULT_QUEUE_SIZE, a.getQueueSize());
33  		assertEquals(McpSubscriptionsConfig.DEFAULT_HEARTBEAT_INTERVAL_MS, a.getHeartbeatIntervalMs());
34  		assertEquals(McpSubscriptionsConfig.DEFAULT_IDLE_TIMEOUT_MS, a.getIdleTimeoutMs());
35  	}
36  
37  	@Test void a02_setMaxConcurrentSubscriptionsZeroThrows() {
38  		var config = new McpSubscriptionsConfig();
39  		var e = assertThrows(IllegalArgumentException.class,
40  			() -> config.setMaxConcurrentSubscriptions(0));
41  		assertEquals("maxConcurrentSubscriptions 0 must be > 0", e.getMessage());
42  	}
43  
44  	@Test void a03_setQueueSizeNegativeThrows() {
45  		var config = new McpSubscriptionsConfig();
46  		var e = assertThrows(IllegalArgumentException.class, () -> config.setQueueSize(-1));
47  		assertEquals("queueSize -1 must be > 0", e.getMessage());
48  	}
49  
50  	@Test void a04_setHeartbeatIntervalMsZeroThrows() {
51  		var config = new McpSubscriptionsConfig();
52  		var e = assertThrows(IllegalArgumentException.class,
53  			() -> config.setHeartbeatIntervalMs(0));
54  		assertEquals("heartbeatIntervalMs 0 must be > 0", e.getMessage());
55  	}
56  
57  	@Test void a05_setIdleTimeoutMsNegativeThrows() {
58  		var config = new McpSubscriptionsConfig();
59  		var e = assertThrows(IllegalArgumentException.class, () -> config.setIdleTimeoutMs(-1));
60  		assertEquals("idleTimeoutMs -1 must be >= 0", e.getMessage());
61  	}
62  
63  	@Test void a06_idleTimeoutMsZeroIsAllowed_meansDisabled() {
64  		assertDoesNotThrow(() -> new McpSubscriptionsConfig().setIdleTimeoutMs(0));
65  	}
66  
67  	@Test void a07_validChainRoundTripsThroughGetters() {
68  		var a = new McpSubscriptionsConfig()
69  			.setMaxConcurrentSubscriptions(10).setQueueSize(64).setHeartbeatIntervalMs(5_000L).setIdleTimeoutMs(30_000L);
70  		assertEquals(10, a.getMaxConcurrentSubscriptions());
71  		assertEquals(64, a.getQueueSize());
72  		assertEquals(5_000L, a.getHeartbeatIntervalMs());
73  		assertEquals(30_000L, a.getIdleTimeoutMs());
74  	}
75  }
