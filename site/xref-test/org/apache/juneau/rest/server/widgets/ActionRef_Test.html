17  package org.apache.juneau.rest.server.widgets;
18  
19  import static org.apache.juneau.test.bct.BctAssertions.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import org.apache.juneau.*;
23  import org.junit.jupiter.api.*;
24  
25  /**
26   * {@link ActionRef} construction and enabled-rule validation.
27   */
28  class ActionRef_Test extends TestBase {
29  
30  	@Test void a01_of_defaults() {
31  		var a = ActionRef.of("approve");
32  		assertEquals("approve", a.id);
33  		assertEquals(ActionRef.Emphasis.SECONDARY, a.emphasis);
34  		assertNull(a.enabledWhen);
35  	}
36  
37  	@Test void a02_of_blankOrNull_rejected() {
38  		assertThrows(IllegalArgumentException.class, () -> ActionRef.of(null));
39  		assertThrows(IllegalArgumentException.class, () -> ActionRef.of("  "));
40  	}
41  
42  	@Test void a03_emphasis_setsAndReturnsThis() {
43  		var a = ActionRef.of("x");
44  		assertSame(a, a.emphasis(ActionRef.Emphasis.PRIMARY));
45  		assertEquals(ActionRef.Emphasis.PRIMARY, a.emphasis);
46  	}
47  
48  	@Test void a04_enabledWhen_valueForm_recordsRulesInOrder() {
49  		var a = ActionRef.of("x")
50  			.enabledWhen("status", Op.EQ, "OPEN", "Must be open")
51  			.enabledWhen("kind", Op.NE, "legacy", "Not for legacy");
52  		assertSize(2, a.enabledWhen);
53  		assertBean(a.enabledWhen.get(0), "field,op,value,reason", "status,EQ,OPEN,Must be open");
54  		assertBean(a.enabledWhen.get(1), "field,op,value,reason", "kind,NE,legacy,Not for legacy");
55  	}
56  
57  	@Test void a05_enabledWhen_presenceForm_recordsRuleWithNullValue() {
58  		var a = ActionRef.of("x").enabledWhen("owner", Op.PRESENT, "Needs an owner");
59  		assertSize(1, a.enabledWhen);
60  		assertBean(a.enabledWhen.get(0), "field,op,reason", "owner,PRESENT,Needs an owner");
61  		assertNull(a.enabledWhen.get(0).value);
62  		a.enabledWhen("owner", Op.ABSENT, "Must be unowned");
63  		assertEquals(Op.ABSENT, a.enabledWhen.get(1).op);
64  	}
65  
66  	@Test void a06_enabledWhen_valueForm_badOp_rejected() {
67  		var a = ActionRef.of("x");
68  		assertThrows(IllegalArgumentException.class, () -> a.enabledWhen("f", Op.IN, "v", "r"));
69  		assertThrows(IllegalArgumentException.class, () -> a.enabledWhen("f", Op.CONTAINS, "v", "r"));
70  		assertThrows(IllegalArgumentException.class, () -> a.enabledWhen("f", Op.PRESENT, "v", "r"));
71  		assertThrows(IllegalArgumentException.class, () -> a.enabledWhen("f", Op.ABSENT, "v", "r"));
72  		assertThrows(IllegalArgumentException.class, () -> a.enabledWhen("f", null, "v", "r"));
73  	}
74  
75  	@Test void a07_enabledWhen_valueForm_nullValue_rejected() {
76  		var a = ActionRef.of("x");
77  		assertThrows(IllegalArgumentException.class, () -> a.enabledWhen("f", Op.EQ, null, "r"));
78  	}
79  
80  	@Test void a08_enabledWhen_presenceForm_badOp_rejected() {
81  		var a = ActionRef.of("x");
82  		assertThrows(IllegalArgumentException.class, () -> a.enabledWhen("f", Op.EQ, "r"));
83  		assertThrows(IllegalArgumentException.class, () -> a.enabledWhen("f", Op.IN, "r"));
84  		assertThrows(IllegalArgumentException.class, () -> a.enabledWhen("f", null, "r"));
85  	}
86  
87  	@Test void a09_enabledWhen_blankFieldOrReason_rejected() {
88  		var a = ActionRef.of("x");
89  		assertThrows(IllegalArgumentException.class, () -> a.enabledWhen(null, Op.EQ, "v", "r"));
90  		assertThrows(IllegalArgumentException.class, () -> a.enabledWhen(" ", Op.PRESENT, "r"));
91  		assertThrows(IllegalArgumentException.class, () -> a.enabledWhen("f", Op.EQ, "v", null));
92  		assertThrows(IllegalArgumentException.class, () -> a.enabledWhen("f", Op.PRESENT, " "));
93  		assertNull(a.enabledWhen);
94  	}
95  }
