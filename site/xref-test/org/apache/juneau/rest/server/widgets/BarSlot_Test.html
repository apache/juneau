17  package org.apache.juneau.rest.server.widgets;
18  
19  import static org.apache.juneau.test.bct.BctAssertions.assertSize;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.util.*;
23  
24  import org.apache.juneau.*;
25  import org.junit.jupiter.api.*;
26  
27  /**
28   * {@link BarSlot} bean contract, the sealed {@link BarWidget} permit set, and fail-closed
29   * {@link BarSlot#validate()} branches.
30   */
31  class BarSlot_Test extends TestBase {
32  
33  	@Test void a01_contractVersion_isOne() {
34  		assertEquals("1", BarSlot.CONTRACT_VERSION);
35  	}
36  
37  	@Test void a02_builder_roundTrip() {
38  		var b = BarSlot.create("bar-main")
39  			.widgets(
40  				BarText.of("ctx", "Editing"),
41  				BarBadge.of("changePending").label("change pending").badge(Badge.count(7).tone(StatusTone.WARNING)))
42  			.refreshUrl("/bar/counts");
43  		assertEquals("bar-main", b.id);
44  		assertSize(2, b.widgets);
45  		b.validate();
46  	}
47  
48  	@Test void a03_sealed_permitsOnlyBarBadgeAndBarText() {
49  		assertTrue(BarWidget.class.isSealed());
50  		var permitted = new HashSet<Class<?>>(Arrays.asList(BarWidget.class.getPermittedSubclasses()));
51  		assertEquals(Set.of(BarBadge.class, BarText.class), permitted);
52  	}
53  
54  	@Test void a04_blankId_rejected() {
55  		var b = BarSlot.create("  ").widgets(BarText.of("t", "x"));
56  		assertThrows(IllegalArgumentException.class, b::validate);
57  	}
58  
59  	@Test void a05_emptySlot_rejected() {
60  		var b1 = BarSlot.create("b");
61  		assertThrows(IllegalArgumentException.class, b1::validate);
62  		var b2 = BarSlot.create("b").widgets();
63  		assertThrows(IllegalArgumentException.class, b2::validate);
64  	}
65  
66  	@Test void a06_duplicateWidgetId_rejected() {
67  		var b = BarSlot.create("b").widgets(BarText.of("dup", "a"), BarText.of("dup", "b"));
68  		var e = assertThrows(IllegalArgumentException.class, b::validate);
69  		assertTrue(e.getMessage().contains("dup"), e::getMessage);
70  	}
71  
72  	@Test void a07_barText_blankText_rejected() {
73  		var b = BarSlot.create("b").widgets(BarText.of("t", "  "));
74  		assertThrows(IllegalArgumentException.class, b::validate);
75  	}
76  
77  	@Test void a08_barBadge_fansOutToBadge() {
78  		var b = BarSlot.create("b").widgets(BarBadge.of("x").badge(Badge.count(-1)));
79  		assertThrows(IllegalArgumentException.class, b::validate);
80  	}
81  
82  	@Test void a09_badContractVersion_rejected() {
83  		var b = BarSlot.create("b").widgets(BarText.of("t", "x"));
84  		b.contractVersion = "9";
85  		assertThrows(IllegalArgumentException.class, b::validate);
86  	}
87  
88  	@Test void a10_refreshUrl_mustBeSameOrigin() {
89  		var b = BarSlot.create("b").widgets(BarText.of("t", "x")).refreshUrl("http://evil/x");
90  		assertThrows(IllegalArgumentException.class, b::validate);
91  	}
92  
93  	@Test void a11_isAWidget() {
94  		assertTrue(Widget.class.isAssignableFrom(BarSlot.class));
95  	}
96  
97  	@Test void a12_nullWidget_rejected() {
98  		var b = BarSlot.create("b").widgets(BarText.of("t", "x"), null);
99  		var e = assertThrows(IllegalArgumentException.class, b::validate);
100 		assertTrue(e.getMessage().contains("must not be null"), e::getMessage);
101 	}
102 
103 	@Test void a13_barBadge_blankId_rejected() {
104 		for (var id : new String[]{null, "  "}) {
105 			var b = BarSlot.create("b").widgets(BarBadge.of(id));
106 			var e = assertThrows(IllegalArgumentException.class, b::validate);
107 			assertTrue(e.getMessage().contains("BarBadge id"), e::getMessage);
108 		}
109 	}
110 
111 	@Test void a14_barText_blankId_rejected() {
112 		for (var id : new String[]{null, "  "}) {
113 			var b = BarSlot.create("b").widgets(BarText.of(id, "x"));
114 			var e = assertThrows(IllegalArgumentException.class, b::validate);
115 			assertTrue(e.getMessage().contains("BarText id"), e::getMessage);
116 		}
117 	}
118 
119 	@Test void a15_barText_nullText_rejected() {
120 		var b = BarSlot.create("b").widgets(BarText.of("t", null));
121 		var e = assertThrows(IllegalArgumentException.class, b::validate);
122 		assertTrue(e.getMessage().contains("text must not be"), e::getMessage);
123 	}
124 
125 	@Test void a16_barBadge_withoutBadge_valid() {
126 		BarSlot.create("b").widgets(BarBadge.of("x").label("pending")).validate();
127 	}
128 
129 	@Test void a17_sameOriginRefreshUrl_valid() {
130 		BarSlot.create("b").widgets(BarText.of("t", "x")).refreshUrl("/bar/counts").validate();
131 	}
132 
133 	@Test void a18_nullId_rejected() {
134 		var b = BarSlot.create(null).widgets(BarText.of("t", "x"));
135 		assertThrows(IllegalArgumentException.class, b::validate);
136 	}
137 }
