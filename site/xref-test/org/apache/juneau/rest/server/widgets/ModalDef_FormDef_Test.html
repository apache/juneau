29   * Contract test for the declarative modal ({@link ModalDef}) + form ({@link FormDef}) payload the modal-open
30   * confirmation fetch returns. This pins the bean half of that contract; the runtime half is
31   * pinned separately in the views module.
32   *
33   * <p>
34   * The confirmation body is typed fields painted with textContent (never innerHTML); this test pins that field
35   * shape, the typed FormDef inputs the client paints (never template markup), plus the omit-when-unset rule for
36   * the optional form and idempotency key.
37   */

88  	@Test void a06_selfTargeted_isAnOptInAndIsOtherwiseOmitted() {
89  		// The row-less dialog seam's opt-in: it tells the runtime to submit the key's OWN value as the target,
90  		// because a dialog with no row has no artifact id to bind to at modal-open.
91  		var unset = ModalDef.create("t").idempotencyKey("k");
92  		assertNull(unset.selfTargeted);
93  		assertFalse(Json.of(unset).contains("selfTargeted"), () -> Json.of(unset));
94  
95  		// Explicit false CLEARS rather than serializing a false: an un-opted-in dialog's payload stays byte-identical
96  		// to one that never set this property, so the property doesn't bump the contract version.
97  		var off = ModalDef.create("t").idempotencyKey("k").selfTargeted(false);
98  		assertNull(off.selfTargeted);
99  		assertFalse(Json.of(off).contains("selfTargeted"), () -> Json.of(off));
100 
101 		var on = ModalDef.create("t").idempotencyKey("k").selfTargeted(true);
102 		assertEquals(Boolean.TRUE, on.selfTargeted);
103 		assertTrue(Json.of(on).contains("\"selfTargeted\":true"), () -> Json.of(on));
104 	}

449 	 * The accepted cost of construction-time stamping: version-presence is guaranteed,
450 	 * structural validity is NOT.  A malformed form that never reached checked() now serializes as contract-VALID
451 	 * and the client will render it, where before the missing version refused it.  This test exists so that
452 	 * behavior is a recorded decision rather than a discovery.  See the serving-path
453 	 * Widget.validate() hook that closes it.
454 	 */
455 	@Test void e14_malformedUncheckedForm_serializesAsContractValid() {
456 		var badForm = FormDef.create().field(FormDef.Input.of("sev", "Severity", "select"));
457 		var modal = ModalDef.create("Pick one").form(badForm);
458 		var json = Json.of(modal);
459 		assertTrue(json.contains("\"contractVersion\":\"2\""), json);
460 		var formMap = (Map<String,Object>) Json.to(json, Map.class).get("form");
461 		assertEquals("2", formMap.get("contractVersion"));
462 	}
