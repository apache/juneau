32  /**
33   * REST-boundary tests for {@link WidgetsMixin.WidgetValidationProcessor} &mdash; the
34   * {@code ServiceLoader}-registered {@link org.apache.juneau.rest.server.processor.ResponseProcessor
35   * ResponseProcessor} that calls {@link Widget#validate()} on any REST response content that is a {@link Widget},
36   * giving the JSON serving path ({@link ModalDef}/{@link FormDef}) the same fail-closed gate the
37   * {@code juneau-rest-server-views} HTML emitters already give every {@code Widget} they consume.
38   *
39   * <p>
40   * This is the REST-boundary inverse of the serialization tests in {@link ModalDef_FormDef_Test}: those pin that a
41   * malformed, never-{@code checked()} modal/form serializes successfully with the current contract version and does not
42   * throw. Here, served over a real REST response instead of a bare {@link Json#of(Object) Json.of(...)} call, the
43   * same malformed shapes now fail loud with a 500 &mdash; which is this item's entire reason to exist (see
44   * {@code h2} below for the pin that {@link Json#of(Object) Json.of(...)} itself is deliberately untouched).
45   *
46   * <p>
47   * No {@code @Rest(mixins=WidgetsMixin.class)} composition is required on any host below except {@code HostC}
48   * (the double-registration guard): the validator is discovered via {@code ServiceLoader} from this module's
49   * {@code META-INF/services} entry on the test classpath, with no consumer opt-in, which is the point of the
50   * design.
51   */

53  	"java:S125", // Comments are explanatory; they are not commented-out code.
54  	"resource" // Closeable MockRestClient fixtures held for the class lifetime.
55  })

63  	/** Row: a {@code select} input with no options. */
64  	private static ModalDef noOptionSelect() {
65  		return ModalDef.create("Ack?").form(
66  			FormDef.create().field(FormDef.Input.of("sev", "Severity", "select")));
67  	}

69  	/** Row: two fields sharing one name - collide on submit. */
70  	private static ModalDef duplicateFieldName() {
71  		return ModalDef.create("Ack?").form(
72  			FormDef.create()
73  				.field(FormDef.Input.of("x", "First", "text"))
74  				.field(FormDef.Input.of("x", "Second", "text")));
75  	}

77  	/** Row: both {@code fields} and {@code sections} declared - mutually exclusive. */
78  	private static ModalDef fieldsAndSections() {
79  		var form = FormDef.create().field(FormDef.Input.of("a", "A", "text"));
80  		form.sections = List.of(FormDef.Section.of("s1", "S1").field(FormDef.Input.of("b", "B", "text")));
81  		return ModalDef.create("Ack?").form(form);
82  	}
83  
84  	/** Row: a non-compiling client validation pattern. */
85  	private static ModalDef nonCompilingPattern() {
86  		return ModalDef.create("Ack?").form(
87  			FormDef.create().field(FormDef.Input.of("p", "P", "text").pattern("[")));
88  	}

90  	/** Row: a blank modal title, bypassing {@link ModalDef#create(String)}'s own blank check. */
91  	private static ModalDef blankTitle() {
92  		var m = new ModalDef();
93  		m.title = "   ";
94  		return m;
95  	}

97  	/** Row (acceptable-today, becomes a 500 under this hook): a confirmation field {@code kind} off the allowlist. */

99  		var m = ModalDef.create("Ack?");
100 		var f = new ModalDef.Field();
101 		f.label = "Severity";
102 		f.kind = "not-a-real-kind";
103 		m.fields = List.of(f);
104 		return m;
105 	}

107 	/** Row (stays acceptable): an {@code action} input whose {@code actionId} is non-blank but resolves to nothing. */
108 	private static ModalDef unknownActionId() {
109 		return ModalDef.create("Ack?").form(
110 			FormDef.create().field(FormDef.Input.of("go", "Go", "action").actionId("no-such-action")));
111 	}

113 	private static ModalDef validNoForm() {
114 		return ModalDef.create("Ack?");
115 	}

197 	/** A {@link Widget} whose {@code validate()} counts calls instead of checking anything - always "valid". */
198 	static final class CountingWidget implements Widget {
199 		static final AtomicInteger COUNT = new AtomicInteger();
200 		@Override public void validate() { COUNT.incrementAndGet(); }
201 	}
202 
203 	@Rest(mixins=WidgetsMixin.class)
204 	public static class HostC extends BasicRestServlet {
205 		private static final long serialVersionUID = 1L;
206 		@RestGet(path="/counting") public CountingWidget counting() { return new CountingWidget(); }
207 		@RestGet(path="/plain") public String plain() { return "plain"; }
208 		@RestGet(path="/map") public Map<String,Object> map() { return Map.of("a", 1); }
209 		@RestGet(path="/list") public List<String> list() { return List.of("x", "y"); }
210 	}
211 
212 	private static final MockRestClient C = MockRestClient.buildLax(HostC.class);
213 
214 	@Test void c1_registeredByServiceLoaderAndMixin_validatesExactlyOnce_notTwice() throws Exception {
215 		// HostC composes WidgetsMixin, so both the ServiceLoader entry and the mixin's own
216 		// responseProcessors + mergeResponseProcessorsIntoHost=true declaration are candidates;
217 		// ResponseProcessorList.Builder de-duplicates by class, so validate() must run exactly once.
218 		var before = CountingWidget.COUNT.get();
219 		C.get("/counting").run().assertStatus(200);
220 		assertEquals(before + 1, CountingWidget.COUNT.get(), "validate() must run exactly once, not once per registration");
221 	}
222 
223 	@Test void c2_nonWidgetContent_servesUnaffected() throws Exception {
224 		C.get("/plain").accept("application/json").run().assertStatus(200).assertContent().asString().is("\"plain\"");
225 		C.get("/map").accept("application/json").run().assertStatus(200).assertContent().asString().isContains("\"a\":1");
226 		C.get("/list").accept("application/json").run().assertStatus(200).assertContent().asString().isContains("\"x\"", "\"y\"");
227 	}

268 	@Test void h1_checkedAndUnchecked_validModal_serveByteIdentical() throws Exception {
269 		// checked() re-validates (the hook validates again); unchecked relies solely on the hook.  Both are valid,
270 		// so the served bytes must be identical - the assertion that keeps ".checked()-vs-hook" drift impossible.
271 		var checked = H.get("/checked").run().assertStatus(200).getContent().asString();
272 		var unchecked = H.get("/unchecked").run().assertStatus(200).getContent().asString();
273 		assertEquals(checked, unchecked);
274 	}
