28  /**
29   * Bean contract for the in-dialog <b>result host</b> opt-in ({@link ModalDef#keepOpenOnSubmit}), the dialog-scoped
30   * <b>child-action catalog</b> ({@link ModalDef#childActions} / {@link ModalDef.ChildAction}), and the confirmation
31   * field <b>display kind</b> ({@link ModalDef.Field#kind} / {@link ModalDef.Field#code(String,String)}).
32   *
33   * <p>
34   * All three are additive to an already-served payload, so the three properties this pins hardest are the ones that
35   * make that claim true rather than merely intended:
36   * <ul class='spaced-list'>
37   * 	<li><b>Appended, never interleaved.</b> The new keys land after the frozen prefix, in {@code @BeanType} order.
38   * 	<li><b>Omitted when unset.</b> An un-opted-in modal's payload is byte-identical to one that never set these
39   * 		keys &mdash; which is what lets an older runtime keep reading it, and lets the golden fixtures in
40   * 		{@code ModalDef_FormDef_Test} / {@code ModalDef_BarSlot_Test} stay untouched.
41   * 	<li><b>No {@link ModalDef#CONTRACT_VERSION} bump.</b> Additive means the version does not move.
42   * </ul>
43   *
44   * <p>
45   * The remaining group is the fail-closed serve-time refusals: a payload that would produce a silently-wrong or
46   * silently-empty client surface is rejected by {@link ModalDef#validate()} at serve time, where the author sees it,
47   * rather than degrading quietly in a browser.
48   */

214 	// A raw list, bypassing the factory - the shape a bean-deserialized payload can actually arrive in.
215 	private static ModalDef withRawChildren(ModalDef.ChildAction... children) {
216 		var m = ModalDef.create("Step 1");
217 		m.childActions = list(children);
218 		return m;
219 	}
220 
221 	private static ModalDef.ChildAction raw(String id, String label) {
222 		var c = new ModalDef.ChildAction();
223 		c.id = id;
224 		c.label = label;
225 		return c;
226 	}
227 
228 	@Test void d01_blankIdIsRefused() {
229 		var e = assertThrows(IllegalArgumentException.class, () -> withRawChildren(raw("  ", "Review")).validate());
230 		assertTrue(e.getMessage().contains("id"), e::getMessage);
231 		assertThrows(IllegalArgumentException.class, () -> withRawChildren(raw(null, "Review")).validate());
232 	}
233 
234 	@Test void d02_blankLabelIsRefused() {
235 		// A child with no label paints a button with nothing on it - unclickable in practice, and invisible in a
236 		// screen reader.  Refused where the author can see it.
237 		var e = assertThrows(IllegalArgumentException.class, () -> withRawChildren(raw("review", "  ")).validate());
238 		assertTrue(e.getMessage().contains("label"), e::getMessage);
239 		assertThrows(IllegalArgumentException.class, () -> withRawChildren(raw("review", null)).validate());
240 	}
241 
242 	@Test void d03_duplicateIdIsRefused() {
243 		// The resolver returns the FIRST match, so a duplicate id means one of the two declared children is
244 		// unreachable with no signal anywhere.
245 		var e = assertThrows(IllegalArgumentException.class,
246 			() -> withRawChildren(raw("review", "Review"), raw("review", "Review again")).validate());
247 		assertTrue(e.getMessage().contains("more than once"), e::getMessage);
248 	}
