30  /**
31   * The mechanical guards on this module's boundaries &mdash; the layering rules a reviewer can miss and the compiler
32   * cannot state.  These were previously spread across individual bean tests; they are consolidated here so there is
33   * one place that answers "what may this module depend on".
34   *
35   * <h5 class='section'>The rules</h5>
36   * <ul>
37   * 	<li>The widgets/views edge is one-way: views composes the widget beans, and widgets never reaches back.  Shared
38   * 		behaviour the two modules both need lives further down the stack instead ({@code juneau-commons},
39   * 		{@code juneau-rest-server}), never in a widgets&rarr;views dependency.
40   * 	<li>The beans stay independent of the REST serving framework and of the HTML5 bean DOM: they are data plus
41   * 		{@code validate()}, and the markup that renders them is emitted elsewhere.  {@link WidgetsMixin} is the single
42   * 		exception, because a mixin is by definition a serving-framework type.
43   * </ul>
44   */
45  class Widgets_ModuleBoundary_Test extends TestBase {
46  
47  	/** The package this module must never reference, assembled from segments so this guard's own source is not a hit. */
48  	private static final String VIEWS_PACKAGE = String.join(".", "org", "apache", "juneau", "rest", "server", "views");
49  
50  	/** The one source file in this module allowed to reference the REST serving framework. */
51  	private static final String MIXIN_FILE = "WidgetsMixin.java";
52  
53  	/** Resolves a source root relative to the module basedir (Surefire's working directory). */
54  	private static Path sourceRoot(String relative) {
55  		var p = Path.of(relative);
56  		assertTrue(Files.isDirectory(p), () -> "Source root not found at '" + p.toAbsolutePath() + "'; the test expects the module basedir as its working directory.");
57  		return p;
58  	}
59  
60  	private static List<Path> javaFilesUnder(String relative) throws IOException {
61  		try (var s = Files.walk(sourceRoot(relative))) {
62  			return s.filter(p -> p.toString().endsWith(".java")).sorted().toList();
63  		}
64  	}
65  
66  	/** Collects every source file under the given roots whose text matches, reported as a path list. */
67  	private static List<String> offenders(Predicate<String> matches, String... roots) throws IOException {
68  		var out = new ArrayList<String>();
69  		for (var root : roots)
70  			for (var f : javaFilesUnder(root))
71  				if (matches.test(Files.readString(f)))
72  					out.add(f.toString());
73  		return out;
74  	}

77  	// a) The widgets -> views edge does not exist, in either the sources or the classpath
78  	//------------------------------------------------------------------------------------------------------------------
79  
80  	@Test void a01_noSourceFileInThisModuleReferencesTheViewsPackage() throws IOException {
81  		// Any reference, not just an import: a fully-qualified name in a signature or a javadoc link is the same
82  		// coupling, and would break the day views is not on the classpath.
83  		var o = offenders(t -> t.contains(VIEWS_PACKAGE), "src/main/java", "src/test/java");
84  		assertTrue(o.isEmpty(), () -> "These files reference '" + VIEWS_PACKAGE + "', which this module must not depend on: " + o);
85  	}
86  
87  	@Test void a02_theViewsPackageIsNotEvenOnThisModulesClasspath() {
88  		// Belt to a01's braces: an import cannot be introduced by accident if the type is not resolvable here.
89  		assertThrows(ClassNotFoundException.class, () -> Class.forName(VIEWS_PACKAGE + ".ViewsMixin"));
90  	}

92  	//------------------------------------------------------------------------------------------------------------------
93  	// b) The beans stay data-only; the mixin is the module's only serving-framework type
94  	//------------------------------------------------------------------------------------------------------------------
95  
96  	@Test void b01_onlyTheMixinReferencesTheRestServingFramework() throws IOException {
97  		// This module does depend on juneau-rest-server, but solely so it can declare a mixin.  Letting that reach
98  		// the beans would make the shared widget contracts unusable outside a REST server.
99  		var framework = Pattern.compile("org\\.apache\\.juneau\\.rest\\.server\\.(?!widgets)");
100 		var o = offenders(t -> framework.matcher(t).find(), "src/main/java");
101 		o.removeIf(p -> p.endsWith(MIXIN_FILE));
102 		assertTrue(o.isEmpty(), () -> "Only " + MIXIN_FILE + " may reference the REST serving framework; these also do: " + o);
103 	}
104 
105 	@Test void b02_beansDoNotReferenceTheHtml5BeanDom() throws IOException {
106 		// The markup that renders these beans is emitted by the module that owns the emitters, not here.
107 		var o = offenders(t -> t.contains("org.apache.juneau.bean.html5."), "src/main/java");
108 		assertTrue(o.isEmpty(), () -> "Widget beans must not depend on the HTML5 bean DOM: " + o);
109 	}

115 	@Test void c01_contractVersionReExports_areAliasesOfTheBeanConstants_notBakedLiterals() throws IOException {
116 		// These fields are compile-time constants, so a baked literal is indistinguishable from an alias at runtime -
117 		// and would silently stop tracking the bean when the bean's wire contract is revised.  The assignment shape
118 		// is therefore checked in the source.
119 		var src = mixinSource();
120 		assertTrue(src.contains("CALENDAR_CONTRACT_VERSION = CalendarDef.CONTRACT_VERSION"), src);
121 		assertTrue(src.contains("HEADER_CONTRACT_VERSION = \"1\""), src);
122 		assertTrue(src.contains("BAR_CONTRACT_VERSION = BarSlot.CONTRACT_VERSION"), src);
123 	}

125 	/**
126 	 * The handler and the bytes arrive together.  This module now ships the three relocated widget runtime assets, so
127 	 * the mixin must declare a serving endpoint for each: a declared mount with no accessor would answer 404, and an
128 	 * accessor with no bytes would answer an empty 200.  Pinning the resource set as well as the endpoint count is
129 	 * what keeps the two halves from drifting apart in either direction.
130 	 */
131 	@Test void c02_theMixinDeclaresOneServingEndpointForEachAssetItShips() throws IOException {
132 		var dir = Path.of("src/main/resources/org/apache/juneau/widgets");
133 		assertTrue(Files.isDirectory(dir), () -> "This module must ship its widget assets at " + dir);
134 		try (var s = Files.list(dir)) {
135 			assertEquals(
136 				List.of("juneau-calendar.css", "juneau-calendar.js"),
137 				s.map(x -> x.getFileName().toString()).sorted().toList());
138 		}
139 		// Anchored to the start of a line so a javadoc mention of the annotation is not a hit.
140 		var m = Pattern.compile("(?m)^\\s*@RestGet").matcher(mixinSource());
141 		var endpoints = 0;
142 		while (m.find())
143 			endpoints++;
144 		assertEquals(2, endpoints, "One serving endpoint per shipped asset, no more and no fewer.");
145 	}

151 	@Test void d01_dialogBeans_resolveFromThisModulesPackage() {
152 		// The declarative dialog contracts are general widget contracts, so they belong to this module.  Resolving
153 		// them by name (rather than referencing the types) is what makes this a genuine relocation guard: a stale
154 		// import elsewhere cannot satisfy it.
155 		for (var n : new String[]{"ModalDef", "FormDef", "FormDef$Input", "FormDef$Section", "ModalDef$Field"})
156 			assertDoesNotThrow(() -> Class.forName("org.apache.juneau.rest.server.widgets." + n), n);
157 	}
158 
159 	@Test void d02_dialogBeans_areGoneFromTheViewsPackage() {
160 		// No shim: a deprecated forwarding class left behind in views would re-create the views/widgets tangle in
161 		// reverse, so the old coordinates must resolve to nothing at all.
162 		for (var n : new String[]{"ModalDef", "FormDef"})
163 			assertThrows(ClassNotFoundException.class, () -> Class.forName(VIEWS_PACKAGE + "." + n), n);
164 	}
165 
166 	@Test void d03_dialogBeans_carryNoDanglingLinkToAViewsType() throws IOException {
167 		// A {@link} from here to a table/row-action type would not resolve and would fail the javadoc gate.  a01
168 		// already forbids the fully-qualified package; these are the simple names those links used, which have to
169 		// have become prose or {@code} instead.
170 		var link = Pattern.compile("\\{@link\\s+(RowAction|IdempotencyKey|ActionResult|RowClassRule)\\b");
171 		var o = offenders(t -> link.matcher(t).find(), "src/main/java", "src/test/java");
172 		assertTrue(o.isEmpty(), () -> "These files {@link} a views type by simple name, which cannot resolve here: " + o);
173 	}

175 	private static String mixinSource() throws IOException {
176 		return Files.readString(sourceRoot("src/main/java").resolve("org/apache/juneau/rest/server/widgets/" + MIXIN_FILE));
