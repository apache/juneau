31  	private static final RunEvent STEP_EVENT = RunEvent.step("tests", "mvn test").withN(4).withState(StepState.WAITING).withRawOffset(120).withSeq(1);

45  	@Test void a02_numberedCopyIncludesSeq() {
46  		var e = RunEvent.done(DoneStatus.OK).withSeq(7);
47  		assertEquals(Kind.DONE, e.kind());
48  		assertEquals(7L, e.seq());
49  		assertEquals(0L, RunEvent.done(DoneStatus.OK).seq());

51  		assertFalse(RunEvent.done(DoneStatus.OK).toContractMap().containsKey("seq"));

68  		assertThrowsWithMessage(IllegalArgumentException.class, "RunEvent step id", () -> RunEvent.step("_x", "t"));
69  		assertThrowsWithMessage(IllegalArgumentException.class, "RunEvent step title", () -> RunEvent.step("a", ""));

74  	@Test void b02_ranges() {
75  		var s = RunEvent.step("a", "T");
76  		assertThrowsWithMessage(IllegalArgumentException.class, "n must be", () -> s.withN(0));
77  		assertThrowsWithMessage(IllegalArgumentException.class, "n must be", () -> s.withN(10000));
78  		assertThrowsWithMessage(IllegalArgumentException.class, "rawOffset", () -> s.withRawOffset(-1));
79  		var e = RunEvent.end("a", EndStatus.OK);
80  		assertThrowsWithMessage(IllegalArgumentException.class, "ms must be", () -> e.withMs(-1));
81  		assertThrowsWithMessage(IllegalArgumentException.class, "ms must be", () -> e.withMs(RunEvent.MAX_SAFE_INT + 1));
82  		assertDoesNotThrow(() -> e.withMs(RunEvent.MAX_SAFE_INT));
83  		assertThrowsWithMessage(IllegalArgumentException.class, "counts.pass", () -> RunEvent.suite("a", "jest", "s", -1, 0, 0));
84  		assertThrowsWithMessage(IllegalArgumentException.class, "counts.skip", () -> RunEvent.suite("a", "jest", "s", 0, 0, 1_000_000_001));
85  		assertThrowsWithMessage(IllegalArgumentException.class, "seq", () -> s.withSeq(-1));
86  	}

119 	@Test void b08_rawOffset() {
120 		var s = RunEvent.step("a", "T").withRawOffset(0);
121 		assertEquals(0L, s.toContractMap().get("rawOffset"));
122 		assertEquals(RunEvent.MAX_SAFE_INT, RunEvent.suite("a", "jest", "s", 1, 0, 0).withRawOffset(RunEvent.MAX_SAFE_INT).toContractMap().get("rawOffset"));
123 		assertEquals(7L, RunEvent.test("a", "jest", "s", "n", TestStatus.PASS).withRawOffset(7).toContractMap().get("rawOffset"));
124 		assertThrowsWithMessage(IllegalArgumentException.class, "rawOffset must be 0-", () -> s.withRawOffset(-1));
125 		assertThrowsWithMessage(IllegalArgumentException.class, "rawOffset must be 0-", () -> s.withRawOffset(RunEvent.MAX_SAFE_INT + 1));
126 		assertThrowsWithMessage(IllegalArgumentException.class, "does not have", () -> RunEvent.end("a", EndStatus.OK).withRawOffset(1));
127 	}

172 	@Test void c08_rawOffsetRoundTripsAndOrders() {
173 		var step = RunEvent.step("a", "T").withRawOffset(120);
174 		var suite = RunEvent.suite("a", "jest", "s", 1, 0, 0).withRawOffset(5);
175 		var test = RunEvent.test("a", "jest", "s", "n", TestStatus.FAIL).withRawOffset(9);
176 		for (var e : List.of(step, suite, test))
177 			assertEquals(e, RunEvent.fromMap(e.toContractMap()));
178 		assertEquals(List.of("ev", "id", "title", "rawOffset"), new ArrayList<>(step.toContractMap().keySet()));
179 		assertEquals(List.of("ev", "step", "fw", "suite", "counts", "rawOffset"), new ArrayList<>(suite.toContractMap().keySet()));
180 		assertEquals(List.of("ev", "step", "fw", "suite", "name", "status", "rawOffset"), new ArrayList<>(test.toContractMap().keySet()));
181 		assertEquals(120L, RunEvent.fromMap(Map.of("ev", "step", "id", "a", "title", "T", "rawOffset", 120.0)).toContractMap().get("rawOffset"));
182 		assertThrowsWithMessage(IllegalArgumentException.class, "rawOffset", () -> RunEvent.fromMap(Map.of("ev", "step", "id", "a", "title", "T", "rawOffset", 1.5)));
183 	}
