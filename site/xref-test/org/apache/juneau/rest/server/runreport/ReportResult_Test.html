17  package org.apache.juneau.rest.server.runreport;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.*;
24  import org.apache.juneau.rest.server.runreport.RunEvent.*;
25  import org.junit.jupiter.api.*;
26  
27  class ReportResult_Test extends TestBase {
28  
29  	private static ReportTest test(String name) {
30  		return new ReportTest("s", name, TestStatus.PASS, 5L, null, null);
31  	}
32  
33  	private static List<String> kinds(List<?> events) {
34  		return events.stream().map(e -> ((org.apache.juneau.rest.server.runreport.RunEvent)e).kind().name().toLowerCase(Locale.ROOT)).toList();
35  	}
36  
37  	@Test void b01_toEventsOrder() {
38  		var r = new ReportResult("jest", List.of(test("t1")), List.of("w1"), false, 0);
39  		var ev = r.toEvents("tests");
40  		assertEquals(List.of("replace", "test", "note"), kinds(ev));
41  		assertEquals("tests", ev.get(0).toContractMap().get("step"));
42  		assertEquals("warn", ev.get(2).toContractMap().get("level"));
43  		assertEquals("tests", ev.get(2).toContractMap().get("step"));
44  	}
45  
46  	@Test void b02_noTestsAndWarningsKeepsPlaceholders() {
47  		var ev = new ReportResult("jest", List.of(), List.of("w"), false, 0).toEvents("tests");
48  		assertEquals(1, ev.size());
49  		assertEquals("note", ev.get(0).toContractMap().get("ev"));
50  	}
51  
52  	@Test void b03_noTestsNoWarningsStillReplaces() {
53  		assertEquals(List.of("replace"), kinds(new ReportResult("jest", List.of(), List.of(), false, 0).toEvents("tests")));
54  	}
55  
56  	@Test void b04_testsAreBoundToStepAndFw() {
57  		var e = new ReportResult("jest", List.of(test("t1")), List.of(), false, 0).toEvents("tests").get(1).toContractMap();
58  		assertEquals("tests", e.get("step"));
59  		assertEquals("jest", e.get("fw"));
60  		assertEquals(5L, e.get("ms"));
61  	}
62  
63  	@Test void b05_emptyNamesAndOverlongTextAreRepaired() {
64  		var t = new ReportTest("", null, TestStatus.FAIL, -1L, "m".repeat(5000), "t".repeat(20000));
65  		var e = new ReportResult("jest", List.of(t), List.of(), false, 0).toEvents("tests").get(1).toContractMap();
66  		assertEquals("(unnamed)", e.get("suite"));
67  		assertEquals("(unnamed)", e.get("name"));
68  		assertFalse(e.containsKey("ms"));
69  		assertEquals(2000, ((String)e.get("msg")).length());
70  		assertEquals(8000, ((String)e.get("trace")).length());
71  	}
72  
73  	@Test void b06_extremeControlCharsShedTrace() {
74  		var ctl = "\u0001";
75  		var t = new ReportTest(ctl.repeat(512), ctl.repeat(512), TestStatus.FAIL, null, ctl.repeat(2000), ctl.repeat(8000));
76  		var ev = new ReportResult("jest", List.of(t), List.of(), false, 0).toEvents("tests");
77  		assertDoesNotThrow(() -> ev.get(1).validate());
78  		assertFalse(ev.get(1).toContractMap().containsKey("trace"));
79  	}
80  
81  	@Test void b07_fwGrammar() {
82  		assertThrows(IllegalArgumentException.class, () -> new ReportResult("Jest", List.of(), List.of(), false, 0));
83  	}
84  
85  	@Test void b08_limitsMustBePositive() {
86  		assertThrows(IllegalArgumentException.class, () -> new ReportLimits(0, 1, 1, 1, 1));
87  		assertEquals(100_000, ReportLimits.DEFAULT.maxTests());
88  	}
89  }
