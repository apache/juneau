19  import static org.apache.juneau.test.bct.BctAssertions.*;
20  import static org.apache.juneau.rest.server.runreport.ReportTestSupport.*;
21  import static org.junit.jupiter.api.Assertions.*;

23  import java.nio.file.*;
24  import java.util.*;
25  
26  import org.apache.juneau.*;
27  import org.junit.jupiter.api.*;
28  import org.junit.jupiter.api.io.*;
29  
30  class PlaywrightJsonReportReader_Test extends TestBase {
31  
32  	@Test void a01_mapping() {
33  		var r = new PlaywrightJsonReportReader().read(resource("playwright-mixed.json"));
34  		assertEquals(List.of("pass", "fail", "pass", "skip", "fail"), statuses(r));
35  		assertBean(r.tests().get(1), "suite,name", "a.spec.ts,Login › breaks [chromium]");
36  		assertEquals("Login › works", r.tests().get(0).name());
37  		assertEquals("playwright", r.fw());
38  	}
39  
40  	@Test void a02_msgAnsiStrippedFirstLine() {
41  		var t = new PlaywrightJsonReportReader().read(resource("playwright-mixed.json")).tests().get(1);
42  		assertEquals("expected visible", t.msg());
43  		assertEquals("expected visible\nCall log: x", t.trace());
44  	}
45  
46  	@Test void a03_durationIsSumOfResults() {
47  		assertEquals(80L, new PlaywrightJsonReportReader().read(resource("playwright-mixed.json")).tests().get(2).ms());
48  	}
49  
50  	@Test void a04_statusFallsBackToLastResult() {
51  		var t = new PlaywrightJsonReportReader().read(resource("playwright-mixed.json")).tests().get(4);
52  		assertBean(t, "suite,name,status,msg", "b.spec.ts,hangs,FAIL,Test timeout of 30000ms exceeded.");
53  	}
54  
55  	@Test void a05_flakyPassHasNoMessage() {
56  		assertNull(new PlaywrightJsonReportReader().read(resource("playwright-mixed.json")).tests().get(2).msg());
57  	}
58  
59  	@Test void b01_truncated() {
60  		var r = new PlaywrightJsonReportReader().read(resource("playwright-truncated.json"));
61  		assertTrue(r.tests().isEmpty());
62  		assertTrue(r.warnings().get(0).startsWith("report unreadable"), r.warnings().get(0));
63  	}
64  
65  	@Test void b02_twelveThousandTests(@TempDir Path d) throws Exception {
66  		var sb = new StringBuilder("{\"suites\":[{\"title\":\"a.spec.ts\",\"file\":\"a.spec.ts\",\"specs\":[");
67  		for (var i = 0; i < 12_000; i++) {
68  			if (i > 0)
69  				sb.append(',');
70  			sb.append("{\"title\":\"s").append(i).append("\",\"tests\":[{\"status\":\"expected\",\"results\":[{\"status\":\"passed\",\"duration\":1}]}]}");
71  		}
72  		var f = d.resolve("big.json");
73  		Files.writeString(f, sb.append("]}]}").toString());
74  		assertEquals(12_000, new PlaywrightJsonReportReader().read(f).tests().size());
75  	}
76  
77  	@Test void b03_maxTests() {
78  		var r = new PlaywrightJsonReportReader().read(resource("playwright-mixed.json"), new ReportLimits(1 << 20, 2, 512, 2000, 8000));
79  		assertEquals(2, r.tests().size());
80  		assertTrue(r.truncated());
81  	}
82  
83  	//-----------------------------------------------------------------------------------------------------------------
84  	// The report juneau-run's own Playwright reporter writes: juneau_run_playwright.test.cjs checks that its report of a
85  	// fake suite equals playwright-juneau-run.json.
86  	//-----------------------------------------------------------------------------------------------------------------
87  
88  	@Test void c01_juneauRunReporter() {
89  		var r = new PlaywrightJsonReportReader().read(resource("playwright-juneau-run.json"));
90  		assertEquals(List.of("fail", "skip", "pass", "pass"), statuses(r));
91  		assertEquals(List.of("top [firefox]", "top [chromium]", "Login › works [chromium]", "Login › flaky one [chromium]"),
92  			r.tests().stream().map(ReportTest::name).toList());
93  		assertBean(r.tests().get(0), "suite,msg,ms", "tests/a.spec.ts,boom,14");
94  		assertEquals("boom\nat a.spec.ts:11", r.tests().get(0).trace());
95  		assertNull(r.tests().get(3).msg());
96  		assertTrue(r.warnings().isEmpty());
97  	}
98  }
