175 	private static String get(String path, int status) throws Exception {
176 		try (var c = MockRestClient.buildLax(Host.class); var rsp = c.get(path).run()) {
177 			var body = rsp.getContent().asString();
178 			assertEquals(status, rsp.getStatusCode(), () -> body);
179 			return body;
180 		}
181 	}

329 	private static int countOccurrences(String haystack, String needle) {
330 		var count = 0;
331 		for (var i = haystack.indexOf(needle); i >= 0; i = haystack.indexOf(needle, i + 1))
332 			count++;
333 		return count;
334 	}
