17  package org.apache.juneau.rest.server.view.freemarker;
18  
19  import org.apache.juneau.*;
20  import org.apache.juneau.commons.inject.*;
21  import org.apache.juneau.rest.mock.classic.*;
22  import org.apache.juneau.rest.server.*;
23  import org.apache.juneau.rest.server.servlet.*;
24  import org.junit.jupiter.api.*;

26  /**
27   * MockRest-level assertions for {@link FreemarkerMixin} composition + route wiring.
28   *
29   * <p>
30   * Like Thymeleaf and Mustache (and unlike JSP), the Apache FreeMarker engine has <b>zero
31   * servlet-container dependencies</b> &mdash; it asks {@link freemarker.template.Configuration}
32   * for a {@code Template} and {@code process(dataModel, writer)} writes directly to the response
33   * writer. That makes the <i>raw-render path</i> (the mixin's own {@code /freemarker/*} handler
34   * that calls {@code cfg.getTemplate(...).process(...)} directly onto the response writer) fully
35   * exerciseable under MockRest, which is what this test covers.
36   *
37   * <h5 class='section'>Deferred &mdash; typed {@link org.apache.juneau.rest.server.server.view.View View} return
38   * path:</h5>
39   *
40   * <p>
41   * Verifying that an {@code @RestOp} method returning a {@link FreemarkerView} renders through
42   * {@link FreemarkerViewRenderer} requires the renderer to run <b>before</b> the default
43   * HTML/JSON/&hellip; serializer processors in the response chain. Today the mixin's
44   * {@link Rest#responseProcessors() @Rest(responseProcessors=...)} declaration appends to the
45   * chain, so the default HTML serializer wins under MockRest and the {@code FreemarkerView} is
46   * serialized as a Juneau-bean HTML table instead of routed through FreeMarker. The response-
47   * processor "prepend" mechanism that fixes this blocks
48   * the typed-handler integration matrix for all four sibling view modules (JSP / Thymeleaf /
49   * Mustache / FreeMarker). The typed-handler test surface lights up alongside the real-container
50   * coverage once that prereq lands &mdash; tracked as the FreeMarker analog of the deferred JSP /
51   * Thymeleaf / Mustache follow-ons.
52   *
53   * <h5 class='figure'>Test resource layout (classpath {@code src/test/resources}):</h5>
54   *
55   * <pre>
56   *   /freemarker-templates/hello.ftlh        ← greets "Hello, ${name}!" (used by future typed-handler tests)
57   *   /freemarker-templates/about.ftlh        ← static "About Juneau"
58   *   /freemarker-templates/nested/inner.ftlh ← validates multi-segment paths
59   * </pre>
60   *
61   * @since 10.0.0
62   */
63  class FreemarkerMixin_MockRest_Test extends TestBase {
64  
65  	/* ---------------------------------------------------------------------------------------- *
66  	 * Resource A: mixin only — default base path "/" + default configuration + no implicit suffix
67  	 * ---------------------------------------------------------------------------------------- */
68  
69  	@Rest(mixins=FreemarkerMixin.class)
70  	public static class A extends BasicRestServlet {
71  		private static final long serialVersionUID = 1L;
72  		@RestGet(path="/items") public String items() { return "items"; }
73  	}
74  
75  	private static final MockRestClient ca = MockRestClient.buildLax(A.class);
76  
77  	@Test void a01_hostEndpointStillReachable() throws Exception {
78  		ca.get("/items")
79  			.accept("application/json")
80  			.run()
81  			.assertStatus(200)
82  			.assertContent().asString().isContains("items");
83  	}
84  
85  	@Test void a02_freemarkerMountInstalled_rendersTemplate() throws Exception {
86  		// Default basePath="/" + no templateSuffix → request path is taken as-is, FreeMarker's
87  		// root-of-classpath loader resolves it directly.
88  		ca.get("/freemarker/freemarker-templates/about.ftlh")
89  			.run()
90  			.assertStatus(200)
91  			.assertContent().asString().isContains("About Juneau");
92  	}
93  
94  	@Test void a04_nonMixinPathFallsThrough() throws Exception {
95  		ca.get("/does-not-exist")
96  			.run()
97  			.assertStatus(404);
98  	}
99  
100 	/* ---------------------------------------------------------------------------------------- *
101 	 * Resource B: configured base path "/freemarker-templates/" + templateSuffix ".ftlh"
102 	 * ---------------------------------------------------------------------------------------- */
103 
104 	@Rest(mixins=FreemarkerMixin.class)
105 	public static class B extends BasicRestServlet {
106 		private static final long serialVersionUID = 1L;
107 
108 		@Bean public FreemarkerMixin freemarker() {
109 			return FreemarkerMixin.create()
110 				.basePath("/freemarker-templates/")
111 				.templateSuffix(".ftlh")
112 				.build();
113 		}
114 	}
115 
116 	private static final MockRestClient cb = MockRestClient.buildLax(B.class);
117 
118 	@Test void b01_renderResolvesUnderConfiguredBasePath() throws Exception {
119 		// basePath = /freemarker-templates/, suffix = .ftlh.
120 		// /freemarker/about → cfg.getTemplate("about.ftlh") → resolves
121 		// /freemarker-templates/about.ftlh on the classpath.
122 		cb.get("/freemarker/about")
123 			.run()
124 			.assertStatus(200)
125 			.assertContent().asString().isContains("About Juneau");
126 	}
127 
128 	@Test void b02_renderResolvesWithExplicitSuffix() throws Exception {
129 		// Same template via the explicit-suffix request form; suffix is idempotent.
130 		cb.get("/freemarker/about.ftlh")
131 			.run()
132 			.assertStatus(200)
133 			.assertContent().asString().isContains("About Juneau");
134 	}
135 
136 	@Test void b03_renderHandlesMultiSegmentPath() throws Exception {
137 		cb.get("/freemarker/nested/inner")
138 			.run()
139 			.assertStatus(200)
140 			.assertContent().asString().isContains("inner template");
141 	}
142 
143 	@Test void b04_contentTypeDefaultsToHtml() throws Exception {
144 		cb.get("/freemarker/about")
145 			.run()
146 			.assertStatus(200)
147 			.assertHeader("Content-Type").isContains("text/html");
148 	}
149 }
