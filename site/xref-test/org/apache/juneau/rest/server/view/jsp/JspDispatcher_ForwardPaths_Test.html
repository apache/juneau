33  /**
34   * Covers {@link JspDispatcher#render render(...)}'s dispatch-succeeded and dispatch-failed branches
35   * (the {@code rd != null} half of the {@code rd == null} check, plus every {@code catch} clause around
36   * {@code rd.forward(...)}).
37   *
38   * <p>
39   * {@link JspMixin_MockRest_Test} exercises a 500 with no custom wiring at all, but MockRest's default
40   * {@code HttpServletRequest#getServletContext()} returns {@code null} outright (no container is present),
41   * so that 500 actually comes from the generic {@code catch (Exception ex)} clause reacting to the
42   * resulting {@code NullPointerException} on {@code ctx.getRequestDispatcher(...)} — not from the
43   * {@code rd == null} branch's own dedicated diagnostic a few lines below. {@link #a00} here hits that
44   * specific branch directly. This class swaps in a hand-rolled {@link ServletContext} (via
45   * {@link MockRestRequest#servletContext} — copied onto the underlying {@code HttpServletRequest} for one
46   * call) whose {@code getRequestDispatcher(...)} returns a {@link RequestDispatcher} test double that
47   * forwards successfully or throws each of the exception types {@code render(...)} specifically catches,
48   * without needing a real JSP engine or servlet container.
49   *
50   * @since 10.0.0
51   */
52  @SuppressWarnings({
53  	"resource" // Closeable test fixtures held in static fields; lifecycle managed by the test/framework, not a real leak.
54  })
55  class JspDispatcher_ForwardPaths_Test extends TestBase {
56  
57  	@Rest(mixins=JspMixin.class)
58  	public static class A extends BasicRestServlet {
59  		private static final long serialVersionUID = 1L;
60  	}
61  
62  	private static final MockRestClient c = MockRestClient.buildLax(A.class);
63  
64  	/** Functional seam for {@link #forward(ServletRequest, ServletResponse)} on the fake dispatcher below. */
65  	@FunctionalInterface
66  	private interface ForwardAction {
67  		void run() throws ServletException, IOException;
68  	}
69  
70  	/** Minimal {@link RequestDispatcher} whose {@code forward(...)} delegates to a configurable action. */
71  	private static final class FakeDispatcher implements RequestDispatcher {
72  		private final ForwardAction action;
73  		FakeDispatcher(ForwardAction action) { this.action = action; }
74  		@Override public void forward(ServletRequest req, ServletResponse res) throws ServletException, IOException { action.run(); }
75  		@Override public void include(ServletRequest req, ServletResponse res) { /* not exercised by render(...) */ }
76  	}
77  
78  	/**
79  	 * Builds a {@link ServletContext} whose only meaningful behavior is
80  	 * {@code getRequestDispatcher(String)} returning {@code dispatcher} (or {@code null}, to pin the
81  	 * already-covered {@code rd == null} branch via the same mechanism as a sanity check). Every other
82  	 * {@link ServletContext} method is never called by {@code render(...)}, so a {@link Proxy} that
83  	 * returns {@code null} for anything else is sufficient without hand-implementing the interface's
84  	 * several dozen other methods.
85  	 */
86  	private static ServletContext fakeServletContext(RequestDispatcher dispatcher) {
87  		return (ServletContext) Proxy.newProxyInstance(
88  			JspDispatcher_ForwardPaths_Test.class.getClassLoader(),
89  			new Class<?>[] { ServletContext.class },
90  			(proxy, method, args) -> eq(method.getName(), "getRequestDispatcher") ? dispatcher : null);
91  	}
92  
93  	@Test void a00_contextPresentButDispatcherNull_reportsNoEngineDiagnostic() throws Exception {
94  		// A real (non-null) ServletContext that itself resolves the target to no dispatcher -- the
95  		// documented rd == null branch, distinct from JspMixin_MockRest_Test's NPE-driven 500 (see class
96  		// javadoc above). render(...) re-throws InternalServerError ahead of the generic catch, so this
97  		// branch's specific "Could not resolve RequestDispatcher... NO_ENGINE_DIAGNOSTIC" message reaches
98  		// the response body intact instead of being re-wrapped into the generic "JSP render failed for" message.
99  		var ctx = fakeServletContext(null);
100 		var res = c.get("/jsp/hello.jsp").servletContext(ctx).run();
101 		res.assertStatus(500);
102 		res.assertContent().asString().isContains("Could not resolve RequestDispatcher");
103 		res.assertContent().asString().isContains("No JSP engine is available on the classpath");
104 	}
105 
106 	@Test void a01_dispatcherFound_forwardSucceeds_returns200() throws Exception {
107 		var ctx = fakeServletContext(new FakeDispatcher(() -> { /* no-op: a well-behaved engine committed the response itself */ }));
108 		c.get("/jsp/hello.jsp").servletContext(ctx).run().assertStatus(200);
109 	}
110 
111 	@Test void a02_dispatcherFound_forwardThrowsIOException_propagatesAs500() throws Exception {
112 		// IOException is declared by render(...) and re-thrown as-is (not wrapped) — the framework's
113 		// default exception-to-status mapping for an undeclared-HTTP-status checked exception is 500.
114 		var ctx = fakeServletContext(new FakeDispatcher(() -> { throw new IOException("disk full"); }));
115 		c.get("/jsp/hello.jsp").servletContext(ctx).run().assertStatus(500);
116 	}
117 
118 	@Test void a03_dispatcherFound_forwardThrowsNotFound_propagatesAs404() throws Exception {
119 		// NotFound is caught alongside IOException and re-thrown as-is (not wrapped) — its own status
120 		// code (404) drives the response, same as if the caller had thrown it directly.
121 		var ctx = fakeServletContext(new FakeDispatcher(() -> { throw new NotFound("no such template"); }));
122 		c.get("/jsp/hello.jsp").servletContext(ctx).run().assertStatus(404);
123 	}
124 
125 	@Test void a04_dispatcherFound_forwardThrowsNoClassDefFoundError_wrappedWithEngineDiagnostic() throws Exception {
126 		// A JSP engine class missing at runtime (e.g. the api jar present but no impl on the classpath)
127 		// surfaces as NoClassDefFoundError from inside forward() -- wrapped with the same
128 		// NO_ENGINE_DIAGNOSTIC text as the rd == null case, rather than a bare 500.
129 		var ctx = fakeServletContext(new FakeDispatcher(() -> { throw new NoClassDefFoundError("org.apache.jasper.JspCompilationContext"); }));
130 		var res = c.get("/jsp/hello.jsp").servletContext(ctx).run();
131 		res.assertStatus(500);
132 		res.assertContent().asString().isContains("jetty-ee11-apache-jsp");
133 	}
134 
135 	@Test void a05_dispatcherFound_forwardThrowsOtherException_wrappedAs500() throws Exception {
136 		// Any other exception from forward() (e.g. a ServletException from a failing JSP compile) falls
137 		// through to the generic catch and is wrapped with the target path in the message, distinct from
138 		// both the IOException/NotFound passthrough and the NoClassDefFoundError engine diagnostic.
139 		var ctx = fakeServletContext(new FakeDispatcher(() -> { throw new ServletException("compile error"); }));
140 		var res = c.get("/jsp/hello.jsp").servletContext(ctx).run();
141 		res.assertStatus(500);
142 		res.assertContent().asString().isContains("JSP render failed for");
143 	}
144 }
