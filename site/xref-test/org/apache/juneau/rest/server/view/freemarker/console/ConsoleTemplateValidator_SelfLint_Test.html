27  /**
28   * Lints the module's own templates: the shipped {@code base.ftlh} and every test fixture that is not intentionally

62  	private static Path basedir() {
63  		return Path.of(System.getProperty("basedir", "."));
64  	}
65  
66  	@Test void a01_testFixtures() {
67  		var findings = ConsoleTemplateValidator.create()
68  			.templateRoot(basedir().resolve("src/test/resources/templates"))
69  			.validateAll()
70  			.stream()
71  			.filter(f -> ! f.template().startsWith("c1/bad-") && ! f.template().startsWith("c1/removed-") && ! f.template().startsWith("c1/dev-bad-"))
72  			.filter(f -> ! INTENTIONALLY_INVALID.contains(f.template()))
73  			.toList();
74  		assertTrue(findings.isEmpty(), () -> String.join("\n", findings.stream().map(Object::toString).toList()));
75  	}
76  
77  	@Test void a02_shippedTemplates() {
78  		var f = ConsoleTemplateValidator.create()
79  			.templateRoot(basedir().resolve("../juneau-rest-server-console-ui-freemarker/src/main/resources"))
80  			.validateAll();
81  		assertTrue(f.isEmpty(), f::toString);
82  	}
