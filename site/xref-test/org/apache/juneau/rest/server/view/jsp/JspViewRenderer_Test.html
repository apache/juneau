17  package org.apache.juneau.rest.server.view.jsp;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.apache.juneau.*;
22  import org.junit.jupiter.api.*;
23  
24  /**
25   * Unit tests for {@link JspViewRenderer}.
26   *
27   * <p>
28   * Covers the path-joining helper and the diagnostic-message constant. The
29   * {@link JspViewRenderer#process process(...)} dispatch path is exercised end-to-end by the
30   * Jetty- / Spring-Boot-flavored integration tests in this package.
31   *
32   * @since 10.0.0
33   */
34  class JspViewRenderer_Test extends TestBase {
35  
36  	/* ---------------------------------------------------------------------------------------- *
37  	 * Section A: joinPath
38  	 * ---------------------------------------------------------------------------------------- */
39  
40  	@Test void a01_joinTrailingSlashLeadingSlash() {
41  		// Both sides have a slash -> the leading slash on template is dropped.
42  		assertEquals("/WEB-INF/views/hello.jsp",
43  			JspViewRenderer.joinPath("/WEB-INF/views/", "/hello.jsp"));
44  	}
45  
46  	@Test void a02_joinNoTrailingNoLeading() {
47  		// Neither side has a slash -> insert one.
48  		assertEquals("/WEB-INF/views/hello.jsp",
49  			JspViewRenderer.joinPath("/WEB-INF/views", "hello.jsp"));
50  	}
51  
52  	@Test void a03_joinTrailingOnly() {
53  		assertEquals("/WEB-INF/views/hello.jsp",
54  			JspViewRenderer.joinPath("/WEB-INF/views/", "hello.jsp"));
55  	}
56  
57  	@Test void a04_joinLeadingOnly() {
58  		assertEquals("/WEB-INF/views/hello.jsp",
59  			JspViewRenderer.joinPath("/WEB-INF/views", "/hello.jsp"));
60  	}
61  
62  	@Test void a05_joinHandlesRootBase() {
63  		assertEquals("/hello.jsp", JspViewRenderer.joinPath("/", "hello.jsp"));
64  		assertEquals("/hello.jsp", JspViewRenderer.joinPath("/", "/hello.jsp"));
65  	}
66  
67  	@Test void a06_joinHandlesNullBase() {
68  		assertEquals("/hello.jsp", JspViewRenderer.joinPath(null, "/hello.jsp"));
69  		assertEquals("/hello.jsp", JspViewRenderer.joinPath(null, "hello.jsp"));
70  	}
71  
72  	@Test void a07_joinHandlesNullTemplate() {
73  		assertEquals("/WEB-INF/views/",
74  			JspViewRenderer.joinPath("/WEB-INF/views/", null));
75  	}
76  
77  	@Test void a08_joinHandlesEmptyTemplate() {
78  		// Neither side has a slash, template is empty -> "/views" + "/" + "" = "/views/"
79  		assertEquals("/views/", JspViewRenderer.joinPath("/views", ""));
80  	}
81  
82  	/* ---------------------------------------------------------------------------------------- *
83  	 * Section B: NO_ENGINE_DIAGNOSTIC text
84  	 * ---------------------------------------------------------------------------------------- */
85  
86  	@Test void b01_diagnosticNamesJettyEngine() {
87  		assertTrue(JspViewRenderer.NO_ENGINE_DIAGNOSTIC.contains("jetty-ee11-apache-jsp"),
88  			"Diagnostic must name the Jetty 12 EE11 engine dependency");
89  	}
90  
91  	@Test void b02_diagnosticNamesTomcatEngine() {
92  		assertTrue(JspViewRenderer.NO_ENGINE_DIAGNOSTIC.contains("tomcat-embed-jasper"),
93  			"Diagnostic must name the embedded Tomcat engine dependency");
94  	}
95  
96  	@Test void b03_diagnosticLinksToDocs() {
97  		assertTrue(JspViewRenderer.NO_ENGINE_DIAGNOSTIC.contains("JspViewSupport"),
98  			"Diagnostic must link to the JSP topic page");
99  	}
100 }
