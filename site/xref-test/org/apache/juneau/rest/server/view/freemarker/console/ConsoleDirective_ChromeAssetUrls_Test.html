37  /**
38   * Nested-path console chrome CSS. A relative or page-prefixed {@code juneau-console/chrome.css}
39   * href on a page at {@code /rest/setup} 404s (the browser asks for
40   * {@code /rest/setup/juneau-console/...}) and the page paints with no chrome. Root-absolute
41   * {@code /juneau-console/chrome.css} is 200 and the chrome paints.
42   *
43   * <p>
44   * This is an HTTP render of a real {@code <@console>} page, not a string-contains check on the
45   * href (that substring also matches the buggy {@code /rest/setup/juneau-console/chrome.css}
46   * form). Juneau has no Playwright e2e suite; this is the closest existing harness
47   * ({@link MockRestClient} against the console-ui-freemarker module).
48   *
49   * @since 10.0.0
50   */

52  	"java:S8786", // Test-only patterns scan small rendered pages; the scan-to-attribute shape needs backtracking.
53  	"resource" // MockRestClient/RestResponse are closed in try-with-resources; fluent assertStatus returns this.

57  	private static final Pattern STYLESHEET_HREF = Pattern.compile(
58  		"<link\\s[^>]*rel=\"stylesheet\"[^>]*href=\"([^\"]+)\"",
59  		Pattern.CASE_INSENSITIVE
60  	);
61  	private static final Pattern THEME_PATH = Pattern.compile(
62  		"^/juneau-console/themes/juneau-theme-[a-z0-9-]+\\.css$"
63  	);
64  
65  	@Rest(mixins={ FreemarkerMixin.class, ConsoleChromeMixin.class }, renderResponseStackTraces="true")
66  	public static class NestedPageHost extends BasicRestServlet {
67  		private static final long serialVersionUID = 1L;
68  		@Bean public FreemarkerMixin freemarker() {
69  			return ConsoleFreemarkerMixin.create()
70  				.basePath("/templates/")
71  				.chromeTemplate("admin/console-chrome-pagetabs.ftlh")
72  				.build();
73  		}
74  		@RestGet(path="/rest/setup")
75  		public View setup() {
76  			return FreemarkerView.of("admin/page-naked.ftlh");
77  		}
78  	}

80  	@Test void a01_nestedPage_chromeAndThemeHrefsAreSiteRootAndFetchable() throws Exception {
81  		try (var c = MockRestClient.buildLax(NestedPageHost.class);
82  			var page = c.get("/rest/setup").run()) {
83  			page.assertStatus(200);
84  			var html = page.getContent().asString();
85  			var pageUri = URI.create("http://localhost/rest/setup");
86  
87  			var chromeHref = requiredHref(html, "chrome.css");
88  			var themeHref = requiredHref(html, "juneau-theme-");
89  			var chromePath = resolvedPath(pageUri, chromeHref);
90  			var themePath = resolvedPath(pageUri, themeHref);
91  
92  			assertEquals("/juneau-console/chrome.css", chromePath,
93  				() -> "chrome.css must be site-root, not under the page path; href=" + chromeHref);
94  			assertTrue(THEME_PATH.matcher(themePath).matches(),
95  				() -> "theme CSS must be site-root, not under the page path; href=" + themeHref + " path=" + themePath);
96  
97  			try (var chrome = c.get(chromePath).run()) {
98  				chrome.assertStatus(200).assertHeader("Content-Type").isContains("text/css");
99  				var css = chrome.getContent().asString();
100 				assertTrue(css.contains(".jc-header"), () -> css);
101 				assertTrue(css.contains(".juneau-page-nav"), () -> css);
102 			}
103 			try (var theme = c.get(themePath).run()) {
104 				theme.assertStatus(200).assertHeader("Content-Type").isContains("text/css");
105 			}
106 
107 			// The shell script is site-root too, and served.
108 			var m = Pattern.compile("<script src=\"([^\"]*juneau-console\\.js[^\"]*)\"").matcher(html);
109 			assertTrue(m.find(), () -> html);
110 			var shellPath = resolvedPath(pageUri, m.group(1));
111 			assertEquals("/juneau-console/juneau-console.js", shellPath, () -> html);
112 			try (var shell = c.get(shellPath).run()) {
113 				shell.assertStatus(200).assertHeader("Content-Type").isContains("javascript");
114 			}
115 
116 			// Home, Items and Admin are three top-level nav entries in the contract.
117 			assertPage(html).isValid().hasNavHref("home", "/home").hasNavHref("items", "/items").hasNavHref("admin", "/admin");
118 			assertSize(() -> html, 3, assertPage(html).contract().getList("nav"));
119 			DualHatGuard.assertClean(html);
120 		}
121 	}

123 	static String requiredHref(String html, String needle) {
124 		for (var href : stylesheetHrefs(html)) {
125 			if (href.contains(needle) && href.contains("juneau-console"))
126 				return href;
127 		}
128 		fail("no juneau-console stylesheet href containing '" + needle + "' in:\n" + html);
129 		return "";
130 	}
131 
132 	static List<String> stylesheetHrefs(String html) {
133 		var out = new ArrayList<String>();
134 		var m = STYLESHEET_HREF.matcher(html);
135 		while (m.find())
136 			out.add(m.group(1));
137 		return out;
138 	}
139 
140 	static String resolvedPath(URI pageUri, String href) {
141 		var resolved = pageUri.resolve(href);
142 		var path = resolved.getPath();
143 		return path == null ? "" : path;
144 	}
