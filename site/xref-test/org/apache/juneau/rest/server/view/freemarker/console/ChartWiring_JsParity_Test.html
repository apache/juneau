33  /**
34   * Pins the Java chart handler to the JS chart row that {@code bus-wiring.cjs} builds from the real handler that
35   * {@code juneau-charts.js} registers: the ops and roles {@link ChartCardType} accepts, and the implicit
36   * {@code zoom:}/{@code legend:} topics it reports, must equal that handler's {@code ops}, {@code roles} and
37   * {@code implicit}.  The one allowed difference is {@code refresh},
38   * the shell's own op (it is handled before the handler's ops are consulted), which Java lists and the JS table does
39   * not.  Gated on {@code node} being on {@code PATH}.
40   */
41  class ChartWiring_JsParity_Test extends TestBase {
42  
43  	private static final Path VIEWS = Path.of(System.getProperty("basedir", "."), "../juneau-rest-server-views");
44  
45  	private static final String SHELL_OP = "refresh";
46  
47  	private static JsonMap report;  // computed once: spawning node three times is wasted work
48  
49  	private static synchronized JsonMap chartReport() throws Exception {
50  		if (report == null)
51  			report = runHarness();
52  		return report;
53  	}
54  
55  	private static JsonMap runHarness() throws Exception {
56  		var harness = VIEWS.resolve("src/test/js/bus-wiring.cjs");
57  		var bus = VIEWS.resolve("src/main/resources/org/apache/juneau/views/juneau-bus.js");
58  		var charts = VIEWS.resolve("../juneau-rest-server-console-ui/src/main/resources/org/apache/juneau/console/juneau-charts.js");
59  		assumeTrue(Files.exists(harness) && Files.exists(bus) && Files.exists(charts), "views or console-ui module sources not found");
60  		try {
61  			assumeTrue(new ProcessBuilder("node", "--version").redirectErrorStream(true).start().waitFor(20, TimeUnit.SECONDS), "node not on PATH");
62  		} catch (java.io.IOException e) {
63  			assumeTrue(false, "node not on PATH");
64  		}
65  		var stdout = Files.createTempFile("chart-parity-stdout-", ".json");
66  		var stderr = Files.createTempFile("chart-parity-stderr-", ".txt");
67  		try {
68  			var p = new ProcessBuilder("node", harness.toAbsolutePath().normalize().toString(), bus.toAbsolutePath().normalize().toString(),
69  				charts.toAbsolutePath().normalize().toString())
70  				.redirectOutput(stdout.toFile()).redirectError(stderr.toFile()).start();
71  			if (! p.waitFor(60, TimeUnit.SECONDS)) {
72  				p.destroyForcibly();
73  				fail("bus-wiring.cjs did not finish within 60s");
74  			}
75  			assertEquals(0, p.exitValue(), () -> "bus-wiring.cjs failed: " + quietRead(stderr));
76  			return JsonMap.ofString(Files.readString(stdout, UTF_8)).getMap("chart");
77  		} finally {
78  			Files.deleteIfExists(stdout);
79  			Files.deleteIfExists(stderr);
80  		}
81  	}
82  
83  	private static String quietRead(Path f) {
84  		try {
85  			return Files.readString(f, UTF_8);
86  		} catch (java.io.IOException e) {
87  			return "(unreadable: " + e + ")";
88  		}
89  	}
90  
91  	@Test void a01_ops_matchTheJsTable() throws Exception {
92  		var js = new TreeSet<String>();
93  		for (var o : chartReport().getList("ops"))
94  			js.add((String)o);
95  		var java = new TreeSet<>(new ChartCardType().acceptedOps());
96  		assertTrue(java.remove(SHELL_OP), "Java lists the shell's refresh op");
97  		assertEquals(js, java, "ChartCardType.acceptedOps() minus the shell's refresh must equal the juneau-charts.js handler's ops");
98  	}
99  
100 	@Test void a02_roles_matchTheJsTable() throws Exception {
101 		var js = new TreeSet<String>();
102 		for (var o : chartReport().getList("roles"))
103 			js.add((String)o);
104 		assertEquals(js, new TreeSet<>(new ChartCardType().acceptedRoles()));
105 	}
106 
107 	@Test void a03_implicitTopics_matchTheJsTable() throws Exception {
108 		var samples = chartReport().getList("samples");
109 		assertSize(6, samples);
110 		for (var s : samples) {
111 			var sample = (Map<?,?>)s;
112 			var card = new JsonMap((Map<?,?>)sample.get("card"));
113 			assertEquals(sample.get("implicit"), new ChartCardType().implicitTopics(card), () -> "card " + card);
114 		}
115 	}

117 	// ---- BusWiringValidator.LEGACY_HELPER_KEYS mirrors ChartDef.HELPER_KEYS (the console-ui module cannot see ChartDef) --------
118 
119 	private static List<String> e74(String key) {
120 		// The value nests the marker one level down so that it is reached for every key, including 'series' (whose map and
121 		// list entries skip their 'data' key, but not 'label').
122 		var contract = JsonMap.ofString("{\"version\":\"1\",\"title\":\"T\",\"nav\":[],\"activeNav\":[],\"cards\":["
123 			+ "{\"id\":\"t\",\"type\":\"chart\",\"kind\":\"line\",\"" + key + "\":{\"label\":{\"$fn\":\"Bad\"}}}]}");
124 		return BusWiringValidator.validate(contract, CardTypeRegistry.standard()).stream()
125 			.filter(p -> "E-74".equals(p.code())).map(Object::toString).toList();
126 	}
127 
128 	@Test void a04_everyHelperKey_isWalkedForFnNames_inTheValidator() {
129 		assertFalse(ChartDef.HELPER_KEYS.isEmpty());
130 		for (var key : new TreeSet<>(ChartDef.HELPER_KEYS))
131 			assertEquals(List.of("E-74 'Bad' is not a valid $fn name"), e74(key), () -> "legacy helper key '" + key + "'");
132 		assertEquals(List.of(), e74("title"), "a non-helper key is not walked");
133 	}
