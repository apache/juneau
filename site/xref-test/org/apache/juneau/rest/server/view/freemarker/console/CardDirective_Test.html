97  	/** The contract entry for card {@code id}. */
98  	static JsonMap card(String body, String id) {
99  		for (var o : assertPage(body).isValid().contract().getList("cards"))
100 			if (id.equals(((JsonMap)o).getString("id")))
101 				return (JsonMap)o;
102 		throw new AssertionError("no card '" + id + "' in " + body);
103 	}

163 	/** The operator {@code name}s of a lifted VIEW_META {@code search.operators} list, in wire order. */
164 	private static List<String> opNames(JsonList operators) {
165 		var l = new ArrayList<String>();
166 		for (var i = 0; i < operators.size(); i++)
167 			l.add(operators.getMap(i).getString("name"));
168 		return l;
169 	}

171 	@Test void b10a_typeDatatables_emitsPerColumnSearchMetadata() throws Exception {
172 		var view = card(get("/card-datatables-search"), "releases").getMap("table");
173 		var cols = view.getList("columns");
174 
175 		// Column 0 (text, list absent) → the full text universe, each operator carrying help; no custom leaked in.
176 		var name = cols.getMap(0).getMap("search");
177 		assertEquals("text", name.getString("type"));
178 		assertEquals(List.of("$eq", "$eqic", "$ne", "$in", "$and", "$or", "$not", "$contains", "$prefix", "$regex", "$blank"),
179 			opNames(name.getList("operators")));
180 		var eq = name.getList("operators").getMap(0);
181 		assertBean(eq, "name,minArgs,maxArgs,combinator,custom", "$eq,1,1,false,false");
182 		assertFalse(eq.containsKey("help"));
183 
184 		// Column 1 (numeric, explicit allow-list) → exactly the named applicable operators, in the author's order.
185 		var count = cols.getMap(1).getMap("search");
186 		assertEquals("numeric", count.getString("type"));
187 		assertEquals(List.of("$eq", "$gt", "$between"), opNames(count.getList("operators")));
188 
189 		// Column 2 (text + custom $near) → the custom is appended after the built-ins, flagged, carrying its help.
190 		var statusOps = cols.getMap(2).getMap("search").getList("operators");
191 		var near = statusOps.getMap(statusOps.size() - 1);
192 		assertBean(near, "name,custom,help", "$near,true,Fuzzy match. Example: $near(jhon)");
193 
194 		// Column 3 (no searchType) → non-searchable: no search block at all.
195 		assertNull(cols.getMap(3).get("search"), cols::toString);
196 	}
