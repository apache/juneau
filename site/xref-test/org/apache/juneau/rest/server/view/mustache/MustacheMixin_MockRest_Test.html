17  package org.apache.juneau.rest.server.view.mustache;
18  
19  import org.apache.juneau.*;
20  import org.apache.juneau.commons.inject.*;
21  import org.apache.juneau.rest.mock.classic.*;
22  import org.apache.juneau.rest.server.*;
23  import org.apache.juneau.rest.server.servlet.*;
24  import org.junit.jupiter.api.*;
25  
26  /**
27   * MockRest-level assertions for {@link MustacheMixin} composition + route wiring.
28   *
29   * <p>
30   * Like Thymeleaf and unlike JSP, mustache.java has <b>zero servlet-container dependencies</b>
31   * &mdash; it compiles a template into a {@code Mustache} object and asks it to
32   * {@code execute(Writer, scope)} directly. That makes the <i>raw-render path</i> (the mixin's
33   * own {@code /mustache/*} handler that calls {@code factory.compile(...).execute(...)} directly
34   * onto the response writer) fully exerciseable under MockRest, which is what this test covers.
35   *
36   * <h5 class='section'>Deferred &mdash; typed {@link org.apache.juneau.rest.server.server.view.View View} return
37   * path:</h5>
38   *
39   * <h5 class='figure'>Test resource layout (classpath {@code src/test/resources}):</h5>
40   *
41   * <pre>
42   *   /mustache-templates/hello.mustache       ← greets "Hello, {{name}}!" (used by future typed-handler tests)
43   *   /mustache-templates/about.mustache       ← static "About Juneau"
44   *   /mustache-templates/nested/inner.mustache ← validates multi-segment paths
45   * </pre>
46   *
47   * @since 10.0.0
48   */
49  class MustacheMixin_MockRest_Test extends TestBase {
50  
51  	/* ---------------------------------------------------------------------------------------- *
52  	 * Resource A: mixin only — default base path "/" + default factory + no implicit suffix
53  	 * ---------------------------------------------------------------------------------------- */
54  
55  	@Rest(mixins=MustacheMixin.class)
56  	public static class A extends BasicRestServlet {
57  		private static final long serialVersionUID = 1L;
58  		@RestGet(path="/items") public String items() { return "items"; }
59  	}
60  
61  	private static final MockRestClient ca = MockRestClient.buildLax(A.class);
62  
63  	@Test void a01_hostEndpointStillReachable() throws Exception {
64  		ca.get("/items")
65  			.accept("application/json")
66  			.run()
67  			.assertStatus(200)
68  			.assertContent().asString().isContains("items");
69  	}
70  
71  	@Test void a02_mustacheMountInstalled_rendersTemplate() throws Exception {
72  		// Default basePath="/" + no templateSuffix → request path is taken as-is, mustache.java's
73  		// no-prefix factory resolves it directly from classpath.
74  		ca.get("/mustache/mustache-templates/about.mustache")
75  			.run()
76  			.assertStatus(200)
77  			.assertContent().asString().isContains("About Juneau");
78  	}
79  
80  	@Test void a04_nonMixinPathFallsThrough() throws Exception {
81  		ca.get("/does-not-exist")
82  			.run()
83  			.assertStatus(404);
84  	}
85  
86  	/* ---------------------------------------------------------------------------------------- *
87  	 * Resource B: configured base path "/mustache-templates/" + templateSuffix ".mustache"
88  	 * ---------------------------------------------------------------------------------------- */
89  
90  	@Rest(mixins=MustacheMixin.class)
91  	public static class B extends BasicRestServlet {
92  		private static final long serialVersionUID = 1L;
93  
94  		@Bean public MustacheMixin mustache() {
95  			return MustacheMixin.create()
96  				.basePath("/mustache-templates/")
97  				.templateSuffix(".mustache")
98  				.build();
99  		}
100 	}
101 
102 	private static final MockRestClient cb = MockRestClient.buildLax(B.class);
103 
104 	@Test void b01_renderResolvesUnderConfiguredBasePath() throws Exception {
105 		// basePath = /mustache-templates/, suffix = .mustache.
106 		// /mustache/about → factory.compile("about.mustache") → resolves
107 		// mustache-templates/about.mustache on the classpath.
108 		cb.get("/mustache/about")
109 			.run()
110 			.assertStatus(200)
111 			.assertContent().asString().isContains("About Juneau");
112 	}
113 
114 	@Test void b02_renderResolvesWithExplicitSuffix() throws Exception {
115 		// Same template via the explicit-suffix request form; suffix is idempotent.
116 		cb.get("/mustache/about.mustache")
117 			.run()
118 			.assertStatus(200)
119 			.assertContent().asString().isContains("About Juneau");
120 	}
121 
122 	@Test void b03_renderHandlesMultiSegmentPath() throws Exception {
123 		cb.get("/mustache/nested/inner")
124 			.run()
125 			.assertStatus(200)
126 			.assertContent().asString().isContains("inner template");
127 	}
128 
129 	@Test void b04_contentTypeDefaultsToHtml() throws Exception {
130 		cb.get("/mustache/about")
131 			.run()
132 			.assertStatus(200)
133 			.assertHeader("Content-Type").isContains("text/html");
134 	}
135 }
