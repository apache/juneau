36  	private static List<ConsoleTemplateValidator.Finding> lint(String source) {
37  		return ConsoleTemplateValidator.create().validateSource("t.ftlh", source);
38  	}
39  
40  	private static void assertOne(String source, int line, int column, String rule, String message) {
41  		var f = lint(source);
42  		assertSize(() -> "findings: " + f, 1, f);
43  		assertEquals(new ConsoleTemplateValidator.Finding("t.ftlh", line, column, rule, message), f.get(0));
44  	}
45  
46  	private static void assertClean(String source) {
47  		var f = lint(source);
48  		assertTrue(f.isEmpty(), () -> "findings: " + f);
49  	}
