31  	private static Map<?,?> map(Section s) {
32  		s.check("d");
33  		return Json5.to(Json.of(s.toMap()), Map.class);

36  	private static String e77(Runnable r) {
37  		return assertThrows(IllegalArgumentException.class, r::run).getMessage();
38  	}

66  	@Test void a05_choice_shapeAndErrors() {
67  		var s = ChoiceSection.fromDim("orders", "reason").label("Reason").max(3).publishes(Topics.filter("reasons"));
68  		assertEquals(Json5.to("{id:'reasons',kind:'choice',topic:'filter:reasons',label:'Reason',source:'orders',dim:'reason',max:3}", Map.class), map(s));
69  		assertEquals(50, ((Number)map(ChoiceSection.fromDim("o", "r").publishes(Topics.filter("c"))).get("max")).intValue());
70  		assertEquals("E-77: dashboard 'd': choice section 'c' source 'a b' is not a valid topic key",
71  			e77(() -> ChoiceSection.fromDim("a b", "r").publishes(Topics.filter("c")).check("d")));
72  		assertEquals("E-77: dashboard 'd': choice section 'c' dim '' is not a valid topic key",
73  			e77(() -> ChoiceSection.fromDim("o", "").publishes(Topics.filter("c")).check("d")));
74  		assertEquals("E-77: dashboard 'd': choice section 'c' max must be at least 1",
75  			e77(() -> ChoiceSection.fromDim("o", "r").max(0).publishes(Topics.filter("c")).check("d")));

92  		var s = CardSection.of(share());
93  		assertEquals(Json5.to("{kind:'card',card:'share'}", Map.class), map(s));
94  		assertEquals("share", s.cards().get(0).id());
95  		assertEquals("E-77: dashboard 'd': card section 'share' publishes through its card; remove publishes(...)",
96  			e77(() -> CardSection.of(share()).publishes(Topics.filter("x")).check("d")));
97  		assertEquals("E-77: dashboard 'd': card 'share' is bare; a dashboard member needs a card host",
98  			e77(() -> CardSection.of(share().toCardSpec().bare(true)).check("d")));
99  		assertEquals("E-77: dashboard 'd': card 'note' has type 'html', which has a dedicated PageSpec method; add it with PageSpec.html(...) and use panel(\"note\")",
100 			e77(() -> CardSection.of(CardSpec.of("html", "note")).check("d")));
101 	}
