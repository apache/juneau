36  	static void e76(String reason, Supplier<Object> build) {
37  		var e = assertThrows(IllegalArgumentException.class, build::get);
38  		assertEquals("E-76: data source 'orders': " + reason, e.getMessage());
39  	}

48  	@Test void a02_fullShape() {
49  		var m = DataSourceDef.create("orders")
50  			.url("/rest/orders/series").format(Format.NDJSON)
51  			.columns("start:time", "status:string", "ms:number", "ok:bool")
52  			.params(range(), Subscription.to(Topics.filter("search")).as("params").map("search", "search"))
53  			.chunk(Chunking.byDay("start", "end")).concurrency(3).debounceMs(100).cache(Cache.indexedDb())
54  			.epoch("/rest/session", "dataEpoch").mode(Mode.AUTO).maxRows(2_000_000)
55  			.aggregateUrl("/rest/orders/series/aggregate")
56  			.aggregate(Aggregate.bucket("auto").groupBy("status").measure("count", "sum(ms)"))
57  			.toMap();
58  		assertEquals("[{\"name\":\"start\",\"type\":\"time\"},{\"name\":\"status\",\"type\":\"string\"},"
59  			+ "{\"name\":\"ms\",\"type\":\"number\"},{\"name\":\"ok\",\"type\":\"bool\"}]",
60  			m.get("columns").toString().replace(" ", ""));
61  		assertEquals("ndjson", m.get("parser"));
62  		assertEquals("{\"unit\":\"day\",\"start\":\"start\",\"end\":\"end\"}", m.get("chunk").toString());
63  		assertEquals("indexedDb", m.get("cache"));
64  		assertEquals("{\"url\":\"/rest/session\",\"path\":\"dataEpoch\"}", m.get("epoch").toString());
65  		assertEquals("auto", m.get("mode"));
66  		assertEquals(2_000_000L, ((Number)m.get("maxRows")).longValue());
67  		assertEquals("{\"bucket\":\"auto\",\"groupBy\":[\"status\"],\"measures\":[\"count\",\"sum(ms)\"]}",
68  			m.get("aggregate").toString());
69  		assertSize(2, ((List<?>)m.get("subscribes")));
70  	}

153 	@Test void a10_aggregateSnapshotAndFields() {
154 		var agg = Aggregate.bucket("day").groupBy("status").measure("sum(ms)");
155 		var d = DataSourceDef.create("orders").url("/x").aggregateUrl("/a").aggregate(agg);
156 		agg.groupBy("a", "b", "c", "d").measure("bogus");
157 		assertEquals("{\"bucket\":\"day\",\"groupBy\":[\"status\"],\"measures\":[\"sum(ms)\"]}",
158 			d.toMap().get("aggregate").toString());
159 		for (var g : new String[]{"a,b", "a b", " ", "", "sum(a)"})
160 			e76("aggregate groupBy field '" + g + "' is not a field name", () -> DataSourceDef.create("orders")
161 				.aggregate(Aggregate.bucket("day").groupBy(g)));
162 		for (var m : new String[]{"sum(a,b)", "sum( )", "sum(a b)", "count(a)"})
163 			e76("aggregate measure '" + m + "' is not count, sum(f), avg(f), min(f) or max(f)",
164 				() -> DataSourceDef.create("orders").aggregate(Aggregate.bucket("day").measure(m)));
165 		assertNotNull(DataSourceDef.create("orders")
166 			.aggregate(Aggregate.bucket("day").groupBy("a.b", "c_d").measure("avg(x-y)")));
