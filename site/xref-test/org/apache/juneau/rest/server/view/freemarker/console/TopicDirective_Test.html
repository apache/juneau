34  /**
35   * Tests for the {@code <@topic>} directive: lowering to the contract and the E-45/E-46/E-50 rejections.
36   *
37   * @since 10.0.0
38   */
39  @SuppressWarnings({
40  	"resource" // MockRestClient/RestResponse are closed in try-with-resources; fluent assertStatus returns this.
41  })
42  class TopicDirective_Test extends TestBase {
43  
44  	@Rest(mixins=FreemarkerMixin.class, renderResponseStackTraces="true")
45  	public static class Host extends BasicRestServlet {
46  		private static final long serialVersionUID = 1L;
47  		@Bean public FreemarkerMixin freemarker() {
48  			return ConsoleFreemarkerMixin.create().basePath("/templates/").chromeTemplate("admin/console-chrome-bare.ftlh").build();
49  		}
50  		@RestGet(path="/topic") public View r0() { return FreemarkerView.of("admin/bus-topic.ftlh"); }
51  		@RestGet(path="/outside") public View r1() { return FreemarkerView.of("admin/bus-topic-outside.ftlh"); }
52  		@RestGet(path="/conflict") public View r2() { return FreemarkerView.of("admin/bus-topic-retain-conflict.ftlh"); }
53  		@RestGet(path="/framework") public View r3() { return FreemarkerView.of("admin/bus-topic-framework.ftlh"); }
54  		@RestGet(path="/noretain") public View r4() { return FreemarkerView.of("admin/bus-topic-noretain.ftlh"); }
55  		@RestGet(path="/nopublisher") public View r5() { return FreemarkerView.of("admin/bus-topic-nopublisher.ftlh"); }
56  		@RestGet(path="/badpublisher") public View r6() { return FreemarkerView.of("admin/bus-topic-badpublisher.ftlh"); }
57  	}
58  
59  	static String ok(String path) throws Exception {
60  		try (var c = MockRestClient.buildLax(Host.class); var rsp = c.get(path).run()) {
61  			rsp.assertStatus(200);
62  			return rsp.getContent().asString();
63  		}
64  	}
65  
66  	static void fails(String path, String message) throws Exception {
67  		try (var c = MockRestClient.buildLax(Host.class); var rsp = c.get(path).run()) {
68  			rsp.assertStatus(500);
69  			var body = rsp.getContent().asString();
70  			assertTrue(body.contains(message), () -> body);
71  		}
72  	}
73  
74  	@Test void a01_topic_lowersToContract() throws Exception {
75  		var a = PageContractAssert.assertPage(ok("/topic"));
76  		a.declaresTopic("app.region-picked").subscribes("region", "app.region-picked", "refresh").hasNoWiringErrors();
77  		var t = (Map<?,?>)a.contract().getList("topics").get(0);
78  		assertEquals("{topic:'app.region-picked',retain:true,publisher:'script'}", Json5.DEFAULT.write(t));
79  	}
80  
81  	@Test void b01_outsideConsoleOrPage_isE50() throws Exception {
82  		fails("/outside", "<@topic> must be inside <@console> or <@page>");
83  	}
84  
85  	@Test void b02_retainConflict_isE46() throws Exception {
86  		fails("/conflict", "topic 'ops.jobs' is declared with retain=false here and retain=true at <@topic>");
87  	}
88  
89  	@Test void b03_frameworkFamily_isE45() throws Exception {
90  		fails("/framework", "'selection' is a framework family; framework topics are implicit and may not be declared");
91  	}
92  
93  	@Test void b04_retainRequired() throws Exception {
94  		fails("/noretain", "<@topic name='app.region-picked'> needs retain=true or retain=false");
95  	}
96  
97  	@Test void b05_publisherRequired() throws Exception {
98  		fails("/nopublisher", "topic 'app.region-picked': a page-level topic needs publisher(SCRIPT|SERVER|RIBBON)");
99  	}
100 
101 	@Test void b06_badPublisher() throws Exception {
102 		fails("/badpublisher", "<@topic name='app.region-picked'> publisher='browser' must be script, server or ribbon");
103 	}
104 }
