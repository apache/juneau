38  /**
39   * Runs the shared same-origin path corpus ({@code $VIEWS/src/test/resources/same-origin-path-corpus.json}) through
40   * the Java callers of {@link SafePathTemplate#isSameOriginPath(String)} that are visible from this module.  The JS half,
41   * {@code same-origin-path.cjs}, runs the same rows through the shipped JS guards, so the two sides cannot drift apart
42   * unnoticed.
43   *
44   * <p>
45   * Covered elsewhere / exempt:
46   * <ul>
47   * 	<li>{@code RunViewChecks.isNoteHref} deliberately accepts an absolute http(s) URL, so the corpus verdict does not apply;
48   * 		its vectors live in {@code RunViewChecks_Test}.  Its non-absolute branch is {@code isSafeLineHref}, covered by a04.
49   * 	<li>{@code RunViewChecks.isRawHrefTemplate} requires exactly one {@code {offset}} placeholder, so the
50   * 		corpus verdict does not apply; its vectors live in {@code RunViewChecks_Test}.  It delegates to {@code isSafeLineHref}.
51   * 	<li>Covered transitively through {@code RegionDef.isSafeDetailEndpoint}: {@code RowDetail} ({id} templates),
52   * 		{@code RunViewDef} eventsUrl and {@code RegionDef} dataUrl.
53   * </ul>
54   */
55  class SameOriginPath_Corpus_Test extends TestBase {
56  
57  	private static final Path CORPUS = Path.of(System.getProperty("basedir", "."),
58  		"../juneau-rest-server-views/src/test/resources/same-origin-path-corpus.json");
59  
60  	/** The corpus holds this many rows; adding or dropping one must be a deliberate edit of this number and of the JS test. */
61  	static final int CASE_COUNT = 45;
62  
63  	private record Row(String name, String input, boolean accepted, boolean crossHost, Map<String,Object> guards) {}
64  
65  	private static List<Row> rows() throws Exception {
66  		var corpus = JsonMap.ofString(Files.readString(CORPUS, UTF_8));
67  		var out = new ArrayList<Row>();
68  		for (var o : corpus.getList("cases")) {
69  			var c = (JsonMap)o;
70  			out.add(new Row(c.getString("name"), c.getString("input"), c.getBoolean("accepted"), c.getBoolean("crossHost"),
71  				c.containsKey("guards") ? c.getMap("guards") : Map.of()));
72  		}
73  		return out;
74  	}
75  
76  	private static boolean throwsIae(Runnable r) {
77  		try {
78  			r.run();
79  			return false;
80  		} catch (IllegalArgumentException e) {
81  			return true;
82  		}
83  	}
84  
85  	private static Map<String,Object> body(Object... kv) {
86  		var m = new LinkedHashMap<String,Object>();
87  		for (var i = 0; i < kv.length; i += 2)
88  			m.put((String)kv[i], kv[i + 1]);
89  		return m;
90  	}
91  
92  	private static Map<String,Function<String,Boolean>> predicates() {
93  		Map<String,Function<String,Boolean>> m = map(
94  			"SafePathTemplate.isSameOriginPath", SafePathTemplate::isSameOriginPath,
95  			"SafePathTemplate.isSafeDocumentUrl", SafePathTemplate::isSafeDocumentUrl,
96  			"RunViewChecks.isSameOriginPath", RunViewChecks::isSameOriginPath,
97  			"RegionDef.isSafeDetailEndpoint", RegionDef::isSafeDetailEndpoint
98  		);
99  		// Each caller below throws exactly when the shared core rejects, so "accepted" is "did not throw".
100 		m.put("ChartDef.onClickHref", s -> ! throwsIae(() -> ChartDef.donut("c").onClickHref(s)));
101 		m.put("ChartCardType onClick.href", s -> ! throwsIae(() -> new ChartCardType().toFragment(CardSource.create("chart", "c")
102 			.bodyMap(body("kind", "donut", "label", "a", "value", "b", "data", List.of(), "onClick", Map.of("href", s))).build())));
103 		m.put("JobLogCardType submit.endpoint", s -> ! throwsIae(() -> new JobLogCardType().toFragment(CardSource.create("job-log", "j")
104 			.bodyMap(body("submit", Map.of("endpoint", s))).build())));
105 		m.put("BadgeDef.src", s -> ! throwsIae(() -> BadgeDef.create("b").src(s).validate()));
106 		m.put("BadgeDef.href", s -> ! throwsIae(() -> BadgeDef.create("b").src("/p").href(s).validate()));
107 		m.put("MenuItem.href", s -> ! throwsIae(() -> MenuItem.link("m", "L", s).validate()));
108 		m.put("BarSlot.refreshUrl", s -> ! throwsIae(() -> BarSlot.create("s").widgets(BarText.of("w", "t")).refreshUrl(s).validate()));
109 		return m;
110 	}

129 		return rows().stream().map(r -> DynamicTest.dynamicTest(r.name(), () -> {

137 	@TestFactory Stream<DynamicTest> a05_cardSrcIsValidatedForEveryCardType() throws Exception {
138 		var registry = CardTypeRegistry.standard();
139 		assertTrue(registry.isRegistered("html"));
140 		assertTrue(registry.isRegistered("chart"));
141 		assertTrue(registry.isRegistered("datatables"));
142 		Map<String,Function<String,CardSource>> cards = map(
143 			"html", s -> CardSource.create("html", "h").src(s).build(),
144 			"chart", s -> CardSource.create("chart", "c").src(s).bodyMap(body("kind", "donut", "label", "a", "value", "b")).build(),
145 			"datatables", s -> CardSource.create("datatables", "d").src(s).build()
146 		);
147 		return rows().stream().flatMap(r -> cards.entrySet().stream().map(e -> DynamicTest.dynamicTest(r.name() + " " + e.getKey(), () -> {
148 			var threw = throwsIae(() -> registry.toCard(e.getValue().apply(r.input())));
149 			if (r.accepted())
150 				assertFalse(threw, () -> e.getKey() + " card src '" + r.name() + "' must pass the page-build check");
151 			else {
152 				var arg = e.getValue().apply(r.input());
153 				var ex = assertThrows(IllegalArgumentException.class, () -> registry.toCard(arg));
154 				assertTrue(ex.getMessage().contains("'" + e.getValue().apply(r.input()).id() + "'"), () -> "names the card id: " + ex.getMessage());
155 				assertTrue(ex.getMessage().contains("src"), ex::getMessage);
156 			}
157 		})));
158 	}

160 	/** The real production path of an html card: PageCapture.addCard, which {@code <@card>} and PageSpec.html both reach. */
161 	@TestFactory Stream<DynamicTest> a06_htmlCardSrcIsCheckedWhereTheCardIsAdded() throws Exception {
162 		return rows().stream().map(r -> DynamicTest.dynamicTest(r.name(), () -> {
163 			var add = (org.junit.jupiter.api.function.Executable)() -> new PageCapture().addCard(CardSpec.html("h").src(r.input()), null);
164 			if (r.accepted())
165 				assertDoesNotThrow(add, r.name());
166 			else {
167 				var ex = assertThrows(freemarker.template.TemplateModelException.class, add, r.name());
168 				assertTrue(ex.getMessage().contains("<@card id='h'>"), ex::getMessage);
169 			}
170 		}));
171 	}

173 	/**
174 	 * CalendarEvent.href treats a blank href as "no href", so the blank corpus rows do not apply to it.
175 	 * CalendarDef.endpoint must also carry {year} and {month}; they are appended as a path suffix, which cannot
176 	 * change the verdict of a row unless that row ends in whitespace, a control character or a slash (a trailing slash would
177 	 * make the suffix a "//" prefix or a new segment), so those rows are skipped.
178 	 */
179 	@TestFactory Stream<DynamicTest> a07_calendarCallersAgreeWithTheCorpus() throws Exception {
180 		return rows().stream().flatMap(r -> {
181 			var tests = new ArrayList<DynamicTest>();
182 			if (! r.input().isBlank())
183 				tests.add(DynamicTest.dynamicTest(r.name() + " CalendarEvent.href", () -> {
184 					var e = CalendarEvent.create().id("e").title("t").start("2026-01-15").href(r.input());
185 					assertEquals(r.accepted(), ! throwsIae(() -> e.validate(Set.of(), null, null)), r.name());
186 				}));
187 			if (! r.input().isEmpty() && r.input().charAt(r.input().length() - 1) > ' ' && ! r.input().endsWith("/"))
188 				tests.add(DynamicTest.dynamicTest(r.name() + " CalendarDef.endpoint", () -> {
189 					var d = CalendarDef.create().id("c").endpoint(r.input() + "/{year}/{month}");
190 					assertEquals(r.accepted(), ! throwsIae(d::validate), r.name());
191 				}));
192 			return tests.stream();
193 		});
194 	}

196 	/** TerminalDef layers "no {} placeholder" on the shared rule, so a templated row is accepted by the core and refused here. */
197 	@TestFactory Stream<DynamicTest> a08_terminalUrlsAgreeWithTheCorpus() throws Exception {
198 		return rows().stream().flatMap(r -> {
199 			var expected = r.accepted() && ! r.input().contains("{");
200 			return Stream.of(
201 				DynamicTest.dynamicTest(r.name() + " TerminalDef.bytesUrl", () ->
202 					assertEquals(expected, ! throwsIae(() -> TerminalDef.create("t").bytesUrl(r.input()).validate()), r.name())),
203 				DynamicTest.dynamicTest(r.name() + " TerminalDef.eventsUrl", () ->
204 					assertEquals(expected, ! throwsIae(() -> TerminalDef.create("t").bytesUrl("/b").eventsUrl(r.input()).validate()), r.name())));
205 		});
