17  package org.apache.juneau.rest.server.view.freemarker.console;
18  
19  import static org.apache.juneau.test.bct.BctAssertions.assertSize;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.util.*;
23  
24  import org.apache.juneau.*;
25  import org.apache.juneau.marshall.collections.*;
26  import org.apache.juneau.rest.server.console.*;
27  import org.junit.jupiter.api.*;
28  
29  class PageSpec_Dashboard_Test extends TestBase {
30  
31  	private static final CardDirectiveModel MODEL = new CardDirectiveModel(
32  		CardRequirements.create().build(new ToolkitPackRegistry()), CardTypeRegistry.standard());
33  
34  	private static final List<Map<String,Object>> ROWS = List.of(Map.of("d", "2026-10-01", "v", 3));
35  
36  	private static DashboardSpec dash() {
37  		return DashboardSpec.create("orders")
38  			.rail(Rail.left().sections(SearchSection.create().publishes(Topics.filter("search"))))
39  			.panel(ChartDef.bar("by-reason").data(ROWS).x("d").series("v", "V"))
40  			.panel("notes", 2);
41  	}
42  
43  	@Test void a01_addsCards_pack_andContract() throws Exception {
44  		var spec = PageSpec.create().html("notes", "Notes", "<p>n</p>").dashboard(dash());
45  		assertEquals(List.of("notes", "by-reason"), spec.cards.stream().map(CardSpec::id).toList());
46  		assertTrue(spec.toolkits.contains(ToolkitPackRegistry.PACK_DASHBOARD));
47  		var cap = new PageCapture();
48  		spec.applyTo(cap, MODEL);
49  		var c = JsonMap.ofString(cap.toContractJson());
50  		assertEquals("orders", c.getMap("dashboard").getString("id"));
51  		assertSize(2, c.getMap("dashboard").getList("panels"));
52  	}
53  
54  	@Test void a02_packOrder_beforeCharts() {
55  		var spec = PageSpec.create().toolkit("views", "charts").html("notes", "<p>n</p>").dashboard(dash());
56  		assertEquals(List.of("views", "dashboard", "charts"), spec.toolkits);
57  		var again = PageSpec.create().toolkit("dashboard").html("notes", "<p>n</p>").dashboard(dash());
58  		assertEquals(List.of("dashboard"), again.toolkits);
59  	}
60  
61  	@Test void a03_secondDashboard_isE77() {
62  		var spec = PageSpec.create().html("notes", "<p>n</p>").dashboard(dash());
63  		var arg = DashboardSpec.create("other");
64  		assertEquals("E-77: dashboard 'other': a page has at most one dashboard",
65  			assertThrows(IllegalArgumentException.class, () -> spec.dashboard(arg)).getMessage());
66  	}
67  
68  	@Test void a04_byIdPanel_mustPrecedeDashboard() {
69  		var target = PageSpec.create();
70  		var arg2 = dash();
71  		assertEquals("E-77: dashboard 'orders': panel 'notes' names no card on this page",
72  			assertThrows(IllegalArgumentException.class, () -> target.dashboard(arg2)).getMessage());
73  	}
74  
75  	@Test void a05_null_rejected() {
76  		var target2 = PageSpec.create();
77  		assertEquals("PageSpec.dashboard requires a dashboard.",
78  			assertThrows(IllegalArgumentException.class, () -> target2.dashboard(null)).getMessage());
79  	}
80  
81  	@Test void a06_earlyWiring_seesDashboard() {
82  		// The search section owns filter:search, so a card subscribing to it wires clean.
83  		// The role is refresh: an html card accepts only params and refresh, so "filter" is E-44.
84  		var spec = PageSpec.create()
85  			.html("notes", "<p>n</p>", c -> c.subscribes(Subscription.to(Topics.filter("search")).as("refresh")))
86  			.dashboard(dash());
87  		assertDoesNotThrow(() -> spec.checkWiringEarly(CardTypeRegistry.standard(), null));
88  	}
89  
90  	@Test void a07_noDashboard_noKey() throws Exception {
91  		var cap = new PageCapture();
92  		PageSpec.create().html("notes", "<p>n</p>").applyTo(cap, MODEL);
93  		assertFalse(JsonMap.ofString(cap.toContractJson()).containsKey("dashboard"));
94  	}
95  }
