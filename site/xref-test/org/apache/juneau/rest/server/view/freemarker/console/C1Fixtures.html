30  /**
31   * Shared host for the C1 fixtures under {@code templates/c1/}.
32   *
33   * <p>
34   * {@code GET /t/{name}} renders {@code c1/{name}.ftlh}. Pages that use {@code <@page>} render under
35   * {@code c1/chrome.ftlh} ({@link Host}) or {@code c1/chrome-selected.ftlh} ({@link SelectedHost}). Fixtures whose
36   * name starts with {@code csrf-} get the loopback CSRF attributes.
37   */
38  @SuppressWarnings({
39  	"resource" // The static MockRestClient HOST/SELECTED fixtures live for the whole test run and are never closed
40  })
41  final class C1Fixtures {
42  
43  	private C1Fixtures() {}
44  
45  	@Rest(mixins=FreemarkerMixin.class, renderResponseStackTraces="true")
46  	public static class Host extends BasicRestServlet {
47  		private static final long serialVersionUID = 1L;
48  		@Bean public FreemarkerMixin freemarker() {
49  			return ConsoleFreemarkerMixin.create().basePath("/templates/").chromeTemplate("c1/chrome.ftlh").build();
50  		}
51  		@RestGet(path="/t/{name}")
52  		public View t(@Path("name") String name, RestRequest req) {
53  			return C1Fixtures.view(name, req);
54  		}
55  	}
56  
57  	@Rest(mixins=FreemarkerMixin.class, renderResponseStackTraces="true")
58  	public static class SelectedHost extends BasicRestServlet {
59  		private static final long serialVersionUID = 1L;
60  		@Bean public FreemarkerMixin freemarker() {
61  			return ConsoleFreemarkerMixin.create().basePath("/templates/").chromeTemplate("c1/chrome-selected.ftlh").build();
62  		}
63  		@RestGet(path="/t/{name}")
64  		public View t(@Path("name") String name, RestRequest req) {
65  			return C1Fixtures.view(name, req);
66  		}
67  	}
68  
69  	@Rest(mixins=FreemarkerMixin.class, renderResponseStackTraces="true")
70  	public static class KpiHost extends BasicRestServlet {
71  		private static final long serialVersionUID = 1L;
72  		@Bean public FreemarkerMixin freemarker() {
73  			return ConsoleFreemarkerMixin.create().basePath("/templates/").chromeTemplate("c1/chrome.ftlh").cardType(new KpiCardType()).build();
74  		}
75  		@RestGet(path="/t/{name}")
76  		public View t(@Path("name") String name, RestRequest req) {
77  			return C1Fixtures.view(name, req);
78  		}
79  	}
80  
81  	static View view(String name, RestRequest req) {
82  		if (name.startsWith("csrf-")) {
83  			req.setAttribute(LoopbackBoundaryFilter.TOKEN_ATTRIBUTE, "tok-123");
84  			req.setAttribute(LoopbackBoundaryFilter.HEADER_ATTRIBUTE, "X-Csrf-Token");
85  		}
86  		return FreemarkerView.of("c1/" + name + ".ftlh");
87  	}
88  
89  	private static final MockRestClient HOST = MockRestClient.buildLax(Host.class);
90  	private static final MockRestClient SELECTED = MockRestClient.buildLax(SelectedHost.class);
91  	private static final MockRestClient KPI = MockRestClient.buildLax(KpiHost.class);
92  
93  	/** Renders {@code c1/{name}.ftlh} under {@code c1/chrome.ftlh} and expects HTTP 200. */
94  	static String render(String name) {
95  		return get(HOST, name, 200);
96  	}
97  
98  	/** Renders {@code c1/{name}.ftlh} under {@code c1/chrome-selected.ftlh} and expects HTTP 200. */
99  	static String renderSelected(String name) {
100 		return get(SELECTED, name, 200);
101 	}
102 
103 	/** Renders {@code c1/{name}.ftlh} on a host whose mixin registered {@link KpiCardType} and expects HTTP 200. */
104 	static String renderKpi(String name) {
105 		return get(KPI, name, 200);
106 	}
107 
108 	/** Renders {@code c1/{name}.ftlh} and expects HTTP 500; returns the error body. */
109 	static String renderError(String name) {
110 		return get(HOST, name, 500);
111 	}
112 
113 	/** Asserts the 500 body for {@code name} contains {@code message} verbatim. */
114 	static void assertError(String name, String message) {
115 		var body = renderError(name);
116 		assertTrue(body.contains(message), () -> "expected '" + message + "' in:\n" + body);
117 	}
118 
119 	private static String get(MockRestClient c, String name, int status) {
120 		try (var rsp = c.get("/t/" + name).run()) {
121 			rsp.assertStatus(status);
122 			return rsp.getContent().asString();
123 		} catch (Exception e) {
124 			throw new AssertionError("GET /t/" + name + ": " + e.getMessage(), e);
125 		}
126 	}
127 }
