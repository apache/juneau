30  	private static Map<?,?> map(ChartDef d) {
31  		return Json5.to(Json.of(d.toCardSpec().toMap()), Map.class);
32  	}
33  
34  	private static String fail(Runnable r) {
35  		return assertThrows(IllegalArgumentException.class, r::run).getMessage();
36  	}
37  
38  	private static ChartDef chart() {
39  		return ChartDef.bar("orders").src("/stats").x("day").series("n", "N");
40  	}
41  
42  	private static RowAction requeue() {
43  		return RowAction.create("requeue").label("Requeue").endpoint("/api/jobs/requeue").method(RowAction.Method.POST);
44  	}
45  
46  	private static Map<?,?> body(ChartDef d) { return (Map<?,?>)map(d); }
47  
48  	@Test void a01_onClickAction_serializesRowActionJson_andAutoDeclaresClickTopic() {
49  		var m = body(chart().onClickAction(requeue()));
50  		var action = (Map<?,?>)((Map<?,?>)m.get("onClick")).get("action");
51  		assertEquals("requeue", action.get("id"));
52  		assertEquals("/api/jobs/requeue", action.get("endpoint"));
53  		assertEquals("POST", action.get("method"));
54  		assertEquals("orders.click", ((Map<?,?>)m.get("on")).get("click"));
55  		assertTrue(((List<?>)m.get("publishes")).toString().contains("orders.click"));
56  	}
57  
58  	@Test void a01b_onClickAction_idNotATopicNamespace_throws() {
59  		assertEquals("ChartDef 'Orders_1' onClickAction cannot derive a click topic from the id; declare one with on(\"click\", topic).",
60  			fail(() -> ChartDef.bar("Orders_1").src("/s").x("d").series("n", "N").onClickAction(requeue())));
61  	}

63  	@Test void a02_onClickAction_keepsExistingClickTopic() {
64  		var m = body(chart().on("click", "pet.selected").onClickAction(requeue()));
65  		assertEquals("pet.selected", ((Map<?,?>)m.get("on")).get("click"));
66  	}

68  	@Test void a03_onTopicAction_undeclaredTopic_throws() {
69  		assertEquals("ChartDef 'orders' onTopicAction topic 'pet.selected' must be one this card publishes through on(...).",
70  			fail(() -> chart().onTopicAction("pet.selected", requeue())));
71  	}
72  
73  	@Test void a04_onTopicAction_declaredTopic_writesTopicActions() {
74  		var m = body(chart().on("markclick", "pet.selected").onTopicAction("pet.selected", requeue()));
75  		var ta = (List<?>)m.get("topicActions");
76  		assertEquals(1, ta.size());
77  		assertEquals("pet.selected", ((Map<?,?>)ta.get(0)).get("topic"));
78  		assertEquals("requeue", ((Map<?,?>)((Map<?,?>)ta.get(0)).get("action")).get("id"));
79  	}

81  	@Test void a05_hrefPlusAction_throws_eitherOrder() {
82  		var msg = "ChartDef 'orders' onClick takes either href or action, not both.";
83  		assertEquals(msg, fail(() -> chart().onClickHref("/x/{id}").onClickAction(requeue())));
84  		assertEquals(msg, fail(() -> chart().onClickAction(requeue()).onClickHref("/x/{id}")));
85  	}

122 	@Test void a13_presentDialog_withAndWithoutForm_passesAndRoundTrips() {
123 		var plain = (Map<?,?>)((Map<?,?>)body(chart().onClickAction(requeue().present(RowAction.Present.DIALOG))).get("onClick")).get("action");
124 		assertEquals("dialog", plain.get("present"));
125 		assertNull(plain.get("form"));
126 		var withForm = (Map<?,?>)((Map<?,?>)body(chart().onClickAction(requeue().present(RowAction.Present.DIALOG).form("/forms/requeue"))).get("onClick")).get("action");
127 		assertEquals("/forms/requeue", withForm.get("form"));
128 	}

130 	@Test void a14_confirmString_roundTripsAsString_noConfirm_roundTripsFalse() {
131 		var s = (Map<?,?>)((Map<?,?>)body(chart().onClickAction(requeue().confirm("Sure?"))).get("onClick")).get("action");
132 		assertEquals("Sure?", s.get("confirm"));
133 		var n = (Map<?,?>)((Map<?,?>)body(chart().onClickAction(requeue().noConfirm())).get("onClick")).get("action");
134 		assertEquals(Boolean.FALSE, n.get("confirm"));
135 	}

141 	@Test void a15_actions_roundTrip_andAccumulateAcrossCalls() {
142 		var m = body(chart().actions(dlg("a")).actions(dlg("b"), dlg("c")));
143 		var l = (List<?>)m.get("actions");

154 	@Test void a17_duplicateId_catalogVsCatalog_vsOnClickAction_vsOnTopicAction_throws() {
155 		var dup = "ChartDef 'orders' action id 'requeue' is declared more than once.";
156 		assertEquals(dup, fail(() -> chart().actions(dlg("requeue"), dlg("requeue"))));
157 		assertEquals(dup, fail(() -> chart().onClickAction(requeue()).actions(dlg("requeue"))));
158 		assertEquals(dup, fail(() -> chart().actions(dlg("requeue")).onClickAction(requeue())));
159 		assertEquals(dup, fail(() -> chart().on("m", "pet.sel").onTopicAction("pet.sel", requeue()).actions(dlg("requeue"))));
160 		assertEquals(dup, fail(() -> chart().on("m", "pet.sel").onClickAction(requeue()).onTopicAction("pet.sel", requeue())));
161 	}

169 	@Test void a19_everyPerActionRule_appliesToCatalogEntries() {
170 		assertTrue(fail(() -> chart().actions(dlg("a").endpoint("servlet:/x"))).contains("actions[0] endpoint must be a same-origin path"));
171 		assertTrue(fail(() -> chart().actions(dlg("a").form("https://evil.example/f"))).contains("actions[0] form must be a same-origin path"));
172 		assertTrue(fail(() -> chart().actions(dlg("a").onSuccess(RowAction.OnSuccess.NAVIGATE))).contains("actions[0] onSuccess must not be 'navigate'"));
173 		assertTrue(fail(() -> chart().actions(dlg("a").endpoint(null))).contains("actions[0] requires a non-blank endpoint"));
174 	}
