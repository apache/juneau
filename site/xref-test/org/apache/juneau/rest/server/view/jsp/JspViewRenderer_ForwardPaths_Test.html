33  /**
34   * Covers {@link JspViewRenderer#process process(...)}'s dispatch-succeeded and dispatch-failed
35   * branches (the {@code rd != null} half of the {@code rd == null} check, plus every {@code catch}
36   * clause around {@code rd.forward(...)}).
37   *
38   * <p>
39   * Mirrors {@link JspDispatcher_ForwardPaths_Test}'s technique (a hand-rolled {@link ServletContext}
40   * swapped in via {@link MockRestRequest#servletContext}) but exercises the sibling code path reached
41   * when an {@code @RestOp} method returns a {@link JspView} rather than a raw {@code /jsp/*} request.
42   *
43   * <p>
44   * Fixture {@code A} declares {@code @Rest(responseProcessors=JspViewRenderer.class)} directly
45   * (the pattern proven to work by {@code JspView_TypedHandler_Test} in {@code juneau-integration-tests}).
46   * A bare {@code @Rest(mixins=JspMixin.class)} now reaches the same code path too, because {@code JspMixin}
47   * declares {@link Rest#mergeResponseProcessorsIntoHost() @Rest(mergeResponseProcessorsIntoHost=true)} on its
48   * own class, folding {@link JspViewRenderer} into the host's own chain — see {@code z01} below.
49   *
50   * @since 10.0.0
51   */
52  @SuppressWarnings({
53  	"resource" // Closeable test fixtures held in static fields; lifecycle managed by the test/framework, not a real leak.
54  })
55  class JspViewRenderer_ForwardPaths_Test extends TestBase {

57  	@Rest(responseProcessors=JspViewRenderer.class)
58  	public static class A extends BasicRestServlet {
59  		private static final long serialVersionUID = 1L;
60  		@RestGet(path="/view")
61  		public View view() {
62  			return JspView.of("hello.jsp");
63  		}
64  		@RestGet(path="/escape")
65  		public View escape() {
66  			// Not user input (see the class-level javadoc on JspViewRenderer.joinPath): a caller-assembled
67  			// template name that itself escapes the configured base path via ".." segments.
68  			return JspView.of("../../etc/passwd");
69  		}
70  		@RestGet(path="/plain")
71  		public String plain() {
72  			return "not a JspView";
73  		}
74  	}

78  	// A bare @Rest(mixins=JspMixin.class) DOES route a HOST-defined @RestGet method's JspView return value
79  	// through JspViewRenderer, because JspMixin declares @Rest(mergeResponseProcessorsIntoHost=true) on its own
80  	// class. That opt-in folds JspMixin's own @Rest(responseProcessors=JspViewRenderer.class) into the host's
81  	// own response-processor chain (response-processor-scoped only), so a host op returning a JspView reaches
82  	// the renderer with no extra wiring. See z01 below, JspMixin's class javadoc, and
83  	// MixinResponseProcessorFold_Test (rest-server) / MixinInheritance_ResponseProcessors_Test
84  	// (juneau-integration-tests), which pin the fold at the RestContext and mock-client levels respectively.
85  	@Rest(mixins=JspMixin.class)
86  	public static class MixinOnly extends BasicRestServlet {
87  		private static final long serialVersionUID = 1L;
88  		@RestGet(path="/view")
89  		public View view() {
90  			return JspView.of("hello.jsp");
91  		}
92  	}
93  
94  	private static final MockRestClient cMixinOnly = MockRestClient.buildLax(MixinOnly.class);
95  
96  	// Non-silent regression guard: a mixin that registers JspViewRenderer but does NOT declare the
97  	// mergeResponseProcessorsIntoHost opt-in stays scoped to its own endpoints -- its responseProcessors are
98  	// NOT folded into the host's chain, so a host op returning a JspView falls back to bean-serialization.
99  	// This is the deliberately-isolated default that JspMixin overrides by opting in (see z03 vs z01).
100 	@Rest(responseProcessors=JspViewRenderer.class)
101 	public static class NoFoldMixin {
102 		@RestGet(path="/mixin-only-endpoint")
103 		public String noop() {
104 			return "noop";
105 		}
106 	}

112 		public View view() {

114 		}

119 	// Opt-in host: adopts JspMixin via the rich mixinDefs form with mergeIntoHost=true, so the mixin's
120 	// @Rest(responseProcessors=JspViewRenderer.class) folds into THIS host's own chain.
121 	@Rest(mixinDefs=@Mixin(type=JspMixin.class, mergeIntoHost=true))
122 	public static class MergeIntoHost extends BasicRestServlet {
123 		private static final long serialVersionUID = 1L;
124 		@RestGet(path="/view")
125 		public View view() {
126 			return JspView.of("hello.jsp");
127 		}
128 	}
129 
130 	private static final MockRestClient cMergeIntoHost = MockRestClient.buildLax(MergeIntoHost.class);

132 	@Test void z01_mixinAlone_routesHostViewReturnThroughRenderer() throws Exception {
133 		// JspMixin declares @Rest(mergeResponseProcessorsIntoHost=true), so a bare @Rest(mixins=JspMixin.class)
134 		// folds JspViewRenderer into the host's own chain: a host @RestGet returning a JspView is CLAIMED by the
135 		// renderer (not bean-serialized). With a well-behaved dispatcher that commits the response, the render
136 		// succeeds end-to-end (200) -- deterministic proof the fold routes AND renders, not merely that it was
137 		// intercepted. The companion null-dispatcher assertion below pins that it reached the renderer rather than
138 		// the default SerializedPojoProcessor (which would have produced a 200 "templateName" body, as in z03).
139 		var okCtx = fakeServletContext(new FakeDispatcher(() -> { /* well-behaved engine committed the response */ }));
140 		cMixinOnly.get("/view").servletContext(okCtx).run().assertStatus(200);
141 
142 		var noEngineCtx = fakeServletContext(null);
143 		var res = cMixinOnly.get("/view").servletContext(noEngineCtx).run();
144 		res.assertStatus(500);
145 		res.assertContent().asString().isContains("Could not resolve RequestDispatcher");
146 		res.assertContent().asString().isContains("No JSP engine is available on the classpath");
147 	}
148 
149 	@Test void z03_nonOptedInMixin_doesNotRouteHostViewReturnThroughRenderer() throws Exception {
150 		// Regression guard for the non-silent contract: NoFoldMixin registers JspViewRenderer but does NOT opt in
151 		// via mergeResponseProcessorsIntoHost, so the renderer stays scoped to the mixin's own endpoints and is
152 		// NOT folded into the host's chain. The host's JspView return therefore falls back to bean-serialization
153 		// (SerializedPojoProcessor won) -- a 200 whose body contains the JspView bean's own "templateName" field,
154 		// which JspViewRenderer's dispatch path never produces. Contrast with z01 (opted-in JspMixin routes).
155 		var res = cNoFoldMixin.get("/view").accept("application/json").run();
156 		res.assertStatus(200);
157 		res.assertContent().asString().isContains("templateName");
158 	}
159 
160 	@Test void z02_mergeIntoHost_routesHostViewReturnThroughRenderer() throws Exception {
161 		// Opt-in @Mixin(mergeIntoHost=true) folds JspMixin's @Rest(responseProcessors=JspViewRenderer.class)
162 		// into the HOST's own chain, so a host @RestGet returning a JspView is now CLAIMED by JspViewRenderer
163 		// (not bean-serialized). With a ServletContext that resolves no dispatcher, the renderer surfaces its
164 		// NO_ENGINE_DIAGNOSTIC 500 -- deterministic proof the JspView reached JspViewRenderer rather than the
165 		// default SerializedPojoProcessor (which would have produced a 200 "templateName" body as in z03).
166 		var ctx = fakeServletContext(null);
167 		var res = cMergeIntoHost.get("/view").servletContext(ctx).run();
168 		res.assertStatus(500);
169 		res.assertContent().asString().isContains("Could not resolve RequestDispatcher");
170 		res.assertContent().asString().isContains("No JSP engine is available on the classpath");
171 	}

173 	/** Functional seam for {@link #forward(ServletRequest, ServletResponse)} on the fake dispatcher below. */
174 	@FunctionalInterface
175 	private interface ForwardAction {
176 		void run() throws ServletException, IOException;
177 	}
178 
179 	/** Minimal {@link RequestDispatcher} whose {@code forward(...)} delegates to a configurable action. */
180 	private static final class FakeDispatcher implements RequestDispatcher {
181 		private final ForwardAction action;
182 		FakeDispatcher(ForwardAction action) { this.action = action; }
183 		@Override public void forward(ServletRequest req, ServletResponse res) throws ServletException, IOException { action.run(); }
184 		@Override public void include(ServletRequest req, ServletResponse res) { /* not exercised by process(...) */ }
185 	}
186 
187 	/** See {@code JspDispatcher_ForwardPaths_Test.fakeServletContext} for why a bare {@link Proxy} suffices. */
188 	private static ServletContext fakeServletContext(RequestDispatcher dispatcher) {
189 		return (ServletContext) Proxy.newProxyInstance(
190 			JspViewRenderer_ForwardPaths_Test.class.getClassLoader(),
191 			new Class<?>[] { ServletContext.class },
192 			(proxy, method, args) -> eq(method.getName(), "getRequestDispatcher") ? dispatcher : null);
193 	}

196 		// The `content instanceof JspView` check's other branch: a non-JspView return value must NOT be
197 		// claimed by this processor -- it returns NEXT so the standard POJO serializer chain handles it.
198 		var res = c.get("/plain").run();
199 		res.assertStatus(200);
200 		res.assertContent().asString().isContains("not a JspView");

203 	@Test void a0b_templateNameEscapesBasePath_wrappedAs500() throws Exception {
204 		// joinPath(...) throws IllegalArgumentException when the template name resolves outside
205 		// basePath; process(...) catches that (distinct from the rd == null / forward() catches below,
206 		// which all run later in the method) and wraps it with a dedicated escape diagnostic.
207 		var res = c.get("/escape").run();
208 		res.assertStatus(500);
209 		res.assertContent().asString().isContains("escapes configured base path");
210 	}

212 	@Test void a00_contextPresentButDispatcherNull_reportsNoEngineDiagnostic() throws Exception {
213 		// The rd == null branch itself (a non-null ServletContext that resolves the target to no
214 		// dispatcher), as distinct from a null ServletContext (which never reaches this class's process()
215 		// call at all under MockRest's defaults). process(...) re-throws InternalServerError ahead of the
216 		// generic catch, so this branch's specific "Could not resolve RequestDispatcher...
217 		// NO_ENGINE_DIAGNOSTIC" message reaches the response body intact instead of being re-wrapped into
218 		// the generic "JSP render failed for" message.
219 		var ctx = fakeServletContext(null);
220 		var res = c.get("/view").servletContext(ctx).run();
221 		res.assertStatus(500);
222 		res.assertContent().asString().isContains("Could not resolve RequestDispatcher");
223 		res.assertContent().asString().isContains("No JSP engine is available on the classpath");
224 	}
