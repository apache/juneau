17  package org.apache.juneau.rest.server.view.freemarker.console;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.io.*;
22  import java.nio.charset.*;
23  import java.nio.file.*;
24  
25  /**
26   * Byte-for-byte golden-file assertions for refactor safety nets.
27   *
28   * <p>
29   * A missing golden is written and the test fails, so a first run never passes silently. Re-generate deliberately
30   * with {@code -Djuneau.golden.update=true}, then review the diff before re-running without it.
31   *
32   * <h5 class='section'>Example:</h5>
33   * <p class='bjava'>
34   * 	GoldenFiles.<jsm>assertGolden</jsm>(<js>"chrome-css"</js>, <js>"gray"</js>, <jv>body</jv>);
35   * 	<jc>// compares against src/test/resources/golden/chrome-css/gray.txt</jc>
36   * </p>
37   */
38  final class GoldenFiles {
39  
40  	private static final boolean UPDATE = Boolean.getBoolean("juneau.golden.update");
41  
42  	private GoldenFiles() {}
43  
44  	static void assertGolden(String dir, String name, String actual) throws IOException {
45  		var file = Path.of("src/test/resources/golden", dir, name + ".txt");
46  		if (UPDATE || ! Files.exists(file)) {
47  			Files.createDirectories(file.getParent());
48  			Files.writeString(file, actual, StandardCharsets.UTF_8);
49  			fail("Golden written: " + file.toAbsolutePath() + " - review it, then re-run without -Djuneau.golden.update.");
50  		}
51  		assertEquals(Files.readString(file, StandardCharsets.UTF_8), actual, () -> "Golden mismatch: " + file);
52  	}
53  }
