19  import static org.apache.juneau.test.bct.BctAssertions.assertSize;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.util.*;
23  
24  import org.apache.juneau.rest.server.views.*;
25  import org.junit.jupiter.api.*;
26  
27  /**
28   * {@code CardSpec.visibleWhen} serialization tests.
29   *
30   * @since 10.0.0
31   */
32  class CardSpec_VisibleWhen_Test {
33  
34  	@Test void a01_absent_omitsKey() {
35  		assertFalse(CardSpec.html("c1").toMap().containsKey("visibleWhen"));
36  	}
37  
38  	@Test void a02_serializesEachRule() {
39  		var m = CardSpec.html("c1").visibleWhen(VisibilityRule.when("role").in("admin", "owner")).toMap();
40  		@SuppressWarnings({
41  			"unchecked" // Casts a parsed JSON value to its known Map/List shape.
42  		})
43  		var rules = (List<Map<String,Object>>)m.get("visibleWhen");
44  		assertSize(1, rules);
45  		assertEquals("role", rules.get(0).get("field"));
46  		assertEquals("in", rules.get(0).get("op"));
47  		assertEquals(List.of("admin", "owner"), rules.get(0).get("value"));
48  	}
49  }
