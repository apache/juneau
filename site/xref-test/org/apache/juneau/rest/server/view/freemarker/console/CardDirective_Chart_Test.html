32  /** The {@code <@card type="chart">} card and its {@code charts} toolkit pack. */
33  class CardDirective_Chart_Test extends TestBase {
34  
35  	private static JsonMap card(String body, String id) {
36  		for (var o : assertPage(body).isValid().contract().getList("cards"))
37  			if (id.equals(((JsonMap)o).getString("id")))
38  				return (JsonMap)o;
39  		throw new AssertionError("no card '" + id + "' in " + body);
40  	}
41  
42  	@Test void a01_srcCard() {
43  		var body = render("ch-card");
44  		assertEquals(
45  			Json5.to("{id:'trend',type:'chart',title:'Orders',src:'/console/store/stats/daily',kind:'line',x:{field:'day'},"
46  				+ "series:[{field:'placed',label:'Placed'}],refreshMs:60000}", Map.class),
47  			Json5.to(Json.of(card(body, "trend")), Map.class));
48  	}
49  
50  	@Test void a02_dataCard() {
51  		assertEquals("donut", card(render("ch-card-data"), "mix").getString("kind"));
52  	}
53  
54  	@Test void a03_chartsScriptEmittedOnce() {
55  		var body = render("ch-card");
56  		var s = "/juneau-console/juneau-charts.js?v=";
57  		assertTrue(body.contains(s), () -> body);
58  		assertEquals(body.indexOf(s), body.lastIndexOf(s), () -> body);
59  	}

65  		"ch-card-both|<@card id='trend'> type='chart' requires exactly one of src=, a data body or a source.",
66  	})
