165 	 * {@code <@tag domain="status" value=release/>} (a raw enum constant bound as a template attribute) produces
166 	 * literal pill markup, not the {@code DefaultObjectWrapper} bean-dump/empty footgun a bare {@code ${...}}

169 	@Test void d01_tagMacro_rendersLiteralPillMarkup_notEscaped() throws Exception {
170 		var c = MockRestClient.buildLax(ConsoleHost.class);
171 		var body = c.get("/mypage").run().assertStatus(200).getContent().asString();
172 		assertFalse(body.contains("&lt;span"), () -> "macro output was HTML-escaped (double-escaped), body:\n" + body);
173 		// find() on an anchor-free pattern (no wrapping .*) avoids the super-linear backtracking risk of
174 		// String.matches() with unbounded quantifiers at both ends.
175 		assertTrue(Pattern.compile("<span(?=[^>]*class=['\"]tag status released['\"])[^>]*>").matcher(body).find(),
176 			() -> "expected literal <span class='tag status released'> markup, body:\n" + body);
177 	}

179 	/**
180 	 * A bare {@code ${Tag.of(...)}} (never done through the macro in this codebase, but documenting the footgun
181 	 * directly) is NOT equivalent to calling the Java factory and stringifying its result through the trusted
182 	 * adapter: {@link org.apache.juneau.rest.server.console.Tag#of} returns an {@code HtmlElement}, and the
183 	 * bridge-default {@code DefaultObjectWrapper} (exposeFields=true) wraps arbitrary objects as bean models, not
184 	 * pre-rendered markup -- so the ONLY supported v1 insertion point is the {@code <@tag>} macro documented on
185 	 * {@link ConsoleFreemarkerMixin}, never a direct method-model call written by hand in a template.
186 	 */
187 	@Test void d02_tagMethodModel_isNotExposedAsABareTemplateVariable() throws Exception {
188 		var console = ConsoleFreemarkerMixin.create().basePath("/templates/").build();
189 		var cfg = console.resolveConfiguration(dummyRequest());
190 		// jcTagHtml is a shared variable on the augmented Configuration -- exists ONLY so base.ftlh's <#macro tag>
191 		// can call it; it is deliberately not part of any public API surface a consumer would discover/write by
192 		// hand (no javadoc, package-private class). This assertion documents that the shared variable is scoped
193 		// to the internal macro plumbing, not a public "insertion point" a consumer should call directly.
194 		assertNotNull(cfg.getSharedVariable(TagMethodModel.NAME), "the <@tag> macro's backing method must be registered");
195 	}

218 	/**
219 	 * Two successive {@code resolveConfiguration(req)} calls on the same {@code ConsoleFreemarkerMixin} (no
220 	 * {@code Configuration} bean) must return the same cached {@code Configuration} instance whose
221 	 * {@code getTemplateLoader()} is a {@code MultiTemplateLoader} of length 2 <b>both times</b> &mdash; not
222 	 * {@code Multi(Multi(base, console), console)} on the second call. Two independent HTTP requests hit the same
223 	 * mixin-bean singleton (one {@code MockRestClient}/{@code RestContext}), each triggering one
224 	 * {@code resolveConfiguration(req)} call.
225 	 */
226 	@Test void e01_secondResolveConfigurationCall_doesNotNestTheLoader() throws Exception {
227 		var c = MockRestClient.buildLax(DoubleWrapProbeHost.class);
228 		var r1 = c.get("/probe").accept("text/plain").run().getContent().asString();
229 		var r2 = c.get("/probe").accept("text/plain").run().getContent().asString();
230 		var p1 = r1.split(":");
231 		var p2 = r2.split(":");
232 		assertEquals(3, p1.length, () -> "expected a MultiTemplateLoader on the first call, got: " + r1);
233 		assertEquals(3, p2.length, () -> "expected a MultiTemplateLoader on the second call, got: " + r2);
234 		assertEquals(p1[2], p2[2], "both calls must return the SAME cached Configuration instance");
235 		assertEquals("2", p1[0], "first call: expected exactly 2 loaders (base + console)");
236 		assertEquals("2", p2[0], () -> "double-wrap: second call's loader count grew past 2 (nested Multi), got " + p2[0]);
237 		assertEquals(p1[1], p2[1], "double-wrap: index-0 loader identity changed between calls (got re-wrapped)");
238 	}

244 	@Rest(mixins=FreemarkerMixin.class)
245 	public static class ConsumerConfigHost extends BasicRestServlet {
246 		private static final long serialVersionUID = 1L;
247 		static final Configuration USER_CFG = new Configuration(Configuration.VERSION_2_3_34);
248 		static {
249 			USER_CFG.setTemplateLoader(new StringTemplateLoader());
250 		}
251 		static final Object ORIGINAL_WRAPPER = USER_CFG.getObjectWrapper();
252 		static final String ORIGINAL_ENCODING = USER_CFG.getDefaultEncoding();
253 		static final Object ORIGINAL_OUTPUT_FORMAT = USER_CFG.getOutputFormat();
254 		static final long ORIGINAL_UPDATE_DELAY = USER_CFG.getTemplateUpdateDelayMilliseconds();
255 		@Bean public Configuration configuration() { return USER_CFG; }
256 		@Bean public FreemarkerMixin freemarker() { return ConsoleFreemarkerMixin.create().build(); }
257 		@RestGet(path="/probe")
258 		public String probe(RestRequest req) {
259 			var mixin = (ConsoleFreemarkerMixin) req.getContext().getBeanStore().getBean(FreemarkerMixin.class).orElseThrow();
260 			var cfg1 = mixin.resolveConfiguration(req);
261 			var cfg2 = mixin.resolveConfiguration(req);
262 			var sameInstance = cfg1 == USER_CFG && cfg2 == USER_CFG;
263 			var settingsUnchanged = cfg1.getObjectWrapper() == ORIGINAL_WRAPPER
264 				&& eq(cfg1.getDefaultEncoding(), ORIGINAL_ENCODING)
265 				&& cfg1.getOutputFormat() == ORIGINAL_OUTPUT_FORMAT
266 				&& cfg1.getTemplateUpdateDelayMilliseconds() == ORIGINAL_UPDATE_DELAY;
267 			var varsFilled = cfg1.getSharedVariable(PageDirectiveModel.NAME) != null
268 				&& cfg1.getSharedVariable(CardDirectiveModel.NAME) != null
269 				&& cfg1.getSharedVariable(NavigationDirectiveModel.NAME) != null
270 				&& cfg1.getSharedVariable(NodeDirectiveModel.NAME) != null
271 				&& cfg1.getSharedVariable(ThemeDirectiveModel.NAME) != null
272 				&& cfg1.getSharedVariable(TagMethodModel.NAME) != null;
273 			var reservedResolves = true;
274 			try {
275 				cfg1.getTemplate(ConsoleFreemarkerMixin.BASE_TEMPLATE_PATH);
276 			} catch (@SuppressWarnings("unused") Exception ex) {
277 				reservedResolves = false;
278 			}
279 			return sameInstance + ":" + settingsUnchanged + ":" + varsFilled + ":" + reservedResolves
280 				+ ":" + loaderCount(cfg1) + ":" + loaderCount(cfg2);
281 		}
282 	}
