32  /**
33   * The {@code <@card type="run-view">} bridge card: a JSON5 {@code {contractVersion, runView}} body, validated
34   * through {@code RunViewDef} at render time and emitted as a {@code cards[]} entry with a {@code runView} object.
35   */
36  class CardDirective_RunView_Test extends TestBase {
37  
38  	/** The contract entry for card {@code id}. */
39  	private static JsonMap card(String body, String id) {
40  		for (var o : assertPage(body).isValid().contract().getList("cards"))
41  			if (id.equals(((JsonMap)o).getString("id")))
42  				return (JsonMap)o;
43  		throw new AssertionError("no card '" + id + "' in " + body);
44  	}
45  
46  	@Test void a01_fullCard() {
47  		var body = render("rv-card");
48  		assertPage(body).isValid().hasCard("run-42", "run-view");
49  		assertEquals(
50  			Json5.to("{id:'run-42',type:'run-view',title:'Run',runView:{eventsUrl:'/juneau-run-view/42/events',"
51  				+ "refreshMs:1000,poll:true,compact:false,title:'Build run',rawHref:'/runs/42/raw#O{offset}'}}", Map.class),
52  			Json5.to(Json.of(card(body, "run-42")), Map.class));
53  		assertFalse(body.contains("<template data-card=\"run-42\">"), () -> body);
54  	}
55  
56  	@Test void a02_minimalCard_hasNoTitleAndOnlyTheUrlAndDefaults() {
57  		var c = card(render("rv-card-minimal"), "run");
58  		assertEquals("run-view", c.getString("type"));
59  		assertNull(c.get("title"));
60  		assertEquals("/runs/1/events", ((JsonMap)c.get("runView")).getString("eventsUrl"));
61  	}
62  
63  	@Test void a03_pollFalse_needsNoUrl() {
64  		var rv = (JsonMap)card(render("rv-card-nopoll"), "run").get("runView");
65  		assertNull(rv.get("eventsUrl"));
66  		assertEquals(false, rv.get("poll"));
67  	}
68  
69  	@Test void a04_requiresIsStillAccepted() {
70  		assertPage(render("rv-card-requires")).isValid().hasCard("run", "run-view");
71  	}
72

74  	@CsvSource(delimiter='|', value={
75  		"rv-card-noid|<@card type=\"run-view\"> requires id=.",
76  		"rv-card-src|<@card id='run'> type='run-view' takes its options as the body; src= is not allowed.",
77  		"rv-card-both|<@card id='run'> type='run-view' takes its options as the body; src= and template= are not allowed.",
78  		"rv-card-template|<@card id='run'> type='run-view' takes its options as the body; template= is not allowed.",
79  		"rv-card-empty|<@card id='run'> type='run-view' requires a JSON5 body { contractVersion: '1', runView: {...} }.",
80  		"rv-card-notjson|<@card id='run'> type='run-view' requires a JSON5 body { contractVersion: '1', runView: {...} }.",
81  		"rv-card-noversion|<@card id='run'> type='run-view' requires contractVersion: '1'; got 'null'.",
82  		"rv-card-badversion|<@card id='run'> type='run-view' requires contractVersion: '1'; got '2'.",
