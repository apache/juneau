27  /**
28   * Enforces the standing Juneau convention that every new public type carries a class-level
29   * {@code <h5 class='section'>Example:</h5>} Javadoc block and {@code @since 10.0.0}, for exactly the top-level
30   * public types the C3 page-builder work created. Reads source text, not compiled Javadoc (doc comments do not
31   * survive into {@code .class} files); a renamed or moved file fails loudly so the audit cannot silently go stale.
32   *
33   * <h5 class='section'>Example:</h5>
34   * <p class='bjava'>
35   * 	<jsm>assertHasExampleBlock</jsm>(<jv>juneauRoot</jv>.resolve(<js>"juneau-rest/.../TableSpec.java"</js>));
36   * </p>
37   *
38   * @since 10.0.0
39   */
40  class PublicTypes_Javadoc_Test {
41  
42  	/** @return The juneau checkout root: the nearest ancestor of the working directory holding {@code juneau-bom}. */
43  	private static Path juneauRoot() {
44  		var cwd = Path.of("").toAbsolutePath();
45  		var p = cwd;
46  		while (p != null && ! Files.exists(p.resolve("juneau-bom")))
47  			p = p.getParent();
48  		if (p == null)
49  			throw new IllegalStateException("Could not locate the juneau checkout root above '" + cwd + "'.");
50  		return p;
51  	}

54  	private static final String CUIF = "juneau-rest/juneau-rest-server-console-ui-freemarker/src/main/java/org/apache/juneau/rest/server/view/freemarker/console/";

60  		CUIF + "PageSpec.java", CUIF + "BadgeDirectiveModel.java", CUIF + "FactsDirectiveModel.java",

67  	private static void assertHasExampleBlock(Path file) throws IOException {
68  		assertTrue(Files.exists(file), () -> "Missing file, update NEW_PUBLIC_TYPE_FILES: '" + file + "'.");
69  		var text = Files.readString(file);
70  		var typeIdx = text.indexOf("public final class");
71  		if (typeIdx < 0)
72  			typeIdx = text.indexOf("public class");
73  		if (typeIdx < 0)
74  			typeIdx = text.indexOf("public interface");
75  		if (typeIdx < 0)
76  			typeIdx = text.indexOf("public abstract class");
77  		assertTrue(typeIdx >= 0, () -> "No public type declaration found in '" + file + "'.");
78  		var javadoc = text.substring(0, typeIdx);
79  		assertTrue(javadoc.contains("<h5 class='section'>Example:</h5>"),
80  			() -> "'" + file + "' is missing the class-level <h5 class='section'>Example:</h5> Javadoc block.");
81  		assertTrue(javadoc.contains("@since 10.0.0"),
82  			() -> "'" + file + "' is missing @since 10.0.0 on its class-level Javadoc.");
83  	}
84  
85  	@Test void everyNewPublicType_hasExampleBlock() throws Exception {
86  		var root = juneauRoot();
87  		for (var rel : NEW_PUBLIC_TYPE_FILES)
88  			assertHasExampleBlock(root.resolve(rel));
89  	}
