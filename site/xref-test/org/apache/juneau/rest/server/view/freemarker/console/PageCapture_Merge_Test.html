17  package org.apache.juneau.rest.server.view.freemarker.console;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.*;
24  import org.junit.jupiter.api.*;
25  
26  import freemarker.template.*;
27  
28  /**
29   * {@link PageCapture}'s merge-aware setters: single-value conflict fields and union fields.
30   *
31   * @since 10.0.0
32   */
33  class PageCapture_Merge_Test extends TestBase {
34  
35  	@Test void a01_mergeTab_blankInputIsNoOp() throws TemplateModelException {
36  		var cap = new PageCapture();
37  		cap.mergeTab("");
38  		assertEquals("", cap.tab());
39  	}
40  
41  	@Test void a02_mergeTab_firstWriterWins() throws TemplateModelException {
42  		var cap = new PageCapture();
43  		cap.mergeTab("slo");
44  		assertEquals("slo", cap.tab());
45  	}
46  
47  	@Test void a03_mergeTab_sameValueTwiceRejected() throws TemplateModelException {
48  		var cap = new PageCapture();
49  		cap.mergeTab("slo");
50  		var ex = assertThrows(TemplateModelException.class, () -> cap.mergeTab("slo"));
51  		assertEquals("PageSpec sets tab='slo' and <@page tab='slo'> also sets it; set it in one place.", ex.getMessage());
52  	}
53  
54  	@Test void a04_mergeTab_conflictRejected() throws TemplateModelException {
55  		var cap = new PageCapture();
56  		cap.mergeTab("slo");
57  		var ex = assertThrows(TemplateModelException.class, () -> cap.mergeTab("activity"));
58  		assertEquals("PageSpec sets tab='slo' and <@page tab='activity'> also sets it; set it in one place.", ex.getMessage());
59  	}
60  
61  	@Test void a05_mergeTheme_conflictRejected() {
62  		var cap = new PageCapture();
63  		cap.theme("light-red");  // As a page spec seeds it.
64  		var ex = assertThrows(TemplateModelException.class, () -> cap.mergeTheme("open"));
65  		assertEquals("PageSpec theme 'light-red' conflicts with <@theme name='open'>; remove name= from <@theme> to inherit the page theme.", ex.getMessage());
66  	}
67  
68  	@Test void a05b_mergeTheme_twoDirectivesDiffer_rejectedWithOwnMessage() throws TemplateModelException {
69  		var cap = new PageCapture();
70  		cap.mergeTheme("light-red");
71  		var ex = assertThrows(TemplateModelException.class, () -> cap.mergeTheme("open"));
72  		assertEquals("<@theme name='open'> conflicts with an earlier <@theme name='light-red'> on this page; use one.", ex.getMessage());
73  	}
74  
75  	@Test void a06_mergeTheme_sameValueTwiceIsNoOp() throws TemplateModelException {
76  		var cap = new PageCapture();
77  		cap.mergeTheme("light-red");
78  		cap.mergeTheme("light-red");
79  		assertDoesNotThrow(() -> cap.mergeTheme("light-red"));
80  	}
81  
82  	@Test void a07_mergeBodyAttrs_unionsDisjointNames() throws TemplateModelException {
83  		var cap = new PageCapture();
84  		cap.mergeBodyAttrs("data-foo=\"1\"");
85  		cap.mergeBodyAttrs("data-bar=\"2\"");
86  		assertEquals("data-foo=\"1\" data-bar=\"2\"", cap.bodyAttrs);
87  	}
88  
89  	@Test void a07b_mergeBodyAttrs_sameNameRejected() throws TemplateModelException {
90  		var cap = new PageCapture();
91  		cap.mergeBodyAttrs("data-foo=\"1\"");
92  		var ex = assertThrows(TemplateModelException.class, () -> cap.mergeBodyAttrs("data-foo=\"2\""));
93  		assertEquals("PageSpec body attribute 'data-foo' is also set by <@body>; set it in one place.", ex.getMessage());
94  	}
95  
96  	@Test void a08_mergeToolkit_unionPreservesOrderAndDedupes() {
97  		var cap = new PageCapture();
98  		cap.toolkit(List.of("views"));
99  		cap.mergeToolkit(List.of("views", "cmd-k"));
100 		assertEquals(List.of("views", "cmd-k"), cap.toolkits());
101 	}
102 
103 	@Test void a09_mergeCss_appendsOnlyNewEntries() {
104 		var cap = new PageCapture();
105 		cap.css(List.of("/css/a.css"));
106 		cap.mergeCss(List.of("/css/a.css", "/css/b.css"));
107 		assertEquals(List.of("/css/a.css", "/css/b.css"), cap.cssHrefs());
108 	}
109 
110 	@Test void a10_mergeInit_appendsOnlyNewEntries() {
111 		var cap = new PageCapture();
112 		cap.init(List.of("/js/a.js"));
113 		cap.mergeInit(List.of("/js/b.js"));
114 		assertEquals(List.of("/js/a.js", "/js/b.js"), cap.initScripts());
115 	}
116 
117 	@Test void a11_mergeTheme_blankIsNoOp() throws TemplateModelException {
118 		var cap = new PageCapture();
119 		cap.mergeTheme("light-red");
120 		cap.mergeTheme("");
121 		cap.mergeTheme(null);
122 		assertDoesNotThrow(() -> cap.mergeTheme("light-red"));
123 	}
124 
125 	@Test void a12_mergeBodyAttrs_blankIsNoOp() throws TemplateModelException {
126 		var cap = new PageCapture();
127 		cap.mergeBodyAttrs("");
128 		cap.mergeBodyAttrs("data-foo=\"1\"");
129 		cap.mergeBodyAttrs(" ");
130 		assertEquals("data-foo=\"1\"", cap.bodyAttrs);
131 	}
132 
133 	@Test void a13_mergeBodyAttrs_sameNameSameValue_stillRejected() throws TemplateModelException {
134 		var cap = new PageCapture();
135 		cap.mergeBodyAttrs("data-foo=\"1\"");
136 		assertThrows(TemplateModelException.class, () -> cap.mergeBodyAttrs("data-foo=\"1\""));
137 	}
138 
139 	@Test void a14_mergeToolkit_sameListTwiceIsIdempotent() {
140 		var cap = new PageCapture();
141 		cap.mergeToolkit(List.of("views", "cmd-k"));
142 		cap.mergeToolkit(List.of("views", "cmd-k"));
143 		assertEquals(List.of("views", "cmd-k"), cap.toolkits());
144 	}
145 
146 	@Test void a15_mergeCss_intoEmptyList() {
147 		var cap = new PageCapture();
148 		cap.mergeCss(List.of("/css/a.css"));
149 		assertEquals(List.of("/css/a.css"), cap.cssHrefs());
150 	}
151 
152 	@Test void a16_mergeInit_intoEmptyList() {
153 		var cap = new PageCapture();
154 		cap.mergeInit(List.of("/js/a.js", "/js/a.js"));
155 		assertEquals(List.of("/js/a.js"), cap.initScripts());
156 	}
157 }
