32  /**
33   * The {@code <@card type="terminal">} card: a JSON5 {@code {contractVersion, terminal}} body, validated through
34   * {@code TerminalDef} at render time and emitted as a {@code cards[]} entry with a {@code terminal} object, plus the
35   * {@code xterm} and {@code terminal} toolkit packs (the terminal pack pulls {@code views} and {@code xterm}).
36   */
37  class CardDirective_Terminal_Test extends TestBase {
38  
39  	/** The contract entry for card {@code id}. */
40  	private static JsonMap card(String body, String id) {
41  		for (var o : assertPage(body).isValid().contract().getList("cards"))
42  			if (id.equals(((JsonMap)o).getString("id")))
43  				return (JsonMap)o;
44  		throw new AssertionError("no card '" + id + "' in " + body);
45  	}

47  	@Test void a01_fullCard_refreshClamped() {
48  		var body = render("tm-card");
49  		assertPage(body).isValid().hasCard("build", "terminal");
50  		assertEquals(
51  			Json5.to("{id:'build',type:'terminal',title:'Build',publishes:[{topic:'term.input',retain:false}],"
52  				+ "subscribes:[{topic:'term.input-status',as:'input-status'}],terminal:{bytesUrl:'/runs/juneau-terminal/r-7/bytes',"
53  				+ "eventsUrl:'/runs/juneau-run-view/r-7/events',refreshMs:1000,title:'mvn test',tailBytes:1048576,input:true,inputId:'r-7'}}", Map.class),
54  			Json5.to(Json.of(card(body, "build")), Map.class));
55  		assertFalse(body.contains("<template data-card=\"build\">"), () -> body);
56  	}
57  
58  	@Test void a02_minimalCard() {
59  		assertEquals(
60  			Json5.to("{id:'log',type:'terminal',terminal:{bytesUrl:'/t/juneau-terminal/r1/bytes'}}", Map.class),
61  			Json5.to(Json.of(card(render("tm-card-minimal"), "log")), Map.class));
62  	}
63  
64  	@Test void a03_pageWithNoToolkitLoadsViewsAndXtermOnceBeforeTheRuntime() {
65  		var body = render("tm-card-minimal");
66  		var xterm = body.indexOf("/lib/xterm.js?v=");
67  		var views = body.indexOf("/juneau-views.js?v=");
68  		var runtime = body.indexOf("/juneau-terminal.js?v=");
69  		assertTrue(xterm >= 0 && xterm < runtime, () -> body);
70  		assertTrue(views >= 0 && views < runtime, () -> body);
71  		for (var s : List.of("/lib/xterm.js?v=", "/css/xterm.css?v=", "/juneau-terminal.js?v=", "/juneau-terminal.css?v="))
72  			assertEquals(body.indexOf(s), body.lastIndexOf(s), () -> s + " emitted once: " + body);
73  		assertTrue(body.indexOf("/css/xterm.css?v=") >= 0, () -> body);
74  	}

76  	/** Renders exactly the FreeMarker snippet published in the TerminalRegion docs page (topic 10.68.12). */
77  	@Test void a05_docsSnippetWiresInput() {
78  		var body = render("tm-input-doc");
79  		assertPage(body).isValid().hasCard("term", "terminal");
80  		assertEquals(
81  			Json5.to("{id:'term',type:'terminal',title:'Release',publishes:[{topic:'term.input',retain:false}],"
82  				+ "subscribes:[{topic:'term.input-status',as:'input-status'}],"
83  				+ "terminal:{bytesUrl:'/t/juneau-terminal/t1/bytes',input:true,inputId:'t1'}}", Map.class),
84  			Json5.to(Json.of(card(body, "term")), Map.class));
85  	}
