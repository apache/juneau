32  /**
33   * Validates the 10.0 bean-store precedence model on a {@link RestContext}.
34   *
35   * <p>
36   * Resolution order, top-to-bottom:
37   * <ol>
38   * 	<li>{@code @Bean} factory methods on the resource.  For non-framework types these are
39   * 		registered as local entries directly; for framework types (e.g. {@link CallLogger},
40   * 		{@link ThrownStore}) the per-bean memoizer captures the {@code @Bean} value, and
41   * 		{@code RestContext} promotes the memoizer-backed supplier into a local entry so that
42   * 		{@code @Bean} uniformly wins.
43   * 	<li>User-supplied bean store from
44   * 		{@code @Bean WritableBeanStore createBeanStore(...)}, including its
45   * 		{@link org.apache.juneau.rest.server.server.springboot.SpringBeanStore}-style fallback to a backing
46   * 		{@code ApplicationContext}.  Consulted only when no {@code @Bean} factory method
47   * 		exists for the type.
48   * 	<li>Memoizer-backed framework defaults (e.g. {@link BasicCallLogger}).  Fire only when neither
49   * 		a per-resource {@code @Bean} method nor a user-supplied bean-store binding exists.
50   * </ol>
51   *
52   * <p>
53   * Net effect: <b>{@code @Bean} factory methods on the resource take precedence over
54   * Spring/user-supplied bindings, which in turn take precedence over framework defaults.</b>
55   * Spring/user-supplied bindings act as drop-in overrides for any type the resource doesn't
56   * customize via {@code @Bean}, with the framework filling in defaults for anything else.
57   */
58  class RestContext_Precedence_Test extends TestBase {

60  	//-----------------------------------------------------------------------------------------------------------------
61  	// Marker beans
62  	//-----------------------------------------------------------------------------------------------------------------
63  
64  	private static final CallLogger SPRING_LOGGER = BasicCallLogger.create(BasicBeanStore.INSTANCE).build();
65  	private static final CallLogger BEAN_LOGGER = BasicCallLogger.create(BasicBeanStore.INSTANCE).build();
66  
67  	//-----------------------------------------------------------------------------------------------------------------
68  	// Spring-substitute bean store
69  	//
70  	// Simulates SpringBeanStore: holds bindings in an internal "Spring app context" map that's
71  	// only consulted from getBean(...) AFTER super.getBean(...) returns empty.  This places the
72  	// "Spring" layer at the BOTTOM of the resolution chain, below local entries and below
73  	// memoizer-backed default suppliers \u2014 exactly where real Spring sits.
74  	//-----------------------------------------------------------------------------------------------------------------
75  
76  	@SuppressWarnings({
77  		"unchecked"  // Unchecked cast required for generic test utility.
78  	})
79  	public static class SpringLikeBeanStore extends BasicBeanStore {
80  
81  		private final Map<Class<?>,Object> springBindings = new HashMap<>();
82  
83  		public SpringLikeBeanStore(BeanStore parent) {
84  			super(parent);
85  		}
86  
87  		public <T> SpringLikeBeanStore bindSpring(Class<T> type, T instance) {
88  			springBindings.put(type, instance);
89  			return this;
90  		}
91  
92  		@Override
93  		public <T> Optional<T> getBean(Class<T> beanType) {
94  			return getBean(beanType, null);
95  		}
96  
97  		@Override
98  		public <T> Optional<T> getBean(Class<T> beanType, String name) {
99  			var o = super.getBean(beanType, name);
100 			if (o.isPresent())
101 				return o;
102 			return springLookup(beanType, name);
103 		}
104 
105 		@Override
106 		public <T> Optional<Supplier<T>> getBeanSupplier(Class<T> beanType) {
107 			return getBeanSupplier(beanType, null);
108 		}
109 
110 		@Override
111 		public <T> Optional<Supplier<T>> getBeanSupplier(Class<T> beanType, String name) {
112 			var o = super.getBeanSupplier(beanType, name);
113 			if (o.isPresent())
114 				return o;
115 			var bound = springLookup(beanType, name).orElse(null);
116 			return bound == null ? oe() : o(() -> bound);
117 		}
118 
119 		private <T> Optional<T> springLookup(Class<T> beanType, String name) {
120 			// "Spring" bindings are unnamed only — anything else falls through.
121 			if (name != null && !name.isEmpty())
122 				return oe();
123 			return o((T) springBindings.get(beanType));
124 		}
125 	}
126 
127 	private static SpringLikeBeanStore springLikeBeanStore() {
128 		return new SpringLikeBeanStore(null).bindSpring(CallLogger.class, SPRING_LOGGER);
129 	}

132 	// 1. @Bean beats the memoizer-backed framework default.
133 	//-----------------------------------------------------------------------------------------------------------------
134 
135 	@Rest
136 	public static class A_BeanBeatsDefault {
137 		@Bean static CallLogger callLoggerCapture;
138 		@Bean public CallLogger callLogger() { return BEAN_LOGGER; }
139 	}
140 
141 	@Test
142 	void a01_restInject_beatsDefault() {
143 		MockRestClient.buildLax(A_BeanBeatsDefault.class);
144 		assertSame(BEAN_LOGGER, A_BeanBeatsDefault.callLoggerCapture);
145 	}

153 		@Bean static CallLogger callLoggerCapture;
154 		@Bean public WritableBeanStore createBeanStore() { return springLikeBeanStore(); }
155 		@Bean public CallLogger callLogger() { return BEAN_LOGGER; }

158 	@Test
159 	void b01_restInject_beatsSpring() {
160 		MockRestClient.buildLax(B_BeanBeatsSpring.class);
161 		assertSame(BEAN_LOGGER, B_BeanBeatsSpring.callLoggerCapture, "@Bean method should win over Spring fallback");
162 	}

164 	//-----------------------------------------------------------------------------------------------------------------
165 	// 3. Spring beats the framework default for framework bean types (when no @Bean method exists).
166 	//
167 	// Without an @Bean CallLogger method, the framework's memoizer-backed default supplier sits
168 	// at level 4 of resolve(), below the user-supplied bean store (parent at level 3).  So Spring
169 	// overrides the framework default.  This is intentional: if the user wired a CallLogger into Spring,
170 	// they meant it to be used in preference to the auto-configured BasicCallLogger.
171 	//-----------------------------------------------------------------------------------------------------------------
172 
173 	@Rest
174 	public static class C_SpringBeatsDefault {
175 		@Bean static CallLogger callLoggerCapture;
176 		@Bean public WritableBeanStore createBeanStore() { return springLikeBeanStore(); }
177 	}
178 
179 	@Test
180 	void c01_spring_beatsDefault_forFrameworkBean() {
181 		MockRestClient.buildLax(C_SpringBeatsDefault.class);
182 		assertSame(SPRING_LOGGER, C_SpringBeatsDefault.callLoggerCapture, "User-supplied bean store binding should win over framework default when no @Bean is declared for the type");
183 	}

185 	//-----------------------------------------------------------------------------------------------------------------
186 	// 4. Spring fallback fills in for non-framework bean types when nothing else has a binding.
187 	//-----------------------------------------------------------------------------------------------------------------
188 
189 	public static class CustomBean {
190 		final String marker;
191 		CustomBean(String marker) { this.marker = marker; }
192 	}
193 
194 	private static final CustomBean SPRING_CUSTOM = new CustomBean("from-spring");
195 
196 	private static SpringLikeBeanStore springLikeBeanStoreWithCustomBean() {
197 		return new SpringLikeBeanStore(null)
198 			.bindSpring(CallLogger.class, SPRING_LOGGER)
199 			.bindSpring(CustomBean.class, SPRING_CUSTOM);
200 	}
201 
202 	@Rest
203 	public static class D_SpringFallbackForUserBean {
204 		@Bean static CustomBean customBeanCapture;
205 		@Bean public WritableBeanStore createBeanStore() { return springLikeBeanStoreWithCustomBean(); }
206 	}
207 
208 	@Test
209 	void d01_spring_fillsInForUserBean() {
210 		MockRestClient.buildLax(D_SpringFallbackForUserBean.class);
211 		assertSame(SPRING_CUSTOM, D_SpringFallbackForUserBean.customBeanCapture, "Spring fallback should provide CustomBean since framework has no default for it and there is no @Bean method");
212 	}
