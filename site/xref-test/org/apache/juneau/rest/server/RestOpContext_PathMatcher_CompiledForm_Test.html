17  package org.apache.juneau.rest.server;
18  
19  import org.apache.juneau.*;
20  import org.apache.juneau.rest.mock.classic.*;
21  import org.junit.jupiter.api.*;
22  import org.junit.jupiter.api.parallel.*;
23  
24  /**
25   * Phase G regression: verifies that the {@link RestOpContext#pathMatchers} Memoizer
26   * exercises the compiled-template seam — every op-path string goes through
27   * {@code vr.compile(p).resolve(session)} (not the legacy one-shot
28   * {@code vr.resolve(p)} call). Production behavior must remain unchanged.
29   *
30   * <p>
31   * The point of the test isn't to assert on internal data structures — those vary
32   * across Memoizer implementations — but to lock in the user-visible contract:
33   * SVL-bearing op paths still resolve correctly under the compiled-form path, with
34   * no extra request-time cost (resolution still happens once at context-build time).
35   */
36  @ResourceLock(Resources.SYSTEM_PROPERTIES)
37  class RestOpContext_PathMatcher_CompiledForm_Test extends TestBase {
38  
39  	@Rest
40  	public static class A01_LiteralPath {
41  		@RestGet(path="/literal/*")
42  		public String hello() { return "ok"; }
43  	}
44  
45  	@Test void a01_literalPathStillWorks() throws Exception {
46  		var c = MockRestClient.buildLax(A01_LiteralPath.class);
47  		c.get("/literal/x").run().assertStatus(200).assertContent().asString().isContains("ok");
48  	}
49  
50  	@Rest
51  	public static class A02_SvlPathDefaultBranch {
52  		@RestGet(path="/${RestOpContext_PathMatcher_CompiledForm_Test.a02:a02-default}/*")
53  		public String hello() { return "a02-hit"; }
54  	}
55  
56  	@Test void a02_svlPath_defaultBranch_compiledFormResolves() throws Exception {
57  		var c = MockRestClient.buildLax(A02_SvlPathDefaultBranch.class);
58  		c.get("/a02-default/anything").run().assertStatus(200).assertContent().asString().isContains("a02-hit");
59  	}
60  
61  	@Rest
62  	public static class A03_SvlPathOverride {
63  		@RestGet(path="/${RestOpContext_PathMatcher_CompiledForm_Test.a03:a03-default}/*")
64  		public String hello() { return "a03-hit"; }
65  	}
66  
67  	@Test void a03_svlPath_overrideBranch_compiledFormResolves() throws Exception {
68  		System.setProperty("RestOpContext_PathMatcher_CompiledForm_Test.a03", "a03-override");
69  		try {
70  			var c = MockRestClient.buildLax(A03_SvlPathOverride.class);
71  			c.get("/a03-override/anything").run().assertStatus(200).assertContent().asString().isContains("a03-hit");
72  			c.get("/a03-default/anything").run().assertStatus(404);
73  		} finally {
74  			System.clearProperty("RestOpContext_PathMatcher_CompiledForm_Test.a03");
75  		}
76  	}
77  
78  	@Rest
79  	public static class A04_MultipleOpsSamePath {
80  		@RestGet(path="/${RestOpContext_PathMatcher_CompiledForm_Test.a04:a04-default}/get")
81  		public String getOp() { return "get-hit"; }
82  
83  		@RestGet(path="/${RestOpContext_PathMatcher_CompiledForm_Test.a04:a04-default}/get2")
84  		public String getOp2() { return "get2-hit"; }
85  	}
86  
87  	@Test void a04_repeatedExpressionAcrossOps_compiledFormReused() throws Exception {
88  		var c = MockRestClient.buildLax(A04_MultipleOpsSamePath.class);
89  		c.get("/a04-default/get").run().assertStatus(200).assertContent().asString().isContains("get-hit");
90  		c.get("/a04-default/get2").run().assertStatus(200).assertContent().asString().isContains("get2-hit");
91  	}
92  }
