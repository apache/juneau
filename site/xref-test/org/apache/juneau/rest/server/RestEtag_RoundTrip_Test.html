17  package org.apache.juneau.rest.server;
18  
19  import java.time.*;
20  import java.util.concurrent.*;
21  
22  import org.apache.juneau.*;
23  import org.apache.juneau.http.*;
24  import org.apache.juneau.marshall.json.*;
25  import org.apache.juneau.rest.mock.classic.*;
26  import org.junit.jupiter.api.*;
27  
28  /**
29   * End-to-end "round trip" tests for the ETag / conditional-GET helpers in a realistic handler:
30   *
31   * <ul>
32   *   <li>First request gets the resource with {@code 200 OK} plus {@code ETag} and {@code Last-Modified} headers.
33   *   <li>Second request re-sends the same {@code ETag} via {@code If-None-Match} and gets {@code 304 Not Modified}.
34   *   <li>An update bumps the version + timestamp; the next request gets {@code 200 OK} with the new tag/date.
35   *   <li>A {@code PUT} with a stale {@code If-Match} is rejected with {@code 412 Precondition Failed}.
36   * </ul>
37   */
38  class RestEtag_RoundTrip_Test extends TestBase {
39  
40  	/** Trivial in-memory order with a version + updated-instant for ETag/Last-Modified. */
41  	public static class Order {
42  		public String id, payload;
43  		public long version;
44  		public Instant updated;
45  		public Order() {}
46  		public Order(String id, String payload, long version, Instant updated) {
47  			this.id = id; this.payload = payload; this.version = version; this.updated = updated;
48  		}
49  	}
50  
51  	@Rest(serializers=JsonSerializer.class, parsers=JsonParser.class)
52  	public static class R {
53  		private final ConcurrentMap<String,Order> repo = new ConcurrentHashMap<>();
54  
55  		public R() {
56  			repo.put("1", new Order("1", "v1-payload", 1L, Instant.parse("2026-05-22T00:00:00Z")));
57  		}
58  
59  		@RestGet("/orders/{id}")
60  		public Order get(@Path("id") String id, RestRequest req, RestResponse res) {
61  			var order = repo.get(id);
62  			res.eTag("\"" + order.version + "\"").lastModified(order.updated);
63  			req.checkPreconditions(res).ifPresent(e -> { throw e; });
64  			return order;
65  		}
66  
67  		@RestPut("/orders/{id}")
68  		public Order put(@Path("id") String id, @Content Order in, RestRequest req, RestResponse res) {
69  			var current = repo.get(id);
70  			res.eTag("\"" + current.version + "\"").lastModified(current.updated);
71  			req.checkPreconditions(res).ifPresent(e -> { throw e; });
72  			var next = new Order(id, in.payload, current.version + 1L, Instant.parse("2026-05-23T00:00:00Z"));
73  			repo.put(id, next);
74  			res.eTag("\"" + next.version + "\"").lastModified(next.updated);
75  			return next;
76  		}
77  	}
78  
79  	@Test void a01_firstGetReturnsEtagAndLastModified() throws Exception {
80  		var c = MockRestClient.buildLax(R.class);
81  		c.get("/orders/1").run().assertStatus(200)
82  			.assertHeader("ETag").is("\"1\"")
83  			.assertHeader("Last-Modified").is("Fri, 22 May 2026 00:00:00 GMT");
84  	}
85  
86  	@Test void a02_secondGetWithIfNoneMatchReturns304() throws Exception {
87  		var c = MockRestClient.buildLax(R.class);
88  		c.get("/orders/1").header("If-None-Match", "\"1\"").run().assertStatus(304);
89  	}
90  
91  	@Test void a03_getWithIfModifiedSinceEqualReturns304() throws Exception {
92  		var c = MockRestClient.buildLax(R.class);
93  		c.get("/orders/1").header("If-Modified-Since", "Fri, 22 May 2026 00:00:00 GMT").run().assertStatus(304);
94  	}
95  
96  	@Test void a04_putWithStaleIfMatchReturns412() throws Exception {
97  		var c = MockRestClient.create(R.class).ignoreErrors().json().build();
98  		// Stale version: client thinks the order is at version 0 (its real version is 1).
99  		c.put("/orders/1", new Order("1", "bad", 0L, Instant.parse("2026-01-01T00:00:00Z")))
100 			.header("If-Match", "\"0\"").run().assertStatus(412);
101 	}
102 
103 	@Test void a05_putWithCorrectIfMatchSucceedsAndBumpsEtag() throws Exception {
104 		var c = MockRestClient.create(R.class).ignoreErrors().json().build();
105 		c.put("/orders/1", new Order("1", "updated", 1L, Instant.parse("2026-01-01T00:00:00Z")))
106 			.header("If-Match", "\"1\"").run().assertStatus(200)
107 			.assertHeader("ETag").is("\"2\"")
108 			.assertHeader("Last-Modified").is("Sat, 23 May 2026 00:00:00 GMT");
109 		// After the bump, the original tag no longer matches — the next GET with the old tag returns 200.
110 		c.get("/orders/1").header("If-None-Match", "\"1\"").run().assertStatus(200);
111 		// And the new tag matches → 304.
112 		c.get("/orders/1").header("If-None-Match", "\"2\"").run().assertStatus(304);
113 	}
114 }
