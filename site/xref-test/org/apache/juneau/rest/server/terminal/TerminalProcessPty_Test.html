34  /**
35   * {@link TerminalProcess} over the real {@code juneau_run.py --pty}: the child sees a terminal of the given size, the
36   * log keeps every byte, and the events carry {@code rawOffset}, the exit code and {@code done}.
37   */
38  @DisabledOnOs(OS.WINDOWS)
39  class TerminalProcessPty_Test extends TestBase {
40  
41  	@TempDir Path dir;
42  
43  	/** {@code juneau-run/src/main/python/juneau_run.py}, from {@code basedir} or the working directory. */
44  	static Path script() {
45  		var basedir = System.getProperty("basedir");
46  		for (var p : List.of(Path.of(basedir == null ? "." : basedir, "../../juneau-run/src/main/python/juneau_run.py"), Path.of("juneau-run/src/main/python/juneau_run.py")))
47  			if (Files.isRegularFile(p))
48  				return p.toAbsolutePath().normalize();
49  		throw new AssertionError("cannot find juneau-run/src/main/python/juneau_run.py");
50  	}
51  
52  	@Test void a01_realRunnerUnderAPty() throws Exception {
53  		var path = System.getenv("PATH");
54  		assumeTrue(TerminalProcess.findOnPath("python3", path) != null, "python3 not on the PATH - skipped");
55  		var host = new Host(System.getProperty("os.name"), script().toString(), null, path);
56  		var cmd = List.of("sh", "-c", "printf '\\033[31mred\\033[0m\\n'; stty size; printf '10%%\\r100%%\\n'; exit 3");
57  		try (var p = TerminalProcess.start("generic", cmd, 100, 30, dir, host)) {
58  			assertEquals(3, p.waitFor(60, TimeUnit.SECONDS));
59  			var chunk = p.terminal().read(0, 1 << 20);
60  			var text = new String(chunk.bytes(), UTF_8);
61  			assertTrue(text.startsWith("\u001b[31mred\u001b[0m\r\n30 100\r\n10%\r100%\r\n"), text);
62  			assertTrue(chunk.done(), "done once the process exits");
63  			assertEquals(100, chunk.cols());
64  			assertEquals(30, chunk.rows());
65  			var events = p.events().page(null, 100).events().stream().map(RunEvent::toContractMap).toList();
66  			var kinds = events.stream().map(e -> e.get("ev")).toList();
67  			assertEquals("step", kinds.get(0), events::toString);
68  			assertEquals(0L, ((Number)events.get(0).get("rawOffset")).longValue(), events::toString);
69  			assertTrue(events.stream().anyMatch(e -> "end".equals(e.get("ev")) && Objects.equals(3L, ((Number)e.getOrDefault("exit", -1L)).longValue())), events::toString);
70  			var last = events.get(events.size() - 1);
71  			assertEquals(List.of("done", "fail"), List.of(last.get("ev"), last.get("status")), events::toString);
72  		}
73  	}
74  }
