29  /**
30   * The terminal region in a real Chromium: the real xterm.js WebJar and the real {@code juneau-terminal.js}, driven by
31   * {@code terminal-browser.cjs}.
32   *
33   * <p>
34   * The prober answers every request itself through Playwright's request routing on a made-up origin, so nothing
35   * listens on a port.  The endpoints' headers are covered by {@link TerminalMixin_Test}; the end-to-end path is
36   * {@link TerminalSmoke_BrowserTest}.
37   *
38   * <h5 class='section'>Off by default:</h5>
39   * <p>
40   * Disabled unless the {@value #GATE} system property is set, which only the module's opt-in {@code js-tests} Maven
41   * profile does.
42   *
43   * @since 10.0.0
44   */
45  @EnabledIfSystemProperty(named=Terminal_BrowserTest.GATE, matches="true",
46  	disabledReason="JS-execution harness is opt-in; run with -Pjs-tests")
47  class Terminal_BrowserTest extends TestBase {
48  
49  	/** System property the {@code js-tests} profile sets to enable this class. */
50  	static final String GATE = "juneau.jsTests";
51  
52  	private static Map<?,?> report;
53  
54  	@BeforeAll static void setUp() throws Exception {
55  		var dir = Path.of(requiredProperty("juneau.jsTests.dir"));
56  		var harness = Path.of(requiredProperty("juneau.jsTests.harnessDir"), "terminal-browser.cjs");
57  		var out = TerminalHarness.run(List.of(System.getProperty("juneau.jsTests.node", "node"), harness.toString()),
58  			TerminalHarness.BROWSER_ASSETS,
59  			Map.of("NODE_PATH", dir.resolve("node_modules").toString(), "PLAYWRIGHT_BROWSERS_PATH", requiredProperty("juneau.jsTests.browsers")),
60  			300);
61  		report = Json.to(out, Map.class);
62  		assertNull(report.get("harnessError"), () -> "prober failed: " + report.get("harnessError"));
63  	}
64  
65  	private static String requiredProperty(String name) {
66  		var v = System.getProperty(name);
67  		assertNotNull(v, () -> "-D" + name + " not set; the js-tests profile is responsible for providing it");
68  		return v;
69  	}
70  
71  	/** Asserts that case {@code name} passed with no page or console errors. */
72  	private static void ok(String name) {
73  		var cases = (Map<?,?>)report.get("cases");
74  		assertEquals(true, cases.get(name), () -> name + " -> " + cases.get(name));
75  		assertEquals(List.of(), ((Map<?,?>)report.get("pageErrors")).get(name), () -> name + " page errors");
76  		assertEquals(List.of(), ((Map<?,?>)report.get("consoleErrors")).get(name), () -> name + " console errors");
77  	}
