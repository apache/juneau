49  	/** Records every accepted frame (copied from {@code ServerBus_Test}, which is test-private). */
50  	static final class RecordingSink implements FrameSink {
51  		final List<String> frames = new ArrayList<>();
52  		int closeCode;
53  
54  		@Override public boolean offer(String frame) {
55  			if (closeCode != 0)
56  				return false;
57  			frames.add(frame);
58  			return true;
59  		}
60  
61  		@Override public void close(int code, String reason) {
62  			closeCode = code;
63  		}
64  
65  		List<JsonMap> decoded() { return frames.stream().map(RecordingSink::parse).toList(); }
66  		JsonMap last() { return parse(frames.get(frames.size() - 1)); }
67  		void reset() { frames.clear(); }
68  
69  		/** The payloads of the {@code term.input-status} frames received. */
70  		List<JsonMap> status() {
71  			return decoded().stream()
72  				.filter(f -> TerminalInput.STATUS_TOPIC.equals(f.get("topic")))
73  				.map(f -> (JsonMap) f.get("payload"))
74  				.toList();
75  		}
76  
77  		static JsonMap parse(String frame) {
78  			try {
79  				return BusFrames.decode(frame);
80  			} catch (BusRefusal e) {
81  				throw new AssertionError(frame, e);
82  			}
83  		}
84  	}

86  	/** A sink that records writes. */
87  	static final class FakeSink implements TerminalSink {
88  		final List<byte[]> writes = new ArrayList<>();
89  		boolean accepts = true, writeResult = true, throwOnWrite, closeOnWrite;
90  
91  		@Override public boolean acceptsInput() { return accepts; }
92  
93  		@Override public boolean write(byte[] bytes) throws IOException {
94  			if (throwOnWrite)
95  				throw new IOException("disk on fire");
96  			if (closeOnWrite)
97  				accepts = false;
98  			if (writeResult)
99  				writes.add(bytes.clone());
100 			return writeResult;
101 		}
102 
103 		List<String> text() { return writes.stream().map(b -> new String(b, UTF_8)).toList(); }
104 	}

166 	/** A request that knows only its principal (copied from {@code ServerBus_Test}). */
167 	private static HttpServletRequest req(String user, String...roles) {
168 		var roleSet = Set.of(roles);
169 		Principal principal = user == null ? null : () -> user;
170 		return (HttpServletRequest)Proxy.newProxyInstance(HttpServletRequest.class.getClassLoader(), new Class<?>[]{HttpServletRequest.class},
171 			(proxy, method, args) -> switch (method.getName()) {
172 				case "getUserPrincipal" -> principal;
173 				case "isUserInRole" -> roleSet.contains(args[0]);
174 				case "toString" -> "FakeRequest[" + user + "]";
175 				case "hashCode" -> System.identityHashCode(proxy);
176 				case "equals" -> proxy == args[0];
177 				default -> method.getReturnType() == boolean.class ? (Object)false : method.getReturnType() == int.class ? (Object)0 : null;
178 			});
179 	}

181 	private static BusSession open(ServerBus bus, String user) throws BusRefusal {
182 		return bus.openSession(req(user), JsonMap.of("v", 1, "bridge", "term", "transport", "websocket",
183 			"downstream", List.of(TerminalInput.STATUS_TOPIC), "upstream", List.of(TerminalInput.TOPIC)));
184 	}

186 	private static RecordingSink connect(ServerBus bus, BusSession s) {
187 		var sink = new RecordingSink();
188 		bus.connect(s, sink);
189 		sink.reset();
190 		return sink;

194 	private static String up(String topic, String payloadJson) {
195 		return "{\"v\":1,\"type\":\"pub\",\"topic\":\"" + topic + "\",\"payload\":" + payloadJson + "}";

198 	private static String frame(String id, String data) {
199 		return up(TerminalInput.TOPIC, Json.of(JsonMap.of("id", id, "data", data)));

352 	@Test void a19_principalEscaped() throws Exception {
353 		var typist = "a b\nc=d\\e%\t";
354 		var watcherOnly = "e=v x";
355 		var unicode = "u\u0085v\u2028w\u2029";
356 		watchers.addAll(Set.of(typist, unicode));
357 		typists.put(typist, fake);
358 		typists.put(unicode, fake);
359 		var s3 = open(bus, unicode);
360 		connect(bus, s3);
361 		var s1 = open(bus, typist);
362 		connect(bus, s1);
363 		var s2 = open(bus, watcherOnly);
364 		connect(bus, s2);
365 		var records = audit(() -> {
366 			send(s1, "t1", "x");
367 			send(s2, "t1", "x");
368 			send(s3, "t1", "x");
369 		});
370 		assertEquals(List.of("principal=a%20b%0Ac%3Dd%5Ce%25%09 terminal=t1 bytes=1 keys=",
371 			"principal=u%C2%85v%E2%80%A8w%E2%80%A9 terminal=t1 bytes=1 keys="), messages(records, Level.INFO),
372 			"C1 controls and U+2028/U+2029 are percent-encoded as their UTF-8 bytes");
373 		assertEquals(List.of("principal=e%3Dv%20x terminal=t1 state=denied"), messages(records, Level.FINE));
374 		for (var r : records) {
375 			assertEquals(4 - (r.getLevel() == Level.FINE ? 1 : 0), r.getMessage().split(" ").length, r::getMessage);
376 			assertEquals(r.getMessage().split(" ").length, r.getMessage().split("=", -1).length - 1, "no forged key=value");
377 		}
378 	}
