41  	@BeforeEach void setup() throws Exception {
42  		log = dir.resolve("run.log");
43  		size("{\"cols\":120,\"rows\":40}");  // A valid sidecar so only the tests about a missing one log a warning.
44  	}
45  
46  	private void append(String s) throws Exception {
47  		Files.write(log, s.getBytes(UTF_8), CREATE, APPEND);
48  	}
49  
50  	private void size(String json) throws Exception {
51  		Files.writeString(dir.resolve("run.log.size"), json);
52  	}
53  
54  	private static String text(TerminalChunk c) {
55  		return new String(c.bytes(), UTF_8);
56  	}
57  
58  	@Test void a01_readFromStartMiddleEndAndPast() throws Exception {
59  		size("{\"cols\":80,\"rows\":24}");
60  		append("\u001b[31mred\u001b[0m\r\n");
61  		var s = FileTerminalSource.create(log).build();
62  		var c = s.read(0, 1000);
63  		assertEquals("\u001b[31mred\u001b[0m\r\n", text(c));
64  		assertEquals(14, c.next());
65  		assertEquals(14, c.end());
66  		assertEquals("red\u001b[0m\r\n", text(s.read(5, 1000)));
67  		assertEquals("\u001b[3", text(s.read(0, 3)));
68  		var atEnd = s.read(14, 1000);
69  		assertEquals(0, atEnd.bytes().length);
70  		assertEquals(14, atEnd.next());
71  		var past = s.read(20, 1000);
72  		assertEquals(0, past.bytes().length);
73  		assertEquals(14, past.end());
74  		assertEquals(80, c.cols());
75  		assertEquals(24, c.rows());
76  		assertFalse(c.truncated());
77  		assertFalse(c.gone());
78  	}

80  	@Test void a02_maxIsClamped() throws Exception {
81  		Files.write(log, new byte[TerminalSource.MAX_READ_BYTES + 10]);
82  		var s = FileTerminalSource.create(log).build();
83  		assertEquals(TerminalSource.MAX_READ_BYTES, s.read(0, Integer.MAX_VALUE).bytes().length);
84  		assertEquals(0, s.read(0, -5).bytes().length);
85  		assertEquals(10, s.read(TerminalSource.MAX_READ_BYTES, Integer.MAX_VALUE).bytes().length);
86  	}

88  	@Test void a03_doneOnlyAtTheEnd() throws Exception {
89  		append("abcdef");
90  		var finished = new AtomicBoolean();
91  		var s = FileTerminalSource.create(log).terminal(finished::get).build();
92  		assertFalse(s.read(6, 10).done());
93  		finished.set(true);
94  		assertFalse(s.read(0, 3).done());
95  		assertTrue(s.read(3, 10).done());
96  		assertTrue(s.read(6, 10).done());
97  	}

99  	@Test void a04_missingOrMalformedSidecarDefaultsAndWarnsOnce() throws Exception {
100 		append("x");
101 		Files.delete(dir.resolve("run.log.size"));
102 		var s = FileTerminalSource.create(log).build();
103 		var records = LogRecordCapture.quietly(FileTerminalSource.class, () -> {
104 			var c = s.read(0, 10);
105 			assertEquals(120, c.cols());
106 			assertEquals(40, c.rows());
107 			size("{\"cols\":0,\"rows\":24}");
108 			assertEquals(120, s.read(0, 10).cols());
109 		});
110 		assertEquals(1, records.size());
111 		assertEquals(Level.WARNING, records.get(0).getLevel());
112 		assertTrue(records.get(0).getMessage().contains("120x40"), records.get(0).getMessage());
113 		size("{\"cols\":100,\"rows\":30}");
114 		assertEquals(100, s.read(0, 10).cols());
115 		assertEquals(30, s.read(0, 10).rows());
116 	}

118 	@Test void a05_parseSize() throws Exception {
119 		var p = dir.resolve("s");
120 		for (var bad : List.of("", "[]", "{\"cols\":80}", "{\"cols\":80.5,\"rows\":24}", "{\"cols\":10000,\"rows\":24}", "{\"cols\":\"80\",\"rows\":24}")) {
121 			Files.writeString(p, bad);
122 			assertNull(FileTerminalSource.parseSize(p), bad);
123 		}
124 		Files.writeString(p, "{\"cols\":9999,\"rows\":1}");
125 		assertArrayEquals(new int[] { 9999, 1 }, FileTerminalSource.parseSize(p));
126 		assertNull(FileTerminalSource.parseSize(dir.resolve("missing")));
127 	}

129 	@Test void a06_deletedLogIsGone() throws Exception {
130 		append("abc");
131 		var s = FileTerminalSource.create(log).build();
132 		assertFalse(s.read(0, 10).gone());
133 		Files.delete(log);
134 		var c = s.read(3, 10);
135 		assertTrue(c.gone());
136 		assertTrue(c.done());
137 		assertEquals(3, c.next());
138 		assertEquals(3, c.end());
139 		assertEquals(1, s.read(1, 10).next());
140 	}

142 	@Test void a10_fileThatShrinksBelowFromIsPastTheEnd() throws Exception {
143 		append("abcdef");
144 		var s = FileTerminalSource.create(log).build();
145 		assertEquals("abcdef", text(s.read(0, 10)));
146 		Files.write(log, "abc".getBytes(UTF_8));
147 		var c = s.read(5, 10);
148 		assertEquals(0, c.bytes().length);
149 		assertEquals(3, c.end());
150 		assertEquals(3, c.next());
151 		assertFalse(c.gone());
152 		assertEquals("c", text(s.read(2, 10)));

171 	@Test void a09_copyToWritesEverything() throws Exception {
172 		var big = new byte[TerminalSource.MAX_READ_BYTES * 2 + 7];
173 		Arrays.fill(big, (byte)'z');
174 		Files.write(log, big);
175 		var out = new ByteArrayOutputStream();
