28  /**
29   * Validates SVL substitution applied per-element to {@code @Rest(paths=...)} annotation literals.
30   *
31   * <p>
32   * Each {@code @Rest(paths=...)} element runs through SVL substitution and the post-SVL value is split on
33   * {@code ,} (trim each piece, drop empties).  This covers the four variable backends that matter for
34   * runtime path substitution:
35   * <ul>
36   * 	<li>{@code $C{key}} &mdash; Juneau {@link Config} (requires a Config bean registered in the bean store).
37   * 	<li>{@code $E{NAME,default}} &mdash; environment variable with literal default.
38   * 	<li>{@code $S{prop,default}} &mdash; system property with literal default.
39   * 	<li>An unresolved SVL reference falls back to the literal element (no exception during construction).
40   * </ul>
41   *
42   * @since 10.0.0
43   */
44  @ResourceLock(Resources.SYSTEM_PROPERTIES)
45  class RestPathsRuntimeOverride_SVL_Test extends TestBase {
46  
47  	private static Config inMemoryConfig(String name, String...kvLines) {
48  		var store = MemoryStore.create().build();
49  		var sb = new StringBuilder();
50  		for (var line : kvLines)
51  			sb.append(line).append('\n');
52  		store.write(name, null, sb.toString());
53  		return Config.create().store(store).name(name).build();
54  	}
55  
56  	private static TestBeanStore overlayWith(Config c) {
57  		return new TestBeanStore().override(Config.class, c);
58  	}

64  	@Rest(paths={"$C{health.paths}"})
65  	public static class A_ConfigKey {}
66  
67  	@Test
68  	void a01_configKey_resolvesAndCommaSplits() throws Exception {
69  		// Single element wraps a Config lookup; the resolved value contains a comma so the post-SVL split
70  		// produces two mount paths.
71  		var cfg = inMemoryConfig("a01.cfg", "health.paths = /healthz, /readyz");
72  		var overlay = overlayWith(cfg);
73  
74  		var args = new RestContext.Args(A_ConfigKey.class, null, null, A_ConfigKey::new, "", null, overlay, null, RestContext.ContextKind.ROOT);
75  		var ctx = new RestContext(args).postInit().postInitChildFirst();
76  
77  		assertArrayEquals(new String[]{"/healthz", "/readyz"}, ctx.getPaths(),
78  			"$C{health.paths} should resolve via Config and the multi-value result should comma-split");
79  	}

100 	@Test
101 	void c01_configMissWithDefault_usesDefault() throws Exception {
102 		// Config has no `paths.missing` key — the $C{key,default} form falls back to the literal default
103 		// after the comma.  This is the supported way to provide an in-annotation fallback.
104 		var cfg = inMemoryConfig("c01.cfg", "some.other.key = /not-this");
105 		var overlay = overlayWith(cfg);
106 
107 		var args = new RestContext.Args(C_ConfigMissWithDefault.class, null, null, C_ConfigMissWithDefault::new, "", null, overlay, null, RestContext.ContextKind.ROOT);
108 		var ctx = new RestContext(args).postInit().postInitChildFirst();
109 
110 		assertArrayEquals(new String[]{"/from-default"}, ctx.getPaths(),
111 			"$C{missing,/default} should fall back to the literal default after the comma");
112 	}

114 	@Rest(paths={"$C{paths.missing}"})
115 	public static class C2_ConfigMissNoDefault {}
116 
117 	@Test
118 	void c02_configMissNoDefault_dropsToEmpty() throws Exception {
119 		// Config has no `paths.missing` key and no in-annotation default — SVL resolves to "" and the
120 		// comma-split drops the empty piece.  Final result: zero mounts.  This matches how every other
121 		// SVL-bearing @Rest member treats unresolved variables (empty substitution).
122 		var cfg = inMemoryConfig("c02.cfg", "some.other.key = /not-this");
123 		var overlay = overlayWith(cfg);
124 
125 		var args = new RestContext.Args(C2_ConfigMissNoDefault.class, null, null, C2_ConfigMissNoDefault::new, "", null, overlay, null, RestContext.ContextKind.ROOT);
126 		var ctx = new RestContext(args).postInit().postInitChildFirst();
127 
128 		assertNotNull(ctx.getPaths());
129 		assertEquals(0, ctx.getPaths().length,
130 			"Unresolved $C{key} with no default should produce zero mounts (the empty SVL substitution gets dropped by the comma-split)");
131 	}
