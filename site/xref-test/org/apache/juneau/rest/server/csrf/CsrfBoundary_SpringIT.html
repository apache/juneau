53  /**
54   * {@link RequestBoundaryFilter} applying a {@link CsrfBoundary} under Spring Boot's embedded Tomcat, over a real
55   * socket: how the boundary's filter order interacts with each {@code server.forward-headers-strategy}, and the
56   * container's session cookie.
57   *
58   * <p>
59   * Every case that differs in properties or filter order runs in its own application context.  The proxied requests
60   * model a proxy that terminated TLS on the same host and port: {@code X-Forwarded-Proto: https} and
61   * {@code X-Forwarded-Port} naming the port the client connected to, so after a correct rewrite the server's own
62   * origin is {@code https://127.0.0.1:<port>}.
63   *
64   * <p>
65   * Binds a socket, so it is off by default.  Run it with {@code -Djuneau.busIt=true}, outside the sandbox.
66   *
67   * @since 10.0.0
68   */

72  @SuppressWarnings({
73  	"resource" // Raw-socket stream chains are closed with the socket and the JDK client is not a resource; the Spring context owns the server lifecycle.

166 	/** One context's view of the application: its port, its write counter and its filter registrations. */
167 	abstract static class Context {
168 
169 		@LocalServerPort int port;
170 		@Autowired Writes writes;
171 		@Autowired ApplicationContext ctx;
172 
173 		String base() { return "http://127.0.0.1:" + port; }
174 
175 		HttpResponse<String> send(String method, String path, String...headers) throws Exception {
176 			var b = HttpRequest.newBuilder(URI.create(base() + path)).timeout(Duration.ofSeconds(10));
177 			for (var i = 0; i < headers.length; i += 2)
178 				b.header(headers[i], headers[i + 1]);
179 			b.method(method, "GET".equals(method) ? HttpRequest.BodyPublishers.noBody() : HttpRequest.BodyPublishers.ofString("{}"));
180 			return HTTP.send(b.build(), BodyHandlers.ofString());
181 		}
182 
183 		/** The forwarded headers of a proxy that terminated TLS in front of this server's own host and port. */
184 		String[] tls() {
185 			return new String[] { "X-Forwarded-Proto", "https", "X-Forwarded-Port", String.valueOf(port) };
186 		}
187 
188 		String token(String...headers) throws Exception {
189 			var res = send("GET", "/token", headers);
190 			assertEquals(200, res.statusCode(), res::body);
191 			assertTrue(res.body().matches("[0-9a-f]{128}"), res::body);
192 			return res.body();
193 		}
194 
195 		/** A same-origin JSON write from the page the proxy served over https. */
196 		HttpResponse<String> tlsWrite(String token) throws Exception {
197 			var h = new ArrayList<>(List.of(tls()));
198 			h.addAll(List.of("Origin", "https://127.0.0.1:" + port, "Content-Type", JSON, CSRF, token));
199 			return send("POST", "/write", h.toArray(String[]::new));
200 		}
201 
202 		/** How many {@link ForwardedHeaderFilter} registrations this context holds. */
203 		long forwardedFilters() {
204 			return ctx.getBeansOfType(FilterRegistrationBean.class).values().stream()
205 				.filter(r -> r.getFilter() instanceof ForwardedHeaderFilter).count();
206 		}
207 
208 		void assertTlsWritePasses() throws Exception {
209 			var res = tlsWrite(token(tls()));
210 			assertEquals(200, res.statusCode(), res::body);
211 			assertEquals(1, writes.count.get());
212 		}
213 	}
214 
215 	static String rejection(HttpResponse<String> res) {
216 		return res.headers().firstValue(RequestBoundaryFilter.REJECTION_HEADER).orElse(null);
217 	}
218 
219 	static String sessionCookie(HttpResponse<String> res) {
220 		for (var c : res.headers().allValues("Set-Cookie"))
221 			if (c.startsWith("JSESSIONID="))
222 				return c.substring(0, c.indexOf(';') < 0 ? c.length() : c.indexOf(';'));
223 		return null;
224 	}
225 
226 	/** Runs {@code r} with the boundary's warnings captured. */
227 	static List<LogRecord> captureWarnings(ThrowingRunnable r) throws Exception {
228 		var records = new ArrayList<LogRecord>();
229 		var h = new Handler() {
230 			@Override public void publish(LogRecord x) { records.add(x); }
231 			@Override public void flush() { /* Nothing buffered. */ }
232 			@Override public void close() { /* Nothing held. */ }
233 		};
234 		h.setLevel(Level.WARNING);
235 		BOUNDARY_LOG.addHandler(h);
236 		try {
237 			r.run();
238 		} finally {
239 			BOUNDARY_LOG.removeHandler(h);
240 		}
241 		return records;
242 	}

245 		void run() throws Exception;
246 	}

400 	/**
401 	 * Sends a hand-encoded {@code GET /echo} over prior-knowledge h2c with the given {@code :authority} (<jk>null</jk>
402 	 * omits it) and extra fields, and returns the response body: the echo of {@code protocol|serverName:port|Host values}
403 	 * when the request got through the boundary, empty when it did not.
404 	 */
405 	static String h2cEcho(int port, String authority, String...extra) throws Exception {
406 		var out = new ByteArrayOutputStream();
407 		out.write("PRI * HTTP/2.0\r\n\r\nSM\r\n\r\n".getBytes(StandardCharsets.US_ASCII));
408 		out.write(frame(0x4, 0, 0, new byte[0]));
409 		var h = new ByteArrayOutputStream();
410 		h.write(0x82);                                 // :method GET
411 		h.write(0x86);                                 // :scheme http
412 		h.write(4);                                    // :path /echo
413 		str(h, "/echo");
414 		if (authority != null) {
415 			h.write(1);                                // :authority
416 			str(h, authority);
417 		}
418 		for (var i = 0; i < extra.length; i += 2) {
419 			h.write(0);
420 			str(h, extra[i]);
421 			str(h, extra[i + 1]);
422 		}
423 		out.write(frame(0x1, 0x5, 1, h.toByteArray())); // HEADERS: END_STREAM | END_HEADERS
424 
425 		var body = new StringBuilder();
426 		try (var socket = new Socket("127.0.0.1", port); var in = new DataInputStream(socket.getInputStream())) {
427 			socket.setSoTimeout(10_000);
428 			socket.getOutputStream().write(out.toByteArray());
429 			socket.getOutputStream().flush();
430 			for (;;) {
431 				var head = new byte[9];
432 				in.readFully(head);
433 				var len = ((head[0] & 0xFF) << 16) | ((head[1] & 0xFF) << 8) | (head[2] & 0xFF);
434 				var payload = new byte[len];
435 				in.readFully(payload);
436 				if (head[3] == 0x4 && (head[4] & 0x1) == 0) {
437 					socket.getOutputStream().write(frame(0x4, 0x1, 0, new byte[0]));
438 					socket.getOutputStream().flush();
439 				} else if (head[3] == 0x1) {
440 					if ((head[4] & 0x1) != 0)
441 						break;
442 				} else if (head[3] == 0x0) {
443 					body.append(new String(payload, StandardCharsets.ISO_8859_1));
444 					if ((head[4] & 0x1) != 0)
445 						break;
446 				} else if (head[3] == 0x3 || head[3] == 0x7) {
447 					break;
448 				}
449 			}
450 		}
451 		return body.toString();
452 	}

454 	private static void str(ByteArrayOutputStream h, String value) throws IOException {
455 		h.write(value.length());
456 		h.write(value.getBytes(StandardCharsets.US_ASCII));
457 	}

459 	private static byte[] frame(int type, int flags, int stream, byte[] payload) {
460 		var b = java.nio.ByteBuffer.allocate(9 + payload.length);
461 		b.put((byte)(payload.length >> 16)).put((byte)(payload.length >> 8)).put((byte)payload.length);
462 		b.put((byte)type).put((byte)flags).putInt(stream).put(payload);
463 		return b.array();
464 	}

478 			assertEquals("HTTP/2.0|app.example:80|[]", h2cEcho(port, "app.example"));
479 			assertEquals("HTTP/2.0|app.example:8443|[]", h2cEcho(port, "app.example:8443"));

481 			assertEquals("HTTP/2.0|localhost:80|[]", h2cEcho(port, null));
482 			// X-Forwarded-* are inert without a forwarded-header strategy.
483 			assertEquals("HTTP/2.0|app.example:80|[]", h2cEcho(port, "app.example", "x-forwarded-host", "evil.example", "x-forwarded-proto", "https"));
484 			// A Host field on HTTP/2 is refused by the container itself: the stream is reset and no response comes back.
485 			assertEquals("", h2cEcho(port, "app.example", "host", "other.example"));
