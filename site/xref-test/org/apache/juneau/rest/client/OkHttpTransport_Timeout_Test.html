17  package org.apache.juneau.rest.client;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.io.*;
22  import java.net.*;
23  import java.nio.charset.*;
24  import java.time.*;
25  import java.util.concurrent.*;
26  
27  import org.apache.juneau.rest.client.okhttp.*;
28  import org.junit.jupiter.api.*;
29  
30  import com.sun.net.httpserver.*;
31  
32  /**
33   * Verifies that {@link TransportRequest#getTimeout()} is honored per request by {@code OkHttpTransport}.
34   *
35   * <p>
36   * Uses an embedded server whose {@code /slow} endpoint delays 2s before responding.
37   */
38  @SuppressWarnings({
39  	"java:S2925", // The server handler sleeps to simulate a slow upstream.
40  	"resource" // Transport/response instances are short-lived test fixtures.
41  })
42  // SEPARATE_THREAD: a socket read blocked on a non-responding peer ignores interrupts, so SAME_THREAD would still hang.
43  @Timeout(value = 60, threadMode = Timeout.ThreadMode.SEPARATE_THREAD)
44  class OkHttpTransport_Timeout_Test {
45  
46  	private static HttpServer server;
47  	private static ExecutorService executor;
48  	private static int port;
49  
50  	@BeforeAll
51  	static void startServer() throws IOException {
52  		// Bind to (and connect via) the explicit IPv4 loopback, not the wildcard address + "localhost": a wildcard bind can be
53  		// handed an ephemeral port another process already holds on 127.0.0.1 (SO_REUSEADDR), which then wins every "localhost" connection.
54  		server = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
55  		port = server.getAddress().getPort();
56  		server.createContext("/slow", exchange -> {
57  			try {
58  				Thread.sleep(2000);
59  				var body = "ok".getBytes(StandardCharsets.UTF_8);
60  				exchange.sendResponseHeaders(200, body.length);
61  				exchange.getResponseBody().write(body);
62  			} catch (InterruptedException | IOException e) {
63  				// Client gave up; nothing to do.
64  			} finally {
65  				exchange.close();
66  			}
67  		});
68  		server.createContext("/fast", exchange -> {
69  			var body = "ok".getBytes(StandardCharsets.UTF_8);
70  			exchange.sendResponseHeaders(200, body.length);
71  			exchange.getResponseBody().write(body);
72  			exchange.close();
73  		});
74  		// A dedicated executor keeps a sleeping /slow handler from blocking /fast on the single default dispatcher thread.
75  		executor = Executors.newCachedThreadPool();
76  		server.setExecutor(executor);
77  		server.start();
78  	}
79  
80  	@AfterAll
81  	static void stopServer() {
82  		server.stop(0);
83  		executor.shutdownNow();
84  	}
85  
86  	private static TransportRequest request(String path, Duration timeout) {
87  		return TransportRequest.builder().method("GET").uri("http://127.0.0.1:" + port + path).timeout(timeout).build();
88  	}
89  
90  	@Test void a01_shortTimeout_timesOutQuickly() {
91  		try (var transport = OkHttpTransport.create()) {
92  			var start = System.nanoTime();
93  			var e = assertThrows(TransportException.class, () -> transport.execute(request("/slow", Duration.ofMillis(200))));
94  			assertTrue(Duration.ofNanos(System.nanoTime() - start).toMillis() < 1800, "should time out well before the 2s server delay");
95  			assertNotNull(e.getCause());
96  		}
97  	}
98  
99  	@Test void a02_noTimeout_waitsForResponse() throws Exception {
100 		try (var transport = OkHttpTransport.create(); var response = transport.execute(request("/slow", null))) {
101 			assertEquals(200, response.getStatusCode());
102 		}
103 	}
104 
105 	@Test void a03_longTimeout_succeeds() throws Exception {
106 		try (var transport = OkHttpTransport.create(); var response = transport.execute(request("/slow", Duration.ofSeconds(10)))) {
107 			assertEquals(200, response.getStatusCode());
108 		}
109 	}
110 
111 	@Test void a04_shortTimeout_fastEndpoint_succeeds() throws Exception {
112 		try (var transport = OkHttpTransport.create(); var response = transport.execute(request("/fast", Duration.ofSeconds(1)))) {
113 			assertEquals(200, response.getStatusCode());
114 		}
115 	}
116 }
