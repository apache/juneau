33  /**
34   * Exercises the SSRF-guard (policy-covered {@code @Remote}) path of {@link OkHttpTransport} against a real
35   * loopback server: a guarded request must always be pinned, even when an unguarded request on the same transport has
36   * left an idle keep-alive connection to the same URL in the pool.
37   *
38   * <p>
39   * The hostname tests use {@code localhost} rather than {@code 127.0.0.1} because OkHttp connects to an IP-literal host
40   * directly, without consulting its {@code Dns}; the transport therefore checks an IP-literal host itself (section C).
41   * Likewise a proxy resolves the target host itself, so a guarded request is refused when the call would use a proxy
42   * (section D).
43   */
44  // SEPARATE_THREAD: a socket read blocked on a non-responding peer ignores interrupts, so SAME_THREAD would still hang.
45  @Timeout(value = 60, threadMode = Timeout.ThreadMode.SEPARATE_THREAD)
46  class OkHttpTransport_SsrfGuard_Test {
47

49  	private static String url;

51  	@BeforeAll
52  	static void startServer() throws IOException {
53  		server = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
54  		server.createContext("/hello", exchange -> {
55  			var body = "hi".getBytes(StandardCharsets.UTF_8);
56  			exchange.sendResponseHeaders(200, body.length);
57  			try (var os = exchange.getResponseBody()) { os.write(body); }
58  			exchange.close();
59  		});
60  		server.createContext("/to-numeric", exchange -> {
61  			exchange.getResponseHeaders().add("Location", "http://2130706433:" + server.getAddress().getPort() + "/hello");
62  			exchange.sendResponseHeaders(302, -1);
63  			exchange.close();
64  		});
65  		server.start();
66  		url = "http://localhost:" + server.getAddress().getPort() + "/hello";
67  	}
68  
69  	@AfterAll
70  	static void stopServer() {
71  		if (server != null)
72  			server.stop(0);
73  	}

185 		"resource" // The stubbed Response and ResponseBody are handed to OkHttp's call chain, which closes them.

187 	void c08_guarded_redirectToNumericHost_rejected() {
188 		var target = "http://2130706433:" + server.getAddress().getPort() + "/hello";
189 		// An application interceptor answers hop 0 (a public-looking name) with a redirect, so no network or Dns is needed.
190 		okhttp3.Interceptor redirector = chain -> {
191 			var req = chain.request();
192 			if (! "redirector.test".equals(req.url().host()))
193 				return chain.proceed(req);
194 			return new okhttp3.Response.Builder().request(req).protocol(okhttp3.Protocol.HTTP_1_1).code(302).message("Found")
195 				.header("Location", target).body(okhttp3.ResponseBody.create(new byte[0], null)).build();
196 		};
197 		var client = new okhttp3.OkHttpClient.Builder().proxy(Proxy.NO_PROXY).addInterceptor(redirector).build();
198 		try (var transport = OkHttpTransport.builder().httpClient(client).build()) {
199 			var req = TransportRequest.builder().method("GET").uri("http://redirector.test/start").remoteUrlPolicy(true, false).build();
200 			var ex = assertThrows(TransportException.class, () -> transport.execute(req).close());
201 			assertTrue(ex.getMessage().contains("No allowed address for host '2130706433'"), ex.getMessage());
202 		}

213 			}
214 			@Override public void connectFailed(URI uri, SocketAddress sa, IOException ioe) { /* no-op */ }
215 		};
216 		var client = new okhttp3.OkHttpClient.Builder().proxySelector(selector).build();
217 		try (var transport = OkHttpTransport.builder().httpClient(client).build()) {
218 			var req = TransportRequest.builder().method("GET").uri("http://my_service:8080/hello").remoteUrlPolicy(true, false).build();
219 			var ex = assertThrows(TransportException.class, () -> transport.execute(req).close());
220 			assertFalse(ex.getMessage().contains("proxy"), ex.getMessage());

386 	/** Minimal SOCKS5 endpoint that records the domain name from the CONNECT request, then closes the connection. */
387 	@SuppressWarnings({
388 		"resource" // The ServerSocket is owned by this class and closed in close().
389 	})
390 	private static final class FakeSocksProxy implements Closeable {
391 		private final ServerSocket ss;
392 		private final AtomicReference<String> host = new AtomicReference<>();
393 		private final CountDownLatch done = new CountDownLatch(1);
394 
395 		FakeSocksProxy() throws IOException {
396 			ss = new ServerSocket(0, 5, InetAddress.getLoopbackAddress());
397 			var t = new Thread(this::serve, "fake-socks");
398 			t.setDaemon(true);
399 			t.start();
400 		}
401 
402 		InetSocketAddress address() {
403 			return new InetSocketAddress(InetAddress.getLoopbackAddress(), ss.getLocalPort());
404 		}
405 
406 		String requestedHost() throws InterruptedException {
407 			done.await(2, TimeUnit.SECONDS);
408 			return host.get();
409 		}
410 
411 		private void serve() {
412 			try (var s = ss.accept()) {
413 				s.setSoTimeout(2000);
414 				try (var in = new DataInputStream(s.getInputStream()); var out = s.getOutputStream()) {
415 					in.readByte(); // version
416 					in.readFully(new byte[in.readUnsignedByte()]); // methods
417 					out.write(new byte[]{5, 0});
418 					out.flush();
419 					in.readFully(new byte[3]); // ver, cmd, rsv
420 					if (in.readUnsignedByte() == 3) { // ATYP domain name
421 						var b = new byte[in.readUnsignedByte()];
422 						in.readFully(b);
423 						host.set(new String(b, StandardCharsets.US_ASCII));
424 					}
425 				}
426 			} catch (IOException e) {
427 				// Expected when no client connects or the client gives up; nothing was recorded.
428 			} finally {
429 				done.countDown();
430 			}
431 		}
432 
433 		@Override
434 		public void close() throws IOException {
435 			ss.close();
436 		}
