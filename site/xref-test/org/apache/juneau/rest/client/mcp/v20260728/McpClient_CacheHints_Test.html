31  @SuppressWarnings({
32  	"resource" // Mock HttpTransport lambdas are short-lived test fixtures whose clients are already closed via try-with-resources at each call site.
33  })
34  class McpClient_CacheHints_Test extends TestBase {
35  
36  	static final class RecordingCache implements McpResponseCache {
37  		String lastScope;
38  		String lastKey;
39  		Object lastValue;
40  		long lastTtl;
41  
42  		@Override
43  		public Optional<Object> get(String scope, String key) {
44  			return Optional.empty();
45  		}
46  
47  		@Override
48  		public void put(String scope, String key, Object value, long ttlMs) {
49  			lastScope = scope;
50  			lastKey = key;
51  			lastValue = value;
52  			lastTtl = ttlMs;
53  		}
54  
55  		@Override
56  		public void clear() {
57  			// Intentional no-op: this test double only records put() calls; clear() is never exercised here.
58  		}
59  	}
60  
61  	@Test
62  	void a01_privateScope_usesPerInstancePartitionPrefix() throws Exception {
63  		var cache = new RecordingCache();
64  		try (
65  			var c1 = McpClient.builder().endpoint("http://x").responseCache(cache).build();
66  			var c2 = McpClient.builder().endpoint("http://x").responseCache(cache).build();
67  		) {
68  			assertNotEquals(c1.privateScopePartitionPrefix(), c2.privateScopePartitionPrefix());
69  		}
70  	}
71  
72  	@Test
73  	void a02_cacheableResultWithNullTtlMs_isNeverWritten() throws Exception {
74  		var cache = new RecordingCache();
75  		var calls = new AtomicInteger();
76  		HttpTransport transport = tReq -> {
77  			calls.incrementAndGet();
78  			// A CacheableResult (ListToolsResult) with no "ttlMs" at all - a legal server response opting out
79  			// of caching for this particular call, distinct from the instanceof check failing entirely.
80  			var wire = "{\"jsonrpc\":\"2.0\",\"id\":\"1\",\"result\":{\"tools\":[{\"name\":\"echo\"}]}}";
81  			return TransportResponse.builder()
82  				.statusCode(200)
83  				.header("Content-Type", "application/json")
84  				.body(new ByteArrayInputStream(wire.getBytes(StandardCharsets.UTF_8)))
85  				.build();
86  		};
87  		try (var c = McpClient.builder().endpoint("http://x/mcp").transport(transport).responseCache(cache).build()) {
88  			c.listTools();
89  			c.listTools();
90  			assertEquals(2, calls.get(), "a null-ttlMs CacheableResult must never be written to the cache, so the second call is not served from it");
91  			assertNull(cache.lastKey);
92  		}
93  	}
94  
95  	@Test
96  	void b01_privateScopeResult_isServedFromCacheOnSecondCall() throws Exception {
97  		var calls = new AtomicInteger();
98  		HttpTransport transport = tReq -> {
99  			calls.incrementAndGet();
100 			var wire = "{\"jsonrpc\":\"2.0\",\"id\":\"1\",\"result\":{\"ttlMs\":60000,\"cacheScope\":\"private\",\"tools\":[{\"name\":\"echo\"}]}}";
101 			return TransportResponse.builder()
102 				.statusCode(200)
103 				.header("Content-Type", "application/json")
104 				.body(new ByteArrayInputStream(wire.getBytes(StandardCharsets.UTF_8)))
105 				.build();
106 		};
107 		try (var c = McpClient.builder()
108 			.endpoint("http://x/mcp")
109 			.transport(transport)
110 			.responseCache(new InMemoryMcpResponseCache())
111 			.build()) {
112 			var a = c.listTools();
113 			var b = c.listTools();
114 			assertEquals(1, calls.get(), "second identical call must be served from the private-scope cache partition");
115 			assertSame(a, b);
116 		}
117 	}
