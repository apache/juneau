53  	@BeforeEach
54  	void startServer() throws IOException {
55  		server = HttpServer.create(new InetSocketAddress(InetAddress.getLoopbackAddress(), 0), 0);
56  		server.createContext("/token", ex -> {
57  			try (var is = ex.getRequestBody()) {
58  				requestBodies.add(new String(is.readAllBytes(), StandardCharsets.UTF_8));
59  			}
60  			var body = Objects.requireNonNull(responses.poll(), "no stubbed response").getBytes(StandardCharsets.UTF_8);
61  			ex.getResponseHeaders().add("Content-Type", "application/json");
62  			ex.sendResponseHeaders(200, body.length);
63  			try (var os = ex.getResponseBody()) {
64  				os.write(body);
65  			}
66  		});
67  		server.start();
68  	}

79  	private static String tokenJson(String at, int expiresIn) {
80  		return "{\"access_token\":\"" + at + "\",\"token_type\":\"Bearer\",\"expires_in\":" + expiresIn + "}";

192 	@Test void b08_clockControlsExpiry() {
193 		// Direct coverage of Builder#clock(Clock), distinct from expirySkew(...) (b04), which forces expiry via a
194 		// large skew instead of advancing the clock.
195 		responses.add(tokenJson("cc-at-1", 60));
196 		responses.add(tokenJson("cc-at-2", 60));
197 		// Anchored to real wall-clock time: OAuthToken's expiresAt is computed from the actual issue instant, so the
198 		// injected clock must track close to it (only diverging once we deliberately advance it below).
199 		var now = new AtomicReference<>(Instant.now());
200 		var clock = new Clock() {
201 			@Override public ZoneId getZone() { return ZoneOffset.UTC; }
202 			@Override public Clock withZone(ZoneId zone) { return this; }
203 			@Override public Instant instant() { return now.get(); }
204 		};
205 		var p = McpTokenProvider.clientCredentials()
206 			.tokenEndpoint(tokenEndpoint()).clientId("id").clientSecret("s").resource(RES)
207 			.clock(clock).build();
208 		assertEquals("cc-at-1", p.get());
209 		assertEquals(1, requestBodies.size());
210 		// Still cached: barely any time has passed.
211 		assertEquals("cc-at-1", p.get());
212 		assertEquals(1, requestBodies.size());
213 		// Advance well past the 60s expiry (+ default 30s skew): must re-acquire.
214 		now.set(now.get().plusSeconds(200));
215 		assertEquals("cc-at-2", p.get());
216 		assertEquals(2, requestBodies.size());
217 	}

242 	@Test void c02_rotatedRefreshTokenCapturedAndReused() {
243 		responses.add(tokenJsonWithRefresh("at-1", "rt-2", 3600));
244 		responses.add(tokenJsonWithRefresh("at-2", "rt-3", 3600));
245 		var p = McpTokenProvider.refreshToken("rt-1")
246 			.tokenEndpoint(tokenEndpoint()).clientId("id").resource(RES)
247 			.expirySkew(Duration.ofHours(2)).build(); // force re-acquire each call
248 		assertEquals("at-1", p.get());
249 		assertEquals("rt-2", p.currentRefreshToken().orElseThrow());
250 		assertEquals("at-2", p.get());
251 		assertEquals("rt-3", p.currentRefreshToken().orElseThrow());
252 		// The second refresh must have used the rotated token rt-2, not the original rt-1.
253 		assertTrue(requestBodies.get(1).contains("refresh_token=rt-2"), () -> requestBodies.get(1));
254 	}

267 	@Test void c04b_refreshErrorOtherThanInvalidGrantDoesNotLatchTerminal() {
268 		// Covers the mode==REFRESH branch of get()'s catch (OAuthFlowException) where the error code is present but
269 		// is NOT "invalid_grant" (e.g. a transient server_error): must propagate as-is without latching the
270 		// terminal invalid_grant state, so a subsequent get() still re-attempts against the IdP.
271 		server.removeContext("/token");
272 		var hits = new AtomicInteger();
273 		server.createContext("/token", ex -> {
274 			hits.incrementAndGet();
275 			var b = "{\"error\":\"server_error\"}".getBytes(StandardCharsets.UTF_8);
276 			ex.getResponseHeaders().add("Content-Type", "application/json");
277 			ex.sendResponseHeaders(400, b.length);
278 			try (var os = ex.getResponseBody()) { os.write(b); }
279 		});
280 		var p = McpTokenProvider.refreshToken("rt-1")
281 			.tokenEndpoint(tokenEndpoint()).clientId("id").resource(RES).build();
282 		assertThrows(OAuthFlowException.class, p::get);
283 		assertThrows(OAuthFlowException.class, p::get);
284 		assertEquals(2, hits.get(), "a non-invalid_grant error must not latch the terminal state");
285 	}

327 	@Test void c04_refreshInvalidGrantLatchesTerminal() {
328 		// M7: an invalid_grant on refresh latches a terminal state; subsequent get() throws McpAuthException
329 		// without re-hitting the IdP with the dead refresh token.
330 		server.removeContext("/token");
331 		var hits = new AtomicInteger();
332 		server.createContext("/token", ex -> {
333 			hits.incrementAndGet();
334 			var b = "{\"error\":\"invalid_grant\"}".getBytes(StandardCharsets.UTF_8);
335 			ex.getResponseHeaders().add("Content-Type", "application/json");
336 			ex.sendResponseHeaders(400, b.length);
337 			try (var os = ex.getResponseBody()) { os.write(b); }
338 		});
339 		var p = McpTokenProvider.refreshToken("rt-dead")
340 			.tokenEndpoint(tokenEndpoint()).clientId("id").resource(RES).build();
341 		var e1 = assertThrows(McpAuthException.class, p::get);
342 		assertTrue(e1.getMessage().contains("invalid_grant"), e1::getMessage);
343 		var e2 = assertThrows(McpAuthException.class, p::get);
344 		assertTrue(e2.getMessage().contains("re-authorization required"), e2::getMessage);
345 		assertEquals(1, hits.get(), "terminal state must not re-hit the IdP");
346 	}
