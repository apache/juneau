33  /**
34   * Coverage for {@link McpClient#callRaw(String, RequestParams)}: the result-type-agnostic dispatch a
35   * Multi-Round-Trip-Request (SEP-2322) caller needs to receive an {@code input_required} response and echo its
36   * {@code requestState}/{@code inputResponses} on a resume call (plan Phase 5, Task 14).
37   */
38  @SuppressWarnings({
39  	"resource" // Mock HttpTransport lambdas and the client(...) test-helper factory (@Owning; callers close via try-with-resources) are short-lived test fixtures.
40  })
41  class McpClient_CallRaw_Test extends TestBase {
42  
43  	private static HttpTransport ok(String wireJson) {
44  		return tReq -> TransportResponse.builder()
45  			.statusCode(200)
46  			.header("Content-Type", "application/json")
47  			.body(new ByteArrayInputStream(wireJson.getBytes(StandardCharsets.UTF_8)))
48  			.build();
49  	}
50  
51  	private static McpClient client(HttpTransport transport) {
52  		return McpClient.builder().endpoint("http://x/mcp").transport(transport).build();
53  	}
54  
55  	@Test
56  	void a01_callRaw_completeResult_returnsRawMapMatchingTypedCall() throws Exception {
57  		var wire = "{\"jsonrpc\":\"2.0\",\"id\":\"1\",\"result\":{\"content\":[{\"type\":\"text\",\"text\":\"hello\"}],\"resultType\":\"complete\"}}";
58  		try (var c = client(ok(wire))) {
59  			var raw = c.callRaw(McpMethods.TOOLS_CALL, new CallToolRequest().setName("echo").setArguments(Map.of("text", "hello")));
60  			assertEquals("complete", raw.get("resultType"));
61  			assertNotNull(raw.get("content"));
62  		}
63  		// The same wire, consumed through the typed method, yields a CallToolResult with matching content --
64  		// proving callRaw returns the identical underlying JSON, only undiscriminated.
65  		try (var c = client(ok(wire))) {
66  			var typed = c.callTool("echo", Map.of("text", "hello"));
67  			assertEquals("hello", ((TextContent) typed.getContent().get(0)).getText());
68  		}
69  	}
70  
71  	@Test
72  	void a02_callRaw_inputRequiredResult_exposesRequestStateAndInputRequests() throws Exception {
73  		var wire = "{\"jsonrpc\":\"2.0\",\"id\":\"1\",\"result\":{\"inputRequests\":{\"q1\":{\"type\":\"elicitation\"}},\"requestState\":\"tok-abc\",\"resultType\":\"input_required\"}}";
74  		try (var c = client(ok(wire))) {
75  			var raw = c.callRaw(McpMethods.TOOLS_CALL, new CallToolRequest().setName("ask"));
76  			assertEquals("input_required", raw.get("resultType"));
77  			assertEquals("tok-abc", raw.get("requestState"));
78  			assertInstanceOf(Map.class, raw.get("inputRequests"));
79  			assertTrue(((Map<?,?>) raw.get("inputRequests")).containsKey("q1"));
80  		}
81  	}
82  
83  	@Test
84  	void a03_callRaw_jsonRpcError_throwsMcpExceptionLikeEveryTypedMethod() throws Exception {
85  		var wire = "{\"jsonrpc\":\"2.0\",\"id\":\"1\",\"error\":{\"code\":-32602,\"message\":\"bad params\"}}";
86  		try (var c = client(ok(wire))) {
87  			var params = new CallToolRequest().setName("ask").setRequestState("stale");
88  			var e = assertThrows(McpException.class, () -> c.callRaw(McpMethods.TOOLS_CALL, params));
89  			assertEquals(-32602, e.getCode());
90  		}
91  	}

93  	/**
94  	 * Reproduces the mixed-API cache-key collision: a typed call caches a non-{@link Map} result bean under a key
95  	 * that {@code callRaw} would compute identically for the same method+params. {@code CallToolResult} (the
96  	 * {@code tools/call} result type) is never actually written to the cache because it doesn't implement
97  	 * {@link CacheableResult}, so {@code resources/read} (whose
98  	 * {@link ReadResourceResult} does implement it) is the type that is genuinely reachable end-to-end: prime the
99  	 * cache via the typed {@link McpClient#readResource} call, then issue {@link McpClient#callRaw} for the same
100 	 * {@code resources/read} URI. Before the fix, {@code callRaw}'s cache READ would return the cached
101 	 * {@code ReadResourceResult} and {@code Map.class.cast(...)} it, throwing {@link ClassCastException}.
102 	 */
103 	@Test
104 	void a04_callRaw_doesNotReadTypedResultFromSharedCacheKey() throws Exception {
105 		var calls = new AtomicInteger();
106 		HttpTransport transport = tReq -> {
107 			var n = calls.incrementAndGet();
108 			var wire = n == 1
109 				? "{\"jsonrpc\":\"2.0\",\"id\":\"1\",\"result\":{\"contents\":[{\"type\":\"resourceText\",\"uri\":\"file:///a\",\"text\":\"cached-body\"}],\"ttlMs\":60000}}"
110 				: "{\"jsonrpc\":\"2.0\",\"id\":\"2\",\"result\":{\"contents\":[{\"type\":\"resourceText\",\"uri\":\"file:///a\",\"text\":\"wire-body\"}]}}";
111 			return TransportResponse.builder()
112 				.statusCode(200)
113 				.header("Content-Type", "application/json")
114 				.body(new ByteArrayInputStream(wire.getBytes(StandardCharsets.UTF_8)))
115 				.build();
116 		};
117 		try (var c = McpClient.builder()
118 			.endpoint("http://x/mcp")
119 			.transport(transport)
120 			.responseCache(new InMemoryMcpResponseCache())
121 			.build()) {
122 			// Primes the cache with a typed ReadResourceResult under the resources/read + uri cache key.
123 			c.readResource("file:///a");
124 			assertEquals(1, calls.get());
125 
126 			// callRaw on the same method+params must not retrieve the cached ReadResourceResult and must not throw
127 			// ClassCastException; it goes to the wire instead, proven by the second (distinct) wire body winning.
128 			var params = new ReadResourceRequest().setUri("file:///a");
129 			var raw = assertDoesNotThrow(() -> c.callRaw(McpMethods.RESOURCES_READ, params));
130 			assertEquals(2, calls.get());
131 			var contents = (List<?>) raw.get("contents");
132 			var content = (Map<?,?>) contents.get(0);
133 			assertEquals("wire-body", content.get("text"));
134 		}
135 	}
