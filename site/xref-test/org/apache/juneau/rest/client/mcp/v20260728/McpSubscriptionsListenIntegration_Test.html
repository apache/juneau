47   * End-to-end coverage for MCP {@code 2026-07-28} SEP-2575 {@code subscriptions/listen} (C8), driven through the
48   * real {@link McpClient#listen} against a live embedded-Jetty server. This is the repo's first real-Jetty test to
49   * exercise a held-open SSE response over the actual {@code AsyncContext}/{@code startAsync()} branch — every prior
50   * SSE test uses {@code MockRestClient}'s synchronous fallback path, which cannot represent a held-open response at
51   * all.
52   *
53   * <p>
54   * The fixture's {@link McpSubscriptionsConfig#getHeartbeatIntervalMs() heartbeatIntervalMs} is tuned down to
55   * {@value #HEARTBEAT_INTERVAL_MS}&nbsp;ms (from the 15s production default) so heartbeat-arrival coverage

132 	/**
133 	 * Bounded polling await shared by a01 and z01's server-side cleanup checks: cleanup is write-failure/close
134 	 * driven on the server side (it only notices a disconnect on its next write attempt), not proactive, so there
135 	 * is no push notification to await deterministically - only the real {@code activeCount} condition, polled on
136 	 * a bounded schedule. Returns the last observed count once it reads {@code "0"} or the deadline elapses.
137 	 */
138 	@SuppressWarnings("java:S2925") // pacing delay between condition polls; no push notification exists for this rollup
139 	private static String awaitActiveCountZero(McpClient client, long timeoutMs, long pollMs) throws Exception {
140 		var deadline = System.currentTimeMillis() + timeoutMs;
141 		String count;
142 		do {
143 			Thread.sleep(pollMs);
144 			count = ((TextContent) client.callTool("activeCount", Map.of()).getContent().get(0)).getText();
145 		} while (! "0".equals(count) && System.currentTimeMillis() < deadline);
146 		return count;
147 	}

153 	@Test void a01_listen_receivesAcknowledgedThenPublishedNotification_thenBrokerCleansUpOnClose() throws Exception {
154 		try (var client = clientBuilder().build()) {
155 			var events = new LinkedBlockingQueue<String>();
156 			var filter = new SubscriptionFilter().setResourceSubscriptions(List.of("file:///a.txt")).setResourcesListChanged(true);
157 			var acknowledged = new LinkedBlockingQueue<SubscriptionFilter>();
158 			var handle = client.listen(filter, new McpSubscriptionListener() {
159 				@Override public void onAcknowledged(SubscriptionFilter honoredFilter) {
160 					acknowledged.add(honoredFilter);
161 					events.add("ack");
162 				}
163 				@Override public void onResourceUpdated(String uri) {
164 					events.add("updated:" + uri);
165 				}
166 				@Override public void onListChanged(McpListChangedKind kind) {
167 					events.add("listChanged:" + kind);
168 				}
169 				@Override public void onError(Throwable t) {
170 					events.add("error:" + t);
171 				}
172 			});
173 			try {
174 				// Assertion 1: no -32600 "Missing required header" error frame - the mandatory acknowledged
175 				// frame is the very first thing received. This is precisely what the McpClient.listen(...)
176 				// header fix enables: without it, this poll would instead see "error:..." (or time out, since
177 				// the client would never receive a well-formed acknowledged frame at all).
178 				assertEquals("ack", events.poll(15, TimeUnit.SECONDS));
179 				var honoredFilter = acknowledged.poll(15, TimeUnit.SECONDS);
180 				assertNotNull(honoredFilter);
181 				// Compared by content (List#equals), not by List#toString() format - the parsed list's
182 				// concrete runtime type renders its toString() as JSON (quoting string elements), which is a
183 				// distinct concern from the (unquoted) String content each element actually carries.
184 				assertEquals(List.of("file:///a.txt"), honoredFilter.getResourceSubscriptions());
185 				assertEquals(Boolean.TRUE, honoredFilter.getResourcesListChanged());
186 
187 				var countBefore = client.callTool("activeCount", Map.of());
188 				assertEquals("1", ((TextContent) countBefore.getContent().get(0)).getText());
189 
190 				// Idle window spanning several heartbeat intervals before publishing: the fixture's live
191 				// heartbeat frames are certainly interleaved on the wire during this wait (proven directly by
192 				// c01, below), so an empty poll here demonstrates the client pump silently skips them rather
193 				// than raising a spurious callback (onError, a bogus "updated", etc.). BlockingQueue#poll's
194 				// bounded wait races against the real background pump thread - it returns the instant a
195 				// (spurious) event arrives, and only blocks the full duration when none does.
196 				assertNull(events.poll(6 * HEARTBEAT_INTERVAL_MS, TimeUnit.MILLISECONDS),
197 					"a heartbeat frame must never surface as a listener callback");
198 
199 				// Assertion 2: a server-emitted change (via the neutral McpSubscriptions SPI, reached through
200 				// the tool handler's BeanStore) round-trips end-to-end to the decoded onResourceUpdated
201 				// callback.
202 				client.callTool("publish", Map.of("uri", "file:///a.txt"));
203 				assertEquals("updated:file:///a.txt", events.poll(15, TimeUnit.SECONDS));
204 			} finally {
205 				handle.close();
206 			}
207 
208 			// Assertion 4: cleanup is write-failure/close-driven, not proactive (spec's "Transport/threading
209 			// detail") - the server only notices the client's disconnect on its next write attempt (heartbeat
210 			// or otherwise), so poll rather than assert immediately. Bounded poll loop, not a fixed sleep for
211 			// synchronization: each iteration's brief sleep is only the between-poll delay, gated by the outer
212 			// deadline.
213 			var countAfter = awaitActiveCountZero(client, 15_000, 200);
214 			assertEquals("0", countAfter);
215 		}
216 	}

222 	@Test void b01_rawHttpClient_holdsOpenSseResponse_deliversAcknowledgedFirst() throws Exception {
223 		var listenId = "raw-1";
224 		// params._meta.protocolVersion/clientCapabilities are mandatory on every v2 request (see
225 		// McpRevision.validateMeta) - McpClient builds this automatically; the raw path must supply it itself.
226 		var body = "{\"jsonrpc\":\"2.0\",\"id\":\"" + listenId + "\",\"method\":\"subscriptions/listen\","
227 			+ "\"params\":{\"notifications\":{\"resourcesListChanged\":true},\"_meta\":{"
228 			+ "\"io.modelcontextprotocol/protocolVersion\":\"2026-07-28\",\"io.modelcontextprotocol/clientCapabilities\":{}}}}";
229 		var req = HttpRequest.newBuilder()
230 			.uri(URI.create(fixture.getRootUrl() + "/"))
231 			.header("Content-Type", "application/json")
232 			.header("Accept", "text/event-stream")
233 			// SEP-2243 Mcp-Method/Mcp-Name headers - McpRevision.dispatch() validates these unconditionally
234 			// before branching on the method (see the McpClient.listen() header fix, above); this raw request
235 			// bypasses McpClient entirely, so it must stamp them itself exactly as McpClient.listen() does.
236 			.header("Mcp-Method", McpMethods.SUBSCRIPTIONS_LISTEN)
237 			.header("Mcp-Name", "")
238 			.timeout(Duration.ofSeconds(15))
239 			.POST(HttpRequest.BodyPublishers.ofString(body))
240 			.build();
241 		var resp = HTTP.send(req, BodyHandlers.ofInputStream());
242 		assertEquals(200, resp.statusCode());
243 		try (var reader = new SseEventReader(new InputStreamReader(resp.body(), StandardCharsets.UTF_8))) {
244 			assertTrue(reader.hasNext());
245 			var first = reader.next();
246 			assertTrue(first.getData() != null && first.getData().contains("notifications/subscriptions/acknowledged"), "" + first.getData());
247 		}
248 	}

254 	/**
255 	 * Opens a raw (client-facade-independent) SSE stream and, on a dedicated reader thread, counts named
256 	 * {@code "ping"} events with no {@code data:} payload arriving after the mandatory acknowledged frame - the
257 	 * fixture's tuned-down {@link #HEARTBEAT_INTERVAL_MS} means at least two must arrive well within the bounded
258 	 * wait below. The read itself is a blocking I/O call (not a sleep-based poll); the bound comes from a
259 	 * {@link CountDownLatch#await(long, TimeUnit)} racing the reader thread, and the connection is force-closed in
260 	 * {@code finally} to unblock that thread if the assertion path never reaches the expected count.
261 	 */
262 	@Test void c01_heartbeat_arrivesRepeatedlyOverLiveConnection_whileIdle() throws Exception {
263 		var body = "{\"jsonrpc\":\"2.0\",\"id\":\"hb-1\",\"method\":\"subscriptions/listen\","
264 			+ "\"params\":{\"notifications\":{},\"_meta\":{"
265 			+ "\"io.modelcontextprotocol/protocolVersion\":\"2026-07-28\",\"io.modelcontextprotocol/clientCapabilities\":{}}}}";
266 		var req = HttpRequest.newBuilder()
267 			.uri(URI.create(fixture.getRootUrl() + "/"))
268 			.header("Content-Type", "application/json")
269 			.header("Accept", "text/event-stream")
270 			// SEP-2243 Mcp-Method/Mcp-Name headers - see b01's comment for why the raw HTTP path must stamp
271 			// these itself.
272 			.header("Mcp-Method", McpMethods.SUBSCRIPTIONS_LISTEN)
273 			.header("Mcp-Name", "")
274 			.timeout(Duration.ofSeconds(20))
275 			.POST(HttpRequest.BodyPublishers.ofString(body))
276 			.build();
277 		var resp = HTTP.send(req, BodyHandlers.ofInputStream());
278 		assertEquals(200, resp.statusCode());
279 
280 		var requiredPings = 2;
281 		var pingsSeen = new CountDownLatch(requiredPings);
282 		var failure = new AtomicReference<Throwable>();
283 		var readerThread = new Thread(() -> {
284 			try (var reader = new SseEventReader(new InputStreamReader(resp.body(), StandardCharsets.UTF_8))) {
285 				assertTrue(reader.hasNext());
286 				var first = reader.next();
287 				assertTrue(first.getData() != null && first.getData().contains("notifications/subscriptions/acknowledged"), first.getData());
288 				while (reader.hasNext() && pingsSeen.getCount() > 0) {
289 					var event = reader.next();
290 					if ("ping".equals(event.getEvent())) {
291 						assertNull(event.getData(), "a heartbeat frame must carry no data: payload");
292 						pingsSeen.countDown();
293 					}
294 				}
295 			} catch (Throwable t) {
296 				failure.set(t);
297 			}
298 		}, "mcp-heartbeat-integration-reader");
299 		readerThread.setDaemon(true);
300 		readerThread.start();
301 		try {
302 			assertTrue(pingsSeen.await(15, TimeUnit.SECONDS),
303 				"expected " + requiredPings + " heartbeat frame(s) over the live connection; saw "
304 					+ (requiredPings - pingsSeen.getCount()));
305 			if (failure.get() != null)
306 				throw new AssertionError(failure.get());
307 		} finally {
308 			resp.body().close();
309 			readerThread.join(5_000);
310 		}
311 	}

324 		var body = "{\"jsonrpc\":\"2.0\",\"id\":\"z-1\",\"method\":\"subscriptions/listen\","
325 			+ "\"params\":{\"notifications\":{},\"_meta\":{"
326 			+ "\"io.modelcontextprotocol/protocolVersion\":\"2026-07-28\",\"io.modelcontextprotocol/clientCapabilities\":{}}}}";
327 		var req = HttpRequest.newBuilder()
328 			.uri(URI.create(fixture.getRootUrl() + "/"))
329 			.header("Content-Type", "application/json")
330 			.header("Accept", "text/event-stream")
331 			.header("Mcp-Method", McpMethods.SUBSCRIPTIONS_LISTEN)
332 			.header("Mcp-Name", "")
333 			.timeout(Duration.ofSeconds(20))
334 			.POST(HttpRequest.BodyPublishers.ofString(body))
335 			.build();
