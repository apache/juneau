34  /**
35   * Coverage for {@link McpClient#connect}, the one-shot build-and-handshake convenience.
36   *
37   * <p>
38   * {@code 2026-07-28} has no {@code initialize} method (see {@link McpClient_Surface_Test}) - its handshake
39   * is {@link McpClient#serverDiscover()} - so {@code connect(...)} performs that call exactly once instead.
40   */
41  @SuppressWarnings({
42  	"resource" // Mock HttpTransport (including the deliberately-`failing` implementations used to pin close-failure/suppressed-exception behavior) and countingTransport(...) instances are short-lived test fixtures; some are intentionally unassigned/never closed since these tests pin McpClient.connect(...)'s own close behavior, not the mock transport's.
43  })
44  class McpClient_Connect_Test extends TestBase {

46  	private static final String DISCOVER_WIRE =
47  		"{\"jsonrpc\":\"2.0\",\"id\":\"1\",\"result\":{\"supportedVersions\":[\"2026-07-28\"],\"capabilities\":{}}}";
48  
49  	private static HttpTransport countingTransport(AtomicInteger calls, String wireJson) {
50  		return req -> {
51  			calls.incrementAndGet();
52  			return TransportResponse.builder()
53  				.statusCode(200)
54  				.header("Content-Type", "application/json")
55  				.body(new ByteArrayInputStream(wireJson.getBytes(StandardCharsets.UTF_8)))
56  				.build();
57  		};
58  	}

60  	@Test
61  	void a01_connect_withEndpoint_buildsAndDiscoversExactlyOnce() throws Exception {
62  		var calls = new AtomicInteger();
63  		var savedTransport = new AtomicReference<HttpTransport>();
64  		// endpoint-only overload builds its own transport internally in production, but tests must inject a fake
65  		// transport - so exercise the Builder overload here and cover the String overload's delegation separately.
66  		savedTransport.set(countingTransport(calls, DISCOVER_WIRE));
67  		try (var c = McpClient.connect(McpClient.builder().endpoint("http://x/mcp").transport(savedTransport.get()))) {
68  			assertNotNull(c);
69  			assertEquals(1, calls.get());
70  		}
71  	}

74  	void a02_connect_cachesDiscoveredServerSoNoSecondRoundTripIsNeeded() throws Exception {
75  		var calls = new AtomicInteger();
76  		try (var c = McpClient.connect(McpClient.builder().endpoint("http://x/mcp").transport(countingTransport(calls, DISCOVER_WIRE)))) {
77  			var result = c.discoveredServer();
78  			assertNotNull(result);
79  			assertEquals("2026-07-28", result.getSupportedVersions().get(0));
80  			assertEquals(1, calls.get());
81  		}

190 	@Test
191 	void e02_connect_endpointOnlyOverload_actuallyDelegatesToBuilderOverload() {
192 		// Port 1 is a reserved/unassigned port on loopback; connecting to it fails fast with no listener
193 		// needed, letting this actually execute connect(String)'s delegation to connect(builder().endpoint(...))
194 		// against the real default transport, rather than just pinning its surface via reflection (see e01).
195 		// This alone does not fully cover connect(String)'s single-statement body: JaCoCo's probe for that
196 		// one-line block sits at the (never-reached, since this throws) return, so a throwing call still shows
197 		// the whole line as uncovered - see e03 below for the one true happy-path exercise of that line.
198 		assertThrows(IOException.class, () -> McpClient.connect("http://127.0.0.1:1/mcp"));
199 	}

201 	@Test
202 	void e03_connect_endpointOnlyOverload_realLoopbackServer_completesTheDelegatingReturnStatement() throws Exception {
203 		// connect(String) builds its own default transport internally (no injection point), so actually
204 		// completing its "return connect(builder().endpoint(endpoint));" statement - as opposed to merely
205 		// entering it and throwing partway through, like e02 - requires a real round trip against a real
206 		// listening server. A loopback com.sun.net.httpserver.HttpServer (same technique already used
207 		// elsewhere in this codebase, e.g. JavaHttpTransport_Test/McpDynamicClientRegistrar_Test) gives that
208 		// without any live external dependency.
209 		var server = HttpServer.create(new InetSocketAddress(InetAddress.getLoopbackAddress(), 0), 0);
210 		server.createContext("/mcp", exchange -> {
211 			var body = DISCOVER_WIRE.getBytes(StandardCharsets.UTF_8);
212 			exchange.getResponseHeaders().add("Content-Type", "application/json");
213 			exchange.sendResponseHeaders(200, body.length);
214 			exchange.getResponseBody().write(body);
215 			exchange.close();
216 		});
217 		server.start();
218 		try (var c = McpClient.connect("http://" + server.getAddress().getHostString() + ":" + server.getAddress().getPort() + "/mcp")) {
219 			assertNotNull(c);
220 			assertNotNull(c.discoveredServer());
221 		} finally {
222 			server.stop(0);
223 		}
224 	}
