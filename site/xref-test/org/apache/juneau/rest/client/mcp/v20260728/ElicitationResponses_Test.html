34  	@Test void a01_toInputResponse_singleAnswer_encodesAcceptWithContent() {
35  		var result = new ElicitResult().setAction(ElicitAction.ACCEPT).putContent("choice", "red");
36  		var encoded = ElicitationResponses.toInputResponse("q1", result);
37  		assertEquals(1, encoded.size());
38  		var q1 = (JsonMap) encoded.get("q1");
39  		assertEquals("accept", q1.getString("action"));
40  		assertEquals("red", q1.getMap("content").getString("choice"));
41  	}

43  	@Test void a02_toInputResponses_multipleAnswers_encodesAll() {
44  		var results = new LinkedHashMap<String,ElicitResult>();
45  		results.put("q1", new ElicitResult().setAction(ElicitAction.ACCEPT).putContent("choice", "red"));
46  		results.put("q2", new ElicitResult().setAction(ElicitAction.DECLINE));
47  		var encoded = ElicitationResponses.toInputResponses(results);
48  		assertEquals(2, encoded.size());
49  		var q1 = (JsonMap) encoded.get("q1");
50  		assertEquals("accept", q1.getString("action"));
51  		assertEquals("red", q1.getMap("content").getString("choice"));
52  		var q2 = (JsonMap) encoded.get("q2");
53  		assertEquals("decline", q2.getString("action"));
54  		assertNull(q2.get("content"));
55  	}

81  	/**
82  	 * Documents the wire-shape symmetry between this client module's {@link ElicitationRequests}/
83  	 * {@link ElicitationResponses} pair and the server module's {@code ElicitationRequests}/
84  	 * {@code ElicitationResponses} (Phase 4) without introducing an illegal cross-module test dependency:
85  	 * {@code juneau-rest-client-mcp-v20260728} must not depend on {@code juneau-rest-server-mcp-v20260728}, so
86  	 * the server-side raw shape is hand-built here to match it exactly (verified by inspection/spec
87  	 * cross-reference against {@code ElicitationRequests.of}/{@code ElicitationResponses.get}, not a live
88  	 * cross-module call).
89  	 */
90  	@Test void a07_roundTrip_requestsToResponses_matchesServerSideHelpers() {
91  		// Mirrors the raw shape ElicitationRequests.of("q1", elicitRequest, ...) (server side) would place into
92  		// McpInputRequiredSignal.getInputRequests(), which callRaw ultimately surfaces as raw's "inputRequests".
93  		var rawInputRequests = JsonMap.of(
94  			"q1", JsonMap.of("message", "Pick one", "requestedSchema", JsonMap.of("type", "object"))
95  		);
96  		var raw = Map.<String,Object>of("resultType", "input_required", "requestState", "tok-abc", "inputRequests", rawInputRequests);
97  
98  		var decodedRequests = ElicitationRequests.requests(raw);
99  		assertEquals("Pick one", decodedRequests.get("q1").getMessage());
100 		assertEquals("tok-abc", ElicitationRequests.requestState(raw));
101 
102 		var answer = new ElicitResult().setAction(ElicitAction.ACCEPT).putContent("choice", "red");
103 		var encodedResponses = ElicitationResponses.toInputResponse("q1", answer);
104 
105 		// This is exactly the shape ElicitationResponses.get(ctx, "q1") (server side) decodes via
106 		// Json.to(Json.of(ctx.inputResponses().get("q1")), ElicitResult.class).
107 		var q1 = (JsonMap) encodedResponses.get("q1");
108 		assertEquals("accept", q1.getString("action"));
109 		assertEquals("red", q1.getMap("content").getString("choice"));
110 	}
