17  package org.apache.juneau.rest.client.mcp.auth;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.net.*;
22  import java.time.*;
23  
24  import org.apache.juneau.*;
25  import org.junit.jupiter.api.*;

27  /**
28   * Tests for {@link EphemeralStore} &mdash; single-use, TTL-bounded, size-bounded {@code state} &rarr;
29   * {@code (codeVerifier, expectedIssuer)} storage (CSRF/replay guard).  Ported and adapted from the
30   * {@code juneau-rest-server-auth-oidc-rp} original to the headless {@link EphemeralStore.PendingAuthorization} shape.
31   *
32   * @since 10.0.0
33   */
34  class EphemeralStore_Test extends TestBase {
35  
36  	private static final URI ISS = URI.create("https://as.example.com");
37  
38  	// Fixed clock seam: these cases don't depend on wall-clock time, so a deterministic clock replaces the system clock.
39  	private static final Clock CLOCK = Clock.fixed(Instant.parse("2026-01-01T00:00:00Z"), ZoneOffset.UTC);
40  
41  	private static EphemeralStore store(Clock clock) {
42  		return new EphemeralStore(Duration.ofMinutes(5), 100, clock);
43  	}
44  
45  	// -----------------------------------------------------------------------------------------------------------------
46  	// A: store / consume single-use
47  	// -----------------------------------------------------------------------------------------------------------------
48  
49  	@Test void a01_storeThenConsume_roundTrips() {
50  		var s = store(CLOCK);
51  		s.store("state-1", "verifier-1", ISS);
52  		var p = s.consume("state-1");
53  		assertTrue(p.isPresent());
54  		assertEquals("verifier-1", p.get().codeVerifier());
55  		assertEquals(ISS, p.get().expectedIssuer());
56  		assertEquals(Instant.parse("2026-01-01T00:00:00Z"), p.get().createdAt());
57  	}
58  
59  	@Test void a02_consume_isSingleUse() {
60  		var s = store(CLOCK);
61  		s.store("state-1", "verifier-1", ISS);
62  		assertTrue(s.consume("state-1").isPresent());
63  		assertTrue(s.consume("state-1").isEmpty(), "second consume of same state must miss (replay defense)");
64  	}
65  
66  	@Test void a03_consume_unknownState_isEmpty() {
67  		var s = store(CLOCK);
68  		assertTrue(s.consume("never-stored").isEmpty());
69  	}
70  
71  	@Test void a04_consume_nullState_isEmpty() {
72  		var s = store(CLOCK);
73  		assertTrue(s.consume(null).isEmpty());
74  	}
75  
76  	@Test void a05_nullExpectedIssuer_allowed() {
77  		var s = store(CLOCK);
78  		s.store("state-1", "verifier-1", null);
79  		var p = s.consume("state-1");
80  		assertTrue(p.isPresent());
81  		assertNull(p.get().expectedIssuer());
82  	}
83  
84  	// -----------------------------------------------------------------------------------------------------------------
85  	// B: TTL expiry
86  	// -----------------------------------------------------------------------------------------------------------------
87  
88  	@Test void b01_expiredEntry_isMissed() {
89  		var clock = new MutableClock(Instant.parse("2026-01-01T00:00:00Z"));
90  		var s = new EphemeralStore(Duration.ofMinutes(5), 100, clock);
91  		s.store("state-1", "verifier-1", ISS);
92  		clock.advance(Duration.ofMinutes(6));
93  		assertTrue(s.consume("state-1").isEmpty(), "entry past TTL must be treated as a miss");
94  	}
95  
96  	@Test void b02_notYetExpired_survives() {
97  		var clock = new MutableClock(Instant.parse("2026-01-01T00:00:00Z"));
98  		var s = new EphemeralStore(Duration.ofMinutes(5), 100, clock);
99  		s.store("state-1", "verifier-1", ISS);
100 		clock.advance(Duration.ofMinutes(4));
101 		assertTrue(s.consume("state-1").isPresent());
102 	}
103 
104 	// -----------------------------------------------------------------------------------------------------------------
105 	// C: LRU size cap
106 	// -----------------------------------------------------------------------------------------------------------------
107 
108 	@Test void c01_sizeCap_evictsEldest() {
109 		var s = new EphemeralStore(Duration.ofMinutes(5), 3, CLOCK);
110 		s.store("s1", "v", ISS);
111 		s.store("s2", "v", ISS);
112 		s.store("s3", "v", ISS);
113 		s.store("s4", "v", ISS);  // evicts s1
114 		assertEquals(3, s.size());
115 		assertTrue(s.consume("s1").isEmpty());
116 		assertTrue(s.consume("s4").isPresent());
117 	}
118 
119 	// -----------------------------------------------------------------------------------------------------------------
120 	// D: constructor / argument validation
121 	// -----------------------------------------------------------------------------------------------------------------
122 
123 	@Test void d01_rejectsNonPositiveTtl() {
124 		assertThrows(IllegalArgumentException.class, () -> new EphemeralStore(Duration.ZERO, 100, CLOCK));
125 	}
126 
127 	@Test void d02_rejectsNegativeTtl() {
128 		var negativeTtl = Duration.ofSeconds(-1);
129 		assertThrows(IllegalArgumentException.class, () -> new EphemeralStore(negativeTtl, 100, CLOCK));
130 	}
131 
132 	@Test void d03_rejectsTtlAboveMax() {
133 		var aboveMaxTtl = EphemeralStore.MAX_TTL.plusSeconds(1);
134 		assertThrows(IllegalArgumentException.class, () -> new EphemeralStore(aboveMaxTtl, 100, CLOCK));
135 	}
136 
137 	@Test void d04_ttlAtMaxAllowed() {
138 		var s = new EphemeralStore(EphemeralStore.MAX_TTL, 100, CLOCK);
139 		s.store("state-1", "verifier-1", ISS);
140 		assertTrue(s.consume("state-1").isPresent());
141 	}
142 
143 	@Test void d05_rejectsNonPositiveMaxEntries() {
144 		var ttl = Duration.ofMinutes(5);
145 		assertThrows(IllegalArgumentException.class, () -> new EphemeralStore(ttl, 0, CLOCK));
146 	}
147 
148 	@Test void d06_rejectsBlankStateOnStore() {
149 		var s = store(CLOCK);
150 		assertThrows(IllegalArgumentException.class, () -> s.store("", "v", ISS));
151 	}
152 
153 	@Test void d07_rejectsBlankVerifierOnStore() {
154 		var s = store(CLOCK);
155 		assertThrows(IllegalArgumentException.class, () -> s.store("state-1", "", ISS));
156 	}
157 
158 	// -----------------------------------------------------------------------------------------------------------------
159 	// E: redaction (M3)
160 	// -----------------------------------------------------------------------------------------------------------------
161 
162 	@Test void e01_toStringRedactsCodeVerifier() {
163 		var p = new EphemeralStore.PendingAuthorization("super-secret-verifier", ISS, Instant.parse("2026-01-01T00:00:00Z"));
164 		assertFalse(p.toString().contains("super-secret-verifier"), p::toString);
165 		assertTrue(p.toString().contains("<redacted>"), p::toString);
166 		assertTrue(p.toString().contains("as.example.com"), p::toString);
167 	}
168 
169 	/** A test clock the test can advance manually. */
170 	static final class MutableClock extends Clock {
171 		private Instant now;
172 		MutableClock(Instant start) { now = start; }
173 		void advance(Duration d) { now = now.plus(d); }
174 		@Override public ZoneId getZone() { return ZoneOffset.UTC; }
175 		@Override public Clock withZone(ZoneId zone) { return this; }
176 		@Override public Instant instant() { return now; }
177 	}
178 }
