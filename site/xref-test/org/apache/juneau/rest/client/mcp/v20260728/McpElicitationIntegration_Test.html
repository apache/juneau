53  	private static ElicitRequest confirmQuestion() {
54  		return new ElicitRequest()
55  			.setMessage("Proceed with deletion?")
56  			.setRequestedSchema(ElicitSchema.create().booleanField("confirm").title("Confirm").build());
57  	}

59  	// The server-side org.apache.juneau.rest.server.mcp.v20260728.ElicitationRequests/ElicitationResponses
60  	// (wildcard-imported above) share their simple names with this test's own client-side package
61  	// (org.apache.juneau.rest.client.mcp.v20260728.ElicitationRequests/ElicitationResponses), where same-package
62  	// visibility wins over a wildcard import - so every server-side call below is qualified with its full
63  	// package to disambiguate, while every client-side call in the test bodies below stays unqualified.
64  	private static McpToolHandler confirm() {
65  		return new McpToolHandler() {
66  			@Override public McpToolSpec descriptor() { return new McpToolSpec().setName("confirm").setDescription("Pauses for one confirmation"); }
67  			@Override public McpToolOutcome call(Map<String,Object> arguments, BeanStore ctx) {
68  				var resume = ctx.getBean(McpMrtrResumeContext.class);
69  				if (resume.isEmpty())
70  					throw org.apache.juneau.rest.server.mcp.v20260728.ElicitationRequests.of("confirm", confirmQuestion(), "cont-1");
71  				var answer = org.apache.juneau.rest.server.mcp.v20260728.ElicitationResponses.get(resume.get(), "confirm");
72  				// Echoes the answer's content (not just its action) and the continuation this tool paused
73  				// with, so the assertions below actually exercise ElicitResult.content marshalling and prove
74  				// the continuation round-trips through McpMrtrResumeContext, instead of only the action enum.
75  				var confirmValue = answer.getContent() == null ? null : answer.getContent().get("confirm");
76  				return McpToolOutcome.text("action:" + answer.getAction() + ",confirm:" + confirmValue
77  					+ ",cont:" + resume.get().continuation());
78  			}
79  		};
80  	}
81  
82  	private static McpToolHandler confirmTwo() {
83  		return new McpToolHandler() {
84  			@Override public McpToolSpec descriptor() { return new McpToolSpec().setName("confirmTwo").setDescription("Pauses for two confirmations in one round"); }
85  			@Override public McpToolOutcome call(Map<String,Object> arguments, BeanStore ctx) {
86  				var resume = ctx.getBean(McpMrtrResumeContext.class);
87  				if (resume.isEmpty()) {
88  					var requests = new LinkedHashMap<String,ElicitRequest>();
89  					requests.put("confirm", confirmQuestion());
90  					requests.put("reason", new ElicitRequest().setMessage("Why?")
91  						.setRequestedSchema(ElicitSchema.create().stringField("reason").build()));
92  					throw org.apache.juneau.rest.server.mcp.v20260728.ElicitationRequests.of(requests, "cont-multi");
93  				}
94  				var answers = org.apache.juneau.rest.server.mcp.v20260728.ElicitationResponses.all(resume.get());
95  				var confirmAnswer = answers.get("confirm");
96  				var reasonAnswer = answers.get("reason");
97  				var confirmValue = confirmAnswer.getContent() == null ? null : confirmAnswer.getContent().get("confirm");
98  				var reasonValue = reasonAnswer.getContent() == null ? null : reasonAnswer.getContent().get("reason");
99  				return McpToolOutcome.text("confirm:" + confirmAnswer.getAction() + "=" + confirmValue
100 					+ ",reason:" + reasonAnswer.getAction() + "=" + reasonValue
101 					+ ",cont:" + resume.get().continuation());
102 			}
103 		};
104 	}

133 	/**
134 	 * Decodes {@code raw} (a {@link McpClient#callRaw} "complete" result) into a {@link CallToolResult} and
135 	 * returns its first content block's text, replacing the raw {@code List}/{@code Map} cast chains every test
136 	 * below previously repeated inline.
137 	 */
138 	private static String firstText(Map<String,Object> raw) {
139 		var result = Json.to(Json.of(raw), CallToolResult.class);
140 		return ((TextContent) result.getContent().get(0)).getText();
141 	}

144 	// A: single-question pause -> resume -> complete, driven entirely through the client-side ElicitationRequests/ElicitationResponses.
145 	// =================================================================================================================
146 
147 	@Test void a01_singleQuestion_pauseResumeAccept_fullLoop() throws Exception {
148 		try (var client = clientBuilder(true).build()) {
149 			var paused = client.callRaw(McpMethods.TOOLS_CALL, new CallToolRequest().setName("confirm").setArguments(Map.of()));
150 			assertTrue(ElicitationRequests.isInputRequired(paused));
151 			var requests = ElicitationRequests.requests(paused);
152 			assertEquals("Proceed with deletion?", requests.get("confirm").getMessage());
153 			// The requestedSchema (built via ElicitSchema.booleanField("confirm").title("Confirm")) must survive
154 			// the wire intact, not just the message - otherwise a real client could not render the right input
155 			// control for the end user.
156 			var confirmProperty = (Map<?,?>) ((Map<?,?>) requests.get("confirm").getRequestedSchema().get("properties")).get("confirm");
157 			assertBean(confirmProperty, "type,title", "boolean,Confirm");
158 
159 			var answer = new ElicitResult().setAction(ElicitAction.ACCEPT).putContent("confirm", true);
160 			var completed = client.callRaw(McpMethods.TOOLS_CALL, new CallToolRequest().setName("confirm")
161 				.setRequestState(ElicitationRequests.requestState(paused))
162 				.setInputResponses(ElicitationResponses.toInputResponse("confirm", answer)));
163 			assertFalse(ElicitationRequests.isInputRequired(completed));
164 			assertEquals("complete", completed.get("resultType"));
165 			assertEquals("action:accept,confirm:true,cont:cont-1", firstText(completed));
166 		}
167 	}

173 	@Test void b01_multiQuestion_pauseResumeBoth_singleRoundTrip() throws Exception {
174 		try (var client = clientBuilder(true).build()) {
175 			var paused = client.callRaw(McpMethods.TOOLS_CALL, new CallToolRequest().setName("confirmTwo").setArguments(Map.of()));
176 			var requests = ElicitationRequests.requests(paused);
177 			assertEquals(2, requests.size());
178 			assertTrue(requests.containsKey("confirm"));
179 			assertTrue(requests.containsKey("reason"));
180 
181 			var answers = new LinkedHashMap<String,ElicitResult>();
182 			answers.put("confirm", new ElicitResult().setAction(ElicitAction.ACCEPT).putContent("confirm", true));
183 			answers.put("reason", new ElicitResult().setAction(ElicitAction.ACCEPT).putContent("reason", "cleanup"));
184 			var completed = client.callRaw(McpMethods.TOOLS_CALL, new CallToolRequest().setName("confirmTwo")
185 				.setRequestState(ElicitationRequests.requestState(paused))
186 				.setInputResponses(ElicitationResponses.toInputResponses(answers)));
187 			assertEquals("complete", completed.get("resultType"));
188 			assertEquals("confirm:accept=true,reason:accept=cleanup,cont:cont-multi", firstText(completed));
189 		}
