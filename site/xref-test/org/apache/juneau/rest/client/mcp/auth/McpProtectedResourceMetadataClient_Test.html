33  /**
34   * Tests for {@link McpProtectedResourceMetadataClient}: PRM parse/fetch and authorization-server discovery + issuer
35   * validation, exercised against an in-JVM stub.

37   * @since 10.0.0

39  @SuppressWarnings({
40  	"resource", // HttpServer held as test fixture; lifecycle managed by @AfterEach
41  	"java:S5976" // Each similarly-shaped fetch()-rejection test pins a distinct documented internal branch; merging
42  	// them would obscure which branch each one exercises.
43  })
44  class McpProtectedResourceMetadataClient_Test extends TestBase {
45  
46  	private HttpServer server;
47  	private volatile String issuerInDoc;
48  
49  	@BeforeEach
50  	void startServer() throws IOException {
51  		server = HttpServer.create(new InetSocketAddress(InetAddress.getLoopbackAddress(), 0), 0);
52  		server.createContext("/.well-known/openid-configuration", ex -> respond(ex, oidcMetadataJson()));
53  		server.createContext("/.well-known/oauth-protected-resource", ex -> respond(ex, prmJson()));
54  		server.start();
55  		issuerInDoc = base();
56  	}
57  
58  	@AfterEach
59  	void stopServer() {
60  		server.stop(0);
61  	}
62  
63  	private static void respond(HttpExchange ex, String body) throws IOException {
64  		var b = body.getBytes(StandardCharsets.UTF_8);
65  		ex.getResponseHeaders().add("Content-Type", "application/json");
66  		ex.sendResponseHeaders(200, b.length);
67  		try (var os = ex.getResponseBody()) {
68  			os.write(b);
69  		}
70  	}
71  
72  	private String base() {
73  		return "http://127.0.0.1:" + server.getAddress().getPort();
74  	}
75  
76  	private String prmJson() {
77  		return "{\"resource\":\"" + base() + "\","
78  			+ "\"authorization_servers\":[\"" + base() + "\"],"
79  			+ "\"scopes_supported\":[\"mcp:read\",\"mcp:write\"]}";
80  	}
81  
82  	private String oidcMetadataJson() {
83  		return "{\"issuer\":\"" + issuerInDoc + "\","
84  			+ "\"authorization_endpoint\":\"" + base() + "/authorize\","
85  			+ "\"token_endpoint\":\"" + base() + "/token\","
86  			+ "\"jwks_uri\":\"" + base() + "/jwks\","
87  			+ "\"response_types_supported\":[\"code\"],"
88  			+ "\"subject_types_supported\":[\"public\"],"
89  			+ "\"id_token_signing_alg_values_supported\":[\"RS256\"]}";
90  	}

136 	@Test void b03_httpRequestConfiguratorAppliedOnFetch() {
137 		var seenHeader = new AtomicReference<String>();
138 		server.createContext("/marker-check", ex -> {
139 			seenHeader.set(ex.getRequestHeaders().getFirst("X-Custom-Marker"));
140 			respond(ex, prmJson());
141 		});
142 		var client = McpProtectedResourceMetadataClient.create()
143 			.httpRequestConfigurator(req -> req.setHeader("X-Custom-Marker", "present"))
144 			.build();
145 		client.fetch(URI.create(base() + "/marker-check"));
146 		assertEquals("present", seenHeader.get());
147 	}
148 
149 	@Test void b04_httpTimeoutSetterValidatesAndApplies() {
150 		var b = McpProtectedResourceMetadataClient.create();
151 		var negativeTimeout = Duration.ofSeconds(-1);
152 		assertThrows(IllegalArgumentException.class, () -> b.httpTimeout(Duration.ZERO));
153 		assertThrows(IllegalArgumentException.class, () -> b.httpTimeout(negativeTimeout));
154 		var prm = McpProtectedResourceMetadataClient.create().httpTimeout(Duration.ofSeconds(5)).build()
155 			.fetch(URI.create(base() + "/.well-known/oauth-protected-resource"));
156 		assertEquals(URI.create(base()), prm.resource());
157 	}

159 	@Test void b05_fetchIoExceptionWrapped() {
160 		// A connection refused (nothing bound on this loopback port) surfaces as an IOException that fetch(...) must
161 		// wrap as an McpAuthException, distinct from the HTTP-status-based b02 case above. Using https:// here also
162 		// exercises requireSecure(...)'s own "https" scheme-check branch directly (short-circuiting before even
163 		// consulting isLoopback(...), unlike every other http://127.0.0.1 test in this class).
164 		var client = McpProtectedResourceMetadataClient.create().build();
165 		var url = URI.create("https://127.0.0.1:1/prm");
166 		var e = assertThrows(McpAuthException.class, () -> client.fetch(url));
167 		assertTrue(e.getMessage().contains("fetch failed"), e::getMessage);
168 	}

213 	@Test void c05b_discoverAuthorizationServerAppliesHttpRequestConfigurator() {
214 		// Covers the non-null arm of discoverAuthorizationServer(...)'s ternary threading httpRequestConfigurator
215 		// into the OidcDiscoveryClient, distinct from every other discovery test above (which builds with none set,
216 		// so it always default to the no-op lambda).
217 		var seenHeader = new AtomicReference<String>();
218 		server.removeContext("/.well-known/openid-configuration");
219 		server.createContext("/.well-known/openid-configuration", ex -> {
220 			seenHeader.set(ex.getRequestHeaders().getFirst("X-Custom-Marker"));
221 			respond(ex, oidcMetadataJson());
222 		});
223 		var client = McpProtectedResourceMetadataClient.create()
224 			.httpRequestConfigurator(req -> req.setHeader("X-Custom-Marker", "present"))
225 			.build();
226 		var prm = client.parse(prmJson(), null);
227 		client.discoverAuthorizationServer(prm);
228 		assertEquals("present", seenHeader.get());
229 	}

231 	@Test void c06_discoverViaRfc8414Only() {
232 		// H4: with only the RFC 8414 oauth-authorization-server endpoint available (no OIDC openid-configuration),
233 		// discovery must still succeed via the AuthorizationServerMetadata path.
234 		server.removeContext("/.well-known/openid-configuration");
235 		server.createContext("/.well-known/oauth-authorization-server", ex -> respond(ex, oidcMetadataJson()));
236 		var client = McpProtectedResourceMetadataClient.create().build();
237 		var prm = client.parse(prmJson(), null);
238 		var md = client.discoverAuthorizationServer(prm);
239 		assertEquals(URI.create(base()), md.issuer());

255 	@Test void d02_allowInsecureHttpOptOut() {
256 		// M8: the opt-out disables the scheme check (loopback http already works; this asserts the flag path builds
257 		// and fetches over the loopback stub).
258 		var prm = McpProtectedResourceMetadataClient.create().allowInsecureHttp(true).build()
259 			.fetch(URI.create(base() + "/.well-known/oauth-protected-resource"));
260 		assertEquals(URI.create(base()), prm.resource());
261 	}

263 	@Test void d03_opaqueUriWithNoHostRejected() {
264 		// isLoopback(...)'s host==null branch: an opaque (no-authority) URI has no host at all, so it can't be
265 		// loopback either, and this http-scheme URI isn't allowInsecureHttp'd — rejected.
266 		var client = McpProtectedResourceMetadataClient.create().build();
267 		var url = URI.create("http:opaque-no-host");
268 		var e = assertThrows(McpAuthException.class, () -> client.fetch(url));
269 		assertTrue(e.getMessage().contains("https"), e::getMessage);
270 	}
271 
272 	@Test void d04_localhostHostnameTreatedAsLoopback() {
273 		// isLoopback(...)'s "localhost" branch: passes the scheme check without allowInsecureHttp, then fails at the
274 		// (unreachable) network hop — proving it got past requireSecure(...) rather than being rejected for scheme.
275 		var client = McpProtectedResourceMetadataClient.create().build();
276 		var url = URI.create("http://localhost:1/prm");
277 		var e = assertThrows(McpAuthException.class, () -> client.fetch(url));
278 		assertFalse(e.getMessage().contains("https"), e::getMessage);
279 	}
280 
281 	@Test void d05_bracketedIpv6LoopbackTreatedAsLoopback() {
282 		// isLoopback(...)'s "[::1]" branch (java.net.URI#getHost() always returns the IPv6 literal WITH brackets —
283 		// see the bare "::1" HTT marker on isLoopback(...) itself for why the unbracketed form is unreachable).
284 		var client = McpProtectedResourceMetadataClient.create().build();
285 		var url = URI.create("http://[::1]:1/prm");
286 		var e = assertThrows(McpAuthException.class, () -> client.fetch(url));
287 		assertFalse(e.getMessage().contains("https"), e::getMessage);
288 	}
