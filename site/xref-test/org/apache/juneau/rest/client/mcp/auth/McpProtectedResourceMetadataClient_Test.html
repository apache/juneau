41  	private HttpServer server;
42  	private final AtomicReference<String> issuerInDoc = new AtomicReference<>();
43  
44  	@BeforeEach
45  	void startServer() throws IOException {
46  		server = HttpServer.create(new InetSocketAddress(InetAddress.getLoopbackAddress(), 0), 0);
47  		server.createContext("/.well-known/openid-configuration", ex -> respond(ex, oidcMetadataJson()));
48  		server.createContext("/.well-known/oauth-protected-resource", ex -> respond(ex, prmJson()));
49  		server.start();
50  		issuerInDoc.set(base());
51  	}
52  
53  	@AfterEach
54  	void stopServer() {
55  		server.stop(0);
56  	}

58  	private static void respond(HttpExchange ex, String body) throws IOException {
59  		var b = body.getBytes(StandardCharsets.UTF_8);
60  		ex.getResponseHeaders().add("Content-Type", "application/json");
61  		ex.sendResponseHeaders(200, b.length);
62  		try (var os = ex.getResponseBody()) {
63  			os.write(b);
64  		}
65  	}
66  
67  	private String base() {
68  		return "http://127.0.0.1:" + server.getAddress().getPort();
69  	}
70  
71  	private String prmJson() {
72  		return "{\"resource\":\"" + base() + "\","
73  			+ "\"authorization_servers\":[\"" + base() + "\"],"
74  			+ "\"scopes_supported\":[\"mcp:read\",\"mcp:write\"]}";
75  	}
76  
77  	private String oidcMetadataJson() {
78  		return "{\"issuer\":\"" + issuerInDoc + "\","
79  			+ "\"authorization_endpoint\":\"" + base() + "/authorize\","
80  			+ "\"token_endpoint\":\"" + base() + "/token\","
81  			+ "\"jwks_uri\":\"" + base() + "/jwks\","
82  			+ "\"response_types_supported\":[\"code\"],"
83  			+ "\"subject_types_supported\":[\"public\"],"
84  			+ "\"id_token_signing_alg_values_supported\":[\"RS256\"]}";
85  	}

88  	private McpProtectedResourceMetadataClient.Builder local() {
89  		return McpProtectedResourceMetadataClient.create().allowPrivateUrls(true).expectedResource(URI.create(base()));

185 		server.removeContext("/.well-known/openid-configuration");
186 		server.createContext("/.well-known/oauth-authorization-server", ex -> respond(ex, oidcMetadataJson()));

189 		var md = client.discoverAuthorizationServer(prm);

218 	private static int closedLoopbackPort() throws IOException {
219 		try (var s = new ServerSocket(0, 0, InetAddress.getLoopbackAddress())) {
220 			return s.getLocalPort();
221 		}
222 	}

224 	@Test void e01_httpTimeoutMustBePositive() {
225 		var b = McpProtectedResourceMetadataClient.create();
226 		assertThrows(IllegalArgumentException.class, () -> b.httpTimeout(Duration.ZERO));
227 		var negative = Duration.ofSeconds(-1);
228 		assertThrows(IllegalArgumentException.class, () -> b.httpTimeout(negative));
229 		assertDoesNotThrow(() -> b.httpTimeout(Duration.ofSeconds(3)));
230 	}

232 	@Test void e02_requestConfiguratorAppliedToFetchAndDiscovery() {
233 		var calls = new AtomicInteger();
234 		var client = local()
235 			.httpTimeout(Duration.ofSeconds(5))
236 			.httpRequestConfigurator(r -> calls.incrementAndGet())
237 			.build();
238 		var prm = client.fetch(URI.create(base() + "/.well-known/oauth-protected-resource"));
239 		assertEquals(1, calls.get(), "configurator must run for the PRM fetch");
240 		client.discoverAuthorizationServer(prm);
241 		assertTrue(calls.get() > 1, "configurator must also run for authorization-server discovery");
242 	}
