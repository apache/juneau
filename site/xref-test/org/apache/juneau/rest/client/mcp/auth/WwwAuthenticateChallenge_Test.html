17  package org.apache.juneau.rest.client.mcp.auth;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.net.*;
22  
23  import org.apache.juneau.*;
24  import org.junit.jupiter.api.*;
25  
26  /**
27   * Tests for {@link WwwAuthenticateChallenge} (RFC 6750 / RFC 9728 challenge parsing).
28   *
29   * @since 10.0.0
30   */
31  class WwwAuthenticateChallenge_Test extends TestBase {
32  
33  	@Test void a01_parsesResourceMetadataPointer() {
34  		var c = WwwAuthenticateChallenge.parse(
35  			"Bearer realm=\"mcp\", resource_metadata=\"https://mcp.example.com/.well-known/oauth-protected-resource\"").orElseThrow();
36  		assertTrue(c.isBearer());
37  		assertEquals("Bearer", c.scheme());
38  		assertEquals(URI.create("https://mcp.example.com/.well-known/oauth-protected-resource"), c.resourceMetadata().orElseThrow());
39  	}
40  
41  	@Test void a02_parsesScopeAndError() {
42  		var c = WwwAuthenticateChallenge.parse("Bearer error=\"insufficient_scope\", scope=\"read write admin\"").orElseThrow();
43  		assertEquals("insufficient_scope", c.error().orElseThrow());
44  		assertEquals(java.util.List.of("read", "write", "admin"), java.util.List.copyOf(c.scopes()));
45  	}
46  
47  	@Test void a03_paramKeysAreCaseInsensitive() {
48  		var c = WwwAuthenticateChallenge.parse("Bearer Resource_Metadata=\"https://x/prm\"").orElseThrow();
49  		assertEquals(URI.create("https://x/prm"), c.resourceMetadata().orElseThrow());
50  		assertEquals("https://x/prm", c.parameter("RESOURCE_METADATA").orElseThrow());
51  	}
52  
53  	@Test void a04_bareTokenValuesAccepted() {
54  		var c = WwwAuthenticateChallenge.parse("Bearer error=invalid_token, scope=read").orElseThrow();
55  		assertEquals("invalid_token", c.error().orElseThrow());
56  		assertEquals(java.util.Set.of("read"), c.scopes());
57  	}
58  
59  	@Test void b01_nullReturnsEmpty() {
60  		assertTrue(WwwAuthenticateChallenge.parse(null).isEmpty());
61  	}
62  
63  	@Test void b02_blankReturnsEmpty() {
64  		assertTrue(WwwAuthenticateChallenge.parse("   ").isEmpty());
65  	}
66  
67  	@Test void b03_schemeOnlyHasNoParams() {
68  		var c = WwwAuthenticateChallenge.parse("Bearer").orElseThrow();
69  		assertTrue(c.isBearer());
70  		assertTrue(c.parameters().isEmpty());
71  		assertTrue(c.resourceMetadata().isEmpty());
72  		assertTrue(c.scopes().isEmpty());
73  		assertTrue(c.error().isEmpty());
74  	}
75  
76  	@Test void b04_nonBearerScheme() {
77  		var c = WwwAuthenticateChallenge.parse("Basic realm=\"x\"").orElseThrow();
78  		assertFalse(c.isBearer());
79  		assertEquals("Basic", c.scheme());
80  	}
81  
82  	@Test void b05_missingResourceMetadataIsEmpty() {
83  		var c = WwwAuthenticateChallenge.parse("Bearer realm=\"mcp\"").orElseThrow();
84  		assertTrue(c.resourceMetadata().isEmpty());
85  	}
86  
87  	@Test void b06_blankResourceMetadataValueIsEmpty() {
88  		// Distinct from b05 (param absent entirely): the param IS present here, but its value is blank.
89  		var c = WwwAuthenticateChallenge.parse("Bearer resource_metadata=\"  \"").orElseThrow();
90  		assertTrue(c.resourceMetadata().isEmpty());
91  	}
92  
93  	@Test void b07_blankScopeValueIsEmptySet() {
94  		// Distinct from b03 (param absent entirely): the param IS present here, but its value is blank.
95  		var c = WwwAuthenticateChallenge.parse("Bearer scope=\"  \"").orElseThrow();
96  		assertTrue(c.scopes().isEmpty());
97  	}
98  
99  	@Test void c01_bareTokenParamWithNoEqualsSignIsSkipped() {
100 		// parseParam(...)'s "bare token, no '=value'" arm (RFC 7235's token68 form is only legal as the whole
101 		// credentials in some schemes, but a real server sending a stray bare token among auth-params must not
102 		// break parsing of the params that follow it).
103 		var c = WwwAuthenticateChallenge.parse("Bearer sometoken68value, error=\"invalid_token\"").orElseThrow();
104 		assertEquals("invalid_token", c.error().orElseThrow());
105 		assertFalse(c.parameters().containsKey("sometoken68value"));
106 	}
107 
108 	@Test void c02_bareTokenAtEndOfStringWithNoTrailingCommaIsSkipped() {
109 		// Distinct from c01 (which ends on a comma): here the bare token runs all the way to the end of the string
110 		// with neither a '=' nor a trailing ',', exercising parseParam(...)'s "i >= n" arm instead of its
111 		// "hit a comma" arm.
112 		var c = WwwAuthenticateChallenge.parse("Bearer justatoken").orElseThrow();
113 		assertTrue(c.parameters().isEmpty());
114 	}
115 
116 	@Test void c03_equalsWithNoValueAtEndOfStringYieldsEmptyValue() {
117 		// parseValue(...)'s "i >= n" arm: a key immediately followed by '=' and then nothing (end of string).
118 		var c = WwwAuthenticateChallenge.parse("Bearer key=").orElseThrow();
119 		assertEquals("", c.parameter("key").orElseThrow());
120 	}
121 
122 	@Test void c04_unterminatedQuotedValueConsumesToEndOfString() {
123 		// parseQuotedValue(...)'s "ran off the end without finding a closing quote" arm; the value is still
124 		// whatever was scanned, and parsing doesn't throw or infinite-loop.
125 		var c = WwwAuthenticateChallenge.parse("Bearer key=\"unterminated").orElseThrow();
126 		assertEquals("unterminated", c.parameter("key").orElseThrow());
127 	}
128 
129 	@Test void c05_escapedQuoteInsideQuotedValueIsUnescaped() {
130 		// parseQuotedValue(...)'s backslash-escape arm: an escaped quote inside the value must not be treated as
131 		// the closing quote.
132 		var c = WwwAuthenticateChallenge.parse("Bearer key=\"a\\\"b\"").orElseThrow();
133 		assertEquals("a\"b", c.parameter("key").orElseThrow());
134 	}
135 
136 	@Test void c07_bareEqualsWithEmptyKeyIsSkipped() {
137 		// parseParam(...)'s "key is present but empty" arm (distinct from c01/c02's "no key at all"): a stray
138 		// leading '=' scans to an empty key substring, which must be skipped rather than stored under "".
139 		var c = WwwAuthenticateChallenge.parse("Bearer =foo, error=bar").orElseThrow();
140 		assertFalse(c.parameters().containsKey(""));
141 		assertEquals("bar", c.error().orElseThrow());
142 	}
143 
144 	@Test void c08_malformedResourceMetadataUriIsEmpty() {
145 		// resourceMetadata()'s catch(IllegalArgumentException) arm: an unescaped space makes URI.create(...) throw.
146 		var c = WwwAuthenticateChallenge.parse("Bearer resource_metadata=\"http://example.com/a b\"").orElseThrow();
147 		assertTrue(c.resourceMetadata().isEmpty());
148 	}
149 
150 	@Test void c06_trailingBackslashAtEndOfUnterminatedQuotedValue() {
151 		// parseQuotedValue(...)'s escape-check guard ("i + 1 < n"): a backslash as the very last character, with no
152 		// following character to escape and no closing quote, must not index out of bounds.
153 		var c = WwwAuthenticateChallenge.parse("Bearer key=\"abc\\").orElseThrow();
154 		assertEquals("abc\\", c.parameter("key").orElseThrow());
155 	}
