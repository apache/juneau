47  	private static OidcMetadata as(URI issuer, boolean withRegEndpoint) {
48  		Map<String,Object> extras = withRegEndpoint
49  			? Map.of("registration_endpoint", issuer + "/register")
50  			: Map.of();
51  		return new OidcMetadata(issuer, URI.create(issuer + "/token"), URI.create(issuer + "/auth"),
52  			null, null, null, null, Set.of(), extras);
53  	}
54  
55  	private static McpClientRegistration reg(URI issuer, String clientId) {
56  		return new McpClientRegistration(clientId, Optional.of("secret-" + clientId), Optional.empty(), Optional.empty(),
57  			Optional.empty(), issuer, List.of(URI.create("http://127.0.0.1/callback")), McpApplicationType.NATIVE, Map.of());
58  	}

60  	/** A counting DCR stub that stamps the issuer of the AS it is asked to register with. */
61  	private static AtomicInteger counter;
62  	private static Function<OidcMetadata,McpClientRegistration> stubDcr() {
63  		counter = new AtomicInteger();
64  		return md -> {
65  			var n = counter.incrementAndGet();
66  			return reg(md.issuer(), "dcr-" + n);
67  		};
68  	}

70  	@Test void a01_onDemandRegistersEveryCall() {
71  		var m = McpClientRegistrationManager.create().registrarFunction(stubDcr()).build();
72  		m.resolve(as(A, true));
73  		m.resolve(as(A, true));
74  		assertEquals(2, counter.get(), "on-demand mode (no store) must register on every resolve");
75  	}

77  	@Test void a02_storeHitReused() {
78  		var store = new InMemoryMcpClientRegistrationStore();
79  		var m = McpClientRegistrationManager.create().store(store).registrarFunction(stubDcr()).build();
80  		var first = m.resolve(as(A, true));
81  		var second = m.resolve(as(A, true));
82  		assertEquals(1, counter.get(), "store-hit must be reused without a second DCR round-trip");
83  		assertEquals(first.clientId(), second.clientId());
84  	}

86  	@Test void a03_migrationReregistersWithNewIssuer() {
87  		var store = new InMemoryMcpClientRegistrationStore();
88  		var m = McpClientRegistrationManager.create().store(store).registrarFunction(stubDcr()).build();
89  		var forA = m.resolve(as(A, true));
90  		var forB = m.resolve(as(B, true));
91  		assertEquals(2, counter.get(), "migration to a new issuer must trigger re-registration");
92  		assertEquals(A, forA.issuer());
93  		assertEquals(B, forB.issuer());
94  		// A's entry is untouched (never reused for B).
95  		assertEquals(forA.clientId(), store.find(A).orElseThrow().clientId());
96  		assertEquals(forB.clientId(), store.find(B).orElseThrow().clientId());
97  	}

141 		// LOW: when both a store and matching pre-registered credentials are configured, the store must still be

143 		var store = new InMemoryMcpClientRegistrationStore();

188 	// McpDynamicClientRegistrar against an in-JVM loopback stub, rather than the stubDcr() override used above.

190 	private static HttpServer loopbackServer(HttpHandler handler) throws IOException {
191 		var server = HttpServer.create(new InetSocketAddress(InetAddress.getLoopbackAddress(), 0), 0);
192 		server.createContext("/register", handler);
193 		server.start();
194 		return server;
195 	}
196 
197 	private static void respond(HttpExchange ex, String body) throws IOException {
198 		var b = body.getBytes(StandardCharsets.UTF_8);
199 		ex.getResponseHeaders().add("Content-Type", "application/json");
200 		ex.sendResponseHeaders(201, b.length);
201 		try (var os = ex.getResponseBody()) {
202 			os.write(b);
203 		}
204 	}

206 	private static Map<String,Object> asExtrasFor(HttpServer server) {
207 		return Map.of("registration_endpoint", "http://127.0.0.1:" + server.getAddress().getPort() + "/register");
208 	}

211 		"resource" // HttpServer is stopped in the finally block below.

213 	@Test void e01_realDcrRoundTripAppliesAllManagerSettings() throws IOException {
214 		var configuratorCalled = Flag.create();
215 		var server = loopbackServer(ex -> respond(ex,
216 			"{\"client_id\":\"dcr-client\",\"client_secret\":\"sek\",\"redirect_uris\":[\"http://127.0.0.1/callback\"],"
217 				+ "\"application_type\":\"web\",\"token_endpoint_auth_method\":\"client_secret_basic\"}"));
218 		try {
219 			var as = new OidcMetadata(A, URI.create(A + "/token"), URI.create(A + "/auth"), null, null, null, null,
220 				Set.of(), asExtrasFor(server));
221 			var m = McpClientRegistrationManager.create()
222 				.applicationType(McpApplicationType.WEB)
223 				.redirectUris(List.of(URI.create("http://127.0.0.1/callback")))
224 				.scope("mcp:read", "mcp:write")
225 				.confidential(true)
226 				.clientName("test-client")
227 				.initialAccessToken("iat-tok")
228 				.httpTimeout(Duration.ofSeconds(5))
229 				.httpRequestConfigurator(req -> configuratorCalled.set())
230 				.build();
231 			var reg = m.resolve(as);
232 			assertEquals("dcr-client", reg.clientId());
233 			assertEquals(McpApplicationType.WEB, reg.applicationType());
234 			assertTrue(reg.isConfidential());
235 			assertTrue(configuratorCalled.isSet(), "httpRequestConfigurator must reach the underlying registrar");
236 		} finally {
237 			server.stop(0);
238 		}
239 	}
