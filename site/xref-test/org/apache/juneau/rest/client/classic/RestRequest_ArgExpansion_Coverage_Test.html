33  /**
34   * Exercises the package-private {@code RestRequest#formDataArg}/{@code #headerArg}/{@code #pathArg}/
35   * {@code #queryArg} dispatch helpers (invoked by {@link RestClient} while processing {@code @FormData}/
36   * {@code @Header}/{@code @Path}/{@code @Query}-annotated remote-proxy method arguments) directly -- same-package
37   * access, no visibility change needed. Each helper multiplexes on the runtime type of the supplied value
38   * (castable single value, {@code PartList}/{@code HeaderList}, {@code Collection}, array, {@code Map}, bean, or
39   * an unrecognized "custom" value) whenever the part/header name is empty, {@code "*"}, or the value is itself a
40   * list/array type; otherwise it takes the single-value path. None of the {@code juneau-integration-tests} remote-
41   * proxy suites exercise every combination, so this fills in the gaps directly.
42   */
43  @SuppressWarnings({
44  	"resource" // req() returns a RestRequest whose ownership is transferred to the caller, who is responsible for closing it; some assigned-but-discarded 'r' locals on exception paths are the same already-tracked instance; Eclipse JDT's @Owning warning is by design.
45  })
46  // SEPARATE_THREAD: a socket read blocked on a non-responding peer ignores interrupts, so SAME_THREAD would still hang.
47  @Timeout(value = 60, threadMode = Timeout.ThreadMode.SEPARATE_THREAD)
48  class RestRequest_ArgExpansion_Coverage_Test {

50  	private static HttpServer server;
51  	private static ExecutorService executor;
52  	private static int port;
53  
54  	@BeforeAll
55  	static void startServer() throws IOException {
56  		// Bind to (and connect via) the explicit IPv4 loopback, not the wildcard address + "localhost": a wildcard bind
57  		// can be handed an ephemeral port that another process already holds on 127.0.0.1 (SO_REUSEADDR), and that
58  		// more-specific listener then wins every "localhost" connection -- the client hangs on its response.
59  		server = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
60  		port = server.getAddress().getPort();
61  		// Without an explicit executor, exchanges run on HttpServer's single internal dispatch thread, which
62  		// is a bottleneck under -T1C parallel load. (The "server failed to respond" flake once blamed on starvation was most likely
63  		// the wildcard-bind port shadowing fixed by binding 127.0.0.1 above; the executor is kept as harmless.)
64  		executor = Executors.newCachedThreadPool();
65  		server.setExecutor(executor);
66  		server.createContext("/echo", exchange -> {
67  			exchange.getRequestBody().readAllBytes();
68  			exchange.sendResponseHeaders(200, -1);
69  			exchange.close();
70  		});
71  		server.start();
72  	}
73  
74  	@AfterAll
75  	static void stopServer() {
76  		if (server != null)
77  			server.stop(0);
78  		if (executor != null)
79  			executor.shutdownNow();
80  	}
81  
82  	private static String url() {
83  		return "http://127.0.0.1:" + port + "/echo";
84  	}

86  	private static RestRequest req() {
87  		try {
88  			return RestClient.create().build().get(url());
89  		} catch (RestCallException e) {
90  			throw new RuntimeException(e);
91  		}
92  	}
