35  /**
36   * Exercises the private {@code ResponseContent#asCursor(ClassMeta)} helper (reached via
37   * {@code ResponseContent#as(Class)} for a {@link RecordReader}/{@link TokenReader} target type) -- entirely
38   * uncovered previously, since none of the {@code juneau-integration-tests} suites request a cursor return type.
39   * Mirrors the equivalent non-classic {@code ResponseBody_Cursor_Test}.
40   */
41  @SuppressWarnings({
42  	"resource" // jsonContent() returns a ResponseContent while the underlying client (and its still-unconsumed response) must stay open for the caller to read the returned cursor; Eclipse JDT's @Owning warning is by design.
43  })
44  // SEPARATE_THREAD: a socket read blocked on a non-responding peer ignores interrupts, so SAME_THREAD would still hang.
45  @Timeout(value = 60, threadMode = Timeout.ThreadMode.SEPARATE_THREAD)
46  class ResponseContent_Cursor_Test {
47  
48  	public static class Bean {
49  		public String name;
50  		public int age;
51  	}
52  
53  	private static HttpServer server;
54  	private static ExecutorService executor;
55  	private static int port;
56  	private static volatile byte[] responseBody = new byte[0];
57  	private static volatile String responseContentType = "application/json";
58  	private static volatile int responseStatus = 200;
59  
60  	@BeforeAll
61  	static void startServer() throws IOException {
62  		// Bind to (and connect via) the explicit IPv4 loopback, not the wildcard address + "localhost": a wildcard bind
63  		// can be handed an ephemeral port that another process already holds on 127.0.0.1 (SO_REUSEADDR), and that
64  		// more-specific listener then wins every "localhost" connection -- the client hangs on its response.
65  		server = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
66  		port = server.getAddress().getPort();
67  		// Without an explicit executor, exchanges run on HttpServer's single internal dispatch thread, which
68  		// is a bottleneck under -T1C parallel load. (The "server failed to respond" flake once blamed on starvation was most likely
69  		// the wildcard-bind port shadowing fixed by binding 127.0.0.1 above; the executor is kept as harmless.)
70  		executor = Executors.newCachedThreadPool();
71  		server.setExecutor(executor);
72  		server.createContext("/echo", exchange -> {
73  			exchange.getRequestBody().readAllBytes();
74  			if (responseContentType != null)
75  				exchange.getResponseHeaders().add("Content-Type", responseContentType);
76  			if (responseBody.length == 0 && responseStatus == 204) {
77  				exchange.sendResponseHeaders(204, -1);
78  			} else {
79  				exchange.sendResponseHeaders(responseStatus, responseBody.length);
80  				exchange.getResponseBody().write(responseBody);
81  			}
82  			exchange.close();
83  		});
84  		server.start();
85  	}
86  
87  	@AfterAll
88  	static void stopServer() {
89  		if (server != null)
90  			server.stop(0);
91  		if (executor != null)
92  			executor.shutdownNow();
93  	}
94  
95  	private static String url() {
96  		return "http://127.0.0.1:" + port + "/echo";
97  	}
98  
99  	private static ResponseContent jsonContent(String json) throws Exception {
100 		responseBody = json.getBytes(StandardCharsets.UTF_8);
101 		responseContentType = "application/json";
102 		responseStatus = 200;
103 		var client = RestClient.create().json().build();
104 		return client.get(url()).run().getContent();
105 	}

147 		// JsonParser produces a JsonTokenReader, which is not assignable to JsonlTokenReader.
148 		var content = jsonContent("{\"name\":\"x\",\"age\":1}").parser(JsonParser.DEFAULT);

153 	@Test void b02_tokenCursorRequestedFromRecordOnlyParser_unsupported() throws Exception {
154 		// IniParser implements RecordReadable but not TokenReadable, so a TokenReader cursor is unsupported
155 		// (TokenReadable extends RecordReadable, so the reverse combination -- RecordReader from a TokenReadable
156 		// parser -- is always supported and cannot exercise this branch).
157 		responseBody = "a=1".getBytes(StandardCharsets.UTF_8);
158 		responseContentType = "text/ini";
159 		responseStatus = 200;
160 		try (var client = RestClient.create().build()) {
161 			try (var req = client.get(url()); var res = req.run()) {
162 				var content = res.getContent().parser(org.apache.juneau.marshall.ini.IniParser.DEFAULT);
163 				var e = assertThrows(RestCallException.class, () -> content.as(TokenReader.class));
164 				assertTrue(e.getMessage().contains("token-reader"), "Unexpected message: " + e.getMessage());
165 			}
166 		}
167 	}

184 	@Test void b02c_ioExceptionOpeningStream_wrappedAsRestCallException() throws Exception {
185 		// A RestCallInterceptor#onConnect substitution of the live entity's getContent() (see the analogous
186 		// technique in ResponseContent_Coverage_Test) makes asInputStream() throw IOException, which asCursor()'s
187 		// own try block must wrap as a RestCallException rather than letting it escape unwrapped.
188 		var itcp = new BasicRestCallInterceptor() {
189 			@Override
190 			public void onConnect(RestRequest req, RestResponse res) {
191 				res.asHttpResponse().setEntity(new HttpEntity() {
192 					@Override public boolean isRepeatable() { return false; }
193 					@Override public boolean isChunked() { return false; }
194 					@Override public long getContentLength() { return -1; }
195 					@Override public Header getContentType() { return null; }
196 					@Override public Header getContentEncoding() { return null; }
197 					@Override public InputStream getContent() { throw new UnsupportedOperationException("Simulated: entity content not available."); }
198 					@Override public void writeTo(OutputStream outstream) { throw new UnsupportedOperationException("Not used by this test."); }
199 					@Override public boolean isStreaming() { return false; }
200 					@Override public void consumeContent() { /* no-op */ }
201 				});
202 			}
203 		};
204 		responseBody = "{}".getBytes(StandardCharsets.UTF_8);
205 		responseContentType = "application/json";
206 		responseStatus = 200;
207 		try (var client = RestClient.create().interceptors(itcp).build()) {
208 			try (var req = client.get(url()); var res = req.run()) {
209 				var content = res.getContent().parser(JsonParser.DEFAULT);
210 				var e = assertThrows(RestCallException.class, () -> content.as(TokenReader.class));
211 				assertTrue(e.getMessage().contains("Could not open cursor"), "Unexpected message: " + e.getMessage());
212 			}
213 		}
214 	}
