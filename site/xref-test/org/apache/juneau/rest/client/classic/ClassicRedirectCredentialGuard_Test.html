17  package org.apache.juneau.rest.client.classic;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.apache.http.*;
22  import org.apache.http.message.*;
23  import org.apache.http.protocol.*;
24  import org.junit.jupiter.api.*;
25  
26  /**
27   * Direct unit tests for {@link ClassicRedirectCredentialGuard}'s edge cases that end-to-end redirect tests
28   * ({@link RestClient_RedirectCredentials_Test}) never reach &mdash; a real HttpClient exchange always
29   * supplies a resolvable {@link HttpHost} target, so a missing/unparsable target host is only reachable
30   * by driving {@link ClassicRedirectCredentialGuard#process(HttpRequest, HttpContext)} directly against a
31   * hand-built {@link HttpContext}. In the same package as the guard since it (and its members) are
32   * package-private.
33   */
34  class ClassicRedirectCredentialGuard_Test {
35  
36  	private final ClassicRedirectCredentialGuard guard = new ClassicRedirectCredentialGuard();
37  
38  	@Test
39  	void a01_noTargetHostInContext_isNoOp() {
40  		var context = new BasicHttpContext();
41  		var request = new BasicHttpRequest("GET", "/");
42  		request.addHeader("Authorization", "Bearer secret");
43  		assertDoesNotThrow(() -> guard.process(request, context));
44  		assertNotNull(request.getFirstHeader("Authorization"), "with no target host to compare against, credentials must be left untouched");
45  	}
46  
47  	@Test
48  	void a02_targetHostToURIThrows_originUnresolvable_isNoOp() {
49  		// "|" is not a legal URI character but passes HttpHost's blanks-only hostname validation, so
50  		// HttpHost.toURI() produces a string that URI.create(...) rejects with IllegalArgumentException —
51  		// exactly the case originOf(...) guards against.
52  		var context = new BasicHttpContext();
53  		HttpCoreContext.adapt(context).setTargetHost(new HttpHost("exa|mple.com", 80, "http"));
54  		var request = new BasicHttpRequest("GET", "/");
55  		request.addHeader("Authorization", "Bearer secret");
56  		assertDoesNotThrow(() -> guard.process(request, context));
57  		assertNotNull(request.getFirstHeader("Authorization"), "an unresolvable origin must not strip credentials");
58  	}
59  
60  	@Test
61  	void a03_firstRequestInExchange_recordsOriginWithoutStrippingCredentials() {
62  		var context = new BasicHttpContext();
63  		HttpCoreContext.adapt(context).setTargetHost(new HttpHost("example.com", 80, "http"));
64  		var request = new BasicHttpRequest("GET", "/");
65  		request.addHeader("Authorization", "Bearer secret");
66  		guard.process(request, context);
67  		assertNotNull(request.getFirstHeader("Authorization"), "the first request of an exchange only records the origin; nothing to compare against yet");
68  	}
69  
70  	@Test
71  	void a04_underscoreHostThenOtherHost_failsClosedAndStrips() {
72  		// java.net.URI reports a null host for an underscore host name; comparing it must not throw and must not forward credentials.
73  		var context = new BasicHttpContext();
74  		var core = HttpCoreContext.adapt(context);
75  		core.setTargetHost(new HttpHost("my_svc", 80, "http"));
76  		var first = new BasicHttpRequest("GET", "/");
77  		first.addHeader("Authorization", "Bearer secret");
78  		assertDoesNotThrow(() -> guard.process(first, context));
79  		core.setTargetHost(new HttpHost("other.example.com", 80, "http"));
80  		var second = new BasicHttpRequest("GET", "/");
81  		second.addHeader("Authorization", "Bearer secret");
82  		assertDoesNotThrow(() -> guard.process(second, context));
83  		assertNull(second.getFirstHeader("Authorization"));
84  	}
85  
86  	@Test
87  	void a05_otherHostThenUnderscoreHost_failsClosedAndStrips() {
88  		var context = new BasicHttpContext();
89  		var core = HttpCoreContext.adapt(context);
90  		core.setTargetHost(new HttpHost("example.com", 80, "http"));
91  		guard.process(new BasicHttpRequest("GET", "/"), context);
92  		core.setTargetHost(new HttpHost("my_svc", 80, "http"));
93  		var second = new BasicHttpRequest("GET", "/");
94  		second.addHeader("Authorization", "Bearer secret");
95  		assertDoesNotThrow(() -> guard.process(second, context));
96  		assertNull(second.getFirstHeader("Authorization"));
97  	}
98  }
