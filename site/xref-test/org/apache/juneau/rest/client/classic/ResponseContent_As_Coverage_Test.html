32  /**
33   * Exercises the {@code HttpResource}/{@code BasicResource} target-type redirect branch of
34   * {@link ResponseContent#as(org.apache.juneau.marshall.httppart.ClassMeta)} -- retargets the requested type to
35   * {@link StreamResource} before the generic {@code HttpResponse}-arg-constructor lookup, since neither
36   * {@code HttpResource} nor {@code BasicResource} is directly instantiable.
37   */
38  @SuppressWarnings({
39  	"resource" // Closeable resources in tests are intentionally unassigned; closing is handled by test infrastructure.
40  })
41  // SEPARATE_THREAD: a socket read blocked on a non-responding peer ignores interrupts, so SAME_THREAD would still hang.
42  @Timeout(value = 60, threadMode = Timeout.ThreadMode.SEPARATE_THREAD)
43  class ResponseContent_As_Coverage_Test {
44  
45  	private static HttpServer server;
46  	private static ExecutorService executor;
47  	private static int port;
48  
49  	@BeforeAll
50  	static void startServer() throws IOException {
51  		// Bind to (and connect via) the explicit IPv4 loopback, not the wildcard address + "localhost": a wildcard bind
52  		// can be handed an ephemeral port that another process already holds on 127.0.0.1 (SO_REUSEADDR), and that
53  		// more-specific listener then wins every "localhost" connection -- the client hangs on its response.
54  		server = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
55  		port = server.getAddress().getPort();
56  		// Without an explicit executor, exchanges run on HttpServer's single internal dispatch thread, which
57  		// is a bottleneck under -T1C parallel load. (The "server failed to respond" flake once blamed on starvation was most likely
58  		// the wildcard-bind port shadowing fixed by binding 127.0.0.1 above; the executor is kept as harmless.)
59  		executor = Executors.newCachedThreadPool();
60  		server.setExecutor(executor);
61  		server.createContext("/echo", exchange -> {
62  			exchange.getRequestBody().readAllBytes();
63  			var body = "hello".getBytes(StandardCharsets.UTF_8);
64  			exchange.getResponseHeaders().add("Content-Type", "text/plain");
65  			exchange.sendResponseHeaders(200, body.length);
66  			exchange.getResponseBody().write(body);
67  			exchange.close();
68  		});
69  		server.createContext("/nullJson", exchange -> {
70  			exchange.getRequestBody().readAllBytes();
71  			var body = "null".getBytes(StandardCharsets.UTF_8);
72  			exchange.getResponseHeaders().add("Content-Type", "application/json");
73  			exchange.sendResponseHeaders(200, body.length);
74  			exchange.getResponseBody().write(body);
75  			exchange.close();
76  		});
77  		server.createContext("/noContentType", exchange -> {
78  			exchange.getRequestBody().readAllBytes();
79  			// Deliberately omits any Content-Type header.
80  			var body = "irrelevant".getBytes(StandardCharsets.UTF_8);
81  			exchange.sendResponseHeaders(200, body.length);
82  			exchange.getResponseBody().write(body);
83  			exchange.close();
84  		});
85  		server.start();
86  	}
87  
88  	@AfterAll
89  	static void stopServer() {
90  		if (server != null)
91  			server.stop(0);
92  		if (executor != null)
93  			executor.shutdownNow();
94  	}
95  
96  	private static String url() {
97  		return "http://127.0.0.1:" + port + "/echo";
98  	}
99  
100 	private static String nullJsonUrl() {
101 		return "http://127.0.0.1:" + port + "/nullJson";
102 	}
103 
104 	private static String noContentTypeUrl() {
105 		return "http://127.0.0.1:" + port + "/noContentType";
106 	}
107 
108 	/** Public no-arg constructor, so the "parser returned null but the type is constructible" fallback applies. */
109 	public static class Widget {
110 		public String name = "default";
111 	}
112 
113 	/** An interface has no constructors at all (so neither the {@code HttpResponse}-arg lookup nor a single-String
114 	 * constructor "string mutater" applies), so {@code as()} falls all the way through to the final
115 	 * Content-Type-negotiation-failure branches. */
116 	public interface UnconstructibleTarget {}
117 
118 	@Test void a01_as_httpResource_returnsStreamResource() throws Exception {
119 		try (var client = RestClient.create().build();
120 				var req = client.get(url());
121 				var res = req.run()) {
122 			var r = res.getContent().as(HttpResource.class);
123 			assertInstanceOf(StreamResource.class, r);
124 			assertArrayEquals("hello".getBytes(StandardCharsets.UTF_8), readAll(r));
125 		}
126 	}
127 
128 	@Test void a02_as_basicResource_returnsStreamResource() throws Exception {
129 		try (var client = RestClient.create().build();
130 				var req = client.get(url());
131 				var res = req.run()) {
132 			var r = res.getContent().as(BasicResource.class);
133 			assertInstanceOf(StreamResource.class, r);
134 			assertArrayEquals("hello".getBytes(StandardCharsets.UTF_8), readAll(r));
135 		}
136 	}
137 
138 	private static byte[] readAll(HttpResource r) throws IOException {
139 		try (var in = r.getContent()) {
140 			return in.readAllBytes();
141 		}
142 	}
143 
144 	@Test void a03_as_noArgCtorFallback_whenParserReturnsNull() throws Exception {
145 		// The JSON parser legitimately reads the "null" literal as a Java null; since Widget has a public no-arg
146 		// constructor (and isn't String), as() falls back to constructing a default instance instead of returning null.
147 		try (var client = RestClient.create().json().build();
148 				var req = client.get(nullJsonUrl());
149 				var res = req.run()) {
150 			var w = res.getContent().as(Widget.class);
151 			assertNotNull(w);
152 			assertEquals("default", w.name);
153 		}
154 	}
155 
156 	@SuppressWarnings({
157 		"unchecked" // generic varargs array (Class<? extends Parser>...) is safe here.
158 	})
159 	@Test void a04_as_noContentTypeHeader_clientHasParsers_throwsContentTypeNotSpecified() throws Exception {
160 		// getMatchingParser() falls back to the sole registered parser when there's exactly one, regardless of media
161 		// type -- registering two (neither matching "text/plain", the implicit default when no header is present)
162 		// forces it to return null despite hasParsers() being true, which is what this branch actually needs.
163 		try (var client = RestClient.create().parsers(JsonParser.class, org.apache.juneau.marshall.xml.XmlParser.class).build();
164 				var req = client.get(noContentTypeUrl());
165 				var res = req.run()) {
166 			var e = assertThrows(RestCallException.class, () -> res.getContent().as(UnconstructibleTarget.class));
167 			var causeMsg = e.getCause().getMessage();
168 			assertTrue(causeMsg.contains("Content-Type not specified"), "Unexpected cause message: " + causeMsg);
169 		}
170 	}
171 
172 	@Test void a05_as_noContentTypeHeader_clientHasNoParsers_throwsUnsupportedMediaType() throws Exception {
173 		try (var client = RestClient.create().build();
174 				var req = client.get(noContentTypeUrl());
175 				var res = req.run()) {
176 			var e = assertThrows(RestCallException.class, () -> res.getContent().as(UnconstructibleTarget.class));
177 			var causeMsg = e.getCause().getMessage();
178 			assertTrue(causeMsg.contains("Unsupported media-type"), "Unexpected cause message: " + causeMsg);
179 		}
180 	}
