32  /**
33   * Exercises remaining {@link ResponseContent} branches not reached by {@code juneau-integration-tests}:
34   * the {@code entity instanceof BasicHttpEntity} fast path in {@code asBytes()}, the
35   * {@code UnsupportedOperationException}-to-{@code IOException} wrapping branch of {@code asInputStream()}, the
36   * {@code asHex()}/{@code asSpacedHex()} convenience methods, and the {@code assertBytes()}/{@code assertObject(...)}/
37   * {@code assertString()} fluent-assertion shortcuts.
38   */
39  @SuppressWarnings({
40  	"resource" // Closeable resources in tests are intentionally unassigned; closing is handled by test infrastructure.
41  })
42  // SEPARATE_THREAD: a socket read blocked on a non-responding peer ignores interrupts, so SAME_THREAD would still hang.
43  @Timeout(value = 60, threadMode = Timeout.ThreadMode.SEPARATE_THREAD)
44  class ResponseContent_Coverage_Test {
45  
46  	private static HttpServer server;
47  	private static ExecutorService executor;
48  	private static int port;
49  
50  	@BeforeAll
51  	static void startServer() throws IOException {
52  		// Bind to (and connect via) the explicit IPv4 loopback, not the wildcard address + "localhost": a wildcard bind
53  		// can be handed an ephemeral port that another process already holds on 127.0.0.1 (SO_REUSEADDR), and that
54  		// more-specific listener then wins every "localhost" connection -- the client hangs on its response.
55  		server = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
56  		port = server.getAddress().getPort();
57  		// Without an explicit executor, exchanges run on HttpServer's single internal dispatch thread, which
58  		// is a bottleneck under -T1C parallel load. (The "server failed to respond" flake once blamed on starvation was most likely
59  		// the wildcard-bind port shadowing fixed by binding 127.0.0.1 above; the executor is kept as harmless.)
60  		executor = Executors.newCachedThreadPool();
61  		server.setExecutor(executor);
62  		server.createContext("/echo", exchange -> {
63  			exchange.getRequestBody().readAllBytes();
64  			var body = "hello".getBytes(StandardCharsets.UTF_8);
65  			exchange.getResponseHeaders().add("Content-Type", "text/plain");
66  			exchange.sendResponseHeaders(200, body.length);
67  			exchange.getResponseBody().write(body);
68  			exchange.close();
69  		});
70  		server.createContext("/truncated", exchange -> {
71  			exchange.getRequestBody().readAllBytes();
72  			var body = "hi".getBytes(StandardCharsets.UTF_8);
73  			exchange.getResponseHeaders().add("Content-Type", "text/plain");
74  			// Declares a larger Content-Length than the bytes actually written, so reading the full declared length
75  			// triggers an IOException partway through -- exercises the IOException-handling branch of asString and toString.
76  			// The explicit flush() forces the partial body onto the wire before close() (which itself throws
77  			// because the declared length wasn't fully written); without it, some JDKs (e.g. 25) tear the
78  			// connection down before anything is sent, producing NoHttpResponseException instead.
79  			exchange.sendResponseHeaders(200, body.length + 1000);
80  			exchange.getResponseBody().write(body);
81  			exchange.getResponseBody().flush();
82  			exchange.close();
83  		});
84  		server.start();
85  	}
86  
87  	@AfterAll
88  	static void stopServer() {
89  		if (server != null)
90  			server.stop(0);
91  		if (executor != null)
92  			executor.shutdownNow();
93  	}

103 	@Test void a01_asBytes_basicHttpEntityFastPath() throws Exception {
104 		// An org.apache.http.HttpResponseInterceptor runs as part of the HttpClient's own response-processing
105 		// pipeline -- before RestClient wraps the raw HttpResponse into a RestResponse/ResponseContent -- so
106 		// replacing the entity here (unlike via a RestCallInterceptor#onConnect, which runs too late) is visible to
107 		// ResponseContent's captured `entity` field, exercising the `entity instanceof BasicHttpEntity<?>` fast path.
108 		var content = "fast-path".getBytes(StandardCharsets.UTF_8);
109 		HttpResponseInterceptor itcp = (response, context) -> response.setEntity(new ByteArrayEntity(null, content));
110 		try (var client = RestClient.create().interceptors(itcp).build();
111 				var req = client.get(url());
112 				var res = req.run()) {
113 			assertArrayEquals(content, res.getContent().asBytes());
114 		}
115 	}

117 	@Test void a02_asInputStream_unsupportedOperationException_wrappedAsIOException() throws Exception {
118 		// asInputStream()'s non-cached branch re-fetches the *live* entity from the raw HttpResponse on every call
119 		// (unlike asBytes(), which uses the field captured at construction), so a RestCallInterceptor#onConnect
120 		// substitution -- which runs after ResponseContent is constructed but before the caller reads the body --
121 		// is late enough to be visible here.
122 		var itcp = new BasicRestCallInterceptor() {
123 			@Override
124 			public void onConnect(RestRequest req, RestResponse res) {
125 				res.asHttpResponse().setEntity(new HttpEntity() {
126 					@Override public boolean isRepeatable() { return false; }
127 					@Override public boolean isChunked() { return false; }
128 					@Override public long getContentLength() { return -1; }
129 					@Override public Header getContentType() { return null; }
130 					@Override public Header getContentEncoding() { return null; }
131 					@Override public InputStream getContent() { throw new UnsupportedOperationException("Simulated: entity content not available."); }
132 					@Override public void writeTo(OutputStream outstream) { throw new UnsupportedOperationException("Not used by this test."); }
133 					@Override public boolean isStreaming() { return false; }
134 					@Override public void consumeContent() { /* no-op */ }
135 				});
136 			}
137 		};
138 		try (var client = RestClient.create().interceptors(itcp).build();
139 				var req = client.get(url());
140 				var res = req.run()) {
141 			assertThrows(IOException.class, () -> res.getContent().asInputStream());
142 		}
143 	}

214 	@Test void a11_nullEntitySentinel_pipeToAndAsBytes_neverDelegateToItsWriteTo() throws Exception {
215 		// ResponseContent's NULL_ENTITY sentinel (used when the raw HttpResponse has no entity)
216 		// has its own writeTo(OutputStream) simplified to fail fast (it's never legitimately invoked -- see the
217 		// in-source comment). Forcing the sentinel into play via an HttpResponseInterceptor (which runs before
218 		// ResponseContent captures `entity`, same as a01 above) and exercising both read paths confirms neither
219 		// pipeTo (which goes through ResponseContent#writeTo, then pipeTo, asInputStream and entity.getContent) nor
220 		// asBytes ever reaches the sentinel's writeTo; if they did, this test would fail with an
221 		// UnsupportedOperationException instead of observing empty content.
222 		HttpResponseInterceptor itcp = (response, context) -> response.setEntity(null);
223 		try (var client = RestClient.create().interceptors(itcp).build();
224 				var req = client.get(url());
225 				var res = req.run()) {
226 			assertArrayEquals(new byte[0], res.getContent().asBytes());
227 		}
228 		try (var client = RestClient.create().interceptors(itcp).build();
229 				var req = client.get(url());
230 				var res = req.run()) {
231 			var os = new ByteArrayOutputStream();
232 			res.getContent().pipeTo(os);
233 			assertArrayEquals(new byte[0], os.toByteArray());
234 		}
