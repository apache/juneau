38   * Exercises {@link RestClient.Builder} shortcut/config methods that aren't reached by the
39   * {@code juneau-integration-tests} module's {@code RestClient_*_Test} suites -- mostly one-line delegate methods
40   * (append/setDefault on the header, form-data, path-data, and query-data lists; media-type shortcuts; OpenAPI
41   * format shortcuts) plus a couple of small branchy methods ({@code interceptors(...)}, {@code rootUrl(Object)}).
42   * A local {@link HttpServer} fixture captures the request actually sent so these can be verified with real
43   * assertions instead of just "didn't throw".
44   */

46  	"resource" // Closeable resources in tests are intentionally unassigned; closing is handled by test infrastructure.
47  })

52  	private static HttpServer server;
53  	private static ExecutorService executor;
54  	private static int port;
55  	private static final AtomicReference<Headers> lastHeaders = new AtomicReference<>();
56  	private static final AtomicReference<String> lastQuery = new AtomicReference<>();

58  	@BeforeAll
59  	static void startServer() throws IOException {
60  		// Bind to (and connect via) the explicit IPv4 loopback, not the wildcard address + "localhost": a wildcard bind
61  		// can be handed an ephemeral port that another process already holds on 127.0.0.1 (SO_REUSEADDR), and that
62  		// more-specific listener then wins every "localhost" connection -- the client hangs on its response.
63  		server = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
64  		port = server.getAddress().getPort();
65  		// Without an explicit executor, exchanges run on HttpServer's single internal dispatch thread, which
66  		// is a bottleneck under -T1C parallel load. (The "server failed to respond" flake once blamed on starvation was most likely
67  		// the wildcard-bind port shadowing fixed by binding 127.0.0.1 above; the executor is kept as harmless.)
68  		executor = Executors.newCachedThreadPool();
69  		server.setExecutor(executor);
70  		server.createContext("/echo", exchange -> {
71  			lastHeaders.set(exchange.getRequestHeaders());
72  			lastQuery.set(exchange.getRequestURI().getQuery());
73  			exchange.getRequestBody().readAllBytes();
74  			exchange.sendResponseHeaders(200, -1);
75  			exchange.close();
76  		});
77  		server.start();
78  	}
79  
80  	@AfterAll
81  	static void stopServer() {
82  		if (server != null)
83  			server.stop(0);
84  		if (executor != null)
85  			executor.shutdownNow();
86  	}

439 	// Package-private accessors on RestClient (isSkipEmptyFormData() etc.) are `protected`, so they're already
440 	// callable directly from this same-package test without any visibility change.
441 	private static boolean isSkipEmptyFormData(RestClient c) { return callProtected(c, "isSkipEmptyFormData"); }
442 	private static boolean isSkipEmptyHeaderData(RestClient c) { return callProtected(c, "isSkipEmptyHeaderData"); }
443 	private static boolean isSkipEmptyQueryData(RestClient c) { return callProtected(c, "isSkipEmptyQueryData"); }
444 
445 	private static boolean callProtected(RestClient c, String method) {
446 		try {
447 			var m = RestClient.class.getDeclaredMethod(method);
448 			m.setAccessible(true);
449 			return (boolean)m.invoke(c);
450 		} catch (ReflectiveOperationException e) {
451 			throw new RuntimeException(e);
452 		}
453 	}

455 	/** Records whether it was instantiated by {@code interceptors(Class...)}. */
456 	public static class RecordingInterceptor extends BasicRestCallInterceptor {
457 		static boolean initCalled = false;
458 		@Override
459 		public void onInit(RestRequest req) {
460 			initCalled = true;
461 		}
