41  /**
42   * Exercises the successful SSRF-guard (policy-covered {@code @Remote}) path of {@link OkHttpTransport}, and the
43   * response-body wiring failure path.
44   *
45   * <p>
46   * A guarded request may only connect to a public address, which a hermetic test cannot reach.  The tests therefore
47   * target public IP literals (OkHttp connects to those without a DNS lookup) and supply an {@link OkHttpClient} whose
48   * {@link SocketFactory} redirects every connect to a loopback server, recording the address the transport asked for.
49   */
50  @SuppressWarnings({
51  	"java:S3415", // Expected value is already first; Sonar mistakes the upper-case constant or field for the expected value
52  	"resource" // Transport/response Closeables are exercised inline; lifecycle is managed by the test.
53  })
54  // SEPARATE_THREAD: a socket read blocked on a non-responding peer ignores interrupts, so SAME_THREAD would still hang.
55  @Timeout(value = 60, threadMode = Timeout.ThreadMode.SEPARATE_THREAD)
56  class OkHttpTransport_PolicyPath_Test {
57  
58  	private static HttpServer server;
59  	private static int port;
60  	private static final List<String> hostHeaders = new CopyOnWriteArrayList<>();
61  
62  	@BeforeAll
63  	static void startServer() throws IOException {
64  		server = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
65  		port = server.getAddress().getPort();
66  		server.createContext("/hello", exchange -> {
67  			hostHeaders.add(exchange.getRequestHeaders().getFirst("Host"));
68  			var body = "hi".getBytes(StandardCharsets.UTF_8);
69  			exchange.sendResponseHeaders(200, body.length);
70  			exchange.getResponseBody().write(body);
71  			exchange.close();
72  		});
73  		server.createContext("/to-public", exchange -> {
74  			hostHeaders.add(exchange.getRequestHeaders().getFirst("Host"));
75  			exchange.getResponseHeaders().add("Location", "http://8.8.4.4:" + port + "/hello");
76  			exchange.sendResponseHeaders(302, -1);
77  			exchange.close();
78  		});
79  		server.createContext("/to-loopback", exchange -> {
80  			hostHeaders.add(exchange.getRequestHeaders().getFirst("Host"));
81  			exchange.getResponseHeaders().add("Location", "http://127.0.0.1:" + port + "/hello");
82  			exchange.sendResponseHeaders(302, -1);
83  			exchange.close();
84  		});
85  		server.start();
86  	}
87  
88  	@AfterAll
89  	static void stopServer() {
90  		if (server != null)
91  			server.stop(0);
92  	}
93  
94  	@BeforeEach
95  	void clearHeaders() {
96  		hostHeaders.clear();
97  	}

110 	@Test
111 	void a02_guarded_publicIpLiteral_sent() throws Exception {
112 		var connects = new CopyOnWriteArrayList<String>();
113 		try (var transport = transport(connects)) {
114 			try (var response = transport.execute(guarded("http://8.8.8.8:" + port + "/hello"))) {
115 				assertEquals(200, response.getStatusCode());
116 				assertEquals("hi", new String(response.getBody().readAllBytes(), StandardCharsets.UTF_8));
117 			}
118 		}
119 		assertEquals(List.of("8.8.8.8:" + port), connects);
120 		assertEquals(List.of("8.8.8.8:" + port), hostHeaders);
121 	}
122 
123 	@Test
124 	void a03_guarded_redirectToPublic_followedOnPinnedClient() throws Exception {
125 		var connects = new CopyOnWriteArrayList<String>();
126 		try (var transport = transport(connects)) {
127 			try (var response = transport.execute(guarded("http://8.8.8.8:" + port + "/to-public"))) {
128 				assertEquals(200, response.getStatusCode());
129 				assertEquals("hi", new String(response.getBody().readAllBytes(), StandardCharsets.UTF_8));
130 			}
131 		}
132 		// Each hop is a separate pinned connect (both Host headers observed by the server).
133 		assertEquals(List.of("8.8.8.8:" + port, "8.8.4.4:" + port), connects);
134 		assertEquals(List.of("8.8.8.8:" + port, "8.8.4.4:" + port), hostHeaders);
135 	}

137 	@Test
138 	void a04_guarded_redirectToLoopback_refusedBeforeConnect() {
139 		var connects = new CopyOnWriteArrayList<String>();
140 		try (var transport = transport(connects)) {
141 			var ex = assertThrows(TransportException.class, () -> transport.execute(guarded("http://8.8.8.8:" + port + "/to-loopback")).close());
142 			assertTrue(ex.getMessage().contains("127.0.0.1"), ex.getMessage());
143 		}
144 		// Only the first hop connected; the private redirect target never did.
145 		assertEquals(List.of("8.8.8.8:" + port), connects);
146 	}

148 	// =================================================================================================================
149 	// B: response-body wiring failure
150 	// =================================================================================================================
151 
152 	@Test
153 	void b01_bodyWiringFails_transportException_andResponseClosed() {
154 		var sourceCalls = new AtomicInteger();
155 		Interceptor breakBody = chain -> {
156 			var response = chain.proceed(chain.request());
157 			var real = response.body();
158 			return response.newBuilder().body(new ResponseBody() {
159 				@Override public MediaType contentType() { return real.contentType(); }
160 				@Override public long contentLength() { return real.contentLength(); }
161 				@Override public BufferedSource source() {
162 					// First call is the transport's byteStream(); later calls come from closing the response.
163 					if (sourceCalls.incrementAndGet() == 1)
164 						throw new IllegalStateException("boom");
165 					return real.source();
166 				}
167 			}).build();
168 		};
169 		var client = new OkHttpClient.Builder().addInterceptor(breakBody).build();
170 		try (var transport = OkHttpTransport.builder().httpClient(client).build()) {
171 			var request = TransportRequest.builder().method("GET").uri("http://127.0.0.1:" + port + "/hello").build();
172 			var ex = assertThrows(TransportException.class, () -> transport.execute(request));
173 			assertEquals("Failed to read response body: boom", ex.getMessage());
174 			assertInstanceOf(IllegalStateException.class, ex.getCause());
175 		}
176 		// The response was closed (its body's source() was requested again) rather than leaking the connection.
177 		assertEquals(2, sourceCalls.get());
178 	}

184 	private static TransportRequest guarded(String url) {
185 		return TransportRequest.builder().method("GET").uri(url).remoteUrlPolicy(true, false).build();
186 	}
187 
188 	// An OkHttp-backed transport whose sockets all connect to the loopback server, recording the requested address.
189 	private static OkHttpTransport transport(List<String> connects) {
190 		var client = new OkHttpClient.Builder()
191 			.proxy(Proxy.NO_PROXY)
192 			.socketFactory(new SocketFactory() {
193 				@Override public Socket createSocket() {
194 					return new Socket() {
195 						@Override public void connect(SocketAddress endpoint, int timeout) throws IOException {
196 							var a = (InetSocketAddress)endpoint;
197 							connects.add(a.getAddress().getHostAddress() + ":" + a.getPort());
198 							super.connect(new InetSocketAddress("127.0.0.1", port), timeout);
199 						}
200 					};
201 				}
202 				@Override public Socket createSocket(String h, int p) { throw new UnsupportedOperationException(); }
203 				@Override public Socket createSocket(String h, int p, InetAddress la, int lp) { throw new UnsupportedOperationException(); }
204 				@Override public Socket createSocket(InetAddress h, int p) { throw new UnsupportedOperationException(); }
205 				@Override public Socket createSocket(InetAddress h, int p, InetAddress la, int lp) { throw new UnsupportedOperationException(); }
206 			})
207 			.build();
208 		return OkHttpTransport.builder().httpClient(client).build();
209 	}
