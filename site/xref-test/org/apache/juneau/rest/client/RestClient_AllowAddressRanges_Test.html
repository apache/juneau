31   * Tests the per-client {@link RestClient.Builder#allowAddressRanges(String...)} option: the same narrow semantics and
32   * validation as the {@code RestClient.allowAddressRanges} system property, threaded to the literal-host check, the
33   * redirect revalidation and the JDK transport's pinned connect.

46  	/** A policy-capable transport that records the requests it is asked to send. */
47  	private static final class Recording implements HttpTransport {
48  		final List<TransportRequest> requests = new ArrayList<>();
49  
50  		@Override
51  		public TransportResponse execute(TransportRequest request) {
52  			requests.add(request);
53  			return TransportResponse.builder().statusCode(200).body(new ByteArrayInputStream(new byte[0])).build();
54  		}
55  
56  		@Override
57  		public boolean supportsUrlPolicy() {
58  			return true;
59  		}
60  	}

69  		for (var bad : new String[]{"10.0.0.0/8", "127.0.0.1", "169.254.169.254", "0.0.0.0/0", "::/0", "example.com", "198.18.0.0/14"}) {

71  			var ex = assertThrows(IllegalArgumentException.class, builder::build, bad);

78  			assertEquals(RemoteUrlPolicy.AllowedRanges.of("198.18.0.0/15,192.0.0.8"), client.getAllowedAddressRanges());
79  		}
80  		try (var client = RestClient.builder().transport(new Recording()).build()) {
81  			assertTrue(client.getAllowedAddressRanges().isEmpty());
82  		}

124 	@Test void b04_unionWithSystemProperty() throws Exception {
125 		var previous = System.getProperty(RemoteUrlPolicy.ALLOW_ADDRESS_RANGES_PROPERTY);
126 		System.setProperty(RemoteUrlPolicy.ALLOW_ADDRESS_RANGES_PROPERTY, "192.0.0.0/24");
127 		try {
128 			var transport = new Recording();
129 			try (var client = RestClient.builder().transport(transport).allowAddressRanges("198.18.0.0/15").build()) {
130 				client.remote(Api.class).get("http://192.0.0.5/x");
131 				client.remote(Api.class).get("http://198.18.0.5/x");
132 				assertSize(2, transport.requests);
133 				var api = client.remote(Api.class);
134 				assertThrows(IllegalArgumentException.class, () -> api.get("http://224.0.0.5/x"));
135 			}
136 		} finally {
137 			if (previous == null)
138 				System.clearProperty(RemoteUrlPolicy.ALLOW_ADDRESS_RANGES_PROPERTY);
139 			else
140 				System.setProperty(RemoteUrlPolicy.ALLOW_ADDRESS_RANGES_PROPERTY, previous);
141 		}

148 	private static final class Hops implements PolicyEnforcedRedirects.HopExecutor {
149 		final List<TransportRequest> hops = new ArrayList<>();
150 		final Deque<String> locations;
151 
152 		Hops(String...locations) {
153 			this.locations = new ArrayDeque<>(List.of(locations));
154 		}
155 
156 		@Override
157 		public TransportResponse execute(TransportRequest hop) {
158 			hops.add(hop);
159 			var location = locations.poll();
160 			if (location == null)
161 				return TransportResponse.builder().statusCode(200).body(new ByteArrayInputStream(new byte[0])).build();
162 			return TransportResponse.builder().statusCode(302).header("Location", location).build();
163 		}
164 	}
165 
166 	private static TransportRequest guarded(String uri, RemoteUrlPolicy.AllowedRanges ranges) {
167 		return TransportRequest.builder().method("GET").uri(uri).remoteUrlPolicy(true, false).allowedAddressRanges(ranges).build();
168 	}
169 
170 	@Test void c01_redirectToAllowedSpecialPurpose_followed_andRangesCarriedAcrossHops() throws Exception {
171 		var ranges = RemoteUrlPolicy.AllowedRanges.of("198.18.0.0/15");
172 		var hops = new Hops("http://198.18.0.9/a", "http://198.18.0.10/b");
173 		PolicyEnforcedRedirects.execute(guarded("http://example.com/start", ranges), hops);
174 		assertSize(3, hops.hops);
175 		for (var h : hops.hops)
176 			assertEquals(ranges, h.getAllowedAddressRanges());
177 	}
178 
179 	@Test void c02_redirectToSpecialPurpose_withoutRanges_refused() {
180 		var hops = new Hops("http://198.18.0.9/a");
181 		var ex = assertThrows(TransportException.class, () -> PolicyEnforcedRedirects.execute(guarded("http://example.com/start", RemoteUrlPolicy.AllowedRanges.NONE), hops));
182 		assertTrue(ex.getMessage().contains("denied by default"), ex.getMessage());
183 	}
184 
185 	@Test void c03_redirectToLoopbackOrMetadata_refused_evenWithRanges() {
186 		var ranges = RemoteUrlPolicy.AllowedRanges.of("198.18.0.0/15");
187 		for (var target : new String[]{"http://127.0.0.1/a", "http://169.254.169.254/a"}) {
188 			var hops = new Hops(target);

198 	@Test void d01_javaHttpTransport_withRanges_notRefusedByPolicy_whileWithoutIsRefused() {
199 		var transport = JavaHttpTransport.create();
200 		var without = assertThrows(TransportException.class, () -> transport.execute(guarded("http://198.18.0.1:1/x", RemoteUrlPolicy.AllowedRanges.NONE)).close());
201 		assertInstanceOf(IllegalArgumentException.class, without.getCause());
202 		assertTrue(without.getMessage().contains("No allowed address for host '198.18.0.1'"), without.getMessage());
203 
204 		var request = TransportRequest.builder().method("GET").uri("http://198.18.0.1:1/x").remoteUrlPolicy(true, false)
205 			.allowedAddressRanges(RemoteUrlPolicy.AllowedRanges.of("198.18.0.0/15")).timeout(java.time.Duration.ofSeconds(2)).build();
206 		try {
207 			transport.execute(request).close();
208 		} catch (TransportException e) {
209 			// The connect itself may fail (nothing listens at 198.18.0.1); it must not be the policy refusal.
210 			assertFalse(e.getMessage().contains("No allowed address"), e.getMessage());
211 		} catch (IOException e) {
212 			// Closing the response is irrelevant here.
213 		}
214 	}

216 	@Test void d02_javaHttpTransport_withRanges_neverReopensLoopback() {
217 		var transport = JavaHttpTransport.create();
218 		for (var host : new String[]{"127.0.0.1", "169.254.169.254"}) {
219 			var request = guarded("http://" + host + ":1/x", RemoteUrlPolicy.AllowedRanges.of("198.18.0.0/15"));
220 			var ex = assertThrows(TransportException.class, () -> transport.execute(request).close(), host);
221 			assertTrue(ex.getMessage().contains("No allowed address for host '" + host + "'"), ex.getMessage());
222 		}
