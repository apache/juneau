17  package org.apache.juneau.rest.client.jetty;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.net.*;
22  
23  import org.eclipse.jetty.client.*;
24  import org.junit.jupiter.api.*;
25  
26  /**
27   * Unit tests for {@link CredentialGuardingHttpClient}, including the defensive fallbacks that fail closed (strip
28   * credentials) when the origins of a redirect cannot be compared.
29   */
30  @SuppressWarnings({
31  	"resource" // HttpClient fixtures are never started; nothing to close.
32  })
33  class CredentialGuardingHttpClient_Test {
34  
35  	private static final String KEY = "X-API-Key";
36  
37  	private final CredentialGuardingHttpClient client = new CredentialGuardingHttpClient();
38  
39  	private Request oldRequest() {
40  		return client.newRequest("http://a.example.com:8080/start").headers(f -> f.add(KEY, "k-1").add("X-Other", "o-1"));
41  	}
42  
43  	@Test
44  	void a01_sameOrigin_keepsCredentialHeaders() {
45  		var copy = client.copyRequest(oldRequest(), URI.create("http://a.example.com:8080/next"));
46  		assertEquals("k-1", copy.getHeaders().get(KEY));
47  		assertEquals("o-1", copy.getHeaders().get("X-Other"));
48  	}
49  
50  	@Test
51  	void a02_differentOrigin_stripsCredentialHeadersOnly() {
52  		var copy = client.copyRequest(oldRequest(), URI.create("http://b.example.com:8080/next"));
53  		assertNull(copy.getHeaders().get(KEY));
54  		assertEquals("o-1", copy.getHeaders().get("X-Other"));
55  	}
56  
57  	@Test
58  	void a03_unparseableOriginalScheme_failsClosedAndStrips() {
59  		// A scheme containing a space cannot be turned back into a URI, so the origin comparison is impossible.
60  		var old = oldRequest().scheme("ht tp");
61  		var copy = client.copyRequest(old, URI.create("http://a.example.com:8080/next"));
62  		assertNull(copy.getHeaders().get(KEY));
63  		assertEquals("o-1", copy.getHeaders().get("X-Other"));
64  	}
65  
66  	@Test
67  	void a04_underscoreHostRedirectTarget_rejectedByJettyBeforeTheOriginCheck() {
68  		// java.net.URI parses an underscore host name with a null host, and Jetty refuses to build a request for a null
69  		// host, so a redirect to such a host fails in Jetty itself and never reaches this client's origin comparison.
70  		var old = oldRequest();
71  		var next = URI.create("http://my_service:8080/next");
72  		var ex = assertThrows(IllegalArgumentException.class, () -> client.copyRequest(old, next));
73  		assertTrue(ex.getMessage().contains("Invalid URI host"), ex.getMessage());
74  	}
75  }
