35  /**
36   * Verifies that an {@link ApacheHc45Transport} built with a caller-supplied {@link CloseableHttpClient} does not
37   * forward caller-set credential headers to a different origin when that client follows a redirect, while still
38   * forwarding them on a same-origin redirect, and without bypassing the client's own cookie store or credentials
39   * provider.
40   */
41  @SuppressWarnings({
42  	"resource" // Transport/client instances are short-lived test fixtures.
43  })
44  // SEPARATE_THREAD: a socket read blocked on a non-responding peer ignores interrupts, so SAME_THREAD would still hang.
45  @Timeout(value = 60, threadMode = Timeout.ThreadMode.SEPARATE_THREAD)
46  class ApacheHc45Transport_CallerClientRedirectCredentials_Test {

49  	private static HttpServer serverB;
50  	private static HttpServer serverC;
51  	private static int portA;
52  	private static int portB;
53  	private static int portC;
54  	private static final AtomicReference<String> firstHopAuth = new AtomicReference<>();
55  	private static final AtomicReference<String> midHopAuth = new AtomicReference<>();

57  	@BeforeAll
58  	static void startServers() throws IOException {
59  		// Bind to (and connect via) the explicit IPv4 loopback; the two servers differ only by port, so they are different origins.
60  		serverB = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
61  		portB = serverB.getAddress().getPort();
62  		serverB.createContext("/echo-creds", ApacheHc45Transport_CallerClientRedirectCredentials_Test::echoCreds);
63  		serverB.createContext("/challenge", ApacheHc45Transport_CallerClientRedirectCredentials_Test::challenge);
64  		serverB.createContext("/bounce-to-a", exchange -> redirect(exchange, 302, "http://127.0.0.1:" + portA + "/echo-creds", null));
65  		serverB.createContext("/bounce-to-a-challenge", exchange -> redirect(exchange, 302, "http://127.0.0.1:" + portA + "/challenge", null));
66  		serverB.createContext("/to-c", exchange -> redirect(exchange, 302, "http://127.0.0.1:" + portC + "/to-a", null));
67  		serverB.createContext("/to-a-then-b", exchange -> redirect(exchange, 302, "http://127.0.0.1:" + portA + "/to-b", null));
68  		serverB.start();
69  
70  		serverC = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
71  		portC = serverC.getAddress().getPort();
72  		serverC.createContext("/to-a", exchange -> redirect(exchange, 302, "http://127.0.0.1:" + portA + "/echo-creds", null));
73  		serverC.start();
74  
75  		serverA = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
76  		portA = serverA.getAddress().getPort();
77  		serverA.createContext("/echo-creds", ApacheHc45Transport_CallerClientRedirectCredentials_Test::echoCreds);
78  		serverA.createContext("/redirect-cross", exchange -> redirect(exchange, 302, "http://127.0.0.1:" + portB + "/echo-creds", null));
79  		serverA.createContext("/redirect-same", exchange -> redirect(exchange, 302, "/echo-creds", null));
80  		serverA.createContext("/redirect-cross-set-cookie", exchange -> redirect(exchange, 302, "http://127.0.0.1:" + portB + "/echo-creds", "store=s-1; Path=/"));
81  		serverA.createContext("/challenge", ApacheHc45Transport_CallerClientRedirectCredentials_Test::challenge);
82  		serverA.createContext("/redirect-same-set-cookie", exchange -> redirect(exchange, 302, "/echo-creds", "store=s-1; Path=/"));
83  		serverA.createContext("/redirect-same-challenge", exchange -> redirect(exchange, 302, "/challenge", null));
84  		serverA.createContext("/redirect-cross-307", exchange -> redirect(exchange, 307, "http://127.0.0.1:" + portB + "/echo-creds", null));
85  		serverA.createContext("/redirect-same-307", exchange -> redirect(exchange, 307, "/echo-creds", null));
86  		serverA.createContext("/redirect-cross-challenge", exchange -> redirect(exchange, 302, "http://127.0.0.1:" + portB + "/challenge", null));
87  		serverA.createContext("/redirect-a-b-a", exchange -> {
88  			firstHopAuth.set(String.valueOf(exchange.getRequestHeaders().getFirst("Authorization")));
89  			redirect(exchange, 302, "http://127.0.0.1:" + portB + "/bounce-to-a", null);
90  		});
91  		serverA.createContext("/redirect-a-b-a-set-cookie", exchange -> redirect(exchange, 302, "http://127.0.0.1:" + portB + "/bounce-to-a", "store=s-1; Path=/"));
92  		serverA.createContext("/redirect-a-b-a-challenge", exchange -> redirect(exchange, 302, "http://127.0.0.1:" + portB + "/bounce-to-a-challenge", null));
93  		serverA.createContext("/redirect-userinfo", exchange -> redirect(exchange, 302, "http://bob:pw@127.0.0.1:" + portA + "/echo-creds", null));
94  		serverA.createContext("/redirect-a-b-c-a", exchange -> redirect(exchange, 302, "http://127.0.0.1:" + portB + "/to-c", null));
95  		serverA.createContext("/redirect-a-b-a-b", exchange -> redirect(exchange, 302, "http://127.0.0.1:" + portB + "/to-a-then-b", null));
96  		serverA.createContext("/to-b", exchange -> {
97  			midHopAuth.set(String.valueOf(exchange.getRequestHeaders().getFirst("Authorization")));
98  			redirect(exchange, 302, "http://127.0.0.1:" + portB + "/echo-creds", null);
99  		});
100 		serverA.start();
101 	}

113 	private static String all(HttpExchange exchange, String name) {
114 		var values = exchange.getRequestHeaders().get(name);
115 		return values == null || values.isEmpty() ? "none" : String.join(",", values);
116 	}
117 
118 	private static void echoCreds(HttpExchange exchange) throws IOException {
119 		respond(exchange, 200, "auth=" + all(exchange, "Authorization") + "|cookie=" + all(exchange, "Cookie")
120 			+ "|key=" + all(exchange, "X-API-Key") + "|trace=" + all(exchange, "X-Trace"));
121 	}
122 
123 	// Answers 401 with a Basic challenge until the request carries Authorization, then echoes it.
124 	private static void challenge(HttpExchange exchange) throws IOException {
125 		if (exchange.getRequestHeaders().getFirst("Authorization") == null) {
126 			exchange.getResponseHeaders().add("WWW-Authenticate", "Basic realm=\"b\"");
127 			respond(exchange, 401, "challenge");
128 			return;
129 		}
130 		echoCreds(exchange);
131 	}
132 
133 	private static void respond(HttpExchange exchange, int status, String text) throws IOException {
134 		var body = text.getBytes(StandardCharsets.UTF_8);
135 		exchange.getResponseHeaders().add("Content-Type", "text/plain");
136 		exchange.sendResponseHeaders(status, body.length);
137 		exchange.getResponseBody().write(body);
138 		exchange.close();
139 	}
140 
141 	private static void redirect(HttpExchange exchange, int status, String location, String setCookie) throws IOException {
142 		exchange.getResponseHeaders().add("Location", location);
143 		if (setCookie != null)
144 			exchange.getResponseHeaders().add("Set-Cookie", setCookie);
145 		exchange.sendResponseHeaders(status, -1);
146 		exchange.close();
147 	}

149 	private static String get(CloseableHttpClient callerClient, String path) throws Exception {
150 		return get(ApacheHc45Transport.builder().httpClient(callerClient).build(), path);
151 	}

153 	private static String get(ApacheHc45Transport transport, String path) throws Exception {
154 		try (var client = RestClient.builder().transport(transport).rootUrl("http://127.0.0.1:" + portA).build();
155 				var response = client.get(path)
156 					.header("Authorization", "Bearer secret-token")
157 					.header("Cookie", "session=abc123")
158 					.header("X-API-Key", "k-1")
159 					.header("X-Trace", "t-1")
160 					.run()) {
161 			assertEquals(200, response.getStatusCode());
162 			return response.getBodyAsString();
163 		}
164 	}

176 	@Test
177 	void a03_crossOrigin_callerClient_cookieStoreCookieStripped() throws Exception {
178 		// J0824: the cookie A sets lives in the caller's cookie store, which would send it on to B (a cookie's domain ignores
179 		// the port).  The guard strips every credential header on a cross-origin hop, so the store's Cookie goes too.
180 		var callerClient = HttpClients.custom().setDefaultCookieStore(new BasicCookieStore()).build();
181 		assertEquals("auth=none|cookie=none|key=none|trace=t-1", get(callerClient, "/redirect-cross-set-cookie"));
182 	}

184 	@Test
185 	void a04_crossOrigin_callerClient_credentialsProviderAuthorizationStripped() throws Exception {
186 		// J0824: the credentials provider would answer B's challenge.  The answer is an Authorization header, so it is stripped;
187 		// B keeps challenging and the caller sees the 401.
188 		var callerClient = HttpClients.custom().setDefaultCredentialsProvider(provider(portB)).build();
189 		assertEquals("401:challenge", getAnyStatus(callerClient, "/redirect-cross-challenge"));
190 	}

210 	private static CredentialsProvider provider(int port) {
211 		var credentials = new BasicCredentialsProvider();
212 		credentials.setCredentials(new AuthScope("127.0.0.1", port), new UsernamePasswordCredentials("bob", "pw"));
213 		return credentials;
214 	}

216 	private static String getAnyStatus(CloseableHttpClient callerClient, String path) throws Exception {
217 		var transport = ApacheHc45Transport.builder().httpClient(callerClient).build();
218 		try (var client = RestClient.builder().transport(transport).rootUrl("http://127.0.0.1:" + portA).build();
219 				var response = client.get(path).header("Authorization", "Bearer secret-token").run()) {
220 			return response.getStatusCode() + ":" + response.getBodyAsString();
221 		}
222 	}
