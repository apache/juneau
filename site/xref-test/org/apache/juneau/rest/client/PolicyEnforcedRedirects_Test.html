27  /**
28   * Verifies that {@link PolicyEnforcedRedirects} forwards caller-set credential headers
29   * ({@link RedirectSecurity#stripOnCrossOrigin()}) only to a same-origin redirect target, and does not forward them
30   * to a different origin or across an {@code https} &rarr; {@code http} downgrade.
31   *
32   * <p>
33   * Drives {@link PolicyEnforcedRedirects#execute(TransportRequest, PolicyEnforcedRedirects.HopExecutor)} directly
34   * with a recording {@link PolicyEnforcedRedirects.HopExecutor}, so no network access is involved.
35   */
36  @SuppressWarnings({
37  	"resource" // TransportResponse objects returned by the scripted HopExecutor wrap in-memory streams and are not closed by the tests
38  })
39  class PolicyEnforcedRedirects_Test {
40  
41  	/** Records every hop and answers each with the next scripted {@code Location} (302), then a final 200. */
42  	private static final class ScriptedHops implements PolicyEnforcedRedirects.HopExecutor {
43  		final List<TransportRequest> hops = new ArrayList<>();
44  		final Deque<String> locations;
45  		int status = 302;
46  
47  		ScriptedHops(String...locations) {
48  			this.locations = new ArrayDeque<>(List.of(locations));
49  		}
50  
51  		@Override
52  		public TransportResponse execute(TransportRequest hopRequest) {
53  			hops.add(hopRequest);
54  			var location = locations.poll();
55  			if (location == null)
56  				return TransportResponse.builder().statusCode(200).body(new ByteArrayInputStream(new byte[0])).build();
57  			return TransportResponse.builder().statusCode(status).header("Location", location).build();
58  		}
59  
60  		TransportRequest last() {
61  			return hops.get(hops.size() - 1);
62  		}
63  	}
64  
65  	private static TransportRequest request(String uri) {
66  		return TransportRequest.builder()
67  			.method("GET")
68  			.uri(uri)
69  			.remoteUrlPolicy(true, false)
70  			.header("Authorization", "Bearer secret-token")
71  			.header("Cookie", "session=abc123")
72  			.header("Proxy-Authorization", "Basic cHJveHk6cHc=")
73  			.header("X-API-Key", "key-123")
74  			.header("Accept", "application/json")
75  			.header("X-Trace", "t-1")
76  			.build();
77  	}
78  
79  	private static List<String> headerNames(TransportRequest r) {
80  		return r.getHeaders().stream().map(TransportHeader::name).toList();
81  	}
82  
83  	private static ScriptedHops run(TransportRequest initial, String...locations) throws Exception {
84  		var hops = new ScriptedHops(locations);
85  		PolicyEnforcedRedirects.execute(initial, hops);
86  		return hops;
87  	}

94  		var hops = run(request("http://a.example.com/start"), "http://b.example.com/echo");

123 	@Test void a05_crossOriginThenBackToOrigin_credentialsRestored() throws Exception {

154 			assertList(headerNames(hops.last()), "Accept", "X-Trace");
155 		}

158 	private static TransportRequest requestWithMethod(String method, String uri) {
159 		return TransportRequest.builder().method(method).uri(uri).remoteUrlPolicy(true, false)
160 			.header("Authorization", "Bearer secret-token").header("Accept", "application/json").build();
161 	}

163 	private static ScriptedHops runWithStatus(TransportRequest initial, int status, String...locations) throws Exception {
164 		var hops = new ScriptedHops(locations);
165 		hops.status = status;
166 		PolicyEnforcedRedirects.execute(initial, hops);
167 		return hops;
168 	}
