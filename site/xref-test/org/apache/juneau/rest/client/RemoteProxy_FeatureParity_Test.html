36  /**
37   * Remote-proxy <b>feature-parity</b> suite for the <b>next-generation</b> REST client engine
38   * ({@link RestClient#remote(Class)} &rarr; {@link org.apache.juneau.rest.server.server.client.remote.RemoteClient}), exercised against
39   * a synthetic {@code @Remote} interface that reproduces a comprehensive remote-proxy feature matrix (F1&ndash;F24) plus
40   * verb-completeness rows (PUT/DELETE/{@code @RemoteOp}).
41   *
42   * <p>
43   * This is the next-gen counterpart to the classic {@code Remote_*_Test} corpus and tracks
44   * The single shared engine
45   * ({@code RemoteClient}) is the system-under-test; it is driven through two columns:
46   * <ul>
47   * 	<li><b>{@link B_MockServerColumn}</b> &mdash; the {@link MockRestClient} in-JVM servlet round-trip
48   * 		(via {@code MockRestClient.create(Fixture.class).getClient().remote(iface)}), asserting the
49   * 		<i>deserialized/echoed response</i>.
50   * 	<li><b>{@link C_RealTransportColumn}</b> &mdash; a next-gen {@link RestClient} built directly over a network-free
51   * 		{@link MockHttpTransport} stub, asserting the <i>serialized request shape</i> the engine emits.
52   * </ul>
53   *
54   * <p>
55   * Cross-walk: each synthetic method's comment names the feature id (F-row) it exercises.
56   */

263 	/** Capture transport for the real-{@link RestClient} column: records every request, replies 200/"result". */
264 	static MockHttpTransport.Builder captureTransport(List<TransportRequest> captured) {
265 		return MockHttpTransport.builder().fallback(req -> {
266 			captured.add(req);
267 			return TransportResponse.builder().statusCode(200).body(new ByteArrayInputStream("result".getBytes())).build();
268 		});
269 	}

271 	static String readBody(TransportRequest req) throws IOException {
272 		var body = req.getBody();
273 		if (body == null)
274 			return null;
275 		var baos = new ByteArrayOutputStream();
276 		body.writeTo(baos);
277 		return baos.toString();
278 	}
