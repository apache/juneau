61  	/** Asserts the planted secret is absent in plaintext, UTF-8 spaced-hex, Base64, and escaped code-unit forms. */
62  	private static void assertAllEncodingsAbsent(String out, String secret) {
63  		assertFalse(out.contains(secret), () -> "plaintext secret present in: " + out);
64  		var bytes = secret.getBytes(StandardCharsets.UTF_8);
65  		assertFalse(out.contains(toSpacedHex(bytes)), () -> "spaced-hex secret present in: " + out);
66  		assertFalse(out.contains(Base64.getEncoder().encodeToString(bytes)), () -> "base64 secret present in: " + out);
67  		var esc = new StringBuilder();
68  		for (var i = 0; i < secret.length(); i++)
69  			esc.append(f("\\u%04X", (int) secret.charAt(i)));
70  		assertFalse(out.contains(esc.toString()), () -> "escaped-code-unit secret present in: " + out);
71  	}

73  	/** Asserts no client-controllable raw control character survived — only formatter-owned {@code \n} separators. */
74  	private static void assertNoRawControlChars(String out) {
75  		assertEquals(-1, out.indexOf('\r'), () -> "raw CR present in: " + out);
76  		assertEquals(-1, out.indexOf('\u0085'), () -> "raw NEL present in: " + out);
77  	}

79  	private static TransportResponse.Builder responseBuilder(int statusCode, String body) {
80  		return TransportResponse.builder()
81  			.statusCode(statusCode)
82  			.reasonPhrase("OK")
83  			.body(new ByteArrayInputStream(body.getBytes(StandardCharsets.UTF_8)));
84  	}

86  	/** Runs a request/response pair through a real client at the given level and returns the captured message. */
87  	private static String runAndCapture(String loggerName, Level level, RestClient.Builder clientBuilder,
88  			java.util.function.Function<RestRequest,RestRequest> requestFn) throws Exception {
89  		var logger = RichLogger.getLogger(loggerName);
90  		var prevLevel = logger.getLevel();
91  		logger.setLevel(level);
92  		try (var c = logger.captureEvents(level)) {
93  			try (var client = clientBuilder.debugLoggerName(loggerName).build();
94  				var res = requestFn.apply(client.get("http://example.com/x")).run()) {
95  				res.body().asString();
96  			}
97  			assertNotNull(c.last());
98  			return c.last().getMessage();
99  		} finally {
100 			logger.setLevel(prevLevel);
101 		}
102 	}

140 	/**
141 	 * Runs a POST (with the given request body/content-type) against a stub transport that returns the given
142 	 * response body/content-type/content-encoding, at {@code FINEST}, through a client using {@code formatter},
143 	 * and returns the captured message. The request side always carries a body so both {@code appendBody} call
144 	 * sites run.
145 	 */
146 	private static String runBodyTest(String loggerName, BasicRestClientDebugFormatter formatter,
147 			String reqBody, String reqContentType, String resBody, String resContentType, String resContentEncoding)
148 			throws Exception {
149 		var logger = RichLogger.getLogger(loggerName);
150 		var prevLevel = logger.getLevel();
151 		logger.setLevel(Level.FINEST);
152 		try (var c = logger.captureEvents(Level.FINEST)) {
153 			try (var client = RestClient.builder()
154 				.transport(req -> {
155 					if (req.getBody() != null) {
156 						try {
157 							req.getBody().writeTo(new ByteArrayOutputStream());
158 						} catch (IOException e) {
159 							throw new RuntimeException(e);
160 						}
161 					}
162 					var b = TransportResponse.builder().statusCode(200).reasonPhrase("OK");
163 					if (resContentType != null)
164 						b.header("Content-Type", resContentType);
165 					if (resContentEncoding != null)
166 						b.header("Content-Encoding", resContentEncoding);
167 					return b.body(new ByteArrayInputStream(resBody.getBytes(StandardCharsets.UTF_8))).build();
168 				})
169 				.debugLoggerName(loggerName)
170 				.debugFormatter(formatter)
171 				.build();
172 				var res = client.post("http://example.com/x").header("Content-Type", reqContentType)
173 					.bodyString(reqBody).run()) {
174 				res.body().asString();
175 			}
176 			assertNotNull(c.last());
177 			return c.last().getMessage();
178 		} finally {
179 			logger.setLevel(prevLevel);
180 		}
181 	}

184 	 * The direct "{@code d02}-equivalent inversion test". Locks all four RED-proof preconditions:
185 	 * {@code FINEST} capture, a non-empty captured body, the placeholder actually present (not just raw-text

189 	@Test
190 	void c01_bodyGateOff_default_suppressesPlaceholder_allEncodingsAbsent() throws Exception {

196 			assertTrue(msg.contains("[body suppressed"), msg);
197 			assertTrue(msg.contains("set JUNEAU_REST_DEBUG_ALLOW_DUMP_BODIES to enable"), msg);
198 			assertFalse(msg.contains("UTF-8---"), msg);
199 			assertFalse(msg.contains("Content Hex"), msg);
200 			assertAllEncodingsAbsent(msg, secret);
201 		} finally {
202 			BasicRestClientDebugFormatter.resetAllowDumpBodiesForTest(null);
203 		}

389 				System.clearProperty(BasicRestClientDebugFormatter.ENV_ALLOW_DUMP_BODIES);
390 			else
391 				System.setProperty(BasicRestClientDebugFormatter.ENV_ALLOW_DUMP_BODIES, prev);
392 			BasicRestClientDebugFormatter.resetAllowDumpBodiesForTest(null);
