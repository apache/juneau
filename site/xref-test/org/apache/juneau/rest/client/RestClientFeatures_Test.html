259 	void b03_finestTier_includesBodies() throws Exception {
260 		var loggerName = getClass().getName() + ".b03";
261 		var logger = org.apache.juneau.commons.logging.RichLogger.getLogger(loggerName);
262 		var previous = logger.getLevel();
263 		logger.setLevel(java.util.logging.Level.FINEST);
264 		// Secret hardening: body dumping is secure-by-default (off) — force the master gate on for this
265 		// test so the FINEST-tier body-inclusion behavior can be observed at all.
266 		BasicRestClientDebugFormatter.resetAllowDumpBodiesForTest(Boolean.TRUE);
267 		try (var capture = logger.captureEvents(java.util.logging.Level.FINEST)) {
268 			try (var client = RestClient.builder()
269 					.transport(MockHttpTransport.builder()
270 						.fallback(req -> {
271 							if (req.getBody() != null) {
272 								try {
273 									req.getBody().writeTo(new ByteArrayOutputStream());
274 								} catch (IOException e) {
275 									throw new RuntimeException(e);
276 								}
277 							}
278 							return TransportResponse.builder()
279 								.statusCode(200)
280 								.header("Content-Type", "text/plain")
281 								.body(new ByteArrayInputStream("uvwxyz".getBytes(StandardCharsets.UTF_8)))
282 								.build();
283 						})
284 						.build())
285 					.rootUrl("http://x.com")
286 					.debugLoggerName(loggerName)
287 					.debugFormatter(new BasicRestClientDebugFormatter().bodyCap(4))
288 					.build();
289 				var response = client.post("/body").bodyString("abcdef").run()) {
290 				assertEquals("uvwxyz", response.body().asString());
291 			}
292 			assertNotNull(capture.last());
293 			// Stable-INFO emission: the record is always stamped INFO; FINEST only widens which
294 			// sections the message body includes (here, the body blocks below).
295 			assertEquals(java.util.logging.Level.INFO, capture.last().getLevel());
296 			var msg = capture.last().getMessage();
297 			assertTrue(msg.contains("---Request Content---"));
298 			assertTrue(msg.contains("abcd"));
299 			assertTrue(msg.contains("---Response Content---"));
300 			assertTrue(msg.contains("uvwx"));
301 		} finally {
302 			logger.setLevel(previous);
303 			BasicRestClientDebugFormatter.resetAllowDumpBodiesForTest(null);
304 		}
305 	}
