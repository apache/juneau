38  /**
39   * Verifies that an {@link ApacheHc5Transport} built with a caller-supplied {@link CloseableHttpClient} never sends
40   * caller-set credential headers to a different origin on a redirect, while the client's own redirect settings decide
41   * which redirects are followed.
42   *
43   * <p>
44   * With {@code DefaultRedirectStrategy} a cross-origin redirect that would replay a credential header is not followed at
45   * all.  With {@code LaxRedirectStrategy}, which follows every redirect, the transport's per-request context removes the
46   * credential headers from the cross-origin hop.
47   */
48  @SuppressWarnings({
49  	"java:S3415", // Expected value is already first; Sonar mistakes the upper-case constant or field for the expected value
50  	"resource" // Transport/client instances are short-lived test fixtures.
51  })
52  // SEPARATE_THREAD: a socket read blocked on a non-responding peer ignores interrupts, so SAME_THREAD would still hang.
53  @Timeout(value = 60, threadMode = Timeout.ThreadMode.SEPARATE_THREAD)
54  class ApacheHc5Transport_CallerClientRedirectCredentials_Test {

56  	private static HttpServer serverA;
57  	private static HttpServer serverB;
58  	private static HttpServer proxy;
59  	private static int portA;
60  	private static int portB;
61  	private static int proxyPort;
62  	private static final List<String> seenByProxy = new CopyOnWriteArrayList<>();
63  	private static final AtomicInteger hitsB = new AtomicInteger();
64  	private static final List<String> seenByB = new CopyOnWriteArrayList<>();
65  	private static final CountDownLatch sseRelease = new CountDownLatch(1);
66  	private static final ExecutorService EXECUTOR = Executors.newCachedThreadPool();

68  	@BeforeAll
69  	static void startServers() throws IOException {
70  		// Bind to (and connect via) the explicit IPv4 loopback; the two servers differ only by port, so they are different origins.
71  		serverB = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
72  		portB = serverB.getAddress().getPort();
73  		serverB.createContext("/echo-creds", exchange -> {
74  			hitsB.incrementAndGet();
75  			echoCreds(exchange);
76  		});
77  		serverB.createContext("/back", exchange -> {
78  			hitsB.incrementAndGet();
79  			redirect(exchange, "http://127.0.0.1:" + portA + "/echo-creds");
80  		});
81  		// Challenges the first request of each test, so a client with credentials for this origin resends it; echoes the resend.
82  		serverB.createContext("/challenge", exchange -> {
83  			seenByB.add(creds(exchange));
84  			if (hitsB.incrementAndGet() > 1) {
85  				echoCreds(exchange);
86  				return;
87  			}
88  			exchange.getResponseHeaders().add("WWW-Authenticate", "Basic realm=\"b\"");
89  			exchange.sendResponseHeaders(401, -1);
90  			exchange.close();
91  		});
92  		// Answers 503 with Retry-After: 0 to the first request of each test, so the client's retry strategy resends it.
93  		// The 503 also closes its connection so the resend always opens a fresh one: HttpClient spends its single retry on the 503,
94  		// so a resend that found a stale pooled keep-alive connection would fail with NoHttpResponseException.
95  		serverB.createContext("/flaky", exchange -> {
96  			seenByB.add(creds(exchange));
97  			if (hitsB.incrementAndGet() > 1) {
98  				echoCreds(exchange);
99  				return;
100 			}
101 			exchange.getResponseHeaders().add("Retry-After", "0");
102 			exchange.getResponseHeaders().add("Connection", "close");
103 			exchange.sendResponseHeaders(503, -1);
104 			exchange.close();
105 		});
106 		serverB.start();
107 
108 		// A plain HTTP forwarding proxy that requires Proxy-Authorization (user "u", password "p").
109 		proxy = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
110 		proxyPort = proxy.getAddress().getPort();
111 		proxy.createContext("/", ApacheHc5Transport_CallerClientRedirectCredentials_Test::proxy);
112 		proxy.start();
113 
114 		serverA = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
115 		portA = serverA.getAddress().getPort();
116 		serverA.createContext("/echo-creds", ApacheHc5Transport_CallerClientRedirectCredentials_Test::echoCreds);
117 		serverA.createContext("/redirect-cross", exchange -> redirect(exchange, "http://127.0.0.1:" + portB + "/echo-creds"));
118 		serverA.createContext("/redirect-same", exchange -> redirect(exchange, "/echo-creds"));
119 		// Same server and port, different host name, so a different origin.
120 		serverA.createContext("/redirect-host", exchange -> redirect(exchange, "http://localhost:" + portA + "/echo-creds"));
121 		serverA.createContext("/redirect-cross-back", exchange -> redirect(exchange, "http://127.0.0.1:" + portB + "/back"));
122 		serverA.createContext("/redirect-flaky", exchange -> redirect(exchange, "http://127.0.0.1:" + portB + "/flaky"));
123 		serverA.createContext("/redirect-challenge", exchange -> redirect(exchange, "http://127.0.0.1:" + portB + "/challenge"));
124 		serverA.createContext("/to-sse", exchange -> redirect(exchange, "/sse"));
125 		// Open-ended SSE stream: sends one event, then holds the connection open until the test class finishes.
126 		serverA.createContext("/sse", exchange -> {
127 			exchange.getResponseHeaders().add("Content-Type", "text/event-stream; charset=UTF-8");
128 			exchange.sendResponseHeaders(200, 0);
129 			exchange.getResponseBody().write("data: foo\n\n".getBytes(StandardCharsets.UTF_8));
130 			exchange.getResponseBody().flush();
131 			try {
132 				sseRelease.await(30, TimeUnit.SECONDS);
133 			} catch (InterruptedException e) {
134 				Thread.currentThread().interrupt();
135 			} finally {
136 				exchange.close();
137 			}
138 		});
139 		// The SSE handler holds its thread, so the other handlers need threads of their own.
140 		serverA.setExecutor(EXECUTOR);
141 		serverA.start();
142 	}

144 	@AfterAll
145 	static void stopServers() {
146 		sseRelease.countDown();
147 		if (serverA != null)
148 			serverA.stop(0);
149 		if (serverB != null)
150 			serverB.stop(0);
151 		if (proxy != null)
152 			proxy.stop(0);
153 		EXECUTOR.shutdownNow();
154 	}

163 	private static String all(HttpExchange exchange, String name) {
164 		var values = exchange.getRequestHeaders().get(name);
165 		return values == null || values.isEmpty() ? "none" : String.join(",", values);
166 	}

168 	private static String creds(HttpExchange exchange) {
169 		return "auth=" + all(exchange, "Authorization") + "|cookie=" + all(exchange, "Cookie")
170 			+ "|key=" + all(exchange, "X-API-Key") + "|trace=" + all(exchange, "X-Trace");
171 	}
172 
173 	private static void echoCreds(HttpExchange exchange) throws IOException {
174 		var body = creds(exchange).getBytes(StandardCharsets.UTF_8);
175 		exchange.getResponseHeaders().add("Content-Type", "text/plain");
176 		exchange.sendResponseHeaders(200, body.length);
177 		exchange.getResponseBody().write(body);
178 		exchange.close();
179 	}

181 	// Records each request it receives as "<target port> <Proxy-Authorization>", answers 407 without the right
182 	// Proxy-Authorization, and otherwise forwards the request (without Proxy-Authorization) to the absolute target URI.
183 	private static void proxy(HttpExchange exchange) throws IOException {
184 		var target = exchange.getRequestURI();
185 		var proxyAuth = all(exchange, "Proxy-Authorization");
186 		seenByProxy.add(target.getPort() + " " + proxyAuth);
187 		if (! proxyAuth.equals("Basic dTpw")) {
188 			exchange.getResponseHeaders().add("Proxy-Authenticate", "Basic realm=\"proxy\"");
189 			exchange.sendResponseHeaders(407, -1);
190 			exchange.close();
191 			return;
192 		}
193 		var connection = (HttpURLConnection)target.toURL().openConnection(Proxy.NO_PROXY);
194 		connection.setInstanceFollowRedirects(false);
195 		connection.setRequestMethod(exchange.getRequestMethod());
196 		for (var name : List.of("Authorization", "Cookie", "X-API-Key", "X-Trace")) {
197 			var values = exchange.getRequestHeaders().get(name);
198 			if (values != null)
199 				for (var value : values)
200 					connection.addRequestProperty(name, value);
201 		}
202 		var code = connection.getResponseCode();
203 		for (var name : List.of("Location", "Content-Type")) {
204 			var value = connection.getHeaderField(name);
205 			if (value != null)
206 				exchange.getResponseHeaders().add(name, value);
207 		}
208 		var in = code >= 400 ? connection.getErrorStream() : connection.getInputStream();
209 		var body = in == null ? new byte[0] : in.readAllBytes();
210 		exchange.sendResponseHeaders(code, body.length == 0 ? -1 : body.length);
211 		if (body.length > 0)
212 			exchange.getResponseBody().write(body);
213 		exchange.close();
214 	}

216 	private static void redirect(HttpExchange exchange, String location) throws IOException {
217 		exchange.getResponseHeaders().add("Location", location);
218 		exchange.sendResponseHeaders(302, -1);
219 		exchange.close();
220 	}

222 	// Returns the status code, followed by the body when the status is 200.
223 	private static String get(String path, String... nameValuePairs) throws Exception {
224 		return get(HttpClients.createDefault(), path, nameValuePairs);
225 	}
226 
227 	private static String get(CloseableHttpClient httpClient, String path, String... nameValuePairs) throws Exception {
228 		return get(httpClient, null, path, nameValuePairs);
229 	}
230 
231 	private static String get(CloseableHttpClient httpClient, Duration timeout, String path, String... nameValuePairs) throws Exception {
232 		var transport = ApacheHc5Transport.builder().httpClient(httpClient).build();
233 		try (var client = RestClient.builder().transport(transport).rootUrl("http://127.0.0.1:" + portA).build()) {
234 			var request = client.get(path);
235 			if (timeout != null)
236 				request.timeout(timeout);
237 			for (var i = 0; i < nameValuePairs.length; i += 2)
238 				request.header(nameValuePairs[i], nameValuePairs[i + 1]);
239 			try (var response = request.run()) {
240 				return response.getStatusCode() + (response.getStatusCode() == 200 ? " " + response.getBodyAsString() : "");
241 			}
242 		}
243 	}

318 	@Test
319 	void b06_crossOrigin_authChallengeResend_laxRedirectStrategy_credentialsStayStripped() throws Exception {
320 		// The other origin challenges, and the client holds credentials for it, so HttpClient resends the hop with its
321 		// headers reset to the unstripped copy; the resend must be stripped too.  Auth caching is off because HttpClient 5.6's
322 		// auth cache fails (NullPointerException) on this exchange, where the caller's own Authorization kept HttpClient from
323 		// choosing an auth scheme; that happens with or without the transport's context.
324 		var client = HttpClients.custom()
325 			.setRedirectStrategy(LaxRedirectStrategy.INSTANCE)
326 			.disableAuthCaching()
327 			.setDefaultCredentialsProvider(CredentialsProviderBuilder.create().add(new HttpHost("http", "127.0.0.1", portB), "u", "p".toCharArray()).build())
328 			.build();
329 		assertEquals(STRIPPED, get(client, "/redirect-challenge", CREDENTIALS));
330 		assertEquals(List.of("auth=none|cookie=none|key=none|trace=t-1", "auth=none|cookie=none|key=none|trace=t-1"), seenByB);
331 	}

333 	@Test
334 	void b07_crossOrigin_throughAuthenticatingProxy_laxRedirectStrategy_keepsProxyAuthorization() throws Exception {
335 		// On a plain (not tunnelled) proxied route, Proxy-Authorization authenticates to the proxy, which every hop goes
336 		// through, so the context keeps the one the client's CredentialsProvider adds; the other credentials are still
337 		// removed from the cross-origin hop.
338 		var client = HttpClients.custom()
339 			.setRedirectStrategy(LaxRedirectStrategy.INSTANCE)
340 			.setProxy(new HttpHost("http", "127.0.0.1", proxyPort))
341 			.setDefaultCredentialsProvider(CredentialsProviderBuilder.create().add(new HttpHost("http", "127.0.0.1", proxyPort), "u", "p".toCharArray()).build())
342 			.build();
343 		assertEquals(STRIPPED, get(client, "/redirect-cross", CREDENTIALS));
344 		assertEquals(1, hitsB.get());
345 		assertEquals(portB + " Basic dTpw", seenByProxy.get(seenByProxy.size() - 1));
346 	}
