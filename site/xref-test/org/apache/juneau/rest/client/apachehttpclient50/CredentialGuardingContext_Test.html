30   * Unit tests for {@link CredentialGuardingContext}: which hops it treats as another origin, and which records of a hop
31   * it inspects.  HttpClient itself drives it in {@code ApacheHc5Transport_CallerClientRedirectCredentials_Test}.

35  	private static HttpRequest request(String method, String uri) {
36  		var request = new BasicHttpRequest(method, java.net.URI.create(uri));
37  		request.addHeader("Authorization", "Bearer t");
38  		request.addHeader("Cookie", "s=1");
39  		request.addHeader("Proxy-Authorization", "Basic p");
40  		request.addHeader("X-API-Key", "k");
41  		request.addHeader("X-Trace", "t-1");
42  		return request;
43  	}
44  
45  	private static HttpRequest request(String uri) {
46  		return request("GET", uri);
47  	}
48  
49  	private static List<String> names(HttpRequest request) {
50  		var names = new ArrayList<String>();
51  		for (var h : request.getHeaders())
52  			names.add(h.getName());
53  		return names;
54  	}
55  
56  	private static final List<String> ALL = List.of("Authorization", "Cookie", "Proxy-Authorization", "X-API-Key", "X-Trace");
57  	private static final List<String> STRIPPED = List.of("X-Trace");
58  
59  	// Records the first request, then the hop, and returns the hop's remaining header names.
60  	private static List<String> hop(String from, String to) {
61  		var context = new CredentialGuardingContext();
62  		context.setRequest(request(from));
63  		var hop = request(to);
64  		context.setRequest(hop);
65  		return names(hop);
66  	}
67  
68  	// Same as hop(from, to), with the route the main exec stage records in the context before the hop.
69  	private static List<String> hop(String from, String to, HttpRoute route) {
70  		var context = new CredentialGuardingContext();
71  		context.setRequest(request(from));
72  		context.setRoute(route);
73  		var hop = request(to);
74  		context.setRequest(hop);
75  		return names(hop);
76  	}

156 	void a12_resendOfSameHop_inspectedAgain() {
157 		// After an authentication challenge the protocol stage resets the hop's headers to its unstripped copy and the main
158 		// stage records the same request object again; that record must strip again.
159 		var context = new CredentialGuardingContext();
160 		context.setRequest(request("http://a.example/x"));
161 		var hop = request("http://b.example/y");
162 		context.setRequest(hop);
163 		hop.addHeader("Authorization", "Bearer t");
164 		context.setRequest(hop);
165 		assertEquals(STRIPPED, names(hop));
166 	}

187 	@Test
188 	void a15_proxiedPlainRoute_keepsProxyAuthorization() {
189 		// Proxy-Authorization on a plain (not tunnelled) proxied route authenticates to the proxy, which every hop goes
190 		// through; stripping it would make the proxy answer 407 again.
191 		var route = new HttpRoute(TARGET_B, null, PROXY, false, TunnelType.PLAIN, LayerType.PLAIN);
192 		assertEquals(List.of("Proxy-Authorization", "X-Trace"), hop("http://a.example:8080/x", "http://b.example:8080/y", route));
193 	}

212 	@Test
213 	void a19_underscoreHostNames_comparedAsHosts() {
214 		// java.net.URI cannot parse a host name with an underscore; the context compares HttpClient's own host objects.
215 		assertEquals(ALL, underscoreHop("my_service", "my_service"));
216 		assertEquals(List.of("X-Trace"), underscoreHop("my_service", "other_service"));
217 	}
218 
219 	private static List<String> underscoreHop(String fromHost, String toHost) {
220 		var context = new CredentialGuardingContext();
221 		context.setRequest(new BasicHttpRequest("GET", new HttpHost("http", fromHost, 8080), "/x"));
222 		var hop = new BasicHttpRequest("GET", new HttpHost("http", toHost, 8080), "/y");
223 		hop.addHeader("Authorization", "Bearer t");
224 		hop.addHeader("Cookie", "s=1");
225 		hop.addHeader("Proxy-Authorization", "Basic p");
226 		hop.addHeader("X-API-Key", "k");
227 		hop.addHeader("X-Trace", "t-1");
228 		context.setRequest(hop);
229 		return names(hop);
230 	}
