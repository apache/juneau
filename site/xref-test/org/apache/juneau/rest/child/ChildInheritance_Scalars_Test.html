29  /**
30   * Phase 4 &mdash; proves the CHILD-WINS scalar shape for {@code @Child(defaultCharset=...)} and
31   * {@code @Child(maxInput=...)}, both of which resolve through {@code RestContext.mergeReplacedStringAttribute}
32   * (last-non-sentinel-wins over the annotation chain), exactly like {@code callLogger}/{@code partSerializer}/
33   * {@code partParser}.
34   *
35   * <p>
36   * {@code defaultAccept}/{@code defaultContentType} are intentionally NOT seed slots on {@code @Child} (see
37   * {@link Child} javadoc): they resolve through the {@code defaultRequestHeaders} memoizer's
38   * {@code HttpHeaderList.setDefault(...)} first-wins semantics rather than {@code mergeReplacedStringAttribute},
39   * which would make the host's seed win over the child's own explicit declaration &mdash; the opposite of the
40   * child-wins contract. That's pre-existing framework behavior unrelated to {@code @Child} (a plain
41   * superclass/subclass {@code @Rest} hierarchy exhibits the same ancestor-wins-over-descendant behavior), so
42   * these two members are deferred rather than shipped as a contract violation.
43   */
44  class ChildInheritance_Scalars_Test extends TestBase {
45  
46  	//==================================================================================================================
47  	// defaultCharset -- true child-wins (via mergeReplacedStringAttribute)
48  	//==================================================================================================================
49  
50  	@Rest(path="/nocharset")
51  	public static class ChildNoCharsetDeclared {
52  		@RestGet(path="/me") public String me() { return "me"; }
53  	}
54  
55  	@Rest(childrenDefs=@Child(type=ChildNoCharsetDeclared.class, defaultCharset="utf-16"))
56  	public static class HostSeedsCharset extends BasicRestServletGroup {
57  		private static final long serialVersionUID = 1L;
58  	}
59  
60  	@Test void a01_charset_childWithNoDeclarationReceivesSeed() {
61  		MockRestClient.buildLax(HostSeedsCharset.class);
62  		var hostCtx = RestContext.getGlobalRegistry().get(HostSeedsCharset.class);
63  		var childCtx = hostCtx.getRestChildren().asMap().get("nocharset");
64  		assertNotNull(childCtx);
65  		var opCtx = childCtx.getRestOperations().getOpContexts().get(0);
66  		assertEquals(StandardCharsets.UTF_16, opCtx.getDefaultCharset(),
67  			"Child with no defaultCharset declaration must receive the host's seeded utf-16");
68  	}
69  
70  	@Rest(path="/owncharset", defaultCharset="utf-16")
71  	public static class ChildWithOwnCharset {
72  		@RestGet(path="/me") public String me() { return "me"; }
73  	}
74  
75  	@Rest(childrenDefs=@Child(type=ChildWithOwnCharset.class, defaultCharset="iso-8859-1"))
76  	public static class HostSeedsCharsetButChildWins extends BasicRestServletGroup {
77  		private static final long serialVersionUID = 1L;
78  	}
79  
80  	@Test void a02_charset_childsOwnDeclarationWins() {
81  		MockRestClient.buildLax(HostSeedsCharsetButChildWins.class);
82  		var hostCtx = RestContext.getGlobalRegistry().get(HostSeedsCharsetButChildWins.class);
83  		var childCtx = hostCtx.getRestChildren().asMap().get("owncharset");
84  		assertNotNull(childCtx);
85  		var opCtx = childCtx.getRestOperations().getOpContexts().get(0);
86  		assertEquals(StandardCharsets.UTF_16, opCtx.getDefaultCharset(),
87  			"Child's own explicit defaultCharset must win over the host's seeded iso-8859-1");
88  	}
89  
90  	//==================================================================================================================
91  	// maxInput -- true child-wins (via mergeReplacedStringAttribute); seed-applies-when-silent already covered by
92  	// ChildContext_Seed_Test.a04 (seed vs. DefaultConfig fallback) -- this file adds the child-wins-when-explicit half.
93  	//==================================================================================================================
94  
95  	@Rest(path="/ownmax", maxInput="111")
96  	public static class ChildWithOwnMaxInput {
97  		@RestGet(path="/me") public String me() { return "me"; }
98  	}
99  
100 	@Rest(childrenDefs=@Child(type=ChildWithOwnMaxInput.class, maxInput="222"))
101 	public static class HostSeedsMaxInputButChildWins extends BasicRestServletGroup {
102 		private static final long serialVersionUID = 1L;
103 	}
104 
105 	@Test void a03_maxInput_childsOwnDeclarationWins() {
106 		MockRestClient.buildLax(HostSeedsMaxInputButChildWins.class);
107 		var hostCtx = RestContext.getGlobalRegistry().get(HostSeedsMaxInputButChildWins.class);
108 		var childCtx = hostCtx.getRestChildren().asMap().get("ownmax");
109 		assertNotNull(childCtx);
110 		var opCtx = childCtx.getRestOperations().getOpContexts().get(0);
111 		assertEquals(111L, opCtx.getMaxInput(),
112 			"Child's own explicit maxInput must win over the host's seeded 222");
113 	}
114 }
