17  package org.apache.juneau.rest.reactive;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.concurrent.*;
22  
23  import org.apache.juneau.marshall.json.*;
24  import org.apache.juneau.rest.mock.classic.*;
25  import org.apache.juneau.rest.server.*;
26  import org.junit.jupiter.api.*;
27  
28  /**
29   * Verifies that a <b>bare</b> {@code juneau-rest-server} (no {@code juneau-rest-server-reactive} /
30   * {@code juneau-rest-server-reactive-reactor} module on the classpath) has <b>zero</b> reactive behavior.
31   *
32   * <p>
33   * This module's test classpath intentionally has no reactive module, so {@code RestContext}'s
34   * {@link java.util.ServiceLoader}-based response-processor discovery finds no provider and
35   * {@code ReactiveResponseProcessor} is never added to the chain. A {@code @RestOp} returning a
36   * {@link Flow.Publisher Flow.Publisher} is therefore treated as an ordinary POJO
37   * return value and serialized through the normal serializer chain &mdash; it is NOT collected into a
38   * JSON array (the reactive buffer shape) and NOT streamed.
39   *
40   * <p>
41   * The positive activation tests live in {@code juneau-integration-tests} (where the reactive module IS on the
42   * classpath); this negative test cannot live there because the module's presence auto-activates the
43   * processor for every resource.
44   */
45  @SuppressWarnings({
46  	"resource" // Closeable test fixtures are intentionally held/unassigned; lifecycle is managed by the test framework.
47  })
48  class ReactiveOptIn_BareServer_Test {
49  
50  	public static final class Pojo {
51  		public final String name;
52  		public final int value;
53  		public Pojo(String name, int value) { this.name = name; this.value = value; }
54  	}
55  
56  	/** Minimal JDK Flow.Publisher; on a bare server it is never subscribed to (no reactive processor). */
57  	static final class ListPublisher<T> implements Flow.Publisher<T> {
58  		@Override public void subscribe(Flow.Subscriber<? super T> sub) {
59  			sub.onSubscribe(new Flow.Subscription() {
60  				@Override public void request(long n) { /* no-op */ }
61  				@Override public void cancel() { /* no-op */ }
62  			});
63  		}
64  	}
65  
66  	@Rest(serializers = JsonSerializer.class)
67  	public static class A {
68  		@RestGet("/flux")
69  		public Flow.Publisher<Pojo> flux() {
70  			return new ListPublisher<>();
71  		}
72  	}
73  
74  	private static final MockRestClient CA = MockRestClient.buildLax(A.class);
75  
76  	@Test void a01_bareServer_doesNotProcessFlowPublisherReactively() throws Exception {
77  		String content;
78  		try (var req = CA.get("/flux")) {
79  			req.accept("application/json");
80  			try (var res = req.run()) {
81  				content = res.getContent().asString();
82  			}
83  		}
84  		// Reactive (buffer-shape) handling would have subscribed to the publisher and produced a JSON
85  		// array of the emitted Pojo elements. With no reactive module on the classpath the publisher is
86  		// instead serialized as a plain (property-less) bean — so the element payload must be absent.
87  		assertFalse(content.contains("\"name\""),
88  			"Bare juneau-rest-server must not reactively process Flow.Publisher returns. Body was: " + content);
89  		assertFalse(content.startsWith("["),
90  			"Bare juneau-rest-server must not buffer a Flow.Publisher into a JSON array. Body was: " + content);
91  	}
92  }
