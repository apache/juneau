18  
19  import org.apache.juneau.*;
20  import org.apache.juneau.rest.mock.classic.*;
21  import org.apache.juneau.rest.server.*;
22  import org.apache.juneau.rest.server.guard.*;
23  import org.apache.juneau.rest.server.servlet.*;
24  import org.junit.jupiter.api.*;
25  
26  /**
27   * Phase 2 regression matrix &mdash; verifies that {@code @Rest(guards=...)} on a mixin class inherits the host's
28   * guard chain by default (security-conservative default: surprises around security err on "too strict") and that
29   * {@code noInherit="guards"} removes host guards from the mixin context.
30   *
31   * <p>
32   * Two scenarios per resource exercise the matrix:
33   * <ul>
34   * 	<li><b>Inherit-by-default protection.</b> A host with {@code @Rest(guards=AllowOnlyHostHeader)} automatically
35   * 		protects mixin endpoints. A request without the host header receives 403 from the mixin endpoint just as
36   * 		it would from the host endpoint.</li>
37   * 	<li><b>noInherit opt-out.</b> A mixin with {@code noInherit="guards"} bypasses the host's guards entirely;
38   * 		the mixin endpoint accepts the request regardless of the host header, while the host endpoint remains
39   * 		guarded.</li>
40   * </ul>
41   */
42  class MixinInheritance_Guards_Test extends TestBase {
43  
44  	public static class AllowOnlyHostHeader extends RestGuard {
45  		@Override public boolean isRequestAllowed(RestRequest req) {
46  			return "yes".equals(req.getHeaderParam("X-Host-Allowed").orElse(null));
47  		}
48  	}
49  
50  	public static class AllowOnlyMixinHeader extends RestGuard {
51  		@Override public boolean isRequestAllowed(RestRequest req) {
52  			return "yes".equals(req.getHeaderParam("X-Mixin-Allowed").orElse(null));
53  		}
54  	}
55  
56  	@Rest
57  	public static class M_InheritsHostGuard {
58  		@RestGet(path="/inh") public String inh() { return "inh"; }
59  	}
60  
61  	@Rest(guards={AllowOnlyMixinHeader.class})
62  	public static class M_AppendsMixinGuard {
63  		@RestGet(path="/app") public String app() { return "app"; }
64  	}
65  
66  	@Rest(noInherit="guards")
67  	public static class M_NoInheritGuard {
68  		@RestGet(path="/free") public String free() { return "free"; }
69  	}
70  
71  	@Rest(guards={AllowOnlyHostHeader.class}, mixins={M_InheritsHostGuard.class, M_AppendsMixinGuard.class, M_NoInheritGuard.class})
72  	public static class Host extends BasicRestServlet {
73  		private static final long serialVersionUID = 1L;
74  		@RestGet(path="/h") public String h() { return "h"; }
75  	}
76  
77  	@Test void a01_hostEndpointGuarded() throws Exception {
78  		var c = MockRestClient.buildLax(Host.class);
79  		c.get("/h").run().assertStatus(403);
80  		c.get("/h").header("X-Host-Allowed", "yes").accept("text/plain").run().assertStatus(200).assertContent("h");
81  	}
82  
83  	@Test void a02_mixinInheritsHostGuardByDefault() throws Exception {
84  		var c = MockRestClient.buildLax(Host.class);
85  		c.get("/inh").run().assertStatus(403);
86  		c.get("/inh").header("X-Host-Allowed", "yes").accept("text/plain").run().assertStatus(200).assertContent("inh");
87  	}
88  
89  	@Test void a03_mixinAppendsGuardAfterInheritedHost() throws Exception {
90  		var c = MockRestClient.buildLax(Host.class);
91  		c.get("/app").run().assertStatus(403);
92  		c.get("/app").header("X-Host-Allowed", "yes").run().assertStatus(403);
93  		c.get("/app").header("X-Mixin-Allowed", "yes").run().assertStatus(403);
94  		c.get("/app").header("X-Host-Allowed", "yes").header("X-Mixin-Allowed", "yes").accept("text/plain").run().assertStatus(200).assertContent("app");
95  	}
96  
97  	@Test void a04_noInheritDropsHostGuardFromMixin() throws Exception {
98  		var c = MockRestClient.buildLax(Host.class);
99  		c.get("/free").accept("text/plain").run().assertStatus(200).assertContent("free");
100 		c.get("/h").run().assertStatus(403);
101 	}
102 }
