28  /**
29   * Phase 2 — exercises {@code @Rest(noInherit={"serializers"})} on a mixin class to verify it blocks the
30   * parent-context walk for that property.
31   *
32   * <p>
33   * Uses uniquely-named test serializers so the {@link BasicRestServlet} default set can't accidentally mask the
34   * assertion that the host's serializers are excluded from the mixin's chain.
35   *
36   * <p>
37   * Acceptance:
38   * <ul>
39   * 	<li>A mixin declaring {@code noInherit={"serializers"}} sees ONLY its own serializers; the host's declared
40   * 		serializer (HostS1) does NOT bleed through.</li>
41   * 	<li>The host's serializer set remains unchanged by the mixin's {@code noInherit} declaration (host still has
42   * 		HostS1; mixin's MixinS1 does NOT leak into the host's set).</li>
43   * </ul>
44   */
45  class MixinInheritance_NoInherit_Test extends TestBase {
46  
47  	public static class HostS1 extends FakeWriterSerializer {
48  		public HostS1(FakeWriterSerializer.Builder b) { super(b.produces("text/host-s1")); }
49  	}
50  
51  	public static class MixinS1 extends FakeWriterSerializer {
52  		public MixinS1(FakeWriterSerializer.Builder b) { super(b.produces("text/mixin-s1")); }
53  	}
54  
55  	@Rest(
56  		noInherit = "serializers",
57  		serializers = {MixinS1.class}
58  	)
59  	public static class M_MixinS1Only {
60  		@RestGet(path="/my") public String my() { return "my"; }
61  	}
62  
63  	@Rest(serializers={HostS1.class}, mixins={M_MixinS1Only.class})
64  	public static class Host extends BasicRestServlet {
65  		private static final long serialVersionUID = 1L;
66  		@RestGet(path="/h") public String h() { return "h"; }
67  	}
68  
69  	@Test void a01_noInheritBlocksParentSerializerWalk() {
70  		MockRestClient.buildLax(Host.class);
71  		var hostCtx = RestContext.getGlobalRegistry().get(Host.class);
72  		var mixinCtx = hostCtx.getMixinContexts().get(M_MixinS1Only.class);
73  		assertNotNull(mixinCtx);
74  
75  		assertTrue(mixinCtx.getSerializers().getSerializer("text/mixin-s1").isPresent(),
76  			"Mixin's own MixinS1 must be present");
77  		assertTrue(mixinCtx.getSerializers().getSerializer("text/host-s1").isEmpty(),
78  			"Mixin with noInherit=\"serializers\" must NOT see the host's HostS1 (parent walk blocked)");
79  	}
80  
81  	@Test void a02_hostUnaffectedByMixinNoInherit() {
82  		MockRestClient.buildLax(Host.class);
83  		var hostCtx = RestContext.getGlobalRegistry().get(Host.class);
84  
85  		assertTrue(hostCtx.getSerializers().getSerializer("text/host-s1").isPresent(),
86  			"Host must retain its declared HostS1 serializer regardless of mixin's noInherit");
87  		assertTrue(hostCtx.getSerializers().getSerializer("text/mixin-s1").isEmpty(),
88  			"Host must NOT pick up MixinS1 from a noInherit-isolated mixin");
89  	}
90  }
