26  /**
27   * Feature tests for the {@link Rest#mixinDefs() @Rest(mixinDefs=@Mixin(...))} host-side override mechanism.
28   *
29   * <p>
30   * Verifies the motivating capability &mdash; a host declaring {@code @Mixin(type=X, guards=AdminGuard)}
31   * applies {@code AdminGuard} to {@code X}'s mixed-in endpoints without {@code X} declaring it &mdash; plus the
32   * resolution rules: host-override + inherited host chain both apply (list-shaped append), the {@code @Mixin}'s
33   * own {@code noInherit} cuts the inherited chain, replace-shaped slots win, host-chosen {@code path} re-mount,
34   * transitive-mixin non-propagation, and bare-class &equiv; empty-override {@code @Mixin} equivalence.
35   */
36  class MixinDefs_Overrides_Test extends TestBase {
37  
38  	public static class AllowOnlyAdmin extends RestGuard {
39  		@Override public boolean isRequestAllowed(RestRequest req) {
40  			return "yes".equals(req.getHeaderParam("X-Admin").orElse(null));
41  		}
42  	}
43  
44  	public static class AllowOnlyHost extends RestGuard {
45  		@Override public boolean isRequestAllowed(RestRequest req) {
46  			return "yes".equals(req.getHeaderParam("X-Host").orElse(null));
47  		}
48  	}
49  
50  	// An unguarded, plain mixin that declares nothing special — overrides come from the host @Mixin.
51  	@Rest
52  	public static class PlainMixin {
53  		@RestGet(path="/plain") public String plain() { return "plain"; }
54  	}

56  	// =================================================================================
57  	// A. Host @Mixin(guards=...) applies a guard the mixin never declared.
58  	// =================================================================================
59  
60  	@Rest(mixinDefs=@Mixin(type=PlainMixin.class, guards=AllowOnlyAdmin.class))
61  	public static class A extends BasicRestServlet {
62  		private static final long serialVersionUID = 1L;
63  		@RestGet(path="/h") public String h() { return "h"; }
64  	}
65  
66  	@Test void a01_hostMixinGuardAppliesToMixinEndpoint() throws Exception {
67  		var c = MockRestClient.buildLax(A.class);
68  		c.get("/plain").run().assertStatus(403);
69  		c.get("/plain").header("X-Admin", "yes").accept("text/plain").run().assertStatus(200).assertContent("plain");
70  	}
71  
72  	@Test void a02_hostEndpointUnaffectedByMixinOverride() throws Exception {
73  		// The override applies only to the mixin's endpoints, not the host's own.
74  		var c = MockRestClient.buildLax(A.class);
75  		c.get("/h").accept("text/plain").run().assertStatus(200).assertContent("h");
76  	}

79  	// B. Override + inherited host guard chain BOTH apply (list-shaped append).
80  	// =================================================================================
81  
82  	@Rest(guards=AllowOnlyHost.class, mixinDefs=@Mixin(type=PlainMixin.class, guards=AllowOnlyAdmin.class))
83  	public static class B extends BasicRestServlet {
84  		private static final long serialVersionUID = 1L;
85  	}
86  
87  	@Test void b01_bothHostChainAndOverrideApply() throws Exception {
88  		var c = MockRestClient.buildLax(B.class);
89  		c.get("/plain").run().assertStatus(403);
90  		c.get("/plain").header("X-Host", "yes").run().assertStatus(403);            // missing admin
91  		c.get("/plain").header("X-Admin", "yes").run().assertStatus(403);           // missing host
92  		c.get("/plain").header("X-Host", "yes").header("X-Admin", "yes").accept("text/plain").run().assertStatus(200).assertContent("plain");
93  	}

99  	@Rest(guards=AllowOnlyHost.class, mixinDefs=@Mixin(type=PlainMixin.class, guards=AllowOnlyAdmin.class, noInherit="guards"))
100 	public static class C extends BasicRestServlet {
101 		private static final long serialVersionUID = 1L;
102 	}
103 
104 	@Test void c01_noInheritCutsHostChainButKeepsOverride() throws Exception {
105 		var c = MockRestClient.buildLax(C.class);
106 		// Host guard cut -> only the @Mixin's own AllowOnlyAdmin applies.
107 		c.get("/plain").header("X-Host", "yes").run().assertStatus(403);            // host header no longer sufficient/required
108 		c.get("/plain").header("X-Admin", "yes").accept("text/plain").run().assertStatus(200).assertContent("plain");
109 	}

181 	// G. Transitive non-propagation — override on the directly-named mixin does NOT reach a transitive mixin.
182 	// =================================================================================
183 
184 	@Rest
185 	public static class InnerMixin {
186 		@RestGet(path="/inner") public String inner() { return "inner"; }
187 	}
188 
189 	// OuterMixin pulls in InnerMixin transitively (flat-linked to the host).
190 	@Rest(mixins=InnerMixin.class)
191 	public static class OuterMixin {
192 		@RestGet(path="/outer") public String outer() { return "outer"; }
193 	}
194 
195 	// Host overrides guards for OuterMixin only; InnerMixin must remain unguarded.
196 	@Rest(mixinDefs=@Mixin(type=OuterMixin.class, guards=AllowOnlyAdmin.class))
197 	public static class G extends BasicRestServlet {
198 		private static final long serialVersionUID = 1L;
199 	}
200 
201 	@Test void g01_overrideAppliesToNamedMixin() throws Exception {
202 		var c = MockRestClient.buildLax(G.class);
203 		c.get("/outer").run().assertStatus(403);
204 		c.get("/outer").header("X-Admin", "yes").accept("text/plain").run().assertStatus(200).assertContent("outer");
205 	}
206 
207 	@Test void g02_overrideDoesNotPropagateToTransitiveMixin() throws Exception {
208 		var c = MockRestClient.buildLax(G.class);
209 		// InnerMixin was not directly named in a @Mixin override, so it stays unguarded.
210 		c.get("/inner").accept("text/plain").run().assertStatus(200).assertContent("inner");
211 	}

297 	// mixinDefs (rich) listed BEFORE the bare mixins= entry for the same class: the rich entry must win
298 	// (a later bare occurrence must not downgrade it).  Exercises the "already discovered, rm is bare" path.
299 	@Rest(mixinDefs=@Mixin(type=PlainMixin.class, guards=AllowOnlyAdmin.class), mixins=PlainMixin.class)
300 	public static class M extends BasicRestServlet {
301 		private static final long serialVersionUID = 1L;
302 	}
303 
304 	@Test void m01_richBeforeBareNotDowngraded() throws Exception {
305 		var c = MockRestClient.buildLax(M.class);
306 		// Rich @Mixin(guards=...) declared first; the later bare entry for the same class does not strip the guard.
307 		c.get("/plain").run().assertStatus(403);
308 		c.get("/plain").header("X-Admin", "yes").accept("text/plain").run().assertStatus(200).assertContent("plain");
309 	}
