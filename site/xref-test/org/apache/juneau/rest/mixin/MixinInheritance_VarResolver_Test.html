27  /**
28   * Phase 2 smoke &mdash; verifies that a mixin sub-context's var resolver reaches the host's Messages bundle
29   * for {@code $L{key}}-style lookups.
30   *
31   * <p>
32   * The mixin's {@link RestContext#getVarResolver() var resolver} is built from the bootstrap resolver plus the
33   * mixin's own {@link org.apache.juneau.marshall.cp.Messages} bean.  Because the mixin's Messages bundle is now chained
34   * to the host's via {@code Messages.chain(...)}, {@code $L{host.key}} resolves through the inherited host
35   * bundle from the mixin context.  This test is a smoke check on that end-to-end seam &mdash; the deeper
36   * acceptance criteria are covered by {@code MixinInheritance_Messages_Test}.
37   */
38  class MixinInheritance_VarResolver_Test extends TestBase {
39  
40  	@Rest
41  	public static class M_NoMessages {
42  		@RestGet(path="/me") public String me() { return "me"; }
43  	}
44  
45  	@Rest(messages="HostMessages", mixins={M_NoMessages.class})
46  	public static class HostWithHostMessages extends BasicRestServlet {
47  		private static final long serialVersionUID = 1L;
48  		@RestGet(path="/h") public String h() { return "h"; }
49  	}
50  
51  	@Test void a01_mixinVarResolverSeesHostLocalizedMessages() {
52  		MockRestClient.buildLax(HostWithHostMessages.class);
53  		var hostCtx = RestContext.getGlobalRegistry().get(HostWithHostMessages.class);
54  		var mixinCtx = hostCtx.getMixinContexts().get(M_NoMessages.class);
55  		assertNotNull(mixinCtx);
56  
57  		assertEquals("host-value", hostCtx.getVarResolver().resolve("$L{host.key}"),
58  			"Host var resolver must resolve $L{host.key} from its own bundle");
59  		assertEquals("host-value", mixinCtx.getVarResolver().resolve("$L{host.key}"),
60  			"Mixin var resolver must resolve $L{host.key} via the inherited host Messages chain");
61  	}
62  }
