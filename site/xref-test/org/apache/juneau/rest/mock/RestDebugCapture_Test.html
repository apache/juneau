50  	 * Sets a level on the named logger for the duration of a test.  Stock JUL leaves a resource logger at the root's
51  	 * inherited {@code INFO}, which no longer emits the per-request line; a level must be configured explicitly.
52  	 */
53  	private static final class ConfiguredLevel implements AutoCloseable {
54  		private final Logger logger;  // strong reference: LogManager holds loggers weakly
55  		private final Level previous;
56  
57  		ConfiguredLevel(Class<?> c, Level level) {
58  			logger = Logger.getLogger(c.getName());
59  			previous = logger.getLevel();
60  			logger.setLevel(level);
61  		}
62  
63  		@Override public void close() {
64  			logger.setLevel(previous);
65  		}
66  	}

164 		try (var lvl = new ConfiguredLevel(B_HostB.class, Level.INFO);

166 				var cb = RichLogger.getLogger(B_HostB.class).captureEvents(Level.FINEST)) {

168 			// .debug() raises ONLY B_HostA's own class logger to FINEST for the duration of the call.

179 			assertEquals(Level.INFO, opA.getLevel());
180 			assertTrue(opA.getMessage().contains("---Request Headers---"),
181 				"HostA debug path should still resolve a fine-grained tier (headers/body), even though emitted records are INFO");

186 			assertNotNull(opB, "expected a mixin op-logger record named <HostB>.who");
187 			assertEquals(Level.INFO, opB.getLevel(),

252 		@Override

255 				records.add(logRecord);

257

259 		public void flush() {
260 			// No-op: records are collected in memory, so there is nothing to flush.
261 		}

264 		public void close() {
265 			// No-op: this handler holds no closeable resources.
266 		}
267

269 			return records;
270 		}

277 	@Test void d00_realHandlerVisibilityTracksHandlerThreshold_notTierLevel() throws Exception {
278 		var logger = Logger.getLogger(D_Resource.class.getName());
279 		var prevLevel = logger.getLevel();
280 		var prevUseParentHandlers = logger.getUseParentHandlers();
281 		var prevHandlers = logger.getHandlers();
282 		var handler = new D00_CollectingHandler();
283 		try {
284 			// Body dumping is off by default; force it on so the FINEST body-visibility proof still holds.
285 			forceOn();
286 			logger.setUseParentHandlers(false);
287 			for (var h : prevHandlers)
288 				logger.removeHandler(h);
289 			handler.setLevel(Level.INFO);
290 			logger.addHandler(handler);
291 			logger.setLevel(Level.FINEST);
292 
293 			var client = org.apache.juneau.rest.mock.classic.MockRestClient.create(D_Resource.class).build();
294 			// Renderable Content-Type so the opted-in body can actually be dumped (absent CT ⇒ non-renderable placeholder).
295 			client.post("/echo", "real-handler-secret").header("Content-Type", "text/plain").run().assertContent("real-handler-secret");
296 
297 			assertSize(() -> "effective logger FINEST with INFO handler should publish one INFO-stamped debug record", 1, handler.records());
298 			var finestRecord = handler.records().get(0);
299 			assertEquals(Level.INFO, finestRecord.getLevel());
300 			assertTrue(finestRecord.getMessage().contains("real-handler-secret"),
301 				"FINEST detail should still render request/response body content when the operator has opted in");
302 
303 			handler.clear();
304 			handler.setLevel(Level.WARNING);
305 			client.post("/echo", "real-handler-secret").header("Content-Type", "text/plain").run().assertContent("real-handler-secret");
306 			assertTrue(handler.records().isEmpty(),
307 				"INFO-stamped records must be filtered by handlers above INFO");
308 
309 			handler.setLevel(Level.INFO);
310 			logger.setLevel(Level.INFO);
311 			client.post("/echo", "real-handler-secret").header("Content-Type", "text/plain").run().assertContent("real-handler-secret");
312 
313 			assertSize(() -> "logger INFO + handler INFO should preserve the prior single visible basic-line behavior", 1, handler.records());
314 			var infoRecord = handler.records().get(0);
315 			assertEquals(Level.INFO, infoRecord.getLevel());
316 			assertTrue(infoRecord.getMessage().contains("[200] HTTP POST /twophase/echo"), infoRecord.getMessage());
317 			assertFalse(infoRecord.getMessage().contains("real-handler-secret"),
318 				"INFO tier does not install capture wrappers, so the body must not render");
319 		} finally {
320 			logger.removeHandler(handler);
321 			for (var h : prevHandlers)
322 				logger.addHandler(h);
323 			logger.setUseParentHandlers(prevUseParentHandlers);
324 			logger.setLevel(prevLevel);
325 		}
326 	}

629 	private List<java.util.logging.LogRecord> e_run(Level level) throws Exception {
630 		var logger = Logger.getLogger(E_Resource.class.getName());
631 		var prevLevel = logger.getLevel();
632 		var prevUseParentHandlers = logger.getUseParentHandlers();
633 		var handler = new D00_CollectingHandler();
634 		try {
635 			logger.setUseParentHandlers(false);
636 			handler.setLevel(Level.ALL);
637 			logger.addHandler(handler);
638 			logger.setLevel(level);
639 			org.apache.juneau.rest.mock.classic.MockRestClient.create(E_Resource.class).build().get("/who").run().assertStatus().asCode().is(200);
640 			return new ArrayList<>(handler.records());
641 		} finally {
642 			logger.removeHandler(handler);
643 			logger.setUseParentHandlers(prevUseParentHandlers);
644 			logger.setLevel(prevLevel);
645 		}
646 	}
