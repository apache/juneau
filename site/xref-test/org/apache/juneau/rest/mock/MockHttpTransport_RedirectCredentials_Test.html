17  package org.apache.juneau.rest.mock;
18  
19  import static org.apache.juneau.test.bct.BctAssertions.*;
20  
21  import java.io.*;
22  import java.util.*;
23  
24  import org.apache.juneau.rest.client.*;
25  import org.junit.jupiter.api.*;
26  
27  /**
28   * Verifies that {@link MockHttpTransport}'s policy-covered redirect loop does not forward caller-set credential
29   * headers to a different origin, while still forwarding them on a same-origin redirect.
30   */
31  @SuppressWarnings({
32  	"resource" // Transport and TransportResponse instances are short-lived in-memory test fixtures.
33  })
34  class MockHttpTransport_RedirectCredentials_Test {
35  
36  	private static MockHttpTransport transport() {
37  		return MockHttpTransport.builder()
38  			.recordRequests()
39  			.on("GET", "/cross", req -> redirect("http://b.example.com/echo"))
40  			.on("GET", "/downgrade", req -> redirect("http://a.example.com/echo"))
41  			.on("GET", "/same", req -> redirect("/echo"))
42  			.on("GET", "/echo", req -> TransportResponse.builder().statusCode(200).body(new ByteArrayInputStream(new byte[0])).build())
43  			.build();
44  	}
45  
46  	private static TransportResponse redirect(String location) {
47  		return TransportResponse.builder().statusCode(302).header("Location", location).build();
48  	}
49  
50  	private static TransportRequest request(String uri) {
51  		return TransportRequest.builder()
52  			.method("GET")
53  			.uri(uri)
54  			.remoteUrlPolicy(true, false)
55  			.header("Authorization", "Bearer secret-token")
56  			.header("Cookie", "session=abc123")
57  			.header("X-API-Key", "key-123")
58  			.header("X-Trace", "t-1")
59  			.build();
60  	}
61  
62  	private static List<String> headerNames(TransportRequest r) {
63  		return r.getHeaders().stream().map(TransportHeader::name).toList();
64  	}
65  
66  	@Test void a01_crossOriginRedirect_dropsCredentialHeaders() throws Exception {
67  		var t = transport();
68  		assertBean(t.execute(request("http://a.example.com/cross")), "statusCode", "200");
69  		var hops = t.getRecordedRequests();
70  		assertList(hops.stream().map(TransportRequest::getUri).toList(), "http://a.example.com/cross", "http://b.example.com/echo");
71  		assertList(headerNames(hops.get(1)), "X-Trace");
72  	}
73  
74  	@Test void a02_httpsToHttpDowngrade_dropsCredentialHeaders() throws Exception {
75  		var t = transport();
76  		t.execute(request("https://a.example.com/downgrade"));
77  		assertList(headerNames(t.getRecordedRequests().get(1)), "X-Trace");
78  	}
79  
80  	@Test void a03_sameOriginRedirect_keepsCredentialHeaders() throws Exception {
81  		var t = transport();
82  		t.execute(request("http://a.example.com/same"));
83  		var hops = t.getRecordedRequests();
84  		assertBean(hops.get(1), "uri", "http://a.example.com/echo");
85  		assertList(headerNames(hops.get(1)), "Authorization", "Cookie", "X-API-Key", "X-Trace");
86  	}
87  
88  	@Test void a04_notPolicyCovered_noRedirectLoop() throws Exception {
89  		// Without the @Remote URL policy the mock transport dispatches once and returns the 302 to the caller.
90  		var t = transport();
91  		var r = TransportRequest.builder().method("GET").uri("http://a.example.com/cross").header("Authorization", "x").build();
92  		assertBean(t.execute(r), "statusCode", "302");
93  		assertList(t.getRecordedRequests().stream().map(TransportRequest::getUri).toList(), "http://a.example.com/cross");
94  	}
95  }
