175 	@Test void c01_getContentLength_nullContentBranch() {
176 		// Setting content with `nn(value)` false produces a null content via no-arg path.
177 		// The simplest way to hit the null-content branch is via reflection-free: call content(null).
178 		// content(null) takes no path so leaves the existing empty-array, but we can pass an
179 		// InputStream that reads zero bytes after content was previously set to null.  Easier:
180 		// set the content to a byte[] of length 0 and check return; null branch uses ternary.
181 		var req = MockServletRequest.create();
182 		// Force content to be non-empty first, then verify length-check returns >= 0.
183 		req.content("");  // sets empty bytes
184 		assertEquals(0, req.getContentLength());
185 		assertEquals(0L, req.getContentLengthLong());
186 	}

189 		var req = MockServletRequest.create();
190 		req.content(null); // nn(value) is false; content remains the default empty byte[]
191 		assertEquals(0, req.getContentLength());
192 	}

294 	@Test void i02_getQueryString_keyOnlyEntryNoEquals() {
295 		// `flag` (no value) -> queryDataMap entry with null value -> branch v == null in lambda.
296 		// Two key-only entries to exercise both `sb.isEmpty()` arms of the ternary in line 474.
297 		var req = MockServletRequest.create().method("GET").uri("/p?flag1&flag2");
298 		req.queryString(null);
299 		var qs = req.getQueryString();
300 		assertNotNull(qs);
301 		assertTrue(qs.contains("flag1"));
302 		assertTrue(qs.contains("flag2"));
303 		// No '=' sign for key-only entries.
304 		assertFalse(qs.contains("flag1="));
305 		assertFalse(qs.contains("flag2="));
306 		// Verify they are joined by `&`.
307 		assertTrue(qs.contains("&"));
308 	}
