173 	@Test void c01_getContentLength_nullContentBranch() {
174 		// Setting content with `nn(value)` false produces a null content via no-arg path.
175 		// The simplest way to hit the null-content branch is via reflection-free: call content(null).
176 		// content(null) takes no path so leaves the existing empty-array, but we can pass an
177 		// InputStream that reads zero bytes after content was previously set to null.  Easier:
178 		// set the content to a byte[] of length 0 and check return; null branch uses ternary.
179 		var req = MockServletRequest.create();
180 		// Force content to be non-empty first, then verify length-check returns >= 0.
181 		req.content("");  // sets empty bytes
182 		assertEquals(0, req.getContentLength());
183 		assertEquals(0L, req.getContentLengthLong());
184 	}

190 	}

292 	@Test void i02_getQueryString_keyOnlyEntryNoEquals() {
293 		// `flag` (no value) -> queryDataMap entry with null value -> branch v == null in lambda.
294 		// Two key-only entries to exercise both `sb.isEmpty()` arms of the ternary in line 474.
295 		var req = MockServletRequest.create().method("GET").uri("/p?flag1&flag2");
296 		req.queryString(null);
297 		var qs = req.getQueryString();
298 		assertNotNull(qs);
299 		assertTrue(qs.contains("flag1"));
300 		assertTrue(qs.contains("flag2"));
301 		// No '=' sign for key-only entries.
302 		assertFalse(qs.contains("flag1="));
303 		assertFalse(qs.contains("flag2="));
304 		// Verify they are joined by `&`.
305 		assertTrue(qs.contains("&"));
306 	}
