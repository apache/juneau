31  /**
32   * Verifies that the <c>No-Trace</c> request header and the <js>"NoTrace"</js> request attribute keep the stack trace off
33   * the access-log record of a failed call, and that the next-gen {@link MockRestClient} can send the header from its
34   * builder and per request.
35   *
36   * @since 10.0.0
37   */
38  @SuppressWarnings({
39  	"resource" // MockRestClient instances are short-lived test fixtures.
40  })
41  class NoTrace_Test {
42  
43  	@Rest(path="/nt")
44  	public static class A_Resource {
45  		@RestGet(path="/bad")
46  		public String bad(@Query(name="n") int n) {
47  			return "x" + n;
48  		}
49  
50  		@RestGet(path="/failPlain")
51  		public String failPlain(RestResponse res) {
52  			res.setException(new NotFound("expected"));
53  			return "x";
54  		}
55  
56  		@RestGet(path="/failAttrFalse")
57  		public String failAttrFalse(RestResponse res) {
58  			res.setNoTrace(false);
59  			res.setException(new NotFound("expected"));
60  			return "x";
61  		}
62  
63  		@RestGet(path="/failAttr")
64  		public String failAttr(RestResponse res) {
65  			res.setNoTrace();
66  			res.setException(new NotFound("expected"));
67  			return "x";
68  		}
69  	}
70  
71  	private static java.util.logging.LogRecord capture(java.util.function.Consumer<MockRestClient> call, MockRestClient client) {
72  		var logger = java.util.logging.Logger.getLogger(A_Resource.class.getName());  // strong reference: LogManager holds loggers weakly
73  		var previous = logger.getLevel();
74  		// Stock JUL leaves the resource logger at the root's inherited INFO, which no longer emits the per-request record.
75  		logger.setLevel(Level.INFO);
76  		try (var c = RichLogger.getLogger(A_Resource.class).captureEvents(Level.FINEST)) {
77  			call.accept(client);
78  			return c.getRecords().stream().filter(r -> r.getLoggerName() != null && r.getLoggerName().startsWith(A_Resource.class.getName())).findFirst().orElseThrow();
79  		} finally {
80  			logger.setLevel(previous);
81  		}
82  	}
83  
84  	private static void run(MockRestClient client, String path, boolean noTrace) {
85  		try {
86  			var req = client.get(path);
87  			if (noTrace)
88  				req.noTrace();
89  			req.run();
90  		} catch (Exception e) {
91  			throw new AssertionError(e);
92  		}
93  	}
94  
95  	@ParameterizedTest
96  	@CsvSource({
97  		"/bad?n=abc,    false",  // No No-Trace anywhere.
98  		"/failPlain,    false",  // Exception set with no NoTrace attribute.
99  		"/failAttrFalse, true"   // The attribute (false) wins over the header (true).
100 	})
101 	void a01_recordCarriesThrown(String path, boolean noTrace) {
102 		var client = MockRestClient.create(A_Resource.class);
103 		var rec = capture(c -> run(c, path, noTrace), client);
104 		assertNotNull(rec.getThrown());
105 	}
106 
107 	@Test void a02_header_viaPerRequestOption_suppressesThrown() {
108 		var client = MockRestClient.create(A_Resource.class);
109 		var rec = capture(c -> run(c, "/bad?n=abc", true), client);
110 		assertNull(rec.getThrown());
111 		assertTrue(rec.getMessage().contains("[400]"), rec.getMessage());
112 	}
113 
114 	@Test void a03_header_viaBuilderOption_suppressesThrown() {
115 		var client = MockRestClient.builder(A_Resource.class).noTrace().build();
116 		var rec = capture(c -> run(c, "/bad?n=abc", false), client);
117 		assertNull(rec.getThrown());
118 	}
119 
120 	@Test void a04_header_false_keepsThrown() {
121 		var client = MockRestClient.create(A_Resource.class);
122 		var rec = capture(c -> {
123 			try {
124 				c.get("/bad?n=abc").header("No-Trace", "false").run();
125 			} catch (Exception e) {
126 				throw new AssertionError(e);
127 			}
128 		}, client);
129 		assertNotNull(rec.getThrown());
130 	}
131 
132 	@Test void a05_requestAttribute_suppressesThrown() {
133 		var client = MockRestClient.create(A_Resource.class);
134 		var rec = capture(c -> run(c, "/failAttr", false), client);
135 		assertNull(rec.getThrown());
136 	}
137 
138 	@Test void a06_builderDefault_isOptIn() {
139 		// ignoreErrors-style defaults do not turn No-Trace on: a plain client still logs the trace.
140 		var client = MockRestClient.builder(A_Resource.class).build();
141 		var rec = capture(c -> run(c, "/bad?n=abc", false), client);
142 		assertNotNull(rec.getThrown());
143 	}
