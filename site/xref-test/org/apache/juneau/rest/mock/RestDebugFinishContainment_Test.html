17  package org.apache.juneau.rest.mock;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.io.*;
23  import java.util.logging.*;
24  
25  import org.apache.juneau.commons.logging.*;
26  import org.apache.juneau.rest.server.*;
27  import org.apache.juneau.rest.server.logging.*;
28  import org.junit.jupiter.api.*;

30  /**
31   * Proves that a formatter (or scrubber) that throws while rendering the debug record during {@link RestSession#finish()}
32   * cannot escape and fail a completed request, and that only a fixed token — never the secret, the message, or a second
33   * formatter pass — is logged.
34   *
35   * @since 10.0.0
36   */
37  @SuppressWarnings({
38  	"resource" // MockRestClient instances are short-lived test fixtures.
39  })
40  class RestDebugFinishContainment_Test {
41  
42  	private static final String FINISH_LOGGER = RestSession.class.getName();
43  
44  	/** A resource that IS its own formatter and throws a {@code RuntimeException} while rendering the body ({@code FINEST}). */
45  	@Rest(path="/finbody")
46  	public static class A_ThrowsRuntimeInBody implements RestDebugFormatter {
47  		@Override public String formatBasic(RestRequest req, RestResponse res) { return "[basic]"; }
48  		@Override public String formatBody(RestRequest req, RestResponse res) {
49  			throw new RuntimeException("secret-in-formatter-BODY");
50  		}
51  		@RestPost(path="/echo")
52  		public String echo(RestRequest req) throws IOException {
53  			return req.getContent().asString();
54  		}
55  	}
56  
57  	@Test void a01_formatterThrowsRuntimeAtFinest_requestStillCompletes_fixedTokenOnly() {
58  		var records = LogRecordCapture.quietly(FINISH_LOGGER, () -> LogRecordCapture.quietly(org.apache.juneau.rest.client.classic.RestClient.class, () -> {
59  			var client = org.apache.juneau.rest.mock.classic.MockRestClient.create(A_ThrowsRuntimeInBody.class).debug().build();
60  
61  			// The request thread must not fail even though the formatter throws during finish().
62  			client.post("/echo", "request-payload").run().assertStatus().asCode().is(200).assertContent("request-payload");
63  		}));
64  
65  		var rec = records.stream()
66  			.filter(r -> eq(r.getLoggerName(), FINISH_LOGGER))
67  			.reduce((a, b) -> b)
68  			.orElse(null);
69  		assertNotNull(rec, "a fixed diagnostic-failure token should be logged");
70  		assertEquals("debug formatter failed", rec.getMessage());
71  		assertFalse(rec.getMessage().contains("secret-in-formatter-BODY"), rec.getMessage());
72  		assertNull(rec.getThrown(), "the failure must not attach the formatter's exception (which carries the body)");
73  	}
74  
75  	/** A resource that IS its own formatter and throws an {@code Error} while rendering the basic line ({@code INFO}). */
76  	@Rest(path="/finbasic")
77  	public static class B_ThrowsErrorInBasic implements RestDebugFormatter {
78  		@Override public String formatBasic(RestRequest req, RestResponse res) {
79  			throw new AssertionError("error-secret-in-basic");
80  		}
81  		@RestPost(path="/echo")
82  		public String echo(RestRequest req) throws IOException {
83  			return req.getContent().asString();
84  		}
85  	}
86  
87  	@Test void a02_formatterThrowsErrorAtInfo_requestStillCompletes_fixedTokenOnly() throws Exception {
88  		var target = Logger.getLogger(B_ThrowsErrorInBasic.class.getName());
89  		var prevLevel = target.getLevel();
90  		target.setLevel(Level.INFO);
91  		try {
92  			var records = LogRecordCapture.quietly(FINISH_LOGGER, () -> {
93  				var client = org.apache.juneau.rest.mock.classic.MockRestClient.create(B_ThrowsErrorInBasic.class).build();
94  
95  				client.post("/echo", "request-payload").run().assertStatus().asCode().is(200).assertContent("request-payload");
96  			});
97  
98  			var rec = records.stream()
99  				.filter(r -> eq(r.getLoggerName(), FINISH_LOGGER))
100 				.reduce((a, b) -> b)
101 				.orElse(null);
102 			assertNotNull(rec);
103 			assertEquals("debug formatter failed", rec.getMessage());
104 			assertFalse(rec.getMessage().contains("error-secret-in-basic"), rec.getMessage());
105 		} finally {
106 			target.setLevel(prevLevel);
107 		}
108 	}
109 }
