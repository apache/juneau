27  /**
28   * End-to-end tests for the status produced when no operation matches and a {@code @RestStartCall} hook has already
29   * written a status on the response.
30   *
31   * <p>
32   * Uses a response that defaults to the servlet-spec {@code 200}, as a real container does, rather than the mock's
33   * coincidental {@code 0}.  An unmatched request must be a 404 (or 405 for a known path with the wrong method) even
34   * when the start call explicitly set {@code 200}; a start call that left a 4xx status other than 404/405/412 must keep
35   * that status; a 5xx left by a start call stays an HTTP 500.
36   *
37   * @since 10.0.0
38   */
39  class RestContext_UnmatchedStartCallStatus_Test {
40  
41  	/** Start call records which resource ran, and sets the status from the {@code X-Start-Status} request header. */
42  	private static void applyStartStatus(String who, HttpServletRequest req, HttpServletResponse res) {
43  		res.addHeader("X-Start", who);
44  		var s = req.getHeader("X-Start-Status");
45  		if (s != null)
46  			res.setStatus(Integer.parseInt(s));
47  	}
48  
49  	@Rest
50  	public static class A_Top {
51  		@RestStartCall public void start(HttpServletRequest req, HttpServletResponse res) { applyStartStatus("top", req, res); }
52  		@RestGet(path="/foo") public String foo() { return "OK"; }
53  	}
54  
55  	@Rest(path="/bar")
56  	public static class B_Child {
57  		@RestStartCall public void start(HttpServletRequest req, HttpServletResponse res) { applyStartStatus("child", req, res); }
58  		@RestGet(path="/bar") public String bar() { return "OK"; }
59  	}
60  
61  	@Rest(children=B_Child.class)
62  	public static class B_Parent {
63  		@RestStartCall public void start(HttpServletRequest req, HttpServletResponse res) { applyStartStatus("parent", req, res); }
64  		@RestGet(path="/foo") public String foo() { return "OK"; }
65  	}
66  
67  	@Rest(children=B_Child.class)
68  	public static class C_Group extends BasicRestServletGroup {
69  		private static final long serialVersionUID = 1L;
70  		@RestStartCall public void start(HttpServletRequest req, HttpServletResponse res) { applyStartStatus("group", req, res); }
71  	}
72  
73  	private static int dispatch(Object resource, String method, String path, Integer startStatus) throws Exception {
74  		return run(resource, method, path, startStatus).getStatus();
75  	}
76  
77  	/** The start-call hooks that fired for the request (comma-separated). */
78  	private static String started(MockServletResponse res) {
79  		return String.join(",", res.getHeaders("X-Start"));
80  	}
81  
82  	private static MockServletResponse run(Object resource, String method, String path, Integer startStatus) throws Exception {
83  		var ctx = new RestContext(new RestContext.Args(resource.getClass(), null, null, () -> resource, "", null, null, null, RestContext.ContextKind.ROOT))
84  			.postInit().postInitChildFirst();
85  		var req = MockServletRequest.create(method, path).header("Accept", "text/plain");
86  		if (startStatus != null)
87  			req.header("X-Start-Status", String.valueOf(startStatus));
88  		var res = new MockServletResponse();
89  		ctx.execute(resource, req, res);
90  		return res;
91  	}

105 	@Test void a03_childResource_unmatchedPathInChild_startSets200_404() throws Exception {
106 		// Routed into the child (only the child's start call fires), where no operation matches.
107 		var res = run(new B_Parent(), "GET", "/bar/nope", 200);
108 		assertEquals(404, res.getStatus());
109 		assertEquals("child", started(res));
110 	}

115 		assertEquals(200, res.getStatus());
116 		assertEquals("child", started(res));
