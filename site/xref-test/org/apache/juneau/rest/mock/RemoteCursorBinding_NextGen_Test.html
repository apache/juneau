31  /**
32   * Integration tests for next-generation client-side cursor binding on {@code @Remote} interfaces and the
33   * {@link RestRequest#streamBodyEntity(RecordStreamBody)} convenience.
34   *
35   * <p>
36   * Verifies that {@code RecordReader} / {@code TokenReader} (and concrete subtypes like {@code JsonTokenReader}) work
37   * as {@code @RemoteOp} return types, and that {@code RecordStreamBody} works as a {@code @Content} parameter and as a
38   * directly-set request body.  Each test builds a next-generation client configured with a JSON parser so the
39   * inbound cursor parser is negotiated from the response {@code Content-Type}.
40   */
41  class RemoteCursorBinding_NextGen_Test {
42  
43  	public static class Bean {
44  		public String name;
45  		public int age;
46  		public Bean() {}
47  		public Bean(String name, int age) { this.name = name; this.age = age; }
48  	}
49  
50  	@Rest(serializers = JsonSerializer.class, parsers = JsonParser.class, defaultAccept = "application/json")
51  	public static class JsonServer {
52  		@RestGet(path = "/bean")
53  		public Bean get() {
54  			return new Bean("alice", 30);
55  		}
56  
57  		@RestPost(path = "/echo")
58  		public Bean echo(@Content Bean b) {
59  			return b;
60  		}
61  	}
62  
63  	@Remote
64  	public interface JsonClientApi {
65  		@RemoteGet("/bean")
66  		RecordReader getBean();
67  
68  		@RemoteGet("/bean")
69  		TokenReader getBeanAsTokens();
70  
71  		@RemoteGet("/bean")
72  		JsonTokenReader getBeanAsJsonTokens();
73  
74  		@RemotePost("/echo")
75  		Bean echo(@Content RecordStreamBody body);
76  	}
77  
78  	// ==========================================================================
79  	// a — return-type cursor binding
80  	// ==========================================================================
81  
82  	@Test
83  	@SuppressWarnings({
84  		"resource" // Inner client returned by getClient() is owned by the MockRestClient, not closed separately.
85  	})
86  	void a01_recordReaderReturnType() throws Exception {
87  		try (var client = MockRestClient.create(JsonServer.class);
88  				var nc = RestClient.builder().transport(client.getClient().getTransport()).parser(JsonParser.DEFAULT).build()) {
89  			var api = nc.remote(JsonClientApi.class);
90  			try (RecordReader r = api.getBean()) {
91  				var b = r.read(Bean.class);
92  				assertEquals("alice", b.name);
93  				assertEquals(30, b.age);
94  			}
95  		}
96  	}
97

99  	@SuppressWarnings({

101 	})

132 
133 	@Test
134 	@SuppressWarnings({
135 		"resource" // Fluent writer calls return the caller-owned writer for chaining; nothing new to close.
136 	})
137 	void b01_recordStreamBody_record() throws Exception {
138 		try (var client = MockRestClient.create(JsonServer.class);
139 				var nc = RestClient.builder().transport(client.getClient().getTransport()).parser(JsonParser.DEFAULT).build()) {
140 			var api = nc.remote(JsonClientApi.class);
141 			Bean got = api.echo(RecordStreamBody.records(w -> {
142 				try {
143 					w.write(new Bean("dave", 99));
144 				} catch (IOException e) {
145 					throw new UncheckedIOException(e);
146 				}
147 			}));
148 			assertEquals("dave", got.name);
149 			assertEquals(99, got.age);
150 		}
151 	}
152 
153 	@Test
154 	@SuppressWarnings({
155 		"resource" // Fluent writer calls return the caller-owned writer for chaining; nothing new to close.
156 	})
157 	void b02_recordStreamBody_token() throws Exception {
158 		try (var client = MockRestClient.create(JsonServer.class);
159 				var nc = RestClient.builder().transport(client.getClient().getTransport()).parser(JsonParser.DEFAULT).build()) {
160 			var api = nc.remote(JsonClientApi.class);
161 			Bean got = api.echo(RecordStreamBody.token(w -> {
162 				try {
163 					w.startObject();
164 					w.fieldName("name");
165 					w.string("eve");
166 					w.fieldName("age");
167 					w.number(45);
168 					w.endObject();
169 				} catch (IOException e) {
170 					throw new UncheckedIOException(e);
171 				}
172 			}));
173 			assertEquals("eve", got.name);
174 			assertEquals(45, got.age);
175 		}
176 	}
177 
178 	// ==========================================================================
179 	// c — streamBodyEntity(...) directly on a request
180 	// ==========================================================================
181 
182 	@Test
183 	@SuppressWarnings({
184 		"resource" // Fluent writer calls return the caller-owned writer for chaining; nothing new to close.
185 	})
186 	void c01_streamBodyEntity_direct() throws Exception {
187 		try (var client = MockRestClient.create(JsonServer.class)) {
188 			try (var resp = client.post("/echo").streamBodyEntity(RecordStreamBody.records(w -> {
189 				try {
190 					w.write(new Bean("frank", 12));
191 				} catch (IOException e) {
192 					throw new UncheckedIOException(e);
193 				}
194 			})).run()) {
195 				var got = JsonParser.DEFAULT.read(resp.body().asString(), Bean.class);
196 				assertEquals("frank", got.name);
197 				assertEquals(12, got.age);
198 			}
199 		}
200 	}
