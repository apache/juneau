32  /**
33   * Drift guard for the RDF-family marshaller streaming static shortcuts (see the marshaller shortcuts design).
34   *
35   * <p>
36   * The RDF formats implement no streaming role, so this asserts each declares <i>zero</i> streaming static
37   * shortcuts.  Capability is derived by reflection ({@code instanceof} on the live parser/serializer) rather
38   * than hardcoded, so the test remains a true drift guard should any RDF format adopt a streaming role.
39   */
40  class RdfMarshallerStreamShortcutDrift_Test extends TestBase {
41  
42  	private static final Set<String> STREAMING_SHORTCUTS = Set.of(
43  		"toTokens", "ofTokens", "toRecords", "ofRecords", "toArrayRecords", "ofArrayRecords");
44  
45  	static List<Class<? extends Marshaller>> marshallerClasses() {
46  		return List.of(
47  			RdfXml.class, RdfXmlAbbrev.class, RdfJson.class, RdfProto.class, RdfThrift.class,
48  			N3.class, NTriple.class, NQuads.class, TriG.class, TriX.class, Turtle.class, JsonLd.class);
49  	}
50  
51  	@ParameterizedTest
52  	@MethodSource("marshallerClasses")
53  	void a01_streamingStaticsMatchDefaultRoles(Class<? extends Marshaller> c) throws Exception {
54  		var def = (Marshaller) c.getField("DEFAULT").get(null);
55  		var parser = def.getParser();
56  		var serializer = def.getSerializer();
57  
58  		var expected = new TreeSet<String>();
59  		if (parser instanceof TokenReadable)
60  			expected.add("toTokens");
61  		if (serializer instanceof TokenWritable)
62  			expected.add("ofTokens");
63  		if (parser instanceof RecordReadable)
64  			expected.add("toRecords");
65  		if (serializer instanceof RecordWritable)
66  			expected.add("ofRecords");
67  		if (parser instanceof ArrayRecordReadable)
68  			expected.add("toArrayRecords");
69  		if (serializer instanceof ArrayRecordWritable)
70  			expected.add("ofArrayRecords");
71  
72  		var actual = Arrays.stream(c.getDeclaredMethods())
73  			.filter(m -> Modifier.isStatic(m.getModifiers()))
74  			.map(Method::getName)
75  			.filter(STREAMING_SHORTCUTS::contains)
76  			.collect(Collectors.toCollection(TreeSet::new));
77  
78  		assertEquals(expected, actual,
79  			() -> c.getSimpleName() + ": declared streaming static shortcuts must exactly match the roles implemented by its DEFAULT parser/serializer");
80  	}
81  }
