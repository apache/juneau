17  package org.apache.juneau.marshaller;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.io.*;
22  
23  import org.apache.juneau.*;
24  import org.apache.juneau.marshall.collections.*;
25  import org.apache.juneau.marshall.markdown.*;
26  import org.apache.juneau.marshall.marshaller.*;
27  import org.apache.juneau.marshall.marshaller.Markdown;
28  import org.junit.jupiter.api.*;
29  
30  /**
31   * Tests for {@link Markdown} marshaller.
32   */
33  class Markdown_Test extends TestBase {
34  
35  	@Test void d01_of() throws Exception {
36  		var bean = JsonMap.of("name", "Alice", "age", 30);
37  		var md = Markdown.of(bean);
38  		assertTrue(md.contains("| Key | Value |") || md.contains("| Property | Value |"), "Expected table header: " + md);
39  		assertTrue(md.contains("| name | Alice |"), "Expected name: " + md);
40  		assertTrue(md.contains("| age | 30 |"), "Expected age: " + md);
41  	}
42  
43  	@Test void d02_write() throws Exception {
44  		var bean = JsonMap.of("name", "Alice", "age", 30);
45  		var md = Markdown.of(bean);
46  		assertTrue(md.contains("| Key | Value |") || md.contains("| Property | Value |"), "Expected table header: " + md);
47  		assertTrue(md.contains("| name | Alice |"), "Expected name: " + md);
48  	}
49  
50  	@Test void d03_docMode() throws Exception {
51  		var bean = JsonMap.of("name", "Alice", "age", 30);
52  		var serializer = MarkdownDocSerializer.create().title("Person").build();
53  		var md = ((CharMarshaller) new MarkdownDoc(serializer, MarkdownDocParser.DEFAULT)).write(bean);
54  		assertTrue(md.contains("# Person"), "Expected H1 title: " + md);
55  		assertTrue(md.contains("| Key | Value |") || md.contains("| Property | Value |"), "Expected table: " + md);
56  		assertTrue(md.contains("| name | Alice |"), "Expected name: " + md);
57  	}
58  
59  	@Test void d04_toOutput() throws Exception {
60  		var bean = JsonMap.of("name", "Alice", "age", 30);
61  		var sw = new StringWriter();
62  		Markdown.DEFAULT.write(bean, sw);
63  		var md = sw.toString();
64  		assertTrue(md.contains("| Key | Value |") || md.contains("| Property | Value |"), "Expected table header: " + md);
65  		assertTrue(md.contains("| name | Alice |"), "Expected name: " + md);
66  	}
67  }
