29  /**
30   * Tests for {@link MethodInfoUtils} package-private validation helpers.
31   *
32   * <p>These helpers are reachable through the public bean-marshaller pipelines (RequestBeanMeta,
33   * ResponseBeanMeta) but are tested here directly since they are package-private and only the
34   * exceptional branches matter for coverage.</p>
35   */
36  class MethodInfoUtils_Test extends TestBase {
37  
38  	// --------------------------------------------------------------------------------
39  	// Fixtures
40  	// --------------------------------------------------------------------------------
41  
42  	@SuppressWarnings({
43  		"unused"  // Unused parameters/variables kept for consistent method signatures across test utilities.
44  	})
45  	static class Fixtures {
46  		// One-arg matching methods
47  		public void oneStringArg(String x) { /* no-op */ }
48  		public void oneIntArg(int x) { /* no-op */ }
49  
50  		// Multi-arg method
51  		public void twoArgs(String a, String b) { /* no-op */ }
52  
53  		// No-arg method
54  		public void noArgs() { /* no-op */ }
55  
56  		// Return-type methods
57  		public void returnsVoid() { /* no-op */ }
58  		public String returnsString() { return ""; }
59  		public int returnsInt() { return 0; }
60  	}
61  
62  	private static MethodInfo m(String name, Class<?>... params) throws Exception {
63  		return MethodInfo.of(Fixtures.class.getMethod(name, params));
64  	}
65  
66  	// --------------------------------------------------------------------------------
67  	// assertArgType
68  	// --------------------------------------------------------------------------------
69  
70  	@Test
71  	void a01_assertArgType_matchingType_returns() throws Exception {
72  		// Should not throw — String matches one of the allowed types.
73  		MethodInfoUtils.assertArgType(m("oneStringArg", String.class), Query.class, String.class, Integer.class);
74  	}
75  
76  	@Test
77  	void a02_assertArgType_matchingTypeSecond_returns() throws Exception {
78  		// Loop iterates and second class matches.
79  		MethodInfoUtils.assertArgType(m("oneIntArg", int.class), Query.class, String.class, int.class);
80  	}
81  
82  	@SuppressWarnings({
83  		"java:S5778" // assertThrows lambdas contain multiple calls; only the primary call throws.
84  	})
85  	@Test
86  	void a03_assertArgType_wrongType_throws() {
87  		// Loop completes without match -> throws.
88  		var thrown = assertThrows(InvalidAnnotationException.class,
89  			() -> MethodInfoUtils.assertArgType(m("oneStringArg", String.class), Query.class, Integer.class, Long.class));
90  		assertTrue(thrown.getMessage().contains("Invalid return type"));
91  	}
