17  package org.apache.juneau.http;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.apache.juneau.*;
22  import org.junit.jupiter.api.*;
23  
24  class HttpProtocolVersion_Test extends TestBase {
25  
26  	@Test void a01_constants() {
27  		assertEquals("HTTP", HttpProtocolVersion.HTTP_1_0.protocol());
28  		assertEquals(1, HttpProtocolVersion.HTTP_1_0.major());
29  		assertEquals(0, HttpProtocolVersion.HTTP_1_0.minor());
30  		assertEquals("HTTP/1.0", HttpProtocolVersion.HTTP_1_0.toString());
31  		assertEquals("HTTP/1.1", HttpProtocolVersion.HTTP_1_1.toString());
32  		assertEquals("HTTP/2.0", HttpProtocolVersion.HTTP_2_0.toString());
33  	}
34  
35  	@Test void a02_of() {
36  		var v = HttpProtocolVersion.of("HTTP", 1, 1);
37  		assertEquals("HTTP", v.protocol());
38  		assertEquals(1, v.major());
39  		assertEquals(1, v.minor());
40  		assertEquals("HTTP/1.1", v.toString());
41  	}
42  
43  	@Test void a03_parse_standardForm() {
44  		var v = HttpProtocolVersion.parse("HTTP/1.1");
45  		assertEquals(HttpProtocolVersion.HTTP_1_1, v);
46  	}
47  
48  	@Test void a04_parse_majorOnlyForm() {
49  		var v = HttpProtocolVersion.parse("HTTP/2");
50  		assertEquals("HTTP", v.protocol());
51  		assertEquals(2, v.major());
52  		assertEquals(0, v.minor());
53  		assertEquals("HTTP/2.0", v.toString());
54  	}
55  
56  	@Test void a05_parse_customProtocol() {
57  		var v = HttpProtocolVersion.parse("RTSP/1.0");
58  		assertEquals("RTSP", v.protocol());
59  		assertEquals(1, v.major());
60  		assertEquals(0, v.minor());
61  	}
62  
63  	@Test void a06_parse_rejectsMissingSlash() {
64  		assertThrows(IllegalArgumentException.class, () -> HttpProtocolVersion.parse("HTTP-1.1"));
65  	}
66  
67  	@Test void a07_parse_rejectsNonNumericMajor() {
68  		assertThrows(IllegalArgumentException.class, () -> HttpProtocolVersion.parse("HTTP/X.0"));
69  	}
70  
71  	@Test void a08_parse_rejectsNonNumericMinor() {
72  		assertThrows(IllegalArgumentException.class, () -> HttpProtocolVersion.parse("HTTP/1.X"));
73  	}
74  
75  	@Test void a09_nullProtocolRejected() {
76  		assertThrows(IllegalArgumentException.class, () -> new HttpProtocolVersion(null, 1, 1));
77  	}
78  
79  	@Test void a10_negativeMajorRejected() {
80  		assertThrows(IllegalArgumentException.class, () -> new HttpProtocolVersion("HTTP", -1, 0));
81  	}
82  
83  	@Test void a11_negativeMinorRejected() {
84  		assertThrows(IllegalArgumentException.class, () -> new HttpProtocolVersion("HTTP", 1, -1));
85  	}
86  
87  	@Test void a12_nullParseInputRejected() {
88  		assertThrows(IllegalArgumentException.class, () -> HttpProtocolVersion.parse(null));
89  	}
90  
91  	@Test void a13_equalsAndHashCode() {
92  		var a = HttpProtocolVersion.of("HTTP", 1, 1);
93  		var b = HttpProtocolVersion.of("HTTP", 1, 1);
94  		var c = HttpProtocolVersion.of("HTTP", 2, 0);
95  		assertEquals(a, b);
96  		assertEquals(a.hashCode(), b.hashCode());
97  		assertNotEquals(a, c);
98  	}
99  }
