221 	private static Object sampleArg(Class<?> t) {
222 		if (t == String.class || t == Object.class)
223 			return "x";
224 		if (t == int.class)
225 			return 201;
226 		if (t == Locale.class)
227 			return Locale.FRANCE;
228 		if (t == HttpEntity.class)
229 			return new StringEntity("body", java.nio.charset.StandardCharsets.UTF_8);
230 		if (t == HeaderList.class)
231 			return HeaderList.create();
232 		if (t == List.class)
233 			return new ArrayList<Header>(List.of(new org.apache.http.message.BasicHeader("A", "b")));
234 		if (t == Header.class)
235 			return new org.apache.http.message.BasicHeader("A", "b");
236 		if (t == Header[].class)
237 			return new Header[]{new org.apache.http.message.BasicHeader("A", "b")};
238 		if (t == Object[].class)
239 			return new Object[0];
240 		if (t == ProtocolVersion.class)
241 			return HttpVersion.HTTP_1_0;
242 		if (t == ReasonPhraseCatalog.class)
243 			return EnglishReasonPhraseCatalog.INSTANCE;
244 		if (t == BasicStatusLine.class)
245 			return BasicStatusLine.create(201, "Made");
246 		if (t == URI.class)
247 			return URI.create("http://localhost/x");
248 		throw new IllegalArgumentException("No sample for " + t);
249 	}

251 	/** Fluent (covariant-returning) public methods declared directly on the leaf type. */
252 	private static List<Method> fluentMethods(Class<?> type) {
253 		var l = new ArrayList<Method>();
254 		for (var m : type.getDeclaredMethods())
255 			if (Modifier.isPublic(m.getModifiers()) && !Modifier.isStatic(m.getModifiers()) && !m.isBridge() && m.getReturnType() == type && m.getParameterCount() > 0)
256 				l.add(m);
257 		return l;
258 	}

260 	private static Object[] sampleArgs(Method m) {
261 		return Arrays.stream(m.getParameterTypes()).map(ClassicResponse_Test::sampleArg).toArray();
262 	}

264 	@ParameterizedTest
265 	@MethodSource("a06_exceptionClasses")
266 	void b01_exceptionFluentSetters(Class<? extends BasicHttpException> type) throws Exception {
267 		var methods = fluentMethods(type);
268 		assertTrue(methods.size() >= 10, type::getName);
269 		for (var m : methods) {
270 			var instance = type.getDeclaredConstructor().newInstance().setHeader2("Pre", "1").setReasonPhrase2("Custom");
271 			var args = sampleArgs(m);
272 			assertSame(instance, m.invoke(instance, args), m::getName);
273 			assertSetterState(instance, m, args);
274 		}
275 	}

277 	/** Asserts the observable state change each fluent setter promises (sample args come from {@link #sampleArg}). */
278 	private static void assertSetterState(BasicHttpException x, Method m, Object[] a) throws Exception {
279 		Supplier<String> what = m::toString;
280 		switch (m.getName()) {
281 			case "setContent" -> {
282 				assertNotNull(x.getEntity(), what);
283 				if (a[0] instanceof HttpEntity e)
284 					assertSame(e, x.getEntity(), what);
285 				else
286 					assertEquals(a[0], org.apache.http.util.EntityUtils.toString(x.getEntity()), what);
287 			}
288 			case "setHeader2" -> assertEquals(a[1], x.getFirstHeader((String)a[0]).getValue(), what);
289 			case "setHeaders" -> {
290 				if (m.getParameterTypes()[0] == List.class)
291 					assertEquals("b", x.getFirstHeader("A").getValue(), what);
292 				else
293 					assertNull(x.getFirstHeader("Pre"), what);  // The empty HeaderList replaced the pre-existing header.
294 			}
295 			case "setHeaders2" -> assertEquals("b", x.getFirstHeader("A").getValue(), what);
296 			case "setLocale2" -> assertEquals(Locale.FRANCE, x.getLocale(), what);
297 			case "setMessage" -> assertEquals("x", x.getMessage(), what);
298 			case "setProtocolVersion" -> assertEquals(HttpVersion.HTTP_1_0, x.getProtocolVersion(), what);
299 			case "setReasonPhrase2" -> assertEquals("x", x.getStatusLine().getReasonPhrase(), what);
300 			case "setReasonPhraseCatalog" -> assertEquals("Custom", x.getStatusLine().getReasonPhrase(), what);
301 			case "setStatusCode2" -> assertEquals(201, x.getStatusLine().getStatusCode(), what);
302 			case "setStatusLine" -> {
303 				assertEquals(201, x.getStatusLine().getStatusCode(), what);
304 				assertEquals("Made", x.getStatusLine().getReasonPhrase(), what);
305 			}
306 			default -> fail("No state assertion for fluent setter " + what.get());
307 		}
308 	}
