48  	@Test void b02_reflectiveGuard_everyPublicMutatorThrows() throws Exception {
49  		var u = new NotFound().setContent("body").setHeader2("X-Test", "v").unmodifiable();
50  		var mutators = 0;
51  		for (var m : NotFound.class.getMethods()) {
52  			var n = m.getName();
53  			// Only funnel-routed mutators declared in our hierarchy — skip JDK-declared set/add/remove such as
54  			// Throwable.setStackTrace(...) and Throwable.addSuppressed(...), which are not routed through modify().
55  			if (Modifier.isStatic(m.getModifiers()) || m.getDeclaringClass().getName().startsWith("java.") || ! (n.startsWith("set") || n.startsWith("add") || n.startsWith("remove")))
56  				continue;
57  			mutators++;
58  			var types = m.getParameterTypes();
59  			var args = new Object[types.length];
60  			for (var i = 0; i < types.length; i++)
61  				args[i] = types[i] == int.class ? Integer.valueOf(0) : null;
62  			var ex = assertThrows(InvocationTargetException.class, () -> m.invoke(u, args), m::getName);
63  			assertInstanceOf(UnsupportedOperationException.class, ex.getCause(), m::getName);
64  		}
65  		assertTrue(mutators >= 10, "Expected at least 10 public mutators, found " + mutators);
66  	}

73  	@Test void b04_contentEquality_equalContentComparesEqual() {
74  		// Content equality — the slices satisfiable on their own.
75  		// Two independently-frozen snapshots of equal content compare equal.
76  		var u1 = new NotFound().setHeader2("X-Test", "v").unmodifiable();
77  		var u2 = new NotFound().setHeader2("X-Test", "v").unmodifiable();
78  		assertEquals(u1, u2);
79  		assertEquals(u1.hashCode(), u2.hashCode());
80  		// Two modifiable beans of equal content compare equal.
81  		var m1 = new NotFound().setHeader2("X-Test", "v");
82  		var m2 = new NotFound().setHeader2("X-Test", "v");
83  		assertEquals(m1, m2);
84  		assertEquals(m1.hashCode(), m2.hashCode());
85  		// Cross-boundary equality: now that HeaderList equality is content-only, a modifiable bean
86  		// compares equal to its own frozen snapshot (over statusLine + headers + getMessage(), excluding the
87  		// value-less content wrapper).
88  		assertEquals(m1, m1.unmodifiable());
89  		assertEquals(m1.hashCode(), m1.unmodifiable().hashCode());
90  	}

193 		var x = new BasicHttpException(200);
194 		x.setStatusCode(201);
195 		assertEquals(201, x.getStatusLine().getStatusCode());
196 		x.setReasonPhrase("Made");
197 		assertEquals("Made", x.getStatusLine().getReasonPhrase());
198 		x.setStatusLine(org.apache.http.HttpVersion.HTTP_1_0, 202);
199 		assertEquals(org.apache.http.HttpVersion.HTTP_1_0, x.getProtocolVersion());
200 		assertEquals(202, x.getStatusLine().getStatusCode());
201 		x.setStatusLine(org.apache.http.HttpVersion.HTTP_1_1, 203, "Other");
202 		assertEquals("Other", x.getStatusLine().getReasonPhrase());
203 		x.setStatusLine(new org.apache.http.message.BasicStatusLine(org.apache.http.HttpVersion.HTTP_1_1, 204, "None"));
204 		assertEquals(204, x.getStatusLine().getStatusCode());
205 		assertEquals("None", x.getStatusLine().getReasonPhrase());
206 		x.setLocale(java.util.Locale.FRANCE);
207 		assertEquals(java.util.Locale.FRANCE, x.getLocale());
208 		x.setParams(null);
209 		assertNull(x.getParams());
210 		x.setEntity(new org.apache.http.entity.StringEntity("e", java.nio.charset.StandardCharsets.UTF_8));
211 		assertNotNull(x.getEntity());
212 		x.setEntity(null);
213 		// Clearing the entity reverts to the lazily-built default entity, whose text is the message.
214 		assertEquals(x.getMessage(), org.apache.http.util.EntityUtils.toString(x.getEntity()));
215 		x.setMessage("hello %s", "world");
216 		assertEquals("hello world", x.getMessage());
217 		assertEquals("hello world", x.toString());

231 		var root = new IllegalStateException("<root> & co");
232 		var mid = new RuntimeException(null, root);
233 		var x = new BasicHttpException(500, mid, "top <msg>");
234 		var raw = x.getFullStackMessage(false);
235 		assertTrue(raw.startsWith("top <msg>"), raw);
236 		assertTrue(raw.contains("Caused by (RuntimeException)\nCaused by (IllegalStateException): <root> & co"), raw);
237 		var scrubbed = x.getFullStackMessage(true);
238 		assertTrue(scrubbed.startsWith("top  msg "), scrubbed);
239 		assertTrue(scrubbed.contains("Caused by (IllegalStateException):  root    co"), scrubbed);
240 		assertSame(root, new BasicHttpException(500, new java.lang.reflect.InvocationTargetException(root)).getRootCause());
241 		assertNull(new BasicHttpException(500).getRootCause());
242 	}
