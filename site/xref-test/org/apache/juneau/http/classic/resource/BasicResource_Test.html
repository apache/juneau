242 	@Test void e02_unmodifiable_returnsFrozenSnapshot() {
243 		var x = new StringResource(null, "foo");
244 		var u = x.unmodifiable();
245 		assertFalse(x.isUnmodifiable());
246 		assertTrue(u.isUnmodifiable());
247 		// The snapshot's direct-field mutator (setHeaders(HeaderList)) throws through the modify() funnel...
248 		var newHeaders = HeaderList.create().append("X", "y");
249 		assertThrows(UnsupportedOperationException.class, () -> u.setHeaders(newHeaders));
250 		// ...and the sub-bean-delegating mutators throw through the frozen entity/headers.
251 		assertThrows(UnsupportedOperationException.class, () -> u.setContent("other"));
252 		assertThrows(UnsupportedOperationException.class, () -> u.addHeader("X", "y"));
253 		assertThrows(UnsupportedOperationException.class, () -> u.setHeader("X", "y"));
254 		assertThrows(UnsupportedOperationException.class, () -> u.setChunked(true));
255 		var frozenHeaders = u.getHeaders();
256 		assertThrows(UnsupportedOperationException.class, () -> frozenHeaders.append("X", "y"));
257 	}

277 	@Test void f01_fluentReturns() {
278 		var x = new StringResource(null, "foo");
279 		assertSame(x, x.addHeader("X-A", "a"));
280 		assertSame(x, x.setHeader("X-B", "b"));
281 		assertSame(x, x.addHeaders(header("X-C", "c")));
282 		assertSame(x, x.setHeaders(header("X-D", "d")));
283 		assertSame(x, x.setHeaders(HeaderList.create().append("X-E", "e")));
284 		assertSame(x, x.setContent("new"));
285 		assertSame(x, x.setContent(() -> "supplied"));
286 		assertSame(x, x.setContentType("text/plain"));
287 		assertSame(x, x.setContentType(contentType("text/html")));
288 		assertSame(x, x.setContentEncoding("identity"));
289 		assertSame(x, x.setContentEncoding(contentEncoding("gzip")));
