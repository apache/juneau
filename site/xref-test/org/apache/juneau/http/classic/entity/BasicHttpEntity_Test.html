17  package org.apache.juneau.http.classic.entity;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.lang.reflect.*;
22  
23  import org.apache.juneau.*;
24  import org.junit.jupiter.api.*;

26  /**
27   * Immutability-paradigm tests for the {@link BasicHttpEntity} hierarchy, using {@link ByteArrayEntity} as a
28   * representative leaf.
29   *
30   * <p>
31   * Mirrors the Phase 2 response tests ({@code BasicHttpResponse_Test} b01–b07): reflective mutator guard, idempotency,
32   * content equality, snapshot independence, and snapshot-construction regression.  Because entities hold no list-backed
33   * sub-bean whose {@code unmodifiable} flag folds into equality, the full D3 assertion {@code bean.equals(bean.unmodifiable())}
34   * holds here and is asserted directly (unlike the response side, which defers it to Phase 5).
35   */
36  class BasicHttpEntity_Test extends TestBase {
37  
38  	private static final byte[] BYTES = "hello world".getBytes();
39  
40  	@Test void b01_unmodifiable_returnsUnmodifiableLeafType() {
41  		var x = new ByteArrayEntity().setContent(BYTES);
42  		assertFalse(x.isUnmodifiable());
43  		var u = x.unmodifiable();
44  		assertTrue(u.isUnmodifiable());
45  		assertInstanceOf(ByteArrayEntity.Unmodifiable.class, u);
46  		assertInstanceOf(ByteArrayEntity.class, u, "Snapshot must preserve the concrete leaf type.");
47  	}
48  
49  	@Test void b02_reflectiveGuard_everyPublicMutatorThrows() throws Exception {
50  		var u = new ByteArrayEntity().setContent(BYTES).setContentType("text/plain").unmodifiable();
51  		var mutators = 0;
52  		for (var m : ByteArrayEntity.class.getMethods()) {
53  			var n = m.getName();
54  			if (Modifier.isStatic(m.getModifiers()) || ! (n.startsWith("set") || n.startsWith("add") || n.startsWith("remove")))
55  				continue;
56  			mutators++;
57  			var types = m.getParameterTypes();
58  			var args = new Object[types.length];
59  			for (var i = 0; i < types.length; i++) {
60  				var t = types[i];
61  				if (t == int.class)
62  					args[i] = Integer.valueOf(0);
63  				else if (t == long.class)
64  					args[i] = Long.valueOf(0);
65  				else if (t == boolean.class)
66  					args[i] = Boolean.FALSE;
67  				else
68  					args[i] = null;
69  			}
70  			var ex = assertThrows(InvocationTargetException.class, () -> m.invoke(u, args), m::getName);
71  			assertInstanceOf(UnsupportedOperationException.class, ex.getCause(), m::getName);
72  		}
73  		assertTrue(mutators >= 10, "Expected at least 10 public mutators, found " + mutators);
74  	}
75  
76  	@Test void b03_idempotency_alreadyUnmodifiableReturnsThis() {
77  		var u = new ByteArrayEntity().setContent(BYTES).unmodifiable();
78  		assertSame(u, u.unmodifiable());
79  	}
80  
81  	@Test void b04_contentEquality_beanEqualsSnapshot() {
82  		// D3 content equality — entities have no list-backed sub-bean whose 'unmodifiable' flag folds into equality,
83  		// so the full bean.equals(bean.unmodifiable()) assertion holds (design spec D3 / Phase 4).
84  		var x = new ByteArrayEntity().setContent(BYTES).setContentType("text/plain");
85  		var u = x.unmodifiable();
86  		assertEquals(x, u);
87  		assertEquals(x.hashCode(), u.hashCode());
88  		// Two independently-frozen snapshots of equal content compare equal.
89  		var u1 = new ByteArrayEntity().setContent(BYTES).setContentType("text/plain").unmodifiable();
90  		var u2 = new ByteArrayEntity().setContent(BYTES).setContentType("text/plain").unmodifiable();
91  		assertEquals(u1, u2);
92  		assertEquals(u1.hashCode(), u2.hashCode());
93  	}
94  
95  	@Test void b05_snapshotIndependence_mutatingOriginalDoesNotAffectSnapshot() throws Exception {
96  		var x = new ByteArrayEntity().setContent(BYTES).setContentType("text/plain");
97  		var u = x.unmodifiable();
98  		x.setContent("changed".getBytes()).setContentType("text/html").setChunked(true);
99  		assertArrayEquals(BYTES, u.asBytes());
100 		assertEquals("text/plain", u.getContentType().getValue());
101 		assertFalse(u.isChunked());
102 	}
103 
104 	@Test void b06_snapshotConstruction_freezeDoesNotThrow() {
105 		// Regression: constructing the snapshot must not route the freeze through the throwing modify().
106 		var x = new ByteArrayEntity().setContent(BYTES).setContentType("text/plain").setChunked(true);
107 		assertDoesNotThrow(x::unmodifiable);
108 	}
109 
110 	@Test void b07_emptyConstant_isUnmodifiable() {
111 		assertTrue(BasicHttpEntity.EMPTY.isUnmodifiable());
112 	}
113 
114 	@Test void b08_crossLeafInequality_equalBaseContentDifferentLeaf() {
115 		// Two unrelated leaves with identical base content must NOT be equal (leaf-type gate), in both directions.
116 		var s = new StringEntity().setContent("x").setContentType("text/plain");
117 		var b = new ByteArrayEntity().setContent("x").setContentType("text/plain");
118 		assertNotEquals(s, b);
119 		assertNotEquals(b, s);
120 	}
121 
122 	@Test void b09_crossLeafInequality_holdsAcrossSnapshots() {
123 		// The leaf-type gate must also hold for the frozen snapshots (Unmodifiable IS-A its leaf).
124 		var s = new StringEntity().setContent("x").unmodifiable();
125 		var b = new ByteArrayEntity().setContent("x").unmodifiable();
126 		assertNotEquals(s, b);
127 		assertNotEquals(b, s);
128 	}
129 
130 	@Test void b10_d3_holdsBothDirections_representativeLeaves() {
131 		// D3: bean.equals(bean.unmodifiable()) AND the reverse, for a representative sampling of leaves.
132 		var se = new StringEntity().setContent("x").setContentType("text/plain");
133 		assertEquals(se, se.unmodifiable());
134 		assertEquals(se.unmodifiable(), se);
135 		assertEquals(se.hashCode(), se.unmodifiable().hashCode());
136 
137 		var be = new ByteArrayEntity().setContent(BYTES).setContentType("text/plain");
138 		assertEquals(be, be.unmodifiable());
139 		assertEquals(be.unmodifiable(), be);
140 		assertEquals(be.hashCode(), be.unmodifiable().hashCode());
141 	}
142 
143 	@Test void b11_sameLeafEquality_differentInstances() {
144 		// Same leaf type + equal base content compare equal (sanity that the gate does not over-restrict).
145 		var a = new StringEntity().setContent("x").setContentType("text/plain");
146 		var b = new StringEntity().setContent("x").setContentType("text/plain");
147 		assertEquals(a, b);
148 		assertEquals(a.hashCode(), b.hashCode());
149 	}
