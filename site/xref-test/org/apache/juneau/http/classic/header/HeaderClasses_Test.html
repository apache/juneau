105 	private static final List<Object> STRING_SAMPLES = List.of("true", "1", "Sun, 6 Nov 1994 08:49:37 GMT", "\"a\"", ZonedDateTime.of(2020, 1, 2, 3, 4, 5, 0, ZoneId.of("GMT")), EntityTag.of("\"a\""));

107 	private static Object sample(Type t, Object cand) {
108 		var s = cand instanceof String str ? str : "1";
109 		if (t instanceof ParameterizedType p) {
110 			var raw = (Class<?>)p.getRawType();
111 			var arg = p.getActualTypeArguments()[0];
112 			if (raw == Supplier.class) {
113 				var v = arg instanceof WildcardType ? cand : sample(arg, cand);
114 				return (Supplier<Object>)() -> v;
115 			}
116 			if (raw == List.class)
117 				return new ArrayList<>(List.of(sample(arg, cand)));
118 			throw new IllegalArgumentException("No sample for " + t);
119 		}
120 		var c = (Class<?>)t;
121 		if (c == String.class || c == Object.class)
122 			return s;
123 		if (c == String[].class)
124 			return new String[]{"a", "b"};
125 		if (c == Throwable[].class)
126 			return new Throwable[]{new IllegalStateException("boom")};
127 		if (c == Throwable.class)
128 			return new IllegalStateException("boom");
129 		if (c == long.class || c == Long.class)
130 			return 1L;
131 		if (c == int.class || c == Integer.class)
132 			return 1;
133 		if (c == boolean.class || c == Boolean.class)
134 			return true;
135 		if (c == Version.class)
136 			return Version.of("1.2.3");
137 		if (c == URI.class)
138 			return URI.create("http://localhost/x");
139 		if (c == ZonedDateTime.class)
140 			return ZonedDateTime.of(2020, 1, 2, 3, 4, 5, 0, ZoneId.of("GMT"));
141 		if (c == EntityTag.class)
142 			return EntityTag.of("\"a\"");
143 		if (c == EntityTags.class)
144 			return EntityTags.of("\"a\"");
145 		if (c == MediaType.class)
146 			return MediaType.of("text/plain");
147 		if (c == MediaRanges.class)
148 			return MediaRanges.of("text/*");
149 		if (c == StringRanges.class)
150 			return StringRanges.of("a");
151 		if (c == Thrown.Part.class)
152 			return new Thrown.Part("a;b");
153 		if (c == Thrown.Part[].class)
154 			return new Thrown.Part[]{new Thrown.Part("a;b")};
155 		throw new IllegalArgumentException("No sample for " + t);
156 	}

158 	private static Object[] sampleArgs(Type[] types, Object s) {
159 		return Arrays.stream(types).map(t -> sample(t, s)).toArray();
160 	}
161 
162 	private static final String NAME_SAMPLE = "X-Sample-Name";
163 
164 	/** Like {@link #sampleArgs}, but a leading String parameter of a multi-parameter signature is a header name, so it gets a distinct sample. */
165 	private static Object[] sampleNamedArgs(Type[] types, Object s) {
166 		var a = sampleArgs(types, s);
167 		if (types.length > 1 && types[0] == String.class)
168 			a[0] = NAME_SAMPLE;
169 		return a;
170 	}

172 	private static boolean sampleable(Type[] types) {
173 		try {
174 			sampleArgs(types, "1");
175 			return true;
176 		} catch (IllegalArgumentException e) {
177 			return false;
178 		}
179 	}

181 	/** Runtime causes of a reflective call surface unchanged (so an NPE regression is not mistaken for a rejected sample); checked causes are wrapped. */
182 	private static RuntimeException unwrap(ReflectiveOperationException e) {
183 		var c = e.getCause() == null ? e : e.getCause();
184 		return c instanceof RuntimeException r ? r : new IllegalArgumentException(c);
185 	}

187 	/** The plain value a header was built from: Suppliers are read, and strings, numbers and booleans are rendered as strings; null when not comparable. */
188 	private static String expectedValue(Object arg) {
189 		if (arg instanceof Supplier<?> sup)
190 			arg = sup.get();
191 		return arg instanceof String || arg instanceof Number || arg instanceof Boolean ? arg.toString() : null;
192 	}

194 	private record Built(org.apache.http.Header header, Object[] args) {}
195 
196 	/** Invokes with each candidate string value until one is accepted. */
197 	private static Built build(Type[] types, boolean named, Function<Object[],Object> call) {
198 		RuntimeException last = null;
199 		for (var s : STRING_SAMPLES) {
200 			var args = named ? sampleNamedArgs(types, s) : sampleArgs(types, s);
201 			try {
202 				var h = (org.apache.http.Header)call.apply(args);
203 				if (h != null)
204 					h.getValue();  // Supplier-backed headers only reject a wrongly-typed value when it is read.
205 				return new Built(h, args);
206 			} catch (IllegalArgumentException | java.time.format.DateTimeParseException e) {
207 				last = e;
208 			} catch (RuntimeException e) {
209 				if (!String.valueOf(e.getMessage()).startsWith("Invalid object type returned by supplier"))
210 					throw e;
211 				// Only a value rejection (parse failure, or a supplier of the wrong type) moves on to the next candidate.
212 				// The wrong-type case is a plain RuntimeException thrown through rex(...) by getValue() in classic
213 				// IfRange.java:226 and RetryAfter.java:217, so it can only be recognised by its message.
214 				last = e;
215 			}
216 		}
217 		throw last;
218 	}

220 	private static String expectedName(Class<?> type) {
221 		var ann = type.getAnnotation(Header.class);
222 		var name = ann == null ? "" : ann.value().isEmpty() ? ann.name() : ann.value();
223 		assertFalse(name.isEmpty(), () -> type.getName() + " must declare its header name.");
224 		return name;
225 	}
226 
227 	/**
228 	 * Asserts the header carries a value: non-null, equal to the supplied string when the only argument was a string,
229 	 * and stable when parsed back through the class's own <c>of(String)</c> / <c>(String)</c> entry point.
230 	 */
231 	private static void assertValue(Class<?> type, Built b, Object what) throws Exception {
232 		var h = b.header();
233 		var v = h.getValue();
234 		assertNotNull(v, () -> "null value: " + what);
235 		if (b.args().length == 1) {
236 			var expected = expectedValue(b.args()[0]);
237 			if (expected != null)
238 				assertEquals(expected, v, () -> "value must round-trip unchanged: " + what);
239 		}
240 
241 		org.apache.http.Header again = null;
242 		for (var m : type.getMethods())
243 			if (m.getName().equals("of") && Modifier.isStatic(m.getModifiers()) && m.getDeclaringClass() == type && Arrays.equals(m.getParameterTypes(), new Class<?>[]{String.class}))
244 				again = (org.apache.http.Header)m.invoke(null, v);
245 		if (again == null) {
246 			try {
247 				again = (org.apache.http.Header)type.getConstructor(String.class).newInstance(v);
248 			} catch (NoSuchMethodException e) { /* no string entry point */ }
249 		}
250 		if (again != null) {
251 			assertEquals(h.getName(), again.getName(), () -> "reparsed name: " + what);
252 			assertEquals(v, again.getValue(), () -> "reparsed value: " + what);
253 		}
254 	}

256 	@ParameterizedTest
257 	@MethodSource
258 	void a01_headerClasses(Class<?> type) throws Exception {
259 		var name = expectedName(type);
260 
261 		var ctors = 0;
262 		for (var c : type.getConstructors()) {
263 			var types = c.getGenericParameterTypes();
264 			if (types.length == 0 || !sampleable(types))
265 				continue;
266 			ctors++;
267 			var b = build(types, false, a -> {
268 				try {
269 					return c.newInstance(a);
270 				} catch (ReflectiveOperationException e) {
271 					throw unwrap(e);
272 				}
273 			});
274 			assertEquals(name, b.header().getName(), c::toString);
275 			assertValue(type, b, c);
276 		}
277 		assertTrue(ctors >= 1, type::getName);
278 
279 		var factories = 0;
280 		for (var m : type.getMethods()) {
281 			if (!m.getName().equals("of") || !Modifier.isStatic(m.getModifiers()) || m.getDeclaringClass() != type)
282 				continue;
283 			var types = m.getGenericParameterTypes();
284 			if (!sampleable(types))
285 				continue;
286 			factories++;
287 			var b = build(types, false, a -> {
288 				try {
289 					return m.invoke(null, a);
290 				} catch (ReflectiveOperationException e) {
291 					throw unwrap(e);
292 				}
293 			});
294 			assertNotNull(b.header(), m::toString);
295 			assertEquals(name, b.header().getName(), m::toString);
296 			assertValue(type, b, m);
297 			if (types.length == 1 && !m.getParameterTypes()[0].isPrimitive() && !m.isVarArgs())
298 				assertNull(m.invoke(null, new Object[]{null}), m::toString);
299 		}
300 		assertTrue(factories >= 1, type::getName);
301 	}

303 	@Test
304 	void a02_httpHeadersFactories() throws Exception {
305 		var covered = 0;
306 		for (var m : HttpHeaders.class.getMethods()) {
307 			if (!Modifier.isStatic(m.getModifiers()) || m.getDeclaringClass() != HttpHeaders.class || m.getParameterCount() == 0 || m.getName().equals("cast") || !org.apache.http.Header.class.isAssignableFrom(m.getReturnType()))
308 				continue;
309 			var types = m.getGenericParameterTypes();
310 			if (!sampleable(types))
311 				continue;
312 			covered++;
313 			var b = build(types, true, a -> {
314 				try {
315 					return m.invoke(null, a);
316 				} catch (ReflectiveOperationException e) {
317 					throw unwrap(e);
318 				}
319 			});
320 			var h = b.header();
321 			assertNotNull(h, m::toString);
322 			assertNotNull(h.getValue(), m::toString);
323 			var ann = m.getReturnType().getAnnotation(Header.class);
324 			var declared = ann == null ? "" : ann.value().isEmpty() ? ann.name() : ann.value();
325 			if (!declared.isEmpty())
326 				assertEquals(declared, h.getName(), m::toString);
327 			else if (types.length > 1 && types[0] == String.class)
328 				assertEquals(NAME_SAMPLE, h.getName(), m::toString);
329 			else
330 				assertFalse(h.getName().isEmpty(), m::toString);
331 			var expected = expectedValue(b.args()[b.args().length - 1]);
332 			if (expected != null && m.getName().equals("contentDispositionAttachment"))
333 				expected = "attachment; filename=\"" + expected + "\"";  // The only factory that derives its value from the argument.
334 			if (expected != null && (types.length == 2 && types[0] == String.class || types.length == 1 && !declared.isEmpty()))
335 				assertEquals(expected, h.getValue(), m::toString);
336 		}
337 		assertEquals(180, covered, "Number of HttpHeaders header factories swept; update when a factory is added or removed.");
338 	}

340 	@Test
341 	void a03_sweepCounts() {
342 		var ctors = 0;
343 		var factories = 0;
344 		var skipped = new TreeSet<String>();
345 		for (var type : (Iterable<Class<?>>)a01_headerClasses()::iterator) {
346 			for (var c : type.getConstructors()) {
347 				var types = c.getGenericParameterTypes();
348 				if (types.length == 0)
349 					continue;
350 				if (sampleable(types))
351 					ctors++;
352 				else
353 					skipped.add(c.toString());
354 			}
355 			for (var m : type.getMethods()) {
356 				if (!m.getName().equals("of") || !Modifier.isStatic(m.getModifiers()) || m.getDeclaringClass() != type)
357 					continue;
358 				if (sampleable(m.getGenericParameterTypes()))
359 					factories++;
360 				else
361 					skipped.add(m.toString());
362 			}
363 		}
364 		assertEquals(Set.of(), skipped, "Every public constructor and of(...) factory must be sampleable by the sweep.");
365 		assertEquals(139, ctors, "Number of header constructors swept.");
366 		assertEquals(145, factories, "Number of header of(...) factories swept.");
367 	}
