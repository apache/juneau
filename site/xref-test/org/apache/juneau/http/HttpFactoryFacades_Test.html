35  /**
36   * Parametric coverage for the static-factory facade classes ({@link HttpHeaders}, {@link HttpBodies},
37   * {@link HttpResponses}).
38   *
39   * <p>
40   * Each facade is a thin DSL over the underlying named-type factories — invoking every public static
41   * method with a type-appropriate sample is the cheapest way to close out coverage on a couple of
42   * hundred otherwise-uncovered delegation lines.
43   */
44  class HttpFactoryFacades_Test extends TestBase {
45  
46  	static Stream<Method> facadeMethods() {
47  		return Stream.of(HttpHeaders.class, HttpBodies.class, HttpResponses.class)
48  			.flatMap(c -> Arrays.stream(c.getDeclaredMethods()))
49  			.filter(m -> Modifier.isPublic(m.getModifiers()))
50  			.filter(m -> Modifier.isStatic(m.getModifiers()));
51  	}
52  
53  	@ParameterizedTest(name = "{0}")
54  	@MethodSource("facadeMethods")
55  	void invokeWithSampleArgs(Method m) throws Exception {
56  		var args = sampleArgs(m);
57  		if (args == null)
58  			return; // unsupported parameter combination — covered elsewhere
59  		var result = m.invoke(null, args);
60  		if (m.getReturnType() != void.class)
61  			assertNotNull(result, m.getDeclaringClass().getSimpleName() + "." + m.getName());
62  	}
63  
64  	private static Object[] sampleArgs(Method m) {
65  		var params = m.getParameterTypes();
66  		var out = new Object[params.length];
67  		for (var i = 0; i < params.length; i++) {
68  			var v = sampleFor(params[i], m);
69  			if (v == NO_SAMPLE)
70  				return null;
71  			out[i] = v;
72  		}
73  		return out;
74  	}
75  
76  	private static final Object NO_SAMPLE = new Object();
77  	private static final ZonedDateTime SAMPLE_DATE = ZonedDateTime.parse("2024-01-15T08:30:00Z");
78  
79  	private static Object sampleFor(Class<?> p, Method m) {
80  		if (p == String.class) return wireStringFor(m);
81  		if (p == int.class || p == Integer.class) return Integer.valueOf(42);
82  		if (p == long.class || p == Long.class) return Long.valueOf(42L);
83  		if (p == boolean.class || p == Boolean.class) return Boolean.TRUE;
84  		if (p == ZonedDateTime.class) return SAMPLE_DATE;
85  		if (p == MediaType.class) return MediaType.of("text/plain");
86  		if (p == MediaRanges.class) return MediaRanges.of("text/plain");
87  		if (p == StringRanges.class) return StringRanges.of("en");
88  		if (p == EntityTag.class) return EntityTag.of("\"foo\"");
89  		if (p == EntityTags.class) return EntityTags.of("\"foo\"");
90  		if (p == URI.class) return URI.create("http://example.com");
91  		if (p == File.class) return new File("pom.xml");
92  		if (p == InputStream.class) return new ByteArrayInputStream(new byte[] { 1, 2, 3 });
93  		if (p == byte[].class) return new byte[] { 1, 2, 3 };
94  		if (p == String[].class) return new String[] { "a", "b" };
95  		if (p == HttpPart[].class) return new HttpPart[0];
96  		if (p == Supplier.class) return supplierFor(m);
97  		return NO_SAMPLE;
98  	}
99  
100 	/** Best-effort wire-format string by method-name keyword. */
101 	private static String wireStringFor(Method m) {
102 		var n = m.getName().toLowerCase();
103 		if (n.contains("date") || n.contains("modified") || n.contains("expires")) return "Wed, 21 Oct 2015 07:28:00 GMT";
104 		if (n.contains("contenttype") || n.contains("accept") && !n.contains("language") && !n.contains("encoding") && !n.contains("charset")) return "text/plain";
105 		if (n.contains("contentlength") || n.contains("age") || n.contains("retry") || n.contains("maxforwards")) return "42";
106 		if (n.contains("location") || n.contains("referer") || n.contains("origin") || n.contains("host")) return "http://example.com";
107 		if (n.contains("etag") || n.contains("ifmatch") || n.contains("ifnonematch") || n.contains("ifrange")) return "\"foo\"";
108 		return "value";
109 	}
110 
111 	/** Best-effort typed supplier by method-name keyword. */
112 	private static Supplier<?> supplierFor(Method m) {
113 		var n = m.getName().toLowerCase();
114 		if (n.contains("lazyparsed")) {
115 			if (n.contains("accept") && !n.contains("language") && !n.contains("encoding") && !n.contains("charset")) return () -> MediaRanges.of("text/plain");
116 			if (n.contains("contenttype")) return () -> MediaType.of("text/plain");
117 			if (n.contains("language") || n.contains("encoding") || n.contains("charset") || n.contains("disposition") || n.contains("te")) return () -> StringRanges.of("en");
118 		}
119 		if (n.contains("lazytokens")) return () -> new String[] { "a", "b" };
120 		if (n.equals("ifrange")) return () -> EntityTag.of("\"foo\"");
121 		if (n.equals("retryafter")) return () -> Integer.valueOf(120);
122 		return () -> "value";
123 	}
