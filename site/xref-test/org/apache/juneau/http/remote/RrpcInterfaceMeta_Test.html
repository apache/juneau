129 	private static java.lang.reflect.Method getMethod(Class<?> c, String name) {
130 		for (var m : c.getMethods())
131 			if (m.getName().equals(name))
132 				return m;
133 		throw new AssertionError("Method not found: " + name);
134 	}

140 	@Test void f01_headerList_set_emitsOneTimeWarning() {
141 		var records = new java.util.concurrent.CopyOnWriteArrayList<java.util.logging.LogRecord>();
142 		var handler = new java.util.logging.Handler() {
143 			@Override public void publish(java.util.logging.LogRecord r) { records.add(r); }
144 			@Override public void flush() { /* Test-harness no-op. */ }
145 			@Override public void close() { /* Test-harness no-op. */ }
146 		};
147 		var log = java.util.logging.Logger.getLogger("org.apache.juneau.http.remote");
148 		var useParent = log.getUseParentHandlers();
149 		log.setUseParentHandlers(false);
150 		log.addHandler(handler);
151 		try {
152 			// First resolution warns; subsequent resolutions are de-duplicated (one-time).
153 			RrpcInterfaceMeta.of(HeaderListIface.class);
154 			RrpcInterfaceMeta.of(HeaderListIface.class);
155 		} finally {
156 			log.removeHandler(handler);
157 			log.setUseParentHandlers(useParent);
158 		}
159 		var matches = records.stream().filter(r -> r.getMessage().contains("headerList")).count();
160 		assertEquals(1L, matches);
161 		var msg = records.stream().filter(r -> r.getMessage().contains("headerList")).findFirst().get().getMessage();
162 		assertTrue(msg.contains(HeaderListIface.class.getName()));
163 		assertTrue(msg.contains("not honored by the next-generation"));
164 	}
165 
166 	@Test void f02_headerList_unset_noWarning() {
167 		var records = new java.util.concurrent.CopyOnWriteArrayList<java.util.logging.LogRecord>();
168 		var handler = new java.util.logging.Handler() {
169 			@Override public void publish(java.util.logging.LogRecord r) { records.add(r); }
170 			@Override public void flush() { /* Test-harness no-op. */ }
171 			@Override public void close() { /* Test-harness no-op. */ }
172 		};
173 		var log = java.util.logging.Logger.getLogger("org.apache.juneau.http.remote");
174 		var useParent = log.getUseParentHandlers();
175 		log.setUseParentHandlers(false);
176 		log.addHandler(handler);
177 		try {
178 			RrpcInterfaceMeta.of(NoHeaderListIface.class);
179 		} finally {
180 			log.removeHandler(handler);
181 			log.setUseParentHandlers(useParent);
182 		}
183 		assertEquals(0L, records.stream().filter(r -> r.getMessage().contains("headerList")).count());
184 	}
