262 	// Raw-byte IPv6 builder: 'prefix' bytes, zero fill, and the IPv4 octets placed at the given byte offsets.
263 	private static InetAddress v6(int[] prefix, int[] offsets, int a, int b, int c, int d) throws UnknownHostException {
264 		var bytes = new byte[16];
265 		for (var i = 0; i < prefix.length; i++)
266 			bytes[i] = (byte) prefix[i];
267 		var v4 = new int[]{a, b, c, d};
268 		for (var i = 0; i < 4; i++)
269 			bytes[offsets[i]] = (byte) v4[i];
270 		return InetAddress.getByAddress(bytes);
271 	}
272 
273 	private static InetAddress nat64(int a, int b, int c, int d) throws UnknownHostException {
274 		return v6(new int[]{0x00, 0x64, 0xff, 0x9b}, new int[]{12, 13, 14, 15}, a, b, c, d);
275 	}
