49  	@Test void a03_factories() {
50  		assertEquals("application/json", ContentType.of("application/json").getValue());
51  		assertEquals("application/json", ContentType.of(MediaType.of("application/json")).getValue());
52  		assertEquals("application/json", ContentType.ofLazyWire(() -> "application/json").getValue());

54  	}

151 	@ParameterizedTest(name = "{0}")
152 	@MethodSource
153 	void b01_constants(String name, ContentType ct, String expected) {
154 		assertNotNull(ct, name + " is null");
155 		assertEquals(expected, ct.getValue(), name + " value mismatch");
156 		assertEquals("Content-Type", ct.getName(), name + " header name mismatch");
157 	}

159 	@Test void c01_noDuplicateMimeValues() {
160 		var seen = new HashSet<String>();
161 		var dupes = new ArrayList<String>();
162 		b01_constants().forEach(args -> {
163 			var v = (String) args.get()[2];
164 			if (! seen.add(v))
165 				dupes.add(v);
166 		});
167 		assertTrue(dupes.isEmpty(), () -> "Duplicate Content-Type values: " + dupes);
168 	}
