17  package org.apache.juneau.http.request;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.apache.juneau.*;
22  import org.apache.juneau.http.*;
23  import org.junit.jupiter.api.*;
24  
25  class HttpRequestLineBean_Test extends TestBase {
26  
27  	@Test void a01_of_defaultsToHttp11() {
28  		var r = HttpRequestLineBean.of("GET", "/users/123");
29  		assertEquals("GET", r.getMethod());
30  		assertEquals("/users/123", r.getUri());
31  		assertEquals(HttpProtocolVersion.HTTP_1_1, r.getProtocolVersion());
32  		assertEquals("GET /users/123 HTTP/1.1", r.toString());
33  	}
34  
35  	@Test void a02_of_typedProtocolVersion() {
36  		var r = HttpRequestLineBean.of("POST", "/api/echo", HttpProtocolVersion.HTTP_2_0);
37  		assertEquals(HttpProtocolVersion.HTTP_2_0, r.getProtocolVersion());
38  		assertEquals("POST /api/echo HTTP/2.0", r.toString());
39  	}
40  
41  	@Test void a03_of_parsedProtocolVersion() {
42  		var r = HttpRequestLineBean.of("GET", "/x", "HTTP/2");
43  		assertEquals(HttpProtocolVersion.of("HTTP", 2, 0), r.getProtocolVersion());
44  	}
45  
46  	@Test void a04_nullMethodRejected() {
47  		assertThrows(IllegalArgumentException.class, () -> HttpRequestLineBean.of(null, "/x"));
48  	}
49  
50  	@Test void a05_nullUriRejected() {
51  		assertThrows(IllegalArgumentException.class, () -> HttpRequestLineBean.of("GET", null));
52  	}
53  
54  	@Test void a06_nullProtocolVersionRejected() {
55  		assertThrows(IllegalArgumentException.class, () -> HttpRequestLineBean.of("GET", "/x", (HttpProtocolVersion)null));
56  		assertThrows(IllegalArgumentException.class, () -> HttpRequestLineBean.of("GET", "/x", (String)null));
57  	}
58  
59  	@Test void a07_equalsAndHashCode() {
60  		var a = HttpRequestLineBean.of("GET", "/x");
61  		var b = HttpRequestLineBean.of("GET", "/x");
62  		var c = HttpRequestLineBean.of("POST", "/x");
63  		var d = HttpRequestLineBean.of("GET", "/y");
64  		var e = HttpRequestLineBean.of("GET", "/x", HttpProtocolVersion.HTTP_2_0);
65  
66  		assertEquals(a, b);
67  		assertEquals(a.hashCode(), b.hashCode());
68  		assertNotEquals(a, c);
69  		assertNotEquals(a, d);
70  		assertNotEquals(a, e);
71  		assertNotEquals("not a request line", a);
72  		assertNotEquals(null, a);
73  	}
74  }
