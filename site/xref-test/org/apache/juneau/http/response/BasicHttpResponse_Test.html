73  	@Test void a07_copyCtor_copiesState() {
74  		var orig = new Ok().addHeader("X-Trace", "1").setLocale(java.util.Locale.GERMAN);
75  		var copy = new Ok(orig);
76  		assertEquals(1, copy.getHeaders().size());
77  		assertEquals(java.util.Locale.GERMAN, copy.getLocale());
78  		// Headers list is a defensive copy -- mutating the original doesn't affect the copy.
79  		orig.addHeader("X-More", "2");
80  		assertEquals(1, copy.getHeaders().size());
81  	}

83  	@Test void a08_equals_sameInstance() {
84  		var r = new Ok();

86  	}

88  	/*
89  	 * BUG (not silently fixed): equals()/hashCode() delegate to the body field's equals(), but
90  	 * StringBody has no equals()/hashCode() override (identity-based), so two independently constructed responses
91  	 * that would otherwise be logically identical (same status, same headers, same body *text*) never compare equal
92  	 * -- here specifically because BasicHttpResponse's own no-body constructor synthesizes a fresh StringBody from
93  	 * the reason phrase, and each Ok() call produces its own distinct StringBody instance. This test pins the
94  	 * current (surprising) behavior rather than the "value equality" contract the javadoc on equals() implies.
95  	 */
96  	@Test void a09_equals_independentInstances_neverEqual_dueToIdentityBasedBodyEquals() {
97  		var r1 = new Ok().addHeader("A", "1");
98  		var r2 = new Ok().addHeader("A", "1");
99  		assertNotEquals(r1, r2);
100 	}
101 
102 	@Test void a09b_equals_copyConstructor_sharesBodyInstance_isEqual() {
103 		var r1 = new Ok().addHeader("A", "1");
104 		var r2 = new Ok(r1);
105 		// Headers compare structurally (HttpHeaderList extends ArrayList), but the copy constructor also shares the
106 		// same body reference, which is what actually makes this pair equal despite the bug noted above.
107 		assertEquals(r1, r2);
108 		assertEquals(r1.hashCode(), r2.hashCode());
109 	}

129 	@Test void a13_equals_null() {
130 		// assertNotEquals(null, r) would call objectsAreEqual(null, r), which short-circuits on the null check
131 		// without ever invoking r.equals(null) -- assert the direction that actually exercises the `instanceof`
132 		// check's false branch.
133 		var r = new Ok();
134 		assertFalse(r.equals(null));
135 	}
136 
137 	@SuppressWarnings({
138 		"unlikely-arg-type" // Intentionally comparing to a mismatched type to cover the equals() type-guard branch.
139 	})
140 	@Test void a14_equals_differentType() {
141 		// Likewise, assertNotEquals("not a response", r) would call "not a response".equals(r), which trivially
142 		// returns false via String.equals() -- never touching BasicHttpResponse.equals() at all.
143 		var r = new Ok();
144 		assertFalse(r.equals("not a response"));
145 	}
