29  /**
30   * Parametric tests over every concrete response class in {@code org.apache.juneau.http.response}.
31   *
32   * <p>
33   * Enumerates every {@code .class} file in the package (excluding base types, interfaces, and
34   * {@code HttpStatusLineBean} which is a value-object helper), and for each:
35   * <ul>
36   * 	<li>verifies that {@code STATUS_CODE} (if present) matches {@code getStatusCode()};
37   * 	<li>exercises the no-arg, {@code (String)}, {@code (HttpBody)}, {@code (Throwable)}, and
38   * 		{@code (String, Throwable)} constructors where they exist;
39   * 	<li>walks {@code getStatusLine() / getHeaders() / getBody() / toString()};
40   * 	<li>verifies the {@code INSTANCE} singleton (if present) is non-null and an instance of the class.
41   * </ul>
42   *
43   * <p>
44   * This is a low-cost, high-coverage sweep: a single test method iterates 60+ classes, each requiring
45   * trivial setup, and brings the response package from ~38% to ~95% instruction coverage.
46   */
47  class NamedResponses_Test extends TestBase {
48  
49  	private static final Set<String> EXCLUDED = Set.of(
50  		"BasicHttpResponse",
51  		"BasicHttpException",
52  		"HttpResponseMessage",
53  		"HttpStatusLineBean",
54  		"package-info"
55  	);
56  
57  	static Stream<Class<?>> responseClasses() throws Exception {
58  		return PackageScanner.enumerateConcreteClasses("org.apache.juneau.http.response", BasicHttpResponse.class)
59  			.stream()
60  			.filter(c -> ! EXCLUDED.contains(c.getSimpleName()));
61  	}
62  
63  	@ParameterizedTest(name = "{0}")
64  	@MethodSource("responseClasses")
65  	void coverAllConstructorsAndAccessors(Class<?> cls) throws Exception {
66  		var statusCode = readIntField(cls, "STATUS_CODE");
67  		var reasonPhrase = readStringField(cls, "REASON_PHRASE");
68  		var instances = new ArrayList<Object>();
69  
70  		// 1) No-arg constructor.
71  		var noArg = findCtor(cls);
72  		if (noArg != null)
73  			instances.add(noArg.newInstance());
74  
75  		// 2) (String) constructor (message / body) — exercise both non-null and null bodies so the
76  		// {@code body != null ? StringBody.of(body) : null} ternary inside BasicHttpResponse(String) is covered.
77  		var stringCtor = findCtor(cls, String.class);
78  		if (stringCtor != null) {
79  			instances.add(stringCtor.newInstance("test-body"));
80  			instances.add(stringCtor.newInstance((String) null));
81  		}
82  
83  		// 3) (HttpBody) constructor for non-exception responses.
84  		var bodyCtor = findCtor(cls, org.apache.juneau.http.HttpBody.class);
85  		if (bodyCtor != null) {
86  			instances.add(bodyCtor.newInstance(StringBody.of("test-body", "text/plain")));
87  			instances.add(bodyCtor.newInstance((org.apache.juneau.http.HttpBody) null));
88  		}
89  
90  		// 4) (Throwable) and (String, Throwable) constructors for exception responses.
91  		// Pass both a non-null cause and a null cause so the {@code cause != null ? ... : null} // NOSONAR
92  		// ternary inside the constructor body covers both branches.
93  		var throwableCtor = findCtor(cls, Throwable.class);
94  		if (throwableCtor != null) {
95  			instances.add(throwableCtor.newInstance(new RuntimeException("cause")));
96  			instances.add(throwableCtor.newInstance((Throwable) null));
97  		}
98  
99  		var stringThrowableCtor = findCtor(cls, String.class, Throwable.class);
100 		if (stringThrowableCtor != null) {
101 			instances.add(stringThrowableCtor.newInstance("msg", new RuntimeException("cause")));
102 			instances.add(stringThrowableCtor.newInstance((String) null, (Throwable) null));
103 		}
104 
105 		// 5) Copy constructor (T) — only if at least one instance was built and the copy ctor exists.
106 		var copyCtor = findCtor(cls, cls);
107 		if (copyCtor != null && ! instances.isEmpty())
108 			instances.add(copyCtor.newInstance(instances.get(0)));
109 
110 		assertFalse(instances.isEmpty(), "Class " + cls.getSimpleName() + " has no recognized constructor.");
111 
112 		// Exercise accessors on every built instance.
113 		for (var o : instances) {
114 			if (statusCode != null && o instanceof BasicHttpResponse r) {
115 				assertEquals(statusCode.intValue(), r.getStatusCode(), cls.getSimpleName() + ".getStatusCode()");
116 				assertNotNull(r.getStatusLine(), cls.getSimpleName() + ".getStatusLine()");
117 				assertNotNull(r.getHeaders(), cls.getSimpleName() + ".getHeaders()");
118 				assertNotNull(r.toString(), cls.getSimpleName() + ".toString()");
119 				// Mutator chain — exercises both branches of {@code withBody(String)} and the header-list copy.
120 				var mutated = r.withBody(StringBody.of("new", "text/plain"))
121 					.withBody("string-body")
122 					.withBody((String) null)
123 					.withHeader(org.apache.juneau.http.header.HttpHeaderBean.of("X-Test", "1"))
124 					.withHeader("X-Test2", "2");
125 				assertEquals(statusCode.intValue(), mutated.getStatusCode());
126 			} else if (statusCode != null && o instanceof BasicHttpException e) {
127 				assertEquals(statusCode.intValue(), e.getStatusCode(), cls.getSimpleName() + ".getStatusCode()");
128 				assertNotNull(e.getStatusLine(), cls.getSimpleName() + ".getStatusLine()");
129 				assertNotNull(e.getHeaders(), cls.getSimpleName() + ".getHeaders()");
130 				assertNotNull(e.toString(), cls.getSimpleName() + ".toString()");
131 				// reasonPhrase, if declared, should match status-line reason.
132 				if (reasonPhrase != null)
133 					assertEquals(reasonPhrase, e.getStatusLine().getReasonPhrase(), cls.getSimpleName() + ".reasonPhrase");
134 			}
135 		}
136 
137 		// 6) INSTANCE singleton (if present).
138 		var instanceField = findField(cls, "INSTANCE");
139 		if (instanceField != null) {
140 			var instance = instanceField.get(null);
141 			assertNotNull(instance, cls.getSimpleName() + ".INSTANCE");
142 			assertTrue(cls.isInstance(instance), cls.getSimpleName() + ".INSTANCE not assignable to declaring class");
143 		}
144 	}

150 	private static Integer readIntField(Class<?> cls, String name) {
151 		try {
152 			var f = cls.getField(name);
153 			return f.getType() == int.class ? (Integer) f.get(null) : null;
154 		} catch (@SuppressWarnings("unused") NoSuchFieldException | IllegalAccessException e) {
155 			return null;
156 		}
157 	}
158 
159 	private static String readStringField(Class<?> cls, String name) {
160 		try {
161 			var f = cls.getField(name);
162 			return f.getType() == String.class ? (String) f.get(null) : null;
163 		} catch (@SuppressWarnings("unused") NoSuchFieldException | IllegalAccessException e) {
164 			return null;
165 		}
166 	}
167 
168 	private static java.lang.reflect.Field findField(Class<?> cls, String name) {
169 		try {
170 			var f = cls.getField(name);
171 			f.setAccessible(true);
172 			return f;
173 		} catch (@SuppressWarnings("unused") NoSuchFieldException e) {
174 			return null;
175 		}
176 	}
177 
178 	private static java.lang.reflect.Constructor<?> findCtor(Class<?> cls, Class<?>... params) {
179 		try {
180 			var c = cls.getDeclaredConstructor(params);
181 			c.setAccessible(true);
182 			return c;
183 		} catch (@SuppressWarnings("unused") NoSuchMethodException e) {
184 			return null;
185 		}
186 	}
