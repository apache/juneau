39  	@Test void a01_stringBody_defaultContentType() throws Exception {
40  		var b = StringBody.of("hello");
41  		assertEquals("text/plain; charset=UTF-8", b.getContentType());
42  		assertEquals(5, b.getContentLength());
43  		assertTrue(b.isRepeatable());
44  		assertEquals("hello", b.toString());
45  		var out = new ByteArrayOutputStream();
46  		b.writeTo(out);
47  		assertEquals("hello", out.toString());
48  	}

55  		assertEquals("{}", out.toString());
56  	}

164 	@Test void f01_multipartBody_buildsAndStreams(@TempDir Path dir) throws Exception {
165 		var report = dir.resolve("report.pdf");
166 		Files.writeString(report, "PDF-CONTENT");
167 		var body = MultipartBody.builder()
168 			.boundary("boundary123")
169 			.field("title", "My Report")
170 			.file("attachment", report.toFile(), "application/pdf")
171 			.part(MultipartBody.MultipartPart.of("notes", null, "text/plain", StringBody.of("ok")))
172 			.build();
173 		assertEquals("multipart/form-data; boundary=boundary123", body.getContentType());
174 		assertEquals(-1, body.getContentLength());
175 		assertTrue(body.isRepeatable(), "all parts (string + file + string) are repeatable");
176 		assertEquals("boundary123", body.getBoundary());
177 		assertEquals(3, body.getParts().size());
178 
179 		var out = new ByteArrayOutputStream();
180 		body.writeTo(out);
181 		var wire = out.toString();
182 		assertTrue(wire.contains("--boundary123"));
183 		assertTrue(wire.contains("Content-Disposition: form-data; name=\"title\""));
184 		assertTrue(wire.contains("My Report"));
185 		assertTrue(wire.contains("filename=\"report.pdf\""));
186 		assertTrue(wire.contains("Content-Type: application/pdf"));
187 		assertTrue(wire.contains("PDF-CONTENT"));
188 		assertTrue(wire.endsWith("--boundary123--\r\n"));
189 	}
