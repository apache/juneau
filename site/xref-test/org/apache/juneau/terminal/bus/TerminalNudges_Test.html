43  	/** Records every accepted frame. */
44  	static final class RecordingSink implements FrameSink {
45  		final List<String> frames = new ArrayList<>();
46  
47  		@Override public boolean offer(String frame) {
48  			frames.add(frame);
49  			return true;
50  		}
51  
52  		@Override public void close(int code, String reason) { /* no-op: the recorder keeps frames only */ }
53  
54  		void reset() { frames.clear(); }
55  
56  		/** The payload JSON of every frame received on a topic. */
57  		List<String> on(String topic) {
58  			var out = new ArrayList<String>();
59  			for (var f : frames) {
60  				try {
61  					var m = BusFrames.decode(f);
62  					if (topic.equals(m.get("topic")))
63  						out.add(Json.of(m.get("payload")));
64  				} catch (BusRefusal e) {
65  					throw new AssertionError(f, e);
66  				}
67  			}
68  			return out;
69  		}
70  	}

72  	/** One fake time source behind both the scheduler and the clock. */
73  	static final class Manual implements TerminalNudges.Scheduler {
74  		long now;
75  		boolean closed, armClockFailure, requireStart, started;
76  		Runnable onStart;
77  		private long seq;
78  		private final PriorityQueue<long[]> queue = new PriorityQueue<>((a, b) -> a[0] == b[0] ? Long.compare(a[1], b[1]) : Long.compare(a[0], b[0]));
79  		private final Map<Long,Runnable> tasks = new HashMap<>();
80  
81  		@Override public void start() {
82  			if (onStart != null)
83  				onStart.run();
84  			started = true;
85  		}
86  
87  		@Override public void schedule(Runnable r, long delayMs) {
88  			if (requireStart && ! started)
89  				return;
90  			var id = seq++;
91  			tasks.put(id, r);
92  			queue.add(new long[]{now + delayMs, id});
93  		}
94  
95  		@Override public void close() { closed = true; }
96  
97  		long clock() {
98  			if (armClockFailure) {
99  				armClockFailure = false;
100 				throw new IllegalStateException("clock failure");
101 			}
102 			return now;
103 		}
104 
105 		int pending() { return queue.size(); }
106 
107 		void advance(long ms) {
108 			now += ms;
109 			while (! queue.isEmpty() && queue.peek()[0] <= now)
110 				tasks.remove(queue.poll()[1]).run();
111 		}
112 	}

119 	private final TerminalAccess access = new TerminalAccess() {
120 		@Override public boolean canWatch(TerminalCaller c, String id) {
121 			if (onWatch != null)
122 				onWatch.run();
123 			if (failHard)
124 				throw new AssertionError("hard failure");
125 			if (throwing.contains(c.principalName()))
126 				throw new IllegalStateException("access failure");
127 			return allowed.contains(c.principalName());
128 		}
129 		@Override public Optional<TerminalSink> inputSink(TerminalCaller c, String id) {
130 			return Optional.empty();
131 		}
132 	};

159 	private static HttpServletRequest req(String user) {
160 		Principal principal = user == null ? null : () -> user;
161 		return (HttpServletRequest)Proxy.newProxyInstance(HttpServletRequest.class.getClassLoader(), new Class<?>[]{HttpServletRequest.class},
162 			(proxy, method, args) -> switch (method.getName()) {
163 				case "getUserPrincipal" -> principal;
164 				case "toString" -> "FakeRequest[" + user + "]";
165 				case "hashCode" -> System.identityHashCode(proxy);
166 				case "equals" -> proxy == args[0];
167 				default -> method.getReturnType() == boolean.class ? (Object)false : method.getReturnType() == int.class ? (Object)0 : null;
168 			});
169 	}

171 	private static BusSession open(ServerBus bus, String user, String...topics) throws BusRefusal {
172 		return bus.openSession(req(user), JsonMap.of("v", 1, "bridge", "term", "transport", "websocket",
173 			"downstream", List.of(topics), "upstream", List.of()));
174 	}
175 
176 	private static RecordingSink connect(ServerBus bus, BusSession s) {
177 		var sink = new RecordingSink();
178 		bus.connect(s, sink);
179 		sink.reset();
180 		return sink;
181 	}
182 
183 	private RecordingSink join(String user, String...topics) throws BusRefusal {
184 		return connect(bus, open(bus, user, topics));
185 	}

515 	private static Thread find() {
516 		return Thread.getAllStackTraces().keySet().stream().filter(t -> t.getName().equals("juneau-terminal-nudges") && t.isAlive()).findFirst().orElse(null);
517 	}
