47  /**
48   * End to end for terminal input: typed keys in a terminal card travel over a real bus WebSocket to
49   * {@link TerminalInput}, are authorized by a {@link TerminalAccess}, reach a job and show up in the terminal.
50   *
51   * <p>
52   * Everything is real: a Jetty on a loopback port hosts the bus ({@link BusWebSockets}), the terminal and
53   * terminal-view mixins and the static assets (the real {@code juneau-bus.js}, {@code juneau-terminal.js} and xterm.js), and
54   * Chromium runs {@code terminal-input-browser.cjs}.  The page mounts the terminal directly, without the console shell: it
55   * declares the two topics, attaches the bridge with {@code attachBridges} and hands {@code mount} a link built from the
56   * page bus.
57   *
58   * <p>
59   * Each job is a {@link TerminalProcess} started with input on, so the keys go through {@code juneau_terminal.py}'s
60   * pseudo-terminal: Enter arrives as CR and the terminal turns it into LF, and {@code ^C} is SIGINT from the terminal's
61   * line discipline.  The test needs {@code python3} on the {@code PATH} and is skipped without it.
62   *
63   * <h5 class='section'>Off by default:</h5>
64   * <p>
65   * Binds a loopback socket, so it is off by default, including under {@code -Pjs-tests}.  Run it with
66   * {@code -Pjs-tests -Dtest='TerminalInput_BrowserTest' -D}{@value #GATE}{@code =true}, outside the sandbox; the profile
67   * supplies the node, harness and browser properties.
68   *
69   * @since 10.0.0
70   */
71  @DisabledOnOs(OS.WINDOWS)

107 		@Override protected void doGet(HttpServletRequest req, jakarta.servlet.http.HttpServletResponse res) throws IOException {
108 			var path = req.getPathInfo();
109 			if (path == null || path.equals("/page")) {
110 				write(res, "text/html;charset=utf-8", page(req.getParameter("id"), "1".equals(req.getParameter("typist")), "1".equals(req.getParameter("input"))).getBytes(UTF_8));
111 				return;
112 			}
113 			var name = path.substring("/a/".length());
114 			var type = name.endsWith(".css") ? "text/css" : "text/javascript";
115 			write(res, type, asset(name));
116 		}

118 		private static void write(jakarta.servlet.http.HttpServletResponse res, String type, byte[] body) throws IOException {
119 			res.setContentType(type);
120 			res.getOutputStream().write(body);
121 		}

124 	@BeforeAll static void setUp() throws Exception {
125 		var path = System.getenv("PATH");
126 		assumeTrue(path != null && Arrays.stream(path.split(File.pathSeparator)).anyMatch(d -> Files.isExecutable(Path.of(d, "python3"))),
127 			"python3 not on the PATH - skipped");
128 		scriptBefore = System.setProperty(TerminalProcess.SCRIPT_PROPERTY, script().toString());
129 		dir = Files.createTempDirectory("terminal-input");
130 		job("t-prompt", "printf 'Continue? [y/N] '; read a; echo \"got $a\"");
131 		job("t-sleep", "echo sleeping; exec sleep 300");
132 		job("t-idle", "echo idle; exec sleep 300");
133 		job("t-tick", "i=0; while true; do echo \"tick $i\"; i=$((i+1)); sleep 0.3; done");
134 
135 		server = new Server();
136 		var connector = new ServerConnector(server);
137 		connector.setHost("127.0.0.1");
138 		connector.setPort(0);
139 		server.addConnector(connector);
140 		connector.open();                       // the port, and so the exact Origin, is known before start
141 		var origin = "http://127.0.0.1:" + connector.getLocalPort();
142 
143 		// Everyone may watch; only a session opened with ?typist=1 gets a sink.
144 		TerminalAccess access = new TerminalAccess() {
145 			@Override public boolean canWatch(TerminalCaller caller, String id) { return JOBS.containsKey(id); }
146 			@Override public Optional<TerminalSink> inputSink(TerminalCaller caller, String id) {
147 				return Boolean.TRUE.equals(caller.attribute("typist")) ? Optional.ofNullable(JOBS.get(id)) : Optional.empty();
148 			}
149 		};
150 		var input = TerminalInput.create(access);
151 		bus = ServerBus.create(BusPolicy.create()
152 			.allowedOrigins(origin)
153 			.upstream(TerminalInput.TOPIC, input)
154 			.downstream(TerminalInput.STATUS_TOPIC, false)
155 			.sessionAttributes(r -> Map.of("typist", "1".equals(r.getParameter("typist"))))
156 			.build());
157 		input.bind(bus);
158 
159 		var ctx = new ServletContextHandler();
160 		ctx.setContextPath("/");
161 		ctx.addServlet(new ServletHolder(new R()), "/rest/*");
162 		ctx.addServlet(new ServletHolder(new Pages()), "/*");
163 		JakartaWebSocketServletContainerInitializer.configure(ctx, (sc, container) -> BusWebSockets.register(container, bus));
164 		server.setHandler(ctx);
165 		server.start();
166 
167 		var js = Path.of(requiredProperty("juneau.jsTests.dir"));
168 		var harness = Path.of(requiredProperty("juneau.jsTests.harnessDir"), "terminal-input-browser.cjs");
169 		var out = runNode(
170 			List.of(System.getProperty("juneau.jsTests.node", "node"), harness.toString(), origin + "/"),
171 			Map.of("NODE_PATH", js.resolve("node_modules").toString(), "PLAYWRIGHT_BROWSERS_PATH", requiredProperty("juneau.jsTests.browsers")),
172 			300);
173 		report = Json.to(out, Map.class);
174 		assertNull(report.get("harnessError"), () -> "prober failed: " + report.get("harnessError"));
175 	}

177 	@AfterAll static void tearDown() throws Exception {
178 		try {
179 			if (server != null)
180 				server.stop();
181 		} finally {
182 			try {
183 				if (bus != null)
184 					bus.close();
185 			} finally {
186 				JOBS.values().forEach(TerminalProcess::close);
187 				if (scriptBefore == null)
188 					System.clearProperty(TerminalProcess.SCRIPT_PROPERTY);
189 				else
190 					System.setProperty(TerminalProcess.SCRIPT_PROPERTY, scriptBefore);
191 				if (dir != null) {
192 					try (var s = Files.walk(dir)) {
193 						for (var p : s.sorted(Comparator.reverseOrder()).toList())
194 							Files.deleteIfExists(p);
195 					}
196 				}
197 			}
198 		}

206 	/** {@code juneau-terminal/src/main/python/juneau_terminal.py}, from {@code basedir} or the working directory. */
207 	private static Path script() {
208 		var basedir = System.getProperty("basedir");
209 		for (var p : List.of(Path.of(basedir == null ? "." : basedir, "../../juneau-terminal/src/main/python/juneau_terminal.py"), Path.of("juneau-terminal/src/main/python/juneau_terminal.py")))
210 			if (Files.isRegularFile(p))
211 				return p.toAbsolutePath().normalize();
212 		throw new AssertionError("cannot find juneau-terminal/src/main/python/juneau_terminal.py");
213 	}

215 	private static String page(String id, boolean typist, boolean input) {
216 		if (id == null || ! ID.matcher(id).matches() || ! JOBS.containsKey(id))
217 			throw new IllegalArgumentException("bad id");
218 		return "<!DOCTYPE html><html><head><meta charset=\"utf-8\"><link rel=\"icon\" href=\"data:,\">"
219 			+ "<link rel=\"stylesheet\" href=\"/a/xterm.css\"><link rel=\"stylesheet\" href=\"/a/juneau-terminal.css\">"
220 			+ "<script src=\"/a/juneau-bus.js\"></script><script src=\"/a/xterm.js\"></script><script src=\"/a/juneau-terminal.js\"></script>"
221 			+ "</head><body><div id=\"host\" style=\"width:1000px\"></div><script>"
222 			+ "const B = window.JuneauViews.bus;"
223 			+ "B.declare('term.input', {retain: false});"
224 			+ "B.declare('term.input-status', {retain: false});"
225 			+ "window.JuneauViews.bus.wiring.attachBridges({bridges: [{id: 'term', transport: 'websocket',"
226 			+ " session: '/rest" + BusEventsMixin.SESSION_PATH + (typist ? "?typist=1" : "") + "',"
227 			+ " downstream: ['term.input-status'], upstream: ['term.input']}]});"
228 			// The console routes the card's input-status role; here the page does it, through the registered card handler.
229 			+ "const role = window.JuneauConsoleCards.find(function (e) { return e[0] === 'terminal'; })[1].roles['input-status'];"
230 			+ "B.subscribe('term.input-status', function (p, m) { role(p, m, {id: '" + id + "'}); });"
231 			+ "window.JuneauTerminal.mount(document.getElementById('host'), {id: '" + id + "', refreshMs: 250,"
232 			+ (input ? " input: true, inputId: '" + id + "'," : "")
233 			+ " bytesUrl: '/rest" + TerminalMixin.TERMINAL_PREFIX + "/" + id + "/bytes',"
234 			+ " eventsUrl: '/rest" + TerminalViewMixin.RUN_VIEW_PREFIX + "/" + id + "/events'},"
235 			+ " {publish: function (p) { B.publish('term.input', p); },"
236 			+ " onConnected: function (fn) { return B.subscribe('bridge:term', function (s) { fn(!!s && s.state === 'open'); }); }});"
237 			+ "</script></body></html>";
238 	}

240 	/** The bytes of a classpath asset: this module's dependencies carry the three scripts, the stylesheet and xterm. */
241 	private static byte[] asset(String name) throws IOException {
242 		var path = switch (name) {
243 			case "xterm.js" -> WebJarResolver.WEBJARS_ROOT + WebJarResolver.resolvePath(TerminalMixin.XTERM_JS_ASSET);
244 			case "xterm.css" -> WebJarResolver.WEBJARS_ROOT + WebJarResolver.resolvePath(TerminalMixin.XTERM_CSS_ASSET);
245 			case "juneau-terminal.js" -> "org/apache/juneau/terminal/juneau-terminal.js";
246 			case "juneau-terminal.css" -> "org/apache/juneau/terminal/juneau-terminal.css";
247 			case "juneau-bus.js" -> "org/apache/juneau/views/juneau-bus.js";
248 			default -> throw new IllegalArgumentException(name);
249 		};
250 		try (var in = TerminalInput_BrowserTest.class.getClassLoader().getResourceAsStream(path)) {
251 			assertNotNull(in, () -> "missing classpath resource: " + path);
252 			return in.readAllBytes();
253 		}
254 	}

256 	/** Runs the token list (never a shell string) and returns its stdout; the child and its descendants are killed in {@code finally}. */
257 	private static String runNode(List<String> cmd, Map<String,String> env, int seconds) throws Exception {
258 		var stdout = Files.createTempFile("terminal-input-stdout-", ".json");
259 		var stderr = Files.createTempFile("terminal-input-stderr-", ".txt");
260 		Process p = null;
261 		try {
262 			var pb = new ProcessBuilder(cmd).redirectOutput(stdout.toFile()).redirectError(stderr.toFile());
263 			pb.environment().putAll(env);
264 			p = pb.start();
265 			if (! p.waitFor(seconds, TimeUnit.SECONDS))
266 				fail(cmd + " did not finish within " + seconds + "s; stderr:\n" + Files.readString(stderr, UTF_8));
267 			if (p.exitValue() != 0)
268 				fail(cmd + " exited " + p.exitValue() + "; stderr:\n" + Files.readString(stderr, UTF_8) + "\nstdout:\n" + Files.readString(stdout, UTF_8));
269 			return Files.readString(stdout, UTF_8);
270 		} finally {
271 			if (p != null && p.isAlive()) {
272 				p.descendants().forEach(ProcessHandle::destroyForcibly);
273 				p.destroyForcibly();
274 			}
275 			Files.deleteIfExists(stdout);
276 			Files.deleteIfExists(stderr);
277 		}
278 	}
