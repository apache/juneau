20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.io.*;
23  import java.nio.file.*;
24  import java.util.*;
25  import java.util.concurrent.*;

27  import org.apache.juneau.marshall.marshaller.*;

30  /**
31   * Node plumbing for this module's harnesses under {@code src/test/js}.
32   *
33   * <p>
34   * The assets a harness reads are classpath resources copied to temp files: this module's
35   * {@code juneau-terminal.js} and {@code .css}, and the xterm.js bundle and stylesheet from the WebJar.
36   */
37  final class TerminalHarness {
38  
39  	/** The six assets, in the order {@code terminal-browser.cjs} takes them. */
40  	static final List<String> BROWSER_ASSETS = List.of("xterm.js", "xterm.css", "juneau-terminal.js", "juneau-terminal.css", "juneau-views.css", "juneau-theme-dark.css");
41  
42  	private static final Map<String,Map<?,?>> CACHE = new ConcurrentHashMap<>();
43  
44  	private TerminalHarness() {}
45  
46  	/**
47  	 * Runs the plain-Node harness {@code terminal.cjs} over the runtime and the real xterm.js, or returns
48  	 * {@code null} when Node is not on the {@code PATH}.
49  	 */
50  	static Map<?,?> nodeReport() {
51  		return CACHE.computeIfAbsent("terminal.cjs", k -> {
52  			if (! nodeAvailable())
53  				return null;
54  			try {
55  				var out = run(List.of("node", locate("terminal.cjs").toString()), List.of("juneau-terminal.js", "xterm.js"), Map.of(), 60);
56  				return Json.to(out, Map.class);
57  			} catch (Exception e) {
58  				throw new AssertionError("could not run terminal.cjs: " + e, e);
59  			}
60  		});
61  	}
62  
63  	/**
64  	 * Runs {@code cmd} with the named assets appended as temp-file arguments and returns its stdout.
65  	 *
66  	 * <p>
67  	 * {@code cmd} is a token list given to {@link ProcessBuilder}, never a shell string.  The child and its descendants are
68  	 * killed in {@code finally}, so a failed or interrupted wait leaves nothing running.
69  	 *
70  	 * @param cmd The command.
71  	 * @param assets Names from {@link #BROWSER_ASSETS}.
72  	 * @param env Extra environment variables.
73  	 * @param seconds The longest wait.
74  	 */
75  	static String run(List<String> cmd, List<String> assets, Map<String,String> env, int seconds) throws Exception {
76  		var temps = new ArrayList<Path>();
77  		var stdout = Files.createTempFile("terminal-harness-stdout-", ".json");
78  		var stderr = Files.createTempFile("terminal-harness-stderr-", ".txt");
79  		Process p = null;
80  		try {
81  			var args = new ArrayList<>(cmd);
82  			for (var a : assets) {
83  				var f = Files.createTempFile("terminal-harness-", "-" + a);
84  				temps.add(f);
85  				Files.write(f, asset(a));
86  				args.add(f.toString());
87  			}
88  			var pb = new ProcessBuilder(args).redirectOutput(stdout.toFile()).redirectError(stderr.toFile());
89  			pb.environment().putAll(env);
90  			p = pb.start();
91  			if (! p.waitFor(seconds, TimeUnit.SECONDS))
92  				fail(cmd + " did not finish within " + seconds + "s; stderr:\n" + quietRead(stderr));
93  			if (p.exitValue() != 0)
94  				fail(cmd + " exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr) + "\nstdout:\n" + quietRead(stdout));
95  			return Files.readString(stdout, UTF_8);
96  		} finally {
97  			if (p != null && p.isAlive()) {
98  				p.descendants().forEach(ProcessHandle::destroyForcibly);
99  				p.destroyForcibly();
100 			}
101 			for (var f : temps)
102 				Files.deleteIfExists(f);
103 			Files.deleteIfExists(stdout);
104 			Files.deleteIfExists(stderr);
105 		}
106 	}
107 
108 	/** The bytes of one of {@link #BROWSER_ASSETS}. */
109 	static byte[] asset(String name) throws IOException {
110 		var path = switch (name) {
111 			case "xterm.js" -> WebJarResolver.WEBJARS_ROOT + WebJarResolver.resolvePath(TerminalMixin.XTERM_JS_ASSET);
112 			case "xterm.css" -> WebJarResolver.WEBJARS_ROOT + WebJarResolver.resolvePath(TerminalMixin.XTERM_CSS_ASSET);
113 			case "juneau-terminal.js" -> TerminalEndpoints.JS_RESOURCE.substring(1);
114 			case "juneau-terminal.css" -> TerminalEndpoints.CSS_RESOURCE.substring(1);
115 			case "juneau-views.css" -> "org/apache/juneau/views/juneau-views.css";
116 			case "juneau-theme-dark.css" -> "org/apache/juneau/console/juneau-theme-dark.css";
117 			default -> throw new IllegalArgumentException(name);
118 		};
119 		try (var in = TerminalHarness.class.getClassLoader().getResourceAsStream(path)) {
120 			assertNotNull(in, () -> "missing classpath resource: " + path);
121 			return in.readAllBytes();
122 		}
123 	}
124 
125 	/** {@code src/test/js/<name>} under the module, from {@code basedir} or the working directory. */
126 	static Path locate(String name) {
127 		var basedir = System.getProperty("basedir");
128 		for (var p : List.of(Path.of(basedir == null ? "." : basedir, "src/test/js", name), Path.of("juneau-rest/juneau-terminal-region/src/test/js", name)))
129 			if (Files.isRegularFile(p))
130 				return p.toAbsolutePath().normalize();
131 		throw new AssertionError("cannot find src/test/js/" + name);
132 	}
133 
134 	private static boolean nodeAvailable() {
135 		try {
136 			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
137 			if (! p.waitFor(5, TimeUnit.SECONDS)) {
138 				p.destroyForcibly();
139 				return false;
140 			}
141 			return p.exitValue() == 0;
142 		} catch (Exception e) {
143 			return false;
144 		}
145 	}
146 
147 	static String quietRead(Path p) {
148 		try {
149 			return Files.readString(p, UTF_8);
150 		} catch (IOException e) {
151 			return "(unreadable: " + e.getMessage() + ")";
152 		}
153 	}
154 }
