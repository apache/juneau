49  	/** Writes a fake python3 that records its argv (one per line) next to itself, then runs the given shell body. */
50  	private void fakePython(String body) throws Exception {
51  		var p = bin.resolve("python3");
52  		Files.writeString(p, "#!/bin/sh\nfor a in \"$@\"; do printf '%s\\n' \"$a\"; done > \"$(dirname \"$0\")/argv.txt\"\n" + body + "\n");
53  		Files.setPosixFilePermissions(p, PosixFilePermissions.fromString("rwxr-xr-x"));
54  	}

60  	private static List<String> events(Path f) throws Exception {

73  		assertEquals("{\"cols\":100,\"rows\":30}", Files.readString(run.resolve(TerminalProcess.LOG_NAME + ".size")));

112 		fakePython("printf '{\"ev\":\"step\",\"id\":\"a\",\"title\":\"A\"}\\n{\"ev\":\"note\",\"level\":\"info\",\"te' > \"${10}\"\nexit 5");
113 		var p = TerminalProcess.start("generic", List.of("true"), 80, 24, run, host("/x.py", null));

123 		assertTrue(records.get(0).getMessage().contains("skipped malformed line 2"), records.get(0).getMessage());
124 		assertTrue(page[0].terminal());

179 		Files.createDirectories(run);
180 		var f = run.resolve("e.jsonl");
181 		var note = "{\"ev\":\"note\",\"level\":\"info\",\"text\":\"" + "x".repeat(100) + "\"}\n";
182 		Files.writeString(f, "{\"ev\":\"done\",\"status\":\"ok\"}\n" + note.repeat(TerminalProcess.TAIL_BYTES / note.length() + 2));
183 		TerminalProcess.ensureDone(f, 3);
184 		var lines = Files.readAllLines(f);
185 		assertEquals("{\"ev\":\"done\",\"status\":\"fail\"}", lines.get(lines.size() - 1));
186 		var g = run.resolve("g.jsonl");
187 		Files.writeString(g, note.repeat(TerminalProcess.TAIL_BYTES / note.length() + 2) + "{\"ev\":\"done\",\"status\":\"ok\"}\n");
188 		var before = Files.size(g);
189 		TerminalProcess.ensureDone(g, 0);
190 		assertEquals(before, Files.size(g));

216 	@DisabledOnOs(OS.WINDOWS)
217 	@Test void b07_closeKillsTheWholeProcessTree() throws Exception {
218 		var pidFile = bin.resolve("pid");
219 		fakePython("sleep 300 &\necho $! > '" + pidFile + ".tmp'\nmv '" + pidFile + ".tmp' '" + pidFile + "'\nwait");
220 		var p = TerminalProcess.start("generic", List.of("true"), 80, 24, run, host("/x.py", null));
221 		long pid = -1;
222 		try {
223 			assertTrue(awaitFile(pidFile, Duration.ofSeconds(10)), "the grandchild never started");
224 			pid = Long.parseLong(Files.readString(pidFile).trim());
225 			var grandchild = ProcessHandle.of(pid);
226 			assertTrue(grandchild.map(ProcessHandle::isAlive).orElse(false), "the grandchild should be running");
227 			p.close();
228 			assertTrue(await(() -> ! grandchild.map(ProcessHandle::isAlive).orElse(false), Duration.ofSeconds(5)), "the grandchild survived close()");
229 		} finally {
230 			if (pid > 0)
231 				ProcessHandle.of(pid).ifPresent(ProcessHandle::destroyForcibly);
232 			p.close();
233 		}
234 	}
