18  
19  import static org.apache.juneau.bct.BctAssertions.*;
20  import static org.apache.juneau.terminal.report.ReportTestSupport.*;
21  import static org.junit.jupiter.api.Assertions.*;
22  
23  import java.nio.file.*;
24  import java.util.*;
25  
26  import org.apache.juneau.*;
27  import org.junit.jupiter.api.*;
28  import org.junit.jupiter.api.io.*;
29  
30  class JestJsonReportReader_Test extends TestBase {
31  
32  	@Test void a01_mapping() {
33  		var r = new JestJsonReportReader().read(resource("jest-mixed.json"));
34  		assertEquals(List.of("pass", "skip", "fail"), statuses(r));
35  		assertBean(r.tests().get(2), "suite,msg", "b.test.js,expected 1");
36  		assertEquals("jest", r.fw());
37  		assertEquals(4L, r.tests().get(0).ms());
38  		assertNull(r.tests().get(1).ms());
39  	}
40  
41  	@Test void a02_traceJoinsAllFailureMessages() {
42  		var t = new JestJsonReportReader().read(resource("jest-mixed.json")).tests().get(2);
43  		assertEquals("expected 1\nat x\nsecond\nat y", t.trace());
44  	}
45  
46  	@Test void a03_nameIsFullNameElseAncestorsPlusTitle() {
47  		var r = new JestJsonReportReader().read(resource("jest-mixed.json"));
48  		assertEquals("math adds", r.tests().get(0).name());
49  		assertEquals("api get returns 200", r.tests().get(2).name());
50  	}
51  
52  	@Test void a04_singleFileKeepsLastTwoSegments() {
53  		var r = new JestJsonReportReader().read(resource("jest-crashed.json"));
54  		assertEquals("src/c.test.js", r.tests().get(0).suite());
55  	}
56  
57  	@Test void b01_crashedFileIsOneErrorTest() {
58  		var r = new JestJsonReportReader().read(resource("jest-crashed.json"));
59  		assertSize(1, r.tests());
60  		assertBean(r.tests().get(0), "name,status,msg", "(suite failed to run),ERROR,Cannot find module './x'");
61  	}
62  
63  	@Test void b02_truncatedIsZeroTestsWithWarning() {
64  		var r = new JestJsonReportReader().read(resource("jest-truncated.json"));
65  		assertTrue(r.tests().isEmpty());
66  		assertTrue(r.warnings().get(0).startsWith("report unreadable"), r.warnings().get(0));
67  		assertTrue(r.warnings().get(0).endsWith("(jest-truncated.json)"), r.warnings().get(0));
68  	}
69  
70  	@Test void b03_oversizeSkipped() {
71  		var r = new JestJsonReportReader().read(resource("jest-mixed.json"), new ReportLimits(10, 100, 512, 2000, 8000));
72  		assertEquals(1, r.skippedFiles());
73  		assertTrue(r.tests().isEmpty());
74  	}
75  
76  	@Test void b04_twelveThousandAssertions(@TempDir Path d) throws Exception {
77  		var sb = new StringBuilder("{\"testResults\":[{\"name\":\"/w/a.test.js\",\"status\":\"failed\",\"assertionResults\":[");
78  		for (var i = 0; i < 12_000; i++) {
79  			if (i > 0)
80  				sb.append(',');
81  			sb.append("{\"title\":\"t").append(i).append("\",\"status\":\"").append(i % 3 == 0 ? "failed" : i % 3 == 1 ? "passed" : "pending").append("\",\"duration\":1,\"failureMessages\":[]}");
82  		}
83  		var f = d.resolve("big.json");
84  		Files.writeString(f, sb.append("]}]}").toString());
85  		var r = new JestJsonReportReader().read(f);
86  		assertEquals(12_000, r.tests().size());
87  		assertEquals(4000, statuses(r).stream().filter("fail"::equals).count());
88  	}
89  
90  	@Test void b05_notAnObject(@TempDir Path d) throws Exception {
91  		var f = d.resolve("x.json");
92  		for (var text : new String[] {"[]", "", "42"}) {
93  			Files.writeString(f, text);
94  			var r = new JestJsonReportReader().read(f);
95  			assertTrue(r.tests().isEmpty(), text);
96  			assertTrue(r.warnings().get(0).startsWith("report unreadable"), text + ": " + r.warnings());
97  		}
98  	}
99  
100 	@Test void b06_maxTests() {
101 		var r = new JestJsonReportReader().read(resource("jest-mixed.json"), new ReportLimits(1 << 20, 2, 512, 2000, 8000));
102 		assertSize(2, r.tests());
103 		assertTrue(r.truncated());
104 		assertEquals(List.of("report truncated at 2 tests"), r.warnings());
105 	}
106 }
