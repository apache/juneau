17  package org.apache.juneau.terminal.report;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.nio.file.*;
22  import java.util.*;
23  
24  import org.apache.juneau.*;
25  import org.junit.jupiter.api.*;
26  
27  class ReportText_Test extends TestBase {
28  
29  	@Test void a01_stripAnsi() {
30  		assertEquals("FAIL x", ReportText.stripAnsi("\u001b[31mFAIL\u001b[0m x"));
31  		assertEquals("", ReportText.stripAnsi(null));
32  	}
33  
34  	@Test void a02_clipAddsEllipsis() {
35  		assertEquals("abcd…", ReportText.clip("abcdefgh", 5));
36  		assertEquals("abc", ReportText.clip("abc", 5));
37  		assertEquals("", ReportText.clip(null, 5));
38  	}
39  
40  	@Test void a04_commonDirStrip() {
41  		assertEquals(List.of("a.test.js", "b/c.test.js"), ReportText.stripCommonDir(List.of("/ci/w/src/a.test.js", "/ci/w/src/b/c.test.js")));
42  		assertEquals(List.of("a.test.js", "b.test.js"), ReportText.stripCommonDir(List.of("/ci/work/repo/src/a.test.js", "/ci/work/repo/src/b.test.js")));
43  	}
44  
45  	@Test void a05_singlePathKeepsLastTwoSegments() {
46  		assertEquals(List.of("src/a.test.js"), ReportText.stripCommonDir(List.of("/ci/w/src/a.test.js")));
47  		assertEquals(List.of("a.js"), ReportText.stripCommonDir(List.of("a.js")));
48  	}
49  
50  	@Test void a06_fileNameNeverAbsolute() {
51  		assertEquals("TEST-a.xml", ReportText.fileName(Path.of("/tmp/x/TEST-a.xml")));
52  		assertEquals("", ReportText.fileName(null));
53  	}
54  
55  	@Test void a07_backslashesAndDivergentRoots() {
56  		assertEquals(List.of("src/a.js", "lib/b.js"), ReportText.stripCommonDir(List.of("C:\\w\\src\\a.js", "C:\\w\\lib\\b.js")));
57  	}
58  
59  	@Test void a08_noCommonDirAndEmpty() {
60  		assertEquals(List.of("a.js", "b.js"), ReportText.stripCommonDir(List.of("a.js", "b.js")));
61  		assertEquals(List.of(), ReportText.stripCommonDir(List.of()));
62  	}
63  }
