29  /**
30   * Tests for config <b>profiles</b> &mdash; {@code <name>-<profile>} overlays activated via
31   * {@code Config.Builder.profiles(...)} (the standalone path; the Spring piggyback is covered in the springboot module).
32   */
33  @SuppressWarnings({
34  	"resource" // MemoryStore/Config are test fixtures; lifecycle managed by the test, not a real leak.
35  })
36  class ConfigProfiles_Test extends TestBase {
37  
38  	private static MemoryStore store(String... namesAndContents) {
39  		var s = MemoryStore.create().build();
40  		for (var i = 0; i < namesAndContents.length; i += 2)
41  			s.update(namesAndContents[i], namesAndContents[i + 1]);
42  		return s;
43  	}
44  
45  	private static Config config(MemoryStore s, String name, String... profiles) {
46  		try {
47  			return Config.create().store(s).name(name).profiles(profiles).build();
48  		} catch (Exception e) {
49  			throw new RuntimeException(e);
50  		}
51  	}
52  
53  	// =================================================================================
54  	// A. Basic overlay — profile wins over base; base-only keys survive.
55  	// =================================================================================
56  
57  	@Test void a01_profileOverridesBase_baseOnlySurvives() {
58  		var s = store(
59  			"App.cfg", "[S]\na = base-a\nb = base-b\n",
60  			"App-stage.cfg", "[S]\na = stage-a\n");
61  		var c = config(s, "App.cfg", "stage");
62  		assertEquals("stage-a", c.get("S/a").orElse(null));  // profile wins
63  		assertEquals("base-b", c.get("S/b").orElse(null));   // base-only survives
64  	}
65  
66  	@Test void a02_noProfilesIsBaseOnly() {
67  		var s = store("App.cfg", "[S]\na = base-a\n", "App-stage.cfg", "[S]\na = stage-a\n");
68  		var c = config(s, "App.cfg");
69  		assertEquals("base-a", c.get("S/a").orElse(null));
70  	}
71  
72  	@Test void a03_newSectionFromProfile() {
73  		var s = store(
74  			"App.cfg", "[S]\na = base-a\n",
75  			"App-stage.cfg", "[T]\nx = stage-x\n");
76  		var c = config(s, "App.cfg", "stage");
77  		assertEquals("base-a", c.get("S/a").orElse(null));
78  		assertEquals("stage-x", c.get("T/x").orElse(null));  // section only in the profile
79  	}
80  
81  	// =================================================================================
82  	// B. Multiple active profiles — last-active-profile wins.
83  	// =================================================================================
84  
85  	@Test void b01_lastActiveProfileWins() {
86  		var s = store(
87  			"App.cfg", "[S]\na = base\n",
88  			"App-one.cfg", "[S]\na = one\n",
89  			"App-two.cfg", "[S]\na = two\n");
90  		var c = config(s, "App.cfg", "one", "two");
91  		assertEquals("two", c.get("S/a").orElse(null));  // two activated last
92  	}
93  
94  	@Test void b02_earlierProfileStillContributesUnsharedKeys() {
95  		var s = store(
96  			"App.cfg", "[S]\na = base\n",
97  			"App-one.cfg", "[S]\nb = one-b\n",
98  			"App-two.cfg", "[S]\na = two-a\n");
99  		var c = config(s, "App.cfg", "one", "two");
100 		assertEquals("two-a", c.get("S/a").orElse(null));   // two wins for a
101 		assertEquals("one-b", c.get("S/b").orElse(null));   // one's unique key survives
102 	}
103 
104 	// =================================================================================
105 	// C. Missing profile file is a no-op overlay (base unchanged).
106 	// =================================================================================
107 
108 	@Test void c01_missingProfileFileNoOp() {
109 		var s = store("App.cfg", "[S]\na = base-a\n");  // no App-stage.cfg seeded
110 		var c = config(s, "App.cfg", "stage");
111 		assertEquals("base-a", c.get("S/a").orElse(null));
112 	}
113 
114 	// =================================================================================
115 	// D. Live reload — a change to a profile overlay file re-fires config-change listeners.
116 	// =================================================================================
117 
118 	@Test void d01_profileFileChangeTriggersReload() throws Exception {
119 		var s = store(
120 			"App.cfg", "[S]\na = base-a\n",
121 			"App-stage.cfg", "[S]\na = stage-a\n");
122 		var c = config(s, "App.cfg", "stage");
123 		assertEquals("stage-a", c.get("S/a").orElse(null));
124 
125 		var latch = new CountDownLatch(1);
126 		var seen = new AtomicReference<String>();
127 		c.addListener((ConfigEventListener) events -> { seen.set("fired"); latch.countDown(); });
128 
129 		// Change the PROFILE file underneath the store.
130 		s.update("App-stage.cfg", "[S]\na = stage-a2\n");
131 
132 		assertTrue(latch.await(5, TimeUnit.SECONDS), "Expected a config-change event from the profile-file update");
133 		assertEquals("stage-a2", c.get("S/a").orElse(null));  // re-merged value visible
134 	}
135 
136 	@Test void d02_baseFileChangeTriggersReload() throws Exception {
137 		var s = store(
138 			"App.cfg", "[S]\na = base-a\nb = base-b\n",
139 			"App-stage.cfg", "[S]\na = stage-a\n");
140 		var c = config(s, "App.cfg", "stage");
141 
142 		var latch = new CountDownLatch(1);
143 		c.addListener((ConfigEventListener) events -> latch.countDown());
144 
145 		s.update("App.cfg", "[S]\na = base-a\nb = base-b2\n");
146 
147 		assertTrue(latch.await(5, TimeUnit.SECONDS), "Expected a config-change event from the base-file update");
148 		assertEquals("stage-a", c.get("S/a").orElse(null));   // profile still wins after base reload
149 		assertEquals("base-b2", c.get("S/b").orElse(null));   // base change visible
150 	}
