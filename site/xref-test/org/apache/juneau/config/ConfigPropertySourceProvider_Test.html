26  /**
27   * Tests for {@link ConfigPropertySourceProvider}.
28   *
29   * <p>
30   * Covers three layers:
31   * <ol>
32   * 	<li>SPI plumbing: the provider returns {@code null} silently when no classpath default exists.
33   * 	<li>SPI plumbing: the provider returns a {@link ConfigPropertySource} when a system default is set.
34   * 	<li>End-to-end: a {@code @Value("${...}")} bean resolves through {@link Settings#addSource(PropertySource)
35   * 		Settings.get().addSource(...)} the same way the microservice / RestContext hooks register their
36   * 		per-resource {@code Config}.
37   * </ol>
38   */
39  @SuppressWarnings({
40  	"java:S2094", // Test fixture / data class, no methods required.
41  	"resource"    // Closeable resources in tests are intentionally unassigned; closing is handled by test infrastructure.
42  })

66  	@Test
67  	void a01_provider_silent_whenNoSystemDefault() {
68  		// Provider always returns a non-null lazy source — eager resolution would create a
69  		// Settings/Config static-initialization cycle (the SPI fires from Settings.<clinit>).
70  		// "Silent when no config" means: every lookup on the source returns missing(), with no
71  		// log noise and no exception.
72  		var previous = Config.getSystemDefault();
73  		Config.setSystemDefault(null);
74  		try {
75  			var provider = new ConfigPropertySourceProvider();
76  			var src = provider.create();
77  			assertNotNull(src, "Provider returns a lazy wrapper unconditionally");
78  			var result = src.get("ConfigPropertySourceProvider_Test/k");
79  			assertFalse(result.isPresent(),
80  				"Lazy wrapper must report missing() when no system-default Config is set");
81  		} finally {
82  			Config.setSystemDefault(previous);
83  		}
84  	}
85  
86  	@Test
87  	void a02_provider_returnsSource_whenSystemDefaultPresent() {
88  		var previous = Config.getSystemDefault();
89  		var cfg = Config.create().memStore().build();
90  		cfg.set("ConfigPropertySourceProvider_Test/k", "from-system-default");
91  		Config.setSystemDefault(cfg);
92  		try {
93  			var provider = new ConfigPropertySourceProvider();
94  			var src = provider.create();
95  			assertNotNull(src, "Provider must return a source when a system default exists");
96  			var result = src.get("ConfigPropertySourceProvider_Test/k");
97  			assertTrue(result.isPresent(), "Source must report present for set key");
98  			assertEquals("from-system-default", result.value().orElse(null));
99  		} finally {
100 			Config.setSystemDefault(previous);
101 		}
102 	}

104 	@Test
105 	void a03_provider_order_lowSoUserAdditionsWin() {
106 		// The provider deliberately sits at order 100 so later programmatic addSource() calls
107 		// (microservice / RestContext) register AFTER it and therefore win at lookup time.
108 		assertEquals(100, new ConfigPropertySourceProvider().order());
109 	}

115 	@Test
116 	void b01_endToEnd_addSource_shadowsDefault() {
117 		var cfg = Config.create().memStore().build();
118 		cfg.set("ConfigPropertySourceProvider_Test/k", "from-explicit-source");
119 		var src = new ConfigPropertySource(cfg);
120 		Settings.get().addSource(src);
121 		try {
122 			var bean = BeanInstantiator.of(ValueBean.class, beanStore).run();
123 			assertEquals("from-explicit-source", bean.value);
124 		} finally {
125 			assertTrue(Settings.get().removeSource(src),
126 				"removeSource must report success when the source is present");
127 		}
128 	}

130 	@Test
131 	void b02_endToEnd_missing_usesDefault() {
132 		// No source pushed — verify the @Value default path fires when nothing wins.
133 		var bean = BeanInstantiator.of(ValueBeanWithDefault.class, beanStore).run();
134 		assertEquals("not-set", bean.value);
135 	}

137 	@Test
138 	void b03_endToEnd_microserviceOverride_winsOverFirstAdd() {
139 		// Simulates the microservice path: a "base" source is registered first, then a "micro"
140 		// source is registered later. The later source wins because Settings walks in reverse.
141 		var base = Config.create().memStore().build();
142 		base.set("ConfigPropertySourceProvider_Test/k", "base-val");
143 		var micro = Config.create().memStore().build();
144 		micro.set("ConfigPropertySourceProvider_Test/k", "micro-val");
145 		var baseSrc = new ConfigPropertySource(base);
146 		var microSrc = new ConfigPropertySource(micro);
147 		Settings.get().addSource(baseSrc);
148 		Settings.get().addSource(microSrc);
149 		try {
150 			var bean = BeanInstantiator.of(ValueBean.class, beanStore).run();
151 			assertEquals("micro-val", bean.value, "Later addSource() must shadow earlier addSource()");
152 		} finally {
153 			Settings.get().removeSource(microSrc);
154 			Settings.get().removeSource(baseSrc);
155 		}
156 	}
