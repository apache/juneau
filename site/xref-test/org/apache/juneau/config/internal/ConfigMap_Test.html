56  	/**
57  	 * A store that delivers change events asynchronously, like a file store with a watcher: each write is preceded
58  	 * by a late event carrying the previous (stale) contents, and the event for the write itself arrives later.
59  	 */
60  	private static class AsyncEventStore extends MemoryStore {
61  		private final String stale;
62  		final AtomicBoolean realEventDelivered = new AtomicBoolean();
63  
64  		AsyncEventStore(String stale) {
65  			super(MemoryStore.create());
66  			this.stale = stale;
67  		}
68  
69  		@Override
70  		public synchronized String write(String name, String expectedContents, String newContents) {
71  			update(name, stale);
72  			CompletableFuture.runAsync(() -> {
73  				realEventDelivered.set(true);
74  				update(name, newContents);
75  			}, CompletableFuture.delayedExecutor(300, TimeUnit.MILLISECONDS));
76  			return null;
77  		}
78  	}
79  
80  	@Test void a04_loadSynchronous_ignoresStaleEvent() throws Exception {
81  		var s = new AsyncEventStore("[S1]\nk1 = stale\n");
82  		var cm = s.getMap("A04.cfg");
83  		cm.load("[S1]\nk1 = v1\n", true);
84  		assertTrue(s.realEventDelivered.get(), "load returned before the event for its own write");
85  		assertEquals("v1", cm.getEntry("S1", "k1").getValue());
86  	}
87  
88  	@Test void a05_loadSynchronous_overProfileStore() throws Exception {
89  		// The profile store notifies listeners with the merged base + overlay text, not the text that was written.
90  		var d = MemoryStore.create().build();
91  		d.update("A05.cfg", "[S1]\na = base\n");
92  		d.update("A05-stage.cfg", "[S1]\na = stage\n");
93  		var s = ProfileConfigStore.create().delegate(d).baseName("A05.cfg").profiles(List.of("stage")).format(IniConfigFormat.INSTANCE).build();
94  		var cm = s.getMap("A05.cfg");
95  		cm.load("[S1]\na = base\nb = 2\n", true);
96  		assertEquals("stage", cm.getEntry("S1", "a").getValue());
97  		assertEquals("2", cm.getEntry("S1", "b").getValue());
98  	}
