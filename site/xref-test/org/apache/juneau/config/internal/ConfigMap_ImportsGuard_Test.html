20  import static org.junit.jupiter.api.Assumptions.*;

28  /**
29   * Pins the empty-imports guard in {@link ConfigMap#getEntry(String, String)}.
30   */
31  @SuppressWarnings({
32  	"resource" // Closeable resources in tests are intentionally unassigned; closing is handled by test infrastructure.
33  })
34  class ConfigMap_ImportsGuard_Test extends TestBase {
35  
36  	@Test void g01a_noImports_missLookup_allocatesUnder32BytesPerOp() throws Exception {
37  		var mx = (com.sun.management.ThreadMXBean)ManagementFactory.getThreadMXBean();
38  		assumeTrue(mx.isThreadAllocatedMemorySupported());
39  		mx.setThreadAllocatedMemoryEnabled(true);
40  		var cm = new ConfigMap(MemoryStore.create().build(), "G01.cfg", "[S1]\nk1 = v1\n");
41  		for (var i = 0; i < 30_000; i++)
42  			cm.getEntry("S1", "absent");  // warmup
43  		var id = Thread.currentThread().getId();
44  		var n = 10_000;
45  		var hits = 0;
46  		var t0 = mx.getThreadAllocatedBytes(id);
47  		for (var i = 0; i < n; i++)
48  			if (cm.getEntry("S1", "absent") != null)  // no assertion inside the timed loop: assertion helpers allocate
49  				hits++;
50  		var perOp = (mx.getThreadAllocatedBytes(id) - t0) / n;
51  		assertEquals(0, hits, "every lookup is a miss");
52  		assertTrue(perOp < 32, "allocated " + perOp + " B per miss; the empty-imports stream must be skipped");
53  	}
54  
55  	@Test void g01b_import_backedLookup_stillResolves() throws Exception {
56  		var ms = MemoryStore.create().build();
57  		ms.write("G01A", "", "[S1]\nk = imported\n");
58  		ms.write("G01B", "", "<G01A>");
59  		var cm = new ConfigMap(ms, "G01B");
60  		assertEquals("imported", cm.getEntry("S1", "k").getValue());
61  	}
62  }
