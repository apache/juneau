31  /**
32   * Cold/warm boot-time benchmark for the Juneau {@link Microservice} + {@link JettyConfiguration} stack.
33   *
34   * <p>
35   * Not a real {@code _Test} — the class name does not match the surefire {@code **&#47;*Test.class} include pattern, so
36   * it will not run as part of a normal {@code mvn test} build.  Invoke explicitly:
37   *
38   * <pre>
39   *   mvn -pl juneau-integration-tests test -Dtest=BootTimeBenchmark -Drat.skip=true
40   * </pre>
41   *
42   * <p>
43   * Or compile and run the {@link #main(String[])} method directly.
44   *
45   * <p>
46   * What it measures, for both a 1-servlet and a 10-servlet config:
47   * <ul>
48   *   <li><b>Build phase:</b> {@code Microservice.create()...build()} — config load, bean-store wiring,
49   *       {@code @Configuration} class scanning, {@link Microservice#init()}.</li>
50   *   <li><b>Start phase:</b> {@code microservice.start()} — Jetty server build (or reuse), servlet mount, port
51   *       bind via {@code Server.start()}.</li>
52   * </ul>
53   *
54   * <p>
55   * Each (servlet-count) variant runs 1 cold iteration + 5 warm iterations.  Output goes to stdout with the prefix
56   * {@code BOOTBENCH:} so it can be greped out of surefire console output.
57   */
58  @SuppressWarnings({
59  	"java:S5976", // Manual benchmark harness, not a real JUnit test — public visibility is intentional.
60  	"java:S2699", // Manual benchmark harness — it measures boot timings and has no assertions by design.
61  	"java:S3577"  // Manual benchmark harness — name intentionally excluded from the surefire *Test include pattern.
62  })

130 	private static void runAndReport() throws Exception {
131 		var hdr = "===== BootTimeBenchmark — Microservice + JettyConfiguration cold/warm boot =====";
132 		println(hdr);
133 		println("JDK: " + System.getProperty("java.runtime.version") + "  on " + System.getProperty("os.name"));
134 		println("Iterations per variant: 1 cold + " + WARM_ITERS + " warm");
135 		println("");
136 
137 		var oneServlet = bench("1 servlet", OneServletConfig.class);
138 		var tenServlets = bench("10 servlets", TenServletsConfig.class);
139 
140 		println("");
141 		println("===== Summary =====");
142 		summarize("1 servlet", oneServlet);
143 		summarize("10 servlets", tenServlets);
144 		println("");
145 
146 		// Linearity check — how much does the 10-servlet warm median exceed 1-servlet warm median?
147 		var w1 = warmMedian(oneServlet);
148 		var w10 = warmMedian(tenServlets);
149 		println(String.format("BOOTBENCH: scaling: warm median 1-servlet=%.1fms, 10-servlets=%.1fms, delta=%.1fms (~%.2f ms/servlet)",
150 			w1, w10, w10 - w1, (w10 - w1) / 9.0));
151 	}

153 	private static Sample[] bench(String label, Class<?> userConfig) throws Exception {
154 		println("BOOTBENCH: --- variant: " + label + " ---");
155 		var samples = new Sample[TOTAL_ITERS];
156 		for (var i = 0; i < TOTAL_ITERS; i++) {
157 			var s = oneBoot(userConfig);
158 			samples[i] = s;
159 			var tag = (i == 0) ? "COLD" : "warm#" + i;
160 			println(String.format("BOOTBENCH: [%s] %s  build=%.1fms  start=%.1fms  total=%.1fms",
161 				label, tag, s.buildMs(), s.startMs(), s.totalMs()));
162 		}
163 		return samples;
164 	}
165 
166 	private static Sample oneBoot(Class<?> userConfig) throws Exception {
167 		var classes = new ArrayList<Class<?>>();
168 		classes.add(userConfig);
169 		classes.add(MicroserviceTestFixture.EphemeralJettyServerConfig.class);
170 		classes.add(JettyConfiguration.class);
171 
172 		var t0 = System.nanoTime();
173 		var ms = Microservice.create()
174 			.configurations(classes.toArray(new Class<?>[0]))
175 			.build();
176 		var t1 = System.nanoTime();
177 		ms.start();
178 		var t2 = System.nanoTime();
179 		try {
180 			return new Sample(nsToMs(t1 - t0), nsToMs(t2 - t1));
181 		} finally {
182 			// Always stop so the bound port is released for the next iteration.
183 			ms.stop();
184 		}
185 	}

191 	private static double nsToMs(long ns) { return ns / 1_000_000.0; }
192 
193 	private static void summarize(String label, Sample[] s) {
194 		var cold = s[0];
195 		var warmBuild = warmMedian(s, true /*build*/);
196 		var warmStart = warmMedian(s, false /*start*/);
197 		var warmTotal = warmMedian(s);
198 		println(String.format("BOOTBENCH: %s : cold total=%.1fms (build=%.1fms, start=%.1fms) | warm-median total=%.1fms (build=%.1fms, start=%.1fms)",
199 			label, cold.totalMs(), cold.buildMs(), cold.startMs(),
200 			warmTotal, warmBuild, warmStart));
201 	}
202 
203 	private static double warmMedian(Sample[] s) {
204 		return median(warmValues(s, Sample::totalMs));
205 	}
206 
207 	private static double warmMedian(Sample[] s, boolean build) {
208 		return median(warmValues(s, build ? Sample::buildMs : Sample::startMs));
209 	}
210 
211 	private static double[] warmValues(Sample[] s, java.util.function.ToDoubleFunction<Sample> fn) {
212 		// Skip index 0 (cold).
213 		var out = new double[s.length - 1];
214 		for (var i = 1; i < s.length; i++) out[i - 1] = fn.applyAsDouble(s[i]);
215 		return out;
216 	}
217 
218 	private static double median(double[] xs) {
219 		var copy = cp(xs);
220 		Arrays.sort(copy);
221 		var n = copy.length;
222 		return (n % 2 == 1) ? copy[n / 2] : (copy[n / 2 - 1] + copy[n / 2]) / 2.0;
223 	}
