341 	@Test void d03_startConsole_thenStopConsole_whenEnabled() throws Exception {
342 		// We need an interactive-looking input that won't EOF immediately.
343 		// Use a piped scanner so the console thread doesn't blow through nextLine() and exit.
344 		var pin = new PipedInputStream();
345 		var pout = new PipedOutputStream(pin);
346 		var sc = new Scanner(pin);
347 		var ms = Microservice.create()
348 			.consoleEnabled(true)
349 			.console(sc, new PrintWriter(new StringWriter()))
350 			.consoleCommands(new HelpCommand())
351 			.build();
352 		try {
353 			assertSame(ms, ms.startConsole());
354 			// Deterministically wait for the console thread to become alive (named "ConsoleThread" in Microservice)
355 			// instead of sleeping a fixed interval, so the stopConsole() interrupt-alive-thread branch is exercised.
356 			var deadline = System.nanoTime() + TimeUnit.SECONDS.toNanos(5);
357 			while (! consoleThreadAlive() && System.nanoTime() < deadline)
358 				Thread.onSpinWait();
359 			assertTrue(consoleThreadAlive(), "Console thread did not start within timeout.");
360 			assertSame(ms, ms.stopConsole());
361 			// Closing the pipe lets the scanner exit cleanly.
362 			pout.close();
363 			pin.close();
364 		} finally {
365 			ms.stop();
366 		}
367 	}

369 	/** Returns <jk>true</jk> if a live thread named "ConsoleThread" (the Microservice console thread) currently exists. */
370 	private static boolean consoleThreadAlive() {
371 		return Thread.getAllStackTraces().keySet().stream().anyMatch(t -> t.isAlive() && "ConsoleThread".equals(t.getName()));
372 	}

391 	@Test void f01_stop_idempotent() throws Exception {
392 		var stopCount = new AtomicInteger();
393 		var listener = new BasicMicroserviceListener() {
394 			@Override public void onStop(Microservice m) { stopCount.incrementAndGet(); }
395 		};
396 		var ms = Microservice.create().listener(listener).build();
397 		ms.stop();
398 		ms.stop();   // second call is a no-op
399 		ms.stop();   // third call is a no-op
400 		assertEquals(1, stopCount.get(), "Listener.onStop must fire exactly once even for repeated stop() calls");
401 	}
