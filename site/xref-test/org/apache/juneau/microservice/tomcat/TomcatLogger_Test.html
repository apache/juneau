19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.logging.*;
22  
23  import org.apache.juneau.*;
24  import org.junit.jupiter.api.*;
25  
26  /**
27   * Tests for {@link TomcatLogger}.
28   */
29  class TomcatLogger_Test extends TestBase {
30  
31  	@Test void a01_defaultConstructor() {
32  		var l = new TomcatLogger();
33  		assertEquals("org.apache.juneau.marshall.microservice.tomcat", l.getName());
34  	}
35  
36  	@Test void a02_namedConstructor() {
37  		var l = new TomcatLogger("test.tomcat.logger.name");
38  		assertEquals("test.tomcat.logger.name", l.getName());
39  	}
40  
41  	//-----------------------------------------------------------------------------------------------------------------
42  	// b — isXEnabled level checks
43  	//-----------------------------------------------------------------------------------------------------------------
44  
45  	@Test void b01_levelChecks_atAll() {
46  		var l = new TomcatLogger("test.tomcat.b01");
47  		Logger.getLogger("test.tomcat.b01").setLevel(Level.ALL);
48  		assertTrue(l.isTraceEnabled());
49  		assertTrue(l.isDebugEnabled());
50  		assertTrue(l.isInfoEnabled());
51  		assertTrue(l.isWarnEnabled());
52  		assertTrue(l.isErrorEnabled());
53  		assertTrue(l.isFatalEnabled());
54  	}
55  
56  	@Test void b02_levelChecks_atOff() {
57  		var l = new TomcatLogger("test.tomcat.b02");
58  		Logger.getLogger("test.tomcat.b02").setLevel(Level.OFF);
59  		assertFalse(l.isTraceEnabled());
60  		assertFalse(l.isDebugEnabled());
61  		assertFalse(l.isInfoEnabled());
62  		assertFalse(l.isWarnEnabled());
63  		assertFalse(l.isErrorEnabled());
64  		assertFalse(l.isFatalEnabled());
65  	}
66  
67  	@Test void b03_levelChecks_atInfo() {
68  		var l = new TomcatLogger("test.tomcat.b03");
69  		Logger.getLogger("test.tomcat.b03").setLevel(Level.INFO);
70  		assertFalse(l.isTraceEnabled());
71  		assertFalse(l.isDebugEnabled());
72  		assertTrue(l.isInfoEnabled());
73  		assertTrue(l.isWarnEnabled());
74  		assertTrue(l.isErrorEnabled());
75  		assertTrue(l.isFatalEnabled());
76  	}
77  
78  	//-----------------------------------------------------------------------------------------------------------------
79  	// c — log methods are loggable (true branch of the level guard)
80  	//-----------------------------------------------------------------------------------------------------------------
81  
82  	@Test void c01_loggable_message() {
83  		var l = new TomcatLogger("test.tomcat.c01");
84  		Logger.getLogger("test.tomcat.c01").setLevel(Level.ALL);
85  		assertDoesNotThrow(() -> l.trace("trace msg"));
86  		assertDoesNotThrow(() -> l.debug("debug msg"));
87  		assertDoesNotThrow(() -> l.info("info msg"));
88  		assertDoesNotThrow(() -> l.warn("warn msg"));
89  		assertDoesNotThrow(() -> l.error("error msg"));
90  		assertDoesNotThrow(() -> l.fatal("fatal msg"));
91  	}
92  
93  	@Test void c02_loggable_messageAndThrowable() {
94  		var l = new TomcatLogger("test.tomcat.c02");
95  		Logger.getLogger("test.tomcat.c02").setLevel(Level.ALL);
96  		var t = new RuntimeException("boom");
97  		assertDoesNotThrow(() -> l.trace("trace msg", t));
98  		assertDoesNotThrow(() -> l.debug("debug msg", t));
99  		assertDoesNotThrow(() -> l.info("info msg", t));
100 		assertDoesNotThrow(() -> l.warn("warn msg", t));
101 		assertDoesNotThrow(() -> l.error("error msg", t));
102 		assertDoesNotThrow(() -> l.fatal("fatal msg", t));
103 	}
104 
105 	@Test void c03_loggable_nullMessage() {
106 		var l = new TomcatLogger("test.tomcat.c03");
107 		Logger.getLogger("test.tomcat.c03").setLevel(Level.ALL);
108 		assertDoesNotThrow(() -> l.info(null));
109 		assertDoesNotThrow(() -> l.error(null, new RuntimeException("x")));
110 	}
111
