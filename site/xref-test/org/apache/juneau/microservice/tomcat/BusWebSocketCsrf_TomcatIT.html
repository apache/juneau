47  /**
48   * The bus's WebSocket handshake under a {@link CsrfBoundary}, over a real Tomcat 11 socket; the counterpart of the
49   * Jetty IT of the same name.
50   *
51   * <p>
52   * Three Tomcat instances share one bus and boundary: a plain one; one with a {@link RemoteIpValve} (protocol header
53   * pinned to {@code X-Forwarded-Proto}), the container's own forwarded-header support; and one whose valve also reads
54   * {@code X-Forwarded-Host}.  A probe endpoint records what {@link HandshakeRequest#getRequestURI()} and {@code Host}
55   * report on each.  The handshake does not consult {@link CsrfBoundary.OriginSource}: its scheme comes from the
56   * (valve-rewritten) handshake URI and its host and port from the {@code Host} header, so the boundary here uses the
57   * default source.  The valved rows model a trusted proxy (loopback is in the valve's {@code internalProxies}), not an
58   * untrusted client; a client that can reach Tomcat directly and is trusted by the valve could forge the forwarded
59   * headers.
60   *
61   * <h5 class='section'>What Tomcat reports (Tomcat 11.0.25), unlike Jetty:</h5>
62   * <p>
63   * Tomcat reports the WebSocket scheme, {@code ws} or {@code wss}, where Jetty reports {@code http} or {@code https}.
64   * With the client's Origin and Host as shown, the request URI is:
65   * <table class='styled'>
66   * 	<tr><th>Case</th><th>Request URI</th><th>Headers visible</th></tr>
67   * 	<tr><td>(a) no valve, plain</td><td>{@code ws://127.0.0.1:<port>/probe}</td><td>{@code Host} {@code 127.0.0.1:<port>}</td></tr>
68   * 	<tr><td>(b) {@code RemoteIpValve}, {@code X-Forwarded-Proto: https}</td><td>{@code wss://127.0.0.1/probe} (scheme
69   * 		rewritten, port dropped)</td><td>{@code Host} {@code 127.0.0.1:<port>}, {@code X-Forwarded-Proto}</td></tr>
70   * 	<tr><td>(c) {@code RemoteIpValve}, no forwarded header</td><td>{@code ws://127.0.0.1:<port>/probe}</td><td>{@code Host}</td></tr>
71   * 	<tr><td>(d) no valve, {@code X-Forwarded-Proto: https}</td><td>{@code ws://127.0.0.1:<port>/probe} (header ignored)</td>
72   * 		<td>{@code Host}, {@code X-Forwarded-Proto}</td></tr>
73   * 	<tr><td>(e) {@code RemoteIpValve} with {@code hostHeader=X-Forwarded-Host}, {@code X-Forwarded-Host: evil.example}</td>
74   * 		<td>{@code ws://evil.example:<port>/probe} (URI host rewritten, scheme and port kept)</td><td>{@code Host} still {@code 127.0.0.1:<port>}</td></tr>
75   * </table>
76   *
77   * <p>
78   * {@link CsrfBoundary#checkHandshake} normalizes {@code ws} to {@code http} and {@code wss} to {@code https}, so the
79   * fail-closed path for an unknown scheme is never reached on Tomcat.  The port always comes from the {@code Host}
80   * header, never from the URI, which is why the valve dropping it in (b) does not matter: an {@code https} Origin
81   * without a port (implied 443) is refused there.  Only the {@code RemoteIpValve} carries {@code X-Forwarded-Proto} into
82   * the scheme.  The default valve ignores {@code X-Forwarded-Host}, so a forged forwarded host is refused.
83   *
84   * <p>
85   * Binds sockets, so it is off by default.  Run it with {@code -Djuneau.busIt=true}, outside the sandbox.
86   */

131 	static final class ProbeConfigurator extends ServerEndpointConfig.Configurator {
132 		@Override public void modifyHandshake(ServerEndpointConfig sec, HandshakeRequest request, HandshakeResponse response) {
133 			PROBED.add(request.getRequestURI() + " Host=" + request.getHeaders().get("Host"));
134 			var lines = new ArrayList<String>();
135 			request.getHeaders().forEach((k, v) -> {
136 				if ("X-Forwarded-Host".equalsIgnoreCase(k))
137 					lines.addAll(v);
138 			});
139 			FORWARDED_HOSTS.add(lines);
140 		}
141 	}

148 	private static Tomcat boot(boolean valve, String hostHeader, HttpServlet rest, ServerBus serverBus) throws Exception {
149 		var tomcat = new Tomcat();
150 		var base = Files.createTempDirectory("tc-bus-it");
151 		TEMP.add(base);
152 		tomcat.setBaseDir(base.toString());
153 		var c = tomcat.getConnector();
154 		c.setPort(0);
155 		c.setProperty("address", "127.0.0.1");
156 		c.addUpgradeProtocol(new org.apache.coyote.http2.Http2Protocol());
157 		tomcat.getHost().setAppBase(base.toString());
158 		if (valve) {
159 			var v = new RemoteIpValve();
160 			v.setProtocolHeader("X-Forwarded-Proto");
161 			v.setProtocolHeaderHttpsValue("https");
162 			if (hostHeader != null)
163 				v.setHostHeader(hostHeader);
164 			tomcat.getEngine().getPipeline().addValve(v);
165 		}
166 		Context ctx = tomcat.addContext("", base.toString());
167 		// The WebSocket filter only runs for a request that maps to a servlet.
168 		var dflt = Tomcat.addServlet(ctx, "default", new HttpServlet() {
169 			private static final long serialVersionUID = 1L;
170 		});
171 		dflt.addMapping("/");
172 		var wrapper = Tomcat.addServlet(ctx, "bus", rest);
173 		wrapper.setLoadOnStartup(1);
174 		wrapper.addMapping("/rest/*");
175 		ctx.addApplicationListener(org.apache.tomcat.websocket.server.WsContextListener.class.getName());
176 		tomcat.start();
177 		var sc = (ServerContainer) ctx.getServletContext().getAttribute(ServerContainer.class.getName());
178 		BusWebSockets.register(sc, serverBus);
179 		sc.addEndpoint(ServerEndpointConfig.Builder.create(ProbeEndpoint.class, "/probe")
180 			.configurator(new ProbeConfigurator())
181 			.build());
182 		return tomcat;
183 	}

236 	private static String token() {
237 		var req = (HttpServletRequest) java.lang.reflect.Proxy.newProxyInstance(HttpServletRequest.class.getClassLoader(), new Class<?>[]{HttpServletRequest.class},
238 			(proxy, method, args) -> method.getReturnType() == boolean.class ? (Object) false : null);
239 		return boundary.tokenFor(req);

242 	/** Opens a WebSocket session through the plain instance and returns its WebSocket path. */
243 	private static String openSession(HttpClient http) throws Exception {
244 		var body = "{\"v\":1,\"bridge\":\"ops\",\"transport\":\"websocket\",\"downstream\":[\"ops.jobs\"],\"upstream\":[]}";
245 		var res = http.send(HttpRequest.newBuilder(URI.create(origin(plainPort) + "/rest" + BusEventsMixin.SESSION_PATH))
246 			.header("Content-Type", "application/json")
247 			.header("Accept", "application/json")
248 			.header("Origin", origin(plainPort))
249 			.header(boundary.csrfHeader(), token())
250 			.POST(HttpRequest.BodyPublishers.ofString(body))
251 			.build(), HttpResponse.BodyHandlers.ofString());
252 		assertEquals(200, res.statusCode(), res::body);
253 		return JsonMap.ofString(res.body()).getMap("paths").getString("websocket");
254 	}

260 	/** Collects whole text frames and the close code. */
261 	static final class Frames implements WebSocket.Listener {
262 		final BlockingQueue<JsonMap> frames = new LinkedBlockingQueue<>();
263 		final CompletableFuture<Integer> closed = new CompletableFuture<>();
264 		private final StringBuilder partial = new StringBuilder();
265 
266 		@Override public CompletionStage<?> onText(WebSocket ws, CharSequence data, boolean last) {
267 			partial.append(data);
268 			if (last) {
269 				try {
270 					frames.add(JsonMap.ofString(partial.toString()));
271 				} catch (Exception e) {
272 					closed.completeExceptionally(e);
273 				}
274 				partial.setLength(0);
275 			}
276 			ws.request(1);
277 			return null;
278 		}
279 
280 		@Override public CompletionStage<?> onClose(WebSocket ws, int statusCode, String reason) {
281 			closed.complete(statusCode);
282 			return null;
283 		}
284 	}

286 	/** The HTTP status of a handshake the container refused before the upgrade. */
287 	private static int handshakeStatus(HttpClient http, URI uri, String...headers) {
288 		var b = http.newWebSocketBuilder();
289 		for (var i = 0; i < headers.length; i += 2)
290 			b.header(headers[i], headers[i + 1]);
291 		var e = assertThrows(ExecutionException.class, () -> b.buildAsync(uri, new Frames()).get(10, TimeUnit.SECONDS));
292 		return assertInstanceOf(WebSocketHandshakeException.class, e.getCause()).getResponse().statusCode();
293 	}

295 	/** Connects with the given headers, expects the first frame {@code resync-begin} and an open socket, then closes. */
296 	private static void assertAdmitted(int port, String...headers) throws Exception {
297 		var http = HttpClient.newHttpClient();
298 		var frames = new Frames();
299 		var b = http.newWebSocketBuilder();
300 		for (var i = 0; i < headers.length; i += 2)
301 			b.header(headers[i], headers[i + 1]);
302 		var ws = b.buildAsync(wsUri(port, openSession(http)), frames).get(10, TimeUnit.SECONDS);
303 		var first = frames.frames.poll(10, TimeUnit.SECONDS);
304 		assertNotNull(first, "no frame within 10s; closed with " + frames.closed.getNow(null));
305 		assertEquals("resync-begin", first.getString("type"));
306 		assertNull(frames.closed.getNow(null), "the socket was closed");
307 		ws.sendClose(WebSocket.NORMAL_CLOSURE, "").get(10, TimeUnit.SECONDS);
308 	}

310 	/** Connects with the given headers and expects the upgrade to complete and the endpoint to close at once with {@code 4403}. */
311 	private static void assertRefused4403(int port, String...headers) throws Exception {
312 		var http = HttpClient.newHttpClient();
313 		var frames = new Frames();
314 		var b = http.newWebSocketBuilder();
315 		for (var i = 0; i < headers.length; i += 2)
316 			b.header(headers[i], headers[i + 1]);
317 		b.buildAsync(wsUri(port, openSession(http)), frames).get(10, TimeUnit.SECONDS);
318 		assertEquals(4403, frames.closed.get(10, TimeUnit.SECONDS));
319 		assertTrue(frames.frames.isEmpty(), "a refused handshake saw a frame");
320 	}

347 	/**
348 	 * A {@code Host} that is not the Origin's host.  The JDK client cannot set {@code Host}, so this speaks the handshake
349 	 * over a raw socket: the upgrade completes and the endpoint's first frame is a close with {@code 4403}.
350 	 */
351 	@Test void a04_hostOriginMismatchCloses4403() throws Exception {
352 		var path = openSession(HttpClient.newHttpClient());
353 		try (var socket = new Socket("127.0.0.1", plainPort); var out = socket.getOutputStream(); var in = new BufferedInputStream(socket.getInputStream())) {
354 			socket.setSoTimeout(10_000);
355 			out.write(("GET " + path + " HTTP/1.1\r\n"
356 				+ "Host: evil.example\r\n"
357 				+ "Origin: " + origin(plainPort) + "\r\n"
358 				+ "Connection: Upgrade\r\n"
359 				+ "Upgrade: websocket\r\n"
360 				+ "Sec-WebSocket-Version: 13\r\n"
361 				+ "Sec-WebSocket-Key: dGhlIHNhbXBsZSBub25jZQ==\r\n\r\n").getBytes(StandardCharsets.US_ASCII));
362 			out.flush();
363 			var head = new StringBuilder();
364 			while (! head.toString().endsWith("\r\n\r\n")) {
365 				var b = in.read();
366 				assertTrue(b >= 0, "connection closed during handshake: " + head);
367 				head.append((char) b);
368 			}
369 			assertTrue(head.toString().startsWith("HTTP/1.1 101"), head::toString);
370 			assertEquals(0x88, in.read(), "first frame is not a close frame");
371 			var len = in.read();
372 			assertTrue(len >= 2 && len < 126, "unexpected close payload length " + len);
373 			assertEquals(4403, (in.read() << 8) | in.read());
374 		}
375 	}

415 	private static String probe(int port, String...headers) throws Exception {
416 		var b = HttpClient.newHttpClient().newWebSocketBuilder();
417 		for (var i = 0; i < headers.length; i += 2)
418 			b.header(headers[i], headers[i + 1]);
419 		PROBED.clear();
420 		b.buildAsync(wsUri(port, "/probe"), new Frames()).get(10, TimeUnit.SECONDS)
421 			.sendClose(WebSocket.NORMAL_CLOSURE, "").get(10, TimeUnit.SECONDS);
422 		return PROBED.poll(10, TimeUnit.SECONDS);
423 	}

439 	 * Trusted forwarding picks the rightmost entry, which relies on the handshake's repeated header lines keeping their

442 	@Test void a09_repeatedForwardedHostLinesKeepWireOrder() throws Exception {
443 		FORWARDED_HOSTS.clear();
444 		try (var socket = new Socket("127.0.0.1", plainPort); var out = socket.getOutputStream()) {
445 			socket.setSoTimeout(10_000);
446 			out.write(("GET /probe HTTP/1.1\r\n"

461 	/** Opens a session through the {@code trustedProxies} instance with a forwarded write, as a proxy on the loopback peer would send it. */
462 	private static String openTrustedSession(HttpClient http) throws Exception {
463 		var body = "{\"v\":1,\"bridge\":\"ops\",\"transport\":\"websocket\",\"downstream\":[\"ops.jobs\"],\"upstream\":[]}";
464 		var req = (HttpServletRequest) java.lang.reflect.Proxy.newProxyInstance(HttpServletRequest.class.getClassLoader(), new Class<?>[]{HttpServletRequest.class},
465 			(proxy, method, args) -> method.getReturnType() == boolean.class ? (Object) false : null);
466 		var res = http.send(HttpRequest.newBuilder(URI.create(origin(trustedPort) + "/rest" + BusEventsMixin.SESSION_PATH))
467 			.header("Content-Type", "application/json")
468 			.header("Accept", "application/json")
469 			.header("Origin", origin(trustedPort))
470 			.header("X-Forwarded-Host", "127.0.0.1:" + trustedPort)
471 			.header("X-Forwarded-Proto", "http")
472 			.header(trustedBoundary.csrfHeader(), trustedBoundary.tokenFor(req))
473 			.POST(HttpRequest.BodyPublishers.ofString(body))
474 			.build(), HttpResponse.BodyHandlers.ofString());
475 		assertEquals(200, res.statusCode(), res::body);
476 		return JsonMap.ofString(res.body()).getMap("paths").getString("websocket");
477 	}

508 	/** What an h2c connection did with an RFC 8441 extended CONNECT: the SETTINGS it sent and how the stream ended. */
509 	record Ext(boolean settingsSeen, boolean connectProtocolAdvertised, String frames) {}
510 
511 	private static void field(ByteArrayOutputStream h, String name, String value) throws IOException {
512 		h.write(0);
513 		str(h, name);
514 		str(h, value);
515 	}
516 
517 	private static void str(ByteArrayOutputStream h, String value) throws IOException {
518 		var v = value.getBytes(StandardCharsets.US_ASCII);
519 		h.write(v.length);
520 		h.write(v);
521 	}
522 
523 	private static byte[] frame(int type, int flags, int stream, byte[] payload) {
524 		var b = java.nio.ByteBuffer.allocate(9 + payload.length);
525 		b.put((byte)(payload.length >> 16)).put((byte)(payload.length >> 8)).put((byte)payload.length);
526 		b.put((byte)type).put((byte)flags).putInt(stream).put(payload);
527 		return b.array();
528 	}
529 
530 	/**
531 	 * Sends a hand-encoded extended CONNECT over prior-knowledge h2c and reads until the stream is reset or the
532 	 * connection ends or goes quiet.  The frame types seen are returned for the record.
533 	 */
534 	private static Ext extendedConnect(int port, String path) throws Exception {
535 		var out = new ByteArrayOutputStream();
536 		out.write("PRI * HTTP/2.0\r\n\r\nSM\r\n\r\n".getBytes(StandardCharsets.US_ASCII));
537 		out.write(frame(0x4, 0, 0, new byte[0]));
538 		var h = new ByteArrayOutputStream();
539 		h.write(2);                                    // :method, name from the static table
540 		str(h, "CONNECT");
541 		field(h, ":protocol", "websocket");
542 		h.write(0x86);                                 // :scheme http
543 		h.write(4);                                    // :path
544 		str(h, path);
545 		h.write(1);                                    // :authority
546 		str(h, "127.0.0.1:" + port);
547 		field(h, "origin", origin(port));
548 		field(h, "sec-websocket-version", "13");
549 		out.write(frame(0x1, 0x4, 1, h.toByteArray()));
550 
551 		var seen = new StringBuilder();
552 		var settings = false;
553 		var advertised = false;
554 		try (var socket = new Socket("127.0.0.1", port); var in = new DataInputStream(socket.getInputStream())) {
555 			socket.setSoTimeout(3_000);
556 			socket.getOutputStream().write(out.toByteArray());
557 			socket.getOutputStream().flush();
558 			try {
559 				for (;;) {
560 					var head = new byte[9];
561 					in.readFully(head);
562 					var len = ((head[0] & 0xFF) << 16) | ((head[1] & 0xFF) << 8) | (head[2] & 0xFF);
563 					var payload = new byte[len];
564 					in.readFully(payload);
565 					seen.append(f("%02x ", head[3]));
566 					if (head[3] == 0x4 && (head[4] & 0x1) == 0) {
567 						settings = true;
568 						for (var i = 0; i + 6 <= len; i += 6)
569 							if ((((payload[i] & 0xFF) << 8) | (payload[i + 1] & 0xFF)) == 0x8 && payload[i + 5] == 1)
570 								advertised = true;
571 						socket.getOutputStream().write(frame(0x4, 0x1, 0, new byte[0]));
572 						socket.getOutputStream().flush();
573 					}
574 					if (head[3] == 0x3 || head[3] == 0x7)
575 						break;
576 				}
577 			} catch (EOFException | SocketTimeoutException e) {
578 				seen.append(e instanceof EOFException ? "eof" : "quiet");
579 			}
580 		}
581 		return new Ext(settings, advertised, seen.toString().trim());
582 	}

586 	 * CONNECT never reaches a WebSocket endpoint, so there is no HTTP/2 handshake for the boundary to judge.
