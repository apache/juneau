17  package org.apache.juneau.microservice;
18  
19  import java.net.*;
20  import java.util.*;
21  
22  import org.apache.juneau.commons.inject.*;
23  import org.apache.juneau.microservice.jetty.*;
24  import org.eclipse.jetty.ee11.servlet.*;
25  import org.eclipse.jetty.server.*;
26  import org.junit.jupiter.api.extension.*;
27  
28  /**
29   * JUnit 5 extension that boots a {@link Microservice} backed by {@link JettyConfiguration} on an ephemeral port for
30   * the duration of a test class.
31   *
32   * <p>
33   * Typical usage:
34   *
35   * <p class='bjava'>
36   * 	<ja>@RegisterExtension</ja>
37   * 	<jk>static</jk> MicroserviceTestFixture <jv>fixture</jv> = MicroserviceTestFixture.<jsm>create</jsm>()
38   * 		.configurations(MyServerConfig.<jk>class</jk>);
39   *
40   * 	<ja>@Configuration</ja>
41   * 	<jk>static class</jk> MyServerConfig {{
42   * 		<ja>@Bean</ja> Servlet myService() {{ <jk>return new</jk> MyRestService(); }}
43   * 	}}
44   *
45   * 	<ja>@Test</ja>
46   * 	<jk>void</jk> exampleTest() {{
47   * 		<jk>var</jk> rootUrl = <jv>fixture</jv>.getRootUrl();
48   * 		<jc>// build a client against rootUrl ...</jc>
49   * 	}}
50   * </p>
51   *
52   * <h5 class='section'>How it works:</h5>
53   * <ul>
54   * 	<li>The fixture installs {@link JettyConfiguration} plus a private {@link EphemeralJettyServerConfig}
55   * 		{@code @Configuration} that contributes a Jetty {@link Server} bound to port {@code 0} (OS-assigned
56   * 		ephemeral port). User-supplied configurations should typically contribute {@code @Bean Servlet} definitions
57   * 		which {@link JettyServerComponent} auto-mounts at {@code @Rest(path=...)}.
58   * 	<li>{@link #beforeAll(ExtensionContext) beforeAll} builds and {@link Microservice#start() starts} the service,
59   * 		then probes the bound port via the {@link Server}'s {@link ServerConnector}.
60   * 	<li>{@link #afterAll(ExtensionContext) afterAll} calls {@link Microservice#stop() stop()} so the bound port is
61   * 		released and {@code @PreDestroy} hooks on bean-store beans fire.
62   * </ul>
63   *
64   * <p>
65   * The fixture is intended to amortize Jetty-startup cost across a whole test class — combined with the dynamic
66   * {@link org.apache.juneau.rest.server.RestChildren#addChild RestChildren.addChild} / {@code removeChild} API, individual
67   * test methods can mount and unmount their own child resources against a single long-running server.
68   */
69  @SuppressWarnings({
70  	"resource" // The Microservice (and bean-store Closeables it owns) is stopped in afterAll(); the builder result is not an unmanaged leak.
71  })
72  public final class MicroserviceTestFixture implements BeforeAllCallback, AfterAllCallback {
73  
74  	private final List<Class<?>> configurations = new ArrayList<>();
75  	private Microservice microservice;
76  	private URI rootUrl;
77  
78  	private MicroserviceTestFixture() {}
79  
80  	/**
81  	 * Creates a new fixture.
82  	 *
83  	 * @return A new fixture instance.
84  	 */
85  	public static MicroserviceTestFixture create() {
86  		return new MicroserviceTestFixture();
87  	}
88  
89  	/**
90  	 * Registers one or more {@code @Configuration} classes whose {@code @Bean Servlet} methods will be auto-mounted
91  	 * by the microservice. May be called multiple times to append.
92  	 *
93  	 * @param cs The configuration classes.
94  	 * @return This fixture (fluent).
95  	 */
96  	public MicroserviceTestFixture configurations(Class<?>... cs) {
97  		Collections.addAll(configurations, cs);
98  		return this;
99  	}
100 
101 	/**
102 	 * Returns the root URI of the running microservice (after {@link #beforeAll(ExtensionContext)} has fired).
103 	 *
104 	 * @return The root URI, e.g. {@code http://localhost:54321/}.
105 	 */
106 	public URI getRootUrl() {
107 		return rootUrl;
108 	}
109 
110 	/**
111 	 * Returns the actual port the server is listening on (post-{@code start()}).
112 	 *
113 	 * @return The bound port.
114 	 */
115 	public int getPort() {
116 		return rootUrl.getPort();
117 	}
118 
119 	/**
120 	 * Returns the underlying microservice instance, useful for fixture-scoped child-resource mutations.
121 	 *
122 	 * @return The microservice.
123 	 */
124 	public Microservice getMicroservice() {
125 		return microservice;
126 	}
127 
128 	/**
129 	 * Returns the {@link JettyServerComponent} bean for the running microservice.
130 	 *
131 	 * @return The Jetty server component.
132 	 */
133 	public JettyServerComponent getJettyServerComponent() {
134 		return microservice.getBeanStore().getBean(JettyServerComponent.class).orElseThrow();
135 	}
136 
137 	@Override
138 	public void beforeAll(ExtensionContext ctx) throws Exception {
139 		// The user's configurations come first (so @Bean Servlet methods are visible); EphemeralJettyServerConfig
140 		// supplies the Jetty Server bound to port 0; JettyConfiguration wires the lifecycle and remaining defaults.
141 		// JettyServerComponent resolves Server from the bean store by type, and BeanStore.getBean(...) returns the
142 		// first registered match (so a user-supplied @Bean Server would still win).
143 		var classes = new ArrayList<>(configurations);
144 		classes.add(EphemeralJettyServerConfig.class);
145 		classes.add(JettyConfiguration.class);
146 		microservice = Microservice.create()
147 			.configurations(classes.toArray(new Class<?>[0]))
148 			.build();
149 		microservice.start();
150 		// JettyServerComponent.getURI() uses InetAddress.getLocalHost().getHostName() (may resolve to a non-loopback
151 		// IP on some machines) and ServerConnector.getPort() (returns the configured port = 0, not the bound port).
152 		// For test fixtures we want a deterministic loopback URL with the actual bound port.
153 		var server = getJettyServerComponent().getServer();
154 		var localPort = -1;
155 		for (var c : server.getConnectors()) {
156 			if (c instanceof ServerConnector sc) {
157 				localPort = sc.getLocalPort();
158 				break;
159 			}
160 		}
161 		if (localPort <= 0)
162 			throw new IllegalStateException("Could not determine local port of ServerConnector after start.");
163 		rootUrl = URI.create("http://localhost:" + localPort);
164 	}
165 
166 	@Override
167 	public void afterAll(ExtensionContext ctx) throws Exception {
168 		if (microservice != null)
169 			microservice.stop();
170 	}
171 
172 	/**
173 	 * Default {@code @Configuration} that supplies a Jetty {@link Server} bound to port {@code 0}.
174 	 *
175 	 * <p>
176 	 * The server has a single {@link ServerConnector} on port 0 (OS-assigned) and a single root
177 	 * {@link ServletContextHandler} at context path {@code "/"}. {@link JettyServerComponent} discovers the handler
178 	 * via the {@code "ServletContextHandler"} server attribute (the same convention used by {@code jetty.xml}).
179 	 */
180 	@Configuration
181 	public static class EphemeralJettyServerConfig {
182 
183 		/**
184 		 * Provides the bean-supplied Jetty {@link Server} that {@link JettyServerComponent} consumes during
185 		 * {@code onStart()}.
186 		 *
187 		 * @return A configured {@link Server} bound to port 0.
188 		 */
189 		@Bean
190 		public Server jettyServer() {
191 			var server = new Server();
192 			var connector = new ServerConnector(server);
193 			connector.setPort(0);
194 			server.addConnector(connector);
195 			var sch = new ServletContextHandler();
196 			sch.setContextPath("/");
197 			server.setAttribute("ServletContextHandler", sch);
198 			server.setHandler(sch);
199 			return server;
200 		}
201 	}
