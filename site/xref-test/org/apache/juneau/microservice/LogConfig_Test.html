17  package org.apache.juneau.microservice;
18  
19  import static org.apache.juneau.test.bct.BctAssertions.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.util.*;
23  import java.util.logging.*;
24  import java.util.logging.Formatter;
25  
26  import org.apache.juneau.*;
27  import org.apache.juneau.microservice.logging.*;
28  import org.junit.jupiter.api.*;
29  
30  /**
31   * Tests for {@link LogConfig}.
32   */
33  class LogConfig_Test extends TestBase {
34  
35  	@Test void a01_create() {
36  		var x = LogConfig.create();
37  		assertNotNull(x);
38  	}
39  
40  	@Test void a02_append() {
41  		var x = LogConfig.create().append();
42  		assertSame(Boolean.TRUE, x.append);
43  	}
44  
45  	@Test void a03_consoleLevel() {
46  		assertBean(LogConfig.create().consoleLevel(Level.WARNING), "consoleLevel", "WARNING");
47  	}
48  
49  	@Test void a04_count() {
50  		assertBean(LogConfig.create().count(5), "count", "5");
51  	}
52  
53  	@Test void a05_fileLevel() {
54  		assertBean(LogConfig.create().fileLevel(Level.SEVERE), "fileLevel", "SEVERE");
55  	}
56  
57  	@Test void a06_formatter() {
58  		Formatter f = new LogEntryFormatter("[{date} {level}] {msg}%n", "yyyy.MM.dd hh:mm:ss", false);
59  		var x = LogConfig.create().formatter(f);
60  		assertSame(f, x.formatter);
61  	}
62  
63  	@Test void a07_level() {
64  		var x = LogConfig.create().level("org.apache.juneau.marshall", Level.FINE);
65  		assertEquals(Level.FINE, x.levels.get("org.apache.juneau.marshall"));
66  	}
67  
68  	@Test void a08_levels_map() {
69  		var map = Map.of("foo.bar", Level.INFO, "baz.qux", Level.FINE);
70  		var x = LogConfig.create().levels(map);
71  		assertEquals(Level.INFO, x.levels.get("foo.bar"));
72  		assertEquals(Level.FINE, x.levels.get("baz.qux"));
73  	}
74  
75  	@Test void a09_limit() {
76  		assertBean(LogConfig.create().limit(1024 * 1024), "limit", "1048576");
77  	}
78  
79  	@Test void a10_logDir() {
80  		assertBean(LogConfig.create().logDir("/var/log/myservice"), "logDir", "/var/log/myservice");
81  	}
82  
83  	@Test void a11_logFile() {
84  		assertBean(LogConfig.create().logFile("myservice.log"), "logFile", "myservice.log");
85  	}
86  
87  	@Test void a12_copy() {
88  		var x = LogConfig.create()
89  			.append()
90  			.consoleLevel(Level.INFO)
91  			.count(3)
92  			.fileLevel(Level.WARNING)
93  			.level("test.logger", Level.FINE)
94  			.limit(500000)
95  			.logDir("/logs")
96  			.logFile("service.log");
97  		var copy = x.copy();
98  		assertNotSame(x, copy);
99  		assertSame(Boolean.TRUE, copy.append);
100 		assertEquals(Level.FINE, copy.levels.get("test.logger"));
101 		assertBean(copy, "consoleLevel,count,fileLevel,limit,logDir,logFile", "INFO,3,WARNING,500000,/logs,service.log");
102 	}
103 
104 	@Test void a13_fluentChaining() {
105 		var x = LogConfig.create()
106 			.logFile("app.log")
107 			.logDir("/tmp")
108 			.limit(10000)
109 			.count(2)
110 			.consoleLevel(Level.INFO)
111 			.fileLevel(Level.FINE)
112 			.append();
113 		assertBean(x, "append,consoleLevel,count,fileLevel,limit,logDir,logFile", "true,INFO,2,FINE,10000,/tmp,app.log");
114 	}
115 }
