35  @SuppressWarnings({

39  class ServletFilterRegistration_Test extends TestBase {
40  
41  	private static final Filter F = new PassThrough();
42  
43  	public static class PassThrough implements Filter {
44  		@Override /* Filter */
45  		public void doFilter(ServletRequest req, ServletResponse res, FilterChain chain) throws IOException, ServletException {
46  			chain.doFilter(req, res);
47  		}
48  	}
49  
50  	private static ServletFilterRegistration reg(int order, String...patterns) {
51  		return ServletFilterRegistration.create(new PassThrough()).urlPatterns(patterns).order(order).build();
52  	}
53  
54  	private static AuthFilterChain auth(BeanStore store) {
55  		return AuthFilterChain.create(store).build();
56  	}

62  	@Test void a01_defaults() {
63  		var r = ServletFilterRegistration.create(F).urlPatterns("/a/*").build();
64  		assertSame(F, r.getFilter());
65  		assertEquals(List.of("/a/*"), r.getUrlPatterns());
66  		assertEquals(Set.of(DispatcherType.REQUEST), r.getDispatcherTypes());
67  		assertEquals(ServletFilterRegistration.DEFAULT_ORDER, r.getOrder());
68  		assertEquals(0, r.getOrder());
69  		assertTrue(r.isAsyncSupported());
70  		assertTrue(r.getName().isEmpty());
71  	}

74  		var r = ServletFilterRegistration.create(F).urlPatterns("/a/*").dispatcherTypes(DispatcherType.ASYNC).build();
75  		assertEquals(Set.of(DispatcherType.ASYNC), r.getDispatcherTypes());
76  	}

168 	@Test void c02_sortedByOrderThenName() {
169 		var store = new BasicBeanStore();
170 		var late = reg(5, "/*");
171 		var early = reg(-5, "/*");
172 		var tieB = reg(0, "/*");
173 		var tieA = reg(0, "/*");
174 		store.addBean(ServletFilterRegistration.class, late, "late");
175 		store.addBean(ServletFilterRegistration.class, tieB, "b");
176 		store.addBean(ServletFilterRegistration.class, early, "early");
177 		store.addBean(ServletFilterRegistration.class, tieA, "a");
178 		var l = ServletFilterRegistration.collect(store);
179 		assertEquals(List.of(early, tieA, tieB, late), l);
180 	}

182 	@Test void c03_authAutoMounted_atAuthOrder_onRequestOnly_async() {
183 		var store = new BasicBeanStore();
184 		var auth = auth(store);
185 		store.addBean(AuthFilterChain.class, auth);
186 		var l = ServletFilterRegistration.collect(store);
187 		assertEquals(1, l.size());
188 		var r = l.get(0);
189 		assertSame(auth, r.getFilter());
190 		assertEquals(ServletFilterRegistration.AUTH_ORDER, r.getOrder());
191 		assertEquals(List.of("/*"), r.getUrlPatterns());
192 		assertEquals(Set.of(DispatcherType.REQUEST), r.getDispatcherTypes());
193 		assertTrue(r.isAsyncSupported());
194 	}

196 	@Test void c04_authBeforeDefaultOrder_afterHighestPrecedence() {
197 		var store = new BasicBeanStore();
198 		var auth = auth(store);
199 		store.addBean(AuthFilterChain.class, auth);
200 		var first = reg(ServletFilterRegistration.HIGHEST_PRECEDENCE, "/console/*");
201 		var dflt = reg(ServletFilterRegistration.DEFAULT_ORDER, "/*");
202 		store.addBean(ServletFilterRegistration.class, dflt, "dflt");
203 		store.addBean(ServletFilterRegistration.class, first, "first");
204 		var l = ServletFilterRegistration.collect(store);
205 		assertSame(first, l.get(0));
206 		assertSame(auth, l.get(1).getFilter());
207 		assertSame(dflt, l.get(2));
208 	}

210 	@Test void c05_unnamedRegistrationAtAuthOrder_authWinsTie() {
211 		var store = new BasicBeanStore();
212 		var auth = auth(store);
213 		store.addBean(AuthFilterChain.class, auth);
214 		var user = reg(ServletFilterRegistration.AUTH_ORDER, "/*");
215 		store.addBean(ServletFilterRegistration.class, user);  // unnamed: bean name "" ties the auth entry on name
216 		var l = ServletFilterRegistration.collect(store);
217 		assertSame(auth, l.get(0).getFilter());
218 		assertSame(user, l.get(1));
219 	}

235 	@Test void d02_differentAuthInstance_isASeparateFilter() {
236 		var store = new BasicBeanStore();
237 		var auth = auth(store);
238 		store.addBean(AuthFilterChain.class, auth);
239 		var other = ServletFilterRegistration.create(auth(store)).urlPatterns("/x/*").build();
240 		store.addBean(ServletFilterRegistration.class, other, "other");
241 		var l = ServletFilterRegistration.collect(store);
242 		assertEquals(2, l.size());
243 		assertSame(auth, l.get(0).getFilter());  // the automatic mount, at AUTH_ORDER, is still there
244 		assertSame(other, l.get(1));
