40  /**
41   * Phase-3 (Tier H14, second pass) lifecycle tests for {@link Microservice}.
42   *
43   * <p>
44   * Phase 2 already covered most builder/runtime surface area; this class targets the residual
45   * uncovered branches concentrated around:
46   * <ul>
47   *   <li>Builder methods left untested in Phase 2 ({@code configStore}, {@code consoleCommands(Class...)} catch path,
48   *       {@code manifest(File|Path|String|Class)}, {@code vars(Class...)}, {@code configurations(List)} null-entry skip).
49   *   <li>Constructor manifest-fallback file-system path (workingDir with META-INF/MANIFEST.MF on disk).
50   *   <li>Constructor config-resolution branches: {@code Main-Config} from manifest, {@code configFile} cmdline arg,
51   *       explicit {@code configStore} hit/miss, {@code workingDir}-scoped {@link FileStore}.
52   *   <li>Console-mode init paths: builder.consoleWriter==null fallback to {@code createLoggerConsoleWriter()},
53   *       {@code Console/commands} config string array (success + bad-class catch).
54   *   <li>{@code init()} with {@code Logging/logFile} set so the file-handler / console-handler path runs, plus
55   *       {@code Logging/levels} and {@code SystemProperties} sections.
56   *   <li>{@code start()} {@code RunningClassWithConfig} branch (config has a name).
57   *   <li>{@code stop()} {@code beanStore.close()} catch path (a {@code @PreDestroy} that throws).
58   *   <li>{@code getCandidateConfigNames}: {@code configFile} cmdline arg branch and {@code Main-Config} manifest branch.
59   *   <li>Protected {@code resolveFile} on the constructed microservice (absolute and relative+workingDir branches).
60   *   <li>{@code createLoggerConsoleWriter} writer mechanics: write/flush/close routed through the buffered logger.
61   * </ul>
62   *
63   * <p>
64   * Tests are kept tests-only: production behaviour is asserted as-is.  Two paths intentionally not exercised here:
65   * {@link Microservice#exit()} and {@link Microservice#kill()} both call {@link System#exit(int)} unconditionally and
66   * are not safely reachable from a unit test without bringing down the JVM.
67   */

311 	@Test void e02_createLoggerConsoleWriter_writeAndFlushAndClose() throws Exception {
312 		// Build an enabled-console microservice without supplying our own writer so consoleWriter is the fallback.
313 		// Then exercise PrintWriter-level write/flush/close so all branches in the inner Writer fire:
314 		//   - newline char ('\n') -> flushBuffer (line 1414)
315 		//   - carriage return ('\r') -> skipped (line 1416 false branch)
316 		//   - other char -> append (line 1417)
317 		//   - flush() -> flushBuffer
318 		//   - close() -> flushBuffer
319 		// init() calls LogManager.reset() which clears handlers, so attach the captor AFTER build().
320 		var captured = new ArrayList<LogRecord>();
321 		var captor = new Handler() {
322 			@Override public void publish(LogRecord r) { captured.add(r); }
323 			@Override public void flush() { /* no-op */ }
324 			@Override public void close() throws SecurityException { /* no-op */ }
325 		};
326 		var ms = Microservice.create().consoleEnabled(true).build();
327 		var logger = ms.getLogger();
328 		logger.addHandler(captor);
329 		try {
330 			try {
331 				// Reach the package-private console writer via reflection on the Microservice class.
332 				var pwField = Microservice.class.getDeclaredField("consoleWriter");
333 				pwField.setAccessible(true);
334 				var pw = (PrintWriter) pwField.get(ms);
335 				assertNotNull(pw);
336 				// Mix \r and \n and ordinary chars in a single write to exercise all three branches.
337 				pw.write("ab\rcd\nef");  // 'ab' buffered, '\r' skipped, 'cd' buffered, '\n' flushes 'abcd', 'ef' buffered
338 				pw.flush();              // flushes 'ef'
339 				// At this point we should have logged at least one buffered chunk via the logger.
340 				assertFalse(captured.isEmpty(), "Inner Writer should have flushed buffered content via the logger");
341 				// Clearing then closing flushes any residual buffer (empty) -> isEmpty() true branch on flushBuffer.
342 				captured.clear();
343 				pw.close();
344 				// No exception expected; close() with empty buffer is a no-op for the logger.
345 			} finally {
346 				ms.stop();
347 			}
348 		} finally {
349 			logger.removeHandler(captor);
350 		}
351 	}

424 		cfg.set("Logging/levels", "{'org.apache.juneau.marshall.microservice.test.lifecycle.g02_cfg':'INFO'}");

466 	public static class ThrowingPreDestroyBean {
467 		@PreDestroy public void onDestroy() {
468 			throw new RuntimeException("deliberate-pre-destroy-failure");
469 		}
470 	}

472 	@Configuration
473 	public static class ThrowingPreDestroyConfig {
474 		@Bean public ThrowingPreDestroyBean throwingBean() { return new ThrowingPreDestroyBean(); }
475 	}
476 	@Test void i01_stop_swallowsBeanStoreCloseFailures() throws Exception {
477 		var ms = Microservice.create().configurations(ThrowingPreDestroyConfig.class).build();
478 		// Force resolution so the bean is registered + will be visited by close().
479 		assertNotNull(ms.getBeanStore().getBean(ThrowingPreDestroyBean.class).orElse(null));
480 		// The throwing @PreDestroy must NOT propagate out of stop(); it should be logged via getLogger().
481 		assertDoesNotThrow(ms::stop);
