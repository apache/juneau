29  /**
30   * READY-394: verifies that {@link JettyServerComponent} publishes one lifecycle-owned {@link ReadinessState}
31   * instance into both {@code ms.getBeanStore()} and the auto-mounted {@link HealthServlet}'s own
32   * {@code RestContext} bean store, instead of every default-configured microservice falling back to the
33   * JVM-wide {@link ReadinessState#shared()} singleton.
34   *
35   * <h5 class='section'>Regression covered:</h5>
36   * <p>
37   * Before this fix, {@code JettyServerComponent.onStart/onStop} flipped {@code ReadinessState.resolve(store)}
38   * without ever registering a bean back into {@code store}, and {@link HealthAggregator} read from the probe
39   * servlet's own (unrelated) bean store.  Both sides therefore always fell back to {@link ReadinessState#shared()}
40   * on the default (no app {@code @Bean ReadinessState}) path, so stopping any one default-configured
41   * microservice flipped {@code /readyz} for every other one in the same JVM.
42   *
43   * <p>
44   * These tests intentionally stay scoped to the bean-store publish and the exact {@link ReadinessState#resolve(BeanStore)}
45   * resolution that {@link HealthAggregator#aggregate} performs, per their own names.  Real-HTTP-level probe-routing
46   * coverage (a real {@code GET /healthz}/{@code /readyz}/{@code /livez} over the Jetty connector) now lives in
47   * {@code HealthServlet_JettyHttpConnector_Test}; the multi-path auto-mount dispatch bug that previously prevented
48   * that round-trip was fixed in the real-HTTP-connector coverage.
49   *
50   * @since 10.0.0
51   */
52  @org.apache.juneau.testing.JettyMicroserviceTest
53  @SuppressWarnings({
54  	"resource" // Microservice/Server instances are test fixtures managed by the test lifecycle; explicit close is not needed for these assertions.
55  })
56  class JettyServerComponent_ReadinessStateDualStorePublish_Test extends TestBase {
57  
58  	private static Microservice create(Class<?>... configurations) throws Exception {
59  		var classes = new Class<?>[configurations.length + 1];
60  		System.arraycopy(configurations, 0, classes, 0, configurations.length);
61  		classes[configurations.length] = JettyConfiguration.class;
62  		return Microservice.create().configurations(classes).build();
63  	}
64  
65  	private static Server ephemeralServer() {
66  		var server = new Server();
67  		var connector = new ServerConnector(server);
68  		connector.setPort(0);
69  		server.addConnector(connector);
70  		var ctx = new ServletContextHandler();
71  		ctx.setContextPath("/");
72  		server.setAttribute("ServletContextHandler", ctx);
73  		server.setHandler(ctx);
74  		server.setStopTimeout(0L);
75  		return server;
76  	}
77  
78  	private static void stopQuietly(Microservice ms) {
79  		try {
80  			ms.stop();
81  		} catch (@SuppressWarnings("unused") Exception e) {
82  			// Best-effort cleanup; the test has already asserted or failed by this point.
83  		}
84  	}
85  
86  	private static HealthServlet findHealthServlet(Microservice ms) {
87  		for (var servlet : ms.getBeanStore().getBeansOfType(jakarta.servlet.Servlet.class).values())
88  			if (servlet instanceof HealthServlet hs)
89  				return hs;
90  		throw new AssertionError("Expected HealthProbeConfiguration to register a HealthServlet bean");
91  	}
92  
93  	//-----------------------------------------------------------------------------------------------------------------
94  	// A.  Two default-configured microservices in one JVM must not share a ReadinessState instance.
95  	//-----------------------------------------------------------------------------------------------------------------
96  
97  	@Configuration
98  	static class A_Config {
99  		@Bean Server jettyServer() { return ephemeralServer(); }
100 	}
101 
102 	@Test
103 	void a01_twoServices_defaultPath_readinessNotSharedAcrossServices() throws Exception {
104 		var msA = create(A_Config.class);
105 		var msB = create(A_Config.class);
106 		var aStopped = false;
107 		try {
108 			msA.start();
109 			msB.start();
110 
111 			var rsA = msA.getBeanStore().getBean(ReadinessState.class)
112 				.orElseThrow(() -> new AssertionError("Expected onStart() to publish a ReadinessState bean into msA's bean store"));
113 			var rsB = msB.getBeanStore().getBean(ReadinessState.class)
114 				.orElseThrow(() -> new AssertionError("Expected onStart() to publish a ReadinessState bean into msB's bean store"));
115 			assertNotSame(rsA, rsB, "Two default-configured microservices in the same JVM must not share one ReadinessState instance");
116 			assertTrue(rsA.isReady(), "A should be ready after its own start()");
117 			assertTrue(rsB.isReady(), "B should be ready after its own start()");
118 
119 			msA.stop();
120 			aStopped = true;
121 			assertFalse(rsA.isReady(), "A should be out of service after its own stop()");
122 			assertTrue(rsB.isReady(), "B must be unaffected by A's stop() -- no cross-service leakage via shared()");
123 		} finally {
124 			if (! aStopped)
125 				stopQuietly(msA);
126 			stopQuietly(msB);
127 		}
128 	}
129 
130 	//-----------------------------------------------------------------------------------------------------------------
131 	// B.  The health-probe servlet's own RestContext bean store holds the SAME instance as ms.getBeanStore(),
132 	//     and resolve() against it reflects flips made through ms.getBeanStore() -- the dual-store publish itself.
133 	//-----------------------------------------------------------------------------------------------------------------
134 
135 	@Configuration
136 	static class B_Config {
137 		@Bean Server jettyServer() { return ephemeralServer(); }
138 	}
139 
140 	@Test
141 	void b01_healthServletBeanStore_holdsSameInstance_asMicroserviceBeanStore() throws Exception {
142 		var ms = create(B_Config.class, HealthProbeConfiguration.class);
143 		try {
144 			ms.start();
145 			var rsFromMsStore = ms.getBeanStore().getBean(ReadinessState.class)
146 				.orElseThrow(() -> new AssertionError("Expected onStart() to publish a ReadinessState bean into ms's bean store"));
147 
148 			var probeBeanStore = findHealthServlet(ms).getContext().getBeanStore();
149 			var rsFromProbeStore = probeBeanStore.getBean(ReadinessState.class)
150 				.orElseThrow(() -> new AssertionError("Expected the probe servlet's own RestContext bean store to hold a ReadinessState bean"));
151 			assertSame(rsFromMsStore, rsFromProbeStore,
152 				"The probe servlet's RestContext bean store must hold the identical instance published into ms.getBeanStore()");
153 		} finally {
154 			stopQuietly(ms);
155 		}
156 	}
157 
158 	@Test
159 	void b02_probeResolution_reflectsInstanceFlippedViaMicroserviceBeanStore() throws Exception {
160 		var ms = create(B_Config.class, HealthProbeConfiguration.class);
161 		try {
162 			ms.start();
163 			var probeBeanStore = findHealthServlet(ms).getContext().getBeanStore();
164 
165 			// This is exactly what HealthAggregator.aggregate() calls for the READY probe.
166 			assertTrue(ReadinessState.resolve(probeBeanStore).isReady(), "Expected the probe to resolve as ready right after start()");
167 
168 			// Flip the SAME per-service instance published into ms.getBeanStore().  If the probe servlet's own
169 			// RestContext bean store didn't also receive this instance, resolve() here would still fall through
170 			// to shared()'s untouched state and stay ready.
171 			ms.getBeanStore().getBean(ReadinessState.class).orElseThrow().markOutOfService();
172 
173 			assertFalse(ReadinessState.resolve(probeBeanStore).isReady(),
174 				"Expected the probe's own bean store to resolve the instance flipped via ms.getBeanStore()");
175 		} finally {
176 			stopQuietly(ms);
177 		}
178 	}
