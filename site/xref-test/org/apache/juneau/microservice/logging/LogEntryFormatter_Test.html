203 		// OTel is not on this module's test classpath, so the trace/span fields render empty - but the
204 		// placeholders must still be recognized (no literal "{traceId}" leaks into the output).
205 		var f = new LogEntryFormatter("[{date} {level}] {msg} trace={traceId} span={spanId}%n", DATE_FORMAT, false);
206 		var r = new LogRecord(Level.INFO, "hello");
207 		var result = f.format(r);
208 		assertNotNull(result);
209 		assertTrue(result.contains("trace= span="), "Expected empty trace/span fields, got:\n" + result);
210 		assertFalse(result.contains("{traceId}"));
211 		assertFalse(result.contains("{spanId}"));
212 	}

214 	@Test void b02_traceIdSpanId_registeredAsParsableFields() {
215 		// {msg} uses a greedy (.*) match, so it must come last; put the trace/span fields ahead of it.
216 		var f = new LogEntryFormatter("[{date} {level}] trace={traceId} span={spanId} {msg}", DATE_FORMAT, false);
217 		var line = "[2023.01.15 10:30:45 INFO] trace=0af7651916cd43dd8448eb211c80319c span=b7ad6b7169203331 hello";
218 		var m = f.getLogEntryPattern().matcher(line);
219 		assertTrue(m.matches(), "Pattern should match a line with populated trace/span ids");
220 		assertEquals("0af7651916cd43dd8448eb211c80319c", f.getField("traceId", m));
221 		assertEquals("b7ad6b7169203331", f.getField("spanId", m));
222 	}

235 	@Test void c01_dateFormat_regexMetacharsEscapedAndMatch() {
236 		// A date format containing regex metacharacters must be escaped (not injected) yet still match its own
237 		// formatted output.  With the previous partial escaping, "(MM)" and "[dd]" produced a stray capture
238 		// group and character class, so the resulting pattern no longer matched the formatted date.
239 		var f = new LogEntryFormatter("[{date}] {msg}", "yyyy(MM)[dd]", false);
240 		var r = new LogRecord(Level.INFO, "hello");
241 		var line = f.format(r);
242 		var m = f.getLogEntryPattern().matcher(line);
243 		assertTrue(m.matches(), "Pattern should match its own formatted output with a metacharacter date format, got:\n" + line);
244 	}
245 
246 	@Test void c02_dateFormat_unbalancedMetacharsDoNotThrow() {
247 		// An unbalanced regex metacharacter in the date format previously produced an invalid pattern that
248 		// threw a PatternSyntaxException during construction; it must now be escaped and compile cleanly.
249 		assertDoesNotThrow(() -> new LogEntryFormatter("[{date}] {msg}", "yyyy(((MM", false));
250 	}

252 	@Test void c03_format_regexMetacharsEscaped() {
253 		// Regex metacharacters in the literal portions of the format string must be escaped, so an adversarial
254 		// format cannot inject regex syntax or break compilation of the log-parsing pattern.
255 		assertDoesNotThrow(() -> new LogEntryFormatter("(([{date}]{level})+ {msg}%n", DATE_FORMAT, false));
