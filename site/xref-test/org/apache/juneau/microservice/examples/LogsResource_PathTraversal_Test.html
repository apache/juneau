33  /**
34   * Regression tests for path-traversal hardening in {@link LogsResource#getFile(String)}.
35   *
36   * <p>
37   * The pre-fix implementation did
38   * {@code new File(logDir.getAbsolutePath() + '/' + path)} and only verified
39   * {@link File#exists()} — the same vulnerability pattern as {@link DirectoryResource}. Each test
40   * below asserts the post-fix behavior: paths that escape the configured log root are rejected
41   * with a 403 (Forbidden); paths that stay inside the root continue to work.
42   *
43   * <p>
44   * The funnel for every public operation on this resource (view, parse, download, delete, info)
45   * is the private {@code getFile(String)} method, so the fix is applied once and verified across
46   * each operation surface here.
47   *
48   * <p>
49   * Test wiring note: {@link LogsResource} initializes its static log-directory field from a
50   * {@link Config} resolved out of the REST bean store via {@code @RestInit}. The test provides
51   * an in-memory Config via {@link MockRestClient.Builder#overridingBeanStore(BeanStore)} so the
52   * static state points at the per-test temp directory.
53   *
54   * @since 10.0.0
55   */
56  class LogsResource_PathTraversal_Test extends TestBase {
57  
58  	@TempDir
59  	static Path tempDir;
60  
61  	static Path logRoot;
62  	static Path outsideSecret;
63  	static Path symlinkInside;
64  	static Path symlinkEscape;
65  
66  	@BeforeAll
67  	static void setup() throws Exception {
68  		logRoot = tempDir.resolve("logs-root");
69  		Files.createDirectories(logRoot);
70  		Files.writeString(logRoot.resolve("inside.log"), "INSIDE_LOG_ROOT");
71  
72  		var nestedDir = logRoot.resolve("a/b");
73  		Files.createDirectories(nestedDir);
74  		Files.writeString(nestedDir.resolve("nested.log"), "NESTED_INSIDE_LOG");
75  
76  		outsideSecret = tempDir.resolve("outside-secret.log");
77  		Files.writeString(outsideSecret, "AUDIT_OUTSIDE_LOG_SECRET");
78  
79  		try {
80  			symlinkInside = logRoot.resolve("link-to-inside.log");
81  			Files.createSymbolicLink(symlinkInside, logRoot.resolve("inside.log"));
82  		} catch (UnsupportedOperationException | IOException e) {
83  			symlinkInside = null;
84  		}
85  		try {
86  			symlinkEscape = logRoot.resolve("link-to-outside.log");
87  			Files.createSymbolicLink(symlinkEscape, outsideSecret);
88  		} catch (UnsupportedOperationException | IOException e) {
89  			symlinkEscape = null;
90  		}
91  	}

110 	// Tests intentionally leave resources open; try-with-resources would obscure the test intent.
111 	@SuppressWarnings({
112 		"resource"  // Closeable resources in tests are intentionally unassigned; closing is handled by test infrastructure.
113 	})
114 	private static MockRestClient buildClient() {
115 		var cfg = Config.create().memStore().build();
116 		cfg.set("Logging/logDir", logRoot.toString());
117 		cfg.set("Logging/allowDeletes", "true");
118 		var overlay = new BasicBeanStore().addBean(Config.class, cfg);
119 		return MockRestClient.create(TestLogsResource.class)
120 			.overridingBeanStore(overlay)
121 			.ignoreErrors()
122 			.noTrace()
123 			.build();
124 	}

213 	@Test void t08_absolutePathInRequest_isNotTreatedAsFilesystemAbsolute() throws Exception {
214 		// On Linux, "/etc/passwd" is an absolute path. Path.resolve(absolute) returns the absolute
215 		// path verbatim, which fails the startsWith(root) boundary check → 403. On Windows the same
216 		// path is interpreted as relative to root → 404 (no such file under root). Either is a
217 		// non-leak outcome.
218 		try (var c = buildClient()) {
219 			@SuppressWarnings({
220 				"resource"  // Closeable resources in tests are intentionally unassigned; closing is handled by test infrastructure.
221 			})
222 			var status = c.get("/etc/passwd").run().getStatusCode();
223 			assertTrue(status == 403 || status == 404, "Status must be 403 or 404, was: " + status);
224 		}
225 	}
226 
227 	@Test void t09_urlEncodedTraversal_doesNotLeak() throws Exception {
228 		// Servlet containers / Apache HttpClient may URL-decode the path before our handler sees
229 		// it; some containers reject "%2e%2e" outright. Either outcome is acceptable as long as
230 		// the outside-root secret is NOT returned.
231 		try (var c = buildClient()) {
232 			@SuppressWarnings({
233 				"resource"  // Closeable resources in tests are intentionally unassigned; closing is handled by test infrastructure.
234 			})
235 			var resp = c.get("/%2e%2e/outside-secret.log").run();
236 			assertFalse(resp.getContent().asString().contains("AUDIT_OUTSIDE_LOG_SECRET"),
237 				"URL-encoded traversal must not leak the outside-root secret. Status was: " + resp.getStatusCode());
238 			assertNotEquals(200, resp.getStatusCode(),
239 				"URL-encoded traversal must not return 200. Status was: " + resp.getStatusCode());
240 		}
241 	}

247 	@Test void t10_deleteTraversal_does_not_delete_outside_root() throws Exception {
248 		assertTrue(Files.exists(outsideSecret), "Pre-condition: outside secret must exist");
249 
250 		try (var c = buildClient()) {
251 			@SuppressWarnings({
252 				"resource"  // Closeable resources in tests are intentionally unassigned; closing is handled by test infrastructure.
253 			})
254 			var status = c.delete("/../outside-secret.log").run().getStatusCode();
255 			assertEquals(403, status, "DELETE /../outside-secret.log must be rejected with 403");
256 			assertTrue(Files.exists(outsideSecret),
257 				"Outside-root file must NOT be deleted by DELETE traversal");
258 			assertEquals("AUDIT_OUTSIDE_LOG_SECRET", Files.readString(outsideSecret),
259 				"Outside-root file content must be unchanged after DELETE traversal attempt");
260 		}
