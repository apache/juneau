59  	private static DataTablesRequest request(String columns, String body) throws Exception {
60  		return Json.to("{\"draw\":1,\"length\":10,\"columns\":" + columns + "," + body + "}", DataTablesRequest.class);
61  	}
62  
63  	private static <T> List<Object> page(DataTablesRequest r, BeanQuerySession<T> s, Function<T,Object> id) {
64  		try (s) {
65  			var res = DataTablesQuery.run(r, s);
66  			assertString("", Objects.toString(res.getError(), ""));  // the query error, if any
67  			var out = new ArrayList<Object>();
68  			out.add(res.getRecordsTotal());
69  			out.add(res.getRecordsFiltered());
70  			res.getData().forEach(x -> out.add(id.apply(x)));
71  			return out;
72  		}
73  	}

94  	@Test void a08_defaultPageIsNotTrivial() throws Exception {
95  		var p = page(request(PET_COLUMNS, "\"start\":0,\"order\":[{\"column\":0,\"dir\":\"asc\"}]"), sql.pets().getSession(), Pet::getId);
96  		assertSize(12, p); // total, filtered, ten ids
97  	}

99  	@Test void a09_columnOverrideFilters() throws Exception {
100 		// The ribbon cases rely on the body's later "columns" key replacing the default one; if it didn't, both
101 		// sides would match unfiltered and the parity check would prove nothing.
102 		var p = page(request(PET_COLUMNS, "\"start\":0,\"order\":[{\"column\":0,\"dir\":\"asc\"}],\"columns\":[{\"data\":\"id\"},{\"data\":\"name\"},{\"data\":\"species\"},{\"data\":\"price\"},{\"data\":\"status\",\"search\":{\"value\":\"$eq(SOLD)\"}}]"), sql.pets().getSession(), Pet::getId);
103 		assertString("true", String.valueOf(((Number) p.get(1)).longValue() < ((Number) p.get(0)).longValue()));
104 	}

160 	@CsvSource(delimiter='|', textBlock="""
161 		status | {"data":"id"},{"data":"petId"},{"data":"shipDate"},{"data":"status","search":{"value":"$eq(PLACED)"}}

165 		// Guards d02/d03: an empty or unfiltered page would match on both sides and prove nothing.
166 		var p = page(request(ORDER_COLUMNS, "\"start\":0,\"order\":[{\"column\":0,\"dir\":\"asc\"}],\"columns\":[" + columns + "]"), sql.orders().getSession(), Order::getId);
