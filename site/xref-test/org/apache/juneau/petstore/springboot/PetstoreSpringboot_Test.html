33   * Integration test that boots the petstore Spring Boot deployment on a random port and exercises the CRUD
34   * surface plus the Spring {@code @Autowired} injection demo.
35   *
36   * <p>
37   * Tagged {@code @SpringbootTest} ({@code @Tag("container")} + {@code @Tag("springboot")}) so it skips under

63  	private static final int READY_TIMEOUT_SECONDS = Integer.parseInt(
64  		System.getProperty("juneau.petstore.springboot.readyTimeoutSeconds", "180"));

66  	/**
67  	 * Primes the root endpoint before the timed test methods run.
68  	 *
69  	 * <p>
70  	 * Spring Boot's {@code RANDOM_PORT} environment only waits for the Tomcat connector to bind — not for the Juneau
71  	 * REST servlet to initialize. The first request to the group resource ({@code /}) forces one-time
72  	 * {@code RestContext} setup of the root <i>and all of its child resources</i> (serializer/parser metadata,
73  	 * {@code HtmlDocSerializer} construction) in a single request. Under a loaded CI agent this cold start can exceed
74  	 * the per-request timeout the test methods use (it has timed out at exactly 10s under a loaded CI run before). Absorbing that
75  	 * startup cost here once (with a generous budget + retry) removes the race from {@code a01} while keeping the
76  	 * per-test timeouts tight. Mirrors the same guard in {@code PetstoreJetty_Test}.
77  	 *
78  	 * <p>
79  	 * The overall deadline is deliberately generous (default {@value #READY_TIMEOUT_SECONDS}s, configurable via the
80  	 * {@code juneau.petstore.springboot.readyTimeoutSeconds} system property) because under a full-reactor build the
81  	 * whole JVM fleet is CPU-starved and Spring Boot startup itself can take far longer than it does in isolation.
82  	 * Each probe still uses a short per-attempt request timeout so a slow-to-bind server fails fast and gets retried
83  	 * instead of burning the whole budget on one stuck request.
84  	 */
85  	@BeforeEach
86  	void warmUpServer() throws Exception {
87  		if (warmedUp)
88  			return;
89  		var deadline = Instant.now().plusSeconds(READY_TIMEOUT_SECONDS);
90  		Exception last = null;
91  		while (Instant.now().isBefore(deadline)) {
92  			try {
93  				var req = HttpRequest.newBuilder()
94  					.uri(URI.create("http://localhost:" + port + "/"))
95  					.timeout(Duration.ofSeconds(5))
96  					.header("Accept", "text/html")
97  					.GET()
98  					.build();
99  				if (HTTP.send(req, BodyHandlers.ofString()).statusCode() == 200) {
100 					warmedUp = true;
101 					return;
102 				}
103 			} catch (IOException e) {
104 				last = e;
105 			}
106 			Thread.sleep(500);
107 		}
108 		throw new IllegalStateException(
109 			"Petstore Spring Boot server did not become ready within " + READY_TIMEOUT_SECONDS + "s", last);
110 	}

112 	private HttpResponse<String> get(String path, String accept) throws Exception {
113 		var req = HttpRequest.newBuilder()
114 			.uri(URI.create("http://localhost:" + port + path))
115 			.timeout(Duration.ofSeconds(30))
116 			.header("Accept", accept)
117 			.GET()
118 			.build();
119 		return HTTP.send(req, BodyHandlers.ofString());
120 	}

122 	private HttpResponse<String> getWithAuth(String path, String accept, String authValue) throws Exception {
123 		var req = HttpRequest.newBuilder()
124 			.uri(URI.create("http://localhost:" + port + path))
125 			.timeout(Duration.ofSeconds(30))
126 			.header("Accept", accept)
127 			.header("Authorization", authValue)
128 			.GET()
129 			.build();
130 		return HTTP.send(req, BodyHandlers.ofString());
131 	}
132 
133 	private HttpResponse<String> post(String path, String contentType, String body) throws Exception {
134 		var req = HttpRequest.newBuilder()
135 			.uri(URI.create("http://localhost:" + port + path))
136 			.timeout(Duration.ofSeconds(30))
137 			.header("Content-Type", contentType)
138 			.header("Accept", "application/json")
139 			.POST(HttpRequest.BodyPublishers.ofString(body))
140 			.build();
141 		return HTTP.send(req, BodyHandlers.ofString());
142 	}

147 			.timeout(Duration.ofSeconds(30))
148 			.DELETE()
149 			.build();
150 		return HTTP.send(req, BodyHandlers.ofString());

289 	private static long extractLong(String body, String key) {
290 		var i = body.indexOf(key);
291 		if (i < 0)
292 			return 0L;
293 		i += key.length();
294 		var end = i;
295 		while (end < body.length() && (Character.isDigit(body.charAt(end)) || body.charAt(end) == '-'))
296 			end++;
297 		if (end == i)
298 			return 0L;
299 		return Long.parseLong(body.substring(i, end));
300 	}
