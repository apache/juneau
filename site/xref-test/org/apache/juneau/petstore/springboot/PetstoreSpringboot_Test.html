87  	/**
88  	 * Primes the root endpoint before the timed test methods run.
89  	 *
90  	 * <p>
91  	 * Spring Boot's {@code RANDOM_PORT} environment only waits for the Tomcat connector to bind — not for the Juneau
92  	 * REST servlet to initialize. The first request to the group resource ({@code /}) forces one-time
93  	 * {@code RestContext} setup of the root <i>and all of its child resources</i> (serializer/parser metadata,
94  	 * {@code HtmlDocSerializer} construction) in a single request. Under a loaded CI agent this cold start can exceed
95  	 * the per-request timeout the test methods use (it has timed out at exactly 10s under a loaded CI run before). Absorbing that
96  	 * startup cost here once (with a generous budget + retry) removes the race from {@code a01} while keeping the
97  	 * per-test timeouts tight. Mirrors the same guard in {@code PetstoreJetty_Test}.
98  	 *
99  	 * <p>
100 	 * The overall deadline is deliberately generous (default {@value #READY_TIMEOUT_SECONDS}s, configurable via the
101 	 * {@code juneau.petstore.springboot.readyTimeoutSeconds} system property) because under a full-reactor build the
102 	 * whole JVM fleet is CPU-starved and Spring Boot startup itself can take far longer than it does in isolation.
103 	 * Each probe still uses a short per-attempt request timeout so a slow-to-bind server fails fast and gets retried
104 	 * instead of burning the whole budget on one stuck request.
105 	 */
106 	@BeforeEach
107 	void warmUpServer() throws Exception {
108 		if (warmedUp)
109 			return;
110 		var deadline = Instant.now().plusSeconds(READY_TIMEOUT_SECONDS);
111 		Exception last = null;
112 		while (Instant.now().isBefore(deadline)) {
113 			try {
114 				var req = HttpRequest.newBuilder()
115 					.uri(URI.create("http://localhost:" + port + "/"))
116 					.timeout(Duration.ofSeconds(5))
117 					.header("Accept", "text/html")
118 					.GET()
119 					.build();
120 				// GET / is a 303 to /console/store (see a01); redirects aren't followed.
121 				if (HTTP.send(req, BodyHandlers.ofString()).statusCode() == 303) {
122 					warmedUp = true;
123 					return;
124 				}
125 			} catch (IOException e) {
126 				last = e;
127 			}
128 			Thread.sleep(500);
129 		}
130 		throw new IllegalStateException(
131 			"Petstore Spring Boot server did not become ready within " + READY_TIMEOUT_SECONDS + "s", last);
132 	}

413 	private static String boundaryReason(HttpResponse<?> resp) {
414 		return resp.headers().firstValue("X-Request-Boundary").orElse(null);
415 	}
416 
417 	private HttpResponse<String> send(String method, String path, String contentType, String origin, String token, String body) throws Exception {
418 		var b = HttpRequest.newBuilder().uri(URI.create(self() + path)).timeout(Duration.ofSeconds(30))
419 			.header("Accept", "application/json").method(method, HttpRequest.BodyPublishers.ofString(body));
420 		if (contentType != null)
421 			b.header("Content-Type", contentType);
422 		if (origin != null)
423 			b.header("Origin", origin);
424 		if (token != null)
425 			b.header("X-Csrf-Token", token);
426 		return HTTP.send(b.build(), BodyHandlers.ofString());
427 	}

510 	private static long extractLong(String body, String key) {
511 		var i = body.indexOf(key);
512 		if (i < 0)
513 			return 0L;
514 		i += key.length();
515 		var end = i;
516 		while (end < body.length() && (Character.isDigit(body.charAt(end)) || body.charAt(end) == '-'))
517 			end++;
518 		if (end == i)
519 			return 0L;
520 		return Long.parseLong(body.substring(i, end));
521 	}
