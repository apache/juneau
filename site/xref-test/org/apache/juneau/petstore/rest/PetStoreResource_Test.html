66  	@Test void a04_createPet_roundTrip() throws Exception {
67  		var c = client();
68  		var created = c.post("/pets", new Pet()
69  				.setName("Wanda")
70  				.setSpecies(Species.RABBIT)
71  				.setPrice(12.50f)
72  				.setStatus(PetStatus.AVAILABLE))
73  			.run()
74  			.assertStatus(200)
75  			.getContent().as(Pet.class);
76  		assertNotEquals(0L, created.getId());
77  
78  		var fetched = c.get("/pets/" + created.getId()).run().assertStatus(200).getContent().as(Pet.class);
79  		assertBean(fetched, "name,species,status", "Wanda,RABBIT,AVAILABLE");
80  	}

156 	private static byte[] pngBytes() throws IOException {
157 		var image = new BufferedImage(2, 2, BufferedImage.TYPE_INT_RGB);
158 		try (var baos = new ByteArrayOutputStream()) {
159 			ImageIO.write(image, "png", baos);
160 			return baos.toByteArray();
161 		}
162 	}
163 
164 	@Test void d01_getPetPhoto_notUploaded_404() throws Exception {
165 		client().get("/pets/1/photo").accept("image/png").run().assertStatus(404);
166 	}

169 		client().get("/pets/99999/photo").accept("image/png").run().assertStatus(404);
170 	}
171 
172 	@Test void d03_putPetPhoto_unknownPet_404() throws Exception {
173 		client().put("/pets/99999/photo", new ByteArrayInputStream(pngBytes())).contentType("image/png")
174 			.run().assertStatus(404);
175 	}
176 
177 	@Test void d04_putPetPhoto_thenGet_roundTrip() throws Exception {
178 		var c = client();
179 		c.put("/pets/1/photo", new ByteArrayInputStream(pngBytes())).contentType("image/png")
180 			.run().assertStatus(200);
181 
182 		var bytes = c.get("/pets/1/photo").accept("image/png").run().assertStatus(200).getContent().asBytes();
183 		assertTrue(bytes.length > 0);
184 
185 		var pet = c.get("/pets/1").run().assertStatus(200).getContent().as(Pet.class);
186 		assertEquals("/petstore/pets/1/photo", pet.getPhoto());
187 	}
