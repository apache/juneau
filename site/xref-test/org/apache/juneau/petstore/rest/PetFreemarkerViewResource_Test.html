17  package org.apache.juneau.petstore.rest;
18  
19  import org.apache.juneau.*;
20  import org.apache.juneau.rest.mock.classic.*;
21  import org.junit.jupiter.api.*;
22  
23  @SuppressWarnings({
24  	"resource" // Closeable test fixtures held in static fields; lifecycle managed by the test/framework, not a real leak.
25  })
26  class PetFreemarkerViewResource_Test extends TestBase {
27  
28  	private static final MockRestClient CLIENT = MockRestClient.buildLax(PetFreemarkerViewResource.class);
29  
30  	@Test void a01_viewPet_rendersTemplateWithPetName() throws Exception {
31  		CLIENT.get("/pets/1/view")
32  			.run()
33  			.assertStatus(200)
34  			.assertContent().asString().isContains("Mr. Frisky")
35  			.assertContent().asString().isContains("Rendered via FreeMarker.");
36  	}
37  
38  	@Test void a02_viewPet_rendersSpeciesAndStatus() throws Exception {
39  		CLIENT.get("/pets/1/view")
40  			.run()
41  			.assertStatus(200)
42  			.assertContent().asString().isContains("CAT")
43  			.assertContent().asString().isContains("AVAILABLE");
44  	}
45  
46  	@Test void a03_viewPet_unknownId_404() throws Exception {
47  		CLIENT.get("/pets/99999/view")
48  			.run()
49  			.assertStatus(404);
50  	}
51  }
