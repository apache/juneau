17  package org.apache.juneau.petstore.marshall;
18  
19  import static org.apache.juneau.bct.BctAssertions.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.awt.image.*;
23  import java.io.*;
24  import java.nio.charset.*;
25  
26  import javax.imageio.*;
27  
28  import org.apache.juneau.*;
29  import org.apache.juneau.marshall.parser.*;
30  import org.junit.jupiter.api.*;
31  
32  class PetPhotoParser_Test extends TestBase {
33  
34  	private static final int W = 7, H = 5;
35  
36  	private static byte[] encode(String format) throws IOException {
37  		var img = new BufferedImage(W, H, BufferedImage.TYPE_INT_RGB);
38  		for (var x = 0; x < W; x++)
39  			for (var y = 0; y < H; y++)
40  				img.setRGB(x, y, (x * 30) << 16 | (y * 40) << 8 | 0x80);
41  		var baos = new ByteArrayOutputStream();
42  		assertTrue(ImageIO.write(img, format, baos));
43  		return baos.toByteArray();
44  	}
45  
46  	private final PetPhotoParser parser = new PetPhotoParser();
47  
48  	@Test void a01_parsesJpeg() throws Exception {
49  		assertTrue(parser.canHandle("image/jpeg"));
50  		assertBean(parser.read(encode("jpeg"), BufferedImage.class), "width,height", "7,5");
51  	}
52  
53  	@Test void a02_parsesPng() throws Exception {
54  		assertTrue(parser.canHandle("image/png"));
55  		assertBean(parser.read(encode("png"), BufferedImage.class), "width,height", "7,5");
56  	}
57  
58  	@Test void a03_mediaTypesIncludeJpegAndLegacyJpg() {
59  		assertList(parser.getMediaTypes(), "image/png", "image/jpeg", "image/jpg");
60  		// Legacy non-standard alias still accepted.
61  		assertTrue(parser.canHandle("image/jpg"));
62  	}
63  
64  	@Test void a04_nonImageInputThrowsParseException() {
65  		var input = "not an image".getBytes(StandardCharsets.UTF_8);
66  		assertThrows(ParseException.class, () -> parser.read(input, BufferedImage.class));
67  	}
68  
69  	@Test void a05_corruptJpegThrowsParseException() throws Exception {
70  		var good = encode("jpeg");
71  		// Valid JPEG header followed by garbage: ImageIO throws instead of returning null.
72  		var bad = new byte[good.length];
73  		System.arraycopy(good, 0, bad, 0, 40);
74  		for (var i = 40; i < bad.length; i++)
75  			bad[i] = (byte)0x5A;
76  		assertThrows(ParseException.class, () -> parser.read(bad, BufferedImage.class));
77  	}
78  
79  	@Test void a06_corruptPngThrowsParseException() throws Exception {
80  		var good = encode("png");
81  		var bad = java.util.Arrays.copyOf(good, 40);
82  		assertThrows(ParseException.class, () -> parser.read(bad, BufferedImage.class));
83  	}
84  }
