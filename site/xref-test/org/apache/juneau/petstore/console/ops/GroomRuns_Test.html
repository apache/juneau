38  	private static final Pattern ANSI = Pattern.compile("\u001b\\[[0-9;]*m");
39  
40  	private static GroomRuns.Run await(GroomRuns.Run r) throws Exception {
41  		for (var i = 0; i < 1000 && ! r.done(); i++)
42  			Thread.sleep(5);
43  		assertTrue(r.done(), () -> "Run '" + r.id() + "' did not finish");
44  		return r;
45  	}
46  
47  	/** Reads the whole source from offset 0. */
48  	private static String read(TerminalSource s) throws IOException {
49  		var out = new ByteArrayOutputStream();
50  		s.copyTo(out);
51  		return out.toString(UTF_8);
52  	}
53  
54  	private static List<String> plain(TerminalSource s) throws IOException {
55  		return Arrays.stream(ANSI.matcher(read(s)).replaceAll("").split("\r\n")).toList();
56  	}

58  	@Test void a01_aRunWritesTheScriptAndFinishes() throws Exception {
59  		try (var g = new GroomRuns(1)) {
60  			var r = await(g.start("Rex", false, null));
61  			var lines = plain(r.terminal());
62  			assertEquals(GroomRuns.SCRIPT.size() + GroomRuns.TRANSCRIPT.size() - 1 + 2, lines.size() - 1, () -> "lines: " + lines);
63  			assertString("Grooming Rex", lines.get(0));
64  			assertTrue(lines.contains("\u2713 All tools ready"), () -> "lines: " + lines);
65  			assertTrue(lines.contains("\u2717 Bow tie: none in stock"), () -> "lines: " + lines);
66  			assertTrue(lines.contains("    at Dryer.heat(Dryer.java:42)"), () -> "lines: " + lines);
67  			assertTrue(read(r.terminal()).contains("\u001b[1m" + "Grooming Rex"), "styled marker");
68  			assertBean(r.terminal().read(0, 1), "done", "false");
69  			assertBean(r.row(), "state,lines", "DONE," + lines.size());
70  		}
71  	}

73  	@Test void a02_progressStripShowsEachWashPass() throws Exception {
74  		try (var g = new GroomRuns(1)) {
75  			var text = read(await(g.start("Rex", false, null)).terminal());
76  			var strip = Arrays.stream(text.split("\r\n")).filter(l -> l.startsWith("Wash passes ")).findFirst().orElseThrow();
77  			assertEquals(GroomRuns.WASH_PASSES, strip.chars().filter(c -> c == '\u25a0').count());
78  			assertEquals(GroomRuns.WASH_PASSES - 2, strip.split("\u001b\\[32m", -1).length - 1);
79  		}
80  	}

102 	@Test void a05_sourcesAreLookedUpByRunId() throws Exception {
103 		try (var g = new GroomRuns(1)) {
104 			var r = await(g.start("Rex", false, null));
105 			assertSame(r.terminal(), g.source(r.id()).orElseThrow());
106 			assertSame(r.fileSource(), g.source(r.id() + GroomRuns.FILE_SUFFIX).orElseThrow());
107 			assertTrue(g.source("nope").isEmpty());
108 			assertTrue(g.source("nope" + GroomRuns.FILE_SUFFIX).isEmpty());
109 			assertTrue(g.source(null).isEmpty());
110 			assertTrue(r.id().matches("[0-9a-f]{32}"), r.id());
111 		}

118 		try (var g = new GroomRuns(1)) {
119 			var r = await(g.start("Rex", false, null));
120 			var c = r.fileSource().read(0, TerminalSource.MAX_READ_BYTES);
121 			assertBean(c, "done,cols,rows", "true," + GroomRuns.COLS + "," + GroomRuns.ROWS);
122 			var lines = plain(r.fileSource());
123 			assertEquals(GroomRuns.SCRIPT.size(), lines.size(), () -> "lines: " + lines);
124 			assertTrue(lines.get(0).endsWith("groomer 2.1 starting"), () -> "lines: " + lines); // The OSC title survives the SGR-only strip.
125 			assertString("groom finished", lines.get(lines.size() - 1));
126 		}

138 	@Test void a08_oldRunsAreDroppedWithTheirFiles() throws Exception {
139 		try (var g = new GroomRuns(60_000)) {
140 			var first = g.start("First", false, null);
141 			GroomRuns.Run last = null;
142 			for (var i = 0; i < GroomRuns.MAX_RUNS; i++)
143 				last = g.start("Pet " + i, false, null);
144 			assertEquals(GroomRuns.MAX_RUNS, g.list().size());
145 			assertTrue(g.get(first.id()).isEmpty());
146 			assertFalse(Files.exists(first.file()));
147 			assertFalse(Files.exists(first.sizeFile));
148 			assertSame(last, g.latest());
149 		}
