17  package org.apache.juneau.petstore.console.dev;
18  
19  import static org.apache.juneau.petstore.console.PetstoreConsoleFixture.*;
20  import static org.apache.juneau.rest.server.console.test.PageContractAssert.*;
21  
22  import org.apache.juneau.*;
23  import org.junit.jupiter.api.*;
24  
25  /**
26   * The Developer and About nav nodes exist on every page, and their routes resolve.
27   */
28  @SuppressWarnings({
29  	"resource" // MockRestClient is a no-op close.
30  })
31  class DevNav_Test extends TestBase {
32  
33  	//-----------------------------------------------------------------------------------------------------------------
34  	// a - nav tree
35  	//-----------------------------------------------------------------------------------------------------------------
36  
37  	@Test void a01_developerNodeAndChildren() throws Exception {
38  		assertPage(page(client(), "/console/store"))
39  			.hasNavPath("dev")
40  			.hasNavChildren("dev", "cards", "themes", "flavors", "secure")
41  			.hasNavChildren("dev/flavors", "html", "freemarker", "mustache", "react")
42  			.hasNavHref("dev", "/console/dev/cards")
43  			.hasNavHref("dev/flavors", "/console/dev/flavors/html")
44  			.hasNavHref("dev/flavors/react", "/console/dev/flavors/react");
45  	}
46  
47  	@Test void a02_aboutNodeExists() throws Exception {
48  		assertPage(page(client(), "/console/store"))
49  			.hasNavPath("about")
50  			.hasNavHref("about", "/console/about");
51  	}
52  
53  	//-----------------------------------------------------------------------------------------------------------------
54  	// b - routes
55  	//-----------------------------------------------------------------------------------------------------------------
56  
57  	@Test void b01_devRootRedirectsToFirstChild() throws Exception {
58  		rawClient().get("/console/dev").run().assertStatus(303).assertHeader("Location").isMatches("*/console/dev/cards");
59  	}
60  
61  	@Test void b02_levelThreePageActivatesFullPath() throws Exception {
62  		assertPage(page(client(), "/console/dev/flavors/mustache")).isValid().hasActiveNav("dev", "flavors", "mustache");
63  	}
64  
65  	@Test void b03_unknownFlavor404() throws Exception {
66  		client().get("/console/dev/flavors/velocity").accept("text/html").run().assertStatus(404);
67  	}
68  }
