35  	private static final String BASE = "/console/dev/dashboard";
36  
37  	private static String csv(String variant, String query) throws Exception {
38  		return client().get(BASE + "/series/" + variant + query).run().assertStatus(200)
39  			.assertHeader("Content-Type").isContains("text/csv").getContent().asString();
40  	}

46  	@Test void a01_dashboardContract() throws Exception {
47  		var html = page(client(), BASE);
48  		var page = assertPage(html).isValid().hasActiveNav("dev").hasNoWiringErrors()
49  			.hasCard("daily", "chart").hasCard("by-reason", "chart").hasCard("by-weekday", "chart").hasCard("share", "chart")
50  			.publishes("daily", "zoom:daily");
51  		var d = page.contract().getMap("dashboard");
52  		assertBean(d, "id,title,grid,status{length,0}", "overview,Order issues,2,{1,data:orders}");
53  		assertBean(d, "panels{length,0{card,span},1{card,span},2{card,span}}", "{3,{daily,2},{by-reason,1},{by-weekday,1}}");
54  		assertBean(d, "rail{side,width,sections{length,0{kind,id,custom,defaultPreset},1{kind,id},2{kind,id,source,dim},3{kind,card},4{kind,id,name}}}",
55  			"{left,280,{5,{range,range,true,30d},{search,q},{choice,reasons,orders,reason},{card,share},{custom,late,petstore.late}}}");
56  		assertBean(d, "persist{ns,version,shareLink,topics{length,0,1,2,3}}",
57  			"{petstore.dashboard,1,true,{4,filter:range,filter:q,filter:reasons,zoom:daily}}");
58  		// The source entry; the page script and the dashboard runtime both load.
59  		assertBean(page.contract().getList("dataSources").getMap(0), "id,kind,url,subscribes{length}", "orders,source,/console/dev/dashboard/series/ok,{4}");
60  		assertContainsAll(html, "/console/dev/dashboard/dashboard.js", "juneau-dashboard.js", BASE + "/series/ok");
61  	}

63  	@Test void a02_builderOnly() throws Exception {
64  		var rsp = client().get(BASE).accept("text/html").run().assertStatus(200);
65  		assertEquals("builder", rsp.getHeader("X-Petstore-Page").orElse(null));
66  	}

68  	@Test void a03_variantAndVersionParams() throws Exception {
69  		assertContains(BASE + "/series/slow", page(client(), BASE + "?variant=slow"));
70  		assertContains(BASE + "/series/ok", page(client(), BASE + "?variant=bogus"));   // unknown is ok
71  		assertBean(assertPage(page(client(), BASE + "?variant=right")).contract().getMap("dashboard").getMap("rail"), "side", "right");
72  		assertBean(assertPage(page(client(), BASE + "?v=3")).contract().getMap("dashboard"), "persist{version}", "{3}");
73  		assertBean(assertPage(page(client(), BASE + "?v=0")).contract().getMap("dashboard"), "persist{version}", "{1}");
74  	}

87  	@Test void b01_oneRowPerUtcDay() throws Exception {
88  		assertString("""
89  			day,reason,orders

91  			2026-10-02T00:00:00Z,late,22
92  			2026-10-03T00:00:00Z,lost,29
93  			""",
94  			csv("ok", "?start=2026-10-01T00:00:00Z&end=2026-10-04T00:00:00Z"));
95  	}

97  	@Test void b02_filters() throws Exception {
98  		var week = "?start=2026-10-01T00:00:00Z&end=2026-10-08T00:00:00Z";
99  		assertString("""
100 			day,reason,orders
101 			2026-10-02T00:00:00Z,late,22
102 			2026-10-03T00:00:00Z,lost,29
103 			2026-10-06T00:00:00Z,late,27
104 			2026-10-07T00:00:00Z,lost,11
105 			""",
106 			csv("ok", week + "&reasons=late&reasons=lost"));
107 		assertString("day,reason,orders\n2026-10-03T00:00:00Z,lost,29\n2026-10-07T00:00:00Z,lost,11\n", csv("ok", week + "&q=OS"));
108 		assertString("day,reason,orders\n2026-10-02T00:00:00Z,late,22\n2026-10-06T00:00:00Z,late,27\n", csv("ok", week + "&lateOnly=true"));
109 		// The search box sends an empty q, which means no filter (a typed String parameter would have answered 400).
110 		assertString(csv("ok", week), csv("ok", week + "&q=&lateOnly=false"));
111 		assertString("day,reason,orders\n", csv("ok", week + "&reasons=late&q=lost"));
112 	}
113 
114 	@Test void b03_truncatedEndsWithTheSentinel() throws Exception {
115 		var lines = csv("truncated", "?start=2026-09-01T00:00:00Z&end=2026-10-11T00:00:00Z").split("\n");
116 		assertEquals(22, lines.length);   // the header, 20 rows, the sentinel
117 		assertString("2026-09-20T00:00:00Z,late,30", lines[20]);
118 		assertString("#truncated,20", lines[21]);
119 	}

123 	}

129 	@Test void b06_everySeriesRequestIsCounted() throws Exception {
130 		var n0 = DashboardRest.SERIES_REQUESTS.get();
131 		csv("ok", "");
132 		client().get(BASE + "/series/error").run();
133 		assertEquals(n0 + 2, DashboardRest.SERIES_REQUESTS.get());
134 		assertContains("\"series\":" + (n0 + 2), client().get(BASE + "/requests").accept("application/json").run()
135 			.assertStatus(200).getContent().asString());
