29  /**
30   * The Custom cards page: a JSON-only {@code kpi} card, a Java-validated {@code inventory-gauge} card, the page script
31   * that registers both, and the two data endpoints they read.
32   */
33  @SuppressWarnings({
34  	"resource" // MockRestClient is a no-op close.
35  })
36  class CustomCardsRest_Test extends TestBase {
37  
38  	private static long count(Species species, PetStatus status) {
39  		return store().getPets().stream().filter(p -> (species == null || p.getSpecies() == species) && p.getStatus() == status).count();
40  	}
41  
42  	private static JsonMap json(String path) throws Exception {
43  		return JsonMap.ofString(client().get(path).accept("application/json").run().assertStatus(200).getContent().asString());
44  	}

50  	@Test void a01_cardsInOrder() throws Exception {
51  		assertPage(page(client(), "/console/dev/cards"))
52  			.isValid()
53  			.hasActiveNav("dev", "cards")
54  			.hasCardOrder("available", "gauge", "bands", "prices", "toolbar", "how");
55  	}

57  	@Test void a02_kpiCardPassesItsBodyThrough() throws Exception {
58  		assertPage(page(client(), "/console/dev/cards"))
59  			.hasCard("available", "kpi")
60  			.hasCardKey("available", "/label", "Pets available")
61  			.hasCardKey("available", "/dataUrl", "/console/dev/cards/stats/available");
62  	}
63  
64  	@Test void a03_gaugeCardCarriesTheShapedFragment() throws Exception {
65  		assertPage(page(client(), "/console/dev/cards"))
66  			.hasCard("gauge", "inventory-gauge")
67  			.hasCardKey("gauge", "/gauge/species", "DOG")
68  			.hasCardKey("gauge", "/gauge/low", 20)
69  			.hasCardKey("gauge", "/gauge/high", 120)
70  			.hasCardKey("gauge", "/gauge/dataUrl", "/console/dev/cards/inventory?species=DOG");
71  	}
72  
73  	@Test void a04_pageLoadsItsScript() throws Exception {
74  		assertContains("/console/dev/cards/petstore-cards.js", page(client(), "/console/dev/cards"));
75  	}
76  
77  	@Test void a05_scriptIsServedAndQueuesBothTypes() throws Exception {
78  		client().get("/console/dev/cards/petstore-cards.js").run()
79  			.assertStatus(200)
80  			.assertHeader("Content-Type").isContains("javascript")
81  			.assertContent().isContains("window.JuneauConsoleCards", "'kpi'", "'inventory-gauge'", "juneau:card-mounted",
82  				"window.JuneauConsoleCharts", "'petstore.stockTotals'", "registerDirective('species'");
83  	}
84  
85  	@Test void a05b_toolbarHasTheShowCatsButton() throws Exception {
86  		assertContains("data-cards-action=\"cat\"", page(client(), "/console/dev/cards"));
87  	}
88  
89  	@Test void a06_bandsIsAStackedBarWithAnAppDecoratorAndClickTopic() throws Exception {
90  		assertPage(page(client(), "/console/dev/cards"))
91  			.hasCard("bands", "chart")
92  			.hasCardSrc("bands", "/console/dev/cards/inventory/all")
93  			.hasCardKey("bands", "/kind", "bar")
94  			.hasCardKey("bands", "/x/field", "species")
95  			.hasCardKey("bands", "/stacked", true)
96  			.hasCardKey("bands", "/orientation", "horizontal")
97  			.hasCardKey("bands", "/series/0/field", "available")
98  			.hasCardKey("bands", "/series/2/field", "sold")
99  			.hasCardKey("bands", "/decorators/0/use", "petstore.stockTotals")
100 			.hasCardKey("bands", "/on/click", "app.band-picked")
101 			.publishes("bands", "app.band-picked")
102 			.hasNoWiringErrors();
103 	}

112 			.publishes("prices", "zoom:prices");

142 	@Test void b06_inventoryForEverySpecies() throws Exception {
143 		var rows = new JsonList(client().get("/console/dev/cards/inventory/all").accept("application/json").run().assertStatus(200).getContent().asString());
144 		assertSize(Species.values().length, rows);
145 		for (var i = 0; i < rows.size(); i++) {
146 			var s = Species.values()[i];
147 			assertBean(rows.getMap(i), "species,available,pending,sold",
148 				s.name() + "," + count(s, PetStatus.AVAILABLE) + "," + count(s, PetStatus.PENDING) + "," + count(s, PetStatus.SOLD));
149 		}
150 	}
151 
152 	@Test void b07_pricesOnePerPetById() throws Exception {
153 		var rows = new JsonList(client().get("/console/dev/cards/prices").accept("application/json").run().assertStatus(200).getContent().asString());
154 		assertSize(store().getPets().size(), rows);
155 		var first = store().getPets().stream().mapToLong(Pet::getId).min().orElseThrow();
156 		assertEquals(first, rows.getMap(0).getLong("id"));
157 		assertEquals((double)store().getPet(first).getPrice(), rows.getMap(0).get("price", Double.class), 0.001);
158 	}
