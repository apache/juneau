17  package org.apache.juneau.petstore.console.dev;
18  
19  import static org.apache.juneau.test.bct.BctAssertions.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import org.apache.juneau.*;
23  import org.apache.juneau.rest.server.console.*;
24  import org.junit.jupiter.api.*;
25  
26  /**
27   * The server half of the inventory-gauge custom card: body validation and fragment shaping.
28   */
29  class InventoryGaugeCardType_Test extends TestBase {
30  
31  	private static final InventoryGaugeCardType H = new InventoryGaugeCardType();
32  
33  	private static CardSource gauge(String body) {
34  		return CardSource.create(InventoryGaugeCardType.TYPE, "gauge").body(body).build();
35  	}
36  
37  	private static String rejected(String body) {
38  		var source = gauge(body);
39  		return assertThrows(IllegalArgumentException.class, () -> H.toFragment(source)).getMessage();
40  	}
41  
42  	//-----------------------------------------------------------------------------------------------------------------
43  	// a - shape
44  	//-----------------------------------------------------------------------------------------------------------------
45  
46  	@Test void a01_type() {
47  		assertString("inventory-gauge", H.type());
48  	}
49  
50  	@Test void a02_fragment() {
51  		assertBean(H.toFragment(gauge("{species:'DOG',low:20,high:120}")), "gauge{species,low,high,dataUrl}",
52  			"{DOG,20,120,/console/dev/cards/inventory?species=DOG}");
53  	}
54  
55  	@Test void a03_speciesIsCaseInsensitive() {
56  		assertBean(H.toFragment(gauge("{species:'cat',low:1,high:2}")), "gauge{species,dataUrl}", "{CAT,/console/dev/cards/inventory?species=CAT}");
57  	}
58  
59  	@Test void a04_lowAndHighDefault() {
60  		assertBean(H.toFragment(gauge("{species:'FISH'}")), "gauge{low,high}", "{0,100}");
61  	}
62  
63  	@Test void a05_fragmentIsOnlyTheGaugeKey() {
64  		assertList(H.toFragment(gauge("{species:'DOG',low:20,high:120}")).keySet(), "gauge");
65  	}
66  
67  	//-----------------------------------------------------------------------------------------------------------------
68  	// b - validation
69  	//-----------------------------------------------------------------------------------------------------------------
70  
71  	@Test void b01_missingSpecies() {
72  		assertString("<@card id='gauge'> type='inventory-gauge' requires 'species'.", rejected("{low:1,high:2}"));
73  	}
74  
75  	@Test void b02_unknownSpecies() {
76  		assertString("<@card id='gauge'> unknown species 'DRAGON'; choose one of: 'BIRD, CAT, DOG, FISH, MOUSE, RABBIT, SNAKE'.",
77  			rejected("{species:'DRAGON',low:1,high:2}"));
78  	}
79  
80  	@Test void b03_lowMustBeBelowHigh() {
81  		assertString("<@card id='gauge'> low '50' must be below high '50'.", rejected("{species:'DOG',low:50,high:50}"));
82  	}
83  
84  	@Test void b04_negativeLow() {
85  		assertString("<@card id='gauge'> low '-1' must not be negative.", rejected("{species:'DOG',low:-1,high:5}"));
86  	}
87  
88  	@Test void b05_markupBodyIsRejected() {
89  		assertContains("type='inventory-gauge' requires a JSON5 object body", rejected("<p>DOG</p>"));
90  	}
91  
92  	@Test void b06_nonIntegerLow() {
93  		assertString("<@card id='gauge'> low 'abc' must be a whole number.", rejected("{species:'DOG',low:'abc',high:5}"));
94  	}
95  
96  	@Test void b07_nonIntegerHigh() {
97  		assertString("<@card id='gauge'> high '1.5' must be a whole number.", rejected("{species:'DOG',low:1,high:1.5}"));
98  	}
99  }
