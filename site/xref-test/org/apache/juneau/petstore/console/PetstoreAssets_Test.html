20  
21  import org.apache.juneau.*;
22  import org.apache.juneau.petstore.console.browser.*;
23  import org.apache.juneau.rest.mock.classic.*;
24  import org.apache.juneau.rest.server.console.test.*;

27  /**
28   * Every stylesheet and script a console page links resolves on the host the runners and the browser tests mount.
29   *
30   * <p>
31   * The console chrome links its assets at context-root URLs ({@code /juneau-console/chrome.css}, the theme pack, the
32   * shell script), so they only load if the host mounts something at those paths.  MockRest page tests assert the
33   * hrefs; this test fetches them with {@link AssetResolutionAssert}.
34   */
35  @SuppressWarnings({
36  	"resource" // MockRestClient is a no-op close.
37  })
38  class PetstoreAssets_Test extends TestBase {
39  
40  	private static final MockRestClient C = MockRestClient.create(PetstoreTestServer.BrowserHost.class).noTrace().ignoreErrors().build();
41  
42  	private static String html(String path) throws Exception {
43  		return C.get(path).accept("text/html").run().assertStatus(200).getContent().asString();
44  	}
45  
46  	private static int status(String url) {
47  		try {
48  			return C.get(url).run().getStatusCode();
49  		} catch (Exception e) {
50  			throw new IllegalStateException(e);
51  		}
52  	}
53  
54  	//-----------------------------------------------------------------------------------------------------------------
55  	// a - every linked asset answers 200
56  	//-----------------------------------------------------------------------------------------------------------------
57  
58  	@Test void a01_everyPageAssetResolves() throws Exception {
59  		for (var path : PetstorePages_ContractTest.PAGE_PATHS)
60  			AssetResolutionAssert.assertAssetsResolve(path, html(path), PetstoreAssets_Test::status);
61  	}
62  
63  	@Test void a02_theChromeLinksItsMountedAssets() throws Exception {
64  		assertContainsAll(String.join("\n", AssetResolutionAssert.assetUrls(html("/console/store"))), "/juneau-console/chrome.css", "/juneau-console/juneau-console.js");
65  	}
66  
67  	@Test void a03_everyPageLoadsTheVersionedPetstoreScript() throws Exception {
68  		var urls = String.join("\n", AssetResolutionAssert.assetUrls(html("/console/store")));
69  		assertContains("/console/petstore.js?v=", urls);
70  		C.get("/console/petstore.js").run().assertStatus(200).assertHeader("Content-Type").asString().isContains("javascript");
71  	}
72  
73  	//-----------------------------------------------------------------------------------------------------------------
74  	// b - the pages' own links and the DataTables dependencies
75  	//-----------------------------------------------------------------------------------------------------------------
76  
77  	@Test void b01_theReactFlavorEmbedsAnAppThatAnswers() throws Exception {
78  		var html = html("/console/dev/flavors/react");
79  		assertContains("src=\"/petstore-ui\"", html);
80  		org.junit.jupiter.api.Assertions.assertEquals(200, status("/petstore-ui"));
81  	}
82  
83  	private static final String JQUERY = "/webjars/jquery/3.7.1/jquery.min.js";
84  	private static final String DATATABLES = "/webjars/datatables.net/2.3.8/js/dataTables.min.js";
85  
86  	@Test void b02_everyDatatablesPageLoadsJqueryAndDataTablesFromTheWebJarsBeforeTheToolkit() throws Exception {
87  		for (var path : java.util.List.of("/console/ops/jobs", "/console/ops/audit")) {
88  			var html = html(path);
89  			var jquery = html.indexOf(JQUERY);
90  			var dataTables = html.indexOf(DATATABLES);
91  			var glue = html.indexOf("juneau-datatables.js");
92  			var toolkit = html.indexOf("juneau-views.js");
93  			org.junit.jupiter.api.Assertions.assertTrue(jquery > 0 && dataTables > jquery && glue > dataTables && toolkit > glue,
94  				() -> path + ": jquery=" + jquery + ", dataTables=" + dataTables + ", glue=" + glue + ", views=" + toolkit);
95  			assertContains("dataTables.dataTables.min.css", html);
96  			org.junit.jupiter.api.Assertions.assertEquals(html.indexOf(JQUERY), html.lastIndexOf(JQUERY), () -> path + ": jQuery loads once");
97  			assertNotContains(() -> path + " must not load a library from a CDN", "cdn.", html);
98  			assertNotContains(() -> path + " still loads the old vendor copies", "/console/vendor/", html);
99  		}
100 		assertNotContains(() -> "a page without a table loads no jQuery", "jquery", html("/console/about"));
101 	}
102 
103 	@Test void b03_theWebJarsAreServed() throws Exception {
104 		for (var url : java.util.List.of("/console/ops/audit" + JQUERY, "/console/ops/audit" + DATATABLES))
105 			C.get(url).run().assertStatus(200).assertHeader("Content-Type").asString().isContains("javascript");
106 	}
107 
108 	@Test void b04_aPathOutsideTheWebJarsDoesNotResolve() {
109 		org.junit.jupiter.api.Assertions.assertEquals(404, status("/console/ops/audit/webjars/../x"));
110 		org.junit.jupiter.api.Assertions.assertEquals(404, status("/console/ops/audit/webjars/no-such/file.js"));
111 		org.junit.jupiter.api.Assertions.assertEquals(404, status("/console/vendor/jquery/3.7.1/jquery.min.js"));
112 	}
113 }
