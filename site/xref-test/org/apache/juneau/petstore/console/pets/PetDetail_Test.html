165 	@Test void c01_inlineEditEnvelopeStages() throws Exception {
166 		// The 'edit' renderer commits {fields:{<column>:value}} through the row action's endpoint.
167 		var c = client();
168 		var id = create(c, "Inline");
169 		var r = new JsonMap(c.post("/console/pets/" + id + "/stage", "{\"action\":\"reprice\",\"targetId\":\"" + id + "\",\"fields\":{\"price\":\"44.5\"}}")
170 			.contentType("application/json").accept("application/json").run().assertStatus(200).getContent().asString());
171 		assertBean(r, "outcome,row{petId,field,newValue,state,author}", "success,{" + id + ",price,44.5,PENDING,anonymous}");
172 		assertBean(json(c, "/petstore/pets/" + id), "name,price", "Inline,3.0");
173 	}

193 	@Test void c04_summaryEditColumnsUseTheStageActions() throws Exception {
194 		var t = assertPage(page(client(), "/console/pets/1/summary")).card("pet").getMap("table");
195 		assertBean(t, "dataMode,rowActions{#{id,endpoint,method}}", "client,{[{rename,/console/pets/{id}/stage,POST},{reprice,/console/pets/{id}/stage,POST},{sell,/console/pets/{id}/sell,POST}]}");
196 		// present=inline: the pencil edit cell commits them and the row menu omits them (and carry no form).
197 		// A null form is omitted from the wire, which BCT reports as a missing property rather than null.
198 		assertFalse(t.getList("rowActions").getMap(0).containsKey("form"));
199 		assertBean(t.getList("rowActions").getMap(0), "present", "inline");
200 		// A null form is omitted from the wire, which BCT reports as a missing property rather than null.
201 		assertFalse(t.getList("rowActions").getMap(1).containsKey("form"));
202 		assertBean(t.getList("rowActions").getMap(1), "present", "inline");
203 		var name = t.getList("columns").getMap(1);
204 		assertBean(name, "key,render{id,meta{action,column}}", "name,{edit,{rename,name}}");
205 	}

235 	private static final String API_TOKEN = "0123456789abcdef".repeat(8);
236 
237 	private static String adminPhotoPage(String suffix, boolean withToken) throws Exception {
238 		var r = client().get("/console/pets/1/photo" + suffix).accept("text/html").header("Authorization", "Bearer petstore-admin");
239 		if (withToken)
240 			r = r.attribute(PetstoreApiBoundary.TOKEN_ATTRIBUTE, API_TOKEN);
241 		return r.run().assertStatus(200).getContent().asString();
242 	}

245 	@Test void d07_photoUploadFormOmitsTheTokenWithoutTheBoundary() throws Exception {
246 		for (var suffix : List.of("", "?builder")) {

249 			assertNotContains(() -> "api token attribute without the boundary" + suffix, "data-petstore-csrf", html);
