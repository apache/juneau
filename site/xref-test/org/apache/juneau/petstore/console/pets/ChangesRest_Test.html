34  	/** A lax client that asks for JSON (the console resources also render HTML for a browser Accept). */
35  	private static MockRestClient client() {
36  		return MockRestClient.create(PetstoreConsoleFixture.Host.class).noTrace().ignoreErrors().header("Accept", "application/json").build();
37  	}

39  	private static JsonMap json(String s) throws Exception {
40  		return new JsonMap(s);
41  	}

43  	private static long newPet(MockRestClient c, String name) throws Exception {
44  		return json(c.post("/console/pets", "{\"name\":\"" + name + "\",\"price\":3}").contentType("application/json").run().assertStatus(200).getContent().asString()).getMap("row").getLong("id");
45  	}

47  	private static long stage(MockRestClient c, long petId, String field, String value) throws Exception {
48  		return json(c.post("/console/pets/" + petId + "/stage", "{\"field\":\"" + field + "\",\"value\":\"" + value + "\"}")
49  			.contentType("application/json").run().assertStatus(200).getContent().asString()).getMap("row").getLong("id");
50  	}

52  	@Test void a01_pendingBadgeSource() throws Exception {
53  		var c = client();
54  		var id = stage(c, 2, "name", "Badgepet");
55  		var r = json(c.get("/console/pets/changes/pending").run().assertStatus(200).getContent().asString());
56  		assertTrue(r.getInt("total") >= 1);
57  		assertContains("\"id\":\"" + id + "\"", r.toString());
58  		assertContains("/console/pets/2/summary", r.toString());
59  	}

61  	@Test void a02_pendingFlagsMineByAuthor() throws Exception {
62  		var c = client();
63  		var petId = newPet(c, "Authorpet");
64  		var id = json(c.post("/console/pets/" + petId + "/stage", "{\"field\":\"name\",\"value\":\"ByAdmin\"}")
65  			.contentType("application/json").header("Authorization", "Bearer petstore-admin").run().assertStatus(200).getContent().asString()).getMap("row").getLong("id");
66  		var anon = mineOf(json(c.get("/console/pets/changes/pending").run().assertStatus(200).getContent().asString()), id);
67  		var admin = mineOf(json(c.get("/console/pets/changes/pending").header("Authorization", "Bearer petstore-admin").run().assertStatus(200).getContent().asString()), id);
68  		assertFalse(anon);
69  		assertTrue(admin);
70  	}
71  
72  	private static boolean mineOf(JsonMap r, long id) {
73  		for (var m : r.getList("items").elements(JsonMap.class))
74  			if (String.valueOf(id).equals(m.getString("id")))
75  				return m.getBoolean("mine");
76  		throw new AssertionError("no pending change '" + id + "'");
77  	}

94  	@Test void b03_applyAfterPetDeletedFailsThenRetry404Stays() throws Exception {
95  		var c = client();
96  		var petId = json(c.post("/console/pets", "{\"name\":\"Doomed\",\"price\":1}").contentType("application/json").run().getContent().asString()).getMap("row").getLong("id");
97  		var id = stage(c, petId, "name", "Ghost");
98  		c.delete("/console/pets/" + petId).run().assertStatus(200);
99  		c.post("/console/pets/changes/" + id + "/apply", "").run().assertStatus(200).assertContent().isContains("FAILED", "Unknown pet '" + petId + "'");
100 		c.post("/console/pets/changes/" + id + "/retry", "").run().assertStatus(200).assertContent().isContains("FAILED");
101 	}

108 		var c = client();
109 		var a = stage(c, newPet(c, "Discard1"), "name", "D1");
110 		var b = stage(c, newPet(c, "Discard2"), "name", "D2");
111 		c.post("/console/pets/changes/" + b + "/apply", "").run().assertStatus(200);
112 		var r = json(c.post("/console/pets/changes/discard", "{\"ids\":[\"" + a + "\",\"" + b + "\",\"999999\"],\"idempotencyKey\":\"k-c01\"}")
113 			.contentType("application/json").run().assertStatus(200).getContent().asString());
114 		assertBean(r, "succeeded,notFound,failed{#{id,message}}", "[" + a + "],[999999],{[{" + b + ",Change '" + b + "' is not pending}]}");
