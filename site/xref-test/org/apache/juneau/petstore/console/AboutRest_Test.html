20  import static org.apache.juneau.rest.server.console.test.PageContractAssert.*;
21  import static org.apache.juneau.test.bct.BctAssertions.*;
22  
23  import org.apache.juneau.*;
24  import org.apache.juneau.petstore.dto.*;
25  import org.apache.juneau.rest.mock.classic.*;
26  import org.junit.jupiter.api.*;
27  
28  /**
29   * The About page and the utility-bean demos that used to be a standalone resource.
30   */
31  @SuppressWarnings({
32  	"resource" // MockRestClient is a no-op close.
33  })
34  class AboutRest_Test extends TestBase {
35  
36  	private static MockRestClient json() {
37  		return MockRestClient.create(PetstoreConsoleFixture.Host.class).noTrace().ignoreErrors().header("Accept", "application/json").build();
38  	}
39  
40  	//-----------------------------------------------------------------------------------------------------------------
41  	// a - page
42  	//-----------------------------------------------------------------------------------------------------------------
43  
44  	@Test void a01_pageRendersProseCardsAndLinks() throws Exception {
45  		var html = page(client(), "/console/about");
46  		assertPage(html).isValid().hasActiveNav("about").hasCardOrder("overview", "links", "beans");
47  		assertContainsAll(html, "jc-prose", "href=\"/petstore/api\"", "href=\"/petstore-ui\"",
48  			"href=\"https://juneau.apache.org/docs/topics/JuneauPetstore\"",
49  			"href=\"/console/about/beans/BeanDescription\"", "href=\"/console/about/beans/Hyperlink\"", "href=\"/console/about/beans/SeeOtherRoot\"");
50  	}
51  
52  	//-----------------------------------------------------------------------------------------------------------------
53  	// b - utility beans
54  	//-----------------------------------------------------------------------------------------------------------------
55  
56  	@Test void b01_beansListsAllThree() throws Exception {
57  		assertContainsAll(json().get("/console/about/beans").run().assertStatus(200).getContent().asString(),
58  			"BeanDescription", "Hyperlink", "SeeOtherRoot");
59  		assertContainsAll(client().get("/console/about/beans").accept("text/html").run().assertStatus(200).getContent().asString(),
60  			"/console/about/beans/BeanDescription", "/console/about/beans/Hyperlink", "/console/about/beans/SeeOtherRoot");
61  	}
62  
63  	@Test void b02_beanDescriptionDescribesPet() throws Exception {
64  		assertContainsAll(json().get("/console/about/beans/BeanDescription").run().assertStatus(200).getContent().asString(),
65  			Pet.class.getName(), "species", "photo");
66  	}
67  
68  	@Test void b03_hyperlinkPointsBackToAbout() throws Exception {
69  		assertContains("/console/about", json().get("/console/about/beans/Hyperlink").run().assertStatus(200).getContent().asString());
70  	}
71  
72  	@Test void b04_seeOtherRootRedirects() throws Exception {
73  		rawClient().get("/console/about/beans/SeeOtherRoot").run().assertStatus(303);
74  	}
75  
76  	@Test void b05_apiLinkIsTheSwaggerPage() throws Exception {
77  		client().get("/petstore/api").accept("text/html").run().assertStatus(200);
78  	}
79  }
