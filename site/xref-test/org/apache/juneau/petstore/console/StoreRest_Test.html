55  	/** The live pet counts by the given key, fetched from the API; keys with no pets are 0. */
56  	private static Map<String,Integer> live(MockRestClient c, String key) throws Exception {
57  		var m = new TreeMap<String,Integer>();
58  		for (var p : list(c, "/petstore/pets").elements(JsonMap.class))
59  			if (p.getString(key) != null) // other tests create pets without a species; the stats rows cannot count them
60  				m.merge(p.getString(key), 1, Integer::sum);
61  		return m;
62  	}
63  
64  	/** The stats rows as key to count, with zero-count keys dropped to match {@link #live}. */
65  	private static Map<String,Integer> counts(JsonList rows, String key) {
66  		var m = new TreeMap<String,Integer>();
67  		for (var r : rows.elements(JsonMap.class))
68  			if (r.getInt("count") > 0)
69  				m.put(r.getString(key), r.getInt("count"));
70  		return m;
71  	}

151 	/**
152 	 * Asserts that row 0 of each chart card's {@code src} has every field the card reads.
153 	 *
154 	 * @return The chart card ids checked, in page order.
155 	 */
156 	static List<String> assertChartFieldsInRows(MockRestClient c, JsonList cards) throws Exception {
157 		var checked = new ArrayList<String>();
158 		for (var o : cards.elements(JsonMap.class)) {
159 			if (! "chart".equals(o.getString("type")))
160 				continue;
161 			var id = o.getString("id");
162 			var fields = new ArrayList<String>();
163 			if (o.containsKey("x"))
164 				fields.add(o.getMap("x").getString("field"));
165 			if (o.containsKey("series"))
166 				for (var x : o.getList("series").elements(JsonMap.class))
167 					fields.add(x.getString("field"));
168 			for (var k : List.of("label", "value"))
169 				if (o.containsKey(k))
170 					fields.add(o.getString(k));
171 			if (o.containsKey("helper") && "treemap".equals(o.getMap("helper").getString("kind"))) {
172 				var h = o.getMap("helper");
173 				fields.add(h.getString("label"));
174 				for (var g : h.getList("groupBy"))
175 					fields.add(String.valueOf(g));
176 				for (var g : h.getList("tileFields"))
177 					fields.add(String.valueOf(g));
178 				if (h.containsKey("color"))
179 					fields.add(h.getMap("color").getString("field"));
180 			}
181 			assertFalse(fields.isEmpty(), () -> id + " names no fields");
182 			var row = list(c, o.getString("src")).getMap(0);
183 			for (var f : fields)
184 				assertTrue(row.containsKey(f), () -> id + " reads '" + f + "' but " + o.getString("src") + " rows have " + row.keySet());
185 			checked.add(id);
186 		}
187 		return checked;
188 	}

281 		var c = json();
282 		var html = page(client(), "/console/store");
283 		var sold = StreamSupport.stream(list(c, "/petstore/pets").elements(JsonMap.class).spliterator(), false).filter(x -> "SOLD".equals(x.getString("status"))).count();

286 		var total = pets.size();
287 		var available = withStatus(pets, "AVAILABLE");
288 		var orders = list(c, "/petstore/orders");
289 		var placed = withStatus(orders, "PLACED");
290 		var approved = withStatus(orders, "APPROVED");
291 		var delivered = withStatus(orders, "DELIVERED");
292 		assertTrue(total > 0);
293 		for (var h : List.of(html, page(client(), "/console/store?builder"))) // the FTL page and its builder twin
294 			assertContainsAll(h, "\"id\":\"orders\",\"type\":\"segmented\"",
295 				"\"label\":\"Placed\",\"count\":" + placed + "}",
296 				"\"label\":\"Approved\",\"count\":" + approved + ",\"tone\":\"info\"",
297 				"\"label\":\"Delivered\",\"count\":" + delivered + ",\"tone\":\"success\"",
298 				"\"id\":\"inventory\",\"type\":\"meter\"",
299 				"\"value\":" + available + ",\"max\":" + total,
300 				"\"text\":\"" + available + " of " + total + "\"");
301 		for (var pg : List.of("/console/store", "/console/store?builder")) {
302 			var segs = (List<?>)((Map<?,?>)stripItems(pg).get(3)).get("segments");
303 			var got = new ArrayList<String>();

309 		}
310 		var rows = list(c, "/console/store/recent-orders");
311 		assertSize(10, rows);
312 		assertTrue(rows.getMap(0).getString("shipDate").compareTo(rows.getMap(9).getString("shipDate")) >= 0);
313 	}

322 	@Test void b06_revenueToday() throws Exception {
323 		// Deliver one priced pet today so a tile stuck at zero cannot pass.
324 		var store = store();
325 		var pet = store.getPets().stream().filter(x -> x.getPrice() > 0).findFirst().orElseThrow();
326 		var today = LocalDate.ofInstant(store.clock().instant(), ZoneOffset.UTC).toString();
327 		var pinned = store.createOrder(new org.apache.juneau.petstore.dto.Order().setPetId(pet.getId()).setStatus(OrderStatus.DELIVERED).setShipDate(today), "test");
328 		try {
329 			var c = json();
330 			var daily = list(c, "/console/store/stats/daily");
331 			assertEquals(today, daily.getMap(daily.size() - 1).getString("day"));
332 			var prices = new HashMap<Long,Double>();
333 			for (var p : list(c, "/petstore/pets").elements(JsonMap.class))
334 				prices.put(p.getLong("id"), p.get("price", Double.class));
335 			var expected = 0d;
336 			var delivered = 0;
337 			for (var o : list(c, "/petstore/orders").elements(JsonMap.class)) {
338 				if ("DELIVERED".equals(o.getString("status")) && today.equals(o.getString("shipDate"))) {
339 					expected += prices.getOrDefault(o.getLong("petId"), 0d);
340 					delivered++;
341 				}
342 			}
343 			assertTrue(expected >= pet.getPrice() - 0.001, () -> "pinned order missing from the expected total");
344 			assertTrue(expected > 0);
345 			assertEquals(daily.getMap(daily.size() - 1).getInt("delivered"), delivered);
346 			var kpi = new JsonMap(c.get("/console/store/kpi/revenue").run().assertStatus(200).getContent().asString());
347 			assertBean(kpi, "label,unit", "Revenue today,USD");
348 			assertEquals(Math.round(expected * 100) / 100d, kpi.get("value", Double.class), 0.001);
349 		} finally {
350 			store.deleteOrder(pinned.getId(), "test");
351 		}
