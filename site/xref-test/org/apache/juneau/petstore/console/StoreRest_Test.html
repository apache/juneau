60  	/** The live pet counts by the given key, fetched from the API; keys with no pets are 0. */
61  	private static Map<String,Integer> live(MockRestClient c, String key) throws Exception {
62  		var m = new TreeMap<String,Integer>();
63  		for (var p : list(c, "/petstore/pets").elements(JsonMap.class))
64  			if (p.getString(key) != null) // other tests create pets without a species; the stats rows cannot count them
65  				m.merge(p.getString(key), 1, Integer::sum);
66  		return m;
67  	}

69  	/** The stats rows as key to count, with zero-count keys dropped to match {@link #live}. */
70  	private static Map<String,Integer> counts(JsonList rows, String key) {
71  		var m = new TreeMap<String,Integer>();
72  		for (var r : rows.elements(JsonMap.class))
73  			if (r.getInt("count") > 0)
74  				m.put(r.getString(key), r.getInt("count"));
75  		return m;
76  	}

190 	/**
191 	 * Asserts that row 0 of each chart card's {@code src} has every field the card reads.
192 	 *
193 	 * @return The chart card ids checked, in page order.
194 	 */
195 	static List<String> assertChartFieldsInRows(MockRestClient c, JsonList cards) throws Exception {
196 		var checked = new ArrayList<String>();
197 		var flat = new ArrayList<JsonMap>();
198 		for (var top : cards.elements(JsonMap.class)) {
199 			if (! "switcher".equals(top.getString("type"))) {
200 				flat.add(top);
201 				continue;
202 			}
203 			for (var v : top.getList("views").elements(JsonMap.class)) {
204 				if (v.containsKey("card"))
205 					flat.add(v.getMap("card"));
206 				else
207 					for (var sec : v.getList("sections").elements(JsonMap.class))
208 						flat.add(sec.getMap("card"));
209 			}
210 		}
211 		for (var o : flat) {
212 			if (! "chart".equals(o.getString("type")))
213 				continue;
214 			var id = o.getString("id");
215 			var fields = new ArrayList<String>();
216 			if (o.containsKey("x"))
217 				fields.add(o.getMap("x").getString("field"));
218 			if (o.containsKey("series"))
219 				for (var x : o.getList("series").elements(JsonMap.class))
220 					fields.add(x.getString("field"));
221 			for (var k : List.of("label", "value"))
222 				if (o.containsKey(k))
223 					fields.add(o.getString(k));
224 			if (o.containsKey("helper") && "treemap".equals(o.getMap("helper").getString("kind"))) {
225 				var h = o.getMap("helper");
226 				fields.add(h.getString("label"));
227 				for (var g : h.getList("groupBy"))
228 					fields.add(String.valueOf(g));
229 				for (var g : h.getList("tileFields"))
230 					fields.add(String.valueOf(g));
231 				if (h.containsKey("color"))
232 					fields.add(h.getMap("color").getString("field"));
233 			}
234 			assertFalse(fields.isEmpty(), () -> id + " names no fields");
235 			var row = list(c, o.getString("src")).getMap(0);
236 			for (var f : fields)
237 				assertTrue(row.containsKey(f), () -> id + " reads '" + f + "' but " + o.getString("src") + " rows have " + row.keySet());
238 			checked.add(id);
239 		}
240 		return checked;
241 	}

381 		try {

399 		} finally {

401 		}

430 		var today = LocalDate.ofInstant(store.clock().instant(), ZoneOffset.UTC);

432 		var made = new ArrayList<Long>();
433 		made.add(store.createOrder(new org.apache.juneau.petstore.dto.Order().setPetId(pet.getId()).setStatus(OrderStatus.DELIVERED).setShipDate(today.toString()), "test").getId());
