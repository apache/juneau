139 	@Test
140 	void a02_p2Body() throws Exception {
141 		var t = assertPage(page(client, "/console/pets")).card("pets").getMap("table");
142 		assertBean(t, "dataMode,dataUrl,selection{rowIdField,scope}", "server,/console/pets/query,{id,persistent}");
143 		assertBean(t, "bulk{actions{#{id,bulkMode}}}", "{{[{sell,perRow}]}}");
144 		var ribbon = t.getList("ribbon");
145 		var status = find(ribbon, "id", "status");
146 		assertBean(status, "type,default,deselectable", "optionGroup,available,true");
147 		assertBean(status.getList("options").getMap(0), "id,value", "available,$eq(AVAILABLE)");
148 		var species = find(ribbon, "id", "species");
149 		assertBean(species, "default,deselectable", "mammals,true");
150 		assertBean(species.getList("options").getMap(0), "value", "$in(DOG,CAT,MOUSE,RABBIT)");
151 		var price = find(t.getList("columns"), "key", "price");
152 		assertBean(price, "label", "Price (USD)");
153 		var export = find(ribbon, "type", "export");
154 		assertBean(export, "buttons,optional", "[copy,csv],[excel,pdf]");
155 	}
156 
157 	private static JsonMap find(JsonList list, String key, String value) {
158 		for (var m : list.elements(JsonMap.class))
159 			if (value.equals(m.getString(key)))
160 				return m;
161 		throw new AssertionError("no element with " + key + "='" + value + "'");
162 	}
