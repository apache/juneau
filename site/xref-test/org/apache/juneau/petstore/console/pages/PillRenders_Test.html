17  package org.apache.juneau.petstore.console.pages;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.*;
24  import org.apache.juneau.console.views.*;
25  import org.apache.juneau.marshall.marshaller.*;
26  import org.junit.jupiter.api.*;
27  
28  /**
29   * The per-column pill renderers: one definition per enum column, so a value has the same colour on every page.
30   */
31  class PillRenders_Test extends TestBase {
32  
33  	private static String json(Render r) {
34  		return Json.of(r);
35  	}
36  
37  	@Test void a01_petStatus() {
38  		assertEquals("{\"id\":\"pill\",\"meta\":{\"tones\":{\"AVAILABLE\":\"success\",\"PENDING\":\"warning\",\"SOLD\":\"neutral\"}}}", json(PillRenders.petStatus()));
39  	}
40  
41  	@Test void a02_orderStatus() {
42  		assertEquals("{\"id\":\"pill\",\"meta\":{\"tones\":{\"PLACED\":\"info\",\"APPROVED\":\"warning\",\"DELIVERED\":\"success\"}}}", json(PillRenders.orderStatus()));
43  	}
44  
45  	@Test void a03_userStatus() {
46  		assertEquals("{\"id\":\"pill\",\"meta\":{\"tones\":{\"ACTIVE\":\"success\",\"INACTIVE\":\"neutral\"}}}", json(PillRenders.userStatus()));
47  	}
48  
49  	@Test void a04_changeState() {
50  		assertEquals("{\"id\":\"pill\",\"meta\":{\"tones\":{\"PENDING\":\"warning\",\"APPLIED\":\"success\",\"FAILED\":\"error\"}}}", json(PillRenders.changeState()));
51  	}
52  
53  	@Test void a05_jobState() {
54  		assertEquals("{\"id\":\"pill\",\"meta\":{\"tones\":{\"RUNNING\":\"info\",\"DONE\":\"success\",\"EXPIRED\":\"neutral\"}}}", json(PillRenders.jobState()));
55  	}
56  
57  	@Test void a06_auditAction() {
58  		assertEquals("{\"id\":\"pill\",\"meta\":{\"tones\":{\"CREATE\":\"success\",\"UPDATE\":\"info\",\"DELETE\":\"error\",\"DISCARD\":\"warning\",\"RESTOCK\":\"info\"}}}", json(PillRenders.auditAction()));
59  	}
60  
61  	@Test void a07_species_pinsAreDistinctAndOffPaletteFree() {
62  		var meta = (Map<?,?>)Json.to(json(PillRenders.species()), Map.class).get("meta");
63  		assertEquals(true, meta.get("categorical"));
64  		var cats = (Map<?,?>)meta.get("categories");
65  		assertEquals(7, cats.size());
66  		assertEquals(7, new HashSet<>(cats.values()).size(), "pinned species slots must be collision-free");
67  		assertFalse(cats.containsValue(3) || cats.containsValue(5), "slots 3 and 5 are reserved from petstore pins");
68  	}
69  
70  	@Test void a08_entity_pinsAreDistinct() {
71  		var meta = (Map<?,?>)Json.to(json(PillRenders.entity()), Map.class).get("meta");
72  		var cats = (Map<?,?>)meta.get("categories");
73  		assertEquals(5, cats.size());
74  		assertEquals(5, new HashSet<>(cats.values()).size());
75  		assertFalse(cats.containsValue(3) || cats.containsValue(5));
76  	}
77  }
