41   * The console's CSRF boundary on a real server: console pages carry the token, console writes without it (or from
42   * another origin) are refused before they reach the resource, and the {@code /petstore} API stays outside this

67  	private static HttpResponse<String> send(String method, String path, String origin, String csrf, String contentType, String body) throws Exception {
68  		var b = HttpRequest.newBuilder(URI.create(base + path)).timeout(Duration.ofSeconds(20)).header("Accept", "application/json");
69  		if (origin != null)
70  			b.header("Origin", origin);
71  		if (csrf != null)
72  			b.header("X-Csrf-Token", csrf);
73  		if (contentType != null)
74  			b.header("Content-Type", contentType);
75  		b.method(method, body == null ? BodyPublishers.noBody() : BodyPublishers.ofString(body));
76  		return HTTP.send(b.build(), BodyHandlers.ofString());
77  	}

79  	private static HttpResponse<String> json(String method, String path, String origin, String csrf) throws Exception {
80  		return send(method, path, origin, csrf, "application/json", "{}");
81  	}

91  	private static long petWithStatus(String status) throws Exception {
92  		for (var o : apiList("/petstore/pets"))
93  			if (status.equals(((JsonMap)o).getString("status")))
94  				return ((JsonMap)o).getLong("id");
95  		throw new IllegalStateException("No seeded pet with status " + status);
96  	}
97  
98  	private static long orderWithStatus(String status) throws Exception {
99  		for (var o : apiList("/petstore/orders"))
100 			if (status.equals(((JsonMap)o).getString("status")))
101 				return ((JsonMap)o).getLong("id");
102 		throw new IllegalStateException("No seeded order with status " + status);
103 	}

105 	private static void assertRefused(HttpResponse<String> r, int status, String reason) {
106 		assertEquals(status, r.statusCode(), r.body());
107 		assertString(reason, r.headers().firstValue("X-Request-Boundary").orElse(null));
108 	}

255 	private static HttpResponse<String> get(String path, String origin, String csrf, String headerName, String headerValue) throws Exception {
256 		var b = HttpRequest.newBuilder(URI.create(base + path)).timeout(Duration.ofSeconds(20)).header("Accept", "application/json");
257 		if (origin != null)
258 			b.header("Origin", origin);
259 		if (csrf != null)
260 			b.header("X-Csrf-Token", csrf);
261 		if (headerName != null)
262 			b.header(headerName, headerValue);
263 		return HTTP.send(b.GET().build(), BodyHandlers.ofString());
264 	}

305 		assertString("POST", org.apache.juneau.rest.server.MethodSafety.effectiveMethod("POST", "method=GET", h -> null));
306 		assertString("GET", org.apache.juneau.rest.server.MethodSafety.effectiveMethod("GET", null, h -> null));
307 		assertString("GET", org.apache.juneau.rest.server.MethodSafety.effectiveMethod("GET", "method=&methods=DELETE", h -> " "));
308 		assertString("DELETE", org.apache.juneau.rest.server.MethodSafety.effectiveMethod("GET", "method=options&method=delete", h -> null));
309 		assertString("PUT", org.apache.juneau.rest.server.MethodSafety.effectiveMethod("GET", "method=get", h -> h.equals("X-HTTP-Method") ? "put" : null));
310 		assertString("%ZZ", org.apache.juneau.rest.server.MethodSafety.effectiveMethod("GET", "method=%ZZ", h -> null));  // undecodable, so unknown, so unsafe
311 		assertString("PATCH", org.apache.juneau.rest.server.MethodSafety.effectiveMethod("HEAD", null, h -> h.equals("X-Method-Override") ? "patch" : null));
312 		assertString("POST", org.apache.juneau.rest.server.MethodSafety.effectiveMethod("post", null, h -> null));  // the shared helper upper-cases the request line too
313 	}
