58  	private static Map<String,Object> m(Object...kv) {
59  		var r = new LinkedHashMap<String,Object>();
60  		for (var i = 0; i < kv.length; i += 2)
61  			r.put((String) kv[i], kv[i + 1]);
62  		return r;
63  	}

65  	private static Map<String,Object> pause(int ms) {

67  	}

298 		var t = assertClean(report, "truncated");
299 		assertString("20 rows · loaded just now · truncated at 20", query(t, "text"));
300 		assertString("warning", query(t, "tone"));
301 		// The failed load is logged by the data source (one console.error per failed generation), so this
302 		// case is not "clean". Every logged failure must be that line; how many generations run before the sections settle
303 		// is the source's debounce, not this test's.
304 		var e = report.get("error");
305 		assertString("200", String.valueOf(e.get("status")));
306 		assertEquals(List.of(), e.get("timeouts"));
307 		assertTrue(((String) query(e, "text")).startsWith("Load failed: "), () -> "error: " + query(e, "text"));
308 		assertString("error", query(e, "tone"));
309 		var failures = (List<?>) e.get("jsFailures");
310 		assertFalse(failures.isEmpty(), "the source logged nothing for a failed load");
311 		for (var f : failures)
312 			assertTrue(String.valueOf(f).startsWith("[juneau-data] E-JS-62: data source 'orders' fetch failed: ")
313 				|| String.valueOf(f).startsWith("[juneau-charts] E-JS-62: chart card 'daily' fetch failed: "), () -> "unexpected failure: " + f);
314 	}
