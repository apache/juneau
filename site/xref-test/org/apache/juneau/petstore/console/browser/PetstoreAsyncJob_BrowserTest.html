31  /**
32   * P6 in the real browser: a double submit with one key returns one job, the job finishes, the polled jobs table
33   * shows it, and an unknown id is 404.  The {@code restock} job-log card streams a restock to its result, re-attaches
34   * after a reload, tells a cancel before the effect from one after it, shows the hard timeout and the job limit, and
35   * the stream refuses a subscriber over its cap.
36   */
37  @Covers(value="async-job", pages="/console/ops/jobs")
38  @EnabledIfSystemProperty(named=PetstoreBrowser.GATE, matches="true", disabledReason=PetstoreBrowser.DISABLED)
39  class PetstoreAsyncJob_BrowserTest extends TestBase {

44  	private static final String SUBMIT_TWICE = "(async () => {"
45  		+ " const key = 'browser-' + Date.now();"
46  		+ " const post = () => fetch('/console/ops/jobs/restock', {method: 'POST', credentials: 'same-origin',"
47  		+ "   headers: {'Content-Type': 'application/json', 'Accept': 'application/json', 'X-Csrf-Token': document.body.dataset.juneauCsrf},"
48  		+ "   body: JSON.stringify({perSpecies: {DOG: 1}, idempotencyKey: key})});"
49  		+ " const a = await post(), b = await post();"
50  		+ " const ja = await a.json(), jb = await b.json();"
51  		+ " window.__jobId = ja.jobId;"
52  		+ " let state = 'RUNNING';"
53  		+ " for (let i = 0; i < 100 && state === 'RUNNING'; i++) {"
54  		+ "   await new Promise(r => setTimeout(r, 100));"
55  		+ "   state = (await (await fetch('/console/ops/jobs/' + ja.jobId, {headers: {'Accept': 'application/json'}})).json()).state;"
56  		+ " }"
57  		+ " return {a: a.status, b: b.status, same: ja.jobId === jb.jobId, state};"
58  		+ "})()";
59  
60  	private static Map<String,Map<String,Object>> report;
61  
62  	// The job-log card's root, whichever element carries the card id.
63  	private static final String J = ":is(#restock, #restock-body) .jc-joblog";
64  	private static final String CSRF = "{'Content-Type': 'application/json', 'Accept': 'application/json', 'X-Csrf-Token': document.body.dataset.juneauCsrf}";
65  
66  	/** JS: starts a restock by fetch and resolves to its job ref. */
67  	private static String start(int slowMs) {
68  		return "fetch('/console/ops/jobs/restock', {method: 'POST', credentials: 'same-origin', headers: " + CSRF + ","
69  			+ " body: JSON.stringify({DOG: 1, slowMs: " + slowMs + ", idempotencyKey: 'browser-' + Math.random()})}).then(r => r.json())";
70  	}
71  
72  	/** JS: cancels the job refs in {@code window.__refs}, with the same request the job-log card sends. */
73  	private static final String CANCEL_REFS = "Promise.all((window.__refs || []).map(r => fetch(r.cancelUrl, {method: 'POST', credentials: 'same-origin', headers: " + CSRF + ", body: '{}'})))";
74  
75  	/** JS: resolves to the number of pets in the store. */
76  	private static final String PET_COUNT = "fetch('/petstore/pets', {headers: {Accept: 'application/json'}}).then(r => r.json()).then(p => p.length)";
77  
78  	// Holds the stream's two subscriber slots open on a slow job, then asks for a third.
79  	private static final String OVER_THE_CAP = "(async () => {"
80  		+ " const ref = await " + start(2500) + ";"
81  		+ " window.__refs = [ref];"
82  		+ " const open = () => new Promise((res, rej) => { const es = new EventSource(ref.streamUrl);"
83  		+ "   es.onopen = () => res(es); setTimeout(() => { es.close(); rej(new Error('stream did not open')); }, 8000); });"
84  		+ " const a = await open(), b = await open();"
85  		+ " const ctl = new AbortController();"
86  		+ " const third = await fetch(ref.streamUrl, {headers: {Accept: 'text/event-stream'}, signal: ctl.signal});"
87  		+ " const out = {open: [a.readyState, b.readyState], third: third.status};"
88  		+ " ctl.abort(); a.close(); b.close();"
89  		+ " await " + CANCEL_REFS + ";"
90  		+ " return out; })()";
91  
92  	private static Map<String,Object> wait(String js, int ms) {
93  		return Map.of("waitFor", js, "timeoutMs", ms);
94  	}
95  
96  	private static Map<String,Object> phase(String p, int ms) {
97  		return wait("document.querySelector('" + J + "[data-phase=" + p + "]')", ms);
98  	}
99  
100 	private static Map<String,Object> slowMs(int ms) {
101 		return Map.of("fill", List.of(J + " .jc-joblog-input[name=slowMs]", String.valueOf(ms)));
102 	}
103 
104 	private static Map<String,Object> holdMs(int ms) {
105 		return Map.of("fill", List.of(J + " .jc-joblog-input[name=holdMs]", String.valueOf(ms)));
106 	}

108 	private static final Map<String,Object> SUBMIT = Map.of("click", J + " .jc-joblog-submit");
109 
110 	/** JS: some output line contains {@code text}. */
111 	private static Map<String,Object> line(String text, int ms) {
112 		return wait("[...document.querySelectorAll('" + J + " .jc-joblog-line')].some(l => l.textContent.includes('" + text + "'))", ms);
113 	}

115 	private static final Map<String,Object> CARD_QUERIES = Map.of(
116 		"lines", "[...document.querySelectorAll('" + J + " .jc-joblog-line')].map(l => l.textContent)",
117 		"banner", "document.querySelector('" + J + " .jc-joblog-banner')?.textContent ?? null",
118 		"search", "location.search");

120 	@BeforeAll
121 	static void probe() throws Exception {
122 		var base = PetstoreTestServer.awaitReady(server);
123 		report = run("async-job", base, List.of(
124 			Map.of("name", "jobs", "path", "/console/ops/jobs", "allowFailedLoads", true,
125 				"actions", List.of(
126 					Map.of("waitFor", "document.querySelector('#jobs-body table')"),
127 					Map.of("evaluate", "window.__submit = " + SUBMIT_TWICE),
128 					Map.of("waitFor", "window.__jobId && document.querySelector('#jobs-body').textContent.includes(window.__jobId)", "timeoutMs", 15000)),
129 				"queries", Map.of("submit", "window.__submit", "jobId", "window.__jobId",
130 					"row", "[...document.querySelectorAll('#jobs-body tbody tr')].find(tr => tr.textContent.includes(window.__jobId))?.textContent ?? null"),
131 				"fetches", Map.of("unknown", "/console/ops/jobs/no-such-job")),
132 			// The unknown stream's 404 is expected.
133 			Map.of("name", "card-submit", "path", "/console/ops/jobs", "allowFailedLoads", true,
134 				"actions", List.of(phase("idle", 8000), SUBMIT, phase("done", 15000)),
135 				"queries", CARD_QUERIES,
136 				"fetches", Map.of("unknownStream", "/console/ops/jobs/juneau-jobs/no-such-job/stream")),
137 			Map.of("name", "reattach", "path", "/console/ops/jobs",
138 				"actions", List.of(phase("idle", 8000), slowMs(1500), SUBMIT,
139 					wait("location.search.includes('job=')", 8000), line("Waiting", 8000),
140 					Map.of("reload", true),
141 					phase("done", 15000)),
142 				"queries", CARD_QUERIES),
143 			Map.of("name", "cancel-early", "path", "/console/ops/jobs",
144 				"actions", List.of(phase("idle", 8000),
145 					Map.of("evaluate", PET_COUNT + ".then(n => { window.__before = n; })"), wait("window.__before != null", 8000),
146 					slowMs(2000), Map.of("evaluate", "window.__posted = Date.now()"), SUBMIT, line("Waiting", 8000),
147 					wait("!document.querySelector('" + J + " .jc-joblog-submit').disabled", 8000), SUBMIT,
148 					phase("done", 10000),
149 					// A worker that ignored the cancel would restock once its wait ran out, so count only after that.
150 					wait("Date.now() - window.__posted > 2500", 8000),
151 					Map.of("evaluate", PET_COUNT + ".then(n => { window.__after = n; })"), wait("window.__after != null", 8000)),
152 				"queries", Map.of("lines", CARD_QUERIES.get("lines"), "banner", CARD_QUERIES.get("banner"),
153 					"pets", "[window.__before, window.__after]")),
154 			Map.of("name", "cancel-late", "path", "/console/ops/jobs",
155 				// The 2.5 s hold keeps the job cancellable past the Added line and ends 2.5 s short of the 5 s hard
156 				// timeout, so the cancel always lands in the hold.
157 				"actions", List.of(phase("idle", 8000), holdMs(2500), SUBMIT, line("Added", 8000), SUBMIT, phase("done", 10000)),
158 				"queries", CARD_QUERIES),
159 			// The third subscriber's 429 is expected.
160 			Map.of("name", "subscriber-cap", "path", "/console/ops/jobs", "allowFailedLoads", true,
161 				"actions", List.of(phase("idle", 8000), Map.of("evaluate", "window.__cap = " + OVER_THE_CAP)),
162 				"queries", Map.of("cap", "window.__cap")),
163 			// The job would run 8 s; the 5 s timeout lands in its hold.  This passes even against a scheduler that fires
164 			// its timeout task a little late; the early- and on-the-deadline firing it once hung on is pinned
165 			// deterministically by AsyncJobRegistry_Test's k group.
166 			Map.of("name", "timeout", "path", "/console/ops/jobs",
167 				"actions", List.of(phase("idle", 8000), slowMs(3000), holdMs(5000), SUBMIT, phase("done", 12000)),
168 				"queries", CARD_QUERIES),
169 			// The two fetched jobs fill the limit, so the card's own submit is refused with 429.
170 			Map.of("name", "limit", "path", "/console/ops/jobs", "allowFailedLoads", true,
171 				"actions", List.of(phase("idle", 8000),
172 					Map.of("evaluate", "window.__started = Promise.all([" + start(2500) + ", " + start(2500) + "]).then(r => { window.__refs = r; })"),
173 					wait("window.__refs && window.__refs.length === 2", 8000),
174 					SUBMIT,
175 					wait("document.querySelector('" + J + " .jc-joblog-banner.jc-joblog-error:not([hidden])')", 8000),
176 					Map.of("evaluate", CANCEL_REFS)),
177 				"queries", CARD_QUERIES)));
178 	}
