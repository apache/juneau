35  /**
36   * URL state in the real browser, through the Custom cards page's own {@code species} directive: {@code ?state=species(...)}
37   * picks the gauge's species, survives a reload, and back/forward restores it.
38   *
39   * <p>
40   * It also covers the pets table's {@code ids} directive: {@code ?state=ids(a,b)} sends {@code ids=a,b} with the table's
41   * server requests and shows only those pets (still under the ribbon's default filters), survives a reload, and follows
42   * history.  The built-in {@code day} and {@code window} directives are on the Orders page ({@link PetstoreOrdersDay_BrowserTest}),
43   * so with {@code species} here every URL-state directive the petstore configures is covered, which retires the {@code url-state}
44   * pending row in {@link PetstoreCoverage_Test}.
45   */
46  @Covers(value="url-state", pages={"/console/pets", "/console/orders", "/console/dev/cards"})
47  @EnabledIfSystemProperty(named=PetstoreBrowser.GATE, matches="true", disabledReason=PetstoreBrowser.DISABLED)
48  class PetstoreUrlState_BrowserTest extends TestBase {
49  
50  	@RegisterExtension
51  	static MicroserviceTestFixture server = PetstoreTestServer.fixture();
52  
53  	private static final String SPECIES = "document.querySelector('#gauge-body meter').getAttribute('data-gauge-species')";
54  
55  	private static String showing(String species) {
56  		return "!!document.querySelector('#gauge-body meter[data-gauge-species=\"" + species + "\"]')";
57  	}
58  
59  	private static final String PETS_QUERY = "/console/pets/query";
60  	private static final String PET_IDS = "[...document.querySelectorAll('#pets tbody tr[data-juneau-row-id]')].map(tr => tr.getAttribute('data-juneau-row-id')).sort((a, b) => a - b).join(',')";
61  	private static final String PET_COUNT = "document.querySelectorAll('#pets tbody tr[data-juneau-row-id]').length";
62  
63  	private static Map<String,Map<String,Object>> report;
64  
65  	/** Three pets the pets page's default ribbon filters (available mammals) show: a and b together, and c alone. */
66  	private static long petA, petB, petC;
67  
68  	/** A sold pet, which the ribbon's available filter hides. */
69  	private static long petSold;
70  
71  	private static String pets(long...ids) {
72  		return Arrays.stream(ids).sorted().mapToObj(String::valueOf).collect(java.util.stream.Collectors.joining(","));
73  	}
74  
75  	private static Map<String,Object> petsShowing(String ids) {
76  		return Map.of("waitFor", "(() => { try { return " + PET_IDS + " === '" + ids + "'; } catch (e) { return false; } })()", "timeoutMs", 15000);
77  	}
78  
79  	private static void findPets(String base) throws Exception {
80  		var body = "{\"draw\":1,\"start\":0,\"length\":3,\"order\":[{\"column\":0,\"dir\":\"asc\"}],\"columns\":["
81  			+ "{\"data\":\"id\"},{\"data\":\"status\",\"search\":{\"value\":\"$eq(AVAILABLE)\"}},"
82  			+ "{\"data\":\"species\",\"search\":{\"value\":\"$in(DOG,CAT,MOUSE,RABBIT)\"}}]}";
83  		var res = HttpClient.newHttpClient().send(PetstoreTestServer.consolePost(base, PETS_QUERY, body), BodyHandlers.ofString());
84  		assertEquals(200, res.statusCode(), res::body);
85  		var rows = new JsonMap(res.body()).getList("data");
86  		assertEquals(3, rows.size());
87  		petA = rows.getMap(0).getLong("id");
88  		petB = rows.getMap(1).getLong("id");
89  		petC = rows.getMap(2).getLong("id");
90  		var soldBody = "{\"draw\":1,\"start\":0,\"length\":1,\"order\":[{\"column\":0,\"dir\":\"asc\"}],\"columns\":[{\"data\":\"id\"},{\"data\":\"status\",\"search\":{\"value\":\"$eq(SOLD)\"}}]}";
91  		var sold = HttpClient.newHttpClient().send(PetstoreTestServer.consolePost(base, PETS_QUERY, soldBody), BodyHandlers.ofString());
92  		assertEquals(200, sold.statusCode(), sold::body);
93  		petSold = new JsonMap(sold.body()).getList("data").getMap(0).getLong("id");
94  	}

221 	@SuppressWarnings("unchecked") // requests is a JSON array of {url, method, postData}.
222 	private static List<String> idParams(Map<String,Object> c) {
223 		return ((List<Map<String,Object>>) c.get("requests")).stream().map(r -> URLDecoder.decode(String.valueOf(r.get("url")), java.nio.charset.StandardCharsets.UTF_8)).toList();
224 	}

226 	@Test void b01_linkShowsOnlyThoseRows() {
227 		var c = assertClean(report, "ids-link");
228 		assertEquals(true, query(c, "registered"));
229 		assertString(pets(petA, petB), query(c, "ids"));
230 		assertTrue(String.valueOf(query(c, "search")).contains("ids(" + petA + "," + petB + ")"), () -> String.valueOf(query(c, "search")));
231 	}
232 
233 	@Test void b02_theRequestCarriesTheIds() {
234 		var c = assertClean(report, "ids-link");
235 		assertTrue(idParams(c).stream().anyMatch(u -> u.contains("ids=" + petA + "," + petB)), () -> idParams(c).toString());
236 	}
237 
238 	@Test void b03_noDirectiveMeansNoFilter() {
239 		var c = assertClean(report, "ids-none");
240 		assertTrue(((Number) query(c, "count")).intValue() > 2);
241 		assertTrue(idParams(c).stream().noneMatch(u -> u.contains("ids=")), () -> idParams(c).toString());
242 	}

244 	@Test void b04_pushBackForwardFollowTheAddress() {
245 		var c = assertClean(report, "ids-history");
246 		assertTrue(String.valueOf(query(c, "afterPush")).contains("ids(" + petC + ")"), () -> String.valueOf(query(c, "afterPush")));
247 		assertTrue(String.valueOf(query(c, "afterBack")).contains("ids(" + petA + "," + petB + ")"), () -> String.valueOf(query(c, "afterBack")));
248 		assertTrue(String.valueOf(query(c, "search")).contains("ids(" + petC + ")"), () -> String.valueOf(query(c, "search")));
249 		assertString(pets(petC), query(c, "ids"));
250 	}
