31  /**
32   * The Custom cards page in the real browser: both custom types render, the card API works, each card mounts once,
33   * and a second registration of a card type fails loudly.
34   *
35   * <p>
36   * The {@code ctx.every} pause while the page is hidden is not asserted here: the gauge polls every 30 s and the shell
37   * has no test hook for a shorter period, so a short wait could not tell a paused timer from a running one.  The
38   * console UI's own tests cover the pause.
39   */
40  @EnabledIfSystemProperty(named=PetstoreBrowser.GATE, matches="true", disabledReason=PetstoreBrowser.DISABLED)
41  class PetstoreCustomCard_BrowserTest extends TestBase {
42  
43  	@RegisterExtension
44  	static MicroserviceTestFixture server = PetstoreTestServer.fixture();
45  
46  	private static Map<String,Map<String,Object>> report;
47  
48  	private static final String RENDERED = "document.querySelector('#gauge-body meter') && document.querySelector('#available-body .petstore-kpi-value')";
49  
50  	// Stubs fetch so the gauge reads window.__stock and the kpi's read aborts, spies on announcements, and puts a
51  	// decoy element carrying the gauge body's id ahead of the card in the document.
52  	private static final String STUB = String.join("\n",
53  		"window.__ann = [];",
54  		"window.JuneauViews = window.JuneauViews || {};",
55  		"window.JuneauViews.announce = function (t) { window.__ann.push(t); };",
56  		"window.__stock = 50;",
57  		"var real = window.fetch;",
58  		"window.fetch = function (url, opts) {",
59  		"  var u = String(url);",
60  		"  if (u.indexOf('/inventory?species=') >= 0)",
61  		"    return Promise.resolve(new Response(JSON.stringify({species:'DOG', available:window.__stock, pending:0, sold:0}),",
62  		"      {status:200, headers:{'Content-Type':'application/json'}}));",
63  		"  if (u.indexOf('/stats/available') >= 0) return Promise.reject(new DOMException('aborted', 'AbortError'));",
64  		"  return real.apply(this, arguments);",
65  		"};",
66  		"var d = document.createElement('div');",
67  		"d.id = 'gauge-body';",
68  		"d.setAttribute('data-decoy', '');",
69  		"document.body.insertBefore(d, document.body.firstChild);");
70  
71  	// Stock 50, then 5 three times, back to 50, then 5 again: two crossings below low (20).
72  	private static final String READS = String.join("\n",
73  		"var api = window.JuneauConsole.cardApi('gauge');",
74  		"var at = function (n) { return function () { window.__stock = n; return api.reload(); }; };",
75  		"Promise.resolve().then(at(50)).then(at(5)).then(at(5)).then(at(5)).then(at(50)).then(at(5))",
76  		"  .then(function () { window.__gaugeDone = true; });",
77  		"Promise.resolve(window.JuneauConsole.refreshCard('available'))",
78  		"  .then(function () { window.__kpiDone = true; }, function () { window.__kpiDone = true; });");
79  
80  	@BeforeAll
81  	static void probe() throws Exception {
82  		var base = PetstoreTestServer.awaitReady(server);
83  		report = run("custom-card", base, List.of(
84  			Map.of("name", "cards", "path", "/console/dev/cards",
85  				"actions", List.of(
86  					Map.of("waitFor", RENDERED),
87  					Map.of("evaluate", "window.__mountedBefore = window.petstoreMounted.slice()"),
88  					Map.of("click", "[data-cards-action=\"refresh\"]"),
89  					Map.of("waitFor", "window.JuneauConsole.cardApi('gauge').reads() >= 2"),
90  					Map.of("click", "[data-cards-action=\"api\"]")),
91  				"queries", Map.of(
92  					"kpi", "document.querySelector('#available-body .petstore-kpi-value').textContent",
93  					"kpiLabel", "document.querySelector('#available-body .petstore-kpi-label').textContent",
94  					"gaugeSpecies", "document.querySelector('#gauge-body meter').getAttribute('data-gauge-species')",
95  					"gaugeText", "document.querySelector('#gauge-body .petstore-gauge-text').textContent",
96  					"apiText", "document.getElementById('cards-api-result').textContent",
97  					"mountedBefore", "window.__mountedBefore",
98  					"types", "window.JuneauConsole.cardTypes()")),
99  			Map.of("name", "dup", "path", "/console/dev/cards", "allowFailedLoads", true,
100 				"actions", List.of(
101 					Map.of("waitFor", RENDERED),
102 					Map.of("evaluate", "try { window.JuneauConsole.registerCard('kpi', {render(){}}); window.__dup = 'none'; } catch (e) { window.__dup = e.code + ' ' + e.message; }")),
103 				"queries", Map.of("dup", "window.__dup")),
104 			Map.of("name", "stubbed", "path", "/console/dev/cards", "allowFailedLoads", true,
105 				"actions", List.of(
106 					Map.of("waitFor", RENDERED),
107 					Map.of("evaluate", STUB),
108 					Map.of("evaluate", READS),
109 					Map.of("waitFor", "window.__gaugeDone === true && window.__kpiDone === true")),
110 				"queries", Map.of(
111 					"lowStock", "window.__ann.filter(function (t) { return t.indexOf('DOG stock is below') === 0; }).length",
112 					"decoyChildren", "document.querySelector('[data-decoy]').childElementCount",
113 					"gaugeText", "document.querySelector('#gauge .petstore-gauge-text').textContent",
114 					"kpiErrors", "document.querySelectorAll('#available .jc-card-error').length"))));
115 	}

121 	@Test void a01_kpiRendersTheServerCount() {
122 		var c = assertClean(report, "cards");
123 		var kpi = String.valueOf(query(c, "kpi"));
124 		assertTrue(kpi.matches("\\d+"), () -> "kpi value is not a count: " + kpi);
125 		assertString("Pets available", query(c, "kpiLabel"));
126 	}
127 
128 	@Test void a02_gaugeRendersTheJavaShapedFragment() {
129 		var c = assertClean(report, "cards");
130 		assertString("DOG", query(c, "gaugeSpecies"));
131 		assertMatchesGlob("* DOG available (*%)", query(c, "gaugeText"));
132 	}
133

138 	@Test void b01_refreshCardAndCardApi() {
139 		assertMatchesGlob("DOG, * reads", query(assertClean(report, "cards"), "apiText"));
140 	}
141 
142 	@Test void b02_eachCardMountsOnce() {
143 		@SuppressWarnings("unchecked") // the query returns a JSON array.
144 		var m = (List<String>) query(assertClean(report, "cards"), "mountedBefore");
145 		assertList(m.stream().filter(x -> x.startsWith("gauge:")).toList(), "gauge:inventory-gauge");
146 		assertList(m.stream().filter(x -> x.startsWith("available:")).toList(), "available:kpi");
147 	}
148 
149 	@Test void b03_bothTypesRegistered() {
150 		assertContainsAll(String.valueOf(query(assertClean(report, "cards"), "types")), "kpi", "inventory-gauge");
151 	}
152 
153 	@Test void b04_handlerDrawsInsideItsOwnCard() {
154 		var c = report.get("stubbed");
155 		assertEquals(0, ((Number)query(c, "decoyChildren")).intValue(), "the gauge drew into a same-id element outside its card");
156 		assertString("5 DOG available (0%)", query(c, "gaugeText"));
157 	}
158 
159 	@Test void b05_lowStockIsAnnouncedOncePerCrossing() {
160 		assertEquals(2, ((Number)query(report.get("stubbed"), "lowStock")).intValue());
161 	}
162 
163 	@Test void b06_abortedKpiReadPaintsNoError() {
164 		assertEquals(0, ((Number)query(report.get("stubbed"), "kpiErrors")).intValue());
165 	}
