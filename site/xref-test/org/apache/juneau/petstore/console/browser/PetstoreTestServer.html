39  /**
40   * The browser tests' server: the {@code /console} tree, the {@code /petstore} API and the {@code /petstore-ui}
41   * React app over one seeded store, in a real Jetty bound to an ephemeral port.
42   *
43   * <h5 class='section'>Example:</h5>
44   * <p class='bjava'>
45   * 	<ja>@RegisterExtension</ja>
46   * 	<jk>static</jk> MicroserviceTestFixture <jv>server</jv> = PetstoreTestServer.<jsm>fixture</jsm>();
47   *
48   * 	<ja>@BeforeAll</ja>
49   * 	<jk>static void</jk> boot() <jk>throws</jk> Exception {
50   * 		PetstoreTestServer.<jsm>awaitReady</jsm>(<jv>server</jv>);
51   * 	}
52   * </p>
53   */
54  @SuppressWarnings({
55  	"java:S2925", // awaitReady() needs a back-off between retries; there is no readiness event to await.

61  	@Rest(children={PetstoreConsoleResource.class, ConsoleAssetsRest.class, PetStoreResource.class, PetstoreUiResource.class})
62  	public static class BrowserHost extends BasicRestServletGroup {
63  		private static final long serialVersionUID = 1L;

65  		/** @return The seeded store shared by every child. */
66  		@Bean public PetStore petStore() {

68  		}

70  		/** @return A 150 ms groom step, so a full groom run takes about 8 s. */
71  		@Bean public JobsRest.GroomPeriod groomPeriod() {
72  			return new JobsRest.GroomPeriod(Duration.ofMillis(150));
73  		}

75  		/**
76  		 * @return A 5 s jobs hard timeout, so a browser test can watch a slow restock time out.  A species hold must
77  		 * 	end well before it (see {@link ShortHoldHost}), or the job is reaped as a hard timeout.
78  		 */
79  		@Bean public JobsRest.HardTimeout jobsHardTimeout() {
80  			return new JobsRest.HardTimeout(Duration.ofSeconds(5));

84  	/** {@link BrowserHost} plus a 2 s species-row hold: long enough to see a hold, short enough to wait out. */
85  	public static class ShortHoldHost extends BrowserHost {
86  		private static final long serialVersionUID = 1L;
87  
88  		/** @return The species-row hold. */
89  		@Bean public JobsRest.SpeciesHold speciesHold() {
90  			return new JobsRest.SpeciesHold(Duration.ofSeconds(2));
91  		}

94  	/**
95  	 * {@link BrowserHost} plus a 7 s species-row hold, so a cancel in any scenario lands in it, and a 12 s hard timeout:
96  	 * a hold must end well before the timeout, or the job is reaped as a hard timeout instead of being cancelled after
97  	 * its effect.
98  	 */
99  	public static class CancelHoldHost extends BrowserHost {
100 		private static final long serialVersionUID = 1L;
101 
102 		/** @return The species-row hold. */
103 		@Bean public JobsRest.SpeciesHold speciesHold() {
104 			return new JobsRest.SpeciesHold(Duration.ofSeconds(7));
105 		}
106 
107 		/** @return A 12 s jobs hard timeout, longer than the hold. */
108 		@Override @Bean public JobsRest.HardTimeout jobsHardTimeout() {
109 			return new JobsRest.HardTimeout(Duration.ofSeconds(12));
110 		}
111 	}

113 	/** Contributes the console and API CSRF boundaries the runners install to the fixture's Jetty; a timings class adds the host. */
114 	@Configuration
115 	public static class BrowserConfig {
116 		// Shared by both boundary beans, as in the runners.
117 		private final PetstoreApiBoundary apiBoundary = new PetstoreApiBoundary();
118 
119 		/** @return The console's CSRF boundary, over {@code /console/*}. */
120 		@Bean(name="consoleBoundary") public ServletFilterRegistration consoleBoundary() {
121 			return ServletFilterRegistration.create(new PetstoreConsoleBoundary(apiBoundary))
122 				.urlPatterns(PetstoreConsoleBoundary.URL_PATTERN)
123 				.dispatcherTypes(DispatcherType.REQUEST, DispatcherType.FORWARD, DispatcherType.ASYNC, DispatcherType.ERROR)
124 				.order(ServletFilterRegistration.HIGHEST_PRECEDENCE)
125 				.build();
126 		}
127 
128 		/** @return The API's CSRF boundary, over {@code /petstore/*}. */
129 		@Bean(name="apiBoundary") public ServletFilterRegistration apiBoundary() {
130 			return ServletFilterRegistration.create(apiBoundary)
131 				.urlPatterns(PetstoreApiBoundary.URL_PATTERN)
132 				.dispatcherTypes(DispatcherType.REQUEST)
133 				.order(ServletFilterRegistration.HIGHEST_PRECEDENCE + 10)
134 				.build();
135 		}

184 	/**
185 	 * Waits until {@code /console/store} answers 200, absorbing the first request's one-time REST setup.
186 	 *
187 	 * @param f The started fixture.
188 	 * @return The base URL, without a trailing slash.
189 	 * @throws Exception If the server is not ready within 30 seconds.
190 	 */
191 	public static String awaitReady(MicroserviceTestFixture f) throws Exception {
192 		var base = f.getRootUrl().toString().replaceAll("/$", "");
193 		var http = HttpClient.newBuilder().connectTimeout(Duration.ofSeconds(5)).build();
194 		var deadline = Instant.now().plusSeconds(30);
195 		Exception last = null;
196 		while (Instant.now().isBefore(deadline)) {
197 			try {
198 				var req = HttpRequest.newBuilder(URI.create(base + "/console/store")).timeout(Duration.ofSeconds(20)).header("Accept", "text/html").GET().build();
199 				if (http.send(req, BodyHandlers.discarding()).statusCode() == 200)
200 					return base;
201 			} catch (IOException e) {  // timeout, refused, reset or EOF while Jetty is still starting: retry until the deadline
202 				last = e;
203 			}
204 			Thread.sleep(250);
205 		}
206 		throw new IllegalStateException("Petstore test server did not become ready within 30s at '" + base + "'", last);
207 	}

209 	private static final Pattern API_TOKEN = Pattern.compile("data-petstore-csrf=\"([0-9a-f]{128})\"");
210 
211 	/**
212 	 * Reads the API's CSRF token the way a page script does: from the admin Photo tab's upload form.
213 	 *
214 	 * @param base The base URL from {@link #awaitReady}.
215 	 * @return The token to send in the API's CSRF header.
216 	 * @throws Exception If the page cannot be read or carries no token.
217 	 */
218 	public static String apiToken(String base) throws Exception {
219 		var page = HttpClient.newHttpClient().send(HttpRequest.newBuilder(URI.create(base + "/console/pets/1/photo"))
220 			.header("Accept", "text/html").header("Authorization", "Bearer petstore-admin").GET().build(), BodyHandlers.ofString());
221 		var m = API_TOKEN.matcher(page.body());
222 		if (! m.find())
223 			throw new IllegalStateException("Admin photo page carries no data-petstore-csrf token");
224 		return m.group(1);
225 	}
226 
227 	private static final Pattern TOKEN = Pattern.compile("<body[^>]*\\sdata-juneau-csrf=\"([^\"]+)\"");
228 
229 	/**
230 	 * Builds a console write the way the page sends one: same-origin, JSON, carrying the token from a console page.
231 	 *
232 	 * @param base The base URL from {@link #awaitReady}.
233 	 * @param path The console path, e.g. {@code /console/ops/jobs/groom}.
234 	 * @param json The JSON body.
235 	 * @return The POST request.
236 	 * @throws Exception If the console page cannot be read or carries no token.
237 	 */
238 	public static HttpRequest consolePost(String base, String path, String json) throws Exception {
239 		var page = HttpClient.newHttpClient().send(HttpRequest.newBuilder(URI.create(base + "/console/store")).header("Accept", "text/html").GET().build(), BodyHandlers.ofString());
240 		var m = TOKEN.matcher(page.body());
241 		if (! m.find())
242 			throw new IllegalStateException("Console page carries no data-juneau-csrf token");
243 		return HttpRequest.newBuilder(URI.create(base + path))
244 			.header("Origin", base).header(LoopbackBoundary.DEFAULT_CSRF_HEADER, m.group(1))
245 			.header("Content-Type", "application/json").header("Accept", "application/json")
246 			.POST(HttpRequest.BodyPublishers.ofString(json)).build();
247 	}
