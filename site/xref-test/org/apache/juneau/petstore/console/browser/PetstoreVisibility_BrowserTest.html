35  /**
36   * Visibility rules in the real browser: a pet's Summary offers Mark sold and the pencils only while the pet is not
37   * sold, and they go once the row redraws as sold; the Photo tab's upload card is rendered for an admin and omitted
38   * for everyone else.
39   */
40  @Covers(value="visibility-rules", pages={"/console/pets/1/summary", "/console/pets/1/photo"})
41  @HookCovers("rules")
42  @EnabledIfSystemProperty(named=PetstoreBrowser.GATE, matches="true", disabledReason=PetstoreBrowser.DISABLED)
43  class PetstoreVisibility_BrowserTest extends TestBase {
44  
45  	@RegisterExtension
46  	static MicroserviceTestFixture server = PetstoreTestServer.fixture();
47  
48  	private static final String TRIGGER = "#pet-body .juneau-view-action-trigger";
49  	private static final String MENU_ITEMS = "[...document.querySelectorAll('.juneau-view-action-item')].map(b => b.dataset.actionId)";
50  	private static final String PENCILS = "[...document.querySelectorAll('#pet-body [data-juneau-edit]')].filter(e => !e.hidden).map(e => e.getAttribute('data-juneau-edit'))";
51  	private static final String CONFIRM = ".juneau-view-dialog-confirm";
52  	private static final String UPLOAD = "form[data-petstore-upload]";
53  	// Whether the Photo tab, fetched with the given bearer token, carries the upload form (inside its card <template>).
54  	private static final String UPLOAD_FOR = "(token => fetch(location.pathname, {headers: token ? {Authorization: 'Bearer ' + token} : {}})"
55  		+ ".then(r => r.text()).then(t => t.includes('data-petstore-upload=')))";
56  
57  	private static String base;
58  	private static long pet;
59  	private static Map<String,Map<String,Object>> report;
60  
61  	private static Map<String,Object> wait(String js) {
62  		return Map.of("waitFor", js, "timeoutMs", 8000);
63  	}
64  
65  	@BeforeAll
66  	static void probe() throws Exception {
67  		base = PetstoreTestServer.awaitReady(server);
68  		for (var o : new JsonList(get("/petstore/pets").body()))
69  			if ("AVAILABLE".equals(((JsonMap)o).getString("status"))) {
70  				pet = ((JsonMap)o).getLong("id");
71  				break;
72  			}
73  		report = run("visibility", base, List.of(
74  			Map.of("name", "sell", "path", "/console/pets/" + pet + "/summary",
75  				"actions", List.of(
76  					wait("document.querySelector('" + TRIGGER + "')"),
77  					Map.of("evaluate", "window.__before = {pencils: " + PENCILS + "}"),
78  					Map.of("click", TRIGGER),
79  					wait("document.querySelector('.juneau-view-action-item')"),
80  					Map.of("evaluate", "window.__before.menu = " + MENU_ITEMS),
81  					Map.of("click", ".juneau-view-action-item[data-action-id=sell]"),
82  					wait("document.querySelector('" + CONFIRM + "')"),
83  					Map.of("click", CONFIRM),
84  					wait("document.querySelector('#pet-body tbody')?.textContent.toUpperCase().includes('SOLD')")),
85  				"queries", Map.of("before", "window.__before", "pencils", PENCILS,
86  					"trigger", "(() => { const t = document.querySelector('" + TRIGGER + "'); return !!t && !t.hidden && !t.disabled; })()")),
87  			Map.of("name", "photo", "waitUntil", "networkidle", "path", "/console/pets/1/photo", "allowFailedLoads", true,
88  				"queries", Map.of("viewer", "!!document.querySelector('" + UPLOAD + "')",
89  					"admin", UPLOAD_FOR + "('petstore-admin')", "user", UPLOAD_FOR + "('petstore-user')"))));
90  	}
91  
92  	private static HttpResponse<String> get(String path) throws Exception {
93  		return HttpClient.newHttpClient().send(HttpRequest.newBuilder(URI.create(base + path)).header("Accept", "application/json").GET().build(), BodyHandlers.ofString());
94  	}
95  
96  	//-----------------------------------------------------------------------------------------------------------------
97  	// a - row actions
98  	//-----------------------------------------------------------------------------------------------------------------
99  
100 	@Test void a01_unsoldPetOffersSellAndThePencils() {
101 		assertBean(query(assertClean(report, "sell"), "before"), "menu,pencils", "[sell],[rename,reprice]");
102 	}
103 
104 	@Test void a02_soldPetLosesThem() throws Exception {
105 		var c = assertClean(report, "sell");
106 		assertString("SOLD", new JsonMap(get("/petstore/pets/" + pet).body()).getString("status"));
107 		assertEmpty(query(c, "pencils"));
108 		assertEquals(false, query(c, "trigger"));
109 	}
110 
111 	//-----------------------------------------------------------------------------------------------------------------
112 	// b - cards
113 	//-----------------------------------------------------------------------------------------------------------------
114 
115 	@Test void b01_uploadCardIsAdminOnly() {
116 		assertBean(assertClean(report, "photo").get("queries"), "viewer,user,admin", "false,false,true");
117 	}
118 }
