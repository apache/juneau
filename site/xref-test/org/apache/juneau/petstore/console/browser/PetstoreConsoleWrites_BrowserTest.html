44  /**
45   * Console writes in the real browser, through the server's CSRF boundary: the page's token rides on every write, so
46   * the P4 pencil stages a change, Pending changes applies and discards them, the Pets dialogs add and edit a pet, bulk
47   * Mark sold sells one, and the Photo tab uploads a photo, which the API refuses without its token.  A form on another
48   * origin that posts to the API changes nothing.
49   */

66  		+ " if (m !== 'GET') window.__writes.push({url: String(u), method: m, status: r.status, body: await r.clone().text()});"
67  		+ " return r; }; })()";

114 		return List.of(

119 			eval("(() => { const s = document.querySelector('.juneau-view-bulk-select'); s.value = '" + action + "'; s.dispatchEvent(new Event('change', {bubbles: true})); })()"),
120 			Map.of("click", ".juneau-view-bulk-go"));
121 	}

269 		// A page on another origin posts a urlencoded form at the API as soon as it loads; the browser sends it with the
270 		// attacker's Origin, and the API's boundary must refuse it.
271 		attacker = new Server();
272 		var connector = new ServerConnector(attacker);
273 		connector.setHost("127.0.0.1");
274 		connector.setPort(0);
275 		attacker.addConnector(connector);
276 		connector.open();
277 		var attackCtx = new ServletContextHandler(ServletContextHandler.NO_SESSIONS);
278 		attackCtx.setContextPath("/");
279 		attackCtx.addServlet(new ServletHolder(new AttackServlet(base)), "/attack");
280 		attacker.setHandler(attackCtx);
281 		attacker.start();
282 		attackOrigin = "http://127.0.0.1:" + connector.getLocalPort();
283 		petsBefore = new JsonList(get("/petstore/pets", "application/json").body()).size();
284 		attack = run("console-writes-cross-origin", attackOrigin, List.of(
285 			Map.of("name", "crossOriginForm", "path", "/attack", "allowFailedLoads", true, "waitUntil", "load",
286 				"actions", List.of(
287 					wait("location.origin === '" + base + "'")),
288 				"queries", Map.of("landed", "location.origin", "path", "location.pathname"))));
289 		petsAfter = new JsonList(get("/petstore/pets", "application/json").body()).size();

332 	@SuppressWarnings("unchecked") // writes is a JSON array of {url, method, status, body}.
333 	private static Map<String,Object> theWrite(String name, String method, String part) {
334 		var c = assertClean(report, name);
335 		var w = ((List<Map<String,Object>>) query(c, "writes")).stream()
336 			.filter(x -> method.equals(x.get("method")) && String.valueOf(x.get("url")).contains(part)).toList();
337 		assertSize(1, w);
338 		assertEquals(200, ((Number) w.get(0).get("status")).intValue(), () -> name + ": " + w.get(0));
339 		return w.get(0);
340 	}
