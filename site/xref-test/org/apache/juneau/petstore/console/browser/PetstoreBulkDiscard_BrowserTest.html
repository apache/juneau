36  /**
37   * The aggregate bulk Discard on Pending changes in the real browser: rows that are not pending can't be selected and
38   * say why, a confirmed discard is one POST carrying the ids and an idempotency key, its summary counts the ids that
39   * were gone or no longer pending, the failed ones stay selected, and a refused request leaves the selection intact
40   * behind a sticky alert.
41   *
42   * <p>
43   * The class seeds its own changes before the browser starts: four pending, one applied, and one failed (its pet
44   * was deleted before it was applied).
45   */
46  @Covers(value="bulk-select", pages="/console/pets/changes")
47  @EnabledIfSystemProperty(named=PetstoreBrowser.GATE, matches="true", disabledReason=PetstoreBrowser.DISABLED)
48  class PetstoreBulkDiscard_BrowserTest extends TestBase {

53  	private static final String READY = "document.querySelector('#changes tbody tr[data-juneau-row-id]')";
54  	private static final String HEADER_BOX = "#changes thead input.juneau-view-select-all-checkbox";
55  	private static final String PENDING_ONLY = "#changes-body .juneau-view-ribbon-btn[aria-label=\"Pending only\"]";
56  	private static final String DLG = "dialog.petstore-typed-confirm";
57  	private static final String OK = DLG + " .petstore-typed-confirm-ok";
58  
59  	/** The selected ids, from the table's retained selection topic. */
60  	private static final String IDS = "((window.JuneauViews.bus.get('selection:changes') || {}).ids || [])";
61  
62  	/** Picks Discard in the bulk action chooser. */
63  	private static final String CHOOSE_DISCARD = "(() => { const s = document.querySelector('#changes-body .juneau-view-bulk-select');"
64  		+ " s.value = 'discard'; s.dispatchEvent(new Event('change')); })()";
65  
66  	/** Types the selection count into the typed confirm, which is what it asks for when several rows are selected. */
67  	private static final String TYPE_COUNT = "(() => { const i = document.querySelector('" + DLG + " input');"
68  		+ " i.value = String(" + IDS + ".length); i.dispatchEvent(new Event('input')); })()";
69  
70  	/**
71  	 * Behind the page's back, discards the first selected change and applies the second, so the bulk run finds one
72  	 * gone and one no longer pending.  The setup discard carries its own key, so it can be told apart.
73  	 */
74  	private static final String RACE = "(async () => { const ids = " + IDS + "; window.__gone = ids[0]; window.__applied = ids[1];"
75  		+ " const h = {'Content-Type': 'application/json', 'Accept': 'application/json', 'X-Csrf-Token': document.body.dataset.juneauCsrf};"
76  		+ " await fetch('/console/pets/changes/discard', {method: 'POST', headers: h, body: JSON.stringify({ids: [ids[0]], idempotencyKey: 'setup'})});"
77  		+ " await fetch('/console/pets/changes/' + ids[1] + '/apply', {method: 'POST', headers: h});"
78  		+ " window.__raced = true; })()";
79  
80  	/** Sends every discard to the demo failure toggle, so the server refuses it with a 503. */
81  	private static final String FAIL_DISCARDS = "const f = window.fetch; window.fetch = (u, o) =>"
82  		+ " f(String(u).endsWith('/discard') ? u + '?fail=discard' : u, o)";
83  
84  	private static final String TOAST = ".juneau-view-bulk-toast";
85  
86  	/** What the page logs and shows when the server refuses the whole discard. */
87  	private static final String REFUSED = "bulk action 'discard' on table 'changes' failed: 'HTTP 503'";

93  	static void probe() throws Exception {
94  		var base = PetstoreTestServer.awaitReady(server);
95  		seed(base);
96  		var confirm = List.<Map<String,Object>>of(
97  			Map.of("waitFor", READY),
98  			Map.of("click", HEADER_BOX),
99  			Map.of("evaluate", "window.__selected = " + IDS + ".slice()"));
100 		var submit = List.<Map<String,Object>>of(
101 			Map.of("evaluate", CHOOSE_DISCARD),
102 			Map.of("click", "#changes-body .juneau-view-bulk-go"),
103 			Map.of("waitFor", "document.querySelector('" + DLG + "')"),
104 			Map.of("evaluate", TYPE_COUNT),
105 			Map.of("click", OK));
106 		report = run("bulk-discard", base, List.of(
107 			Map.of("name", "disabled", "path", "/console/pets/changes",
108 				"actions", List.of(
109 					Map.of("waitFor", READY),
110 					Map.of("click", PENDING_ONLY),
111 					Map.of("waitFor", "document.querySelector('#changes tbody input.juneau-view-select-checkbox:disabled')")),
112 				"queries", Map.of(
113 					"reasons", "Array.from(document.querySelectorAll('#changes tbody input.juneau-view-select-checkbox:disabled')).map(b => b.title)",
114 					"states", "Array.from(document.querySelectorAll('#changes tbody input.juneau-view-select-checkbox:disabled'))"
115 						+ ".map(b => b.closest('tr').textContent.includes('APPLIED') ? 'APPLIED' : b.closest('tr').textContent.includes('FAILED') ? 'FAILED' : '?')")),
116 			// Runs before 'discard', which changes the store; this one is refused and changes nothing.
117 			Map.of("name", "discard-fail", "waitUntil", "networkidle", "path", "/console/pets/changes", "allowFailedLoads", true,
118 				"actions", concat(List.of(Map.of("evaluate", FAIL_DISCARDS)), confirm, submit,
119 					List.of(Map.of("waitFor", "document.querySelector('" + TOAST + "[role=alert]')"))),
120 				"queries", Map.of(
121 					"selected", "window.__selected",
122 					"alert", "document.querySelector('" + TOAST + "[role=alert]').textContent",
123 					"left", IDS)),
124 			Map.of("name", "discard", "path", "/console/pets/changes", "capture", "/console/pets/changes/discard",
125 				"actions", concat(confirm,
126 					List.of(Map.of("evaluate", RACE), Map.of("waitFor", "window.__raced === true")),
127 					submit,
128 					List.of(Map.of("waitFor", "document.querySelector('" + TOAST + "')"))),
129 				"queries", Map.of(
130 					"selected", "window.__selected",
131 					"gone", "window.__gone",
132 					"applied", "window.__applied",
133 					"toast", "document.querySelector('" + TOAST + "').textContent",
134 					"role", "document.querySelector('" + TOAST + "').getAttribute('role')",
135 					"left", IDS))));
136 	}

138 	@SafeVarargs
139 	private static List<Map<String,Object>> concat(List<Map<String,Object>>...parts) {
140 		var l = new ArrayList<Map<String,Object>>();
141 		for (var p : parts)
142 			l.addAll(p);
143 		return l;
144 	}

146 	/** Stages the changes the cases work on, through the same console endpoints the pages use. */
147 	private static void seed(String base) throws Exception {
148 		var http = HttpClient.newHttpClient();
149 		pending = new ArrayList<>();
150 		for (var i = 1; i <= 4; i++)
151 			pending.add(String.valueOf(stage(http, base, newPet(http, base, "Discardable " + i), "Renamed " + i)));
152 		var applied = stage(http, base, newPet(http, base, "Applied"), "Applied twice");
153 		assertEquals("APPLIED", post(http, base, "/console/pets/changes/" + applied + "/apply", "{}").getMap("row").getString("state"));
154 		var doomed = newPet(http, base, "Doomed");
155 		var failed = stage(http, base, doomed, "Never written");
156 		var del = http.send(HttpRequest.newBuilder(URI.create(base + "/petstore/pets/" + doomed)).DELETE().build(), BodyHandlers.ofString());
157 		assertEquals(200, del.statusCode(), del::body);
158 		assertEquals("FAILED", post(http, base, "/console/pets/changes/" + failed + "/apply", "{}").getMap("row").getString("state"));
159 	}

161 	private static long newPet(HttpClient http, String base, String name) throws Exception {
162 		return post(http, base, "/console/pets", "{\"name\":\"" + name + "\",\"price\":3}").getMap("row").getLong("id");
163 	}
164 
165 	private static long stage(HttpClient http, String base, long petId, String name) throws Exception {
166 		return post(http, base, "/console/pets/" + petId + "/stage", "{\"field\":\"name\",\"value\":\"" + name + "\"}").getMap("row").getLong("id");
167 	}
168 
169 	private static JsonMap post(HttpClient http, String base, String path, String json) throws Exception {
170 		var res = http.send(PetstoreTestServer.consolePost(base, path, json), BodyHandlers.ofString());
171 		assertEquals(200, res.statusCode(), res::body);
172 		return new JsonMap(res.body());
173 	}

182 	@Test void a01_rowsThatAreNotPendingCannotBeSelectedAndSayWhy() {
183 		var c = assertClean(report, "disabled");
184 		assertList(new TreeSet<>(ids(c, "states")), "APPLIED", "FAILED");
185 		for (var r : ids(c, "reasons"))
186 			assertString("Only pending changes can be discarded or applied.", r);
187 	}
188 
189 	@Test void a02_selectAllTakesOnlyThePendingRows() {
190 		var c = assertClean(report, "discard");
191 		assertEquals(new TreeSet<>(pending), new TreeSet<>(ids(c, "selected")));
192 	}
193 
194 	@SuppressWarnings({
195 		"unchecked" // requests is a JSON array of {url, method, postData}.
196 	})
197 	@Test void b01_discardIsOnePostWithTheIdsAndAKey() throws Exception {
198 		var c = assertClean(report, "discard");
199 		var runs = ((List<Map<String,Object>>) c.get("requests")).stream()
200 			.filter(r -> "POST".equals(r.get("method")) && ! String.valueOf(r.get("postData")).contains("\"setup\"")).toList();
201 		assertSize(1, runs);
202 		var body = Json.to((String) runs.get(0).get("postData"), Map.class);
203 		assertEquals(new TreeSet<>(pending), new TreeSet<>((List<String>) body.get("ids")));
204 		assertNotEmpty(body.get("idempotencyKey"));
205 	}

207 	@Test void b02_summaryCountsTheGoneAndTheNoLongerPending() {
208 		var c = assertClean(report, "discard");
209 		assertString("Discard: 2 succeeded, 1 not found, 1 failed (ids: " + query(c, "applied") + ")Close", query(c, "toast"));
210 		assertString("alert", query(c, "role"));
211 	}

213 	@Test void b03_onlyTheFailedIdStaysSelected() {
214 		var c = assertClean(report, "discard");
215 		assertList(ids(c, "left"), String.valueOf(query(c, "applied")));
216 	}
217 
218 	@Test void b04_aRefusedDiscardKeepsTheSelectionBehindAStickyAlert() {
219 		var c = report.get("discard-fail");
220 		assertNotNull(c, () -> "No report for case 'discard-fail'; cases: " + report.keySet());
221 		// The whole-run failure is logged to the console on purpose; nothing else may be.
222 		assertList(c.get("jsFailures"), REFUSED);
223 		assertEquals(List.of(), c.get("timeouts"), () -> "timed out waiting for: " + c.get("timeouts"));
224 		assertString(REFUSED + "Close", query(c, "alert"));
225 		assertEquals(new TreeSet<>(pending), new TreeSet<>(ids(c, "selected")));
226 		assertEquals(new TreeSet<>(pending), new TreeSet<>(ids(c, "left")));
227 	}
