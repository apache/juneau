32  /**
33   * The Users and Pets row Delete actions declare a {@code confirm} but no {@code present(DIALOG)}; in the real browser
34   * each must still prompt with its row-filled text.  Cancel sends no DELETE; Confirm sends exactly one, for that row,
35   * and the row leaves the table.
36   */
37  @EnabledIfSystemProperty(named=PetstoreBrowser.GATE, matches="true", disabledReason=PetstoreBrowser.DISABLED)
38  class PetstoreDeleteConfirm_BrowserTest extends TestBase {
39  
40  	@RegisterExtension
41  	static MicroserviceTestFixture server = PetstoreTestServer.fixture();
42  
43  	private static final String DLG = ".juneau-view-dialog";
44  	private static final String TITLE = DLG + " .juneau-view-dialog-title";
45  	private static final String NO_DLG = "!document.querySelector('" + DLG + "')";
46  
47  	private static Map<String,Map<String,Object>> report;
48  
49  	/** JS: some row in {@code body} has a cell whose text is exactly the quoted name (so "Rex 10" never matches "Rex 100"). */
50  	private static String rowWithName(String body) {
51  		return "Array.from(document.querySelectorAll('" + body + " tbody tr')).some(tr => Array.from(tr.cells).some(td => td.textContent.trim() === window.__name))";
52  	}
53  
54  	private static Map<String,Object> wait(String js) {
55  		return Map.of("waitFor", js, "timeoutMs", 5000);
56  	}
57  
58  	/**
59  	 * Opens the first row's action menu on {@code page}, clicks Delete, records the prompt (and the quoted name in it),
60  	 * then clicks {@code button} ({@code cancel} or {@code confirm}).
61  	 */
62  	private static Map<String,Object> theCase(String name, String page, String button) {
63  		var body = "#" + page + "-body";
64  		List<Map<String,Object>> actions = list(
65  			wait("document.querySelector('" + body + " .juneau-view-action-trigger')"),
66  			Map.of("click", body + " .juneau-view-action-trigger"),
67  			Map.of("click", ".juneau-view-action-item[data-action-id=delete]"),
68  			wait("document.querySelector('" + TITLE + "')")
69  		);
70  		actions.add(Map.of("evaluate", "(() => { const t = document.querySelector('" + TITLE + "').textContent; window.__title = t;"
71  			+ " const m = /'([^']+)'/.exec(t); window.__name = m && m[1]; })()"));
72  		actions.add(Map.of("click", DLG + " .juneau-view-dialog-" + button));
73  		actions.add(wait(NO_DLG));
74  		if (button.equals("confirm"))
75  			actions.add(wait("!" + rowWithName(body)));
76  		return Map.of("name", name, "path", "/console/" + page, "capture", "/console/" + page + "/", "actions", actions,
77  			"queries", Map.of("title", "window.__title", "name", "window.__name",
78  				"rowStill", rowWithName(body)));
79  	}
80  
81  	@BeforeAll
82  	static void probe() throws Exception {
83  		var base = PetstoreTestServer.awaitReady(server);
84  		report = run("delete-confirm", base, List.of(
85  			theCase("usersCancel", "users", "cancel"),
86  			theCase("usersConfirm", "users", "confirm"),
87  			theCase("petsCancel", "pets", "cancel"),
88  			theCase("petsConfirm", "pets", "confirm")));
89  	}
90  
91  	@SuppressWarnings({
92  		"unchecked" // requests is a JSON array of {url, method, postData}.
93  	})
94  	private static List<String> deletes(Map<String,Object> c) {
95  		return ((List<Map<String,Object>>) c.get("requests")).stream()
96  			.filter(r -> "DELETE".equals(r.get("method")))
97  			.map(r -> String.valueOf(r.get("url")).replaceFirst("^https?://[^/]+", ""))
98  			.toList();
99  	}
100 
101 	private static void assertCancelled(String name, String prompt) {
102 		var c = assertClean(report, name);
103 		var who = (String) query(c, "name");
104 		assertNotEmpty(who);
105 		assertEquals(prompt.replace("{}", who), query(c, "title"));
106 		assertEmpty(deletes(c));
107 		assertEquals(true, query(c, "rowStill"));
108 	}
109 
110 	/** {@code url} is a glob in which {@code {}} stands for the quoted name the prompt showed. */
111 	private static void assertConfirmed(String name, String prompt, String url) {
112 		var c = assertClean(report, name);
113 		var who = (String) query(c, "name");
114 		assertNotEmpty(who);
115 		assertEquals(prompt.replace("{}", who), query(c, "title"));
116 		var d = deletes(c);
117 		assertSize(1, d);
118 		assertMatchesGlob(url.replace("{}", who), d.get(0));
119 		assertEquals(false, query(c, "rowStill"));
120 	}
121 
122 	@Test void a01_usersDeletePromptsAndCancelSendsNothing() {
123 		assertCancelled("usersCancel", "Delete user '{}'?");
124 	}
125 
126 	@Test void a02_usersDeleteConfirmSendsOneDelete() {
127 		assertConfirmed("usersConfirm", "Delete user '{}'?", "/console/users/{}");
128 	}
129 
130 	@Test void a03_petsDeletePromptsAndCancelSendsNothing() {
131 		assertCancelled("petsCancel", "Delete '{}'?");
132 	}
133 
134 	@Test void a04_petsDeleteConfirmSendsOneDelete() {
135 		assertConfirmed("petsConfirm", "Delete '{}'?", "/console/pets/*");
136 	}
