35  /**
36   * The Orders row actions end to end in the real browser: Approve shows only on a placed order and redraws the row
37   * into an approved one, Cancel opens its server-built form dialog and sends the reason with its {@code DELETE}, and
38   * Delete goes through the petstore {@code typed} confirm renderer before it deletes.  The renderer's own rules are in
39   * {@link PetstoreTypedConfirm_BrowserTest}.  The table's View Settings dialog offers only the tabs the page's
40   * {@code columnConfig.tabs} names (a per-view tab restriction), not the ribbon's tabs.
41   */
42  @Covers(value={"confirm-dialogs", "per-view-tab-restriction"}, pages="/console/orders")
43  @EnabledIfSystemProperty(named=PetstoreBrowser.GATE, matches="true", disabledReason=PetstoreBrowser.DISABLED)
44  class PetstoreHooks_BrowserTest extends TestBase {

51  	// Records every non-GET fetch the page makes, with its status and body.
52  	private static final String RECORD = "(() => { window.__writes = []; const f = window.fetch;"
53  		+ " window.fetch = async (u, o) => { const r = await f(u, o); const m = (o && o.method) || 'GET';"
54  		+ " if (m !== 'GET') window.__writes.push({url: String(u), method: m, status: r.status, sent: o.body || null, body: await r.clone().text()});"
55  		+ " return r; }; })()";
56  	private static final String DLG = "[role=dialog]";
57  	private static final String TYPED = "dialog.petstore-typed-confirm";
58  	private static final String MENU_ITEMS = "[...document.querySelectorAll('.juneau-view-action-item')].map(b => b.dataset.actionId)";
59  
60  	private static String base;
61  	private static long approved, cancelled, deleted;
62  	private static Map<String,Map<String,Object>> report;
63  
64  	private static Map<String,Object> wait(String js) {
65  		return Map.of("waitFor", js, "timeoutMs", 8000);
66  	}
67  
68  	private static String row(long id) {
69  		return "#orders-body tr[data-juneau-row-id='" + id + "']";
70  	}
71  
72  	private static String wrote(String method, long id) {
73  		return "(window.__writes || []).some(w => w.method === '" + method + "' && w.url.includes('/console/orders/" + id + "'))";
74  	}

76  	/** Opens a row's action menu and records which actions it offers. */
77  	private static List<Map<String,Object>> openMenu(long id) {
78  		return List.of(
79  			wait("document.querySelector(\"" + row(id) + " .juneau-view-action-trigger\")"),
80  			Map.of("evaluate", RECORD),
81  			Map.of("click", row(id) + " .juneau-view-action-trigger"),
82  			wait("document.querySelector('.juneau-view-action-item')"),
83  			Map.of("evaluate", "window.__menu = " + MENU_ITEMS));
84  	}
85  
86  	@SafeVarargs
87  	private static List<Map<String,Object>> concat(List<Map<String,Object>> head, Map<String,Object>...tail) {
88  		var l = new ArrayList<>(head);
89  		l.addAll(List.of(tail));
90  		return l;
91  	}

93  	@BeforeAll
94  	static void probe() throws Exception {
95  		base = PetstoreTestServer.awaitReady(server);
96  		// One order to approve (the only placed one), and two others to cancel and to delete.
97  		var ids = new ArrayList<Long>();
98  		for (var o : new JsonList(get("/petstore/orders").body())) {
99  			var m = (JsonMap)o;
100 			if ("PLACED".equals(m.getString("status")) && approved == 0)
101 				approved = m.getLong("id");
102 			else
103 				ids.add(m.getLong("id"));
104 		}
105 		Collections.sort(ids);
106 		cancelled = ids.get(0);
107 		deleted = ids.get(1);
108 		report = run("hooks", base, List.of(
109 			Map.of("name", "approve", "path", "/console/orders",
110 				"actions", concat(openMenu(approved),
111 					Map.of("click", ".juneau-view-action-item[data-action-id=approve]"),
112 					wait(wrote("POST", approved)),
113 					Map.of("evaluate", "document.body.click()"),
114 					wait("!document.querySelector('.juneau-view-action-item')"),
115 					wait("document.querySelector(\"" + row(approved) + "\")?.textContent.toUpperCase().includes('APPROVED')"),
116 					Map.of("click", row(approved) + " .juneau-view-action-trigger"),
117 					wait("document.querySelector('.juneau-view-action-item')")),
118 				"queries", Map.of("before", "window.__menu", "after", MENU_ITEMS, "writes", "window.__writes")),
119 			Map.of("name", "cancel", "path", "/console/orders",
120 				"actions", concat(openMenu(cancelled),
121 					Map.of("click", ".juneau-view-action-item[data-action-id=cancel]"),
122 					wait("document.querySelector('" + DLG + " [data-juneau-form-field=reason]')"),
123 					Map.of("evaluate", "window.__dialog = {title: document.querySelector('" + DLG + " .juneau-view-dialog-title').textContent,"
124 						+ " fields: [...document.querySelectorAll('" + DLG + " [data-juneau-form-field]')].map(f => f.getAttribute('data-juneau-form-field'))}"),
125 					Map.of("fill", List.of(DLG + " [data-juneau-form-field=reason]", "Customer changed their mind")),
126 					Map.of("click", DLG + " .juneau-view-dialog-confirm"),
127 					wait(wrote("DELETE", cancelled)),
128 					wait("!document.querySelector(\"" + row(cancelled) + "\")")),
129 				"queries", Map.of("dialog", "window.__dialog", "writes", "window.__writes")),
130 			Map.of("name", "delete", "path", "/console/orders",
131 				"actions", concat(openMenu(deleted),
132 					Map.of("click", ".juneau-view-action-item[data-action-id=delete]"),
133 					wait("document.querySelector('" + TYPED + "')"),
134 					Map.of("evaluate", "window.__label = document.querySelector('" + TYPED + " input').getAttribute('aria-label')"),
135 					Map.of("fill", List.of(TYPED + " input", String.valueOf(deleted))),
136 					Map.of("click", TYPED + " .petstore-typed-confirm-ok"),
137 					wait(wrote("DELETE", deleted)),
138 					wait("!document.querySelector(\"" + row(deleted) + "\")")),
139 				"queries", Map.of("label", "window.__label", "writes", "window.__writes")),
140 			Map.of("name", "settings", "path", "/console/orders",
141 				"actions", List.of(
142 					wait("document.querySelector('#orders-body .juneau-config-chooser-btn')"),
143 					Map.of("click", "#orders-body .juneau-config-chooser-btn"),
144 					wait("document.querySelector('.juneau-config-dialog [role=tab]')")),
145 				"queries", Map.of(
146 					"tabs", "[...document.querySelectorAll('.juneau-config-dialog [role=tab]')].map(t => t.textContent.trim())",
147 					"names", "[...document.querySelectorAll('.juneau-config-dialog [role=tab]')].map(t => t.dataset.tab)"))));
148 	}

150 	private static HttpResponse<String> get(String path) throws Exception {
151 		return HTTP.send(HttpRequest.newBuilder(URI.create(base + path)).header("Accept", "application/json").GET().build(), BodyHandlers.ofString());
152 	}
153 
154 	@SuppressWarnings("unchecked") // writes is a JSON array of {url, method, status, sent, body}.
155 	private static Map<String,Object> theWrite(Map<String,Object> c, String method) {
156 		var w = ((List<Map<String,Object>>) query(c, "writes")).stream().filter(x -> method.equals(x.get("method"))).toList();
157 		assertSize(1, w);
158 		assertEquals(200, ((Number) w.get(0).get("status")).intValue(), () -> String.valueOf(w.get(0)));
159 		return w.get(0);
160 	}

171 	@Test void a02_approveRedrawsTheRow() throws Exception {
172 		var c = assertClean(report, "approve");
173 		theWrite(c, "POST");
174 		assertList(query(c, "after"), "deliver", "cancel", "delete");
175 		assertString("APPROVED", new JsonMap(get("/petstore/orders/" + approved).body()).getString("status"));
176 	}
