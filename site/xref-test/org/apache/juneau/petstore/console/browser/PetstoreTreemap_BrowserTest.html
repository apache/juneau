36  /**
37   * The Store page's pets treemap in the real browser: one tile per pet grouped by species then status, a pet tile
38   * click (or Enter on the keyboard-walked tile) opens the pet edit dialog, saving it sends a CSRF-protected PUT and
39   * redraws the treemap, and a group tile click does nothing.
40   */
41  @Covers(value="charts", pages="/console/store")
42  @HookCovers({"actions", "csrf"})
43  @EnabledIfSystemProperty(named=PetstoreBrowser.GATE, matches="true", disabledReason=PetstoreBrowser.DISABLED)
44  class PetstoreTreemap_BrowserTest extends TestBase {

53  	// Records every non-GET fetch the page makes, with its status and its CSRF header.
54  	private static final String RECORD = "(() => { window.__writes = []; const f = window.fetch;"
55  		+ " window.fetch = async (u, o) => { const r = await f(u, o); const m = (o && o.method) || 'GET';"
56  		+ " if (m !== 'GET') { const h = (o && o.headers) || {}; const t = h instanceof Headers ? h.get('X-Csrf-Token') : (h['X-Csrf-Token'] || h['x-csrf-token'] || null);"
57  		+ " window.__writes.push({url: String(u), method: m, status: r.status, csrf: t, body: String(o && o.body || '')}); }"
58  		+ " return r; }; })()";

60  	// Reads the drawn treemap: every node in data-index order with its pixel box, and clicks by real mouse events on the canvas.
61  	private static final String TM = """
62  		(() => {
63  		  const chart = () => window.JuneauConsole.cardApi('pets-map').chart();
64  		  const data = c => c.getModel().getSeriesByIndex(0).getData();
65  		  const box = (c, i) => { const el = data(c).getItemGraphicEl(i); if (!el) return null;
66  		    const r = el.getBoundingRect().clone(), m = el.getComputedTransform(); if (m) r.applyTransform(m);
67  		    return { x: r.x, y: r.y, w: r.width, h: r.height }; };
68  		  const nodes = () => { const c = chart(), d = data(c), out = [];
69  		    for (let i = 0; i < d.count(); i++) { const n = d.tree.getNodeByDataIndex(i), l = n && n.getLayout();
70  		      if (n && l) out.push({ i, name: n.name, leaf: n.children.length === 0, status: n.parentNode ? n.parentNode.name : null, species: n.parentNode && n.parentNode.parentNode ? n.parentNode.parentNode.name : null, box: box(c, i) }); }
71  		    return out; };
72  		  const fire = (type, x, y) => { const el = chart().getZr().painter.getViewportRoot(), r = el.getBoundingClientRect();
73  		    el.dispatchEvent(new MouseEvent(type, { bubbles: true, cancelable: true, clientX: r.left + x, clientY: r.top + y })); };
74  		  const click = (x, y) => { fire('mousemove', x, y); fire('mousedown', x, y); fire('mouseup', x, y); fire('click', x, y); };
75  		  const leaf = name => nodes().find(n => n.leaf && n.name === name);
76  		  window.__tm = {
77  		    nodes,
78  		    placed: () => nodes().filter(n => n.leaf).map(n => ({ name: n.name, status: n.status, species: n.species })),
79  		    leaves: () => nodes().filter(n => n.leaf).map(n => n.name),
80  		    groups: () => nodes().filter(n => !n.leaf && n.name).map(n => n.name),
81  		    speciesOf: name => { const n = leaf(name); return n ? n.species : null; },
82  		    clickLeaf: name => { const n = leaf(name); click(n.box.x + n.box.w / 2, n.box.y + n.box.h / 2); },
83  		    clickGroup: name => { const n = nodes().find(x => !x.leaf && x.name === name); click(n.box.x + n.box.w / 2, n.box.y + 3); }
84  		  };
85  		})()""";

87  	private static final String READY = "(() => { try { const t = window.__tm; return !!t && t.leaves().length > 3 && !!document.querySelector('#pets-map canvas'); } catch (e) { return false; } })()";

108 	private static String get(String path) throws Exception {
109 		return HttpClient.newHttpClient().send(HttpRequest.newBuilder(URI.create(base + path)).header("Accept", "application/json").GET().build(), BodyHandlers.ofString()).body();
110 	}

112 	/** The steps that wait for the treemap to draw and install the page helper. */
113 	private static List<Map<String,Object>> open() {
114 		return List.of(wait(DRAWN), eval(TM), wait(READY), eval(RECORD));
115 	}

132 	/** Edits the target pet's species through its tile, then waits for the redrawn tree to show it under {@code species}. */
133 	private static Map<String,Object> save(String name, String path, String species) {
134 		var n = js(target.getString("name"));
135 		return pick(name, path, then(open(),
136 			eval("window.__tm.clickLeaf(" + n + ")"),
137 			wait("document.querySelector('" + NAME + "')?.value"),
138 			eval("(() => { const s = document.querySelector('" + DLG + " [data-juneau-form-field=species]'); s.value = '" + species + "'; s.dispatchEvent(new Event('change', {bubbles: true})); })()"),
139 			Map.of("click", CONFIRM),
140 			wait("(window.__writes || []).some(w => w.method === 'PUT')"),
141 			wait("document.querySelectorAll('" + DLG + "').length === 0"),
142 			wait("(() => { try { return window.__tm.speciesOf(" + n + ") === '" + species + "'; } catch (e) { return false; } })()")), Map.of(
143 			"writes", "window.__writes",
144 			"dialogs", "document.querySelectorAll('" + DLG + "').length",
145 			"backdrops", "document.querySelectorAll('dialog[open], .juneau-view-dialog-backdrop').length",
146 			"species", "window.__tm.speciesOf(" + n + ")",
147 			"banner", "(document.querySelector('#pets-map [data-testid=chart-action-banner]') || {}).textContent || ''"));
148 	}

163 		var n = js(target.getString("name"));

165 		report = run("treemap", base, List.of(

202 	@Test void a01_storePage_rendersPetsTreemap_groupedBySpeciesThenStatus() {
203 		var c = assertClean(report, "render");
204 		var leaves = new TreeSet<String>();
205 		for (var p : pets.elements(JsonMap.class))
206 			leaves.add(p.getString("name"));
207 		assertEquals(leaves, new TreeSet<>(asStrings(query(c, "leaves"))));
208 		@SuppressWarnings("unchecked") // placed is a JSON array of {name, status, species}.
209 		var placed = (List<Map<String,Object>>)query(c, "placed");
210 		for (var p : pets.elements(JsonMap.class))
211 			assertTrue(placed.stream().anyMatch(x -> p.getString("name").equals(x.get("name"))
212 				&& p.getString("status").equals(x.get("status")) && p.getString("species").equals(x.get("species"))),
213 				() -> "pet " + p.getString("name") + " is not under status " + p.getString("status") + " under species " + p.getString("species") + ": " + placed);
214 		assertEquals(4, ((Number)query(c, "live")).intValue());
215 	}

234 	private void assertSaved(String name, String species) {
235 		var c = assertClean(report, name);
236 		@SuppressWarnings("unchecked") // writes is a JSON array of {url, method, status, csrf, body}.
237 		var w = (List<Map<String,Object>>)query(c, "writes");
238 		assertSize(1, w);
239 		assertString("PUT", w.get(0).get("method"));
240 		assertContains("/console/pets/" + target.getLong("id"), String.valueOf(w.get(0).get("url")));
241 		assertEquals(200, ((Number)w.get(0).get("status")).intValue(), String.valueOf(w.get(0)));
242 		assertNotNull(w.get(0).get("csrf"), "the PUT carried no CSRF token");
243 		assertFalse(String.valueOf(w.get(0).get("csrf")).isBlank());
244 		assertContains("\"" + species + "\"", String.valueOf(w.get(0).get("body")));
245 		assertEquals(0, ((Number)query(c, "dialogs")).intValue());
246 		assertEquals(0, ((Number)query(c, "backdrops")).intValue());
247 		assertContains("Saved '" + target.getString("name") + "'", String.valueOf(query(c, "banner")));
248 		assertString(species, query(c, "species"));
249 	}
