31  /**
32   * Smoke test for the browser harness: one console page loads in Chromium from the live test server with no
33   * script errors.
34   */
35  @EnabledIfSystemProperty(named=PetstoreBrowser.GATE, matches="true", disabledReason=PetstoreBrowser.DISABLED)
36  class PetstoreHarness_BrowserTest extends TestBase {
37  
38  	@RegisterExtension
39  	static MicroserviceTestFixture server = PetstoreTestServer.fixture();
40  
41  	private static Map<String,Map<String,Object>> report;
42  
43  	@BeforeAll
44  	static void probe() throws Exception {
45  		var base = PetstoreTestServer.awaitReady(server);
46  		report = run("harness", base, List.of(
47  			Map.of("name", "store", "waitUntil", "networkidle", "path", "/console/store",
48  				"queries", Map.of("title", "document.title", "pages", "document.querySelectorAll('#juneau-page').length")),
49  			Map.of("name", "actions", "path", "/console/about", "viewport", List.of(800, 600), "waitUntil", "load",
50  				"initScript", "window.__init = (window.__init || 0) + 1;",
51  				"actions", List.of(
52  					// The nav badge polls on every page; a reload that lands mid-fetch aborts it and logs E-JS-65 'Failed to fetch'.
53  					// A 'load' case therefore waits for the badge's fetch to finish before it navigates away.
54  					Map.of("waitFor", "performance.getEntriesByType('resource').some(r => r.name.includes('/changes/pending') && r.responseEnd > 0)"),
55  					Map.of("evaluate", "sessionStorage.setItem('h', '1')"),
56  					Map.of("reload", true),
57  					Map.of("goto", "/console/about?x=1"),
58  					Map.of("follow", "location.origin + '/console/about?y=2'")),
59  				"queries", Map.of("width", "window.innerWidth", "init", "window.__init", "search", "location.search",
60  					"kept", "sessionStorage.getItem('h')"))));
61  	}
62  
63  	@Test void a01_storeLoadsCleanly() {
64  		var c = assertClean(report, "store");
65  		assertString("1", String.valueOf(query(c, "pages")));
66  		assertContains("Juneau Petstore", query(c, "title"));
67  	}
68  
69  	@Test void a02_caseKeysAndNavigationActions() {
70  		var c = assertClean(report, "actions");
71  		assertString("800", String.valueOf(query(c, "width")));
72  		assertString("1", String.valueOf(query(c, "init")));   // the init script ran again in the new document
73  		assertString("?y=2", query(c, "search"));
74  		assertString("1", query(c, "kept"));   // reload, goto and follow stay in the case's one context
75  	}
76  
77  	@Test void a03_everyCaseReportsItsWallTime() {
78  		assertTrue(((Number) assertClean(report, "store").get("ms")).longValue() > 0);
79  	}
80  }
