33  /**
34   * The petstore {@code typed} confirm renderer in the real browser: what the user must type (the row count, one
35   * row's id, or a ribbon-hosted action's own id), the fail-closed refusal when there is nothing to type, Cancel and
36   * Escape resolving false, and the Orders row Delete requiring its exact order id.
37   */
38  @HookCovers({"rowActions", "dialogs"})
39  @EnabledIfSystemProperty(named=PetstoreBrowser.GATE, matches="true", disabledReason=PetstoreBrowser.DISABLED)
40  class PetstoreTypedConfirm_BrowserTest extends TestBase {
41  
42  	@RegisterExtension
43  	static MicroserviceTestFixture server = PetstoreTestServer.fixture();
44  
45  	private static final String DLG = "dialog.petstore-typed-confirm";
46  	private static final String INPUT = DLG + " input";
47  	private static final String OK = DLG + " .petstore-typed-confirm-ok";
48  	private static final String CANCEL = DLG + " .petstore-typed-confirm-cancel";
49  
50  	// Records the typed dialog's label and Confirm state under a key, so a case can observe several steps.
51  	private static final String SNAP = "window.__snap = k => {"
52  		+ " const d = document.querySelector('" + DLG + "');"
53  		+ " (window.__log = window.__log || {})[k] = d ? {label: d.querySelector('input').getAttribute('aria-label'), title: d.querySelector('h2').textContent,"
54  		+ "   disabled: d.querySelectorAll('button')[0].disabled} : null; }";
55  
56  	private static String open(String request) {
57  		return "(() => { window.__res = 'pending'; window.__warns = []; console.warn = (...a) => window.__warns.push(a.join(' '));"
58  			+ " JuneauViews.dialogs.confirm(Object.assign({renderer: 'typed', title: 'Sure?', confirmLabel: 'Go'}, " + request + "))"
59  			+ ".then(v => { window.__res = v; }, () => { window.__res = 'rejected'; }); })()";
60  	}
61  
62  	private static Map<String,Object> theCase(String name, String request, List<Map<String,Object>> steps) {
63  		List<Map<String,Object>> actions = list(
64  			Map.of("waitFor", "window.JuneauViews && JuneauViews.dialogs"),
65  			Map.of("evaluate", SNAP),
66  			Map.of("evaluate", open(request))
67  		);
68  		actions.addAll(steps);
69  		return Map.of("name", name, "path", "/console/orders", "actions", actions,
70  			"queries", Map.of("log", "window.__log", "res", "window.__res", "warns", "window.__warns", "open", "!!document.querySelector('" + DLG + "')"));
71  	}
72  
73  	private static Map<String,Object> snap(String k) {
74  		return Map.of("evaluate", "window.__snap('" + k + "')");
75  	}
76  
77  	private static Map<String,Object> wait(String js) {
78  		return Map.of("waitFor", js, "timeoutMs", 5000);
79  	}
80  
81  	private static final String RES_TRUE = "window.__res === true";
82  	private static final String RES_FALSE = "window.__res === false";
83  
84  	private static Map<String,Map<String,Object>> report;
85  
86  	@BeforeAll
87  	static void probe() throws Exception {
88  		var base = PetstoreTestServer.awaitReady(server);
89  		report = run("typed-confirm", base, List.of(
90  			theCase("count", "{rows: [{id: 1}, {id: 2}, {id: 3}]}", List.of(
91  				wait("document.querySelector('" + DLG + "')"), snap("start"),
92  				Map.of("fill", List.of(INPUT, "2")), snap("wrong"),
93  				Map.of("fill", List.of(INPUT, "3")), snap("right"),
94  				Map.of("click", OK), wait(RES_TRUE))),
95  			theCase("oneRow", "{rows: [{id: 42}]}", List.of(
96  				wait("document.querySelector('" + DLG + "')"), snap("start"),
97  				Map.of("fill", List.of(INPUT, "4")), snap("wrong"),
98  				Map.of("fill", List.of(INPUT, "42")), snap("right"),
99  				Map.of("click", OK), wait(RES_TRUE))),
100 			theCase("noRow", "{rows: [], action: {id: 'reset'}}", List.of(
101 				wait("document.querySelector('" + DLG + "')"), snap("start"),
102 				Map.of("fill", List.of(INPUT, "rese")), snap("wrong"),
103 				Map.of("fill", List.of(INPUT, "reset")), snap("right"),
104 				Map.of("click", OK), wait(RES_TRUE))),
105 			theCase("empty", "{rows: []}", List.of(
106 				wait(RES_FALSE))),
107 			theCase("nullId", "{rows: [{id: null}]}", List.of(wait(RES_FALSE))),
108 			theCase("missingId", "{rows: [{name: 'no id here'}]}", List.of(wait(RES_FALSE))),
109 			theCase("blankId", "{rows: [{id: '   '}]}", List.of(wait(RES_FALSE))),
110 			theCase("cancel", "{rows: [{id: 7}]}", List.of(
111 				wait("document.querySelector('" + DLG + "')"),
112 				Map.of("fill", List.of(INPUT, "7")),
113 				Map.of("click", CANCEL), wait(RES_FALSE))),
114 			theCase("escape", "{rows: [{id: 7}]}", List.of(
115 				wait("document.querySelector('" + DLG + "')"),
116 				Map.of("press", List.of(INPUT, "Escape")), wait(RES_FALSE))),
117 			Map.of("name", "ordersDelete", "waitUntil", "networkidle", "path", "/console/orders", "capture", "/console/orders/",
118 				"actions", List.of(
119 					Map.of("evaluate", SNAP),
120 					wait("document.querySelector('#orders-body .juneau-view-action-trigger')"),
121 					Map.of("click", "#orders-body .juneau-view-action-trigger"),
122 					Map.of("click", ".juneau-view-action-item[data-action-id=delete]"),
123 					wait("document.querySelector('" + DLG + "')"), snap("start"),
124 					Map.of("fill", List.of(INPUT, "not-the-id")), snap("wrong"),
125 					// Type the id the dialog names in its own prompt, one character short, then exactly.
126 					Map.of("evaluate", "(() => { const m = /'([^']+)'/.exec(document.querySelector('" + DLG + " h2').textContent); window.__id = m && m[1]; })()"),
127 					Map.of("evaluate", "(() => { const i = document.querySelector('" + DLG + " input'); i.value = window.__id.slice(0, -1); i.dispatchEvent(new Event('input')); })()"), snap("short"),
128 					Map.of("evaluate", "(() => { const i = document.querySelector('" + DLG + " input'); i.value = window.__id; i.dispatchEvent(new Event('input')); })()"), snap("right"),
129 					Map.of("click", CANCEL), wait("!document.querySelector('" + DLG + "')")),
130 				"queries", Map.of("log", "window.__log", "id", "window.__id"))));
131 	}

133 	@SuppressWarnings({
134 		"unchecked" // log is a JSON object of {label, disabled} snapshots.
135 	})
136 	private static Map<String,Object> log(String name, String key) {
137 		var log = (Map<String,Object>) query(assertClean(report, name), "log");
138 		return (Map<String,Object>) log.get(key);
139 	}

141 	@Test void a01_severalRowsTypeTheCount() {
142 		assertBean(log("count", "start"), "label,disabled", "Type 3 to confirm,true");
143 		assertBean(log("count", "wrong"), "disabled", "true");
144 		assertBean(log("count", "right"), "disabled", "false");
145 		assertEquals(true, query(assertClean(report, "count"), "res"));
146 	}

169 	@Test void a04b_aRowWithoutAUsableIdFailsClosed() {
170 		for (var name : List.of("nullId", "missingId", "blankId")) {
171 			var c = assertClean(report, name);
172 			assertEquals(false, query(c, "res"), name);
173 			assertEquals(false, query(c, "open"), name);
174 			assertContains("typed confirm", String.valueOf(query(c, "warns")));
175 		}
176 	}

186 	@SuppressWarnings({
187 		"unchecked" // requests is a JSON array of {url, method, postData}.
188 	})
189 	@Test void a07_ordersDeleteNeedsTheExactId() {
190 		var c = assertClean(report, "ordersDelete");
191 		var id = (String) query(c, "id");
192 		assertNotEmpty(id);
193 		assertBean(log("ordersDelete", "start"), "label,disabled", "Type " + id + " to confirm,true");
194 		assertBean(log("ordersDelete", "wrong"), "disabled", "true");
195 		assertBean(log("ordersDelete", "short"), "disabled", "true");
196 		assertBean(log("ordersDelete", "right"), "disabled", "false");
197 		for (var r : (List<Map<String,Object>>) c.get("requests"))
198 			assertNotEquals("DELETE", r.get("method"), "Cancel must not send the delete");
199 	}
