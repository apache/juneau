32  /**
33   * P3 Sold in the real browser: the client-mode table's ribbon filters the rows it already holds, with no new request
34   * for rows.
35   */
36  @EnabledIfSystemProperty(named=PetstoreBrowser.GATE, matches="true", disabledReason=PetstoreBrowser.DISABLED)
37  class PetstoreClientRibbon_BrowserTest extends TestBase {
38  
39  	@RegisterExtension
40  	static MicroserviceTestFixture server = PetstoreTestServer.fixture();
41  
42  	/**
43  	 * The distinct species on the shown page (the third cell of each row).  Body cells carry no column hook, so the
44  	 * column is found by position, which the page's fixed column list pins.
45  	 */
46  	private static final String SPECIES = "[...new Set([...document.querySelectorAll('#sold tbody tr[data-juneau-row-id] td:nth-child(3)')].map(td => td.textContent.trim()))]";
47  
48  	/** The paging pill's total, from its {@code 1-25 of 87} text. */
49  	private static final String TOTAL = "Number(document.querySelector('#sold-body .juneau-view-pagingpill-info').textContent.split(' of ')[1])";
50  
51  	private static final String DOGS = "#sold-body .juneau-view-ribbon-btn[aria-label=\"Dogs\"]";
52  
53  	private static Map<String,Map<String,Object>> report;
54  
55  	@BeforeAll
56  	static void probe() throws Exception {
57  		var base = PetstoreTestServer.awaitReady(server);
58  		report = run("client-ribbon", base, List.of(
59  			Map.of("name", "sold", "path", "/console/pets/sold", "capture", "/console/pets/sold/rows",
60  				"actions", List.of(
61  					Map.of("waitFor", "document.querySelector('#sold tbody tr[data-juneau-row-id]')"),
62  					Map.of("evaluate", "window.__before = " + TOTAL + "; window.__beforeSpecies = " + SPECIES),
63  					Map.of("click", DOGS),
64  					Map.of("waitFor", TOTAL + " < window.__before")),
65  				"queries", Map.of(
66  					"before", "window.__before",
67  					"beforeSpecies", "window.__beforeSpecies.length",
68  					"after", TOTAL,
69  					"species", SPECIES,
70  					"pressed", "document.querySelector('" + DOGS + "').getAttribute('aria-pressed')"),
71  				"fetches", Map.of("pets", "/petstore/pets"))));
72  	}
73  
74  	@Test void a01_ribbonFiltersClientSide() {
75  		var c = assertClean(report, "sold");
76  		assertTrue(((Number) query(c, "beforeSpecies")).intValue() > 1, () -> "the unfiltered page shows one species: " + c.get("queries"));
77  		assertList(query(c, "species"), "DOG");
78  		assertString("true", query(c, "pressed"));
79  	}
80  
81  	@Test void a02_noNewRowRequest() {
82  		var c = assertClean(report, "sold");
83  		assertSize(1, (List<?>) c.get("requests"));
84  	}
85  
86  	@Test void a03_totalIsTheSoldDogs() throws Exception {
87  		var c = assertClean(report, "sold");
88  		@SuppressWarnings({
89  			"unchecked" // fetches are {status, body} objects.
90  		})
91  		var f = (Map<String,Object>)((Map<String,Object>)c.get("fetches")).get("pets");
92  		assertEquals(200, ((Number) f.get("status")).intValue(), () -> String.valueOf(f));
93  		var dogs = 0;
94  		for (var p : new JsonList(String.valueOf(f.get("body"))).elements(JsonMap.class))
95  			if ("SOLD".equals(p.getString("status")) && "DOG".equals(p.getString("species")))
96  				dogs++;
97  		assertTrue(dogs > 0, "the seed sold no dogs");
98  		assertEquals(dogs, ((Number) query(c, "after")).intValue());
99  		assertTrue(dogs < ((Number) query(c, "before")).intValue(), () -> String.valueOf(c.get("queries")));
100 	}
101 }
