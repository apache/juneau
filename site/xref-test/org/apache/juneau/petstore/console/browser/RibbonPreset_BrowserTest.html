31   * The standard ribbon preset in the real browser: slot order, a single column button inside the ribbon, distinct filter
32   * icons, the export buttons, and the conditional collapse and columns items.
33   */

40  	private static String ids(String body) {
41  		return "[...document.querySelectorAll('#" + body + "-body [data-juneau-ribbon-id]')].map(e => e.getAttribute('data-juneau-ribbon-id')).join(',')";
42  	}
43  
44  	private static String icons(String body) {
45  		return "[...document.querySelectorAll('#" + body + "-body .juneau-view-ribbon-btn use')].map(u => u.getAttribute('href') || u.getAttribute('xlink:href'))";
46  	}

48  	private static Map<String,Object> page(String name, String path, String body) {
49  		return Map.of("name", name, "path", path,
50  			"actions", List.of(Map.of("waitFor", "document.querySelector('#" + body + " tbody tr[data-juneau-row-id]')")),
51  			"queries", Map.of(
52  				"ids", ids(body),
53  				"icons", icons(body),
54  				"strayChooser", "document.querySelectorAll('#" + body + "-body .juneau-config-chooser-btn').length",
55  				"chooserInRibbon", "document.querySelectorAll('#" + body + "-body [data-juneau-ribbon-id=columns]').length"));
56  	}
57  
58  	private static final String HEADERS = "[...document.querySelectorAll('#%s thead tr:first-child th')].map(t => t.textContent.trim()).join('|')";
59  
60  	private static final String GEOM = "(() => {"
61  		+ "const r = id => { const b = document.querySelector('#%s-body [data-juneau-ribbon-id=' + id + ']'); const g = b && b.closest('.juneau-view-ribbon-group'); if (!g) return null;"
62  		+ " const k = [...g.querySelectorAll('.juneau-view-ribbon-btn')]; const f = getComputedStyle(k[0]), l = getComputedStyle(k[k.length - 1]);"
63  		+ " return [f.borderTopLeftRadius, f.borderBottomLeftRadius, l.borderTopRightRadius, l.borderBottomRightRadius, k.length > 1 ? getComputedStyle(k[1]).borderTopLeftRadius : '0px'].join(','); };"
64  		+ "return { exportGroup: r('copy'), viewingGroup: r('columns') }; })()";
65  
66  	/** Opens the ribbon's columns button, removes a column in the dialog, applies, and reloads. */
67  	private static Map<String,Object> columnsCase(String name, String path, String body, String col, String label) {
68  		var rows = "document.querySelector('#" + body + " tbody tr[data-juneau-row-id]')";
69  		var headers = String.format(HEADERS, body);
70  		return Map.of("name", name, "path", path,
71  			"actions", List.of(
72  				Map.of("waitFor", rows),
73  				Map.of("click", "#" + body + "-body [data-juneau-ribbon-id=columns]"),
74  				Map.of("waitFor", "document.querySelector('.juneau-config-dialog')"),
75  				Map.of("click", ".juneau-config-col-row[data-col=\"" + col + "\"] .juneau-config-col-remove"),
76  				Map.of("click", ".juneau-config-apply"),
77  				Map.of("waitFor", "!" + headers + ".split('|').includes('" + label + "')"),
78  				Map.of("evaluate", "window.__hiddenOk = true"),
79  				Map.of("reload", true),
80  				Map.of("waitFor", rows)),
81  			"queries", Map.of("headers", headers));
82  	}

99  	private static Map<String,Object> geometry(String name, String path, String body) {
100 		return Map.of("name", name + "Geometry", "path", path,
101 			"actions", List.of(Map.of("waitFor", "document.querySelector('#" + body + " tbody tr[data-juneau-row-id]')")),
102 			"queries", Map.of(
103 				"geom", String.format(GEOM, body),
104 				"collapseAnywhere", "document.querySelectorAll('[data-juneau-ribbon-id=collapse]').length",
105 				"stale", "(() => { const e = document.querySelector('#" + body + "-body .juneau-view-staleness'); return e ? [e.getAttribute('role'), e.getAttribute('aria-label'), e.getAttribute('data-jc-tip')].join('|') : null; })()",
106 				"filterIcons", "[...document.querySelectorAll('#" + body + "-body .juneau-view-ribbon-btn[data-juneau-ribbon-id]')].filter(b => b.querySelector('use') && !['copy','csv','excel','pdf','link','columns','reset','refresh','search','add-pet'].includes(b.getAttribute('data-juneau-ribbon-id'))).map(b => b.getAttribute('data-juneau-ribbon-id') + '=' + b.querySelector('use').getAttribute('href'))"));
107 	}

109 	private static String q(String name, String key) {
110 		return String.valueOf(query(assertClean(report, name), key));

113 	@Test void a01_petsSlotOrderHasNoCollapse() {
114 		for (var n : new String[] {"pets", "petsBuilder"}) {
115 			var ids = q(n, "ids");
116 			assertFalse(ids.contains("collapse"), () -> n + ": " + ids);
117 			var order = List.of("search", "add-pet", "status", "species", "copy", "csv", "excel", "pdf", "link", "columns", "reset", "refresh");
118 			var last = -1;
119 			for (var id : order) {
120 				var i = Arrays.asList(ids.split(",")).indexOf(id);
121 				if (i < 0 && (id.equals("status") || id.equals("species")))
122 					continue;
123 				assertTrue(i > last, () -> n + ": '" + id + "' out of order in " + ids);
124 				last = i;
125 			}
126 		}
127 	}

143 	@Test void a05_iconsAreDistinctAndExportButtonsPresent() {
144 		for (var n : new String[] {"pets", "petsBuilder"}) {
145 			var ids = q(n, "ids");
146 			assertTrue(ids.contains("excel") && ids.contains("pdf"), () -> n + ": " + ids);
147 			var icons = (List<?>)query(assertClean(report, n), "icons");
148 			assertFalse(icons.isEmpty(), n);
149 			assertTrue(icons.stream().allMatch(x -> x != null && ! String.valueOf(x).isEmpty()), () -> n + ": " + icons);
150 		}
151 	}

153 	@Test void a06_ribbonColumnsButtonOpensDialogAndHideSurvivesReload() {
154 		for (var n : new String[] {"petsColumns", "storeColumns"}) {
155 			var c = assertClean(report, n);
156 			assertFalse(c.get("timeouts") != null && ! ((List<?>)c.get("timeouts")).isEmpty(), () -> n + " timeouts: " + c.get("timeouts"));
157 			var h = String.valueOf(query(c, "headers"));
158 			assertFalse(h.contains("Price (USD)") && n.startsWith("pets"), () -> n + ": " + h);
159 			assertFalse(h.contains("Status") && n.startsWith("store"), () -> n + ": " + h);
160 		}
161 	}
