21  import static org.junit.jupiter.api.Assertions.*;

23  import java.util.*;
24  
25  import org.apache.juneau.*;

31  /**
32   * Bulk selection on P2 Pets in the real browser: the selection adds up across pages and survives a search, and the
33   * per-row Mark sold action runs at most four writes at a time and ends in one summary toast.
34   */
35  @Covers(value="bulk-select", pages="/console/pets")
36  @EnabledIfSystemProperty(named=PetstoreBrowser.GATE, matches="true", disabledReason=PetstoreBrowser.DISABLED)
37  class PetstoreBulkSelect_BrowserTest extends TestBase {
38  
39  	@RegisterExtension
40  	static MicroserviceTestFixture server = PetstoreTestServer.fixture();
41  
42  	private static final String READY = "document.querySelector('#pets tbody tr[data-juneau-row-id]')";
43  	private static final String HEADER_BOX = "#pets thead input.juneau-view-select-all-checkbox";
44  	private static final String ROW_BOX = "input.juneau-view-select-checkbox";
45  
46  	/** The selected count, from the table's retained selection topic. */
47  	private static final String COUNT = "(window.JuneauViews.bus.get('selection:pets') || {}).count";
48  
49  	/** The header checkbox state: mixed, true or false. */
50  	private static final String HEADER = "(() => { const h = document.querySelector('" + HEADER_BOX + "'); return h.indeterminate ? 'mixed' : String(h.checked); })()";
51  
52  	/** Wraps fetch to record the peak and total number of sell requests in flight. */
53  	private static final String COUNT_SELLS = "window.__inflight = 0; window.__peak = 0; window.__sells = 0; const f = window.fetch;"
54  		+ " window.fetch = async (...a) => { const sell = String(a[0]).endsWith('/sell'); if (sell) { window.__sells++; window.__inflight++;"
55  		+ " window.__peak = Math.max(window.__peak, window.__inflight); } try { return await f(...a); } finally { if (sell) window.__inflight--; } }";
56  
57  	private static Map<String,Map<String,Object>> report;
58  
59  	@BeforeAll
60  	static void probe() throws Exception {
61  		var base = PetstoreTestServer.awaitReady(server);
62  		report = run("bulk-select", base, List.of(
63  			Map.of("name", "across-pages", "path", "/console/pets", "capture", "/console/pets/query",
64  				"actions", List.of(
65  					Map.of("waitFor", READY),
66  					Map.of("click", HEADER_BOX),
67  					Map.of("evaluate", "window.__pageOne = " + COUNT + "; window.__rows = document.querySelectorAll('#pets tbody tr[data-juneau-row-id]').length"),
68  					Map.of("click", "#pets-body .juneau-view-pagingpill-btn[aria-label=\"Next page\"]"),
69  					Map.of("waitForCaptured", 2),
70  					Map.of("waitFor", "document.querySelector('#pets tbody " + ROW_BOX + ":not(:checked)')"),
71  					Map.of("click", "#pets tbody tr:first-child " + ROW_BOX),
72  					Map.of("evaluate", "window.__count = " + COUNT + "; window.__header = " + HEADER),
73  					Map.of("fill", List.of("#pets-body input[type=search]", "a")),
74  					Map.of("waitForCaptured", 3, "timeoutMs", 5000),
75  					Map.of("waitFor", READY)),
76  				"queries", Map.of(
77  					"pageOne", "window.__pageOne",
78  					"rows", "window.__rows",
79  					"count", "window.__count",
80  					"header", "window.__header",
81  					"countAfterSearch", COUNT,
82  					"toolbar", "document.querySelector('#pets-body .juneau-view-bulk-count').textContent")),
83  			Map.of("name", "mark-sold", "path", "/console/pets",
84  				"actions", List.of(
85  					Map.of("waitFor", READY),
86  					Map.of("evaluate", COUNT_SELLS),
87  					Map.of("click", HEADER_BOX),
88  					Map.of("evaluate", "window.__selected = " + COUNT),
89  					Map.of("click", "#pets-body .juneau-view-bulk-action-btn"),
90  					Map.of("waitFor", "document.querySelector('[role=dialog]')"),
91  					Map.of("evaluate", "window.__title = document.querySelector('[role=dialog] .juneau-view-dialog-title').textContent"),
92  					Map.of("click", "[role=dialog] .juneau-view-dialog-confirm"),
93  					Map.of("waitFor", "document.querySelector('.juneau-view-bulk-toast[role=status]')")),
94  				"queries", Map.of(
95  					"selected", "window.__selected",
96  					"title", "window.__title",
97  					"peak", "window.__peak",
98  					"sells", "window.__sells",
99  					"toast", "document.querySelector('.juneau-view-bulk-toast[role=status]').textContent",
100 					"left", COUNT))));
101 	}
102 
103 	/** Page one's rows, which the header checkbox selects, plus the one row picked on page two. */
104 	private static int expected(Map<String,Object> c) {
105 		var pageOne = ((Number) query(c, "pageOne")).intValue();
106 		assertTrue(pageOne > 0, "the header checkbox selected nothing");
107 		assertEquals(((Number) query(c, "rows")).intValue(), pageOne);
108 		return pageOne + 1;
109 	}
110 
111 	@Test void a01_selectionAddsUpAcrossPages() {
112 		var c = assertClean(report, "across-pages");
113 		assertString("mixed", query(c, "header"));
114 		assertEquals(expected(c), ((Number) query(c, "count")).intValue());
115 	}
116 
117 	@Test void a02_selectionSurvivesSearch() {
118 		var c = assertClean(report, "across-pages");
119 		var n = expected(c);
120 		assertEquals(n, ((Number) query(c, "countAfterSearch")).intValue());
121 		assertString(n + " selected ·", query(c, "toolbar"));
122 	}
123 
124 	@Test void a03_perRowMarkSoldIsAtMostFourConcurrent() {
125 		var c = assertClean(report, "mark-sold");
126 		var peak = ((Number) query(c, "peak")).intValue();
127 		assertTrue(peak >= 1 && peak <= 4, () -> "peak " + peak);
128 		assertEquals(((Number) query(c, "selected")).intValue(), ((Number) query(c, "sells")).intValue());
129 	}
130 
131 	@Test void a04_runEndsInOneSummaryAndClearsTheSelection() {
132 		var c = assertClean(report, "mark-sold");
133 		assertString("Mark pets sold", query(c, "title"));
134 		assertString("Mark sold: " + query(c, "selected") + " succeeded", query(c, "toast"));
135 		assertEquals(0, ((Number) query(c, "left")).intValue());
136 	}
137 }
