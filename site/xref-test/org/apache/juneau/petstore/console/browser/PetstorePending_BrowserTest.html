34  /**
35   * The header pending-changes badge in the real browser: hidden while nothing is staged, counting a change the pencil
36   * stages (without touching the pet), linking to Pending changes, and hidden again once that change is discarded.
37   */
38  @Covers(value="pending-changes-badge", pages={"/console/store", "/console/pets/1/summary", "/console/pets/changes"})
39  @EnabledIfSystemProperty(named=PetstoreBrowser.GATE, matches="true", disabledReason=PetstoreBrowser.DISABLED)
40  class PetstorePending_BrowserTest extends TestBase {
41  
42  	@RegisterExtension
43  	static MicroserviceTestFixture server = PetstoreTestServer.fixture();
44  
45  	private static final String BADGE = "[data-juneau-badge-id=pending]";
46  	private static final String PENCIL = "#pet-body [data-juneau-edit=\"rename\"]";
47  	private static final String INPUT = "#pet-body .juneau-edit-input";
48  	private static final String TYPED = "dialog.petstore-typed-confirm";
49  	private static final String STATE = "(() => { const b = document.querySelector('" + BADGE + "'); return b ? {hidden: b.hidden, text: b.textContent, total: b.getAttribute('data-jc-total')} : null; })()";
50  	private static final String PENDING = "fetch('/console/pets/changes/pending', {headers: {Accept: 'application/json'}}).then(r => r.json()).then(b => b.total)";
51  
52  	// Flags the page once its discard POST has come back.
53  	private static final String DISCARDED = "(() => { const f = window.fetch; window.fetch = async (u, o) => { const r = await f(u, o);"
54  		+ " if (o && o.method === 'POST' && String(u).includes('/discard')) window.__discarded = r.status; return r; }; })()";
55  
56  	// Records the total of every badge poll that has come back, so a case can tell a polled "nothing staged" (the badge
57  	// rendered hidden) from a page whose poll has not answered yet.  Runs at the start of every document.
58  	private static final String POLLS = "(() => { window.__polls = []; const f = window.fetch; window.fetch = async (u, o) => {"
59  		+ " const r = await f(u, o);"
60  		+ " if (String(u).includes('/console/pets/changes/pending') && !(o && o.method && o.method !== 'GET'))"
61  		+ "   r.clone().json().then(b => window.__polls.push(b.total), () => window.__polls.push('error'));"
62  		+ " return r; }; })()";
63  
64  	// A badge poll has answered and the badge has rendered that answer.
65  	private static final String POLLED = "window.__polls && window.__polls.length > 0 && document.querySelector('" + BADGE + "')";
66  
67  	// The total the most recent badge poll answered (the badge renders the latest one; a page may poll more than once).
68  	private static final String LAST_POLL = "window.__polls[window.__polls.length - 1]";
69  
70  	private static String base, before;
71  	private static long pet;
72  	private static Map<String,Map<String,Object>> report;
73  
74  	private static Map<String,Object> wait(String js) {
75  		return Map.of("waitFor", js, "timeoutMs", 8000);
76  	}
77  
78  	@BeforeAll
79  	static void probe() throws Exception {
80  		base = PetstoreTestServer.awaitReady(server);
81  		for (var o : new JsonList(get("/petstore/pets").body()))
82  			if ("AVAILABLE".equals(((JsonMap)o).getString("status"))) {
83  				pet = ((JsonMap)o).getLong("id");
84  				break;
85  			}
86  		before = petName();
87  		var row = "[...document.querySelectorAll('#changes-body tbody tr')].find(tr => tr.textContent.includes('Badged " + pet + "'))";
88  		report = run("pending", base, List.of(
89  			Map.of("name", "empty", "path", "/console/store", "initScript", POLLS,
90  				"actions", List.of(wait(POLLED)),
91  				"queries", Map.of("badge", STATE, "lastPoll", LAST_POLL)),
92  			Map.of("name", "staged", "path", "/console/pets/" + pet + "/summary",
93  				"actions", List.of(
94  					wait("document.querySelector('" + PENCIL + "')"),
95  					Map.of("click", PENCIL),
96  					wait("document.querySelector('" + INPUT + "')"),
97  					Map.of("fill", List.of(INPUT, "Badged " + pet)),
98  					Map.of("press", List.of(INPUT, "Enter")),
99  					wait("document.querySelector('.juneau-view-bulk-toast')"),
100 					Map.of("goto", "/console/store"),
101 					wait("document.querySelector('" + BADGE + ":not([hidden])')")),
102 				"queries", Map.of("badge", STATE)),
103 			Map.of("name", "follow", "path", "/console/store",
104 				"actions", List.of(
105 					wait("document.querySelector('" + BADGE + ":not([hidden])')"),
106 					Map.of("click", BADGE),
107 					wait("location.pathname === '/console/pets/changes'")),
108 				"queries", Map.of("path", "location.pathname")),
109 			Map.of("name", "cleared", "path", "/console/pets/changes", "initScript", POLLS,
110 				"actions", List.of(
111 					wait(row),
112 					Map.of("evaluate", DISCARDED),
113 					Map.of("evaluate", row + ".querySelector('.juneau-view-select-checkbox').click()"),
114 					wait("document.querySelector('.juneau-view-bulk-toolbar:not([hidden]) .juneau-view-bulk-select')"),
115 					Map.of("evaluate", "(() => { const s = document.querySelector('.juneau-view-bulk-select'); s.value = 'discard'; s.dispatchEvent(new Event('change', {bubbles: true})); })()"),
116 					Map.of("click", ".juneau-view-bulk-go"),
117 					wait("document.querySelector('" + TYPED + "')"),
118 					Map.of("evaluate", "(() => { const i = document.querySelector('" + TYPED + " input'); i.value = /Type (.+) to confirm/.exec(i.getAttribute('aria-label'))[1]; i.dispatchEvent(new Event('input')); })()"),
119 					Map.of("click", TYPED + " .petstore-typed-confirm-ok"),
120 					wait("window.__discarded"),
121 					Map.of("goto", "/console/store"),
122 					wait(POLLED)),
123 				"queries", Map.of("badge", STATE, "lastPoll", LAST_POLL, "pending", PENDING))));
124 	}
125 
126 	private static HttpResponse<String> get(String path) throws Exception {
127 		return HttpClient.newHttpClient().send(HttpRequest.newBuilder(URI.create(base + path)).header("Accept", "application/json").GET().build(), BodyHandlers.ofString());
128 	}
129 
130 	private static String petName() throws Exception {
131 		return new JsonMap(get("/petstore/pets/" + pet).body()).getString("name");
132 	}

138 	@Test void a01_hiddenWhileNothingIsStaged() {
139 		var c = assertClean(report, "empty");
140 		assertString("0", String.valueOf(query(c, "lastPoll")));   // a poll answered "nothing staged"...
141 		assertBean(query(c, "badge"), "hidden", "true");   // ...and the badge rendered that, not a pre-poll default
142 	}
143 
144 	@Test void a02_countsTheStagedChange() {
145 		assertBean(query(assertClean(report, "staged"), "badge"), "hidden,text,total", "false,1 change pending (1 yours),1");
146 	}
147 
148 	@Test void a03_stagingLeavesThePetAlone() throws Exception {
149 		assertClean(report, "staged");
150 		assertString(before, petName());
151 	}
152 
153 	@Test void a04_linksToPendingChanges() {
154 		assertString("/console/pets/changes", query(assertClean(report, "follow"), "path"));
155 	}
156 
157 	@Test void a05_hiddenAgainOnceDiscarded() {
158 		var c = assertClean(report, "cleared");
159 		assertString("0", String.valueOf(query(c, "pending")));
160 		assertString("0", String.valueOf(query(c, "lastPoll")));   // the store page's own poll saw the discard
161 		assertBean(query(c, "badge"), "hidden", "true");
162 	}
