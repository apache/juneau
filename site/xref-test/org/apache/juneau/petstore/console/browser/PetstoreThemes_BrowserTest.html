24  import org.apache.juneau.*;

30  /**
31   * P9 in the real browser: each stock theme resolves to a distinct set of token values, the custom theme's leaf and
32   * alias resolve, and every theme is screenshotted to {@code target/js/snapshots} (CI artifacts only, OQ9).
33   */
34  @EnabledIfSystemProperty(named=PetstoreBrowser.GATE, matches="true", disabledReason=PetstoreBrowser.DISABLED)
35  class PetstoreThemes_BrowserTest extends TestBase {
36  
37  	@RegisterExtension
38  	static MicroserviceTestFixture server = PetstoreTestServer.fixture();
39  
40  	private static final List<String> STOCK = List.of("open", "light-red", "light-brown", "red", "gray");
41  	// Tokens every stock pack defines and that differ between open and gray (checked against juneau-theme-*.css).
42  	private static final String TOKENS = "(() => { const s = getComputedStyle(document.documentElement);"
43  		+ " return ['--jc-chrome-bg', '--jc-border', '--jc-text', '--jc-accent-selected', '--jc-pill-red-bg'].map(n => s.getPropertyValue(n).trim()); })()";
44  	// The custom theme's own list: --jc-tab-bar-bg is an alias declared only by the custom <@theme> block.
45  	private static final String CUSTOM_TOKENS = "(() => { const s = getComputedStyle(document.documentElement);"
46  		+ " return ['--jc-pill-red-bg', '--jc-tab-bar-bg', '--jc-card-bg'].map(n => s.getPropertyValue(n).trim()); })()";
47  
48  	// The page's initial scheme, then its scheme and body background after the header toggle's Dark button is clicked.
49  	private static final String SCHEME_STATE = "(() => ({ scheme: document.documentElement.getAttribute('data-jc-scheme'),"
50  		+ " bg: getComputedStyle(document.body).backgroundColor }))()";
51  
52  	private static Map<String,Map<String,Object>> report;
53  
54  	@BeforeAll
55  	static void probe() throws Exception {
56  		var base = PetstoreTestServer.awaitReady(server);
57  		var cases = new ArrayList<Map<String,Object>>();
58  		for (var t : concat(STOCK, "custom"))
59  			cases.add(Map.of("name", t, "path", "/console/dev/themes?theme=" + t,
60  				"actions", List.of(Map.of("screenshot", "theme-" + t + ".png")),
61  				"queries", Map.of("tokens", "custom".equals(t) ? CUSTOM_TOKENS : TOKENS)));
62  		cases.add(Map.of("name", "scheme-dark", "path", "/console/store",
63  			"actions", List.of(
64  				Map.of("waitFor", "!!document.querySelector('.jc-scheme-toggle button[data-scheme=dark]')"),
65  				Map.of("click", ".jc-scheme-toggle button[data-scheme=dark]"),
66  				Map.of("screenshot", "scheme-dark.png")),
67  			"queries", Map.of("state", SCHEME_STATE)));
68  		report = run("themes", base, cases);
69  	}
70  
71  	private static List<String> concat(List<String> l, String x) {
72  		var out = new ArrayList<>(l);
73  		out.add(x);
74  		return out;
75  	}
76  
77  	@SuppressWarnings("unchecked") // tokens is a JSON array of strings.
78  	private static List<String> tokens(String theme) {
79  		return (List<String>) query(assertClean(report, theme), "tokens");
80  	}
81  
82  	@Test void a01_everyTokenResolvesInEveryTheme() {
83  		for (var t : concat(STOCK, "custom"))
84  			for (var v : tokens(t))
85  				assertNotEmpty(() -> "Theme '" + t + "' left a token unresolved: " + tokens(t), v);
86  	}
87  
88  	@Test void a02_stockThemesAreDistinct() {
89  		var seen = new HashSet<List<String>>();
90  		for (var t : STOCK)
91  			seen.add(tokens(t));
92  		assertSize(STOCK.size(), seen);
93  	}
94  
95  	@Test void a03_customLeafAndAliasResolve() {
96  		var c = tokens("custom");
97  		assertString("#fdeceb", c.get(0));  // --jc-pill-red-bg leaf override
98  		assertString(c.get(2), c.get(1));   // --jc-tab-bar-bg follows --jc-card-bg
99  	}
100 
101 	@SuppressWarnings("unchecked") // state is a JSON object of strings.
102 	@Test void a04_headerToggleSwitchesToDark() {
103 		var state = (Map<String,Object>) query(assertClean(report, "scheme-dark"), "state");
104 		assertString("dark", state.get("scheme"));
105 		// The dark body background is a dark colour, not the light page white.
106 		var bg = String.valueOf(state.get("bg"));
107 		var m = java.util.regex.Pattern.compile("rgba?\\((\\d+), (\\d+), (\\d+)").matcher(bg);
108 		org.junit.jupiter.api.Assertions.assertTrue(m.find(), () -> "unparseable body background: " + bg);
109 		org.junit.jupiter.api.Assertions.assertTrue(Integer.parseInt(m.group(1)) + Integer.parseInt(m.group(2)) + Integer.parseInt(m.group(3)) < 3 * 96, () -> "body background is not dark: " + bg);
110 	}
111 }
