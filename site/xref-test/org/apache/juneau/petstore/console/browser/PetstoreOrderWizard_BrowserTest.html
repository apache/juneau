32  /**
33   * The Orders "+ New order" wizard in the real browser: the ribbon item opens the compact first step, its Review
34   * button stacks the second step with the first step's inputs, and that step's Confirm places the order (it posts to
35   * the endpoint and method its child action declares), closes both steps, shows the success banner under the
36   * ribbon and redraws the table.
37   */
38  @EnabledIfSystemProperty(named=PetstoreBrowser.GATE, matches="true", disabledReason=PetstoreBrowser.DISABLED)
39  class PetstoreOrderWizard_BrowserTest extends TestBase {

44  	private static final String TOP = "(() => { const d = document.querySelectorAll('[role=dialog]'); return d[d.length - 1]; })()";

46  	private static final String NEW_ORDER = "[...document.querySelectorAll('.juneau-view-ribbon-btn')].find(b => b.getAttribute('aria-label') === '+ New order')";

48  	private static Map<String,Object> wait(String js) {
49  		return Map.of("waitFor", js, "timeoutMs", 5000);

52  	private static Map<String,Object> eval(String js) {
53  		return Map.of("evaluate", js);

60  	/** The table's total row count, read from its "1-25 of 2,001" summary. */
61  	private static final String TOTAL = "Number(/of (\\d+(?:,\\d{3})*)/.exec(document.body.textContent)[1].replace(/,/g, ''))";

65  	/** One full pass through the wizard at the given viewport, shipping on the given date. */
66  	private static Map<String,Object> wizard(String name, List<Integer> viewport, String shipDate) {
67  		return Map.of("name", name, "path", "/console/orders", "capture", "/console/orders", "viewport", viewport,
68  			"actions", List.of(
69  				wait(NEW_ORDER + " && " + ROWS + " > 0"),
70  				eval("window.__totalBefore = " + TOTAL),
71  				eval(NEW_ORDER + ".click()"),
72  				wait("document.querySelector('[data-juneau-form-field=review]')"),
73  				eval("window.__step1 = " + TOP + ".querySelector('h2, [class*=title]').textContent"),
74  				// Choose the second available pet and a ship date, as a user would.
75  				eval("(() => { const f = " + FIELD + "('petId'); f.selectedIndex = 1; window.__petId = f.value; window.__petName = f.options[1].textContent;"
76  					+ " f.dispatchEvent(new Event('input', {bubbles: true})); f.dispatchEvent(new Event('change', {bubbles: true}));"
77  					+ " const d = " + FIELD + "('shipDate'); d.value = '" + shipDate + "';"
78  					+ " d.dispatchEvent(new Event('input', {bubbles: true})); d.dispatchEvent(new Event('change', {bubbles: true})); })()"),
79  				eval(FIELD + "('review').click()"),
80  				wait("[...document.querySelectorAll('[role=dialog]')].length > 1 && " + TOP + ".querySelector('.juneau-view-dialog-confirm')"),
81  				eval("window.__step2 = " + TOP + ".textContent"),
82  				eval("window.__dialogs = document.querySelectorAll('[role=dialog]').length"),
83  				eval("window.__compact = document.querySelectorAll('.petstore-dialog-compact').length"),
84  				// The stacked step sits inside the viewport, not spilling off a narrow screen.
85  				eval("window.__rect = JSON.stringify(" + TOP + ".getBoundingClientRect()) + ' vs ' + window.innerWidth + 'x' + window.innerHeight"),
86  				eval("window.__offset = " + TOP + ".getBoundingClientRect().left - document.querySelectorAll('[role=dialog]')[0].getBoundingClientRect().left"),
87  				eval("window.__fits = (r => r.left >= 0 && r.right <= window.innerWidth + 1 && r.top >= 0 && r.bottom <= window.innerHeight + 1)(" + TOP + ".getBoundingClientRect())"),
88  				eval("window.__fontPx = parseFloat(getComputedStyle(document.querySelectorAll('[role=dialog]')[0]).fontSize)"),
89  				eval("window.__innerWidth = window.innerWidth"),
90  				eval("window.__firstRect = JSON.stringify(document.querySelectorAll('[role=dialog]')[0].getBoundingClientRect())"),
91  				eval("window.__firstFits = (r => r.left >= 0 && r.right <= window.innerWidth + 1 && r.top >= 0 && r.bottom <= window.innerHeight + 1)(document.querySelectorAll('[role=dialog]')[0].getBoundingClientRect())"),
92  				eval("window.__firstWidth = document.querySelectorAll('[role=dialog]')[0].getBoundingClientRect().width"),
93  				eval(TOP + ".querySelector('.juneau-view-dialog-confirm').click()"),
94  				wait("document.querySelector('.juneau-view-ribbon-action-outcome[data-state=success]') && " + TOTAL + " === window.__totalBefore + 1"),
95  				// Sort by id, newest first, so the placed order is the first row.  With the View Settings gear the sort controls are Juneau's .juneau-view-col-sort-icon; the Id column is first.
96  				eval("window.__orderId = /Placed order '(\\d+)'/.exec(document.body.textContent)[1]"),
97  				eval("document.querySelector('th .juneau-view-col-sort-icon').click()"),
98  				wait("document.querySelector('table tbody tr td')?.textContent.trim() === window.__orderId")),
99  			"queries", Map.ofEntries(
100 				Map.entry("step1", "window.__step1"), Map.entry("step2", "window.__step2"), Map.entry("compact", "window.__compact"),
101 				Map.entry("dialogs", "window.__dialogs"), Map.entry("fits", "window.__fits"), Map.entry("firstFits", "window.__firstFits"), Map.entry("firstWidth", "window.__firstWidth"), Map.entry("fontPx", "window.__fontPx"), Map.entry("innerWidth", "window.__innerWidth"), Map.entry("firstRect", "window.__firstRect"), Map.entry("rect", "window.__rect"), Map.entry("offset", "window.__offset"), Map.entry("petId", "window.__petId"),
102 				Map.entry("petName", "window.__petName"), Map.entry("totalBefore", "window.__totalBefore"), Map.entry("totalAfter", TOTAL),
103 				Map.entry("firstRow", "document.querySelector('table tbody tr').textContent"), Map.entry("orderId", "window.__orderId"),
104 				Map.entry("backdrops", "document.querySelectorAll('.juneau-view-dialog-backdrop').length"),
105 				Map.entry("banner", "document.querySelector('.juneau-view-ribbon-action-outcome[data-state=success]')?.textContent ?? null"),
106 				Map.entry("all", "document.body.textContent")));
107 	}

117 	/** The order this places is left in the throwaway test server's store; nothing else reads it. */
118 	@SuppressWarnings({
119 		"unchecked" // requests is a JSON array of {url, method, postData}.
120 	})
121 	private static void verify(String name, String shipDate, String shown) {
122 		var c = assertClean(report, name);
123 		assertContains("New order", String.valueOf(query(c, "step1")));
124 		// The second step reads back the first step's inputs: the chosen pet and the ship date.
125 		var step2 = String.valueOf(query(c, "step2"));
126 		assertContains("Review order", step2);
127 		assertContains(String.valueOf(query(c, "petName")), step2);
128 		assertContains(shipDate, step2);
129 		assertEquals(2, ((Number)query(c, "dialogs")).intValue(), "both steps are stacked");
130 		assertTrue((Integer)query(c, "compact") > 0);
131 		assertTrue(((Number)query(c, "offset")).doubleValue() > 0, "the second step is offset from the first, so the two read as a stack");
132 		assertEquals(Boolean.TRUE, query(c, "fits"), () -> "the stacked step must fit the viewport: " + query(c, "rect"));
133 		assertEquals(Boolean.TRUE, query(c, "firstFits"), () -> "the compact first step must fit the viewport: " + query(c, "firstRect"));
134 		var em = ((Number)query(c, "fontPx")).doubleValue();
135 		var width = ((Number)query(c, "firstWidth")).doubleValue();
136 		var offset = ((Number)query(c, "offset")).doubleValue();
137 		assertEquals(1.5 * em, offset, 1.0, "the stack offset is one 1.5em shift");
138 		if ("wide".equals(name))
139 			assertTrue(width >= 28 * em - 1, "wide geometry keeps the 28em minimum width: " + width);
140 		else
141 			assertEquals(((Number)query(c, "innerWidth")).doubleValue() - 3 * em, width, 2.0, "narrow: the compact step fills the viewport less the 3em slack");
142 		var posted = ((List<Map<String,Object>>)c.get("requests")).stream()
143 			.filter(r -> "POST".equals(r.get("method")) && String.valueOf(r.get("url")).endsWith("/console/orders")).toList();
144 		assertSize(1, posted);
145 		// A successful submit redraws the table, which re-queries it after the order was posted.
146 		var urls = ((List<Map<String,Object>>)c.get("requests")).stream().filter(r -> "POST".equals(r.get("method"))).map(r -> String.valueOf(r.get("url"))).toList();
147 		assertTrue(urls.lastIndexOf(posted.get(0).get("url").toString() + "/query") > urls.indexOf(posted.get(0).get("url").toString()));
148 		var body = new org.apache.juneau.marshall.collections.JsonMap(String.valueOf(posted.get(0).get("postData")));
149 		assertEquals(String.valueOf(query(c, "petId")), body.getMap("fields").getString("petId"));
150 		assertEquals(shipDate, body.getMap("fields").getString("shipDate"));
151 		// The committed write closes the whole stack - the first step too, not just the one that submitted - and
152 		// its outcome is the banner under the ribbon.
153 		assertEquals(0, ((Number)query(c, "backdrops")).intValue());
154 		assertContains("Placed order", String.valueOf(query(c, "banner")));
155 		// The new order is in the redrawn table: the total grew by one and the newest row is that order, still placed
156 		// and carrying the chosen pet and ship date.
157 		assertEquals(((Number)query(c, "totalBefore")).intValue() + 1, ((Number)query(c, "totalAfter")).intValue());
158 		var row = String.valueOf(query(c, "firstRow"));
159 		assertTrue(row.startsWith(String.valueOf(query(c, "orderId")) + query(c, "petId")), row);
160 		// A date-only value renders as the calendar day it names, in any browser time zone.
161 		assertTrue(row.matches(".*" + shown + ".*"), row);
162 		assertTrue(row.endsWith("PLACED"), row);
163 	}
