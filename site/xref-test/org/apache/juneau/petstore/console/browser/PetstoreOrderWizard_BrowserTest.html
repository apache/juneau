32  /**
33   * The Orders "+ New order" wizard in the real browser: the ribbon item opens the compact dialog, the single
34   * compact dialog takes a pet and a date, its Confirm places the order, closes the dialog, shows the success banner
35   * under the ribbon and redraws the table.
36   */

38  class PetstoreOrderWizard_BrowserTest extends TestBase {

43  	private static final String TOP = "(() => { const d = document.querySelectorAll('[role=dialog]'); return d[d.length - 1]; })()";
44  
45  	private static final String NEW_ORDER = "[...document.querySelectorAll('.juneau-view-ribbon-btn')].find(b => b.getAttribute('aria-label') === '+ New order')";
46  
47  	private static Map<String,Object> wait(String js) {
48  		return Map.of("waitFor", js, "timeoutMs", 5000);
49  	}
50  
51  	private static Map<String,Object> eval(String js) {
52  		return Map.of("evaluate", js);
53  	}

55  	private static final String FIELD = "(n => document.querySelector('[data-juneau-form-field=' + n + ']'))";
56  
57  	private static final String ROWS = "document.querySelectorAll('table tbody tr').length";
58  
59  	/**
60  	 * The table's total row count, read from its "1-25 of 2,001" summary element (0 until a summary is drawn).  Scoping to
61  	 * the summary matters: the page text runs it straight into the page-size menu ("0-0 of 0" + "25 rows"), so a regex
62  	 * over the whole body reads that placeholder as "of 025".
63  	 */
64  	private static final String TOTAL = "Number((/of (\\d+(?:,\\d{3})*)/.exec(document.querySelector('.juneau-view-pagingpill-info')?.textContent ?? '')?.[1] ?? '0').replace(/,/g, ''))";

68  	/** One full pass through the wizard at the given viewport, shipping on the given date. */
69  	private static Map<String,Object> wizard(String name, List<Integer> viewport, String shipDate) {
70  		return Map.of("name", name, "path", "/console/orders", "capture", "/console/orders", "viewport", viewport,
71  			"actions", List.of(
72  				// The empty "0-0 of 0" placeholder already has a row, so wait for the loaded total, not just a row.
73  				wait(NEW_ORDER + " && " + ROWS + " > 0 && " + TOTAL + " > 0"),
74  				eval("window.__totalBefore = " + TOTAL),
75  				eval(NEW_ORDER + ".click()"),
76  				wait("document.querySelector('[data-juneau-form-field=petId]')"),
77  				eval("window.__step1 = " + TOP + ".querySelector('h2, [class*=title]').textContent"),
78  				// Choose the second available pet and a ship date, as a user would.
79  				eval("(() => { const f = " + FIELD + "('petId'); f.selectedIndex = 1; window.__petId = f.value; window.__petName = f.options[1].textContent;"
80  					+ " f.dispatchEvent(new Event('input', {bubbles: true})); f.dispatchEvent(new Event('change', {bubbles: true}));"
81  					+ " const d = " + FIELD + "('shipDate'); d.value = '" + shipDate + "';"
82  					+ " d.dispatchEvent(new Event('input', {bubbles: true})); d.dispatchEvent(new Event('change', {bubbles: true})); })()"),
83  				wait(TOP + ".querySelector('.juneau-view-dialog-confirm')"),
84  				eval("window.__dateType = " + FIELD + "('shipDate').type"),
85  				eval("window.__fontPx = parseFloat(getComputedStyle(document.querySelectorAll('[role=dialog]')[0]).fontSize)"),
86  				eval("window.__innerWidth = window.innerWidth"),
87  				eval("window.__firstRect = JSON.stringify(document.querySelectorAll('[role=dialog]')[0].getBoundingClientRect())"),
88  				eval("window.__firstFits = (r => r.left >= 0 && r.right <= window.innerWidth + 1 && r.top >= 0 && r.bottom <= window.innerHeight + 1)(document.querySelectorAll('[role=dialog]')[0].getBoundingClientRect())"),
89  				eval("window.__firstWidth = document.querySelectorAll('[role=dialog]')[0].getBoundingClientRect().width"),
90  				eval(TOP + ".querySelector('.juneau-view-dialog-confirm').click()"),
91  				wait("document.querySelector('.juneau-view-ribbon-action-outcome[data-state=success]') && " + TOTAL + " === window.__totalBefore + 1"),
92  				// Sort by id, newest first, so the placed order is the first row.  With the View Settings gear the sort controls are Juneau's .juneau-view-col-sort-icon; the Id column is first.
93  				eval("window.__orderId = /Placed order '(\\d+)'/.exec(document.body.textContent)[1]"),
94  				eval("document.querySelector('th .juneau-view-col-sort-icon').click()"),
95  				wait("document.querySelector('table tbody tr td:nth-child(2)')?.textContent.trim() === window.__orderId")),
96  			"queries", Map.ofEntries(
97  				Map.entry("step1", "window.__step1"), Map.entry("dateType", "window.__dateType"),
98  				Map.entry("firstFits", "window.__firstFits"), Map.entry("firstWidth", "window.__firstWidth"), Map.entry("fontPx", "window.__fontPx"), Map.entry("innerWidth", "window.__innerWidth"), Map.entry("firstRect", "window.__firstRect"), Map.entry("petId", "window.__petId"),
99  				Map.entry("petName", "window.__petName"), Map.entry("totalBefore", "window.__totalBefore"), Map.entry("totalAfter", TOTAL),
100 				Map.entry("firstRow", "document.querySelector('table tbody tr').textContent"), Map.entry("orderId", "window.__orderId"),
101 				Map.entry("backdrops", "document.querySelectorAll('.juneau-view-dialog-backdrop').length"),
102 				Map.entry("banner", "document.querySelector('.juneau-view-ribbon-action-outcome[data-state=success]')?.textContent ?? null"),
103 				Map.entry("all", "document.body.textContent")));
104 	}

114 	/** The order this places is left in the throwaway test server's store; nothing else reads it. */
115 	@SuppressWarnings({
116 		"unchecked" // requests is a JSON array of {url, method, postData}.
117 	})
118 	private static void verify(String name, String shipDate, String shown) {
119 		var c = assertClean(report, name);
120 		assertContains("New order", String.valueOf(query(c, "step1")));
121 		assertTrue(! String.valueOf(query(c, "petName")).isEmpty(), "a pet was chosen");
122 		assertEquals("date", query(c, "dateType"), "the ship date is a native date control");
123 		assertTrue(! String.valueOf(query(c, "all")).contains("Review order"), "there is no review step");
124 		assertEquals(Boolean.TRUE, query(c, "firstFits"), () -> "the compact first step must fit the viewport: " + query(c, "firstRect"));
125 		var em = ((Number)query(c, "fontPx")).doubleValue();
126 		var width = ((Number)query(c, "firstWidth")).doubleValue();
127 		if ("wide".equals(name))
128 			assertTrue(width >= 28 * em - 1, "wide geometry keeps the 28em minimum width: " + width);
129 		else
130 			assertEquals(((Number)query(c, "innerWidth")).doubleValue() - 3 * em, width, 2.0, "narrow: the compact step fills the viewport less the 3em slack");
131 		var posted = ((List<Map<String,Object>>)c.get("requests")).stream()
132 			.filter(r -> "POST".equals(r.get("method")) && String.valueOf(r.get("url")).endsWith("/console/orders")).toList();
133 		assertSize(1, posted);
134 		// A successful submit redraws the table, which re-queries it after the order was posted.
135 		var urls = ((List<Map<String,Object>>)c.get("requests")).stream().filter(r -> "POST".equals(r.get("method"))).map(r -> String.valueOf(r.get("url"))).toList();
136 		assertTrue(urls.lastIndexOf(posted.get(0).get("url").toString() + "/query") > urls.indexOf(posted.get(0).get("url").toString()));
137 		var body = new org.apache.juneau.marshall.collections.JsonMap(String.valueOf(posted.get(0).get("postData")));
138 		assertEquals(String.valueOf(query(c, "petId")), body.getMap("fields").getString("petId"));
139 		assertEquals(shipDate, body.getMap("fields").getString("shipDate"));
140 		// The committed write closes the dialog and
141 		// its outcome is the banner under the ribbon.
142 		assertEquals(0, ((Number)query(c, "backdrops")).intValue());
143 		assertContains("Placed order", String.valueOf(query(c, "banner")));
144 		// The new order is in the redrawn table: the total grew by one and the newest row is that order, still placed
145 		// and carrying the chosen pet and ship date.
146 		assertEquals(((Number)query(c, "totalBefore")).intValue() + 1, ((Number)query(c, "totalAfter")).intValue());
147 		var row = String.valueOf(query(c, "firstRow"));
148 		assertTrue(row.startsWith(String.valueOf(query(c, "orderId")) + query(c, "petId")), row);
149 		// A date-only value renders as the calendar day it names, in any browser time zone.
150 		assertTrue(row.matches(".*" + shown + ".*"), row);
151 		assertTrue(row.endsWith("PLACED"), row);
152 	}
