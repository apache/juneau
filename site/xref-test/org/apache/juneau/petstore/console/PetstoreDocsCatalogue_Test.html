29  /**
30   * Fails when something the petstore docs tour (pages 19.01-19.18) documents disappears from the code, or when a
31   * FailLoud test is added without a docs row.
32   *
33   * <p>
34   * The manifest {@code docs-catalogue.txt} has two row forms (blank lines and {@code #} comments are ignored):
35   * <ul>
36   * 	<li>{@code page|<P-id>|<path>|<docs file>} for each documented console page.
37   * 	<li>{@code failloud|<test method>|<fixture file or ->|<asserted code or ->} for each {@link PetstoreFailLoud_Test} row.
38   * </ul>
39   *
40   * <p>
41   * Rules (a)-(e) always run.  Rule (f), the docs-checkout check, runs only when the system property
42   * {@code juneau.docs.dir} names a juneau-docs checkout.
43   */
44  class PetstoreDocsCatalogue_Test {
45  
46  	private static final String MANIFEST = "/org/apache/juneau/petstore/console/docs-catalogue.txt";
47  	private static final String FIXTURE_DIR = "org/apache/juneau/petstore/console/failloud/";
48  	private static final Path FAILLOUD_SOURCE = Path.of("src/test/java/org/apache/juneau/petstore/console/PetstoreFailLoud_Test.java");
49  	private static final String FAILLOUD_DOC = "19.18.PetstoreFailLoud.md";
50  
51  	record PageRow(String id, String path, String docsFile) {}
52  	record FailLoudRow(String method, String fixture, String code) {}
53  
54  	private static final List<PageRow> PAGE_ROWS = new ArrayList<>();
55  	private static final List<FailLoudRow> FAILLOUD_ROWS = new ArrayList<>();
56  
57  	@BeforeAll static void load() throws IOException {
58  		PAGE_ROWS.clear();
59  		FAILLOUD_ROWS.clear();
60  		try (var in = PetstoreDocsCatalogue_Test.class.getResourceAsStream(MANIFEST)) {
61  			assertNotNull(in, "Missing manifest " + MANIFEST);
62  			var n = 0;
63  			for (var line : new String(in.readAllBytes(), StandardCharsets.UTF_8).split("\n")) {
64  				n++;
65  				var l = line.strip();
66  				if (l.isEmpty() || l.startsWith("#"))
67  					continue;
68  				var f = l.split("\\|", -1);
69  				if (f.length == 4 && f[0].equals("page"))
70  					PAGE_ROWS.add(new PageRow(f[1], f[2], f[3]));
71  				else if (f.length == 4 && f[0].equals("failloud"))
72  					FAILLOUD_ROWS.add(new FailLoudRow(f[1], f[2], f[3]));
73  				else
74  					fail("Manifest line " + n + " is malformed: " + l);
75  			}
76  		}
77  	}

145 	// disabledReason (unlike a failed assumption) lands in the skipped test's message in the Surefire report.
146 	@Test
147 	@EnabledIfSystemProperty(named="juneau.docs.dir", matches=".+", disabledReason="juneau.docs.dir is not set; skipping the docs-checkout check")
148 	void c01_docsCheckoutMentionsEveryRow() throws IOException {
149 		var dir = System.getProperty("juneau.docs.dir");
150 		var topics = Path.of(dir, "pages", "topics");
151 		var problems = new ArrayList<String>();
152 		for (var r : PAGE_ROWS)
153 			if (! Files.isRegularFile(topics.resolve(r.docsFile())))
154 				problems.add("missing docs file " + r.docsFile() + " (" + r.id() + ")");
155 		var fl = topics.resolve(FAILLOUD_DOC);
156 		if (! Files.isRegularFile(fl)) {
157 			problems.add("missing docs file " + FAILLOUD_DOC);
158 		} else {
159 			var text = Files.readString(fl);
160 			for (var r : FAILLOUD_ROWS) {
161 				var id = r.method().substring(0, r.method().indexOf('_'));
162 				if (! text.contains(id))
163 					problems.add(FAILLOUD_DOC + " does not mention test id " + id);
164 				if (! r.code().equals("-") && ! text.contains(r.code()))
165 					problems.add(FAILLOUD_DOC + " does not mention code " + r.code());
166 			}
167 		}
168 		assertTrue(problems.isEmpty(), "Docs checkout out of step with docs-catalogue.txt: " + problems);
169 	}

175 	/** Removes {@code //} and block comments, leaving string and char literals intact. */
176 	static String stripComments(String src) {
177 		var sb = new StringBuilder();
178 		var i = 0;
179 		var n = src.length();
180 		while (i < n) {
181 			var c = src.charAt(i);
182 			if (c == '"' || c == '\'') {
183 				var j = i + 1;
184 				while (j < n && src.charAt(j) != c) {
185 					if (src.charAt(j) == '\\')
186 						j++;
187 					j++;
188 				}
189 				sb.append(src, i, Math.min(j + 1, n));
190 				i = j + 1;
191 			} else if (c == '/' && i + 1 < n && src.charAt(i + 1) == '/') {
192 				while (i < n && src.charAt(i) != '\n')
193 					i++;
194 			} else if (c == '/' && i + 1 < n && src.charAt(i + 1) == '*') {
195 				var end = src.indexOf("*/", i + 2);
196 				i = end < 0 ? n : end + 2;
197 			} else {
198 				sb.append(c);
199 				i++;
200 			}
201 		}
202 		return sb.toString();
203 	}
204 
205 	/** Statements (split on {@code ;}) that contain an {@code assert} call. */
206 	static List<String> assertionStatements(String code) {
207 		var out = new ArrayList<String>();
208 		for (var s : code.split(";"))
209 			if (s.contains("assert"))
210 				out.add(s);
211 		return out;
212 	}
