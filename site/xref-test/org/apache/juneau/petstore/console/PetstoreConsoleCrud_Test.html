28  /**
29   * Writes made on the console are visible to the {@code /petstore} API and the other way round: one shared store.
30   */
31  @SuppressWarnings({
32  	"resource" // MockRestClient is a no-op close.
33  })
34  class PetstoreConsoleCrud_Test extends TestBase {
35  
36  	private static MockRestClient json() {
37  		return MockRestClient.create(Host.class).noTrace().ignoreErrors().header("Accept", "application/json").build();
38  	}
39  
40  	@Test void a01_consolePetIsOnTheApi() throws Exception {
41  		var c = json();
42  		var id = new JsonMap(c.post("/console/pets", "{\"name\":\"Crudpet\",\"price\":3}").contentType("application/json").run().assertStatus(200).getContent().asString()).getMap("row").getLong("id");
43  		c.get("/petstore/pets/" + id).run().assertStatus(200).assertContent().isContains("Crudpet");
44  		c.delete("/petstore/pets/" + id).run().assertStatus(200);
45  		c.get("/console/pets/" + id + "/row").run().assertStatus(404);
46  	}
47  
48  	@Test void b01_consoleApprovedOrderReadsBackApproved() throws Exception {
49  		var c = json();
50  		var id = new JsonMap(c.post("/petstore/orders", "{\"petId\":1,\"status\":\"PLACED\"}").contentType("application/json").run().assertStatus(200).getContent().asString()).getLong("id");
51  		c.post("/console/orders/" + id + "/approve", "").run().assertStatus(200);
52  		c.get("/petstore/orders/" + id).run().assertStatus(200).assertContent().isContains("APPROVED");
53  	}
54  
55  	@Test void c01_passwordIsKeptAcrossAConsoleEdit() throws Exception {
56  		var c = json();
57  		c.post("/petstore/users", "{\"username\":\"crud1\",\"firstName\":\"C\",\"password\":\"kept\"}").contentType("application/json").run().assertStatus(200);
58  		c.put("/console/users/crud1", "{\"firstName\":\"Changed\"}").contentType("application/json").run().assertStatus(200);
59  		c.get("/petstore/users/crud1").run().assertContent().isContains("Changed");
60  		assertBean(store().getUser("crud1"), "password", "kept");
61  	}
62  
63  	private static String body(MockRestClient c, String path) throws Exception {
64  		return c.get(path).run().assertStatus(200).getContent().asString();
65  	}
66  
67  	@Test void c02_apiNeverReturnsPasswordButAcceptsIt() throws Exception {
68  		var c = json();
69  		var created = c.post("/petstore/users", "{\"username\":\"crud2\",\"firstName\":\"D\",\"password\":\"hunter2\"}").contentType("application/json").run().assertStatus(200).getContent().asString();
70  		assertNotContains("hunter2", created);
71  		assertBean(store().getUser("crud2"), "password", "hunter2");
72  		c.get("/petstore/users/crud2").run().assertStatus(200).assertContent().isContains("crud2");
73  		assertNotContains("password", body(c, "/petstore/users/crud2"));
74  		assertNotContains("hunter2", body(c, "/petstore/users/crud2"));
75  		c.get("/petstore/users").run().assertStatus(200).assertContent().isContains("mwatson");
76  		for (var x : new String[]{"hunter2", "123456", "password"})
77  			assertNotContains(x, body(c, "/petstore/users"));
78  		c.put("/petstore/users/crud2", "{\"firstName\":\"E\",\"password\":\"hunter3\"}").contentType("application/json").run().assertStatus(200).assertContent().isNotContains("hunter3");
79  		assertBean(store().getUser("crud2"), "password", "hunter3");
80  	}
81  
82  	@Test void c03_apiPasswordIsHiddenInEveryFormat() throws Exception {
83  		var c = MockRestClient.create(Host.class).noTrace().ignoreErrors().build();
84  		// Formats the users endpoint does not serve (e.g. text/ini answers 500, text/prototext 406) fail the same way
85  		// without this patch, so only 200 responses are checked; the password VALUE must never appear in any of them.
86  		var checked = 0;
87  		for (var accept : new String[]{"application/json", "text/json5", "application/jsonl", "text/html", "text/html+stripped", "text/xml", "text/xml+soap", "text/uon", "application/x-www-form-urlencoded", "text/plain", "text/yaml", "text/csv", "text/markdown", "octal/msgpack", "application/cbor", "application/bson", "text/ini", "text/prototext"}) {
88  			for (var path : new String[]{"/petstore/users/mwatson", "/petstore/users"}) {
89  				var r = c.get(path).accept(accept).run();
90  				if (r.getStatusCode() != 200)
91  					continue;
92  				checked++;
93  				var body = r.getContent().asString();
94  				assertContains("mwatson", body);
95  				assertNotContains("123456", body);
96  			}
97  		}
98  		assertEquals(33, checked, "of 36 format/path combinations, exactly 33 are served (200)");
99  	}

101 	@Test void c04_getThenPutKeepsThePassword() throws Exception {
102 		var c = json();
103 		c.post("/petstore/users", "{\"username\":\"crud4\",\"firstName\":\"F\",\"password\":\"keepme\"}").contentType("application/json").run().assertStatus(200);
104 		var got = body(c, "/petstore/users/crud4");
105 		c.put("/petstore/users/crud4", got).contentType("application/json").run().assertStatus(200);
106 		assertBean(store().getUser("crud4"), "password", "keepme");
107 		c.put("/petstore/users/crud4", "{\"firstName\":\"F\",\"password\":\" \"}").contentType("application/json").run().assertStatus(200);
108 		assertBean(store().getUser("crud4"), "password", "keepme");
109 		c.put("/petstore/users/crud4", "{\"firstName\":\"F\",\"password\":\"fresh\"}").contentType("application/json").run().assertStatus(200);
110 		assertBean(store().getUser("crud4"), "password", "fresh");
111 	}
