40  /**
41   * Broken templates fail loudly at render time, with the console's own message.  Rows whose upstream E-code has not
42   * landed are {@code @Disabled} with the gate that blocks them; none is invented here.
43   *
44   * <p>
45   * The live rows assert the exact production message text, because these messages carry no E-code to assert on.
46   * One row stays disabled: a08 (the production chrome has a fixed nav, so no petstore fixture can duplicate a sibling;
47   * the console bridge tests cover it).
48   */

102 	 * Renders {@code failloud/chrome-page.ftlh} inside a small test chrome that includes one fixture: the directives

109 		@Bean public FreemarkerMixin freemarker() {

113 		@RestGet(path="/c/{name}")

121 		}

220 	@Test void b01_scriptInPetNameIsEscaped() throws Exception {
221 		var c = PetstoreConsoleFixture.client();
222 		var hostile = "</script><script>alert(1)</script>";
223 		var created = c.post("/console/pets", JsonMap.of("name", hostile, "price", 1)).contentType("application/json").header("Accept", "application/json").run().assertStatus(200).getContent().asString();
224 		var id = new JsonMap(created).getMap("row").getLong("id");
225 		try {
226 			var body = c.get("/console/pets/" + id + "/summary").accept("text/html").run().assertStatus(200).getContent().asString();
227 			assertNotContains(hostile, body);
228 			assertContains("alert(1)", body);
229 		} finally {
230 			// The store is shared across the fork; leave it as found so count-based tests are not skewed.
231 			c.delete("/console/pets/" + id).header("Accept", "application/json").run();
232 		}
233 	}
