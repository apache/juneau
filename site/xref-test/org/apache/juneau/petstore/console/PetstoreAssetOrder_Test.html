29  /**
30   * Build-time guard for the stylesheet/script asset order the console chrome emits for every petstore page.
31   *
32   * <p>
33   * Ported from the deleted {@code AssetLoadOrderBand_Test}.  The console chrome (not this sample) owns the emitted
34   * order, and it differs from the five-position band that test enforced: the chrome stylesheet and theme are linked
35   * <i>first</i> (their {@code html:root} override block makes the cascade order irrelevant), then vendor, then
36   * {@code juneau-views.css}, then widget-layer stylesheets, then any page-local {@code <style>}:
37   *
38   * <pre>
39   * 1. console chrome.css and theme            (emitted first by the chrome)
40   * 2. vendor stylesheets                 (the DataTables WebJar stylesheet)
41   * 3. juneau-views.css                        (the views base layer)
42   * 4. first-party widget stylesheets          (e.g. juneau-datatables.css, juneau-config.css)
43   * 5. page-local &lt;style&gt;
44   * </pre>
45   *
46   * <p>
47   * The {@code b0x} methods are synthetic: they prove the guard can fail.
48   */
49  @SuppressWarnings({
50  	"resource" // MockRestClient is a no-op close.
51  })
52  class PetstoreAssetOrder_Test extends TestBase {
53  
54  	private static final int CHROME_THEME = 1;
55  	private static final int VENDOR = 2;
56  	private static final int VIEWS_BASE = 3;
57  	private static final int WIDGET = 4;
58  	private static final int PAGE_LOCAL = 5;
59  
60  	private static final Pattern LINK_HREF = Pattern.compile("<link\\s+rel=\"stylesheet\"\\s+href=\"([^\"]*)\"");
61  	private static final Pattern SCRIPT_SRC = Pattern.compile("<script\\s+src=\"([^\"]*)\"");
62  	private static final Pattern STYLE_TAG = Pattern.compile("<style>");
63  
64  	/** One classified asset found in an emitted page, in document order. */
65  	private record BandHit(int position, String label, int index) {}
66  
67  	/** Classifies a console stylesheet onto the band; unknown assets fail, so new ones get placed. */
68  	private static int cssBandPosition(String href) {
69  		if (href.contains("/juneau-console/chrome.css") || href.contains("/juneau-console/themes/"))
70  			return CHROME_THEME;
71  		if (href.contains("dataTables.dataTables") || href.contains("/webjars/"))
72  			return VENDOR;
73  		if (href.contains("/juneau-views.css"))
74  			return VIEWS_BASE;
75  		if (href.contains("/juneau-datatables.css") || href.contains("/juneau-widgets") || href.contains("/juneau-calendar.css") || href.contains("/juneau-config.css") || href.contains("/juneau-terminal.css"))
76  			return WIDGET;
77  		throw new AssertionError("Unclassified stylesheet - the band doesn't name this asset: " + href);
78  	}
79  
80  	private static List<BandHit> cssBandSequence(String html) {
81  		var out = new ArrayList<BandHit>();
82  		var links = LINK_HREF.matcher(html);
83  		while (links.find())
84  			out.add(new BandHit(cssBandPosition(links.group(1)), links.group(1), links.start()));
85  		var style = STYLE_TAG.matcher(html);
86  		if (style.find())
87  			out.add(new BandHit(PAGE_LOCAL, "<style> (page-local)", style.start()));
88  		out.sort(Comparator.comparingInt(BandHit::index));
89  		return out;
90  	}
91  
92  	private static List<BandHit> scriptSequence(String html, Map<String,Integer> roleOf) {
93  		var out = new ArrayList<BandHit>();
94  		var scripts = SCRIPT_SRC.matcher(html);
95  		while (scripts.find()) {
96  			var src = scripts.group(1);
97  			for (var e : roleOf.entrySet())
98  				if (src.contains(e.getKey()))
99  					out.add(new BandHit(e.getValue(), src, scripts.start()));
100 		}
101 		out.sort(Comparator.comparingInt(BandHit::index));
102 		return out;
103 	}
104 
105 	private static void assertBandOrder(List<BandHit> hits, String pageLabel) {
106 		for (var i = 1; i < hits.size(); i++) {
107 			var prev = hits.get(i - 1);
108 			var cur = hits.get(i);
109 			assertTrue(prev.position() <= cur.position(),
110 				() -> pageLabel + ": asset order violates the band - \"" + cur.label() + "\" (position "
111 					+ cur.position() + ") must not load before \"" + prev.label() + "\" (position "
112 					+ prev.position() + ")");
113 		}
114 	}

116 	@Test void a01_storeCssBandOrder() throws Exception {
117 		var hits = cssBandSequence(page(client(), "/console/store"));
118 		assertTrue(hits.size() >= 3, () -> "expected the chrome, theme and views stylesheets: " + hits);
119 		assertBandOrder(hits, "P1 Store");
120 	}

123 		var c = client();
124 		for (var path : PetstorePages_ContractTest.PAGE_PATHS)
125 			assertBandOrder(cssBandSequence(page(c, path)), path);
126 	}

128 	/** The chrome script loads first, then the views layer, then the region runtime, then the helpers that reuse both. */
129 	@Test void a03_scriptOrder() throws Exception {
130 		var hits = scriptSequence(page(client(), "/console/store"),
131 			Map.of("/juneau-console.js", 1, "/juneau-views.js", 2, "/juneau-regions.js", 3, "/juneau-helpers.js", 4));
132 		assertSize(4, hits);
133 		assertBandOrder(hits, "P1 script order (console, views, regions, helpers)");
134 	}

136 	/** The head carries the scheme script first, then the chrome, the dark layer, the palette and the page-local style. */
137 	@Test void a04_headOrderSchemeFirst() throws Exception {
138 		var html = page(client(), "/console/dev/themes?theme=custom");
139 		var script = html.indexOf("juneau-console:scheme");
140 		var chrome = html.indexOf("/juneau-console/chrome.css");
141 		var dark = html.indexOf("/juneau-console/themes/juneau-theme-dark.css");
142 		var palette = html.indexOf("/juneau-console/themes/juneau-theme-light-brown.css");
143 		var style = html.indexOf("<style");
144 		assertTrue(script >= 0 && chrome >= 0 && dark >= 0 && palette >= 0 && style >= 0,
145 			() -> "missing head piece: script=" + script + " chrome=" + chrome + " dark=" + dark + " palette=" + palette + " style=" + style);
146 		assertTrue(script < chrome && chrome < dark && dark < palette && palette < style,
147 			() -> "head order must be script, chrome, dark, palette, style: " + script + "," + chrome + "," + dark + "," + palette + "," + style);
148 	}

150 	@Test void b01_syntheticCssBandViolation_isCaught() {
151 		var reversed = "<link rel=\"stylesheet\" href=\"/juneau-config.css\">"
152 			+ "<link rel=\"stylesheet\" href=\"/juneau-views.css\">";
153 		var seq = cssBandSequence(reversed);
154 		var err = assertThrows(AssertionError.class, () -> assertBandOrder(seq, "synthetic"));
155 		assertTrue(err.getMessage().contains("juneau-views.css"), err.getMessage());
156 	}

158 	@Test void b02_syntheticScriptBandViolation_isCaught() {
159 		var reversed = "<script src=\"/juneau-regions.js\"></script><script src=\"/juneau-views.js\"></script>";
160 		var hits = scriptSequence(reversed, Map.of("/juneau-views.js", 2, "/juneau-regions.js", 3));
161 		var err = assertThrows(AssertionError.class, () -> assertBandOrder(hits, "synthetic"));
162 		assertTrue(err.getMessage().contains("juneau-views.js"), err.getMessage());
163 	}

165 	@Test void b03_conformingSyntheticFullBand_isNotFlagged() {
