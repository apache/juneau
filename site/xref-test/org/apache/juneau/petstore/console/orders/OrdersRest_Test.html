50  	private static final String ALL = "{\"draw\":1,\"length\":-1,\"columns\":[{\"data\":\"id\"},{\"data\":\"status\"},{\"data\":\"shipDate\"}]}";
51  	/** Newest first, so the server's page cap keeps the orders a test just created however many the shared store holds. */
52  	private static final String NEWEST = "{\"draw\":1,\"length\":-1,\"order\":[{\"column\":0,\"dir\":\"desc\"}],\"columns\":[{\"data\":\"id\"},{\"data\":\"status\"},{\"data\":\"shipDate\"}]}";
53  	private static final String PLACED_ONLY = "{\"draw\":1,\"length\":-1,\"columns\":[{\"data\":\"id\"},{\"data\":\"status\",\"search\":{\"value\":\"$eq(PLACED)\"}},{\"data\":\"shipDate\"}]}";
54  
55  	private static JsonMap queryOn(String qs, String body) throws Exception {
56  		return new JsonMap(client().post("/console/orders/query" + qs, body).contentType("application/json").run().assertStatus(200).getContent().asString());
57  	}

59  	private static Set<Long> ids(JsonMap r) {
60  		var s = new HashSet<Long>();
61  		for (var row : r.getList("data").elements(JsonMap.class))
62  			s.add(row.getLong("id"));
63  		return s;
64  	}

66  	@Test void a03_dayKeepsOnlyThatDaysOrders() throws Exception {
67  		var c = client();
68  		var in = newOrderOn(c, "PLACED", "2031-03-04");
69  		var before = newOrderOn(c, "PLACED", "2031-03-03");
70  		var after = newOrderOn(c, "PLACED", "2031-03-05");
71  		var r = queryOn("?day=2031-03-04", ALL);
72  		var ids = ids(r);
73  		Assertions.assertTrue(ids.contains(in), "the day's own order is shown");
74  		Assertions.assertFalse(ids.contains(before), "the previous day is before the window");
75  		Assertions.assertFalse(ids.contains(after), "the next day is not in the day");
76  		for (var row : r.getList("data").elements(JsonMap.class))
77  			assertBean(row, "shipDate", "2031-03-04");
78  		Assertions.assertEquals(ids.size(), r.getInt("recordsFiltered"), "the count is of the filtered orders");
79  	}

82  		var c = client();
83  		var utc = newOrderOn(c, "PLACED", "2031-06-04T12:00:00Z");
84  		var offset = newOrderOn(c, "PLACED", "2031-06-04T23:30:00+02:00");
85  		var local = newOrderOn(c, "PLACED", "2031-06-04T08:15:00");
86  		var next = newOrderOn(c, "PLACED", "2031-06-05T00:00:00Z");
87  		var ids = ids(queryOn("?day=2031-06-04", ALL));
88  		Assertions.assertTrue(ids.containsAll(Set.of(utc, offset, local)), "datetime ship dates on the day are kept");
89  		Assertions.assertFalse(ids.contains(next), "a datetime on the next day is not");
90  		Assertions.assertEquals(ids, ids(queryOn("?day=%202031-06-04%20", ALL)), "the day param is trimmed");
91  	}

105 		var c = client();
106 		newOrderOn(c, "PLACED", "2031-05-04");
107 		newOrderOn(c, "PLACED", "2031-05-05");
108 		var everything = queryOn("", ALL);
109 		var all = ids(everything);
110 		Assertions.assertTrue(queryOn("?day=2031-05-04", ALL).getInt("recordsFiltered") < everything.getInt("recordsFiltered"), "a real day does narrow");
111 		for (var bad : List.of("garbage", "2031-02-30", "2031-5-4", "2031-05-04T00:00:00Z", "", "%20", "20310504"))
112 			try {
113 				Assertions.assertEquals(all, ids(queryOn("?day=" + bad, ALL)), "day '" + bad + "' is ignored");
114 			} catch (RuntimeException e) {
115 				throw new AssertionError("day '" + bad + "': " + e.getMessage(), e);
116 			}

120 		var c = client();
121 		var before = newOrderOn(c, "PLACED", "2032-03-09");
122 		var first = newOrderOn(c, "PLACED", "2032-03-10");
123 		var middle = newOrderOn(c, "PLACED", "2032-03-12T15:00:00Z");
124 		var last = newOrderOn(c, "PLACED", "2032-03-14");
125 		var after = newOrderOn(c, "PLACED", "2032-03-15");
126 		var r = queryOn("?start=2032-03-10&end=2032-03-14", ALL);
127 		var ids = ids(r);
128 		Assertions.assertTrue(ids.containsAll(Set.of(first, middle, last)), "both ends and the inside are kept");
129 		Assertions.assertFalse(ids.contains(before), "a day before start is out");
130 		Assertions.assertFalse(ids.contains(after), "a day after end is out");
131 		Assertions.assertEquals(ids.size(), r.getInt("recordsFiltered"), "the count is of the filtered orders");
132 	}

135 		var c = client();
136 		var early = newOrderOn(c, "PLACED", "2032-04-01");
137 		var late = newOrderOn(c, "PLACED", "2032-04-30");

141 		var to = queryOn("?end=2032-04-15", NEWEST);

144 		Assertions.assertTrue(to.getInt("recordsFiltered") > 1);
145 		Assertions.assertTrue(to.getInt("recordsFiltered") < queryOn("", NEWEST).getInt("recordsFiltered"), "the bound narrows");

276 	/** Creates a pet of our own through the console API (sold when asked); the caller removes it with {@link #dropPet}. */
277 	private static long newPet(MockRestClient c, boolean sold) throws Exception {
278 		var created = new JsonMap(c.post("/console/pets", "{\"name\":\"Wizpet\",\"price\":3}").contentType("application/json").run().assertStatus(200).getContent().asString());
279 		var id = created.getMap("row").getLong("id");
280 		if (sold)
281 			c.post("/console/pets/" + id + "/sell", "").run().assertStatus(200);
282 		return id;
283 	}

286 	private static void dropPet(MockRestClient c, long id) throws Exception {
287 		c.delete("/console/pets/" + id).run().assertStatus(200);
288 	}
