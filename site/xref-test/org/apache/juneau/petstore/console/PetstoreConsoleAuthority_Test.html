37  /**
38   * The {@value PetstoreConsoleBoundary#AUTHORITY_PROPERTY} setting on a real server: set before the first console
39   * request, it replaces {@code localhost:<port>} as the one {@code Host} (and {@code Origin}) the console accepts.
40   */
41  @SuppressWarnings({
42  	"java:S2925", // start() polls a starting server; without a back-off a refused connection would busy-spin.
43  	"java:S8692" // start() polls a real HTTP server against a wall-clock deadline.
44  })
45  class PetstoreConsoleAuthority_Test extends TestBase {
46  
47  	@RegisterExtension
48  	static MicroserviceTestFixture server = PetstoreTestServer.fixture();
49  
50  	private static final HttpClient HTTP = HttpClient.newBuilder().connectTimeout(Duration.ofSeconds(5)).followRedirects(HttpClient.Redirect.NEVER).build();
51  	private static final Pattern TOKEN = Pattern.compile("<body[^>]*\\sdata-juneau-csrf=\"([^\"]+)\"");
52  
53  	private static String base;
54  
55  	@BeforeAll
56  	static void start() throws Exception {
57  		// The server has bound but served no console request yet, so the boundary has not fixed its authority.
58  		base = "http://127.0.0.1:" + server.getPort();
59  		System.setProperty(PetstoreConsoleBoundary.AUTHORITY_PROPERTY, "127.0.0.1:" + server.getPort());
60  		var deadline = Instant.now().plusSeconds(30);
61  		for (var r = page(base); r == null || r.statusCode() != 200; r = page(base)) {
62  			if (Instant.now().isAfter(deadline))
63  				throw new IllegalStateException("console did not answer 200 at " + base + ": "
64  					+ (r == null ? "no response" : r.statusCode() + " " + r.headers().firstValue("X-Request-Boundary").orElse("") + " " + r.body()));
65  			Thread.sleep(250);
66  		}
67  	}
68  
69  	@AfterAll
70  	static void stop() {
71  		System.clearProperty(PetstoreConsoleBoundary.AUTHORITY_PROPERTY);
72  	}
73  
74  	private static HttpResponse<String> page(String root) throws Exception {
75  		try {
76  			return HTTP.send(HttpRequest.newBuilder(URI.create(root + "/console/store")).timeout(Duration.ofSeconds(20)).header("Accept", "text/html").GET().build(), BodyHandlers.ofString());
77  		} catch (IOException e) {  // still starting
78  			return null;
79  		}
80  	}
81  
82  	private static long availablePet() throws Exception {
83  		var list = new JsonList(HTTP.send(HttpRequest.newBuilder(URI.create(base + "/petstore/pets")).header("Accept", "application/json").GET().build(), BodyHandlers.ofString()).body());
84  		for (var o : list)
85  			if ("AVAILABLE".equals(((JsonMap)o).getString("status")))
86  				return ((JsonMap)o).getLong("id");
87  		throw new IllegalStateException("No seeded AVAILABLE pet");
88  	}
89  
90  	@Test void a01_theConfiguredAuthorityIsServed() throws Exception {
91  		var r = page(base);
92  		assertEquals(200, r.statusCode());
93  		assertTrue(TOKEN.matcher(r.body()).find(), "no data-juneau-csrf on <body>");
94  	}
95  
96  	@Test void a02_theDefaultAuthorityIsNowRefused() throws Exception {
97  		var r = page("http://localhost:" + server.getPort());
98  		assertEquals(421, r.statusCode(), r.body());
99  		assertString("HOST_MISMATCH", r.headers().firstValue("X-Request-Boundary").orElse(null));
100 	}
101 
102 	@Test void a03_writesExpectTheConfiguredOrigin() throws Exception {
103 		var m = TOKEN.matcher(page(base).body());
104 		assertTrue(m.find());
105 		var id = availablePet();
106 		var w = HttpRequest.newBuilder(URI.create(base + "/console/pets/" + id + "/sell")).timeout(Duration.ofSeconds(20))
107 			.header("Content-Type", "application/json").header("Accept", "application/json").header("X-Csrf-Token", m.group(1));
108 		var wrong = HTTP.send(w.copy().header("Origin", "http://localhost:" + server.getPort()).POST(BodyPublishers.ofString("{}")).build(), BodyHandlers.ofString());
109 		assertEquals(403, wrong.statusCode(), wrong.body());
110 		assertString("ORIGIN_MISMATCH", wrong.headers().firstValue("X-Request-Boundary").orElse(null));
111 		var right = HTTP.send(w.copy().header("Origin", base).POST(BodyPublishers.ofString("{}")).build(), BodyHandlers.ofString());
112 		assertEquals(200, right.statusCode(), right.body());
113 	}
114 }
