26  @SuppressWarnings({
27  	"resource" // MockRestClient is a no-op close.
28  })
29  class PetstoreConsoleShell_Test extends TestBase {
30  
31  	@Test void a01_storeRendersInChrome() throws Exception {
32  		assertPage(page(client(), "/console/store"))
33  			.isValid()
34  			.hasContractVersion("1")
35  			.hasActiveNav("store")
36  			.hasHeaderTitle("Juneau Petstore")
37  			.hasFooterText("Apache Juneau petstore sample");
38  	}
39  
40  	@Test void a02_staticNavTree() throws Exception {
41  		assertPage(page(client(), "/console/store"))
42  			.hasNavPath("store").hasNavPath("pets").hasNavPath("orders").hasNavPath("users").hasNavPath("ops").hasNavPath("dev").hasNavPath("about")
43  			.hasNavChildren("pets", "sold", "changes")
44  			.hasNavChildren("ops", "jobs", "audit")
45  			.hasNavHref("pets/sold", "/console/pets/sold");
46  	}
47  
48  	@Test void a03_consoleRootRedirectsToStore() throws Exception {
49  		rawClient().get("/console").run().assertStatus(303).assertHeader("Location").isMatches("*/console/store");
50  	}
51  
52  	@Test void a04_unknownConsolePage404() throws Exception {
53  		client().get("/console/nope").accept("text/html").run().assertStatus(404);
54  	}
55  
56  	@Test void a05_everyNavLinkIs200() throws Exception {
57  		var c = client();
58  		var nav = assertPage(page(c, "/console/store"));
59  		var links = new java.util.LinkedHashMap<String,String>();
60  		links.put("store", "/console/store");
61  		links.put("ops/jobs", "/console/ops/jobs");
62  		links.put("ops/audit", "/console/ops/audit");
63  		for (var f : new String[]{"html", "freemarker", "mustache", "react"})
64  			links.put("dev/flavors/" + f, "/console/dev/flavors/" + f);
65  		links.put("pets", "/console/pets");
66  		links.put("pets/sold", "/console/pets/sold");
67  		links.put("orders", "/console/orders");
68  		links.put("users", "/console/users");
69  		links.put("dev/cards", "/console/dev/cards");
70  		links.forEach(nav::hasNavHref);
71  		for (var href : links.values())
72  			c.get(href).accept("text/html").run().assertStatus(200);
73  	}
74  
75  	@Test void a06_customCardsPageIsNotAComingSoonStub() throws Exception {
76  		assertNotContains("Coming soon", page(client(), "/console/dev/cards"));
77  	}
78  
79  	@Test void a07_missingStoreBeanFailsLoudly() {
80  		var r = new StoreRest();
81  		var e = org.junit.jupiter.api.Assertions.assertThrows(IllegalStateException.class, r::store);
82  		assertContains("No PetStore bean is wired", e.getMessage());
83  	}
84  }
