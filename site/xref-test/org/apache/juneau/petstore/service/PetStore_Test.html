33  	@Test void a01_seededPetsLoaded() {

38  	@Test void a02_seededOrdersLoaded() {
39  		var s = new PetStore();
40  		assertEquals(3, s.getOrders().size());

43  	@Test void a03_seededUsersLoaded() {
44  		var s = new PetStore();
45  		assertEquals(3, s.getUsers().size());
46  	}
47  
48  	@Test void a04_seededPetByIdHasCorrectShape() {
49  		var s = new PetStore();
50  		// First seeded pet — id auto-assigned to 1.
51  		assertBean(s.getPet(1L), "name,species,price,status", "Mr. Frisky,CAT,39.99,AVAILABLE");
52  	}
53  
54  	@Test void a05_seededOrderById() {
55  		var s = new PetStore();
56  		assertBean(s.getOrder(101L), "id,petId,shipDate,status", "101,101,2018-01-01,PLACED");
57  	}
58  
59  	@Test void a06_seededUserByUsername() {
60  		var s = new PetStore();
61  		assertBean(s.getUser("mwatson"), "username,firstName,lastName,userStatus", "mwatson,Marie,Watson,ACTIVE");
62  	}

64  	//------------------------------------------------------------------------------------------------------------------
65  	// b — pets CRUD
66  	//------------------------------------------------------------------------------------------------------------------
67  
68  	@Test void b01_createPet_assignsId() {
69  		var s = new PetStore();
70  		var p = s.createPet(new Pet().setName("Newpet").setSpecies(Species.DOG).setPrice(10f).setStatus(PetStatus.AVAILABLE));
71  		assertNotEquals(0L, p.getId());
72  		assertEquals(p, s.getPet(p.getId()));
73  	}
74  
75  	@Test void b02_updatePet_replacesContent() {
76  		var s = new PetStore();
77  		var p = s.getPet(1L).setName("Renamed");
78  		s.updatePet(p);
79  		assertEquals("Renamed", s.getPet(1L).getName());
80  	}
81  
82  	@Test void b03_updatePet_unknownId_throws() {
83  		var s = new PetStore();
84  		var ghost = new Pet().setId(99999L).setName("Ghost").setSpecies(Species.CAT).setStatus(PetStatus.AVAILABLE);
85  		assertThrows(PetstoreNotFoundException.class, () -> s.updatePet(ghost));
86  	}
87  
88  	@Test void b04_deletePet_removes() {
89  		var s = new PetStore();
90  		s.deletePet(1L);
91  		assertNull(s.getPet(1L));
92  	}
93  
94  	@Test void b05_deletePet_unknownId_throws() {
95  		var s = new PetStore();
96  		assertThrows(PetstoreNotFoundException.class, () -> s.deletePet(99999L));
97  	}

143 	}
