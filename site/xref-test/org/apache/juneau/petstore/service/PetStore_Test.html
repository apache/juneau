158 	private static final Instant T0 = Instant.parse("2026-10-01T12:00:00Z");
159 
160 	private static PetStore fixedStore() {
161 		return new PetStore(Clock.fixed(T0, ZoneOffset.UTC));
162 	}
163 
164 	private static Pet newPet(String name) {
165 		return new Pet().setName(name).setSpecies(Species.DOG).setPrice(10f).setStatus(PetStatus.AVAILABLE);
166 	}

172 	@Test void e02_createPet_apiActorByDefault() {
173 		var s = fixedStore();
174 		var p = s.createPet(newPet("Newpet"));
175 		assertBeans(s.getAudit(), "id,at,actor,entity,entityId,action,detail",
176 			"1,2026-10-01T12:00:00Z,api,Pet," + p.getId() + ",CREATE,Newpet");

220 		var s = fixedStore();

223 		s.createPet(newPet("B"));
224 		assertSize(1, snap);

229 	@Test void e09_load_preservesIdsAndBumpsCounters() {
230 		var s = fixedStore();
231 		s.load(
232 			List.of(new Pet().setId(700).setName("Loaded").setSpecies(Species.CAT).setStatus(PetStatus.SOLD)),
233 			List.of(new Order().setId(900).setPetId(700).setStatus(OrderStatus.PLACED)),
234 			List.of(new User().setUsername("loaded")),
235 			List.of(new AuditEntry().setId(40).setAt(T0).setActor("api").setEntity("Pet").setEntityId("700").setAction("CREATE")));
236 		assertBean(s.getPet(700), "id,name", "700,Loaded");
237 		assertBean(s.getOrder(900), "id,petId", "900,700");
238 		assertBean(s.getUser("loaded"), "username", "loaded");
239 		assertEquals(701L, s.createPet(newPet("Next")).getId());
240 		assertEquals(901L, s.createOrder(new Order().setPetId(1).setStatus(OrderStatus.PLACED)).getId());
241 		assertBeans(s.getAudit(), "id", "40", "41", "42");
242 	}

250 	@Test void e11_restock_addsPetsAndOneSummaryAudit() {
251 		var s = new PetStore();
252 		var before = s.getAudit().size();
253 		var added = s.restock(new LinkedHashMap<>(Map.of(Species.DOG, 2)), "job:restock");
254 		assertSize(2, added);
255 		assertEquals(before + 3, s.getAudit().size());
256 		var last = s.getAudit().get(s.getAudit().size() - 1);
257 		assertBean(last, "actor,entity,entityId,action,detail", "job:restock,Job,restock,RESTOCK,2 pets");
258 	}
