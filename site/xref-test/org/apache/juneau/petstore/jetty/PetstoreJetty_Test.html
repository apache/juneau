42  	"java:S8692", // warmUpServer() polls a real HTTP server against a genuine wall-clock deadline; a fixed clock would break the retry loop.

51  	private static final HttpClient HTTP = HttpClient.newBuilder()
52  		.connectTimeout(Duration.ofSeconds(5))
53  		.followRedirects(HttpClient.Redirect.NEVER)
54  		.build();
55  
56  	/**
57  	 * Primes the root endpoint before the timed test methods run.
58  	 *
59  	 * <p>
60  	 * The fixture's {@code beforeAll} only waits for the Jetty connector to bind — not for the REST servlet to
61  	 * initialize. The first request to a Juneau REST resource triggers one-time {@code RestContext} setup
62  	 * (serializer/parser metadata, {@code HtmlDocSerializer} construction) which, under a loaded CI agent, can
63  	 * exceed the tight 10s per-request timeout the test methods use. Absorbing that cold-start here (with a
64  	 * generous budget + retry) removes the startup race from {@code a01} while keeping the per-test timeouts tight.
65  	 */
66  	@BeforeAll
67  	static void warmUpServer() throws Exception {
68  		var deadline = Instant.now().plusSeconds(30);
69  		Exception last = null;
70  		while (Instant.now().isBefore(deadline)) {
71  			try {
72  				var req = HttpRequest.newBuilder()
73  					.uri(URI.create(fixture.getRootUrl() + "/"))
74  					.timeout(Duration.ofSeconds(20))
75  					.header("Accept", "text/html")
76  					.GET()
77  					.build();
78  				if (HTTP.send(req, BodyHandlers.ofString()).statusCode() == 200)
79  					return;
80  			} catch (HttpTimeoutException | ConnectException e) {
81  				last = e;
82  			}
83  			Thread.sleep(250);
84  		}
85  		throw new IllegalStateException("Petstore Jetty server did not become ready within 30s", last);
86  	}

276 	/**
277 	 * Cheap brittle JSON int extractor — finds the first occurrence of {@code key} (e.g. {@code "id":}) and reads
278 	 * the integer that follows. Sufficient for the simple CRUD round-trip assertions; we don't pull in a full JSON
279 	 * parser just for this test class.
280 	 */
281 	private static long extractLong(String body, String key) {
282 		var i = body.indexOf(key);
283 		if (i < 0)
284 			return 0L;
285 		i += key.length();
286 		var end = i;
287 		while (end < body.length() && (Character.isDigit(body.charAt(end)) || body.charAt(end) == '-'))
288 			end++;
289 		if (end == i)
290 			return 0L;
291 		return Long.parseLong(body.substring(i, end));
292 	}
