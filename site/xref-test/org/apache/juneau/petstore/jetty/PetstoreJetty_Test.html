65  	/**
66  	 * Primes the root endpoint before the timed test methods run.
67  	 *
68  	 * <p>
69  	 * The fixture's {@code beforeAll} only waits for the Jetty connector to bind — not for the REST servlet to
70  	 * initialize. The first request to a Juneau REST resource triggers one-time {@code RestContext} setup
71  	 * (serializer/parser metadata, {@code HtmlDocSerializer} construction) which, under a loaded CI agent, can
72  	 * exceed the tight 10s per-request timeout the test methods use. Absorbing that cold-start here (with a
73  	 * generous budget + retry) removes the startup race from {@code a01} while keeping the per-test timeouts tight.
74  	 */
75  	private static void warmUpServer() throws Exception {
76  		var deadline = Instant.now().plusSeconds(30);
77  		Exception last = null;
78  		while (Instant.now().isBefore(deadline)) {
79  			try {
80  				var req = HttpRequest.newBuilder()
81  					.uri(URI.create(fixture.getRootUrl() + "/"))
82  					.timeout(Duration.ofSeconds(20))
83  					.header("Accept", "text/html")
84  					.GET()
85  					.build();
86  				// GET / is a 303 to /console/store (see a01); redirects aren't followed.
87  				if (HTTP.send(req, BodyHandlers.ofString()).statusCode() == 303)
88  					return;
89  			} catch (HttpTimeoutException | ConnectException e) {
90  				last = e;
91  			}
92  			Thread.sleep(250);
93  		}
94  		throw new IllegalStateException("Petstore Jetty server did not become ready within 30s", last);
95  	}

369 	private static String boundaryReason(HttpResponse<?> resp) {
370 		return resp.headers().firstValue("X-Request-Boundary").orElse(null);

373 	private static HttpResponse<String> send(String method, String path, String contentType, String origin, String token, String body) throws Exception {
374 		var b = HttpRequest.newBuilder().uri(URI.create(fixture.getRootUrl() + path)).timeout(Duration.ofSeconds(30))
375 			.header("Accept", "application/json").method(method, HttpRequest.BodyPublishers.ofString(body));
376 		if (contentType != null)
377 			b.header("Content-Type", contentType);
378 		if (origin != null)
379 			b.header("Origin", origin);
380 		if (token != null)
381 			b.header("X-Csrf-Token", token);
382 		return HTTP.send(b.build(), BodyHandlers.ofString());
383 	}

490 	/**
491 	 * Cheap brittle JSON int extractor — finds the first occurrence of {@code key} (e.g. {@code "id":}) and reads
492 	 * the integer that follows. Sufficient for the simple CRUD round-trip assertions; we don't pull in a full JSON
493 	 * parser just for this test class.
494 	 */
495 	private static long extractLong(String body, String key) {
496 		var i = body.indexOf(key);
497 		if (i < 0)
498 			return 0L;
499 		i += key.length();
500 		var end = i;
501 		while (end < body.length() && (Character.isDigit(body.charAt(end)) || body.charAt(end) == '-'))
502 			end++;
503 		if (end == i)
504 			return 0L;
505 		return Long.parseLong(body.substring(i, end));
506 	}
