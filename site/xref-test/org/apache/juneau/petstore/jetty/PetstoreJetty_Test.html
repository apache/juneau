68  	/** Every exit the shutdown resource asks for while this class runs; nothing here may stop the test JVM. */
69  	private static final BlockingQueue<Integer> EXITS = new LinkedBlockingQueue<>();
70  
71  	/** Every exit the shutdown resource has asked for, including those drained from {@link #EXITS}. */
72  	private static final AtomicInteger EXIT_COUNT = new AtomicInteger();
73  
74  	/** The shutdown resource mounted in place of the deployment's, so a shutdown is recorded instead of stopping the JVM. */
75  	private static final ShutdownResource SHUTDOWN = new ShutdownResource(code -> {
76  		EXIT_COUNT.incrementAndGet();
77  		EXITS.add(code);
78  	});
79  
80  	/** The exits that were expected, because a request the boundary admits was sent. */
81  	private static final AtomicInteger EXPECTED_EXITS = new AtomicInteger();
82  
83  	/** A refused shutdown request has this long, from the moment it was sent, to show an exit: the resource exits about a second after it answers. */
84  	private static final long EXIT_WINDOW_NANOS = TimeUnit.MILLISECONDS.toNanos(1500);
85  
86  	/** When the latest refused shutdown request was sent, and which requests were refused. */
87  	private static volatile long lastRefusedNanos = System.nanoTime();
88  	private static final List<String> REFUSED = new CopyOnWriteArrayList<>();

95  	/**
96  	 * Primes the root endpoint before the timed test methods run.
97  	 *
98  	 * <p>
99  	 * The fixture's {@code beforeAll} only waits for the Jetty connector to bind — not for the REST servlet to
100 	 * initialize. The first request to a Juneau REST resource triggers one-time {@code RestContext} setup
101 	 * (serializer/parser metadata, {@code HtmlDocSerializer} construction) which, under a loaded CI agent, can
102 	 * exceed the tight 10s per-request timeout the test methods use. Absorbing that cold-start here (with a
103 	 * generous budget + retry) removes the startup race from {@code a01} while keeping the per-test timeouts tight.
104 	 */
105 	private static void warmUpServer() throws Exception {
106 		var deadline = Instant.now().plusSeconds(30);
107 		Exception last = null;
108 		while (Instant.now().isBefore(deadline)) {
109 			try {
110 				var req = HttpRequest.newBuilder()
111 					.uri(URI.create(fixture.getRootUrl() + "/"))
112 					.timeout(Duration.ofSeconds(20))
113 					.header("Accept", "text/html")
114 					.GET()
115 					.build();
116 				// GET / is a 303 to /console/store (see a01); redirects aren't followed.
117 				if (HTTP.send(req, BodyHandlers.ofString()).statusCode() == 303)
118 					return;
119 			} catch (HttpTimeoutException | ConnectException e) {
120 				last = e;
121 			}
122 			Thread.sleep(250);
123 		}
124 		throw new IllegalStateException("Petstore Jetty server did not become ready within 30s", last);
125 	}

127 	/**
128 	 * Replaces the deployment's shutdown resource with {@link #SHUTDOWN}, and refuses to run any test unless that took.
129 	 *
130 	 * <p>
131 	 * The boundary filters sit in front of the root servlet by path, so they cover the replacement exactly as they cover
132 	 * the original.  The {@code GET} answers {@code 405} and does not exit.
133 	 */
134 	private static void mountTestShutdown() throws Exception {
135 		get("/", "text/html");  // Initializes the root servlet before the shutdown resource is mounted.
136 		RootResources root = null;
137 		for (var h : fixture.getJettyServerComponent().getServletContextHandler().getServletHandler().getServlets())
138 			if (h.getServlet() instanceof RootResources r)
139 				root = r;
140 		assertNotNull(root, "no RootResources mounted");
141 		root.getContext().getRestChildren().addChild(SHUTDOWN, true);
142 		var mounted = root.getContext().getRestChildren().asMap().values().stream()
143 			.filter(c -> c.getResource() instanceof ShutdownResource).toList();
144 		assertEquals(1, mounted.size());
145 		assertSame(SHUTDOWN, mounted.get(0).getResource(), "The test's shutdown resource must be the mounted one");
146 		assertEquals(405, get("/shutdown", "application/json").statusCode());
147 	}

156 	/**
157 	 * Records that a shutdown request which must not exit was just sent.
158 	 *
159 	 * <p>
160 	 * The check that no exit follows is made once, in {@link #assertNoStrayExits()}, after the last of these has had its
161 	 * full window; waiting out that window in every test would add it up, one test after another.
162 	 */
163 	private static void refusedShutdown(String what) {
164 		REFUSED.add(what);
165 		lastRefusedNanos = System.nanoTime();
166 	}
167 
168 	/** Fails the class if any exit happened that an admitted request did not account for. */
169 	@AfterAll
170 	static void assertNoStrayExits() throws Exception {
171 		var remaining = EXIT_WINDOW_NANOS - (System.nanoTime() - lastRefusedNanos);
172 		if (remaining > 0)
173 			TimeUnit.NANOSECONDS.sleep(remaining);
174 		assertEquals(EXPECTED_EXITS.get(), EXIT_COUNT.get(), "Refused and GET requests must not schedule an exit; refused: " + REFUSED);
175 	}

447 	private static HttpResponse<String> send(String method, String path, String contentType, String origin, String token, String body) throws Exception {
448 		var b = HttpRequest.newBuilder().uri(URI.create(fixture.getRootUrl() + path)).timeout(Duration.ofSeconds(30))
449 			.header("Accept", "application/json").method(method, HttpRequest.BodyPublishers.ofString(body));
450 		if (contentType != null)
451 			b.header("Content-Type", contentType);
452 		if (origin != null)
453 			b.header("Origin", origin);
454 		if (token != null)
455 			b.header("X-Csrf-Token", token);
456 		return HTTP.send(b.build(), BodyHandlers.ofString());
457 	}

645 	// Sends a GET with a Host header the HTTP client would not let us set, and returns "<status> <boundary reason>".
646 	private static String rawGet(String path, String host) throws Exception {
647 		var root = fixture.getRootUrl();
648 		try (var s = new Socket(root.getHost(), root.getPort()); var out = s.getOutputStream();
649 			var in = new BufferedReader(new InputStreamReader(s.getInputStream(), StandardCharsets.US_ASCII))) {
650 			s.setSoTimeout(30_000);
651 			out.write(("GET " + path + " HTTP/1.1\r\nHost: " + host + "\r\nAccept: text/plain\r\nConnection: close\r\n\r\n")
652 				.getBytes(StandardCharsets.US_ASCII));
653 			out.flush();
654 			var status = in.readLine().split(" ")[1];
655 			String reason = null;
656 			for (var line = in.readLine(); line != null && ! line.isEmpty(); line = in.readLine())
657 				if (line.regionMatches(true, 0, RequestBoundaryFilter.REJECTION_HEADER + ":", 0, RequestBoundaryFilter.REJECTION_HEADER.length() + 1))
658 					reason = line.substring(line.indexOf(':') + 1).trim();
659 			return status + " " + reason;
660 		}
661 	}

735 	/**
736 	 * Cheap brittle JSON int extractor — finds the first occurrence of {@code key} (e.g. {@code "id":}) and reads
737 	 * the integer that follows. Sufficient for the simple CRUD round-trip assertions; we don't pull in a full JSON
738 	 * parser just for this test class.
739 	 */
740 	private static long extractLong(String body, String key) {
741 		var i = body.indexOf(key);
742 		if (i < 0)
743 			return 0L;
744 		i += key.length();
745 		var end = i;
746 		while (end < body.length() && (Character.isDigit(body.charAt(end)) || body.charAt(end) == '-'))
747 			end++;
748 		if (end == i)
749 			return 0L;
750 		return Long.parseLong(body.substring(i, end));
751 	}
