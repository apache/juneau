34  	private static final JsonSerializer MCP_JSON =
35  		JsonSerializer.create()
36  			.addBeanTypes()
37  			.typePropertyName(Content.class, "type")
38  			.typePropertyName(ResourceContents.class, "type")
39  			.typePropertyName(CompletionReference.class, "type")
40  			.build();
41  
42  	private static final JsonParser MCP_JSON_PARSER =
43  		JsonParser.create()
44  			.typePropertyName(Content.class, "type")
45  			.typePropertyName(ResourceContents.class, "type")
46  			.typePropertyName(CompletionReference.class, "type")
47  			.build();
48  
49  	private static void assertJsonRoundTrip(Object bean, Class<?> type) {
50  		var j1 = MCP_JSON.write(bean);
51  		var copy = MCP_JSON_PARSER.read(j1, type);
52  		var j2 = MCP_JSON.write(copy);
53  		assertEquals(j1, j2, () -> "Round-trip JSON mismatch for " + type.getName() + ": " + j1);
54  	}
55  
56  	private static void assertJsonRoundTripPlain(Object bean, Class<?> type) {
57  		var ser = JsonSerializer.DEFAULT;
58  		var par = JsonParser.DEFAULT;
59  		var j1 = ser.write(bean);
60  		var copy = par.read(j1, type);
61  		var j2 = ser.write(copy);
62  		assertEquals(j1, j2, () -> "Round-trip JSON mismatch for " + type.getName() + ": " + j1);
63  	}
