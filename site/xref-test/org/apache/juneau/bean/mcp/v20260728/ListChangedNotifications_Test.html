28  class ListChangedNotifications_Test {
29  
30  	static Stream<Class<?>> types() {
31  		return Stream.of(ResourcesListChangedNotification.class, ToolsListChangedNotification.class,
32  			PromptsListChangedNotification.class);
33  	}
34  
35  	@ParameterizedTest
36  	@MethodSource("types")
37  	void a01_extendsRequestParams(Class<?> type) {
38  		assertTrue(RequestParams.class.isAssignableFrom(type), () -> type.getName() + " must extend RequestParams");
39  	}
40  
41  	@ParameterizedTest
42  	@MethodSource("types")
43  	void a02_declaresNoMembersBeyondInheritedMeta(Class<?> type) {
44  		assertEquals(0, type.getDeclaredFields().length, () -> type.getName() + " must declare no own fields");
45  	}
46  
47  	@Test void a03_metaRoundTrip_resourcesListChanged() throws Exception {
48  		roundTripMeta(new ResourcesListChangedNotification());
49  	}
50  
51  	@Test void a04_metaRoundTrip_toolsListChanged() throws Exception {
52  		roundTripMeta(new ToolsListChangedNotification());
53  	}
54  
55  	@Test void a05_metaRoundTrip_promptsListChanged() throws Exception {
56  		roundTripMeta(new PromptsListChangedNotification());
57  	}
58  
59  	private static void roundTripMeta(Object bean) throws Exception {
60  		bean.getClass().getMethod("setMeta", RequestMeta.class)
61  			.invoke(bean, new RequestMeta().setProtocolVersion("2026-07-28"));
62  		var json = JsonSerializer.DEFAULT.write(bean);
63  		assertTrue(json.contains("\"_meta\":{"), () -> bean.getClass().getSimpleName() + " must nest _meta: " + json);
64  		var copy = JsonParser.DEFAULT.read(json, bean.getClass());
65  		assertEquals(json, JsonSerializer.DEFAULT.write(copy));
66  	}
67  }
