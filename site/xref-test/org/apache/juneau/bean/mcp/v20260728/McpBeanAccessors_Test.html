29   * Coverage for the plain accessors and the lazily-initialised {@code put}/{@code add} helpers on the MCP
30   * 2026-07-28 request, result, capability, and content beans.

40  	@Test void a01_callToolRequest_mapHelpers() {
41  		var x = new CallToolRequest().putArgument("a", 1).putArgument("b", 2)
42  			.putInputResponse("i1", "x").putInputResponse("i2", "y");
43  		assertBean(x, "arguments{a,b},inputResponses{i1,i2}", "{1,2},{x,y}");
44  	}

46  	@Test void a02_getPromptRequest_accessorsAndMapHelpers() {
47  		var x = new GetPromptRequest().setName("p").putArgument("a", 1).putArgument("b", 2)
48  			.putInputResponse("i1", "x").putInputResponse("i2", "y");
49  		assertBean(x, "name,arguments{a,b},inputResponses{i1,i2}", "p,{1,2},{x,y}");
50  		x.setArguments(new LinkedHashMap<>(Map.of("c", 3)));
51  		assertBean(x, "arguments{c}", "{3}");
52  	}
