29  /**
30   * Coverage for the {@code 2026-07-28} SEP-2549 cache-hint carriers: {@link McpCacheScope} and the
31   * {@link CacheableResult} base shared by every cacheable v2 list/read result.
32   */
33  class McpCacheableResults_Test {
34  
35  	private static Stream<CacheableResult<?>> results() {
36  		return Stream.of(new ListToolsResult(), new ListPromptsResult(),
37  			new ListResourcesResult(), new ReadResourceResult(), new ListResourceTemplatesResult());
38  	}
39  
40  	@ParameterizedTest @MethodSource("results")
41  	void a01_defaultsOmitBothFields(CacheableResult<?> value) {
42  		assertNull(value.getTtlMs());
43  		assertNull(value.getCacheScope());
44  		assertFalse(Json.of(value).contains("ttlMs"));
45  		assertFalse(Json.of(value).contains("cacheScope"));
46  	}
47  
48  	@ParameterizedTest @MethodSource("results")
49  	void a02_inheritedFieldsRoundTrip(CacheableResult<?> value) {
50  		value.setTtlMs(0).setCacheScope(McpCacheScope.PRIVATE);
51  		var json = Json.of(value);
52  		assertTrue(json.contains("\"ttlMs\":0"));
53  		assertTrue(json.contains("\"cacheScope\":\"private\""));
54  		var copy = JsonParser.DEFAULT.read(json, value.getClass());
55  		assertEquals(0, ((CacheableResult<?>)copy).getTtlMs());
56  		assertEquals(McpCacheScope.PRIVATE, ((CacheableResult<?>)copy).getCacheScope());
57  	}
58  
59  	@Test void a03_scopeUsesExactLowercaseWireTokens() {
60  		assertEquals("\"public\"", Json.of(McpCacheScope.PUBLIC));
61  		assertEquals("\"private\"", Json.of(McpCacheScope.PRIVATE));
62  		assertEquals(McpCacheScope.PUBLIC, JsonParser.DEFAULT.read("\"public\"", McpCacheScope.class));
63  		assertEquals(McpCacheScope.PRIVATE, JsonParser.DEFAULT.read("\"private\"", McpCacheScope.class));
64  		assertThrows(Exception.class, () -> JsonParser.DEFAULT.read("\"bogus\"", McpCacheScope.class));
65  		assertArrayEquals(new McpCacheScope[] { McpCacheScope.PUBLIC, McpCacheScope.PRIVATE }, McpCacheScope.values());
66  	}
67  
68  	@Test void a04_concreteFluentChainsCompileAndPreservePayload() {
69  		var tools = new ListToolsResult().setTtlMs(5).setCacheScope(McpCacheScope.PUBLIC)
70  			.setTools(new Tool().setName("t")).setNextCursor("n");
71  		var read = new ReadResourceResult().setCacheScope(McpCacheScope.PRIVATE).setTtlMs(7)
72  			.setContents(new TextResourceContents().setUri("file:///a").setText("x"));
73  		assertEquals("t", tools.getTools().get(0).getName());
74  		assertEquals("n", tools.getNextCursor());
75  		assertEquals("x", ((TextResourceContents)read.getContents().get(0)).getText());
76  	}
77  
78  	@Test void a05_wireBeanPreservesNegativeParsedTtl() {
79  		var value = JsonParser.DEFAULT.read("{\"ttlMs\":-1,\"tools\":[]}", ListToolsResult.class);
80  		assertEquals(-1, value.getTtlMs());
81  		assertEquals("complete", value.getResultType());
82  		assertEquals("{\"resultType\":\"complete\",\"tools\":[],\"ttlMs\":-1}", Json.of(value));
83  	}
84  
85  	@Test void a06_payloadPlusInheritedResultType_listToolsResult() {
86  		var value = new ListToolsResult().setTools(new Tool().setName("t"));
87  		var json = Json.of(value);
88  		assertEquals("{\"resultType\":\"complete\",\"tools\":[{\"name\":\"t\"}]}", json);
89  		var copy = JsonParser.DEFAULT.read(json, ListToolsResult.class);
90  		assertEquals(json, Json.of(copy));
91  		assertFalse(json.contains("ttlMs"));
92  		assertFalse(json.contains("cacheScope"));
93  	}
94  
95  	@Test void a07_payloadPlusInheritedResultType_listPromptsResult() {
96  		var value = new ListPromptsResult().setPrompts(new Prompt().setName("p"));
97  		var json = Json.of(value);
98  		assertEquals("{\"prompts\":[{\"name\":\"p\"}],\"resultType\":\"complete\"}", json);
99  		var copy = JsonParser.DEFAULT.read(json, ListPromptsResult.class);
100 		assertEquals(json, Json.of(copy));
101 		assertFalse(json.contains("ttlMs"));
102 		assertFalse(json.contains("cacheScope"));

107 		var json = Json.of(value);
108 		assertEquals("{\"resources\":[{\"uri\":\"file:///a\"}],\"resultType\":\"complete\"}", json);

115 	@Test void a09_payloadPlusInheritedResultType_readResourceResult() {
116 		var value = new ReadResourceResult().setContents(new TextResourceContents().setUri("file:///a").setText("x"));
117 		var json = Json.of(value);
118 		assertEquals("{\"contents\":[{\"text\":\"x\",\"uri\":\"file:///a\"}],\"resultType\":\"complete\"}", json);

124 	void a10_cacheableResults_extendResult(CacheableResult<?> value) {
125 		assertTrue(Result.class.isAssignableFrom(value.getClass()));
126 		assertEquals("complete", value.getResultType());
127 	}
128 
129 	@Test void a11_cacheableResult_classExtendsResult() {
130 		assertTrue(Result.class.isAssignableFrom(CacheableResult.class));
131 	}
132 
133 	@Test void a12_fluentChain_mixesOwnCacheAndInheritedResultSetters() {
134 		var value = new ListToolsResult()
135 			.setTools(new Tool().setName("t"))
136 			.setTtlMs(5)
137 			.setCacheScope(McpCacheScope.PUBLIC)
138 			.setResultType("complete")
139 			.setMeta(new ResultMeta().setServerInfo(new Implementation().setName("s").setVersion("1")));
140 		assertEquals("t", value.getTools().get(0).getName());
141 		assertEquals(5, value.getTtlMs());
142 		assertEquals(McpCacheScope.PUBLIC, value.getCacheScope());
143 		assertEquals("complete", value.getResultType());
144 		assertEquals("s", value.getMeta().getServerInfo().getName());
145 	}
