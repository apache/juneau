77  	@Test void a06_reDeclaringFieldName_lastFieldWins() {
78  		// Re-declaring "name" via a second stringField(...) call must discard the first declaration's
79  		// modifiers (title) entirely, leaving only what was applied after the second declaration.
80  		var schema = ElicitSchema.create()
81  			.stringField("name").title("First").description("first description")
82  			.stringField("name").format("email")
83  			.build();
84  		var property = schema.getMap("properties").getMap("name");
85  		assertEquals(Map.of("type", "string", "format", "email"), property);
86  	}

302 	@Test void d01_build_returnsSchemaIndependentOfFurtherBuilderMutation() {
303 		// Pins the fix for the shallow-copy aliasing bug: build() must deep-copy each per-field map so that
304 		// applying further modifiers to the builder after build() cannot mutate an already-returned schema.
305 		var builder = ElicitSchema.create().stringField("name").title("Name");
306 		var first = builder.build();
307 		builder.description("added after build");
308 		builder.integerField("age").min(0);
309 		assertEquals("{\"type\":\"object\",\"properties\":{\"name\":{\"type\":\"string\",\"title\":\"Name\"}}}",
310 			first.toString());
311 	}

340 	@Test void e05_required_reDeclaredField_markingSurvives() {
341 		// required(...) is a builder-level, order-independent set (see class Javadoc): redeclaring "name" via
342 		// a second stringField(...) call discards its per-field modifiers (last-field-wins) but must NOT
343 		// discard its earlier required(...) marking.
344 		var schema = ElicitSchema.create()
345 			.stringField("name").required("name")
346 			.stringField("name")
347 			.build();
348 		assertEquals(List.of("name"), schema.get("required"));
349 	}
