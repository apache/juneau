54  	private static void assertJsonRoundTrip(Object bean, Class<?> type) {
55  		var j1 = MCP_JSON.write(bean);
56  		var copy = MCP_JSON_PARSER.read(j1, type);
57  		var j2 = MCP_JSON.write(copy);
58  		assertEquals(j1, j2, () -> "Round-trip JSON mismatch for " + type.getName() + ": " + j1);
59  	}
