51  	private static void assertJsonRoundTrip(Object bean, Class<?> type) {
52  		var j1 = MCP_JSON.write(bean);
53  		var copy = MCP_JSON_PARSER.read(j1, type);
54  		var j2 = MCP_JSON.write(copy);
55  		assertEquals(j1, j2, () -> "Round-trip JSON mismatch for " + type.getName() + ": " + j1);
56  	}
