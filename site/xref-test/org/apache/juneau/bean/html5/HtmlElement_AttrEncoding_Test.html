19  import static org.apache.juneau.bean.html5.HtmlBuilder.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import org.apache.juneau.*;
23  import org.apache.juneau.marshall.html.*;
24  import org.junit.jupiter.api.*;

26  /**
27   * Tests that attribute values set on an {@link HtmlElement} are entity-escaped when the element is serialized.
28   *
29   * <p>
30   * Element text children have always been escaped; these lock in the matching guarantee for attribute values, which is
31   * what makes it safe to route dynamic text (labels, tooltips, ids) into an attribute.
32   */
33  class HtmlElement_AttrEncoding_Test extends TestBase {
34  
35  	@Test void a01_aQuoteBearingValueCannotBreakOutOfItsAttribute() throws Exception {
36  		var r = HtmlSerializer.DEFAULT_SQ.write(div().attr("aria-label", "\"><script>alert(1)</script>"));
37  		assertFalse(r.contains("<script>"), r);
38  		assertTrue(r.contains("aria-label='&quot;&gt;&lt;script&gt;alert(1)&lt;/script&gt;'"), r);
39  	}
40  
41  	@Test void a02_theXmlSpecialCharactersBecomeEntities() throws Exception {
42  		var r = HtmlSerializer.DEFAULT_SQ.write(div().attr("title", "&\"<>'"));
43  		assertTrue(r.contains("title='&amp;&quot;&lt;&gt;&apos;'"), r);
44  	}
45  
46  	@Test void a03_ordinaryValuesArePassedThroughUnchanged() throws Exception {
47  		var r = HtmlSerializer.DEFAULT_SQ.write(div().class_("btn btn-primary").id("c1"));
48  		assertTrue(r.contains("class='btn btn-primary'"), r);
49  		assertTrue(r.contains("id='c1'"), r);
50  		assertFalse(r.contains("&amp;"), r);
51  	}
52  
53  	@Test void a04_aQueryStringHrefIsEncodedOnceNotTwice() throws Exception {
54  		var r = HtmlSerializer.DEFAULT_SQ.write(a("http://x/y?a=1&b=2", "text"));
55  		assertEquals("<a href='http://x/y?a=1&amp;b=2'>text</a>", r);
56  	}
57  }
