20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.io.*;
23  
24  import org.apache.juneau.*;

29  /**
30   * Tests for the re-serializable, String-backed {@link HtmlBuilder#rawText(String)} affordance.
31   *
32   * <p>
33   * Unlike a one-shot {@code Reader}, a {@link RawText} is backed by a reusable {@code String} and therefore survives
34   * repeated serialization.  Its content is emitted <b>verbatim</b> (no entity/whitespace encoding), mirroring the
35   * existing raw {@code Reader}/{@code InputStream} path.
36   */
37  class HtmlBuilder_RawText_Test extends TestBase {
38  
39  	// A rawText body containing '<' and '&' is emitted verbatim, AND the element re-serializes identically
40  	// (proves it is not one-shot like a Reader).
41  	@Test void a01_reSerializable() throws Exception {
42  		var s = HtmlSerializer.DEFAULT_SQ;
43  		var e = script().text(rawText("a<b & c"));
44  		var r1 = s.write(e);
45  		var r2 = s.write(e);
46  		assertEquals("<script>a<b & c</script>", r1);
47  		assertEquals(r1, r2);
48  	}
49  
50  	// rawText(s) produces output identical to the established one-shot Reader raw path for the same input.
51  	@Test void a02_vsReaderParity() throws Exception {
52  		var s = HtmlSerializer.DEFAULT_SQ;
53  		var body = "if (a < b) { alert('x & y'); }";
54  		var rawTextOut = s.write(script().text(rawText(body)));
55  		var readerOut = s.write(script().text(new StringReader(body)));
56  		assertEquals(readerOut, rawTextOut);
57  		assertEquals("<script>if (a < b) { alert('x & y'); }</script>", rawTextOut);
58  	}
59  
60  	// Also emits verbatim through the XML serializer (RawText is recognized by both).
61  	@Test void a03_xmlSerializerVerbatim() throws Exception {
62  		var s = XmlSerializer.DEFAULT_SQ;
63  		var r = s.write(script().text(rawText("x < y & z")));
64  		assertEquals("<script>x < y & z</script>", r);
65  	}
66  
67  	// SECURITY: RAWTEXT/rawText is verbatim by contract -- it does NOT silently neutralize a '</script>'
68  	// end-tag sequence in the body.  Safety is the caller's responsibility; callers embedding a JSON payload should
69  	// use StringUtils.escapeForScript(String) rather than hand-rolling it.  This test documents that contract.
70  	@Test void c01_scriptEndTagIsCallerResponsibility() throws Exception {
71  		var s = HtmlSerializer.DEFAULT_SQ;
72  
73  		// Verbatim: a literal </script> in the body is emitted as-is (would prematurely close the element in a browser).
74  		var unsafe = s.write(script().text(rawText("var x = '</script>';")));
75  		assertEquals("<script>var x = '</script>';</script>", unsafe);
76  		assertEquals(2, countOccurrences(unsafe, "</script>"));
77  
78  		// Caller neutralizes by escaping '<' -> \u003c (valid JS); the only real </script> is the true close tag.
79  		var safe = s.write(script().text(rawText("var x = '\\u003c/script>';")));
80  		assertEquals("<script>var x = '\\u003c/script>';</script>", safe);
81  		assertEquals(1, countOccurrences(safe, "</script>"));
82  	}
83  
84  	private static int countOccurrences(String haystack, String needle) {
85  		var count = 0;
86  		var i = 0;
87  		while ((i = haystack.indexOf(needle, i)) != -1) {
88  			count++;
89  			i += needle.length();
90  		}
91  		return count;
92  	}
93  }
