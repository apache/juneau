29  /**
30   * Coverage for the JSON-RPC 2.0 envelope beans and {@link JsonRpcResponse}'s static helpers.
31   */
32  class JsonRpcBeans_Test {
33  
34  	private static void assertRoundTrip(Object bean, Class<?> type) {
35  		var a = JsonSerializer.DEFAULT.write(bean);
36  		var b = JsonSerializer.DEFAULT.write(JsonParser.DEFAULT.read(a, type));
37  		assertEquals(a, b, () -> "Round-trip JSON mismatch for " + type.getName() + ": " + a);
38  	}
39  
40  	@Test
41  	void a01_request_roundTrip() {
42  		var a = new JsonRpcRequest().setJsonrpc("2.0").setId("abc").setMethod("tools/list").setParams(JsonMap.of("cursor", "1"));
43  		assertRoundTrip(a, JsonRpcRequest.class);
44  	}
45  
46  	@Test
47  	void a02_response_resultRoundTrip() {
48  		var a = new JsonRpcResponse().setJsonrpc("2.0").setId(42).setResult(JsonMap.of("x", 1));
49  		assertRoundTrip(a, JsonRpcResponse.class);
50  	}
51  
52  	@Test
53  	void a03_response_errorRoundTrip() {
54  		var a = new JsonRpcResponse().setJsonrpc("2.0").setId(7)
55  			.setError(new JsonRpcError().setCode(-32600).setMessage("Invalid Request").setData(JsonMap.of("detail", "x")));
56  		assertRoundTrip(a, JsonRpcResponse.class);
57  	}
58  
59  	@Test
60  	void a04_error_roundTrip() {
61  		assertRoundTrip(new JsonRpcError().setCode(-32603).setMessage("boom").setData(JsonMap.of("type", "X")), JsonRpcError.class);
62  	}
63  
64  	@Test
65  	void b01_notification_isNullId() {
66  		assertTrue(JsonRpcResponse.notification(null));
67  		assertFalse(JsonRpcResponse.notification(1));
68  		assertFalse(JsonRpcResponse.notification("abc"));
69  	}
70  
71  	@Test
72  	void b02_ok_setsVersionIdAndResult() {
73  		var a = JsonRpcResponse.ok(1, JsonMap.of("k", "v"));
74  		assertEquals("2.0", a.getJsonrpc());
75  		assertEquals(1, a.getId());
76  		assertNotNull(a.getResult());
77  		assertNull(a.getError());
78  	}
79  
80  	@Test
81  	void b03_errorResponse_fourArg_carriesData() {
82  		var a = JsonRpcResponse.errorResponse(1, -32603, "boom", JsonMap.of("type", "X"));
83  		assertEquals("2.0", a.getJsonrpc());
84  		assertEquals(1, a.getId());
85  		assertNull(a.getResult());
86  		assertEquals(-32603, a.getError().getCode());
87  		assertEquals("boom", a.getError().getMessage());
88  		assertNotNull(a.getError().getData());
89  	}
90  
91  	@Test
92  	void b04_errorResponse_threeArg_leavesDataNull() {
93  		var a = JsonRpcResponse.errorResponse(null, -32600, "Request envelope is null");
94  		assertNull(a.getId());
95  		assertEquals(-32600, a.getError().getCode());
96  		assertNull(a.getError().getData());
97  	}

99  	private static Set<String> instanceFieldNames(Class<?> type) {
100 		return Arrays.stream(type.getDeclaredFields())
101 			.filter(f -> ! Modifier.isStatic(f.getModifiers()))
102 			.map(Field::getName)
103 			.collect(Collectors.toSet());
104 	}
105 
106 	@Test
107 	void c01_request_declaredFieldsAreExactlyJsonRpcEnvelope() {
108 		assertEquals(Set.of("jsonrpc", "id", "method", "params"), instanceFieldNames(JsonRpcRequest.class));
109 	}
110 
111 	@Test
112 	void c02_response_declaredFieldsAreExactlyJsonRpcEnvelope() {
113 		assertEquals(Set.of("jsonrpc", "id", "result", "error"), instanceFieldNames(JsonRpcResponse.class));
114 	}
115 
116 	@Test
117 	void c03_envelopesRemainMcpNeutral() {
118 		for (var type : List.of(JsonRpcRequest.class, JsonRpcResponse.class)) {
119 			var names = Arrays.stream(type.getMethods()).map(Method::getName).toList();
120 			assertFalse(names.contains("getProtocolVersion"));
121 			assertFalse(names.contains("getClientCapabilities"));
122 			assertFalse(names.contains("getMeta"));
123 			assertFalse(names.contains("setMeta"));
124 		}
125 	}
126 
127 	@Test
128 	void c04_noTopLevelMetaIsEmitted() {
129 		var reqJson = JsonSerializer.DEFAULT.write(new JsonRpcRequest().setJsonrpc("2.0").setId(1).setMethod("tools/list").setParams(JsonMap.of("cursor", "1")));
130 		assertFalse(reqJson.contains("_meta"));
131 		var respJson = JsonSerializer.DEFAULT.write(new JsonRpcResponse().setJsonrpc("2.0").setId(1).setResult(JsonMap.of("x", 1)));
132 		assertFalse(respJson.contains("_meta"));
133 	}
