17  package org.apache.juneau.bean.jsonschema;

19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.io.*;
22  import java.net.*;
23  
24  import org.apache.juneau.*;
25  import org.junit.jupiter.api.*;

27  @SuppressWarnings({
28  	"removal",   // Tests deprecated getId() / setId() for backward compatibility
29  	"java:S5778", // assertThrows lambdas with chained calls; intermediate invocations do not throw in practice
30  	"resource",  // Reader fixtures returned by getReader() overrides are test stubs (null / throwing); no resource to close.
31  	"rawtypes",  // JsonSchema/JsonSchemaProperty are self-typed CRTP roots; direct instantiation is intentionally raw (accepted 10.0.0 tradeoff).
32  	"unchecked"  // See rawtypes rationale above.
33  })
34  class JsonSchemaMap_Test extends TestBase {
35  
36  	@Test void a01_add_withId() {
37  		var map = new JsonSchemaMap() {};
38  		var schema = new JsonSchema().setId("http://example.org/schema");
39  		var result = map.add(schema);
40  		assertSame(map, result);
41  		assertNotNull(map.get(URI.create("http://example.org/schema")));
42  	}
43  
44  	@Test void a02_add_withoutId_throws() {
45  		var map = new JsonSchemaMap() {};
46  		var schema = new JsonSchema(); // no ID set
47  		assertThrows(IllegalArgumentException.class, () -> map.add(schema));
48  	}
49  
50  	@Test void a03_get_existingSchema_returnsDirectly() {
51  		var map = new JsonSchemaMap() {};
52  		var uri = URI.create("http://example.org/schema");
53  		var schema = new JsonSchema().setId(uri.toString());
54  		map.add(schema);
55  
56  		// Second get should return the cached entry without going through load()
57  		var result = map.get(uri);
58  		assertSame(schema, result);
59  	}
60  
61  	@Test void a04_get_notFound_callsLoad_returnsNull() {
62  		// Default load() implementation calls getReader() which returns null → load() returns null
63  		var map = new JsonSchemaMap() {};
64  		var result = map.get(URI.create("http://missing.org/schema"));
65  		assertNull(result);
66  	}
67  
68  	@Test void a05_get_notFound_callsLoad_returnsSchema() {
69  		var loadedSchema = new JsonSchema().setType(JsonType.OBJECT);
70  		var targetUri = URI.create("http://dynamic.org/schema");
71  
72  		var map = new JsonSchemaMap() {
73  			@Override
74  			public JsonSchema load(URI uri) {
75  				if (uri.equals(targetUri))
76  					return loadedSchema;
77  				return null;
78  			}
79  		};
80  
81  		var result = map.get(targetUri);
82  		assertSame(loadedSchema, result);
83  		// Schema should now be cached
84  		assertSame(loadedSchema, map.get(targetUri));
85  	}
86  
87  	@Test void a06_load_withReader() {
88  		var schemaJson = "{type:'string'}";
89  		var targetUri = URI.create("http://example.org/via-reader");
90  
91  		var map = new JsonSchemaMap() {
92  			@Override
93  			public Reader getReader(URI uri) {
94  				return new StringReader(schemaJson);
95  			}
96  		};
97  
98  		var result = map.load(targetUri);
99  		assertNotNull(result);
100 	}
101 
102 	@Test void a07_getReader_default_returnsNull() {
103 		var map = new JsonSchemaMap() {};
104 		assertNull(map.getReader(URI.create("http://example.org/schema")));
105 	}
106 
107 	@Test void a08_load_withExceptionInReader_wrapsAsRuntimeException() {
108 		var map = new JsonSchemaMap() {
109 			@Override
110 			public Reader getReader(URI uri) {
111 				// Return a reader that throws when read
112 				return new Reader() {
113 					@Override
114 					public int read(char[] cbuf, int off, int len) throws IOException {
115 						throw new IOException("Simulated IO error");
116 					}
117 					@Override
118 					public void close() { /* intentionally empty */ }
119 				};
120 			}
121 		};
122 		assertThrows(RuntimeException.class, () -> map.load(URI.create("http://example.org/schema")));
123 	}
124 }
