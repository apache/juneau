36  	@Test void a01_primitives() {
37  		var g = JsonSchemaBeanGenerator.DEFAULT;
38  		assertEquals("integer", g.generate(int.class).getTypeAsJsonType().toString().toLowerCase(Locale.ROOT));
39  		assertEquals("int32", g.generate(int.class).getFormat());
40  		assertEquals("number", g.generate(float.class).getTypeAsJsonType().toString().toLowerCase(Locale.ROOT));
41  		assertEquals("float", g.generate(float.class).getFormat());
42  		assertEquals("boolean", g.generate(boolean.class).getTypeAsJsonType().toString().toLowerCase(Locale.ROOT));
43  		assertEquals("string", g.generate(String.class).getTypeAsJsonType().toString().toLowerCase(Locale.ROOT));
44  		assertEquals("uri", g.generate(URI.class).getFormat());
45  	}

47  	@Test void a02_bean_collection_map_enum() {
48  		var g = JsonSchemaBeanGenerator.DEFAULT;
49  		assertEquals("object", g.generate(SimpleBean.class).getTypeAsJsonType().toString().toLowerCase(Locale.ROOT));
50  		assertNotNull(g.generate(SimpleBean.class).getProperty("name"));
51  		assertEquals("array", g.generate(BeanList.class).getTypeAsJsonType().toString().toLowerCase(Locale.ROOT));
52  		assertEquals("object", g.generate(BeanList.class).getItemsAsSchema().getTypeAsJsonType().toString().toLowerCase(Locale.ROOT));
53  		assertEquals(Boolean.TRUE, g.generate(BeanSet.class).getUniqueItems());
54  		assertNotNull(g.generate(BeanMap.class).getAdditionalPropertiesAsSchema());
55  		assertEquals("string", g.generate(TinyEnum.class).getTypeAsJsonType().toString().toLowerCase(Locale.ROOT));
56  		assertEquals(List.of("one","two"), g.generate(TinyEnum.class).getEnum());
57  	}

66  	@Test void a05_descriptions_and_examples() {
67  		var bean = JsonSchemaBeanGenerator.create()
68  			.addDescriptionsTo(TypeCategory.ANY)
69  			.addExamplesTo(TypeCategory.ANY)
70  			.allowNestedDescriptions()
71  			.allowNestedExamples()
72  			.build()
73  			.generate(ExampleBean.class);
74  
75  		assertEquals("object", bean.getTypeAsJsonType().toString().toLowerCase(Locale.ROOT));
76  		assertEquals("org.apache.juneau.bean.jsonschema.JsonSchemaBeanGenerator_Test$ExampleBean", bean.getDescription());
77  		assertEquals("java.lang.String", bean.getProperty("value").getDescription());
78  	}

80  	@Test void a06_roundTripParityWithJsonMapOutput() throws Exception {
81  		var map = JsonSchemaGenerator.DEFAULT.getSession().getSchema(SimpleBean.class);
82  		var bean = JsonSchemaBeanGenerator.toBean(map);
83  		var mapJson = JsonSerializer.DEFAULT.write(map);
84  		var beanJson = JsonSerializer.DEFAULT.write(bean);
85  		assertEquals(
86  			JsonParser.DEFAULT.read(mapJson, JsonMap.class),
87  			JsonParser.DEFAULT.read(beanJson, JsonMap.class)
88  		);
89  	}
