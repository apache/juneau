17  package org.apache.juneau.bean.jsonapi;
18  
19  import static org.apache.juneau.test.bct.BctAssertions.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.util.*;
23  
24  import org.apache.juneau.*;
25  import org.junit.jupiter.api.*;
26  
27  /**
28   * Tests for the lazy-initializing {@code putMeta} / {@code putLink} helpers and the varargs / collection
29   * overloads on the {@code juneau-bean-jsonapi} beans.
30   */
31  class JsonApi_PutAndCollections_Test extends TestBase {
32  
33  	@Test
34  	void a01_documentPutMeta() {
35  		var d = new JsonApiDocument().putMeta("a", 1).putMeta("b", 2);
36  		assertEquals(Map.of("a", 1, "b", 2), d.getMeta());
37  	}
38  
39  	@Test
40  	void a02_documentPutLink() {
41  		var d = new JsonApiDocument().putLink("self", "/x").putLink("next", new JsonApiLink("/y"));
42  		assertSize(2, d.getLinks());
43  		assertEquals("/x", d.getLinks().get("self"));
44  		assertEquals("/y", ((JsonApiLink)d.getLinks().get("next")).getHref());
45  	}
46  
47  	@Test
48  	void a03_documentSetErrorsVarargs() {
49  		var d = new JsonApiDocument().setErrors(new JsonApiError().setTitle("x"), new JsonApiError().setTitle("y"));
50  		assertSize(2, d.getErrors());
51  		assertBean(d.getErrors().get(1), "title", "y");
52  	}
53  
54  	@Test
55  	void a04_documentSetIncludedVarargs() {
56  		var d = new JsonApiDocument().setIncluded(new JsonApiResource("a", "1"), new JsonApiResource("a", "2"));
57  		assertSize(2, d.getIncluded());
58  		assertBean(d.getIncluded().get(1), "type,id", "a,2");
59  	}
60  
61  	@Test
62  	void a05_documentAddErrorsCollectionInitializesAndAppends() {
63  		var d = new JsonApiDocument().addErrors(List.of(new JsonApiError().setTitle("a")));
64  		assertSize(1, d.getErrors());
65  		d.addErrors(List.of(new JsonApiError().setTitle("b")));
66  		assertSize(2, d.getErrors());
67  		assertEquals("b", d.getErrors().get(1).getTitle());
68  	}
69  
70  	@Test
71  	void a06_documentAddIncludedCollectionInitializesAndAppends() {
72  		var d = new JsonApiDocument().addIncluded(List.of(new JsonApiResource("a", "1")));
73  		assertSize(1, d.getIncluded());
74  		d.addIncluded(List.of(new JsonApiResource("a", "2")));
75  		assertSize(2, d.getIncluded());
76  		assertEquals("2", d.getIncluded().get(1).getId());
77  	}
78  
79  	@Test
80  	void b01_errorPutLinkAndMeta() {
81  		var e = new JsonApiError().putLink("about", "/e").putLink("more", "/m").putMeta("a", 1).putMeta("b", 2);
82  		assertEquals(Map.of("about", "/e", "more", "/m"), e.getLinks());
83  		assertEquals(Map.of("a", 1, "b", 2), e.getMeta());
84  	}
85  
86  	@Test
87  	void c01_linkPutMeta() {
88  		var l = new JsonApiLink("/u").putMeta("a", 1).putMeta("b", 2);
89  		assertEquals(Map.of("a", 1, "b", 2), l.getMeta());
90  	}
91  
92  	@Test
93  	void d01_relationshipPutLinkAndMeta() {
94  		var r = new JsonApiRelationship().putLink("self", "/s").putLink("related", "/r").putMeta("a", 1).putMeta("b", 2);
95  		assertEquals(Map.of("self", "/s", "related", "/r"), r.getLinks());
96  		assertEquals(Map.of("a", 1, "b", 2), r.getMeta());
97  	}
98  
99  	@Test
100 	void e01_resourcePutLinkAndMeta() {
101 		var r = new JsonApiResource("a", "1").putLink("self", "/s").putLink("o", "/o").putMeta("a", 1).putMeta("b", 2);
102 		assertEquals(Map.of("self", "/s", "o", "/o"), r.getLinks());
103 		assertEquals(Map.of("a", 1, "b", 2), r.getMeta());
104 	}
105 
106 	@Test
107 	void f01_resourceIdentifierPutMeta() {
108 		var r = new JsonApiResourceIdentifier("a", "1").putMeta("a", 1).putMeta("b", 2);
109 		assertEquals(Map.of("a", 1, "b", 2), r.getMeta());
110 	}
111 
112 	@Test
113 	void g01_versionPutMeta() {
114 		var v = new JsonApiVersion("1.1").putMeta("a", 1).putMeta("b", 2);
115 		assertEquals(Map.of("a", 1, "b", 2), v.getMeta());
116 	}
117 }
