31  	private static final JsonSerializer SER =
32  		JsonSerializer.create()
33  			.addBeanTypes()
34  			.addRootType()
35  			.typePropertyName(JsonPatchOperation.class, "op")
36  			.build();
37  
38  	private static final JsonParser PAR =
39  		JsonParser.create()
40  			.typePropertyName(JsonPatchOperation.class, "op")
41  			.build();
42  
43  	private static void assertJsonRoundTrip(Object bean, Class<?> type) {
44  		var j1 = SER.write(bean);
45  		var copy = PAR.read(j1, type);
46  		var j2 = SER.write(copy);
47  		assertEquals(j1, j2, () -> "Round-trip JSON mismatch for " + type.getName() + ": " + j1);
48  	}

152 	@Nested
153 	class C_PolymorphicDispatch {
154 
155 		@Test
156 		void c01_allSixOpsRoundTripViaPatchDocument() {
157 			var patch = new JsonPatch()
158 				.append(new AddOp("/a/b/c", "foo"))
159 				.append(new RemoveOp("/a/b/c"))
160 				.append(new ReplaceOp("/a/b/c", 42))
161 				.append(new MoveOp("/a/b/c", "/a/b/d"))
162 				.append(new CopyOp("/a/b/c", "/a/b/e"))
163 				.append(new TestOp("/a/b/c", "foo"));
164 			assertJsonRoundTrip(patch, JsonPatch.class);
165 		}
166 
167 		@Test
168 		void c02_parserPicksRightSubclassForEachOp() {
169 			var patch = new JsonPatch()
170 				.append(new AddOp("/a", 1))
171 				.append(new RemoveOp("/b"))
172 				.append(new ReplaceOp("/c", "x"))
173 				.append(new MoveOp("/d", "/e"))
174 				.append(new CopyOp("/f", "/g"))
175 				.append(new TestOp("/h", true));
176 			var j = SER.write(patch);
177 			var back = PAR.read(j, JsonPatch.class);
178 			assertEquals(6, back.size());
179 			assertInstanceOf(AddOp.class, back.get(0));
180 			assertInstanceOf(RemoveOp.class, back.get(1));
181 			assertInstanceOf(ReplaceOp.class, back.get(2));
182 			assertInstanceOf(MoveOp.class, back.get(3));
183 			assertInstanceOf(CopyOp.class, back.get(4));
184 			assertInstanceOf(TestOp.class, back.get(5));
185 			assertEquals("/a", back.get(0).getPath());
186 		}
187 
188 		@Test
189 		void c03_wireFormatIsJsonArrayOfOperationObjects() {
190 			var patch = new JsonPatch().append(new AddOp("/x", 1));
191 			var j = SER.write(patch);
192 			assertTrue(j.startsWith("["), () -> "Top-level should be a JSON array: " + j);
193 			assertTrue(j.contains("\"op\":\"add\""), () -> j);
194 			assertTrue(j.contains("\"path\":\"/x\""), () -> j);
195 			assertTrue(j.contains("\"value\":1"), () -> j);
196 		}
