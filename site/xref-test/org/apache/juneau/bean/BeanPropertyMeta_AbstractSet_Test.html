28  /**
29   * Tests for materialization of abstract {@link Collection} bean properties that have no setter and no
30   * {@code @BeanProp(type=...)} override.
31   *
32   * <p>
33   * Such fields are populated from a JSON array source by
34   * {@code BeanPropertyMeta.setPropertyValue(...)}, which instantiates a concrete collection via
35   * {@code createDefaultCollectionForAbstractType(...)}.  These tests pin the shape-based default table
36   * ({@code Set}&rarr;{@code LinkedHashSet}, {@code SortedSet}&rarr;{@code TreeSet},
37   * {@code List}/raw {@code Collection}&rarr;{@code ArrayList}, {@code Deque}/{@code Queue}&rarr;{@code ArrayDeque}),
38   * the still-honored {@code @BeanProp(type=...)} override, and the best-effort-materialize-then-fallback
39   * behavior for custom abstract-set subclasses.
40   */
41  class BeanPropertyMeta_AbstractSet_Test extends TestBase {
42  
43  	public enum Tag { URGENT, PRIORITY, LOW }
44  
45  	//------------------------------------------------------------------------------------------------------------------
46  	// a01: abstract Set<EnumType> field, no setter, no @BeanProp -> LinkedHashSet (order-preserving), elements coerced.
47  	//------------------------------------------------------------------------------------------------------------------
48  
49  	public static class A01_Bean { public Set<Tag> tags; }
50  
51  	@Test void a01_setOfEnum_materializesLinkedHashSet() {
52  		var a = json("{tags:['URGENT','PRIORITY']}", A01_Bean.class);
53  		assertInstanceOf(LinkedHashSet.class, a.tags);
54  		assertEquals(List.of(Tag.URGENT, Tag.PRIORITY), new ArrayList<>(a.tags));  // Insertion order preserved.
55  	}
56  
57  	//------------------------------------------------------------------------------------------------------------------
58  	// a02: abstract SortedSet<EnumType> field -> TreeSet (natural ordering of the enum).
59  	//------------------------------------------------------------------------------------------------------------------
60  
61  	public static class A02_Bean { public SortedSet<Tag> tags; }
62  
63  	@Test void a02_sortedSetOfEnum_materializesTreeSet() {
64  		var a = json("{tags:['PRIORITY','URGENT','LOW']}", A02_Bean.class);
65  		assertInstanceOf(TreeSet.class, a.tags);
66  		// TreeSet orders by enum ordinal: URGENT(0), PRIORITY(1), LOW(2).
67  		assertEquals(List.of(Tag.URGENT, Tag.PRIORITY, Tag.LOW), new ArrayList<>(a.tags));
68  	}
69  
70  	//------------------------------------------------------------------------------------------------------------------
71  	// a03: abstract List<EnumType> field, no setter -> ArrayList.
72  	//------------------------------------------------------------------------------------------------------------------
73  
74  	public static class A03_Bean { public List<Tag> tags; }
75  
76  	@Test void a03_listOfEnum_materializesArrayList() {
77  		var a = json("{tags:['LOW','URGENT']}", A03_Bean.class);
78  		assertInstanceOf(ArrayList.class, a.tags);
79  		assertEquals(List.of(Tag.LOW, Tag.URGENT), a.tags);
80  	}
81  
82  	//------------------------------------------------------------------------------------------------------------------
83  	// a04: abstract Deque<String> field -> ArrayDeque.
84  	//------------------------------------------------------------------------------------------------------------------
85  
86  	public static class A04_Bean { public Deque<String> items; }
87  
88  	@Test void a04_dequeOfString_materializesArrayDeque() {
89  		var a = json("{items:['x','y']}", A04_Bean.class);
90  		assertInstanceOf(ArrayDeque.class, a.items);
91  		assertEquals(List.of("x", "y"), new ArrayList<>(a.items));
92  	}
93  
94  	//------------------------------------------------------------------------------------------------------------------
95  	// a05: @BeanProp(type=TreeSet.class) override still honored (wins over the LinkedHashSet shape default for Set).
96  	//------------------------------------------------------------------------------------------------------------------
97  
98  	public static class A05_Bean {
99  		@BeanProp(type=TreeSet.class, params={Tag.class})
100 		public Set<Tag> tags;
101 	}
102 
103 	@Test void a05_beanPropTypeOverride_honored() {
104 		var a = json("{tags:['PRIORITY','URGENT']}", A05_Bean.class);
105 		assertInstanceOf(TreeSet.class, a.tags);  // Override wins; not the LinkedHashSet default.
106 		// params={Tag.class} carries the element type, so elements are coerced and TreeSet orders by enum ordinal.
107 		assertEquals(List.of(Tag.URGENT, Tag.PRIORITY), new ArrayList<>(a.tags));
108 	}
109 
110 	//------------------------------------------------------------------------------------------------------------------
111 	// a06: OQA-4 - a custom *instantiable* AbstractSet subclass (public no-arg ctor) is materialized as ITSELF.
112 	//------------------------------------------------------------------------------------------------------------------
113 
114 	public static class A06_InstantiableSet<E> extends AbstractSet<E> {
115 		private final Set<E> delegate = new LinkedHashSet<>();
116 		@Override public boolean add(E e) { return delegate.add(e); }
117 		@Override public Iterator<E> iterator() { return delegate.iterator(); }
118 		@Override public int size() { return delegate.size(); }
119 	}
120 
121 	public static class A06_Bean { public A06_InstantiableSet<Tag> tags; }
122 
123 	@Test void a06_instantiableCustomAbstractSet_materializedAsItself() {
124 		var a = json("{tags:['URGENT','LOW']}", A06_Bean.class);
125 		assertInstanceOf(A06_InstantiableSet.class, a.tags);
126 		assertEquals(List.of(Tag.URGENT, Tag.LOW), new ArrayList<>(a.tags));
127 	}
128 
129 	//------------------------------------------------------------------------------------------------------------------
130 	// a07: OQA-4 - a genuinely *non-instantiable* abstract set field (AbstractSet) falls back to
131 	//      LinkedHashSet (the Set shape default; LinkedHashSet is-a AbstractSet, so the field assignment succeeds).
132 	//------------------------------------------------------------------------------------------------------------------
133 
134 	public static class A07_Bean { public AbstractSet<Tag> tags; }
135 
136 	@Test void a07_nonInstantiableAbstractSet_fallsBackToLinkedHashSet() {
137 		var a = json("{tags:['URGENT','PRIORITY']}", A07_Bean.class);
138 		assertInstanceOf(LinkedHashSet.class, a.tags);
139 		assertEquals(List.of(Tag.URGENT, Tag.PRIORITY), new ArrayList<>(a.tags));
140 	}
