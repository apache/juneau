17  package org.apache.juneau.bean.rfc7807;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.apache.juneau.*;
22  import org.junit.jupiter.api.*;
23  
24  /**
25   * Tests {@link ProblemException} — construction, factory, getter round-trip, and throw/catch behavior.
26   */
27  class ProblemException_Test extends TestBase {
28  
29  	@Test
30  	void a01_constructor_storesProblem() {
31  		var problem = Problem.fromStatus(404, "Not Found", "Order 1 missing");
32  		var ex = new ProblemException(problem);
33  		assertSame(problem, ex.getProblem());
34  	}
35  
36  	@Test
37  	void a02_constructor_nullProblem_isAllowed() {
38  		var ex = new ProblemException(null);
39  		assertNull(ex.getProblem());
40  		assertNull(ex.getMessage());
41  	}
42  
43  	@Test
44  	void a03_message_fromDetail() {
45  		var problem = Problem.fromStatus(500, "Internal Server Error", "boom");
46  		var ex = new ProblemException(problem);
47  		assertEquals("boom", ex.getMessage());
48  	}
49  
50  	@Test
51  	void a04_message_fallsBackToTitle_whenDetailNull() {
52  		var problem = new Problem().setStatus(403).setTitle("Forbidden");
53  		var ex = new ProblemException(problem);
54  		assertEquals("Forbidden", ex.getMessage());
55  	}
56  
57  	@Test
58  	void a05_message_nullWhenBothDetailAndTitleNull() {
59  		var problem = new Problem().setStatus(418);
60  		var ex = new ProblemException(problem);
61  		assertNull(ex.getMessage());
62  	}
63  
64  	@Test
65  	void a06_factory_of_returnsNewInstance() {
66  		var problem = Problem.fromStatus(409, "Conflict", "Version mismatch");
67  		var ex = ProblemException.of(problem);
68  		assertSame(problem, ex.getProblem());
69  		assertEquals("Version mismatch", ex.getMessage());
70  	}
71  
72  	@Test
73  	void a07_throwCatch_roundTrip() {
74  		var problem = Problem.fromStatus(429, "Too Many Requests", "Slow down");
75  		try {
76  			throw new ProblemException(problem);
77  		} catch (ProblemException caught) {
78  			assertSame(problem, caught.getProblem());
79  			assertEquals(Integer.valueOf(429), caught.getProblem().getStatus());
80  			assertEquals("Too Many Requests", caught.getProblem().getTitle());
81  			assertEquals("Slow down", caught.getProblem().getDetail());
82  		}
83  	}
84  
85  	@Test
86  	void a08_isRuntimeException() {
87  		var ex = new ProblemException(Problem.fromStatus(500, "Internal Server Error", "x"));
88  		assertTrue(ex instanceof RuntimeException);
89  	}
90  
91  	@Test
92  	void a09_preservesExtensionFields() {
93  		var problem = Problem.fromStatus(403, "Insufficient credit", "Balance 30 < cost 50")
94  			.set("balance", 30)
95  			.set("accounts", "[a,b]");
96  		var ex = new ProblemException(problem);
97  		var recovered = ex.getProblem();
98  		assertEquals(30, recovered.get("balance"));
99  		assertEquals("[a,b]", recovered.get("accounts"));
100 	}
101 }
