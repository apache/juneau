26  /**
27   * Tests {@link ProblemAdapters#fromException(BasicHttpException)} across the 8 most-common
28   * {@link BasicHttpException} subclasses, plus null safety and the {@code detail}-vs-reason-phrase fallback.
29   */
30  class ProblemAdapters_Test extends TestBase {
31  
32  	// -----------------------------------------------------------------------------------------------------------------
33  	// A: per-subclass status + title mapping
34  	// -----------------------------------------------------------------------------------------------------------------
35  
36  	@Test
37  	void a01_badRequest_noMessage() {
38  		var p = ProblemAdapters.fromException(new BadRequest());
39  		assertEquals(Integer.valueOf(400), p.getStatus());
40  		assertEquals("Bad Request", p.getTitle());
41  		assertNull(p.getDetail(), "detail should be null when message == reason phrase");
42  		assertNull(p.getType(), "type must not be synthesized");
43  	}
44  
45  	@Test
46  	void a02_unauthorized_noMessage() {
47  		var p = ProblemAdapters.fromException(new Unauthorized());
48  		assertEquals(Integer.valueOf(401), p.getStatus());
49  		assertEquals("Unauthorized", p.getTitle());
50  		assertNull(p.getDetail());
51  	}

101 	// -----------------------------------------------------------------------------------------------------------------
102 	// B: with-message variants — detail should be populated when distinct from reason phrase
103 	// -----------------------------------------------------------------------------------------------------------------
104 
105 	@Test
106 	void b01_badRequest_withMessage() {
107 		var p = ProblemAdapters.fromException(new BadRequest("Field %s is invalid", "balance"));
108 		assertEquals(Integer.valueOf(400), p.getStatus());
109 		assertEquals("Bad Request", p.getTitle());
110 		assertEquals("Field balance is invalid", p.getDetail());
111 	}
112 
113 	@Test
114 	void b02_notFound_withFormatArgs() {
115 		var p = ProblemAdapters.fromException(new NotFound("Order %s not found", 42));
116 		assertEquals(Integer.valueOf(404), p.getStatus());
117 		assertEquals("Not Found", p.getTitle());
118 		assertEquals("Order 42 not found", p.getDetail());
119 	}
120 
121 	@Test
122 	void b03_internalServerError_withMessage() {
123 		var p = ProblemAdapters.fromException(new InternalServerError("DB down"));
124 		assertEquals(Integer.valueOf(500), p.getStatus());
125 		assertEquals("Internal Server Error", p.getTitle());
126 		assertEquals("DB down", p.getDetail());
127 	}
128 
129 	@Test
130 	void b04_serviceUnavailable_withMessageMatchingReasonPhrase_omitsDetail() {
131 		// Explicit message that exactly matches the reason phrase should still be suppressed
132 		// (the bare-`new ServiceUnavailable()` case echoes the reason phrase via BasicHttpException.getMessage()).
133 		var p = ProblemAdapters.fromException(new ServiceUnavailable("Service Unavailable"));
134 		assertEquals(Integer.valueOf(503), p.getStatus());
135 		assertEquals("Service Unavailable", p.getTitle());
136 		assertNull(p.getDetail());
137 	}

139 	// -----------------------------------------------------------------------------------------------------------------
140 	// C: with-cause variants — detail flows from cause.getMessage() when no explicit message is given
141 	// -----------------------------------------------------------------------------------------------------------------
142 
143 	@Test
144 	void c01_internalServerError_withCause() {
145 		var cause = new IllegalStateException("boom");
146 		var p = ProblemAdapters.fromException(new InternalServerError(cause));
147 		assertEquals(Integer.valueOf(500), p.getStatus());
148 		assertEquals("Internal Server Error", p.getTitle());
149 		assertEquals("boom", p.getDetail());
150 	}
151 
152 	@Test
153 	void c02_notFound_withCauseAndMessage() {
154 		var cause = new IllegalArgumentException("ignored");
155 		var p = ProblemAdapters.fromException(new NotFound(cause, "Order %s missing", 7));
156 		assertEquals(Integer.valueOf(404), p.getStatus());
157 		assertEquals("Not Found", p.getTitle());
158 		assertEquals("Order 7 missing", p.getDetail());
159 	}
160 
161 	@Test
162 	void c03_forbidden_withNullCause() {
163 		var p = ProblemAdapters.fromException(new Forbidden((Throwable) null));
164 		assertEquals(Integer.valueOf(403), p.getStatus());
165 		assertEquals("Forbidden", p.getTitle());
166 		assertNull(p.getDetail());
167 	}
168 
169 	@Test
170 	void c04_conflict_withCauseHavingNullMessage() {
171 		var cause = new RuntimeException((String) null);
172 		var p = ProblemAdapters.fromException(new Conflict(cause));
173 		assertEquals(Integer.valueOf(409), p.getStatus());
174 		assertEquals("Conflict", p.getTitle());
175 		// cause.getMessage() == null → BasicHttpException.getMessage() falls back to the reason phrase,
176 		// which the adapter then suppresses.
177 		assertNull(p.getDetail());
178 	}
179 
180 	// -----------------------------------------------------------------------------------------------------------------
181 	// D: null-input safety + type / instance never set
182 	// -----------------------------------------------------------------------------------------------------------------
183 
184 	@Test
185 	void d01_nullException_returnsNull() {
186 		assertNull(ProblemAdapters.fromException(null));
187 	}
188 
189 	@Test
190 	void d02_type_isNeverSynthesized() {
191 		// Locked decision Q7: never set type (don't synthesize about:blank on the wire).
192 		var p = ProblemAdapters.fromException(new BadRequest("oops"));
193 		assertNull(p.getType());
194 		assertNull(p.getInstance());
195 	}
196 
197 	@Test
198 	void d03_arbitrary_basicHttpException_status() {
199 		// Direct BasicHttpException construction with a custom status code that doesn't have a dedicated subclass.
200 		var p = ProblemAdapters.fromException(new BasicHttpException(418, "I'm a teapot", "Short and stout"));
201 		assertEquals(Integer.valueOf(418), p.getStatus());
202 		assertEquals("I'm a teapot", p.getTitle());
203 		assertEquals("Short and stout", p.getDetail());
204 	}
