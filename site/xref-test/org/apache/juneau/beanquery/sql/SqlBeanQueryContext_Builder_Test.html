50  	@Test
51  	void a01_buildRequiresDialectTableConnectionAndColumns() {
52  		assertIse("SqlBeanQueryContext requires a dialect.", SqlBeanQueryContext.create(Object.class).table("t").column("a", TEXT).connectionSupplier(() -> null));
53  		assertIse("SqlBeanQueryContext requires a table.", SqlBeanQueryContext.create(Object.class).dialect(D).column("a", TEXT).connectionSupplier(() -> null));
54  		assertIse("SqlBeanQueryContext requires a connection source.", SqlBeanQueryContext.create(Object.class).dialect(D).table("t").column("a", TEXT));
55  		assertIse("At least one column must be declared.", SqlBeanQueryContext.create(Object.class).dialect(D).table("t").connectionSupplier(() -> null));
56  		assertThrows(IllegalArgumentException.class, () -> SqlBeanQueryContext.create(null));
57  	}

77  	@Test
78  	void a03_renderRowsUsesResolvedQuery() {
79  		var c = builder().columnSql("name", "full_name").build();
80  		var s = c.renderRows(c.resolve(new BeanQuery().setSearch("age=$gt(21)").setSort("name-").setView("name").setPosition(10).setLimit(5)));
81  		assertBean(s, "sql,binds,columns", "SELECT \"full_name\" FROM \"person\" WHERE \"age\" > ? ORDER BY \"full_name\" DESC LIMIT 5 OFFSET 10,[21],[name]");
82  		assertEquals("SELECT \"full_name\", \"age\" FROM \"person\" LIMIT 100", c.renderRows(c.resolve(new BeanQuery())).sql());
83  	}

85  	@Test
86  	void a04_guardsAreParenthesizedAndAnded() {
87  		var c = builder().guard(SqlFragment.of("\"tenant\" = ? OR \"tenant\" = ?", "t1", "t2")).build();
88  		var q = c.resolve(new BeanQuery().setSearch("age=$gt(21)"));
89  		var total = c.renderCount(q, false);
90  		assertEquals("SELECT count(*) FROM \"person\" WHERE (\"tenant\" = ? OR \"tenant\" = ?)", total.sql());
91  		var matched = c.renderCount(q, true, SqlFragment.of("\"age\" < ?", 65));
92  		assertBean(matched, "sql,binds", "SELECT count(*) FROM \"person\" WHERE (\"tenant\" = ? OR \"tenant\" = ?) AND (\"age\" < ?) AND \"age\" > ?,[t1,t2,65,21]");
93  	}

95  	@Test
96  	void a05_copyRoundTrips() {
97  		var c = builder().columnSql("name", "full_name").guard(SqlFragment.of("\"x\" = 1")).build();
98  		var d = c.copy().table("people").build();
99  		var q = new BeanQuery().setView("name");
100 		assertEquals("SELECT \"full_name\" FROM \"people\" WHERE (\"x\" = 1) LIMIT 100", d.renderRows(d.resolve(q)).sql());
101 		assertEquals("SELECT \"full_name\" FROM \"person\" WHERE (\"x\" = 1) LIMIT 100", c.renderRows(c.resolve(q)).sql());  // Unchanged.
102 		assertSame(Object.class, c.getType());
103 		assertSame(Object.class, d.getType());  // Carried through copy().
104 	}

113 		var e = assertThrows(BeanQueryExecutionException.class, c::getSession);

188 	@Test
189 	void d06_explicitRowMapperWinsOverDefault() {
190 		SqlRowMapper<Person> explicit = (rs, cols) -> null;
191 		assertSame(explicit, builderOf(Person.class).rowMapper(explicit).build().rowMapper());
192 		@SuppressWarnings({
193 			"unchecked" // Class literals are raw; TreeMap.class is safe to view as the parameterized type.
194 		})
195 		var treeType = (Class<TreeMap<String,Object>>)(Class<?>)TreeMap.class;
196 		SqlRowMapper<TreeMap<String,Object>> explicitTree = (rs, cols) -> new TreeMap<>();
197 		assertSame(explicitTree, builderOf(treeType).rowMapper(explicitTree).build().rowMapper());
198 	}
