82  	/** Asserts the body fails with a {@link BeanQueryExecutionException} and that the session logged the failure once. */
83  	private static BeanQueryExecutionException assertFailsLogged(org.junit.jupiter.api.function.Executable body) {
84  		var thrown = new BeanQueryExecutionException[1];
85  		var records = LogRecordCapture.quietly(SqlBeanQuerySession.class, () -> thrown[0] = assertThrows(BeanQueryExecutionException.class, body));
86  		assertSize(records::toString, 1, records);
87  		assertTrue(records.get(0).getMessage().startsWith("Bean query failed: "), records.get(0).getMessage());
88  		return thrown[0];
89  	}

399 	void s29_queryTimeoutOverflowIsClampedNotThrown() {
400 		try (var s = builder().queryTimeout(Duration.ofSeconds(Long.MAX_VALUE)).build().getSession()) {
401 			s.find(new BeanQuery());  // Must not throw ArithmeticException from Duration.toMillis() overflowing, nor
402 			// trip a driver's own internal seconds*1000 overflow (confirmed real in H2 2.3.232's setQueryTimeout(int)).
403 		}
404 		// Clamp bound is Integer.MAX_VALUE / 1000, not Integer.MAX_VALUE: some JDBC drivers multiply the seconds
405 		// value by 1000 internally in 32-bit int math, so clamping at the full Integer.MAX_VALUE would overflow
406 		// that downstream multiplication. Integer.MAX_VALUE / 1000 (~24.8 days) is still effectively "no timeout".
407 		assertEquals(List.of("queryTimeout " + (Integer.MAX_VALUE / 1000)), r.events("queryTimeout"));
408 	}

586 	@Test
587 	void n01_columnLevelNot_matchesNullCell() {
588 		try (var s = nullableContext().getSession()) {
589 			// In-memory: $gt(5) is false for a null cell, so $not($gt(5)) is true for it.
590 			assertEquals(Arrays.asList(null, "Bob"), names(s.find(new BeanQuery().setSearch("age=$not($gt(5))").setSort("age")).rows()));
591 			assertEquals(Arrays.asList(null, "Bob"), names(s.find(new BeanQuery().setSearch("name=$not($eq(Alice))").setSort("age")).rows()));
592 		}
593 	}

595 	@Test
596 	void n02_topLevelFilterNot_matchesNullCell() throws SQLException {
597 		var ctx = nullableContext();
598 		var op = SearchOperatorSet.standard();
599 		var leaf = new Filter.Leaf("name", SearchExpressionParser.parse("$eq(Alice)", op), SearchType.TEXT);
600 		var where = SqlSearchCompiler.create(TestDialect.INSTANCE, ctx).compile(new Filter.Not(leaf));
601 		var names = new ArrayList<String>();
602 		try (var c = open(); var ps = c.prepareStatement("SELECT \"name\", \"age\" FROM \"nullable\" WHERE " + where.sql() + " ORDER BY \"age\"")) {
603 			ps.setObject(1, where.binds().get(0));
604 			try (var rs = ps.executeQuery()) {
605 				while (rs.next())
606 					names.add(rs.getString(1));
607 			}
608 		}
609 		assertEquals(Arrays.asList(null, "Bob"), names);
