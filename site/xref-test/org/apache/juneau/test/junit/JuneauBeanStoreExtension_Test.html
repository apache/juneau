25  /**
26   * Tests for {@link JuneauBeanStoreExtension} discovery and per-test isolation.
27   *
28   * <p>
29   * Phase 2 covers {@code @TestBean} field/method discovery, per-test overlay isolation, parameter resolution,
30   * named-bean qualifiers, and explicit-type overrides.  Class-scope behavior is exercised by
31   * {@code JuneauBeanStoreExtension_ClassScope_Test} (Phase 3).
32   *
33   * <p>
34   * The host class itself is wired via {@code @ExtendWith(JuneauBeanStoreExtension.class)} so the lifecycle
35   * callbacks (beforeAll/afterAll/beforeEach/afterEach/ParameterResolver) are exercised at runtime by every test.
36   * Error-condition tests use the package-private {@code buildClassScopeStore} / {@code buildMethodScopeStore}
37   * helpers to drive the discovery layer directly &mdash; that lets them assert on failure messages without
38   * spawning a nested JUnit launcher.
39   */
40  @ExtendWith(JuneauBeanStoreExtension.class)
41  @SuppressWarnings({
42  	"java:S5778",  // assertThrows lambdas with chained calls; intermediate invocations do not throw in practice
43  	"resource" // Bean stores/overlays are Closeables whose lifecycle is owned by the extension/test; Eclipse JDT @Owning warning is by design.
44  })

267 	@Test
268 	void h01_getStore_scope_method_isPopulatedDuringTestMethod(TestBeanStore store) {
269 		// The injected store is the method-scope store, which must be populated with the host class's
270 		// @TestBean beans during the test method.
271 		assertNotNull(store, "Injected method-scope store should be populated during the test method.");
272 		assertTrue(store.hasBean(Svc.class), "Method-scope store should contain the @TestBean Svc bean during the test method.");
273 
274 		// The extension is registered on this host class; calling create() yields a new, unbound extension.
275 		// Asking for either scope on the unbound extension must be empty.
276 		var unbound = JuneauBeanStoreExtension.create();
277 		assertTrue(unbound.getStore(Scope.METHOD).isEmpty());
278 		assertTrue(unbound.getStore(Scope.CLASS).isEmpty());
279 		assertThrows(IllegalStateException.class, unbound::getStore);
280 	}

315 	@Test
316 	void i04_resolveParameter_fallsBackToClassScope() {
317 		var ext = JuneauBeanStoreExtension.create();
318 		var classCtx = org.apache.juneau.test.junit.testsupport.StubExtensionContext.of(F02_NullableField.class, null);
319 		ext.beforeAll(classCtx);
320 		try {
321 			// No beforeEach -> no method-scope store; resolveParameter must fall back to class-scope.
322 			var resolved = ext.resolveParameter(null, classCtx);
