114 	@SuppressWarnings({
115 		"java:S125"  // Explanatory prose on overload-resolution; not dead code.
116 	})
117 	@Test
118 	void c01_supplier_invokedLazily() {
119 		var calls = new AtomicInteger();
120 		var produced = new Svc("produced");
121 		// Pre-typed supplier disambiguates between override(Class<T>, T) and override(Class<T>, Supplier<T>);
122 		// Java's overload resolver carries the instance-form candidate forward when the lambda itself
123 		// is the second argument, even though Svc is not a functional interface.
124 		Supplier<Svc> supplier = () -> {
125 			calls.incrementAndGet();
126 			return produced;
127 		};
128 		var store = new TestBeanStore().override(Svc.class, supplier);
129 
130 		assertEquals(0, calls.get(), "Supplier should not have fired before lookup");
131 
132 		assertSame(produced, store.getBean(Svc.class).get());
133 		assertEquals(1, calls.get(), "Supplier should have fired exactly once on first lookup");
134 	}

147 		// addSupplier is by spec re-invoked per lookup; verifies wiring through to BasicBeanStore.
148 		var counter = new AtomicInteger();
149 		Supplier<Svc> supplier = () -> new Svc("call-" + counter.incrementAndGet());
150 		var store = new TestBeanStore().override(Svc.class, supplier);
151 
152 		var a = store.getBean(Svc.class).get();
153 		var b = store.getBean(Svc.class).get();
154 		assertEquals("call-1", a.tag);
155 		assertEquals("call-2", b.tag);

162 	@Test
163 	void d01_asOverridingParent_shadowsProductionEntries() {
164 		// Simulate the production shape: a BasicBeanStore configured with @Bean factory entries
165 		// (tier 2) and a TestBeanStore in the overridingParent slot (tier 1).
166 		var fromBeanFactory = new Svc("from-bean-factory");
167 		var fromTestOverride = new Svc("from-test-override");
168 
169 		var overlay = new TestBeanStore().override(Svc.class, fromTestOverride);
170 		var production = new BasicBeanStore(null, overlay).addBean(Svc.class, fromBeanFactory);
171 
172 		assertSame(fromTestOverride, production.getBean(Svc.class).get(),
173 			"TestBeanStore in overridingParent slot should shadow local @Bean entries");
174 	}
