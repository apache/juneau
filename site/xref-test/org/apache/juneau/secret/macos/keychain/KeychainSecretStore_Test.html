31  /**
32   * Coverage for {@link KeychainSecretStore}.
33   *
34   * <p>
35   * The round-trip tests run against an in-memory {@link FakeKeychain} (via {@link KeychainSecretStore}'s
36   * package-private {@code CommandRunner} seam) rather than the real {@code security} CLI, so they exercise this
37   * store's real argument construction and output parsing deterministically on every OS &mdash; no
38   * {@code /usr/bin/security}, no login-keychain mutation, and no GUI unlock/auth prompt.  The {@link FailMode} and
39   * validation tests use a deliberately bad binary path so they also run deterministically on any platform without
40   * touching a real keychain.  The only test against the real {@code security} binary is {@link #f01_realKeychainRoundTrip},
41   * which mutates the login keychain (a throwaway item, always deleted) and so runs only on macOS, only when
42   * {@code /usr/bin/security} exists, and only when opted in with {@code -Djuneau.keychain.integration=true}.
43   */

54  	/**
55  	 * An in-memory fake of the real {@code security} CLI, keyed by service+account, that models the
56  	 * {@code add-generic-password}/{@code find-generic-password}/{@code delete-generic-password} argument shapes
57  	 * and exit-code/stdout contract that {@link KeychainSecretStore} depends on &mdash; including {@code -U}
58  	 * update-in-place and the doubled value/confirmation {@code -w} stdin payload &mdash; so tests can exercise the
59  	 * store's real argument-construction and output-parsing logic without shelling out to a real keychain.
60  	 */
61  	private static final class FakeKeychain implements KeychainSecretStore.CommandRunner {
62  
63  		private final Map<String,char[]> entries = new HashMap<>();
64  
65  		@Override
66  		public KeychainSecretStore.Result run(byte[] stdin, String[] args) {
67  			// store() runs "security -i" and sends the add-generic-password command line on stdin.
68  			if (args.length == 1 && args[0].equals("-i"))
69  				return interactive(stdin);
70  			return switch (args[0]) {
71  				case "find-generic-password" -> findGenericPassword(args);
72  				case "delete-generic-password" -> deleteGenericPassword(args);
73  				default -> throw new IllegalArgumentException("Unsupported command: " + args[0]);
74  			};
75  		}
76  
77  		private KeychainSecretStore.Result interactive(byte[] stdin) {
78  			var line = new String(stdin, UTF_8);
79  			if (! line.endsWith("\n") || line.indexOf('\n') != line.length() - 1)
80  				throw new IllegalArgumentException("Expected exactly one newline-terminated command line.");
81  			var args = tokenize(line.substring(0, line.length() - 1));
82  			if (! args[0].equals("add-generic-password"))
83  				throw new IllegalArgumentException("Unsupported command: " + args[0]);
84  			entries.put(key(args), opt(args, "-w").toCharArray());
85  			return new KeychainSecretStore.Result(0, new byte[0], "");
86  		}
87  
88  		/** Mirrors the verified {@code security -i} parser: whitespace-separated, double-quoted tokens, backslash escapes the next char. */
89  		static String[] tokenize(String line) {
90  			var tokens = new ArrayList<String>();
91  			var sb = new StringBuilder();
92  			var inToken = false;
93  			var quoted = false;
94  			for (var i = 0; i < line.length(); i++) {
95  				var c = line.charAt(i);
96  				if (c == '\\') {
97  					sb.append(line.charAt(++i));
98  					inToken = true;
99  				} else if (c == '"') {
100 					quoted = ! quoted;
101 					inToken = true;
102 				} else if (Character.isWhitespace(c) && ! quoted) {
103 					if (inToken)
104 						tokens.add(sb.toString());
105 					sb.setLength(0);
106 					inToken = false;
107 				} else {
108 					sb.append(c);
109 					inToken = true;
110 				}
111 			}
112 			assertFalse(quoted, "unterminated quote");
113 			if (inToken)
114 				tokens.add(sb.toString());
115 			return tokens.toArray(new String[0]);
116 		}
117 
118 		private KeychainSecretStore.Result findGenericPassword(String[] args) {
119 			var value = entries.get(key(args));
120 			if (value == null)
121 				return notFound();
122 			if (! Arrays.asList(args).contains("-g"))
123 				return new KeychainSecretStore.Result(0, new byte[0], "");
124 			return new KeychainSecretStore.Result(0, "keychain: \"/fake/login.keychain-db\"\n".getBytes(UTF_8), passwordLine(value));
125 		}
126 
127 		/** Mirrors the verified {@code security -g} stderr: plain printable text is quoted, anything else is hex plus an escaped rendering. */
128 		static String passwordLine(char[] value) {
129 			var bytes = new String(value).getBytes(UTF_8);
130 			if (bytes.length == 0)
131 				return "password: \n";
132 			var plain = true;
133 			for (var b : bytes)
134 				if (b < 0x20 || b > 0x7e || b == '"' || b == '\\')
135 					plain = false;
136 			if (plain)
137 				return "password: \"" + new String(bytes, UTF_8) + "\"\n";
138 			var hex = new StringBuilder("password: 0x");
139 			for (var b : bytes)
140 				hex.append(f("%02X", b));
141 			return hex + "  \"<escaped>\"\n";
142 		}
143 
144 		private KeychainSecretStore.Result deleteGenericPassword(String[] args) {
145 			if (entries.remove(key(args)) == null)
146 				return notFound();
147 			return new KeychainSecretStore.Result(0, new byte[0], "");
148 		}
149 
150 		private static KeychainSecretStore.Result notFound() {
151 			return new KeychainSecretStore.Result(KeychainSecretStore.NOT_FOUND, new byte[0], "The specified item could not be found in the keychain.");
152 		}
153 
154 		private static String key(String[] args) {
155 			return opt(args, "-s") + "\u0000" + opt(args, "-a");
156 		}
157 
158 		private static String opt(String[] args, String flag) {
159 			for (var i = 0; i < args.length - 1; i++)
160 				if (args[i].equals(flag))
161 					return args[i + 1];
162 			return null;
163 		}
164 	}

201 	@Test void a03_secretNeverAppearsOnArgv() throws IOException {
202 		assumeTrue(posixShellAvailable(), "POSIX shell not available");
203 		var argvDump = File.createTempFile("juneau-keychain-argv", ".txt");
204 		var stdinDump = File.createTempFile("juneau-keychain-stdin", ".txt");
205 		argvDump.deleteOnExit();
206 		stdinDump.deleteOnExit();
207 		var script = fakeSecurity(""
208 			+ "printf '%s\\n' \"$@\" > '" + argvDump.getAbsolutePath() + "'\n"
209 			+ "cat > '" + stdinDump.getAbsolutePath() + "'\n"
210 			+ "exit 0");
211 		try {
212 			var secret = "arg\u00eev-h0stile s3cr3t \t \u00fc";
213 			var store = new KeychainSecretStore("svc", FailMode.FAIL_CLOSED, 5L, script.getAbsolutePath());
214 			store.store("acct", secret.toCharArray());
215 
216 			var argvContent = Files.readString(argvDump.toPath());
217 			assertFalse(argvContent.contains(secret), "secret must not appear in the child process's argv");
218 			assertEquals("-i\n", argvContent, "store() must invoke 'security -i' with no other arguments (no bare -w prompt)");
219 
220 			var stdinContent = Files.readString(stdinDump.toPath());
221 			assertEquals("add-generic-password -U -a \"acct\" -s \"svc\" -w \"" + secret + "\"\n", stdinContent, "command line should be delivered via stdin");
222 		} finally {
223 			argvDump.delete();
224 			stdinDump.delete();
225 			script.delete();
226 		}
227 	}

367 		assertArrayEquals("pw".toCharArray(), stderrReturning("password: \"pw\"\n").find("k").orElseThrow());
368 		assertArrayEquals("pw".toCharArray(), stderrReturning("warn: x\npassword: \"pw\"\r\nmore\n").find("k").orElseThrow());
369 		assertArrayEquals("pw".toCharArray(), stderrReturning("password: \"pw\"").find("k").orElseThrow());

374 		// Hex form (tab, non-ASCII, quote/backslash), including the trailing escaped rendering.

379 		// Looks like hex or a hex prefix but is quoted text, so it is literal.
380 		assertArrayEquals("abcd".toCharArray(), stderrReturning("password: \"abcd\"\n").find("k").orElseThrow());

476 	@Test void f01_realKeychainRoundTrip() {
477 		assumeTrue("true".equals(System.getProperty("juneau.keychain.integration")), "Set -Djuneau.keychain.integration=true to run against the real login keychain");
478 		assumeTrue(System.getProperty("os.name", "").toLowerCase().contains("mac"), "macOS only");
479 		assumeTrue(new File("/usr/bin/security").canExecute(), "/usr/bin/security not available");
480 		var service = "org.apache.juneau.test.j0840." + UUID.randomUUID();
481 		var store = new KeychainSecretStore(service, FailMode.FAIL_CLOSED, 20L);
482 		var acct = "it acct \"q\" \\";
483 		var secret = "pa ss\"w\\rd 'x' $y";
484 		try {
485 			assertFalse(store.exists(acct));
486 			store.store(acct, secret.toCharArray());
487 			assertTrue(store.exists(acct));
488 			assertArrayEquals(secret.toCharArray(), store.find(acct).orElseThrow());
489 			// Tab, non-ASCII, hex-looking, and empty secrets (the CLI's -w flag returns a hex dump for some of these).
490 			for (var v : new String[] {"a\tb", "\u00e9 \u00fc \u6f22", "abcd", "0x41", "610962", ""}) {
491 				store.store(acct, v.toCharArray());
492 				assertArrayEquals(v.toCharArray(), store.find(acct).orElseThrow(), "value=" + v);
493 			}
494 			store.store(acct, "second".toCharArray());
495 			assertArrayEquals("second".toCharArray(), store.find(acct).orElseThrow());
496 		} finally {
497 			store.delete(acct);
498 		}
499 		assertFalse(store.exists(acct));
