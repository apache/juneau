39  	@Test void a01_beanFactory_create() throws Exception {
40  		BeanFactory<String> a = () -> "hello";
41  		assertEquals("hello", a.create());
42  	}

44  	@SuppressWarnings({
45  		"rawtypes"  // Raw types required for generic test utility.
46  	})
47  	@Test void a02_beanFactory_void_throws() throws Exception {
48  		var ctor = BeanFactory.Void.class.getDeclaredConstructor();
49  		ctor.setAccessible(true);
50  		var instance = (BeanFactory) ctor.newInstance();
51  		assertThrows(UnsupportedOperationException.class, instance::create);
52  	}

57  
58  	@Test void b01_beanConsumer_defaultLifecycle() throws Exception {
59  		var received = new ArrayList<String>();
60  		BeanConsumer<String> a = received::add;
61  		a.begin();
62  		a.acceptThrows("x");
63  		a.acceptThrows("y");
64  		a.complete();
65  		assertEquals(List.of("x", "y"), received);
66  	}
67  
68  	@Test void b02_beanConsumer_onError_rethrows_by_default() {
69  		BeanConsumer<String> a = item -> {};
70  		assertThrows(RuntimeException.class, () -> a.onError(new RuntimeException("oops")));
71  	}
72  
73  	@Test void b03_beanConsumer_accept_wraps_checked_exception() {
74  		BeanConsumer<String> a = item -> { throw new Exception("checked"); };
75  		assertThrows(RuntimeException.class, () -> a.accept("x"));
76  	}
77  
78  	@Test void b04_beanConsumer_onError_absorb_allows_continue() throws Exception {
79  		var skipped = new ArrayList<String>();
80  		BeanConsumer<String> a = new BeanConsumer<>() {
81  			@Override public void acceptThrows(String item) throws Exception {
82  				if ("bad".equals(item)) throw new Exception("bad item");
83  			}
84  			@Override public void onError(Exception e) {
85  				skipped.add(e.getMessage());
86  			}
87  		};
88  		a.begin();
89  		a.acceptThrows("good");
90  		try { a.acceptThrows("bad"); } catch (Exception e) { a.onError(e); }
91  		a.acceptThrows("good2");
92  		a.complete();
93  		assertEquals(List.of("bad item"), skipped);
94  	}

100 	@Test void c01_beanSupplier_defaultLifecycle() throws Exception {
101 		var data = List.of("a", "b", "c");
102 		BeanSupplier<String> a = data::iterator;
103 		a.begin();
104 		var result = new ArrayList<String>();
105 		a.iterator().forEachRemaining(result::add);
106 		a.complete();
107 		assertEquals(data, result);

124 	@Test void d01_beanChannel_extendsConsumerAndSupplier() {
125 		ListBeanChannel<String> a = new ListBeanChannel<>();
126 		assertTrue(a instanceof BeanConsumer);
127 		assertTrue(a instanceof BeanSupplier);
128 		assertTrue(a instanceof BeanChannel);
129 		assertTrue(a instanceof Iterable);
130 	}
131 
132 	@Test void d02_beanChannel_defaultLifecycleMethods() throws Exception {
133 		BeanChannel<String> a = new BeanChannel<>() {
134 			@Override public Iterator<String> iterator() { return Collections.emptyIterator(); }
135 			@Override public void acceptThrows(String item) { /* intentionally empty */ }
136 		};
137 		// begin and complete are no-ops by default
138 		a.begin();
139 		a.complete();
140 		// onError rethrows by default
141 		var x = new Exception("x");
142 		var thrown = assertThrows(Exception.class, () -> a.onError(x));
143 		assertSame(x, thrown);
144 	}

150 	@Test void e01_listBeanChannel_collectAndIterate() {
151 		var a = new ListBeanChannel<String>();
152 
153 		a.acceptThrows("one");
154 		a.acceptThrows("two");
155 		a.acceptThrows("three");
156 
157 		assertEquals(List.of("one", "two", "three"), a.getList());
158 
159 		var result = new ArrayList<String>();
160 		a.iterator().forEachRemaining(result::add);
161 		assertEquals(List.of("one", "two", "three"), result);
162 	}
