29   * Tests for {@link BeanStore#invokeBeanMethod(Class, MethodInfo, Object, Object...)},
30   * {@link BeanAnnotation#find(MethodInfo)} and the second (non-public, {@code @Bean}-annotated) pass of
31   * {@link BasicBeanStore#createBeanFromMethod(Class, Object, java.util.function.Predicate, Object...)}.
32   */
33  @SuppressWarnings({
34  	"java:S114", // Nested fixture classes are named for the scenario they exercise (Pass2PkgOnly, FindBase, ...), not to a type-naming convention
35  	"java:S1144", // Unused private fixture methods are invoked reflectively
36  	"java:S1172", // Unused fixture parameters
37  	"java:S2094", // Fixture classes such as ChildOfPrivate and CovChild are intentionally empty subclasses that exist only to drive inheritance lookups
38  	"resource", // The BasicBeanStore instances created per test are in-memory and never need closing
39  	"unused" // Fixture methods declare parameters/locals that are never read because they are only invoked reflectively via invokeBeanMethod
40  })

43  	static class Widget {
44  		final String tag;
45  		Widget(String tag) { this.tag = tag; }
46  		String getTag() { return tag; }
47  	}
48  
49  	static class SubWidget extends Widget {
50  		SubWidget(String tag) { super(tag); }
51  	}
52  
53  	private static MethodInfo m(Class<?> c, String name) {
54  		return ClassInfo.of(c).getDeclaredMethods().stream().filter(x -> x.getNameSimple().equals(name)).findFirst().orElseThrow();
55  	}
56  
57  	private static Widget[] tags(Widget...w) { return w; }
