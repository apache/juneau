68  	/**
69  	 * A Spring-style {@code @Value} replica that {@link JsrSupport} matches by FQN.
70  	 *
71  	 * <p>
72  	 * The real Spring annotation is at {@code org.springframework.beans.factory.annotation.Value}; we
73  	 * cannot directly import that here without a compile-time Spring dependency.  Instead, this
74  	 * private annotation mirrors the FQN that {@link JsrSupport#SPRING_VALUE} matches but lives in
75  	 * the test fixture package, so {@code isValueAnnotation} alone cannot tell them apart.  We
76  	 * instead exercise both Juneau {@code @Value} and {@code @Value}+{@code @Inject} conflict paths
77  	 * directly — the Spring FQN match is exercised in the actual Spring Boot integration test
78  	 * fixture (Phase 4).
79  	 */
80  	@Retention(RetentionPolicy.RUNTIME)
81  	@Target({ElementType.FIELD, ElementType.PARAMETER, ElementType.METHOD, ElementType.CONSTRUCTOR})
82  	@interface MarkerAnnotation {}

410 	@Test
411 	void f02_valueExpression_extractsValue() {
412 		var f = org.apache.juneau.commons.reflect.ClassInfo.of(StringFieldBean.class)
413 			.getDeclaredField(x -> "greeting".equals(x.getName())).orElseThrow();
414 		var expr = f.getAnnotations().stream()
415 			.map(JsrSupport::valueExpression)
416 			.filter(Objects::nonNull)
417 			.findFirst()
418 			.orElse(null);
419 		assertEquals("${" + P_STRING + ":hello}", expr);
420 	}

422 	@Test
423 	void f03_valueExpression_returnsNull_forNonValueAnnotation() {
424 		var f = org.apache.juneau.commons.reflect.ClassInfo.of(ConflictFieldBean.class)
425 			.getDeclaredField(x -> "mixed".equals(x.getName())).orElseThrow();
426 		var injectAnno = f.getAnnotations().stream()
427 			.filter(a -> JsrSupport.isInjectAnnotation(a) && ! JsrSupport.isValueAnnotation(a))
428 			.findFirst()
429 			.orElseThrow();
430 		assertNull(JsrSupport.valueExpression(injectAnno));
431 	}
