77  	// B. @ConfigProperties and @Value coexist on the same class
78  	// =================================================================================
79  
80  	@ConfigProperties(prefix = "MyServiceB01")
81  	public static class B01_Bean {
82  		public String host = "localhost";
83  
84  		// Deliberately resolves a DIFFERENT underlying key than the binder's implicit "MyServiceB01.label"
85  		// candidate (prefix + field name) — see b01 below: overriding that candidate key only reaches this
86  		// field if the binder incorrectly treats a @Value-owned field as bindable.
87  		@Value("${MyServiceB01.actualLabelKey:default-label}")
88  		public String label;
89  	}
90  
91  	@Test void b01_configPropertiesAndValueCoexistOnSameClass() {
92  		var settings = Settings.get();
93  		settings.setGlobal("MyServiceB01.host", "example.com");
94  		// Sets the config key the binder WOULD bind "label" from (prefix + field name) were it to incorrectly
95  		// treat this @Value-owned field as bindable. A correct binder skips it (isBindable()), so @Value's own,
96  		// differently-keyed expression still resolves to its default and the field is left untouched end-to-end.
97  		settings.setGlobal("MyServiceB01.label", "config-should-not-win");
98  		var store = new BasicBeanStore(null);
99  		var bean = BeanInstantiator.of(B01_Bean.class, store).run();
100 		assertEquals("example.com", bean.host); // via ConfigPropertiesBinder
101 		assertEquals("default-label", bean.label); // via @Value, independently resolved; NOT clobbered by the config bind
102 	}

174 	/**
175 	 * Minimal read-only {@link BeanStore} facade used to prove that a non-{@link WritableBeanStore} parent store
176 	 * still allows binding while auto-registration is silently skipped.
177 	 */
178 	static final class F02_ReadOnlyStore implements BeanStore {
179 		private final BeanStore delegate;
180 
181 		F02_ReadOnlyStore(BeanStore delegate) {
182 			this.delegate = delegate;
183 		}
184 
185 		@Override public <T> Optional<T> getBean(Class<T> beanType) { return delegate.getBean(beanType); }
186 		@Override public <T> Optional<T> getBean(Class<T> beanType, String name) { return delegate.getBean(beanType, name); }
187 		@Override public <T> Map<String,T> getBeansOfType(Class<T> beanType) { return delegate.getBeansOfType(beanType); }
188 		@Override public boolean hasBean(Class<?> beanType) { return delegate.hasBean(beanType); }
189 		@Override public boolean hasBean(Class<?> beanType, String name) { return delegate.hasBean(beanType, name); }
190 		@Override public <T> Optional<Supplier<T>> getBeanSupplier(Class<T> beanType) { return delegate.getBeanSupplier(beanType); }
191 		@Override public <T> Optional<Supplier<T>> getBeanSupplier(Class<T> beanType, String name) { return delegate.getBeanSupplier(beanType, name); }
192 	}
193 
194 	@Test void f02_readOnlyParentStoreStillBindsAndSkipsRegistration() {
195 		var settings = Settings.get();
196 		settings.setGlobal("MyServiceA01.host", "example.com");
197 		var delegate = new BasicBeanStore(null);
198 		var readOnly = new F02_ReadOnlyStore(delegate);
199 		var bean = BeanInstantiator.of(F01_Bean.class, readOnly).run();
200 		assertEquals("example.com", bean.host); // binds even though the parent store is not writable
201 		assertFalse(delegate.getBean(F01_Bean.class).isPresent()); // and auto-registration into the wrapped delegate is skipped
202 	}

236 	// =================================================================================

238 	@ConfigProperties(prefix = "MyServiceI01")
239 	public static class I01_Base {
240 		public String host = "localhost";
241 	}
242 
243 	public static class I01_Sub extends I01_Base {}
244 
245 	@Test void i01_subtypeViaTypeStillBindsAndRegistersUnderDeclaredBaseType() {
246 		// Pins the actual declared-type semantics of the hook when combined with .type(...): the "declared
247 		// target type" used for both the @ConfigProperties lookup and auto-registration is the type passed to
248 		// of(...) (the base), not the .type(...) subtype — the mirror image of e01's not-inherited case, where
249 		// a subclass passed directly to of(...) does NOT trigger binding at all.
250 		var settings = Settings.get();
251 		settings.setGlobal("MyServiceI01.host", "example.com");
252 		var store = new BasicBeanStore(null);
253 		var bean = BeanInstantiator.of(I01_Base.class, store).type(I01_Sub.class).run();
254 		assertInstanceOf(I01_Sub.class, bean); // the requested subtype was actually instantiated
255 		assertEquals("example.com", bean.host); // bound under the BASE type's prefix
256 		assertSame(bean, store.getBean(I01_Base.class).orElse(null)); // registered under the declared BASE type
257 		assertFalse(store.getBean(I01_Sub.class).isPresent()); // NOT registered under the subtype
258 	}
