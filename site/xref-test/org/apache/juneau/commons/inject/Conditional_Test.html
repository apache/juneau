25  /**
26   * Coverage for {@code @Conditional}, {@code @ConditionalOnClass},
27   * {@code @ConditionalOnMissingBean}, and {@code @ConditionalOnProperty} on both
28   * {@code @Configuration} types (class-level) and {@code @Bean} members (member-level).
29   *
30   * <p>The evaluation semantics asserted here mirror the plan:
31   * <ul>
32   * 	<li>Class-level skip cascades to imports.
33   * 	<li>Member-level skip applies only to that specific {@code @Bean} member.
34   * 	<li>Evaluation is eager at registration time.
35   * </ul>
36   */

40  	"resource"    // Closeable resources in tests are intentionally unassigned; closing is handled by test infrastructure.

68  	/**
69  	 * Custom condition that exercises every {@link ConditionContext} accessor at least once.
70  	 *
71  	 * <p>Used to give the test suite coverage of {@code beanStore()}, {@code classLoader()}, and
72  	 * {@code annotatedElement()} (in addition to {@code settings()}, which is reached by
73  	 * {@link HasBeanProperty}).
74  	 */
75  	public static class FullContextSnapshot implements Condition {
76  		public FullContextSnapshot() { /* intentionally empty */ }
77  		@Override public boolean matches(ConditionContext ctx) {
78  			return ctx.beanStore() != null
79  				&& ctx.classLoader() != null
80  				&& ctx.annotatedElement() != null
81  				&& ctx.settings() != null;
82  		}
83  	}

396 	// Tests - logger-fine lambda coverage.
397 	//
398 	// BasicBeanStore.matchesConditions() emits two LOGGER.fine(...) messages for skip events.  When
399 	// the BasicBeanStore logger is at FINE level, the lambdas actually execute and their message-
400 	// construction bodies become observable to JaCoCo.  The tests below temporarily raise the level
401 	// and verify the skip still happens (the captured handler is asserted on too).
402 	//------------------------------------------------------------------------------------------------
403 
404 	@Configuration
405 	@ConditionalOnClass("definitely.does.not.Exist")
406 	public static class ClassLevelOnClassMissingConfig {
407 		public ClassLevelOnClassMissingConfig() { /* intentionally empty */ }
408 		@Bean public SvcA svcA() { return new SvcA("never"); }
409 	}
410 
411 	@Test
412 	void h01_loggerFine_messageLambdasExecute() {
413 		var logger = java.util.logging.Logger.getLogger(BasicBeanStore.class.getName());
414 		var originalLevel = logger.getLevel();
415 		var originalUseParent = logger.getUseParentHandlers();
416 		var captured = new java.util.ArrayList<String>();
417 		var handler = new java.util.logging.Handler() {
418 			@Override public void publish(java.util.logging.LogRecord r) { captured.add(r.getMessage()); }
419 			@Override public void flush() { /* no-op */ }
420 			@Override public void close() { /* no-op */ }
421 		};
422 		handler.setLevel(java.util.logging.Level.FINE);
423 		try {
424 			logger.setLevel(java.util.logging.Level.FINE);
425 			logger.setUseParentHandlers(false);
426 			logger.addHandler(handler);
427 
428 			// 1) AlwaysFalse @Conditional path → "Skipping conditional element due to @Conditional: …"
429 			var store1 = new BasicBeanStore(null);
430 			store1.registerConfiguration(ClassLevelFalseConfig.class);
431 			assertFalse(store1.getBean(SvcA.class).isPresent());
432 
433 			// 2) @ConditionalOnClass("definitely.does.not.Exist") → "Skipping conditional element due to missing class …"
434 			var store2 = new BasicBeanStore(null);
435 			store2.registerConfiguration(ClassLevelOnClassMissingConfig.class);
436 			assertFalse(store2.getBean(SvcA.class).isPresent());
437 
438 			assertTrue(captured.stream().anyMatch(m -> m.contains("@Conditional")),
439 				"Expected a @Conditional skip log; got: " + captured);
440 			assertTrue(captured.stream().anyMatch(m -> m.contains("missing class")),
441 				"Expected a missing-class skip log; got: " + captured);
442 		} finally {
443 			logger.removeHandler(handler);
444 			logger.setLevel(originalLevel);
445 			logger.setUseParentHandlers(originalUseParent);
446 		}
447 	}
