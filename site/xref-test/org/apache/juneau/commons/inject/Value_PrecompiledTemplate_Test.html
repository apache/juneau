17  package org.apache.juneau.commons.inject;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.apache.juneau.commons.*;
22  import org.apache.juneau.commons.settings.*;
23  import org.apache.juneau.commons.svl.*;
24  import org.junit.jupiter.api.*;
25  
26  /**
27   * Phase G regression: verifies that {@code @Value("${...}")} expressions are tokenized
28   * <i>once</i> per distinct expression and cached as {@link VarTemplate} instances on
29   * {@link ValueResolver}'s internal cache.
30   *
31   * <p>
32   * Without the precompile retrofit, every bean construction re-tokenized the expression
33   * via {@code VarResolver.DEFAULT.resolve(expr)}. After the retrofit, the first lookup
34   * caches the compiled {@link VarTemplate} and subsequent lookups return the same
35   * instance, skipping the tokenizer + var-registry-lookup pipeline.
36   */
37  @SuppressWarnings({
38  	"resource" // BasicBeanStore is a test-only fluent AutoCloseable with no real resources to release
39  })
40  class Value_PrecompiledTemplate_Test extends TestBase {
41  
42  	private static final String P_KEY = "Value_PrecompiledTemplate_Test.key";
43  
44  	public static class GreetingBean {
45  		@Value("${Value_PrecompiledTemplate_Test.key:hello}")
46  		String greeting;
47  	}
48  
49  	@BeforeEach
50  	void setUp() {
51  		ValueResolver.clearTemplateCache();
52  	}
53  
54  	@AfterEach
55  	void tearDown() {
56  		Settings.get().unsetGlobal(P_KEY);
57  		ValueResolver.clearTemplateCache();
58  	}
59  
60  	@Test void a01_sameExpressionReturnsSameTemplate() {
61  		var t1 = ValueResolver.getCompiledTemplate("${" + P_KEY + ":hello}");
62  		var t2 = ValueResolver.getCompiledTemplate("${" + P_KEY + ":hello}");
63  		assertSame(t1, t2, "Cached VarTemplate must be returned on second lookup");
64  	}
65  
66  	@Test void a02_distinctExpressionsCachedSeparately() {
67  		var t1 = ValueResolver.getCompiledTemplate("${a:1}");
68  		var t2 = ValueResolver.getCompiledTemplate("${b:2}");
69  		assertNotSame(t1, t2, "Distinct expressions must produce distinct VarTemplate instances");
70  		assertSame(t1, ValueResolver.getCompiledTemplate("${a:1}"));
71  		assertSame(t2, ValueResolver.getCompiledTemplate("${b:2}"));
72  	}
73  
74  	@Test void a03_repeatedFieldInjectionReusesCachedTemplate() {
75  		// First bean construction should populate the cache; second construction must hit the
76  		// cached VarTemplate (verified by getCompiledTemplate returning the same instance).
77  		var bs = new BasicBeanStore(null);
78  		var bean1 = BeanInstantiator.of(GreetingBean.class, bs).run();
79  		var bean2 = BeanInstantiator.of(GreetingBean.class, bs).run();
80  		assertEquals("hello", bean1.greeting);
81  		assertEquals("hello", bean2.greeting);
82  
83  		var t1 = ValueResolver.getCompiledTemplate("${" + P_KEY + ":hello}");
84  		var t2 = ValueResolver.getCompiledTemplate("${" + P_KEY + ":hello}");
85  		assertSame(t1, t2);
86  	}
87  
88  	@Test void a04_cachedTemplateRespectsSettingsUpdate() {
89  		var bs = new BasicBeanStore(null);
90  		var bean1 = BeanInstantiator.of(GreetingBean.class, bs).run();
91  		assertEquals("hello", bean1.greeting);
92  
93  		Settings.get().setGlobal(P_KEY, "world");
94  		var bean2 = BeanInstantiator.of(GreetingBean.class, bs).run();
95  		assertEquals("world", bean2.greeting,
96  			"Cached template re-resolves against current Settings on each injection (Settings/PropertyVar is intentionally not stable-folded)");
97  	}
98  
99  	@Test void a05_literalExpressionCachedAsLiteralTemplate() {
100 		var t = ValueResolver.getCompiledTemplate("plain literal");
101 		assertTrue(t.isLiteral(), "Literal expressions fold to a single LiteralSegment at compile time");
102 		assertSame(t, ValueResolver.getCompiledTemplate("plain literal"));
103 	}
104 }
