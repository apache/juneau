82  		var logger = getLogger("b01");

215 	@Test void f01_close_removesListener() {
216 		var logger = getLogger("f01");
217 		LogRecordCapture capture = null;
218 		try {
219 			capture = logger.captureEvents();
220 			logger.info("Message 1");
221 			assertSize(1, capture.getRecords());
222 		} finally {
223 			if (capture != null) {
224 				capture.close();
225 			}
226 		}
227 
228 		// After close, new capture should only see new messages
229 		try (var capture2 = logger.captureEvents()) {
230 			logger.info("Message 2");
231 			assertSize(1, capture2.getRecords());
232 			assertEquals("Message 2", capture2.getRecords().get(0).getMessage());
233 		}
234 	}

254 	@Test void g01_concurrentLogging_capturesAll() throws InterruptedException {
255 		var logger = getLogger("g01");
256 		try (var capture = logger.captureEvents()) {
257 			var thread1 = new Thread(() -> logger.info("Thread 1"));
258 			var thread2 = new Thread(() -> logger.info("Thread 2"));
259 			var thread3 = new Thread(() -> logger.info("Thread 3"));
260 
261 			thread1.start();
262 			thread2.start();
263 			thread3.start();
264 
265 			thread1.join();
266 			thread2.join();
267 			thread3.join();
268 
269 			// All messages should be captured
270 			var records = capture.getRecords();
271 			assertEquals(3, records.size());
272 		}
273 	}
