915 		"java:S2925" // Polling for an async GC event has no latch/await equivalent; the sleep is load-bearing for the poll interval.
916 	})
917 	private static boolean awaitCollected(WeakReference<?> ref) throws Exception {
918 		for (int i = 0; i < 20; i++) {
919 			System.gc();
920 			Thread.sleep(20);
921 			if (ref.get() == null)
922 				return true;
923 		}
924 		return false;
