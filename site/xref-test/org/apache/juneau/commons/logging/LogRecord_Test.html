309 	private static LogRecord roundTrip(LogRecord rec) throws Exception {
310 		byte[] data;
311 		try (var baos = new ByteArrayOutputStream(); var oos = new ObjectOutputStream(baos)) {
312 			oos.writeObject(rec);
313 			oos.flush();
314 			data = baos.toByteArray();
315 		}
316 		try (var bais = new ByteArrayInputStream(data); var ois = new ObjectInputStream(bais)) {
317 			return (LogRecord)ois.readObject();
318 		}
319 	}
