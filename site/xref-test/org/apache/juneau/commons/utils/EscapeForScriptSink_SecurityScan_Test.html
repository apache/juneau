27  /**
28   * Script-sink gate: the framework's {@code <script>}-JSON sinks must route through
29   * {@link StringUtils#escapeForScript(String)}.
30   *
31   * <p>
32   * Pairs a set of unit RED/GREEN checks on {@link ScriptJsonSinkScanner} (proving it flags an unescaped sink, passes
33   * an escaped one, ignores non-{@code <script>} raw text, and is not fooled by a comment mention of the escaper) with
34   * a live scan of the real source tree that both asserts zero violations <i>and</i> asserts the scanner still finds
35   * the known-good {@code RegionTable} sink &mdash; so a green result can never mean "nothing was
36   * examined".
37   */
38  class EscapeForScriptSink_SecurityScan_Test extends TestBase {
39  
40  	// -----------------------------------------------------------------------------------------------------------
41  	// Unit RED/GREEN on the scanner itself
42  	// -----------------------------------------------------------------------------------------------------------
43  
44  	/** RED: a serialized payload written into a raw-text {@code <script>} without escapeForScript is a violation. */
45  	@Test void a01_unescapedScriptRawText_isFlagged() {
46  		var src = """
47  			class X {
48  			  Object of(Object bean) {
49  			    var json = Json.of(bean);
50  			    return script().type("application/json").text(rawText(json));
51  			  }
52  			}
53  			""";
54  		var r = ScriptJsonSinkScanner.scan(src);
55  		assertSize(() -> "sinks: " + r.sinks(), 1, r.sinks());
56  		assertSize(() -> "violations: " + r.violations(), 1, r.violations());
57  		assertFalse(r.sinks().get(0).escaped());
58  	}
59  
60  	/** GREEN: the ViewTable shape - escaped via a same-source variable assignment - passes. */
61  	@Test void a02_escapedViaVariable_passes() {
62  		var src = """
63  			class X {
64  			  Object of(Object bean) {
65  			    var json = escapeForScript(Json.of(bean));
66  			    return script().type("application/json").text(rawText(json));
67  			  }
68  			}
69  			""";
70  		var r = ScriptJsonSinkScanner.scan(src);
71  		assertSize(() -> "sinks: " + r.sinks(), 1, r.sinks());
72  		assertEquals(java.util.List.of(), r.violations());
73  		assertTrue(r.sinks().get(0).escaped());
74  	}
75  
76  	/** GREEN: inline escapeForScript(...) directly inside rawText(...) passes. */
77  	@Test void a03_escapedInline_passes() {
78  		var src = """
79  			class X {
80  			  Object of(Object bean) {
81  			    return script().text(rawText(escapeForScript(Json.of(bean))));
82  			  }
83  			}
84  			""";
85  		var r = ScriptJsonSinkScanner.scan(src);
86  		assertSize(() -> "sinks: " + r.sinks(), 1, r.sinks());
87  		assertEquals(java.util.List.of(), r.violations());
88  	}
89  
90  	/** A raw-text write that is not into a {@code <script>} element is not a sink (JSON-in/JSON-out scope, not widened). */
91  	@Test void a04_nonScriptRawText_isNotASink() {
92  		var src = """
93  			class X {
94  			  Object of(Object html) {
95  			    return div().text(rawText(html));
96  			  }
97  			}
98  			""";
99  		var r = ScriptJsonSinkScanner.scan(src);
100 		assertEquals(java.util.List.of(), r.sinks(), () -> "sinks: " + r.sinks());
101 		assertEquals(java.util.List.of(), r.violations());
102 	}
103 
104 	/**
105 	 * A {@code escapeForScript} mention that lives only in a comment must NOT satisfy the guard - this is the exact
106 	 * silent-decay failure mode the scanner exists to prevent, so it is asserted directly.
107 	 */
108 	@Test void a05_commentMentionOfEscaper_doesNotSatisfyGuard() {
109 		var src = """
110 			class X {
111 			  // We used to call escapeForScript(...) here; the code below no longer does.
112 			  Object of(Object bean) {
113 			    var json = Json.of(bean); // escapeForScript would go here
114 			    return script().type("application/json").text(rawText(json));
115 			  }
116 			}
117 			""";
118 		var r = ScriptJsonSinkScanner.scan(src);
119 		assertSize(() -> "sinks: " + r.sinks(), 1, r.sinks());
120 		assertSize(() -> "comment mention must not satisfy the guard: " + r.violations(), 1, r.violations());
121 	}
122 
123 	/** {@code new RawText(...)} is recognized as the same raw-text primitive as {@code rawText(...)}. */
124 	@Test void a06_newRawTextConstructor_isAlsoASink() {
125 		var escaped = ScriptJsonSinkScanner.scan("""
126 			class X { Object of(Object b) { return script().text(new RawText(escapeForScript(Json.of(b)))); } }
127 			""");
128 		assertSize(1, escaped.sinks());
129 		assertEquals(java.util.List.of(), escaped.violations());
130 
131 		var unescaped = ScriptJsonSinkScanner.scan("""
132 			class X { Object of(Object b) { return script().text(new RawText(Json.of(b))); } }
133 			""");
134 		assertSize(1, unescaped.sinks());
135 		assertSize(1, unescaped.violations());
136 	}

138 	// -----------------------------------------------------------------------------------------------------------
139 	// Live gate against the real framework tree
140 	// -----------------------------------------------------------------------------------------------------------
141 
142 	private static final String REGION_TABLE = "juneau-rest/juneau-console-views/src/main/java/org/apache/juneau/console/views/RegionTable.java";
143 
144 	/** GREEN: every {@code <script>}-JSON sink in the real framework tree routes through escapeForScript. */
145 	@Test void a10_realTree_hasNoViolations() throws Exception {
146 		var root = requireRepoRoot();
147 		var r = ScriptJsonSinkScanner.scanTree(root);
148 		assertEquals(java.util.List.of(), r.violations(),
149 			() -> "framework <script> JSON sinks missing escapeForScript:\n  " + String.join("\n  ", r.violations()));
150 	}
151 
152 	/**
153 	 * Anti-vacuous: the scan must actually be finding the real, known-good sinks. If this ever drops to zero the
154 	 * "no violations" result above is meaningless - the scanner would have quietly stopped matching.
155 	 */
156 	@Test void a11_realTree_findsKnownSinks_notVacuous() throws Exception {
157 		var root = requireRepoRoot();
158 		var r = ScriptJsonSinkScanner.scanTree(root);
159 		assertTrue(r.sinks().size() >= 1, () -> "expected >=1 framework <script> sinks, found: " + r.sinks());
160 		assertTrue(r.sinks().stream().anyMatch(s -> s.file().replace('\\', '/').equals(REGION_TABLE)),
161 			() -> "RegionTable sink not found; scanner may have stopped matching. sinks: " + r.sinks());
162 	}
163 
164 	/**
165 	 * Mutation check: the real {@code RegionTable} passes as-is, but removing the escaper turns it into a violation.
166 	 * This proves the guard is genuinely exercising that sink rather than passing it by accident.
167 	 *
168 	 * <p>
169 	 * {@code RegionTable} currently has one escaped {@code <script>}-JSON sidecar. ViewTable's VIEW_META /
170 	 * BulkMutateDef sidecars retired with the MOVE Java delete. The blanket
171 	 * {@code escapeForScript}&rarr;{@code noEscape} mutation must flag that remaining sink.
172 	 */
173 	@Test void a12_realRegionTable_passesButFailsWhenEscaperRemoved() throws Exception {
174 		var root = requireRepoRoot();
175 		var source = Files.readString(root.resolve(REGION_TABLE));
176 
177 		var clean = ScriptJsonSinkScanner.scan(REGION_TABLE, source);
178 		assertSize(() -> "sinks: " + clean.sinks(), 1, clean.sinks());
179 		assertEquals(java.util.List.of(), clean.violations());
180 
181 		var mutated = ScriptJsonSinkScanner.scan(REGION_TABLE, source.replace("escapeForScript", "noEscape"));
182 		assertSize(() -> "sinks: " + mutated.sinks(), 1, mutated.sinks());
183 		assertSize(() -> "removing escapeForScript from RegionTable should be flagged: " + mutated.violations(), 1, mutated.violations());
184 	}
185 
186 	private static Path requireRepoRoot() {
187 		var root = ScriptJsonSinkScanner.locateRepoRoot();
188 		assertNotNull(root, "could not locate the Juneau repo root from the working directory - the security gate "
189 			+ "must fail loudly rather than pass vacuously when it cannot find the sources it guards");
190 		return root;
191 	}
