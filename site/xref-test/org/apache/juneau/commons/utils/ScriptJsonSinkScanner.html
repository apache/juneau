25  /**
26   * Test-only, <b>private</b> source scanner enforcing the {@code <script>}-JSON adoption of
27   * {@link StringUtils#escapeForScript(String)} across the Juneau framework's own code.
28   *
29   * <p>
30   * Not shipped and deliberately not public API: it guards the framework's own raw-text {@code <script>} sinks and is
31   * <b>not</b> a tool consumers are expected to run. It is the durable half of the {@code escapeForScript} adoption
32   * sweep &mdash; a one-time migration decays, so this fails the build if a framework source ever writes a serialized
33   * JSON payload into a raw-text {@code <script>} element without routing it through {@code escapeForScript} first.
34   *
35   * <h5 class='section'>What counts as a sink</h5>
36   * <p>
37   * The framework's single mechanism for inserting <i>verbatim</i> (non-entity-encoded) content into HTML output is the
38   * {@code juneau-bean-html5} raw-text primitive &mdash; {@link org.apache.juneau.bean.html5.HtmlBuilder#rawText(String)}
39   * and {@code new RawText(...)}. A {@code <script type="application/json">} sidecar's body is exactly such content.
40   * A <b>sink</b> is therefore a statement that builds a {@code script(...)} element <i>and</i> inserts a
41   * {@code rawText(...)} / {@code new RawText(...)} payload &mdash; which is precisely the shape
42   * {@code ViewTable} use. For each sink the payload argument must resolve to an
43   * {@code escapeForScript(...)} result (directly, or via a same-file {@code var = escapeForScript(...)} assignment).
44   *
45   * <h5 class='section'>Why it must also assert it still finds the known-good sites</h5>
46   * <p>
47   * A scanner that silently stops matching reads as a passing test, which is worse than no scanner. Comments are
48   * therefore stripped before analysis (so a {@code escapeForScript} mention in a javadoc cannot vacuously satisfy the
49   * guard), and the accompanying test asserts the scan still finds the real {@code ViewTable} sinks
50   * and that removing the escaper from one of them turns it into a violation &mdash; so "zero violations" can never mean
51   * "zero files examined".
52   *
53   * <h5 class='section'>Contract boundary</h5>
54   * <p>
55   * This enforces the JSON-in/JSON-out contract only; it does not widen {@code escapeForScript} to arbitrary raw-text
56   * payloads and does not attempt to detect hand-built {@code <script>} JSON assembled by raw string concatenation
57   * (the framework does not do that &mdash; it goes through the html5 builder). It scans the framework's Java sources on
58   * disk; it does not and cannot see consumer applications.
59   */
60  final class ScriptJsonSinkScanner {
61  
62  	private ScriptJsonSinkScanner() {}
63  
64  	/** One raw-text {@code <script>} payload site found in a source. */
65  	record Sink(String file, boolean escaped, String detail) {}
66  
67  	/** The outcome of a scan: every sink found, and the subset that are violations. */
68  	record Result(List<Sink> sinks, List<String> violations) {}
69  
70  	/** Matches a {@code rawText(} or {@code new RawText(} call (the html5 raw-text primitive). */
71  	private static final Pattern RAWTEXT_CALL = Pattern.compile("(?:new\\s+RawText|\\brawText)\\s*\\(");
72  
73  	/** An argument that is (directly, possibly fully-qualified) an {@code escapeForScript(...)} call. */
74  	private static final Pattern DIRECT_ESCAPE = Pattern.compile("(?s)^[A-Za-z0-9_.]*escapeForScript\\(.*");
75  
76  	/** A bare Java identifier (the variable-indirection case). */
77  	private static final Pattern IDENT = Pattern.compile("[A-Za-z_][A-Za-z0-9_]*");
78  
79  	/**
80  	 * Scans a single Java source string.
81  	 *
82  	 * @param file A label for the source (used in violation messages).
83  	 * @param source The Java source text.
84  	 * @return The sinks found and any violations.
85  	 */
86  	static Result scan(String file, String source) {
87  		var sinks = new ArrayList<Sink>();
88  		var violations = new ArrayList<String>();
89  		var code = stripCommentsAndStrings(source);
90  
91  		var m = RAWTEXT_CALL.matcher(code);
92  		while (m.find()) {
93  			var openParen = m.end() - 1; // the '(' of the rawText/RawText call
94  			var stmtStart = statementStart(code, m.start());
95  			var stmt = code.substring(stmtStart, m.start());
96  			if (! stmt.contains("script("))
97  				continue; // a raw-text write that is not into a <script> element - out of scope (JSON-in/JSON-out only)
98  
99  			var arg = balancedArg(code, openParen).trim();
100 			var escaped = isEscaped(arg, code);
101 			sinks.add(new Sink(file, escaped, "rawText(" + arg + ")"));
102 			if (! escaped)
103 				violations.add(file + ": <script> raw-text payload not routed through escapeForScript: rawText(" + arg + ")");
104 		}
105 		return new Result(sinks, violations);
106 	}
107 
108 	/** Convenience overload for ad-hoc source strings. */
109 	static Result scan(String source) {
110 		return scan("(source)", source);
111 	}
112 
113 	/**
114 	 * Walks a repository tree and scans every {@code src/main/java} Java source under it.
115 	 *
116 	 * @param repoRoot The repository root.
117 	 * @return The aggregated sinks and violations across the tree.
118 	 * @throws IOException If the tree cannot be walked.
119 	 */
120 	static Result scanTree(Path repoRoot) throws IOException {
121 		var sinks = new ArrayList<Sink>();
122 		var violations = new ArrayList<String>();
123 		var mainJava = File.separator + "src" + File.separator + "main" + File.separator + "java" + File.separator;
124 		var files = new ArrayList<Path>();
125 		Files.walkFileTree(repoRoot, new SimpleFileVisitor<Path>() {
126 			@Override
127 			public FileVisitResult preVisitDirectory(Path dir, BasicFileAttributes attrs) {
128 				// Never enter target/: forks create and delete temp files there while this scan runs.
129 				var name = dir.getFileName();
130 				return !dir.equals(repoRoot) && name != null && name.toString().equals("target") ? FileVisitResult.SKIP_SUBTREE : FileVisitResult.CONTINUE;
131 			}
132 
133 			@Override
134 			public FileVisitResult visitFile(Path f, BasicFileAttributes attrs) {
135 				if (f.toString().endsWith(".java") && f.toString().contains(mainJava))
136 					files.add(f);
137 				return FileVisitResult.CONTINUE;
138 			}
139 
140 			@Override
141 			public FileVisitResult visitFileFailed(Path f, IOException e) throws IOException {
142 				if (e instanceof NoSuchFileException)
143 					return FileVisitResult.CONTINUE;
144 				throw e;
145 			}
146 		});
147 		files.sort(null);
148 		for (var f : files) {
149 			var r = scan(repoRoot.relativize(f).toString(), Files.readString(f));
150 			sinks.addAll(r.sinks());
151 			violations.addAll(r.violations());
152 		}
153 		return new Result(sinks, violations);
154 	}
155 
156 	/**
157 	 * Locates the Juneau repository root by walking up from the working directory until a directory holding both
158 	 * {@code juneau-core} and {@code juneau-rest} is found.
159 	 *
160 	 * @return The repository root, or {@code null} if not found.
161 	 */
162 	static Path locateRepoRoot() {
163 		for (var d = Path.of("").toAbsolutePath(); d != null; d = d.getParent())
164 			if (Files.isDirectory(d.resolve("juneau-core")) && Files.isDirectory(d.resolve("juneau-rest")))
165 				return d;
166 		return null;
167 	}
168 
169 	/** True if {@code arg} is an escapeForScript result, directly or via a same-source {@code arg = escapeForScript(...)} assignment. */
170 	private static boolean isEscaped(String arg, String code) {
171 		if (DIRECT_ESCAPE.matcher(arg).matches())
172 			return true;
173 		if (IDENT.matcher(arg).matches()) {
174 			var p = Pattern.compile("\\b" + Pattern.quote(arg) + "\\s*=\\s*[A-Za-z0-9_.]*escapeForScript\\(");
175 			return p.matcher(code).find();
176 		}
177 		return false;
178 	}
179 
180 	/** Index just past the previous statement/block boundary ({@code ;}, <code>{</code>, or <code>}</code>) before {@code from}. */
181 	private static int statementStart(String s, int from) {
182 		for (var i = from - 1; i >= 0; i--) {
183 			var c = s.charAt(i);
184 			if (c == ';' || c == '{' || c == '}')
185 				return i + 1;
186 		}
187 		return 0;
188 	}
189 
190 	/** Returns the (comma-and-nesting-preserving) argument text between the balanced parens starting at {@code openParen}. */
191 	private static String balancedArg(String s, int openParen) {
192 		var depth = 0;
193 		var sb = new StringBuilder();
194 		for (var i = openParen; i < s.length(); i++) {
195 			var c = s.charAt(i);
196 			if (c == '(') {
197 				depth++;
198 				if (depth == 1)
199 					continue;
200 			} else if (c == ')') {
201 				depth--;
202 				if (depth == 0)
203 					break;
204 			}
205 			sb.append(c);
206 		}
207 		return sb.toString();
208 	}

210 	/**
211 	 * Blanks out comment and string/char-literal <i>content</i> (keeping quote delimiters and newlines) so pattern
212 	 * matching sees only the code skeleton. Critical for correctness: a {@code escapeForScript} or {@code <script}
213 	 * mention inside a javadoc or comment must not be mistaken for the code doing the right (or wrong) thing.
214 	 */
215 	private static String stripCommentsAndStrings(String s) {
216 		var out = new StringBuilder(s.length());
217 		var n = s.length();
218 		var i = 0;
219 		while (i < n) {
220 			var c = s.charAt(i);
221 			if (c == '/' && i + 1 < n && s.charAt(i + 1) == '/') {
222 				while (i < n && s.charAt(i) != '\n')
223 					i++;
224 				continue;
225 			}
226 			if (c == '/' && i + 1 < n && s.charAt(i + 1) == '*') {
227 				i += 2;
228 				while (i + 1 < n && ! (s.charAt(i) == '*' && s.charAt(i + 1) == '/')) {
229 					if (s.charAt(i) == '\n')
230 						out.append('\n');
231 					i++;
232 				}
233 				i = Math.min(n, i + 2);
234 				continue;
235 			}
236 			if (c == '"' && i + 2 < n && s.charAt(i + 1) == '"' && s.charAt(i + 2) == '"') {
237 				out.append("\"\"\"");
238 				i += 3;
239 				while (i + 2 < n && ! (s.charAt(i) == '"' && s.charAt(i + 1) == '"' && s.charAt(i + 2) == '"')) {
240 					if (s.charAt(i) == '\n')
241 						out.append('\n');
242 					i++;
243 				}
244 				if (i + 2 < n) {
245 					out.append("\"\"\"");
246 					i += 3;
247 				} else {
248 					i = n;
249 				}
250 				continue;
251 			}
252 			if (c == '"') {
253 				out.append('"');
254 				i++;
255 				while (i < n && s.charAt(i) != '"') {
256 					if (s.charAt(i) == '\\' && i + 1 < n) {
257 						i += 2;
258 						continue;
259 					}
260 					if (s.charAt(i) == '\n')
261 						out.append('\n');
262 					i++;
263 				}
264 				out.append('"');
265 				i++;
266 				continue;
267 			}
268 			if (c == '\'') {
269 				out.append('\'');
270 				i++;
271 				while (i < n && s.charAt(i) != '\'') {
272 					if (s.charAt(i) == '\\' && i + 1 < n) {
273 						i += 2;
274 						continue;
275 					}
276 					i++;
277 				}
278 				out.append('\'');
279 				i++;
280 				continue;
281 			}
282 			out.append(c);
283 			i++;
284 		}
285 		return out.toString();
286 	}
