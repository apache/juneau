35  	private static SearchExpression fn(String opName, String...args) {
36  		var op = OPS.get(opName);
37  		var argNodes = new ArrayList<SearchExpression>();
38  		for (var a : args)
39  			argNodes.add(SearchExpression.literal(a, false));
40  		return SearchExpression.func(op, argNodes);
41  	}

275 	/** One value per type that every applicable operator can be driven against without throwing. */
276 	private static Object matchingCell(SearchType type) {
277 		return switch (type) {
278 			case TEXT, ID -> "needle";
279 			case NUMERIC -> 5;
280 			case TIMESTAMP -> "2024-06-15T00:00:00Z";
281 			case VERSION -> "1.2.3";
282 			case ENUM -> "OPEN";
283 			case BOOLEAN -> true;
284 		};
285 	}
286 
287 	private static String matchingArg(SearchType type) {
288 		return switch (type) {
289 			case TEXT, ID -> "needle";
290 			case ENUM -> "OPEN";  // Matches matchingCell(ENUM) for consistency.
291 			case NUMERIC -> "5";
292 			case TIMESTAMP -> "2024-06-15T00:00:00Z";
293 			case VERSION -> "1.2.3";
294 			case BOOLEAN -> "true";
295 		};
296 	}
297 
298 	@Test
299 	void h01_everyApplicableOperatorTypeCombinationRunsWithoutThrowing() {
300 		for (var op : OPS.operators()) {
301 			if (op.isCombinator())
302 				continue;  // Combinators wrap sub-expressions; exercised structurally by sections a-e, not per-leaf.
303 			for (var type : SearchType.values()) {
304 				if (! op.appliesTo(type))
305 					continue;
306 				var cell = matchingCell(type);
307 				var arg = matchingArg(type);
308 				SearchExpression node;
309 				if (op.minArgs() == 0)
310 					node = fn(op.name());
311 				else if (op.name().equals("$between"))
312 					node = fn(op.name(), arg, arg);
313 				else
314 					node = fn(op.name(), arg);
315 				// Not asserting the boolean result here (each operator already has its own dedicated test in
316 				// sections a-e above) -- this matrix's job is purely to drive every applicable (operator, type)
317 				// pair through matches() at least once, covering the per-type branches in eq()/contains()/
318 				// wildcard()/prefix()/between()/cmp() that today's hand-picked tests do not all reach.
319 				assertDoesNotThrow(() -> InMemoryMatch.matches(node, cell, type), () -> op.name() + " on " + type);
320 			}
321 		}
322 	}

325 	// i - matchLeaf's default case must only mean "$between"; a custom=false operator that reaches in-memory
326 	// evaluation under a name matchLeaf doesn't recognize must fail clearly instead of silently being treated
327 	// as $between. This used to be reachable via copy()+rename of a real built-in, but SearchOperator.Builder#
328 	// build() now rejects renaming a built-in copy (design ruling, see SearchOperator_Test#f01), so the only way
329 	// left to construct a custom=false operator under an unrecognized name is the package-private
330 	// SearchOperator#builtin(...) factory directly (this test class shares its package). This covers both ways
331 	// the old unconditional fallthrough was wrong: an unrecognized operator with FEWER than 2 args crashed
332 	// (args.get(1) on a too-short list, i01); one with EXACTLY 2 args never crashed at all and would have
333 	// silently been evaluated as if it were a real $between call (i02) - the more dangerous of the two, since it
334 	// fails silently rather than loudly.
335 	//====================================================================================================
336 
337 	@Test
338 	void i01_unrecognizedBuiltinName_fewerArgsThanBetween_throwsClearException() {
339 		// Shaped like $gt (1 arg); matchLeaf's default must not fall into the "$between" branch and call
340 		// args.get(1) on a 1-element list.
341 		var unrecognized = SearchOperator.builtin("$notreal1", 1, 1, false, SearchType.NUMERIC);
342 		var node = SearchExpression.func(unrecognized, List.of(SearchExpression.literal("5", false)));
343 		var e = assertThrows(IllegalArgumentException.class, () -> InMemoryMatch.matches(node, 10, SearchType.NUMERIC));
344 		assertEquals("Unsupported operator '$notreal1' for in-memory evaluation.", e.getMessage());
345 	}
346 
347 	@Test
348 	void i02_unrecognizedBuiltinName_sameArityAsBetween_doesNotSilentlyMatchAsBetween() {
349 		// Shaped like $between (2 args), so the old default branch wouldn't have crashed at all -- it would have
350 		// silently evaluated this operator as if it were a real $between call. The fix's unconditional throw must
351 		// catch this case too, not just the lower-arity crash in i01.
352 		var unrecognized = SearchOperator.builtin("$notreal2", 2, 2, false, SearchType.NUMERIC);
353 		var node = SearchExpression.func(unrecognized, List.of(SearchExpression.literal("1", false), SearchExpression.literal("5", false)));
354 		var e = assertThrows(IllegalArgumentException.class, () -> InMemoryMatch.matches(node, 3, SearchType.NUMERIC));
355 		assertEquals("Unsupported operator '$notreal2' for in-memory evaluation.", e.getMessage());
356 	}

362 	/** Types a bare literal or a leaf operator's arguments the way QueryResolver does, to test InMemoryMatch in isolation. */
363 	private static SearchExpression typed(SearchExpression node, SearchType type) {
364 		if (node.isLiteral())
365 			return node.withTypedValue(type.parse(node.value()));
366 		var newArgs = new ArrayList<SearchExpression>();
367 		for (var a : node.args())
368 			newArgs.add(a.isLiteral() ? a.withTypedValue(type.parse(a.value())) : typed(a, type));
369 		return SearchExpression.func(node.operator(), newArgs);
370 	}
