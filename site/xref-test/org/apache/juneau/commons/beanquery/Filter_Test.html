17  package org.apache.juneau.commons.beanquery;
18  
19  import static org.apache.juneau.commons.TestAssertions.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.util.*;
23  
24  import org.apache.juneau.commons.TestBase;
25  import org.junit.jupiter.api.*;
26  
27  class Filter_Test extends TestBase {
28  
29  	private static final SearchOperatorSet OPS = SearchOperatorSet.standard();
30  
31  	private static SearchExpression expr(String raw) {
32  		return SearchExpressionParser.parse(raw, OPS);
33  	}
34  
35  	@Test
36  	void a01_leaf_fields() {
37  		var leaf = new Filter.Leaf("name", expr("$eq(\"bob\")"), SearchType.TEXT);
38  		assertBean(leaf, "column,type", "name,TEXT");
39  		assertTrue(leaf.expression().isFunction());
40  	}
41  
42  	@Test
43  	void a02_and_defensiveCopy() {
44  		var items = new ArrayList<Filter>();
45  		items.add(new Filter.Leaf("a", expr("$eq(\"1\")"), SearchType.TEXT));
46  		items.add(new Filter.Leaf("b", expr("$eq(\"2\")"), SearchType.TEXT));
47  		var and = new Filter.And(items);
48  		items.add(new Filter.Leaf("c", expr("$eq(\"3\")"), SearchType.TEXT));
49  		assertSize(2, and.items());
50  		var hoistedTarget1 = and.items();
51  		var hoistedArg1 = new Filter.Leaf("d", expr("$eq(\"4\")"), SearchType.TEXT);
52  		assertThrows(UnsupportedOperationException.class, () -> hoistedTarget1.add(hoistedArg1));
53  	}
54  
55  	@Test
56  	void a03_or_and_not_equality() {
57  		var leaf = new Filter.Leaf("a", expr("$blank()"), SearchType.TEXT);
58  		assertEquals(new Filter.Not(leaf), new Filter.Not(leaf));
59  		var or1 = new Filter.Or(List.of(leaf, leaf));
60  		var or2 = new Filter.Or(List.of(leaf, leaf));
61  		assertEquals(or1, or2);
62  	}
63  
64  	@Test
65  	void a04_emptyAnd_matchesEveryRow_byConvention() {
66  		// An empty top-level And is what an empty search resolves to (design §7); Filter itself carries no such
67  		// short-circuit — that behavior lives in the evaluator (InMemoryBeanQuerySession / SqlSearchCompiler).
68  		var and = new Filter.And(List.of());
69  		assertTrue(and.items().isEmpty());
70  	}
71  }
