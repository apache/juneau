45  				super(copyFrom);
46  			}

49  			public StubContext build() {
50  				return new StubContext(this);
51  			}
52  		}
53  
54  		StubContext(Builder b) {
55  			super(b);
56  		}
57  
58  		@Override /* BeanQueryContext */
59  		public Builder copy() {
60  			return new Builder(this);
61  		}
62  
63  		@Override /* BeanQueryContext */
64  		public StubSession.Builder createSession() {
65  			return new StubSession.Builder(this);
66  		}
67  	}

69  	static final class StubSession extends BeanQuerySession<Object> {
70  
71  		static final class Builder extends BeanQuerySession.Builder<Object,Builder> {
72  			Builder(StubContext context) {
73  				super(context);
74  			}
75  
76  			@Override /* BeanQuerySession.Builder */
77  			public StubSession build() {
78  				return new StubSession(this);
79  			}
80  		}
81  
82  		String hook;
83  		ResolvedQuery last;
84  
85  		StubSession(Builder b) {
86  			super(b);
87  		}
88  
89  		private void capture(String name, ResolvedQuery q) {
90  			hook = name;
91  			last = q;
92  		}
93  
94  		@Override protected Page<Object> doFind(ResolvedQuery q) { capture("doFind", q); return page(List.of(), q, () -> 10, () -> 3); }
95  		@Override protected Page<Map<String,Object>> doFindValues(ResolvedQuery q) { capture("doFindValues", q); return page(List.of(), q, () -> 10, () -> 3); }
96  		@Override protected Counts doCount(ResolvedQuery q) { capture("doCount", q); return Counts.of(10, 3); }
97  		@Override protected Stream<Object> doStream(ResolvedQuery q) { capture("doStream", q); return Stream.empty(); }
98  		@Override protected Stream<Map<String,Object>> doStreamValues(ResolvedQuery q) { capture("doStreamValues", q); return Stream.empty(); }
99  		@Override public void close() { /* Nothing to release. */ }
100 	}

102 	/** name TEXT, age NUMERIC. */
103 	private static StubContext.Builder ctx() {
104 		return new StubContext.Builder().column("name", TEXT).column("age", NUMERIC);
105 	}
106 
107 	private static StubSession session(StubContext.Builder b) {
108 		return b.build().createSession().build();
109 	}
110 
111 	private static ResolvedQuery find(StubContext.Builder b, BeanQuery q) {
112 		try (var s = session(b)) {
113 			s.find(q);
114 			return s.last;
115 		}
116 	}
117 
118 	private static void assertSyntax(String message, StubContext.Builder b, BeanQuery q) {
119 		try (var s = session(b)) {
120 			assertThrowsWithMessage(BeanQuerySyntaxException.class, message, () -> s.find(q));
121 		}
122 	}

187 		@Test void b01_finalMethodsDispatch() {
188 			try (var s = session(ctx())) {
189 				var q = new BeanQuery();
190 				s.find(q);
191 				assertEquals("doFind", s.hook);
192 				s.findValues(q);
193 				assertEquals("doFindValues", s.hook);
194 				assertEquals("Counts(total=10, matched=3)", s.count(q).toString());
195 				assertEquals("doCount", s.hook);
196 				s.stream(q);
197 				assertEquals("doStream", s.hook);
198 				s.streamValues(q);
199 				assertEquals("doStreamValues", s.hook);
200 			}
201 		}
202 
203 		@Test void b02_pageHelperFollowsPolicy() {
204 			try (var s = session(ctx())) {
205 				assertTrue(s.find(new BeanQuery()).matched().isEmpty());
206 				var m = s.find(new BeanQuery().setOpts("counts=matched"));
207 				assertTrue(m.matched().isPresent());
208 				assertTrue(m.total().isEmpty());
209 				assertEquals(3, m.matched().getAsLong());
210 				var b = s.find(new BeanQuery().setOpts("counts=both"));
211 				assertEquals(10, b.total().getAsLong());
212 				assertEquals(3, b.matched().getAsLong());
213 			}
214 		}
215 
216 		@Test void b03_sessionOverrides() {
217 			var c = ctx().build();
218 			try (var s = c.createSession().defaultLimit(5).maxLimit(5).countPolicy(CountPolicy.BOTH).build()) {
219 				s.find(new BeanQuery().setLimit(50));
220 				assertEquals(5, s.last.limit().intValue());
221 				assertEquals(CountPolicy.BOTH, s.last.counts());
222 			}
223 			try (var s = c.createSession().maxSearchLength(3).maxSearchClauses(1).maxSortKeys(1).maxExpressionDepth(1).regexTimeout(Duration.ofSeconds(1)).build()) {
224 				var hoistedArg1 = new BeanQuery().setSearch("name=x");
225 				assertThrowsWithMessage(BeanQuerySyntaxException.class, "Search string exceeds 3 characters.", () -> s.find(hoistedArg1));
226 			}
227 			var hoistedTarget1 = c.createSession().defaultLimit(2000);
228 			assertThrowsWithMessage(IllegalStateException.class, "maxLimit (1000) must not be less than defaultLimit (2000).", hoistedTarget1::build);
229 		}
230 
231 		@Test void b04_restrictColumnsNarrowsOnly() {
232 			var c = ctx().build();
233 			try (var s = c.createSession().restrictColumns("age", "name").restrictColumns("age").build()) {
234 				assertList(s.columns(), "age");
235 				var hoistedArg2 = new BeanQuery().setSearch("name=x");
236 				assertThrowsWithMessage(BeanQuerySyntaxException.class, "Unknown column 'name' in search.", () -> s.find(hoistedArg2));
237 			}
238 			var hoistedTarget2 = c.createSession();
239 			assertThrowsWithMessage(IllegalArgumentException.class, "Column 'x' is not a column of this context; a session can only narrow the columns.", () -> hoistedTarget2.restrictColumns("x"));
240 			var hoistedTarget3 = c.createSession().restrictColumns();
241 			assertThrowsWithMessage(IllegalStateException.class, "At least one column must be declared.", hoistedTarget3::build);
242 			assertList(c.columns(), "name", "age");  // The context is unchanged.
243 		}

256 		@Test void c01_copyRoundTrips() {
257 			var c = ctx().allowRegex(false).defaultLimit(7).columnOperators("age", SearchOperatorSet.standard().without("$eq")).build();
258 			var d = c.copy().build();
259 			assertList(d.columns(), "name", "age");
260 			assertFalse(d.operatorSet().contains("$regex"));  // allowRegex(false) survives the copy.
261 			assertFalse(d.operatorSet("age").contains("$eq"));
262 			assertEquals(7, d.resolve(new BeanQuery()).limit().intValue());
263 			var e = c.copy().defaultLimit(9).build();  // Changing the copy leaves the original alone.
264 			assertEquals(7, c.resolve(new BeanQuery()).limit().intValue());
265 			assertEquals(9, e.resolve(new BeanQuery()).limit().intValue());
266 		}

268 		@Test void c02_buildRejectsMissingOrBadSettings() {
269 			var hoistedTarget4 = new StubContext.Builder();
270 			assertThrowsWithMessage(IllegalStateException.class, "At least one column must be declared.", hoistedTarget4::build);
271 			// defaultLimit is explicit here so maxLimit(10) can't auto-narrow past it -- it's still a real conflict.
272 			var hoistedTarget5 = ctx().defaultLimit(50).maxLimit(10);
273 			assertThrowsWithMessage(IllegalStateException.class, "must not be less than defaultLimit", hoistedTarget5::build);
274 			var hoistedTarget6 = ctx();
275 			assertThrowsWithMessage(IllegalArgumentException.class, "defaultLimit must be positive: 0", () -> hoistedTarget6.defaultLimit(0));
276 			var hoistedTarget7 = ctx();
277 			assertThrowsWithMessage(IllegalArgumentException.class, "maxLimit must be positive or null: -1", () -> hoistedTarget7.maxLimit(-1));
278 			var hoistedTarget8 = ctx();
279 			assertThrowsWithMessage(IllegalArgumentException.class, "maxSortKeys must be positive: 0", () -> hoistedTarget8.maxSortKeys(0));
280 			var hoistedTarget9 = ctx();
281 			assertThrowsWithMessage(IllegalArgumentException.class, "regexTimeout must be positive", () -> hoistedTarget9.regexTimeout(Duration.ZERO));
282 			var hoistedTarget10 = ctx();
283 			assertThrows(IllegalArgumentException.class, () -> hoistedTarget10.column(null, TEXT));
284 			var hoistedTarget11 = ctx();
285 			assertThrows(IllegalArgumentException.class, () -> hoistedTarget11.column("x", null));
286 			var hoistedTarget12 = ctx();
287 			assertThrows(IllegalArgumentException.class, () -> hoistedTarget12.operators(null));
288 			var hoistedTarget13 = ctx();
289 			assertThrows(IllegalArgumentException.class, () -> hoistedTarget13.countPolicy(null));
290 			var hoistedTarget14 = ctx();
291 			var hoistedArg3 = SearchOperatorSet.standard();
292 			assertThrowsWithMessage(IllegalArgumentException.class, "Column 'x' is not declared.", () -> hoistedTarget14.columnOperators("x", hoistedArg3));
293 		}

295 		@Test void c03_columnsAndExclude() {
296 			assertList(ctx().columns("age", "name").build().columns(), "age", "name");
297 			assertList(ctx().exclude("name").build().columns(), "age");
298 			var hoistedTarget15 = ctx();
299 			assertThrowsWithMessage(IllegalArgumentException.class, "Column 'x' is not declared.", () -> hoistedTarget15.columns("x"));
300 			var hoistedTarget16 = ctx();
