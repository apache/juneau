17  package org.apache.juneau.commons.beanquery;
18  
19  import static org.apache.juneau.commons.TestAssertions.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.math.*;
23  import java.util.*;
24  
25  import org.apache.juneau.commons.*;
26  import org.junit.jupiter.api.*;
27  
28  class SearchExpression_Test extends TestBase {
29  
30  	private static final SearchOperatorSet OPS = SearchOperatorSet.standard();
31  
32  	private static SearchExpression parse(String s) {
33  		return SearchExpressionParser.parse(s, OPS);
34  	}
35  
36  	@Test void a01_literal_untyped_typedValueIsText() {
37  		assertEquals("Bill", parse("Bill").typedValue());
38  	}
39  
40  	@Test void a02_withTypedValue_returnsCopyNotMutation() {
41  		var e = parse("42");
42  		var typed = e.withTypedValue(new BigDecimal("42"));
43  		assertEquals("42", e.typedValue());
44  		assertEquals(new BigDecimal("42"), typed.typedValue());
45  		assertEquals("42", typed.value());
46  	}
47  
48  	@Test void a03_withTypedValue_preservesQuoted() {
49  		var e = parse("\"a*\"");
50  		assertTrue(e.isQuoted());
51  		assertTrue(e.withTypedValue("x").isQuoted());
52  	}
53  
54  	@Test void a04_withTypedValue_rejectsNull() {
55  		var e = parse("42");
56  		assertThrows(NullPointerException.class, () -> e.withTypedValue(null));
57  	}
58  
59  	@Test void a05_function_typedArgs_mapsEachArgsTypedValue() {
60  		var e = parse("$eq(42)");
61  		var typedArg = e.args().get(0).withTypedValue(new BigDecimal("42"));
62  		assertEquals(List.of(new BigDecimal("42")), SearchExpression.func(e.operator(), List.of(typedArg)).typedArgs());
63  	}
64  
65  	@Test void a06_function_typedArgs_untypedFallsBackToLiteralText() {
66  		assertEquals(List.of("a", "b", "c"), parse("$in(a,b,c)").typedArgs());
67  	}
68  
69  	@Test void a07_between_typedArgs_bothArguments() {
70  		var e = parse("$between(1,100)");
71  		var typed = SearchExpression.func(e.operator(), List.of(
72  			e.args().get(0).withTypedValue(new BigDecimal("1")),
73  			e.args().get(1).withTypedValue(new BigDecimal("100"))));
74  		assertEquals(List.of(new BigDecimal("1"), new BigDecimal("100")), typed.typedArgs());
75  	}
76  
77  	@Test void a08_withTypedValue_onFunction_throws() {
78  		var e = parse("$eq(42)");
79  		var hoistedArg1 = new BigDecimal("42");
80  		assertThrows(IllegalStateException.class, () -> e.withTypedValue(hoistedArg1));
81  	}
82  
83  	@Test void a09_typedArgs_nestedFunction_contributesName() {
84  		var e = parse("$and($eq(1),$gt(2))");
85  		assertList(e.typedArgs(), "$eq", "$gt");
86  		assertList(e.literalArgs(), "$eq", "$gt");
87  		assertEquals(e.argCount(), e.typedArgs().size());
88  	}
89  
90  	@Test void a10_argLists_areImmutable() {
91  		var e = parse("$in(a,b)");
92  		var hoistedTarget1 = e.literalArgs();
93  		assertThrows(UnsupportedOperationException.class, () -> hoistedTarget1.add("c"));
94  		var hoistedTarget2 = e.typedArgs();
95  		assertThrows(UnsupportedOperationException.class, () -> hoistedTarget2.add("c"));
96  		var hoistedTarget3 = e.args();
97  		assertThrows(UnsupportedOperationException.class, () -> hoistedTarget3.add(e));
98  	}
99  
100 	@Test
101 	@SuppressWarnings({
102 		"java:S3415" // Deliberate: the parsed expression must be the equals() receiver when compared to a String.
103 	})
104 	void a11_equalsHashCode_valueBased() {
105 		assertEquals(parse("$eq(42)"), parse("$eq(42)"));
106 		assertEquals(parse("$eq(42)").hashCode(), parse("$eq(42)").hashCode());
107 		assertNotEquals(parse("$eq(42)"), parse("$eq(43)"));
108 		assertNotEquals(parse("$eq(42)"), parse("$ne(42)"));
109 		assertNotEquals(parse("42"), "42");
110 	}
111 
112 	@Test void a12_equalsHashCode_quotedAndTyped() {
113 		assertEquals(parse("\"a\""), parse("\"a\""));
114 		assertNotEquals(parse("\"a\""), parse("a"));
115 		var typed = parse("42").withTypedValue(new BigDecimal("42"));
116 		assertEquals(typed, parse("42").withTypedValue(new BigDecimal("42")));
117 		assertEquals(typed.hashCode(), parse("42").withTypedValue(new BigDecimal("42")).hashCode());
118 		assertNotEquals(typed, parse("42"));
119 		assertNotEquals(typed, parse("42").withTypedValue(new BigDecimal("43")));
120 	}
121 }
