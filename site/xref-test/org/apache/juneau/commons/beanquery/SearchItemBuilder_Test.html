17  package org.apache.juneau.commons.beanquery;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.commons.TestBase;
24  import org.junit.jupiter.api.*;
25  
26  class SearchItemBuilder_Test extends TestBase {
27  
28  	@Test
29  	void a01_addLeaf_simple() {
30  		var b = new SearchItemBuilder();
31  		b.addLeaf("name", "$eq(\"bob\")");
32  		assertEquals(List.of(new SearchItem.Leaf("name", "$eq(\"bob\")")), b.items());
33  	}
34  
35  	@Test
36  	void a02_addLeaf_sameColumnTwice_wrapsInAnd() {
37  		var b = new SearchItemBuilder();
38  		b.addLeaf("age", "$gt(\"5\")");
39  		b.addLeaf("age", "$lt(\"10\")");
40  		assertEquals(List.of(new SearchItem.Leaf("age", "$and($gt(\"5\"),$lt(\"10\"))")), b.items());
41  	}
42  
43  	@Test
44  	void a03_addLeaf_sameColumnThreeTimes_appendsIntoSameAnd() {
45  		var b = new SearchItemBuilder();
46  		b.addLeaf("age", "$gt(\"5\")");
47  		b.addLeaf("age", "$lt(\"10\")");
48  		b.addLeaf("age", "$ne(\"7\")");
49  		assertEquals(List.of(new SearchItem.Leaf("age", "$and($gt(\"5\"),$lt(\"10\"),$ne(\"7\"))")), b.items());
50  	}
51  
52  	@Test
53  	void a04_addLeaf_differentColumns_bothKept_inOrder() {
54  		var b = new SearchItemBuilder();
55  		b.addLeaf("name", "$eq(\"bob\")");
56  		b.addLeaf("age", "$gt(\"5\")");
57  		assertEquals(List.of(new SearchItem.Leaf("name", "$eq(\"bob\")"), new SearchItem.Leaf("age", "$gt(\"5\")")), b.items());
58  	}
59  
60  	@Test
61  	void a05_addGroup_appendsDistinctSiblings_neverMerges() {
62  		var b = new SearchItemBuilder();
63  		var or1 = new SearchItem.Or(List.of(new SearchItem.Leaf("a", "$eq(\"1\")"), new SearchItem.Leaf("b", "$eq(\"2\")")));
64  		b.addGroup(or1);
65  		b.addGroup(or1);
66  		assertEquals(List.of(or1, or1), b.items());
67  	}
68  
69  	@Test
70  	void a06_addGroup_null_addsNothing() {
71  		var b = new SearchItemBuilder();
72  		b.addGroup(null);
73  		assertEquals(List.of(), b.items());
74  	}
75  
76  	@Test
77  	void a07_seed_roundTrips() {
78  		var parsed = SearchParser.parse("name=$eq(\"bob\"),$or(a=$eq(\"1\"),b=$eq(\"2\"))");
79  		var b = new SearchItemBuilder();
80  		b.seed(parsed);
81  		assertEquals(parsed, b.items());
82  	}
83  
84  	@Test
85  	void a08_seed_thenAddLeaf_mergesWithSeededColumn() {
86  		var b = new SearchItemBuilder();
87  		b.seed(SearchParser.parse("age=$gt(\"5\")"));
88  		b.addLeaf("age", "$lt(\"10\")");
89  		assertEquals(List.of(new SearchItem.Leaf("age", "$and($gt(\"5\"),$lt(\"10\"))")), b.items());
90  	}
91  
92  	@Test
93  	void a09_seed_thenAddLeafTwice_appendsIntoSameAnd() {
94  		var b = new SearchItemBuilder();
95  		b.seed(SearchParser.parse("age=$gt(\"5\")"));
96  		b.addLeaf("age", "$lt(\"10\")");
97  		b.addLeaf("age", "$ne(\"7\")");
98  		assertEquals(List.of(new SearchItem.Leaf("age", "$and($gt(\"5\"),$lt(\"10\"),$ne(\"7\"))")), b.items());
99  	}
100 }
