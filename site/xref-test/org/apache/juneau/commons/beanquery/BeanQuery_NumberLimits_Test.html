17  package org.apache.juneau.commons.beanquery;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.math.*;
22  import java.util.*;
23  
24  import org.apache.juneau.commons.*;
25  import org.junit.jupiter.api.*;
26  
27  /**
28   * Numeric comparison and formatting in bean queries treat decimals beyond the size limits of
29   * <c>StringUtils.parseBigDecimal(String)</c> (more than 1,000 characters, or a
30   * decimal exponent larger than 10,000 in magnitude) as non-numeric.
31   */
32  class BeanQuery_NumberLimits_Test extends TestBase {
33  
34  	private static final SearchOperatorSet OPS = SearchOperatorSet.standard();
35  
36  	private static final String LARGE_EXPONENT = "1e100000";
37  	private static final String MAX_EXPONENT = "1e10000";
38  	private static final String LONG_DIGITS = "1".repeat(1001);
39  	private static final String MAX_DIGITS = "1".repeat(1000);
40  
41  	private static SearchExpression typedFn(String opName, String...args) {
42  		var argNodes = new ArrayList<SearchExpression>();
43  		for (var a : args)
44  			argNodes.add(SearchExpression.literal(a, false).withTypedValue(SearchType.NUMERIC.parse(a)));
45  		return SearchExpression.func(OPS.get(opName), argNodes);
46  	}
47  
48  	//====================================================================================================
49  	// a - InMemoryMatch comparison
50  	//====================================================================================================
51  
52  	@Test
53  	void a01_gt_largeExponentCell_doesNotMatch() {
54  		var gt = typedFn("$gt", "5");
55  		assertFalse(InMemoryMatch.matches(gt, LARGE_EXPONENT, SearchType.NUMERIC));
56  		assertFalse(InMemoryMatch.matches(gt, "-" + LARGE_EXPONENT, SearchType.NUMERIC));
57  		assertFalse(InMemoryMatch.matches(typedFn("$lt", "5"), "1e-100000", SearchType.NUMERIC));
58  	}
59  
60  	@Test
61  	void a02_gt_tooLongCell_doesNotMatch() {
62  		assertFalse(InMemoryMatch.matches(typedFn("$gt", "5"), LONG_DIGITS, SearchType.NUMERIC));
63  	}
64  
65  	@Test
66  	void a03_between_overLimitCell_doesNotMatch() {
67  		var between = typedFn("$between", "1", "100");
68  		assertFalse(InMemoryMatch.matches(between, LARGE_EXPONENT, SearchType.NUMERIC));
69  		assertFalse(InMemoryMatch.matches(between, LONG_DIGITS, SearchType.NUMERIC));
70  	}
71  
72  	@Test
73  	void a04_gt_atLimitCell_stillCompares() {
74  		var gt = typedFn("$gt", "5");
75  		assertTrue(InMemoryMatch.matches(gt, MAX_EXPONENT, SearchType.NUMERIC));
76  		assertTrue(InMemoryMatch.matches(gt, MAX_DIGITS, SearchType.NUMERIC));
77  	}
78  
79  	@Test
80  	void a05_compareCells_overLimitSortsAfterNumbers() {
81  		assertTrue(InMemoryMatch.compareCells("9", LARGE_EXPONENT, SearchType.NUMERIC) < 0);
82  		assertTrue(InMemoryMatch.compareCells(LARGE_EXPONENT, "9", SearchType.NUMERIC) > 0);
83  		assertTrue(InMemoryMatch.compareCells("9", LONG_DIGITS, SearchType.NUMERIC) < 0);
84  		assertTrue(InMemoryMatch.compareCells(MAX_EXPONENT, "9", SearchType.NUMERIC) > 0);  // In range: compared by value.
85  	}
86  
87  	//====================================================================================================
88  	// b - ValueFormat formatting
89  	//====================================================================================================
90  
91  	@Test
92  	void b01_format_largeExponent_rejected() {
93  		var e = assertThrows(IllegalArgumentException.class, () -> ValueFormat.format(new BigDecimal(LARGE_EXPONENT)));
94  		assertTrue(e.getMessage().contains("Unsupported search value type"), e.getMessage());
95  		assertThrows(IllegalArgumentException.class, () -> ValueFormat.format(new BigDecimal("1e-100000")));
96  	}
97  
98  	@Test
99  	void b02_format_tooLong_rejected() {
100 		assertThrows(IllegalArgumentException.class, () -> ValueFormat.format(new BigDecimal(LONG_DIGITS)));
101 	}
102 
103 	@Test
104 	void b03_format_atLimit_rendersPlain() {
105 		assertEquals(MAX_DIGITS, ValueFormat.format(new BigDecimal(MAX_DIGITS)));
106 		assertEquals("1" + "0".repeat(20), ValueFormat.format(new BigDecimal("1e20")));
107 	}
108 }
