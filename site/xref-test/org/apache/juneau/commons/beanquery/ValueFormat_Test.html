73  	@Test
74  	void a07_instantFamily() {
75  		var instant = Instant.parse("2026-09-30T12:00:00Z");
76  		assertEquals("2026-09-30T12:00:00Z", ValueFormat.format(instant));
77  		assertEquals("2026-09-30T12:00:00Z", ValueFormat.format(instant.atOffset(java.time.ZoneOffset.UTC)));
78  		assertEquals("2026-09-30T12:00:00Z", ValueFormat.format(instant.atZone(java.time.ZoneOffset.UTC)));
79  		assertEquals("2026-09-30T12:00:00Z", ValueFormat.format(Date.from(instant)));
80  		var cal = GregorianCalendar.from(instant.atZone(java.time.ZoneOffset.UTC));
81  		assertEquals("2026-09-30T12:00:00Z", ValueFormat.format(cal));
82  	}

132 	/** A Number subtype that is none of the built-in boxed types, to exercise the "other Number" row. */
133 	static final class AtomicIntegerLike extends Number {
134 		private static final long serialVersionUID = 1L;
135 		private final int v;
136 		AtomicIntegerLike(int v) { this.v = v; }
137 		@Override public int intValue() { return v; }
138 		@Override public long longValue() { return v; }
139 		@Override public float floatValue() { return (float)v; }
140 		@Override public double doubleValue() { return v; }
141 		@Override public String toString() { return String.valueOf(v); }
142 	}
143 
144 	/** A Number subtype whose toString() is not parseable by {@code new BigDecimal(String)}, to exercise the catch branch. */
145 	static final class UnparseableNumber extends Number {
146 		private static final long serialVersionUID = 1L;
147 		@Override public int intValue() { return 0; }
148 		@Override public long longValue() { return 0; }
149 		@Override public float floatValue() { return 0f; }
150 		@Override public double doubleValue() { return 0; }
151 		@Override public String toString() { return "abc"; }
152 	}
