267 		private static SearchExpression expr(QuerySettings s, String search) {
268 			var f = resolve(s, new BeanQuery().setSearch(search)).filter();
269 			if (f instanceof Filter.And and && and.items().size() == 1)

377 		private static final Clock CLOCK = Clock.fixed(Instant.parse("2026-09-30T12:00:00Z"), ZoneOffset.UTC);
378 
379 		private static QuerySettings timestampSettings() {
380 			var s = settings();
381 			s.columns.put("createdAt", TIMESTAMP);
382 			s.columns.put("endedAt", TIMESTAMP);
383 			return s;

385 
386 		private static List<Filter.Leaf> leaves(String search, QuerySettings s, Clock clock) {
387 			var out = new ArrayList<Filter.Leaf>();
388 			collect(QueryResolver.resolve(new BeanQuery().setSearch(search), s, null, clock).filter(), out);
389 			return out;
390 		}
391 
392 		private static void collect(Filter f, List<Filter.Leaf> out) {
393 			if (f instanceof Filter.Leaf l)
394 				out.add(l);
395 			else if (f instanceof Filter.And and)
396 				and.items().forEach(i -> collect(i, out));
397 			else if (f instanceof Filter.Or or)
398 				or.items().forEach(i -> collect(i, out));
399 		}
