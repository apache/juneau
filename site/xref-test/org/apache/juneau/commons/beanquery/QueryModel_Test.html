38  	public static class Person {
39  		public String nickName;                                   // Public field: readable and writable.
40  		public final String code = "c";                           // Public final field: readable only.
41  		private String name;
42  		private int age;
43  		public String getName() { return name; }
44  		public void setName(String v) { name = v; }
45  		public int getAge() { return age; }                       // Getter only: readable, not writable.
46  		public void setSecretAge(int v) { age = v; }              // Setter only: writable, not readable.
47  		public boolean isActive() { return true; }
48  		public Color getColor() { return Color.RED; }
49  		@BeanProp("born") public Instant getBirthday() { return Instant.EPOCH; }
50  		@BeanIgnore public String getPassword() { return "x"; }
51  		public String getBoom() { throw new IllegalStateException("boom"); }
52  		@BeanProp(wo="true") public String getSecret() { return "s"; }
53  		public void setSecret(@SuppressWarnings("unused") String v) { /* stored nowhere */ }
54  		@BeanProp(ro="true") public String getLocked() { return "l"; }
55  		public void setLocked(@SuppressWarnings("unused") String v) { /* ignored */ }
56  		public void setExplode(@SuppressWarnings("unused") String v) { throw new IllegalStateException("bang"); }
57  		public String getExplode() { return "e"; }
58  	}

69  	private static List<String> describe(Map<String,QueryProperty> m) {
70  		return m.values().stream().map(p -> p.getName() + ":" + p.getType().getSimpleName() + ":" + p.getSearchType()).toList();
71  	}

77  	@Test
78  	void a01_readableProperties() {
79  		var m = QueryModel.of(Person.class);
80  		assertList(describe(m.getReadableProperties()),
81  			"active:boolean:BOOLEAN", "age:int:NUMERIC", "boom:String:TEXT", "born:Instant:TIMESTAMP",
82  			"code:String:TEXT", "color:Color:ENUM", "explode:String:TEXT", "locked:String:TEXT", "name:String:TEXT", "nickName:String:TEXT");
83  		assertNull(m.getReadableProperty("secret"));      // wo=true: not a column.
84  		assertNull(m.getReadableProperty("password"));    // @BeanIgnore.
85  		assertNull(m.getReadableProperty("secretAge"));   // Write-only: not a column.
86  		assertNull(m.getReadableProperty("class"));       // getClass() is declared on Object.
87  		assertSame(Person.class, m.getBeanClass());
88  	}
