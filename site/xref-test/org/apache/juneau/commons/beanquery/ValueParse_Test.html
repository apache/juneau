135 	@Test void b01_valueFormatRoundTrip() {
136 		var instantNanos = Instant.parse("2026-09-30T12:00:00.123456789Z");
137 		var utc = ZoneOffset.UTC;
138 		var cal = GregorianCalendar.from(Instant.parse("2026-09-30T12:00:00.250Z").atZone(utc));
139 		var date = Date.from(Instant.parse("2026-09-30T12:00:00.250Z"));
140 		var zoned = ZonedDateTime.parse("2026-09-30T12:00:00+02:00[Europe/Paris]");
141 		var offset = OffsetDateTime.parse("2026-09-30T12:00:00-04:00");
142 
143 		// Numbers normalise to BigDecimal (compareTo).
144 		for (Object n : List.<Object>of(42, 42L, BigInteger.valueOf(42), 0.5d, 0.5f, new BigDecimal("1E+3"), new BigDecimal("-3.50")))
145 			assertEquals(0, new BigDecimal(ValueFormat.format(n)).compareTo((BigDecimal)NUMERIC.parse(ValueFormat.format(n))), "number " + n);
146 		assertEquals(0, new BigDecimal("1000").compareTo((BigDecimal)NUMERIC.parse(ValueFormat.format(new BigDecimal("1E+3")))));
147 		assertEquals(0, new BigDecimal("0.5").compareTo((BigDecimal)NUMERIC.parse(ValueFormat.format(0.5f))));
148 
149 		assertEquals(Boolean.TRUE, BOOLEAN.parse(ValueFormat.format(true)));
150 		assertEquals(Boolean.FALSE, BOOLEAN.parse(ValueFormat.format(false)));
151 
152 		// Instant-like values are emitted as UTC instants, so each parses back as the same instant at UTC.
153 		assertEquals(instantNanos.atOffset(utc), TIMESTAMP.parse(ValueFormat.format(instantNanos)));
154 		assertEquals(date.toInstant().atOffset(utc), TIMESTAMP.parse(ValueFormat.format(date)));
155 		assertEquals(cal.toInstant().atOffset(utc), TIMESTAMP.parse(ValueFormat.format(cal)));
156 		assertEquals(zoned.toInstant().atOffset(utc), TIMESTAMP.parse(ValueFormat.format(zoned)));
157 		assertEquals(offset.toInstant().atOffset(utc), TIMESTAMP.parse(ValueFormat.format(offset)));
158 		assertEquals(offset.toInstant(), ((OffsetDateTime)TIMESTAMP.parse(ValueFormat.format(offset))).toInstant());
159 
160 		// Local forms round-trip unchanged.
161 		var ldt = LocalDateTime.parse("2026-09-30T12:00:00.5");
162 		var ld = LocalDate.parse("2026-09-30");
163 		assertEquals(ldt, TIMESTAMP.parse(ValueFormat.format(ldt)));
164 		assertEquals(ld, TIMESTAMP.parse(ValueFormat.format(ld)));
165 	}

287 		var rd = RelativeDuration.of(-24, RelativeDuration.Unit.H);
288 		assertEquals(new BigDecimal("-86400000"), NUMERIC.parse(ValueFormat.format(rd)));
289 		assertEquals(OffsetDateTime.parse("2026-09-29T12:00:00Z"), TIMESTAMP.parse(ValueFormat.format(rd), T0));

293 				assertEquals(T0.toEpochMilli() + ((BigDecimal)NUMERIC.parse(text)).longValue(), ((OffsetDateTime)TIMESTAMP.parse(text, T0)).toInstant().toEpochMilli(), text);
294 			}
