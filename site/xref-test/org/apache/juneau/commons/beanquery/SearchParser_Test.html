17  package org.apache.juneau.commons.beanquery;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.commons.TestBase;
24  import org.junit.jupiter.api.*;
25  
26  class SearchParser_Test extends TestBase {
27  
28  	@Test
29  	void a01_emptyOrBlank() {
30  		assertEquals(List.of(), SearchParser.parse(null));
31  		assertEquals(List.of(), SearchParser.parse(""));
32  		assertEquals(List.of(), SearchParser.parse("   "));
33  	}
34  
35  	@Test
36  	void a02_singleLeaf() {
37  		var items = SearchParser.parse("name=$eq(\"bob\")");
38  		assertEquals(List.of(new SearchItem.Leaf("name", "$eq(\"bob\")")), items);
39  	}
40  
41  	@Test
42  	void a03_multipleTopLevelLeaves() {
43  		var items = SearchParser.parse("name=$eq(\"bob\"),age=$gt(\"5\")");
44  		assertEquals(List.of(new SearchItem.Leaf("name", "$eq(\"bob\")"), new SearchItem.Leaf("age", "$gt(\"5\")")), items);
45  	}
46  
47  	@Test
48  	void a04_orGroup() {
49  		var items = SearchParser.parse("$or(name=$contains(\"x\"),email=$contains(\"x\"))");
50  		assertEquals(List.of(new SearchItem.Or(List.of(
51  			new SearchItem.Leaf("name", "$contains(\"x\")"),
52  			new SearchItem.Leaf("email", "$contains(\"x\")")))), items);
53  	}
54  
55  	@Test
56  	void a05_notGroup() {
57  		var items = SearchParser.parse("$not(name=$blank())");
58  		assertEquals(List.of(new SearchItem.Not(new SearchItem.Leaf("name", "$blank()"))), items);
59  	}
60  
61  	@Test
62  	void a06_nestedGroups() {
63  		var items = SearchParser.parse("$and(a=$eq(\"1\"),$or(b=$eq(\"2\"),c=$eq(\"3\")))");
64  		assertEquals(List.of(new SearchItem.And(List.of(
65  			new SearchItem.Leaf("a", "$eq(\"1\")"),
66  			new SearchItem.Or(List.of(new SearchItem.Leaf("b", "$eq(\"2\")"), new SearchItem.Leaf("c", "$eq(\"3\")")))))), items);
67  	}
68  
69  	@Test
70  	void a07_orArityError() {
71  		var e = assertThrows(BeanQuerySyntaxException.class, () -> SearchParser.parse("$or(a=$eq(\"1\"))"));
72  		assertTrue(e.getMessage().contains("$or takes at least two items"));
73  	}
74  
75  	@Test
76  	void a08_andArityError() {
77  		assertThrows(BeanQuerySyntaxException.class, () -> SearchParser.parse("$and(a=$eq(\"1\"))"));
78  	}
79  
80  	@Test
81  	void a09_notArityError() {
82  		var e = assertThrows(BeanQuerySyntaxException.class, () -> SearchParser.parse("$not(a=$eq(\"1\"),b=$eq(\"2\"))"));
83  		assertTrue(e.getMessage().contains("$not takes exactly one item"));
84  	}
85  
86  	@Test
87  	void a10_unbalancedGroup_unclosed() {
88  		assertThrows(BeanQuerySyntaxException.class, () -> SearchParser.parse("$or(a=$eq(\"1\"),b=$eq(\"2\")"));
89  	}
90  
91  	@Test
92  	void a11_unbalancedGroup_extraClose() {
93  		assertThrows(BeanQuerySyntaxException.class, () -> SearchParser.parse("$or(a=$eq(\"1\"),b=$eq(\"2\")))"));
94  	}
95  
96  	@Test
97  	void a12_duplicateTopLevelColumn_rejected() {
98  		var e = assertThrows(BeanQuerySyntaxException.class, () -> SearchParser.parse("name=$eq(\"1\"),name=$eq(\"2\")"));
99  		assertTrue(e.getMessage().contains("appears more than once"));
100 	}
101 
102 	@Test
103 	void a13_duplicateColumnInsideGroup_allowed() {
104 		assertDoesNotThrow(() -> SearchParser.parse("$or(name=$eq(\"1\"),name=$eq(\"2\"))"));
105 	}
106 
107 	@Test
108 	void a14_duplicateColumn_topPlusInsideGroup_allowed() {
109 		assertDoesNotThrow(() -> SearchParser.parse("name=$eq(\"1\"),$or(name=$eq(\"2\"),email=$eq(\"3\"))"));
110 	}
111 
112 	@Test
113 	void a15_blankLeaf_preserved() {
114 		var items = SearchParser.parse("name=,age=$gt(\"5\")");
115 		assertEquals(List.of(new SearchItem.Leaf("name", ""), new SearchItem.Leaf("age", "$gt(\"5\")")), items);
116 	}
117 
118 	@Test
119 	void a16_allBlankGroup_preserved() {
120 		var items = SearchParser.parse("$or(a=,b=),age=$gt(\"5\")");
121 		assertEquals(List.of(
122 			new SearchItem.Or(List.of(new SearchItem.Leaf("a", ""), new SearchItem.Leaf("b", ""))),
123 			new SearchItem.Leaf("age", "$gt(\"5\")")), items);
124 	}
125 
126 	@Test
127 	void a17_wholeSearchBlank_parsesAsLeaf() {
128 		assertEquals(List.of(new SearchItem.Leaf("name", "")), SearchParser.parse("name="));
129 	}
130 
131 	@Test
132 	void a18_whitespaceIgnored() {
133 		var items = SearchParser.parse(" $or( a=$eq(\"1\") , b=$eq(\"2\") ) ");
134 		assertEquals(List.of(new SearchItem.Or(List.of(new SearchItem.Leaf("a", "$eq(\"1\")"), new SearchItem.Leaf("b", "$eq(\"2\")")))), items);
135 	}
136 
137 	@Test
138 	void a19_commaInsideQuotedValue_notASplit() {
139 		var items = SearchParser.parse("name=$contains(\"a,b\")");
140 		assertEquals(List.of(new SearchItem.Leaf("name", "$contains(\"a,b\")")), items);
141 	}
142 
143 	@Test
144 	void a20_doubledQuoteInsideValue_notASplit() {
145 		var items = SearchParser.parse("name=$eq(\"a\"\"b\"),age=$gt(\"5\")");
146 		assertEquals(List.of(new SearchItem.Leaf("name", "$eq(\"a\"\"b\")"), new SearchItem.Leaf("age", "$gt(\"5\")")), items);
147 	}
148 
149 	@Test
150 	void a21_escapedLeadingDollarColumn() {
151 		var items = SearchParser.parse("\\$or=$eq(\"1\")");
152 		assertEquals(List.of(new SearchItem.Leaf("$or", "$eq(\"1\")")), items);
153 	}
154 
155 	@Test
156 	void a22_renderRoundTrip() {
157 		List<SearchItem> items = List.of(
158 			new SearchItem.Leaf("name", "$eq(\"bob\")"),
159 			new SearchItem.Or(List.of(new SearchItem.Leaf("a", "$eq(\"1\")"), new SearchItem.Leaf("b", "$eq(\"2\")"))),
160 			new SearchItem.Not(new SearchItem.Leaf("c", "$blank()")));
161 		String rendered = SearchParser.render(items);
162 		assertEquals("name=$eq(\"bob\"),$or(a=$eq(\"1\"),b=$eq(\"2\")),$not(c=$blank())", rendered);
163 		assertEquals(items, SearchParser.parse(rendered));
164 	}
165 
166 	@Test
167 	void a23_renderEscapesLeadingDollarColumn() {
168 		assertEquals("\\$or=$eq(\"1\")", SearchParser.render(List.of(new SearchItem.Leaf("$or", "$eq(\"1\")"))));
169 	}
