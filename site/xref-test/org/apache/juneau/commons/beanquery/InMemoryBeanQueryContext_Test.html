32  	public static class Rec {
33  		private final String name;
34  		private final int age;
35  		private final String version;
36  
37  		Rec(String name, int age, String version) {
38  			this.name = name;
39  			this.age = age;
40  			this.version = version;
41  		}
42  
43  		public String getName() { return name; }
44  		public int getAge() { return age; }
45  		public String getVersion() { return version; }
46  
47  		@Override public String toString() { return name; }
48  	}
49  
50  	private static final List<Rec> ROWS = List.of(
51  		new Rec("Alice", 30, "10.0.1"),
52  		new Rec("Bob", 25, "10.1.0"),
53  		new Rec("Carol", 40, "9.5.0"),
54  		new Rec("Dave", 17, "10.0.0")
55  	);
56  
57  	/** Bean-derived columns age (NUMERIC), name (TEXT), version (overridden to VERSION). */
58  	private static InMemoryBeanQueryContext.Builder<Rec> builder() {
59  		return InMemoryBeanQueryContext.create(Rec.class).column("version", SearchType.VERSION);
60  	}
61  
62  	private static InMemoryBeanQueryContext<Rec> context() {
63  		return builder().build();
64  	}
65  
66  	private static List<String> names(List<Rec> rows) {
67  		return rows.stream().map(Rec::getName).toList();
68  	}
