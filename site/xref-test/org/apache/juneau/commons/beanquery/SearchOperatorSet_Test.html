27  	@Test
28  	void a01_withoutRemovesOneOperator() {
29  		var std = SearchOperatorSet.standard();
30  		var s = std.without("$regex");
31  		assertFalse(s.contains("$regex"));
32  		assertTrue(s.contains("$eq"));
33  		assertEquals(std.operators().size() - 1, s.operators().size());
34  		assertTrue(std.contains("$regex"));  // The original is unchanged.
35  	}

38  	void a02_withoutUnknownOrNullReturnsSameSet() {
39  		var std = SearchOperatorSet.standard();

41  		assertSame(std, std.without(null));
42  	}

78  	@Test
79  	void c01_of_rejectsExactDuplicate() {
80  		var a = SearchOperator.create("$near", "help").predicate((cell, args) -> true).build();
81  		var b = SearchOperator.create("$near", "help2").predicate((cell, args) -> true).build();
82  		assertThrows(IllegalArgumentException.class, () -> SearchOperatorSet.of(a, b));
83  	}
84  
85  	@Test
86  	void c02_of_rejectsCaseOnlyDuplicate() {
87  		var a = SearchOperator.create("$near", "help").predicate((cell, args) -> true).build();
88  		var b = SearchOperator.create("$Near", "help2").predicate((cell, args) -> true).build();
89  		assertThrows(IllegalArgumentException.class, () -> SearchOperatorSet.of(a, b));
90  	}
91  
92  	@Test
93  	void c03_with_replacesIgnoringCase() {
94  		var a = SearchOperator.create("$near", "help").predicate((cell, args) -> true).build();
95  		var set = SearchOperatorSet.of(a);
96  		var b = SearchOperator.create("$Near", "help2").predicate((cell, args) -> true).build();
97  		var replaced = set.with(b);
98  		assertEquals(1, replaced.operators().size());
99  		assertEquals("$Near", replaced.get("$near").name());
100 	}

106 	@Test
107 	void d01_standard_hasEqic() {
108 		var std = SearchOperatorSet.standard();
109 		assertTrue(std.contains("$eqic"));
110 		var eqic = std.get("$eqic");
111 		assertBean(eqic, "minArgs,maxArgs", "1,1");
112 		assertTrue(eqic.appliesTo(SearchType.TEXT));
113 		assertTrue(eqic.appliesTo(SearchType.ID));
114 		assertTrue(eqic.appliesTo(SearchType.ENUM));
115 		assertFalse(eqic.appliesTo(SearchType.NUMERIC));
116 	}

122 	}
