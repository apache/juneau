17  package org.apache.juneau.commons.concurrent;
18  
19  import static org.apache.juneau.commons.concurrent.ReplayCache.FailMode.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.util.concurrent.atomic.*;
23  
24  import org.apache.juneau.commons.*;
25  import org.junit.jupiter.api.*;
26  
27  /**
28   * Coverage for the {@link ReplayCache} fail-mode policy applied by
29   * {@link ReplayCache#checkAndRecord(String, long, ReplayCache.FailMode, java.util.function.Consumer)}.
30   */
31  class ReplayCache_Test extends TestBase {
32  
33  	/** A cache whose backing "store" is unavailable: every {@code checkAndRecord} throws (after counting the call). */
34  	private static final class ThrowingReplayCache implements ReplayCache {
35  		final AtomicInteger calls = new AtomicInteger();
36  
37  		@Override public boolean checkAndRecord(String id, long expiresAtMs) {
38  			calls.incrementAndGet();
39  			throw new IllegalStateException("store unavailable");
40  		}
41  	}
42  
43  	@Test void a01_failOpen_allowsWhenStoreUnavailable() {
44  		// A thrown failure resolved with FAIL_OPEN is treated as first-seen (allow).
45  		var cache = new ThrowingReplayCache();
46  		assertTrue(cache.checkAndRecord("id-1", 0L, FAIL_OPEN));
47  		assertEquals(1, cache.calls.get());  // the cache WAS consulted -- not vacuous
48  	}
49  
50  	@Test void a02_failClosed_rejectsWhenStoreUnavailable() {
51  		// A thrown failure resolved with FAIL_CLOSED is treated as a replay (reject).
52  		var cache = new ThrowingReplayCache();
53  		assertFalse(cache.checkAndRecord("id-1", 0L, FAIL_CLOSED));
54  		assertEquals(1, cache.calls.get());
55  	}
56  
57  	@Test void a03_onFailureHandlerReceivesException() {
58  		var cache = new ThrowingReplayCache();
59  		var captured = new AtomicReference<Exception>();
60  		assertTrue(cache.checkAndRecord("id-1", 0L, FAIL_OPEN, captured::set));
61  		assertNotNull(captured.get());
62  		assertEquals("store unavailable", captured.get().getMessage());
63  	}
64  
65  	@Test void a04_onFailureNotInvokedWhenNoThrow() {
66  		var cache = new InMemoryReplayCache();
67  		var invoked = new AtomicInteger();
68  		assertTrue(cache.checkAndRecord("id-1", System.currentTimeMillis() + 60_000L, FAIL_CLOSED, e -> invoked.incrementAndGet()));
69  		assertEquals(0, invoked.get());
70  	}
71  
72  	@Test void a05_normalOutcomePassesThroughUnderEitherMode() {
73  		// When the delegate does not throw, the fail-mode is irrelevant: the real first-seen/replay outcome wins.
74  		var cache = new InMemoryReplayCache();
75  		var expiresAtMs = System.currentTimeMillis() + 60_000L;
76  		assertTrue(cache.checkAndRecord("id-1", expiresAtMs, FAIL_CLOSED));
77  		assertFalse(cache.checkAndRecord("id-1", expiresAtMs, FAIL_OPEN));
78  	}
79  
80  	@Test void a06_nullFailModeRejected() {
81  		var cache = new InMemoryReplayCache();
82  		assertThrows(IllegalArgumentException.class, () -> cache.checkAndRecord("id-1", 0L, null));
83  	}
84  }
