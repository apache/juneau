29  /**
30   * Coverage tests for the marshalling-only-path branches of {@link BeanPropertyMeta#add}, {@link BeanPropertyMeta#set}
31   * (and its private {@code setPropertyValue} helper), and {@code setArray} / {@code createDefaultCollectionForAbstractType}.
32   *
33   * <p>
34   * These methods require a non-null {@link BeanPropertyMeta.Builder#rawMetaType(BeanInfo) rawTypeMeta} and (for
35   * type-conversion) a {@link BeanMap#setBeanSession(BeanSession) BeanSession} — both normally supplied by the
36   * marshalling layer.  This test class fakes both seams ({@link BeanTestFakes.FakeBeanInfo},
37   * {@link BeanTestFakes.FakeBeanSession}) entirely within {@code juneau-commons}, driving
38   * {@link BeanPropertyMeta.Builder} directly (package-private constructor/fields — same package as production code)
39   * instead of going through the commons-only {@link BeanMeta#of(Class)} discovery path, which never populates
40   * {@code rawTypeMeta}.
41   */
42  @SuppressWarnings({
43  	"unused"  // Test POJO fields are read reflectively through BeanPropertyMeta, not directly.
44  })
45  class BeanPropertyMeta_Marshalling_Coverage_Test extends TestBase {

118 	/**
119 	 * Builds a {@link BeanMeta} whose {@link BeanMeta#getBeanInfo()} is non-null (a {@link FakeBeanInfo} wrapping
120 	 * {@code beanClass}), by routing through the marshalling-side {@link BeanMeta#create(BeanInfo, ClassInfo)}
121 	 * factory instead of the commons-only {@link BeanMeta#of(Class)}.  Needed because several error paths in
122 	 * {@link BeanPropertyMeta#add} / {@code setPropertyValue} read {@code beanMeta.getBeanInfo()} directly (for
123 	 * diagnostic messages), which is <jk>null</jk> — and thus NPEs — on a plain {@link BeanMeta#of(Class)} instance.
124 	 * Property discovery still runs in commons mode ({@link BeanMeta#create} passes a <jk>null</jk> marshalling
125 	 * context internally), so this does not affect any other behavior under test.
126 	 */
127 	private static <T> BeanMeta<T> marshallingBeanMeta(Class<T> beanClass) {
128 		return BeanMeta.create(new FakeBeanInfo<>(beanClass), null).beanMeta();
129 	}
130 
131 	/** Builds a field-only property (no getter/setter method — exercises the field-based invoke paths). */
132 	private static BeanPropertyMeta.Builder fieldProperty(Class<?> beanClass, String fieldName) throws Exception {
133 		var beanMeta = marshallingBeanMeta(beanClass);
134 		var f = info(beanClass.getField(fieldName));
135 		return BeanPropertyMeta.builder(beanMeta, fieldName).setField(f).canRead().canWrite();
136 	}

138 	@SuppressWarnings("unchecked")
139 	private static <T> BeanMap<T> mapOf(T bean) {
140 		var bm = BeanMap.of(bean, BeanMeta.of((Class<T>) bean.getClass()));
141 		bm.setBeanSession(new FakeBeanSession());
142 		return bm;
143 	}
