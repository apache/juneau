32  /**
33   * Coverage tests for the marshalling-only-path branches of {@link BeanPropertyMeta#add}, {@link BeanPropertyMeta#set}
34   * (and its private {@code setPropertyValue} helper), and {@code setArray} / {@code createDefaultCollectionForAbstractType}.
35   *
36   * <p>
37   * These methods require a non-null {@link BeanPropertyMeta.Builder#rawMetaType(BeanInfo) rawTypeMeta} and (for
38   * type-conversion) a {@link BeanMap#setBeanSession(BeanSession) BeanSession} — both normally supplied by the
39   * marshalling layer.  This test class fakes both seams ({@link BeanTestFakes.FakeBeanInfo},
40   * {@link BeanTestFakes.FakeBeanSession}) entirely within {@code juneau-commons}, driving
41   * {@link BeanPropertyMeta.Builder} directly (package-private constructor/fields — same package as production code)
42   * instead of going through the commons-only {@link BeanMeta#of(Class)} discovery path, which never populates
43   * {@code rawTypeMeta}.
44   */

198 	/**
199 	 * Builds a {@link BeanMeta} whose {@link BeanMeta#getBeanInfo()} is non-null (a {@link FakeBeanInfo} wrapping
200 	 * {@code beanClass}), by routing through the marshalling-side {@link BeanMeta#create(BeanInfo, ClassInfo)}
201 	 * factory instead of the commons-only {@link BeanMeta#of(Class)}.  Needed because several error paths in
202 	 * {@link BeanPropertyMeta#add} / {@code setPropertyValue} read {@code beanMeta.getBeanInfo()} directly (for
203 	 * diagnostic messages), which is <jk>null</jk> — and thus NPEs — on a plain {@link BeanMeta#of(Class)} instance.
204 	 * Property discovery still runs in commons mode ({@link BeanMeta#create} passes a <jk>null</jk> marshalling
205 	 * context internally), so this does not affect any other behavior under test.
206 	 */
207 	private static <T> BeanMeta<T> marshallingBeanMeta(Class<T> beanClass) {
208 		return BeanMeta.create(new FakeBeanInfo<>(beanClass), null).beanMeta();
209 	}
210 
211 	/** Overload allowing a custom {@link BeanConfigContext} (e.g. to flip ignore-* flags) to flow through {@link FakeBeanInfo#getBeanConfigContext()} into {@link BeanMeta#getConfig()}. */
212 	private static <T> BeanMeta<T> marshallingBeanMeta(Class<T> beanClass, BeanConfigContext config) {
213 		return BeanMeta.create(new FakeBeanInfo<>(beanClass, config), null).beanMeta();
214 	}
215 
216 	/** Builds a field-only property (no getter/setter method — exercises the field-based invoke paths). */
217 	private static BeanPropertyMeta.Builder fieldProperty(Class<?> beanClass, String fieldName) throws Exception {
218 		var beanMeta = marshallingBeanMeta(beanClass);
219 		var f = info(beanClass.getField(fieldName));
220 		return BeanPropertyMeta.builder(beanMeta, fieldName).setField(f).canRead().canWrite();
221 	}

223 	@SuppressWarnings({
224 		"unchecked" // bean.getClass() yields Class<? extends Object>, cast to Class<T> for the test helper.
225 	})
226 	private static <T> BeanMap<T> mapOf(T bean) {
227 		var bm = BeanMap.of(bean, BeanMeta.of((Class<T>) bean.getClass()));
228 		bm.setBeanSession(new FakeBeanSession());
229 		return bm;
230 	}
