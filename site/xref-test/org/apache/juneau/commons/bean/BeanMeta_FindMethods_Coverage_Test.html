52  	// resolves to property "foo" (lowercased by the PropertyNamer at line ~1436) - but this bare-@BeanProp
53  	// void-returning getter resolves to "Foo" (capital F, unlowercased). Root cause: the bpName-fallback
54  	// branch (~1391-1396) assigns the name using the pre-namer stripped name ("Foo"), and that raw value
55  	// then overrides the already-namered name again a few lines later, undoing the lowercasing that just
56  	// happened one line earlier for every other bpName-driven getter shape. Pinning current (buggy)
57  	// behavior here rather than silently fixing it.

60  		var bm = BeanMeta.of(BareBeanPropZeroParamBean.class);

62  		assertFalse(bm.getProperties().containsKey("foo"));
63  	}
