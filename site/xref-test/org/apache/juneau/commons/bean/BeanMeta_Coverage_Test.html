30   * <p>

33   * getGetterProps, getSetterProps, isUnsortedProperties, getBeanProxyInvocationHandler),

219 	void b01_fixedOrder_propertiesAvailable() {
220 		// @BeanType(properties = "y,x") is processed only on the marshalling-side path (where a BeanFilter
221 		// is built).  On the commons-only path, the annotation isn't consulted so the natural alphabetical
222 		// order is preserved — just verify both properties are present.
223 		var bm = BeanMeta.of(FixedOrder.class);
224 		var keys = new ArrayList<>(bm.getProperties().keySet());
