42  	private static StringFormat fs(String pattern) {
43  		return StringFormat.ofPrintf(pattern);
44  	}
45  
46  	private static String stringify(ThrowingSupplier<String> supplier) {
47  		try {
48  			return supplier.get();
49  		} catch (Throwable t) {
50  			return cns(t) + ": " + t.getLocalizedMessage();
51  		}
52  	}
53  
54  	private static void assertStringFormat(String pattern, Locale locale, Object... args) {
55  		var expected = stringify(()->String.format(locale, pattern, args));
56  		var actual = "";
57  		var fmt = (StringFormat)null;
58  		try {
59  			var fmt2 = fs(pattern);
60  			fmt = fmt2;
61  			actual = stringify(()->fmt2.format(locale, args));
62  		} catch (Throwable t) {
63  			actual = cns(t) + ": " + t.getLocalizedMessage();
64  		}
65  		if (!expected.equals(actual)) {
66  			System.out.println("Pattern: " + pattern);
67  			var toPattern = o(fmt).map(x -> x.toPattern()).orElse(null);
68  			System.out.println("toPattern(): " + toPattern);
69  			fail("Pattern: " + pattern + ", toPattern(): " + toPattern + ", expected: <" + expected + "> but was: <" + actual + ">");
70  		}
71  	}
72

189 		// { and ' are literal text on the printf path (java.util.Formatter semantics).
190 		assertEquals("Hello {0} world", fs("Hello {0} world").format("ignored-extra-arg"));
191 		assertEquals("a '{0}' b", fs("a '{0}' b").format("ignored"));
192 
193 		// BigDecimal with %d - String.format throws exception, but our optimized code handles it via longValue().
194 		assertEquals("Number: 42", fs("Number: %d").format(new BigDecimal("42")));
195 
196 		// Lenient null handling for simple string conversions (renders "null" rather than throwing).
197 		assertEquals("Value: null", fs("Value: %s").format((Object)null));

250 	@Test void a09_toPattern() {
251 		// Literal tokens ({ and ' are literal on the printf path)
252 		assertEquals("[L:Hello ]", fs("Hello ").toPattern());
253 		assertEquals("[L:a '{0}' b]", fs("a '{0}' b").toPattern());
254 		assertEquals("[L:Hello {0}]", fs("Hello {0}").toPattern());
255 
256 		// StringFormat (printf) tokens
257 		assertEquals("[L:Hello ][S:s0:%s]", fs("Hello %s").toPattern());  // Simple format: 's'
258 		assertEquals("[L:Number: ][S:d0:%d]", fs("Number: %d").toPattern());  // Simple format: 'd'
259 		assertEquals("[L:Hex: ][S:x0:%x]", fs("Hex: %x").toPattern());  // Simple format: 'x'
260 		assertEquals("[L:Float: ][S:z0:%.2f]", fs("Float: %.2f").toPattern());  // Complex format: 'z' (other)
261 		assertEquals("[L:ID: ][S:z0:%05d]", fs("ID: %05d").toPattern());  // Complex format: 'z' (other)
262 		assertEquals("[L:Hello ][S:s0:%s][L: ][S:s1:%s]", fs("Hello %s %s").toPattern());
263 
264 		// Time conversions (2-character) - 't' or 'T' handling
265 		assertEquals("[L:Month: ][S:z0:%tm]", fs("Month: %tm").toPattern());  // %tm is 2-character time conversion
266 		assertEquals("[L:Year: ][S:z0:%tY]", fs("Year: %tY").toPattern());  // %tY is 2-character time conversion
267 		assertEquals("[L:Date: ][S:z0:%TD]", fs("Date: %TD").toPattern());  // %TD is 2-character time conversion
268 
269 		// %n doesn't consume an argument - it's handled as a LiteralToken
270 		var lineSep = System.lineSeparator();
271 		assertEquals("[L:Line 1][L:" + lineSep + "][L:Line 2]", fs("Line 1%nLine 2").toPattern());
272 		assertEquals("[S:s0:%s][L: ][L:" + lineSep + "][L: ][S:s1:%s]", fs("%s %n %s").toPattern());  // %n is a literal, so second %s gets index 1
273 	}
