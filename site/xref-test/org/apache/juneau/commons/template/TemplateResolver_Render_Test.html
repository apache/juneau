17  package org.apache.juneau.commons.template;

19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.commons.*;

27  /** Tests for {@link TemplateResolver}. */
28  class TemplateResolver_Render_Test extends TestBase {
29  
30  	private static String r(TemplateResolver tr, String t, Map<String,?> scope) {
31  		return tr.render(tr.parse(t), scope);
32  	}
33  
34  	private static Map<String,Object> row(String id, String msg) {
35  		var m = new LinkedHashMap<String,Object>();
36  		m.put("id", id);
37  		m.put("msg", msg);
38  		return m;
39  	}
40  
41  	@Test void a01_scopeVarReadsScope() {
42  		assertEquals("Hi Bob", r(TemplateResolver.DEFAULT, "Hi ${name}", Map.of("name", "Bob")));
43  	}
44  
45  	@Test void a02_absentNameIsEmpty() {
46  		assertEquals("Hi ", r(TemplateResolver.DEFAULT, "Hi ${name}", Map.of()));
47  	}
48  
49  	@Test void a03_noSystemVarsByDefault() {
50  		// "S" is not a registered var, so this is the scope key "S" with default "user.home"; no system property is read.
51  		assertEquals("user.home", r(TemplateResolver.DEFAULT, "${S:user.home}", Map.of()));
52  		assertNotEquals(System.getProperty("user.home"), r(TemplateResolver.DEFAULT, "${S:user.home}", Map.of()));
53  	}
54  
55  	@Test void a04_pCanBeRebound() {
56  		var tr = TemplateResolver.create().vars(new DefaultingVar("P") {
57  			@Override public String resolve(VarResolverSession s, String key) { return "rebound-" + key; }
58  		}).build();
59  		assertEquals("rebound-x", r(tr, "${x}", Map.of("x", "scope")));
60  	}
61  
62  	@Test void a05_projectionReadsRowFields() {
63  		var rows = List.of(row("1", "a"), row("2", "b"));
64  		assertEquals("1:a,2:b", r(TemplateResolver.DEFAULT, "#{join(',', rows.![id + ':' + msg])}", Map.of("rows", rows)));
65  	}
66  
67  	@Test void a06_presentNullFieldDoesNotFallBackToRoot() {
68  		var m = row("1", null);
69  		assertEquals("[x]", r(TemplateResolver.DEFAULT, "#{rows.![coalesce(msg, 'x')]}".replace("rows.![coalesce(msg, 'x')]", "'['+rows.![msg ?: 'x'][0]+']'"), Map.of("rows", List.of(m), "msg", "ROOT")));
70  	}
71  
72  	@Test void a07_denyListedFunctionLenientFallsThrough() {
73  		assertEquals("#{exec('x')}", r(TemplateResolver.DEFAULT, "#{exec('x')}", Map.of()));
74  	}
75  
76  	@Test void a08_strictUnknownFunctionThrows() {
77  		var tr = TemplateResolver.create().strict(true).build();
78  		assertThrows(ExprException.class, () -> r(tr, "#{exec('x')}", Map.of()));
79  	}
80  
81  	@Test void a09_injectionInDataNotReparsed() {
82  		assertEquals("${S:user.home} #{1+1}", r(TemplateResolver.DEFAULT, "${a}", Map.of("a", "${S:user.home} #{1+1}")));
83  	}
84  
85  	@Test void a10_classAccessBlocked() {
86  		assertThrows(ExprException.class, () -> r(TemplateResolver.DEFAULT, "#{a.class}", Map.of("a", "x")));
87  	}
88  
89  	@Test void a11_maxCollectionSizeLenientTruncates() {
90  		var tr = TemplateResolver.create().maxCollectionSize(2).build();
91  		var out = r(tr, "#{list(1,2,3,4)}", Map.of());
92  		assertFalse(out.contains("4"), out);
93  	}
94  
95  	@Test void a12_maxCollectionSizeStrictThrows() {
96  		var tr = TemplateResolver.create().maxCollectionSize(2).strict(true).build();
97  		assertThrows(RuntimeException.class, () -> r(tr, "#{list(1,2,3,4)}", Map.of()));
98  	}

100 	@Test void a13_explicitEncoderOverridesDestination() {
101 		var tr = TemplateResolver.create().destination(Destinations.HTML).build();
102 		var t = tr.parse("${a}");
103 		assertEquals("<b>", tr.render(t, Map.of("a", "<b>"), s -> s));
104 		assertEquals("&lt;b&gt;", tr.render(t, Map.of("a", "<b>")));
105 	}
106 
107 	@Test void a14_nullScopeAndTemplateLiteral() {
108 		assertEquals("plain", TemplateResolver.DEFAULT.render(TemplateResolver.DEFAULT.parse("plain"), null));
109 	}
110 
111 	@Test void a15_foreignParsedTemplateRejected() {
112 		var other = TemplateResolver.create().build();
113 		var parsed = other.parse("x");
114 		assertThrows(IllegalArgumentException.class, () -> TemplateResolver.DEFAULT.render(parsed, Map.of()));
115 	}
116 
117 	@Test void a16_allowFunctionsIsAdditive() {
118 		var tr = TemplateResolver.create().allowFunctions("repeat").build();
119 		assertEquals("ababX", r(tr, "#{repeat('ab', 2)}#{upper('x')}", Map.of()));
120 	}
121 
122 	@Test void a17_allowOnlyFunctionsReplaces() {
123 		var tr = TemplateResolver.create().allowOnlyFunctions("upper", "$op.*").build();
124 		assertEquals("X #{join(',', list('a'))}", r(tr, "#{upper('x')} #{join(',', list('a'))}", Map.of()));
125 	}

127 	@Test void a18_wildcardRejected() {
128 		var b = TemplateResolver.create();
129 		assertThrows(IllegalArgumentException.class, () -> b.allowFunctions("*"));
130 		assertThrows(IllegalArgumentException.class, () -> b.allowOnlyFunctions("*"));
131 	}

133 	@Test void a19_mapRowSizeShadowsScope() {
134 		// Pins current behavior: a map row's synthetic size does not fall back to a scope entry named size.
135 		var rows = List.of(row("1", "a"));
136 		assertEquals("2", r(TemplateResolver.DEFAULT, "#{join(',', rows.![size])}", Map.of("rows", rows, "size", "OUTER")));
137 	}

139 	@Test void a20_noDestinationDoesNotEncode() {
140 		assertEquals("<b>", r(TemplateResolver.DEFAULT, "${a}", Map.of("a", "<b>")));
141 	}
142 
143 	@Test void a21_ulLenientTruncates() {
144 		var tr = TemplateResolver.create().maxCollectionSize(2).destination(Destinations.PLAIN_TEXT).build();
145 		assertEquals("- 1\n- 2", r(tr, "#{ul(xs)}", Map.of("xs", List.of(1, 2, 3, 4))));
146 	}
147 
148 	@Test void a22_ulStrictThrows() {
149 		var tr = TemplateResolver.create().maxCollectionSize(2).strict(true).build();
150 		assertThrows(RuntimeException.class, () -> r(tr, "#{ul(xs)}", Map.of("xs", List.of(1, 2, 3))));
151 	}
152 
153 	@Test void a23_olTableTakeJoinCapped() {
154 		var tr = TemplateResolver.create().maxCollectionSize(2).destination(Destinations.PLAIN_TEXT).build();
155 		var xs = Map.<String,Object>of("xs", List.of(1, 2, 3, 4));
156 		assertEquals("1. 1\n2. 2".replace("\\n", "\n"), r(tr, "#{ol(xs)}", xs));
157 		assertEquals("1,2", r(tr, "#{join(',', xs)}", xs));
158 		assertEquals("[1,2]", r(TemplateResolver.create().maxCollectionSize(2).build(), "#{take(xs, 3)}", xs));
159 		var rows = Map.<String,Object>of("rows", List.of(List.of("a"), List.of("b"), List.of("c")));
160 		assertEquals(2, r(tr, "#{table(list('H'), rows)}", rows).split("\n").length - 2);
161 	}
162 
163 	@Test void a24_joinAndTakeStrictThrow() {
164 		var tr = TemplateResolver.create().maxCollectionSize(2).strict(true).build();
165 		var xs = Map.<String,Object>of("xs", List.of(1, 2, 3));
166 		assertThrows(RuntimeException.class, () -> r(tr, "#{join(',', xs)}", xs));
167 		assertThrows(RuntimeException.class, () -> r(tr, "#{take(xs, 1)}", xs));
168 	}
