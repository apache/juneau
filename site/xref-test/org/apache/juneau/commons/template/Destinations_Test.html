29  	private static String r(Destination d, String t, Map<String,?> scope) {
30  		var tr = TemplateResolver.create().destination(d).build();
31  		return tr.render(tr.parse(t), scope);
32  	}

34  	@Test void a01_htmlEscapesValuesNotTemplate() {
35  		assertEquals("<p>&lt;b&gt; &amp; &quot;q&quot;</p>", r(Destinations.HTML, "<p>${a}</p>", Map.of("a", "<b> & \"q\"")));
36  	}

38  	@Test void a02_htmlList() {
39  		assertEquals("<ul><li>&lt;a&gt;</li><li>b</li></ul>", r(Destinations.HTML, "#{ul(xs)}", Map.of("xs", List.of("<a>", "b"))));
40  	}

43  		var rows = List.of(Map.of("id", "<1>", "n", "x"));
44  		assertEquals("<table><thead><tr><th>ID</th><th>N</th></tr></thead><tbody><tr><td>&lt;1&gt;</td><td>x</td></tr></tbody></table>",

52  	@Test void a05_markdownLeadingBlockMarkers() {
53  		assertEquals("\\# h", r(Destinations.MARKDOWN, "${a}", Map.of("a", "# h")));
54  		assertEquals("1\\. x", r(Destinations.MARKDOWN, "${a}", Map.of("a", "1. x")));
55  		assertEquals("1\\) x", r(Destinations.MARKDOWN, "${a}", Map.of("a", "1) x")));
56  		assertEquals("\\- x", r(Destinations.MARKDOWN, "${a}", Map.of("a", "- x")));
57  	}

85  	@Test void a12_markdownCorpusNeverStartsBlockOrInjectsLinks() {
86  		var corpus = List.of("# h", "> q", "+ x", "= x", "1) x", "    code", "```", "<script>", "[a](http://x)", "https://evil.example/x", "{.raw}", "!img", "{javascript}", "a\n\n# b", "***", "~~x~~", "&amp;");
87  		for (var s : corpus) {
88  			var out = r(Destinations.MARKDOWN, "${a}", Map.of("a", s));
89  			assertFalse(out.contains("\n"), out);
90  			assertFalse(out.matches("^\\s*[#>+=].*"), out);
91  			assertFalse(out.matches(".*(?<!\\\\)[\\[\\]<>`*~{}!].*"), out);
92  			assertFalse(out.matches("(?s).*(?<!\\\\)://.*"), out);
93  		}
94  	}

96  	private static final List<Destination> ALL = List.of(Destinations.HTML, Destinations.MARKDOWN, Destinations.PLAIN_TEXT, Destinations.JSON_STRING, Destinations.URL_COMPONENT);

98  	@Test void b01_dataIsNeverReparsedOnAnyPreset() {
99  		for (var d : ALL) {
100 			assertFalse(r(d, "${a}", Map.of("a", "${S:user.home}")).contains(System.getProperty("user.home")), d.name());
101 			assertFalse(r(d, "${a}", Map.of("a", "#{1+1}")).contains("2") && ! r(d, "${a}", Map.of("a", "#{1+1}")).contains("1"), d.name());
102 		}
103 		assertEquals("#{1+1}", r(Destinations.PLAIN_TEXT, "${a}", Map.of("a", "#{1+1}")));
104 		assertEquals("${x}", r(Destinations.PLAIN_TEXT, "${a}", Map.of("a", "${x}")));
105 	}

145 	@Test void b10_deniedFunctionsDoNotRunOnAnyPreset() {
146 		for (var d : ALL) {
147 			var tr = TemplateResolver.create().destination(d).build();
148 			for (var f : List.of("jsonPath('x','y')", "get('x','y')", "match('a','a')", "replaceRegex('a','a','b')", "htmlUnescape('&lt;')", "repeat('a', 3)")) {
149 				var t = "#{" + f + "}";
150 				var out = tr.render(tr.parse(t), Map.of());
151 				assertFalse(out.contains("<") || out.equals("b") || out.equals("aaa"), d.name() + " " + f + " -> " + out);
152 			}
153 		}
154 	}
