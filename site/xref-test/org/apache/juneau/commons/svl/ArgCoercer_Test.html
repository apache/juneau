17  package org.apache.juneau.commons.svl;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.commons.*;
24  import org.junit.jupiter.api.*;
25  
26  /**
27   * Tests the {@link ArgCoercer} argument-coercion table for {@code #{...}} function dispatch.
28   *
29   * <p>
30   * Drives {@link ArgCoercer} indirectly through {@link TypedFunction} subclasses so the integration
31   * path (function dispatch → coercion → typed invoke) is exercised end-to-end.
32   */
33  @SuppressWarnings({
34  	"java:S5778"  // assertThrows lambdas with chained calls; intermediate invocations do not throw in practice
35  })
36  class ArgCoercer_Test extends TestBase {
37  
38  	public static class IntFn extends TypedFunction {
39  		@Override public String name() { return "intfn"; }
40  		public String invoke(int x) { return "int=" + x; }
41  	}
42  
43  	public static class LongFn extends TypedFunction {
44  		@Override public String name() { return "longfn"; }
45  		public String invoke(long x) { return "long=" + x; }
46  	}
47  
48  	public static class DoubleFn extends TypedFunction {
49  		@Override public String name() { return "dblfn"; }
50  		public String invoke(double x) { return "dbl=" + x; }
51  	}
52  
53  	public static class BoolFn extends TypedFunction {
54  		@Override public String name() { return "boolfn"; }
55  		public String invoke(boolean x) { return "bool=" + x; }
56  	}
57  
58  	public static class ArrFn extends TypedFunction {
59  		@Override public String name() { return "arrfn"; }
60  		public String invoke(String[] xs) { return "arr=" + Arrays.toString(xs); }
61  	}
62  
63  	/**
64  	 * Function with a non-final {@code String[]} param so it triggers JSON-array shortcut
65  	 * coercion (rather than variadic gather).
66  	 */
67  	public static class JsonArrFn extends TypedFunction {
68  		@Override public String name() { return "jsonArr"; }
69  		public String invoke(String[] xs, String tag) { return tag + "=" + Arrays.toString(xs); }
70  	}
71  
72  	public static class ObjFn extends TypedFunction {
73  		@Override public String name() { return "objfn"; }
74  		public String invoke(Object o) { return "obj=" + o; }
75  	}
76  
77  	private VarResolver vr() {
78  		return VarResolver.create()
79  			.functions(new IntFn(), new LongFn(), new DoubleFn(), new BoolFn(),
80  				new ArrFn(), new JsonArrFn(), new ObjFn())
81  			.build();
82  	}
83  
84  	@Test void a01_intCoercion() {
85  		assertEquals("int=42", vr().resolve("#{intfn(42)}"));
86  		assertEquals("int=-7", vr().resolve("#{intfn(-7)}"));
87  	}
88  
89  	@Test void a02_intCoercionFailsOnNonNumeric() {
90  		var ex = assertThrows(IllegalArgumentException.class, () -> vr().resolve("#{intfn(notanumber)}"));
91  		assertTrue(ex.getMessage().contains("cannot coerce"), ex.getMessage());
92  	}
93  
94  	@Test void a03_longCoercion() {
95  		assertEquals("long=999999999999", vr().resolve("#{longfn(999999999999)}"));
96  	}
97  
98  	@Test void a04_doubleCoercion() {
99  		assertEquals("dbl=3.14", vr().resolve("#{dblfn(3.14)}"));
100 	}
101 
102 	@Test void a05_booleanCoercionTrueTokens() {
103 		assertEquals("bool=true", vr().resolve("#{boolfn(true)}"));
104 		assertEquals("bool=true", vr().resolve("#{boolfn(TRUE)}"));
105 		assertEquals("bool=true", vr().resolve("#{boolfn(1)}"));
106 		assertEquals("bool=true", vr().resolve("#{boolfn(yes)}"));
107 		assertEquals("bool=true", vr().resolve("#{boolfn(YES)}"));
108 		assertEquals("bool=true", vr().resolve("#{boolfn(on)}"));
109 		assertEquals("bool=true", vr().resolve("#{boolfn(ON)}"));
110 	}
111 
112 	@Test void a06_booleanCoercionFalseTokens() {
113 		assertEquals("bool=false", vr().resolve("#{boolfn(false)}"));
114 		assertEquals("bool=false", vr().resolve("#{boolfn(0)}"));
115 		assertEquals("bool=false", vr().resolve("#{boolfn(no)}"));
116 		assertEquals("bool=false", vr().resolve("#{boolfn(off)}"));
117 		// Empty / quoted-empty coerces to false.
118 		assertEquals("bool=false", vr().resolve("#{boolfn(\"\")}"));
119 	}
120 
121 	@Test void a07_booleanCoercionRejectsUnknown() {
122 		var ex = assertThrows(IllegalArgumentException.class, () -> vr().resolve("#{boolfn(perhaps)}"));
123 		assertTrue(ex.getMessage().contains("cannot coerce 'perhaps' to boolean"), ex.getMessage());
124 	}
125 
126 	@Test void a08_stringArrayJsonShortcutInNonFinalSlot() {
127 		// JSON-array shortcut form must be quoted in the script body so the recursive-descent
128 		// parser doesn't split on the inner commas. Once delivered to ArgCoercer as a single
129 		// String, the JSON-array parse handles the rest. The String[] slot must be NON-final
130 		// for JSON-array coercion to apply (a final String[] is variadic per the "final
131 		// slot collects excess" rule).
132 		assertEquals("X=[a, b, c]", vr().resolve("#{jsonArr(\"[\\\"a\\\",\\\"b\\\",\\\"c\\\"]\", X)}"));
133 	}
134 
135 	@Test void a09_stringArrayVariadicGather() {
136 		// Final String[] slot — args are gathered, not JSON-parsed.
137 		assertEquals("arr=[a, b, c]", vr().resolve("#{arrfn(a, b, c)}"));
138 	}
139 
140 	@Test void a09b_stringArrayBareWrappedInNonFinalSlot() {
141 		assertEquals("X=[hello]", vr().resolve("#{jsonArr(hello, X)}"));
142 	}
143 
144 	@Test void a10_objectPassthrough() {
145 		assertEquals("obj=hello", vr().resolve("#{objfn(hello)}"));
146 	}
147 }
