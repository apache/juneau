17  package org.apache.juneau.commons.svl;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.commons.*;
24  import org.junit.jupiter.api.*;
25  
26  /** Tests for {@link Structure}. */
27  class Structure_Test extends TestBase {
28  
29  	@Test void a01_ul() {
30  		var s = Structure.ul(List.of("a", "b"));
31  		assertEquals(Structure.Kind.UL, s.kind());
32  		assertEquals(List.of("a", "b"), s.items());
33  		assertEquals("- a\n- b", s.toString());
34  		assertTrue(s.options().isEmpty());
35  		assertTrue(s.rows().isEmpty());
36  	}
37  
38  	@Test void a02_ol() {
39  		var s = Structure.ol(List.of("a", "b"));
40  		assertEquals(Structure.Kind.OL, s.kind());
41  		assertEquals("1. a\n2. b", s.toString());
42  	}
43  
44  	@Test void a03_block() {
45  		var s = Structure.block("hello");
46  		assertEquals(Structure.Kind.BLOCK, s.kind());
47  		assertEquals(List.of("hello"), s.items());
48  		assertEquals("hello", s.toString());
49  	}
50  
51  	@Test void a04_table() {
52  		var s = Structure.table(List.of("n", "v"), List.of(List.of("a", 1), List.of("b", 2)));
53  		assertEquals(Structure.Kind.TABLE, s.kind());
54  		assertEquals(List.of("n", "v"), s.columns());
55  		assertEquals(2, s.rows().size());
56  		assertEquals(List.of("b", 2), s.rows().get(1));
57  		assertTrue(s.items().isEmpty());
58  		assertEquals("n\tv\na\t1\nb\t2", s.toString());
59  	}
60  
61  	@Test void b01_defensiveCopies() {
62  		var items = new ArrayList<Object>(List.of("a"));
63  		var opts = new HashMap<String,Object>(Map.of("k", "v"));
64  		var s = Structure.ul(items, opts);
65  		items.add("b");
66  		opts.put("x", "y");
67  		assertEquals(List.of("a"), s.items());
68  		assertEquals(Map.of("k", "v"), s.options());
69  
70  		var row = new ArrayList<Object>(List.of("a"));
71  		var rows = new ArrayList<List<?>>(List.of(row));
72  		var t = Structure.table(List.of("c"), rows);
73  		row.add("z");
74  		rows.clear();
75  		assertEquals(List.of(List.of("a")), t.rows());
76  	}
77  
78  	@Test void b02_immutable() {
79  		var s = Structure.ol(List.of("a"), Map.of("k", 1));
80  		assertThrows(UnsupportedOperationException.class, () -> ((List<Object>)s.items()).add("b"));
81  		assertThrows(UnsupportedOperationException.class, () -> s.options().put("x", 2));
82  		var t = Structure.table(List.of("c"), List.of(List.of("a")));
83  		assertThrows(UnsupportedOperationException.class, () -> t.columns().add("d"));
84  		assertThrows(UnsupportedOperationException.class, () -> ((List<Object>)t.rows().get(0)).add("b"));
85  	}
86  
87  	@Test void b03_optionsOverloads() {
88  		assertEquals(Map.of("a", 1), Structure.ul(List.of(), Map.of("a", 1)).options());
89  		assertEquals(Map.of("a", 1), Structure.ol(List.of(), Map.of("a", 1)).options());
90  		assertEquals(Map.of("a", 1), Structure.block("x", Map.of("a", 1)).options());
91  		assertEquals(Map.of("a", 1), Structure.table(List.of(), List.of(), Map.of("a", 1)).options());
92  	}
93  
94  	@Test void b04_nullsRejected() {
95  		assertThrows(NullPointerException.class, () -> Structure.ul(null));
96  		assertThrows(NullPointerException.class, () -> Structure.table(null, List.of()));
97  		assertThrows(NullPointerException.class, () -> Structure.table(List.of(), null));
98  	}
99  
100 	@Test void c01_emptyAndNullElements() {
101 		assertEquals("", Structure.ul(List.of()).toString());
102 		assertEquals("- null", Structure.ul(Arrays.asList((Object)null)).toString());
103 		assertEquals("null", Structure.block(null).toString());
104 	}
105 
106 	@Test void d01_nullRowAndNullOptions() {
107 		var rows = new ArrayList<List<?>>();
108 		rows.add(null);
109 		var e = assertThrows(NullPointerException.class, () -> Structure.table(List.of("c"), rows));
110 		assertTrue(e.getMessage().contains("rows"), e.getMessage());
111 		assertTrue(Structure.ul(List.of("a"), null).options().isEmpty());
112 		assertTrue(Structure.table(List.of(), List.of(), null).options().isEmpty());
113 	}
114 }
