17  package org.apache.juneau.commons.svl;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.commons.*;
24  import org.junit.jupiter.api.*;
25  
26  /**
27   * Phase E discovery wiring tests — verify that {@link VarResolver#DEFAULT} auto-registers the
28   * built-in catalog and that explicit registrations override built-ins on name collision.
29   */
30  class VarFunctionDiscovery_Test extends TestBase {
31  
32  	/** Override built-in {@code upper(...)} to force-uppercase + bang for collision testing. */
33  	public static class BangUpper extends TypedFunction {
34  		@Override public String name() { return "upper"; }
35  		public String invoke(String s) { return s.toUpperCase(Locale.ROOT) + "!"; }
36  	}
37  
38  	@Test void defaultResolverHasBuiltins() {
39  		// Sample one function from each Phase D category.
40  		assertEquals("FOO", VarResolver.DEFAULT.resolve("#{upper(foo)}"));    // string
41  		assertEquals("3", VarResolver.DEFAULT.resolve("#{toInt(3)}"));        // type-conv
42  		assertEquals("5", VarResolver.DEFAULT.resolve("#{add(2, 3)}"));       // arithmetic
43  		assertEquals("true", VarResolver.DEFAULT.resolve("#{eq(a, a)}"));     // boolean
44  		assertEquals("yes", VarResolver.DEFAULT.resolve("#{if(true, yes, no)}"));  // conditional
45  		assertEquals("true", VarResolver.DEFAULT.resolve("#{match(abc, \"a.c\")}")); // regex
46  		assertEquals("Zm9v", VarResolver.DEFAULT.resolve("#{base64Encode(foo)}"));   // encoding
47  	}
48  
49  	@Test void explicitRegistrationOverridesBuiltin() {
50  		var vr = VarResolver.create().defaultFunctions().functions(BangUpper.class).build();
51  		assertEquals("HELLO!", vr.resolve("#{upper(hello)}"));
52  	}
53  
54  	@Test void unknownFunctionFailsLazily() {
55  		// Unknown function name resolves to an exception at resolve-time.
56  		var vr = VarResolver.DEFAULT;
57  		var ex = assertThrows(IllegalArgumentException.class, () -> vr.resolve("#{nosuchfn(x)}"));
58  		assertTrue(ex.getMessage().contains("nosuchfn"), () -> "Expected message to mention function name; got: " + ex.getMessage());
59  	}
60  
61  	@Test void emptyCatalogResolverHasNoBuiltins() {
62  		// VarResolver.create().build() — no .defaultFunctions() — still resolves Vars but not
63  		// any function references.
64  		var vr = VarResolver.create().defaultVars().build();
65  		var ex = assertThrows(IllegalArgumentException.class, () -> vr.resolve("#{upper(foo)}"));
66  		assertTrue(ex.getMessage().contains("upper"), () -> "Expected message to mention 'upper'; got: " + ex.getMessage());
67  	}
68  
69  	@Test void defaultFunctionsRegistersAllCategories() {
70  		// Spot-check that every Phase D category contributes at least one function.
71  		var b = VarResolver.create().defaultFunctions();
72  		var seen = new HashSet<String>();
73  		// Build to inspect the resulting map indirectly via resolve calls.
74  		var vr = b.build();
75  		for (var template : List.of(
76  				"#{upper(x)}",          // StringFunctions
77  				"#{toInt(1)}",          // TypeConversionFunctions
78  				"#{add(1, 2)}",         // ArithmeticFunctions
79  				"#{and(true, true)}",   // BooleanFunctions
80  				"#{coalesce(\"\", x)}", // ConditionalFunctions
81  				"#{match(abc, abc)}",   // RegexFunctions
82  				"#{urlEncode(x)}",      // EncodingFunctions
83  				"#{formatDate(0)}",     // DateFunctions
84  				"#{uuid()}",            // RandomFunctions
85  				"#{size(\"[]\")}"       // JsonFunctions
86  		)) {
87  			var v = vr.resolve(template);
88  			assertNotNull(v, () -> "Template '" + template + "' returned null");
89  			seen.add(template);
90  		}
91  		assertEquals(10, seen.size());
92  	}
93  }
