54  		public EchoVar() { super("E"); }
55  		@Override public String resolve(VarResolverSession session, String arg) { return "[" + arg + "]"; }

59  	public static class RecursiveStableVar extends SimpleVar {
60  		public RecursiveStableVar() { super("RS"); }
61  		@Override public String resolve(VarResolverSession session, String arg) { return "$E{" + arg + "}"; }
62  		@Override protected boolean isStable() { return true; }

733 	@Test void o01_stableVarFoldsAtCompileTime() {
734 		// CountingStable returns its arg uppercased. Stable -> folded once.
735 		var counter = new AtomicInteger();
736 		var vr = VarResolver.create().vars(new CountingStable(counter)).build();
737 		var tpl = vr.compile("hello $CS{world}");
738 		assertEquals(1, counter.get(), "Stable var folded once at compile time");
739 		for (var i = 0; i < 5; i++)
740 			assertEquals("hello WORLD", tpl.resolve(vr.createSession()));

744 	public static class CountingStable extends SimpleVar {
745 		final AtomicInteger calls;
746 		public CountingStable(AtomicInteger calls) {
747 			super("CS");
748 			this.calls = calls;
749 		}
750 		@Override public String resolve(VarResolverSession session, String arg) {
751 			calls.incrementAndGet();
752 			return arg == null ? "" : arg.toUpperCase(Locale.ROOT);
753 		}
754 		@Override protected boolean isStable() { return true; }
755 	}
