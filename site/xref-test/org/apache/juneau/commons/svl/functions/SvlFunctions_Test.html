201 	private final VarResolver json = VarResolver.create().functions(JsonFunctions.ALL).build();
202 
203 	private static String arg(String raw) {
204 		var sb = new StringBuilder("\"");
205 		for (var c : raw.toCharArray()) {
206 			if (c == '"' || c == '\\') sb.append('\\');
207 			sb.append(c);
208 		}
209 		sb.append('"');
210 		return sb.toString();
211 	}
212 
213 	@Test void i01_jsonPath_rootObject() {
214 		var doc = "{\"a\":1}";
215 		// Path "/" returns the root
216 		var result = json.resolve("#{jsonPath(" + arg(doc) + ", \"/\")}");
217 		assertNotNull(result);
218 	}
219 	@Test void i02_get_outOfBoundsArray() {
220 		assertEquals("", json.resolve("#{get(" + arg("[\"a\",\"b\"]") + ", 99)}"));
221 	}
222 	@Test void i03_get_negativeIndex() {
223 		assertEquals("", json.resolve("#{get(" + arg("[\"a\",\"b\"]") + ", -1)}"));
224 	}
