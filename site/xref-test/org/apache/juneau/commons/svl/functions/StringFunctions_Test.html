17  package org.apache.juneau.commons.svl.functions;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.apache.juneau.commons.*;
22  import org.apache.juneau.commons.svl.*;
23  import org.junit.jupiter.api.*;
24  
25  /** Tests for {@link StringFunctions}. */
26  class StringFunctions_Test extends TestBase {
27  
28  	private final VarResolver vr = VarResolver.create().functions(StringFunctions.ALL).build();
29  
30  	@Test void a01_substring_2arg() { assertEquals("llo", vr.resolve("#{substring(hello, 2)}")); }
31  	@Test void a02_substring_3arg() { assertEquals("ell", vr.resolve("#{substring(hello, 1, 4)}")); }
32  	@Test void a03_substring_clamps() { assertEquals("hello", vr.resolve("#{substring(hello, -5, 99)}")); }
33  
34  	@Test void a04_upper() { assertEquals("HELLO", vr.resolve("#{upper(hello)}")); }
35  	@Test void a05_lower() { assertEquals("hello", vr.resolve("#{lower(HELLO)}")); }
36  	@Test void a06_trim() { assertEquals("hi", vr.resolve("#{trim(\"  hi  \")}")); }
37  	@Test void a07_stripLeading() { assertEquals("hi  ", vr.resolve("#{stripLeading(\"  hi  \")}")); }
38  	@Test void a08_stripTrailing() { assertEquals("  hi", vr.resolve("#{stripTrailing(\"  hi  \")}")); }
39  	@Test void a09_stripSlashes() { assertEquals("a/b", vr.resolve("#{stripSlashes(\"/a/b/\")}")); }
40  
41  	@Test void a10_pathToken_bare() { assertEquals("jsp", vr.resolve("#{pathToken(jsp)}")); }
42  	@Test void a11_pathToken_leading() { assertEquals("jsp", vr.resolve("#{pathToken(/jsp)}")); }
43  	@Test void a12_pathToken_trailingSlash() { assertEquals("jsp", vr.resolve("#{pathToken(/jsp/)}")); }
44  	@Test void a13_pathToken_trailingStar() { assertEquals("jsp", vr.resolve("#{pathToken(/jsp/*)}")); }
45  	@Test void a14_pathToken_bareTrailingStar() { assertEquals("jsp", vr.resolve("#{pathToken(jsp/*)}")); }
46  	@Test void a15_pathToken_multi() { assertEquals("api/v1", vr.resolve("#{pathToken(/api/v1/*)}")); }
47  	@Test void a16_pathToken_emptyJustSlash() { assertEquals("", vr.resolve("#{pathToken(/)}")); }
48  	@Test void a17_pathToken_starOnly() { assertEquals("", vr.resolve("#{pathToken(/*)}")); }
49  
50  	@Test void a18_len() { assertEquals("5", vr.resolve("#{len(hello)}")); }
51  	@Test void a19_len_empty() { assertEquals("0", vr.resolve("#{len(\"\")}")); }
52  
53  	@Test void a20_replace() { assertEquals("hxllo", vr.resolve("#{replace(hello, e, x)}")); }
54  	@Test void a21_contains_true() { assertEquals("true", vr.resolve("#{contains(hello, ell)}")); }
55  	@Test void a22_contains_false() { assertEquals("false", vr.resolve("#{contains(hello, xyz)}")); }
56  	@Test void a23_startsWith() { assertEquals("true", vr.resolve("#{startsWith(hello, hel)}")); }
57  	@Test void a24_endsWith() { assertEquals("true", vr.resolve("#{endsWith(hello, llo)}")); }
58  
59  	@Test void a25_concat() { assertEquals("abc", vr.resolve("#{concat(a, b, c)}")); }
60  	@Test void a26_repeat() { assertEquals("abcabcabc", vr.resolve("#{repeat(abc, 3)}")); }
61  	@Test void a27_repeat_zero() { assertEquals("", vr.resolve("#{repeat(abc, 0)}")); }
62  	@Test void a28_reverse() { assertEquals("olleh", vr.resolve("#{reverse(hello)}")); }
63  
64  	@Test void a29_format_string() { assertEquals("hi-007", vr.resolve("#{format(\"%s-%03d\", hi, 7)}")); }
65  	@Test void a30_split() { assertEquals("[\"a\",\"b\",\"c\"]", vr.resolve("#{split(\"a,b,c\", \",\")}")); }
66  	@Test void a31_join() { assertEquals("a/b/c", vr.resolve("#{join(\"/\", a, b, c)}")); }
67  
68  	@Test void a32_compose_join_split() {
69  		assertEquals("X-Y-Z", vr.resolve("#{join(\"-\", X, Y, Z)}"));
70  	}
71  }
