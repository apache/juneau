31  	private static final VarResolver LENIENT = VarResolver.create().defaultFunctions().build();
32  	private static final VarResolver STRICT = VarResolver.create().defaultFunctions().strict(true).build();
33  
34  	private static Object op(VarResolver vr, String name, Object...args) {
35  		var s = vr.createSession();
36  		try {
37  			return s.findFunction("$op." + name).invokeValue(s, Arrays.asList(args));
38  		} catch (RuntimeException e) {
39  			throw e;
40  		} catch (Exception e) {
41  			throw new RuntimeException(e);
42  		}
43  	}
44  
45  	private static Object op(String name, Object...args) { return op(LENIENT, name, args); }

293 	@Test void f12_matchesUsesSharedCache() {
294 		var re = "uniq-" + System.nanoTime() + "[a-z]+";
295 		var before = RegexFunctions.PATTERN_CACHE.size();
296 		assertEquals(true, op("matches", re.replace("[a-z]+", "abc"), re));
297 		assertEquals(before + 1, RegexFunctions.PATTERN_CACHE.size());
298 		// Cached by RegexFunctions' own function: no growth when reused via match().
299 		assertEquals("true", LENIENT.resolve("#{match(\"" + re.replace("[a-z]+", "abc") + "\", \"" + re + "\")}"));
300 		assertEquals(before + 1, RegexFunctions.PATTERN_CACHE.size());
301 	}
