17  package org.apache.juneau.commons.svl.functions;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.apache.juneau.commons.*;
22  import org.apache.juneau.commons.svl.*;
23  import org.junit.jupiter.api.*;
24  
25  /** Tests for {@link JsonFunctions}. */
26  class JsonFunctions_Test extends TestBase {
27  
28  	private final VarResolver vr = VarResolver.create().functions(JsonFunctions.ALL).build();
29  
30  	// JSON document with double-quotes used internally; the helper escapes them for embedding
31  	// inside a `"`-quoted script argument (so the script parser sees the entire JSON as one
32  	// argument despite the inner commas and quotes).
33  	private static final String JSON_DOC =
34  		"{\"name\":\"alice\",\"age\":30,\"tags\":[\"a\",\"b\",\"c\"],\"address\":{\"city\":\"sf\"}}";
35  
36  	/** Wrap a string for use as a {@code "..."}-quoted script arg by escaping {@code "} and {@code \\}. */
37  	private static String arg(String raw) {
38  		var sb = new StringBuilder("\"");
39  		for (var c : raw.toCharArray()) {
40  			if (c == '"' || c == '\\') sb.append('\\');
41  			sb.append(c);
42  		}
43  		sb.append('"');
44  		return sb.toString();
45  	}
46  
47  	@Test void a01_jsonPath_topLevel() {
48  		assertEquals("alice", vr.resolve("#{jsonPath(" + arg(JSON_DOC) + ", \"/name\")}"));
49  	}
50  
51  	@Test void a02_jsonPath_arrayIndex() {
52  		assertEquals("b", vr.resolve("#{jsonPath(" + arg(JSON_DOC) + ", \"/tags/1\")}"));
53  	}
54  
55  	@Test void a03_jsonPath_nested() {
56  		assertEquals("sf", vr.resolve("#{jsonPath(" + arg(JSON_DOC) + ", \"/address/city\")}"));
57  	}
58  
59  	@Test void a04_jsonPath_missing_emptyDefault() {
60  		assertEquals("", vr.resolve("#{jsonPath(" + arg(JSON_DOC) + ", \"/nope\")}"));
61  	}
62  
63  	@Test void a05_jsonPath_missing_explicitDefault() {
64  		assertEquals("fallback", vr.resolve("#{jsonPath(" + arg(JSON_DOC) + ", \"/nope\", fallback)}"));
65  	}
66  
67  	@Test void a06_get_fromObject() {
68  		assertEquals("alice", vr.resolve("#{get(" + arg(JSON_DOC) + ", name)}"));
69  	}
70  
71  	@Test void a07_get_fromArray() {
72  		assertEquals("a", vr.resolve("#{get(" + arg("[\"a\",\"b\"]") + ", 0)}"));
73  	}
74  
75  	@Test void a08_get_missing() {
76  		assertEquals("", vr.resolve("#{get(" + arg(JSON_DOC) + ", nope)}"));
77  	}
78  
79  	@Test void a09_keys_object() {
80  		assertEquals("[\"name\",\"age\",\"tags\",\"address\"]",
81  			vr.resolve("#{keys(" + arg(JSON_DOC) + ")}"));
82  	}
83  
84  	@Test void a10_keys_nonObject() {
85  		assertEquals("[]", vr.resolve("#{keys(" + arg("[\"a\",\"b\"]") + ")}"));
86  	}
87  
88  	@Test void a11_values_object() {
89  		var s = vr.resolve("#{values(" + arg("{\"a\":1,\"b\":2}") + ")}");
90  		assertEquals("[\"1\",\"2\"]", s);
91  	}
92  
93  	@Test void a12_values_array() {
94  		var s = vr.resolve("#{values(" + arg("[\"a\",\"b\"]") + ")}");
95  		assertEquals("[\"a\",\"b\"]", s);
96  	}
97  
98  	@Test void a13_size_object() {
99  		assertEquals("4", vr.resolve("#{size(" + arg(JSON_DOC) + ")}"));
100 	}
101 
102 	@Test void a14_size_array() {
103 		assertEquals("2", vr.resolve("#{size(" + arg("[1,2]") + ")}"));
104 	}
105 
106 	@Test void a15_size_string() {
107 		assertEquals("5", vr.resolve("#{size(" + arg("\"hello\"") + ")}"));
108 	}
109 }
