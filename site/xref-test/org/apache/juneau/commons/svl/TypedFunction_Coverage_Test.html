17  package org.apache.juneau.commons.svl;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.apache.juneau.commons.*;
22  import org.junit.jupiter.api.*;
23  
24  /**
25   * Targeted coverage tests for {@link TypedFunction}, filling gaps not exercised by
26   * {@link VarFunction_Test} / {@link ArgCoercer_Test} (multi-overload dispatch selection,
27   * null/checked-exception invoke results, and the "no invoke method" construction failure).
28   */
29  @SuppressWarnings({
30  	"java:S1172", // invoke(String) parameter is required by TypedFunction's reflective signature matching.
31  	"java:S5778", // assertThrows lambdas with chained calls; intermediate invocations do not throw in practice
32  	"unused" // Parameter required to match the 1-arg invoke() signature dispatched to via reflection.
33  })
34  class TypedFunction_Coverage_Test extends TestBase {
35  
36  	/** Two fixed-arity overloads - dispatch must skip the (sorted-first) 3-arg one to reach the 2-arg one. */
37  	public static class Substring extends TypedFunction {
38  		@Override public String name() { return "substring"; }
39  		public String invoke(String s, int start) { return s.substring(start); }
40  		public String invoke(String s, int start, int end) { return s.substring(start, end); }
41  	}
42  
43  	/** Fixed-arity overload combined with a variadic overload of a smaller fixed count. */
44  	public static class MultiVariadic extends TypedFunction {
45  		@Override public String name() { return "multi"; }
46  		public String invoke(String a, String b) { return "pair:" + a + "," + b; }
47  		public String invoke(String[] parts) { return "variadic:" + parts.length; }
48  	}
49  
50  	/** Returns null to exercise the null-result-becomes-empty-string branch. */
51  	public static class NullReturning extends TypedFunction {
52  		@Override public String name() { return "nullfn"; }
53  		public String invoke(String s) { return null; }
54  	}
55  
56  	/** Throws a checked (non-RuntimeException) exception from invoke. */
57  	public static class CheckedThrower extends TypedFunction {
58  		@Override public String name() { return "checkedthrow"; }
59  		public String invoke(String s) throws java.io.IOException { throw new java.io.IOException("boom"); }
60  	}
61  
62  	/** Throws an unchecked exception from invoke - should propagate as-is, not get wrapped. */
63  	public static class RuntimeThrower extends TypedFunction {
64  		@Override public String name() { return "runtimethrow"; }
65  		public String invoke(String s) { throw new IllegalStateException("kaboom"); }
66  	}
67  
68  	/** No invoke(...) method declared at all - construction must fail. */
69  	public static class NoInvoke extends TypedFunction {
70  		@Override public String name() { return "noinvoke"; }
71  	}
72  
73  	@Test
74  	void a01_multiOverload_dispatchSkipsNonMatchingArityToFindLaterMatch() {
75  		var vr = VarResolver.create().functions(new Substring()).build();
76  		assertEquals("llo", vr.resolve("#{substring(hello, 2)}"));
77  		assertEquals("el", vr.resolve("#{substring(hello, 1, 3)}"));
78  	}
79  
80  	@Test
81  	void a02_fixedArityOverloadCombinedWithVariadic_fixedArityPreferredWhenMatching() {
82  		var vr = VarResolver.create().functions(new MultiVariadic()).build();
83  		assertEquals("pair:a,b", vr.resolve("#{multi(a, b)}"));
84  		assertEquals("variadic:3", vr.resolve("#{multi(a, b, c)}"));
85  	}
86  
87  	@Test
88  	void a03_variadicOverload_arityBelowFixedCount_notAccepted() {
89  		// MultiVariadic's variadic overload has fixedCount 0 (String[] alone), so any n >= 0 matches it,
90  		// but the 2-arg fixed overload takes precedence for n==2. Verify a 0-arg call still resolves via
91  		// the variadic overload (n=0 >= fixedCount=0), confirming acceptsArity's variadic branch works both ways.
92  		var vr = VarResolver.create().functions(new MultiVariadic()).build();
93  		assertEquals("variadic:0", vr.resolve("#{multi()}"));
94  	}
95  
96  	@Test
97  	void a04_nullInvokeResult_resolvesToEmptyString() {
98  		var vr = VarResolver.create().functions(new NullReturning()).build();
99  		assertEquals("", vr.resolve("#{nullfn(x)}"));
100 	}
101 
102 	@Test
103 	void a05_checkedExceptionFromInvoke_wrappedAsIllegalArgumentException() {
104 		var vr = VarResolver.create().functions(new CheckedThrower()).build();
105 		var ex = assertThrows(IllegalArgumentException.class, () -> vr.resolve("#{checkedthrow(x)}"));
106 		assertTrue(ex.getMessage().contains("checkedthrow"), ex.getMessage());
107 		assertTrue(ex.getMessage().contains("boom"), ex.getMessage());
108 	}
109 
110 	@Test
111 	void a06_runtimeExceptionFromInvoke_propagatedUnwrapped() {
112 		var vr = VarResolver.create().functions(new RuntimeThrower()).build();
113 		var ex = assertThrows(IllegalStateException.class, () -> vr.resolve("#{runtimethrow(x)}"));
114 		assertEquals("kaboom", ex.getMessage());
115 	}
116 
117 	@Test
118 	void a07_noInvokeMethodDeclared_constructionThrows() {
119 		var ex = assertThrows(IllegalArgumentException.class, NoInvoke::new);
120 		assertTrue(ex.getMessage().contains("must declare a public invoke"), ex.getMessage());
121 	}
122 }
