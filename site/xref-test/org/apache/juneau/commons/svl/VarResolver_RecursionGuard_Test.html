17  package org.apache.juneau.commons.svl;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.apache.juneau.commons.TestBase;
22  import org.junit.jupiter.api.*;
23  
24  /**
25   * Tests the recursion depth guard in {@link VarResolver}.
26   */
27  class VarResolver_RecursionGuard_Test extends TestBase {
28  
29  	public static class SelfVar extends SimpleVar {
30  		public SelfVar() {
31  			super("SELF");
32  		}
33  
34  		@Override
35  		public String resolve(VarResolverSession s, String arg) {
36  			return "${SELF:" + arg + "}";
37  		}
38  	}
39  
40  	/** Test var that returns a system property and, like most vars, allows its output to be re-resolved. */
41  	public static class PropVar extends SimpleVar {
42  		public PropVar() {
43  			super("PV");
44  		}
45  
46  		@Override
47  		public String resolve(VarResolverSession s, String arg) {
48  			return System.getProperty(arg);
49  		}
50  	}
51  
52  	@Test void a01_selfReferenceThrowsInsteadOfOverflow() {
53  		var r = VarResolver.create().vars(SelfVar.class).maxRecursionDepth(8).build();
54  		var e = assertThrows(VarResolverException.class, () -> r.resolve("${SELF:x}"));
55  		assertTrue(e.getMessage().contains("recursion depth"), e.getMessage());
56  		assertTrue(e.getMessage().startsWith("Maximum variable recursion depth"), e.getMessage());
57  		assertNull(e.getCause());
58  	}
59  
60  	@Test void a02_normalNestingUnaffected() {
61  		var r = VarResolver.create().vars(PropVar.class).maxRecursionDepth(8).build();
62  		System.setProperty("svl.test.a02a", "${PV:svl.test.a02b}");
63  		System.setProperty("svl.test.a02b", "v");
64  		try {
65  			assertEquals("v", r.resolve("${PV:svl.test.a02a}"));
66  		} finally {
67  			System.clearProperty("svl.test.a02a");
68  			System.clearProperty("svl.test.a02b");
69  		}
70  	}
71  
72  	@Test void a03_defaultIs32() {
73  		assertEquals(32, VarResolver.DEFAULT_MAX_RECURSION_DEPTH);
74  	}
75  
76  	@Test void a04_builderRejectsNonPositive() {
77  		var b = VarResolver.create();
78  		assertThrows(IllegalArgumentException.class, () -> b.maxRecursionDepth(0));
79  		assertThrows(IllegalArgumentException.class, () -> b.maxRecursionDepth(-1));
80  	}
81  
82  	@Test void a05_exactBoundary() {
83  		// a -> ${PV:b} -> ${PV:c} -> v : the output of a and b are each re-resolved, so depth 2.
84  		System.setProperty("svl.test.a05a", "${PV:svl.test.a05b}");
85  		System.setProperty("svl.test.a05b", "${PV:svl.test.a05c}");
86  		System.setProperty("svl.test.a05c", "v");
87  		try {
88  			var ok = VarResolver.create().vars(PropVar.class).maxRecursionDepth(2).build();
89  			assertEquals("v", ok.resolve("${PV:svl.test.a05a}"));
90  			var tooSmall = VarResolver.create().vars(PropVar.class).maxRecursionDepth(1).build();
91  			assertThrows(VarResolverException.class, () -> tooSmall.resolve("${PV:svl.test.a05a}"));
92  		} finally {
93  			System.clearProperty("svl.test.a05a");
94  			System.clearProperty("svl.test.a05b");
95  			System.clearProperty("svl.test.a05c");
96  		}
97  	}
98  
99  	@Test void a06_sessionReusableAfterGuardThrows() {
100 		var r = VarResolver.create().vars(SelfVar.class).defaultVars().maxRecursionDepth(4).build();
101 		var session = r.createSession();
102 		assertThrows(VarResolverException.class, () -> session.resolve("${SELF:x}"));
103 		assertThrows(VarResolverException.class, () -> session.resolve("${SELF:x}"));
104 		assertEquals("plain", session.resolve("plain"));
105 	}
106 
107 	@Test void a07_copyKeepsMaxRecursionDepth() {
108 		var r = VarResolver.create().defaultVars().maxRecursionDepth(8).build();
109 		assertEquals(8, r.copy().build().maxRecursionDepth);
110 	}
111 }
