32  	/** Function returning a fixed typed value. */
33  	static VarFunction fn(String name, Supplier<Object> s) {
34  		return new VarFunction() {
35  			@Override public String name() { return name; }
36  			@Override public int minArity() { return 0; }
37  			@Override public int maxArity() { return 0; }
38  			@Override public String invoke(VarResolverSession session, List<Object> args) { return "legacy"; }
39  			@Override public Object invokeValue(VarResolverSession session, List<Object> args) { return s.get(); }
40  		};
41  	}

43  	static VarResolver.Builder base() {
44  		return VarResolver.create().defaultFunctions()
45  			.functions(fn("names", () -> List.of("a", "b")))
46  			.functions(fn("nums", () -> List.of(1, 2.5, 3.0, true)))
47  			.functions(fn("themap", () -> new LinkedHashMap<>(Map.of("k", "v"))))
48  			.functions(fn("nested", () -> List.of(List.of("x"), Map.of("m", Arrays.asList(1, null)))))
49  			.functions(fn("arr", () -> new int[]{1, 2}))
50  			.functions(fn("esc", () -> List.of("q\"b\\s\nt\u0001é")))
51  			.functions(fn("nul", () -> null))
52  			.functions(fn("dbl", () -> 4.0))
53  			.functions(fn("struct", () -> Structure.ul(List.of("p", "q"))))
54  			.functions(fn("cyc", () -> { var l = new ArrayList<Object>(); l.add(l); return l; }))
55  			.functions(fn("big", () -> new ArrayList<>(Collections.nCopies(10, "z"))));
56  	}

218 	@Test void g01_stableCollectionNotFoldedSoOverrideApplies() {
219 		var vr = base().build();
220 		// Collections are never folded (only stable literal/var expressions fold, and those yield scalars here), so a compiled template honours the override.
221 		var tpl = vr.compile("#{names()}");
222 		assertEquals("[\"a\",\"b\"]", tpl.resolve(vr.createSession()));
223 		assertEquals("L2", tpl.resolve(vr.createSession().valueFormatter(o -> "L" + ((Collection<?>)o).size())));
224 	}
225 	/** Stable var whose typed value is a List. */
226 	public static class StableList extends DefaultingVar {
227 		public StableList() { super("SL"); }
228 		@Override public String resolve(VarResolverSession session, String arg) { return "text"; }
229 		@Override protected Object resolveValue(VarResolverSession session, String arg) { return List.of("p", "q"); }
230 		@Override protected boolean isStable() { return true; }
231 	}
232 
233 	@Test void g03_stableCollectionExpressionHonoursOverride() {
234 		var vr = base().vars(new StableList()).build();
235 		var tpl = vr.compile("#{${SL:x}}");
236 		assertEquals("[\"p\",\"q\"]", tpl.resolve(vr.createSession()));
237 		assertEquals("L2", tpl.resolve(vr.createSession().valueFormatter(o -> "L" + ((Collection<?>)o).size())));
238 	}
