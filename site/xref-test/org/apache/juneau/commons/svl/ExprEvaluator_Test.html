349 		// A single projection of exactly maxCollectionSize elements works.
350 		assertEquals(1000, ((List<?>)ev("big.![this]")).size());
351 		assertEquals(1000, ((List<?>)evs("big.![this]")).size());
352 		// Nested iteration shares ONE budget per evaluation.
353 		assertThrows(ExprException.class, () -> evs("big.![big.![this]]"));
354 		var r = (List<?>)ev("big.![big.![this]]");
355 		// The first outer element spends 1 of the 1000, its inner projection gets the other 999, then the outer loop stops.
356 		assertEquals(1, r.size());
357 		assertEquals(999, ((List<?>)r.get(0)).size());
358 		// Budget is per evaluation, not per context: a fresh evaluation starts over.
359 		assertEquals(1000, ((List<?>)ev("big.![this]")).size());
