34  	/** Delegate that counts calls per name; names starting with "present." resolve to themselves, "nullval." to an explicit null. */
35  	private static final class CountingDelegate implements PropertySource {
36  		final ConcurrentHashMap<String,AtomicInteger> calls = new ConcurrentHashMap<>();
37  		final AtomicInteger total = new AtomicInteger();
38  
39  		@Override
40  		public PropertyLookupResult get(String name) {
41  			total.incrementAndGet();
42  			if (name != null)
43  				calls.computeIfAbsent(name, k -> new AtomicInteger()).incrementAndGet();
44  			if (name == null)
45  				return PropertyLookupResult.missing();
46  			if (name.startsWith("present."))
47  				return PropertyLookupResult.present(o(name));
48  			if (name.startsWith("nullval."))
49  				return PropertyLookupResult.present(Optional.<String>empty());
50  			return PropertyLookupResult.missing();
51  		}
52  
53  		int callsFor(String name) {
54  			var c = calls.get(name);
55  			return c == null ? 0 : c.get();
56  		}
57  	}
58  
59  	@Test void a01_hitAndMiss_eachServedFromCacheAfterFirstCall() {
60  		var d = new CountingDelegate();
61  		var s = CachingPropertySource.of(d);
62  		for (var i = 0; i < 5; i++) {
63  			assertEquals("present.a", s.get("present.a").value().orElse(null));
64  			assertFalse(s.get("absent.a").isPresent());
65  		}
66  		assertEquals(1, d.callsFor("present.a"));
67  		assertEquals(1, d.callsFor("absent.a"));
68  	}
69  
70  	@Test void a02_explicitNull_cachedAsPresent() {
71  		var d = new CountingDelegate();
72  		var s = CachingPropertySource.of(d);
73  		for (var i = 0; i < 3; i++) {
74  			var r = s.get("nullval.a");
75  			assertTrue(r.isPresent());
76  			assertTrue(r.value().isEmpty());
77  		}
78  		assertEquals(1, d.callsFor("nullval.a"));
79  	}
80  
81  	@Test void a03_capReached_newNamesPassThrough_firstSeenStayCached() {
82  		var d = new CountingDelegate();
83  		var s = CachingPropertySource.of(d);
84  		for (var i = 0; i < CachingPropertySource.MAX_ENTRIES; i++)
85  			s.get("present.n" + i);
86  		assertEquals(CachingPropertySource.MAX_ENTRIES, s.size());
87  		// A name beyond the cap is answered correctly, uncached: the delegate is called every time.
88  		for (var i = 0; i < 3; i++)
89  			assertEquals("present.extra", s.get("present.extra").value().orElse(null));
90  		assertEquals(3, d.callsFor("present.extra"));
91  		assertEquals(CachingPropertySource.MAX_ENTRIES, s.size());
92  		// A first-seen name is still served from cache.
93  		s.get("present.n0");
94  		s.get("present.n0");
95  		assertEquals(1, d.callsFor("present.n0"));
96  	}

98  	@Test void a04_nullName_skipsCache_reachesDelegate() {
99  		var d = new CountingDelegate();
100 		var s = CachingPropertySource.of(d);
101 		assertFalse(s.get(null).isPresent());
102 		assertFalse(s.get(null).isPresent());
103 		assertEquals(2, d.total.get());
104 		assertEquals(0, s.size());
105 		var thrower = CachingPropertySource.of(name -> {
106 			if (name == null)
107 				throw new IllegalStateException("null name");
108 			return PropertyLookupResult.missing();
109 		});
110 		assertThrows(IllegalStateException.class, () -> thrower.get(null));
111 	}

113 	@Test void a05_concurrent_firstLookups_andSoftCapBound() throws Exception {
114 		var threads = 8;
115 		var pool = Executors.newFixedThreadPool(threads);
116 		try {
117 			// Part 1: N threads race on one name; all get an equal result and it is cached afterwards.
118 			var d1 = new CountingDelegate();
119 			var s1 = CachingPropertySource.of(d1);
120 			var start1 = new CountDownLatch(1);
121 			var futures = new ArrayList<Future<String>>();
122 			for (var t = 0; t < threads; t++)
123 				futures.add(pool.submit(() -> {
124 					start1.await();
125 					return s1.get("present.race").value().orElse(null);
126 				}));
127 			start1.countDown();
128 			for (var f : futures)
129 				assertEquals("present.race", f.get(30, TimeUnit.SECONDS));
130 			var before = d1.callsFor("present.race");
131 			assertTrue(before >= 1 && before <= threads);
132 			s1.get("present.race");
133 			assertEquals(before, d1.callsFor("present.race"), "cached after the race");
134 
135 			// Part 2: N threads fill the cache at once; the documented soft cap bounds the overshoot by the thread count.
136 			var d2 = new CountingDelegate();
137 			var s2 = CachingPropertySource.of(d2);
138 			var start2 = new CountDownLatch(1);
139 			var fills = new ArrayList<Future<?>>();
140 			for (var t = 0; t < threads; t++) {
141 				var id = t;
142 				fills.add(pool.submit(() -> {
143 					start2.await();
144 					for (var i = 0; i < CachingPropertySource.MAX_ENTRIES * 2; i++)
145 						s2.get("present.t" + id + "." + i);
146 					return null;
147 				}));
148 			}
149 			start2.countDown();
150 			for (var f : fills)
151 				f.get(60, TimeUnit.SECONDS);
152 			assertTrue(s2.size() <= CachingPropertySource.MAX_ENTRIES + threads, "size=" + s2.size());
153 		} finally {
154 			pool.shutdownNow();
155 		}
156 	}
