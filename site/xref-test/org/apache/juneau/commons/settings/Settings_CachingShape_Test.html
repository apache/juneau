17  package org.apache.juneau.commons.settings;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.commons.*;
24  import org.junit.jupiter.api.*;
25  import org.junit.jupiter.api.parallel.*;
26  
27  /**
28   * Pins which sources are wrapped in a {@link CachingPropertySource} and which are not.
29   */
30  @ResourceLock(Resources.SYSTEM_PROPERTIES)
31  @ResourceLock(value = "org.apache.juneau.commons.settings.Settings.singleton", mode = ResourceAccessMode.READ)  // reads the singleton chain
32  class Settings_CachingShape_Test extends TestBase {
33  
34  	private static final Set<Class<?>> RUN_CONSTANT = Set.of(
35  		SystemEnvPropertySource.class, DotenvPropertySource.class, ManifestFilePropertySource.class, ArgsPropertySource.class);
36  
37  	private static PropertySource unwrap(PropertySource s) {
38  		return s instanceof CachingPropertySource c ? c.delegate() : s;
39  	}
40  
41  	@Test void e01a_relaxedEnvConstant_isCachingOverRelaxed() {
42  		assertInstanceOf(CachingPropertySource.class, Settings.RELAXED_SYSTEM_ENV_SOURCE);
43  		assertInstanceOf(RelaxedPropertySource.class, ((CachingPropertySource)Settings.RELAXED_SYSTEM_ENV_SOURCE).delegate());
44  	}
45  
46  	@Test void e01b_singleton_runConstantSources_areCaching() {
47  		var seen = new HashSet<Class<?>>();
48  		for (var s : Settings.get().sources()) {
49  			var cls = unwrap(s).getClass();
50  			if (RUN_CONSTANT.contains(cls)) {
51  				assertInstanceOf(CachingPropertySource.class, s, cls.getSimpleName() + " must be wrapped");
52  				seen.add(cls);
53  			}
54  		}
55  		// All four providers are in META-INF/services and their create() never returns null, so all four are always there.
56  		assertEquals(RUN_CONSTANT, seen, "every run-constant source must be on the singleton chain, wrapped");
57  	}
58  
59  	@Test void e02_singleton_noSystemPropertySourceIsCaching() {
60  		for (var s : Settings.get().sources()) {
61  			if (s instanceof CachingPropertySource c) {
62  				assertNotSame(Settings.SYSTEM_PROPERTY_SOURCE, c.delegate());
63  				assertFalse(c.delegate() instanceof SystemPropertyPropertySource);
64  			}
65  		}
66  	}
67  
68  	@Test void e04_providers_wrapExactlyTheRunConstantOnes() {
69  		assertWrapped(new SystemEnvPropertySourceProvider().create(), SystemEnvPropertySource.class);
70  		assertWrapped(new DotenvPropertySourceProvider().create(), DotenvPropertySource.class);
71  		assertWrapped(new ManifestFilePropertySourceProvider().create(), ManifestFilePropertySource.class);
72  		assertWrapped(new ArgsPropertySourceProvider().create(), ArgsPropertySource.class);
73  		// The system-property provider is never wrapped: system properties are mutable.
74  		assertEquals(SystemPropertyPropertySource.class, new SystemPropertyPropertySourceProvider().create().getClass());
75  	}
76  
77  	private static void assertWrapped(PropertySource s, Class<?> expectedDelegate) {
78  		assertInstanceOf(CachingPropertySource.class, s);
79  		assertEquals(expectedDelegate, ((CachingPropertySource)s).delegate().getClass());
80  	}
81  
82  	@Test void e05_wrapped_source_stillShadowedByLocalOverride() {
83  		var settings = Settings.create()
84  			.addSource(CachingPropertySource.of(name -> name.equals("e05.k") ? PropertyLookupResult.present("env") : PropertyLookupResult.missing()))
85  			.build();
86  		assertEquals("env", settings.get("e05.k").orElse(null));
87  		settings.setLocal("e05.k", "local");
88  		try {
89  			assertEquals("local", settings.get("e05.k").orElse(null));
90  		} finally {
91  			settings.clearLocal();
92  		}
93  		assertEquals("env", settings.get("e05.k").orElse(null));
94  	}
95  }
