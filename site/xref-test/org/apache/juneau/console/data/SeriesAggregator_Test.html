32  	static final Instant T0 = Instant.parse("2026-10-05T00:00:00Z"); // a Monday
33  	static final long DAY = 86_400_000L, HOUR = 3_600_000L;
34  
35  	static Map<String,List<String>> q(String...kv) {
36  		var m = new LinkedHashMap<String,List<String>>();
37  		for (var i = 0; i < kv.length; i += 2)
38  			m.computeIfAbsent(kv[i], k -> new ArrayList<>()).add(kv[i + 1]);
39  		return m;
40  	}
41  
42  	static SeriesSource.Row row(Instant t, String org, double ms) {
43  		Map<String,Object> m = map("time", t, "org", org, "ms", ms);
44  		return SeriesSource.Row.of(m);
45  	}
46  
47  	/** Hour h has org "a" when h is even, else "b", and ms = h. */
48  	static Stream<SeriesSource.Row> hourly(int hours) {
49  		return IntStream.range(0, hours).mapToObj(h -> row(T0.plusMillis(HOUR * h), h % 2 == 0 ? "a" : "b", h));
50  	}
51  
52  	static SeriesAggregator agg() {
53  		return SeriesAggregator.create().timeField("time").allowFields("org", "ms");
54  	}
55  
56  	static Map<String,List<String>> valueless(String name, String...kv) {
57  		var m = q(kv);
58  		m.put(name, Collections.singletonList(null));
59  		return m;
60  	}

62  	static String iso(long ms) { return Instant.ofEpochMilli(ms).toString(); }

64  	@Test void a01_dayCountAndSum() {
65  		var r = agg()
66  			.aggregate(SeriesAggregator.Request.of(q("start", T0.toString(), "end", iso(T0.toEpochMilli() + 2 * DAY),
67  			"bucket", "day", "measure", "count", "measure", "sum(ms)")), hourly(48));
68  		assertEquals(List.of("bucket", "count", "sum(ms)"), r.dims());
69  		assertEquals(List.of(T0.toEpochMilli(), T0.toEpochMilli() + DAY), r.columns().get("bucket"));
70  		assertEquals(List.of(24L, 24L), r.columns().get("count"));
71  		assertEquals(List.of(276.0, 852.0), r.columns().get("sum(ms)"));
72  		assertFalse(r.truncated());
73  	}

105 	@Test void a05_weekStartsAndMonthEdges() {
106 		var day = SeriesAggregator.Request.of(q("start", T0.toString(), "end", iso(T0.toEpochMilli() + DAY), "bucket",
107 			"week"));
108 		assertEquals(List.of(T0.toEpochMilli()), agg().aggregate(day, hourly(1)).columns().get("bucket"),
109 			"Monday by default");
110 		assertEquals(List.of(T0.toEpochMilli() - DAY), agg().weekStart(DayOfWeek.SUNDAY).aggregate(day, hourly(1))
111 			.columns().get("bucket"));
112 		var jan31 = Instant.parse("2026-01-31T23:00:00Z");
113 		var r = agg()
114 			.aggregate(SeriesAggregator.Request.of(q("start", "2026-01-01T00:00:00Z", "end", "2026-03-01T00:00:00Z",
115 			"bucket", "month")),
116 			Stream.of(row(jan31, "a", 1), row(jan31.plusMillis(HOUR), "a", 1)));
117 		assertEquals(List.of(Instant.parse("2026-01-01T00:00:00Z").toEpochMilli(), Instant.parse("2026-02-01T00:00:00Z")
118 			.toEpochMilli()), r.columns().get("bucket"));
119 	}

151 	@Test void a09_rejected() {
152 		var w = new String[] { "start", T0.toString(), "end", iso(T0.toEpochMilli() + 1) };
153 		var cases = new LinkedHashMap<Map<String,List<String>>,String>();
154 		cases.put(q("end", T0.toString()), "'start' is required");
155 		cases.put(q("start", T0.toString(), "end", T0.toString()), "'end' must be after 'start'");
156 		cases.put(q("start", "nope", "end", T0.toString()),
157 			"'start' is not an ISO-8601 instant or epoch milliseconds: 'nope'");
158 		cases.put(q(w[0], w[1], w[2], w[3], "bucket", "year"), "unknown bucket 'year'");
159 		cases.put(q(w[0], w[1], w[2], w[3], "tz", "America/New_York"),
160 			"only tz 'UTC' is supported, not 'America/New_York'");
161 		cases.put(q(w[0], w[1], w[2], w[3], "groupBy", "a,b,c,d"), "at most 3 groupBy fields, not 4");
162 		cases.put(q(w[0], w[1], w[2], w[3], "measure", "median(ms)"), "unknown measure 'median(ms)'");
163 		cases.put(q(w[0], w[1], w[2], w[3], "measure", "sum:ms"), "unknown measure 'sum:ms'");
164 		cases.put(q("start", "99999999999999999999", "end", T0.toString()),
165 			"'start' is not an ISO-8601 instant or epoch milliseconds: '99999999999999999999'");
166 		cases.put(q(w[0], w[1], w[2], w[3], "groupBy", "org,org"), "duplicate groupBy field 'org'");
167 		cases.put(q(w[0], w[1], w[2], w[3], "groupBy", "bucket"), "groupBy field 'bucket' is reserved");
168 		cases.put(q(w[0], w[1], w[2], w[3], "groupBy", "count"), "groupBy field 'count' collides with a measure");
169 		cases.put(q(w[0], w[1], w[2], w[3], "measure", "count", "measure", "count"), "duplicate measure 'count'");
170 		cases.put(valueless("start"), "'start' is required");
171 		cases.forEach((query, reason) -> {
172 			var e = assertThrows(SeriesAggregator.RequestException.class, () -> SeriesAggregator.Request.of(query));
173 			assertEquals("E-75: aggregate request rejected: " + reason, e.getMessage());
174 			assertEquals(400, e.getStatusCode());
175 		});
176 		var e = assertThrows(SeriesAggregator.RequestException.class, () -> SeriesAggregator.create().aggregate(
177 			SeriesAggregator.Request.of(q(w[0], w[1], w[2], w[3], "groupBy", "org")), Stream.empty()));
178 		assertEquals("E-75: aggregate request rejected: field 'org' is not allowed", e.getMessage(),
179 			"no allowFields rejects every field");
180 	}
