19  import static org.apache.juneau.bct.BctAssertions.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import org.apache.juneau.*;
23  import org.junit.jupiter.api.*;
24  import org.junit.jupiter.params.*;
25  import org.junit.jupiter.params.provider.*;

27  /**
28   * Tests for {@link Theme#create(String)} / {@link Theme.Builder#token(String, String)} identifier REJECT
29   * guards.
30   *
31   * <p>
32   * These tests are written to run against a deliberately-unvalidated {@code Theme} skeleton first (proving the guard
33   * is genuinely absent, RED), and again once the two anchored {@code String.matches(...)} guards are added (GREEN).
34   * Token <i>values</i> are intentionally NOT covered here &mdash; that is a separate, later gate (the allowlist
35   * grammar).
36   */
37  class Theme_IdentifierGuards_Test extends TestBase {
38  
39  	//-----------------------------------------------------------------------------------------------------------------
40  	// Token name REJECT
41  	//-----------------------------------------------------------------------------------------------------------------
42  
43  	/**
44  	 * The third vector is the anchoring proof: an unanchored {@code find()}-based regex would incorrectly ACCEPT
45  	 * {@code --jc-foo;--bar} because {@code --jc-foo} matches as a substring; only a full-string
46  	 * {@code matches("^--jc-[a-z0-9-]+$")} rejects it.
47  	 */
48  	@ParameterizedTest
49  	@ValueSource(strings = {
50  		"rm-accent",         // no --jc- prefix
51  		"--jc-bad name",     // interior space
52  		"--jc-foo;--bar",    // anchoring proof (see javadoc)
53  	})
54  	void a01_tokenName_invalidNames_rejects(String name) {
55  		var b = Theme.create("acme");
56  		assertThrows(IllegalArgumentException.class, () -> b.token(name, "red"));
57  	}
58  
59  	//-----------------------------------------------------------------------------------------------------------------
60  	// Theme name REJECT
61  	//-----------------------------------------------------------------------------------------------------------------
62  
63  	@Test void a02_themeName_uppercase_rejects() {
64  		assertThrows(IllegalArgumentException.class, () -> Theme.create("Acme"));
65  	}
66  
67  	/** Path-shaped names are rejected here so a later fast-follow that interpolates the name into an asset path doesn't need to re-gate. */
68  	@Test void a03_themeName_pathShaped_rejects() {
69  		assertThrows(IllegalArgumentException.class, () -> Theme.create("../evil"));
70  	}
71  
72  	@Test void a04_themeName_empty_rejects() {
73  		assertThrows(IllegalArgumentException.class, () -> Theme.create(""));
74  	}
75  
76  	//-----------------------------------------------------------------------------------------------------------------
77  	// Positive control
78  	//-----------------------------------------------------------------------------------------------------------------
79  
80  	@Test void a05_positiveControl_roundTrips() {
81  		var theme = Theme.create("acme").token("--jc-accent", "#1589EE").build();
82  		assertBean(theme, "name,tokens{--jc-accent}", "acme,{#1589EE}");
83  	}
84  }
