28  /** Schema coverage for the client object model additions: {@code model}, {@code handlerRef}, {@code scripts}, handler subscriptions. */
29  class PageContractSchema_Model_Test extends TestBase {
30  
31  	private static String page(String extraRoot, String cards) {
32  		return "{\"contractVersion\":\"1\",\"title\":\"T\",\"nav\":[],\"activeNav\":[]," + extraRoot + "\"cards\":[" + cards + "]}";
33  	}
34  
35  	private static String html(String extra) {
36  		return "{\"id\":\"c1\",\"type\":\"html\",\"src\":\"/x\"" + extra + "}";
37  	}
38  
39  	private static List<String> errors(String json) {
40  		return PageContractSchema.get().validate(json);
41  	}
42  
43  	private static void valid(String json) {
44  		var e = errors(json);
45  		assertTrue(e.isEmpty(), () -> e + " for " + json);
46  	}
47  
48  	private static void invalid(String json) {
49  		assertFalse(errors(json).isEmpty(), () -> "expected errors for " + json);
50  	}
51  
52  	@Test void root_accepts_modelVersion_scripts_allowInline_model() {
53  		valid(page("\"modelVersion\":\"1\",\"scripts\":[\"/a.js\"],\"allowInlineHandlers\":true,\"model\":{\"on\":{\"loaded\":\"app.hello\"}},", ""));
54  	}
55  
56  	@Test void root_rejects_modelVersion_other_than_1() {
57  		invalid(page("\"modelVersion\":\"2\",", ""));
58  	}
59  
60  	@Test void root_model_page_only_on() {
61  		invalid(page("\"model\":{\"actions\":[{\"type\":\"handler\",\"id\":\"a\",\"handler\":\"app.x\"}]},", ""));
62  		invalid(page("\"model\":{\"subscribes\":[{\"topic\":\"data:x\",\"handler\":\"app.x\"}]},", ""));
63  	}
64  
65  	@ParameterizedTest
66  	@ValueSource(strings = { "loaded", "rendered", "expanded", "collapsed", "activated" })
67  	void model_on_event_names_enum(String ev) {
68  		valid(page("", html(",\"model\":{\"on\":{\"" + ev + "\":\"app.x\"}}")));
69  	}
70  
71  	@Test void model_on_unknown_event_rejected() {
72  		invalid(page("", html(",\"model\":{\"on\":{\"clicked\":\"app.x\"}}")));
73  	}
74  
75  	@Test void model_unknown_key_rejected() {
76  		invalid(page("", html(",\"model\":{\"bogus\":1}")));
77  	}
78  
79  	@Test void handlerAction_requires_type_id_handler() {
80  		invalid(page("", html(",\"model\":{\"actions\":[{\"id\":\"a\",\"handler\":\"app.x\"}]}")));
81  		invalid(page("", html(",\"model\":{\"actions\":[{\"type\":\"handler\",\"handler\":\"app.x\"}]}")));
82  		invalid(page("", html(",\"model\":{\"actions\":[{\"type\":\"handler\",\"id\":\"a\"}]}")));
83  		valid(page("", html(",\"model\":{\"actions\":[{\"type\":\"handler\",\"id\":\"a\",\"handler\":\"app.x\",\"icon\":\"i\",\"payload\":{}}]}")));
84  	}
85  
86  	@Test void handlerAction_type_const_handler() {
87  		invalid(page("", html(",\"model\":{\"actions\":[{\"type\":\"link\",\"id\":\"a\",\"handler\":\"app.x\"}]}")));
88  	}
89  
90  	@Test void handlerRef_name_or_inline() {
91  		valid(page("", html(",\"model\":{\"on\":{\"loaded\":{\"inline\":\"return 1;\"}}}")));
92  		invalid(page("", html(",\"model\":{\"on\":{\"loaded\":{\"inline\":\"\"}}}")));
93  		invalid(page("", html(",\"model\":{\"on\":{\"loaded\":{\"inline\":\"x\",\"extra\":1}}}")));
94  	}

96  	@Test void handlerRef_bad_name_rejected() {
97  		invalid(page("", html(",\"model\":{\"on\":{\"loaded\":\"BadName\"}}")));

99  	}

117 	@Test void templatedTopic_allows_placeholders() {
118 		valid(page("", html(",\"model\":{\"subscribes\":[{\"topic\":\"detail:{rowKey}\",\"handler\":\"app.x\"}]}")));

123 	@Test void every_closed_card_def_accepts_model() {
124 		var m = ",\"model\":{\"on\":{\"loaded\":\"app.x\"}}";
125 		valid(page("", html(m)));
126 		valid(page("", "{\"id\":\"d\",\"type\":\"datatables\",\"src\":\"/x\"" + m + "}"));
127 		valid(page("", "{\"id\":\"c\",\"type\":\"chart\",\"src\":\"/x\",\"kind\":\"line\"" + m + "}"));
128 		valid(page("", "{\"id\":\"j\",\"type\":\"job-log\",\"submit\":{\"endpoint\":\"/j\"}" + m + "}"));
129 		valid(page("", "{\"id\":\"tv\",\"type\":\"terminal-view\",\"terminalView\":{\"poll\":false}" + m + "}"));
130 		valid(page("", "{\"id\":\"t\",\"type\":\"terminal\",\"terminal\":{\"bytesUrl\":\"/b\",\"eventsUrl\":\"/e\"}" + m + "}"));
131 	}
