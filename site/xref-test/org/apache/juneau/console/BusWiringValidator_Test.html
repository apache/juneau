39  	/** A stub adopter type: {@code roles}/{@code ops} of null mean "unknown to Java". */
40  	private record Stub(String type, Set<String> roles, Set<String> ops, List<String> publishes) implements CardTypeHandler {
41  		@Override public JsonMap toFragment(CardSource source) { return new JsonMap(); }
42  		@Override public List<String> implicitTopics(JsonMap card) { return publishes; }
43  		@Override public Set<String> acceptedRoles() { return roles; }
44  		@Override public Set<String> acceptedOps() { return ops; }

47  	private static final CardTypeRegistry REGISTRY = CardTypeRegistry.standard().copy()
48  		.add(new Stub("grid", Set.of("highlight"), Set.of("reload"), List.of("grid.picked")))
49  		.add(new Stub("meter", null, null, List.of()))
50  		.build();

52  	private static List<String> problems(String contract) throws Exception {
53  		return BusWiringValidator.validate(JsonMap.ofString(contract), REGISTRY).stream().map(Object::toString).toList();
54  	}

184 	/** A page with one terminal card with input on; each argument is the wiring fragment to include (null to omit). */
185 	private static String terminalPage(String topics, String bridges, String publishes, String subscribes) {
186 		var card = Stream.of("\"id\":\"term\",\"type\":\"terminal\",\"terminal\":{\"bytesUrl\":\"/t/juneau-terminal/t1/bytes\",\"input\":true,\"inputId\":\"t1\"}", publishes, subscribes)

353 	private static String derived(String... idFrom) {
354 		var sb = new StringBuilder("\"dataSources\":[{\"id\":\"src\",\"kind\":\"source\",\"url\":\"/x\"}");
355 		for (var i = 0; i < idFrom.length; i += 2)
356 			sb.append(",{\"id\":\"").append(idFrom[i]).append("\",\"kind\":\"derived\",\"from\":\"").append(idFrom[i + 1]).append("\",\"transforms\":[]}");
