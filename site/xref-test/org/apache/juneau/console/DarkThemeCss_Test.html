32   * Guards for the shipped {@code juneau-theme-dark.css} neutral layer: its shape, that it declares only known

42  	private static final String DARK_RESOURCE = "/org/apache/juneau/console/juneau-theme-dark.css";
43  
44  	private static final Pattern COMMENT = Pattern.compile("/\\*.*?\\*/", Pattern.DOTALL);
45  
46  	private static final Pattern WRAPPER = Pattern.compile(
47  		"^\\s*@media not print\\s*\\{\\s*html:root\\[data-jc-scheme=\"dark\"\\]\\s*\\{(.*)\\}\\s*\\}\\s*$", Pattern.DOTALL);
48  
49  	/**
50  	 * Colour tokens the dark layer deliberately leaves alone: accents (owned by the theme or app dark block), the
51  	 * danger/success fills, the white and on-* inks, and derived aliases that follow a live {@code var()} to a
52  	 * token this layer does redefine or deliberately leaves.
53  	 */
54  	private static final Set<String> ACCENT_OR_INVARIANT = Set.of(
55  		"--jc-accent", "--jc-accent-wash", "--jc-link", "--jc-focus", "--jc-accent-selected",
56  		"--jc-btn-primary", "--jc-btn-primary-hover", "--jc-avatar-bg", "--jc-danger", "--jc-success", "--jc-white",
57  		"--jc-on-accent", "--jc-on-btn-primary",
58  		// Derived aliases that stay live.
59  		"--jc-btn-primary-bg", "--jc-btn-primary-bg-hover", "--jc-page-nav-accent", "--jc-nav-bg",
60  		"--jc-header-icon-text", "--jc-tone-info", "--jc-tone-success", "--jc-tone-warning", "--jc-tone-error",
61  		"--jc-chrome-button-border", "--jc-chrome-disabled-border", "--jc-chrome-control-border");

63  	private static String read(String resource) throws IOException {
64  		try (var in = DarkThemeCss_Test.class.getResourceAsStream(resource)) {
65  			assertNotNull(in, () -> "missing classpath resource: " + resource);
66  			return new String(in.readAllBytes(), UTF_8);
67  		}
68  	}
69  
70  	private static String readSibling(String relative) throws IOException {
71  		var f = Path.of(System.getProperty("basedir", "."), relative);
72  		assertTrue(Files.isRegularFile(f), () -> "missing sibling resource: " + f.toAbsolutePath().normalize());
73  		return Files.readString(f, UTF_8);
74  	}

76  	private static final String VIEWS_CSS =
77  		"../juneau-console-views/src/main/resources/org/apache/juneau/views/juneau-views.css";
78  
79  	/** The declarations of the dark block, in source order. */
80  	private static LinkedHashMap<String,String> darkDecls() throws IOException {
81  		var css = COMMENT.matcher(read(DARK_RESOURCE)).replaceAll("");
82  		var m = WRAPPER.matcher(css);
83  		assertTrue(m.matches(), () -> "dark file is not exactly one @media not print { html:root[data-jc-scheme=\"dark\"] { ... } }:\n" + css);
84  		return decls(m.group(1));
85  	}
86  
87  	private static LinkedHashMap<String,String> decls(String body) {
88  		var out = new LinkedHashMap<String,String>();
89  		for (var decl : body.split(";")) {
90  			var d = decl.trim();
91  			if (d.isEmpty())
92  				continue;
93  			var colon = d.indexOf(':');
94  			assertTrue(colon > 0, () -> "malformed declaration: '" + d + "'");
95  			// A later duplicate wins, as in the cascade.
96  			out.put(d.substring(0, colon).trim(), d.substring(colon + 1).trim());
97  		}
98  		return out;
99  	}

101 	/** The first {@code :root{}} block of the views stylesheet. */
102 	private static LinkedHashMap<String,String> viewsRoot() throws IOException {
103 		var css = COMMENT.matcher(readSibling(VIEWS_CSS)).replaceAll("");
104 		var m = Pattern.compile("(?m)^:root\\s*\\{").matcher(css);
105 		assertTrue(m.find(), "views stylesheet has no :root block");
106 		var start = m.end();
107 		return decls(css.substring(start, css.indexOf("\n}", start)));
108 	}

114 	private static Set<String> chartsFallbackKeys() throws IOException {
115 		var js = read("/org/apache/juneau/console/juneau-charts.js");
116 		var m = Pattern.compile("const TOKEN_FALLBACKS = Object\\.freeze\\(\\{(.*?)\\}\\);", Pattern.DOTALL).matcher(js);
117 		assertTrue(m.find(), "juneau-charts.js has no TOKEN_FALLBACKS");
118 		var out = new LinkedHashSet<String>();
119 		var k = Pattern.compile("\"(--jc-[a-z0-9-]+)\"\\s*:").matcher(m.group(1));
120 		while (k.find())
121 			out.add(k.group(1));
122 		return out;
123 	}

125 	private static final Pattern VAR_ONLY = Pattern.compile("^var\\((--jc-[a-z0-9-]+)\\)$");
126 	private static final Pattern VAR_WITH_FALLBACK = Pattern.compile("^var\\((--jc-[a-z0-9-]+)\\s*,\\s*(.+)\\)$");
127 	private static final Pattern COLOUR = Pattern.compile(
128 		"^(#[0-9a-fA-F]{3,8}|(?i:rgba?|hsla?)\\([^)]*\\)|transparent|linear-gradient\\(.*\\))$");
129 
130 	/** True when the value is a colour or a {@code var()} chain that ends in one. */
131 	private static boolean isColour(String value, Map<String,String> all, int depth) {
132 		if (depth > 10)
133 			return false;
134 		var v = value.trim();
135 		var m = VAR_ONLY.matcher(v);
136 		if (m.matches()) {
137 			var target = all.get(m.group(1));
138 			return target != null && isColour(target, all, depth + 1);
139 		}
140 		m = VAR_WITH_FALLBACK.matcher(v);
141 		if (m.matches()) {
142 			var target = all.get(m.group(1));
143 			return isColour(target != null ? target : m.group(2), all, depth + 1);
144 		}
145 		return COLOUR.matcher(v).matches();
146 	}

159 	@Test void a03_valuesPassGrammar() throws Exception {
160 		for (var e : darkDecls().entrySet()) {
161 			if (e.getKey().equals("color-scheme") || VAR_ONLY.matcher(e.getValue()).matches())
162 				continue;
163 			var key = e.getKey();
164 			var val = e.getValue();
165 			assertDoesNotThrow(() -> CssValueGrammar.normalizeAndValidate(val), () -> key + ": " + val);
166 		}
167 	}

169 	@Test void a04_namesKnown() throws Exception {
170 		var known = new LinkedHashSet<String>(Theme.OPEN.getTokens().keySet());
171 		known.addAll(aliases().keySet());
172 		known.addAll(viewsRoot().keySet());
173 		known.addAll(chartsFallbackKeys());
174 		for (var name : darkDecls().keySet())
175 			if (! name.equals("color-scheme"))
176 				assertTrue(known.contains(name), () -> "unknown token declared in the dark file: " + name);
177 	}

180 		var dim = Pattern.compile("\\d(px|rem|em)\\b");
181 		for (var e : darkDecls().entrySet()) {
182 			if (e.getKey().endsWith("-shadow"))
183 				continue;
184 			assertFalse(dim.matcher(e.getValue()).find(), () -> "dimension in " + e.getKey() + ": " + e.getValue());
185 		}

188 	@Test void a06_completeness() throws Exception {
189 		var all = new LinkedHashMap<String,String>(Theme.OPEN.getTokens());
190 		all.putAll(aliases());
191 		all.putAll(viewsRoot());
192 		var declared = darkDecls().keySet();
193 		var missing = new TreeSet<String>();
194 		for (var e : all.entrySet())
195 			if (isColour(e.getValue(), all, 0) && ! declared.contains(e.getKey()) && ! ACCENT_OR_INVARIANT.contains(e.getKey()))
196 				missing.add(e.getKey());
197 		assertEquals(Set.of(), missing, "colour tokens neither redefined in the dark file nor on the accent/invariant list");
198 	}

201 		var d = darkDecls();
202 		var count = 0;
203 		for (var c : List.of("green", "blue", "amber", "neutral", "red"))
