29  /**
30   * WCAG 1.4.11 gate for the "selected = hover" indicator (ribbon toggles, dialog section tabs).
31   *
32   * <p>
33   * The selected/hover state is an accent edge plus the accent wash, so the edge carries the 3:1 non-text contrast.  For
34   * every stock palette in both schemes, {@code --jc-accent} must be at least 3:1 against the rest background
35   * ({@code --jc-card-bg}) and against the washed button ({@code --jc-accent-wash} composited over {@code --jc-card-bg}).
36   * A failing palette fixes its {@code --jc-accent} (darker in light, lighter in dark), never this rule.
37   *
38   * <p>
39   * The secondary-button edge ({@code --jc-control-border}, which is {@code --jc-border-2} in light) must reach 3:1 against the card.
40   *
41   * <p>
42   * The label on the washed button is text, so {@code --jc-link} must reach 4.5:1 against the same composite.
43   */
44  class Theme_ContrastGate_Test extends TestBase {
45  
46  	private static final double MIN = 3.0;
47  	private static final double MIN_TEXT = 4.5;
48  
49  	private static final Pattern HEX = Pattern.compile("^#([0-9a-fA-F]{6})$");
50  	private static final Pattern RGBA = Pattern.compile("^rgba\\(\\s*(\\d+)\\s*,\\s*(\\d+)\\s*,\\s*(\\d+)\\s*,\\s*([0-9.]+)\\s*\\)$");
51  
52  	/** The value of one declaration in the shared dark neutral layer. */
53  	private static String darkLayer(String name) throws IOException {
54  		try (var in = Theme_ContrastGate_Test.class.getResourceAsStream("/org/apache/juneau/console/juneau-theme-dark.css")) {
55  			assertNotNull(in);
56  			var m = Pattern.compile("(?m)^\\s*" + Pattern.quote(name) + "\\s*:\\s*([^;]+);").matcher(new String(in.readAllBytes(), UTF_8));
57  			assertTrue(m.find(), () -> "dark layer has no " + name);
58  			return m.group(1).trim();
59  		}
60  	}
61  
62  	/** An opaque colour as {r, g, b} in 0..255. */
63  	private static double[] rgb(String v) {
64  		var m = HEX.matcher(v.trim());
65  		assertTrue(m.matches(), () -> "not an opaque #rrggbb colour: " + v);
66  		var h = Integer.parseInt(m.group(1), 16);
67  		return new double[] { h >> 16 & 0xff, h >> 8 & 0xff, h & 0xff };
68  	}
69  
70  	/** {@code over} (an rgba() or opaque hex) composited onto the opaque {@code base}. */
71  	private static double[] over(String over, double[] base) {
72  		var m = RGBA.matcher(over.trim());
73  		if (! m.matches())
74  			return rgb(over);
75  		var a = Double.parseDouble(m.group(4));
76  		var out = new double[3];
77  		for (var i = 0; i < 3; i++)
78  			out[i] = Double.parseDouble(m.group(i + 1)) * a + base[i] * (1 - a);
79  		return out;
80  	}
81  
82  	private static double luminance(double[] c) {
83  		var l = new double[3];
84  		for (var i = 0; i < 3; i++) {
85  			var s = c[i] / 255;
86  			l[i] = s <= 0.03928 ? s / 12.92 : Math.pow((s + 0.055) / 1.055, 2.4);
87  		}
88  		return 0.2126 * l[0] + 0.7152 * l[1] + 0.0722 * l[2];
89  	}
90  
91  	private static double ratio(double[] a, double[] b) {
92  		var la = luminance(a);
93  		var lb = luminance(b);
94  		return (Math.max(la, lb) + 0.05) / (Math.min(la, lb) + 0.05);
95  	}
96  
97  	private static void assertGate(Theme theme, boolean dark) throws IOException {
98  		var t = new LinkedHashMap<>(theme.getTokens());
99  		var card = t.get("--jc-card-bg");
100 		if (dark) {
101 			t.putAll(theme.getDarkTokens());
102 			card = darkLayer("--jc-card-bg");
103 		}
104 		var scheme = theme.getName() + (dark ? " dark" : " light");
105 		var accent = rgb(t.get("--jc-accent"));
106 		var cardRgb = rgb(card);
107 		var washed = over(t.get("--jc-accent-wash"), cardRgb);
108 		var vsCard = ratio(accent, cardRgb);
109 		var vsWash = ratio(accent, washed);
110 		assertTrue(vsCard >= MIN, () -> scheme + ": --jc-accent vs --jc-card-bg is " + String.format("%.2f", vsCard) + ":1");
111 		assertTrue(vsWash >= MIN, () -> scheme + ": --jc-accent vs the washed button is " + String.format("%.2f", vsWash) + ":1");
112 		// The label on that washed button is --jc-link (WCAG 1.4.3 text), not --jc-accent.
113 		var link = rgb(t.get("--jc-link"));
114 		var linkVsWash = ratio(link, washed);
115 		assertTrue(linkVsWash >= MIN_TEXT, () -> scheme + ": --jc-link vs the washed button is " + String.format("%.2f", linkVsWash) + ":1");
116 		// Folder tabs: the unselected label (--jc-pill-neutral-text) over the neutral tab fill is text
117 		// (WCAG 1.4.3); the selected tab's accent outline over the card face is a UI boundary (1.4.11).
118 		var tabFill = rgb(dark ? darkLayer("--jc-pill-neutral-bg") : t.get("--jc-pill-neutral-bg"));
119 		var tabLabel = rgb(dark ? darkLayer("--jc-pill-neutral-text") : t.get("--jc-pill-neutral-text"));
120 		var tabLabelVsFill = ratio(tabLabel, tabFill);
121 		assertTrue(tabLabelVsFill >= MIN_TEXT, () -> scheme + ": unselected tab label vs --jc-pill-neutral-bg is " + String.format("%.2f", tabLabelVsFill) + ":1");
122 		// Secondary buttons: their fill is the surface, so the edge alone identifies them (WCAG 1.4.11).
123 		var edge = rgb(dark ? darkLayer("--jc-control-border") : t.get("--jc-border-2"));
124 		var edgeVsCard = ratio(edge, cardRgb);
125 		assertTrue(edgeVsCard >= MIN, () -> scheme + ": --jc-control-border vs --jc-card-bg is " + String.format("%.2f", edgeVsCard) + ":1");
126 	}
