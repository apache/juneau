25  /**
26   * {@code var(--jc-name)} Theme-layer reference recognition and composition-time ({@code build()}) resolution.
27   *
28   * <p>
29   * The load-bearing security property here is that {@code var()} is <b>never</b> a {@code CssValueGrammar} value
30   * shape: it is recognized one layer above the grammar and resolved to a concrete literal at {@code build()} time,
31   * so {@code getTokens()} never carries the substring {@code var(} (see {@code d02}). Resolution is fail-closed:
32   * an unknown reference, a cycle, or a chain past the depth cap is a loud {@code build()} failure with its own
33   * distinct message, never a silent fall-through (see the {@code c} group).
34   */
35  class Theme_VarReferences_Test extends TestBase {

41  	@Test void a01_ownMapReference_resolvesToTheReferencedLiteral() {
42  		// The acceptance case (mirrors ReleaseManagerTheme): both tokens on the SAME builder, so this exercises
43  		// own-map resolution, not the Theme.OPEN fallback.
44  		var theme = Theme.create("derived")
45  			.token("--jc-danger", "#c23934")
46  			.token("--jc-tag-red-text", "var(--jc-danger)")
47  			.build();
48  		assertBean(theme.getTokens(), "--jc-tag-red-text,--jc-danger", "#c23934,#c23934");
49  	}

51  	@Test void a02_forwardReference_resolvesRegardlessOfDeclarationOrder() {
52  		// Reference declared BEFORE its target - only resolvable at build(), the whole point of deferring.
53  		var theme = Theme.create("fwd")
54  			.token("--jc-a", "var(--jc-b)")
55  			.token("--jc-b", "#abcabc")
56  			.build();
57  		assertEquals("#abcabc", theme.getTokens().get("--jc-a"));
58  	}

62  		var theme = Theme.create("fb").token("--jc-a", "var(--jc-danger)").build();
63  		assertEquals("#c23934", theme.getTokens().get("--jc-a"));
64  	}

69  			.token("--jc-b", "var(--jc-c)")
70  			.token("--jc-c", "#0a0b0c")
71  			.build();

75  	@Test void a05_recognitionVariants_shareTheSameNormalizationBelt() {
76  		// Comment-stripping, case-insensitivity, and interior whitespace are handled by the SAME belt every other
77  		// value runs through - there is no second, reference-specific normalization pass.
78  		var theme = Theme.create("rec")
79  			.token("--jc-x", "#123456")
80  			.token("--jc-comment", "var/**/(--jc-x)")
81  			.token("--jc-inner-comment", "var(--jc-/**/x)")
82  			.token("--jc-upper", "VAR(--jc-x)")
83  			.token("--jc-space", "var( --jc-x )")
84  			.build();
85  		assertBean(theme.getTokens(), "--jc-comment,--jc-inner-comment,--jc-upper,--jc-space", "#123456,#123456,#123456,#123456");

98  		// Resolution runs on a COPY: a failed build() must not half-resolve the builder's own map.
99  		var b = Theme.create("atom").token("--jc-a", "var(--jc-missing)");
100 		assertThrows(IllegalArgumentException.class, b::build);

102 		var theme = b.token("--jc-missing", "#123123").build();
103 		assertEquals("#123123", theme.getTokens().get("--jc-a"));

110 	@Test void c01_unknownReference_isABuildFailure_namingBothTokens() {
111 		var b = Theme.create("u").token("--jc-a", "var(--jc-missing)");
112 		var ex = assertThrows(IllegalArgumentException.class, b::build);
113 		assertTrue(ex.getMessage().contains("--jc-a"), () -> "message must name the defining token: " + ex.getMessage());
114 		assertTrue(ex.getMessage().contains("--jc-missing"), () -> "message must name the missing target: " + ex.getMessage());
115 		assertTrue(ex.getMessage().contains("unknown"), () -> "message: " + ex.getMessage());
116 		// Must NOT reuse the pre-existing "not one of the allowed CSS value shapes" message.
117 		assertFalse(ex.getMessage().contains("allowed CSS value shapes"), () -> "message: " + ex.getMessage());
118 	}

120 	@Test void c02_shadowedBrokenOverride_failsInsteadOfSilentlyFallingBackToThemeOpen() {
121 		// --jc-danger EXISTS on Theme.OPEN (#c23934). Shadowing it here with a broken reference must FAIL the build,
122 		// NOT silently resolve through to Theme.OPEN's value - the exact bug the reference feature exists to catch.
123 		var b = Theme.create("shadow").token("--jc-danger", "var(--jc-nonexistent)");
124 		var ex = assertThrows(IllegalArgumentException.class, b::build);
125 		assertTrue(ex.getMessage().contains("--jc-nonexistent"), () -> "message: " + ex.getMessage());
126 		assertTrue(ex.getMessage().contains("unknown"), () -> "message: " + ex.getMessage());
127 	}

131 		var ex = assertThrows(IllegalArgumentException.class, b::build);
132 		assertTrue(ex.getMessage().contains("cyclic"), () -> "message: " + ex.getMessage());

145 	@Test void c05_depthCapBreach_isADistinctFailure_fromCycleAndUnknown() {
146 		// A long ACYCLIC chain (all names distinct, terminal token defined) must still be bounded by the hop cap -
147 		// so this fails for depth, not cycle (no repeat) and not unknown (--jc-c10 exists).
148 		var b = Theme.create("deep");
149 		for (var i = 0; i < 10; i++)
150 			b.token("--jc-c" + i, "var(--jc-c" + (i + 1) + ")");
151 		b.token("--jc-c10", "#000000");
152 		var ex = assertThrows(IllegalArgumentException.class, b::build);
153 		assertTrue(ex.getMessage().contains("maximum reference depth"), () -> "message: " + ex.getMessage());
154 		assertFalse(ex.getMessage().contains("cyclic"), () -> "depth-cap message must be distinct from the cycle message: " + ex.getMessage());
155 		assertFalse(ex.getMessage().contains("unknown"), () -> "depth-cap message must be distinct from the unknown-name message: " + ex.getMessage());
156 	}

162 	@Test void d01_resolvedLiteralThatTheGrammarRejects_wouldFailTheReValidation() {
163 		// The resolved literal (never an intermediate var()) is re-validated by the unchanged grammar. Through the
164 		// public API every literal is already grammar-validated at token() time, so this path cannot be driven to a
165 		// failure here - it is exercised (every resolved value passes back through normalizeAndValidate) rather than
166 		// tripped. This test pins that a reference to a plain literal round-trips byte-for-byte through that
167 		// re-validation.
168 		var theme = Theme.create("dd")
169 			.token("--jc-a", "linear-gradient(135deg, rgb(21,137,238), rgba(0,0,0,0.2))")
170 			.token("--jc-b", "var(--jc-a)")
171 			.build();
172 		assertEquals("linear-gradient(135deg, rgb(21,137,238), rgba(0,0,0,0.2))", theme.getTokens().get("--jc-b"));
173 	}

175 	@Test void d02_afterBuild_noTokenValueContainsTheVarSubstring() {
176 		var theme = Theme.create("inv")
177 			.token("--jc-danger", "#c23934")
178 			.token("--jc-tag-red-text", "var(--jc-danger)")   // own-map reference
179 			.token("--jc-a", "var(--jc-accent)")               // Theme.OPEN fallback reference
180 			.token("--jc-b", "var(--jc-danger)")               // chained onto another own token
181 			.token("--jc-lit", "#1589EE")                      // plain literal, untouched
182 			.build();
183 		for (var v : theme.getTokens().values())
184 			assertFalse(v.contains("var("), () -> "resolved token value still contains 'var(': " + v);
185 	}

188 	// e) Seeding a builder from another theme's tokens: the shadowing branch a builder's OWN tokens now take
189 	//-----------------------------------------------------------------------------------------------------------------

191 	private static Theme.Builder seedFrom(String name, Theme seed) {
192 		var b = Theme.create(name);
193 		for (var e : seed.getTokens().entrySet())
194 			b.token(e.getKey(), e.getValue());
195 		return b;
196 	}

198 	@Test void e01_seededBuilder_overriddenReferencedToken_referenceResolvesToTheOverride() {
199 		// The seed's --jc-danger is shadowed by the caller's override, so a reference to it must resolve to the
200 		// override - exactly as it would for a plain create("x") that defines --jc-danger itself.
201 		var theme = seedFrom("x", Theme.OPEN)
202 			.token("--jc-danger", "#900000")
203 			.token("--jc-tag-red-text", "var(--jc-danger)")
204 			.build();
205 		assertEquals("#900000", theme.getTokens().get("--jc-tag-red-text"));
206 	}
207 
208 	@Test void e02_seededBuilder_unoverriddenReferencedToken_resolvesAgainstTheSeededCopy_notTheOpenFallback() {
209 		// The seed here is deliberately NOT Theme.OPEN and gives --jc-danger a value OPEN does not have. If
210 		// resolution incorrectly fell through to the Theme.OPEN fallback (instead of using this builder's OWN,
211 		// seeded copy of --jc-danger), this would resolve to OPEN's "#c23934" rather than the seed's "#111111" -
212 		// so this pins the case that looks identical to the fallback case but is reached by a different code path.
213 		var seed = Theme.create("custom-seed").token("--jc-danger", "#111111").build();
214 		var theme = seedFrom("x", seed)
215 			.token("--jc-tag-red-text", "var(--jc-danger)")
216 			.build();
217 		assertEquals("#111111", theme.getTokens().get("--jc-tag-red-text"));
218 	}
219 
220 	@Test void e03_seededBuilder_seedAuthoredReference_isASeededLiteral_notRelinkedByAnOverride() {
221 		// What seeding copies is the seed's VALUES, not its reference relationships: the seed's own build()
222 		// already flattened var(--jc-danger) to "#111111", so overriding --jc-danger on the seeded builder does
223 		// not re-flow into it. Only a reference the caller authors on the seeded builder resolves live - which
224 		// --jc-b does, to the override.
225 		var seed = Theme.create("custom-seed")
226 			.token("--jc-danger", "#111111")
227 			.token("--jc-tag-red-text", "var(--jc-danger)")
228 			.build();
229 		var theme = seedFrom("x", seed)
230 			.token("--jc-danger", "#900000")
231 			.token("--jc-b", "var(--jc-danger)")
232 			.build();
233 		assertBean(theme.getTokens(), "--jc-tag-red-text,--jc-b", "#111111,#900000");
234 	}
