20  import static org.junit.jupiter.api.Assertions.*;

29  /**
30   * Every pill palette text/background pair (status and categorical) must reach WCAG 1.4.3 text contrast, in the light
31   * and the dark scheme of every stock theme.
32   */
33  class Theme_PillPalette_ContrastGate_Test extends TestBase {
34  
35  	private static final double MIN_TEXT = 4.5;
36  
37  	private static final Pattern HEX = Pattern.compile("^#([0-9a-fA-F]{6})$");
38  
39  	private static final List<String> NAMES = List.of(
40  		"green", "blue", "amber", "neutral", "red",
41  		"cat-1", "cat-2", "cat-3", "cat-4", "cat-5", "cat-6", "cat-7", "cat-8", "cat-9", "cat-10");
42  
43  	/** The value of one declaration in the shared dark neutral layer. */
44  	private static String darkLayer(String name) throws IOException {
45  		try (var in = Theme_PillPalette_ContrastGate_Test.class.getResourceAsStream("/org/apache/juneau/console/juneau-theme-dark.css")) {
46  			assertNotNull(in);
47  			var m = Pattern.compile("(?m)^\\s*" + Pattern.quote(name) + "\\s*:\\s*([^;]+);").matcher(new String(in.readAllBytes(), UTF_8));
48  			assertTrue(m.find(), () -> "dark layer has no " + name);
49  			return m.group(1).trim();
50  		}
51  	}
52  
53  	/** An opaque colour as {r, g, b} in 0..255. */
54  	private static double[] rgb(String v) {
55  		var m = HEX.matcher(v.trim());
56  		assertTrue(m.matches(), () -> "not an opaque #rrggbb colour: " + v);
57  		var h = Integer.parseInt(m.group(1), 16);
58  		return new double[] { h >> 16 & 0xff, h >> 8 & 0xff, h & 0xff };
59  	}
60  
61  	private static double luminance(double[] c) {
62  		var l = new double[3];
63  		for (var i = 0; i < 3; i++) {
64  			var s = c[i] / 255;
65  			l[i] = s <= 0.03928 ? s / 12.92 : Math.pow((s + 0.055) / 1.055, 2.4);
66  		}
67  		return 0.2126 * l[0] + 0.7152 * l[1] + 0.0722 * l[2];
68  	}
69  
70  	private static double ratio(double[] a, double[] b) {
71  		var la = luminance(a);
72  		var lb = luminance(b);
73  		return (Math.max(la, lb) + 0.05) / (Math.min(la, lb) + 0.05);
74  	}
75  
76  	private static void assertPairs(Theme theme, boolean dark) throws IOException {
77  		var scheme = theme.getName() + (dark ? " dark" : " light");
78  		for (var n : NAMES) {
79  			var bg = dark ? darkLayer("--jc-pill-" + n + "-bg") : theme.getTokens().get("--jc-pill-" + n + "-bg");
80  			var text = dark ? darkLayer("--jc-pill-" + n + "-text") : theme.getTokens().get("--jc-pill-" + n + "-text");
81  			assertNotNull(bg, () -> scheme + ": no --jc-pill-" + n + "-bg");
82  			assertNotNull(text, () -> scheme + ": no --jc-pill-" + n + "-text");
83  			var r = ratio(rgb(text), rgb(bg));
84  			assertTrue(r >= MIN_TEXT, () -> scheme + ": --jc-pill-" + n + "-text vs -bg is " + String.format("%.2f", r) + ":1");
85  		}
86  	}
87  
88  	@Test void a01_open() throws Exception { assertPairs(Theme.OPEN, false); assertPairs(Theme.OPEN, true); }
89  	@Test void a02_gray() throws Exception { assertPairs(Theme.GRAY, false); assertPairs(Theme.GRAY, true); }
90  	@Test void a03_lightBrown() throws Exception { assertPairs(Theme.LIGHT_BROWN, false); assertPairs(Theme.LIGHT_BROWN, true); }
91  	@Test void a04_lightRed() throws Exception { assertPairs(Theme.LIGHT_RED, false); assertPairs(Theme.LIGHT_RED, true); }
92  	@Test void a05_red() throws Exception { assertPairs(Theme.RED, false); assertPairs(Theme.RED, true); }
93  }
