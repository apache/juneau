17  package org.apache.juneau.console;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import java.util.*;
22  import java.util.regex.*;
23  
24  /**
25   * Test-only golden-file scanner for {@code chrome.css}.
26   *
27   * <p>
28   * Not shipped &mdash; this is a build-time regression check on the static classpath resource, not a runtime
29   * feature. Fails a CSS source if:
30   * <ul class='spaced-list'>
31   * 	<li>{@code --jc-logo} appears anywhere (the logo is a hard-coded static rule, never a token);
32   * 	<li>{@code var(--jc-} appears nested inside any function call other than the {@code var(} call itself (e.g.
33   * 		nested in {@code linear-gradient(...)}/{@code rgb(...)}/{@code url(...)});
34   * 	<li>a {@code var(--jc-*)} is the value of a url-capable property, UNLESS the property is
35   * 		{@code background-image} and the token is one of the two whitelisted gradient tokens
36   * 		({@code --jc-page-bg}, {@code --jc-avatar-bg}).
37   * </ul>
38   *
39   * <p>
40   * Does NOT fail on the static {@code url("data:image/svg+xml;...")} literal in {@code .jc-logo} &mdash; there is no
41   * {@code var(--jc-} there.
42   */
43  @SuppressWarnings({
44  	"java:S8786" // Scanner regex is intentional; tightening would change match semantics.
45  })
46  final class ChromeCssScanner {
47  
48  	private ChromeCssScanner() {}
49  
50  	/** The url-capable CSS properties denylist(extended list). */
51  	private static final Set<String> URL_CAPABLE_PROPERTIES = Set.of(
52  		"background", "background-image", "background-size", "border", "cursor", "list-style", "content",
53  		"border-image", "border-image-source", "list-style-image", "mask-image", "filter", "clip-path");
54  
55  	/** The only (property, token) pairs allowed to put a var(--jc-*) on a url-capable property. */
56  	private static final Set<String> GRADIENT_TOKEN_ALLOWLIST = Set.of("--jc-page-bg", "--jc-avatar-bg");
57  
58  	private static final Pattern COMMENT = Pattern.compile("/\\*.*?\\*/", Pattern.DOTALL);
59  	private static final Pattern WHOLE_VALUE_VAR_JC =
60  		Pattern.compile("([a-zA-Z-]++)\\s*+:\\s*+var\\((--jc-[a-z0-9-]++)\\)\\s*+;");
61  
62  	/**
63  	 * Scans the specified CSS source and returns a list of human-readable violation descriptions (empty if the CSS
64  	 * is clean).
65  	 *
66  	 * @param css The CSS source to scan.
67  	 * @return A list of violations, empty if none.
68  	 */
69  	static List<String> scan(String css) {
70  		var violations = new ArrayList<String>();
71  		var stripped = COMMENT.matcher(css).replaceAll("");
72  
73  		if (stripped.contains("--jc-logo"))
74  			violations.add("'--jc-logo' custom property must not exist - the logo is a hard-coded static rule.");
75  
76  		scanNestedVarJc(stripped, violations);
77  		scanUrlCapableSinks(stripped, violations);
78  
79  		return violations;
80  	}
81  
82  	/** Fails if {@code var(--jc-} is found nested inside any function call other than the {@code var(} call itself. */
83  	private static void scanNestedVarJc(String css, List<String> violations) {
84  		var i = 0;
85  		while (true) {
86  			var idx = css.indexOf('(', i);
87  			if (idx < 0)
88  				break;
89  			var j = idx - 1;
90  			while (j >= 0 && (Character.isLetterOrDigit(css.charAt(j)) || css.charAt(j) == '-'))
91  				j--;
92  			var name = css.substring(j + 1, idx).toLowerCase(Locale.ROOT);
93  
94  			var depth = 1;
95  			var k = idx + 1;
96  			while (k < css.length() && depth > 0) {
97  				var c = css.charAt(k);
98  				if (c == '(')
99  					depth++;
100 				else if (c == ')')
101 					depth--;
102 				k++;
103 			}
104 			var inner = css.substring(idx + 1, Math.max(idx + 1, k - 1));
105 
106 			if (neq(name, "var") && inner.contains("var(--jc-"))
107 				violations.add("'var(--jc-' nested inside '" + name + "(...)': " + name + "(" + inner + ")");
108 
109 			i = idx + 1;
110 		}
111 	}
112 
113 	/** Fails if a var(--jc-*) is the WHOLE value of a url-capable property, unless it's an allowlisted gradient token on background-image. */
114 	private static void scanUrlCapableSinks(String css, List<String> violations) {
115 		var m = WHOLE_VALUE_VAR_JC.matcher(css);
116 		while (m.find()) {
117 			var property = m.group(1).toLowerCase(Locale.ROOT);
118 			var token = m.group(2);
119 			if (! URL_CAPABLE_PROPERTIES.contains(property))
120 				continue;
121 			var allowed = eq(property, "background-image") && GRADIENT_TOKEN_ALLOWLIST.contains(token);
122 			if (! allowed)
123 				violations.add("'" + token + "' sinks into url-capable property '" + property + "'.");
124 		}
125 	}
126 }
