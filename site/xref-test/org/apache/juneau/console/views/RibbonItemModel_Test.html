17  package org.apache.juneau.console.views;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.*;
24  import org.apache.juneau.console.*;
25  import org.junit.jupiter.api.*;
26  
27  /** {@link RibbonItem} as a {@link Handled} element: action items, model blocks and required ids. */
28  class RibbonItemModel_Test extends TestBase {
29  
30  	private static TableSpec table(RibbonItem... items) {
31  		return TableSpec.create("t1").dataUrl("/d").columns(Column.create("id")).ribbon(items);
32  	}
33  
34  	@Test void a01_actionFactoryRequiresId() {
35  		assertThrows(IllegalArgumentException.class, () -> RibbonItem.action(" "));
36  		assertThrows(IllegalArgumentException.class, () -> RibbonItem.action((String)null));
37  	}
38  
39  	@Test void a02_actionShape() {
40  		var m = RibbonItem.action("reload").icon("refresh-cw").action("click", null, "juneau.refresh").toMap();
41  		assertEquals("action", m.get("type"));
42  		assertEquals("reload", m.get("id"));
43  		assertEquals("refresh-cw", m.get("icon"));
44  		assertTrue(m.get("model").toString().contains("juneau.refresh"), m.toString());
45  	}
46  
47  	@Test void a03_stockDefaults() {
48  		assertEquals("refresh", RibbonItem.refresh().toMap().get("id"));
49  		assertEquals("collapseAll", RibbonItem.collapseAll().toMap().get("id"));
50  		assertEquals("pausePolling", RibbonItem.pausePolling().toMap().get("id"));
51  		assertEquals("resetDefaults", RibbonItem.resetDefaults().toMap().get("id"));
52  		assertEquals("export", RibbonItem.export("copy").toMap().get("id"));
53  		assertEquals("add", RibbonItem.dialog("add").toMap().get("id"));
54  		assertEquals("opt", RibbonItem.option("opt").column("c").value("v").toMap().get("id"));
55  	}
56  
57  	@Test void a04_secondStockOfSameKindNeedsExplicitId() {
58  		var r1 = RibbonItem.refresh();
59  		var r2 = RibbonItem.refresh();
60  		var e = assertThrows(IllegalArgumentException.class, () -> table(r1, r2));
61  		assertTrue(e.getMessage().contains("t1"), e.getMessage());
62  		table(RibbonItem.refresh(), RibbonItem.refresh().id("refresh2"));
63  	}
64  
65  	@Test void a05_publishRequiresExplicitId() {
66  		var noId = RibbonItem.publish("T", "app.x", null);
67  		assertThrows(IllegalArgumentException.class, () -> table(noId));
68  		table(RibbonItem.publish("T", "app.x", null).id("p"));
69  	}
70  
71  	@Test void a06_dividerNeedsNoId() {
72  		table(RibbonItem.divider(), RibbonItem.divider(), RibbonItem.refresh());
73  		assertFalse(RibbonItem.divider().toMap().containsKey("id"));
74  	}
75  
76  	@Test void a07_duplicateIdsThrow() {
77  		var a1 = RibbonItem.action("a");
78  		var a2 = RibbonItem.action("a");
79  		assertThrows(IllegalArgumentException.class, () -> table(a1, a2));
80  	}
81  
82  	@Test void a08_modelKeyAllowedOnEveryType() {
83  		for (var r : List.of(RibbonItem.refresh(), RibbonItem.collapseAll(), RibbonItem.export("copy"), RibbonItem.divider(),
84  				RibbonItem.dialog("d"), RibbonItem.action("a"), RibbonItem.publish("T", "app.x", null).id("p")))
85  			assertDoesNotThrow(() -> r.on("activated", "a.b").toMap(), r.toMap().toString());
86  	}
87  
88  	@Test void a09_onlyActivatedEvent() {
89  		var sut1 = RibbonItem.refresh();
90  		assertThrows(IllegalArgumentException.class, () -> sut1.on("loaded", "a.b"));
91  	}
92  
93  	@Test void a10_inlineSurfacesThroughTable() {
94  		assertTrue(table(RibbonItem.action("a").on("activated", HandlerRef.inline("return 1"))).hasInlineHandlers());
95  	}
96  }
