365 	/**
366 	 * The surface page under {@code htmlAttrs}, with an optional inline style block and an optional chart card.  The
367 	 * DataTables vendor stylesheet sits where the toolkit emits it: after the theme, before the views stylesheet.
368 	 */
369 	private static String surfacePage(String htmlAttrs, String style, boolean chart) throws Exception {
370 		var head = "<link rel=\"stylesheet\" href=\"" + CONFIG_CSS + "\">\n<link rel=\"stylesheet\" href=\"" + CALENDAR_CSS + "\">\n" + style;
371 		var before = chart ? "<script src=\"" + ECHARTS + "\"></script>\n" + CHART_PROBE : "";
372 		var after = viewsPack() + (chart ? "<script src=\"" + CHARTS + "\"></script>\n" : "") + OVERLAYS;
373 		var views = "<link rel=\"stylesheet\" href=\"" + VIEWS_CSS + "\">";
374 		var page = contractPage(contract(chart), KIT, head + before, after, htmlAttrs, true);
375 		assertTrue(page.contains(views), "the views stylesheet is linked");
376 		return page.replace(views, "<link rel=\"stylesheet\" href=\"" + DT_CSS + "\">\n" + views);
377 	}

462 	/**
463 	 * Every surface in {@code caseName} rendered, painted a colour, and met {@code min}; an allowlisted surface instead
464 	 * measured exactly its recorded ratio.
465 	 */
466 	private static void assertContrast(String caseName, String[][] surfaces, double min, Map<String,Double> allowed) {
467 		var bad = new ArrayList<String>();
468 		for (var s : surfaces) {
469 			var errors = errors(caseName, s[0]);
470 			if (! errors.isEmpty()) {
471 				bad.add(s[0] + " (" + s[2] + "): " + errors);
472 				continue;
473 			}
474 			var w = worst(caseName, s[0]);
475 			if (w == null) {
476 				bad.add(s[0] + " (" + s[2] + "): nothing rendered");
477 				continue;
478 			}
479 			var r = (Number)w.get("ratio");
480 			if (r == null) {
481 				bad.add(s[0] + " (" + s[2] + "): no " + s[1] + " colour painted: " + w);
482 				continue;
483 			}
484 			var a = allowed.get(s[0]);
485 			if (a != null) {
486 				if (Math.abs(r.doubleValue() - a) > 0.005)
487 					bad.add(s[0] + " (" + s[2] + "): allowlisted at " + a + " but measured " + r + ": " + w);
488 			} else if (r.doubleValue() < min) {
489 				bad.add(s[0] + " (" + s[2] + "): " + r + " < " + min + ": " + w);
490 			}
491 		}
492 		assertEquals(List.of(), bad, caseName);
493 	}

495 	/** Prints each surface's measurement and returns the lines; never fails on a ratio. */
496 	private static List<String> reportOnly(String caseName, String[][] surfaces) {
497 		var lines = new ArrayList<String>();
498 		for (var s : surfaces) {
499 			var w = worst(caseName, s[0]);
500 			var errors = errors(caseName, s[0]);
501 			lines.add(caseName + " " + s[0] + " (" + s[2] + "): " + (errors.isEmpty() ? w == null ? "nothing rendered" : w.get("fg") + " on " + w.get("bg") + " = " + w.get("ratio") : "errors " + errors));
502 		}
503 		lines.forEach(System.out::println);
504 		return lines;
505 	}

699 	/** WCAG 2 contrast over colours parsed from {@code #rgb}, {@code #rrggbb} or {@code rgb[a](...)}. */
700 	private static final class Wcag {
701 
702 		/** {r, g, b, a} with channels 0-255 and alpha 0-1, or null. */
703 		static double[] parse(String s) {
704 			if (s == null)
705 				return null;
706 			s = s.trim();
707 			if (s.startsWith("#")) {
708 				var h = s.substring(1);
709 				if (h.length() == 3)
710 					h = "" + h.charAt(0) + h.charAt(0) + h.charAt(1) + h.charAt(1) + h.charAt(2) + h.charAt(2);
711 				if (h.length() != 6)
712 					return null;
713 				var v = Integer.parseInt(h, 16);
714 				return new double[] {v >> 16 & 0xff, v >> 8 & 0xff, v & 0xff, 1};
715 			}
716 			var m = java.util.regex.Pattern.compile("rgba?\\(\\s*([\\d.]+)[,\\s]+([\\d.]+)[,\\s]+([\\d.]+)(?:\\s*[,/]\\s*([\\d.]+))?\\s*\\)").matcher(s);
717 			if (! m.find())
718 				return null;
719 			return new double[] {Double.parseDouble(m.group(1)), Double.parseDouble(m.group(2)), Double.parseDouble(m.group(3)), m.group(4) == null ? 1 : Double.parseDouble(m.group(4))};
720 		}
721 
722 		/** {@code top} composited over the opaque {@code under}. */
723 		static double[] over(double[] top, double[] under) {
724 			var a = top[3];
725 			return new double[] {top[0] * a + under[0] * (1 - a), top[1] * a + under[1] * (1 - a), top[2] * a + under[2] * (1 - a), 1};
726 		}
727 
728 		static double ratio(double[] a, double[] b) {
729 			var x = lum(a);
730 			var y = lum(b);
731 			return Math.round((Math.max(x, y) + 0.05) / (Math.min(x, y) + 0.05) * 100) / 100.0;
732 		}
733 
734 		private static double lum(double[] c) {
735 			return 0.2126 * channel(c[0]) + 0.7152 * channel(c[1]) + 0.0722 * channel(c[2]);
736 		}
737 
738 		private static double channel(double v) {
739 			v /= 255;
740 			return v <= 0.03928 ? v / 12.92 : Math.pow((v + 0.055) / 1.055, 2.4);
741 		}
742 	}
