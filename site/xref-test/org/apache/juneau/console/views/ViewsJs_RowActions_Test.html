226 		var body = viewsJs();

228 		var safeMethodIdx = fn.indexOf("isSafeMethod(action.method)");
229 		var tokenIdx = fn.indexOf("isBlankToken(token)");
230 		var noEndpointIdx = fn.indexOf("reason: \"no-endpoint\"");
231 		var emptySubIdx = fn.indexOf("reason: \"empty-substitution\"");

267 	/** Extracts the bare regex pattern text (no leading/trailing `/`, no flags) immediately following {@code marker}. */
268 	private static String extractRegexPattern(String body, String marker) {
269 		var markerIdx = body.indexOf(marker);
270 		assertTrue(markerIdx >= 0, () -> marker + " not found:\n" + body);
271 		var slashStart = body.indexOf('/', markerIdx);
272 		var slashEnd = body.indexOf('/', slashStart + 1);
273 		assertTrue(slashStart >= 0 && slashEnd > slashStart, () -> "regex literal not found after " + marker);
274 		return body.substring(slashStart + 1, slashEnd);
275 	}

302 		if (!nodeAvailable())

306 			return;
307 		var viewsFile = Files.createTempFile("juneau-views-", ".js");

309 		try {
310 			Files.writeString(viewsFile, viewsJs(), UTF_8);

312 			report = Json.to(runNode(harness, viewsFile, rendersFile), Map.class);
313 			reportNoRendersJs = Json.to(runNode(harness, viewsFile, null), Map.class);
314 		} finally {
315 			Files.deleteIfExists(viewsFile);
316 			Files.deleteIfExists(rendersFile);
317 		}
318 	}

320 	private static boolean nodeAvailable() {
321 		try {
322 			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
323 			if (!p.waitFor(5, TimeUnit.SECONDS)) {
324 				p.destroyForcibly();
325 				return false;
326 			}
327 			return p.exitValue() == 0;
328 		} catch (Exception e) {
329 			return false;
330 		}
331 	}

349 	private static String runNode(Path harness, Path viewsJs, Path rendersJs) throws Exception {
350 		var stdout = Files.createTempFile("row-action-endpoint-stdout-", ".json");
351 		var stderr = Files.createTempFile("row-action-endpoint-stderr-", ".txt");
352 		try {
353 			var args = new ArrayList<String>(List.of("node", harness.toString(), viewsJs.toString()));
354 			if (rendersJs != null) args.add(rendersJs.toString());
355 			var pb = new ProcessBuilder(args)
356 				.redirectOutput(stdout.toFile())
357 				.redirectError(stderr.toFile());
358 			var p = pb.start();
359 			if (!p.waitFor(30, TimeUnit.SECONDS)) {
360 				p.destroyForcibly();
361 				fail("row-action-endpoint.cjs did not finish within 30s; stderr:\n" + quietRead(stderr));
362 			}
363 			if (p.exitValue() != 0)
364 				fail("row-action-endpoint.cjs exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr)
365 					+ "\nstdout:\n" + quietRead(stdout));
366 			return Files.readString(stdout, UTF_8);
367 		} finally {
368 			Files.deleteIfExists(stdout);
369 			Files.deleteIfExists(stderr);
370 		}
371 	}

373 	private static String quietRead(Path p) {
374 		try { return Files.readString(p, UTF_8); }
375 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
376 	}

465 	@Test void b10_dotDotRowValueRefuses() {
466 		// The headline finding: encodeURIComponent('..') === '..' (`.` is RFC 3986 unreserved, never escaped), so
467 		// a row value of `..` would otherwise turn `servlet:/incidents/{id}/ack` into
468 		// `servlet:/incidents/../ack`, which browser URL resolution normalizes to `/ack` - a DIFFERENT, undeclared
469 		// mutating endpoint - and submitRowAction would fire it as an authenticated, CSRF-headered POST.
470 		assertRefusal((Map<?,?>) report().get("dotdot_rowValue"), "unsafe-endpoint");
471 	}
