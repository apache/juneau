30  /**
31   * Pins the exact set of {@code <symbol>} stem ids shipped in {@code juneau-symbols.svg}.
32   *
33   * <p>
34   * A third-party host page can override one of these glyphs by looking the stem id up by name and swapping in a
35   * replacement drawing for it. That lookup is name-based only, so if a stem id here is renamed - or quietly added
36   * or removed - without a matching change on the consuming side, the override stops finding its target. The
37   * result is not an error: the glyph simply renders as nothing, with no signal anywhere in this module. Guarding
38   * against that silent failure is the entire point of this test, so the check below is set equality rather than a
39   * one-directional containment check - a containment check would happily let a new id slip in unnoticed.
40   */
41  class SymbolSprite_StemIds_Test extends TestBase {
42  
43  	/**
44  	 * The complete set of stem ids as of the time this guard was written. This set is intentionally frozen:
45  	 * any difference from what is actually shipped must be a deliberate, reviewed decision, not an accident of
46  	 * an artwork or naming change.
47  	 */
48  	private static final Set<String> EXPECTED_IDS = Set.of(
49  		"juneau-sym-cancel", "juneau-sym-check", "juneau-sym-chevrondown", "juneau-sym-chevronleft",
50  		"juneau-sym-chevronright", "juneau-sym-chevronup", "juneau-sym-close", "juneau-sym-collapse_all",
51  		"juneau-sym-columns", "juneau-sym-copy", "juneau-sym-csv", "juneau-sym-download", "juneau-sym-edit",
52  		"juneau-sym-filter", "juneau-sym-first_page", "juneau-sym-forceStop", "juneau-sym-last_page",
53  		"juneau-sym-link", "juneau-sym-more", "juneau-sym-new", "juneau-sym-openPr", "juneau-sym-pause", "juneau-sym-pdf", "juneau-sym-print",
54  		"juneau-sym-push", "juneau-sym-refresh", "juneau-sym-search", "juneau-sym-settings", "juneau-sym-sort",
55  		"juneau-sym-spreadsheet", "juneau-sym-stop", "juneau-sym-toggle-deleted",
56  		"juneau-sym-toggle_column_search"
57  	);
58  
59  	private static final Pattern SYMBOL_ID_PATTERN = Pattern.compile("<symbol\\s+id=\"([^\"]+)\"");
60  
61  	private static Set<String> actualIds() throws IOException {
62  		String svg;
63  		try (var in = ViewsMixin.class.getResourceAsStream(ViewsMixin.SYMBOLS_SVG_RESOURCE)) {
64  			assertNotNull(in, () -> "missing classpath resource: " + ViewsMixin.SYMBOLS_SVG_RESOURCE);
65  			svg = new String(in.readAllBytes(), UTF_8);
66  		}
67  		var ids = new LinkedHashSet<String>();
68  		var m = SYMBOL_ID_PATTERN.matcher(svg);
69  		while (m.find())
70  			ids.add(m.group(1));
71  		return ids;
72  	}
73  
74  	@SuppressWarnings({
75  		"java:S4030" // removed/added are read only by the failure-message lambda, which Sonar does not count as a use.
76  	})
77  	@Test void a01_stemIdSetIsPinned() throws Exception {
78  		var actual = actualIds();
79  		var removed = new TreeSet<>(EXPECTED_IDS);
80  		removed.removeAll(actual);
81  		var added = new TreeSet<>(actual);
82  		added.removeAll(EXPECTED_IDS);
83  
84  		assertList(() -> "juneau-symbols.svg <symbol> stem ids no longer match the pinned set - removed: " + removed
85  			+ ", added: " + added + ". A stem id is a lookup key a host page can override a glyph by; changing,"
86  			+ " adding, or removing one here (even one renamed to something equivalent-looking) breaks that"
87  			+ " lookup silently, with no error on either side. If this change is intentional, update"
88  			+ " EXPECTED_IDS to match.", new TreeSet<>(actual), new TreeSet<>(EXPECTED_IDS).toArray());
89  	}
90  }
