30  /**
31   * Pins the exact set of {@code <symbol>} stem ids shipped in {@code juneau-symbols.svg}.
32   *
33   * <p>
34   * A third-party host page can override one of these glyphs by looking the stem id up by name and swapping in a
35   * replacement drawing for it. That lookup is name-based only, so if a stem id here is renamed - or quietly added
36   * or removed - without a matching change on the consuming side, the override stops finding its target. The
37   * result is not an error: the glyph simply renders as nothing, with no signal anywhere in this module. Guarding
38   * against that silent failure is the entire point of this test, so the check below is set equality rather than a
39   * one-directional containment check - a containment check would happily let a new id slip in unnoticed.
40   */
41  class SymbolSprite_StemIds_Test extends TestBase {
42  
43  	/**
44  	 * The complete set of stem ids as of the time this guard was written. This set is intentionally frozen:
45  	 * any difference from what is actually shipped must be a deliberate, reviewed decision, not an accident of
46  	 * an artwork or naming change.
47  	 */

61  	/** Ids intentionally present in more than one sprite file (the default pack, its authoring key, the material pack). */
62  	private static final String LUCIDE_RESOURCE = "/org/apache/juneau/views/juneau-symbols-lucide.svg";
63  	private static final String MATERIAL_RESOURCE = "/org/apache/juneau/views/juneau-symbols-material.svg";
64  
65  	/** The material pack is not edited by this work; its id set stays as it was. */
66  	private static final Set<String> MATERIAL_IDS = Set.of(
67  		"juneau-sym-copy", "juneau-sym-csv", "juneau-sym-spreadsheet", "juneau-sym-pdf",
68  		"juneau-sym-refresh", "juneau-sym-toggle_column_search", "juneau-sym-collapse_all",
69  		"juneau-sym-settings", "juneau-sym-first_page", "juneau-sym-chevronleft",
70  		"juneau-sym-chevronright", "juneau-sym-last_page", "juneau-sym-filter",
71  		"juneau-sym-chevrondown"
72  	);
73  
74  	/** Ids intentionally in more than one file: the material pack overlays the default pack by stem (pack("material")). */
75  	private static final Set<String> OVERLAY_ALLOWLIST = MATERIAL_IDS;
76  
77  	private static final Pattern SYMBOL_ID_PATTERN = Pattern.compile("<symbol\\s+id=\"([^\"]+)\"");
78  
79  	private static List<String> idList(String resource) throws IOException {
80  		String svg;
81  		try (var in = ViewsMixin.class.getResourceAsStream(resource)) {
82  			assertNotNull(in, () -> "missing classpath resource: " + resource);
83  			svg = new String(in.readAllBytes(), UTF_8);
84  		}
85  		var ids = new ArrayList<String>();
86  		var m = SYMBOL_ID_PATTERN.matcher(svg);
87  		while (m.find())
88  			ids.add(m.group(1));
89  		return ids;
90  	}
91  
92  	private static Set<String> actualIds() throws IOException {
93  		return new LinkedHashSet<>(idList(ViewsMixin.SYMBOLS_SVG_RESOURCE));
94  	}
95  
96  	private static Set<String> duplicates(List<String> ids) {
97  		var seen = new HashSet<String>();
98  		var dup = new TreeSet<String>();
99  		for (var id : ids)
100 			if (!seen.add(id))
101 				dup.add(id);
102 		return dup;
103 	}

105 	@SuppressWarnings({
106 		"java:S4030" // removed/added are read only by the failure-message lambda, which Sonar does not count as a use.
107 	})
108 	@Test void a01_stemIdSetIsPinned() throws Exception {
109 		var actual = actualIds();
110 		var removed = new TreeSet<>(EXPECTED_IDS);
111 		removed.removeAll(actual);
112 		var added = new TreeSet<>(actual);
113 		added.removeAll(EXPECTED_IDS);
114 
115 		assertList(() -> "juneau-symbols.svg <symbol> stem ids no longer match the pinned set - removed: " + removed
116 			+ ", added: " + added + ". A stem id is a lookup key a host page can override a glyph by; changing,"
117 			+ " adding, or removing one here (even one renamed to something equivalent-looking) breaks that"
118 			+ " lookup silently, with no error on either side. If this change is intentional, update"
119 			+ " EXPECTED_IDS to match.", new TreeSet<>(actual), new TreeSet<>(EXPECTED_IDS).toArray());
120 	}

122 	@Test void a02_idsAreUniqueWithinEachSpriteFile() throws Exception {
123 		for (var resource : List.of(ViewsMixin.SYMBOLS_SVG_RESOURCE, MATERIAL_RESOURCE, LUCIDE_RESOURCE))
124 			assertEmpty(() -> "duplicate <symbol> ids within " + resource, duplicates(idList(resource)));
125 	}

127 	@Test void a03_idsSharedAcrossSpriteFilesAreAllowlisted() throws Exception {
128 		var shared = new TreeSet<>(actualIds());
129 		shared.retainAll(new HashSet<>(idList(MATERIAL_RESOURCE)));
130 		shared.removeAll(OVERLAY_ALLOWLIST);
131 		assertEmpty(() -> "ids present in more than one sprite file but not in OVERLAY_ALLOWLIST", shared);
132 	}

135 		var shared = new TreeSet<>(actualIds());
136 		shared.retainAll(new HashSet<>(idList(LUCIDE_RESOURCE)));
137 		assertEmpty(() -> "ids in both the shipped and Lucide sprite files", shared);
138 	}

144 	@Test void a05_tuneIsTheJuneauSlidersNotTheColumnsGear() throws Exception {
145 		String svg;
146 		try (var in = ViewsMixin.class.getResourceAsStream(ViewsMixin.SYMBOLS_SVG_RESOURCE)) {
147 			svg = new String(in.readAllBytes(), UTF_8);
148 		}
149 		var tune = Pattern.compile("<symbol\\s+id=\"juneau-sym-tune\".*?</symbol>", Pattern.DOTALL).matcher(svg);
150 		assertTrue(tune.find());
151 		var body = tune.group();
152 		assertEquals(3, body.split("<circle", -1).length - 1, "three slider knobs: " + body);
153 		assertFalse(body.contains("M9.594"), "tune must not carry the columns gear path: " + body);
154 	}
