32  /**
33   * The <b>"popups never clip"</b> half of the module's JavaScript-execution harness: runs the real served
34   * {@code juneau-views.js} / {@code juneau-renders.js} / {@code juneau-views.css} in headless Chromium and proves that
35   * a surface anchored in a table cell escapes the table's scroll region instead of being clipped by it.
36   *
37   * <h5 class='section'>The matrix:</h5>
38   * <p>
39   * Three surfaces (row-action menu, cell popover, timestamp popup) across three scroll boundaries, each with the
40   * anchoring cell deliberately scrolled half out of view:
41   * <ul class='spaced-list'>
42   * 	<li><b>DT1</b> &mdash; the JS-inserted {@code .juneau-view-table-scroll} wrap.
43   * 	<li><b>DT2</b> &mdash; the flex {@code .dt-layout-cell} the scoped CSS rule turns into the scroll box.
44   * 	<li><b>Nested</b> &mdash; a nested table inside a row-detail panel's own scrolling
45   * 		{@code .juneau-view-detail-nested} wrapper.
46   * </ul>
47   *
48   * <h5 class='section'>Mechanical, never visual:</h5>
49   * <p>
50   * "Not clipped" is asserted as mechanism, so it cannot rot into a judgement call.  For each case the layer must be
51   * {@code position: fixed}, parented to {@code document.body}, outside the scroll box, and have <b>no clipping
52   * ancestor at all</b> &mdash; that last one is stronger than "outside that box", because it also catches a newly
53   * introduced clipping ancestor somewhere else in the lineage.  Each case additionally proves its box really is a
54   * clipping, actually-overflowing region, so a fixture that silently stopped scrolling cannot pass vacuously.
55   *
56   * <h5 class='section'>Scrolling RETAINS the layer:</h5>
57   * <p>
58   * Scrolling the region underneath an open layer neither repositions nor dismisses it &mdash; the layer keeps the same
59   * viewport rect and stays in the document.  This matches the shipped {@code h4} menu behavior, and stating it as an
60   * assertion is what stops a future "reposition on scroll" change from landing silently.
61   *
62   * <h5 class='section'>Off by default:</h5>
63   * <p>
64   * Disabled unless the {@value #GATE} system property is set, which only the module's opt-in {@code js-tests} Maven
65   * profile does.  It reuses that profile's provisioned Node + Playwright browser and derives its own prober
66   * ({@code table-clip-free-browser.cjs}) from the profile's {@code juneau.jsTests.harness} directory, so no pom change
67   * is needed.
68   */
69  @EnabledIfSystemProperty(named=TableClipFree_BrowserTest.GATE, matches="true",
70  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -f juneau-rest/juneau-console-views/pom.xml test`")
71  class TableClipFree_BrowserTest extends TestBase {

78  	private static String resource(String path) throws IOException {
79  		try (var in = ViewsMixin.class.getResourceAsStream(path)) {
80  			assertNotNull(in, () -> "missing classpath resource: " + path);
81  			return new String(in.readAllBytes(), UTF_8);
82  		}
83  	}

85  	@BeforeAll
86  	static void probe() throws Exception {
87  		var dir = Path.of(requiredProperty("juneau.jsTests.dir"));
88  		var harness = Path.of(requiredProperty("juneau.jsTests.harness")).getParent()
89  			.resolve("table-clip-free-browser.cjs");
90  
91  		// The CSS matters here: the DT1/DT2/nested scroll boxes and the timestamp popup's position:fixed all come
92  		// from the real stylesheet, so a fixture without it would prove nothing about the shipped contract.
93  		var fixture = "<!DOCTYPE html><html><head><meta charset=\"utf-8\"><style>\n"
94  			+ resource(ViewsMixin.VIEWS_CSS_RESOURCE)
95  			+ "\n</style></head><body>\n<script>\n"
96  			+ resource(ViewsMixin.RENDERS_JS_RESOURCE)
97  			+ "\n</script>\n<script>\n"
98  			+ resource(ViewsMixin.VIEWS_JS_RESOURCE)
99  			+ "\n</script></body></html>";
100 		var fixtureFile = Files.createDirectories(dir.resolve("fixtures")).resolve("table-clip-free.html");
101 		Files.write(fixtureFile, fixture.getBytes(UTF_8));
102 
103 		report = Json.to(run(dir, harness, fixtureFile), Map.class);
104 	}

106 	private static String requiredProperty(String name) {
107 		var v = System.getProperty(name);
108 		assertNotNull(v, () -> "-D" + name + " not set; the js-tests profile is responsible for providing it");
109 		return v;
110 	}

112 	private static String run(Path dir, Path harness, Path fixture) throws Exception {
113 		var cmd = List.of(System.getProperty("juneau.jsTests.node", "node"), harness.toString(), fixture.toString());
114 		var stdout = dir.resolve("table-clip-free-stdout.json");
115 		var stderr = dir.resolve("table-clip-free-stderr.txt");
116 		var pb = new ProcessBuilder(cmd).redirectOutput(stdout.toFile()).redirectError(stderr.toFile());
117 		pb.environment().put("NODE_PATH", dir.resolve("node_modules").toString());
118 		pb.environment().put("PLAYWRIGHT_BROWSERS_PATH", requiredProperty("juneau.jsTests.browsers"));
119 
120 		var p = pb.start();
121 		if (!p.waitFor(3, TimeUnit.MINUTES)) {
122 			p.destroyForcibly();
123 			fail("prober did not finish within 3m; stderr:\n" + quietRead(stderr));
124 		}
125 		assertEquals(0, p.exitValue(), () -> "prober exited non-zero; stderr:\n" + quietRead(stderr));
126 		return Files.readString(stdout);
127 	}
128 
129 	private static String quietRead(Path p) {
130 		try {
131 			return Files.readString(p);
132 		} catch (IOException e) {
133 			return "<unreadable: " + e + ">";
134 		}
135 	}

137 	@SuppressWarnings({
138 		"unchecked" // The prober report is a JSON-parsed Object tree; sub() casts a known entry to Map<String,Object>
139 	})
140 	private static Map<String,Object> sub(String key) {
141 		var m = (Map<String,Object>) report.get(key);
142 		assertNotNull(m, () -> "the prober produced no `" + key + "` case: " + report);
143 		return m;
144 	}

146 	/**
147 	 * The one shared body of assertions, applied to every cell in the matrix.  Kept as a single helper on purpose:
148 	 * the contract is identical for every surface and every scroll boundary, and writing it once is what makes that
149 	 * sameness checkable rather than aspirational.
150 	 */
151 	private static void assertClipFree(String caseName) {
152 		var m = sub(caseName);
153 		assertEquals(Boolean.TRUE, m.get("opened"), () -> caseName + ": the surface never opened: " + m);
154 		// The fixture is honest: a real clipping box that really overflows.
155 		assertEquals(Boolean.TRUE, m.get("boxClips"), () -> caseName + ": the scroll box does not clip: " + m);
156 		assertEquals(Boolean.TRUE, m.get("boxActuallyScrolled"),
157 			() -> caseName + ": the scroll box is not actually overflowing, so it proves nothing: " + m);
158 		// h4 C: portalled to the body as position:fixed.
159 		assertEquals(Boolean.TRUE, m.get("onBody"), () -> caseName + ": the layer is not parented to <body>: " + m);
160 		assertEquals(Boolean.TRUE, m.get("positionFixed"), () -> caseName + ": the layer is not position:fixed: " + m);
161 		assertEquals(Boolean.TRUE, m.get("escapedScrollBox"),
162 			() -> caseName + ": the layer is still a descendant of the scroll box: " + m);
163 		assertEquals(List.of(), m.get("clippingAncestors"),
164 			() -> caseName + ": the layer has a clipping ancestor, so its overflow can still be cut: " + m);
165 		assertEquals(Boolean.TRUE, m.get("hasArea"), () -> caseName + ": the layer has no painted area: " + m);
166 		// Scrolling the region RETAINS the layer - it neither moves nor disappears.
167 		assertEquals(Boolean.TRUE, m.get("retainedInDom"), () -> caseName + ": scrolling dismissed the layer: " + m);
168 		assertEquals(Boolean.TRUE, m.get("retainedVisible"), () -> caseName + ": scrolling hid the layer: " + m);
169 		assertEquals(Boolean.TRUE, m.get("notRepositioned"),
170 			() -> caseName + ": scrolling repositioned the layer; the contract is that it is RETAINED: " + m);
171 	}

203 	/**
204 	 * The clip default must not maim the row-detail panel.  DataTables' native child row wraps the panel in a plain
205 	 * {@code <td colspan>}, and that cell is a descendant of the {@code .juneau-view-table} &mdash; so without an
206 	 * explicit opt-out the table's own {@code overflow:hidden} / {@code nowrap} / {@code max-width} would flatten a
207 	 * whole expanded detail panel into a single ellipsised line.
208 	 */
209 	@Test void e01_expandedDetailPanelCellIsNotClipped() {
210 		var m = sub("detailPanelCell");
211 		assertEquals(Boolean.TRUE, m.get("expanded"), () -> "the detail row never expanded: " + m);
212 		// The premise: the panel really is hosted by a <td> inside a clip-stamped table.
213 		assertEquals(Boolean.TRUE, m.get("hostIsTd"), () -> "the panel is not hosted by a <td>: " + m);
214 		assertEquals(Boolean.TRUE, m.get("insideViewTable"),
215 			() -> "the host cell is not inside a .juneau-view-table, so this proves nothing: " + m);
216 		assertEquals(Boolean.TRUE, m.get("wraps"), () -> "the detail panel's cell is nowrap - the panel is flattened: " + m);
217 		assertEquals(Boolean.TRUE, m.get("notOverflowHidden"), () -> "the detail panel's cell clips its overflow: " + m);
218 		assertEquals(Boolean.TRUE, m.get("noMaxWidthCap"), () -> "the detail panel's cell is width-capped: " + m);
219 		assertEquals(Boolean.TRUE, m.get("panelHasArea"), () -> "the detail panel has no height: " + m);
220 	}
