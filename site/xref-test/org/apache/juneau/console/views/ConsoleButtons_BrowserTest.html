82  	private static final String PROBE = """
83  		<script>
84  		(function () {
85  			var PROPS = ['background-color', 'color', 'border-top-color', 'border-left-color', 'border-top-width', 'opacity',
86  				'cursor', 'pointer-events', 'z-index', 'margin-left', 'width', 'height', 'border-top-left-radius',
87  				'text-decoration-line', 'filter', 'box-shadow'];
88  			var rec = {};
89  			function snap(id) {
90  				var cs = getComputedStyle(document.getElementById(id));
91  				var o = {};
92  				PROPS.forEach(function (p) { o[p] = cs.getPropertyValue(p); });
93  				return o;
94  			}
95  			window.__rec = function (key, id) { rec[key] = snap(id); };
96  			window.__probe = { toJSON: function () {
97  				var tok = {};
98  				var t = document.getElementById('tokprobe');
99  				%s.forEach(function (n) { t.style.color = 'var(' + n + ')'; tok[n] = getComputedStyle(t).color; });
100 				var rest = {};
101 				['sec','sec2','pri','des','dan','lnk','dis','g1','g2','tib','tibp','ico','bare','bareapp','rb','rbp','ds1','ds2','st1','st2','tab','sub','dc','dx','dk','dd','outside']
102 					.forEach(function (id) { rest[id] = snap(id); });
103 				return { tok: tok, rest: rest, hover: rec };
104 			} };
105 		})();
106 		</script>
107 		""";

111 	private static String caseName(String palette, String scheme, String order) {
112 		return palette + "-" + scheme + "-" + order;
113 	}
114 
115 	private static String page(String palette, String scheme, String order) {
116 		var chrome = "<link rel=\"stylesheet\" href=\"" + CHROME_CSS + "\">\n";
117 		var views = "<link rel=\"stylesheet\" href=\"" + VIEWS_CSS + "\">\n";
118 		var themes = "<link rel=\"stylesheet\" href=\"" + DARK_CSS + "\">\n<link rel=\"stylesheet\" href=\"/juneau-console/juneau-theme-" + palette + ".css\">\n";
119 		var attrs = "data-jc-scheme=\"" + scheme + "\" data-jc-scheme-pin=\"" + scheme + "\"";
120 		var tokens = new StringBuilder("[");
121 		for (var t : TOKENS)
122 			tokens.append(tokens.length() == 1 ? "" : ",").append('\'').append(t).append('\'');
123 		tokens.append(']');
124 		return "<!DOCTYPE html>\n<html lang=\"en\" " + attrs + "><head><meta charset=\"utf-8\"><link rel=\"icon\" href=\"data:,\">\n"
125 			+ (order.equals("cv") ? chrome + themes + views : views + chrome + themes)
126 			+ "<style>button[data-app] { background-color: rgb(255, 0, 0); }</style>\n"
127 			+ "</head><body>\n" + BODY + PROBE.formatted(tokens) + "</body></html>\n";
128 	}

130 	@BeforeAll
131 	static void run() throws Exception {
132 		var f = ConsoleBrowserFixture.create("buttons");
133 		for (var p : PALETTES)
134 			f.asset("/juneau-console/juneau-theme-" + p + ".css", resource("/org/apache/juneau/console/juneau-theme-" + p + ".css"), "text/css");
135 		for (var p : PALETTES)
136 			for (var s : SCHEMES)
137 				for (var o : ORDERS) {
138 					f.page(caseName(p, s, o), ORIGIN + "/" + caseName(p, s, o), page(p, s, o));
139 					for (var id : HOVERED)
140 						f.hover("#" + id).eval("window.__rec('" + id + "', '" + id + "')");
141 					// Park the pointer off every button so the rest snapshot is not hovered.
142 					f.hover("#park");
143 				}
144 		report = f.run();
145 	}

147 	private record Probe(String name, Map<String,String> tok, Map<String,Map<String,String>> rest, Map<String,Map<String,String>> hover) {
148 		String tok(String n) { return tok.get(n); }
149 		String rest(String id, String prop) { return rest.get(id).get(prop); }
150 		String hover(String id, String prop) { return hover.get(id).get(prop); }
151 	}
152 
153 	private static List<Probe> probes() {
154 		var out = new ArrayList<Probe>();
155 		for (var p : PALETTES)
156 			for (var s : SCHEMES)
157 				for (var o : ORDERS) {
158 					var name = caseName(p, s, o);
159 					var r = report.get(name);
160 					assertNotNull(r, () -> "no report for " + name);
161 					assertEquals(List.of(), r.get("pageErrors"), () -> name + ": page errors");
162 					var pr = probe(r);
163 					out.add(new Probe(name, (Map<String,String>)pr.get("tok"), (Map<String,Map<String,String>>)pr.get("rest"),
164 						(Map<String,Map<String,String>>)pr.get("hover")));
165 				}
166 		return out;
167 	}
