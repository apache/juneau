29  /**
30   * Design test <b>33</b>: the <b>emitter-purity scan</b> &mdash; every {@code rawText(...)} call site in
31   * the <b>main</b> Java of the views and widgets trees hands the verbatim writer a JSON-sidecar payload (or provably
32   * nothing), and the content-bearing sites are a closed, individually-named set that <b>cannot grow</b>.
33   *
34   * <p>
35   * This is the test the design nominates as proving the headline claim of the emitter surgery: emitters stopped being
36   * markup authors and became descriptor authors. It shares {@link RawContentSinkScanner}'s stripper and argument extractor with
37   * {@link ShippedJs_SinkScan_Test}, which guards the shipped JavaScript against HTML sinks.
38   *
39   * <h5 class='section'>The design's universal is now the pin</h5>
40   * <p>
41   * The contract is universal &mdash; <i>every</i> {@code rawText} takes a sidecar payload, i.e. the
42   * content-bearing count is <b>zero</b>.  That is now true: the page/card declarative-content emitters
43   * ({@code Tab}/{@code Subtab}/{@code CardContent}) are gone, {@link #KNOWN_CONTENT_SINKS} is empty, and a
44   * <b>new</b> content-bearing {@code rawText} anywhere in either tree fails this test.
45   *
46   * <h5 class='section'>Why the anti-vacuity checks come first</h5>
47   * <p>
48   * Every assertion below &mdash; a floor, a universal, an exact-set pin &mdash; is satisfied <i>more</i> easily by a
49   * scan that finds nothing than by one that finds the real tree. {@link RawContentSinkScanner}'s own javadoc records
50   * this hazard biting it once already: the widgets tree is reached by a relative path and a walk skips a missing
51   * directory in silence. So the checks proving the scan still sees both trees are as load-bearing as the violation
52   * check itself, and they run first.
53   */
54  @SuppressWarnings({
55  	"java:S3415" // The Set.of/List.of fixture is the expected value; the constant/field is the actual under test.
56  })
57  class EmitterPurity_ScanTest extends TestBase {
58  
59  	/**
60  	 * The content-bearing {@code rawText(...)} sinks that survive the emitter surgery, named individually because an
61  	 * unnamed allowance is indistinguishable from a hole.
62  	 *
63  	 * <p>
64  	 * Empty: the page/card declarative-content emitters are gone.  <b>This set must stay empty.</b>
65  	 */
66  	private static final Set<String> KNOWN_CONTENT_SINKS = Set.of();
67  
68  	/**
69  	 * Floor on the total number of {@code rawText(...)} sites, matching the remaining sidecar write
70  	 * ({@code RegionTable}). ViewTable's VIEW_META / bulk sidecars retired with the MOVE Java delete.
71  	 *
72  	 * <p>
73  	 * Deliberately slack: sidecars legitimately come and go as emitters are refactored, so pinning the total exactly
74  	 * would make this a change-detector. Its only job is to fail loudly if the scan stops seeing the tree at all.
75  	 */
76  	private static final int MINIMUM_EXPECTED_SITES = 1;
77  
78  	private static Path moduleRoot() {
79  		var root = RawContentSinkScanner.locateModuleRoot();
80  		assertNotNull(root, "could not locate the juneau-console-views module root; the scan would see nothing");
81  		return root;
82  	}
83  
84  	private static List<RawContentSinkScanner.RawTextSite> sites() throws Exception {
85  		return RawContentSinkScanner.scanRawTextTree(moduleRoot());
86  	}

93  		var sites = sites();
94  		assertTrue(sites.size() >= MINIMUM_EXPECTED_SITES,

96  				+ sites.size() + " rawText site(s): " + sites);
97  	}

99  	@Test void a02_scanReachesTheSiblingWidgetsTree() {
100 		// Asserted separately from every other check here because all of those are floors or universals, and both
101 		// are satisfied MORE easily by a smaller set.  This is the exact hazard RawContentSinkScanner's javadoc
102 		// records having bitten it before.
103 		var widgets = RawContentSinkScanner.widgetsModuleRoot(moduleRoot()).resolve("src/main/java");
104 		assertTrue(Files.isDirectory(widgets),
105 			() -> "the widgets main tree is not where this scan looks (" + widgets + "), so its emitters would drop "
106 				+ "out of every scan silently while the detection logic stayed perfectly correct");
107 	}

112 			() -> "expected the RegionTable sidecar payload shape to still exist: " + args);

114 	}

116 	@Test void a04_scanFindsTheKnownContentSinks() throws Exception {
117 		// If the classifier silently stopped recognizing content, b01/b02 would both pass vacuously.
118 		var keys = sites().stream().map(RawContentSinkScanner.RawTextSite::key).toList();
119 		for (var known : KNOWN_CONTENT_SINKS)
120 			assertTrue(keys.contains(known),
121 				() -> "the scan no longer finds the known content sink '" + known + "'.  If it was genuinely removed "
122 					+ "(the deprecation window closing) drop it from KNOWN_CONTENT_SINKS in the same commit; "
123 					+ "otherwise the scan has gone blind.  Found: " + keys);
124 	}

126 	@Test void a05_argumentExtractionIsBalanced() {
127 		// Must be balanced-paren, or rawText(escapeForScript(nestedJson(v, tokenless))) truncates to
128 		// "escapeForScript(nestedJson(v" and gets silently RECLASSIFIED as content-bearing.
129 		var found = RawContentSinkScanner.scanRawText("X.java",
130 			"class X { void f() { q.text(rawText(escapeForScript(nestedJson(v, tokenless)))); } }");
131 		assertSize(found::toString, 1, found);
132 		assertEquals("escapeForScript(nestedJson(v, tokenless))", found.get(0).arg());
133 		assertTrue(found.get(0).isSidecarPayload());
134 	}

136 	@Test void a06_commentMentionsAreNotCounted() {
137 		// The real tree mentions rawText(...) in javadoc on ViewTable and siblings.  Counting prose would both
138 		// inflate the floor and manufacture phantom violations.
139 		var found = RawContentSinkScanner.scanRawText("X.java", """
140 			class X {
141 				/** Writes via {@link HtmlBuilder#rawText(String) rawText} verbatim. */
142 				// rawText(t.content) mentioned here
143 				void f() {}
144 			}
145 			""");
146 		assertTrue(found.isEmpty(), found::toString);
147 	}
148 
149 	@Test void a07_classifierSeparatesTheThreeShapes() {
150 		var sidecar = new RawContentSinkScanner.RawTextSite("A.java", 1, "json");
151 		var inline = new RawContentSinkScanner.RawTextSite("A.java", 2, "escapeForScript(nestedJson(v))");
152 		var empty = new RawContentSinkScanner.RawTextSite("A.java", 3, "\"\"");
153 		var content = new RawContentSinkScanner.RawTextSite("A.java", 4, "t.content");
154 
155 		assertTrue(sidecar.isSidecarPayload() && !sidecar.isContentBearing());
156 		assertTrue(inline.isSidecarPayload() && !inline.isContentBearing());
157 		assertTrue(empty.isEmptyLiteral() && !empty.isContentBearing());
158 		assertTrue(content.isContentBearing(), "a bare field read is the shape a live-data body would take");
159 	}

165 	@Test void b01_everyRawTextIsASidecarPayloadEmptyOrANamedContentSink() throws Exception {
166 		var violations = sites().stream()
167 			.filter(RawContentSinkScanner.RawTextSite::isContentBearing)
168 			.filter(s -> !KNOWN_CONTENT_SINKS.contains(s.key()))
169 			.map(RawContentSinkScanner.RawTextSite::toString)
170 			.toList();
171 		assertTrue(violations.isEmpty(),
172 			() -> "a NEW content-bearing rawText(...) sink appeared.  Emitters author DESCRIPTORS, not markup: every "
173 				+ "rawText in main Java must hand the verbatim writer a JSON-sidecar payload.  The only content-bearing "
174 				+ "exceptions are the deprecated declarative-content sinks named in KNOWN_CONTENT_SINKS, and that set "
175 				+ "exists to shrink to empty - never to grow:\n  " + String.join("\n  ", violations));
176 	}

178 	@Test void b02_theContentBearingSurfaceIsEmpty() throws Exception {
179 		var content = sites().stream()
180 			.filter(RawContentSinkScanner.RawTextSite::isContentBearing)
181 			.map(RawContentSinkScanner.RawTextSite::toString)
182 			.toList();
183 		assertEquals(List.of(), content,
184 			() -> "the content-bearing rawText surface closed with the page/card emitters.  It must stay empty:\n  "
185 				+ String.join("\n  ", content));

188 	@Test void b03_theRegionEmitterIsPureFromDayOne() throws Exception {
189 		// The emitter this item ADDS must be on the right side of the line from the start.  A region container is an
190 		// EMPTY div plus a descriptor sidecar; if RegionTable ever writes content verbatim, the whole
191 		// "configuration rides the sidecar, not the markup" claim is gone.
192 		var region = sites().stream().filter(s -> eq(s.file(), "RegionTable.java")).toList();
193 		assertFalse(region.isEmpty(),
194 			"RegionTable writes a descriptor sidecar, so at least one rawText site is expected; finding none means "
195 				+ "the scan is not reaching it");
196 		region.forEach(s -> assertTrue(s.isSidecarPayload(),
197 			() -> "the region emitter must never be a content sink: " + s));
198 	}

200 	@Test void b04_viewTableIsConstantsOnly_noRawText() throws Exception {
201 		var viewTable = sites().stream().filter(s -> eq(s.file(), "ViewTable.java")).toList();
202 		assertEquals(List.of(), viewTable,
203 			() -> "ViewTable is constants-only after MOVE delete; it must not write rawText: " + viewTable);
204 	}

206 	@Test void b05_everyContentSinkIsInTheDeprecatedDeclarativePath() throws Exception {
207 		// Pins WHERE the four live, not just how many.  A content sink migrating into a different emitter would keep
208 		// the count at four while quietly widening the blast radius.
209 		var files = sites().stream()
210 			.filter(RawContentSinkScanner.RawTextSite::isContentBearing)
211 			.map(RawContentSinkScanner.RawTextSite::file)
212 			.distinct()
213 			.sorted()
214 			.toList();
215 		assertEquals(List.of(), files,
216 			"content-bearing rawText must not return: the page/card declarative-content emitters are gone");
217 	}

220 	// c) Proof the guard BITES.  a01-a07 prove the scan sees the tree; these prove that seeing it is not the same as
221 	//    a passing test being meaningful.  Both mutate REAL main source text, so they cannot drift from the shapes
222 	//    the emitters actually use the way a hand-written fixture silently would.
223 	//------------------------------------------------------------------------------------------------------------------
224 
225 	@Test void c01_mutatingARealSidecarIntoAFieldRead_becomesAViolation() throws Exception {
226 		var regionTable = moduleRoot().resolve("src/main/java/org/apache/juneau/console/views/RegionTable.java");
227 		var src = Files.readString(regionTable);
228 		assertTrue(src.contains("text(rawText(json))"), "the real sidecar shape moved; this mutation no longer applies");
229 
230 		var mutated = src.replace("text(rawText(json))", "text(rawText(row.liveValue))");
231 		var sites = RawContentSinkScanner.scanRawText("RegionTable.java", mutated);
232 		var offenders = sites.stream().filter(RawContentSinkScanner.RawTextSite::isContentBearing).toList();
233 
234 		assertSize(() -> "turning one real sidecar write into a live-data field read must produce exactly one content-bearing "
235 				+ "site; if it produces none, b01/b02 are passing for free: " + sites, 1, offenders);
236 		assertFalse(KNOWN_CONTENT_SINKS.contains(offenders.get(0).key()),
237 			"the mutated site must NOT be absorbed by the allowlist - otherwise the allowlist is a hole");
238 	}
239 
240 	@Test void c02_aNewContentSinkInTheWidgetsTreeWouldBeCaught() {
241 		// The widgets tree is the one reached by a relative path, so pair a02's "the directory is there" with proof
242 		// that a violation IN it is actually classified as one.
243 		var sites = RawContentSinkScanner.scanRawText("CardContent.java",
244 			"class CardContent { Object f() { return div(rawText(this.userSuppliedBody)); } }");
245 		assertSize(sites::toString, 1, sites);
246 		assertTrue(sites.get(0).isContentBearing(), () -> "not classified as content: " + sites.get(0));
247 		assertFalse(KNOWN_CONTENT_SINKS.contains(sites.get(0).key()), "must not be pre-absorbed by the allowlist");
248 	}
