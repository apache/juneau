27  /**
28   * Always-on source-shape coverage for the {@code juneau-views.js} async-job + SSE-streaming plumbing.
29   * Mirrors {@link ViewsJs_ModalResult_Test}'s served-script substring style: proves the load-bearing pieces of the
30   * async contract are present in the shipped asset without booting a browser (the behavioral proof lives in the opt-in
31   * {@code AsyncJob_BrowserTest} canary).
32   *
33   * <p>
34   * The single most important invariant asserted here is that the job-running affordance is a DISTINCT marker
35   * ({@code data-juneau-job}) that {@code hasInFlightRow} does NOT read, so a long job never freezes the table's
36   * polling the way the synchronous {@code data-juneau-inflight} marker does.
37   */
38  @SuppressWarnings({
39  	"resource" // Closeable test fixture held in a static field; lifecycle managed by the test/framework, not a real leak.
40  })
41  class ViewsJs_AsyncJobs_Test extends TestBase {
42  
43  	@Rest(mixins=ViewsMixin.class)
44  	public static class WithMixin extends BasicRestServlet {
45  		private static final long serialVersionUID = 1L;
46  		@RestGet(path="/items") public String items() { return "items"; }
47  	}
48  
49  	private static final MockRestClient c = MockRestClient.buildLax(WithMixin.class);
50  
51  	private static String viewsJs() throws Exception {
52  		return c.get(ViewsMixin.VIEWS_JS_PATH).run().assertStatus(200).getContent().asString();
53  	}
54  
55  	private static String fn(String body, String signature) {
56  		var start = body.indexOf(signature);
57  		assertTrue(start >= 0, () -> signature + " not found:\n" + body);
58  		var end = body.indexOf("\n\t}", start);
59  		return body.substring(start, end < 0 ? body.length() : end);
60  	}

62  	//------------------------------------------------------------------------------------------------------------------
63  	// a) the job-running affordance is a DISTINCT marker that never freezes polling
64  	//------------------------------------------------------------------------------------------------------------------
65  
66  	@Test void a01_setRowJobRunning_setsDistinctJobMarker_neverTheInflightMarker() throws Exception {
67  		var s = fn(viewsJs(), "function setRowJobRunning(");
68  		assertTrue(s.contains("dataset.juneauJob"), s);
69  		assertTrue(s.contains("delete tr.dataset.juneauJob"), s);
70  		// The whole point of the distinct marker: a job must NEVER set the synchronous in-flight marker that freezes
71  		// polling.
72  		assertFalse(s.contains("data-juneau-inflight"), () -> "job affordance must NOT touch data-juneau-inflight:\n" + s);
73  		// A second job can't be launched on the same row while one runs.
74  		assertTrue(s.contains("trigger.disabled"), s);
75  	}
76  
77  	@Test void a02_hasInFlightRow_ignoresTheJobMarker_soPollingKeepsRunningDuringAJob() throws Exception {
78  		// hasInFlightRow reads ONLY data-juneau-inflight; it must NOT read data-juneau-job, or a job would freeze
79  		// the whole table's polling for up to the 120s hard timeout (hiding a resurrected incident) - exactly the
80  		// failure the distinct marker exists to prevent.
81  		var h = fn(viewsJs(), "function hasInFlightRow(");
82  		assertTrue(h.contains("data-juneau-inflight"), h);
83  		assertFalse(h.contains("data-juneau-job"), () -> "hasInFlightRow must ignore the job marker:\n" + h);
84  	}
85  
86  	@Test void a03_startJobStream_usesTheJobAffordance_neverTheInflightMarker() throws Exception {
87  		var s = fn(viewsJs(), "function startJobStream(");
88  		assertTrue(s.contains("host.setRunning(true)"), s);
89  		assertTrue(s.contains("host.setRunning(false)"), s);
90  		// It must not co-opt the synchronous in-flight lifecycle.
91  		assertFalse(s.contains("setRowInFlight("), () -> "startJobStream must not use the sync in-flight marker:\n" + s);
92  		assertFalse(s.contains("data-juneau-inflight"), s);
93  	}

96  	// b) The async start is detected by RESPONSE SHAPE (a job pointer), not a new RowAction wire field
97  	//------------------------------------------------------------------------------------------------------------------
98  
99  	@Test void b01_parseJobStarted_requiresANonBlankStreamUrl_andIsDisjointFromATypedResult() throws Exception {
100 		var p = fn(viewsJs(), "function parseJobStarted(");
101 		assertTrue(p.contains("isBlankToken(o.streamUrl)"), p);
102 		// The two 2xx shapes are disjoint: a job pointer is recognized by streamUrl, a terminal result by outcome.
103 		var settle = fn(viewsJs(), "function settleActionResponse(");
104 		assertTrue(settle.contains("parseJobStarted(text)"), settle);
105 		assertTrue(settle.contains("startJobStream("), settle);
106 		// The job branch is checked BEFORE the typed-result branch.
107 		assertTrue(settle.indexOf("parseJobStarted(text)") < settle.indexOf("parseActionResult(text)"), settle);
108 	}
109 
110 	@Test void b02_settleClearsInflightBeforeRoutingToTheJobStream_soPollingResumesFirst() throws Exception {
111 		// The in-flight marker is cleared as the FIRST statement of settle (417), so table polling has already
112 		// resumed before the (possibly 120s) job even starts streaming - the job then only ever carries the
113 		// distinct data-juneau-job marker.
114 		var settle = fn(viewsJs(), "function settleActionResponse(");
115 		var open = settle.indexOf("{");
116 		var firstStmt = settle.substring(open + 1, settle.indexOf(";", open));
117 		assertTrue(firstStmt.contains("setRowInFlight(tr, false)"), () -> "settle must clear the in-flight marker first:\n" + settle);
118 	}

120 	//------------------------------------------------------------------------------------------------------------------
121 	// c) The stream URL IS the capability: an EventSource GET, no CSRF header on the stream
122 	//------------------------------------------------------------------------------------------------------------------
123 
124 	@Test void c01_startJobStream_opensAnEventSourceOnTheCapabilityStreamUrl() throws Exception {
125 		var s = fn(viewsJs(), "function startJobStream(");
126 		assertTrue(s.contains("new EventSource(started.streamUrl)"), s);
127 		// A browser EventSource cannot set headers, so the stream carries NO CSRF header - unguessability is the gate.
128 		assertFalse(s.contains(DEFAULT_CSRF_HEADER_LITERAL), () -> "the SSE GET must not attach a CSRF header:\n" + s);
129 		// Progress + a single terminal result event; a stream error is itself a non-optimistic terminal outcome.
130 		assertTrue(s.contains("addEventListener(\"progress\""), s);
131 		assertTrue(s.contains("addEventListener(\"result\""), s);
132 		assertTrue(s.contains("addEventListener(\"error\""), s);
133 	}
134 
135 	/** The exact header token the runtime uses for the CSRF header (so c01's negative assertion is meaningful). */
136 	private static final String DEFAULT_CSRF_HEADER_LITERAL = "X-Csrf-Token";
137 
138 	@Test void c02_missingEventSourceDegradesToAVisibleUnknown_notASilentHang() throws Exception {

140 		assertTrue(s.contains("typeof EventSource === \"undefined\""), s);
141 		assertTrue(s.contains("outcome: \"unknown\""), s);
142 	}

171 	@Test void e02_actionOutcomeMessage_rendersBothReservedAsyncOutcomes() throws Exception {
172 		// The frozen ActionResult contract reserved cancelled / cancelled-after-effect; the runtime renders both
173 		// without any new UI (the two are DIFFERENT outcomes and must not be collapsed).
174 		var body = viewsJs();
175 		var m = fn(body, "function actionOutcomeMessage(");
176 		assertTrue(m.contains("case \"cancelled\":"), m);
177 		assertTrue(m.contains("case \"cancelled-after-effect\":"), m);
178 		// And they are recognized outcomes (not normalized away to unknown).
179 		assertTrue(body.contains("\"cancelled\": 1"), body);
180 		assertTrue(body.contains("\"cancelled-after-effect\": 1"), body);
181 	}
