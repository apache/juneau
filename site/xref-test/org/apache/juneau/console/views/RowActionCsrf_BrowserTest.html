34  /**
35   * The row-action half of the module's <b>JavaScript-execution harness</b>: runs the REAL served
36   * {@code juneau-views.js} in a real headless browser and asserts the fail-closed CSRF submit behaves as a user
37   * would experience it &mdash; a blank/absent/whitespace token issues NO request and renders a VISIBLE refusal, and
38   * a valid token issues a JSON {@code POST} carrying the {@code X-Csrf-Token} header.
39   *
40   * <h5 class='section'>Why this exists (beyond the always-on source-shape test):</h5>
41   * <p>
42   * {@link ViewsJs_RowActions_Test} proves the shipped script <i>contains</i> the fail-closed logic; it cannot prove
43   * the logic actually suppresses a request and surfaces a visible refusal at runtime.  This canary drives the real
44   * runtime in Chromium against a stubbed {@code fetch}, so "the request was not sent and a banner became visible" is
45   * measured rather than inferred.  The client refusal is defense-against-consumer-omission; the server-side
46   * {@link LoopbackBoundary} is the real control.
47   *
48   * <h5 class='section'>Off by default:</h5>
49   * <p>
50   * Disabled unless the {@value #GATE} system property is set, which only the module's opt-in {@code js-tests} Maven
51   * profile does.  It reuses that profile's provisioned Node + Playwright browser, and derives its own prober
52   * ({@code row-actions.cjs}) from the profile's {@code juneau.jsTests.harness} directory, so no pom change is needed
53   * to add this second canary.
54   *
55   */
56  @EnabledIfSystemProperty(named=RowActionCsrf_BrowserTest.GATE, matches="true",
57  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -f juneau-rest/juneau-console-views/pom.xml test`")
58  class RowActionCsrf_BrowserTest extends TestBase {

63  	private static Map<?,?> report;

65  	private static String resource(String path) throws IOException {
66  		try (var in = ViewsMixin.class.getResourceAsStream(path)) {
67  			assertNotNull(in, () -> "missing classpath resource: " + path);
68  			return new String(in.readAllBytes(), UTF_8);
69  		}
70  	}
71  
72  	@BeforeAll
73  	static void probe() throws Exception {
74  		var dir = Path.of(requiredProperty("juneau.jsTests.dir"));
75  		var harness = Path.of(requiredProperty("juneau.jsTests.harness"));
76  
77  		// The fixture restates nothing under test: it loads the REAL served juneau-views.js.  No jQuery/DataTables
78  		// is needed - the prober drives the runtime's exposed pure + DOM helpers directly.
79  		var fixture = "<!DOCTYPE html><html><head><meta charset=\"utf-8\"></head><body>\n<script>\n"
80  			+ resource(ViewsMixin.VIEWS_JS_RESOURCE)
81  			+ "\n</script></body></html>";
82  		var fixtureFile = Files.createDirectories(dir.resolve("fixtures")).resolve("row-actions.html");
83  		Files.write(fixtureFile, fixture.getBytes(UTF_8));
84  
85  		report = Json.to(run(dir, harness, fixtureFile), Map.class);
86  	}
87  
88  	private static String requiredProperty(String name) {
89  		var v = System.getProperty(name);
90  		assertNotNull(v, () -> "-D" + name + " not set; the js-tests profile is responsible for providing it");
91  		return v;
92  	}
93  
94  	/** Runs the prober, failing with its stderr attached (its exit code alone is not a diagnosis). */
95  	private static String run(Path dir, Path harness, Path fixture) throws Exception {
96  		var cmd = List.of(System.getProperty("juneau.jsTests.node", "node"), harness.toString(), fixture.toString());
97  		var stdout = dir.resolve("row-actions-stdout.json");
98  		var stderr = dir.resolve("row-actions-stderr.txt");
99  		var pb = new ProcessBuilder(cmd).redirectOutput(stdout.toFile()).redirectError(stderr.toFile());
100 		pb.environment().put("NODE_PATH", dir.resolve("node_modules").toString());
101 		pb.environment().put("PLAYWRIGHT_BROWSERS_PATH", requiredProperty("juneau.jsTests.browsers"));
102 
103 		var p = pb.start();
104 		if (!p.waitFor(3, TimeUnit.MINUTES)) {
105 			p.destroyForcibly();
106 			fail("prober did not finish within 3m; stderr:\n" + quietRead(stderr));
107 		}
108 		assertEquals(0, p.exitValue(), () -> "prober exited non-zero; stderr:\n" + quietRead(stderr));
109 		return Files.readString(stdout);
110 	}
111 
112 	private static String quietRead(Path p) {
113 		try {
114 			return Files.readString(p);
115 		} catch (IOException e) {
116 			return "<unreadable: " + e + ">";
117 		}
118 	}
