17  package org.apache.juneau.console.views;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.*;
24  import org.junit.jupiter.api.*;
25  
26  /** {@link RowAction#link(String,String)}: the header link action and its validation. */
27  class RowAction_Link_Test extends TestBase {
28  
29  	@Test void a01_link_emitsHref_noEndpointMethodForm() {
30  		var a = RowAction.link("open", "/console/pets/{id}/summary").label("Open").icon("link");
31  		assertEquals("/console/pets/{id}/summary", a.href);
32  		assertNull(a.endpoint);
33  		assertNull(a.method);
34  		assertNull(a.form);
35  	}
36  
37  	@Test void a02_unsafeHref_rejected() {
38  		for (var bad : List.of("https://evil.example/x", "//evil.example/x", "javascript:alert(1)", "x/y"))
39  			assertThrows(IllegalArgumentException.class, () -> RowAction.link("o", bad), bad);
40  	}
41  
42  	@Test void a03_hrefWithEndpointFormConfirm_rejected() {
43  		var withEndpoint = RowDetail.inline().action(RowAction.link("o","/a").label("O").endpoint("/b"));
44  		var withConfirm = RowDetail.inline().action(RowAction.link("o","/a").label("O").confirm("sure"));
45  		assertThrows(IllegalArgumentException.class, withEndpoint::toMap);
46  		assertThrows(IllegalArgumentException.class, withConfirm::toMap);
47  	}
48  
49  	@Test void a04_iconOnlyAction_needsLabel() {
50  		var iconOnly = RowDetail.inline().action(RowAction.link("o","/a").icon("link"));
51  		assertThrows(IllegalArgumentException.class, iconOnly::toMap);
52  	}
53  
54  	@Test void a05_rowMenu_ignoresLinkActions() {
55  		var t = TableSpec.create("t").dataUrl("/rows").columns(Column.create("id"))
56  			.rowActions(RowAction.link("o", "/a/{id}").label("Open"));
57  		t.validate();
58  		assertFalse(t.toCardBody().containsKey("rowActions"));
59  	}
60  
61  	@Test void a06_rowMenu_keepsSubmitActions() {
62  		var t = TableSpec.create("t").dataUrl("/rows").columns(Column.create("id"))
63  			.rowActions(RowAction.link("o", "/a/{id}").label("Open"),
64  				RowAction.create("d").label("Del").endpoint("/a/{id}").method(RowAction.Method.DELETE));
65  		var ra = (List<?>)t.toCardBody().get("rowActions");
66  		assertEquals(1, ra.size());
67  	}
68  }
