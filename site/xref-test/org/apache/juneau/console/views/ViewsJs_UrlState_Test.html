141 	@BeforeAll
142 	static void probeIfNodeAvailable() throws Exception {
143 		if (!nodeAvailable())
144 			return;
145 		var harness = locateHarness("url-state.cjs");
146 		if (harness != null) {
147 			var jsFile = Files.createTempFile("juneau-urlstate-", ".js");
148 			try {
149 				Files.writeString(jsFile, urlStateJs(), UTF_8);
150 				report = Json.to(runNode(harness, List.of(jsFile.toString())), Map.class);
151 			} finally {
152 				Files.deleteIfExists(jsFile);
153 			}
154 		}
155 		var live = locateHarness("url-state-live.cjs");
156 		if (live != null) {
157 			var urlState = writeTempJs("juneau-urlstate-", urlStateJs());
158 			var renders = writeTempResource("juneau-renders-", ViewsMixin.RENDERS_JS_RESOURCE);
159 			var views = writeTempResource("juneau-views-", ViewsMixin.VIEWS_JS_RESOURCE);
160 			try {
161 				liveReport = Json.to(runNode(live, List.of(
162 					urlState.toString(), renders.toString(), views.toString()
163 				)), Map.class);
164 			} finally {
165 				Files.deleteIfExists(urlState);
166 				Files.deleteIfExists(renders);
167 				Files.deleteIfExists(views);
168 			}
169 		}
170 	}

187 	private static boolean nodeAvailable() {
188 		try {
189 			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
190 			if (!p.waitFor(5, TimeUnit.SECONDS)) {
191 				p.destroyForcibly();
192 				return false;
193 			}
194 			return p.exitValue() == 0;
195 		} catch (Exception e) {
196 			return false;
197 		}
198 	}

200 	private static Path locateHarness(String name) {
201 		var basedir = System.getProperty("basedir");
202 		if (basedir != null) {
203 			var p = Path.of(basedir, "src/test/js", name);
204 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
205 		}
206 		for (var rel : List.of(
207 			"src/test/js/" + name,
208 			"juneau-rest/juneau-console-views/src/test/js/" + name
209 		)) {
210 			var p = Path.of(rel);
211 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
212 		}
213 		return null;
214 	}

216 	private static String runNode(Path harness, List<String> args) throws Exception {
217 		var stdout = Files.createTempFile("url-state-stdout-", ".json");
218 		var stderr = Files.createTempFile("url-state-stderr-", ".txt");
219 		try {
220 			var cmd = new ArrayList<String>();
221 			cmd.add("node");
222 			cmd.add(harness.toString());
223 			cmd.addAll(args);
224 			var pb = new ProcessBuilder(cmd)
225 				.redirectOutput(stdout.toFile())
226 				.redirectError(stderr.toFile());
227 			var p = pb.start();
228 			if (!p.waitFor(30, TimeUnit.SECONDS)) {
229 				p.destroyForcibly();
230 				fail(harness.getFileName() + " did not finish within 30s; stderr:\n" + quietRead(stderr));
231 			}
232 			if (p.exitValue() != 0)
233 				fail(harness.getFileName() + " exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr)
234 					+ "\nstdout:\n" + quietRead(stdout));
235 			return Files.readString(stdout, UTF_8);
236 		} finally {
237 			Files.deleteIfExists(stdout);
238 			Files.deleteIfExists(stderr);
239 		}
240 	}

242 	private static String quietRead(Path p) {
243 		try { return Files.readString(p, UTF_8); }
244 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
245 	}

261 			"tab(setup);filter(status=$in(OPEN,CLOSED));filter(name=$eq(a,b));sort(name=desc),setup,"

265 	@Test void b01a_clauseGrammar_escapesRoundTrip_multiClauseBody_firstUnescapedEquals() {
266 		var r = report();
267 		// A column name carrying the grammar's own comma/equals is escaped on encode and restored on decode.
268 		// One filter body with multiple top-level clauses -> one filter each; a comma inside $in(...) is protected.
269 		// Only the FIRST unescaped '=' splits key/value; a later '=' stays verbatim in the value.
270 		assertBean(r, "escapedColEncoded,escapedColDecoded,multiClauseFilters,firstEqualsSplit",
271 			"filter(a\\,b\\=c=$eq(1)),[a,b=c|$eq(1)],[status=$eq(OPEN),name=$in(a,b),score=$gt(5)],expr|$eq(a=b)");
272 	}
