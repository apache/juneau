48  	/** Extracts a named function's body: from `function <name>(` to the next top-level `\n\t}`. */
49  	private static String functionBody(String body, String signature) {
50  		var start = body.indexOf(signature);
51  		assertTrue(start >= 0, () -> "'" + signature + "' not found:\n" + body);
52  		var end = body.indexOf("\n\t}", start);
53  		return body.substring(start, end < 0 ? body.length() : end);
54  	}

56  	@Test void a01_ribbonJs_hasDefaultIconsMapWithAnEntryPerNamedGlyph() throws Exception {
57  		var body = cWithMixin.get(ViewsMixin.RIBBON_JS_PATH).run().assertStatus(200).getContent().asString();
58  		assertTrue(body.contains("DEFAULT_ICONS"), body);
59  		assertTrue(body.contains("copy: \"content_copy\""), body);
60  		assertTrue(body.contains("csv: \"csv\""), body);
61  		assertTrue(body.contains("excel: \"table\""), body);
62  		assertTrue(body.contains("pdf: \"picture_as_pdf\""), body);
63  		assertTrue(body.contains("refresh: \"refresh\""), body);
64  		assertFalse(body.contains("columnSearchToggle"), body);
65  		assertTrue(body.contains("collapse: \"unfold_less\""), body);
66  		assertTrue(body.contains("dialog: \"new\""), body);
67  	}

69  	/**
70  	 * Naming the row-less dialog trigger's glyph is load-bearing for the same reason the pause control's is (see
71  	 * {@code TablePolling_Wiring_Test.e06}): an unnamed type falls through to {@code resolveButtonIcon}'s neutral
72  	 * "tune" default, which paints the settings gear the column chooser already owns - so an icon-only ribbon
73  	 * carrying both would show two identical gears, one opening a create dialog and one opening the column list.
74  	 */
75  	@Test void a03_defaultIcons_dialogDoesNotCollideWithTheColumnChooserGear() throws Exception {
76  		var ribbon = cWithMixin.get(ViewsMixin.RIBBON_JS_PATH).run().assertStatus(200).getContent().asString();
77  		var start = ribbon.indexOf("const DEFAULT_ICONS = {");
78  		var defaults = ribbon.substring(start, ribbon.indexOf("};", start));
79  		assertTrue(defaults.contains("dialog:"), defaults);
80  		assertFalse(defaults.contains("dialog: \"tune\""), defaults);
81  		assertFalse(defaults.contains("dialog: \"settings\""), defaults);
82  	}
83  
84  	@ParameterizedTest
85  	@MethodSource("a02_functionBodyContainsProvider")
86  	void a02_functionBodyContainsExpectedSubstring(String functionSignature, String expectedSubstring) throws Exception {
87  		var body = cWithMixin.get(ViewsMixin.RIBBON_JS_PATH).run().assertStatus(200).getContent().asString();
88  		var fnBody = functionBody(body, functionSignature);
89  		assertTrue(fnBody.contains(expectedSubstring), fnBody);
90  	}
91  
92  	static Stream<Arguments> a02_functionBodyContainsProvider() {
93  		return Stream.of(
94  			// resolveButtonIcon: an explicit symbol wins over the default.
95  			Arguments.of("function resolveButtonIcon(", ".symbol"),
96  			// resolveButtonIcon: a custom option with no symbol falls back to "tune".
97  			Arguments.of("function resolveButtonIcon(", "\"tune\""),
98  			// ribbonStorageKey: the storage-key layout is unchanged.
99  			Arguments.of("function ribbonStorageKey(", "\"juneau.view.\" + viewId + \".ribbon.\" + optionId"));
100 	}

102 	@Test void a05_loadPersistedState_routesThroughSyncPersistenceSpi_sameExactKeys() throws Exception {
103 		var body = cWithMixin.get(ViewsMixin.RIBBON_JS_PATH).run().assertStatus(200).getContent().asString();
104 		var load = functionBody(body, "function loadPersistedState(");
105 		assertTrue(load.contains("storageGet(ribbonStorageKey(viewDef.id, a.id))"), load);
106 		assertFalse(load.contains("store.getItem("), load);
107 		assertTrue(load.contains("a.default"), load);
108 		var persistFn = functionBody(body, "function persist(");
109 		assertTrue(persistFn.contains("storageSet(ribbonStorageKey(viewDef.id, id), String(value))"), persistFn);
110 		assertFalse(persistFn.contains("store.setItem("), persistFn);
111 		var getFn = functionBody(body, "function storageGet(");
112 		assertTrue(getFn.contains("NS.persistence.getItem(key)"), getFn);
113 		assertFalse(getFn.contains("Promise"), getFn);
114 		assertFalse(getFn.contains(".then("), getFn);
115 		var setFn = functionBody(body, "function storageSet(");
116 		assertTrue(setFn.contains("NS.persistence.setItem(key, value)"), setFn);
117 		assertFalse(setFn.contains("Promise"), setFn);
118 	}
