18  
19  import static org.apache.juneau.console.views.ConsoleBrowserFixture.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.util.*;
23  
24  import org.apache.juneau.*;
25  import org.junit.jupiter.api.*;
26  import org.junit.jupiter.api.condition.*;

28  /**
29   * One minimal "it boots in a browser" check for the richest parity case (case 19,
30   * {@code pagespec-corpus/19-visible-when-passthrough.json}: a table with a {@code visibleWhen} row action and detail region):
31   * no console error or page error, and the row action's label is on the opened menu. The contract itself is owned by
32   * {@code PageSpec_Parity_Test}.
33   */
34  @EnabledIfSystemProperty(named=ConsoleBrowserFixture.GATE, matches="true",
35  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -f juneau-rest/juneau-console-views/pom.xml test`")
36  class PageSpecParity_BrowserTest extends TestBase {
37  
38  	private static final String ROWS = """
39  		[{"id":"1","state":"open"}]
40  		""";
41  
42  	private static Map<String,Map<String,Object>> report;
43  
44  	@BeforeAll
45  	static void probe() throws Exception {
46  		report = ConsoleBrowserFixture.create("pagespec-parity")
47  			.asset("/rest/t/data", ROWS, "application/json")
48  			.page("parity", ORIGIN + "/page", corpusPage("19-visible-when-passthrough", "", viewsPack()),
49  				"label", "body .juneau-view-action-menu .juneau-view-action-item")
50  			.waitFor("#t-body tbody .juneau-view-action-trigger")
51  			.click("#t-body tbody .juneau-view-action-trigger")
52  			.run();
53  	}
54  
55  	@Test void a01_bootsCleanly() {
56  		assertNoShellErrors(report.get("parity"));
57  	}
58  
59  	@Test void a02_rowActionLabelIsOnTheRow() {
60  		assertEquals("Close", queries(report.get("parity")).get("label"), () -> report.get("parity").toString());
61  	}
62  }
