36  /**
37   * Node-sandbox integration coverage for {@code juneau-console.js}: loads the real production
38   * shell into a hand-rolled DOM shim (no jsdom) per case, exercising the full {@code #juneau-page} contract surface -
39   * header/nav/cards/footer rendering, nav-depth and longest-prefix matching, every {@code E-JS-1}..{@code
40   * E-JS-14} loud-failure codes, custom card-type registration, the pending-card path, and the terminal-view bridge - plus a permanent regression
41   * pin for the {@code isSafeHref}/{@code isProtocolRelativeUrl} security hardening reviewed earlier (a
42   * scheme-bypass and a keydown-listener-leak bug, both already fixed in the shipped shell; this is the test that was
43   * missing).
44   *
45   * <p>
46   * Gated on {@code node} being on {@code PATH} (skipped otherwise - no {@code -Pjs-tests} required), mirroring
47   * {@link ViewsJs_ChromeMenu_Test}'s harness-probing structure.

51  	private static String resource(String name) throws IOException {
52  		try (var in = ConsoleChromeMixin.class.getResourceAsStream(name)) {
53  			assertNotNull(in, () -> "missing classpath resource: " + name);
54  			return new String(in.readAllBytes(), UTF_8);
55  		}
56  	}

58  	private static Map<?,?> report;

60  	@BeforeAll
61  	static void probeIfNodeAvailable() throws Exception {
62  		if (!nodeAvailable())
63  			return;
64  		var harness = locateHarness();
65  		if (harness == null)
66  			return;
67  		var shellFile = Files.createTempFile("juneau-console-", ".js");
68  		try {
69  			Files.writeString(shellFile, resource(ConsoleChromeMixin.CONSOLE_JS_RESOURCE), UTF_8);
70  			report = Json.to(runNode(harness, shellFile), Map.class);
71  		} finally {
72  			Files.deleteIfExists(shellFile);
73  		}
74  	}

76  	private static boolean nodeAvailable() {
77  		try {
78  			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
79  			if (!p.waitFor(5, TimeUnit.SECONDS)) {
80  				p.destroyForcibly();
81  				return false;
82  			}
83  			return p.exitValue() == 0;
84  		} catch (Exception e) {
85  			return false;
86  		}
87  	}

89  	private static Path locateHarness() {
90  		var basedir = System.getProperty("basedir");
91  		if (basedir != null) {
92  			var p = Path.of(basedir, "src/test/js/juneau-console.cjs");
93  			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
94  		}
95  		for (var rel : List.of(
96  			"src/test/js/juneau-console.cjs",
97  			"juneau-rest/juneau-console-views/src/test/js/juneau-console.cjs"
98  		)) {
99  			var p = Path.of(rel);
100 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
101 		}
102 		return null;
103 	}

105 	private static String runNode(Path harness, Path shell) throws Exception {
106 		var stdout = Files.createTempFile("juneau-console-stdout-", ".json");
107 		var stderr = Files.createTempFile("juneau-console-stderr-", ".txt");
108 		try {
109 			var pb = new ProcessBuilder(List.of("node", harness.toString(), shell.toString()))
110 				.redirectOutput(stdout.toFile())
111 				.redirectError(stderr.toFile());
112 			var p = pb.start();
113 			if (!p.waitFor(30, TimeUnit.SECONDS)) {
114 				p.destroyForcibly();
115 				fail("juneau-console.cjs did not finish within 30s; stderr:\n" + quietRead(stderr));
116 			}
117 			if (p.exitValue() != 0)
118 				fail("juneau-console.cjs exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr)
119 					+ "\nstdout:\n" + quietRead(stdout));
120 			return Files.readString(stdout, UTF_8);
121 		} finally {
122 			Files.deleteIfExists(stdout);
123 			Files.deleteIfExists(stderr);
124 		}
125 	}
126 
127 	private static String quietRead(Path p) {
128 		try { return Files.readString(p, UTF_8); }
129 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
130 	}

288 	 * href (not even one with a missing/empty {@code href}) &mdash; the harness mounts all 8 obfuscated-scheme and

290 	 * {@code JaVaScRiPt:}, {@code data:}, {@code vbscript:}, {@code //evil.example/x}, {@code /\evil.example/x}) as
291 	 * nav items keyed {@code rej0}..{@code rej7} in {@code hrefSafety.nav}.
292 	 */
