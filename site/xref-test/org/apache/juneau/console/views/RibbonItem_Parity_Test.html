28  /**
29   * Pins {@link RibbonItem#toMap()} to the shared ribbon corpus ({@code ribbon-corpus.json}), the same contract
30   * {@code juneau-ribbon.js} is tested against: for every corpus case, the builder-made list must equal the case's
31   * {@code ribbon}.  A case with no builder here fails, so a new corpus case cannot go unchecked.
32   */
33  class RibbonItem_Parity_Test extends TestBase {
34  
35  	private static RibbonItem dropped() {
36  		return RibbonItem.option("dropped-only").column("status").value("$eq(DROPPED)");
37  	}
38  
39  	private static RibbonItem pending() {
40  		return RibbonItem.option("pending").column("phase").value("$in(Waiting,\"Partially reviewed\")");
41  	}
42  
43  	private static RibbonItem done() {
44  		return RibbonItem.option("done").column("phase").value("$in(\"Ready to push\",Completed)");
45  	}
46  
47  	private static RibbonItem all() {
48  		return RibbonItem.option("all").title("All");
49  	}
50  
51  	private static RibbonItem phaseGroup() {
52  		return RibbonItem.optionGroup("phase", pending(), done());
53  	}

111 	private static List<Map<String,Object>> normalized(List<Map<String,Object>> ribbon) {
112 		// The corpus case still opens with the retired columnSearchToggle entry; the builder has no such item.
113 		return ribbon.stream().filter(e -> ! "columnSearchToggle".equals(e.get("type"))).toList();
114 	}
115 
116 	@Test void a01_everyCorpusCaseMatches() throws Exception {
117 		var cases = RibbonCorpus_Test.corpus().cases;
118 		assertFalse(cases.isEmpty());
119 		for (var c : cases) {
120 			var builder = BUILDERS.get(c.name);
121 			assertNotNull(builder, "No builder for corpus case '" + c.name + "'");
122 			var actual = builder.get().stream().map(i -> (Map<String,Object>)i.toMap()).toList();
123 			assertEquals(normalized(c.ribbon), actual, c.name);
124 		}
125 	}
126 
127 	@Test void a02_noStaleBuilders() throws Exception {
128 		var names = RibbonCorpus_Test.corpus().cases.stream().map(c -> c.name).collect(Collectors.toSet());
129 		assertEquals(names, BUILDERS.keySet());
130 	}
