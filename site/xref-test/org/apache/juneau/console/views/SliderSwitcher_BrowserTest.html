27  /**
28   * The slider switcher in Chromium, light and dark: no stray border on any button, an oval hover, an even inset for
29   * the thumb, and a thumb shadow that does not bleed past the track in the dark scheme.
30   */
31  @EnabledIfSystemProperty(named=ConsoleBrowserFixture.GATE, matches="true",
32  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -pl juneau-rest/juneau-console-views -am test` from the repo root")
33  class SliderSwitcher_BrowserTest extends org.apache.juneau.TestBase {
34  
35  	private static final List<String> SCHEMES = List.of("light", "dark");
36  
37  	private static final String BODY = """
38  		<div class="juneau-switcher-selector" data-juneau-switcher-style="slider">
39  		<div class="juneau-view-ribbon-group" role="tablist" id="g">
40  		<span class="juneau-switcher-thumb" id="th" aria-hidden="true" style="width:60px"></span>
41  		<button type="button" class="juneau-view-ribbon-btn" role="tab" id="b1" aria-selected="true">One</button><button type="button" class="juneau-view-ribbon-btn" role="tab" id="b2" aria-selected="false">Two</button><button type="button" class="juneau-view-ribbon-btn" role="tab" id="b3" aria-selected="false">Three</button></div></div>
42  		<div class="juneau-switcher-selector" data-juneau-switcher-style="slider">
43  		<div class="juneau-view-ribbon-group" role="tablist"><button type="button" class="juneau-view-ribbon-btn" role="tab" id="o1" aria-selected="true">Only</button></div></div>
44  		<div id="park" style="height:40px"></div>
45  		""";
46  
47  	private static final String PROBE = """
48  		<script>
49  		(function () {
50  			var PROPS = ['border-top-width', 'border-right-width', 'border-bottom-width', 'border-left-width', 'border-top-color',
51  				'border-top-left-radius', 'border-top-right-radius', 'border-bottom-left-radius', 'border-bottom-right-radius', 'background-color', 'box-shadow'];
52  			function snap(id) {
53  				var el = document.getElementById(id), cs = getComputedStyle(el), o = {};
54  				PROPS.forEach(function (p) { o[p] = cs.getPropertyValue(p); });
55  				var r = el.getBoundingClientRect(); o.top = r.top; o.bottom = r.bottom; o.height = r.height;
56  				return o;
57  			}
58  			var rec = {};
59  			window.__rec = function (key, id) { rec[key] = snap(id); };
60  			window.__probe = { toJSON: function () {
61  				var g = document.getElementById('g').getBoundingClientRect();
62  				return { rest: { b1: snap('b1'), b2: snap('b2'), b3: snap('b3'), o1: snap('o1'), th: snap('th') },
63  					hover: rec, group: { top: g.top, bottom: g.bottom } };
64  			} };
65  		})();
66  		</script>
67  		""";
68  
69  	private static Map<String,Map<String,Object>> report;
70  
71  	private static String page(String scheme) {
72  		var attrs = "data-jc-scheme=\"" + scheme + "\" data-jc-scheme-pin=\"" + scheme + "\"";
73  		return "<!DOCTYPE html>\n<html lang=\"en\" " + attrs + "><head><meta charset=\"utf-8\"><link rel=\"icon\" href=\"data:,\">\n"
74  			+ "<link rel=\"stylesheet\" href=\"" + CHROME_CSS + "\">\n<link rel=\"stylesheet\" href=\"" + DARK_CSS + "\">\n"
75  			+ "<link rel=\"stylesheet\" href=\"" + THEME_CSS + "\">\n<link rel=\"stylesheet\" href=\"" + VIEWS_CSS + "\">\n"
76  			+ "</head><body>\n" + BODY + PROBE + "</body></html>\n";
77  	}
78  
79  	@BeforeAll
80  	static void run() throws Exception {
81  		var f = ConsoleBrowserFixture.create("slider-switcher");
82  		for (var s : SCHEMES) {
83  			f.page(s, ORIGIN + "/" + s, page(s));
84  			for (var id : List.of("b1", "b3", "o1"))
85  				f.hover("#" + id).eval("window.__rec('" + id + "', '" + id + "')");
86  			f.hover("#park");
87  		}
88  		report = f.run();
89  	}
90  
91  	private static Map<String,Object> probe(String scheme) {
92  		var r = report.get(scheme);
93  		assertEquals(List.of(), r.get("pageErrors"), () -> scheme + ": page errors");
94  		return ConsoleBrowserFixture.probe(r);
95  	}
96  
97  	@SuppressWarnings("unchecked")
98  	private static Map<String,Object> sub(Map<String,Object> p, String a, String b) {
99  		return (Map<String,Object>)((Map<String,Object>)p.get(a)).get(b);
100 	}
101 
102 	@Test void a01_noButtonCarriesABorder() {
103 		for (var s : SCHEMES) {
104 			var p = probe(s);
105 			for (var id : List.of("b1", "b2", "b3", "o1"))
106 				for (var side : List.of("top", "right", "bottom", "left"))
107 					assertEquals("0px", sub(p, "rest", id).get("border-" + side + "-width"), s + ": " + id + " " + side);
108 		}
109 	}
110 
111 	@Test void a02_hoverIsAnOvalWithNoBorder() {
112 		for (var s : SCHEMES) {
113 			var p = probe(s);
114 			for (var id : List.of("b1", "b3", "o1")) {
115 				var h = sub(p, "hover", id);
116 				assertNotEquals("rgba(0, 0, 0, 0)", h.get("background-color"), s + ": " + id + " hover wash");
117 				for (var c : List.of("top-left", "top-right", "bottom-left", "bottom-right"))
118 					assertTrue(Double.parseDouble(String.valueOf(h.get("border-" + c + "-radius")).replace("px", "")) >= 9, s + ": " + id + " " + c + " " + h.get("border-" + c + "-radius"));
119 				for (var side : List.of("top", "right", "bottom", "left"))
120 					assertEquals("0px", h.get("border-" + side + "-width"), s + ": " + id + " hover " + side);
121 			}
122 		}
123 	}
124 
125 	@Test void a03_thumbIsInsetEvenlyTopAndBottom() {
126 		for (var s : SCHEMES) {
127 			var p = probe(s);
128 			@SuppressWarnings("unchecked") var g = (Map<String,Object>)p.get("group");
129 			var th = sub(p, "rest", "th");
130 			var top = ((Number)th.get("top")).doubleValue() - ((Number)g.get("top")).doubleValue();
131 			var bottom = ((Number)g.get("bottom")).doubleValue() - ((Number)th.get("bottom")).doubleValue();
132 			assertEquals(top, bottom, 0.5, s + ": thumb inset");
133 			assertTrue(top >= 2.5, s + ": thumb inset " + top);
134 		}
135 	}
136 
137 	@Test void a04_darkThumbShadowDoesNotBleedPastTheTrack() {
138 		var light = String.valueOf(sub(probe("light"), "rest", "th").get("box-shadow"));
139 		var dark = String.valueOf(sub(probe("dark"), "rest", "th").get("box-shadow"));
140 		assertNotEquals("none", light);
141 		assertEquals("none", dark, "dark thumb shadow");
142 	}
