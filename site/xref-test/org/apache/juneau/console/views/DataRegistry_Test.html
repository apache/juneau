27  /** The DataRegistry {@code juneau-data.js} under the views DOM shim.  Runs when node is on PATH. */
28  class DataRegistry_Test extends TestBase {
29  	@Test void a01_registry() {
30  		var r = NodeHarness.report("data-registry.cjs", ViewsMixin.BUS_JS_RESOURCE, ViewsMixin.DATA_JS_RESOURCE);
31  		assumeTrue(r != null, "node (or data-registry.cjs) not available - JS layer skipped");
32  		assertEquals(1, ((Number)r.get("a01_put_returnsVersion1")).intValue());
33  		for (var k : List.of("a02_get_sameColumnsByReference", "a03_previousHeldDuringSwap", "a04_twoBackDropped",
34  				"a05_unknownSource_null", "a06_partial", "a07_partialCleared_null", "a09_release"))
35  			assertEquals(true, r.get(k), k);
36  		assertEquals(List.of(true, false, false, false), r.get("a08_isHandle"));
37  	}
38  }
