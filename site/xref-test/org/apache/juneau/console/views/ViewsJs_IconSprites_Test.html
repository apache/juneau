33  /**
34   * Always-on coverage for {@code juneau-icons.js}'s page-level sprite layering API.
35   *
36   * <p>The runtime resolves every chrome glyph through up to three sprite layers - per-icon override, then the
37   * app's replacement sprite, then Juneau's shipped set - merged once into a single in-document
38   * {@code <symbol id="juneau-sym-{stem}">} sprite so hosts can {@code <use href="#juneau-sym-{stem}"/>}.  This
39   * test pins the module boundary with always-on source-shape assertions, then exercises the actual resolution,
40   * failure, and late-registration behavior in a Node VM sandbox (skipped when Node is absent) over the REAL
41   * shipped sprite so the fixtures cannot drift from what ships.
42   */
43  class ViewsJs_IconSprites_Test extends TestBase {

123 	@BeforeAll
124 	static void probeIfNodeAvailable() throws Exception {
125 		if (!nodeAvailable())
126 			return;
127 		var harness = locateHarness();
128 		if (harness == null)
129 			return;
130 		var iconsFile = Files.createTempFile("juneau-icons-", ".js");
131 		var symbolsFile = Files.createTempFile("juneau-symbols-", ".svg");
132 		try {
133 			Files.writeString(iconsFile, iconsJs(), UTF_8);
134 			Files.writeString(symbolsFile, symbolsSvg(), UTF_8);
135 			report = Json.to(runNode(harness, iconsFile, symbolsFile), Map.class);
136 		} finally {
137 			Files.deleteIfExists(iconsFile);
138 			Files.deleteIfExists(symbolsFile);
139 		}
140 	}

142 	private static boolean nodeAvailable() {
143 		try {
144 			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
145 			if (!p.waitFor(5, TimeUnit.SECONDS)) {
146 				p.destroyForcibly();
147 				return false;
148 			}
149 			return p.exitValue() == 0;
150 		} catch (Exception e) {
151 			return false;
152 		}
153 	}

155 	private static Path locateHarness() {
156 		var basedir = System.getProperty("basedir");
157 		if (basedir != null) {
158 			var p = Path.of(basedir, "src/test/js/icons.cjs");
159 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
160 		}
161 		for (var rel : List.of(
162 			"src/test/js/icons.cjs",
163 			"juneau-rest/juneau-console-views/src/test/js/icons.cjs"
164 		)) {
165 			var p = Path.of(rel);
166 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
167 		}
168 		return null;
169 	}

171 	private static String runNode(Path harness, Path iconsJs, Path symbolsSvg) throws Exception {
172 		var stdout = Files.createTempFile("icons-stdout-", ".json");
173 		var stderr = Files.createTempFile("icons-stderr-", ".txt");
174 		try {
175 			var pb = new ProcessBuilder(List.of("node", harness.toString(), iconsJs.toString(), symbolsSvg.toString()))
176 				.redirectOutput(stdout.toFile())
177 				.redirectError(stderr.toFile());
178 			var p = pb.start();
179 			if (!p.waitFor(30, TimeUnit.SECONDS)) {
180 				p.destroyForcibly();
181 				fail("icons.cjs did not finish within 30s; stderr:\n" + quietRead(stderr));
182 			}
183 			if (p.exitValue() != 0)
184 				fail("icons.cjs exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr)
185 					+ "\nstdout:\n" + quietRead(stdout));
186 			return Files.readString(stdout, UTF_8);
187 		} finally {
188 			Files.deleteIfExists(stdout);
189 			Files.deleteIfExists(stderr);
190 		}
191 	}
192 
193 	private static String quietRead(Path p) {
194 		try { return Files.readString(p, UTF_8); }
195 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
196 	}
