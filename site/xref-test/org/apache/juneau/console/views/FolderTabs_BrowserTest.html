27  /**
28   * The folder-tab look in Chromium, light and dark: the selected tab's fill, accent top bar and accent edge, the
29   * white unselected tabs, labels that share one top, the padded panel, and the shorter row (including the compact
30   * chart tabs).
31   */

34  class FolderTabs_BrowserTest extends org.apache.juneau.TestBase {

50  	private static final String PROBE = """
51  		<script>
52  		(function () {
53  			var PROPS = ['background-color', 'color', 'border-top-color', 'border-top-width', 'border-bottom-width', 'border-bottom-color',
54  				'border-left-color', 'font-weight', 'padding-top', 'padding-left', 'box-shadow'];
55  			function snap(id) {
56  				var el = document.getElementById(id), cs = getComputedStyle(el), o = {};
57  				PROPS.forEach(function (p) { o[p] = cs.getPropertyValue(p); });
58  				var r = el.getBoundingClientRect(), rg = document.createRange();
59  				rg.selectNodeContents(el);
60  				o.height = r.height; o.top = r.top; o.labelTop = rg.getBoundingClientRect().top;
61  				return o;
62  			}
63  			function tok(expr, prop) {
64  				var t = document.getElementById('tokprobe'); t.style[prop] = expr;
65  				return getComputedStyle(t)[prop];
66  			}
67  			window.__probe = { toJSON: function () {
68  				var sel = document.getElementById('t1').getBoundingClientRect(), pn = document.getElementById('pn').getBoundingClientRect();
69  				var hit = document.elementFromPoint(sel.left + sel.width / 2, pn.top + 0.5);
70  				return {
71  					cover: { selBottom: sel.bottom, paneTop: pn.top, hit: hit ? hit.id : null },
72  					tabs: { t1: snap('t1'), t2: snap('t2'), t3: snap('t3'), c1: snap('c1'), c2: snap('c2') },
73  					pane: snap('pn'),
74  					tok: {
75  						accent: tok('var(--jc-accent)', 'color'), border: tok('var(--jc-border)', 'color'), card: tok('var(--jc-card-bg)', 'color'),
76  						wash: tok('var(--jc-tab-selected-bg)', 'backgroundColor'), link: tok('var(--jc-link)', 'color')
77  					}
78  				};
79  			} };
80  		})();
81  		</script>
82  		""";

86  	private static String page(String scheme) {
87  		var attrs = "data-jc-scheme=\"" + scheme + "\" data-jc-scheme-pin=\"" + scheme + "\"";
88  		return "<!DOCTYPE html>\n<html lang=\"en\" " + attrs + "><head><meta charset=\"utf-8\"><link rel=\"icon\" href=\"data:,\">\n"
89  			+ "<link rel=\"stylesheet\" href=\"" + CHROME_CSS + "\">\n<link rel=\"stylesheet\" href=\"" + DARK_CSS + "\">\n"
90  			+ "<link rel=\"stylesheet\" href=\"" + THEME_CSS + "\">\n<link rel=\"stylesheet\" href=\"" + VIEWS_CSS + "\">\n"
91  			+ "</head><body>\n" + BODY + PROBE + "</body></html>\n";
92  	}

94  	@BeforeAll
95  	static void run() throws Exception {
96  		var f = ConsoleBrowserFixture.create("folder-tabs");
97  		for (var s : SCHEMES)
98  			f.page(s, ORIGIN + "/" + s, page(s));
99  		report = f.run();
100 	}

102 	@SuppressWarnings("unchecked")
103 	private static Map<String,Object> probe(String scheme) {
104 		var r = report.get(scheme);
105 		assertEquals(List.of(), r.get("pageErrors"), () -> scheme + ": page errors");
106 		return ConsoleBrowserFixture.probe(r);
107 	}

109 	@SuppressWarnings("unchecked")
110 	private static Map<String,Object> tab(Map<String,Object> p, String id) {
111 		return (Map<String,Object>)((Map<String,Object>)p.get("tabs")).get(id);
112 	}
113 
114 	@SuppressWarnings("unchecked")
115 	private static String tok(Map<String,Object> p, String n) {
116 		return String.valueOf(((Map<String,Object>)p.get("tok")).get(n));
117 	}
118 
119 	private static double num(Map<String,Object> m, String k) {
120 		return ((Number)m.get(k)).doubleValue();
121 	}
122 
123 	@Test void a01_selectedTabIsWashFilledWithAccentBarAndEdge() {
124 		for (var s : SCHEMES) {
125 			var p = probe(s);
126 			var t = tab(p, "t1");
127 			assertEquals(tok(p, "wash"), t.get("background-color"), s + ": selected fill");
128 			assertEquals("600", t.get("font-weight"), s);
129 			assertEquals("3px", t.get("border-top-width"), s);
130 			assertEquals(tok(p, "accent"), t.get("border-top-color"), s);
131 			assertEquals(tok(p, "accent"), t.get("border-left-color"), s);
132 			assertEquals("rgba(0, 0, 0, 0)", t.get("border-bottom-color"), s + ": selected tab has no visible bottom edge");
133 			assertEquals("none", t.get("box-shadow"), s + ": the bar is the top border, not an inset shadow");
134 		}
135 	}
136 
137 	@Test void a02_unselectedTabsAreWhiteWithLightEdgeAndNeutralTopBorder() {
138 		for (var s : SCHEMES) {
139 			var p = probe(s);
140 			for (var id : List.of("t2", "t3")) {
141 				var t = tab(p, id);
142 				assertEquals(tok(p, "card"), t.get("background-color"), s + ": " + id + " fill");
143 				assertEquals(tok(p, "border"), t.get("border-left-color"), s + ": " + id + " edge");
144 				assertEquals("3px", t.get("border-top-width"), s + ": " + id + " reserves the bar");
145 				assertEquals(tok(p, "border"), t.get("border-top-color"), s + ": " + id + " has a neutral top border");
146 			}
147 		}
148 	}
149 
150 	@Test void a06_selectedTabCoversThePanelTopBorder() {
151 		for (var s : SCHEMES) {
152 			@SuppressWarnings("unchecked") var c = (Map<String,Object>)probe(s).get("cover");
153 			assertTrue(num(c, "selBottom") >= num(c, "paneTop") + 1 - 0.01, s + ": selected tab overlaps the pane's 1px top border " + c);
154 			assertEquals("t1", c.get("hit"), s + ": the selected tab, not the pane border, paints at the pane's top edge");
155 		}
156 	}
157 
158 	@Test void a03_labelsShareOneTopAndUnselectedHeightsMatch() {
159 		for (var s : SCHEMES) {
160 			var p = probe(s);
161 			assertEquals(num(tab(p, "t1"), "labelTop"), num(tab(p, "t2"), "labelTop"), 0.5, s + ": selected vs unselected label");
162 			assertEquals(num(tab(p, "t2"), "labelTop"), num(tab(p, "t3"), "labelTop"), 0.5, s);
163 			assertEquals(num(tab(p, "t2"), "height"), num(tab(p, "t3"), "height"), 0.5, s);
164 		}
165 	}
166 
167 	@Test void a04_panelIsCardFilledAccentBorderedAndPadded() {
168 		for (var s : SCHEMES) {
169 			var p = probe(s);
170 			@SuppressWarnings("unchecked") var pane = (Map<String,Object>)p.get("pane");
171 			assertEquals(tok(p, "card"), pane.get("background-color"), s);
172 			assertEquals(tok(p, "accent"), pane.get("border-top-color"), s);
173 			assertEquals("12px", pane.get("padding-top"), s);
174 			assertEquals("12px", pane.get("padding-left"), s);
175 		}
176 	}
177 
178 	@Test void a05_rowIsShorter() {
179 		for (var s : SCHEMES) {
180 			var p = probe(s);
181 			assertEquals("4px", tab(p, "t2").get("padding-top"), s + ": default tab padding");
182 			assertEquals("2px", tab(p, "c2").get("padding-top"), s + ": compact chart tab padding");
183 		}
