17  package org.apache.juneau.console.views;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  import static org.junit.jupiter.api.Assumptions.*;
21  
22  import java.util.*;
23  
24  import org.apache.juneau.*;
25  import org.apache.juneau.marshall.marshaller.*;
26  import org.junit.jupiter.api.*;
27  
28  /**
29   * Node check that the {@code window} and {@code ids} built-ins are configured from a view's own {@code urlState}
30   * entry, with nothing registered by hand, end to end from {@link TableSpec} output to the live request.
31   */
32  class ViewsJs_UrlStateBuiltinConfig_Test extends TestBase {
33  
34  	private static Map<?,?> report() {
35  		var urlState = TableSpec.create("x")
36  			.urlState("ids", Map.of("param", "pid", "max", 2))
37  			.urlState("window", Map.of("params", Map.of("start", "from", "end", "to")))
38  			.toCardBody().get("urlState");
39  		var r = NodeHarness.report(Map.of("URLSTATE_VIEW_JSON", Json.of(urlState)), "urlstate-builtin-config.cjs", ViewsMixin.RENDERS_JS_RESOURCE, ViewsMixin.VIEWS_JS_RESOURCE, ViewsMixin.URLSTATE_JS_RESOURCE);
40  		assumeTrue(r != null, "node (or urlstate-builtin-config.cjs) not available - behavioral layer skipped");
41  		return r;
42  	}
43  
44  	@Test void a01_builtinsAreRegisteredOnlyByTheViewConfig() {
45  		var r = report();
46  		assertEquals("false|false", r.get("registeredBefore"));
47  		assertEquals("true|true", r.get("registeredAfter"));
48  	}
49  
50  	@Test void a02_windowConfigRenamesTheRequestParams() {
51  		var r = report();
52  		assertEquals("2026-10-01T00:00:00Z", r.get("windowStart"));
53  		assertEquals("2026-10-02T00:00:00Z", r.get("windowEnd"));
54  		var params = (String)r.get("requestParams");
55  		assertTrue(params.contains("\"from\":\"2026-10-01T00:00:00Z\"") && params.contains("\"to\":\"2026-10-02T00:00:00Z\""), params);
56  	}
57  
58  	@Test void a03_idsConfigRenamesTheParamAndCapsAtMax() {
59  		var r = report();
60  		assertEquals("[\"a\",\"b\"]", r.get("seededIds"));
61  		assertEquals(true, r.get("idsTruncated"));
62  		assertEquals("Showing the first 2 ids", r.get("idsCapMessage"));
63  		var params = (String)r.get("requestParams");
64  		assertTrue(params.contains("\"pid\":\"a,b\""), params);
65  	}
66  }
