27  /**
28   * The row-id containment a row-detail region depends on: <b>{@code expandDetailRow} stamps the row's id onto the
29   * detail panel</b>.
30   *
31   * <h5 class='section'>The gap</h5>
32   * <p>
33   * DataTables inserts a child row as a <b>sibling</b> {@code <tr>} of the row it belongs to, never as a descendant of
34   * it. The detail panel &mdash; and every region container cloned into it &mdash; therefore sits <i>outside</i> the
35   * {@code <tr>} carrying {@code data-juneau-row-id}. Since {@code juneau-regions.js}'s {@code readIds} resolves every
36   * enclosing identity with {@code el.closest("[data-juneau-row-id]")}, a panel region would read a null row id, and
37   * {@code substituteDeclaredUrl} would then refuse the {@code {id}} substitution outright &mdash; or, for an author's own
38   * populate doing its own interpolation, build a literal {@code /.../null/...} path and collect a 400 from the
39   * server's path-param resolver. Both surface to a user as an empty or broken region body.
40   *
41   * <h5 class='section'>Why behavioral rather than a source-shape pin</h5>
42   * <p>
43   * The claim is not "a {@code setAttribute} call exists" &mdash; it is that a region inside the panel resolves the
44   * <b>right</b> id and fetches the <b>right</b> url. That spans two runtimes ({@code expandDetailRow} in
45   * {@code juneau-views.js} writes the stamp; {@code readIds}/{@code substituteDeclaredUrl} in {@code juneau-regions.js}
46   * read it), so only running both against one DOM proves they agree. The harness drives the real production path:
47   * {@code initDetailsExpander}'s delegated click &rarr; {@code toggleDetailRow} &rarr; {@code expandDetailRow} &rarr;
48   * {@code NS.regions.enrolIn(panel)}.
49   *
50   * <h5 class='section'>The claims are paired with controls</h5>
51   * <p>
52   * "The region resolved the right id" is a claim a <b>broken harness</b> can satisfy by accident &mdash; most simply
53   * by building a fixture where the panel was never a sibling in the first place, making the stamp a redundant
54   * restatement of something {@code closest} would have found anyway. So the harness also proves the containment gap is
55   * real ({@code gap_*}), and re-runs the identical enrolment against a panel whose stamp has been removed
56   * ({@code unstamped_*}) to show it degrades exactly as described. The pair is the evidence; either half alone is not.
57   *
58   * <p>
59   * Runs whenever {@code node} is on {@code PATH} and is skipped (not failed) otherwise, so it exercises on any
60   * developer or CI machine with node without becoming a hard toolchain dependency.
61   */
62  class ViewsJs_DetailRegionRowId_Test extends TestBase {
63  
64  	private static final String HARNESS = "detail-region-row-id.cjs";
65  
66  	private static Map<?,?> report() {
67  		var r = RegionsHarness.report(HARNESS);
68  		assumeTrue(r != null, "node not available or " + HARNESS + " not found - behavioral layer skipped");
69  		assertEquals(true, r.get("hasFixture"), () -> "the harness never built a detail panel: " + r);
70  		return r;
71  	}
72  
73  	private static void assertAllTrue(Map<?,?> r, String... keys) {
74  		for (var k : keys)
75  			assertEquals(true, r.get(k), () -> k + " -> " + r.get(k) + " in " + r);
76  	}
77  
78  	/**
79  	 * The stamp itself, and the read that depends on it: the panel carries the row's id, and a region cloned into the
80  	 * panel resolves that id the way every consumer resolves it &mdash; by {@code closest(...)}.
81  	 */
82  	@Test void a01_panelCarriesTheRowIdAndItsRegionResolvesIt() {
83  		assertAllTrue(report(), "panelCarriesRowId", "regionResolvesRowIdByClosest");
84  	}
85  
86  	/**
87  	 * End to end, and the part a user would have seen break: the region's declared {@code {id}} url is fetched with
88  	 * the real id substituted in &mdash; never with a literal {@code {id}} still in the path, and never as
89  	 * {@code /.../null/...}.
90  	 */
91  	@Test void a02_regionFetchesTheSubstitutedUrl() {
92  		assertAllTrue(report(),
93  			"regionFetchedSubstitutedUrl", "regionNeverFetchedNull", "regionNeverFetchedLiteralTemplate");
94  	}
95  
96  	/**
97  	 * The fixture really did drive the production expand path rather than a stub of it: the panel issued its OWN
98  	 * detail GET, substituted, which only {@code expandDetailRow} does.
99  	 */
100 	@Test void a03_theFixtureDroveTheRealExpandPath() {
101 		assertAllTrue(report(), "panelFetchedItsOwnDetailUrl");
102 	}
103 
104 	/**
105 	 * CONTROL 1 &mdash; the containment gap is real. The panel is genuinely not inside the row's {@code <tr>}, which
106 	 * is what makes the stamp load-bearing instead of redundant. Were this to start failing, {@code a01} would be
107 	 * passing for a reason that has nothing to do with the stamp.
108 	 */
109 	@Test void a04_control_panelIsNotADescendantOfItsRow() {
110 		assertAllTrue(report(), "gap_panelIsNotInsideTheRowTr", "gap_panelIsASiblingSubtree");
111 	}
112 
113 	/**
114 	 * CONTROL 2 &mdash; removing the stamp degrades the same enrolment: the region resolves no row id at all, and the
115 	 * declared fetch it would have issued is refused rather than sent. This is the observable difference the stamp
116 	 * makes, and without it a harness whose region never fetched at all would look identical to a working one.
117 	 */
118 	@Test void a05_control_withoutTheStampTheRegionResolvesNothing() {
119 		assertAllTrue(report(), "unstamped_resolvesNoRowId", "unstamped_neverFetchedSubstitutedUrlTwice");
120 	}
121 }
