28  /**
29   * The header theme selector: it renders only when the page opts in, switches the palette link, persists the pick
30   * across a reload, leaves the dark scheme layer in place, and clears the pick through the "Page default" entry.
31   */
32  @SuppressWarnings({
33  	"unchecked" // The probe report is parsed into raw Maps and Lists.
34  })
35  @EnabledIfSystemProperty(named=ConsoleBrowserFixture.GATE, matches="true",
36  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -f juneau-rest/juneau-console-views/pom.xml test`")
37  class ThemeSelector_BrowserTest extends TestBase {
38  
39  	private static final String KEY = "juneau-console:theme";
40  	private static final String ON = "data-jc-theme-selector data-jc-themes=\"open red gray\"";
41  	private static final String SIMPLE = """
42  		{"contractVersion":"1","title":"S","header":{"title":"T"},"cards":[]}
43  		""";
44  
45  	private static final String EVENTS = """
46  		<script>
47  		window.__events = [];
48  		document.addEventListener('juneau:theme-changed', function (e) { window.__events.push(e.detail); });
49  		</script>
50  		""";
51  
52  	private static final String PROBE = """
53  		<script>
54  		window.__probe = { toJSON: function () {
55  			var h = document.documentElement, s = document.querySelector('.jc-theme-select select'), l = document.querySelector('link[data-jc-theme-link]'), out = {};
56  			out.hasControl = !!s;
57  			out.value = s ? s.value : null;
58  			out.options = s ? Array.prototype.map.call(s.options, function (o) { return o.value; }) : null;
59  			out.href = l ? l.getAttribute('href') : null;
60  			out.scheme = h.getAttribute('data-jc-scheme');
61  			out.dark = !!document.querySelector('link[href*="juneau-theme-dark.css"]');
62  			out.events = window.__events.slice();
63  			try { out.stored = localStorage.getItem('juneau-console:theme'); } catch (x) { out.stored = 'ERR'; }
64  			return out;
65  		} };
66  		</script>
67  		""";
68  
69  	/** Picks a value and resolves once the palette sheet has loaded and the change was announced (or after a second). */
70  	private static final String PICK = "new Promise(function(r){document.addEventListener('juneau:theme-changed',function(){r()},{once:true});setTimeout(r,3500);"
71  		+ "var s=document.querySelector('.jc-theme-select select');s.value='%s';s.dispatchEvent(new Event('change',{bubbles:true}))})";
72  
73  	private static Map<String,Map<String,Object>> report;
74  
75  	private static String page(String attrs) {
76  		var html = contractPage(SIMPLE, "", EVENTS, PROBE, attrs, true);
77  		return html.replace("href=\"" + THEME_CSS + "\"", "href=\"" + THEME_CSS + "\"" + (attrs.contains("theme-selector") ? " data-jc-theme-link" : ""));
78  	}
79  
80  	@BeforeAll
81  	static void runPages() throws Exception {
82  		var fx = ConsoleBrowserFixture.create("theme-selector");
83  		for (var n : List.of("red", "gray"))
84  			fx.asset("/juneau-console/juneau-theme-" + n + ".css", "html:root{}", "text/css");
85  		report = fx
86  			.page("off", ORIGIN + "/off", page(""))
87  			.page("on", ORIGIN + "/on", page(ON))
88  			.page("pick", ORIGIN + "/pick", page(ON)).eval(PICK.formatted("red"))
89  			.page("stored", ORIGIN + "/stored", page(ON)).storage(Map.of(KEY, "gray"))
90  			.page("garbage", ORIGIN + "/garbage", page(ON)).storage(Map.of(KEY, "nope"))
91  			.page("clear", ORIGIN + "/clear", page(ON)).storage(Map.of(KEY, "gray")).eval(PICK.formatted(""))
92  			.page("darkPick", ORIGIN + "/darkPick", page(ON + " data-jc-scheme=\"dark\" data-jc-scheme-pin=\"dark\"")).eval(PICK.formatted("red"))
93  			.run();
94  	}
95  
96  	private static Map<String,Object> p(String name) {
97  		var r = report.get(name);
98  		assertNoShellErrors(r);
99  		return probe(r);
100 	}
101 
102 	@Test void a01_flagOffMeansNoControl() {
103 		assertEquals(false, p("off").get("hasControl"));
104 	}
105 
106 	@Test void a02_controlListsPageDefaultThenThemes() {
107 		var pr = p("on");
108 		assertEquals(true, pr.get("hasControl"));
109 		assertEquals(List.of("", "open", "red", "gray"), pr.get("options"));
110 		assertEquals("", pr.get("value"));
111 		assertEquals(THEME_CSS, pr.get("href"));
112 	}
113 
114 	@Test void a03_pickSwitchesPaletteAndStores() {
115 		var pr = p("pick");
116 		assertEquals("/juneau-console/juneau-theme-red.css", pr.get("href"));
117 		assertEquals("red", pr.get("stored"));
118 		assertEquals("red", pr.get("value"));
119 		assertEquals(List.of(Map.of("scheme", "light", "theme", "red")), pr.get("events"));
120 	}
121 
122 	@Test void a04_storedPickAppliesOnLoad() {
123 		var pr = p("stored");
124 		assertEquals("/juneau-console/juneau-theme-gray.css", pr.get("href"));
125 		assertEquals("gray", pr.get("value"));
126 	}
127 
128 	@Test void a05_unknownStoredPickIgnored() {
129 		var pr = p("garbage");
130 		assertEquals(THEME_CSS, pr.get("href"));
131 		assertEquals("", pr.get("value"));
132 	}
133 
134 	@Test void a06_pageDefaultClearsThePick() {
135 		var pr = p("clear");
136 		assertEquals(THEME_CSS, pr.get("href"));
137 		assertNull(pr.get("stored"));
138 		assertEquals("", pr.get("value"));
139 	}
140 
141 	@Test void a07_darkOverlayStillApplies() {
142 		var pr = p("darkPick");
143 		assertEquals("dark", pr.get("scheme"));
144 		assertEquals(true, pr.get("dark"));
145 		assertEquals("/juneau-console/juneau-theme-red.css", pr.get("href"));
146 	}
147 }
