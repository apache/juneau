29  /**
30   * Guards the shared chrome scale contract on the served {@code juneau-views.css}: the steps are declared once
31   * each, and no surface writes a literal that duplicates one.
32   *
33   * <p>
34   * Driven by {@link ChromeScaleScanner}, which carries the contract's own step table and the reasoning behind
35   * what it does and does not evaluate.
36   */
37  class ChromeScale_ContractScan_Test extends TestBase {
38  
39  	private static String viewsCss() throws IOException {
40  		try (var in = ChromeScale_ContractScan_Test.class.getResourceAsStream("/org/apache/juneau/views/juneau-views.css")) {
41  			assertNotNull(in, "juneau-views.css not found on the test classpath");
42  			return new String(in.readAllBytes(), StandardCharsets.UTF_8);
43  		}
44  	}
45  
46  	//-----------------------------------------------------------------------------------------------------------------
47  	// a) Every step is declared exactly once, at the value the contract states
48  	//-----------------------------------------------------------------------------------------------------------------
49  
50  	@Test void a01_everyStepIsDeclaredExactlyOnce() throws IOException {
51  		var problems = ChromeScaleScanner.checkDeclaredExactlyOnce(viewsCss());
52  		assertTrue(problems.isEmpty(), () -> "chrome scale declaration problems:\n  " + String.join("\n  ", problems));
53  	}
54  
55  	/**
56  	 * The ladders are declared whole, so a step with no consumer today is still declared. Asserting the count
57  	 * pins that: a future edit that prunes an "unused" step has to come through this test.
58  	 */
59  	@Test void a02_ladderIsDeclaredWhole_includingStepsWithNoConsumerYet() throws IOException {
60  		var css = viewsCss();
61  		assertTrue(css.contains("--jc-space-5: 24px;"), "the spacing ladder is declared whole, unused steps included");
62  		assertTrue(css.contains("--jc-space-6: 32px;"), "the spacing ladder is declared whole, unused steps included");
63  		assertSize(17, ChromeScaleScanner.scale());
64  	}
65  
66  	//-----------------------------------------------------------------------------------------------------------------
67  	// b) No literal duplicates a declared step
68  	//-----------------------------------------------------------------------------------------------------------------
69  
70  	@Test void b01_noLiteralDuplicatesADeclaredStep() throws IOException {
71  		var r = ChromeScaleScanner.scan(viewsCss());
72  		assertTrue(r.violations().isEmpty(), () -> "literals duplicating a declared step:\n  "
73  			+ String.join("\n  ", r.violations()));
74  	}
75  
76  	/**
77  	 * A scanner that silently stops matching reads as a passing test, which is worse than no scanner. The scan
78  	 * must actually reach a meaningful number of declarations on the properties the scale covers.
79  	 */
80  	@Test void b02_scanIsNotVacuous() throws IOException {
81  		var r = ChromeScaleScanner.scan(viewsCss());
82  		assertTrue(r.checked().size() > 20, () -> "scan examined only " + r.checked().size()
83  			+ " declarations - it has probably stopped matching rather than found a clean file");
84  	}
85  
86  	/**
87  	 * The recorded-exception table is a ratchet, not an allowlist that may rot: an entry matching nothing means
88  	 * the declaration it excused has moved or been fixed, and the entry must go rather than sit there quietly
89  	 * widening what the scan will tolerate.
90  	 */
91  	@Test void b03_everyRecordedExceptionStillMatchesSomething() throws IOException {
92  		var r = ChromeScaleScanner.scan(viewsCss());
93  		var stale = ChromeScaleScanner.recordedLiterals().stream()
94  			.filter(x -> !r.matchedRecords().contains(x))
95  			.map(x -> x.selector() + " { " + x.property() + ": " + x.value() + " }")
96  			.toList();
97  		assertTrue(stale.isEmpty(), () -> "recorded exceptions that no longer match any declaration - remove them:\n  "
98  			+ String.join("\n  ", stale));
99  	}
100 
101 	/** Every recorded exception states why it is not spending a step; a bare entry is not auditable. */
102 	@Test void b04_everyRecordedExceptionCarriesAReason() {
103 		ChromeScaleScanner.recordedLiterals()
104 			.forEach(x -> assertFalse(x.reason() == null || x.reason().isBlank(),
105 				() -> "recorded exception with no reason: " + x.selector() + " { " + x.property() + " }"));
106 	}

108 	/**
109 	 * Proves the scan can actually fail. Without this, (b01) passing tells you nothing about whether the check
110 	 * works or merely never fires.
111 	 */
112 	@Test void b05_syntheticDuplicateLiteralIsCaught() {
113 		var r = ChromeScaleScanner.scan(".synthetic-surface { height: 31px; }");
114 		assertSize(() -> "expected the synthetic duplicate to be caught: " + r.violations(), 1, r.violations());
115 		assertTrue(r.violations().get(0).contains("--jc-chrome-control-height"), r.violations().get(0));
116 	}
117 
118 	/**
119 	 * An {@code em} value must never be treated as a step match. Its computed value depends on a font-size
120 	 * context this file does not have one of, and on form controls that context is a UA/system font that differs
121 	 * by browser - so an engine-specific computation would fail on one platform and pass on another.
122 	 */
123 	@Test void b06_emValuesAreNeverTreatedAsAStepMatch() {
124 		// 0.9em computes to exactly 12px against Chrome's form-control font and to nothing near a step elsewhere.
125 		var r = ChromeScaleScanner.scan(".synthetic-toggle::after { width: 0.9em; height: 0.9em; }");
126 		assertTrue(r.violations().isEmpty(), () -> "an em value was evaluated as a step match: " + r.violations());
127 	}
128 
129 	/**
130 	 * A provisional step has no rendered consumer yet, so an existing literal must not be bound to its value.
131 	 *
132 	 * <p>
133 	 * {@code SCALE} carries zero provisional steps now that {@code --jc-chrome-font-size-2} (its own fixture's
134 	 * former value) has flipped from provisional to confirmed, so this exercises the exemption branch
135 	 * against a synthetic still-provisional step via the {@code scan(String, List)} test seam rather than
136 	 * against a real one - see that overload's javadoc.
137 	 */
138 	@Test void b07_provisionalStepValuesDoNotTriggerAViolation() {
139 		var provisional = List.of(
140 			new ChromeScaleScanner.Step("--jc-synthetic-provisional", "0.8125rem", false, ChromeScaleScanner.Family.FONT_SIZE));
141 		var r = ChromeScaleScanner.scan(".synthetic-label { font-size: 0.8125rem; }", provisional);
142 		assertTrue(r.violations().isEmpty(), () -> "a provisional step value was enforced: " + r.violations());
143 	}
144 
145 	/** A declaration already spending a token is compliant, fallback and all. */
146 	@Test void b08_tokenFallbackIsNotReadAsABareLiteral() {
147 		var r = ChromeScaleScanner.scan(".synthetic-grid { gap: var(--jc-card-gap, 1rem); }");
148 		assertTrue(r.violations().isEmpty(), () -> "a var() fallback was read as a bare literal: " + r.violations());
149 	}
150 
151 	/** {@code width} is a scale property on an SVG glyph and a content measure everywhere else. */
152 	@Test void b09_widthIsOnlyCheckedOnGlyphs() {
153 		assertTrue(ChromeScaleScanner.scan(".synthetic-cell { width: 16px; }").violations().isEmpty());
154 		assertSize(1, ChromeScaleScanner.scan(".synthetic-btn svg { width: 18px; }").violations());
155 	}
156 
157 	/**
158 	 * {@code CONTROL_PADDING_X} was structurally unscored before this fix - no property mapped onto it in
159 	 * {@code CHECKED_PROPERTIES}, so a duplicated value could never be caught regardless of a step's
160 	 * {@code confirmed} flag. Exercises both newly-wired longhand properties.
161 	 */
162 	@Test void b10_controlPaddingXCatchesADuplicatedValue() {
163 		var r = ChromeScaleScanner.scan(".synthetic-control { padding-left: 10px; padding-right: 14px; }");
164 		assertSize(() -> "expected both padding-left and padding-right to be caught: "
165 			+ r.violations(), 2, r.violations());
166 		assertTrue(r.violations().get(0).contains("--jc-chrome-control-padding-x (10px)"), r.violations().get(0));
167 		assertTrue(r.violations().get(1).contains("--jc-chrome-control-padding-x-wide (14px)"), r.violations().get(1));
168 	}
169 
170 	/**
171 	 * {@code LINE_HEIGHT}'s only step value is unitless, so the general px/rem literal matcher could never bind
172 	 * to it before this fix - {@code line-height} was already mapped in {@code CHECKED_PROPERTIES}, but nothing
173 	 * could ever match. Proves the widened, {@code line-height}-scoped value-matching now catches it.
174 	 */
175 	@Test void b11_lineHeightCatchesADuplicatedValue() {
176 		var r = ChromeScaleScanner.scan(".synthetic-control { line-height: 1.2; }");
177 		assertSize(() -> "expected the duplicated line-height value to be caught: "
178 			+ r.violations(), 1, r.violations());
179 		assertTrue(r.violations().get(0).contains("--jc-chrome-line-height (1.2)"), r.violations().get(0));
180 	}
181 
182 	/**
183 	 * {@code font-size}'s three chrome-scale steps are declared in {@code rem}, not {@code px} - the general
184 	 * px-only literal matcher could never bind to a {@code rem}-valued {@code font-size} duplicate before this
185 	 * fix (the font-size rem-reachability carve-out), so a tenth un-tokenized {@code 0.75rem}/{@code 0.8125rem}/{@code 0.875rem} site
186 	 * introduced today would have gone unnoticed exactly the same way the original twelve did. Proves the
187 	 * widened, {@code font-size}-scoped {@code rem} carve-out now catches it.
188 	 */
189 	@Test void b12_fontSizeRemCarveOutCatchesADuplicatedValue() {
190 		var r = ChromeScaleScanner.scan(".synthetic-label { font-size: 0.875rem; }");
191 		assertSize(() -> "expected the rem-valued font-size duplicate to be caught: "
192 			+ r.violations(), 1, r.violations());
193 		assertTrue(r.violations().get(0).contains("--jc-chrome-font-size-3 (0.875rem)"), r.violations().get(0));
194 	}
195 
196 	/**
197 	 * The three DataTables density literals ({@code table[data-juneau-view]}/{@code thead}/{@code tbody} cell
198 	 * type context) moved from {@code 0.75rem} (a duplicate of {@code --jc-chrome-font-size-1}) to
199 	 * {@code 0.8333rem} - the reference body type step. {@code 0.8333rem} does not equal any
200 	 * declared step, so the {@code font-size} {@code rem} carve-out never even reaches
201 	 * {@link ChromeScaleScanner#recordedLiterals()} for these three sites any more: no duplicate, no violation,
202 	 * no exception needed. Reproduces the three pinned sites synthetically, by selector shape rather than by
203 	 * reading the shipped stylesheet's own line numbers, so this proves the new value is clean of the scale
204 	 * independent of where the sites currently sit in the file - and would start failing loudly (a real,
205 	 * unexcused violation, since these are no longer in the recorded table) if the value ever drifted back onto
206 	 * an existing step.
207 	 */
208 	@Test void b13_movedTableCellPinNoLongerDuplicatesAStepOrNeedsAnException() {
209 		var r = ChromeScaleScanner.scan(
210 			"table[data-juneau-view], table.dataTable { font-size: 0.8333rem; }"
211 			+ " table[data-juneau-view] > thead > tr > th { font-size: 0.8333rem; }"
212 			+ " table[data-juneau-view] > tbody > tr > th { font-size: 0.8333rem; }");
213 		assertTrue(r.violations().isEmpty(), () -> "0.8333rem does not equal any declared step and must not be "
214 			+ "reported as a violation: " + r.violations());
215 		assertTrue(r.matchedRecords().isEmpty(), () -> "0.8333rem does not duplicate a step, so it needs no "
216 			+ "recorded exception any more: " + r.matchedRecords());
217 	}
