13  package org.apache.juneau.console.views;
14  
15  import static org.junit.jupiter.api.Assertions.*;
16  import static org.junit.jupiter.api.Assumptions.*;
17  
18  import java.util.*;
19  
20  import org.junit.jupiter.api.*;
21  
22  /** Client-rendered tabs: the renderer registry and its context, run through the Node harness. */
23  class DetailRegister_Test {
24  
25  	private static Map<?,?> report() {
26  		var r = RegionsHarness.reportWith("detail-register.cjs", ViewsMixin.HELPERS_JS_RESOURCE, ViewsMixin.BUS_JS_RESOURCE,
27  			"/org/apache/juneau/views/juneau-model.js");
28  		assumeTrue(r != null, "node not available or detail-register.cjs not found - behavioral layer skipped");
29  		return r;
30  	}
31  
32  	@Test void a01_registered_calledOnce_withHandleAndFrozenData() {
33  		var r = report();
34  		assertEquals(true, r.get("reg_calledOnce"));
35  		assertEquals("c:detailtab", r.get("reg_tab"));
36  		assertEquals("fail,fetch,invalidate,pane,refresh,row,rowId,signal", r.get("reg_dataKeys"));
37  		assertEquals(true, r.get("reg_dataFrozen"));
38  		assertEquals("s-7:Rex:secret", r.get("reg_row"));
39  		assertEquals("initial", r.get("reg_refresh"));
40  		assertEquals(true, r.get("reg_paneSignal"));
41  		assertEquals(true, r.get("reg_painted"));
42  	}
43  
44  	@Test void a02_helpers_frozenWhitelist() {
45  		assertEquals("fieldGrid,renderAsyncStatus:true:function:function:fieldGrid,renderAsyncStatus", report().get("reg_helpers"));
46  	}
47  
48  	@Test void a03_blankNameOrNonFunction_refused() {
49  		assertEquals(3, ((Number)report().get("refuse_errors")).intValue());
50  	}
51  
52  	@Test void a04_lastWriteWins() {
53  		assertEquals(true, report().get("lastWins"));
54  	}
55  
56  	@Test void a05_unregistered_showsError() {
57  		assertEquals("Renderer 'nope' is not registered", report().get("unregistered"));
58  	}
59  
60  	@Test void a06_fetch_sameOrigin_substitutes_csrf_accept() {
61  		var r = report();
62  		assertEquals("/api/pet/s-7/labels", r.get("fetch_url"));
63  		assertEquals("application/json", r.get("fetch_accept"));
64  		assertEquals(true, r.get("fetch_csrf"));
65  		assertEquals(true, r.get("fetch_cross"));
66  		assertEquals(true, r.get("fetch_proto"));
67  		assertEquals(true, r.get("fetch_noEvilRequest"));
68  		assertEquals("ok:{\"contractVersion\":\"1\",\"labels\":[\"a\"]}", r.get("fetch_good"));
69  	}
70  
71  	@Test void a07_fetch_rejectsNon2xx_andWrongContract() {
72  		var r = report();
73  		assertEquals(true, r.get("fetch_bad"));
74  		assertEquals(true, r.get("fetch_contract"));
75  	}
76  
77  	@Test void a08_throw_showsError_withRetry_thatReruns() {
78  		var r = report();
79  		assertEquals("kaput", r.get("throw_status"));
80  		assertEquals(true, r.get("throw_hasRetry"));
81  		assertEquals(true, r.get("retry_reran"));
82  	}
83  
84  	@Test void a09_failAndRejection_showError() {
85  		var r = report();
86  		assertEquals("nope", r.get("fail_status"));
87  		assertEquals(true, r.get("fail_retry"));
88  		assertEquals("async boom", r.get("reject_status"));
89  	}
90  
91  	@Test void a10_promise_loadingThenEmptyText() {
92  		var r = report();
93  		assertEquals("Loading...", String.valueOf(r.get("promise_loading")).replace("\u2026", "..."));
94  		assertEquals("Nothing here", r.get("promise_empty"));
95  	}
96  }
