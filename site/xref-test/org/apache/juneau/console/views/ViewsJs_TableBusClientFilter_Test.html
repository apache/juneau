17  package org.apache.juneau.console.views;
18  
19  import static org.apache.juneau.console.views.ViewsJs_TableBus_Test.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  import static org.junit.jupiter.api.Assumptions.*;
22  
23  import java.util.*;
24  
25  import org.apache.juneau.*;
26  import org.junit.jupiter.api.*;
27  
28  /**
29   * A CLIENT-mode {@code set-filter} drives the table's in-memory ribbon row filter (the one the ribbon installs on
30   * {@code $.fn.dataTable.ext.search}) and then {@code draw()}s - never a DataTables column search for the option, and
31   * never an ajax reload.  SERVER mode reloads, and the options reach the request through the ribbon's own query-param
32   * and column-search derivation.
33   */
34  class ViewsJs_TableBusClientFilter_Test extends TestBase {
35  
36  	static Map<String,Object> report;
37  
38  	@BeforeAll static void runHarness() {
39  		report = BusHarness.run("table-bus.cjs",
40  			ViewsMixin.BUS_JS_RESOURCE, ViewsMixin.RENDERS_JS_RESOURCE, ViewsMixin.VIEWS_JS_RESOURCE, ViewsMixin.RIBBON_JS_RESOURCE, ViewsMixin.SEARCH_JS_RESOURCE);
41  	}
42  
43  	@BeforeEach void needsNode() {
44  		assumeTrue(report != null, "node not on PATH");
45  		assertNull(report.get("crash"), () -> "harness crashed: " + report.get("crash"));
46  		assertEquals(true, report.get("hasTableBus"), "NS.tableBus is not exported");
47  	}
48  
49  	@Test void a01_bindDoesNotColumnSearchAnOption() {
50  		assertEquals("[]", report.get("bindSearches"));
51  	}
52  
53  	@Test void b01_setFilterOptionRefreshesRowFilterAndDraws() {
54  		var m = map(report.get("clientNew"));
55  		assertEquals("[]", m.get("searches"), "an option is a row predicate, not a DataTables column search");
56  		assertEquals(1, m.get("rowFilterRefreshes"));
57  		assertEquals(1, m.get("draws"));
58  		assertEquals(0, m.get("reloads"), "CLIENT mode filters in memory; it must not refetch");
59  		assertEquals("new", m.get("activeAge"));
60  		assertEquals("{\"schemaVersion\":1,\"viewId\":\"changes\",\"search\":\"\",\"columns\":{},\"options\":{\"mine\":false,\"openOnly\":false,\"age\":\"new\"}}", m.get("filter"),
61  			"an option is reported under options, never under columns");
62  	}
63  
64  	@Test void b02_valuelessGroupMemberStillRefreshesTheRowFilter() {
65  		var m = map(report.get("clientAll"));
66  		assertEquals(2, m.get("rowFilterRefreshes"));
67  		assertEquals("all", m.get("filterAge"));
68  	}
69  
70  	@Test void c01_plainPageWithoutABusFiltersToo() {
71  		assertNull(report.get("plainBus"));
72  		var m = map(report.get("plain"));
73  		assertEquals(1, m.get("rowFilterRefreshes"));
74  		assertEquals("new", m.get("lastAge"));
75  		assertEquals(1, m.get("draws"));
76  		assertEquals(2, m.get("seen"), "the starting filter, then the patch");
77  	}
78  
79  	@Test void d01_serverModeUsesRequestDerivationNotRowFilter() {
80  		var m = map(report.get("server"));
81  		assertEquals("[]", m.get("searches"));
82  		assertEquals(1, m.get("reloads"));
83  		assertEquals(0, m.get("draws"));
84  		assertEquals("{\"mine\":\"true\"}", m.get("params"));
85  		assertEquals("{\"3\":\"open\"}", m.get("columnSearches"));
86  		assertEquals(true, m.get("filterOpenOnly"));
87  	}
88  }
