27  /**
28   * Java-to-JS parity for tabbed row details: the Java serialisation of a detail covering every tab kind is pinned to a
29   * golden file, and the same file is loaded by {@code juneau-views.js} in a Node sandbox.
30   */
31  class RowDetailTabs_Parity_Test extends TestBase {
32  
33  	private static final String GOLDEN = "row-detail-all-kinds.golden.json";
34  
35  	private static Map<String,Object> detail() {
36  		var nested = TableSpec.create("orders").dataUrl("/rest/orders/rows").columns(Column.create("id"), Column.create("status"));
37  		var d = RowDetail.inline()
38  			.title("Pet {id}")
39  			.icon("paw")
40  			.action(RowAction.link("open", "/pets/{id}").label("Open"))
41  			.tab(DetailTab.fields("Summary", "name", "status").id("summary"))
42  			.tab(DetailTab.table("Orders", nested).id("orders").countFromRows())
43  			.tab(DetailTab.chart("Trend", RegionDef.chart("trend", "/rest/trend/{id}").kind("line").x("name")).id("trend"))
44  			.tab(DetailTab.client("Custom", "pet-labels").id("custom"));
45  		return d.toMap("pets", List.of(Column.create("id"), Column.create("name")), "id");
46  	}
47  
48  	private static String golden() throws IOException {
49  		try (var in = RowDetailTabs_Parity_Test.class.getResourceAsStream(GOLDEN)) {
50  			assertNotNull(in, "missing golden " + GOLDEN);
51  			return new String(in.readAllBytes(), UTF_8).strip();
52  		}
53  	}
54  
55  	@Test void a01_javaSerialisation_matchesGolden() throws Exception {
56  		var json = Json.of(detail());
57  		Files.createDirectories(Path.of("target"));
58  		Files.writeString(Path.of("target", "row-detail-all-kinds.actual.json"), json + "\n", UTF_8);
59  		assertEquals(golden(), json);
60  	}
61  
62  	@Test void a02_jsConsumesTheGolden() {
63  		var r = RegionsHarness.reportWith("row-detail-parity.cjs", ViewsMixin.HELPERS_JS_RESOURCE,
64  			"/org/apache/juneau/console/views/" + GOLDEN);
65  		assumeTrue(r != null, "node not available or row-detail-parity.cjs not found - behavioral layer skipped");
66  		assertEquals("summary|orders|trend|custom", r.get("tabIds"));
67  		assertEquals("Summary|Orders|Trend|Custom", r.get("tabLabels"));
68  		assertEquals("open", r.get("headerLinkId"));
69  		assertEquals("/pets/s-7", r.get("headerLinkHref"));
70  		assertEquals(true, r.get("noFetch"));
71  	}
72  
73  	@Test void a03_tabKinds_matchJavaEnum() {
74  		var r = RegionsHarness.reportWith("row-detail-parity.cjs", ViewsMixin.HELPERS_JS_RESOURCE,
75  			"/org/apache/juneau/console/views/" + GOLDEN);
76  		assumeTrue(r != null, "node not available or row-detail-parity.cjs not found - behavioral layer skipped");
77  		var java = new ArrayList<String>();
78  		for (var k : DetailTab.Kind.values())
79  			java.add(k.wire());
80  		assertEquals(java, r.get("kinds"));
81  	}
82  }
