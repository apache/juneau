28  /**
29   * Test-only, <b>private</b> reader for {@code juneau-symbols.svg} and its provenance manifest
30   * ({@code juneau-symbols-provenance.md}), modeled on {@link RawContentSinkScanner}'s
31   * {@link RawContentSinkScanner#locateModuleRoot() locateModuleRoot()} / scan-the-module's-own-tree idiom.
32   *
33   * <h5 class='section'>What it is for</h5>
34   * <p>
35   * Each glyph in this sprite is either Juneau-original or artwork cleared for copy from another sprite (see
36   * the manifest's {@code origin} column); the manifest exists as a guard: it pins the <i>approved</i> artwork so that a future paste of foreign path data over a glyph fails the build
37   * until someone deliberately edits the manifest row. This class does the reading and the hashing;
38   * {@link SymbolSprite_Provenance_Test} does the asserting.
39   *
40   * <h5 class='section'>Be honest about what a fingerprint can and cannot do</h5>
41   * <p>
42   * It reliably catches <b>drift</b> &mdash; artwork changing without a reviewed manifest edit. It cannot recognise
43   * foreign artwork as foreign: an author who pastes a proprietary glyph <i>and</i> updates the row passes. That is
44   * not a defect to be engineered around, it is the boundary of the mechanism: the guard's job is to make the paste
45   * require a visible, reviewable second act rather than ride along inside an artwork commit. The manifest's
46   * <i>Authoring rules</i> section is what that second act is reviewed against.
47   *
48   * <h5 class='section'>What the fingerprint covers, and why it is the whole element</h5>
49   * <p>
50   * SHA-256 over the exact UTF-8 bytes of the whole {@code <symbol ...>...</symbol>} element &mdash; opening tag
51   * included, so {@code viewBox} is inside the hash. That is deliberate rather than incidental: a glyph's
52   * {@code viewBox} sets the lattice its stroke centrelines snap to, because {@code juneau-icons.js} hard-codes the
53   * host {@code <svg>} at {@code viewBox="0 0 24 24"} and the grid-fit lattice is defined in <i>host</i> units. A
54   * glyph arriving at a foreign modulus is therefore an artwork change in every sense that matters, and hashing only
55   * the path data would leave it invisible to every guard in the tree.
56   *
57   * <h5 class='section'>Why it reads the source tree rather than the classpath</h5>
58   * <p>
59   * The original-pack manifest is not a registered servable resource &mdash; only the two sprites are (see
60   * {@link ViewsMixin}) &mdash; so reading it through the resource loader would assert against a copy no consumer
61   * ever fetches. Both files are read from the module's own {@code src/main/resources}, the same way
62   * {@link SymbolsKey_Staleness_Test} reads the sprite and its authoring key.
63   */
64  @SuppressWarnings({
65  	"java:S8786" // Scanner regex is intentional; tightening would change match semantics.
66  })
67  final class SymbolProvenanceScanner {
68  
69  	static final String RESOURCE_DIR = "src/main/resources/org/apache/juneau/views";
70  	static final String SPRITE = "juneau-symbols.svg";
71  	static final String MANIFEST = "juneau-symbols-provenance.md";
72  
73  	/** The id prefix every glyph carries; the stem is what follows it. */
74  	static final String STEM_PREFIX = "juneau-sym-";
75  
76  	/** One whole {@code <symbol>} element, non-greedy so consecutive symbols do not collapse into one match. */
77  	private static final Pattern SYMBOL =
78  		Pattern.compile("<symbol\\s+id=\"" + Pattern.quote(STEM_PREFIX) + "([A-Za-z0-9_-]+)\".*?</symbol>", Pattern.DOTALL);
79  
80  	/** One manifest fingerprint row: {@code | `stem` | `origin` | `hex` |}. */
81  	private static final Pattern MANIFEST_ROW =
82  		Pattern.compile("^\\|\\s*`([A-Za-z0-9_-]+)`\\s*\\|\\s*`([a-z-]+)`\\s*\\|\\s*`([0-9a-f]{64})`\\s*\\|\\s*$",
83  			Pattern.MULTILINE);
84  
85  	/** The three document-family members (csv / pdf / spreadsheet); the manifest pins their origin, not a shared frame path. */
86  	static final List<String> FRAMED_FAMILY = List.of("csv", "pdf", "spreadsheet");
87  
88  	/** A {@code d} attribute value. */
89  	private static final Pattern PATH_D = Pattern.compile("\\bd=\"([^\"]*)\"");
90  
91  	/** A {@code fill} or {@code stroke} paint value. */
92  	private static final Pattern PAINT = Pattern.compile("\\b(fill|stroke)=\"([^\"]*)\"");
93  
94  	/**
95  	 * A themable paint: a custom property whose fallback is {@code currentColor}, e.g.
96  	 * {@code var(--jc-sort-asc-fill, currentColor)}.  Allowed deliberately so a glyph can expose a per-part tint
97  	 * hook (custom properties inherit into a {@code <use>} shadow tree, where document selectors cannot reach)
98  	 * while still defaulting to the host's {@code color}; a literal colour in the fallback is still a breach.
99  	 */
100 	private static final Pattern THEMABLE_PAINT = Pattern.compile("var\\(--[A-Za-z0-9_-]+,\\s*currentColor\\)");
101 
102 	/** One element of a symbol body, with its attributes. */
103 	private static final Pattern ELEMENT = Pattern.compile("<(\\w+)([^>]*?)/?>");
104 
105 	private SymbolProvenanceScanner() {}
106

111 	/** Reads one file out of the module's own {@code src/main/resources}; {@code null} if the module cannot be located. */
112 	static String read(String fileName) throws IOException {
113 		var root = RawContentSinkScanner.locateModuleRoot();
114 		if (root == null)
115 			return null;
116 		var f = root.resolve(RESOURCE_DIR).resolve(fileName);
117 		return Files.isRegularFile(f) ? Files.readString(f, UTF_8) : null;
118 	}
119 
120 	/** Stem to the verbatim bytes of its whole {@code <symbol>} element, in document order. */
121 	static Map<String,String> symbols(String svg) {
122 		var out = new LinkedHashMap<String,String>();
123 		var m = SYMBOL.matcher(svg);
124 		while (m.find())
125 			out.put(m.group(1), m.group());
126 		return out;
127 	}
128 
129 	/** Stem to its declared {@code origin} in the manifest, in manifest order. */
130 	static Map<String,String> manifestOrigins(String md) {
131 		var out = new LinkedHashMap<String,String>();
132 		var m = MANIFEST_ROW.matcher(md);
133 		while (m.find())
134 			out.put(m.group(1), m.group(2));
135 		return out;
136 	}
137 
138 	/** Stem to its declared fingerprint in the manifest, in manifest order. */
139 	static Map<String,String> manifestFingerprints(String md) {
140 		var out = new LinkedHashMap<String,String>();
141 		var m = MANIFEST_ROW.matcher(md);
142 		while (m.find())
143 			out.put(m.group(1), m.group(3));
144 		return out;
145 	}

149 	//------------------------------------------------------------------------------------------------------------------
150 
151 	/** SHA-256, lower-case hex, over the UTF-8 bytes of {@code s}. */
152 	static String fingerprint(String s) {
153 		try {
154 			var digest = MessageDigest.getInstance("SHA-256").digest(s.getBytes(UTF_8));
155 			var sb = new StringBuilder(digest.length * 2);
156 			for (var b : digest)
157 				sb.append(Character.forDigit((b >> 4) & 0xf, 16)).append(Character.forDigit(b & 0xf, 16));
158 			return sb.toString();
159 		} catch (NoSuchAlgorithmException e) {
160 			throw new IllegalStateException("SHA-256 is required of every JRE", e);
161 		}
162 	}
163 
164 	/** Every {@code d} attribute value in a symbol element, in document order. */
165 	static List<String> pathData(String symbol) {
166 		var out = new ArrayList<String>();
167 		var m = PATH_D.matcher(symbol);
168 		while (m.find())
169 			out.add(m.group(1));
170 		return out;
171 	}
172 
173 	/**
174 	 * The document-family frame path of a symbol: its <b>first</b> {@code d} attribute.
175 	 *
176 	 * <p>
177 	 * Positional rather than pattern-matched, and that is the point. The family spec declares the frame as the
178 	 * first path of each member, so "the first {@code d}" is the spec being read back rather than a heuristic that
179 	 * might latch onto an interior mark. A member that reordered its paths so the frame stopped being first would
180 	 * fail the byte-identity assertion, which is the correct outcome: the spec was not followed.
181 	 */
182 	static String framePath(String symbol) {
183 		var d = pathData(symbol);
184 		return d.isEmpty() ? null : d.get(0);
185 	}
186 
187 	/** Every {@code fill}/{@code stroke} value in the sprite that is not {@code none}, {@code currentColor} or {@code var(--x, currentColor)}. */
188 	static List<String> offContractPaints(String svg) {
189 		var out = new ArrayList<String>();
190 		var m = PAINT.matcher(svg);
191 		while (m.find()) {
192 			var v = m.group(2);
193 			if (!eqa(v, "none", "currentColor") && !THEMABLE_PAINT.matcher(v).matches())
194 				out.add(m.group(1) + "=\"" + v + "\"");
195 		}
196 		return out;
197 	}
198 
199 	/**
200 	 * Elements of a symbol that paint a stroke without declaring its width.
201 	 *
202 	 * <p>
203 	 * Returned as {@code tag attributes} strings so a failure names the offending element rather than only its
204 	 * glyph.
205 	 */
206 	static List<String> strokedWithoutWidth(String symbol) {
207 		var out = new ArrayList<String>();
208 		var m = ELEMENT.matcher(symbol);
209 		while (m.find()) {
210 			var tag = m.group(1);
211 			var attrs = m.group(2);
212 			if (eq(tag, "symbol"))
213 				continue;
214 			if (attrs.contains("stroke=\"currentColor\"") && !attrs.contains("stroke-width=\""))
215 				out.add(tag + attrs.stripTrailing());
216 		}
217 		return out;
218 	}
219 
220 	/** The {@code viewBox} an element declares, or {@code null}. */
221 	static String viewBox(String element) {
222 		var m = Pattern.compile("\\bviewBox=\"([^\"]*)\"").matcher(element);
223 		return m.find() ? m.group(1) : null;
224 	}
