28  /**
29   * Behavioral coverage for the {@code juneau-bus.js} core.  The Node
30   * harness {@code bus-core.cjs} loads the bus standalone into the DOM shim and reports one section per subject; every
31   * assertion is here, so a failure names the subject in surefire.  Gated on {@code node} being on {@code PATH}.

35  	private static final String CORE_END = "// ---- stock sources: session handshake";
36  
37  	private static Map<String,Object> report;
38  
39  	@BeforeAll
40  	static void runHarness() {
41  		report = BusHarness.run("bus-core.cjs", ViewsMixin.BUS_JS_RESOURCE);
42  	}
43  
44  	@SuppressWarnings({
45  		"unchecked" // Casts a parsed JSON value to its known Map/List shape.
46  	})
47  	private static Map<String,Object> section(String name) {
48  		assumeTrue(report != null, "node not on PATH");
49  		var s = (Map<String,Object>) report.get(name);
50  		assertNotNull(s, "harness section " + name + " missing; report=" + report);
51  		return s;
52  	}
53  
54  	@SuppressWarnings({
55  		"unchecked" // Casts a parsed JSON value to its known Map/List shape.
56  	})
57  	private static Map<String,Object> map(Map<String,Object> m, String key) {
58  		return (Map<String,Object>) m.get(key);
59  	}
60  
61  	private static List<?> list(Map<String,Object> m, String key) {
62  		return (List<?>) m.get(key);
63  	}
64  
65  	private static int num(Map<String,Object> m, String key) {
66  		return ((Number) m.get(key)).intValue();
67  	}
68  
69  	private static String owned(String topic, String owner, String who, String key) {
70  		return "'" + topic + "' is owned by '" + owner + "'; '" + who + "' may not publish or clear it (publish cmd:" + key + " instead)";
71  	}

77  	@Test void a00_sourceShape() throws Exception {
78  		var js = BusHarness.source(ViewsMixin.BUS_JS_RESOURCE);
79  		assertTrue(js.contains("juneau-bus.js"), "the header comment names the asset");
80  		var end = js.indexOf(CORE_END);
81  		assertTrue(end > 0, "the stock-sources section marker is present");
82  		var core = js.substring(0, end);
83  		assertFalse(core.contains("addEventListener("), "the core attaches no listeners");
84  		assertFalse(core.contains("document."), "the core, the bridge runtime and the codec never touch the DOM");
85  		assertTrue(core.contains("NS.bus = {"), "the core defines exactly JuneauViews.bus");
86  		assertFalse(js.toLowerCase().contains("slds"), "no SLDS anywhere");
87  	}
