17  package org.apache.juneau.console.views;
18  
19  import static org.apache.juneau.bct.BctAssertions.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.util.*;
23  
24  import org.apache.juneau.*;
25  import org.junit.jupiter.api.*;
26  
27  /**
28   * {@link RegionTable#of(RegionDef)} attribute stamping: identity, and the declared-descriptor attribute that appears
29   * only when the region carries {@code params}.
30   */
31  class RegionTable_Test extends TestBase {
32  
33  	@Test void a01_noParams_noDeclaredAttribute() {
34  		var html = RegionTable.of(RegionDef.create("side")).toString();
35  		assertContains(() -> html, "data-juneau-region='side'", html);
36  		assertFalse(html.contains(RegionTable.REGION_DECLARED_ATTR), html);
37  	}
38  
39  	@Test void a02_emptyParams_noDeclaredAttribute() {
40  		var html = RegionTable.of(RegionDef.create("side").params(new LinkedHashMap<String,Object>())).toString();
41  		assertFalse(html.contains(RegionTable.REGION_DECLARED_ATTR), html);
42  	}
43  
44  	@Test void a03_params_stampDeclaredDescriptor() {
45  		var r = RegionDef.create("side").params("run", "7");
46  		var html = RegionTable.of(r).toString();
47  		assertContains(() -> html, RegionTable.REGION_DECLARED_ATTR + "='", html);
48  		assertContains(() -> html, "contractVersion", html);
49  	}
50  
51  	@Test void a04_declaredDescriptor_isAttributeEscaped() {
52  		var r = RegionDef.create("side").params("q", "it's <b>&\"x\"");
53  		var html = RegionTable.of(r).toString();
54  		var start = html.indexOf(RegionTable.REGION_DECLARED_ATTR);
55  		assertTrue(start >= 0, html);
56  		var attr = html.substring(start);
57  		assertFalse(attr.contains("<b>"), attr);
58  	}
59  }
