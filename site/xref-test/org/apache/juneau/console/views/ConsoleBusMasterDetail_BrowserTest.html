17  package org.apache.juneau.console.views;
18  
19  import static org.apache.juneau.console.views.ConsoleBusBrowserSupport.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.util.*;
23  
24  import org.apache.juneau.*;
25  import org.junit.jupiter.api.*;
26  import org.junit.jupiter.api.condition.*;
27  
28  /**
29   * Master/detail over the bus in real Chromium: the {@code tasks} card follows {@code selection:changes} as a {@code params} subscription,
30   * shows its {@code emptyText} while nothing is selected, refetches with the encoded id on each pick, and clears again on deselect.
31   */
32  @EnabledIfSystemProperty(named=ConsoleBusBrowserSupport.GATE, matches="true", disabledReason=ConsoleBusBrowserSupport.DISABLED)
33  class ConsoleBusMasterDetail_BrowserTest extends TestBase {
34  
35  	private static Map<String,Object> r;
36  
37  	@BeforeAll
38  	static void probe() throws Exception {
39  		r = run("master-detail", "master-detail.json", true, "{declare:[{topic:'app.region-picked',retain:true}]}", "");
40  	}
41  
42  	@Test void a01_noJsFailures() {
43  		assertEquals(List.of(), list(r.get("jsFailures")), () -> r.toString());
44  		assertEquals(List.of(), list(r.get("consoleErrors")), () -> r.toString());
45  	}
46  
47  	@Test void a02_awaitingParamsShowsEmptyTextAndFetchesNothing() {
48  		var initial = map(r.get("initial"));
49  		assertEquals("Select a change to see its tasks.", initial.get("tasksEmpty"));
50  		assertEquals(0, num(initial.get("tasksFetches")));
51  		assertEquals(0, num(map(initial.get("selection")).get("count")));
52  		assertEquals(List.of("mounted", "pending", "mounted"), list(r.get("cardStates")));
53  	}
54  
55  	@Test void a03_pickingARowFetchesTheDetailForThatId() {
56  		assertEquals(Boolean.TRUE, r.get("selectClicked"));
57  		var sel = map(r.get("selected"));
58  		assertEquals(List.of("/api/changes/c-17/tasks"), list(sel.get("tasksPaths")));
59  		assertEquals(List.of("c-17"), list(map(sel.get("selection")).get("ids")));
60  		assertEquals(Boolean.TRUE, sel.get("tasksHasRow"));
61  	}
62  
63  	@Test void a04_deselectingClearsTheDetailBackToEmptyText() {
64  		var c = map(r.get("cleared"));
65  		assertEquals("Select a change to see its tasks.", c.get("tasksEmpty"));
66  		assertEquals(1, num(c.get("tasksFetches")), "clearing must not fetch");
67  		assertEquals(0, num(map(c.get("selection")).get("count")));
68  	}
69  
70  	@Test void a05_theIdIsUrlEncodedInThePath() {
71  		assertEquals("/api/changes/c%2F9%20x/tasks", map(r.get("encoded")).get("lastTasksPath"));
72  	}
73  }
