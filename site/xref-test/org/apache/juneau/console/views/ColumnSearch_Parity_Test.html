33  /**
34   * Always-on parity check that {@code juneau-search.js} filters byte-for-byte the way the server-side
35   * {@link InMemoryBeanQueryContext} does.
36   *
37   * <p>
38   * The Node harness ({@code src/test/js/column-search.cjs}) loads the served {@code juneau-search.js} asset into a
39   * {@code vm} sandbox and re-runs the SAME shared {@code search-corpus.json} corpus that {@code SearchCorpus_Test}
40   * runs against the Java engine, emitting one pass/fail flag per case id. Case ids are read directly from the
41   * corpus resource (never hardcoded here), so adding or renaming a case in the corpus does not require touching
42   * this file. Every case id is asserted here, so a divergence between the JS and Java engines names the exact case
43   * that drifted. When Node is unavailable (or the harness script is not on any working-directory path), the whole
44   * class skips.
45   */
46  class ColumnSearch_Parity_Test extends TestBase {
47  
48  	private static Map<?,?> report() {
49  		var r = run();
50  		assumeTrue(r != null, "node not available or column-search.cjs not found - JS parity layer skipped");
51  		return r;
52  	}
53  
54  	@Test void a01_everyCorpusCaseMatchesServerSide() {
55  		var r = report();
56  		var ids = caseIds();
57  		// Guard against a harness that silently dropped cases: the count must match the ids asserted below.
58  		assertEquals((long) ids.size(), ((Number) r.get("caseCount")).longValue(),
59  			() -> "harness emitted a different case count than expected: " + r);
60  		for (var id : ids)
61  			assertEquals(true, r.get(id), () -> "column-search parity case " + id + " diverged; failures=" + r.get("failures"));
62  		assertEquals(true, r.get("allPass"), () -> "column-search parity failures: " + r.get("failures"));
63  	}
64  
65  	@Test void z01_everyJavaBuiltinHasJsHelpText() {
66  		var r = report();
67  		assertEquals(List.of(), r.get("helpMissing"), "built-in operators with no help text in juneau-search.js");
68  	}

70  	// -----------------------------------------------------------------------------------------------------------------
71  	// Corpus case ids (mirror of SearchCorpus_Test.loadCorpus(), but only the "id" field is needed here).
72  	// -----------------------------------------------------------------------------------------------------------------
73  
74  	@SuppressWarnings({
75  		"unchecked" // The (List<Map<String,Object>>) cast of the parsed JSON corpus is safe because the corpus file is a JSON array of objects.
76  	})
77  	private static List<String> caseIds() {
78  		try (var in = ColumnSearch_Parity_Test.class.getResourceAsStream(CORPUS)) {
79  			assertNotNull(in, CORPUS);
80  			var json = new String(in.readAllBytes(), UTF_8);
81  			var cases = (List<Map<String,Object>>) (List<?>) Json.to(json, List.class);
82  			var out = new ArrayList<String>(cases.size());
83  			for (var kase : cases)
84  				out.add((String) kase.get("id"));
85  			return out;
86  		} catch (IOException e) {
87  			throw new UncheckedIOException(e);
88  		}
89  	}

91  	// -----------------------------------------------------------------------------------------------------------------
92  	// Node plumbing (sibling of HelpersHarness, two assets: the rendered juneau-search.js plus the shared corpus).
93  	// -----------------------------------------------------------------------------------------------------------------
94  
95  	private static final String HARNESS = "column-search.cjs";
96  	private static final String CORPUS = "search-corpus.json";
97  	private static Map<?,?> cached;
98  	private static boolean failed;
99  
100 	private static synchronized Map<?,?> run() {
101 		if (failed)
102 			return null;
103 		if (cached != null)
104 			return cached;
105 		try {
106 			if (!nodeAvailable())
107 				return markUnavailable();
108 			var harness = locate("src/test/js/" + HARNESS);
109 			var corpus = locate("src/test/resources/org/apache/juneau/console/views/" + CORPUS);
110 			if (harness == null || corpus == null)
111 				return markUnavailable();
112 			cached = Json.to(exec(harness, corpus), Map.class);
113 			return cached;
114 		} catch (Exception e) {
115 			throw new AssertionError("could not run " + HARNESS + ": " + e, e);
116 		}
117 	}
118 
119 	private static Map<?,?> markUnavailable() {
120 		failed = true;
121 		return null;
122 	}
123 
124 	private static String exec(Path harness, Path corpus) throws Exception {
125 		var search = Files.createTempFile("juneau-search-", ".js");
126 		var stdout = Files.createTempFile("column-search-stdout-", ".json");
127 		var stderr = Files.createTempFile("column-search-stderr-", ".txt");
128 		var names = Files.createTempFile("column-search-names-", ".json");
129 		try {
130 			Files.writeString(search, asset(ViewsMixin.SEARCH_JS_RESOURCE), UTF_8);
131 			Files.writeString(names, Json.of(SearchOperatorSet.standard().operators().stream().map(SearchOperator::name).toList()), UTF_8);
132 			var p = new ProcessBuilder(List.of("node", harness.toString(), search.toString(), corpus.toString(), names.toString()))
133 				.redirectOutput(stdout.toFile())
134 				.redirectError(stderr.toFile())
135 				.start();
136 			if (!p.waitFor(60, TimeUnit.SECONDS)) {
137 				p.destroyForcibly();
138 				fail(HARNESS + " did not finish within 60s; stderr:\n" + quietRead(stderr));
139 			}
140 			if (p.exitValue() != 0)
141 				fail(HARNESS + " exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr)
142 					+ "\nstdout:\n" + quietRead(stdout));
143 			return Files.readString(stdout, UTF_8);
144 		} finally {
145 			for (var f : List.of(search, stdout, stderr, names))
146 				Files.deleteIfExists(f);
147 		}
148 	}
149 
150 	private static String asset(String resource) throws IOException {
151 		try (var in = ViewsMixin.class.getResourceAsStream(resource)) {
152 			assertNotNull(in, resource);
153 			return new String(in.readAllBytes(), UTF_8);
154 		}
155 	}
156 
157 	private static boolean nodeAvailable() {
158 		try {
159 			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
160 			if (!p.waitFor(5, TimeUnit.SECONDS)) {
161 				p.destroyForcibly();
162 				return false;
163 			}
164 			return p.exitValue() == 0;
165 		} catch (Exception e) {
166 			return false;
167 		}
168 	}

170 	/** Locates a file given its path relative to this module's root, trying {@code basedir} first (Surefire's
171 	 * working directory under multi-module reactors), then a couple of relative fallbacks for IDE/ad-hoc runs. */
172 	private static Path locate(String relToModule) {
173 		var basedir = System.getProperty("basedir");
174 		if (basedir != null) {
175 			var p = Path.of(basedir, relToModule);
176 			if (Files.isRegularFile(p))
177 				return p.toAbsolutePath().normalize();
178 		}
179 		for (var rel : List.of(
180 			relToModule,
181 			"juneau-rest/juneau-console-views/" + relToModule
182 		)) {
183 			var p = Path.of(rel);
184 			if (Files.isRegularFile(p))
185 				return p.toAbsolutePath().normalize();
186 		}
187 		return null;
188 	}
189 
190 	private static String quietRead(Path p) {
191 		try { return Files.readString(p, UTF_8); }
192 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
193 	}
