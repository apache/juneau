254 		var body = minimal().urlState("ids", "day").urlState("ids", Map.of("param", "a")).urlState("ids", Map.of("max", 5)).urlState("ids", Map.of("param", "b")).toCardBody();

263 		assertEquals(List.of("day", "ids"), List.copyOf(((Map<?,?>)minimal().urlState("day").urlState("ids", Map.of("max", 3)).toCardBody().get("urlState")).keySet()));

267 		var t = minimal();
268 		for (var bad : List.<Map<String,Object>>of(
269 			Map.of("bogus", 1), Map.of("param", ""), Map.of("param", "a&b"), Map.of("param", 5), Map.of("param", "1x"),
270 			Map.of("max", 0), Map.of("max", -1), Map.of("max", 1.5), Map.of("max", "50"), Map.of("max", 2.0)))
271 			assertThrows(IllegalArgumentException.class, () -> t.urlState("ids", bad), "ids " + bad);
272 		for (var bad : List.<Map<String,Object>>of(
273 			Map.of("param", "a b"), Map.of("param", "x".repeat(65)), Map.of("max", 5), Map.of("bogus", "x")))
274 			assertThrows(IllegalArgumentException.class, () -> t.urlState("day", bad), "day " + bad);
275 		for (var bad : List.<Map<String,Object>>of(
276 			Map.of("param", "x"), Map.of("params", "x"), Map.of("params", Map.of("middle", "m")), Map.of("params", Map.of("start", "")),
277 			Map.of("params", Map.of("end", "a&b")), Map.of("params", Map.of("start", 1))))
278 			assertThrows(IllegalArgumentException.class, () -> t.urlState("window", bad), "window " + bad);
279 		Map<String,Object> arg2 = Map.of();
280 		assertThrows(IllegalArgumentException.class, () -> t.urlState(" ", arg2));
281 		Map<String,Object> arg3 = Map.of();
282 		assertThrows(IllegalArgumentException.class, () -> t.urlState((String)null, arg3));
283 		var arg4 = (Map<String,Object>)null;
284 		assertThrows(IllegalArgumentException.class, () -> t.urlState("ids", arg4));
285 		// A refused call leaves nothing behind.
286 		assertFalse(t.toCardBody().containsKey("urlState"));
287 		var partial = minimal().urlState("ids", Map.of("param", "ok"));
288 		Map<String,Object> invalid = Map.of("param", "also-ok", "max", 0);
289 		assertThrows(IllegalArgumentException.class, () -> partial.urlState("ids", invalid));
290 		assertEquals(Map.of("ids", Map.of("param", "ok")), partial.toCardBody().get("urlState"));

321 	@Test void b02k_urlStateConfig_isDeepCopiedOnInputAndOnEmit() {
322 		var params = new java.util.HashMap<String,Object>();
323 		params.put("start", "from");
324 		var cfg = new java.util.HashMap<String,Object>();
325 		cfg.put("params", params);
326 		var t = minimal().urlState("window", cfg);
327 		params.put("start", "mutated");
328 		params.put("end", "extra");
329 		cfg.put("bogus", 1);
330 		var expected = Map.of("window", Map.of("params", Map.of("start", "from")));
331 		assertEquals(expected, t.toCardBody().get("urlState"));
332 		// Mutating what toCardBody() returned must not reach the spec's own state.
333 		var emitted = (Map<String,Object>)t.toCardBody().get("urlState");
334 		((Map<String,Object>)((Map<String,Object>)emitted.get("window")).get("params")).put("start", "emitted");
335 		((Map<String,Object>)emitted.get("window")).put("param", "x");
336 		assertEquals(expected, t.toCardBody().get("urlState"));
337 		// Same for a flat config.
338 		var flat = new java.util.HashMap<String,Object>();
339 		flat.put("param", "pid");
340 		var t2 = minimal().urlState("ids", flat);
341 		flat.put("param", "mutated");
342 		((Map<String,Object>)((Map<String,Object>)t2.toCardBody().get("urlState")).get("ids")).put("param", "emitted");
343 		assertEquals(Map.of("ids", Map.of("param", "pid")), t2.toCardBody().get("urlState"));
344 	}

379 	private static void assertPlain(Object o) {
380 		if (o instanceof Map<?,?> m)
381 			m.values().forEach(TableSpec_ToCardBody_Test::assertPlain);
382 		else if (o instanceof List<?> l)
383 			l.forEach(TableSpec_ToCardBody_Test::assertPlain);
384 		else
385 			assertTrue(o == null || o instanceof String || o instanceof Number || o instanceof Boolean, String.valueOf(o));
386 	}
