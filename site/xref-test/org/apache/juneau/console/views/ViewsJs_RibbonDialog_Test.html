32  /**
33   * The row-less (ribbon-hosted) dialog seam: a {@link RibbonAction#dialog(String) dialog} ribbon action opened with
34   * NO {@code <tr>} behind it, reusing the whole already-reviewed dialog machinery.
35   *
36   * <p>Two layers, same as {@code ViewsJs_SelfTargetedIdempotency_Test}:
37   * <ul class='spaced-list'>
38   * 	<li><b>Default-gate source pins</b> (a-group): the one-way module hop the ribbon runtime makes, the catalog
39   * 		isolation that keeps a ribbon action out of every row's action menu, and - the guard that matters most -
40   * 		the fact that the ROW render paths were NOT made null-tolerant to accommodate a row-less caller.
41   * 	<li><b>A behavioral harness</b> ({@code ribbon-dialog.cjs}) that loads BOTH runtimes into one window and drives
42   * 		real clicks, since the seam spans the two files and the interesting failures (a refusal painting into a
43   * 		row's actions cell that does not exist) are DOM-shaped, not source-shaped.  Runs when {@code node} is on
44   * 		{@code PATH}, skipped otherwise.
45   * </ul>
46   */
47  class ViewsJs_RibbonDialog_Test extends TestBase {

49  	private static String resource(String name) throws IOException {
50  		try (var in = ViewsMixin.class.getResourceAsStream(name)) {
51  			assertNotNull(in, () -> "missing classpath resource: " + name);
52  			return new String(in.readAllBytes(), UTF_8);
53  		}
54  	}

60  	private static String functionBody(String body, String signature) {
61  		var start = body.indexOf(signature);
62  		assertTrue(start >= 0, () -> "'" + signature + "' not found:\n" + body);
63  		var end = body.indexOf("\n\t}", start);
64  		return body.substring(start, end < 0 ? body.length() : end);
65  	}

81  	@Test void a02_catalogsStayDisjoint() throws Exception {
82  		var views = viewsJs();
83  		// The ribbon resolver reads the RIBBON catalog and only the ribbon catalog...
84  		var ribbonResolver = functionBody(views, "function findRibbonDialogAction(");
85  		assertTrue(ribbonResolver.contains("ctx?.viewDef?.ribbon"), ribbonResolver);
86  		assertFalse(ribbonResolver.contains("rowActions"),
87  			() -> "a ribbon dialog action must not be resolvable from the row-action catalog:\n" + ribbonResolver);
88  		// ...and the row-action resolver was left alone, so a ribbon action can never surface in a row menu or pill.
89  		var rowResolver = functionBody(views, "function dialogActionIsOpenable(");
90  		assertTrue(rowResolver.contains("ctx?.viewDef?.rowActions"), rowResolver);
91  		assertFalse(rowResolver.contains("viewDef?.ribbon"),
92  			() -> "widening the ROW catalog to the ribbon is exactly what the separate resolver avoids:\n" + rowResolver);
93  	}

95  	@Test void a03_rowRenderPathsWereNotMadeNullTolerant() throws Exception {
96  		var views = viewsJs();
97  		// The row renderers keep dereferencing `tr` unguarded: a row-less caller must be ROUTED to the row-less
98  		// host, not silently absorbed by a null-tolerant row renderer that paints nowhere.
99  		assertTrue(functionBody(views, "function renderRowActionRefusal(").contains("tr.querySelector"), views);
100 		assertTrue(functionBody(views, "function renderActionOutcome(").contains("tr.querySelector"), views);
101 		// The routing lives in the dispatchers instead.
102 		var refusalFor = functionBody(views, "function renderActionRefusalFor(");
103 		assertTrue(refusalFor.contains("if (tr) renderRowActionRefusal(tr, action, reason);"), refusalFor);
104 		assertTrue(refusalFor.contains("else renderRibbonActionRefusal(ctx, action, reason);"), refusalFor);
105 		var outcomeFor = functionBody(views, "function renderActionOutcomeFor(");
106 		assertTrue(outcomeFor.contains("if (tr) renderActionOutcome(tr, cls);"), outcomeFor);
107 		assertTrue(outcomeFor.contains("else renderRibbonActionOutcome(ctx, cls);"), outcomeFor);
108 		// A row-less mergeRow can't merge (mergeRowFromResult dereferences `tr`), so it redraws instead.
109 		var apply = functionBody(views, "function applySuccessBehavior(");
110 		assertTrue(apply.contains("&& tr"), () -> "mergeRow must be gated on having a row:\n" + apply);
111 	}

128 	@BeforeAll
129 	static void probeIfNodeAvailable() throws Exception {
130 		if (!nodeAvailable())
131 			return;
132 		var harness = locateHarness();
133 		if (harness == null)
134 			return;
135 		var rendersFile = Files.createTempFile("juneau-renders-", ".js");
136 		var viewsFile = Files.createTempFile("juneau-views-", ".js");
137 		var ribbonFile = Files.createTempFile("juneau-ribbon-", ".js");
138 		try {
139 			Files.writeString(rendersFile, rendersJs(), UTF_8);
140 			Files.writeString(viewsFile, viewsJs(), UTF_8);
141 			Files.writeString(ribbonFile, ribbonJs(), UTF_8);
142 			report = Json.to(runNode(harness, rendersFile, viewsFile, ribbonFile), Map.class);
143 		} finally {
144 			Files.deleteIfExists(rendersFile);
145 			Files.deleteIfExists(viewsFile);
146 			Files.deleteIfExists(ribbonFile);
147 		}
148 	}

150 	private static boolean nodeAvailable() {
151 		try {
152 			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
153 			if (!p.waitFor(5, TimeUnit.SECONDS)) {
154 				p.destroyForcibly();
155 				return false;
156 			}
157 			return p.exitValue() == 0;
158 		} catch (Exception e) {
159 			return false;
160 		}
161 	}

163 	private static Path locateHarness() {
164 		var basedir = System.getProperty("basedir");
165 		if (basedir != null) {
166 			var p = Path.of(basedir, "src/test/js/ribbon-dialog.cjs");
167 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
168 		}
169 		for (var rel : List.of(
170 			"src/test/js/ribbon-dialog.cjs",
171 			"juneau-rest/juneau-console-views/src/test/js/ribbon-dialog.cjs"
172 		)) {
173 			var p = Path.of(rel);
174 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
175 		}
176 		return null;
177 	}

179 	private static String runNode(Path harness, Path rendersJs, Path viewsJs, Path ribbonJs) throws Exception {
180 		var stdout = Files.createTempFile("ribbon-dialog-stdout-", ".json");
181 		var stderr = Files.createTempFile("ribbon-dialog-stderr-", ".txt");
182 		try {
183 			var pb = new ProcessBuilder(List.of(
184 					"node", harness.toString(), rendersJs.toString(), viewsJs.toString(), ribbonJs.toString()))
185 				.redirectOutput(stdout.toFile())
186 				.redirectError(stderr.toFile());
187 			var p = pb.start();
188 			if (!p.waitFor(30, TimeUnit.SECONDS)) {
189 				p.destroyForcibly();
190 				fail("ribbon-dialog.cjs did not finish within 30s; stderr:\n" + quietRead(stderr));
191 			}
192 			if (p.exitValue() != 0)
193 				fail("ribbon-dialog.cjs exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr)
194 					+ "\nstdout:\n" + quietRead(stdout));
195 			return Files.readString(stdout, UTF_8);
196 		} finally {
197 			Files.deleteIfExists(stdout);
198 			Files.deleteIfExists(stderr);
199 		}
200 	}
201 
202 	private static String quietRead(Path p) {
203 		try { return Files.readString(p, UTF_8); }
204 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
205 	}
