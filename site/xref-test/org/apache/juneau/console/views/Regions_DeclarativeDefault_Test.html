28  /**
29   * Always-on coverage for the declarative default: the {@code RegionDef} descriptor's client twin ({@code ctx.declared}),
30   * {@code ctx.fetchDeclared()}'s handshake-checked fetch, the reserved {@code "default"} populator's
31   * normative algorithm, R14a's pre-fetch matrix, the poll lifecycle, and - the ACCEPTANCE
32   * GATE for this child - the non-privilege proof.
33   *
34   * <p>
35   * <b>TESTS 11 AND 12 ARE A STOP GATE.</b> If either fails, the declarative default has a privileged path that an
36   * ordinary consumer populator cannot reach, and every child built on top of this one (host enrolment walks, the
37   * three real callers, the deprecation-window close) is building on a broken foundation.
38   *
39   * <p>
40   * The behavioral half runs the real runtime source - {@code juneau-regions.js} AND {@code juneau-helpers.js}, since
41   * the reserved default paints through {@code ctx.helpers[...]} - under a DOM shim and a controllable clock (see
42   * {@code src/test/js/regions-declarative-default.cjs}).
43   */

47  		var r = RegionsHarness.reportWithHelpers("regions-declarative-default.cjs");

52  	private static void assertAllTrue(Map<?,?> r, String... keys) {
53  		for (var k : keys)
54  			assertEquals(true, r.get(k), () -> k + " -> " + r.get(k) + " in " + r);
55  	}

57  	private static int intOf(Map<?,?> r, String key) {
58  		return ((Number)r.get(key)).intValue();
59  	}

122 	/**
123 	 * `dataUrl` set -> the region's OWN first populate call issues the fetch BEFORE the populate ever runs, and hands
124 	 * the unwrapped values map to the populate as {@code ctx.data}.  A later re-populate (here, {@code ctx.refresh()})
125 	 * does NOT pre-fetch: {@code ctx.data} is null and no second fetch is issued unless the populate asks for one.
126 	 */
127 	@Test void t15a_preFetch_firstPopulateGetsCtxDataBeforeItRuns_repopulateDoesNot() {
128 		var r = report();
129 		assertEquals(1, intOf(r, "t15a_fetchCountBeforeFirstPopulate"),
130 			"the pre-fetch must already be IN FLIGHT before the first populate call runs");
131 		assertEquals(Map.of("status", "ok"), r.get("t15a_ctxDataOnFirstPopulate"), r::toString);
132 		assertNull(r.get("t15a_ctxDataOnRepopulate"), "R14a: only the first-ever populate is pre-fetched");
133 		assertEquals(1, intOf(r, "t15a_fetchCountAfterRepopulate"),
134 			"a re-populate must not trigger a second automatic pre-fetch");
135 	}

194 	/**
195 	 * A consumer-registered populator that DELEGATES to {@code ctx.defaultPopulate} reproduces the reserved
196 	 * default's own DOM byte-for-byte, AND a second consumer populator that hand-reproduces the exact same
197 	 * algorithm using only the documented ctx contract (no call to {@code ctx.defaultPopulate} at all) ALSO
198 	 * reproduces it.  If the reserved default had a privileged path - a framework-internal field, a branch keyed on
199 	 * the reserved name, anything not on the documented ctx contract - one of these two ordinary registry entries
200 	 * would fail to reproduce the output and this test would catch it.
201 	 *
202 	 * <p>
203 	 * STOP GATE: if this fails, do not proceed - the layering this gate exists to prove is wrong.
204 	 */
205 	@Test void t11_STOP_GATE_consumerPopulatorsReproduceTheDefaultByteForByte() {
206 		var r = report();
207 		assertAllTrue(r, "t11_defaultVsDelegateIdentical", "t11_defaultVsHandRolledIdentical");
208 		// Sanity: the compared output is non-trivial (a real painted field grid), not two empty strings agreeing.
209 		assertTrue(((String)r.get("t11_sample")).contains("juneau-view-detail-field"), r::toString);
210 		assertTrue(((String)r.get("t11_sample")).contains(">Status<"), r::toString);
211 	}
