85  	// Registration-time failures cannot ride a contract island: an inline script calls the API directly. The caught
86  	// error goes to window.__probe, which console-shell.cjs copies into the report as "probe".
87  	private static String caught(String call) {
88  		return "<script>window.__probe = {}; try { " + call + " } catch (e) { window.__probe.caught = { name: e.name, code: e.code, message: e.message }; }</script>";
89  	}
90  
91  	private static void probeRegister() throws Exception {
92  		var shell = "<script src=\"" + SHELL + "\"></script>\n";
93  		var badType = shell + caught("JuneauConsole.registerCard('Bad_Type', function(){});");
94  		var badHandler = shell + caught("JuneauConsole.registerCard('ok', 'nope');");
95  		var notMounted = shell + caught("JuneauConsole.refreshCard('never-mounted');");
96  		var badQueueEntry = "<script>(window.JuneauConsoleCards = window.JuneauConsoleCards || []).push(['oops', 'nope', 'extra']);</script>\n" + shell;
97  		registerReport = ConsoleBrowserFixture.create("loud-failure-register")
98  			.page("e20", ORIGIN + "/e20", noIslandPage(badType))
99  			.page("e21", ORIGIN + "/e21", noIslandPage(badHandler))
100 			.page("e22", ORIGIN + "/e22", noIslandPage(notMounted))
101 			.page("e23", ORIGIN + "/e23", noIslandPage(badQueueEntry))
102 			.run();
103 	}

210 	private static void assertCardLevelFailure(Map<String,Object> r, String message) {
211 		assertEquals(List.of(), r.get("banner"), () -> "card-level failure must not reach the page banner: " + r);
212 		assertEquals(message, queries(r).get("cardError"), () -> "card body does not show the failure: " + r);
213 		assertTrue(((List<?>)r.get("consoleErrors")).contains(message), () -> "console.error did not log '" + message + "': " + r);
214 		assertNotNull(r.get("mounted"), () -> "the mount must still finish: " + r);
215 	}
