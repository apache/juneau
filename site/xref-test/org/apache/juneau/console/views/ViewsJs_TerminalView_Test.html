65  	static Map<?,?> report(String harness) {
66  		var r = RegionsHarness.reportWith(harness, ViewsMixin.HELPERS_JS_RESOURCE, ViewsMixin.RUN_VIEW_JS_RESOURCE, VECTORS);
67  		assumeTrue(r != null, "node not available or " + harness + " not found - skipped");
68  		return r;
69  	}
70  
71  	static void assertAllTrue(Map<?,?> r, String... keys) {
72  		for (var k : keys)
73  			assertEquals(true, r.get(k), () -> k + " -> " + r.get(k) + " in " + r);
74  	}
75  
76  	private static Map<?,?> r() { return report("terminal-view.cjs"); }
