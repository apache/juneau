17  package org.apache.juneau.console.views;
18  
19  import static org.apache.juneau.console.views.ConsoleBrowserFixture.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.util.*;
23  
24  import org.junit.jupiter.api.*;
25  import org.junit.jupiter.api.condition.*;
26  
27  /**
28   * The field grid of a real row detail in Chromium: each label sits beside its value on one row, and rows are
29   * separated by a hairline.
30   */
31  @EnabledIfSystemProperty(named=ConsoleBrowserFixture.GATE, matches="true",
32  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -pl juneau-rest/juneau-console-views -am test` from the repo root")
33  class DetailFieldGrid_BrowserTest extends org.apache.juneau.TestBase {
34  
35  	private static final String PAGE = """
36  		{"contractVersion":"1","title":"Pets","cards":[
37  		 {"id":"t1","type":"datatables","table":{
38  		   "rows":[{"id":"p1","name":"Rex","species":"dog"}],
39  		   "columns":[{"key":"id","label":"Id"},{"key":"name","label":"Name"},{"key":"species","label":"Species"}],
40  		   "detail":{"mode":"inline","title":"Pet {id}","tabs":[
41  		     {"id":"main","kind":"fields","title":"Main","fields":["name","species"]}]}}}]}
42  		""";
43  
44  	private static final String PROBE = """
45  		(async function () {
46  		  var p = window.__probe = window.__probe || {};
47  		  try {
48  		    for (var i = 0; i < 100 && !document.querySelector('#t1-body .juneau-view-detail-field'); i++)
49  		      await new Promise(function (r) { setTimeout(r, 50); });
50  		    var f = document.querySelector('#t1-body .juneau-view-detail-field');
51  		    var t = f.querySelector('.juneau-view-detail-field-title').getBoundingClientRect();
52  		    var v = f.querySelector('.juneau-view-detail-field-value').getBoundingClientRect();
53  		    var cs = getComputedStyle(f);
54  		    p.labelTop = t.top; p.valueTop = v.top; p.labelRight = t.right; p.valueLeft = v.left;
55  		    p.sepWidth = cs.borderBottomWidth; p.sepColor = cs.borderBottomColor;
56  		    p.gridClass = f.parentElement.className;
57  		    p.done = true;
58  		  } catch (e) { p.err = String(e && e.stack || e); }
59  		})();
60  		""";
61  
62  	private static Map<String,Map<String,Object>> report;
63  
64  	@BeforeAll
65  	static void run() throws Exception {
66  		var views = viewsPack();
67  		report = ConsoleBrowserFixture.create("detail-field-grid")
68  			.page("grid", ORIGIN + "/grid", contractPage(PAGE, "", "", views))
69  			.waitFor("#t1-body tbody td.sorting_1")
70  			.click("#t1-body .juneau-view-detail-toggle")
71  			.eval(PROBE)
72  			.waitFor("body")
73  			.run();
74  	}
75  
76  	private static Map<String,Object> p() {
77  		var r = report.get("grid");
78  		assertNoShellErrors(r);
79  		var p = probe(r);
80  		assertEquals(Boolean.TRUE, p.get("done"), () -> p.get("err") + " " + p);
81  		return p;
82  	}
83  
84  	private static double d(Map<String,Object> p, String k) {
85  		return ((Number)p.get(k)).doubleValue();
86  	}
87  
88  	@Test void a01_labelAndValueShareARow() {
89  		var p = p();
90  		assertEquals(d(p, "labelTop"), d(p, "valueTop"), 2.0, p::toString);
91  		assertTrue(d(p, "valueLeft") >= d(p, "labelRight"), p::toString);
92  	}
93  
94  	@Test void a02_rowsHaveASeparator() {
95  		var p = p();
96  		assertEquals("1px", p.get("sepWidth"), p::toString);
97  		assertNotEquals("rgba(0, 0, 0, 0)", p.get("sepColor"), p::toString);
98  	}
99  
100 	@Test void a03_gridCarriesTheInlineLayout() {
101 		var p = p();
102 		assertTrue(String.valueOf(p.get("gridClass")).contains("juneau-view-detail-fields-inline"), p::toString);
103 	}
104 }
