40  	@SuppressWarnings({
41  		"java:S8786" // Test-only regex over a short fixture.
42  	})
43  	private static final Pattern COMMENT = Pattern.compile("/\\*.*?\\*/", Pattern.DOTALL);
44  
45  	@SuppressWarnings({
46  		"java:S5852", // Possessive quantifiers: no backtracking; input is the module's own bundled stylesheet.
47  		"java:S8786" // Test-only regex over a short fixture.
48  	})
49  	private static final Pattern RULE = Pattern.compile("([^{}]++)\\{([^}]*+)\\}");
50  
51  	private static final Pattern LITERAL_COLOR = Pattern.compile("#[0-9a-fA-F]{3,8}\\b|\\brgba?\\(|\\bhsla?\\(");
52  
53  	private static final Pattern VAR_CALL = Pattern.compile("var\\([^()]*+\\)");
54  
55  	private static String read(Class<?> anchor, String path) throws IOException {
56  		try (var in = anchor.getResourceAsStream(path)) {
57  			assertNotNull(in, path);
58  			return new String(in.readAllBytes(), UTF_8);
59  		}
60  	}

74  	private static String norm(String s) {
75  		return s.trim().replaceAll("\\s++", " ");
76  	}
77  
78  	// Every declaration of 'prop' in a rule whose selector list contains exactly 'selector', whitespace-normalized.
79  	private static List<String> decls(String css, String selector, String prop) {
80  		var out = new ArrayList<String>();
81  		var m = RULE.matcher(COMMENT.matcher(css).replaceAll(""));
82  		while (m.find()) {
83  			var matches = false;
84  			for (var s : m.group(1).split(","))
85  				if (norm(s).equals(selector))
86  					matches = true;
87  			if (! matches)
88  				continue;
89  			for (var d : m.group(2).split(";")) {
90  				var i = d.indexOf(':');
91  				if (i > 0 && d.substring(0, i).trim().equals(prop))
92  					out.add(norm(d.substring(i + 1)));
93  			}
94  		}
95  		return out;
96  	}
97  
98  	private static void assertTokenSpent(String css, String selector, String prop) {
99  		var values = decls(css, selector, prop);
100 		assertFalse(values.isEmpty(), () -> selector + " declares no " + prop);
101 		for (var v : values) {
102 			assertTrue(v.contains("var(--jc-"), () -> selector + " { " + prop + ": " + v + " } does not spend a --jc- token");
103 			var bare = VAR_CALL.matcher(v).replaceAll("");
104 			assertFalse(LITERAL_COLOR.matcher(bare).find(), () -> selector + " { " + prop + ": " + v + " } carries a bare colour literal");
105 		}
106 	}

108 	// The compact value of every :root declaration in the stylesheet.
109 	private static Map<String,String> rootDecls(String css) {
110 		var out = new LinkedHashMap<String,String>();
111 		var m = RULE.matcher(COMMENT.matcher(css).replaceAll(""));
112 		while (m.find()) {
113 			if (! norm(m.group(1)).equals(":root"))
114 				continue;
115 			for (var d : m.group(2).split(";")) {
116 				var i = d.indexOf(':');
117 				if (i > 0)
118 					out.put(d.substring(0, i).trim(), d.substring(i + 1).replaceAll("\\s++", ""));
119 			}
120 		}
121 		return out;
122 	}

165 	@Test void a03_textRolesUsed() throws Exception {
166 		var c = chrome();
167 		assertEquals(List.of("var(--jc-danger-text)"), decls(c, ".jc-btn-danger", "color"));
168 		assertEquals(List.of("var(--jc-success-text)"), decls(c, ".jc-btn-success", "color"));
169 		assertEquals(List.of("var(--jc-danger-text)"), decls(c, ".jc-editable-field-error", "color"));
170 		assertEquals(List.of("var(--jc-danger-text)"), decls(c, ".jc-toast.is-error", "color"));
171 		assertEquals(List.of("var(--jc-success-text)"), decls(c, ".jc-toast.is-success", "color"));
172 		assertEquals(List.of("var(--jc-danger-text)"), decls(c, ".juneau-view-dialog-dismiss:hover", "color"));
173 		assertEquals(List.of("var(--jc-danger-text)"), decls(c, ".jc-dash-error", "color"));
174 		assertEquals(List.of("var(--jc-danger)"), decls(c, ".juneau-view-dialog-confirm.juneau-view-dialog-confirm-danger", "background-color"));
175 		assertEquals(List.of("var(--jc-success)"), decls(c, ".juneau-view-toggle:checked", "background-color"));
