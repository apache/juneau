28  /**
29   * The status dot in a real browser (spec section 6): its colour per state for each stock palette in both schemes, the
30   * dashed ring that marks paused, and that the dot shows no text.
31   *
32   * <p>
33   * The markup is the DOM {@code wireTablePolling} (juneau-views.js) builds in the {@code status} group, written out
34   * statically with one dot per state.
35   */
36  @SuppressWarnings({
37  	"unchecked" // The probe report is parsed into raw Maps.
38  })
39  @EnabledIfSystemProperty(named=ConsoleBrowserFixture.GATE, matches="true",
40  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -pl juneau-rest/juneau-console-views -am test` from the repo root")
41  class Status_BrowserTest extends TestBase {
42  
43  	private static final List<String> PALETTES = List.of("open", "gray", "light-brown", "light-red", "red");
44  	private static final List<String> SCHEMES = List.of("light", "dark");
45  	private static final List<String> STATES = List.of("idle", "refreshing", "error", "paused");
46  
47  	private static final String BODY = """
48  		<main class="jc-main"><div class="jc-card"><div class="juneau-view-toolbar-right">
49  		<span class="juneau-view-ribbon-group" data-juneau-group="status">
50  		<span class="juneau-view-staleness" id="idle" role="img" data-state="idle" data-jc-tip="Updated just now"></span>
51  		<span class="juneau-view-staleness" id="refreshing" role="img" data-state="refreshing" data-jc-tip="Refreshing"></span>
52  		<span class="juneau-view-staleness" id="error" role="img" data-state="error" data-jc-tip="Refresh failed"></span>
53  		<span class="juneau-view-staleness" id="paused" role="img" data-state="paused" data-jc-tip="Paused"></span>
54  		</span></div></div></main>
55  		<div id="tokprobe" style="display:none"></div>
56  		""";
57  
58  	private static final String PROBE = """
59  		<script>
60  		(function () {
61  			window.__probe = { toJSON: function () {
62  				var t = document.getElementById('tokprobe');
63  				var tok = {};
64  				['--jc-text-muted', '--jc-success', '--jc-danger-text'].forEach(function (n) {
65  					t.style.color = 'var(' + n + ')'; tok[n] = getComputedStyle(t).color;
66  				});
67  				var dots = {};
68  				['idle', 'refreshing', 'error', 'paused'].forEach(function (id) {
69  					var el = document.getElementById(id), cs = getComputedStyle(el), r = el.getBoundingClientRect();
70  					dots[id] = { color: cs.color, bg: cs.backgroundColor, borderStyle: cs.borderTopStyle,
71  						opacity: cs.opacity, text: el.textContent, width: r.width, height: r.height };
72  				});
73  				return { tok: tok, dots: dots };
74  			} };
75  		})();
76  		</script>
77  		""";
78  
79  	private static Map<String,Map<String,Object>> report;
80  
81  	private static String caseName(String palette, String scheme) {
82  		return palette + "-" + scheme;
83  	}
84  
85  	private static String page(String palette, String scheme) {
86  		var attrs = "data-jc-scheme=\"" + scheme + "\" data-jc-scheme-pin=\"" + scheme + "\"";
87  		return "<!DOCTYPE html>\n<html lang=\"en\" " + attrs + "><head><meta charset=\"utf-8\"><link rel=\"icon\" href=\"data:,\">\n"
88  			+ "<link rel=\"stylesheet\" href=\"" + CHROME_CSS + "\">\n<link rel=\"stylesheet\" href=\"" + DARK_CSS + "\">\n"
89  			+ "<link rel=\"stylesheet\" href=\"/juneau-console/juneau-theme-" + palette + ".css\">\n"
90  			+ "<link rel=\"stylesheet\" href=\"" + VIEWS_CSS + "\">\n"
91  			+ "</head><body>\n" + BODY + PROBE + "</body></html>\n";
92  	}
93  
94  	@BeforeAll
95  	static void run() throws Exception {
96  		var f = ConsoleBrowserFixture.create("status");
97  		for (var p : PALETTES)
98  			f.asset("/juneau-console/juneau-theme-" + p + ".css", resource("/org/apache/juneau/console/juneau-theme-" + p + ".css"), "text/css");
99  		for (var p : PALETTES)
100 			for (var s : SCHEMES)
101 				f.page(caseName(p, s), ORIGIN + "/" + caseName(p, s), page(p, s));
102 		report = f.run();
103 	}
104 
105 	private static void eachCase(java.util.function.BiConsumer<String,Map<String,Object>> check) {
106 		for (var p : PALETTES)
107 			for (var s : SCHEMES) {
108 				var name = caseName(p, s);
109 				var r = report.get(name);
110 				assertNotNull(r, () -> "no report for " + name);
111 				assertEquals(List.of(), r.get("pageErrors"), () -> name + ": page errors");
112 				check.accept(name, probe(r));
113 			}
114 	}
115 
116 	private static String dot(Map<String,Object> pr, String state, String prop) {
117 		return String.valueOf(((Map<String,Map<String,Object>>)pr.get("dots")).get(state).get(prop));
118 	}
119 
120 	private static String tok(Map<String,Object> pr, String n) {
121 		return ((Map<String,String>)pr.get("tok")).get(n);
122 	}
123 
124 	@Test void a01_dotColourPerState() {
125 		eachCase((name, pr) -> {
126 			assertEquals(tok(pr, "--jc-text-muted"), dot(pr, "idle", "color"), name + ": idle is muted");
127 			assertEquals(tok(pr, "--jc-success"), dot(pr, "refreshing", "color"), name + ": refreshing is success");
128 			assertEquals(tok(pr, "--jc-danger-text"), dot(pr, "error", "color"), name + ": error is danger-text");
129 			assertEquals(tok(pr, "--jc-text-muted"), dot(pr, "paused", "color"), name + ": paused is muted");
130 			assertEquals(tok(pr, "--jc-success"), dot(pr, "refreshing", "bg"), name + ": refreshing fills with its colour");
131 		});
132 	}
133 
134 	@Test void a02_statesAreDistinguishable() {
135 		eachCase((name, pr) -> {
136 			assertNotEquals(dot(pr, "idle", "color"), dot(pr, "refreshing", "color"), name);
137 			assertNotEquals(dot(pr, "idle", "color"), dot(pr, "error", "color"), name);
138 			assertNotEquals(dot(pr, "refreshing", "color"), dot(pr, "error", "color"), name);
139 		});
140 	}
141 
142 	@Test void a03_pausedIsADashedRingNotAFilledDot() {
143 		eachCase((name, pr) -> {
144 			assertEquals("dashed", dot(pr, "paused", "borderStyle"), name);
145 			assertEquals("rgba(0, 0, 0, 0)", dot(pr, "paused", "bg"), name + ": paused is hollow");
146 			assertEquals("0.7", dot(pr, "paused", "opacity"), name);
147 			for (var s : List.of("idle", "refreshing", "error"))
148 				assertEquals("none", dot(pr, s, "borderStyle"), name + ": " + s + " has no ring");
149 		});
150 	}
151 
152 	@Test void a04_dotIsSmallAndShowsNoText() {
153 		eachCase((name, pr) -> {
154 			for (var s : STATES) {
155 				assertEquals("", dot(pr, s, "text"), name + ": " + s + " has no visible text");
156 				assertTrue(Double.parseDouble(dot(pr, s, "width")) < 20, name + ": " + s + " width " + dot(pr, s, "width"));
157 				assertEquals(dot(pr, s, "width"), dot(pr, s, "height"), name + ": " + s + " is round");
158 			}
159 		});
160 	}
