13  package org.apache.juneau.console.views;
14  
15  import static org.junit.jupiter.api.Assertions.*;
16  import static org.junit.jupiter.api.Assumptions.*;
17  
18  import java.util.*;
19  
20  import org.junit.jupiter.api.*;
21  
22  /** The {@code chart} region type and its built-in populator, run through the Node harness. */
23  class RegionsChart_Test {
24  
25  	private static Map<?,?> report() {
26  		var r = RegionsHarness.reportWithHelpers("regions-chart.cjs");
27  		assumeTrue(r != null, "node not available or regions-chart.cjs not found - behavioral layer skipped");
28  		assertNull(r.get("crash"), () -> String.valueOf(r.get("crash")));
29  		return r;
30  	}
31  
32  	@Test void a01_chartIsARegionType() {
33  		assertTrue(String.valueOf(report().get("types")).contains("chart"));
34  	}
35  
36  	@Test void a02_fetchesItsOwnDataUrl_andMountsMappedRows() {
37  		var r = report();
38  		assertEquals(List.of("/pets/s-7/price-series"), r.get("fetches"));
39  		assertEquals(1, ((Number)r.get("mountCount")).intValue());
40  		assertEquals(true, r.get("sameEl"));
41  		assertEquals(List.of(Map.of("day", "Mon", "price", 1), Map.of("day", "Tue", "price", 2)), r.get("rows"));
42  		var card = (Map<?,?>)r.get("card");
43  		assertEquals("line", card.get("kind"));
44  		assertEquals(240, ((Number)card.get("height")).intValue());
45  		assertEquals("day", card.get("x"));
46  		assertEquals("price", ((Map<?,?>)((List<?>)card.get("series")).get(0)).get("field"));
47  		assertEquals("ok", r.get("state"));
48  		assertEquals(List.of(), r.get("errors"));
49  	}
50  
51  	@Test void a03_emptyRecords_showEmptyText() {
52  		var r = report();
53  		assertEquals(0, ((Number)r.get("emptyMounts")).intValue());
54  		assertEquals("No prices", r.get("emptyText"));
55  		assertEquals("No data", r.get("emptyDefault"));
56  	}
57  
58  	@Test void a04_chartingAbsent_showsStatusWithoutThrowing() {
59  		var r = report();
60  		assertEquals("charting is not loaded", r.get("absentText"));
61  	}
62  }
