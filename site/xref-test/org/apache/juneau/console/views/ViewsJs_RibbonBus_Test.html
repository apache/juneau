145 	private static String resource(String name) throws IOException {
146 		try (var in = ViewsMixin.class.getResourceAsStream(name)) {
147 			assertNotNull(in, name);
148 			return new String(in.readAllBytes(), UTF_8);
149 		}
150 	}

152 	/** Balanced-brace extractor. */
153 	private static String functionBody(String body, String signature) {
154 		var start = body.indexOf(signature);
155 		assertTrue(start >= 0, () -> "'" + signature + "' not found");
156 		var i = body.indexOf('{', start);
157 		var depth = 0;
158 		for (var j = i; j < body.length(); j++) {
159 			var c = body.charAt(j);
160 			if (c == '{') depth++;
161 			else if (c == '}' && --depth == 0) return body.substring(start, j + 1);
162 		}
163 		return fail("'" + signature + "' is unbalanced");
164 	}
