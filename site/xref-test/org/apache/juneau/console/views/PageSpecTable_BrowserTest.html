17  package org.apache.juneau.console.views;
18  
19  import static org.apache.juneau.console.views.ConsoleBrowserFixture.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.util.*;
23  
24  import org.apache.juneau.*;
25  import org.junit.jupiter.api.*;
26  import org.junit.jupiter.api.condition.*;
27  
28  /**
29   * A {@code PageSpec}-built {@code table(...)} card (parity case 03, {@code pagespec-corpus/03-table-every-feature.json})
30   * mounts a live DataTable in Chromium: a header cell for every visible column, none for the {@code defaultVisible(false)}
31   * column, the ribbon's buttons, the {@code linked} render and the {@code rowClass} rule applied to a fetched row. The contract
32   * shape is owned by {@code PageSpec_Parity_Test}; this only confirms the browser agrees.
33   */
34  @EnabledIfSystemProperty(named=ConsoleBrowserFixture.GATE, matches="true",
35  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -f juneau-rest/juneau-console-views/pom.xml test`")
36  class PageSpecTable_BrowserTest extends TestBase {
37  
38  	private static final String PROBE = """
39  		<script>
40  		window.__probe = {
41  		  get ths() { return Array.from(document.querySelectorAll("#slo-body table thead th")).map(function (e) { return e.textContent.trim(); }); },
42  		  get ribbon() { return Array.from(document.querySelectorAll("#slo-body [data-testid=ribbon] button")).map(function (e) { return e.getAttribute("aria-label"); }); }
43  		};
44  		</script>
45  		""";
46  
47  	private static final String ROWS = """
48  		[{"id":"1","pod":"p1","incidentKey":"I-1","incidentUrl":"/inc/1","rootCause":"disk","analysisState":"degraded"}]
49  		""";
50  
51  	private static Map<String,Map<String,Object>> report;
52  
53  	@BeforeAll
54  	static void probe() throws Exception {
55  		report = ConsoleBrowserFixture.create("pagespec-table")
56  			.asset("/rest/slo/data", ROWS, "application/json")
57  			.page("table", ORIGIN + "/page", corpusPage("03-table-every-feature", PROBE, viewsPack()),
58  				"row", "#slo-body tbody tr.row-degraded td:nth-child(2)",
59  				"link", "#slo-body tbody tr.row-degraded td:nth-child(3)")
60  			.waitFor("#slo-body tbody tr.row-degraded")
61  			.run();
62  	}
63  
64  	@SuppressWarnings({
65  		"unchecked" // Casts a parsed JSON value to its known Map/List shape.
66  	})
67  	private static List<String> probeList(String key) {
68  		return (List<String>)ConsoleBrowserFixture.probe(report.get("table")).get(key);
69  	}
70  
71  	@Test void a01_mountsWithoutErrors() {
72  		assertNoShellErrors(report.get("table"));
73  	}
74  
75  	// defaultVisible(false) only takes effect with columnConfig on (juneau-config.js); corpus 03 has no columnConfig, so the table keeps every column.
76  	@Test void a02_headerHasEveryColumn_withLabelsLiftedToTitles() {
77  		var ths = probeList("ths");
78  		assertTrue(ths.containsAll(List.of("Pod", "Incident", "Root cause")), ths::toString);
79  	}
80  
81  	@Test void a03_ribbonButtonsPresent() {
82  		var labels = probeList("ribbon");
83  		assertTrue(labels.contains("Refresh"), () -> "ribbon buttons: " + report.get("table"));
84  	}
85  
86  	@Test void a04_fetchedRow_isRenderedWithRowClassAndLinkedRender() {
87  		var q = queries(report.get("table"));
88  		assertEquals("p1", q.get("row"));
89  		assertEquals("I-1", q.get("link"));
90  	}
91  }
