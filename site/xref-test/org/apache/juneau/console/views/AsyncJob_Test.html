29  /**
30   * Unit test for the in-memory one-shot {@link AsyncJob}.
31   *
32   * <p>
33   * Covers the server-side hard limits that are BOTH heap bounds and data-egress disclosure bounds:
34   * the per-job output cap and the hard timeout; the reserved async terminal outcomes (cancelled vs
35   * cancelled-after-effect); the subscriber cap; the progress buffering / await used by the SSE stream; and the
36   * resume cursor the stream picks from a {@code Last-Event-ID} / {@code lastEventId} value.
37   */
38  class AsyncJob_Test extends TestBase {

40  	private static AsyncJob job() {
41  		return job(AsyncJobRegistry.MAX_OUTPUT_BYTES, AsyncJobRegistry.MAX_SUBSCRIBERS_PER_JOB, Duration.ofSeconds(120));
42  	}

44  	private static AsyncJob job(long maxOutputBytes, int maxSubscribers, Duration timeout) {
45  		return new AsyncJob("cap-id", Instant.parse("2026-08-19T00:00:00Z"), timeout, maxOutputBytes, maxSubscribers);
46  	}

86  	@Test void b03_cancelAfterEffectIsCancelledAfterEffect() {

102 	@Test void b05_markEffectStartedReportsWhetherTheEffectMayRun() {
103 		// The mark and the check are one lock acquisition, so a producer cannot start its effect on a settled job.
104 		var running = job();
105 		assertTrue(running.markEffectStarted());
106 		assertTrue(running.markEffectStarted());           // idempotent while running
107 		var cancelled = job();
108 		cancelled.cancel();
109 		assertFalse(cancelled.markEffectStarted());
110 		assertFalse(cancelled.effectStarted());
111 		assertEquals("cancelled", cancelled.result().outcome);
112 		var timedOut = job();
113 		timedOut.enforceTimeout(timedOut.deadline().plusMillis(1));
114 		assertFalse(timedOut.markEffectStarted());
115 	}

121 	@Test void c01_outputCapTerminatesJobAndDropsExcess() {
122 		var j = job(10, 2, Duration.ofSeconds(120));
123 		assertTrue(j.progress("12345"));           // 5 bytes -> ok
124 		assertFalse(j.progress("123456"));         // +6 -> 11 > 10 -> terminate, drop this event
125 		assertTrue(j.isTerminal());
126 		assertEquals("failure", j.result().outcome);
127 		assertTrue(j.result().message.contains("output-limit"), j.result().message);
128 		assertEquals(1, j.eventCount(), "the over-cap event must be dropped, not streamed");
129 		// Once terminal, further progress is a no-op.
130 		assertFalse(j.progress("x"));
131 		assertEquals(1, j.eventCount());
132 	}

141 	@Test void c03_emptyProgressEventsHitTheCapAndStopGrowing() {
142 		var j = job(100, 2, Duration.ofSeconds(120));
143 		for (var i = 0; i < 1000 && ! j.isTerminal(); i++)
144 			j.progress("");
145 		assertTrue(j.isTerminal(), "empty events must count toward the cap");
146 		assertEquals("failure", j.result().outcome);
147 		assertTrue(j.result().message.contains("output-limit"), j.result().message);
148 		var count = j.eventCount();
149 		assertTrue(count <= 100, "events must be bounded by the cap, was " + count);
150 		assertFalse(j.progress(""));
151 		assertEquals(count, j.eventCount());
152 	}

189 	@Test void e01_awaitUpdateReturnsBufferedEventsAndResultForTerminalJob() {
190 		var j = job();
191 		j.progress("p1");
192 		j.progress("p2");
193 		j.complete(ActionResult.success(null));
194 		var u = j.awaitUpdate(0, Duration.ofMillis(10));
195 		assertEquals(java.util.List.of("p1", "p2"), u.events());
196 		assertNotNull(u.result());
197 		assertEquals("success", u.result().outcome);
198 	}
199 
200 	@Test void e02_awaitUpdateBlocksThenWakesOnLiveEvent() {
201 		var j = job();
202 		var pool = Executors.newSingleThreadScheduledExecutor();
203 		try {
204 			pool.schedule(() -> j.progress("live"), 50, TimeUnit.MILLISECONDS);
205 			var u = j.awaitUpdate(0, Duration.ofSeconds(2));   // blocks until the scheduled progress arrives
206 			assertEquals(java.util.List.of("live"), u.events());
207 			assertNull(u.result());
208 		} finally {
209 			pool.shutdownNow();
210 		}
211 	}

246 	// h) resumeCursor: the stream's start position from a Last-Event-ID / lastEventId value
247 	//------------------------------------------------------------------------------------------------------------------

249 	private static AsyncJob jobWith(int events) {
250 		var j = job();
251 		for (var i = 0; i < events; i++)
252 			j.progress("e" + i);
253 		return j;
254 	}
255 
256 	@Test void h01_noValue_isLiveOnly_currentEventCount() {
257 		var j = jobWith(3);
258 		assertEquals(3, j.resumeCursor(null));
259 		assertEquals(3, j.resumeCursor(""));
260 		assertEquals(3, j.resumeCursor(" \t"));
261 		assertEquals(0, job().resumeCursor(null));
262 	}
263 
264 	@Test void h02_validValues_returnTheCursor() {
265 		var j = jobWith(3);
266 		assertEquals(0, j.resumeCursor("0"));
267 		assertEquals(2, j.resumeCursor("2"));
268 		assertEquals(3, j.resumeCursor("3"));
269 		assertEquals(2, j.resumeCursor(" 2 "));
270 		assertEquals(2, j.resumeCursor("\t2\t"));
271 		assertEquals(7, jobWith(8).resumeCursor("007"));
272 		assertEquals(0, job().resumeCursor("0"));
273 	}

278 		var j = jobWith(3);
