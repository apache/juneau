45  	/** Extracts a named function's body: from `function <name>(` to the next top-level `\n\t}`. */
46  	private static String functionBody(String body, String signature) {
47  		var start = body.indexOf(signature);
48  		assertTrue(start >= 0, () -> "'" + signature + "' not found:\n" + body);
49  		var end = body.indexOf("\n\t}", start);
50  		return body.substring(start, end < 0 ? body.length() : end);
51  	}
