212 	/** JSON numbers come back as Integer/Long/Double depending on the parser; compare them as ints. */
213 	private static Object normalize(Object o) {
214 		if (o instanceof Map<?,?> m) {
215 			var out = new LinkedHashMap<String,Object>();
216 			m.forEach((k, v) -> out.put((String)k, normalize(v)));
217 			return out;
218 		}
219 		if (o instanceof Number n && n.doubleValue() == n.intValue())
220 			return n.intValue();
221 		return o;
222 	}
