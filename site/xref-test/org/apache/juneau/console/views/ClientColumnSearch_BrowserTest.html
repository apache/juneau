33  /**
34   * The real-browser canary for client-mode column-search DSL evaluation: real DataTables 2.1.8, a
35   * client-side table, and {@code JuneauViews.init.setColumnExpr(...)} installing the {@code juneau-dsl}
36   * {@code column().search.fixed} predicate built by {@code juneau-search.js}.
37   *
38   * <p>
39   * The node-only {@link ViewsJs_ClientColumnSearch_Test} covers the semantics against fake column APIs; this canary
40   * proves the one thing a fake cannot - that the real library's {@code search.fixed} contract (argument order, raw
41   * row data, composition with the global search box, {@code col.search()} staying empty) matches what the views
42   * runtime assumes.  {@code client-column-search-browser.cjs} drives it with headless Chromium.
43   *
44   * <h5 class='section'>Off by default:</h5>
45   * <p>
46   * Disabled unless the {@value #GATE} system property is set, which only the module's opt-in {@code js-tests} Maven
47   * profile does: {@code mvn -Pjs-tests -f juneau-rest/juneau-console-views/pom.xml test}.  That profile also
48   * installs the {@code jquery} and {@code datatables.net} npm packages the fixture page inlines.
49   *
50   * <h5 class='section'>See Also:</h5>
51   * <ul>
52   * 	<li class='jc'>{@link DataTablesAjax_BrowserTest} &mdash; the sibling real-DataTables canary (server mode).
53   * </ul>
54   */
55  @EnabledIfSystemProperty(named=ClientColumnSearch_BrowserTest.GATE, matches="true",
56  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -f juneau-rest/juneau-console-views/pom.xml test`")
57  class ClientColumnSearch_BrowserTest extends TestBase {

64  	private static String resource(String path) throws IOException {
65  		try (var in = ViewsMixin.class.getResourceAsStream(path)) {
66  			assertNotNull(in, () -> "missing classpath resource: " + path);
67  			return new String(in.readAllBytes(), UTF_8);
68  		}
69  	}
70  
71  	private static String requiredProperty(String name) {
72  		var v = System.getProperty(name);
73  		assertNotNull(v, () -> "-D" + name + " not set; the js-tests profile is responsible for providing it");
74  		return v;
75  	}

77  	@BeforeAll
78  	static void probe() throws Exception {
79  		var dir = Path.of(requiredProperty("juneau.jsTests.dir"));
80  		var harness = Path.of(requiredProperty("juneau.jsTests.harness")).getParent().resolve("client-column-search-browser.cjs");
81  		var nodeModules = dir.resolve("node_modules");
82  		var fixture = "<!DOCTYPE html><html><head><meta charset=\"utf-8\"></head><body>\n"
83  			+ "<table id=\"t\"></table>\n"
84  			+ "<script>\n" + Files.readString(nodeModules.resolve("jquery/dist/jquery.min.js")) + "\n</script>\n"
85  			+ "<script>\n" + Files.readString(nodeModules.resolve("datatables.net/js/dataTables.min.js")) + "\n</script>\n"
86  			+ "<script>\n" + resource(ViewsMixin.RENDERS_JS_RESOURCE) + "\n</script>\n"
87  			+ "<script>\n" + resource(ViewsMixin.VIEWS_JS_RESOURCE) + "\n</script>\n"
88  			+ "<script>\n" + resource(ViewsMixin.URLSTATE_JS_RESOURCE) + "\n</script>\n"
89  			+ "<script>\n" + resource(ViewsMixin.SEARCH_JS_RESOURCE) + "\n</script>\n"
90  			+ "<script>\n"
91  			+ "window.__clientDt = new DataTable('#t', { data: ["
92  			+ "{id:'r1',status:'Triaged',name:'alpha'},{id:'r2',status:'New',name:'beta'},"
93  			+ "{id:'r3',status:'Closed',name:'gamma'},{id:'r4',status:'Open',name:'delta'},"
94  			+ "{id:'r5',status:'Tri',name:'epsilon'}],"
95  			+ " columns: [{data:'id',title:'ID'},{data:'status',title:'Status'},{data:'name',title:'Name'}] });\n"
96  			+ "</script>\n"
97  			+ "</body></html>";
98  		var fixtureFile = Files.createDirectories(dir.resolve("fixtures")).resolve("client-column-search.html");
99  		Files.write(fixtureFile, fixture.getBytes(UTF_8));
100 		report = Json.to(run(dir, harness, fixtureFile), Map.class);
101 	}
102 
103 	/** Runs the prober, failing with its stderr attached (its exit code alone is not a diagnosis). */
104 	private static String run(Path dir, Path harness, Path fixture) throws Exception {
105 		var cmd = List.of(System.getProperty("juneau.jsTests.node", "node"), harness.toString(), fixture.toString());
106 		var stdout = dir.resolve("client-column-search-stdout.json");
107 		var stderr = dir.resolve("client-column-search-stderr.txt");
108 		var pb = new ProcessBuilder(cmd).redirectOutput(stdout.toFile()).redirectError(stderr.toFile());
109 		pb.environment().put("NODE_PATH", dir.resolve("node_modules").toString());
110 		pb.environment().put("PLAYWRIGHT_BROWSERS_PATH", requiredProperty("juneau.jsTests.browsers"));
111 		var p = pb.start();
112 		if (!p.waitFor(3, TimeUnit.MINUTES)) {
113 			p.destroyForcibly();
114 			fail("prober did not finish within 3m; stderr:\n" + quietRead(stderr));
115 		}
116 		assertEquals(0, p.exitValue(), () -> "prober exited non-zero; stderr:\n" + quietRead(stderr));
117 		return Files.readString(stdout);
118 	}
119 
120 	private static String quietRead(Path p) {
121 		try {
122 			return Files.readString(p);
123 		} catch (IOException e) {
124 			return "<unreadable: " + e + ">";
125 		}
126 	}
