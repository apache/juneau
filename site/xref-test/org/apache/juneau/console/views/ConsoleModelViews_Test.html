15  import static org.junit.jupiter.api.Assertions.*;
16  import static org.junit.jupiter.api.Assumptions.*;
17  
18  import java.util.*;
19  import java.util.stream.*;
20  
21  import org.apache.juneau.*;
22  import org.junit.jupiter.api.*;

24  /**
25   * Node-sandbox coverage for the hooks the client object model has in {@code juneau-views.js} and
26   * {@code juneau-ribbon.js} (table handle, draw cycle, detail rows, ribbon), through {@code model-views.cjs}.
27   */
28  class ConsoleModelViews_Test extends TestBase {
29  
30  	private static final String MODEL_JS = "/org/apache/juneau/views/juneau-model.js";
31  
32  	private static Map<String,Object> report;
33  
34  	@BeforeAll
35  	static void run() {
36  		report = BusHarness.run("model-views.cjs", ViewsMixin.BUS_JS_RESOURCE, ViewsMixin.RENDERS_JS_RESOURCE, ViewsMixin.VIEWS_JS_RESOURCE,
37  			ViewsMixin.RIBBON_JS_RESOURCE, ViewsMixin.SEARCH_JS_RESOURCE, MODEL_JS);
38  	}
39  
40  	private static final String[] KEYS = {
41  		"tables_handle_is_model_Table",
42  		"tables_handle_four_members",
43  		"tables_ready_resolves_with_table",
44  		"draw_hooks_bound_to_own_table_not_nested",
45  		"draw_keyed_expanded_detaches_not_destroys",
46  		"draw_keyed_default_destroys_detail",
47  		"draw_positional_destroyed",
48  		"draw_missing_row_evicted",
49  		"registry_stable_after_draws",
50  		"expand_creates_detail_instance_and_tabs",
51  		"collapse_destroys_detail_keeps_row",
52  		"row_expanded_collapsed_events",
53  		"detail_actions_painted_and_click_runs_handler",
54  		"ribbon_item_handles_by_id",
55  		"ribbon_missing_id_custom_item_E121",
56  		"ribbon_stock_default_ids_client_mirror",
57  		"ribbon_duplicate_stock_kind_E121",
58  		"ribbon_new_item_types_have_handles",
59  		"ribbon_hidden_items_leave_no_handle",
60  		"detail_tab_setLabel_on_strip_button",
61  		"ribbon_action_item_click_handler",
62  		"ribbon_stock_refresh_unchanged",
63  		"table_loaded_event_count",
64  	};
65  
66  	@TestFactory
67  	Stream<DynamicTest> checks() {
68  		return Arrays.stream(KEYS).map(k -> DynamicTest.dynamicTest(k, () -> {
69  			assumeTrue(report != null, "node (or the harness) is not available");
70  			assertTrue(report.containsKey(k), "missing report key: " + k);
71  			assertEquals(true, report.get(k));
72  		}));
73  	}
74  }
