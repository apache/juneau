17  package org.apache.juneau.console.views;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.*;
24  import org.junit.jupiter.api.*;
25  
26  /** {@link NestedSlot}: the nested-table slot emitted for a table tab. */
27  @SuppressWarnings("unchecked")
28  class NestedSlot_Test extends TestBase {
29  
30  	private static TableSpec orders(String id) {
31  		return TableSpec.create(id).dataUrl("/orders/rows").columns(Column.create("id"), Column.create("status"));
32  	}
33  
34  	@Test void a01_shape() {
35  		var n = NestedSlot.of(orders("orders"), "petId");
36  		assertEquals("2", n.get("contractVersion"));
37  		assertEquals("petId", n.get("parentScopeParam"));
38  		var view = (Map<String,Object>) n.get("view");
39  		assertEquals("orders", view.get("id"));
40  		assertNotNull(view.get("columns"));
41  		var meta = (Map<String,Object>) n.get("meta");
42  		assertEquals(ViewsMixin.CONTRACT_VERSION, meta.get("contractVersion"));
43  		assertEquals("/orders/rows", meta.get("dataUrl"));
44  		var col = ((List<Map<String,Object>>) meta.get("columns")).get(0);
45  		assertEquals("id", col.get("data"));
46  		assertNull(col.get("key"));
47  	}
48  
49  	@Test void a02_noRowsDataUrl_rejected() {
50  		assertThrows(IllegalArgumentException.class, () -> NestedSlot.of(TableSpec.create("t"), "petId"));
51  	}
52  
53  	@Test void a03_depthOver2_rejected() {
54  		var inner = orders("inner");
55  		var mid = orders("mid").detail(RowDetail.inline().tab(DetailTab.table("Inner", inner)));
56  		assertThrows(IllegalArgumentException.class, () -> NestedSlot.of(mid, "petId"));
57  	}
58  
59  	@Test void a04_duplicateViewId_rejected() {
60  		var d = RowDetail.inline()
61  			.tab(DetailTab.table("A", orders("same")).id("a"))
62  			.tab(DetailTab.table("B", orders("same")).id("b"));
63  		assertThrows(IllegalArgumentException.class, d::toMap);
64  	}
65  
66  	@Test void a05_selectionAndDetailCarried() {
67  		var t = orders("o").selection(SelectionDef.create("id")).detail(RowDetail.inline());
68  		var n = NestedSlot.of(t, "petId");
69  		assertNotNull(n.get("selection"));
70  		assertNotNull(n.get("detail"));
71  	}
72  
73  	@Test void a06_tableTab_emitsNestedAndRowsCount() {
74  		var m = DetailTab.table("Orders", orders("o")).countFromRows().toMap();
75  		assertEquals("rows", m.get("count"));
76  		assertNotNull(m.get("nested"));
77  	}
78  }
