40  /**
41   * The real-browser, real-server canary for View Settings restore end to end: a column hidden through the View Settings
42   * dialog
43   * <i>stays</i> hidden after a real page reload, in both server-mode and client-mode tables.
44   *
45   * <p>
46   * {@code ViewsJs_ConfigApplication_Test} and {@code ViewsJs_Reinit_Test} prove, against a recording fake DataTables,
47   * that the persisted blob reaches the constructor options.  Only a real browser proves the whole loop: a real click
48   * on the gear, a real checkbox, a real Apply, the blob landing in real {@code localStorage}, and the next page load
49   * (same {@code http://} origin) rebuilding the real DataTables grid without the hidden column.
50   *
51   * <h5 class='section'>Off by default:</h5>
52   * <p>
53   * Disabled unless the {@value #GATE} system property is set, which only the module's opt-in {@code js-tests} Maven
54   * profile does: {@code mvn -Pjs-tests -f juneau-rest/juneau-console-views/pom.xml test}.
55   *
56   * <h5 class='section'>See Also:</h5>
57   * <ul>
58   * 	<li class='jc'>{@link ViewSettingsA11y_BrowserTest} &mdash; the sibling canary for the dialog's a11y behavior.
59   * 	<li class='jc'>{@link ViewsServerNested_BrowserTest} &mdash; the sibling real-server canary this one borrows its
60   * 		HTTP fixture from.
61   * </ul>
62   *
63   * @since 10.0.0
64   */

94  	private int port;

96  	@BeforeEach
97  	void startServer() throws IOException {
98  		server = HttpServer.create(new InetSocketAddress(InetAddress.getLoopbackAddress(), 0), 0);
99  		port = server.getAddress().getPort();
100 		var context = InMemoryBeanQueryContext.create(Row.class).columns("id", "name", "status").build();
101 
102 		server.createContext("/query", ex -> {
103 			try {
104 				if (neq(ex.getRequestMethod(), "POST")) {
105 					ex.sendResponseHeaders(405, -1);
106 					return;
107 				}
108 				String body;
109 				try (var in = ex.getRequestBody()) {
110 					body = new String(in.readAllBytes(), UTF_8);
111 				}
112 				String json;
113 				try (var s = context.getSession(rows())) {
114 					json = Json.of(DataTablesQuery.run(Json.to(body, DataTablesRequest.class), s));
115 				}
116 				write(ex, "application/json", json);
117 			} catch (Exception e) {
118 				sendFailure(ex, e);
119 			}
120 		});
121 		server.createContext("/data", ex -> write(ex, "application/json", Json.of(rows())));
122 		server.createContext("/server", ex -> {
123 			try {
124 				write(ex, "text/html;charset=utf-8", fixturePage("server", "/query"));
125 			} catch (Exception e) {
126 				sendFailure(ex, e);
127 			}
128 		});
129 		server.createContext("/client", ex -> {
130 			try {
131 				write(ex, "text/html;charset=utf-8", fixturePage("client", "/data"));
132 			} catch (Exception e) {
133 				sendFailure(ex, e);
134 			}
135 		});
136 		server.start();
137 	}

139 	@AfterEach
140 	void stopServer() {
141 		server.stop(0);
142 	}

144 	private static void sendFailure(HttpExchange ex, Exception e) throws IOException {
145 		var sw = new StringWriter();
146 		e.printStackTrace(new PrintWriter(sw));
147 		System.err.println(sw);
148 		var bytes = sw.toString().getBytes(UTF_8);
149 		ex.getResponseHeaders().add("Content-Type", "text/plain;charset=utf-8");
150 		ex.sendResponseHeaders(500, bytes.length);
151 		try (var os = ex.getResponseBody()) {
152 			os.write(bytes);
153 		}
154 	}
155 
156 	private static void write(HttpExchange ex, String contentType, String body) throws IOException {
157 		var bytes = body.getBytes(UTF_8);
158 		ex.getResponseHeaders().add("Content-Type", contentType);
159 		ex.sendResponseHeaders(200, bytes.length);
160 		try (var os = ex.getResponseBody()) {
161 			os.write(bytes);
162 		}
163 	}

165 	private static String resource(Class<?> anchor, String path) throws IOException {
166 		try (var in = anchor.getResourceAsStream(path)) {
167 			assertNotNull(in, () -> "missing classpath resource: " + path + " (relative to " + anchor.getName() + ")");
168 			return new String(in.readAllBytes(), UTF_8);
169 		}

172 	/** Load order per the module's own doc comment: renders, views, urlstate, search, pagestate, config. */
173 	private static String fixturePage(String dataMode, String dataUrl) throws IOException {
174 		var nodeModules = Path.of(requiredProperty("juneau.jsTests.dir")).resolve("node_modules");
175 		var jquery = Files.readString(nodeModules.resolve("jquery/dist/jquery.min.js"));
176 		var dataTablesJs = Files.readString(nodeModules.resolve("datatables.net/js/dataTables.min.js"));
177 		var meta = "{\"contractVersion\":\"5\",\"id\":\"t1\",\"dataMode\":\"" + dataMode + "\",\"dataUrl\":\"" + dataUrl + "\","
178 			+ "\"columnConfig\":{},"
179 			+ "\"columns\":[{\"data\":\"id\",\"title\":\"ID\"},{\"data\":\"name\",\"title\":\"Name\"},"
180 			+ "{\"data\":\"status\",\"title\":\"Status\"}]}";
181 		return "<!DOCTYPE html><html><head><meta charset=\"utf-8\"><style>\n"
182 			+ resource(ViewsMixin.class, ViewsMixin.VIEWS_CSS_RESOURCE) + "\n"
183 			+ resource(ViewsMixin.class, ViewsMixin.CONFIG_CSS_RESOURCE)
184 			+ "\n</style></head><body>\n"
185 			+ "<table data-juneau-view=\"t1\" id=\"t1\"></table>\n"
186 			+ "<script type=\"application/json\" id=\"juneau-view:t1\">" + meta + "</script>\n"
187 			+ "<script>\n" + jquery + "\n</script>\n"
188 			+ "<script>\n" + dataTablesJs + "\n</script>\n"
189 			+ "<script>\n" + resource(DataTablesMixin.class, "juneau-datatables.js") + "\n</script>\n"
190 			+ "<script>\n" + resource(ViewsMixin.class, ViewsMixin.RENDERS_JS_RESOURCE) + "\n</script>\n"
191 			+ "<script>\n" + resource(ViewsMixin.class, ViewsMixin.VIEWS_JS_RESOURCE) + "\n</script>\n"
192 			+ "<script>\n" + resource(ViewsMixin.class, ViewsMixin.URLSTATE_JS_RESOURCE) + "\n</script>\n"
193 			+ "<script>\n" + resource(ViewsMixin.class, ViewsMixin.SEARCH_JS_RESOURCE) + "\n</script>\n"
194 			+ "<script>\n" + resource(ViewsMixin.class, ViewsMixin.PAGESTATE_JS_RESOURCE) + "\n</script>\n"
195 			+ "<script>\n" + resource(ViewsMixin.class, ViewsMixin.CONFIG_JS_RESOURCE) + "\n</script>\n"
196 			+ "</body></html>";
197 	}

205 	private Map<?,?> runHarness() throws Exception {
206 		var dir = Path.of(requiredProperty("juneau.jsTests.dir"));
207 		var harness = Path.of(requiredProperty("juneau.jsTests.harness")).getParent().resolve("view-settings-reload-browser.cjs");
208 		var baseUrl = "http://127.0.0.1:" + port + "/";
209 
210 		var cmd = List.of(System.getProperty("juneau.jsTests.node", "node"), harness.toString(), baseUrl);
211 		var stdout = dir.resolve("view-settings-reload-stdout.json");
212 		var stderr = dir.resolve("view-settings-reload-stderr.txt");
213 		var pb = new ProcessBuilder(cmd).redirectOutput(stdout.toFile()).redirectError(stderr.toFile());
214 		pb.environment().put("NODE_PATH", dir.resolve("node_modules").toString());
215 		pb.environment().put("PLAYWRIGHT_BROWSERS_PATH", requiredProperty("juneau.jsTests.browsers"));
216 
217 		var p = pb.start();
218 		if (! p.waitFor(3, TimeUnit.MINUTES)) {
219 			p.destroyForcibly();
220 			fail("prober did not finish within 3m; stderr:\n" + quietRead(stderr));
221 		}
222 		assertEquals(0, p.exitValue(), () -> "prober exited non-zero; stderr:\n" + quietRead(stderr));
223 		return Json.to(Files.readString(stdout), Map.class);
224 	}

226 	private static String quietRead(Path p) {
227 		try {
228 			return Files.readString(p);
229 		} catch (IOException e) {
230 			return "<unreadable: " + e + ">";
231 		}
232 	}
