30  /**
31   * Row-detail tabs that own live resources (a chart, a nested table) in a real browser: collapsing a row and resetting
32   * a tab must both release what the tab built, so repeated cycles leave the live chart, table and region counts where
33   * they started.
34   */

36  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -f juneau-rest/juneau-console-views/pom.xml test`")

49  	private static String page() throws Exception {
50  		var orders = TableSpec.create("pet-orders").dataMode(TableSpec.DataMode.CLIENT).dataUrl("/api/orders")
51  			.columns(Column.create("id").label("Id"));
52  		var trend = RegionDef.chart("trend", "/api/trend").x("day").series("placed", "Placed").height(160);
53  		var detail = RowDetail.create("/api/pets/{id}/detail").title("Pet {id}")
54  			.tab(DetailTab.fields("Info", "name").id("info"))
55  			.tab(DetailTab.chart("Trend", trend).id("trend"))
56  			.tab(DetailTab.table("Orders", orders).id("orders").scopeParam("petId"));
57  		var t = TableSpec.create("pets").dataMode(TableSpec.DataMode.CLIENT).dataUrl("/api/pets")
58  			.selection(SelectionDef.create("id"))
59  			.columns(Column.create("id").label("Id"), Column.create("name").label("Name"))
60  			.detail(detail);
61  		var card = new LinkedHashMap<String,Object>();
62  		card.put("id", "pets");
63  		card.put("type", "datatables");
64  		card.put("title", "Pets");
65  		card.put("table", t.toCardBody());
66  		var page = new LinkedHashMap<String,Object>();
67  		page.put("contractVersion", "1");
68  		page.put("modelVersion", "1");
69  		page.put("title", "Pets");
70  		page.put("cards", List.of(card));
71  		return contractPage(Json.of(page), "", "", viewsPack()
72  			+ "<script src=\"" + ECHARTS + "\"></script>\n<script src=\"" + CHARTS + "\"></script>\n");
73  	}

144 	private static Map<String,Object> p(String name, String done) {
145 		var r = report.get(name);
146 		assertEquals(List.of(), r.get("pageErrors"), r::toString);
147 		var m = ConsoleBrowserFixture.probe(r);
148 		assertEquals(Boolean.TRUE, m.get(done), () -> String.valueOf(m.get("err")) + " " + m);
149 		return m;
150 	}

152 	@SuppressWarnings("unchecked")
153 	@Test void a01_collapsingRowsRepeatedly_releasesCharts_tablesAndRegions() {
154 		var m = p("collapse", "collapseDone");
155 		for (var s : (List<Map<String,Object>>)m.get("afterCollapse")) {
156 			assertEquals(0L, ((Number)s.get("c")).longValue(), m::toString);
157 			assertEquals(0L, ((Number)s.get("r")).longValue(), m::toString);
158 			assertEquals(((Number)((Map<String,Object>)m.get("base")).get("t")).longValue(), ((Number)s.get("t")).longValue(), m::toString);
159 		}
160 	}
161 
162 	@SuppressWarnings("unchecked")
163 	@Test void a02_resettingChartTab_keepsOneLiveChart() {
164 		var m = p("reset", "resetDone");
165 		var base = ((Number)((Map<String,Object>)m.get("resetBase")).get("c")).longValue();
166 		for (var s : (List<Map<String,Object>>)m.get("resetSeries"))
167 			assertEquals(base, ((Number)s.get("c")).longValue(), m::toString);
168 	}

170 	@Test void a03_resettingTableTab_keepsLiveTableCountFlat() {

173 		for (var n : (List<?>)m.get("tblSeries"))
174 			assertEquals(base, ((Number)n).longValue(), m::toString);
175 	}
