31  /**
32   * Option snapshots of the {@code juneau-charts.js} pipeline (frame, helper, decorators, patch, render mode,
33   * large-data defaults, {@code $fn}) against goldens under {@code src/test/resources/charts-goldens}.
34   *
35   * <p>
36   * The goldens follow the console-ui {@code GoldenFiles} rule: a missing golden is written and the test fails, and
37   * {@code -Djuneau.golden.update=true} rewrites them all.  Review the diff before re-running without it.  A golden file
38   * with no case in {@code NAMES} fails the test (delete it).
39   *
40   * <p>
41   * Besides the goldens the runner reports the warnings each case logged, the error each failing case threw, and a set of
42   * boolean behaviour checks (the mode a {@code $fn} is built in, {@code build()} returning a promise, frozen tokens,
43   * {@code __proto__} keys); each is asserted here.
44   *
45   * <p>
46   * The whole JS layer is skipped (via {@code assumeTrue}, the {@link NodeHarness} convention) when {@code node} is not
47   * installed, so on a node-less build these goldens guard nothing.
48   */
49  class ConsoleChartsOptions_Test extends TestBase {

96  	private static Map<?,?> report() {
97  		// The fn outputs (number, datetime) depend on the locale and the time zone.
98  		var r = NodeHarness.report(Map.of("LC_ALL", "en_US.UTF-8", "TZ", "UTC"), "charts-options.cjs", ConsoleChromeMixin.CHARTS_JS_RESOURCE);
99  		assumeTrue(r != null, "node (or charts-options.cjs) not available - JS layer skipped");
100 		return r;
101 	}

103 	// Unlike GoldenFiles.assertGolden, every missing or updated golden is written before the one failure.
104 	@Test void a01_snapshots_matchGoldens() throws IOException {
105 		var goldens = (Map<?,?>)report().get("goldens");
106 		assertEquals(NAMES, new ArrayList<>(goldens.keySet()));
107 		var written = new ArrayList<String>();
108 		for (var name : NAMES) {
109 			var file = Path.of("src/test/resources/charts-goldens", name + ".json");
110 			var actual = (String)goldens.get(name);
111 			if (UPDATE || ! Files.exists(file)) {
112 				Files.createDirectories(file.getParent());
113 				Files.writeString(file, actual, StandardCharsets.UTF_8);
114 				written.add(file.toAbsolutePath().toString());
115 			} else {
116 				assertEquals(Files.readString(file, StandardCharsets.UTF_8), actual, () -> "Golden mismatch: " + file);
117 			}
118 		}
119 		// A golden whose case was removed or renamed must be deleted.
120 		try (var files = Files.list(Path.of("src/test/resources/charts-goldens"))) {
121 			var onDisk = files.map(f -> f.getFileName().toString()).sorted().toList();
122 			assertEquals(NAMES.stream().map(n -> n + ".json").sorted().toList(), onDisk);
123 		}
124 		// The legacy flat form draws the same option as the helper form.
125 		assertEquals(goldens.get("line"), goldens.get("legacy-line"));
126 		assertEquals(goldens.get("scatter-symbol-size"), goldens.get("legacy-scatter-symbol-size"));
127 		if (! written.isEmpty())
128 			fail("Goldens written: " + written + " - review them, then re-run without -Djuneau.golden.update.");
129 	}

131 	@Test void a02_warningsAndErrors_perCase() {
132 		var r = report();
133 		// The throwing-fn-blank case calls a fn that throws: it logs E-JS-98 once and formats as "".
134 		assertEquals(List.of("[juneau-charts] E-JS-98: 'c' fn 'app.boom' failed: 'boom'"), r.get("errors"));
135 		var caseWarnings = (Map<?,?>)r.get("caseWarnings");
136 		for (var name : NAMES)
137 			assertEquals(WARNINGS.getOrDefault(name, List.of()), caseWarnings.get(name), name);
138 		assertEquals(THROWN, r.get("thrown"));
139 	}

142 		var checks = (Map<?,?>)report().get("checks");
143 		assertFalse(checks.isEmpty());
144 		var failed = checks.entrySet().stream().filter(e -> ! Boolean.TRUE.equals(e.getValue())).map(Map.Entry::getKey).toList();
145 		assertEquals(List.of(), failed);
