17  package org.apache.juneau.console.views;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.*;
24  import org.apache.juneau.commons.beanquery.*;

27  /** {@link Column}'s display properties: render, href, className, defaultVisible, toCatalogMap. */
28  class Column_DisplayProperties_Test extends TestBase {
29  
30  	@Test void a01_toCatalogMap_keyAndLabelFallback() {
31  		var m = Column.create("status").toCatalogMap();
32  		assertEquals("status", m.get("key"));
33  		assertEquals("status", m.get("label")); // label falls back to name when unset
34  		assertFalse(m.containsKey("render"));
35  		assertFalse(m.containsKey("href"));
36  		assertFalse(m.containsKey("className"));
37  		assertFalse(m.containsKey("defaultVisible"));
38  	}
39  
40  	@Test void a02_toCatalogMap_explicitLabelWins() {
41  		var m = Column.create("status").label("Status").toCatalogMap();
42  		assertEquals("Status", m.get("label"));
43  	}
44  
45  	@Test void a03_render_stringSugar() {
46  		var m = Column.create("status").render("tag:status").toCatalogMap();
47  		assertEquals("tag", ((Render)m.get("render")).id);
48  		assertEquals("status", ((Render)m.get("render")).meta.get("field"));
49  	}
50  
51  	@Test void a04_render_object() {
52  		var m = Column.create("age").render(Render.of("age-duration")).toCatalogMap();
53  		assertEquals("age-duration", ((Render)m.get("render")).id);
54  	}
55  
56  	@Test void a05_href_withoutRender_impliesLinked() {
57  		var m = Column.create("incidentKey").href("{incidentUrl}").toCatalogMap();
58  		assertEquals("linked", ((Render)m.get("render")).id);
59  		assertEquals("{incidentUrl}", m.get("href"));
60  	}
61  
62  	@Test void a05b_href_withoutRender_doesNotMutateColumn() {
63  		var c = Column.create("k").href("{u}");
64  		c.toCatalogMap();
65  		c.render("tag:status");
66  		assertEquals("tag", ((Render)c.toCatalogMap().get("render")).id);
67  	}
68  
69  	@Test void a06_href_withRender_emitted() {
70  		var m = Column.create("incidentKey").render("linked").href("{incidentUrl}").toCatalogMap();
71  		assertEquals("{incidentUrl}", m.get("href"));
72  	}
73  
74  	@Test void a07_className_emitted() {
75  		var m = Column.create("pod").className("ssc-id-cell").toCatalogMap();
76  		assertEquals("ssc-id-cell", m.get("className"));
77  	}
78  
79  	@Test void a08_defaultVisible_falseEmitted() {
80  		var m = Column.create("rootCause").defaultVisible(false).toCatalogMap();
81  		assertEquals(false, m.get("defaultVisible"));
82  	}
83  
84  	@Test void a09_defaultVisible_trueOmitted() {
85  		var m = Column.create("rootCause").defaultVisible(true).toCatalogMap();
86  		assertFalse(m.containsKey("defaultVisible")); // true is the implicit default; only false is worth a byte on the wire
87  	}
88  
89  	@Test void a10_toCatalogMap_search_inheritedOperators_emitsTypeOnly() {
90  		var m = Column.create("status").searchType(SearchType.ENUM).toCatalogMap();
91  		assertEquals("enum", m.get("searchType"));
92  		assertFalse(m.containsKey("searchOperators"));
93  		assertFalse(m.containsKey("customOperators"));
94  	}
95  
96  	@Test void a10b_toCatalogMap_search_ownOperators_emitsNames() {
97  		var std = SearchOperatorSet.standard();
98  		var m = Column.create("status").searchType(SearchType.ID)
99  			.operators(SearchOperatorSet.of(std.get("$eq"), std.get("$in"))).toCatalogMap();
100 		assertEquals(List.of("$eq", "$in"), m.get("searchOperators"));
101 		assertFalse(m.containsKey("customOperators"));
102 	}
103 
104 	@Test void a10c_toCatalogMap_search_customOperator_emitsNameAndHelp() {
105 		var mine = SearchOperator.create("$mine", "Assigned to me").types(SearchType.ID).build();
106 		var m = Column.create("owner").searchType(SearchType.ID)
107 			.operators(SearchOperatorSet.of(SearchOperatorSet.standard().get("$eq"), mine)).toCatalogMap();
108 		assertEquals(List.of("$eq", "$mine"), m.get("searchOperators"));
109 		assertEquals(List.of(Map.of("name", "$mine", "help", "Assigned to me")), m.get("customOperators"));
110 	}
111 
112 	@Test void a10m_toCatalogMapWithTableDefault_inheritedColumn_emitsTableSet() {
113 		var table = SearchOperatorSet.of(SearchOperatorSet.standard().get("$eq"));
114 		var m = Column.create("owner").searchType(SearchType.ID).toCatalogMap(table);
115 		assertEquals(List.of("$eq"), m.get("searchOperators"));
116 		assertFalse(m.containsKey("customOperators"));
117 	}
118 
119 	@Test void a10n_toCatalogMapWithTableDefault_ownSetWins() {
120 		var table = SearchOperatorSet.of(SearchOperatorSet.standard().get("$eq"));
121 		var own = SearchOperatorSet.of(SearchOperatorSet.standard().get("$ne"));
122 		var m = Column.create("owner").searchType(SearchType.ID).operators(own).toCatalogMap(table);
123 		assertEquals(List.of("$ne"), m.get("searchOperators"));
124 	}
125 
126 	@Test void a10o_toCatalogMapWithTableDefault_noSearchType_emitsNothing() {
127 		var m = Column.create("note").toCatalogMap(SearchOperatorSet.standard());
128 		assertFalse(m.containsKey("searchType"));
129 		assertFalse(m.containsKey("searchOperators"));
130 	}
131 
132 	@Test void a10d_toCatalogMap_noSearchType_noSearchKeys() {
133 		var m = Column.create("note").operators(SearchOperatorSet.standard()).toCatalogMap();
134 		assertFalse(m.containsKey("searchType"));
135 		assertFalse(m.containsKey("searchOperators"));
136 	}
137 
138 	@Test void a10e_toCatalogMap_resolvesThroughSearchMeta() {
139 		// The names/custom shape is what Column.searchMeta(name, type, allowList, customs) takes back.
140 		var mine = SearchOperator.create("$mine", "Assigned to me").types(SearchType.ID).build();
141 		var c = Column.create("owner").searchType(SearchType.ID).operators(SearchOperatorSet.of(SearchOperatorSet.standard().get("$eq"), mine));
142 		var m = c.toCatalogMap();
143 		@SuppressWarnings({
144 			"unchecked" // Casts a parsed JSON value to its known Map/List shape.
145 		})
146 		var round = Column.searchMeta("owner", (String)m.get("searchType"), (List<String>)m.get("searchOperators"), (List<Map<String,String>>)m.get("customOperators"));
147 		assertEquals(c.searchMeta(), round);
148 	}
149 
150 	@Test void a10f_render_null_clearsRender_hrefThenImpliesLinked() {
151 		var c = Column.create("k").render("tag:status").render((Render)null).href("{u}");
152 		assertEquals("linked", ((Render)c.toCatalogMap().get("render")).id);
153 	}

162 		assertFalse(m.containsKey("href"));
163 	}
