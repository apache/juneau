30  /**
31   * Asserts that the authoring key sheet ({@code juneau-symbols-key.svg}) still agrees with the sprite it documents
32   * ({@code juneau-symbols.svg}).
33   *
34   * <h5 class='section'>The drift this exists to catch:</h5>
35   * <p>
36   * The key sheet is not served - only the sprite is a registered servable resource - so nothing else in the build
37   * reads it, and nothing else would notice it going stale. It inlines a copy of every {@code <symbol>} definition
38   * so the sheet can be opened on its own, which means every artwork change has to be made in two places. Worse,
39   * each glyph's {@code viewBox} appears <b>twice</b> in the key sheet: once on the inlined {@code <symbol>} and
40   * again on the per-cell wrapper {@code <svg>} that {@code <use>}s it. A glyph normalised in the sprite but not on
41   * its wrapper renders at the wrong scale on the sheet only, so the artwork looks wrong in review while being
42   * perfectly correct in the shipped sprite - which is the most confusing version of this failure.
43   *
44   * <h5 class='section'>What this asserts, and what it deliberately does not:</h5>
45   * <p>
46   * It asserts the <i>property</i>: the two files carry the same {@code <symbol>} elements byte for byte and in the
47   * same order, and every per-cell wrapper declares the same {@code viewBox} as the symbol it references. It does
48   * <b>not</b> pin the sheet's generated layout. {@code scripts/gen-symbols-key.py --check} does that - it compares
49   * the whole file against what it would generate - and it is strictly stronger. The gap is real and is accepted
50   * rather than hidden: a hand-edit to the grid arithmetic or the sheet size passes this test and fails that script.
51   *
52   * <h5 class='section'>Why it reads the source tree rather than the classpath:</h5>
53   * <p>
54   * The key sheet is not a servable resource, so reading it through {@link ViewsMixin}'s resource loader would
55   * assert against a copy that no consumer ever fetches. Both files are read from the module's own
56   * {@code src/main/resources} instead, the way {@link RawContentSinkScanner#locateModuleRoot()} already does for
57   * this module's shipped assets.
58   */
59  @SuppressWarnings({
60  	"java:S8786" // Test regex is intentional; tightening would change match semantics.
61  })
62  class SymbolsKey_Staleness_Test extends TestBase {
63  
64  	private static final String RESOURCE_DIR = "src/main/resources/org/apache/juneau/views";
65  	private static final String SPRITE = "juneau-symbols.svg";
66  	private static final String KEY = "juneau-symbols-key.svg";
67  
68  	/** How to put it back, quoted in every failure message rather than left for the reader to find. */
69  	private static final String REGENERATE = "Regenerate the sheet with `python3 scripts/gen-symbols-key.py`.";
70  
71  	private static final Pattern SYMBOL = Pattern.compile("<symbol\\s.*?</symbol>", Pattern.DOTALL);
72  	private static final Pattern SYMBOL_ID = Pattern.compile("<symbol\\s+id=\"([^\"]+)\"");
73  	private static final Pattern SYMBOL_VIEWBOX = Pattern.compile("<symbol\\s+[^>]*viewBox=\"([^\"]+)\"");
74  
75  	/** One per-cell wrapper of the key sheet's grid: its own declared viewBox, and the symbol id it references. */
76  	private static final Pattern CELL_WRAPPER =
77  		Pattern.compile("<svg\\s+x=\"[^\"]*\"\\s+y=\"[^\"]*\"[^>]*viewBox=\"([^\"]+)\"><use\\s+href=\"#([^\"]+)\"/>");
78  
79  	private static String read(String fileName) throws IOException {
80  		var root = RawContentSinkScanner.locateModuleRoot();
81  		assertNotNull(root, "could not locate the juneau-console-views module root");
82  		var f = root.resolve(RESOURCE_DIR).resolve(fileName);
83  		assertTrue(Files.isRegularFile(f), () -> "missing resource: " + f);
84  		return Files.readString(f, UTF_8);
85  	}
86  
87  	/** Every {@code <symbol>} element of an SVG document, verbatim and in document order. */
88  	private static List<String> symbols(String svg) {
89  		var out = new ArrayList<String>();
90  		var m = SYMBOL.matcher(svg);
91  		while (m.find())
92  			out.add(m.group());
93  		return out;
94  	}
95  
96  	/** Index of the first symbol that differs between the two documents, or {@code -1} if they agree. */
97  	private static int firstDrift(List<String> sprite, List<String> key) {
98  		for (var i = 0; i < Math.min(sprite.size(), key.size()); i++)
99  			if (!sprite.get(i).equals(key.get(i)))
100 				return i;
101 		return sprite.size() == key.size() ? -1 : Math.min(sprite.size(), key.size());
102 	}
103 
104 	private static String group(Pattern p, String s) {
105 		var m = p.matcher(s);
106 		assertTrue(m.find(), () -> "no match for " + p.pattern() + " in: " + s);
107 		return m.group(1);
108 	}
109 
110 	/**
111 	 * Appends a redundant {@code Z} to the first path in a document - a byte change that is valid in any path.
112 	 *
113 	 * <p>
114 	 * The needle carries its leading space deliberately: {@code id="} contains {@code d="}, so searching for the
115 	 * bare attribute name finds every glyph's <i>id</i> first. That still produces a detectable byte change, so
116 	 * b01 would pass either way - and would be asserting that a renamed stem is detected rather than that changed
117 	 * artwork is, which is the other test's job.
118 	 */
119 	private static String mutateFirstPath(String svg) {
120 		var at = svg.indexOf(" d=\"");
121 		assertTrue(at >= 0, "no path data to mutate; this file's anti-vacuous checks cannot run");
122 		var close = svg.indexOf('"', at + 4);
123 		assertTrue(close > at, "unterminated path data");
124 		return svg.substring(0, close) + " Z" + svg.substring(close);
125 	}

127 	//------------------------------------------------------------------------------------------------------------------
128 	// a: the two files agree
129 	//------------------------------------------------------------------------------------------------------------------
130 
131 	@Test void a01_keySheetInlinesTheSpriteSymbolsByteForByte() throws Exception {
132 		var sprite = symbols(read(SPRITE));
133 		var key = symbols(read(KEY));
134 		assertFalse(sprite.isEmpty(), "no <symbol> elements found in the sprite - this test would be vacuous");
135 		assertEquals(sprite.size(), key.size(),
136 			() -> "the key sheet inlines " + key.size() + " symbols but the sprite ships " + sprite.size()
137 				+ ". " + REGENERATE);
138 		var at = firstDrift(sprite, key);
139 		if (at < 0)
140 			return;
141 		fail("the key sheet's copy of a glyph has drifted from the sprite at position " + at + " (sprite id="
142 			+ group(SYMBOL_ID, sprite.get(at)) + ", key id=" + group(SYMBOL_ID, key.get(at))
143 			+ "). The sprite is the source of truth. " + REGENERATE);
144 	}
145 
146 	@Test void a02_everyCellWrapperDeclaresItsSymbolsViewBox() throws Exception {
147 		var key = read(KEY);
148 		var declared = new LinkedHashMap<String,String>();
149 		for (var s : symbols(key))
150 			declared.put(group(SYMBOL_ID, s), group(SYMBOL_VIEWBOX, s));
151 
152 		var seen = new ArrayList<String>();
153 		var m = CELL_WRAPPER.matcher(key);
154 		while (m.find()) {
155 			var wrapperViewBox = m.group(1);
156 			var id = m.group(2);
157 			seen.add(id);
158 			assertEquals(declared.get(id), wrapperViewBox,
159 				() -> "the key sheet's grid cell for " + id + " wraps it in a viewBox its <symbol> does not"
160 					+ " declare, so it renders at the wrong scale on the sheet only. " + REGENERATE);
161 		}
162 		// Without this, a sheet whose grid stopped being emitted at all would satisfy the loop above trivially.
163 		assertEquals(new ArrayList<>(declared.keySet()), seen,
164 			"every inlined symbol must appear exactly once in the sheet's grid, in order. " + REGENERATE);
165 	}

169 	//
170 	// A staleness test that silently stopped seeing artwork reads as a passing test, which is worse than no test
171 	// at all - the same discipline ShippedJs_SinkScan_Test applies to its scanner.

174 	@Test void b01_aOneCharacterArtworkChangeIsDetected() throws Exception {
175 		var real = read(SPRITE);
176 		var drifted = symbols(mutateFirstPath(real));
177 		assertEquals(symbols(real).size(), drifted.size(), "the mutation must not change the symbol count");
178 		assertEquals(0, firstDrift(symbols(real), drifted),
179 			"a single added path command went undetected, so a01 is comparing something other than artwork"
180 				+ " - a whitespace-normalising comparison, or a regex capturing only the opening tag, both look"
181 				+ " like this");
182 	}
183 
184 	@Test void b02_aMissingSymbolIsDetected() throws Exception {
185 		var sprite = symbols(read(SPRITE));
186 		var short_ = sprite.subList(0, sprite.size() - 1);
187 		assertTrue(firstDrift(sprite, short_) >= 0, "a dropped glyph went undetected");
188 	}
189 
190 	@Test void b03_theWrapperPatternMatchesEveryCellOfTheRealSheet() throws Exception {
191 		// a02's assertion is inside a while-loop, so a pattern that stopped matching would make it vacuous.  The
192 		// count is asserted against the symbol count rather than a literal so it survives a glyph being added.
193 		var key = read(KEY);
194 		var count = 0;
195 		var m = CELL_WRAPPER.matcher(key);
196 		while (m.find())
197 			count++;
198 		assertEquals(symbols(key).size(), count,
199 			"the per-cell wrapper pattern no longer matches every cell of the real sheet");
200 	}
