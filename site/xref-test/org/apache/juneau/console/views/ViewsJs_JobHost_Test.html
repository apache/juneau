19  import static org.junit.jupiter.api.Assertions.*;
20  import static org.junit.jupiter.api.Assumptions.*;
21

26  /** Node behavioral checks for async jobs started through {@code JuneauViews.actions.run} (no row, banner-hosted) and for {@code cancelJob}'s contract. */
27  class ViewsJs_JobHost_Test {
28  
29  	private static Map<?,?> r() {
30  		var r = NodeHarness.report("job-host.cjs", ViewsMixin.RENDERS_JS_RESOURCE, ViewsMixin.VIEWS_JS_RESOURCE);
31  		assumeTrue(r != null, "node (or job-host.cjs) not available - behavioral layer skipped");
32  		return r;
33  	}
34  
35  	private static String s(String k) { return (String)r().get(k); }
36  	private static void yes(String k) { assertEquals(true, r().get(k), k); }
37  
38  	@Test void r01_runStaysPendingWhileTheJobStreamsThenResolvesWithItsResult() {
39  		yes("r01_pending"); yes("r01_sourceOpened"); yes("r01_progressShown"); yes("r01_cancelButton");
40  		assertEquals("Cancel Run job", s("r01_cancelLabel"));
41  		assertEquals("40%", s("r01_progressText"));
42  		assertEquals("success", s("r01_outcome"));
43  		assertEquals("all done", s("r01_message"));
44  		yes("r01_progressGone"); yes("r01_closed"); yes("r01_painted");
45  	}
46  
47  	@Test void r02_streamErrorResolvesUnknown() {
48  		assertEquals("unknown", s("r02_outcome"));
49  		assertEquals("the progress stream was interrupted", s("r02_message"));
50  	}
51  
52  	@Test void r03_noEventSourceResolvesUnknownWithTheBrowserNotice() {
53  		assertEquals("unknown", s("r03_outcome"));
54  		assertEquals("live progress is unavailable in this browser", s("r03_message"));
55  	}
56  
57  	@Test void r04_offSiteStreamUrlOpensNothing() {
58  		assertEquals(0, ((Number)r().get("r04_opened")).intValue());
59  		assertEquals("unknown", s("r04_outcome"));
60  		assertEquals("the job started, but its progress URL is not on this site, so progress is not shown", s("r04_message"));
61  	}
62  
63  	@Test void r05_throwingEventSourceResolvesUnknown() {
64  		assertEquals("unknown", s("r05_outcome"));
65  		assertEquals("the progress stream could not be opened", s("r05_message"));
66  	}
67  
68  	@Test void r06_offSiteCancelUrlShowsNoCancelButton() { yes("r06_progress"); yes("r06_noCancel"); }
69  
70  	@Test void r07_cancelPostsOnceWithCsrfAndDisablesTheButton() {
71  		assertEquals(1, ((Number)r().get("r07_cancelPosts")).intValue());
72  		yes("r07_csrf"); yes("r07_disabled");
73  		assertEquals("cancelled", s("r07_outcome"));
74  	}
75  
76  	@Test void r08_failedCancelReEnablesAndIsNotTheRunOutcome() {
77  		yes("r08_reEnabled");
78  		assertEquals("success", s("r08_outcome"));
79  	}
80  
81  	@Test void r09_releaseClosesTheStreamAndSettlesUnknown() {
82  		yes("r09_closed"); yes("r09_bannerGone"); yes("r09_noBannerHostRebuilt"); yes("r09_noLateText");
83  		assertEquals("unknown", s("r09_outcome"));
84  	}
85  
86  	@Test void r10_secondRunOpensAFreshStream() {
87  		yes("r10_secondStream");
88  		assertEquals("failure", s("r10_outcome"));
89  	}
90  
91  	@Test void r04b_aStreamUrlWithTrimmableCharactersOpensNothing() {
92  		assertEquals(0, ((Number)r().get("r04b_opened")).intValue());
93  		assertEquals("unknown", s("r04b_outcome"));
94  	}
95  	@Test void r11_eachJobKeepsItsOwnProgressElement() {
96  		assertEquals(2, ((Number)r().get("r11_count")).intValue());
97  		assertEquals("one-again,two", s("r11_texts"));
98  	}
99  	@Test void r12_unsafeJobUrlShapesAreNeverStreamedOrCancelled() {
100 		for (var k : new String[]{"backslash", "lead", "c0", "newline", "proto", "scheme"}) {
101 			var st = (java.util.Map<?,?>)r().get("r12_stream_" + k);
102 			assertEquals(0, ((Number)st.get("opened")).intValue(), k);
103 			assertEquals("unknown", st.get("outcome"), k);
104 			var ca = (java.util.Map<?,?>)r().get("r12_cancel_" + k);
105 			assertEquals(true, ca.get("noButton"), k);
106 			yes("r12_cancelJob_" + k);
107 			yes("r12_noCancelPost_" + k);
108 		}
109 		assertEquals(false, r().get("r12_missingCancel"));
110 		assertTrue(s("r12_untrustedCancelText").contains("not on this site"), s("r12_untrustedCancelText"));
111 	}

131 	}
