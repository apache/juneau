31  /**
32   * Static backstop for the Node floor (see {@code juneau.node.floor} in the root pom): fails when a shipped script
33   * ({@code src/main/resources/**}/*.js) or a Node harness ({@code src/test/js/**}/*.cjs) in any module calls a builtin
34   * that is newer than the floor.
35   *
36   * <p>
37   * The harness runs themselves already strip these builtins through {@code src/test/js/node-floor-preload.cjs}; this
38   * scan additionally covers shipped scripts that no Node harness exercises, and pins the same list in one readable
39   * place.  Keep it in step with the preload.
40   *
41   * <p>
42   * A call that is genuinely safe (for example a method of the same name on an unrelated object) is exempted by putting
43   * {@code node-floor-ok} and a reason in a comment on the same line.
44   */
45  class NodeFloorScan_Test {
46  
47  	private static final String ALLOW_MARKER = "node-floor-ok";
48  
49  	/**
50  	 * Method calls, matched as {@code .name(} so a local variable, property or keyword of the same name is not a hit.
51  	 */
52  	private static final Pattern METHOD_CALL = Pattern.compile(
53  		"\\.(toSorted|toReversed|toSpliced|with|isWellFormed|toWellFormed|transfer|transferToFixedLength"
54  		+ "|union|intersection|difference|symmetricDifference|isSubsetOf|isSupersetOf|isDisjointFrom)\\s*\\(");
55  
56  	/** Static builtins, matched as {@code Owner.name} followed by a call or a reference. */
57  	private static final Pattern STATIC_BUILTIN = Pattern.compile(
58  		"\\b(Object\\.groupBy|Map\\.groupBy|Promise\\.withResolvers|Promise\\.try|Array\\.fromAsync|RegExp\\.escape|Error\\.isError)\\b");
59  
60  	/** A line comment: {@code //} at the start of the line or after whitespace (so {@code http://} is left alone). */
61  	private static final Pattern LINE_COMMENT = Pattern.compile("(^|\\s)//.*$");
62  
63  	private static final Set<String> SKIPPED_DIRS = Set.of("target", "node_modules", ".git", "vendor");
64  
65  	/** Logs the build agent's Node version once, so the real floor can be read off the CI log; never fails the test. */
66  	private static void logNodeVersion() {
67  		try {
68  			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
69  			String out;
70  			try (var in = p.getInputStream()) {
71  				out = new String(in.readAllBytes(), UTF_8).trim();
72  			}
73  			p.waitFor(10, java.util.concurrent.TimeUnit.SECONDS);
74  			System.out.println("[NodeFloorScan] node --version: " + out);
75  		} catch (Exception e) {
76  			System.out.println("[NodeFloorScan] node --version unavailable: " + e);
77  		}
78  	}
79  
80  	@Test void a01_noBuiltinsNewerThanTheNodeFloor() throws IOException {
81  		logNodeVersion();
82  		var root = repoRoot();
83  		assumeTrue(root != null, "repository root not found");
84  		var hits = new ArrayList<String>();
85  		var scanned = new int[1];
86  		Files.walkFileTree(root, new SimpleFileVisitor<Path>() {
87  			@Override
88  			public FileVisitResult preVisitDirectory(Path dir, BasicFileAttributes attrs) {
89  				// Never enter target/ and friends: sibling forks create and delete temp files there while this scan runs.
90  				var name = dir.getFileName();
91  				return !dir.equals(root) && name != null && SKIPPED_DIRS.contains(name.toString()) ? FileVisitResult.SKIP_SUBTREE : FileVisitResult.CONTINUE;
92  			}
93  
94  			@Override
95  			public FileVisitResult visitFile(Path f, BasicFileAttributes attrs) {
96  				if (isScanned(f)) {
97  					scanned[0]++;
98  					scan(root, f, hits);
99  				}
100 				return FileVisitResult.CONTINUE;
101 			}
102 
103 			@Override
104 			public FileVisitResult visitFileFailed(Path f, IOException e) throws IOException {
105 				if (e instanceof NoSuchFileException)
106 					return FileVisitResult.CONTINUE;
107 				throw e;
108 			}
109 		});
110 		assertTrue(scanned[0] > 0, "no scripts found under " + root);
111 		assertTrue(hits.isEmpty(), () -> "Builtins newer than the Node floor (the CI Node lacks them; use sort()/slice().sort() style equivalents, "
112 			+ "or add a 'node-floor-ok: <reason>' comment on the line):\n  " + String.join("\n  ", hits));
113 	}
114 
115 	@Test void a02_patternsMatchOnlyCalls() {
116 		assertTrue(METHOD_CALL.matcher("urls.toSorted()").find());
117 		assertTrue(METHOD_CALL.matcher("xs.with (1, 2)").find());
118 		assertTrue(METHOD_CALL.matcher("a.union(b)").find());
119 		assertFalse(METHOD_CALL.matcher("with (obj) { x }").find());
120 		assertFalse(METHOD_CALL.matcher("const with = 1; o.with = 2;").find());
121 		assertFalse(METHOD_CALL.matcher("o.difference").find());
122 		assertTrue(STATIC_BUILTIN.matcher("Object.groupBy(rows, f)").find());
123 		assertFalse(STATIC_BUILTIN.matcher("groupBy: ['a']").find());
124 		assertFalse(STATIC_BUILTIN.matcher("function groupByTransform(frame)").find());
125 	}
126 
127 	private static boolean isScanned(Path f) {
128 		if (!Files.isRegularFile(f))
129 			return false;
130 		var s = f.toString().replace('\\', '/');
131 		for (var d : SKIPPED_DIRS)
132 			if (s.contains("/" + d + "/"))
133 				return false;
134 		if (s.endsWith(".min.js") || s.endsWith("/node-floor-preload.cjs"))  // the preload names the builtins it removes
135 			return false;
136 		return (s.endsWith(".js") && s.contains("/src/main/resources/")) || (s.endsWith(".cjs") && s.contains("/src/test/js/"));
137 	}
138 
139 	private static void scan(Path root, Path f, List<String> hits) {
140 		try {
141 			var lines = Files.readAllLines(f, UTF_8);
142 			for (var i = 0; i < lines.size(); i++) {
143 				var line = lines.get(i);
144 				if (line.contains(ALLOW_MARKER))
145 					continue;
146 				var code = LINE_COMMENT.matcher(line).replaceFirst("");
147 				var m = METHOD_CALL.matcher(code);
148 				var s = STATIC_BUILTIN.matcher(code);
149 				if (m.find() || s.find())
150 					hits.add(root.relativize(f) + ":" + (i + 1) + ": " + line.trim());
151 			}
152 		} catch (IOException e) {
153 			throw new UncheckedIOException(e);
154 		}
155 	}
156 
157 	/** The directory holding the root pom: the nearest ancestor of the module directory that has a {@code juneau-core} sibling module. */
158 	private static Path repoRoot() {
159 		var basedir = System.getProperty("basedir");
160 		var p = Path.of(basedir == null ? "." : basedir).toAbsolutePath().normalize();
161 		for (; p != null; p = p.getParent())
162 			if (Files.isDirectory(p.resolve("juneau-core")) && Files.isDirectory(p.resolve("juneau-rest")))
163 				return p;
164 		return null;
165 	}
