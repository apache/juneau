32  /**
33   * The sectioned-dialog half of the module's <b>JavaScript-execution harness</b>: runs the REAL served
34   * {@code juneau-views.js} <b>and</b> {@code juneau-views.css} in a real headless browser and asserts a
35   * {@code FormDef.sections} dialog behaves as a user experiences it &mdash; exactly one section renders, a value typed
36   * into one survives a round trip through another, an inline error stays inside the section that owns it, confirm
37   * reveals the offending section before focusing it, Tab runs ribbon-then-body without leaking into a hidden section,
38   * and Escape closes the whole <b>dialog</b> through the shared layer stack while restoring focus to the invoker.
39   *
40   * <h5 class='section'>Why this exists (beyond the always-on source-shape + Node harness tests):</h5>
41   * <p>
42   * {@link ViewsJs_RibbonStrip_Test} proves the shipped script builds the ribbon and hides the right panes under a DOM
43   * shim, where {@code hidden} is just a property.  In a real browser it is a cascade question, and the pane rule is
44   * {@code display: flex} &mdash; an <b>author</b> rule that outranks the UA stylesheet's
45   * {@code [hidden] { display: none }}.  Only a real browser with the real stylesheet can prove an unselected section is
46   * genuinely not rendered rather than merely flagged, and only a real browser proves the focus trap, real Tab order and
47   * real Escape.
48   *
49   * <h5 class='section'>Off by default:</h5>
50   * <p>
51   * Disabled unless the {@value #GATE} system property is set, which only the module's opt-in {@code js-tests} Maven
52   * profile does.  It reuses that profile's provisioned Node + Playwright browser and derives its own prober
53   * ({@code sectioned-dialog-browser.cjs}) from the profile's {@code juneau.jsTests.harness} directory, so no pom change
54   * is needed.
55   */
56  @EnabledIfSystemProperty(named=SectionedDialog_BrowserTest.GATE, matches="true",
57  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -f juneau-rest/juneau-console-views/pom.xml test`")
58  class SectionedDialog_BrowserTest extends TestBase {

65  	private static String resource(String path) throws IOException {
66  		try (var in = ViewsMixin.class.getResourceAsStream(path)) {
67  			assertNotNull(in, () -> "missing classpath resource: " + path);
68  			return new String(in.readAllBytes(), UTF_8);
69  		}
70  	}

72  	@BeforeAll
73  	static void probe() throws Exception {
74  		var dir = Path.of(requiredProperty("juneau.jsTests.dir"));
75  		var harness = Path.of(requiredProperty("juneau.jsTests.harness")).getParent()
76  			.resolve("sectioned-dialog-browser.cjs");
77  
78  		// The CSS is load-bearing here, not decoration: the pane's author-level `display: flex` competes with the UA
79  		// `[hidden]` rule, so a fixture without the real stylesheet would prove nothing about the shipped contract.
80  		var fixture = "<!DOCTYPE html><html><head><meta charset=\"utf-8\"><style>\n"
81  			+ resource(ViewsMixin.VIEWS_CSS_RESOURCE)
82  			+ "\n</style></head><body>\n<script>\n"
83  			+ resource(ViewsMixin.RENDERS_JS_RESOURCE)
84  			+ "\n</script>\n<script>\n"
85  			+ resource(ViewsMixin.VIEWS_JS_RESOURCE)
86  			+ "\n</script></body></html>";
87  		var fixtureFile = Files.createDirectories(dir.resolve("fixtures")).resolve("sectioned-dialog.html");
88  		Files.write(fixtureFile, fixture.getBytes(UTF_8));
89  
90  		report = Json.to(run(dir, harness, fixtureFile), Map.class);
91  	}

93  	private static String requiredProperty(String name) {
94  		var v = System.getProperty(name);
95  		assertNotNull(v, () -> "-D" + name + " not set; the js-tests profile is responsible for providing it");
96  		return v;
97  	}

99  	private static String run(Path dir, Path harness, Path fixture) throws Exception {
100 		var cmd = List.of(System.getProperty("juneau.jsTests.node", "node"), harness.toString(), fixture.toString());
101 		var stdout = dir.resolve("sectioned-dialog-stdout.json");
102 		var stderr = dir.resolve("sectioned-dialog-stderr.txt");
103 		var pb = new ProcessBuilder(cmd).redirectOutput(stdout.toFile()).redirectError(stderr.toFile());
104 		pb.environment().put("NODE_PATH", dir.resolve("node_modules").toString());
105 		pb.environment().put("PLAYWRIGHT_BROWSERS_PATH", requiredProperty("juneau.jsTests.browsers"));
106 
107 		var p = pb.start();
108 		if (!p.waitFor(3, TimeUnit.MINUTES)) {
109 			p.destroyForcibly();
110 			fail("prober did not finish within 3m; stderr:\n" + quietRead(stderr));
111 		}
112 		assertEquals(0, p.exitValue(), () -> "prober exited non-zero; stderr:\n" + quietRead(stderr));
113 		return Files.readString(stdout);
114 	}
115 
116 	private static String quietRead(Path p) {
117 		try {
118 			return Files.readString(p);
119 		} catch (IOException e) {
120 			return "<unreadable: " + e + ">";
121 		}
122 	}
