28  /**
29   * Guards the folder-tab look, the switcher slider/in-switcher rules, the chart data-tab rules and the popover rules
30   * written into {@code chrome.css} and {@code juneau-views.css}.
31   */
32  class FolderTabs_Css_Test extends TestBase {
33  
34  	private static final String TAB_GROUP = ".juneau-view-ribbon-group[data-juneau-strip-mode=\"tab\"]";
35  	private static final String SEL = TAB_GROUP + " .juneau-view-ribbon-btn[aria-selected=\"true\"]";
36  
37  	private static String css(String path) throws IOException {
38  		try (var in = FolderTabs_Css_Test.class.getResourceAsStream(path)) {
39  			assertNotNull(in, path + " not found on the test classpath");
40  			return new String(in.readAllBytes(), StandardCharsets.UTF_8);
41  		}
42  	}
43  
44  	private static String viewsCss() throws IOException {
45  		return css("/org/apache/juneau/views/juneau-views.css");
46  	}
47  
48  	private static String chromeCss() throws IOException {
49  		return css("/org/apache/juneau/console/chrome.css");
50  	}
51  
52  	/** Returns the body of the first rule whose selector list is exactly {@code selector} (at the start of a line). */
53  	private static String ruleBody(String css, String selector) {
54  		var m = Pattern.compile("(?m)^" + Pattern.quote(selector) + "\\s*\\{([^}]*)\\}").matcher(css);
55  		return m.find() ? m.group(1) : "";
56  	}
57  
58  	@Test void a01_selectedTab_hasAccentTopBar() throws IOException {
59  		var body = ruleBody(chromeCss(), SEL);
60  		assertTrue(body.contains("box-shadow: inset 0 3px 0 0 var(--jc-accent)"), body);
61  	}
62  
63  	@Test void a02_selectedTab_whiteFill_noTransparentBottom() throws IOException {
64  		var body = ruleBody(chromeCss(), SEL);
65  		assertTrue(body.contains("background-color: var(--jc-card-bg)"), body);
66  		assertFalse(body.contains("border-bottom-color: transparent"), body);
67  		assertFalse(ruleBody(viewsCss(), SEL).contains("border-bottom-color: transparent"));
68  	}
69  
70  	@Test void a03_selectedTab_overlapsPanelByOnePixel() throws IOException {
71  		assertTrue(ruleBody(viewsCss(), SEL).contains("margin-bottom: -1px"));
72  	}
73  
74  	@Test void a04_unselectedTab_greyFill() throws IOException {
75  		var body = ruleBody(chromeCss(), TAB_GROUP + " .juneau-view-ribbon-btn");
76  		assertTrue(body.contains("background-color: var(--jc-pill-neutral-bg)"), body);
77  	}
78  
79  	@Test void a05_panel_hasGreyBorderNotAccent() throws IOException {
80  		var body = ruleBody(chromeCss(), ".juneau-view-helper-tabstrip-wrapper[data-juneau-strip-panel] > .juneau-view-helper-tabstrip-panes");
81  		assertTrue(body.contains("border-color: var(--jc-border)"), body);
82  		assertFalse(body.contains("var(--jc-accent)"), body);
83  	}
84  
85  	@Test void a06_compactTokensDeclared() throws IOException {
86  		var css = viewsCss();
87  		assertTrue(css.contains("--jc-tab-compact-font:") && css.contains("--jc-tab-compact-pad-y:") && css.contains("--jc-tab-compact-pad-x:"));
88  	}
89  
90  	@Test void a07_detailTabsUseSharedRules() throws IOException {
91  		var chrome = chromeCss();
92  		assertFalse(chrome.contains(".juneau-view-detail-tabs {") || chrome.contains(".juneau-view-detail-tabs["), "detail tabs must not get their own colour rules");
93  		assertFalse(chrome.contains(".juneau-view-dialog-sections {") || chrome.contains(".juneau-view-dialog-sections["), "dialog sections must not get their own colour rules");
94  	}

96  	private static final String SLIDER = ".juneau-switcher-selector[data-juneau-switcher-style=\"slider\"]";
97  
98  	@Test void b01_sliderTrackAndThumbRules() throws IOException {
99  		var chrome = chromeCss();
100 		var track = ruleBody(chrome, SLIDER + " > .juneau-view-ribbon-group");
101 		assertTrue(track.contains("background-color: var(--jc-pill-neutral-bg)") && track.contains("border-radius"), track);
102 		var thumb = ruleBody(chrome, ".juneau-switcher-thumb");
103 		assertTrue(thumb.contains("background-color: var(--jc-card-bg)") && thumb.contains("transition: transform"), thumb);
104 	}

106 	@Test void b02_reducedMotion() throws IOException {
107 		var m = Pattern.compile("@media \\(prefers-reduced-motion: reduce\\)\\s*\\{[^@]*?\\.juneau-switcher-thumb\\s*\\{\\s*transition: none;").matcher(chromeCss());
108 		assertTrue(m.find());
109 	}

114 	}

120 	@Test void c02_srOnlyPattern() throws IOException {
121 		var body = ruleBody(chromeCss(), ".jc-chart-sr");
122 		for (var d : new String[]{"position: absolute", "width: 1px", "height: 1px", "overflow: hidden", "clip-path: inset(50%)", "white-space: nowrap"})

124 		assertFalse(body.contains("display: none"), body);
125 	}
