17  package org.apache.juneau.console.views.markdown;
18  
19  import static org.apache.juneau.BasicTestUtils.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import org.apache.juneau.*;
23  import org.junit.jupiter.api.*;
24  
25  /**
26   * Behavioural tests for {@link CommonmarkMarkdownRenderer} - the shipped commonmark-java default with the GFM
27   * tables extension enabled.
28   */
29  class CommonmarkMarkdownRenderer_Test extends TestBase {
30  
31  	private static final CommonmarkMarkdownRenderer RENDERER = new CommonmarkMarkdownRenderer();
32  
33  	//-----------------------------------------------------------------------------------------------------------------
34  	// a) Core CommonMark rendering
35  	//-----------------------------------------------------------------------------------------------------------------
36  
37  	@Test void a01_paragraph() {
38  		assertEquals("<p>Hello world.</p>\n", RENDERER.toHtml("Hello world."));
39  	}
40  
41  	@Test void a02_heading() {
42  		assertEquals("<h1>Runbook</h1>\n", RENDERER.toHtml("# Runbook"));
43  	}
44  
45  	@Test void a03_unorderedList() {
46  		assertEquals("<ul>\n<li>one</li>\n<li>two</li>\n</ul>\n", RENDERER.toHtml("- one\n- two"));
47  	}
48  
49  	@Test void a04_fencedCodeBlock() {
50  		var html = RENDERER.toHtml("```\ncode();\n```");
51  		assertTrue(html.contains("<pre><code>"), () -> "expected a code block, got:\n" + html);
52  		assertTrue(html.contains("code();"), () -> "expected code content, got:\n" + html);
53  	}
54  
55  	@Test void a05_inlineEmphasisAndLink() {
56  		assertEquals("<p><em>hi</em> <a href=\"https://x\">y</a></p>\n", RENDERER.toHtml("*hi* [y](https://x)"));
57  	}
58  
59  	//-----------------------------------------------------------------------------------------------------------------
60  	// b) GFM tables extension - the reason this ships enabled by default
61  	//-----------------------------------------------------------------------------------------------------------------
62  
63  	@Test void b01_pipeTable_rendersAsTable_notLiteralPipes() {
64  		var md = "| A | B |\n| --- | --- |\n| 1 | 2 |";
65  		var html = RENDERER.toHtml(md);
66  		assertTrue(html.contains("<table>"), () -> "GFM tables extension not enabled - got:\n" + html);
67  		assertTrue(html.contains("<th>A</th>"), () -> "expected header cell, got:\n" + html);
68  		assertTrue(html.contains("<td>1</td>"), () -> "expected body cell, got:\n" + html);
69  		assertFalse(html.contains("| A | B |"), () -> "table rendered as literal pipes, got:\n" + html);
70  	}
71  
72  	//-----------------------------------------------------------------------------------------------------------------
73  	// c) Edge cases
74  	//-----------------------------------------------------------------------------------------------------------------
75  
76  	@Test void c01_emptyString_rendersEmpty() {
77  		assertEquals("", RENDERER.toHtml(""));
78  	}
79  
80  	@Test void c02_null_throws() {
81  		assertThrowsWithMessage(IllegalArgumentException.class, "Markdown source must not be null.", () -> RENDERER.toHtml(null));
82  	}
83  
84  	@Test void c03_rendererIsReusableAcrossCalls() {
85  		assertEquals("<p>one</p>\n", RENDERER.toHtml("one"));
86  		assertEquals("<p>two</p>\n", RENDERER.toHtml("two"));
87  	}
88  
89  	//-----------------------------------------------------------------------------------------------------------------
90  	// d) XSS: raw HTML escaped, javascript:/data: URLs stripped
91  	//-----------------------------------------------------------------------------------------------------------------
92  
93  	@Test void d01_rawScript_isEscapedNotAnElement() {
94  		var html = RENDERER.toHtml("hello <script>alert(1)</script>");
95  		assertFalse(html.contains("<script"), () -> html);
96  		assertTrue(html.contains("&lt;script"), () -> html);
97  	}
98  
99  	@Test void d02_imgOnerror_isEscaped() {
100 		var html = RENDERER.toHtml("<img src=x onerror=alert(1)>");
101 		assertFalse(html.contains("<img"), () -> html);
102 		assertTrue(html.contains("&lt;img"), () -> html);
103 	}
104 
105 	@Test void d03_javascriptHref_isStripped() {
106 		var html = RENDERER.toHtml("[x](javascript:alert(1))");
107 		assertFalse(html.contains("javascript:"), () -> html);
108 		assertTrue(html.contains("<a>") || html.contains("<a >") || html.contains("<a>x</a>"), () -> html);
109 	}
110 
111 	@Test void d04_httpsHref_isKept() {
112 		var html = RENDERER.toHtml("[y](https://example.com)");
113 		assertTrue(html.contains("href=\"https://example.com\""), () -> html);
114 	}
115 
116 	@Test void d05_dataUrl_isRejected() {
117 		assertFalse(CommonmarkMarkdownRenderer.isSafeUrl("data:text/html,<script>"));
118 		assertFalse(CommonmarkMarkdownRenderer.isSafeUrl("javascript:alert(1)"));
119 		assertFalse(CommonmarkMarkdownRenderer.isSafeUrl("vbscript:msg"));
120 		assertTrue(CommonmarkMarkdownRenderer.isSafeUrl("https://x"));
121 		assertTrue(CommonmarkMarkdownRenderer.isSafeUrl("http://x"));
122 		assertTrue(CommonmarkMarkdownRenderer.isSafeUrl("mailto:a@b"));
123 		assertTrue(CommonmarkMarkdownRenderer.isSafeUrl("#frag"));
124 		assertTrue(CommonmarkMarkdownRenderer.isSafeUrl("/rest/skills"));
125 		assertTrue(CommonmarkMarkdownRenderer.isSafeUrl("relative/path"));
126 		assertFalse(CommonmarkMarkdownRenderer.isSafeUrl(null));
127 		assertFalse(CommonmarkMarkdownRenderer.isSafeUrl(""));
128 	}
129 }
