17  package org.apache.juneau.console.views;

19  import static org.apache.juneau.console.views.ConsoleBrowserFixture.*;
20  import static org.junit.jupiter.api.Assertions.*;

22  import java.util.*;
23  
24  import org.junit.jupiter.api.*;

27  /**
28   * The column-search popover frame in Chromium: a thin bordered, rounded, shadowed card; a filled Apply with plain
29   * Cancel and Clear at the input's height; and the help mark at the bottom right of the footer.
30   */
31  @EnabledIfSystemProperty(named=ConsoleBrowserFixture.GATE, matches="true",
32  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -pl juneau-rest/juneau-console-views -am test` from the repo root")
33  class ColumnSearchPopoverStyle_BrowserTest extends org.apache.juneau.TestBase {
34  
35  	private static final List<String> SCHEMES = List.of("light", "dark");
36  
37  	private static final String BODY = """
38  		<div class="juneau-view-col-search-popover" id="pop" style="position:absolute;left:40px;top:40px">
39  		<div class="juneau-view-col-search-popover-title" id="ttl">Status</div>
40  		<input type="text" class="juneau-view-col-search-popover-input" id="inp" value="x">
41  		<div class="juneau-view-col-search-popover-footer" id="ftr">
42  		<button type="button" class="juneau-view-col-search-popover-apply jc-btn jc-btn-primary" id="apply">Apply</button>
43  		<button type="button" class="juneau-view-col-search-popover-cancel jc-btn" id="cancel">Cancel</button>
44  		<button type="button" class="juneau-view-col-search-popover-clear jc-btn" id="clear">Clear</button>
45  		<button type="button" class="juneau-view-col-search-popover-more" id="more">?</button>
46  		</div></div>
47  		""";
48  
49  	private static final String PROBE = """
50  		<script>
51  		(function () {
52  			function cs(id) { return getComputedStyle(document.getElementById(id)); }
53  			function r(id) { return document.getElementById(id).getBoundingClientRect(); }
54  			window.__probe = { toJSON: function () {
55  				var p = cs('pop');
56  				var pr = r('pop'), mr = r('more'), fr = r('ftr');
57  				return { border: p.borderTopWidth, borderStyle: p.borderTopStyle, radius: parseFloat(p.borderTopLeftRadius), shadow: p.boxShadow,
58  					applyBg: cs('apply').backgroundColor, cancelBg: cs('cancel').backgroundColor, clearBg: cs('clear').backgroundColor,
59  					inputH: r('inp').height, applyH: r('apply').height, cancelH: r('cancel').height, clearH: r('clear').height,
60  					moreRight: pr.right - mr.right, moreBottom: fr.bottom - mr.bottom, moreIsRight: mr.left > r('clear').right,
61  					titleColor: cs('ttl').color, accent: (function () { var t = document.createElement('i'); t.style.color = 'var(--jc-accent)'; document.body.appendChild(t); return getComputedStyle(t).color; })() };
62  			} };
63  		})();
64  		</script>
65  		""";
66  
67  	private static Map<String,Map<String,Object>> report;
68  
69  	private static String page(String scheme) {
70  		var attrs = "data-jc-scheme=\"" + scheme + "\" data-jc-scheme-pin=\"" + scheme + "\"";
71  		return "<!DOCTYPE html>\n<html lang=\"en\" " + attrs + "><head><meta charset=\"utf-8\"><link rel=\"icon\" href=\"data:,\">\n"
72  			+ "<link rel=\"stylesheet\" href=\"" + CHROME_CSS + "\">\n<link rel=\"stylesheet\" href=\"" + DARK_CSS + "\">\n"
73  			+ "<link rel=\"stylesheet\" href=\"" + THEME_CSS + "\">\n<link rel=\"stylesheet\" href=\"" + VIEWS_CSS + "\">\n"
74  			+ "</head><body>\n" + BODY + PROBE + "</body></html>\n";
75  	}
76  
77  	@BeforeAll
78  	static void run() throws Exception {
79  		var f = ConsoleBrowserFixture.create("column-search-popover-style");
80  		for (var s : SCHEMES)
81  			f.page(s, ORIGIN + "/" + s, page(s));
82  		report = f.run();
83  	}
84  
85  	private static Map<String,Object> probe(String scheme) {
86  		var r = report.get(scheme);
87  		assertEquals(List.of(), r.get("pageErrors"), () -> scheme + ": page errors");
88  		return ConsoleBrowserFixture.probe(r);
89  	}
90  
91  	private static double num(Map<String,Object> p, String k) {
92  		return ((Number)p.get(k)).doubleValue();
93  	}
94  
95  	@Test void a01_frameIsAThinRoundedShadowedCard() {
96  		for (var s : SCHEMES) {
97  			var p = probe(s);
98  			assertEquals("1px", p.get("border"), s);
99  			assertEquals("solid", p.get("borderStyle"), s);
100 			assertTrue(num(p, "radius") >= 4, s + ": radius " + p.get("radius"));
101 			assertNotEquals("none", p.get("shadow"), s + ": shadow");
102 		}
103 	}
104 
105 	@Test void b01_applyIsFilledAndTheOthersAreNot() {
106 		for (var s : SCHEMES) {
107 			var p = probe(s);
108 			assertNotEquals(p.get("cancelBg"), p.get("applyBg"), s + ": apply vs cancel");
109 			assertEquals(p.get("cancelBg"), p.get("clearBg"), s + ": cancel vs clear");
110 		}
111 	}
112 
113 	@Test void b02_buttonsMatchTheInputHeight() {
114 		for (var s : SCHEMES) {
115 			var p = probe(s);
116 			for (var k : List.of("applyH", "cancelH", "clearH"))
117 				assertEquals(num(p, "inputH"), num(p, k), 1.0, s + ": " + k);
118 		}
119 	}
120 
121 	@Test void c01_helpMarkSitsAtTheBottomRight() {
122 		for (var s : SCHEMES) {
123 			var p = probe(s);
124 			assertEquals(Boolean.TRUE, p.get("moreIsRight"), s);
125 			assertTrue(num(p, "moreRight") <= 16, s + ": gap to the right edge " + p.get("moreRight"));
126 			assertTrue(num(p, "moreBottom") <= 4, s + ": gap to the footer bottom " + p.get("moreBottom"));
127 		}
128 	}
129 
130 	@Test void d01_titleUsesBodyText() {
131 		for (var s : SCHEMES)
132 			assertNotEquals(probe(s).get("accent"), probe(s).get("titleColor"), s);
133 	}
134 }
