41  	private static String resource(String name) throws IOException {
42  		try (var in = ViewsMixin.class.getResourceAsStream(name)) {
43  			assertNotNull(in, () -> "missing classpath resource: " + name);
44  			return new String(in.readAllBytes(), UTF_8);
45  		}
46  	}

48  	private static String viewsJs() throws IOException { return resource(ViewsMixin.VIEWS_JS_RESOURCE); }

50  	private static String functionBody(String body, String signature) {
51  		var start = body.indexOf(signature);
52  		assertTrue(start >= 0, () -> signature + " not found:\n" + body);
53  		var end = body.indexOf("\n\t}", start);
54  		return body.substring(start, end < 0 ? body.length() : end);
55  	}

127 	@BeforeAll
128 	static void probeIfNodeAvailable() throws Exception {
129 		if (!nodeAvailable())
130 			return;
131 		var harness = locateHarness();
132 		if (harness == null)
133 			return;
134 		var rendersFile = Files.createTempFile("juneau-renders-", ".js");
135 		var viewsFile = Files.createTempFile("juneau-views-", ".js");
136 		try {
137 			Files.writeString(rendersFile, resource(ViewsMixin.RENDERS_JS_RESOURCE), UTF_8);
138 			Files.writeString(viewsFile, resource(ViewsMixin.VIEWS_JS_RESOURCE), UTF_8);
139 			report = Json.to(runNode(harness, rendersFile, viewsFile), Map.class);
140 		} finally {
141 			Files.deleteIfExists(rendersFile);
142 			Files.deleteIfExists(viewsFile);
143 		}
144 	}

146 	private static boolean nodeAvailable() {
147 		try {
148 			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
149 			if (!p.waitFor(5, TimeUnit.SECONDS)) {
150 				p.destroyForcibly();
151 				return false;
152 			}
153 			return p.exitValue() == 0;
154 		} catch (@SuppressWarnings("unused") Exception e) {
155 			return false;
156 		}
157 	}
158 
159 	private static Path locateHarness() {
160 		var basedir = System.getProperty("basedir");
161 		if (basedir != null) {
162 			var p = Path.of(basedir, "src/test/js/row-action-form.cjs");
163 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
164 		}
165 		for (var rel : List.of(
166 			"src/test/js/row-action-form.cjs",
167 			"juneau-rest/juneau-console-views/src/test/js/row-action-form.cjs"
168 		)) {
169 			var p = Path.of(rel);
170 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
171 		}
172 		return null;
173 	}
174 
175 	private static String runNode(Path harness, Path rendersJs, Path viewsJs) throws Exception {
176 		var stdout = Files.createTempFile("row-action-form-stdout-", ".json");
177 		var stderr = Files.createTempFile("row-action-form-stderr-", ".txt");
178 		try {
179 			var pb = new ProcessBuilder(List.of("node", harness.toString(), rendersJs.toString(), viewsJs.toString()))
180 				.redirectOutput(stdout.toFile())
181 				.redirectError(stderr.toFile());
182 			var p = pb.start();
183 			if (!p.waitFor(30, TimeUnit.SECONDS)) {
184 				p.destroyForcibly();
185 				fail("row-action-form.cjs did not finish within 30s; stderr:\n" + quietRead(stderr));
186 			}
187 			if (p.exitValue() != 0)
188 				fail("row-action-form.cjs exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr)
189 					+ "\nstdout:\n" + quietRead(stdout));
190 			return Files.readString(stdout, UTF_8);
191 		} finally {
192 			Files.deleteIfExists(stdout);
193 			Files.deleteIfExists(stderr);
194 		}
195 	}
196 
197 	private static String quietRead(Path p) {
198 		try { return Files.readString(p, UTF_8); }
199 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
200 	}

207 	/**
208 	 * CRITICAL CAVEAT: a missing key reads as {@code false} through {@code flag()}, but the SAME missing key
209 	 * read directly off the report map (as {@code assertBean} does) renders as the literal string {@code <null>},
210 	 * not {@code false}. Folding a {@code flag()}-backed boolean into an {@code assertBean} call is therefore only
211 	 * safe where the harness unconditionally assigns that key on every path (never omits it) - verify that in
212 	 * {@code row-action-form.cjs} before converting a new site, not just by analogy to this one.
213 	 */
214 	private static boolean flag(String key) {
215 		return Boolean.TRUE.equals(report().get(key));
216 	}
