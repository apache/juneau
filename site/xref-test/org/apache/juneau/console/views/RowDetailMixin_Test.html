30  /** Serving test for {@link RowDetailMixin}. */
31  @SuppressWarnings({
32  	"unchecked", // JSON maps
33  	"resource" // Mock client held in a static field
34  })
35  class RowDetailMixin_Test extends TestBase {
36  
37  	public static class Pet {
38  		private final String id;
39  		public Pet(String id) { this.id = id; }
40  		public String getId() { return id; }
41  		public String getName() { return "Rex-" + id; }
42  		public String getEmail() { return "x@y.z"; }
43  		@org.apache.juneau.commons.bean.BeanIgnore
44  		public String getHidden() { return "h"; }
45  	}
46  
47  	public static class Owner {
48  		public String getId() { return "o"; }
49  		public String getName() { return "n"; }
50  	}
51  
52  	static final RowDetail PET = RowDetail.fromBean(Pet.class, id -> "p1".equals(id) ? new Pet(id) : null)
53  		.title("{name}")
54  		.section("Info", "id", "name")
55  		.tab(DetailTab.fields("More", "email").count(id -> 7));
56  
57  	static final RowDetail OWNER = RowDetail.fromBean(Owner.class, id -> new Owner()).key("owner");
58  
59  	@Rest
60  	public static class R extends BasicRestServlet implements RowDetailMixin {
61  		private static final long serialVersionUID = 1L;
62  		@Override public Collection<RowDetail> rowDetails() { return List.of(PET, OWNER); }
63  	}
64  
65  	static final RowDetail SHARED = RowDetail.fromBean(Pet.class, id -> "p1".equals(id) ? new Pet(id) : null).key("shared");
66  
67  	static final RowDetail BOOM = RowDetail.fromBean(Pet.class, id -> { throw new IllegalStateException("lookup failed"); }).key("boom")
68  		.tab(DetailTab.fields("Info", "name").count(id -> { throw new IllegalStateException("count failed"); }));
69  
70  	static final RowDetail BADCOUNT = RowDetail.fromBean(Pet.class, id -> new Pet(id)).key("badcount")
71  		.tab(DetailTab.fields("Info", "name").count(id -> { throw new IllegalStateException("count failed"); }));
72  
73  	@Rest
74  	public static class R2 extends BasicRestServlet implements RowDetailMixin {
75  		private static final long serialVersionUID = 1L;
76  		@Override public Collection<RowDetail> rowDetails() { return List.of(SHARED, BOOM, BADCOUNT); }
77  	}
78  
79  	private static final MockRestClient c = MockRestClient.buildLax(R.class);
80  	private static final MockRestClient c2 = MockRestClient.buildLax(R2.class);
81  
82  	private static void assertSame404(org.apache.juneau.rest.client.classic.RestResponse r) throws Exception {
83  		r.assertStatus(404);
84  		assertEquals("no-store", r.getStringHeader("Cache-Control").orElse(null));
85  		assertEquals("{}", r.getContent().asString().replaceAll("\\s", ""));
86  	}
87  
88  	@Test void a01_detailShape() throws Exception {
89  		var body = c.get("/row-detail/pet/p1").header("Accept", "application/json").run().assertStatus(200).getContent().asString();
90  		var m = Json.to(body, Map.class);
91  		assertEquals("1", m.get("contractVersion"));
92  		var f = (Map<String,Object>) m.get("fields");
93  		assertEquals("Rex-p1", f.get("name"));
94  		assertNotNull(m.get("meta"));
95  		assertEquals(7, ((Map<String,Object>) m.get("counts")).get("more"));
96  	}
97  
98  	@Test void a02_tabShape() throws Exception {
99  		var body = c.get("/row-detail/pet/p1/tab/more").header("Accept", "application/json").run().assertStatus(200).getContent().asString();
100 		var f = (Map<String,Object>) Json.to(body, Map.class).get("fields");
101 		assertEquals("x@y.z", f.get("email"));
102 		assertNull(f.get("name"));
103 	}
104 
105 	@Test void a03_countsShape() throws Exception {
106 		var body = c.get("/row-detail/pet/p1/counts").header("Accept", "application/json").run().assertStatus(200).getContent().asString();
107 		assertEquals(7, ((Map<String,Object>) Json.to(body, Map.class).get("counts")).get("more"));
108 	}
109 
110 	@Test void a04_identical404() throws Exception {
111 		for (var p : List.of("/row-detail/nope/p1", "/row-detail/pet/zzz", "/row-detail/pet/p1/tab/nope",
112 				"/row-detail/pet/..", "/row-detail/pet/bad%20id", "/row-detail/pet/p1/tab/Bad_Id"))
113 			assertSame404(c.get(p).header("Accept", "application/json").run());
114 	}
115 
116 	@Test void a05_unnamedPropertiesNotServed() throws Exception {
117 		var body = c.get("/row-detail/pet/p1/tab/more").header("Accept", "application/json").run().getContent().asString();
118 		assertFalse(body.contains("Rex"));
119 	}
120 
121 	@Test void a06_duplicateKey_failsStartup() {
122 		var mixin = new RowDetailMixin() {
123 			@Override public Collection<RowDetail> rowDetails() { return List.of(PET, RowDetail.fromBean(Pet.class, id -> null)); }
124 		};
125 		assertThrows(IllegalStateException.class, mixin::rowDetailIndex);
126 	}
127 
128 	@Test void z01_oneDetailOnTwoTables_servesFieldsOfBoth() throws Exception {
129 		TableSpec.create("ta").dataUrl("/a").columns(Column.create("id"), Column.create("name")).detail(SHARED).toCardBody();
130 		TableSpec.create("tb").dataUrl("/b").columns(Column.create("id"), Column.create("email")).detail(SHARED).toCardBody();
131 		var body = c2.get("/row-detail/shared/p1").header("Accept", "application/json").run().assertStatus(200).getContent().asString();
132 		var f = (Map<String,Object>)Json.to(body, Map.class).get("fields");
133 		assertTrue(f.containsKey("name"), body);
134 		assertTrue(f.containsKey("email"), body);
135 	}
136 
137 	@Test void z02_throwingLookup_isTheSameEmpty404() throws Exception {
138 		assertSame404(c2.get("/row-detail/boom/p1").header("Accept", "application/json").run());
139 		assertSame404(c2.get("/row-detail/boom/p1/counts").header("Accept", "application/json").run());
140 		assertSame404(c2.get("/row-detail/boom/p1/tab/info").header("Accept", "application/json").run());
141 	}
142 
143 	@Test void z03_throwingCounter_dropsItsCount_notTheDetail() throws Exception {
144 		var body = c2.get("/row-detail/badcount/p1").header("Accept", "application/json").run().assertStatus(200).getContent().asString();
145 		var counts = (Map<String,Object>)Json.to(body, Map.class).get("counts");
146 		assertFalse(counts.containsKey("info"), body);
147 	}
