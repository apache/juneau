28  /**
29   * Drives {@code terminal-view-poll.cjs}: polling against a fake fetch and clock, {@code mount}, the {@code terminal-view} region
30   * populator and {@code of}.  One method per case, plus a catch-all that fails when the harness reports a case without
31   * a method of its own or a method names a case the harness does not have.
32   */
33  class ViewsJs_TerminalViewPoll_Test extends TestBase {
34  
35  	static final String VECTORS = "/org/apache/juneau/console/views/terminal-view-vectors.json";
36  
37  	/** Every case name the harness is expected to report; each task appends its own. */
38  	private static final Set<String> EXPECTED = new TreeSet<>(List.of(
39  		"poll01_firstRequestHasNoQuery_thenAfterNext", "poll02_urlWithQueryUsesAmpersand", "poll03_moreRefetchesAtOnce",
40  		"poll04_idleWaitsRefreshMs", "poll05_stopsOnTerminal", "poll06_stopsOnDoneWithoutMore",
41  		"poll07_abortStops", "poll08_hiddenTabPausesAndVisibleResumes", "poll09_backoffOnOutage_andRecovers",
42  		"poll10_4xxIsFatal", "poll11_408And429Retry", "poll12_410WithTokenResetsAndRefetches",
43  		"poll13_410WithoutTokenIsFatal", "poll14_contractMismatchIsFatal", "poll15_nonIncreasingSeqIsFatal",
44  		"poll16_badNextIsFatal", "poll17_oneRequestInFlight", "poll18_neverRejects",
45  		"poll19_badEventsUrlIsE1", "poll20_emptyMorePageDoesNotSpin", "poll21_replayedPageIsHarmless",
46  		"mount01_returnsCleanupSynchronously", "mount02_cleanupTearsDownInOrder", "mount03_idSubstitution",
47  		"mount04_pollFalseNeverFetches", "mount05_populatorRegistered", "mount06_emitsTerminalViewState",
48  		"of01_byElementAndById", "of02_unknownIsNull", "of03_sseStyleAppend",
49  		"chk99_noStubsLeft"
50  	));
51  
52  	static Map<?,?> report(String harness) {
53  		var r = RegionsHarness.reportWith(harness, ViewsMixin.HELPERS_JS_RESOURCE, ViewsMixin.RUN_VIEW_JS_RESOURCE, VECTORS);
54  		assumeTrue(r != null, "node not available or " + harness + " not found - skipped");
55  		return r;
56  	}
57  
58  	static void assertAllTrue(Map<?,?> r, String... keys) {
59  		for (var k : keys)
60  			assertEquals(true, r.get(k), () -> k + " -> " + r.get(k) + " in " + r);
61  	}
62  
63  	private static Map<?,?> r() { return report("terminal-view-poll.cjs"); }
64

98  		var r = r();

100 		for (var e : r.entrySet())
101 			assertEquals(true, e.getValue(), () -> e.getKey() + " -> " + e.getValue());
