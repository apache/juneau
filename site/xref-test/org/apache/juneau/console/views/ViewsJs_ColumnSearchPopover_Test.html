33  /**
34   * Behavioral coverage for the per-column search popover commit semantics (operator help and the evaluation
35   * model).  A Node harness ({@code column-search-popover.cjs}) loads {@code juneau-search.js} into the same sandbox
36   * as views, drives {@code openColumnSearchPopover(...)} through each commit path, and reports what reached the grid:
37   * help rendered from VIEW_META, an incomplete {@code $}-draft leaving the grid untouched, an out-of-set operator
38   * rejected, a bare quick-filter previewing live on a client table but deferred on a server table, a {@code $}
39   * expression deferred until Enter, and Esc reverting a live preview.  Source-shape pins live in
40   * {@link ViewsJs_HeaderSortSearch_Test}.  Gated on {@code node} being on {@code PATH}.
41   *
42   * <p>
43   * Test groups: {@code a01}-{@code a06} the commit/preview/revert semantics above; {@code a07} the per-type bare-value
44   * help rendered before the operator help (and absent without search metadata); {@code b01} the per-table live
45   * announcer (politely live, outside the wrapper, reused on repeat); {@code c01}-{@code c05} the dismiss announcements
46   * (a revert names the column title, a never-previewed {@code $}-draft still announces, an Enter commit or a typed-then-
47   * restored value never announces, and two reverts in a row both announce); {@code d01} the client-mode DSL store
48   * path (writes go to the per-table store and one {@code search.fixed} predicate, never native search).
49   */
50  class ViewsJs_ColumnSearchPopover_Test extends TestBase {

52  	private static String resource(String name) throws IOException {
53  		try (var in = ViewsMixin.class.getResourceAsStream(name)) {
54  			assertNotNull(in, name);
55  			return new String(in.readAllBytes(), UTF_8);
56  		}
57  	}

59  	private static Map<?,?> report;
60  
61  	@BeforeAll
62  	static void probeIfNodeAvailable() throws Exception {
63  		if (!nodeAvailable())
64  			return;
65  		var harness = locateHarness();
66  		if (harness == null)
67  			return;
68  		var viewsFile = Files.createTempFile("juneau-views-", ".js");
69  		var rendersFile = Files.createTempFile("juneau-renders-", ".js");
70  		var searchFile = Files.createTempFile("juneau-search-", ".js");
71  		try {
72  			Files.writeString(viewsFile, resource(ViewsMixin.VIEWS_JS_RESOURCE), UTF_8);
73  			Files.writeString(rendersFile, resource(ViewsMixin.RENDERS_JS_RESOURCE), UTF_8);
74  			Files.writeString(searchFile, resource(ViewsMixin.SEARCH_JS_RESOURCE), UTF_8);
75  			report = Json.to(runNode(harness, rendersFile, viewsFile, searchFile), Map.class);
76  		} finally {
77  			Files.deleteIfExists(viewsFile);
78  			Files.deleteIfExists(rendersFile);
79  			Files.deleteIfExists(searchFile);
80  		}
81  	}

83  	private static boolean nodeAvailable() {
84  		try {
85  			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
86  			if (!p.waitFor(5, TimeUnit.SECONDS)) {
87  				p.destroyForcibly();
88  				return false;
89  			}
90  			return p.exitValue() == 0;
91  		} catch (Exception e) {
92  			return false;
93  		}
94  	}

96  	private static Path locateHarness() {
97  		var basedir = System.getProperty("basedir");
98  		if (basedir != null) {
99  			var p = Path.of(basedir, "src/test/js/column-search-popover.cjs");
100 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
101 		}
102 		for (var rel : List.of(
103 			"src/test/js/column-search-popover.cjs",
104 			"juneau-rest/juneau-console-views/src/test/js/column-search-popover.cjs"
105 		)) {
106 			var p = Path.of(rel);
107 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
108 		}
109 		return null;
110 	}

112 	private static String runNode(Path harness, Path rendersJs, Path viewsJs, Path searchJs) throws Exception {
113 		var stdout = Files.createTempFile("column-search-popover-stdout-", ".json");
114 		var stderr = Files.createTempFile("column-search-popover-stderr-", ".txt");
115 		try {
116 			var pb = new ProcessBuilder(List.of(
117 					"node", harness.toString(), rendersJs.toString(), viewsJs.toString(), searchJs.toString()))
118 				.redirectOutput(stdout.toFile())
119 				.redirectError(stderr.toFile());
120 			var p = pb.start();
121 			if (!p.waitFor(30, TimeUnit.SECONDS)) {
122 				p.destroyForcibly();
123 				fail("column-search-popover.cjs did not finish within 30s; stderr:\n" + quietRead(stderr));
124 			}
125 			if (p.exitValue() != 0)
126 				fail("column-search-popover.cjs exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr)
127 					+ "\nstdout:\n" + quietRead(stdout));
128 			return Files.readString(stdout, UTF_8);
129 		} finally {
130 			Files.deleteIfExists(stdout);
131 			Files.deleteIfExists(stderr);
132 		}
133 	}
134 
135 	private static String quietRead(Path p) {
136 		try { return Files.readString(p, UTF_8); }
137 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
138 	}

145 	@Test void a01_operatorHelp_renderedFromViewMeta() {
146 		var r = report();
147 		// Every effective operator gets a help row: its "$name" then the operator API's help text.
148 		assertBean(r, "hasOpenColumnSearchPopover,hasSearchEngine,helpOps,helpTexts",
149 			"true,true,[$eq,$in,$contains],[Exact match.,Any of the listed values.,Case-insensitive substring.]");
150 	}

153 		var r = report();

155 		// operator's own help text still wins.
156 		assertBean(r, "fallbackOps,fallbackTexts",
157 			"[$eq,$contains,$near],[Exact, case-sensitive match. Example: $eq(OPEN),Case-insensitive substring match. Example: $contains(err),Within N units. Example: $near(5)]");
