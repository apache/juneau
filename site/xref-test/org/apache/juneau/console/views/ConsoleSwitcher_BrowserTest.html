205 	private static String sections(String id, String... cards) {
206 		var sb = new StringBuilder("{\"id\":\"" + id + "\",\"label\":\"" + id + "\",\"sections\":[");
207 		for (var i = 0; i < cards.length; i++)
208 			sb.append(i == 0 ? "" : ",").append("{\"id\":\"s").append(i + 1).append("\",\"label\":\"S").append(i + 1).append("\",\"card\":").append(cards[i]).append('}');
209 		return sb.append("]}").toString();
210 	}

236 	static void runPages() throws Exception {
237 		var kbd = switcher("sw1", "mv", sections("mv", leaf("m1"), leaf("m2")), view("other", leaf("o1")));
238 		var two = switcher("swa", null, view("a1", leaf("x1")), view("a2", leaf("x2")), view("a3", leaf("x3")))
239 			+ "," + switcher("swb", null, view("b1", leaf("y1")), view("b2", leaf("y2")), view("b3", leaf("y3")));
240 		var seededView = Map.of(VIEW_PREF, "\"data\"");
241 		var seededSection = Map.of(VIEW_PREF, "\"multi\"", SECTION_PREF, "\"s2\"");
242 		var urlBeatsPrefs = Map.of(VIEW_PREF, "\"multi\"");
243 		report = ConsoleBrowserFixture.create("switcher")
244 			.asset(CHARTS, resource(ConsoleChromeMixin.CHARTS_JS_RESOURCE), "text/javascript")
245 			.asset(ECHARTS, resource(ECHARTS_RESOURCE), "text/javascript")
246 			.page("main", ORIGIN + "/main", page(main(), MAIN_SCRIPT))
247 			.waitFor("#done")
248 			.page("kbd", ORIGIN + "/kbd", page(kbd, KBD_SCRIPT))
249 			.waitFor("#done")
250 			.eval("document.querySelector('#sw1 .juneau-switcher-selector [aria-selected=\"true\"]').focus();" + rec("k0"))
251 			.press("ArrowRight")
252 			.eval(rec("k1"))
253 			.press("Tab")
254 			.eval(rec("k2"))
255 			.eval("JuneauConsole.cardApi('sw1').select('mv'); document.querySelector('#sw1 [data-juneau-strip-mode=\"tab\"] [data-tab-id=\"s1\"]').focus();")
256 			.press("Shift")
257 			.eval(rec("k3"))
258 			.asset("/api/nt", "[{\"id\":\"p1\",\"name\":\"Rex\"}]", "application/json")
259 			.asset("/api/nt/p1/detail", "{\"contractVersion\":\"1\",\"fields\":{\"name\":\"Rex\"}}", "application/json")
260 			.page("layout", ORIGIN + "/layout", page(layoutCards(), LAYOUT_SCRIPT))
261 			.waitFor("#done")
262 			.waitFor("table[data-juneau-view=\"nt\"] tbody td.sorting_1")
263 			.eval("var host = document.querySelector('.juneau-switcher-view [data-juneau-in-switcher] table[data-juneau-view=\"nt\"]');"
264 				+ "window.__probe.inSwitcher = !!host;"
265 				+ "window.__probe.statStrip = !!document.querySelector('[data-juneau-stat=\"q1\"]');"
266 				+ "window.__probe.ribbon = !!document.querySelector('[data-juneau-ribbon-id=collapseAll]');"
267 				+ "document.querySelector('table[data-juneau-view=\"nt\"] .juneau-view-detail-toggle').click();")
268 			.eval("(async function () { await new Promise(function (r) { setTimeout(r, 1500); });"
269 				+ "window.__probe.detailPanels = document.querySelectorAll('.juneau-view-detail-panel').length;"
270 				+ "})()")
271 			.page("two", ORIGIN + "/two", page(two, TWO_SCRIPT))
272 			.waitFor("#done")
273 			.page("prefView", ORIGIN + "/prefview", page(main(), PROBE_SCRIPT))
274 			.storage(seededView)
275 			.waitFor("#done")
276 			.page("prefSection", ORIGIN + "/prefsection", page(main(), PROBE_SCRIPT))
277 			.storage(seededSection)
278 			.waitFor("#done")
279 			.page("sharedUrl", ORIGIN + "/shared?state=" + URLEncoder.encode("sw(sw1:data)", StandardCharsets.UTF_8), page(main(), PROBE_SCRIPT))
280 			.storage(urlBeatsPrefs)
281 			.waitFor("#done")
282 			.run();
283 	}
