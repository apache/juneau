32  /**
33   * The behavioral tripwire for what {@code submitActionDialog} ACTUALLY TRANSMITS as its target: the row's id, or -
34   * for a modal that has explicitly opted in with {@code selfTargeted} - the self-bound idempotency key's own value.
35   *
36   * <p>A source-text presence check cannot carry this property.  The literals {@code "idempotencyKey"} and
37   * {@code "targetId"} appear in the function body under EITHER a correct opt-in-gated rule or a regressed blanket
38   * "the key always wins" one, so a check that only greps for them (see {@code ViewsJs_ModalResult_Test.g01}) stays
39   * green through the exact regression it looks like it is guarding: an unconditional precedence silently stops
40   * sending the real row id that an artifact-bound key was minted against, making that key's own binding check
41   * vacuous for every existing {@code present=dialog} consumer.
42   *
43   * <p>So this class has two layers:
44   * <ul class='spaced-list'>
45   * 	<li><b>A default-gate SHAPE pin</b> (a01/a02, no Node required): the whole {@code targetId} statement, pinned
46   * 		exactly, plus the absence of any {@code tr}-null conditional around the opt-in branch.  Substring presence
47   * 		is deliberately not enough - {@code modal?.selfTargeted &&} stays present even if a {@code tr == null}
48   * 		guard is wrapped around the branch it gates, which is precisely how the row-less case could silently
49   * 		regress to omitting the target entirely.
50   * 	<li><b>A behavioral harness</b> ({@code selftargeted-idempotency.cjs}) that INVOKES the function once per
51   * 		branch and reads the transmitted request body.  Runs when {@code node} is on {@code PATH} (skipped
52   * 		otherwise - no {@code -Pjs-tests} required, mirroring {@code ViewsJs_DetailActionDialogRouting_Test}).
53   * </ul>
54   */
55  class ViewsJs_SelfTargetedIdempotency_Test extends TestBase {

59  		"const targetId = (modal?.selfTargeted && modal?.idempotencyKey != null) "
60  		+ "? modal.idempotencyKey : (tr?.dataset?.juneauRowId ?? null);";

76  	/** Extracts a named function's body: from `function <name>(` to the next top-level `\n\t}`. */
77  	private static String functionBody(String body, String signature) {
78  		var start = body.indexOf(signature);
79  		assertTrue(start >= 0, () -> "'" + signature + "' not found:\n" + body);
80  		var end = body.indexOf("\n\t}", start);
81  		return body.substring(start, end < 0 ? body.length() : end);
82  	}

88  	@Test void a01_sourceShape_targetIdStatementIsExactlyTheOptInGatedTernary() throws Exception {
89  		var fnBody = functionBody(viewsJs(), "function submitActionDialog(");
90  		var start = fnBody.indexOf("const targetId");
91  		assertTrue(start >= 0, () -> "submitActionDialog must compute a targetId:\n" + fnBody);
92  		var end = fnBody.indexOf(';', start);
93  		assertTrue(end >= 0, fnBody);
94  		// Whitespace-normalized so the statement may wrap across lines; everything else is pinned exactly - which
95  		// is what forbids `tr` appearing anywhere except the non-opt-in arm, guard included.
96  		var statement = fnBody.substring(start, end + 1).replaceAll("\\s+", " ");
97  		assertEquals(TARGET_ID_STATEMENT, statement,
98  			"the opt-in-gated precedence rule must have exactly this shape - a `tr` read in (or a guard around) "
99  			+ "the selfTargeted arm is the row-less regression this pin exists to catch");
100 	}
101 
102 	@Test void a02_sourceShape_noRowNullConditionalAnywhereInTheSubmitSeam() throws Exception {
103 		var fnBody = functionBody(viewsJs(), "function submitActionDialog(");
104 		for (var guard : new String[]{"tr == null", "tr != null", "tr === null", "tr !== null", "!tr", "! tr"})
105 			assertFalse(fnBody.contains(guard),
106 				() -> "a `" + guard + "` conditional in submitActionDialog would gate the opt-in branch on the "
107 					+ "presence of a row, reviving the original row-less bug:\n" + fnBody);
108 		// Exactly one target computation, and exactly one attachment of it: a second computation further down could
109 		// quietly undo the first.
110 		assertTrue(fnBody.contains("if (targetId != null) extra.targetId = targetId;"), fnBody);
111 		var withoutAttachment = fnBody.replace("if (targetId != null) extra.targetId = targetId;", "");
112 		assertEquals(1, withoutAttachment.split("targetId =", -1).length - 1, withoutAttachment);
113 		// The rest of the seam is unchanged: the key attachment and the delegation to the row submit path.
114 		assertTrue(fnBody.contains("if (modal?.idempotencyKey != null) extra.idempotencyKey = modal.idempotencyKey;"), fnBody);
115 		assertTrue(fnBody.contains("submitRowAction(action, table, tr, ctx, extra);"), fnBody);
116 	}

124 	@BeforeAll
125 	static void probeIfNodeAvailable() throws Exception {
126 		if (!nodeAvailable())
127 			return;
128 		var harness = locateHarness();
129 		if (harness == null)
130 			return;
131 		var viewsFile = Files.createTempFile("juneau-views-", ".js");
132 		var rendersFile = Files.createTempFile("juneau-renders-", ".js");
133 		try {
134 			Files.writeString(viewsFile, viewsJs(), UTF_8);
135 			Files.writeString(rendersFile, rendersJs(), UTF_8);
136 			report = Json.to(runNode(harness, rendersFile, viewsFile), Map.class);
137 		} finally {
138 			Files.deleteIfExists(viewsFile);
139 			Files.deleteIfExists(rendersFile);
140 		}
141 	}

143 	private static boolean nodeAvailable() {
144 		try {
145 			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
146 			if (!p.waitFor(5, TimeUnit.SECONDS)) {
147 				p.destroyForcibly();
148 				return false;
149 			}
150 			return p.exitValue() == 0;
151 		} catch (Exception e) {
152 			return false;
153 		}
154 	}

156 	private static Path locateHarness() {
157 		var basedir = System.getProperty("basedir");
158 		if (basedir != null) {
159 			var p = Path.of(basedir, "src/test/js/selftargeted-idempotency.cjs");
160 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
161 		}
162 		for (var rel : List.of(
163 			"src/test/js/selftargeted-idempotency.cjs",
164 			"juneau-rest/juneau-console-views/src/test/js/selftargeted-idempotency.cjs"
165 		)) {
166 			var p = Path.of(rel);
167 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
168 		}
169 		return null;
170 	}

172 	private static String runNode(Path harness, Path rendersJs, Path viewsJs) throws Exception {
173 		var stdout = Files.createTempFile("selftargeted-idempotency-stdout-", ".json");
174 		var stderr = Files.createTempFile("selftargeted-idempotency-stderr-", ".txt");
175 		try {
176 			var pb = new ProcessBuilder(List.of("node", harness.toString(), rendersJs.toString(), viewsJs.toString()))
177 				.redirectOutput(stdout.toFile())
178 				.redirectError(stderr.toFile());
179 			var p = pb.start();
180 			if (!p.waitFor(30, TimeUnit.SECONDS)) {
181 				p.destroyForcibly();
182 				fail("selftargeted-idempotency.cjs did not finish within 30s; stderr:\n" + quietRead(stderr));
183 			}
184 			if (p.exitValue() != 0)
185 				fail("selftargeted-idempotency.cjs exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr)
186 					+ "\nstdout:\n" + quietRead(stdout));
187 			return Files.readString(stdout, UTF_8);
188 		} finally {
189 			Files.deleteIfExists(stdout);
190 			Files.deleteIfExists(stderr);
191 		}
192 	}
193 
194 	private static String quietRead(Path p) {
195 		try { return Files.readString(p, UTF_8); }
196 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
197 	}

209 	/** Case (i): the opt-in wins over a real, DIFFERENT row id. */
210 	@Test void b01_optIn_withARow_transmitsTheKeyNotTheRowId() {
211 		var r = report();
212 		assertEquals(true, r.get("optIn_withRow_rowIdWasDifferent"), "the fixture must not accidentally agree");
213 		assertEquals("k-0123456789abcdef", r.get("optIn_withRow_targetId"),
214 			() -> "a selfTargeted modal's key must be transmitted as the target even when a row id exists: " + r);
215 		assertEquals("k-0123456789abcdef", r.get("optIn_withRow_idempotencyKey"), r::toString);
216 	}

218 	/**
219 	 * Case (ii): the direct regression guard.  A key with NO opt-in - the shape the example app's ack form uses,
220 	 * minted against a real row id - must keep transmitting that row's id, or its own binding check is vacuous.
221 	 */
222 	@Test void b02_noOptIn_withARow_transmitsTheRowIdNotTheKey() {
223 		var r = report();
224 		assertEquals("row-42", r.get("noOptIn_withRow_targetId"),
225 			() -> "an artifact-bound key must NOT hijack the target: this is the framework-wide regression that "
226 				+ "an unconditional precedence rule would cause for every existing present=dialog consumer: " + r);
227 		assertEquals("k-0123456789abcdef", r.get("noOptIn_withRow_idempotencyKey"),
228 			() -> "the key itself still rides the submit, unchanged: " + r);
229 	}
230 
231 	/**
232 	 * Case (iii): the direct regression guard for the ORIGINAL row-less bug.  With no row there is nothing to fall
233 	 * back to, so a {@code tr == null} guard reintroduced around the opt-in branch would silently go back to
234 	 * omitting the target - and cases (i)/(ii) would both stay green while it did.
235 	 */
236 	@Test void b03_optIn_rowless_transmitsTheKeyWithNoRowToFallBackOn() {
237 		var r = report();
238 		assertEquals("k-0123456789abcdef", r.get("optIn_rowless_targetId"),
239 			() -> "a row-less selfTargeted submit must carry the key as its target, not omit the target: " + r);
240 		assertEquals("k-0123456789abcdef", r.get("optIn_rowless_idempotencyKey"), r::toString);
241 	}
242 
243 	/** The fallbacks that must not move: no key, and an opt-in with no key to honour it with. */
244 	@Test void b04_fallbacks_areUnchanged() {
245 		var r = report();
246 		assertEquals("row-42", r.get("noKey_withRow_targetId"), r::toString);
247 		assertEquals(false, r.get("noKey_withRow_hasIdempotencyKey"), r::toString);
248 		assertEquals(false, r.get("noKey_rowless_hasTargetId"),
249 			() -> "with no key and no row there is no target to invent: " + r);
250 		assertEquals("row-42", r.get("optInNoKey_withRow_targetId"),
251 			() -> "the gate requires BOTH the opt-in and a key; with no key it falls back to the row id: " + r);
252 		assertEquals(false, r.get("optInNoKey_rowless_hasTargetId"), r::toString);
253 	}
