33  /**
34   * {@link Render#pill()} / {@link Render#pill(String)} factory sugar: canonical {@code {id:"pill",meta:…}} wire form
35   * and boxed-omission rules (dot default on, absent/blank tone omitted).  These factories are pure sugar over
36   * {@link Render#of(String)} &mdash; they carry no host-specific behavior of their own; they are not tied to any host or sink.
37   */
38  class Render_Pill_Test extends TestBase {
39  
40  	private static Map<?,?> wire(Render r) {
41  		return Json.to(Json.of(r), Map.class);
42  	}
43  
44  	@Test void a01_pill_noArgs_isBareIdWithNoMeta() {
45  		var r = Render.pill();
46  		assertEquals("pill", r.id);
47  		assertNull(r.meta, "dot default is on in JS; the factory must not emit a dot key");
48  		assertEquals(Map.of("id", "pill"), wire(r));
49  	}
50  
51  	@Test void a02_pill_withTone_emitsToneMetaOnly() {
52  		for (var tone : new String[]{"info", "success", "warning", "error", "neutral"}) {
53  			var r = Render.pill(tone);
54  			assertEquals("pill", r.id);
55  			assertEquals(Map.of("tone", tone), r.meta, tone);
56  			assertEquals(Map.of("id", "pill", "meta", Map.of("tone", tone)), wire(r), tone);
57  		}
58  	}
59  
60  	@Test void a03_pill_nullOrBlankTone_omitsToneMeta() {
61  		for (var tone : new String[]{null, "", "   "}) {
62  			var r = Render.pill(tone);
63  			assertEquals("pill", r.id);
64  			assertNull(r.meta, () -> "blank tone must be omitted, identical to pill(): " + Json.of(r));
65  			assertEquals(Map.of("id", "pill"), wire(r));
66  		}
67  	}
68  
69  	@Test void a03b_pill_typedTone_matchesWireToken() {
70  		for (var tone : StatusTone.values()) {
71  			var r = Render.pill(tone);
72  			assertBean(r, "id,meta", "pill,{tone=" + tone.wire() + "}");
73  			assertBean(wire(r), "id,meta", "pill,{tone=" + tone.wire() + "}");
74  		}
75  		var n = Render.pill((StatusTone)null);
76  		assertNull(n.meta);
77  		assertEquals(Map.of("id", "pill"), wire(n));
78  	}
79  
80  	@Test void a04_chainedMeta_carriesFieldDotAction() {
81  		var r = Render.pill("success").meta("field", "state").meta("dot", "off").meta("action", "ack");
82  		assertEquals(Map.of("tone", "success", "field", "state", "dot", "off", "action", "ack"), r.meta);
83  		assertEquals(
84  			Map.of("id", "pill", "meta", Map.of("tone", "success", "field", "state", "dot", "off", "action", "ack")),
85  			wire(r));
86  	}
87  
88  	@Test void a05_stringSugarUnaffected_bareIdAndColonField() {
89  		// Render.parse still treats "id:field" the same; "pill" gains no special parsing.
90  		assertEquals("pill", Render.parse("pill").id);
91  		assertNull(Render.parse("pill").meta);
92  		var withField = Render.parse("pill:state");
93  		assertEquals("pill", withField.id);
94  		assertEquals(Map.of("field", "state"), withField.meta);
95  	}

97  	/**
98  	 * Companion pin: the {@code pill} block of the shipped {@code juneau-renders.js} must carry NO hardcoded
99  	 * hex colour.  A pill chip's colour comes exclusively from the console-ui {@code --jc-pill-*} palette classes
100 	 * (via {@code chrome.css}); the renderer only stamps the palette classes, never a literal colour.  This is the
101 	 * source-scan pin; the shipped renderer must never stamp a literal colour.
102 	 */
103 	@Test void a06_pillBlockOfRendersJs_carriesNoHardcodedHex() throws IOException {
104 		// The test CWD is the module root; the shipped renderer lives under src/main/resources.
105 		var src = Paths.get("src/main/resources/org/apache/juneau/views/juneau-renders.js");
106 		assertTrue(Files.exists(src), () -> "shipped juneau-renders.js not found at " + src.toAbsolutePath());
107 		var js = Files.readString(src, StandardCharsets.UTF_8);
108 
109 		// Isolate the pill renderer block: from its banner comment to the next renderer section (TIMESTAMP POPUP).
110 		var start = js.indexOf("--- pill renderer");
111 		assertTrue(start >= 0, "could not locate the pill renderer block banner in juneau-renders.js");
112 		var end = js.indexOf("TIMESTAMP POPUP", start);
113 		assertTrue(end > start, "could not locate the end of the pill renderer block in juneau-renders.js");
114 		var block = js.substring(start, end);
115 
116 		var hex = Pattern.compile("#[0-9a-fA-F]{3,8}\\b").matcher(block);
117 		assertFalse(hex.find(),
118 			() -> "pill block of juneau-renders.js must contain no hardcoded hex colour; found '"
119 				+ hex.group() + "' - colour must come from the --jc-pill-* palette classes only.");
120 	}
121 
122 	@Test void a20_meta_numericValue_serializesAsNumber() {
123 		var r = Render.of("decimal").meta("places", 2).meta("field", "price");
124 		assertEquals("{\"id\":\"decimal\",\"meta\":{\"places\":2,\"field\":\"price\"}}", Json.of(r));
125 		assertEquals(2, ((Map<?,?>)wire(r).get("meta")).get("places"));
126 	}
