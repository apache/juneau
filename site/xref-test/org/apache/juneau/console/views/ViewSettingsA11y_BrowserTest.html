33  /**
34   * The end-to-end half of the module's <b>JavaScript-execution harness</b>: runs the REAL served
35   * {@code juneau-views.js} / {@code juneau-urlstate.js} / {@code juneau-search.js} / {@code juneau-pagestate.js} /
36   * {@code juneau-config.js} in a real headless browser and proves, together in one page, four scenarios plus a fifth, Copy-link: a malformed {@code ?state=} tab id applied alongside a real
37   * filter clause; the View Settings dialog driven keyboard-only; a column-search popover's Escape-revert
38   * announcement and focus return; and a version-mismatched {@code schemaVersion} blob's one-time reset notice surviving a real
39   * page reload; and a real click on the Copy-link toolbar button copying and announcing the built URL.
40   *
41   * <h5 class='section'>Why this exists (beyond the Node harnesses):</h5>
42   * <p>
43   * {@code ViewsJs_UrlState_Test}, {@code ViewsJs_ColumnSearchPopover_Test}, {@code ViewsJs_ConfigChooser_Test}, and
44   * {@code ViewsJs_ConfigChooserA11y_Test} each prove one fix against a hand-rolled Node DOM shim. None of them can
45   * prove a real {@code KeyboardEvent} bubbles to a document-level handler, that {@code Enter} on a focused
46   * {@code <button>} triggers its native click, that {@code location.search} is decoded by a real browser's URL
47   * handling, or that a stored blob survives a real {@code page.reload()} against real {@code localStorage}.
48   *
49   * <h5 class='section'>Off by default:</h5>
50   * <p>
51   * Disabled unless the {@value #GATE} system property is set, which only the module's opt-in {@code js-tests}
52   * Maven profile does. Run with {@code mvn -Pjs-tests -f juneau-rest/juneau-console-views/pom.xml test}.
53   *
54   * <h5 class='section'>See Also:</h5>
55   * <ul>
56   * 	<li class='jc'>{@link ConfigPersistence_BrowserTest} &mdash; the sibling canary this one borrows its
57   * 		fixture-building and prober-running conventions from.
58   * </ul>
59   */

72  	private static String resource(String path) throws IOException {
73  		try (var in = ViewsMixin.class.getResourceAsStream(path)) {
74  			assertNotNull(in, () -> "missing classpath resource: " + path);
75  			return new String(in.readAllBytes(), UTF_8);
76  		}
77  	}

79  	@BeforeAll
80  	static void probe() throws Exception {
81  		var dir = Path.of(requiredProperty("juneau.jsTests.dir"));
82  		// The pom's js-tests profile provisions ONE harness property; this canary lives beside it in src/test/js,
83  		// so it is derived from that property's directory rather than adding a new pom property.
84  		var harness = Path.of(requiredProperty("juneau.jsTests.harness")).getParent().resolve("view-settings-a11y-browser.cjs");
85  
86  		// Dependency order per the module's own doc comment: renders -> views -> urlstate -> search -> pagestate ->
87  		// config.  No CSS is loaded: selectConfigTab toggles the native `hidden` property and the popover sets its
88  		// own inline `display`, so neither juneau-views.css nor juneau-config.css affects any assertion here.
89  		var fixture = "<!DOCTYPE html><html><head><meta charset=\"utf-8\"></head><body>\n<script>\n"
90  			+ resource(ViewsMixin.RENDERS_JS_RESOURCE)
91  			+ "\n</script>\n<script>\n"
92  			+ resource(ViewsMixin.VIEWS_JS_RESOURCE)
93  			+ "\n</script>\n<script>\n"
94  			+ resource(ViewsMixin.URLSTATE_JS_RESOURCE)
95  			+ "\n</script>\n<script>\n"
96  			+ resource(ViewsMixin.SEARCH_JS_RESOURCE)
97  			+ "\n</script>\n<script>\n"
98  			+ resource(ViewsMixin.PAGESTATE_JS_RESOURCE)
99  			+ "\n</script>\n<script>\n"
100 			+ resource(ViewsMixin.CONFIG_JS_RESOURCE)
101 			+ "\n</script></body></html>";
102 		var fixtureFile = Files.createDirectories(dir.resolve("fixtures")).resolve("view-settings-a11y.html");
103 		Files.write(fixtureFile, fixture.getBytes(UTF_8));
104 
105 		report = Json.to(run(dir, harness, fixtureFile), Map.class);
106 	}

108 	private static String requiredProperty(String name) {
109 		var v = System.getProperty(name);
110 		assertNotNull(v, () -> "-D" + name + " not set; the js-tests profile is responsible for providing it");
111 		return v;
112 	}

114 	/** Runs the prober, failing with its stderr attached (its exit code alone is not a diagnosis). */
115 	private static String run(Path dir, Path harness, Path fixture) throws Exception {
116 		var cmd = List.of(System.getProperty("juneau.jsTests.node", "node"), harness.toString(), fixture.toString());
117 		var stdout = dir.resolve("view-settings-a11y-stdout.json");
118 		var stderr = dir.resolve("view-settings-a11y-stderr.txt");
119 		var pb = new ProcessBuilder(cmd).redirectOutput(stdout.toFile()).redirectError(stderr.toFile());
120 		pb.environment().put("NODE_PATH", dir.resolve("node_modules").toString());
121 		pb.environment().put("PLAYWRIGHT_BROWSERS_PATH", requiredProperty("juneau.jsTests.browsers"));
122 
123 		var p = pb.start();
124 		if (!p.waitFor(3, TimeUnit.MINUTES)) {
125 			p.destroyForcibly();
126 			fail("prober did not finish within 3m; stderr:\n" + quietRead(stderr));
127 		}
128 		assertEquals(0, p.exitValue(), () -> "prober exited non-zero; stderr:\n" + quietRead(stderr));
129 		return Files.readString(stdout);
130 	}
131 
132 	private static String quietRead(Path p) {
133 		try {
134 			return Files.readString(p);
135 		} catch (IOException e) {
136 			return "<unreadable: " + e + ">";
137 		}
138 	}

156 	@Test void f1_01_theBackslashTabIdDecodesExactlyAndOnlyThatTabIsClicked() {
157 		// Exactly the tab whose dataset.juneauStripTab equals the decoded id is clicked, never 'plain'.
158 		// A literal backslash is BCT's escape character, so these two values are compared as plain strings.
159 		assertEquals("a\\", obj("f1").get("decodedTab"), report::toString);
160 		assertEquals(List.of("a\\"), obj("f1").get("clicked"), report::toString);
161 	}

165 	}

167 	@Test void f1_03_onAClientDslColumnTheFilterLandsInTheStoreAndPredicate() {
168 		// real juneau-search.js validates the restored $eq(OK); it installs the "juneau-dsl" predicate and
169 		// the per-table store, never native col.search().
170 		assertBean(report::toString, obj("f1"), "dslStoredExpr,dslNativeExpr,dslPredicateMatchesOk,dslPredicateRejectsOther",
