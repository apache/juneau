32  /**
33   * The async-job + SSE-streaming half of the module's <b>JavaScript-execution harness</b>: runs the REAL served
34   * {@code juneau-views.js} in a real headless browser and asserts the async-job behavior as a user would experience it
35   * &mdash; an async row action's start POST returns a job pointer, the row picks up the DISTINCT
36   * {@code data-juneau-job} affordance (never {@code data-juneau-inflight}, so the table KEEPS polling), streamed
37   * progress renders in a visible {@code textContent} banner, the single terminal {@code result} event settles the row
38   * to success / cancelled / cancelled-after-effect, a stream error settles it to a non-optimistic unknown, and Cancel
39   * issues a fail-closed CSRF POST to the job's cancel URL.
40   *
41   * <h5 class='section'>Why this exists (beyond the always-on source-shape test):</h5>
42   * <p>
43   * {@link ViewsJs_AsyncJobs_Test} proves the shipped script <i>contains</i> the async/streaming logic; it cannot
44   * prove that at runtime a running job leaves table polling live (the load-bearing polling fact), that progress and the
45   * terminal outcome actually paint, or that a stream error degrades to a visible unknown.  This canary drives the real
46   * runtime in Chromium against a controllable fake {@code EventSource}, so those user-visible facts are measured
47   * rather than inferred.
48   *
49   * <h5 class='section'>Off by default:</h5>
50   * <p>
51   * Disabled unless the {@value #GATE} system property is set, which only the module's opt-in {@code js-tests} Maven
52   * profile does.  It reuses that profile's provisioned Node + Playwright browser and derives its own prober
53   * ({@code async-job.cjs}) from the profile's {@code juneau.jsTests.harness} directory, so no pom change is needed.
54   *
55   * <h5 class='section'>See Also:</h5>
56   * <ul>
57   * 	<li class='jc'>{@link ModalResult_BrowserTest} &mdash; the sibling declarative-modal/typed-result canary.
58   * </ul>
59   */
60  @EnabledIfSystemProperty(named=AsyncJob_BrowserTest.GATE, matches="true",
61  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -f juneau-rest/juneau-console-views/pom.xml test`")
62  @SuppressWarnings({
63  	"unchecked" // Browser report values are cast to Map<String,Object>
64  })
65  class AsyncJob_BrowserTest extends TestBase {

72  	private static String resource(String path) throws IOException {
73  		try (var in = ViewsMixin.class.getResourceAsStream(path)) {
74  			assertNotNull(in, () -> "missing classpath resource: " + path);
75  			return new String(in.readAllBytes(), UTF_8);
76  		}
77  	}
78  
79  	@BeforeAll
80  	static void probe() throws Exception {
81  		var dir = Path.of(requiredProperty("juneau.jsTests.dir"));
82  		var harness = Path.of(requiredProperty("juneau.jsTests.harness")).getParent().resolve("async-job.cjs");
83  
84  		var fixture = "<!DOCTYPE html><html><head><meta charset=\"utf-8\"></head><body>\n<script>\n"
85  			+ resource(ViewsMixin.VIEWS_JS_RESOURCE)
86  			+ "\n</script></body></html>";
87  		var fixtureFile = Files.createDirectories(dir.resolve("fixtures")).resolve("async-job.html");
88  		Files.write(fixtureFile, fixture.getBytes(UTF_8));
89  
90  		report = Json.to(run(dir, harness, fixtureFile), Map.class);
91  	}
92  
93  	private static String requiredProperty(String name) {
94  		var v = System.getProperty(name);
95  		assertNotNull(v, () -> "-D" + name + " not set; the js-tests profile is responsible for providing it");
96  		return v;
97  	}
98  
99  	private static String run(Path dir, Path harness, Path fixture) throws Exception {
100 		var cmd = List.of(System.getProperty("juneau.jsTests.node", "node"), harness.toString(), fixture.toString());
101 		var stdout = dir.resolve("async-job-stdout.json");
102 		var stderr = dir.resolve("async-job-stderr.txt");
103 		var pb = new ProcessBuilder(cmd).redirectOutput(stdout.toFile()).redirectError(stderr.toFile());
104 		pb.environment().put("NODE_PATH", dir.resolve("node_modules").toString());
105 		pb.environment().put("PLAYWRIGHT_BROWSERS_PATH", requiredProperty("juneau.jsTests.browsers"));
106 
107 		var p = pb.start();
108 		if (!p.waitFor(3, TimeUnit.MINUTES)) {
109 			p.destroyForcibly();
110 			fail("prober did not finish within 3m; stderr:\n" + quietRead(stderr));
111 		}
112 		assertEquals(0, p.exitValue(), () -> "prober exited non-zero; stderr:\n" + quietRead(stderr));
113 		return Files.readString(stdout);
114 	}
115 
116 	private static String quietRead(Path p) {
117 		try {
118 			return Files.readString(p);
119 		} catch (IOException e) {
120 			return "<unreadable: " + e + ">";
121 		}
122 	}

125 		return (Map<String,Object>) report.get(key);

138 	//------------------------------------------------------------------------------------------------------------------
139 	// b) a running job uses the DISTINCT marker and does NOT freeze the table's polling
140 	//------------------------------------------------------------------------------------------------------------------
141 
142 	@Test void b01_runningJobUsesDistinctMarker_andDoesNotFreezePolling() {
143 		var r = sub("running");
144 		assertEquals(Boolean.TRUE, r.get("hasJobMarker"), () -> "row did not pick up data-juneau-job: " + report);
145 		// The whole point of the distinct marker: a job must NOT set the synchronous in-flight marker that freezes
146 		// polling.
147 		assertEquals(Boolean.FALSE, r.get("hasInflightMarker"), () -> "job wrongly set data-juneau-inflight: " + report);
148 		assertEquals(Boolean.FALSE, r.get("pollingFrozen"), () -> "polling was frozen during the job: " + report);
149 		assertEquals(Boolean.TRUE, r.get("progressVisible"), () -> "job progress banner not visible: " + report);
150 		assertEquals("Working\u2026", r.get("progressText"), () -> report.toString());
151 		assertEquals(Boolean.TRUE, r.get("triggerDisabled"), () -> "trigger not disabled while job runs: " + report);
152 		assertEquals(Boolean.TRUE, r.get("esOpened"), () -> "no EventSource opened: " + report);
153 		assertEquals("/juneau-jobs/cap/stream", r.get("esUrl"), () -> "EventSource opened on the wrong (capability) URL: " + report);
154 	}
155 
156 	@Test void b02_progressEventUpdatesBanner_andPollingStaysLive() {
157 		var p = sub("progress");
158 		assertEquals("step 2 of 3", p.get("text"), () -> "streamed progress did not render: " + report);
159 		assertEquals(Boolean.FALSE, p.get("pollingFrozenDuringProgress"), () -> "polling froze during progress: " + report);
160 		assertEquals(Boolean.FALSE, p.get("outcomeYet"), () -> "an outcome rendered before the terminal event: " + report);
161 	}

165 	//------------------------------------------------------------------------------------------------------------------

167 	@Test void c01_terminalResultSettlesRowToSuccess() {
168 		var s = sub("settledSuccess");
169 		assertEquals(Boolean.TRUE, s.get("esClosed"), () -> "EventSource not closed on settle: " + report);
170 		assertEquals(Boolean.TRUE, s.get("jobMarkerCleared"), () -> "data-juneau-job not cleared on settle: " + report);
171 		assertEquals(Boolean.TRUE, s.get("progressCleared"), () -> "progress banner not cleared on settle: " + report);
172 		assertEquals(Boolean.TRUE, s.get("triggerReEnabled"), () -> "trigger not re-enabled on settle: " + report);
173 		assertEquals(Boolean.FALSE, s.get("pollingFrozen"), () -> "polling frozen after settle: " + report);
174 		assertNotNull(s.get("mergedRow"), () -> "success did not re-render from the authoritative result row: " + report);
175 		var outcome = (Map<String,Object>) s.get("outcome");
176 		assertEquals(Boolean.TRUE, outcome.get("visible"), () -> "success banner not visible: " + report);
177 		assertEquals("success", outcome.get("state"), () -> report.toString());
178 		assertEquals("status", outcome.get("role"), () -> report.toString());
179 	}

185 	@Test void d01_cancelIssuesFailClosedCsrfPost() {
186 		var cancel = sub("cancel");
187 		assertEquals(Boolean.TRUE, cancel.get("cancelBtnVisible"), () -> "cancel button not visible: " + report);
188 		assertEquals(Boolean.TRUE, cancel.get("postIssued"), () -> "cancel did not issue a POST: " + report);
189 		assertEquals("/juneau-jobs/cap/cancel", cancel.get("url"), () -> "cancel POST hit the wrong URL: " + report);
190 		assertEquals("POST", cancel.get("method"), () -> "cancel must be a non-safe POST: " + report);
191 		assertEquals("tok-xyz", cancel.get("csrfHeader"), () -> "cancel POST did not carry the CSRF token: " + report);
192 	}

194 	@Test void d02_cancelledOutcomeArrivesOverStream() {
195 		var cancel = sub("cancel");
196 		var outcome = (Map<String,Object>) cancel.get("outcome");
197 		assertEquals("cancelled", outcome.get("state"), () -> report.toString());
198 		assertEquals(Boolean.TRUE, outcome.get("visible"), () -> "cancelled banner not visible: " + report);
199 		assertTrue(String.valueOf(outcome.get("text")).contains("Cancelled"), () -> report.toString());
200 		assertEquals(Boolean.TRUE, cancel.get("jobMarkerCleared"), () -> "job marker not cleared on cancel: " + report);
201 	}

203 	@Test void d03_cancelledAfterEffectIsADistinctOutcome() {
204 		// cancelled-after-effect must NOT be collapsed into plain cancelled - it is its own rendered outcome.
