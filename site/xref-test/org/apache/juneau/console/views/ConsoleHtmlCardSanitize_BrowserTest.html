20  import static org.junit.jupiter.api.Assertions.*;

28  /**
29   * An {@code html} card with {@code src} and {@code sanitize:true} paints the fetched fragment through the views
30   * attribute-allowlist sanitizer: script, {@code on*} handlers, unsafe URLs and other unlisted attributes are dropped
31   * while benign markup survives.  Without the flag the fragment is inserted as served.
32   */
33  @EnabledIfSystemProperty(named=ConsoleBrowserFixture.GATE, matches="true",
34  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -f juneau-rest/juneau-console-views/pom.xml test`")
35  class ConsoleHtmlCardSanitize_BrowserTest extends TestBase {
36  
37  	private static final String HOSTILE = """
38  		<script>window.__fired = 1;</script>\
39  		<img src="x" alt="pic" onerror="window.__fired = 1" onclick="window.__fired = 1" style="width:1px" class="c" id="evil">\
40  		<a href="javascript:window.__fired=1" onclick="window.__fired = 1">js-link</a>\
41  		<a href="data:text/html,x">data-link</a>\
42  		<a href="/ok" target="_blank" onmouseover="window.__fired = 1" title="t">ok-link</a>\
43  		<iframe src="/frame"></iframe>\
44  		<b>survivor</b>\
45  		<table><tr><td colspan="2" onclick="window.__fired = 1">cell</td></tr></table>
46  		""";
47  
48  	private static final String BENIGN = "<p>Hello <b>world</b></p><table><tr><td>1</td><td>2</td></tr></table><a href=\"/ok\">go</a>";
49  
50  	// A stray end tag must not close the sanitizer's wrapper and swallow what follows it.
51  	private static final String STRAY = "<p>one</p></div></body></html><p>two</p><b>after</b>";
52  
53  	private static final String CONTRACT_TRUE = card("\"sanitize\":true");
54  	private static final String CONTRACT_ABSENT = card("\"bare\":false");
55  	private static final String CONTRACT_FALSE = card("\"sanitize\":false");
56  
57  	private static String card(String extra) {
58  		return "{\"contractVersion\":\"1\",\"title\":\"S\",\"cards\":[{\"id\":\"h\",\"type\":\"html\",\"title\":\"Frag\",\"src\":\"/frag\","
59  			+ extra + "}]}";
60  	}
61  
62  	// console-shell.cjs reads window.__probe when the case is probed; a getter inspects the painted card at that moment.
63  	private static final String PROBE = """
64  		<script>
65  		Object.defineProperty(window, '__probe', { configurable: true, get: function () {
66  			var host = document.getElementById('h');
67  			var els = host ? Array.prototype.slice.call(host.querySelectorAll('*')).filter(function (e) { return !e.classList.contains('jc-card-title'); }) : [];
68  			return {
69  				fired: window.__fired || 0,
70  				tags: els.map(function (e) { return e.tagName; }),
71  				attrs: els.map(function (e) { return e.tagName + ':' + Array.prototype.map.call(e.attributes, function (a) { return a.name; }).sort().join(','); }),
72  				hrefs: els.filter(function (e) { return e.tagName === 'A'; }).map(function (e) { return e.getAttribute('href'); }),
73  				title: host && host.querySelector('.jc-card-title') ? host.querySelector('.jc-card-title').textContent : null
74  			};
75  		} });
76  		</script>
77  		""";
78  
79  	private static Map<String,Map<String,Object>> report;
80  	private static Map<String,Map<String,Object>> benignReport;
81  	private static Map<String,Map<String,Object>> lateReport;
82  	private static Map<String,Map<String,Object>> strayReport;
83  
84  	@BeforeAll
85  	static void load() throws Exception {
86  		var views = viewsPack();
87  		report = ConsoleBrowserFixture.create("html-sanitize")
88  			.asset("/frag", HOSTILE, "text/html")
89  			.page("on", ORIGIN + "/on", contractPage(CONTRACT_TRUE, "", PROBE, views)).waitFor("#h b")
90  			.page("absent", ORIGIN + "/absent", contractPage(CONTRACT_ABSENT, "", PROBE, views)).waitFor("#h b")
91  			.page("false", ORIGIN + "/false", contractPage(CONTRACT_FALSE, "", PROBE, views)).waitFor("#h b")
92  			.page("on-no-views", ORIGIN + "/on-no-views", contractPage(CONTRACT_TRUE, "", PROBE, ""),
93  				"fragment", "#h b").waitFor("#h b, .jc-console-error-item")
94  			.run();
95  		benignReport = ConsoleBrowserFixture.create("html-sanitize-benign")
96  			.asset("/frag", BENIGN, "text/html")
97  			.page("on", ORIGIN + "/on", contractPage(CONTRACT_TRUE, "", PROBE, views)).waitFor("#h table")
98  			.run();
99  		// The views toolkit arrives as a slow external script, so the parser is still blocked (readyState "loading") when
100 		// the fragment fetch resolves: the sanitizer is not there yet, but will be by DOMContentLoaded.
101 		var lateViews = views.replace("<script>\n", "").replace("\n</script>\n", "\n;\n");
102 		lateReport = ConsoleBrowserFixture.create("html-sanitize-late")
103 			.asset("/frag", BENIGN, "text/html")
104 			.delayedAsset("/late-views.js", lateViews, "text/javascript", 1500)
105 			.page("on", ORIGIN + "/on", contractPage(CONTRACT_TRUE, "", PROBE, "<script src=\"/late-views.js\"></script>"))
106 				.waitFor("#h table, .jc-console-error-item")
107 			.run();
108 		strayReport = ConsoleBrowserFixture.create("html-sanitize-stray")
109 			.asset("/frag", STRAY, "text/html")
110 			.page("on", ORIGIN + "/on", contractPage(CONTRACT_TRUE, "", PROBE, views)).waitFor("#h p")
111 			.run();
112 	}

114 	@Test void a01_sanitizeTrue_dropsScriptHandlersAndUnsafeUrls() {
115 		var r = report.get("on");
116 		assertNoShellErrors(r);
117 		var p = probe(r);
118 		assertEquals(0, ((Number)p.get("fired")).intValue(), () -> "nothing executed: " + p);
119 		assertFalse(((List<?>)p.get("tags")).contains("SCRIPT"), () -> "script dropped: " + p);
120 		assertFalse(((List<?>)p.get("tags")).contains("IFRAME"), () -> "iframe dropped: " + p);
121 		for (var a : (List<?>)p.get("attrs"))
122 			assertFalse(String.valueOf(a).matches(".*\\b(on\\w+|style|class|id|target)\\b.*"), () -> "unlisted attribute survived: " + p);
123 		assertEquals(List.of("/ok"), ((List<?>)p.get("hrefs")).stream().filter(Objects::nonNull).toList(), () -> "only the safe href survives: " + p);
124 	}
125 
126 	@Test void a02_sanitizeTrue_keepsBenignMarkupAndTheCardTitle() {
127 		var r = report.get("on");
128 		var p = probe(r);
129 		var tags = (List<?>)p.get("tags");
130 		assertTrue(tags.contains("B") && tags.contains("TABLE") && tags.contains("TD"), () -> "benign markup survives: " + p);
131 		assertTrue(((List<?>)p.get("attrs")).contains("TD:colspan"), () -> "allowlisted colspan survives: " + p);
132 		assertTrue(((List<?>)p.get("attrs")).contains("A:href,title"), () -> "allowlisted href/title survive: " + p);
133 		assertEquals("Frag", p.get("title"), () -> "sanitized paint must not clear the card title: " + p);
134 		var b = probe(benignReport.get("on"));
135 		assertEquals(List.of("P", "B", "TABLE", "TBODY", "TR", "TD", "TD", "A"), b.get("tags"), b::toString);
136 	}
137 
138 	@Test void a03_flagAbsent_fragmentIsInsertedAsServed() {
139 		for (var name : List.of("absent", "false")) {
140 			var r = report.get(name);
141 			assertNoShellErrors(r);
142 			var p = probe(r);
143 			assertTrue(((List<?>)p.get("tags")).contains("SCRIPT"), () -> name + ": script element kept: " + p);
144 			assertTrue(((List<?>)p.get("tags")).contains("IFRAME"), () -> name + ": iframe kept: " + p);
145 			assertTrue(((List<?>)p.get("attrs")).stream().anyMatch(a -> String.valueOf(a).startsWith("IMG:") && String.valueOf(a).contains("onerror")),
146 				() -> name + ": onerror kept: " + p);
147 		}
148 	}
149 
150 	@Test void a04_sanitizeTrue_withoutViews_failsClosedWithE_JS_8() {
151 		var r = report.get("on-no-views");
152 		assertNull(queries(r).get("fragment"), () -> "nothing painted unsanitized: " + r);
153 		assertFailure(r, "E-JS-8", "card 'h' (type 'html') handler threw: 'html card sanitize:true needs JuneauViews; load the views toolkit'");
154 	}
155 
156 	@Test void a05_viewsArrivingAfterTheFragment_stillPaintsSanitized() {
157 		var r = lateReport.get("on");
158 		assertNoShellErrors(r);
159 		var p = probe(r);
160 		assertEquals(List.of("P", "B", "TABLE", "TBODY", "TR", "TD", "TD", "A"), p.get("tags"), () -> "painted once views loaded: " + p);
161 	}
162 
163 	@Test void a06_strayEndTag_doesNotDropTheRestOfTheFragment() {
164 		var r = strayReport.get("on");
165 		assertNoShellErrors(r);
166 		var p = probe(r);
167 		assertEquals(List.of("P", "P", "B"), p.get("tags"), () -> "everything after the stray end tag survives: " + p);
168 	}
