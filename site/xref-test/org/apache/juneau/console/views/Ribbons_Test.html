29  	private static List<Object> ids(List<RibbonItem> items) {
30  		return items.stream().map(i -> i.toMap().get("id")).toList();
31  	}
32  
33  	private static List<Object> groups(List<RibbonItem> items) {
34  		return items.stream().map(i -> i.toMap().get("group")).toList();
35  	}
36  
37  	private static RibbonItem find(List<RibbonItem> items, String id) {
38  		return items.stream().filter(i -> id.equals(i.toMap().get("id"))).findFirst().orElse(null);
39  	}

42  		var items = Ribbons.standard().build();

44  		assertEquals(List.of("search", "export", "export", "viewing", "viewing", "viewing", "page"), groups(items));

48  		var items = Ribbons.standard()
49  			.create(RibbonItem.dialog("add-pet").title("Add pet"))
50  			.filter(RibbonItem.optionGroup("status", RibbonItem.option("a").title("A")))
51  			.build();
52  		assertEquals(List.of("search", "add-pet", "status", "export", "link", "collapse", "columns", "reset", "refresh"), ids(items));
53  		assertEquals("entity", items.get(1).toMap().get("group"));
54  		assertEquals("filter", items.get(2).toMap().get("group"));
55  	}

58  		var items = Ribbons.standard()
59  			.create(RibbonItem.dialog("c1")).create(RibbonItem.dialog("c2"))
60  			.filter(RibbonItem.optionGroup("f1", RibbonItem.option("a"))).filter(RibbonItem.optionGroup("f2", RibbonItem.option("b")))

130 	@Test void z01_preset_order_matches_the_runtime_canonical_group_order() throws Exception {
131 		String js;
132 		try (var in = ViewsMixin.class.getResourceAsStream(ViewsMixin.RIBBON_JS_RESOURCE)) {
133 			js = new String(in.readAllBytes(), java.nio.charset.StandardCharsets.UTF_8);
134 		}
135 		var m = java.util.regex.Pattern.compile("const CANONICAL_GROUPS = \\[([^\\]]*)\\]").matcher(js);
136 		assertTrue(m.find());
137 		var canonical = Arrays.stream(m.group(1).split(",")).map(x -> x.trim().replace("\"", "")).toList();
138 		var items = Ribbons.standard().create(RibbonItem.dialog("c")).filter(RibbonItem.optionGroup("f", RibbonItem.option("a")))
139 			.export("excel").build();
140 		int last = -1;
141 		for (var g : groups(items)) {
142 			int r = canonical.indexOf(g);
143 			assertTrue(r >= 0, "group not canonical: " + g);
144 			assertTrue(r >= last, "preset order disagrees with the runtime at group " + g);
145 			last = r;
146 		}
147 	}
