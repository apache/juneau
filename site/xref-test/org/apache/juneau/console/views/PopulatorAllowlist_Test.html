70  	/** Locates {@code PopulatorAllowlist.java}'s own source on disk, the same way {@code locateHarness()} does in
71  	 * {@link ViewsJs_ContractVersion_Test}. */
72  	private static Path locateSource() {
73  		var basedir = System.getProperty("basedir");
74  		if (basedir != null) {
75  			var p = Path.of(basedir, "src/main/java/org/apache/juneau/console/views/PopulatorAllowlist.java");
76  			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
77  		}
78  		for (var rel : List.of(
79  				"src/main/java/org/apache/juneau/console/views/PopulatorAllowlist.java",
80  				"juneau-rest/juneau-console-views/src/main/java/org/apache/juneau/console/views/PopulatorAllowlist.java")) {
81  			var p = Path.of(rel);
82  			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
83  		}
84  		return null;
85  	}

87  	/**
88  	 * Guards the honest-limit warning in {@link PopulatorAllowlist}'s class Javadoc against silent deletion by a
89  	 * future editor.  Unlike {@link SinkRenderAllowlist}, a populator allowlist controls only which name is
90  	 * resolved and nothing about what the resolved populator then does &mdash; a fact that is easy to lose once the
91  	 * class exists and looks, on its shape alone, exactly like a real security boundary.  This test reads the
92  	 * class's own source file off disk and fails if either load-bearing phrase is missing, so deleting the warning
93  	 * (rather than merely editing code around it) breaks the build.
94  	 */
95  	@Test void a08_classJavadoc_keepsTheHonestLimitWarning() throws IOException {
96  		var src = locateSource();
97  		assertNotNull(src, "could not locate PopulatorAllowlist.java on disk to check its class Javadoc");
98  		var body = Files.readString(src, StandardCharsets.UTF_8);
99  		var classStart = body.indexOf("public final class PopulatorAllowlist");
100 		assertTrue(classStart > 0, () -> "could not find 'public final class PopulatorAllowlist' in:\n" + src);
101 		var javadoc = body.substring(0, classStart).toLowerCase(Locale.ROOT);
102 		assertTrue(javadoc.contains("authoring-discipline") || javadoc.contains("authoring discipline"),
103 			() -> "PopulatorAllowlist's class Javadoc must keep stating that this allowlist is an authoring-discipline "
104 				+ "control, not a security boundary. A future edit removed that language - restore it so nobody "
105 				+ "transfers SinkRenderAllowlist's security value onto this class by mistake.");
106 		assertTrue(javadoc.contains("not a security boundary"),
107 			() -> "PopulatorAllowlist's class Javadoc must keep stating that this allowlist is NOT a security "
108 				+ "boundary. A future edit removed that language - restore it so nobody transfers "
109 				+ "SinkRenderAllowlist's security value onto this class by mistake.");
110 	}
