17  package org.apache.juneau.console.views;
18  
19  import static org.apache.juneau.console.views.ConsoleBrowserFixture.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.util.*;
23  
24  import org.apache.juneau.*;
25  import org.junit.jupiter.api.*;
26  import org.junit.jupiter.api.condition.*;

28  /**
29   * Header and footer slots render in place; an inline script inside a template runs exactly once at mount; a JSON
30   * island inside a template is readable by a later script and by a DOMContentLoaded listener; and a {@code replace}-
31   * kind slot swaps the entire stock element rather than filling it.
32   */
33  @EnabledIfSystemProperty(named=ConsoleBrowserFixture.GATE, matches="true",
34  	disabledReason="JS-execution harness is opt-in; run with `mvn -Pjs-tests -f juneau-rest/juneau-console-views/pom.xml test`")
35  class ConsoleSlots_BrowserTest extends TestBase {
36  
37  	private static final String SLOTS = """
38  		{"contractVersion":"1","title":"Slots",
39  		 "header":{"title":"T","slots":{"brand":"header.brand","actions":"header.actions","banner":"header.banner"}},
40  		 "footer":{"slots":{"content":"footer.content"}},
41  		 "cards":[{"id":"jc-seg-1","type":"html","template":"jc-seg-1","bare":true}]}
42  		""";
43  
44  	private static final String SLOT_TEMPLATES = """
45  		<template data-slot="header.brand"><span class="probe-brand">B</span></template>
46  		<template data-slot="header.actions"><button type="button" class="probe-action">A</button></template>
47  		<template data-slot="header.banner"><div class="probe-banner">Demo</div></template>
48  		<template data-slot="footer.content"><em class="probe-foot">F</em></template>
49  		<template data-card="jc-seg-1"><div id="seg">seg</div>
50  		<script>window.__probe = window.__probe || {}; window.__probe.runs = (window.__probe.runs || 0) + 1;</script>
51  		<script type="application/json" id="island">{"k":42}</script></template>
52  		""";
53  
54  	private static final String AFTER = """
55  		<script>
56  		window.__probe = window.__probe || {};
57  		window.__probe.templatesLeft = document.querySelectorAll('template').length;
58  		window.__probe.islandLater = JSON.parse(document.getElementById('island').textContent).k;
59  		document.addEventListener('DOMContentLoaded', function () {
60  			window.__probe.islandReady = JSON.parse(document.getElementById('island').textContent).k;
61  		});
62  		</script>
63  		""";
64  
65  	private static final String REPLACE = """
66  		{"contractVersion":"1","title":"Replace","header":{"title":"ignored","slots":{"replace":"header.replace"}},"cards":[]}
67  		""";
68  
69  	private static final String CHROME_ON = """
70  		{"contractVersion":"1","title":"Chrome","header":{"title":"T","chrome":true,"slots":{"banner":"header.banner"}},
71  		 "nav":[{"id":"home","label":"Home","href":"/chrome"}],"activeNav":["home"],"cards":[]}
72  		""";
73  
74  	private static final String CHROME_OFF = """
75  		{"contractVersion":"1","title":"NoChrome","header":{"title":"T","slots":{"banner":"header.banner"}},
76  		 "nav":[{"id":"home","label":"Home","href":"/nochrome"}],"activeNav":["home"],"cards":[]}
77  		""";
78  
79  	private static final String BANNER_ONLY = """
80  		<template data-slot="header.banner"><div class="probe-banner">Demo</div></template>
81  		""";
82  
83  	private static Map<String,Map<String,Object>> report;
84  
85  	@BeforeAll
86  	static void probe() throws Exception {
87  		report = ConsoleBrowserFixture.create("slots")
88  			.page("slots", ORIGIN + "/slots", contractPage(SLOTS, SLOT_TEMPLATES, "", AFTER),
89  				"brand", "header.jc-header > .jc-brand > .probe-brand",
90  				"actions", "header.jc-header > .jc-header-actions > .probe-action",
91  				"banner", "body > .probe-banner",
92  				"footer", "footer.jc-page-footer > .probe-foot",
93  				"order", "header.jc-header + .probe-banner + main.jc-main + footer.jc-page-footer",
94  				"segment", "main.jc-main > #seg")
95  			.page("chromeOn", ORIGIN + "/chrome", contractPage(CHROME_ON, BANNER_ONLY),
96  				"order", "body > div.jc-chrome > header.jc-header + .probe-banner + nav.juneau-page-nav",
97  				"strayBanner", "body > .probe-banner")
98  			.page("chromeOff", ORIGIN + "/nochrome", contractPage(CHROME_OFF, BANNER_ONLY),
99  				"order", "body > header.jc-header + .probe-banner + nav.juneau-page-nav")
100 			.page("replace", ORIGIN + "/replace",
101 				contractPage(REPLACE, "<template data-slot=\"header.replace\"><header class=\"my-header\">WHOLE</header></template>"),
102 				"replaced", "body > header.my-header",
103 				"stock", ".jc-header")
104 			.run();
105 	}
106 
107 	@Test void a01_slotsRenderInPlace() {
108 		var r = report.get("slots");
109 		assertNoShellErrors(r);
110 		var q = queries(r);
111 		assertEquals("B", q.get("brand"));
112 		assertEquals("A", q.get("actions"));
113 		assertEquals("Demo", q.get("banner"));
114 		assertEquals("F", q.get("footer"));
115 		assertEquals("F", q.get("order"), "header, banner, main, footer must be siblings in that order");
116 		assertEquals("seg", q.get("segment"), "a bare html card is inserted without a .jc-card wrapper");
117 	}
118 
119 	@Test void a01b_bannerSitsBetweenHeaderAndNav_insideChromeWhenOn() {
120 		var on = report.get("chromeOn");
121 		assertNoShellErrors(on);
122 		assertEquals("Home", queries(on).get("order"),
123 			"chrome on: .jc-chrome must hold header, banner, nav as consecutive siblings in that order");
124 		assertNull(queries(on).get("strayBanner"), "chrome on: the banner must not be a direct child of <body>");
125 		var off = report.get("chromeOff");
126 		assertNoShellErrors(off);
127 		assertEquals("Home", queries(off).get("order"), "chrome off: header, banner, nav are body siblings in that order");
128 	}
129 
130 	@Test void a02_inlineScriptRunsOnce_andTemplatesAreConsumed() {
131 		var p = ConsoleBrowserFixture.probe(report.get("slots"));
132 		assertEquals(1, ((Number)p.get("runs")).intValue());
133 		assertEquals(0, ((Number)p.get("templatesLeft")).intValue());
134 	}
135 
136 	@Test void a03_jsonIslandIsReadableLaterAndOnReady() {
137 		var p = ConsoleBrowserFixture.probe(report.get("slots"));
138 		assertEquals(42, ((Number)p.get("islandLater")).intValue());
139 		assertEquals(42, ((Number)p.get("islandReady")).intValue());
140 	}
141 
142 	@Test void a04_replaceSlot_replacesTheWholeHeader() {
143 		var r = report.get("replace");
144 		assertNoShellErrors(r);
145 		assertEquals("WHOLE", queries(r).get("replaced"));
146 		assertNull(queries(r).get("stock"));
147 	}
148 }
