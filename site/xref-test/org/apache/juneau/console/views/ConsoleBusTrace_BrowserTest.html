20  import static org.junit.jupiter.api.Assertions.*;

22  import java.util.*;
23  
24  import org.apache.juneau.*;

28  /**
29   * The bus trace flag in real Chromium: {@code ?juneau-bus-trace} logs each publish as a {@code [bus]} console.debug line, and the
30   * history ring records the master/detail traffic, including a suppressed unchanged publish, with or without the flag.
31   */
32  @EnabledIfSystemProperty(named=ConsoleBusBrowserSupport.GATE, matches="true", disabledReason=ConsoleBusBrowserSupport.DISABLED)
33  class ConsoleBusTrace_BrowserTest extends TestBase {
34  
35  	private static Map<String,Object> traced;
36  	private static Map<String,Object> quiet;
37  
38  	@BeforeAll
39  	static void probe() throws Exception {
40  		traced = run("trace", "master-detail.json", true, "{declare:[{topic:'app.region-picked',retain:true}]}", "?juneau-bus-trace");
41  		quiet = run("trace", "master-detail.json", true, "{declare:[{topic:'app.region-picked',retain:true}]}", "");
42  	}
43  
44  	private static List<String> topics(Map<String,Object> report) {
45  		return list(report.get("history")).stream().map(h -> map(h).get("topic").toString()).toList();
46  	}
47  
48  	/**
49  	 * Sorts each run of adjacent {@code redraw:} topics.  Tables load over independent requests, so which one finishes
50  	 * first (and publishes its redraw first) is a network race, not part of the contract; every other topic keeps its place.
51  	 */
52  	private static List<String> redrawRunsUnordered(List<String> topics) {
53  		var out = new ArrayList<String>();
54  		for (var i = 0; i < topics.size();) {
55  			if (! topics.get(i).startsWith("redraw:")) {
56  				out.add(topics.get(i++));
57  				continue;
58  			}
59  			var run = new ArrayList<String>();
60  			while (i < topics.size() && topics.get(i).startsWith("redraw:"))
61  				run.add(topics.get(i++));
62  			run.sort(null);
63  			out.addAll(run);
64  		}
65  		return out;
66  	}
67  
68  	@Test void a01_theFlagLogsEveryPublishAsABusDebugLine() {
69  		var lines = list(traced.get("busDebug")).stream().map(Object::toString).toList();
70  		assertFalse(lines.isEmpty());
71  		assertTrue(lines.stream().allMatch(l -> l.startsWith("[bus] ")), lines::toString);
72  		for (var t : List.of("card:changes", "filter:changes", "selection:changes", "redraw:tasks"))
73  			assertTrue(lines.stream().anyMatch(l -> l.contains(" " + t + " ")), () -> t + " not traced in " + lines);
74  		assertTrue(lines.stream().anyMatch(l -> l.contains("suppressed (unchanged)")), lines::toString);
75  		assertEquals(List.of(), list(traced.get("jsFailures")));
76  	}
77  
78  	@Test void a02_noFlagMeansNoDebugLinesButTheHistoryStillFills() {
79  		assertEquals(List.of(), list(quiet.get("busDebug")));
80  		var t = topics(quiet);
81  		assertTrue(t.contains("selection:changes") && t.contains("redraw:tasks"), t::toString);
82  		assertTrue(list(quiet.get("history")).stream().anyMatch(h -> Boolean.TRUE.equals(map(h).get("suppressed"))), () -> quiet.toString());
83  		assertEquals(redrawRunsUnordered(topics(traced)), redrawRunsUnordered(t), "the flag must not change what is published");
84  	}
85  }
