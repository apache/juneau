83  	private static final String PROBE = """
84  		<script>
85  		(function () {
86  			var PROPS = ['background-color', 'color', 'border-top-color', 'border-bottom-color', 'border-top-width',
87  				'border-bottom-width', 'border-top-left-radius', 'border-bottom-left-radius', 'font-weight', 'height',
88  				'width', 'justify-self', 'justify-content', 'column-gap', 'padding-top', 'padding-left', 'margin-bottom', 'flex-grow', 'box-shadow'];
89  			var rec = {};
90  			function snap(id) {
91  				var el = document.getElementById(id);
92  				var cs = getComputedStyle(el);
93  				var o = {};
94  				PROPS.forEach(function (p) { o[p] = cs.getPropertyValue(p); });
95  				var r = el.getBoundingClientRect();
96  				o.left = r.left; o.right = r.right;
97  				return o;
98  			}
99  			function after(id, prop) { return getComputedStyle(document.getElementById(id), '::after').getPropertyValue(prop); }
100 			window.__rec = function (key, id) { rec[key] = snap(id); };
101 			window.__probe = { toJSON: function () {
102 				var tok = {};
103 				var t = document.getElementById('tokprobe');
104 				%s.forEach(function (n) { t.style.color = 'var(' + n + ')'; tok[n] = getComputedStyle(t).color; });
105 				var rest = {};
106 				['hd','dx','ps','p1','p2','fa','cb','tg','atg','ft','fc','fk','fs','f1','f2','f3','rail','m1','m2','m3','wft','wc','wsp','wb','wn']
107 					.forEach(function (id) { rest[id] = snap(id); });
108 				var bars = { s1: after('s1', 'border-top-color'), s2: after('s2', 'border-top-color'), s3: after('s3', 'content') };
109 				return { tok: tok, rest: rest, hover: rec, bars: bars };
110 			} };
111 		})();
112 		</script>

117 	private static String caseName(String palette, String scheme, String order) {
118 		return palette + "-" + scheme + "-" + order;
119 	}

121 	private static String page(String palette, String scheme, String order) {
122 		var chrome = "<link rel=\"stylesheet\" href=\"" + CHROME_CSS + "\">\n";
123 		var views = "<link rel=\"stylesheet\" href=\"" + VIEWS_CSS + "\">\n";
124 		var themes = "<link rel=\"stylesheet\" href=\"" + DARK_CSS + "\">\n<link rel=\"stylesheet\" href=\"/juneau-console/juneau-theme-" + palette + ".css\">\n";
125 		var attrs = "data-jc-scheme=\"" + scheme + "\" data-jc-scheme-pin=\"" + scheme + "\"";
126 		var tokens = new StringBuilder("[");
127 		for (var t : TOKENS)
128 			tokens.append(tokens.length() == 1 ? "" : ",").append('\'').append(t).append('\'');
129 		tokens.append(']');
130 		return "<!DOCTYPE html>\n<html lang=\"en\" " + attrs + "><head><meta charset=\"utf-8\"><link rel=\"icon\" href=\"data:,\">\n"
131 			+ (order.equals("cv") ? chrome + themes + views : views + chrome + themes)
132 			// An app's own toggle size, at single-class specificity and loaded last (as SSC's console.css does).
133 			+ "<style>.app-toggle { width: 50px; }</style>\n"
134 			+ "</head><body>\n" + BODY + PROBE.formatted(tokens) + "</body></html>\n";
135 	}

137 	@BeforeAll
138 	static void run() throws Exception {
139 		var f = ConsoleBrowserFixture.create("dialog-frame");
140 		for (var p : PALETTES)
141 			f.asset("/juneau-console/juneau-theme-" + p + ".css", resource("/org/apache/juneau/console/juneau-theme-" + p + ".css"), "text/css");
142 		for (var p : PALETTES)
143 			for (var s : SCHEMES)
144 				for (var o : ORDERS) {
145 					f.page(caseName(p, s, o), ORIGIN + "/" + caseName(p, s, o), page(p, s, o));
146 					for (var id : HOVERED)
147 						f.hover("#" + id).eval("window.__rec('" + id + "', '" + id + "')");
148 					// Park the pointer off every control so the rest snapshot is not hovered.
149 					f.hover("#park");
150 				}
151 		report = f.run();
152 	}

154 	private record Probe(String name, Map<String,String> tok, Map<String,Map<String,Object>> rest, Map<String,Map<String,Object>> hover, Map<String,String> bars) {
155 		String tok(String n) { return tok.get(n); }
156 		String rest(String id, String prop) { return String.valueOf(rest.get(id).get(prop)); }
157 		double x(String id, String edge) { return ((Number)rest.get(id).get(edge)).doubleValue(); }
158 		String hover(String id, String prop) { return String.valueOf(hover.get(id).get(prop)); }
159 	}
160 
161 	private static List<Probe> probes() {
162 		var out = new ArrayList<Probe>();
163 		for (var p : PALETTES)
164 			for (var s : SCHEMES)
165 				for (var o : ORDERS) {
166 					var name = caseName(p, s, o);
167 					var r = report.get(name);
168 					assertNotNull(r, () -> "no report for " + name);
169 					assertEquals(List.of(), r.get("pageErrors"), () -> name + ": page errors");
170 					var pr = probe(r);
171 					out.add(new Probe(name, (Map<String,String>)pr.get("tok"), (Map<String,Map<String,Object>>)pr.get("rest"),
172 						(Map<String,Map<String,Object>>)pr.get("hover"), (Map<String,String>)pr.get("bars")));
173 				}
174 		return out;
175 	}

177 	private static void eq(Probe p, Object expected, Object actual, String what) {
178 		assertEquals(expected, actual, () -> p.name() + ": " + what);
179 	}
