32  /**
33   * Regression coverage: a detail-header/detail-panel ActionBar ActionRef click
34   * ({@code handleDetailActionRefClick}, {@code juneau-views.js}) must honor a {@code present=dialog} action the
35   * SAME way the row-action menu / cell-pill paths already do ({@code isDialogAction(action) ?
36   * openActionDialog(...) : submitRowAction(...)} - see {@code buildRowActionMenu} / {@code activatePillAction}).
37   *
38   * <p>Before the fix, the detail path called {@code submitRowAction(action, table, parentTr, ctx)} directly and
39   * unconditionally, so a dialog-declared detail action never opened its confirmation/form, never attached
40   * {@code extra.targetId}, and never carried a server-minted idempotency key into the submit -
41   * {@code submitActionDialog} (the ONLY place that attaches them) was skipped entirely.
42   *
43   * <p>Source-shape (always runs, no Node required) plus a behavioral Node harness
44   * ({@code detail-action-dialog-routing.cjs}) that runs when {@code node} is on {@code PATH} (skipped otherwise -
45   * no {@code -Pjs-tests} required, mirroring {@code ViewsJs_RowDetail_Test} / {@code ViewsJs_ContractVersion_Test}).
46   */
47  class ViewsJs_DetailActionDialogRouting_Test extends TestBase {

63  	/** Extracts a named function's body: from `function <name>(` to the next top-level `\n\t}`. */
64  	private static String functionBody(String body, String signature) {
65  		var start = body.indexOf(signature);
66  		assertTrue(start >= 0, () -> "'" + signature + "' not found:\n" + body);
67  		var end = body.indexOf("\n\t}", start);
68  		return body.substring(start, end < 0 ? body.length() : end);
69  	}

71  	@Test void a01_sourceShape_detailPathRoutesDialogDeclaredActionsThroughOpenActionDialog() throws Exception {
72  		var js = viewsJs();
73  		var fnBody = functionBody(js, "function handleDetailActionRefClick(");
74  		// The detail path takes the shared row-action dispatch the menu and pill paths take.
75  		assertTrue(fnBody.contains("dispatchRowAction(action, table, parentTr, ctx)"), fnBody);
76  		assertTrue(fnBody.contains("panel?._juneauParentTr"), fnBody);
77  		var dispatch = functionBody(js, "function dispatchRowAction(");
78  		assertTrue(dispatch.contains("if (isDialogAction(action)) return openActionDialog(action, table, tr, ctx);"), dispatch);
79  		// The non-dialog, no-confirm branch keeps going through the pre-existing direct submit, unchanged.
80  		assertTrue(dispatch.contains("return submitRowAction(action, table, tr, ctx, extra);"), dispatch);
81  	}

83  	/**
84  	 * The routing fix above only ROUTES INTO {@code submitActionDialog} - it changes nothing inside it.  What has to
85  	 * hold is the seam itself: the submit is assembled into one {@code extra} envelope and handed to the shared
86  	 * {@code submitRowAction}, so every activation surface settles through the same already-reviewed path.
87  	 *
88  	 * <p>The {@code targetId} computation is the one part of this body that is NOT frozen here - it is owned by the
89  	 * row-less dialog seam, whose {@code ViewsJs_SelfTargetedIdempotency_Test} pins its exact shape and, more to the
90  	 * point, drives all three of its branches behaviorally.  Duplicating that pin here would just mean two places to
91  	 * update; what this test cares about is that a target is still computed and attached at all.
92  	 */
93  	@Test void a02_submitActionDialogIsStillTheOneSubmitSeam() throws Exception {
94  		var fnBody = functionBody(viewsJs(), "function submitActionDialog(");
95  		assertTrue(fnBody.contains("if (targetId != null) extra.targetId = targetId;"), fnBody);
96  		assertTrue(fnBody.contains("if (modal?.idempotencyKey != null) extra.idempotencyKey = modal.idempotencyKey;"), fnBody);
97  		assertTrue(fnBody.contains("submitRowAction(action, table, tr, ctx, extra);"), fnBody);
98  	}

102 	@BeforeAll
103 	static void probeIfNodeAvailable() throws Exception {
104 		if (!nodeAvailable())
105 			return;
106 		var harness = locateHarness();
107 		if (harness == null)
108 			return;
109 		var viewsFile = Files.createTempFile("juneau-views-", ".js");
110 		var rendersFile = Files.createTempFile("juneau-renders-", ".js");
111 		try {
112 			Files.writeString(viewsFile, viewsJs(), UTF_8);
113 			Files.writeString(rendersFile, rendersJs(), UTF_8);
114 			report = Json.to(runNode(harness, rendersFile, viewsFile), Map.class);
115 		} finally {
116 			Files.deleteIfExists(viewsFile);
117 			Files.deleteIfExists(rendersFile);
118 		}
119 	}

121 	private static boolean nodeAvailable() {
122 		try {
123 			var p = new ProcessBuilder("node", "--version").redirectErrorStream(true).start();
124 			if (!p.waitFor(5, TimeUnit.SECONDS)) {
125 				p.destroyForcibly();
126 				return false;
127 			}
128 			return p.exitValue() == 0;
129 		} catch (Exception e) {
130 			return false;
131 		}
132 	}

134 	private static Path locateHarness() {
135 		var basedir = System.getProperty("basedir");
136 		if (basedir != null) {
137 			var p = Path.of(basedir, "src/test/js/detail-action-dialog-routing.cjs");
138 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
139 		}
140 		for (var rel : List.of(
141 			"src/test/js/detail-action-dialog-routing.cjs",
142 			"juneau-rest/juneau-console-views/src/test/js/detail-action-dialog-routing.cjs"
143 		)) {
144 			var p = Path.of(rel);
145 			if (Files.isRegularFile(p)) return p.toAbsolutePath().normalize();
146 		}
147 		return null;
148 	}

150 	private static String runNode(Path harness, Path rendersJs, Path viewsJs) throws Exception {
151 		var stdout = Files.createTempFile("detail-action-dialog-routing-stdout-", ".json");
152 		var stderr = Files.createTempFile("detail-action-dialog-routing-stderr-", ".txt");
153 		try {
154 			var pb = new ProcessBuilder(List.of("node", harness.toString(), rendersJs.toString(), viewsJs.toString()))
155 				.redirectOutput(stdout.toFile())
156 				.redirectError(stderr.toFile());
157 			var p = pb.start();
158 			if (!p.waitFor(30, TimeUnit.SECONDS)) {
159 				p.destroyForcibly();
160 				fail("detail-action-dialog-routing.cjs did not finish within 30s; stderr:\n" + quietRead(stderr));
161 			}
162 			if (p.exitValue() != 0)
163 				fail("detail-action-dialog-routing.cjs exited " + p.exitValue() + "; stderr:\n" + quietRead(stderr)
164 					+ "\nstdout:\n" + quietRead(stdout));
165 			return Files.readString(stdout, UTF_8);
166 		} finally {
167 			Files.deleteIfExists(stdout);
168 			Files.deleteIfExists(stderr);
169 		}
170 	}
171 
172 	private static String quietRead(Path p) {
173 		try { return Files.readString(p, UTF_8); }
174 		catch (IOException e) { return "(unreadable: " + e.getMessage() + ")"; }
175 	}
