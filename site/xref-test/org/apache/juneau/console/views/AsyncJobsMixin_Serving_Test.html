37  /**
38   * Serving test for {@link AsyncJobsMixin} &mdash; the SSE progress stream and cancel endpoints.
39   *
40   * <p>
41   * Drives the two endpoints through a {@link MockRestClient}: a fresh attach is live-only (an ended job sends only its
42   * terminal {@code result} event); a {@code Last-Event-ID} header (or, failing a non-empty one, a {@code lastEventId}
43   * query parameter) resumes from that position, with each progress event carrying {@code id: i+1}; a bad or
44   * past-the-end resume value is a {@code 400}; an unknown capability id is a {@code 404} (enumeration-resistant);
45   * a job already at its subscriber cap is a {@code 429}; and cancel returns the terminal
46   * {@code cancelled} / {@code cancelled-after-effect} outcome.
47   *
48   * <p>
49   * Most SSE assertions complete the job <b>before</b> subscribing, so the stream drains in one deterministic pass.  The
50   * running-job cases run the stream GET on a background thread and only produce progress once
51   * {@link AsyncJob#subscriberCount()} shows the subscriber attached: the stream fixes its cursor before it takes the
52   * subscriber slot, so from that point on the outcome is deterministic.
53   */
54  @SuppressWarnings({
55  	"resource" // Closeable test fixture held in a static field; lifecycle managed by the test/framework, not a real leak.
56  })
57  class AsyncJobsMixin_Serving_Test extends TestBase {

59  	@Rest
60  	public static class R extends BasicRestServlet implements AsyncJobsMixin {
61  		private static final long serialVersionUID = 1L;
62  
63  		// No background scheduler: the serving tests do not exercise wall-clock timeout, so nothing runs off-thread.
64  		static final AsyncJobRegistry REGISTRY = new AsyncJobRegistry(Clock.systemUTC(), Duration.ofSeconds(120),
65  			AsyncJobRegistry.MAX_OUTPUT_BYTES, AsyncJobRegistry.MAX_SUBSCRIBERS_PER_JOB, null);
66  
67  		static volatile AsyncJob lastJob;
68  
69  		@Override public AsyncJobRegistry asyncJobRegistry() { return REGISTRY; }
70  
71  		/** Test-only: starts a job (as a consumer's async row-action POST would) and returns its AsyncJobRef pointer. */
72  		@RestPost(path="/start") public AsyncJobRef start() {
73  			var job = REGISTRY.create();
74  			lastJob = job;
75  			return AsyncJobRef.of(job);
76  		}
77  	}

79  	private static final MockRestClient c = MockRestClient.buildLax(R.class);

81  	private static String startJob() throws Exception {
82  		var body = c.post("/start", "").header("Accept", "application/json").run().assertStatus(200).getContent().asString();
83  		return String.valueOf(Json.to(body, Map.class).get("jobId"));
84  	}
85  
86  	/** Starts a job, buffers {@code e0..e(n-1)} and completes it. */
87  	private static String endedJob(int n) throws Exception {
88  		var jobId = startJob();
89  		var job = R.lastJob;
90  		for (var i = 0; i < n; i++)
91  			job.progress("e" + i);
92  		job.complete(ActionResult.success(Map.of("id", "INC-1")));
93  		return jobId;
94  	}

96  	/** GETs the stream with an optional Last-Event-ID header and lastEventId query value (null = not sent). */
97  	private static org.apache.juneau.rest.client.classic.RestResponse stream(String jobId, String header, String query) throws Exception {
98  		var req = c.get(AsyncJobsMixin.streamPath(jobId)).header("Accept", "text/event-stream");
99  		if (header != null)
100 			req.header(new BasicHeader("Last-Event-ID", header));
101 		if (query != null)
102 			req.queryData("lastEventId", query);
103 		return req.run();
104 	}
105 
106 	private static List<SseEvent> parse(String body) throws IOException {
107 		try (var r = new SseEventReader(new StringReader(body))) {
108 			return r.toList();
109 		}
110 	}
111 
112 	/** Each event as {@code type#id=data} ({@code #id} omitted when the event has no id), result data dropped. */
113 	private static List<String> events(String body) throws IOException {
114 		return parse(body).stream()
115 			.map(e -> e.getEvent() + (e.getId() == null ? "" : "#" + e.getId()) + ("result".equals(e.getEvent()) ? "" : "=" + e.getData()))
116 			.toList();
117 	}
118 
119 	private static List<String> streamEvents(String jobId, String header, String query) throws Exception {
120 		return events(stream(jobId, header, query).assertStatus(200).getContent().asString());
121 	}

127 	/** Waits (bounded by the caller's preemptive timeout) until the job has {@code n} subscribers. */
128 	@SuppressWarnings({
129 		"java:S2925" // Bounded poll helper; the caller's preemptive timeout caps the wait.
130 	})
131 	private static void awaitSubscribers(AsyncJob job, int n) throws InterruptedException {
132 		while (job.subscriberCount() != n)
133 			Thread.sleep(5);
134 	}

140 	@Test void a01_startReturnsAsyncJobRefWithCapabilityUrls() throws Exception {
141 		var body = c.post("/start", "").header("Accept", "application/json").run().assertStatus(200).getContent().asString();
142 		var ref = Json.to(body, Map.class);
143 		var jobId = String.valueOf(ref.get("jobId"));
144 		assertTrue(jobId.matches("[0-9a-f]{64}"), jobId);
145 		// streamUrl/cancelUrl are @Uri-annotated: Juneau's serializer resolves the "servlet:"
146 		// pseudo-scheme against the request's UriContext before the value ever reaches the wire, so under this
147 		// root-mounted MockRestClient (no context/servlet path) the resolved, browser-usable form is identical
148 		// to the un-prefixed mixin path - see a02 below for the literal-scheme regression guard.
149 		assertEquals(AsyncJobsMixin.streamPath(jobId), ref.get("streamUrl"));
150 		assertEquals(AsyncJobsMixin.cancelPath(jobId), ref.get("cancelUrl"));
151 		// The pointer carries NO ActionResult outcome discriminator - the two 2xx shapes are disjoint.
152 		assertNull(ref.get("outcome"));
153 	}
154 
155 	@Test void a02_streamUrlAndCancelUrl_areResolved_notLiteralServletScheme() throws Exception {
156 		// AsyncJobRef.streamUrl/cancelUrl used to be plain, un-annotated Strings, so
157 		// no serializer ever resolved the "servlet:" pseudo-scheme - the browser EventSource received the literal
158 		// "servlet:/juneau-jobs/<id>/stream" and silently rejected the unsupported scheme (Hank's report). @Uri
159 		// opts these fields into the same per-request URI resolution ViewsMixin's asset URLs already rely on
160 		// (SerializedPojoProcessor wires req.getUriContext() into every serializer session, not just HTML's).
161 		var body = c.post("/start", "").header("Accept", "application/json").run().assertStatus(200).getContent().asString();
162 		var ref = Json.to(body, Map.class);
163 		var streamUrl = String.valueOf(ref.get("streamUrl"));
164 		var cancelUrl = String.valueOf(ref.get("cancelUrl"));
165 		assertFalse(streamUrl.startsWith("servlet:"), streamUrl);
166 		assertFalse(cancelUrl.startsWith("servlet:"), cancelUrl);
167 		// Root-relative ("/...") or absolute (has a scheme+authority) are both browser-usable EventSource targets.
168 		assertTrue(streamUrl.startsWith("/") || streamUrl.matches("^[a-zA-Z][a-zA-Z0-9+.-]*://.*"), streamUrl);
169 		assertTrue(cancelUrl.startsWith("/") || cancelUrl.matches("^[a-zA-Z][a-zA-Z0-9+.-]*://.*"), cancelUrl);
170 	}

294 	@Test void b12_idIsIndexPlusOne_whateverTheStartCursor() throws Exception {
295 		var n = 3;
296 		var jobId = endedJob(n);
297 		var buffered = R.lastJob.eventsFrom(0);
298 		for (var k = 0; k <= n; k++) {
299 			var body = stream(jobId, String.valueOf(k), null).assertStatus(200).getContent().asString();
300 			var progress = parse(body).stream().filter(e -> "progress".equals(e.getEvent())).toList();
301 			assertEquals(n - k, progress.size(), body);
302 			for (var i = 0; i < progress.size(); i++) {
303 				var e = progress.get(i);
304 				var id = Integer.parseInt(e.getId());
305 				assertEquals(k + i + 1, id, body);
306 				assertEquals(buffered.get(id - 1), e.getData(), body);
307 			}
308 		}
309 	}

311 	@Test void b13_freshAttachToRunningJob_isLiveOnly() {
312 		assertTimeoutPreemptively(Duration.ofSeconds(10), () -> {
313 			var jobId = startJob();
314 			var job = R.lastJob;
315 			job.progress("early");
316 			var pool = Executors.newSingleThreadExecutor();
317 			try {
318 				var a = pool.submit(() -> streamEvents(jobId, null, null));
319 				awaitSubscribers(job, 1);
320 				job.progress("live");
321 				job.complete(ActionResult.success(null));
322 				assertEquals(List.of("progress#2=live", "result"), a.get());
323 			} finally {
324 				pool.shutdownNow();
325 			}
326 			assertEquals(0, job.subscriberCount());
327 		});
328 	}

331 		assertTimeoutPreemptively(Duration.ofSeconds(10), () -> {
332 			var jobId = startJob();
333 			var job = R.lastJob;
334 			job.progress("early");
335 			var pool = Executors.newFixedThreadPool(2);
336 			try {

346 				pool.shutdownNow();
