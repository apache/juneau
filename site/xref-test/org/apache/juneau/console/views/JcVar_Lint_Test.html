32  /**
33   * Fails when a bundled stylesheet or script references a {@code --jc-*} custom property that nothing defines.
34   *
35   * <p>
36   * A typo in {@code var(--jc-...)} otherwise degrades silently to the fallback (or to nothing) at runtime.
37   * A name counts as defined when a bundled stylesheet declares it, a theme token ({@link Theme#OPEN}) or chrome role alias ({@link ConsoleChromeMixin#roleAliasNames()}) provides it,
38   * or a script stamps it with {@code style.setProperty}. A use is checked from {@code var(...)} in CSS and JS and from
39   * {@code getPropertyValue(...)} in JS.
40   */
41  class JcVar_Lint_Test extends TestBase {
42  
43  	/**
44  	 * Modules whose bundled {@code .css} and {@code .js} are scanned, relative to this module's base directory.
45  	 * Read from the source trees (not the classpath) so a module this one does not depend on is still covered;
46  	 * Maven and Eclipse both run tests with the module directory as the working directory.
47  	 */
48  	private static final List<String> MODULES = List.of(
49  		"../juneau-console/src/main/resources",
50  		"src/main/resources",
51  		"../juneau-console-widgets/src/main/resources",
52  		"../juneau-terminal-region/src/main/resources");
53  
54  	/**
55  	 * Names used only as {@code var(--jc-x, fallback)} override hooks: no bundled sheet defines them, an embedding
56  	 * application may.  Each entry needs a reason.  Currently empty.
57  	 */
58  	private static final Set<String> FALLBACK_ONLY = Set.of();
59  
60  	/** Third-party and generated files that are not ours to lint. */
61  	private static boolean vendored(Path p) {
62  		var n = p.getFileName().toString();
63  		return n.endsWith(".min.js") || n.endsWith(".min.css") || p.toString().contains("/vendor/") || p.toString().contains("/META-INF/");
64  	}
65  
66  	private static List<Path> resources() throws IOException {
67  		var base = Path.of(System.getProperty("basedir", "")).toAbsolutePath();
68  		var files = new ArrayList<Path>();
69  		for (var m : MODULES) {
70  			var dir = base.resolve(m).normalize();
71  			assertTrue(Files.isDirectory(dir), dir + " not found; run from the juneau-console-views module directory");
72  			try (var s = Files.walk(dir)) {
73  				s.filter(p -> p.toString().endsWith(".css") || p.toString().endsWith(".js"))
74  					.filter(p -> ! vendored(p))
75  					.forEach(files::add);
76  			}
77  		}
78  		assertTrue(files.size() > 30, "scan found only " + files.size() + " files; the directory scan has probably stopped matching");
79  		return files;
80  	}
81  
82  	private static final Pattern BLOCK_COMMENT = Pattern.compile("/\\*.*?\\*/", Pattern.DOTALL);
83  	private static final Pattern LINE_COMMENT = Pattern.compile("(?m)^\\s*//.*$");
84  	private static final Pattern DECLARATION = Pattern.compile("(--jc-[a-z0-9-]+)\\s*:");
85  	private static final Pattern SET_PROPERTY = Pattern.compile("setProperty\\(\\s*[\"'](--jc-[a-z0-9-]+)[\"']");
86  	private static final Pattern VAR_USE = Pattern.compile("var\\(\\s*(--jc-[a-z0-9-]+)");
87  	private static final Pattern GET_PROPERTY = Pattern.compile("getPropertyValue\\(\\s*[\"'](--jc-[a-z0-9-]+)[\"']");
88  
89  	private static String read(Path path) throws IOException {
90  		var s = Files.readString(path, StandardCharsets.UTF_8);
91  		s = BLOCK_COMMENT.matcher(s).replaceAll("");
92  		return path.toString().endsWith(".js") ? LINE_COMMENT.matcher(s).replaceAll("") : s;
93  	}
94  
95  	private static Set<String> find(Pattern p, String text) {
96  		var names = new TreeSet<String>();
97  		var m = p.matcher(text);
98  		while (m.find())
99  			names.add(m.group(1));
100 		return names;
101 	}
102 
103 	private static TreeSet<String> definedByTheme() {
104 		var names = new TreeSet<String>(Theme.OPEN.getTokens().keySet());
105 		names.addAll(ConsoleChromeMixin.roleAliasNames());
106 		return names;
107 	}
108 
109 	@Test void a01_everyUsedNameIsDefined() throws IOException {
110 		var defined = definedByTheme();
111 		var used = new TreeMap<String,Path>();
112 		for (var path : resources()) {
113 			var text = read(path);
114 			defined.addAll(find(DECLARATION, text));
115 			defined.addAll(find(SET_PROPERTY, text));
116 			for (var n : find(VAR_USE, text))
117 				used.putIfAbsent(n, path);
118 			for (var n : find(GET_PROPERTY, text))
119 				used.putIfAbsent(n, path);
120 		}
121 		var undefined = used.entrySet().stream()
122 			.filter(e -> ! defined.contains(e.getKey()) && ! FALLBACK_ONLY.contains(e.getKey()))
123 			.map(e -> e.getKey() + " (" + e.getValue() + ")")
124 			.collect(Collectors.toList());
125 		assertTrue(undefined.isEmpty(), "Undefined --jc-* custom properties: " + undefined);
126 	}
127 
128 	@Test void a02_fallbackAllowlistHasNoStaleEntries() throws IOException {
129 		var defined = definedByTheme();
130 		var used = new TreeSet<String>();
131 		for (var path : resources()) {
132 			var text = read(path);
133 			defined.addAll(find(DECLARATION, text));
134 			defined.addAll(find(SET_PROPERTY, text));
135 			used.addAll(find(VAR_USE, text));
136 		}
137 		for (var n : FALLBACK_ONLY) {
138 			assertTrue(used.contains(n), n + " is allowlisted but no longer used; remove it");
139 			assertFalse(defined.contains(n), n + " is now defined; remove it from the allowlist");
140 		}
141 	}
