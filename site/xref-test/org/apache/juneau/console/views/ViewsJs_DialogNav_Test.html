27  /**
28   * Node behavioral checks for the dialog frame's inner navigation: {@code data-nav}, the wizard step rail and
29   * per-step validation (driven by {@code dialog-nav.cjs}).
30   */
31  class ViewsJs_DialogNav_Test extends TestBase {
32  
33  	private static Map<?,?> report() {
34  		var r = NodeHarness.report("dialog-nav.cjs", ViewsMixin.RENDERS_JS_RESOURCE, ViewsMixin.VIEWS_JS_RESOURCE);
35  		assumeTrue(r != null, "node (or dialog-nav.cjs) not available - behavioral layer skipped");
36  		return r;
37  	}
38  
39  	private static Map<?,?> map(String key) {
40  		return (Map<?,?>)report().get(key);
41  	}
42  
43  	@Test void a01_dataNav_pillByDefault() {
44  		assertEquals("pill", map("plain").get("nav"));
45  		assertEquals(false, map("plain").get("rail"));
46  	}
47  
48  	@Test void a02_dataNav_pillAndFolderFromModalNav() {
49  		assertEquals("pill", map("pill").get("nav"));
50  		assertEquals(true, map("pill").get("strip"));
51  		assertEquals("folder", map("folder").get("nav"));
52  		assertEquals(true, map("folder").get("strip"));
53  	}
54  
55  	@Test void a03_dataNav_unknownFallsBackToPillSilently() {
56  		assertEquals("pill", map("unknown").get("nav"));
57  		assertEquals(List.of(), report().get("unknownLogged"));
58  	}
59  
60  	@Test void a04_dataNav_stepsMakeAWizard() {
61  		var w = map("wizardWinsOverNav");
62  		assertEquals("wizard", w.get("nav"));
63  		assertEquals(true, w.get("rail"));
64  	}
65  
66  	@Test void b01_rail_numberedMarksAndActiveStep() {
67  		var r = map("rail");
68  		assertEquals("OL", r.get("tag"));
69  		assertEquals("Steps", r.get("ariaLabel"));
70  		assertEquals(true, r.get("railAfterHeader"));
71  		assertEquals(List.of(
72  			Map.of("state", "active", "current", "step", "mark", "1", "label", "Pick"),
73  			Map.of("state", "todo", "mark", "2", "label", "two"),
74  			Map.of("state", "todo", "mark", "3", "label", "Done")
75  		), stripNulls((List<?>)r.get("at0")));
76  	}
77  
78  	@Test void b02_rail_completedStepShowsCheckMark() {
79  		var r = map("rail");
80  		assertEquals(List.of(
81  			Map.of("state", "done", "mark", "✓", "label", "Pick"),
82  			Map.of("state", "active", "current", "step", "mark", "2", "label", "two"),
83  			Map.of("state", "todo", "mark", "3", "label", "Done")
84  		), stripNulls((List<?>)r.get("at1")));
85  	}
86  
87  	@Test void b03_wizard_onePanePerStepOnlyCurrentShown() {
88  		var r = map("rail");
89  		assertEquals(3, ((Number)r.get("paneCount")).intValue());
90  		assertEquals(List.of("two"), r.get("visiblePanes"));
91  	}
92  
93  	@Test void b04_wizard_footerCancelSpacerBackConfirm() {
94  		assertEquals(List.of("juneau-view-dialog-cancel", "juneau-view-dialog-actions-spacer", "juneau-view-dialog-back", "juneau-view-dialog-confirm"),
95  			map("rail").get("footer"));
96  	}
97  
98  	@Test void b05_wizard_consumesOneDepthSlot() {
99  		assertEquals(1, ((Number)map("rail").get("depth")).intValue());
100 	}
101 
102 	@Test void c01_next_validatesOnlyTheCurrentStep() {
103 		var v = map("validation");
104 		assertEquals("one", v.get("blockedStep"));
105 		assertEquals(true, v.get("focusedA"));
106 		assertEquals("true", v.get("aInvalid"));
107 		assertEquals(true, v.get("bUntouched"));
108 		assertEquals("two", v.get("advancedStep"));
109 		assertEquals(true, v.get("bUntouchedAfterNext"));
110 	}
111 
112 	@Test void c02_finish_validatesEveryStepAndRevealsTheFirstInvalid() {
113 		var v = map("validation");
114 		assertEquals("one", v.get("finishRevealedStep"));
115 		assertEquals(true, v.get("finishFocusedA"));
116 		assertEquals(true, v.get("paneOneShown"));
117 		assertEquals(true, v.get("paneTwoHidden"));
118 		assertEquals(1, ((Number)v.get("stillOpen")).intValue());
119 		assertEquals("submitted", v.get("closed"));
120 	}
121 
122 	private static List<Map<?,?>> stripNulls(List<?> l) {
123 		var out = new ArrayList<Map<?,?>>();
124 		for (var o : l) {
125 			var m = new LinkedHashMap<Object,Object>((Map<?,?>)o);
126 			m.values().removeIf(Objects::isNull);
127 			out.add(m);
128 		}
129 		return out;
130 	}
