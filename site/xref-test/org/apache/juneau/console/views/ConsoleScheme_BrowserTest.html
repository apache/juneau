75  	/** Records every change event from before the shell runs, so events fired while the page loads are seen too. */
76  	private static final String EVENTS = """
77  		<script>
78  		window.__events = [];
79  		document.addEventListener('juneau:theme-changed', function (e) { window.__events.push(e.detail.scheme); });
80  		</script>
81  		""";

83  	/** Publishes the page state lazily, so the probe reads it after any clicks and steps. */
84  	private static final String PROBE = """
85  		<script>
86  		window.__probe = { toJSON: function () {
87  			var h = document.documentElement, t = document.querySelector('.jc-scheme-toggle'), out = {};
88  			out.scheme = h.getAttribute('data-jc-scheme');
89  			out.events = window.__events.slice();
90  			try { out.stored = localStorage.getItem('juneau-console:scheme'); } catch (x) { out.stored = 'ERR'; }
91  			out.apiOk = window.__apiOk === true;
92  			out.hasToggle = !!t;
93  			if (t) {
94  				out.role = t.getAttribute('role');
95  				out.label = t.getAttribute('aria-label');
96  				out.buttons = Array.prototype.map.call(t.querySelectorAll('button[role=radio]'), function (b) {
97  					return { scheme: b.getAttribute('data-scheme'), checked: b.getAttribute('aria-checked'), tab: b.getAttribute('tabindex'), title: b.getAttribute('title') };
98  				});
99  				var f = document.activeElement;
100 				out.focused = f ? f.getAttribute('data-scheme') : null;
101 				var on = t.querySelector('[aria-checked=true]');
102 				if (on) {
103 					var cs = getComputedStyle(on), p = document.createElement('i');
104 					document.body.appendChild(p);
105 					var tok = function (prop, name) { p.style[prop] = 'var(' + name + ')'; return getComputedStyle(p)[prop]; };
106 					out.style = { bg: cs.backgroundColor, wash: tok('backgroundColor', '--jc-accent-wash'), color: cs.color, text: tok('color', '--jc-text'),
107 						border: cs.borderBottomColor, accent: tok('color', '--jc-accent') };
108 					p.remove();
109 				}
110 			}
111 			var ha = document.querySelector('.jc-header-actions');
112 			out.actions = ha ? Array.prototype.map.call(ha.children, function (c) { return c.className.split(' ')[0] || c.tagName.toLowerCase(); }) : null;
113 			return out;
114 		} };
115 		</script>
116 		""";

121 		if (ha) { var b = document.createElement('div'); b.className = 'jc-header-badges'; ha.appendChild(b); }
122 		</script>

131 	/** A toggle page whose head carries no scheme script, as when a CSP blocks it. */
132 	private static String pageWithoutHeadScript(String json, String attrs) {
133 		var withScript = page(json, attrs);
134 		var without = withScript.replace("<script>" + ConsoleChromeMixin.SCHEME_SCRIPT + "</script>", "");
135 		assertNotEquals(withScript, without, "the fixture no longer emits the head script");
136 		return without;
137 	}

144 	static void runPages() throws Exception {
145 		report = ConsoleBrowserFixture.create("scheme")
146 			.page("plain", ORIGIN + "/plain", page(SIMPLE, ""))
147 			.page("render", ORIGIN + "/render", page(SIMPLE, TOGGLE))
148 			.page("order", ORIGIN + "/order", page(FULL, FULL_TEMPLATES, TOGGLE, BADGE))
149 			.page("only", ORIGIN + "/only", page(TOGGLE_ONLY, TOGGLE))
150 			.page("clickDark", ORIGIN + "/clickDark", page(SIMPLE, TOGGLE)).click("[data-scheme=dark]")
151 			.page("clickDarkTwice", ORIGIN + "/clickDarkTwice", page(SIMPLE, TOGGLE)).click("[data-scheme=dark]").click("[data-scheme=dark]")
152 			.page("autoClears", ORIGIN + "/autoClears", page(SIMPLE, TOGGLE)).storage(Map.of(KEY, "dark")).colorScheme("light").click("[data-scheme=auto]")
153 			.page("pinned", ORIGIN + "/pinned", page(SIMPLE, "data-jc-scheme=\"dark\" data-jc-scheme-pin=\"dark\" " + TOGGLE))
154 			.page("keyRight", ORIGIN + "/keyRight", page(SIMPLE, TOGGLE)).click("[data-scheme=auto]").press("ArrowRight")
155 			.page("keyEnd", ORIGIN + "/keyEnd", page(SIMPLE, TOGGLE)).click("[data-scheme=auto]").press("End")
156 			.page("keyHome", ORIGIN + "/keyHome", page(SIMPLE, TOGGLE)).click("[data-scheme=dark]").press("Home")
157 			.page("osAuto", ORIGIN + "/osAuto", page(SIMPLE, "")).colorScheme("light").emulate("dark")
158 			.page("osPinned", ORIGIN + "/osPinned", page(SIMPLE, "data-jc-scheme=\"light\" data-jc-scheme-pin=\"light\"")).colorScheme("light").emulate("dark")
159 			.page("osStored", ORIGIN + "/osStored", page(SIMPLE, TOGGLE)).storage(Map.of(KEY, "light")).colorScheme("light").emulate("dark")
160 			.page("storageToggle", ORIGIN + "/storageToggle", page(SIMPLE, TOGGLE)).colorScheme("light")
161 				.eval("window.dispatchEvent(new StorageEvent('storage', {key: '" + KEY + "', newValue: 'dark'}))")
162 			.page("storagePlain", ORIGIN + "/storagePlain", page(SIMPLE, "")).colorScheme("light")
163 				.eval("window.dispatchEvent(new StorageEvent('storage', {key: '" + KEY + "', newValue: 'dark'}))")
164 			.page("noHead", ORIGIN + "/noHead", pageWithoutHeadScript(SIMPLE, TOGGLE)).storage(Map.of(KEY, "dark")).colorScheme("light")
165 			.page("noHeader", ORIGIN + "/noHeader", pageWithoutHeadScript(NO_HEADER, "data-jc-scheme=\"light\" " + TOGGLE)).storage(Map.of(KEY, "dark")).colorScheme("light")
166 			.page("api", ORIGIN + "/api", page(REPLACE, REPLACE_TEMPLATES, TOGGLE,
167 				"<script>var c = JuneauConsole.schemeControl(document); document.body.appendChild(c); window.__apiOk = c.matches('.jc-scheme-toggle[role=radiogroup]');</script>"))
168 				.click("[data-scheme=dark]")
169 			.run();
170 	}

172 	private static Map<String,Object> p(String name) {
173 		var r = report.get(name);
174 		assertNoShellErrors(r);
175 		return probe(r);
176 	}

178 	private static List<Map<String,Object>> buttons(Map<String,Object> pr) {
179 		return (List<Map<String,Object>>)pr.get("buttons");
180 	}
181 
182 	private static String checked(Map<String,Object> pr) {
183 		var on = buttons(pr).stream().filter(b -> "true".equals(b.get("checked"))).toList();
184 		assertSize(() -> "exactly one checked: " + pr, 1, on);
185 		assertEquals("0", on.get(0).get("tab"));
186 		assertEquals(1, buttons(pr).stream().filter(b -> "0".equals(b.get("tab"))).count());
187 		return (String)on.get(0).get("scheme");
188 	}
