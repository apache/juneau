13  package org.apache.juneau.console.views;
14  
15  import static org.junit.jupiter.api.Assertions.*;
16  
17  import org.apache.juneau.*;
18  import org.apache.juneau.console.*;
19  import org.apache.juneau.marshall.marshaller.*;
20  import org.junit.jupiter.api.*;
21  
22  /** Detail tabs of every kind, and a kind-less tab, validate against the page contract schema. */
23  class RowDetailTabs_Schema_Test extends TestBase {
24  
25  	private static String page(RowDetail detail) {
26  		var card = TableSpec.create("t").dataUrl("/rest/t/data").columns(Column.create("id"), Column.create("name")).detail(detail).toCardBody();
27  		var json = Json.of(card);
28  		return "{\"contractVersion\":\"1\",\"title\":\"T\",\"nav\":[],\"activeNav\":[],\"cards\":[{\"id\":\"t\",\"type\":\"datatables\",\"table\":" + json + "}]}";
29  	}
30  
31  	private static void assertValid(String json) {
32  		var errors = PageContractSchema.get().validate(json);
33  		assertTrue(errors.isEmpty(), () -> errors + " in " + json);
34  	}
35  
36  	@Test void a01_javaBuiltTabs_emitKindAndValidate() {
37  		var nested = TableSpec.create("n").dataUrl("/rest/n/data").columns(Column.create("id"));
38  		var detail = RowDetail.inline()
39  			.tab(DetailTab.fields("Summary", "name"))
40  			.tab(DetailTab.table("Items", nested))
41  			.tab(DetailTab.chart("Trend", RegionDef.chart("trend", "/rest/trend/{id}").kind("line").x("name")))
42  			.tab(DetailTab.client("Custom", "custom-renderer"));
43  		var json = page(detail);
44  		for (var k : new String[]{"fields", "table", "chart", "client"})
45  			assertTrue(json.contains("\"kind\":\"" + k + "\""), () -> k + " missing in " + json);
46  		assertValid(json);
47  	}
48  
49  	@Test void a02_kindlessTab_validates() {
50  		var json = "{\"contractVersion\":\"1\",\"title\":\"T\",\"nav\":[],\"activeNav\":[],\"cards\":[{\"id\":\"t\",\"type\":\"datatables\",\"table\":"
51  			+ "{\"dataUrl\":\"/rest/t/data\",\"columns\":[{\"key\":\"id\",\"label\":\"Id\"}],\"detail\":{\"mode\":\"inline\",\"tabs\":[{\"id\":\"x\",\"title\":\"X\"}]}}}]}";
52  		assertValid(json);
53  	}
54  }
