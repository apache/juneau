26  /**
27   * Contract text cannot break out of the {@code #juneau-page} island: every {@code <} is written as the JSON unicode
28   * escape, and the parsed value is the author's literal string.
29   *
30   * @since 10.0.0
31   */
32  class PageCapture_ScriptSafety_Test extends TestBase {
33  
34  	@Test void a01_titleWithScriptCloseIsEscapedAndRoundTrips() {
35  		var html = render("page-script-break");
36  		var start = html.indexOf("<script type=\"application/json\" id=\"juneau-page\">");
37  		var end = html.indexOf("</script>", start);
38  		var island = html.substring(start, end);
39  		assertFalse(island.contains("<b>"), island);
40  		assertTrue(island.contains("\\u003c/script>"), island);
41  		var c = PageCapture_Segments_Test.card(assertPage(html).isValid(), "x");
42  		assertEquals("</script><b>t</b>", c.getString("title"));
43  	}
44  
45  	@Test void a02_noSidecarsAnywhere() {
46  		for (var name : new String[] {"page-segments", "page-card-title", "page-card-src", "console-full"})
47  			assertFalse(render(name).contains("juneau-card-sidecar"), name);
48  	}
49  }
