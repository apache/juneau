32  	private static Map<String,Object> frag(String body) {
33  		var b = CardSource.create("job-log", "j");
34  		if (body != null)
35  			b.body(body);
36  		return new JobLogCardType().toFragment(b.build());
37  	}

39  	@Test void a01_type_andNoBusRoles() {
40  		var h = new JobLogCardType();
41  		assertEquals("job-log", h.type());
42  		assertEquals(Set.of(), h.acceptedRoles());
43  		assertEquals(Set.of(), h.acceptedOps());
44  	}

48  			+ "height:320,urlParam:'job',clearOnSubmit:false}";
49  		assertEquals(Json5.to(body, Map.class), Json5.to(Json.of(frag(body)), Map.class));

110 		// Its own test: the message holds the CSV delimiter.
111 		var e = assertThrows(IllegalArgumentException.class, () -> frag("{submit:{endpoint:'/a'},output:{target:{field:'a'},from:'row.a.b'}}"));
112 		assertEquals("<@card id='j'> type='job-log' output.from must match 'message|row\\.[A-Za-z_][A-Za-z0-9_]{0,63}'; got 'row.a.b'.", e.getMessage());
