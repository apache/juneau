34  /**
35   * Directive emission of the client object model scripts and the inline-handler CSP check: the model script follows
36   * the bus in the runtime pack, the inline compiler is a separate tag only for a page that opts in, and an opted-in
37   * page fails fast under a response CSP that lacks {@code 'unsafe-eval'}.
38   *
39   * @since 10.0.0
40   */
41  @SuppressWarnings({
42  	"resource" // MockRestClient/RestResponse are closed in try-with-resources; fluent assertStatus returns this.
43  })
44  class InlineHandlerCsp_Test extends TestBase {
45  
46  	@Rest(mixins={FreemarkerMixin.class, ViewsMixin.class}, renderResponseStackTraces="true")
47  	public static class Host extends BasicRestServlet {
48  		private static final long serialVersionUID = 1L;
49  		@Bean public FreemarkerMixin freemarker() { return ConsoleFreemarkerMixin.create().basePath("/templates/").build(); }
50  
51  		private static PageSpec spec(boolean inline) {
52  			var s = PageSpec.create().template("admin/theme-nameless.ftlh").toolkit("views");
53  			return inline ? s.allowInlineHandlers(true).on("loaded", HandlerRef.inline("return 1")) : s;
54  		}
55  		@RestGet(path="/plain") public View plain(RestRequest req) { return spec(false).view(req); }
56  		@RestGet(path="/inline") public View inline(RestRequest req) { return spec(true).view(req); }
57  		@RestGet(path="/csp-strict") public View cspStrict(RestRequest req) {
58  			req.getResponse().setHeader("Content-Security-Policy", "script-src 'self'");
59  			return spec(true).view(req);
60  		}
61  		@RestGet(path="/csp-eval") public View cspEval(RestRequest req) {
62  			req.getResponse().setHeader("Content-Security-Policy", "script-src 'self' 'unsafe-eval'");
63  			return spec(true).view(req);
64  		}
65  		@RestGet(path="/csp-default") public View cspDefault(RestRequest req) {
66  			req.getResponse().setHeader("Content-Security-Policy", "default-src 'self'");
67  			return spec(true).view(req);
68  		}
69  		@RestGet(path="/csp-default-eval") public View cspDefaultEval(RestRequest req) {
70  			req.getResponse().setHeader("Content-Security-Policy", "default-src 'self' 'unsafe-eval'");
71  			return spec(true).view(req);
72  		}
73  		@RestGet(path="/csp-multi") public View cspMulti(RestRequest req) {
74  			req.getResponse().addHeader("Content-Security-Policy", "script-src 'self' 'unsafe-eval'");
75  			req.getResponse().addHeader("Content-Security-Policy", "script-src 'self'");
76  			return spec(true).view(req);
77  		}
78  		@RestGet(path="/csp-eval-mixed-case") public View cspEvalMixedCase(RestRequest req) {
79  			req.getResponse().setHeader("Content-Security-Policy", "script-src 'self' 'Unsafe-Eval'");
80  			return spec(true).view(req);
81  		}
82  		@RestGet(path="/late-table-inline") public View lateTableInline(RestRequest req) {
83  			var t = TableSpec.create("t1").dataUrl("/api/t").columns(Column.create("id"));
84  			var page = spec(false).table(t, (java.util.function.Consumer<CardSpec>)null);
85  			t.on("loaded", HandlerRef.inline("return 1"));   // added after table(...) captured the table
86  			return page.view(req);
87  		}
88  		@RestGet(path="/xfo") public View xfo(RestRequest req) {
89  			req.getResponse().setHeader("X-Frame-Options", "DENY");
90  			return spec(false).view(req);
91  		}
92  		@RestGet(path="/own-ancestors") public View ownAncestors(RestRequest req) {
93  			req.getResponse().setHeader("Content-Security-Policy", "frame-ancestors https://partner.example");
94  			return spec(false).view(req);
95  		}
96  		@RestGet(path="/csp-elem-only") public View cspElemOnly(RestRequest req) {
97  			req.getResponse().setHeader("Content-Security-Policy", "script-src 'self' 'unsafe-eval'; script-src-elem 'self'");
98  			return spec(true).view(req);
99  		}
100 	}
101 
102 	private static String get(String path, int status) throws Exception {
103 		try (var c = MockRestClient.buildLax(Host.class); var rsp = c.get(path).run()) {
104 			rsp.assertStatus(status);
105 			return rsp.getContent().asString();
106 		}
107 	}
108 
109 	private static List<String> cspHeaders(String path) throws Exception {
110 		try (var c = MockRestClient.buildLax(Host.class); var rsp = c.get(path).run()) {
111 			rsp.assertStatus(200);
112 			return Arrays.stream(rsp.getHeaders("Content-Security-Policy")).map(h -> h.getValue()).toList();
113 		}
114 	}
115 
116 	@Test void c01_pageDefaultsToSameOriginFraming() throws Exception {
117 		assertEquals(List.of("frame-ancestors 'self'"), cspHeaders("/plain"));
118 	}
119 
120 	@Test void c02_appFramingPolicyIsKept() throws Exception {
121 		assertEquals(List.of("frame-ancestors https://partner.example"), cspHeaders("/own-ancestors"));
122 		assertEquals(List.of(), cspHeaders("/xfo"));
123 	}

125 	@Test void a01_modelJs_followsBus() throws Exception {
126 		var b = get("/plain", 200);
127 		var bus = b.indexOf("juneau-bus.js");
128 		var model = b.indexOf("juneau-model.js");
129 		assertTrue(bus >= 0 && model > bus, () -> b);
130 		assertTrue(model < b.indexOf("juneau-renders.js"), () -> b);
131 	}
132 
133 	@Test void a02_notOptedIn_noInlineFile() throws Exception {
134 		assertFalse(get("/plain", 200).contains("juneau-model-inline.js"));
135 	}

137 	@Test void a03_optIn_inlineAfterModel() throws Exception {
138 		var b = get("/inline", 200);
139 		assertTrue(b.indexOf("juneau-model-inline.js") > b.indexOf("juneau-model.js"), () -> b);
140 		assertTrue(b.contains("\"allowInlineHandlers\":true"), () -> b);
141 	}
