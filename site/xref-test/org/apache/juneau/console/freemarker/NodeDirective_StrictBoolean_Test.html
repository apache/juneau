19  import static org.apache.juneau.console.test.PageContractAssert.*;
20  import static org.apache.juneau.console.freemarker.C1Fixtures.*;

27  /**
28   * Strict booleans (E-5) for {@code <@node visible|selected>} and {@code <@console chrome>}.
29   *
30   * @since 10.0.0
31   */
32  class NodeDirective_StrictBoolean_Test extends TestBase {
33  
34  	@ParameterizedTest
35  	@CsvSource({"bool-visible-true,true", "bool-visible-false,false", "bool-visible-model,false"})
36  	void a01_visible(String fixture, boolean shown) {
37  		var a = assertPage(render(fixture)).isValid();
38  		assertEquals(shown ? 1 : 0, a.contract().getList("nav").size());
39  	}
40  
41  	@ParameterizedTest
42  	@CsvSource({"bool-selected-true", "bool-selected-model"})
43  	void a02_selected(String fixture) {
44  		assertPage(render(fixture)).isValid().hasActiveNav("a");
45  	}
46  
47  	@ParameterizedTest
48  	@CsvSource({"bool-chrome-true,true", "bool-chrome-model,true", "bool-chrome-false,false"})
49  	void a03_chrome(String fixture, boolean on) {
50  		var h = assertPage(render(fixture)).contract().getMap("header");
51  		assertEquals(on, h != null && Boolean.TRUE.equals(h.get("chrome")));
52  	}
53  
54  	@ParameterizedTest
55  	@CsvSource(delimiter='|', value={
56  		"bool-visible-tru|<@node> visible= must be true or false; got 'tru'.",
57  		"bool-visible-empty|<@node> visible= must be true or false; got ''.",
58  		"bool-selected-tru|<@node> selected= must be true or false; got 'tru'.",
59  		"bool-selected-empty|<@node> selected= must be true or false; got ''.",
60  		"bool-chrome-tru|<@console> chrome= must be true or false; got 'tru'.",
61  		"bool-chrome-empty|<@console> chrome= must be true or false; got ''."
62  	})
63  	void a04_rejected_E5(String fixture, String message) {
64  		assertError(fixture, message);
65  	}
66  }
