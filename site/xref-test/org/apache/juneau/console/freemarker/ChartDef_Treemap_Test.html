27  class ChartDef_Treemap_Test extends TestBase {
28  
29  	private static Object j(String json5) { return Json5.to(json5, Object.class); }
30  
31  	private static Map<?,?> map(ChartDef d) {
32  		return Json5.to(Json.of(d.toCardSpec().toMap()), Map.class);
33  	}
34  
35  	private static String fail(Runnable r) {
36  		return assertThrows(IllegalArgumentException.class, r::run).getMessage();
37  	}
38  
39  	private static ChartDef fleet() {
40  		return ChartDef.treemap("fleet").src("/api/hosts").groupBy("region", "cluster").label("host");
41  	}
42  
43  	@Test void a01_treemap_emitsHelperUnderHelperKey() {
44  		var d = fleet()
45  			.colorBy("health", ChartDef.legendItem("ok", "Healthy", "ok"), ChartDef.legendItem("warn", "Degraded", "warning"),
46  				ChartDef.legendItem("down", "Down", "danger"))
47  			.tileFields("host", "health", "cpu");
48  		assertEquals(j("{id:'fleet',type:'chart',src:'/api/hosts',helper:{kind:'treemap',groupBy:['region','cluster'],label:'host',"
49  			+ "color:{field:'health',legend:[{value:'ok',label:'Healthy',tone:'ok'},{value:'warn',label:'Degraded',tone:'warning'},"
50  			+ "{value:'down',label:'Down',tone:'danger'}]},tileFields:['host','health','cpu']}}"), map(d));
51  	}
52  
53  	@Test void a02_treemap_isAKind() {
54  		assertTrue(ChartDef.KINDS.contains("treemap"));
55  		assertFalse(ChartDef.LEGACY_KINDS.contains("treemap"));
56  	}
57  
58  	@Test void a03_groupBy_emptyOrBlank_rejected() {
59  		assertEquals("ChartDef 'fleet' groupBy requires at least one non-blank field.", fail(() -> ChartDef.treemap("fleet").groupBy()));
60  		assertEquals("ChartDef 'fleet' groupBy requires at least one non-blank field.", fail(() -> ChartDef.treemap("fleet").groupBy("a", " ")));
61  	}
62  
63  	@Test void a04_colorBy_unknownTone_rejected() {
64  		assertEquals("ChartDef legendItem tone 'success' must be 'ok', 'warning', 'danger', '--jc-chart-1'..'--jc-chart-8' or null.",
65  			fail(() -> ChartDef.legendItem("x", "X", "success")));
66  		assertEquals("ChartDef legendItem tone '--jc-chart-9' must be 'ok', 'warning', 'danger', '--jc-chart-1'..'--jc-chart-8' or null.",
67  			fail(() -> ChartDef.legendItem("x", "X", "--jc-chart-9")));
68  	}
69  
70  	@Test void a05_legendItem_nullTone_isNeutral() {
71  		var m = ChartDef.legendItem("x", "X", null);
72  		assertEquals(Map.of("value", "x", "label", "X"), m);
73  		assertFalse(m.containsKey("tone"));
74  		assertEquals("--jc-chart-3", ChartDef.legendItem("x", "X", "--jc-chart-3").get("tone"));
75  		assertEquals("x", ChartDef.legendItem("x", null, null).get("label"));
76  		assertEquals("ChartDef legendItem requires a non-blank value.", fail(() -> ChartDef.legendItem(" ", "X", null)));
77  	}
78  
79  	@Test void a06_size_optional_andNamed() {
80  		var none = (Map<?,?>)map(fleet()).get("helper");
81  		assertFalse(none.containsKey("size"));
82  		var sized = (Map<?,?>)map(fleet().size("cpu")).get("helper");
83  		assertEquals("cpu", sized.get("size"));
84  		assertEquals("ChartDef 'fleet' size requires a non-blank field.", fail(() -> fleet().size(" ")));
85  	}
86  
87  	@Test void a07_label_required() {
88  		assertEquals("ChartDef 'fleet' kind 'treemap' requires label(field).",
89  			fail(() -> ChartDef.treemap("fleet").src("/s").groupBy("region").toCardSpec()));
90  		assertEquals("ChartDef 'fleet' kind 'treemap' requires groupBy(...).",
91  			fail(() -> ChartDef.treemap("fleet").src("/s").label("host").toCardSpec()));
92  	}
93  
94  	@Test void a08_tileFields_invalidName_rejected() {
95  		assertEquals("ChartDef 'fleet' tileFields entry 'a b' must match [A-Za-z0-9_.-]+.", fail(() -> ChartDef.treemap("fleet").tileFields("a b")));
96  		assertEquals("ChartDef 'fleet' tileFields entry 'a{b}' must match [A-Za-z0-9_.-]+.", fail(() -> ChartDef.treemap("fleet").tileFields("ok", "a{b}")));
97  	}
98  
99  	@Test void a09_set_treemapKeys_rejected() {
100 		for (var k : List.of("groupBy", "size", "color", "tileFields"))
101 			assertEquals("ChartDef 'fleet' key '" + k + "' has a dedicated method.", fail(() -> fleet().set(k, "x")));
102 	}

104 	@Test void a10_kinds_matchBothSchemaKindEnums() throws Exception {
105 		try (var in = ChartDef.class.getResourceAsStream("/org/apache/juneau/console/juneau-page.schema.json")) {
106 			assertNotNull(in, "page schema on the classpath");
107 			var schema = Json5.to(new String(in.readAllBytes(), java.nio.charset.StandardCharsets.UTF_8), Map.class);
108 			var chart = (Map<?,?>)((Map<?,?>)schema.get("$defs")).get("chartCard");
109 			var props = (Map<?,?>)chart.get("properties");
110 			var cardKinds = (List<?>)((Map<?,?>)props.get("kind")).get("enum");
111 			var helperProps = (Map<?,?>)((Map<?,?>)props.get("helper")).get("properties");
112 			var helperKinds = (List<?>)((Map<?,?>)helperProps.get("kind")).get("enum");
113 			assertEquals(new HashSet<>(ChartDef.KINDS), new HashSet<>(cardKinds));
114 			assertEquals(new HashSet<>(ChartDef.KINDS), new HashSet<>(helperKinds));
115 			for (var k : List.of("groupBy", "size", "color", "tileFields"))
116 				assertTrue(helperProps.containsKey(k), "schema helper property " + k);
117 		}
118 	}
