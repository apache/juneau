200 	@Test void a05_toolkitViews_noCards_noVendor() throws Exception {
201 		var body = toolkitBody("/toolkit");
202 		// The "views" pack is emitted (marked so a consumer chrome can find it); page-cards is no longer part of it.
203 		assertTrue(body.contains("data-toolkit-js"), () -> body);
204 		assertTrue(body.contains("juneau-views.js"), () -> body);
205 		assertFalse(body.contains("juneau-page-cards.js"), () -> body);
206 		assertTrue(body.indexOf("juneau-views.js") < body.indexOf("juneau-helpers.js"), () -> body);
207 		// No table card: no glue, no DataTables, no jQuery.
208 		assertFalse(body.toLowerCase().contains("datatables"), () -> body);
209 		assertFalse(body.contains("/webjars/jquery/"), () -> body);
210 		DualHatGuard.assertClean(body);
211 	}

213 	@Test void a05d_toolkitViews_emitsBusFirst_beforeEveryBusConsumer() throws Exception {
214 		var body = toolkitBody("/toolkit");
215 		// juneau-bus.js is dependency-free and every bus consumer reads JuneauViews.bus, so it must precede them all.
216 		var bus = body.indexOf("juneau-bus.js");
217 		assertTrue(bus >= 0, () -> body);
218 		assertTrue(bus < body.indexOf("juneau-renders.js"), () -> body);
219 		assertTrue(bus < body.indexOf("juneau-views.js"), () -> body);
220 		assertTrue(bus < body.indexOf("juneau-regions.js"), () -> body);
221 		// It is the first toolkit script tag.
222 		var firstToolkitScript = body.lastIndexOf("<script", body.indexOf("data-toolkit-js"));
223 		assertTrue(firstToolkitScript >= 0 && body.indexOf("juneau-bus.js", firstToolkitScript) < body.indexOf(">", firstToolkitScript), () -> body);
224 	}

235 	// Every datatables card, client or server mode, pulls jQuery, DataTables and the glue, once each, before the runtime.
236 	private static void assertTableVendor(String path, String body) {
237 		var jq = body.indexOf("/jquery.min.js");
238 		var dt = body.indexOf("/js/dataTables.min.js");
239 		var glue = body.indexOf("juneau-datatables.js");
240 		var views = body.indexOf("juneau-views.js");
241 		assertTrue(jq >= 0 && jq < dt && dt < glue && glue < views, () -> path + ": " + body);
242 		assertTrue(body.contains("/css/dataTables.dataTables.min.css"), () -> path + ": " + body);
243 		for (var s : java.util.List.of("/jquery.min.js", "/js/dataTables.min.js", "juneau-datatables.js"))
244 			assertEquals(body.indexOf(s), body.lastIndexOf(s), () -> path + ": " + s + " emitted once: " + body);
245 	}
