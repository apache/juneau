17  package org.apache.juneau.console.freemarker;
18  
19  import static org.apache.juneau.bct.BctAssertions.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.util.*;
23  
24  import org.apache.juneau.*;
25  import org.apache.juneau.marshall.collections.*;
26  import org.apache.juneau.marshall.marshaller.*;
27  import org.apache.juneau.console.*;
28  import org.junit.jupiter.api.*;
29  
30  class Decorators_Test extends TestBase {
31  
32  	private static Object j(String json5) { return Json5.to(json5, Map.class); }
33  
34  	@Test void a01_fn_ref() {
35  		assertEquals(j("{'$fn':'compact'}"), Fn.ref("compact"));
36  		assertEquals(j("{'$fn':'template',args:{text:'{name}'}}"), Fn.ref("template", JsonMap.of("text", "{name}")));
37  		assertEquals(j("{'$fn':'app.x'}"), Fn.ref("app.x"));
38  	}
39  
40  	@Test void a02_fn_badName_isE74() {
41  		var e = assertThrows(IllegalArgumentException.class, () -> Fn.ref("Bad"));
42  		assertString("E-74: 'Bad' is not a valid $fn name", e.getMessage());
43  		assertThrows(IllegalArgumentException.class, () -> Fn.ref(null));
44  	}
45  
46  	@Test void a03_markers() {
47  		assertEquals(j("{use:'markers',src:'/rest/releases',at:'date',label:'name',topic:'app.release'}"),
48  			Markers.from("/rest/releases").at("date").label("name").onClick(Topics.custom("app", "release")).toJson());
49  		assertEquals(j("{use:'markers',values:[{at:'2026-10-01',label:'Freeze'}],style:'dashed'}"),
50  			Markers.values(List.of(Map.of("at", "2026-10-01", "label", "Freeze"))).style("dashed").toJson());
51  		assertEquals(j("{use:'markers',source:'data:releases',at:'date',label:'name'}"),
52  			Markers.source(Topics.data("releases")).at("date").label("name").toJson());
53  		assertEquals(j("{use:'markers',now:{label:'Today'}}"), Markers.now("Today").toJson());
54  	}
55  
56  	@Test void a04_markers_rejects() {
57  		assertThrows(IllegalArgumentException.class, () -> Markers.from("https://evil/x"));
58  		assertThrows(IllegalArgumentException.class, () -> Markers.source("filter:x"));
59  		var target = Markers.now("x");
60  		assertThrows(IllegalArgumentException.class, () -> target.style("wavy"));
61  	}
62  
63  	@Test void a05_bands() {
64  		assertEquals(j("{use:'bands',src:'/rest/outages',start:'from',end:'to',label:'Outage'}"),
65  			Bands.from("/rest/outages").start("from").end("to").label("Outage").toJson());
66  		assertEquals(j("{use:'bands',auto:'gaps',label:'No data'}"), Bands.gaps().label("No data").toJson());
67  		assertEquals(j("{use:'bands',auto:'gaps',minGap:2}"), Bands.gaps().minGap(2).toJson());
68  		var target2 = Bands.gaps();
69  		assertThrows(IllegalArgumentException.class, () -> target2.minGap(0));
70  	}
71  
72  	@Test void a06_thresholds() {
73  		assertEquals(j("{use:'thresholds',values:[{y:500,label:'SLO',tone:'danger'}]}"),
74  			Thresholds.at(500).label("SLO").tone("danger").toJson());
75  		var target3 = Thresholds.at(1);
76  		assertThrows(IllegalArgumentException.class, () -> target3.tone("pink"));
77  	}
78  
79  	@Test void a07_annotations() {
80  		assertEquals(j("{use:'annotations',presets:['max','min'],label:'{name}'}"),
81  			Annotations.max().withMin().label("{name}").toJson());
82  		assertEquals(j("{use:'annotations',values:[{x:'Mon',y:3,label:'Peak'}]}"),
83  			Annotations.points(List.of(Map.of("x", "Mon", "y", 3, "label", "Peak"))).toJson());
84  	}
85  
86  	@Test void a08_toJson_isACopy() {
87  		var m = Markers.now("x");
88  		m.toJson().put("use", "other");
89  		assertEquals("markers", m.toJson().get("use"));
90  	}
91  }
