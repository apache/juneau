17  package org.apache.juneau.console.freemarker;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.*;
24  import org.apache.juneau.console.*;
25  import org.apache.juneau.console.views.*;
26  import org.junit.jupiter.api.*;

28  /**
29   * Guards byte-identical contract output for builders that do not use the model API: no {@code model} key,
30   * {@code modelVersion}, {@code scripts} or {@code allowInlineHandlers} may appear.
31   */
32  class Golden_NoModelJson_Test extends TestBase {
33  
34  	private static TableSpec table() {
35  		return TableSpec.create("t1").dataUrl("/api/t").columns(Column.create("id")).detail(RowDetail.create("/d/{id}").title("T {id}"));
36  	}
37  
38  	@Test void a01_card() {
39  		var m = CardSpec.html("c1").title("X").src("/x").publishes().subscribes(Subscription.to("selection:t1").as("params")).toMap();
40  		assertEquals("{id=c1, type=html, title=X, src=/x, subscribes=[{\"topic\":\"selection:t1\",\"as\":\"params\"}]}", m.toString());
41  	}
42  
43  	@Test void a02_table() {
44  		assertEquals("{\"dataUrl\":\"/api/t\",\"columns\":[{\"key\":\"id\",\"label\":\"id\"}],\"detail\":{\"endpoint\":\"/d/{id}\",\"title\":\"T {id}\"}}", table().toCardBody().toString());
45  	}
46  
47  	@Test void a03_detail() {
48  		assertEquals("{\"endpoint\":\"/d/{id}\",\"title\":\"T\",\"icon\":\"i\"}", RowDetail.create("/d/{id}").title("T").icon("i").toMap().toString());
49  	}
50  
51  	@Test void a04_pageContract() throws Exception {
52  		var cap = new PageCapture();
53  		cap.title("T");
54  		cap.addCard(CardSpec.html("c1").title("X").src("/x"), null);
55  		assertEquals("{\"contractVersion\":\"1\",\"title\":\"T\",\"nav\":[],\"activeNav\":[],\"cards\":[{\"id\":\"c1\",\"type\":\"html\",\"title\":\"X\",\"src\":\"/x\"}]}", cap.toContractJson());
56  	}
57  
58  	@Test void a05_modelKeyInsideAPayloadIsNotAModelBlock() throws Exception {
59  		var cap = new PageCapture();
60  		cap.title("T");
61  		var payload = Map.of("model", "legacy", "type", "handler", "handler", "h");
62  		cap.addCard(CardSpec.of("datatables", "t1").body("table", Map.of("dataUrl", "/api/t",
63  			"ribbon", List.of(Map.of("type", "publish", "id", "p", "topic", "x", "payload", payload)))), null);
64  		var json = cap.toContractJson();
65  		assertFalse(json.contains("modelVersion"), json);
66  	}
67  }
