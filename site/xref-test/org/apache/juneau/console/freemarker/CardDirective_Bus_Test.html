89  	static String ok(String path) throws Exception {
90  		try (var c = MockRestClient.buildLax(Host.class); var rsp = c.get(path).run()) {
91  			rsp.assertStatus(200);
92  			return rsp.getContent().asString();
93  		}
94  	}
95  
96  	static void fails(String path, String message) throws Exception {
97  		try (var c = MockRestClient.buildLax(Host.class); var rsp = c.get(path).run()) {
98  			rsp.assertStatus(500);
99  			var body = rsp.getContent().asString();
100 			assertTrue(body.contains(message), () -> body);
101 		}
102 	}

104 	@Test void a01_attributes_shorthandAndHash() throws Exception {
105 		var a = PageContractAssert.assertPage(ok("/wiring"))
106 			.publishes("focus", "ssc.focus")
107 			.publishes("focus", "ssc.hot")
108 			.subscribes("detail", "ssc.focus", "refresh")
109 			.subscribes("hot", "ssc.hot", "refresh")
110 			.hasNoWiringErrors();
111 		assertEquals("[{topic:'ssc.focus',retain:true},{topic:'ssc.hot',retain:false}]",
112 			Json5.DEFAULT.write(a.card("focus").getList("publishes")));
113 		assertEquals("[{topic:'ssc.hot',as:'refresh',whenEmpty:'keep',emptyText:'Nothing hot.'}]",
114 			Json5.DEFAULT.write(a.card("hot").getList("subscribes")));
115 	}
