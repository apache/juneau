33  /**
34   * Placing PageSpec cards from a page template with {@code <@card ref>}: placement and append order, and the
35   * already-placed, unknown-card and mixed-attribute rejections.
36   *
37   * @since 10.0.0
38   */
39  @SuppressWarnings({
40  	"resource" // MockRestClient/RestResponse are closed in try-with-resources; fluent assertStatus returns this.
41  })
42  class CardRef_Test extends TestBase {
43  
44  	@Rest(mixins=FreemarkerMixin.class, renderResponseStackTraces="true")
45  	public static class Host extends BasicRestServlet {
46  		private static final long serialVersionUID = 1L;
47  		@Bean public FreemarkerMixin freemarker() {
48  			return ConsoleFreemarkerMixin.create().basePath("/templates/")
49  				.chromeTemplate("admin/console-chrome-bare.ftlh").build();
50  		}
51  		private static PageSpec spec(String template) {
52  			return PageSpec.create().template(template).html("a", "<p>a</p>").html("b", "<p>b</p>").html("c", "<p>c</p>");
53  		}
54  		@RestGet(path="/combo")
55  		public View combo(RestRequest req) { return spec("admin/card-ref-combo.ftlh").view(req); }
56  		@RestGet(path="/local-only")
57  		public View localOnly(RestRequest req) { return spec("admin/card-ref-local-only.ftlh").view(req); }
58  		@RestGet(path="/no-template")
59  		public View noTemplate(RestRequest req) {
60  			return PageSpec.create().html("a", "<p>a</p>").html("b", "<p>b</p>").html("c", "<p>c</p>").view(req);
61  		}
62  		@RestGet(path="/duplicate")
63  		public View duplicate(RestRequest req) { return spec("admin/card-ref-duplicate.ftlh").view(req); }
64  		@RestGet(path="/unknown")
65  		public View unknown(RestRequest req) {
66  			return PageSpec.create().html("x", "<p>x</p>").html("y", "<p>y</p>")
67  				.template("admin/card-ref-unknown.ftlh").view(req);
68  		}
69  		@RestGet(path="/withattr")
70  		public View withAttr(RestRequest req) { return spec("admin/card-ref-withattr.ftlh").view(req); }
71  		@RestGet(path="/withbody")
72  		public View withBody(RestRequest req) { return spec("admin/card-ref-withbody.ftlh").view(req); }
73  		@RestGet(path="/empty")
74  		public View empty(RestRequest req) { return spec("admin/card-ref-empty.ftlh").view(req); }
75  		@RestGet(path="/blankbody")
76  		public View blankBody(RestRequest req) { return spec("admin/card-ref-blankbody.ftlh").view(req); }
77  		@RestGet(path="/outside")
78  		public View outside(RestRequest req) { return spec("admin/card-ref-outside.ftlh").view(req); }
79  		@RestGet(path="/nested")
80  		public View nested(RestRequest req) { return spec("admin/card-ref-nested.ftlh").view(req); }
81  		@RestGet(path="/macro")
82  		public View macro(RestRequest req) { return spec("admin/card-ref-macro.ftlh").view(req); }
83  		@RestGet(path="/manyattrs")
84  		public View manyAttrs(RestRequest req) { return spec("admin/card-ref-manyattrs.ftlh").view(req); }
85  	}
86  
87  	private static String get(String path, int status) throws Exception {
88  		try (var c = MockRestClient.buildLax(Host.class); var rsp = c.get(path).run()) {
89  			rsp.assertStatus(status);
90  			return rsp.getContent().asString();
91  		}
92  	}
93  
94  	private static List<String> order(String path) throws Exception {
95  		var ids = new ArrayList<String>();
96  		for (var o : assertPage(get(path, 200)).isValid().contract().getList("cards"))
97  			ids.add(String.valueOf(((Map<?,?>)o).get("id")));
98  		return ids;
99  	}

101 	@Test void a01_refPlacesBuiltCardBetweenAuthoredCards_restAppended() throws Exception {
102 		assertEquals(List.of("x1", "b", "x2", "a", "c"), order("/combo"));
103 	}

105 	@Test void a02_authoredCardsOnly_builtCardsAppendedInDeclarationOrder() throws Exception {
106 		assertEquals(List.of("local", "a", "b", "c"), order("/local-only"));
107 	}

110 		assertEquals(List.of("a", "b", "c"), order("/no-template"));
