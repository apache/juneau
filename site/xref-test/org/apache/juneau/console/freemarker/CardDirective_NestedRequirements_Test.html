15  import static org.junit.jupiter.api.Assertions.*;
16  
17  import java.util.*;
18  
19  import org.apache.juneau.*;
20  import org.apache.juneau.console.*;
21  import org.junit.jupiter.api.*;
22  
23  /**
24   * The toolkit packs a container card records for the cards nested inside it.
25   */
26  class CardDirective_NestedRequirements_Test extends TestBase {
27  
28  	private static final CardDirectiveModel MODEL = new CardDirectiveModel(
29  		CardRequirements.create().build(new ToolkitPackRegistry()), CardTypeRegistry.standard());
30  
31  	private static List<String> packs(String id, String type, Map<String,Object> body) throws Exception {
32  		var cap = new PageCapture();
33  		MODEL.recordRequirements(cap, type, id, List.of(), body);
34  		return List.copyOf(cap.requiredPacks());
35  	}
36  
37  	@Test void a01_switcherHoldingAChart_recordsTheChartPackBeforeTheSwitcherPack() throws Exception {
38  		var body = CardTypeRegistry.standard().toCard(CardSource.create("switcher", "sw").body(
39  			"{views:[{id:'a',label:'A',card:{id:'c1',type:'chart',src:'/x',kind:'line',x:{field:'d'},series:['n']}}]}").build());
40  		var p = packs("sw", "switcher", body);
41  		assertTrue(p.contains("charts"), p.toString());
42  		assertTrue(p.contains("switcher"), p.toString());
43  		assertTrue(p.indexOf("charts") < p.indexOf("switcher"), p.toString());
44  	}
45  
46  	@Test void a02_rawUnliftedNestedMaps_stillRecordEveryNestedPack() throws Exception {
47  		// Maps as an app would hand them over: no lifting, a nested table still carrying its dataUrl at the top.
48  		var chart = map("id", "c1", "type", "chart", "src", "/x", "kind", "line");
49  		var table = map("id", "d1", "type", "datatables", "dataMode", "client", "dataUrl", "/x");
50  		var views = List.of(
51  			map("id", "a", "label", "A", "card", chart),
52  			map("id", "t", "label", "T", "sections", List.of(map("id", "s1", "label", "S1", "card", table))));
53  		var p = packs("sw", "switcher", map("views", views));
54  		assertTrue(p.contains("charts"), p.toString());
55  		assertTrue(p.contains("datatables-glue"), p.toString());
56  		assertTrue(p.indexOf("charts") < p.indexOf("switcher"), p.toString());
57  		assertTrue(p.indexOf("datatables-glue") < p.indexOf("switcher"), p.toString());
58  	}
59  
60  	private static Map<String,Object> map(Object...kv) {
61  		var m = new LinkedHashMap<String,Object>();
62  		for (var i = 0; i < kv.length; i += 2)
63  			m.put((String)kv[i], kv[i + 1]);
64  		return m;
65  	}
66  }
