17  package org.apache.juneau.console.freemarker;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.apache.juneau.*;
22  import org.junit.jupiter.api.*;
23  
24  class PageCapture_BodyAttrs_Test extends TestBase {
25  
26  	private static PageCapture cap(String attrs) throws Exception {
27  		return new PageCapture().mergeBodyAttrs(attrs);
28  	}
29  
30  	private static String strip(String attrs) throws Exception {
31  		return cap(attrs).bodyAttrsWithout("data-juneau-csrf");
32  	}
33  
34  	@Test void a01_doubleQuoted() throws Exception {
35  		assertEquals("data-k=\"v\"", strip("data-juneau-csrf=\"tok\" data-k=\"v\""));
36  	}
37  
38  	@Test void a02_singleQuoted() throws Exception {
39  		assertEquals("data-k=\"v\"", strip("data-juneau-csrf='tok' data-k=\"v\""));
40  	}
41  
42  	@Test void a03_unquoted() throws Exception {
43  		assertEquals("data-k=\"v\"", strip("data-juneau-csrf=tok data-k=\"v\""));
44  	}
45  
46  	@Test void a04_upperCaseName() throws Exception {
47  		assertEquals("data-k=\"v\"", strip("DATA-JUNEAU-CSRF=\"tok\" data-k=\"v\""));
48  	}
49  
50  	@Test void a05_whitespaceAroundEquals() throws Exception {
51  		assertEquals("data-k=\"v\"", strip("data-juneau-csrf = 'tok' data-k=\"v\""));
52  	}
53  
54  	@Test void a06_headerNameKept() throws Exception {
55  		assertEquals("data-juneau-csrf-header=\"X-H\"", strip("data-juneau-csrf=\"tok\" data-juneau-csrf-header=\"X-H\""));
56  	}
57  
58  	@Test void a07_prefixedNameKept() throws Exception {
59  		assertEquals("x-data-juneau-csrf=\"keep\"", strip("x-data-juneau-csrf=\"keep\" data-juneau-csrf=tok"));
60  	}
61  
62  	@Test void a08_valuelessKept() throws Exception {
63  		assertEquals("data-juneau-csrf data-k=\"v\"", strip("data-juneau-csrf data-k=\"v\""));
64  	}
65  
66  	@Test void a09_valueWithGtAndSpacesInQuotes() throws Exception {
67  		assertEquals("data-k=\"v\"", strip("data-juneau-csrf='a > b' data-k=\"v\""));
68  	}
69  
70  	@Test void a10_hasBodyAttr_allFormsAndCase() throws Exception {
71  		assertTrue(cap("data-juneau-csrf=\"t\"").hasBodyAttr("data-juneau-csrf"));
72  		assertTrue(cap("data-juneau-csrf='t'").hasBodyAttr("data-juneau-csrf"));
73  		assertTrue(cap("data-juneau-csrf=t").hasBodyAttr("data-juneau-csrf"));
74  		assertTrue(cap("Data-Juneau-Csrf = 't'").hasBodyAttr("data-juneau-csrf"));
75  		assertFalse(cap("data-juneau-csrf-header='t'").hasBodyAttr("data-juneau-csrf"));
76  		assertFalse(cap("x-data-juneau-csrf='t'").hasBodyAttr("data-juneau-csrf"));
77  		assertFalse(cap("data-juneau-csrf").hasBodyAttr("data-juneau-csrf"));
78  	}
79  
80  	@Test void a11_noAttrs() {
81  		assertNull(new PageCapture().bodyAttrsWithout("data-juneau-csrf"));
82  	}
83  
84  	@Test void a12_nameInsideAnotherValueIsNotAnAttribute() throws Exception {
85  		assertEquals("title=\"data-juneau-csrf=x\" data-k=\"v\"", strip("title=\"data-juneau-csrf=x\" data-k=\"v\""));
86  		assertEquals("title='a data-juneau-csrf=x'", strip("title='a data-juneau-csrf=x'"));
87  		assertEquals("title=\"data-juneau-csrf=x\"", strip("data-juneau-csrf=tok title=\"data-juneau-csrf=x\""));
88  	}
89  
90  	@Test void a13_hasBodyAttr_ignoresNameInsideAnotherValue() throws Exception {
91  		assertFalse(cap("title=\"data-juneau-csrf=x\"").hasBodyAttr("data-juneau-csrf"));
92  	}
93  }
