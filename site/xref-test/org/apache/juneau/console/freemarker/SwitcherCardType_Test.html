17  package org.apache.juneau.console.freemarker;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.*;
24  import org.apache.juneau.console.*;
25  import org.junit.jupiter.api.*;
26  
27  class SwitcherCardType_Test extends TestBase {
28  
29  	private static final CardTypeRegistry REG = CardTypeRegistry.standard();
30  
31  	private static CardSource src(String body) {
32  		return CardSource.create("switcher", "sw").body(body).build();
33  	}
34  
35  	private static final String OK = "{views:[{id:'a',label:'A',card:{id:'c1',type:'chart',src:'/x',kind:'line',x:{field:'d'},series:['n']}},"
36  		+ "{id:'t',label:'T',sections:[{id:'s1',label:'S1',card:{id:'d1',type:'datatables',dataMode:'client',dataUrl:'/x',columns:[{key:'k',label:'K'}]}}]}]}";
37  
38  	@Test void a01_type() {
39  		assertEquals("switcher", new SwitcherCardType().type());
40  	}
41  
42  	@Test void a02_registeredAndReserved() {
43  		assertTrue(REG.isRegistered("switcher"));
44  		assertTrue(CardTypeRegistry.RESERVED.contains("switcher"));
45  	}
46  
47  	@Test void a03_customHandlerCannotClaimSwitcher() {
48  		var b = REG.copy();
49  		var imposter = new CardTypeHandler() {
50  			@Override public String type() { return "switcher"; }
51  			@Override public org.apache.juneau.marshall.collections.JsonMap toFragment(CardSource s) { return new org.apache.juneau.marshall.collections.JsonMap(); }
52  		};
53  		var e = assertThrows(IllegalArgumentException.class, () -> b.add(imposter));
54  		assertTrue(e.getMessage().startsWith("Card type 'switcher' is reserved"), e.getMessage());
55  	}
56  
57  	@Test void a04_toCard_keepsViews() {
58  		var card = REG.toCard(src(OK));
59  		assertEquals("switcher", card.get("type"));
60  		assertEquals(2, ((List<?>)card.get("views")).size());
61  	}
62  
63  	@Test void a05_nestedCards_bothForms() {
64  		var card = REG.toCard(src(OK));
65  		var ids = new SwitcherCardType().nestedCards(card).stream().map(m -> m.get("id")).toList();
66  		assertEquals(List.of("c1", "d1"), ids);
67  	}
68  
69  	@Test void a06_nestedDatatablesLiftedToTable() {
70  		var card = REG.toCard(src(OK));
71  		var d = new SwitcherCardType().nestedCards(card).get(1);
72  		assertEquals("datatables", d.get("type"));
73  		assertEquals("d1", d.get("id"));
74  		var table = (Map<?,?>)d.get("table");
75  		assertNotNull(table);
76  		assertEquals("/x", table.get("dataUrl"));
77  		assertFalse(d.containsKey("dataUrl"));
78  	}
79  
80  	@Test void a07_authoredBodyNotMutated() {
81  		var s = src(OK);
82  		REG.toCard(s);
83  		var again = new SwitcherCardType().toFragment(s);
84  		var d = new SwitcherCardType().nestedCards(again).get(1);
85  		assertEquals("/x", d.get("dataUrl"));
86  	}
87  
88  	@Test void a08_invalid() {
89  		var empty = src("{views:[]}");
90  		var badId = src("{views:[{id:'a.b',label:'A',card:{id:'c',type:'html'}}]}");
91  		var nested = src("{views:[{id:'a',label:'A',card:{id:'c',type:'switcher'}}]}");
92  		var e = assertThrows(IllegalArgumentException.class, () -> REG.toCard(empty));
93  		assertEquals("<@card id='sw'> SwitcherDef 'sw' requires at least one view.", e.getMessage());
94  		e = assertThrows(IllegalArgumentException.class, () -> REG.toCard(badId));
95  		assertTrue(e.getMessage().contains("view id 'a.b' must match"), e.getMessage());
96  		e = assertThrows(IllegalArgumentException.class, () -> REG.toCard(nested));
97  		assertTrue(e.getMessage().contains("cannot hold a switcher"), e.getMessage());
98  	}
99  
100 	@Test void a09_nestedBadSrcRejectedByItsOwnHandler() {
101 		var bad = src(
102 			"{views:[{id:'a',label:'A',card:{id:'c1',type:'chart',src:'http://evil/x',kind:'line',x:{field:'d'},series:['n']}}]}");
103 		var e = assertThrows(IllegalArgumentException.class, () -> REG.toCard(bad));
104 		assertTrue(e.getMessage().contains("same-origin"), e.getMessage());
105 	}
106 
107 	@Test void a10_requiresJsonBody() {
108 		var plain = src("plain");
109 		var e = assertThrows(IllegalArgumentException.class, () -> REG.toCard(plain));
110 		assertTrue(e.getMessage().contains("type='switcher' requires a JSON5 body."), e.getMessage());
111 	}
112 
113 	@Test void style_a01_jsonBodyAccepted() {
114 		var card = REG.toCard(src("{style:'slider'," + OK.substring(1)));
115 		assertEquals("slider", card.get("style"));
116 	}
117 
118 	@Test void style_a02_jsonBodyBadValueSameMessage() {
119 		var e = assertThrows(IllegalArgumentException.class, () -> REG.toCard(src("{style:'pills'," + OK.substring(1))));
120 		assertTrue(e.getMessage().contains("SwitcherDef 'sw' style 'pills' must be one of tabs, slider."), e.getMessage());
121 	}

123 	private static final String STACKED = "{views:[{id:'a',label:'A',cards:[{id:'c1',type:'chart',src:'/x',kind:'line',x:{field:'d'},series:['n']},"
124 		+ "{id:'d1',type:'datatables',dataMode:'client',dataUrl:'/x',columns:[{key:'k',label:'K'}]}]}]}";
125 
126 	@Test void cards_a01_jsonBodyLifted() {
127 		var card = REG.toCard(src(STACKED));
128 		var v0 = (Map<?,?>)((List<?>)card.get("views")).get(0);
129 		assertEquals(2, ((List<?>)v0.get("cards")).size());
130 	}
131 
132 	@Test void cards_a02_nestedCardsWalked() {
133 		var card = REG.toCard(src(STACKED));
134 		var ids = new SwitcherCardType().nestedCards(card).stream().map(m -> m.get("id")).toList();
135 		assertEquals(List.of("c1", "d1"), ids);
136 	}
