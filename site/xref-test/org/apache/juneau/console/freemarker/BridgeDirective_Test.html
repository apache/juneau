32  /**
33   * Tests for the {@code <@bridge>} directive: lowering to the contract, session resolution and the E-50/E-51 rejections.
34   *
35   * @since 10.0.0
36   */
37  @SuppressWarnings({
38  	"resource" // MockRestClient/RestResponse are closed in try-with-resources; fluent assertStatus returns this.
39  })
40  class BridgeDirective_Test extends TestBase {
41  
42  	@Rest(mixins=FreemarkerMixin.class, renderResponseStackTraces="true")
43  	public static class Host extends BasicRestServlet {
44  		private static final long serialVersionUID = 1L;
45  		@Bean public FreemarkerMixin freemarker() {
46  			return ConsoleFreemarkerMixin.create().basePath("/templates/").chromeTemplate("admin/console-chrome-bare.ftlh").build();
47  		}
48  		@RestGet(path="/sse") public View r0() { return FreemarkerView.of("admin/bus-bridge.ftlh"); }
49  		@RestGet(path="/ws") public View r1() { return FreemarkerView.of("admin/bus-bridge-ws.ftlh"); }
50  		@RestGet(path="/crossorigin") public View r2() { return FreemarkerView.of("admin/bus-bridge-crossorigin.ftlh"); }
51  		@RestGet(path="/transport") public View r3() { return FreemarkerView.of("admin/bus-bridge-transport.ftlh"); }
52  		@RestGet(path="/maxattempts") public View r4() { return FreemarkerView.of("admin/bus-bridge-maxattempts.ftlh"); }
53  		@RestGet(path="/outside") public View r5() { return FreemarkerView.of("admin/bus-bridge-outside.ftlh"); }
54  	}
55  
56  	static String ok(String path) throws Exception {
57  		try (var c = MockRestClient.buildLax(Host.class); var rsp = c.get(path).run()) {
58  			rsp.assertStatus(200);
59  			return rsp.getContent().asString();
60  		}
61  	}
62  
63  	static void fails(String path, String message) throws Exception {
64  		try (var c = MockRestClient.buildLax(Host.class); var rsp = c.get(path).run()) {
65  			rsp.assertStatus(500);
66  			var body = rsp.getContent().asString();
67  			assertTrue(body.contains(message), () -> body);
68  		}
69  	}
70  
71  	@Test void a01_sse_sessionResolved() throws Exception {
72  		var a = PageContractAssert.assertPage(ok("/sse")).hasBridge("ops").declaresTopic("ops.jobs").hasNoWiringErrors();
73  		var b = (org.apache.juneau.marshall.collections.JsonMap)a.contract().getList("bridges").get(0);
74  		assertEquals("sse", b.getString("transport"));
75  		assertEquals("[\"ops.jobs\"]", b.getList("downstream").toString());
76  		var session = b.getString("session");
77  		assertFalse(session.startsWith("servlet:"), session);
78  		assertTrue(session.startsWith("/") && session.endsWith("/juneau-bus/session"), session);
79  		assertFalse(b.containsKey("upstream"));
80  	}
81  
82  	@Test void a02_websocket_listsAndMaxAttempts() throws Exception {
83  		var b = PageContractAssert.assertPage(ok("/ws")).hasNoWiringErrors().contract().getList("bridges").get(0);
84  		assertEquals("{id:'ops',transport:'websocket',session:'/rest/ops/juneau-bus/session',downstream:['ops.jobs'],upstream:['ops.cancel-all'],maxAttempts:5}",
85  			Json5.DEFAULT.write(b));
86  	}
87  
88  	@Test void b01_crossOrigin_isE51() throws Exception {
89  		fails("/crossorigin", "bridge 'ops': session 'https://elsewhere.example/s' must be a same-origin path or a servlet:/context: URI");
90  	}
91  
92  	@Test void b02_badTransport_isE51() throws Exception {
93  		fails("/transport", "bridge 'ops': transport 'poll' must be sse or websocket");
94  	}
95  
96  	@Test void b03_badMaxAttempts_isE51() throws Exception {
97  		fails("/maxattempts", "bridge 'ops': maxAttempts must be an integer; got 'many'");
98  	}
99  
100 	@Test void b04_outside_isE50() throws Exception {
101 		fails("/outside", "<@bridge> must be inside <@console> or <@page>");
102 	}
103 }
