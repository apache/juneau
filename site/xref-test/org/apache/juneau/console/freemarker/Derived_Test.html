28  class Derived_Test extends TestBase {
29  
30  	@Test void a01_shape() {
31  		var d = Derived.from("orgRequests")
32  			.transform("filter", JsonMap.of("field", "status", "in", List.of("{reasons}")))
33  			.transform("bucket", JsonMap.of("field", "start", "unit", "auto"))
34  			.params(Subscription.to(Topics.filter("reasons")).as("params").map("reasons", "columns.reason"))
35  			.as("trendBuckets");
36  		assertEquals("trendBuckets", d.id());
37  		assertEquals("{\"id\":\"trendBuckets\",\"kind\":\"derived\",\"from\":\"orgRequests\",\"transforms\":["
38  			+ "{\"use\":\"filter\",\"args\":{\"field\":\"status\",\"in\":[\"{reasons}\"]}},"
39  			+ "{\"use\":\"bucket\",\"args\":{\"field\":\"start\",\"unit\":\"auto\"}}],"
40  			+ "\"subscribes\":[{\"topic\":\"filter:reasons\",\"as\":\"params\","
41  			+ "\"map\":{\"reasons\":\"columns.reason\"}}]}",
42  			d.toMap().toString());
43  	}
44  
45  	@Test void a02_errors() {
46  		var target = Derived.from("a");
47  		var e = assertThrows(IllegalArgumentException.class, () -> target.transform("Bad", null));
48  		assertEquals("E-74: 'Bad' is not a valid transform name", e.getMessage());
49  		var target2 = Derived.from("a");
50  		var s = assertThrows(IllegalStateException.class, target2::toMap);
51  		assertEquals("Derived.from('a'): as(id) not called", s.getMessage());
52  		var target3 = Derived.from("a");
53  		e = assertThrows(IllegalArgumentException.class, () -> target3.as("a"));
54  		assertEquals("E-76: data source 'a': a derived source cannot read its own topic", e.getMessage());
55  		var target4 = Derived.from("a")
56  			.params(Subscription.to("app.x").as("refresh")).as("b");
57  		e = assertThrows(IllegalArgumentException.class, target4::toMap);
58  		assertEquals("E-76: data source 'b': subscription to 'app.x' has role 'refresh'; "
59  			+ "a derived source accepts params", e.getMessage());
60  		assertThrows(IllegalArgumentException.class, () -> Derived.from("bad id"));
61  	}
62  
63  	@Test void a03_copies() {
64  		var args = JsonMap.of("field", "x");
65  		var d = Derived.from("a").transform("filter", args).as("b");
66  		args.put("field", "changed");
67  		var m = d.toMap();
68  		assertEquals("{\"field\":\"x\"}", ((JsonMap)((List<?>)m.get("transforms")).get(0)).get("args").toString());
69  		((JsonMap)((List<?>)m.get("transforms")).get(0)).put("use", "zzz");
70  		assertEquals("filter", ((JsonMap)((List<?>)d.toMap().get("transforms")).get(0)).get("use"));
71  		var target5 = Derived.from("a");
72  		var e = assertThrows(IllegalArgumentException.class, () -> target5.params((Subscription)null));
73  		assertEquals("E-76: data source 'a': subscription is null", e.getMessage());
74  	}
75  
76  	@Test void a99_idsMustMatchCardIdGrammar() {
77  		var msg = "E-76: data source '%s': id must match ^[A-Za-z][A-Za-z0-9_-]{0,63}$";
78  		assertEquals(msg.formatted("1st"), assertThrows(IllegalArgumentException.class, () -> Derived.from("1st"))
79  			.getMessage());
80  		var target6 = Derived.from("a");
81  		assertEquals(msg.formatted("a.b"),
82  			assertThrows(IllegalArgumentException.class, () -> target6.as("a.b")).getMessage());
83  	}
