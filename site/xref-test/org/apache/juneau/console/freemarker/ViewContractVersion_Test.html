31  /**
32   * A pre-built SLOT_META body fails at render time with E-28; the static {@code view-contract-version} validator rule
33   * still flags a stale pinned version.
34   *
35   * @since 10.0.0
36   */
37  @SuppressWarnings({
38  	"resource" // Closeable resources in tests are intentionally unassigned; closing is handled by test infrastructure.
39  })
40  class ViewContractVersion_Test extends TestBase {
41  
42  	private static final String RUNTIME = ViewsMixin.CONTRACT_VERSION;
43  
44  	//-----------------------------------------------------------------------------------------------------------------
45  	// End to end through <@card type="datatables">
46  	//-----------------------------------------------------------------------------------------------------------------
47  
48  	@Rest(mixins=FreemarkerMixin.class, renderResponseStackTraces="true")
49  	public static class DevHost extends BasicRestServlet {
50  		private static final long serialVersionUID = 1L;
51  		@Bean public FreemarkerMixin freemarker() {
52  			return ConsoleFreemarkerMixin.create().devMode(true).basePath("/templates/").chromeTemplate("c1/chrome.ftlh").build();
53  		}
54  		@RestGet(path="/t/{name}")
55  		public View t(@Path("name") String name) {
56  			return FreemarkerView.of("c1/" + name + ".ftlh");
57  		}
58  	}
59  
60  	@Test void b03_render_prebuiltSlotMeta_failsWithE28() throws Exception {
61  		try (var c = MockRestClient.buildLax(DevHost.class); var rsp = c.get("/t/cv-pinned-stale").run()) {
62  			rsp.assertStatus(500);
63  			assertContainsAll(rsp.getContent().asString(), "c1/cv-pinned-stale.ftlh", "body is a pre-built SLOT_META envelope");
64  		}
65  	}
66  
67  	//-----------------------------------------------------------------------------------------------------------------
68  	// Static validator rule
69  	//-----------------------------------------------------------------------------------------------------------------
70  
71  	private static List<ConsoleTemplateValidator.Finding> lint(String source) {
72  		// Only the version rule is under test.
73  		return ConsoleTemplateValidator.create().validateSource("t.ftlh", source).stream().filter(x -> "view-contract-version".equals(x.rule())).toList();
74  	}
75  
76  	@Test void c01_validator_flagsStalePin() {
77  		var f = lint("<@page>\n<@card type=\"datatables\" id=\"t\">\n{ contractVersion:'1', quickStats:{contractVersion:'1'}, view:{ contractVersion:'4', id:'t' } }\n</@card>\n</@page>");
78  		assertSize(f::toString, 1, f);
79  		assertEquals("view-contract-version", f.get(0).rule());
80  		assertEquals(3, f.get(0).line(), "anchored on the stale pin inside the card body");
81  	}
82  
83  	@Test void c02_validator_acceptsOmittedAndMatchingAndNestedVersions() {
84  		assertTrue(lint("<@page>\n<@card type=\"datatables\" id=\"t\">\n{ contractVersion:'1', view:{ id:'t', detail:{contractVersion:'1'} } }\n</@card>\n</@page>").isEmpty());
85  		assertTrue(lint("<@page>\n<@card type=\"datatables\" id=\"t\">\n{ contractVersion:'1', view:{ contractVersion:'" + RUNTIME + "', id:'t' } }\n</@card>\n</@page>").isEmpty());
86  	}
87  }
