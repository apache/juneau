310 		var a = CardSpec.html("c").visibleWhen(VisibilityRule.when("a").present());

316 		assertSize(() -> "an empty other leaves the rules alone", 1, ((List<?>)a.toMap().get("visibleWhen")));

356 	private static int count(String body, String part) {
357 		var n = 0;
358 		for (var i = body.indexOf(part); i >= 0; i = body.indexOf(part, i + 1))
359 			n++;
360 		return n;
361 	}
