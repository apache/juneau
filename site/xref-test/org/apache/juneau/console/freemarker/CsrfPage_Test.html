31  /**
32   * A console page rendered behind a {@link CsrfBoundary} carries the token, so no cache may keep it.
33   *
34   * <p>
35   * The mock REST client never runs a servlet filter, so the page resource calls
36   * {@link RequestBoundaryFilter#doFilter(jakarta.servlet.ServletRequest, jakarta.servlet.ServletResponse,
37   * jakarta.servlet.FilterChain) doFilter} on the real filter itself, with a no-op chain, before it renders.  That
38   * publishes the same request attributes a deployed filter would.  What this proves is the page's own response
39   * headers; whether the filter blocks anything is proven elsewhere, over a mocked chain and a real socket.
40   *
41   * @since 10.0.0
42   */
43  @SuppressWarnings({
44  	"resource" // MockRestClient/RestResponse are closed in try-with-resources; fluent assertStatus returns this.
45  })
46  class CsrfPage_Test extends TestBase {
47  
48  	static final CsrfBoundary BOUNDARY = CsrfBoundary.create().build();
49  
50  	@Rest(mixins=FreemarkerMixin.class, renderResponseStackTraces="true")
51  	public static class PageHost extends BasicRestServlet {
52  		private static final long serialVersionUID = 1L;
53  		@Bean public FreemarkerMixin freemarker() {
54  			return ConsoleFreemarkerMixin.create()
55  				.basePath("/templates/")
56  				.chromeTemplate("admin/console-chrome.ftlh")
57  				.build();
58  		}
59  		@RestGet(path="/page")
60  		public View page(RestRequest req, RestResponse res) throws Exception {
61  			// An application default that a token-bearing page must not inherit.
62  			res.setHeader("Cache-Control", "public, max-age=600");
63  			new RequestBoundaryFilter(BOUNDARY).doFilter(req.getHttpServletRequest(), res.getHttpServletResponse(), (q, s) -> {});
64  			return FreemarkerView.of("admin/page-naked.ftlh");
65  		}
66  	}
67  
68  	@Test void a04_tokenBearingPageNeverCacheable() throws Exception {
69  		try (var c = MockRestClient.buildLax(PageHost.class);
70  			var rsp = c.get("/page").header("Accept", "text/html").header("Host", "app.example").run()) {
71  			rsp.assertStatus(200);
72  			var body = rsp.getContent().asString();
73  			assertTrue(body.matches("(?s).*<meta name=\"csrf-token\" content=\"[0-9a-f]{128}\".*"), body);
74  			assertTrue(body.matches("(?s).* data-juneau-csrf=\"[0-9a-f]{128}\".*"), body);
75  			var cacheControl = rsp.getHeaders("Cache-Control");
76  			assertEquals(1, cacheControl.length, () -> java.util.Arrays.toString(cacheControl));
77  			assertEquals("private, no-store", cacheControl[0].getValue());
78  			assertFalse(cacheControl[0].getValue().contains("public"));
79  			assertEquals(0, rsp.getHeaders("Expires").length);
80  		}
81  	}
82  }
