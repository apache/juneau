31  /**
32   * Runs the Java side of the shared model validation corpus ({@code model-validation-corpus.json} in the views module).
33   * The JS side (JSON mount and imperative calls) is {@code model.cjs}.  Here, every case that has a {@code build} step
34   * on an element is replayed against {@link ModelBlock}/{@link HandlerRef}/{@link ModelAction}, which must refuse it
35   * (the early, friendlier duplicate of the client rules), and every {@code schema} case must fail
36   * {@link PageContractSchema}.  Cases that only the client can judge (versions, sibling ids, opt-in) are listed in
37   * {@link #CLIENT_ONLY}; the opt-in case is Java-checked at page build in {@code ModelSerialization_Test}.
38   */
39  class ModelValidation_Corpus_Test extends TestBase {
40  
41  	private static final Path CORPUS = Path.of(System.getProperty("basedir", "."),
42  		"../juneau-console-views/src/test/resources/model-validation-corpus.json");
43  
44  	private static final Set<String> CLIENT_ONLY = Set.of("modelVersion", "contractVersion", "siblingIdUnique", "inlineWithoutOptIn",
45  		"subscriptionRole", "cardReservedKeys", "pageHeader");
46  
47  	private static String kindOfPath(String path) {
48  		if (path.endsWith("/table/detail") || path.endsWith("/detail"))
49  			return "rowdetail";
50  		if (path.contains("/tab:"))
51  			return "detailtab";
52  		if (path.contains("/item:"))
53  			return "ribbonitem";
54  		if (path.contains("pagetab:"))
55  			return "pagetab";
56  		if (path.endsWith("/table"))
57  			return "table";
58  		if (path.startsWith("page/") && ! path.contains("/card"))
59  			return path.indexOf('/', 5) < 0 ? "page" : "card";
60  		return "card";
61  	}
62  
63  	private static String kindOfCtor(String ctor) {
64  		return switch (ctor) {
65  			case "Row" -> "row";
66  			case "SwitcherView" -> "view";
67  			case "Chart" -> "chart";
68  			default -> null;
69  		};
70  	}
71  
72  	private static Object handlerOf(Object arg) {
73  		if (arg instanceof Map<?,?> m)
74  			return HandlerRef.inline((String)m.get("inline"));
75  		return HandlerRef.named((String)arg);
76  	}
77  
78  	@SuppressWarnings({
79  		"unchecked" // Casts a parsed JSON value to its known Map/List shape.
80  	})
81  	@TestFactory Stream<DynamicTest> a01_corpus() throws Exception {
82  		var corpus = JsonMap.ofString(Files.readString(CORPUS, UTF_8));
83  		return corpus.getList("cases").stream().map(o -> (JsonMap)o).map(c -> DynamicTest.dynamicTest(c.getString("name"), () -> {
84  			var name = c.getString("name");
85  			if (Boolean.TRUE.equals(c.get("schema"))) {
86  				var errs = PageContractSchema.get().validate(c.getMap("contract").toString());
87  				assertFalse(errs.isEmpty(), name + " must fail the schema");
88  				return;
89  			}
90  			if (CLIENT_ONLY.contains(name) || c.get("build") == null)
91  				return;
92  			var kinds = new HashMap<String,String>();
93  			var blocks = new HashMap<String,ModelBlock>();
94  			var last = "";
95  			var steps = (List<Map<String,Object>>)c.get("build");
96  			// Replay: each step must be accepted until the last one, which must be refused.
97  			for (var i = 0; i < steps.size(); i++) {
98  				var s = steps.get(i);
99  				var ctor = (String)s.get("ctor");
100 				if (ctor != null) {
101 					var k = kindOfCtor(ctor);
102 					if (k != null) {
103 						kinds.put("@" + s.get("as"), k);
104 						blocks.put("@" + s.get("as"), new ModelBlock(k, ctor));
105 					}
106 					continue;
107 				}
108 				var on = (String)s.get("on");
109 				var args = (List<Object>)s.get("args");
110 				var block = blocks.computeIfAbsent(on, p -> new ModelBlock(kinds.getOrDefault(p, kindOfPath(p)), p));
111 				Runnable r = switch ((String)s.get("call")) {
112 					case "bind" -> () -> block.on((String)args.get(0), (HandlerRef)handlerOf(args.get(1)));
113 					case "action" -> () -> block.action(ModelAction.of((String)args.get(0)).handler((String)args.get(2)));
114 					case "listen" -> () -> block.subscribe((String)args.get(0), (HandlerRef)handlerOf(args.get(1)));
115 					default -> null;
116 				};
117 				if (r == null)
118 					return;
119 				if (i < steps.size() - 1)
120 					r.run();
121 				else
122 					assertThrows(IllegalArgumentException.class, r::run, name + " must be refused by the Java builders");
123 			}
124 		}));
125 	}
