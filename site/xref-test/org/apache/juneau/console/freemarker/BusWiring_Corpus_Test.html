31  /**
32   * Runs the shared R-10 / R-11 corpus ({@code $VIEWS/src/test/resources/bus-wiring-corpus.json}) through
33   * {@link BusWiringValidator}, against the real registry: {@code html}, the ServiceLoader-registered
34   * {@code DatatablesCardType}, and the corpus's own {@code cardTypes} stubs.  The views module's {@code bus-wiring.cjs} runs
35   * the same cases through {@code JuneauViews.bus.wiring.validate} and checks {@code expectJs}.
36   */
37  class BusWiring_Corpus_Test extends TestBase {
38  
39  	private static final Path CORPUS = Path.of(System.getProperty("basedir", "."),
40  		"../juneau-console-views/src/test/resources/bus-wiring-corpus.json");
41  
42  	/** A corpus {@code cardTypes} entry; {@code null} roles/ops mean "unknown to Java". */
43  	private record CorpusType(String type, Set<String> roles, Set<String> ops, List<String> publishes) implements CardTypeHandler {
44  		@Override public JsonMap toFragment(CardSource source) { return new JsonMap(); }
45  		@Override public List<String> implicitTopics(JsonMap card) { return publishes; }
46  		@Override public Set<String> acceptedRoles() { return roles; }
47  		@Override public Set<String> acceptedOps() { return ops; }
48  	}
49  
50  	@SuppressWarnings({
51  		"unchecked" // Casts a parsed JSON value to its known Map/List shape.
52  	})
53  	private static Set<String> set(Object list) {
54  		return list == null ? null : Set.copyOf((List<String>)list);
55  	}
56  
57  	@SuppressWarnings({
58  		"unchecked" // Casts a parsed JSON value to its known Map/List shape.
59  	})
60  	@TestFactory Stream<DynamicTest> a01_corpus() throws Exception {
61  		var corpus = JsonMap.ofString(Files.readString(CORPUS, UTF_8));
62  		assertTrue(CardTypeRegistry.standard().isRegistered("datatables"), "DatatablesCardType is ServiceLoader-registered");
63  		var b = CardTypeRegistry.standard().copy();
64  		for (var e : corpus.getMap("cardTypes").entrySet()) {
65  			var t = (Map<String,Object>)e.getValue();
66  			b.add(new CorpusType(e.getKey(), set(t.get("roles")), set(t.get("ops")), (List<String>)t.get("publishes")));
67  		}
68  		var registry = b.build();
69  		return corpus.getList("cases").stream().map(o -> (JsonMap)o).map(c -> DynamicTest.dynamicTest(c.getString("name"), () -> {
70  			var expected = c.getList("expect").stream().map(o -> (Map<String,Object>)o)
71  				.map(m -> new BusWiringValidator.Problem((String)m.get("code"), (String)m.get("message"))).toList();
72  			assertEquals(expected, BusWiringValidator.validate(c.getMap("contract"), registry));
73  		}));
74  	}
75  
76  	@Test void a02_corpusCoversEveryJavaCode() throws Exception {
77  		var corpus = JsonMap.ofString(Files.readString(CORPUS, UTF_8));
78  		var codes = new TreeSet<String>();
79  		for (var c : corpus.getList("cases"))
80  			for (var p : ((JsonMap)c).getList("expect"))
81  				codes.add(((Map<?,?>)p).get("code").toString());
82  		assertEquals(new TreeSet<>(List.of("E-40", "E-41", "E-42", "E-43", "E-44", "E-45", "E-46", "E-47", "E-48", "E-49",
83  			"E-51", "E-52", "E-53", "E-54", "E-58", "E-71", "E-72", "E-73", "E-74", "E-76", "E-77")), codes, "E-50 is FTL-only");
84  	}
