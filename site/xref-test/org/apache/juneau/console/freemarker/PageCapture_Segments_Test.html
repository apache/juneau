31  /**
32   * Free markup in a {@code <@page>} body becomes ordered {@code jc-seg-N} bare html cards between the authored cards
33   *, and {@code <@card>} without {@code id=} gets {@code jc-card-N}.
34   *
35   * @since 10.0.0
36   */
37  class PageCapture_Segments_Test extends TestBase {
38  
39  	@Test void a01_segmentsKeepAuthoredOrder() {
40  		var a = assertPage(render("page-segments")).isValid().hasCardOrder("jc-seg-1", "a", "jc-seg-2", "b")
41  			.templateContains("jc-seg-1", "<p>s1</p>")
42  			.templateContains("jc-seg-2", "<p>s2</p>")
43  			.templateContains("a", "<p>A</p>");
44  		assertEquals(Boolean.TRUE, card(a, "jc-seg-1").get("bare"));
45  		assertNull(card(a, "a").get("bare"));
46  	}
47  
48  	@Test void a02_whitespaceBetweenCardsIsNotASegment() {
49  		assertPage(render("page-cards-only")).isValid().hasCardOrder("a", "b");
50  	}
51  
52  	@Test void a03_autoIds() {
53  		assertPage(render("page-autoid")).isValid().hasCardOrder("jc-card-1", "jc-seg-1", "jc-card-2")
54  			.templateContains("jc-card-1", "<p>one</p>")
55  			.templateContains("jc-card-2", "<p>two</p>");
56  	}
57  
58  	@Test void a04_title() {
59  		assertEquals("Totals", card(assertPage(render("page-card-title")).isValid(), "t").getString("title"));
60  	}
61  
62  	@Test void a05_srcOnly() {
63  		var c = card(assertPage(render("page-card-src")).isValid(), "s");
64  		assertEquals("/frag/s.html", c.getString("src"));
65  		assertNull(c.get("template"));
66  	}
67  
68  	@Test void a06_templateReference() {
69  		var a = assertPage(render("page-card-template-ref")).isValid();
70  		assertBeans(List.of(card(a, "again"), card(a, "body1")), "template", "body1", "body1");
71  		assertEquals(1, render("page-card-template-ref").split("<template data-card=", -1).length - 1);
72  	}
73  
74  	static JsonMap card(PageContractAssert a, String id) {
75  		for (var o : a.contract().getList("cards"))
76  			if (id.equals(((JsonMap)o).getString("id")))
77  				return (JsonMap)o;
78  		throw new AssertionError("no card '" + id + "'");
79  	}
80  }
