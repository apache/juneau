39  /**
40   * Proves a {@link PageSpec}-built page and the equivalent hand-written FTL page produce the same
41   * {@code #juneau-page} contract and the same {@code <template>} set, for every feature the page builder exposes.
42   *
43   * <p>
44   * Each case is one FTL fixture under {@code templates/parity/c3} and one {@link Cases} factory; both render through
45   * the shared {@code parity-chrome.ftlh}.  The comparison is {@link PageContractAssert#isTwinOf}: key-sorted JSON (list
46   * order is kept, because card, nav and template order is part of what parity must prove), then template ids and
47   * stripped template markup.
48   *
49   * <h5 class='section'>Example:</h5>
50   * <p class='bjava'>
51   * 	var <jv>ftl</jv> = <jsm>assertPage</jsm>(<jsm>renderFtl</jsm>(<js>"03-table-every-feature"</js>)).isValid();
52   * 	var <jv>java</jv> = <jsm>assertPage</jsm>(<jsm>renderSpec</jsm>(<js>"03-table-every-feature"</js>)).isValid();
53   * 	<jv>ftl</jv>.isTwinOf(<jv>java</jv>);
54   * </p>
55   *
56   * @since 10.0.0
57   */

118 	private static void assertParity(String ftlFixture, String specCase, Consumer<PageContractAssert> javaGuard) throws Exception {
119 		var ftl = assertPage(get("/ftl/" + ftlFixture)).isValid();
120 		var java = assertPage(get("/spec/" + specCase)).isValid();
121 		javaGuard.accept(java);  // Non-vacuity: the Java side really produced the thing being compared.
122 		try {
123 			ftl.isTwinOf(java);
124 		} catch (AssertionError e) {
125 			throw new AssertionError("case '" + specCase + "' (ftl vs java): " + e.getMessage(), e);
126 		}
127 	}

130 		assertParity(caseName, caseName, javaGuard);
131 	}

135 		return a -> {

137 			for (var k : tableKeys)

139 		};
140 	}

142 	/** Java templates carry no unresolved {@code ref=}, and the cards are in the stated order. */
143 	private static Consumer<PageContractAssert> placed(String... order) {
144 		return a -> {
145 			a.hasCardOrder(order);
146 			a.templates().forEach((k, v) -> assertFalse(java.util.regex.Pattern.compile("<@card[^>]*\bref=").matcher(v).find(), () -> "template '" + k + "' has an unresolved ref: " + v));
147 		};
148 	}

239 	private static List<String> toolkitTags(String html) {
240 		return html.lines().filter(l -> l.contains("data-toolkit-js") || l.contains("data-toolkit-css")).map(String::strip).toList();
