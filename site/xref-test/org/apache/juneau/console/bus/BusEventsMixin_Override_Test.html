21  import java.util.*;
22  import java.util.concurrent.atomic.*;

24  import org.apache.juneau.*;
25  import org.apache.juneau.marshall.collections.*;
26  import org.apache.juneau.marshall.marshaller.*;
27  import org.apache.juneau.rest.mock.classic.*;

29  import org.apache.juneau.rest.server.servlet.*;
30  import org.junit.jupiter.api.*;

32  /**
33   * An application guard on the session POST: an override of {@link BusEventsMixin#openBusSession} that calls
34   * {@code BusEventsMixin.super.openBusSession}.
35   *
36   * <p>
37   * Pins that re-declaring {@code @RestPost(path=SESSION_PATH)} on the override registers the route once, and that an
38   * override without the annotation still inherits it.
39   */
40  @SuppressWarnings({
41  	"resource" // The shared bus and MockRestClients live for the whole class; the bus is closed in @AfterAll.
42  })
43  class BusEventsMixin_Override_Test extends TestBase {
44  
45  	static final ServerBus BUS = ServerBus.create(BusPolicy.create().downstream("ops.jobs", true).build());
46  	static final AtomicInteger CALLS = new AtomicInteger();
47  	static final String BODY = "{\"v\":1,\"bridge\":\"ops\",\"transport\":\"sse\",\"downstream\":[\"ops.jobs\"]}";
48  
49  	/** Lists the Java methods behind every registered operation, as {@code DeclaringClass.method}. */
50  	static List<String> ops(RestRequest req) {
51  		return req.getContext().getRestOperations().getOpContexts().stream()
52  			.map(o -> o.getJavaMethod().getDeclaringClass().getSimpleName() + "." + o.getJavaMethod().getName())
53  			.toList();
54  	}
55  
56  	@Rest
57  	public static class Guarded extends BasicRestServlet implements BusEventsMixin {
58  		private static final long serialVersionUID = 1L;
59  
60  		@Override public ServerBus serverBus() { return BUS; }
61  
62  		@Override
63  		@RestPost(path=SESSION_PATH)
64  		public JsonMap openBusSession(RestRequest req, RestResponse res) {
65  			CALLS.incrementAndGet();
66  			if (! "yes".equals(req.getHttpServletRequest().getHeader("X-App-Guard"))) {
67  				res.setStatus(403);
68  				return JsonMap.of("code", "app:guard", "message", "refused by the application guard");
69  			}
70  			return BusEventsMixin.super.openBusSession(req, res);
71  		}
72  
73  		@RestGet("/ops")
74  		public List<String> listOps(RestRequest req) {
75  			return ops(req);
76  		}
77  	}
78  
79  	@Rest
80  	public static class Plain extends BasicRestServlet implements BusEventsMixin {
81  		private static final long serialVersionUID = 1L;
82  
83  		@Override public ServerBus serverBus() { return BUS; }
84  
85  		@Override
86  		public JsonMap openBusSession(RestRequest req, RestResponse res) {
87  			CALLS.incrementAndGet();
88  			return BusEventsMixin.super.openBusSession(req, res);
89  		}
90  
91  		@RestGet("/ops")
92  		public List<String> listOps(RestRequest req) {
93  			return ops(req);
94  		}
95  	}
96  
97  	private static final MockRestClient GUARDED = MockRestClient.buildLax(Guarded.class);
98  	private static final MockRestClient PLAIN = MockRestClient.buildLax(Plain.class);
99  
100 	@AfterAll static void closeBus() {
101 		BUS.close();
102 	}
103 
104 	@BeforeEach void reset() {
105 		CALLS.set(0);
106 	}
107 
108 	@SuppressWarnings({
109 		"unchecked" // Casts a parsed JSON value to its known Map/List shape.
110 	})
111 	private static List<String> sessionOps(MockRestClient c) throws Exception {
112 		var text = c.get("/ops").header("Accept", "application/json").run().assertStatus(200).getContent().asString();
113 		return ((List<String>) Json.to(text, List.class)).stream().filter(x -> x.endsWith(".openBusSession")).toList();
114 	}
115 
116 	@Test void a01_aReDeclaredRestPostRegistersOnce() throws Exception {
117 		assertEquals(List.of("Guarded.openBusSession"), sessionOps(GUARDED));
118 	}
119 
120 	@Test void a02_theGuardRunsOncePerRequest() throws Exception {
121 		GUARDED.post(BusEventsMixin.SESSION_PATH).contentString(BODY).contentType("application/json")
122 			.header("Accept", "application/json").run().assertStatus(403);
123 		assertEquals(1, CALLS.get());
124 
125 		var text = GUARDED.post(BusEventsMixin.SESSION_PATH).contentString(BODY).contentType("application/json")
126 			.header("Accept", "application/json").header("X-App-Guard", "yes").run().assertStatus(200).getContent().asString();
127 		assertEquals(2, CALLS.get());
128 		assertTrue(text.contains("sessionId"), text);
129 	}
130 
131 	@Test void b01_anUnannotatedOverrideInheritsTheRoute() throws Exception {
132 		assertEquals(List.of("Plain.openBusSession"), sessionOps(PLAIN));
133 		PLAIN.post(BusEventsMixin.SESSION_PATH).contentString(BODY).contentType("application/json")
134 			.header("Accept", "application/json").run().assertStatus(200);
135 		assertEquals(1, CALLS.get());
136 	}
137 }
