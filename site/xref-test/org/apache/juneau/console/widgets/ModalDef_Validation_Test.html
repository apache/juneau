19  import static org.apache.juneau.bct.BctAssertions.assertSize;

27  /**
28   * {@link ModalDef#validate()} fail-closed branches: fields, child-action catalog, nested form and bar slot.
29   */
30  class ModalDef_Validation_Test extends TestBase {
31  
32  	private static void assertRejected(Runnable r, String fragment) {
33  		var e = assertThrows(IllegalArgumentException.class, r::run);
34  		assertTrue(e.getMessage().contains(fragment), () -> "expected '" + fragment + "' in: " + e.getMessage());
35  	}
36  
37  	@Test void a01_title_blank_rejected() {
38  		var blank = ModalDef.create("T");
39  		blank.title = " ";
40  		assertRejected(blank::validate, "title must not be null or blank");
41  		var m = ModalDef.create("T");
42  		m.title = null;
43  		assertRejected(m::validate, "title must not be null or blank");
44  	}
45  
46  	@Test void a02_field_nullBlankLabelAndBadKind_rejected() {
47  		var m = ModalDef.create("T").field("A", "1");
48  		m.fields.add(null);
49  		assertRejected(m::validate, "field must not be null");
50  
51  		var blank = ModalDef.create("T");
52  		blank.fields = new ArrayList<>();
53  		var f = ModalDef.Field.of("A", "1");
54  		f.label = " ";
55  		blank.fields.add(f);
56  		assertRejected(blank::validate, "label must not be null or blank");
57  
58  		var badKind = ModalDef.create("T").field("A", "1");
59  		badKind.fields.get(0).kind = "html";
60  		assertRejected(badKind::validate, "kind must be");
61  	}
62  
63  	@Test void a03_field_textCodeAndUnsetKind_accepted() {
64  		var m = ModalDef.create("T").field("A", "1").codeField("B", "2");
65  		m.fields.add(ModalDef.Field.of("C", null));
66  		m.fields.get(2).kind = null;
67  		m.validate();
68  	}
69  
70  	@Test void a04_codeField_appendsToExistingFields() {
71  		var m = ModalDef.create("T").field("A", "1").codeField("B", "2");
72  		assertSize(2, m.fields);
73  		assertEquals(ModalDef.FIELD_KIND_CODE, m.fields.get(1).kind);
74  	}
75  
76  	@Test void a05_field_nullLabel_rejected() {
77  		var m = ModalDef.create("T").field("A", "1");
78  		m.fields.get(0).label = null;
79  		assertRejected(m::validate, "label must not be null or blank");
80  	}
81  
82  	@Test void b01_childAction_null_rejected() {
83  		assertRejected(() -> ModalDef.create("T").childAction(null), "ChildAction must not be null");
84  	}
85  
86  	@Test void b02_childAction_accumulates_andDuplicateIdRejected() {
87  		var m = ModalDef.create("T")
88  			.childAction(ModalDef.ChildAction.of("a", "A"))
89  			.childAction(ModalDef.ChildAction.of("b", "B"));
90  		assertSize(2, m.childActions);
91  		m.validate();
92  		m.childAction(ModalDef.ChildAction.of("a", "Again"));
93  		assertRejected(m::validate, "more than once");
94  	}
95  
96  	@Test void b03_childAction_nullBlankIdOrLabel_rejected() {
97  		var nul = ModalDef.create("T").childAction(ModalDef.ChildAction.of("a", "A"));
98  		nul.childActions.add(null);
99  		assertRejected(nul::validate, "ChildAction must not be null");
100 
101 		var noId = ModalDef.create("T").childAction(ModalDef.ChildAction.of("a", "A"));
102 		noId.childActions.get(0).id = " ";
103 		assertRejected(noId::validate, "id must not be null or blank");
104 
105 		var noLabel = ModalDef.create("T").childAction(ModalDef.ChildAction.of("a", "A"));
106 		noLabel.childActions.get(0).label = null;
107 		assertRejected(noLabel::validate, "label must not be null or blank");
108 	}
109 
110 	@Test void b04_carryDrafts_needsModalFormAndChildFormUrl() {
111 		var noModalForm = ModalDef.create("T")
112 			.childAction(ModalDef.ChildAction.of("a", "A").form("/f").carryDrafts(true));
113 		assertRejected(noModalForm::validate, "no form");
114 
115 		var noChildForm = ModalDef.create("T").form(FormDef.create().field(FormDef.Input.of("n", "N", "text")))
116 			.childAction(ModalDef.ChildAction.of("a", "A").carryDrafts(true));
117 		assertRejected(noChildForm::validate, "no form URL");
118 
119 		var blankChildForm = ModalDef.create("T").form(FormDef.create().field(FormDef.Input.of("n", "N", "text")))
120 			.childAction(ModalDef.ChildAction.of("a", "A").form(" ").carryDrafts(true));
121 		assertRejected(blankChildForm::validate, "no form URL");
122 
123 		ModalDef.create("T").form(FormDef.create().field(FormDef.Input.of("n", "N", "text")))
124 			.childAction(ModalDef.ChildAction.of("a", "A").form("/f").carryDrafts(true)).validate();
125 	}
126 
127 	@Test void c01_nestedForm_isValidated() {
128 		var m = ModalDef.create("T").form(FormDef.create().field(FormDef.Input.of("n", "N", "text")).field(FormDef.Input.of("n", "N2", "text")));
129 		assertRejected(m::validate, "duplicate field name");
130 	}
131 
132 	@Test void c02_barSlot_setter_andNestedValidation() {
133 		var good = ModalDef.create("T").barSlot(BarSlot.create("b").widgets(BarText.of("t", "x")));
134 		assertNotNull(good.barSlot);
135 		good.validate();
136 		var bad = ModalDef.create("T").barSlot(BarSlot.create("b"));
137 		assertRejected(bad::validate, "at least one widget");
138 		assertNull(ModalDef.create("T").barSlot(null).barSlot);
139 	}
