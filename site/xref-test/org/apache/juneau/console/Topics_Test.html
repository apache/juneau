53  	@Test void a02_badKey_isE40() {
54  		var e = assertThrows(IllegalArgumentException.class, () -> Topics.selection("a b"));
55  		assertEquals(f(E40, "selection:a b"), e.getMessage());
56  		e = assertThrows(IllegalArgumentException.class, () -> Topics.job(null));
57  		assertEquals(f(E40, "job:null"), e.getMessage());
58  		var arg = "x".repeat(129);
59  		e = assertThrows(IllegalArgumentException.class, () -> Topics.cmd(arg));
60  		assertEquals(f(E40, "cmd:" + "x".repeat(129)), e.getMessage());
61  	}

63  	@Test void a03_custom() {
64  		assertEquals("ssc.focus", Topics.custom("ssc", "focus"));
65  		assertEquals("ssc.alert:east", Topics.custom("ssc", "alert", "east"));
66  		assertEquals("ssc.alert:*", Topics.custom("ssc", "alert", "*"));
67  		var e = assertThrows(IllegalArgumentException.class, () -> Topics.custom("", "focus"));
68  		assertEquals("custom topic family 'focus' must be namespaced (e.g. 'app.focus')", e.getMessage());
69  		e = assertThrows(IllegalArgumentException.class, () -> Topics.custom("SSC", "focus"));
70  		assertEquals(f(E40, "SSC.focus"), e.getMessage());
71  		e = assertThrows(IllegalArgumentException.class, () -> Topics.custom("ssc", "alert", "a b"));
72  		assertEquals(f(E40, "ssc.alert:a b"), e.getMessage());
73  	}

81  	@Test void a05_frameworkFamilies_matchJuneauBusJs() throws Exception {
82  		var js = Files.readString(Path.of(System.getProperty("basedir", "."),
83  			"../juneau-console-views/src/main/resources/org/apache/juneau/views/juneau-bus.js"), UTF_8);
84  		var m = Pattern.compile("FRAMEWORK_FAMILIES\\s*[:=]\\s*(?:Object\\.freeze\\()?\\[([^\\]]*)\\]").matcher(js);
85  		assertTrue(m.find(), "FRAMEWORK_FAMILIES array not found in juneau-bus.js");
86  		var jsFamilies = new LinkedHashSet<String>();
87  		for (var part : m.group(1).split(","))
88  			if (! part.isBlank())
89  				jsFamilies.add(part.trim().replaceAll("^['\"]|['\"]$", ""));
90  		assertEquals(Topics.FRAMEWORK_FAMILIES, jsFamilies);
91  	}

93  	@Test void a06_patternsMatchSchema() throws Exception {
94  		try (var in = Topics.class.getResourceAsStream("/org/apache/juneau/console/juneau-page.schema.json")) {
95  			var defs = JsonMap.ofString(new String(in.readAllBytes(), UTF_8)).getMap("$defs");
96  			assertEquals(defs.getMap("topic").getString("pattern"), Topics.TOPIC.pattern());
97  			assertEquals(defs.getMap("topicDecl").getMap("properties").getMap("topic").getString("pattern"), Topics.TOPIC_DECL.pattern());
98  			assertEquals(defs.getMap("publication").getMap("properties").getMap("topic").getString("pattern"), Topics.PUBLICATION.pattern());
99  			assertEquals(defs.getMap("subscription").getMap("properties").getMap("as").getString("pattern"), Topics.ROLE.pattern());
100 		}
