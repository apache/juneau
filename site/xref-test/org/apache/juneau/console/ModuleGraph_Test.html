30  /**
31   * Mechanical, {@code basedir}-relative assertion of the three-module dependency graph.
32   *
33   * <p>
34   * This is an <b>invariant/regression assertion, not a shown-failing-first TDD RED</b>: the
35   * {@code view-freemarker ↛ -datatables} half is already true on {@code main} before this feature exists. It fails
36   * only if someone later wires a forbidden edge (e.g. points {@code console-ui-freemarker} at {@code -datatables}),
37   * and passes once the module graph is exactly what is documented:
38   *
39   * <pre>
40   * console-ui                       → rest-server
41   * console-ui-freemarker            → console-ui + view-freemarker     (NOT -datatables)
42   * datatables                       → rest-server + bean-html5          (NOT console-ui — stays generic)
43   * view-freemarker                  → rest-server + freemarker          (NOT -datatables — untouched)
44   * </pre>
45   *
46   * <p>
47   * Reads sibling {@code pom.xml} files by a {@code basedir}-relative file read (Maven Surefire's default working
48   * directory is the module's own {@code basedir}, so {@code user.dir}'s parent is {@code juneau-rest/}) rather than
49   * as test <i>resources</i>, which would be fragile (resources are copied per-module, not shared across siblings).
50   */
51  class ModuleGraph_Test extends TestBase {
52  
53  	private static String pomOf(String moduleArtifactId) throws IOException {
54  		var restDir = new File(System.getProperty("user.dir")).getParentFile();
55  		var pom = new File(restDir, moduleArtifactId + "/pom.xml");
56  		assertTrue(pom.isFile(), () -> "Expected sibling pom not found: " + pom.getAbsolutePath());
57  		return Files.readString(pom.toPath());
58  	}
59  
60  	private static boolean hasDependency(String pomXml, String artifactId) {
61  		return pomXml.contains("<artifactId>" + artifactId + "</artifactId>");
62  	}
63  
64  	@ParameterizedTest
65  	@ValueSource(strings = {
66  		"juneau-rest-server-view-freemarker",
67  		"juneau-console-freemarker",
68  		"juneau-console",
69  	})
70  	void a01_moduleDoesNotDependOnDatatables(String moduleArtifactId) throws IOException {
71  		var pom = pomOf(moduleArtifactId);
72  		assertFalse(hasDependency(pom, "juneau-console-datatables"));
73  	}
74  
75  	@Test void a04_datatables_doesNotDependOnConsoleUiFamily() throws IOException {
76  		var pom = pomOf("juneau-console-datatables");
77  		assertFalse(hasDependency(pom, "juneau-console"));
78  		assertFalse(hasDependency(pom, "juneau-console-freemarker"));
79  	}
80  
81  	@Test void a06_consoleUi_dependsOnlyOnRestServer() throws IOException {
82  		var pom = pomOf("juneau-console");
83  		assertTrue(hasDependency(pom, "juneau-rest-server"));
84  	}
85  
86  	@Test void a07_consoleUiFreemarker_dependsOnConsoleUiAndViewFreemarker() throws IOException {
87  		var pom = pomOf("juneau-console-freemarker");
88  		assertContainsAll(pom, "<artifactId>juneau-console</artifactId>", "<artifactId>juneau-rest-server-view-freemarker</artifactId>");
89  	}
90  
91  	/**
92  	 * {@code console-ui} stays engine- and toolkit-agnostic: it must not depend on the {@code views} toolkit.
93  	 */
94  	@Test void a12_consoleUi_doesNotDependOnViews() throws IOException {
95  		var pom = pomOf("juneau-console");
96  		assertFalse(hasDependency(pom, "juneau-console-views"));
97  	}
98  
99  	/** C4 deleted the theme-packs module; custom palettes are FTL-authored, so nothing may re-introduce it. */
100 	@Test void a13_restParentPom_listsNoThemePacksModule() throws IOException {
101 		var restPom = Files.readString(new File(new File(System.getProperty("user.dir")).getParentFile(), "pom.xml").toPath());
102 		assertFalse(restPom.contains("<module>juneau-rest-server-theme-packs</module>"));
103 	}
104 
105 	/** The console-ui-freemarker-datatables module (the old <@datatable> macro) was deleted; datatables cards go through the card registry. */
106 	@Test void a14_restParentPom_listsNoConsoleUiFreemarkerDatatablesModule() throws IOException {
107 		var restPom = Files.readString(new File(new File(System.getProperty("user.dir")).getParentFile(), "pom.xml").toPath());
108 		assertFalse(restPom.contains("<module>juneau-rest-server-console-ui-freemarker-datatables</module>"));
109 	}

111 	/**
112 	 * {@code PageContractAssert} (C1-D3) ships at main scope so adopters can use it under any test framework;
113 	 * it must not pull in JUnit or opentest4j.
114 	 */
115 	@Test void a20_pageContractAssert_hasNoTestFrameworkImports() throws Exception {
116 		var src = java.nio.file.Files.readString(java.nio.file.Path.of(System.getProperty("user.dir"),
117 			"src/main/java/org/apache/juneau/console/test/PageContractAssert.java"));
118 		assertFalse(src.contains("org.junit"), "PageContractAssert must not import JUnit");
119 		assertFalse(src.contains("org.opentest4j"), "PageContractAssert must not import opentest4j");
120 	}
121 
122 	/**
123 	 * {@code views} now composes the console shell script by taking a compile-scope dependency on
124 	 * {@code console-ui}; that edge is one-directional &mdash; {@code console-ui} stays engine/toolkit-agnostic and
125 	 * must still not depend back on {@code views} (see {@link #a12_consoleUi_doesNotDependOnViews()}).
126 	 */
127 	@Test void a21_viewsDependsOnConsoleUi_consoleUiStillNotOnViews() throws IOException {
128 		var viewsPom = pomOf("juneau-console-views");
129 		assertTrue(hasDependency(viewsPom, "juneau-console"));
130 		var consoleUiPom = pomOf("juneau-console");
131 		assertFalse(hasDependency(consoleUiPom, "juneau-console-views"));
132 	}
