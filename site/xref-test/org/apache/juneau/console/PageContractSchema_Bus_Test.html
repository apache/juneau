30  /**
31   * The bus keys of schema 1.1: top-level {@code topics} / {@code bridges}, and
32   * {@code publishes} / {@code subscribes} on every card branch.  Rule checks (R-10, R-11, E-40..E-54) belong to
33   * {@link BusWiringValidator}, not the schema; this class pins only shape.
34   */
35  class PageContractSchema_Bus_Test extends TestBase {
36  
37  	private static String fixture(String name) throws IOException {
38  		try (var in = PageContractSchema_Bus_Test.class.getResourceAsStream("/contracts/" + name)) {
39  			assertNotNull(in, () -> "missing fixture " + name);
40  			return new String(in.readAllBytes(), UTF_8);
41  		}
42  	}
43  
44  	@Test void a01_validBusWiringPasses() throws Exception {
45  		var errors = PageContractSchema.get().validate(fixture("valid-bus-wiring.json"));
46  		assertTrue(errors.isEmpty(), () -> "valid-bus-wiring.json: " + errors);
47  	}
48  
49  	@ParameterizedTest
50  	@CsvSource(delimiter = '|', value = {
51  		"invalid-bus-topic-decl-retain.json|$.topics[0]: missing required 'retain'",
52  		"invalid-bus-topic-decl-family.json|$.topics[0].topic: does not match",
53  		"invalid-bus-topic-decl-publisher.json|$.topics[0].publisher: must be one of",
54  		"invalid-bus-bridge-transport.json|$.bridges[0].transport: must be one of",
55  		"invalid-bus-bridge-extra.json|$.bridges[0].url: not allowed",
56  		"invalid-bus-bridge-maxattempts.json|$.bridges[0].maxAttempts: expected integer",
57  		"invalid-bus-bridge-missing.json|$.bridges[0]: missing required 'session'",
58  		"invalid-bus-sub-as.json|$.cards[0].subscribes[0].as: does not match",
59  		"invalid-bus-sub-wildcard.json|$.cards[0].subscribes[0].topic: does not match",
60  		"invalid-bus-sub-whenempty.json|$.cards[0].subscribes[0].whenEmpty: must be one of",
61  		"invalid-bus-sub-map.json|$.cards[0].subscribes[0].map.x: does not match",
62  		"invalid-bus-pub-framework.json|$.cards[0].publishes[0].topic: does not match",
63  		"invalid-bus-pub-retain.json|$.cards[0].publishes[0]: missing required 'retain'",
64  		"invalid-bus-pub-topic-terminal-view.json|$.cards[0].publishes[0].topic: does not match",
65  	})
66  	void a02_invalidBusShapesFail(String name, String expected) throws Exception {
67  		var errors = PageContractSchema.get().validate(fixture(name));
68  		assertTrue(errors.stream().anyMatch(e -> e.contains(expected)), () -> name + " expected '" + expected + "' in " + errors);
69  	}
70  
71  	@Test void a03_defsPresentAndClosed() throws Exception {
72  		var defs = JsonMap.ofString(PageContractSchema.get().schemaJson()).getMap("$defs");
73  		for (var name : new String[] {"bridgeDecl", "topic", "topicDecl", "publication", "subscription", "ribbonPublish"})
74  			assertTrue(defs.containsKey(name), name);
75  		for (var name : new String[] {"bridgeDecl", "topicDecl", "publication", "subscription", "ribbonPublish"})
76  			assertEquals(false, defs.getMap(name).get("additionalProperties"), name + " is closed");
77  		for (var branch : new String[] {"card", "htmlCard", "datatablesCard", "terminalViewCard", "terminalCard"}) {
78  			var props = defs.getMap(branch).getMap("properties");
79  			assertTrue(props.containsKey("publishes") && props.containsKey("subscribes"), branch + " admits the wiring keys");
80  		}
81  	}
82  }
