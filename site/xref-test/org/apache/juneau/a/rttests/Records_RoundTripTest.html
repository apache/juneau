106 	@ParameterizedTest
107 	@MethodSource("testers")
108 	void a01_simpleRecord(RoundTrip_Tester t) throws Exception {
109 		var in = new Person("John", 30);
110 		var out = t.roundTrip(in, Person.class);
111 		assertEquals("John", out.name());
112 		assertEquals(30, out.age());
113 	}

115 	@ParameterizedTest
116 	@MethodSource("testers")
117 	void a02_nestedRecord(RoundTrip_Tester t) throws Exception {
118 		var person = new Person("Jane", 25);
119 		var in = new Nested(person, "developer");
120 		var out = t.roundTrip(in, Nested.class);
121 		assertEquals("Jane", out.person().name());
122 		assertEquals(25, out.person().age());
123 		assertEquals("developer", out.role());
124 	}

126 	@ParameterizedTest

128 	void a03_recordWithList(RoundTrip_Tester t) throws Exception {
129 		var members = List.of(new Person("Alice", 30), new Person("Bob", 25));
130 		var in = new Team("devs", members);
131 		var out = t.roundTrip(in, Team.class);
132 		assertEquals("devs", out.name());
133 		assertEquals(2, out.members().size());
