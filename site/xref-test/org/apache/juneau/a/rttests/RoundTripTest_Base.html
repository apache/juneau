191 	/**
192 	 * Returns true if the object contains structures Parquet cannot serialize due to inherent format limitations:
193 	 * recursive schemas, annotation-based reconstruction, null beans in collections, etc.
194 	 */
195 	private static boolean isParquetIncompatibleBeanOrCollection(Object o) {
196 		if (o == null) return false;
197 		var cls = o.getClass();
198 		var name = cls.getName();
199 		// JsonSchema has recursive structures (properties that contain other JsonSchema instances)
200 		if (name.contains("JsonSchema"))
201 			return true;
202 		// Beans with @NameProperty or @ParentProperty - Parquet parser hits ArrayIndexOutOfBoundsException
203 		// in readMapKeyValueColumnChunk due to nested map schema mismatch
204 		if (name.contains("NameProperty_RoundTripTest") || name.contains("ParentProperty_RoundTripTest"))
205 			return true;
206 		// Collection containing Class or other incompatible elements
207 		if (o instanceof Collection<?> c) {
208 			boolean hasNull = false;
209 			Object firstNonNull = null;
210 			for (var elem : c) {
211 				if (elem == null) { hasNull = true; continue; }
212 				if (elem instanceof Class || isParquetIncompatibleBeanOrCollection(elem)) return true;
213 				if (firstNonNull == null) firstNonNull = elem;
214 			}
215 			// Parquet cannot encode a null bean in a list (no row-null sentinel in flat schema)
216 			if (hasNull && firstNonNull != null && isUserDefinedBeanInstance(firstNonNull))
217 				return true;
218 		}
219 		// Map containing Class keys/values or other incompatible elements
220 		if (o instanceof Map<?,?> m) {
221 			for (var k : m.keySet())
222 				if (k instanceof Class || (k != null && isParquetIncompatibleBeanOrCollection(k)))
223 					return true;
224 			for (var v : m.values())
225 				if (v instanceof Class || (v != null && isParquetIncompatibleBeanOrCollection(v)))
226 					return true;
227 		}
228 		return false;
229 	}

231 	/**
232 	 * Returns true if the object is a user-defined bean instance (not a scalar, array, collection, or JDK type).
233 	 * Used to detect collections of beans that contain null elements, which Parquet cannot encode faithfully.
234 	 */
235 	private static boolean isUserDefinedBeanInstance(Object o) {
236 		if (o == null) return false;
237 		var cls = o.getClass();
238 		return !cls.isPrimitive()
239 			&& !cls.isArray()
240 			&& !(o instanceof Number)
241 			&& !(o instanceof String)
242 			&& !(o instanceof Boolean)
243 			&& !(o instanceof Character)
244 			&& !(o instanceof Optional)
245 			&& !(o instanceof Collection)
246 			&& !(o instanceof Map)
247 			&& !cls.getName().startsWith("java.")
248 			&& !cls.getName().startsWith("javax.");
249 	}

251 	/**
252 	 * Returns true if the object can be serialized to CSV without error.
253 	 *
254 	 * <p>
255 	 * CSV supports byte[], primitive arrays ([1;2;3]), nested structures (when enabled),
256 	 * and type discriminators. Only 2D+ primitive arrays are excluded.
257 	 */
258 	protected static boolean isCsvSerializableInput(Object o) {
259 		if (o == null) return false;
260 		var cls = o.getClass();
261 		// Skip 2D+ primitive arrays (int[][], etc.) - not supported
262 		if (cls.isArray()) {
263 			var ct = cls.getComponentType();
264 			if (ct.isArray() && ct.getComponentType().isPrimitive()) return false;
265 		}
266 		return true;
267 	}
