17  package org.apache.juneau.examples.guides.testanendpoint;
18  
19  // region test
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import org.apache.juneau.marshall.json.*;
23  import org.apache.juneau.rest.mock.*;
24  import org.junit.jupiter.api.*;
25  
26  class GreetingResource_Test {
27  
28  	@Test
29  	void a01_getReturnsJson() throws Exception {
30  		try (var mock = MockRestClient.create(GreetingResource.class);
31  				var resp = mock.get("/Ada").header("Accept", "application/json").run()) {
32  			resp.assertThat().isOk().body().contains("\"message\":\"Hello, Ada!\"");
33  		}
34  	}
35  
36  	@Test
37  	void a02_postParsesBodyIntoBean() throws Exception {
38  		try (var mock = MockRestClient.builder(GreetingResource.class).defaultParser(JsonParser.DEFAULT).build();
39  				var resp = mock.post("/").body(new Greeting("Grace")).run()) {
40  			resp.assertOk();
41  			var greeting = resp.body().as(Greeting.class);
42  			assertEquals("Hello, Grace!", greeting.message);
43  		}
44  	}
45  
46  	@Test
47  	void a03_blankNameIsRejected() throws Exception {
48  		try (var mock = MockRestClient.builder(GreetingResource.class).noTrace().build();
49  				var resp = mock.post("/").bodyString("{\"name\":\"\"}").header("Content-Type", "application/json").run()) {
50  			resp.assertStatus(400);
51  		}
52  	}
53  }
54  // endregion test
