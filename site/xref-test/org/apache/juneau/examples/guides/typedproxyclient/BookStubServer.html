17  package org.apache.juneau.examples.guides.typedproxyclient;

19  // region stub
20  import static java.nio.charset.StandardCharsets.*;
21  
22  import java.io.*;
23  import java.net.*;
24  
25  import com.sun.net.httpserver.*;
26  
27  /**
28   * A stand-in for the real books API, on the JDK's built-in HTTP server.  Answers with fixed JSON, and echoes a
29   * posted book back with id 3.
30   */
31  public class BookStubServer {
32  
33  	/**
34  	 * Serves the stand-in API on port 8080 until the process is stopped.
35  	 *
36  	 * @param args Unused.
37  	 * @throws IOException If the port cannot be bound.
38  	 */
39  	public static void main(String[] args) throws IOException {
40  		start(8080);
41  		System.out.println("Books API stand-in on http://localhost:8080/books");
42  	}
43  
44  	/**
45  	 * Starts the server.
46  	 *
47  	 * @param port The port, or {@code 0} for any free port.
48  	 * @return The running server; call {@code stop(0)} to shut it down.
49  	 * @throws IOException If the port cannot be bound.
50  	 */
51  	public static HttpServer start(int port) throws IOException {
52  		var server = HttpServer.create(new InetSocketAddress("localhost", port), 0);
53  		server.createContext("/books", x -> {
54  			var path = x.getRequestURI().getPath();
55  			var method = x.getRequestMethod();
56  			if (method.equals("GET") && path.equals("/books"))
57  				reply(x, 200, "[{\"id\":1,\"title\":\"Dune\",\"author\":\"Frank Herbert\"},"
58  					+ "{\"id\":2,\"title\":\"Emma\",\"author\":\"Jane Austen\"}]");
59  			else if (method.equals("GET") && path.equals("/books/1"))
60  				reply(x, 200, "{\"id\":1,\"title\":\"Dune\",\"author\":\"Frank Herbert\"}");
61  			else if (method.equals("POST") && path.equals("/books"))
62  				reply(x, 200, new String(x.getRequestBody().readAllBytes(), UTF_8).replace("\"id\":0", "\"id\":3"));
63  			else
64  				reply(x, 404, "{\"error\":\"Not found\"}");
65  		});
66  		server.start();
67  		return server;
68  	}
69  
70  	private static void reply(HttpExchange x, int status, String json) throws IOException {
71  		var bytes = json.getBytes(UTF_8);
72  		x.getResponseHeaders().set("Content-Type", "application/json");
73  		x.sendResponseHeaders(status, bytes.length);
74  		try (var out = x.getResponseBody()) {
75  			out.write(bytes);
76  		}
77  	}
78  }
