17  package org.apache.juneau.examples.guides;
18  
19  import static java.nio.charset.StandardCharsets.*;
20  
21  import java.io.*;
22  import java.util.*;
23  import java.util.regex.*;
24  
25  /**
26   * Reads a named region out of a guide resource file, normalized the same way the docs checker normalizes fences.
27   *
28   * <p>
29   * A guide page shows a resource file (expected output, a sample POM) through a fence such as
30   * <c>```text guide=SpringBootRestApi/run-output.txt#get-pet</c>.  The docs build checks that the fence equals the
31   * region; the guide's test calls {@link #read(String, String)} to check that the region equals what the code really
32   * produces.  Together they keep the page, the resource and the code in step.
33   *
34   * <p>
35   * A region is the block between <c>region NAME</c> and <c>endregion NAME</c> marker lines, written as a
36   * <c>//</c>, <c>#</c> or <c>&lt;!-- --&gt;</c> comment.  Normalization: CRLF to LF, tabs expanded to four-column stops, common
37   * leading indentation removed, trailing whitespace stripped from every line, leading and trailing blank lines
38   * dropped.
39   */
40  public final class GuideRegion {
41  
42  	private static final String MARKER = "^\\s*(?://|#|<!--)\\s*%s\\s+(%s)\\s*(?:-->)?\\s*$";
43  
44  	private GuideRegion() {}
45  
46  	/**
47  	 * Returns the normalized body of a region in a classpath resource.
48  	 *
49  	 * @param resourcePath The resource path relative to the classpath root (for example
50  	 * 	<js>"SpringBootRestApi/run-output.txt"</js>).
51  	 * @param region The region name.
52  	 * @return The normalized region body, lines joined with <c>\n</c>.
53  	 * @throws IllegalArgumentException If the resource or the region is missing, or the region is unterminated.
54  	 */
55  	public static String read(String resourcePath, String region) {
56  		var open = Pattern.compile(String.format(MARKER, "region", Pattern.quote(region)));
57  		var close = Pattern.compile(String.format(MARKER, "endregion", Pattern.quote(region)));
58  		var lines = readLines(resourcePath);
59  		var body = new ArrayList<String>();
60  		var inside = false;
61  		for (var line : lines) {
62  			if (! inside) {
63  				inside = open.matcher(line).matches();
64  			} else if (close.matcher(line).matches()) {
65  				return normalize(body);
66  			} else {
67  				body.add(line);
68  			}
69  		}
70  		throw new IllegalArgumentException((inside ? "Region unterminated: " : "Region not found: ") + resourcePath + "#" + region);
71  	}
72  
73  	/**
74  	 * Normalizes text the same way {@link #read(String, String)} normalizes a region body.
75  	 *
76  	 * <p>
77  	 * Tests use this on captured program output before comparing it with a region.
78  	 *
79  	 * @param text The text to normalize.
80  	 * @return The normalized text, lines joined with <c>\n</c>.
81  	 */
82  	public static String normalize(String text) {
83  		return normalize(Arrays.asList(text.replace("\r\n", "\n").split("\n", -1)));
84  	}
85  
86  	private static List<String> readLines(String resourcePath) {
87  		try (var in = GuideRegion.class.getClassLoader().getResourceAsStream(resourcePath)) {
88  			if (in == null)
89  				throw new IllegalArgumentException("Resource not found: " + resourcePath);
90  			return Arrays.asList(new String(in.readAllBytes(), UTF_8).replace("\r\n", "\n").split("\n", -1));
91  		} catch (IOException e) {
92  			throw new UncheckedIOException(e);
93  		}
94  	}
95  
96  	// Same as Python's str.expandtabs(4): a tab advances to the next multiple of four columns.
97  	private static String expandTabs(String line) {
98  		var sb = new StringBuilder();
99  		for (var c : line.toCharArray()) {
100 			if (c == '\t') {
101 				do {
102 					sb.append(' ');
103 				} while (sb.length() % 4 != 0);
104 			} else {
105 				sb.append(c);
106 			}
107 		}
108 		return sb.toString();
109 	}
110 
111 	private static String normalize(List<String> body) {
112 		var lines = new ArrayList<String>();
113 		for (var line : body)
114 			lines.add(expandTabs(line).stripTrailing());
115 		while (! lines.isEmpty() && lines.get(0).isEmpty())
116 			lines.remove(0);
117 		while (! lines.isEmpty() && lines.get(lines.size() - 1).isEmpty())
118 			lines.remove(lines.size() - 1);
119 		var indent = lines.stream().filter(x -> ! x.isEmpty()).mapToInt(x -> x.length() - x.stripLeading().length()).min().orElse(0);
120 		var sb = new StringBuilder();
121 		for (var line : lines) {
122 			if (sb.length() > 0)
123 				sb.append('\n');
124 			sb.append(line.isEmpty() ? line : line.substring(indent));
125 		}
126 		return sb.toString();
127 	}
128 }
