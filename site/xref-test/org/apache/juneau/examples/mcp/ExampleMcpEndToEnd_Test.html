30  /**
31   * Proves the example actually works: boots {@link ExampleServer} in-process on an ephemeral port and drives
32   * the real {@link McpClient} through every surface the example demonstrates &mdash; discovery, listing, a
33   * tool call, a templated resource read, a completion, a prompt, a subscription change notification, and an
34   * elicitation (MRTR) round-trip &mdash; asserting the observable outcome of each.
35   *
36   * <p>
37   * It reuses {@link ExampleClient#connect(String)} so the exact client wiring the walkthrough uses is what is
38   * being tested. Each test publishes its own distinctly-titled note (rather than relying on notes published by
39   * other tests), so tests remain independent even though the server/client pair is shared across the class.
40   */
41  @SuppressWarnings({
42  	"resource" // server/client are opened in setUp() and closed in tearDown(); lifecycle spans the whole test class.
43  })

151 	@Test
152 	void f01_subscription_deliversResourceUpdatedOnPublish() throws Exception {
153 		var noteUri = NoteStore.uriFor("subscribed");
154 		var updates = new LinkedBlockingQueue<String>();
155 		var acknowledged = new CountDownLatch(1);
156 
157 		var handle = client.listen(
158 			new SubscriptionFilter().setResourceSubscriptions(List.of(noteUri)),
159 			new McpSubscriptionListener() {
160 				@Override public void onAcknowledged(SubscriptionFilter honoredFilter) { acknowledged.countDown(); }
161 				@Override public void onResourceUpdated(String uri) { updates.add(uri); }
162 			});
163 		try {
164 			assertTrue(acknowledged.await(15, TimeUnit.SECONDS), "subscription was not acknowledged");
165 			client.callTool("publishNote", Map.of("title", "subscribed", "body", "hi"));
166 			assertEquals(noteUri, updates.poll(15, TimeUnit.SECONDS));
167 		} finally {
168 			handle.close();
169 		}
170 	}

172 	// -------- deleteNote elicitation (MRTR) --------
173 
174 	@Test
175 	void g01_deleteWithElicitation_confirmsThenDeletes() throws Exception {
176 		client.callTool("publishNote", Map.of("title", "trash", "body", "delete me"));
177 
178 		var result = client.callToolWithElicitation("deleteNote", Map.of("title", "trash"), requests -> {
179 			var answers = new LinkedHashMap<String,ElicitResult>();
180 			requests.keySet().forEach(id -> answers.put(id,
181 				new ElicitResult().setAction(ElicitAction.ACCEPT).putContent("confirm", true)));
182 			return answers;
183 		});
184 		assertEquals("Deleted note 'trash'.", result.firstText());
185 
186 		// And it is really gone from the index.
187 		var contents = client.readResource(NoteStore.SCHEME + "index").getContents();
188 		assertFalse(((TextResourceContents)contents.get(0)).getText().contains("trash"));
189 	}
190 
191 	@Test
192 	void g02_deleteWithElicitation_declineKeepsNote() throws Exception {
193 		client.callTool("publishNote", Map.of("title", "keep", "body", "keep me"));
194 
195 		var result = client.callToolWithElicitation("deleteNote", Map.of("title", "keep"), requests -> {
196 			var answers = new LinkedHashMap<String,ElicitResult>();
197 			requests.keySet().forEach(id -> answers.put(id, new ElicitResult().setAction(ElicitAction.DECLINE)));
198 			return answers;
199 		});
200 		assertEquals("Deletion of 'keep' was cancelled.", result.firstText());
201 
202 		var contents = client.readResource(NoteStore.SCHEME + "index").getContents();
203 		assertTrue(((TextResourceContents)contents.get(0)).getText().contains("keep"));
204 	}
