32  /**
33   * Regression tests for the Phase-2 CBOR conformance fixes (175fc).
34   *
35   * <p>
36   * Each category guards one audited gap:
37   * <ul>
38   * 	<li><b>a</b> — GAP-1: 64-bit unsigned integers / negatives beyond signed {@code long}.
39   * 	<li><b>b</b> — GAP-2: lossless {@link BigInteger} / {@link BigDecimal} round-trips.
40   * 	<li><b>c</b> — GAP-3: indefinite-length text/byte strings (RFC 8949 §3.2.3).
41   * 	<li><b>d</b> — GAP-4: half-precision (float16) subnormal decode.
42   * 	<li><b>e</b> — GAP-8: unpaired UTF-16 surrogate on encode.
43   * </ul>
44   */
45  @SuppressWarnings({
46  	"resource" // Token readers are closed via try-with-resources; JDT's flow analysis over chained factory calls yields false-positive leak reports.
47  })
48  class CborConformanceFixes_Test extends TestBase {
49  
50  	private static String enc(Object input) throws Exception {
51  		return toSpacedHex(CborSerializer.DEFAULT.write(input));
52  	}
53  
54  	private static <T> T rt(Object input, Class<T> type) throws Exception {
55  		return CborParser.DEFAULT.read(CborSerializer.DEFAULT.write(input), type);
56  	}
57  
58  	//------------------------------------------------------------------------------------------------
59  	// GAP-1 — uint64 / negatives beyond signed long.
60  	//------------------------------------------------------------------------------------------------
61  
62  	@Test
63  	void a01_uint64MaxDecodesToBigInteger() throws Exception {
64  		// 1B FF FF FF FF FF FF FF FF == CBOR 18446744073709551615 (uint64 max).
65  		var b = fromHex("1BFFFFFFFFFFFFFFFF");
66  		assertEquals(new BigInteger("18446744073709551615"), CborParser.DEFAULT.read(b, BigInteger.class));
67  	}
68  
69  	@Test
70  	void a02_uint64MaxIntoLongKeepsRawBits() throws Exception {
71  		// Per the type-driven contract a long field keeps the raw 64-bit bits (uint64 max -> -1).
72  		var b = fromHex("1BFFFFFFFFFFFFFFFF");
73  		assertEquals(-1L, CborParser.DEFAULT.read(b, Long.class));
74  	}
75  
76  	@Test
77  	void a03_nintMinDecodesToBigInteger() throws Exception {
78  		// 3B FF FF FF FF FF FF FF FF == CBOR -18446744073709551616 (nint min).
79  		var b = fromHex("3BFFFFFFFFFFFFFFFF");
80  		assertEquals(new BigInteger("-18446744073709551616"), CborParser.DEFAULT.read(b, BigInteger.class));
81  	}
82  
83  	@Test
84  	void a04_uint64MaxEncodesWithoutTruncation() throws Exception {
85  		// Previously truncated to CBOR -1 (0x20); must now emit the full 8-byte magnitude.
86  		assertEquals("1B FF FF FF FF FF FF FF FF", enc(new BigInteger("18446744073709551615")));
87  	}
88  
89  	@Test
90  	void a05_pow63EncodesAsNativeUint() throws Exception {
91  		// 2^63 exceeds signed long but fits unsigned 64; emit major type 0 with the raw magnitude.
92  		assertEquals("1B 80 00 00 00 00 00 00 00", enc(new BigInteger("9223372036854775808")));
93  	}
94  
95  	@Test
96  	void a06_nintMinEncodesAsNativeNint() throws Exception {
97  		// -2^64 is the smallest CBOR negative integer; emit major type 1 with argument 2^64-1.
98  		assertEquals("3B FF FF FF FF FF FF FF FF", enc(new BigInteger("-18446744073709551616")));
99  	}

102 	void a07_uint64RoundTrip() throws Exception {
103 		var v = new BigInteger("18446744073709551615");
104 		assertEquals(v, rt(v, BigInteger.class));
105 	}

144 	// GAP-3 — indefinite-length text / byte strings (RFC 8949 §3.2.3).
145 	//------------------------------------------------------------------------------------------------
146 
147 	@Test
148 	void c01_indefiniteByteStringDecodes() throws Exception {
149 		// 5F 42 01 02 43 03 04 05 FF == indefinite byte string of chunks {01 02} + {03 04 05}.
150 		var b = fromHex("5F42010243030405FF");
151 		assertArrayEquals(new byte[]{1, 2, 3, 4, 5}, CborParser.DEFAULT.read(b, byte[].class));
152 	}
153 
154 	@Test
155 	void c02_indefiniteTextStringDecodes() throws Exception {
156 		// 7F 61 61 61 62 FF == indefinite text string of chunks "a" + "b".
157 		var b = fromHex("7F61616162FF");
158 		assertEquals("ab", CborParser.DEFAULT.read(b, String.class));
159 	}
160 
161 	@Test
162 	void c03_indefiniteEmptyTextStringDecodes() throws Exception {
163 		// 7F FF == indefinite text string with zero chunks -> empty string.
164 		var b = fromHex("7FFF");
165 		assertEquals("", CborParser.DEFAULT.read(b, String.class));
166 	}
167 
168 	@Test
169 	void c04_indefiniteByteStringViaTokenReader() throws Exception {
170 		var b = fromHex("5F42010243030405FF");
171 		try (var r = CborParser.DEFAULT.readTokens(b)) {
172 			assertEquals(TokenType.VALUE_BINARY, r.next());
173 			assertArrayEquals(new byte[]{1, 2, 3, 4, 5}, r.getBinary());
174 		}
175 	}
176 
177 	@Test
178 	void c05_nestedIndefiniteChunkRejectedCleanly() throws Exception {
179 		// 5F 5F FF == a byte-string chunk that is itself indefinite; rejected per §3.2.3 with a clean
180 		// IOException rather than a NegativeArraySizeException.
181 		var b = fromHex("5F5FFF");
182 		try (var r = CborParser.DEFAULT.readTokens(b)) {
183 			assertThrowsWithMessage(IOException.class, "Nested indefinite-length string chunk not allowed", r::next);
184 		}
185 	}
186 
187 	@Test
188 	void c06_mismatchedChunkMajorTypeRejectedCleanly() throws Exception {
189 		// 5F 61 61 FF == a text-string chunk inside an indefinite byte string; rejected cleanly.
190 		var b = fromHex("5F6161FF");
191 		try (var r = CborParser.DEFAULT.readTokens(b)) {
192 			assertThrowsWithMessage(IOException.class, "Invalid chunk major type", r::next);
193 		}
194 	}
195 
196 	@Test
197 	void c07_truncatedIndefiniteStringRejectedCleanly() throws Exception {
198 		// 5F 42 01 02 (no BREAK, EOF mid-stream) must not loop or crash with a runtime exception.
199 		var b = fromHex("5F42010 2".replace(" ", ""));
200 		try (var r = CborParser.DEFAULT.readTokens(b)) {
201 			assertThrowsWithMessage(IOException.class, "Unexpected end of CBOR input in indefinite-length string", r::next);
202 		}
203 	}

209 	@Test
210 	void d01_smallestPositiveSubnormal() throws Exception {
211 		// F9 00 01 == smallest positive half subnormal == 2^-24 (~5.96e-8), previously decoded ~0.
212 		var b = fromHex("F90001");
213 		assertEquals(1.0 / 16777216.0, CborParser.DEFAULT.read(b, Double.class), 0.0);
214 	}
215 
216 	@Test
217 	void d02_largestSubnormal() throws Exception {
218 		// F9 03 FF == largest half subnormal == 1023 * 2^-24.
219 		var b = fromHex("F903FF");
220 		assertEquals(1023.0 / 16777216.0, CborParser.DEFAULT.read(b, Double.class), 0.0);
221 	}
222 
223 	@Test
224 	void d03_negativeSubnormalKeepsSign() throws Exception {
225 		// F9 80 01 == negative smallest subnormal.
226 		var b = fromHex("F98001");
227 		assertEquals(-1.0 / 16777216.0, CborParser.DEFAULT.read(b, Double.class), 0.0);
228 	}
229 
230 	@Test
231 	void d04_zeroStillDecodesToZero() throws Exception {
232 		// F9 00 00 == +0.0 (mant == 0 branch must still be exact zero).
233 		var b = fromHex("F90000");
234 		assertEquals(0.0, CborParser.DEFAULT.read(b, Double.class), 0.0);
235 	}

241 	@Test
242 	void e01_loneHighSurrogateEmitsReplacement() throws Exception {
243 		// A trailing/unpaired high surrogate must emit U+FFFD (EF BF BD) rather than throwing.
244 		assertEquals("63 EF BF BD", enc("\uD83D"));
245 		assertEquals("\uFFFD", rt("\uD83D", String.class));
246 	}
247 
248 	@Test
249 	void e02_loneLowSurrogateEmitsReplacement() throws Exception {
250 		assertEquals("63 EF BF BD", enc("\uDC00"));
251 		assertEquals("\uFFFD", rt("\uDC00", String.class));
252 	}
253 
254 	@Test
255 	void e03_highSurrogateFollowedByNonLowSurrogate() throws Exception {
256 		// High surrogate followed by 'a' (not a low surrogate): replacement char then 'a'.
257 		assertEquals("\uFFFDa", rt("\uD83Da", String.class));
258 	}
259 
260 	@Test
261 	void e04_validSurrogatePairStillRoundTrips() throws Exception {
262 		// Regression guard: a proper astral character (U+1F600) must still encode as a 4-byte
263 		// sequence and round-trip unchanged.
264 		assertEquals("64 F0 9F 98 80", enc("\uD83D\uDE00"));
265 		assertEquals("\uD83D\uDE00", rt("\uD83D\uDE00", String.class));
266 	}
267 
268 	@Test
269 	void e05_threeByteBmpCharStillRoundTrips() throws Exception {
270 		// Regression guard for the non-surrogate branch: a 3-byte BMP char (U+4E2D) must encode as
271 		// E4 B8 AD and round-trip unchanged (no replacement applied).
272 		assertEquals("63 E4 B8 AD", enc("\u4E2D"));
273 		assertEquals("\u4E2D", rt("\u4E2D", String.class));
274 	}
