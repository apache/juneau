19  import static org.apache.juneau.commons.utils.StringUtils.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import org.apache.juneau.*;
23  import org.apache.juneau.marshall.collections.*;
24  import org.junit.jupiter.api.*;
25  
26  /**
27   * CBOR decoding verification for each encoding tier.
28   */
29  class CborInputStream_Test extends TestBase {
30  
31  	@Test
32  	void b01_decodePositiveIntegers() throws Exception {
33  		assertEquals(0, CborParser.DEFAULT.read(hex("00"), Integer.class));
34  		assertEquals(23, CborParser.DEFAULT.read(hex("17"), Integer.class));
35  		assertEquals(24, CborParser.DEFAULT.read(hex("18 18"), Integer.class));
36  		assertEquals(255, CborParser.DEFAULT.read(hex("18 FF"), Integer.class));
37  		assertEquals(256, CborParser.DEFAULT.read(hex("19 01 00"), Integer.class));
38  		assertEquals(65535, CborParser.DEFAULT.read(hex("19 FF FF"), Integer.class));
39  		assertEquals(65536, CborParser.DEFAULT.read(hex("1A 00 01 00 00"), Long.class));
40  		assertEquals(0xFFFFFFFFL, CborParser.DEFAULT.read(hex("1A FF FF FF FF"), Long.class));
41  	}
42  
43  	@Test
44  	void b02_decodeNegativeIntegers() throws Exception {
45  		assertEquals(-1, CborParser.DEFAULT.read(hex("20"), Integer.class));
46  		assertEquals(-24, CborParser.DEFAULT.read(hex("37"), Integer.class));
47  		assertEquals(-25, CborParser.DEFAULT.read(hex("38 18"), Integer.class));
48  		assertEquals(-256, CborParser.DEFAULT.read(hex("38 FF"), Integer.class));
49  		assertEquals(-257, CborParser.DEFAULT.read(hex("39 01 00"), Integer.class));
50  	}
51  
52  	@Test
53  	void b03_decodeBoolean() throws Exception {
54  		assertEquals(false, CborParser.DEFAULT.read(hex("F4"), Boolean.class));
55  		assertEquals(true, CborParser.DEFAULT.read(hex("F5"), Boolean.class));
56  	}
57  
58  	@Test
59  	void b04_decodeNull() throws Exception {
60  		assertNull(CborParser.DEFAULT.read(hex("F6"), Object.class));
61  	}
62  
63  	@Test
64  	void b05_decodeFloat32() throws Exception {
65  		assertEquals(0.0f, CborParser.DEFAULT.read(hex("FA 00 00 00 00"), Float.class));
66  		assertEquals(1.0f, CborParser.DEFAULT.read(hex("FA 3F 80 00 00"), Float.class));
67  	}
68  
69  	@Test
70  	void b06_decodeFloat64() throws Exception {
71  		assertEquals(0.0, CborParser.DEFAULT.read(hex("FB 00 00 00 00 00 00 00 00"), Double.class));
72  		assertEquals(1.1, CborParser.DEFAULT.read(hex("FB 3F F1 99 99 99 99 99 9A"), Double.class));
73  	}
74  
75  	@Test
76  	void b07_decodeStrings() throws Exception {
77  		assertEquals("", CborParser.DEFAULT.read(hex("60"), String.class));
78  		assertEquals("a", CborParser.DEFAULT.read(hex("61 61"), String.class));
79  		var s24 = "x".repeat(24);
80  		assertEquals(s24, CborParser.DEFAULT.read(CborSerializer.DEFAULT.write(s24), String.class));
81  	}
82  
83  	@Test
84  	void b08_decodeBinary() throws Exception {
85  		assertArrayEquals(new byte[0], CborParser.DEFAULT.read(hex("40"), byte[].class));
86  		assertArrayEquals(new byte[] { 1, 2, 3 }, CborParser.DEFAULT.read(hex("43 01 02 03"), byte[].class));
87  	}
88  
89  	@Test
90  	void b09_decodeArrays() throws Exception {
91  		var empty = CborParser.DEFAULT.read(hex("80"), JsonList.class);
92  		assertNotNull(empty);
93  		assertTrue(empty.isEmpty());
94  		var arr = CborParser.DEFAULT.read(hex("83 01 02 03"), JsonList.class);
95  		assertEquals(3, arr.size());
96  		assertEquals(1, arr.getInt(0));
97  		assertEquals(2, arr.getInt(1));
98  		assertEquals(3, arr.getInt(2));
99  	}
100 
101 	@Test
102 	void b10_decodeMaps() throws Exception {
103 		var m = CborParser.DEFAULT.read(hex("A0"), JsonMap.class);
104 		assertTrue(m.isEmpty());
105 		var m2 = CborParser.DEFAULT.read(hex("A2 61 61 01 61 62 02"), JsonMap.class);
106 		assertEquals(1, m2.getInt("a"));
107 		assertEquals(2, m2.getInt("b"));
108 	}
109 
110 	@Test
111 	void b11_decodeUnicodeStrings() throws Exception {
112 		assertEquals("\u00E9", CborParser.DEFAULT.read(hex("62 C3 A9"), String.class));
113 		assertEquals("\u20AC", CborParser.DEFAULT.read(hex("63 E2 82 AC"), String.class));
114 	}
115 
116 	@Test
117 	void b12_dataTypeDetection() {
118 		assertEquals(0, CborInputStream.getMajorType(0x00));
119 		assertEquals(1, CborInputStream.getMajorType(0x20));
120 		assertEquals(3, CborInputStream.getMajorType(0x61));
121 		assertEquals(4, CborInputStream.getMajorType(0x80));
122 		assertEquals(5, CborInputStream.getMajorType(0xA0));
123 		assertEquals(7, CborInputStream.getMajorType(0xF6));
124 		assertEquals(22, CborInputStream.getAdditionalInfo(0xF6));
125 	}
126 
127 	private static byte[] hex(String spaced) {
128 		return fromHex(spaced.replace(" ", ""));
129 	}
