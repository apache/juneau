42  	private static byte[] bytes(java.util.function.Consumer<ProtobufWriter> fn) {
43  		var out = new ByteArrayOutputStream();

58  	@Test
59  	void a01_taggedRepeatedMapElement_hitsIsMapBranch() throws Exception {
60  		// The serializer cannot itself produce this shape (writeSingle's message path requires a bean, and a bare
61  		// Map isn't one -- see writeMessageField), so the bytes are hand-built directly. On the parse side,
62  		// readMessage's newBeanMap(outer, Map.class) returns null for a non-bean type (no BeanMeta), and reading
63  		// still proceeds to the unconditional final "return m.getBean();", so this exercises the elType.isMap()
64  		// dispatch (L152) but blows up with an NPE deeper in readMessage -- a real bug (flagged, not fixed).
65  		var elBytes = bytes(w -> { /* empty sub-message; content is irrelevant, the NPE occurs before any field is read */ });
66  		var msg = bytes(w -> { w.writeTag(1, WireType.LEN); w.writeLenDelimited(elBytes); });
67  		var ex = assertThrows(org.apache.juneau.marshall.parser.ParseException.class, () -> ProtobufParser.DEFAULT.read(msg, MapList.class));
68  		assertTrue(ex.getCause() instanceof NullPointerException);
69  	}

146 	@Test
147 	void a04_unknownFieldNumberInMapEntry_isSkipped() throws Exception {
148 		// Hand-built map entry: field 1 (key)="a", field 3 (unknown)=varint 999, field 2 (value)=varint 1.
149 		var entry = bytes(w -> {
150 			w.writeTag(1, WireType.LEN); w.writeString("a");
151 			w.writeTag(3, WireType.VARINT); w.writeVarint(999);
152 			w.writeTag(2, WireType.VARINT); w.writeVarint(1);
153 		});
154 		var msg = bytes(w -> { w.writeTag(1, WireType.LEN); w.writeLenDelimited(entry); });
155 		var b = ProtobufParser.DEFAULT.read(msg, StrMap.class);
156 		assertEquals(Integer.valueOf(1), b.m.get("a"));
157 	}

184 	@Test
185 	void a06_enumOrdinalNegative_fallsBackToRawOrdinal() throws Exception {
186 		// A varint whose low 32 bits are all set decodes to ordinal -1 after the (int) narrowing cast in
187 		// decodeScalar's ENUM_INT arm, hitting the "ordinal >= 0" false branch (L272) -- falls back to the raw
188 		// ordinal (L274) rather than indexing into the enum's constants array. The raw Integer -1 is then
189 		// rejected by the bean-property setter's own int-to-enum conversion, which is a separate, expected
190 		// failure downstream of the branch under test here.
191 		var msg = bytes(w -> { w.writeTag(1, WireType.VARINT); w.writeVarint(0xFFFFFFFFL); });
192 		var ex = assertThrows(org.apache.juneau.marshall.parser.ParseException.class, () -> ProtobufParser.DEFAULT.read(msg, EnumBean.class));
193 		assertTrue(ex.getCause() instanceof InvalidDataConversionException);
194 	}
195 
196 	@Test
197 	void a07_enumOrdinalOutOfRange_fallsBackToRawOrdinal() throws Exception {
198 		// Ordinal 99 exceeds E's 3 constants, hitting the "ordinal < constants.length" false branch (L272) --
199 		// falls back to the raw ordinal (L274) rather than indexing out of bounds. As in a06, the raw Integer is
200 		// then rejected by the setter's int-to-enum conversion, which is expected and outside this branch's scope.
201 		var msg = bytes(w -> { w.writeTag(1, WireType.VARINT); w.writeVarint(99); });
202 		var ex = assertThrows(org.apache.juneau.marshall.parser.ParseException.class, () -> ProtobufParser.DEFAULT.read(msg, EnumBean.class));
203 		assertTrue(ex.getCause() instanceof InvalidDataConversionException);
204 	}

218 	@Test
219 	void a08_beanRuntimeExceptionFromSetterPropagates() throws Exception {
220 		// Missing setters are silently ignored by default (MarshallingContext.Builder#disableIgnoreMissingSetters()
221 		// defaults to false), so disableIgnoreMissingSetters() is needed to force BeanPropertyMeta.set() to actually
222 		// throw for the getter-only "name" property, exercising setProperty's catch/rethrow (L233-237).
223 		var ser = ProtobufSerializer.create().build();
224 		var par = ProtobufParser.create().disableIgnoreMissingSetters().build();
225 		var bytes = ser.write(new GetterOnlyBean(5));
226 		var ex = assertThrows(org.apache.juneau.marshall.parser.ParseException.class, () -> par.read(bytes, GetterOnlyBean.class));
227 		assertTrue(ex.getCause() instanceof BeanRuntimeException);
228 	}
