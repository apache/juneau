308 	// Temporal values carried as map values flow un-swapped, exercising the date/calendar/temporal/duration/period
309 	// scalar string paths in both the serializer and parser sessions (bean-property values are swapped by BeanMap).
310 	public static class C03_TemporalMaps {
311 		public java.util.Map<String,java.util.Date> dates;
312 		public java.util.Map<String,java.util.Calendar> cals;
313 		public java.util.Map<String,LocalDate> lds;
314 		public java.util.Map<String,Duration> durs;
315 		public java.util.Map<String,Period> pers;
316 		public C03_TemporalMaps() { /* Public no-arg constructor required for Juneau bean recognition. */ }
317 	}
318 
319 	@Test
320 	void c03_temporalMapValuesRoundTrip() throws Exception {
321 		var a = new C03_TemporalMaps();
322 		var d = new java.util.Date(1_700_000_000_000L);
323 		var cal = java.util.GregorianCalendar.from(ZonedDateTime.parse("2024-01-02T03:04:05Z"));
324 		a.dates = java.util.Map.of("k", d);
325 		a.cals = java.util.Map.of("k", cal);
326 		a.lds = java.util.Map.of("k", LocalDate.of(2024, Month.JANUARY, 2));
327 		a.durs = java.util.Map.of("k", Duration.ofSeconds(90));
328 		a.pers = java.util.Map.of("k", Period.of(1, 2, 3));
329 		var bytes = ProtobufSerializer.DEFAULT.write(a);
330 		var b = ProtobufParser.DEFAULT.read(bytes, C03_TemporalMaps.class);
331 		assertEquals(d, b.dates.get("k"));
332 		assertEquals(cal.toInstant(), b.cals.get("k").toInstant());
333 		assertEquals(LocalDate.of(2024, Month.JANUARY, 2), b.lds.get("k"));
334 		assertEquals(Duration.ofSeconds(90), b.durs.get("k"));
335 		assertEquals(Period.of(1, 2, 3), b.pers.get("k"));
336 	}
