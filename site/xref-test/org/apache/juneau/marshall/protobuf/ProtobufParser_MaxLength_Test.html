19  import static org.apache.juneau.BasicTestUtils.*;
20  import static org.apache.juneau.test.bct.BctAssertions.*;
21  import static org.junit.jupiter.api.Assertions.*;
22  
23  import java.io.*;
24  
25  import org.apache.juneau.*;
26  import org.apache.juneau.marshall.protobuf.ProtobufSerializer_Test.*;
27  import org.junit.jupiter.api.*;

29  /**
30   * Bounds-check tests for the configurable wire-length maximum on {@link ProtobufParser}, mirroring the
31   * BSON/MsgPack/CBOR coverage.
32   *
33   * <p>
34   * A length-delimited block carries a varint length prefix used to size a {@code byte[]} before any payload
35   * byte is read.  These tests confirm that an oversized (or wrapped-negative) declared length is rejected up
36   * front rather than driving a large allocation.
37   */
38  class ProtobufParser_MaxLength_Test extends TestBase {
39  
40  	private static byte[] bytes(int...v) {
41  		var b = new byte[v.length];
42  		for (var i = 0; i < v.length; i++)
43  			b[i] = (byte)v[i];
44  		return b;
45  	}
46  
47  	// ================================================================
48  	// Reader-level guard
49  	// ================================================================
50  
51  	@Test
52  	void a01_oversizedDeclaredLengthRejected() {
53  		// Length-delimited block declaring 2^31-1 bytes (varint FF FF FF FF 07); a ~5-byte input must not
54  		// size a giant buffer.
55  		var r = new ProtobufReader(bytes(0xFF, 0xFF, 0xFF, 0xFF, 0x07));
56  		assertThrowsWithMessage(IOException.class, "exceeds maximum", r::readLenDelimited);
57  	}
58  
59  	@Test
60  	void a02_wrappedNegativeLengthRejected() {
61  		// A varint that decodes to a negative long (bit 63 set) must be rejected as negative rather than
62  		// truncating to a bogus positive int.
63  		var r = new ProtobufReader(bytes(0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0x01));
64  		assertThrowsWithMessage(IOException.class, "negative", r::readLenDelimited);
65  	}
66  
67  	@Test
68  	void a03_configurableMaxLengthOnReader() throws Exception {
69  		// A 4-byte block parses under a generous cap but is rejected under a tiny cap.
70  		var lenient = new ProtobufReader(bytes(0x04, 0xAA, 0xBB, 0xCC, 0xDD));
71  		assertEquals(4, lenient.readLenDelimited().length);
72  
73  		var strict = new ProtobufReader(bytes(0x04, 0xAA, 0xBB, 0xCC, 0xDD));
74  		strict.setMaxLength(2);
75  		assertThrowsWithMessage(IOException.class, "exceeds maximum", strict::readLenDelimited);
76  	}
77  
78  	@Test
79  	void a04_maxLengthZeroDisablesCap() throws Exception {
80  		var r = new ProtobufReader(bytes(0x04, 0xAA, 0xBB, 0xCC, 0xDD));
81  		r.setMaxLength(0);
82  		assertEquals(4, r.readLenDelimited().length);
83  	}
84  
85  	@Test
86  	void a05_subReaderInheritsMaxLength() throws Exception {
87  		// Outer block (length 2) is within the cap, but its inner varint declares 300, which exceeds the
88  		// propagated cap.  Without propagation the sub-reader would default to 16 MiB and mis-report.
89  		var outer = new ProtobufReader(bytes(0x02, 0xAC, 0x02));
90  		outer.setMaxLength(2);
91  		var sub = outer.readLenDelimitedReader();
92  		assertThrowsWithMessage(IOException.class, "exceeds maximum", sub::readLenDelimited);
93  	}
94  
95  	// ================================================================
96  	// End-to-end parser
97  	// ================================================================
98  
99  	@Test
100 	void b01_oversizedLenFieldRejectedEndToEnd() {
101 		// Field 2 (name, LEN/string) declaring a 2^31-1 byte length; a ~6-byte hostile payload must not OOM.
102 		assertThrowsWithMessage(IOException.class, "exceeds maximum",
103 			() -> ProtobufParser.DEFAULT.read(bytes(0x12, 0xFF, 0xFF, 0xFF, 0xFF, 0x07), Simple.class));
104 	}
105 
106 	@Test
107 	void b02_legitimatePayloadUnaffected() throws Exception {
108 		// A normal small string field still parses under the default cap.
109 		var b = ProtobufParser.DEFAULT.read(bytes(0x12, 0x07, 0x74, 0x65, 0x73, 0x74, 0x69, 0x6E, 0x67), Simple.class);
110 		assertBean(b, "name", "testing");
111 	}
112 
113 	// ================================================================
114 	// Configurable cap wiring
115 	// ================================================================
116 
117 	@Test
118 	void c01_maxLengthAffectsCacheKey() {
119 		// Different maxLength values must NOT collide in the parser cache (hashKey wiring).
120 		var p1 = ProtobufParser.create().maxLength(100).build();
121 		var p2 = ProtobufParser.create().maxLength(200).build();
122 		var p3 = ProtobufParser.create().maxLength(100).build();
123 		assertNotSame(p1, p2);
124 		assertSame(p1, p3);
125 		assertEquals(100, p1.getMaxLength());
126 		assertEquals(200, p2.getMaxLength());
127 	}
128 }
