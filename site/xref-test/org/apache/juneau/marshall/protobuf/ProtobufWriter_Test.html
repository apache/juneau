29  /**
30   * Golden-byte tests for {@link ProtobufWriter} wire primitives.
31   */
32  @SuppressWarnings({
33  	"resource" // w wraps an in-memory ByteArrayOutputStream fixture; flush() is sufficient and closure is irrelevant to these golden-byte assertions
34  })
35  class ProtobufWriter_Test extends TestBase {
36  
37  	private static String write(ThrowingConsumer<ProtobufWriter> c) {
38  		var baos = new ByteArrayOutputStream();
39  		var w = new ProtobufWriter(baos);
40  		c.accept(w);
41  		w.flush();
42  		return toSpacedHex(baos.toByteArray());
43  	}
44  
45  	@Test
46  	void a01_varintSmall() {
47  		assertEquals("00", write(w -> w.writeVarint(0)));
48  		assertEquals("01", write(w -> w.writeVarint(1)));
49  		assertEquals("96 01", write(w -> w.writeVarint(150)));
50  		assertEquals("AC 02", write(w -> w.writeVarint(300)));
51  	}
52  
53  	@Test
54  	void a02_varintNegativeInt32Is10Bytes() {
55  		// int32 -1 sign-extends to 64 bits -> 10-byte varint.
56  		assertEquals("FF FF FF FF FF FF FF FF FF 01", write(w -> w.writeVarint(-1L)));
57  	}
58  
59  	@Test
60  	void a03_zigzag32() {
61  		assertEquals("00", write(w -> w.writeZigZag32(0)));
62  		assertEquals("01", write(w -> w.writeZigZag32(-1)));
63  		assertEquals("02", write(w -> w.writeZigZag32(1)));
64  		assertEquals("03", write(w -> w.writeZigZag32(-2)));
65  		assertEquals("04", write(w -> w.writeZigZag32(2)));
66  	}
67  
68  	@Test
69  	void a04_zigzag64() {
70  		assertEquals("00", write(w -> w.writeZigZag64(0L)));
71  		assertEquals("01", write(w -> w.writeZigZag64(-1L)));
72  		assertEquals("02", write(w -> w.writeZigZag64(1L)));
73  	}
74  
75  	@Test
76  	void a05_fixed() {
77  		// little-endian
78  		assertEquals("00 00 80 3F", write(w -> w.writeFixed32(Float.floatToIntBits(1.0f))));
79  		assertEquals("00 00 00 00 00 00 F0 3F", write(w -> w.writeFixed64(Double.doubleToLongBits(1.0))));
80  	}
81  
82  	@Test
83  	void a06_tag() {
84  		assertEquals("08", write(w -> w.writeTag(1, WireType.VARINT)));
85  		assertEquals("12", write(w -> w.writeTag(2, WireType.LEN)));
86  		assertEquals("22", write(w -> w.writeTag(4, WireType.LEN)));
87  		// field 16 -> 2-byte tag.
88  		assertEquals("82 01", write(w -> w.writeTag(16, WireType.LEN)));
89  	}
90  
91  	@Test
92  	void a07_stringLenDelimited() {
93  		assertEquals("07 74 65 73 74 69 6E 67", write(w -> w.writeLenDelimited("testing".getBytes(UTF_8))));
94  		assertEquals("07 74 65 73 74 69 6E 67", write(w -> w.writeString("testing")));
95  		assertEquals("00", write(w -> w.writeString("")));
96  	}
97  }
