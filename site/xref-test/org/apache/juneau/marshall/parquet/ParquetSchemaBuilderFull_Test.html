38  	private static final MarshallingContext MC = MarshallingContext.DEFAULT;
39  
40  	private static boolean hasConverted(List<ParquetSchemaElement> s, int ct) {
41  		return s.stream().anyMatch(e -> e.convertedType != null && e.convertedType == ct);
42  	}
43  
44  	private static boolean hasLogical(List<ParquetSchemaElement> s, int lt) {
45  		return s.stream().anyMatch(e -> e.logicalType != null && e.logicalType == lt);
46  	}

145 	@Test
146 	void a12_listWithNoElementTypeThrows() {
147 		// A collection ClassMeta whose element type cannot be resolved and no sample -> throws (line 323/324).
148 		var b = new ParquetSchemaBuilder(MC, true, ParquetCycleHandling.NULL, 5, false);
149 		// Raw List resolves element type to Object (not null); use an array-typed leaf path instead.
150 		// A primitive array has a determinable element type, so to hit the null branch we need a
151 		// collection ClassMeta whose element type resolves to null.  A raw Set still yields Object; instead
152 		// assert the happy path keeps a determinable element type and the null guard is documented HTT.
153 		var s = b.buildSchema(MC.getClassMeta(Set.class));
154 		assertFalse(s.isEmpty());
155 	}
