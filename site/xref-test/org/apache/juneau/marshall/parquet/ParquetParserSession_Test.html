73  	@Test void a03_badMagicAtEnd_throws() {
74  		// MAGIC at start but not at end => "Invalid Parquet magic"
75  		var bytes = new byte[16];
76  		bytes[0] = 'P';
77  		bytes[1] = 'A';
78  		bytes[2] = 'R';
79  		bytes[3] = '1';

82  	}

84  	@Test void a04_badFooterLen_throws() {
85  		// MAGIC at both ends but footerLen too large -> footerStart < 4
86  		var bytes = new byte[16];
87  		bytes[0] = 'P'; bytes[1] = 'A'; bytes[2] = 'R'; bytes[3] = '1';
88  		bytes[12] = 'P'; bytes[13] = 'A'; bytes[14] = 'R'; bytes[15] = '1';
89  		// readLe4 of bytes[8..12] is zero by default → footerStart = 16 - 8 - 0 = 8 OK
90  		// To force footerStart < 4, set footerLen to something huge.
91  		bytes[8] = (byte)0xFF;
92  		bytes[9] = (byte)0xFF;
93  		bytes[10] = (byte)0xFF;
94  		bytes[11] = (byte)0x7F;
95  		assertThrows(ParseException.class, () -> ParquetParser.DEFAULT.read(bytes, Object.class));
96  	}

98  	@Test void a05_corruptFooter_throws() throws Exception {
99  		// Magic OK but footer body bytes are zero / invalid Thrift => readFileMetaData throws
100 		var bytes = new byte[16];
101 		bytes[0] = 'P'; bytes[1] = 'A'; bytes[2] = 'R'; bytes[3] = '1';
102 		bytes[12] = 'P'; bytes[13] = 'A'; bytes[14] = 'R'; bytes[15] = '1';
103 		// footerLen=0 => empty footer → readFileMetaData with empty bytes throws or returns no-rows
104 		// Just exercise the end-to-end path; outcome may be ParseException OR returns null/empty.
105 		try {
106 			var result = ParquetParser.DEFAULT.read(bytes, Object.class);
107 			// If it doesn't throw, allow null/empty.
108 			assertTrue(result == null || result instanceof Map || result instanceof List);
109 		} catch (@SuppressWarnings("unused") ParseException expected) {
110 			// Acceptable - we're hitting the error path.
111 		}
112 	}

114 	private static byte[] buildParquetWithNumRows(long numRows) throws IOException {
115 		var footer = ThriftCompactEncoder.encodeToBytes(enc -> {
116 			enc.writeStructBegin();
117 			enc.writeFieldBegin(ThriftCompactEncoder.I64, 3); // field 3 = num_rows
118 			enc.writeI64(numRows);
119 			enc.writeStructEnd();
120 		});
121 		var out = new ByteArrayOutputStream();
122 		out.write(new byte[]{'P', 'A', 'R', '1'});
123 		out.write(footer);
124 		int flen = footer.length;
125 		out.write(flen & 0xFF);
126 		out.write((flen >> 8) & 0xFF);
127 		out.write((flen >> 16) & 0xFF);
128 		out.write((flen >> 24) & 0xFF);
129 		out.write(new byte[]{'P', 'A', 'R', '1'});
130 		return out.toByteArray();
131 	}
