28  /**
29   * Additional coverage for {@link ParquetParserSession#doRead}'s {@code type.isMap()} branch family: the
30   * single-row {@code {value: Map}} unwrap distinct from the {@code {root: Map}} case, the non-string-key
31   * key/value-pair-format detection when the actual data is NOT in that shape, and the {@code valueType==null}
32   * fallback to {@code Object} when reading a non-string-keyed map without an explicit value type.
33   */
34  @SuppressWarnings({
35  	"unchecked" // ParquetParser.read() returns a raw Map that the tests cast to the expected Map<K,V>.
36  })
37  class ParquetParserSessionMapTarget_Test extends TestBase {
38  
39  	@Test void a01_singleKeyLiterallyValue_unwrapsInnerMap() throws Exception {
40  		// A root Map whose only entry's key is literally "value" (not "root") pointing to another Map.
41  		// Exercises the m.containsKey("value") && m.size()==1 branch (distinct from the {root:{...}} case).
42  		var inner = new LinkedHashMap<String,Object>();
43  		inner.put("a", 1);
44  		var in = new LinkedHashMap<String,Object>();
45  		in.put("value", inner);
46  		var bytes = ParquetSerializer.DEFAULT.write(in);
47  		var out = (Map<?,?>) ParquetParser.DEFAULT.read(bytes, Map.class);
48  		assertNotNull(out);
49  	}
50  
51  	@Test void a02_singleKeyValue_pointingToNonMap_fallsThroughToNormalPath() throws Exception {
52  		// Row has containsKey("value") but the map has more than one key, so the single-key-unwrap
53  		// guard (m.size()==1) is false; falls through to the generic prepareMapForBean(m, type) path.
54  		var in = new LinkedHashMap<String,Object>();
55  		in.put("value", "x");
56  		in.put("other", "y");
57  		var bytes = ParquetSerializer.DEFAULT.write(in);
58  		var out = (Map<String,Object>) ParquetParser.DEFAULT.read(bytes, Map.class, String.class, Object.class);
59  		assertEquals("x", out.get("value"));
60  		assertEquals("y", out.get("other"));
61  	}
62  
63  	@Test void b01_nonStringKeyMap_withoutExplicitValueType_defaultsToObject() throws Exception {
64  		// Map<Integer,?> read with only the key type specified (no value type argument): exercises the
65  		// valueType==null -> ctx.getMarshallingContext().getClassMeta(Object.class) fallback.
66  		var in = new LinkedHashMap<Integer,String>();
67  		in.put(1, "one");
68  		in.put(2, "two");
69  		var bytes = ParquetSerializer.DEFAULT.write(in);
70  		var out = (Map<Integer,Object>) ParquetParser.DEFAULT.read(bytes, Map.class, Integer.class);
71  		assertEquals("one", out.get(1));
72  		assertEquals("two", out.get(2));
73  	}
74  
75  	@Test void c01_nonStringKeyType_dataNotInKeyValuePairFormat_skipsPairBranch() throws Exception {
76  		// The actual wire data is a plain String-keyed map (flat columns "1"/"2", numeric so the eventual
77  		// key conversion to Integer succeeds), but we deliberately request a non-String key type.
78  		// isKeyValuePairFormat(rows) inspects the *actual* row shape (first row lacks "key"/"value" keys
79  		// entirely) and returns false, so the isMap() block falls through to the plain single-row-map
80  		// unwrap instead of the key/value-pair reconstruction.
81  		var in = new LinkedHashMap<String,Object>();
82  		in.put("1", "one");
83  		in.put("2", "two");
84  		var bytes = ParquetSerializer.DEFAULT.write(in);
85  		var out = (Map<Integer,String>) ParquetParser.DEFAULT.read(bytes, Map.class, Integer.class, String.class);
86  		assertNotNull(out);
87  		assertEquals("one", out.get(1));
88  	}
89  
90  	@Test void d01_intKeyMap_multipleRows_keyValuePairFormat_valueTypeExplicit() throws Exception {
91  		// Sanity check that the key/value-pair branch (valueType != null) still round-trips correctly
92  		// alongside the valueType==null variant above.
93  		Map<Integer,String> in = map(5, "five", 6, "six", 7, "seven");
94  		var bytes = ParquetSerializer.DEFAULT.write(in);
95  		var out = (Map<Integer,String>) ParquetParser.DEFAULT.read(bytes, Map.class, Integer.class, String.class);
96  		assertSize(3, out);
97  		assertEquals("seven", out.get(7));
98  	}
99  }
