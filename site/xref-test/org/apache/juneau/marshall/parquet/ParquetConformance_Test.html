37  	// GAP-12: a statically-typed Number leaf must not be silently narrowed to INT32.
38  	//-----------------------------------------------------------------------------------------------------------------
39  
40  	public static class NumBean {
41  		public Number n;
42  	}
43  
44  	@Test
45  	void gap12_numberFieldAboveIntRangeNotTruncated() throws Exception {
46  		// > Integer.MAX_VALUE: the old INT32 mapping corrupted this to 705032704.  DOUBLE preserves it exactly.
47  		var big = new NumBean();
48  		big.n = 5_000_000_000L;
49  		var bytes = ParquetSerializer.DEFAULT.write(big);
50  		var parsed = (List<NumBean>) ParquetParser.DEFAULT.read(bytes, List.class, NumBean.class);
51  
52  		assertEquals(1, parsed.size());
53  		assertNotNull(parsed.get(0).n);
54  		assertEquals(5_000_000_000L, parsed.get(0).n.longValue());
55  	}
56  
57  	@Test
58  	void gap12_fractionalNumberFieldNotTruncated() throws Exception {
59  		// The old INT32 mapping truncated 3.14 to 3.
60  		var frac = new NumBean();
61  		frac.n = 3.14d;
62  		var bytes = ParquetSerializer.DEFAULT.write(frac);
63  		var parsed = (List<NumBean>) ParquetParser.DEFAULT.read(bytes, List.class, NumBean.class);
64  
65  		assertEquals(3.14d, parsed.get(0).n.doubleValue(), 0.0);
66  	}

69  	// GAP-13: acyclic data flowing through a recursive *type* must round-trip as nested beans, not toString garbage.
70  	//-----------------------------------------------------------------------------------------------------------------
71  
72  	public static class Node {
73  		public String val;
74  		public Node next;
75  	}
76  
77  	@Test
78  	void gap13_acyclicRecursiveTypeRoundTrips() throws Exception {
79  		var a = new Node();
80  		a.val = "a";
81  		a.next = new Node();
82  		a.next.val = "b";
83  		a.next.next = new Node();
84  		a.next.next.val = "c";
85  
86  		var bytes = ParquetSerializer.DEFAULT.write(a);
87  		var parsed = (List<Node>) ParquetParser.DEFAULT.read(bytes, List.class, Node.class);
88  
89  		assertEquals(1, parsed.size());
90  		var n = parsed.get(0);
91  		assertEquals("a", n.val);
92  		assertNotNull(n.next);
93  		assertEquals("b", n.next.val);
94  		assertNotNull(n.next.next);
95  		assertEquals("c", n.next.next.val);
96  		assertNull(n.next.next.next);
97  	}

99  	@Test
100 	void gap13_recursionBeyondMaxDepthTruncatesToNull() throws Exception {
101 		// Depth-2 limit: only the root + one level of "next" survive; deeper links truncate to null.
102 		var ser = ParquetSerializer.create().maxRecursionDepth(2).build();
103 		var a = new Node();
104 		a.val = "a";
105 		a.next = new Node();
106 		a.next.val = "b";
107 		a.next.next = new Node();
108 		a.next.next.val = "c";
109 
110 		var bytes = ser.write(a);
111 		var parsed = (List<Node>) ParquetParser.DEFAULT.read(bytes, List.class, Node.class);
112 
113 		var n = parsed.get(0);
114 		assertEquals("a", n.val);
115 		assertNotNull(n.next);
116 		assertEquals("b", n.next.val);
117 		assertNull(n.next.next);
118 	}

121 	// GAP-14: multi-level optional nesting must distinguish a null intermediate group from a present group with a
122 	// null leaf.
123 	//-----------------------------------------------------------------------------------------------------------------
124 
125 	public static class B {
126 		public String v;
127 	}
128 
129 	public static class A {
130 		public B b;
131 	}
132 
133 	@Test
134 	void gap14_multiLevelOptionalNullDistinction() throws Exception {
135 		var nullGroup = new A(); // a.b == null
136 		var nullLeaf = new A();
137 		nullLeaf.b = new B(); // a.b != null, a.b.v == null
138 		var present = new A();
139 		present.b = new B();
140 		present.b.v = "x";
141 
142 		var bytes = ParquetSerializer.DEFAULT.write(list(nullGroup, nullLeaf, present));
143 		var parsed = (List<A>) ParquetParser.DEFAULT.read(bytes, List.class, A.class);
144 
145 		assertEquals(3, parsed.size());
146 		assertNull(parsed.get(0).b, "null intermediate group must reconstruct as null");
147 		assertNotNull(parsed.get(1).b, "present group with null leaf must not collapse to null group");
148 		assertNull(parsed.get(1).b.v);
149 		assertNotNull(parsed.get(2).b);
150 		assertEquals("x", parsed.get(2).b.v);
151 	}

154 	// GAP-4: PLAIN BOOLEAN must be bit-packed 1 bit/value (LSB-first), not 1 byte/value.
155 	//-----------------------------------------------------------------------------------------------------------------
156 
157 	private static ParquetSchemaElement boolColumn() {
158 		return new ParquetSchemaElement("b", ParquetSchemaElement.TYPE_BOOLEAN, null, ParquetSchemaElement.REQUIRED, null, null, null, null, null, "b");
159 	}
160 
161 	@Test
162 	void gap4_booleanPlainIsBitPacked() {
163 		var w = new ParquetColumnWriter(boolColumn());
164 		// true,false,true,false,false,false,false,false -> bits 0b00000101 = 0x05
165 		var pattern = new boolean[]{true, false, true, false, false, false, false, false};
166 		for (var v : pattern)
167 			w.writeBoolean(v);
168 		var page = w.finalizePage();
169 		assertEquals(1, page.length, "8 booleans must pack into a single byte");
170 		assertEquals((byte) 0x05, page[0]);
171 	}
172 
173 	@Test
174 	void gap4_booleanPartialTrailingByteFlushed() {
175 		var w = new ParquetColumnWriter(boolColumn());
176 		w.writeBoolean(true);
177 		w.writeBoolean(true);
178 		w.writeBoolean(false);
179 		var page = w.finalizePage();
180 		assertEquals(1, page.length, "3 booleans still occupy exactly one packed byte");
181 		assertEquals((byte) 0x03, page[0]);
182 	}
183 
184 	@Test
185 	void gap4_booleanBitPackRoundTrip() throws Exception {
186 		var w = new ParquetColumnWriter(boolColumn());
187 		var pattern = new boolean[]{true, true, false, true, false, false, true, false, true, true};
188 		for (var v : pattern)
189 			w.writeBoolean(v);
190 		var page = w.finalizePage();
191 		assertEquals(2, page.length, "10 booleans pack into 2 bytes");
192 
193 		var r = new ParquetColumnReader(page, pattern.length, 0);
194 		for (var expected : pattern) {
195 			assertTrue(r.hasNext());
196 			r.advance();
197 			assertEquals(expected, r.readBoolean());
198 		}
199 	}

205 	@Test
206 	void gap5_supportedCodecsResolve() throws Exception {
207 		assertEquals(CompressionCodec.UNCOMPRESSED, CompressionCodec.fromThrift(0));
208 		assertEquals(CompressionCodec.GZIP, CompressionCodec.fromThrift(2));
209 		// SNAPPY (1) now resolves — decode-only support was added for external-read interop.
210 		assertEquals(CompressionCodec.SNAPPY, CompressionCodec.fromThrift(1));
211 	}
212 
213 	@Test
214 	void gap5_unsupportedCodecThrows() {
215 		// Every known-but-unsupported codec id (and an unknown id) is a hard error with a descriptive name.
216 		assertCodecError(3, "LZO");
217 		assertCodecError(4, "BROTLI");
218 		assertCodecError(5, "LZ4");
219 		assertCodecError(6, "ZSTD");
220 		assertCodecError(7, "LZ4_RAW");
221 		var unknown = assertThrows(IOException.class, () -> CompressionCodec.fromThrift(99));
222 		assertTrue(unknown.getMessage().contains("99"), "Expected raw id in message: " + unknown.getMessage());
223 	}
224 
225 	private static void assertCodecError(int thriftValue, String expectedName) {
226 		var ex = assertThrows(IOException.class, () -> CompressionCodec.fromThrift(thriftValue));
227 		assertTrue(ex.getMessage().contains(expectedName), "Expected '" + expectedName + "' in message: " + ex.getMessage());
228 	}
