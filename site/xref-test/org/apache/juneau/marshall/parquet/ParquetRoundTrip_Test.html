116 	void a07a_schemaWithSampleForListOfBean() {
117 		var members = List.of(new Person("Alice", 30), new Person("Bob", 25));
118 		var in = new Team("devs", members);
119 		var bc = MarshallingContext.DEFAULT;
120 		var beanMap = bc.toBeanMap(in);
121 		var schema = new ParquetSchemaBuilder(bc, false, ParquetCycleHandling.NULL).buildSchema(bc.getClassMeta(Team.class), beanMap);
122 		var leaves = ParquetSchemaBuilder.getLeafColumns(schema);
123 		var leafPaths = leaves.stream().map(e -> e.path).sorted().toList();
124 		// With sample, expect root.name, root.members.list.element.name, root.members.list.element.age
125 		assertEquals(List.of("root.members.list.element.age", "root.members.list.element.name", "root.name"), leafPaths);
126 	}

152 	@Test
153 	void a09_scalarAndArrayUnwrap() throws Exception {
154 		// 2.1: Top-level scalar/array - parser unwraps {value: X} to X
155 		assertEquals("foobar", ParquetParser.DEFAULT.read(ParquetSerializer.DEFAULT.write("foobar"), String.class));
156 		assertEquals(123, ParquetParser.DEFAULT.read(ParquetSerializer.DEFAULT.write(123), Integer.class));
157 		assertEquals(123, ParquetParser.DEFAULT.read(ParquetSerializer.DEFAULT.write(123), int.class).intValue());
158 		assertEquals(true, ParquetParser.DEFAULT.read(ParquetSerializer.DEFAULT.write(true), Boolean.class));
159 		var intArr = ParquetParser.DEFAULT.read(ParquetSerializer.DEFAULT.write(new int[]{1, 2, 3}), int[].class);
160 		assertArrayEquals(new int[]{1, 2, 3}, intArr);
161 		var strArr = ParquetParser.DEFAULT.read(ParquetSerializer.DEFAULT.write(new String[]{"a", "b"}), String[].class);
162 		assertArrayEquals(new String[]{"a", "b"}, strArr);
163 	}

256 	@Test
257 	void a12_beanWithNestedMapRoundTrip() throws Exception {
258 		// 2.1: Nested Map in beans - key_value repeated group
259 		var empty = new BeanWithNestedMap();
260 		empty.name = "empty";
261 		empty.statusMap = map();
262 		var withEntries = new BeanWithNestedMap();
263 		withEntries.name = "withEntries";
264 		withEntries.statusMap = new LinkedHashMap<>();
265 		withEntries.statusMap.put(TestEnum.FOO, TestEnum.BAR);
266 		withEntries.statusMap.put(TestEnum.BAR, TestEnum.FOO);
267 
268 		var bytesEmpty = ParquetSerializer.DEFAULT.write(list(empty));
269 		var parsedEmpty = (List<BeanWithNestedMap>) ParquetParser.DEFAULT.read(bytesEmpty, List.class, BeanWithNestedMap.class);
270 		assertBeans(parsedEmpty, "name,statusMap", "empty,{}");
271 
272 		var bytesWithEntries = ParquetSerializer.DEFAULT.write(list(withEntries));
273 		var parsedWithEntries = (List<BeanWithNestedMap>) ParquetParser.DEFAULT.read(bytesWithEntries, List.class, BeanWithNestedMap.class);
274 		assertEquals(1, parsedWithEntries.size());
275 		assertEquals("withEntries", parsedWithEntries.get(0).name);
276 		var m = parsedWithEntries.get(0).statusMap;
277 		assertEquals(2, m.size());
278 		assertEquals(TestEnum.BAR, m.get(TestEnum.FOO));
279 		assertEquals(TestEnum.FOO, m.get(TestEnum.BAR));
280 	}
