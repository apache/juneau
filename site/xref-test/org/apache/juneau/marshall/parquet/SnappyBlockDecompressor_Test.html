31  	private static byte[] literalBlock(byte[] payload) {
32  		// varint length (single byte for <128) + inline literal tag + payload.
33  		var b = new byte[2 + payload.length];
34  		b[0] = (byte) payload.length;
35  		b[1] = (byte) ((payload.length - 1) << 2);
36  		System.arraycopy(payload, 0, b, 2, payload.length);
37  		return b;
38  	}

40  	@Test
41  	void a01_expectedSizeNegativeSkipsMismatchCheck() throws Exception {
42  		// expectedSize < 0 bypasses the length-equality guard (line 61 false branch).
43  		var payload = "abc".getBytes();
44  		var decoded = SnappyBlockDecompressor.decompress(literalBlock(payload), -1);
45  		assertArrayEquals(payload, decoded);
46  	}

48  	@Test
49  	void a02_fourByteOffsetCopy() throws Exception {
50  		// 8-byte literal then a 4-byte-offset copy(len=8, offset=8) — exercises the elemType==3 branch.
51  		var lit = new byte[]{0, 1, 2, 3, 4, 5, 6, 7};
52  		var expected = new byte[16];
53  		System.arraycopy(lit, 0, expected, 0, 8);
54  		System.arraycopy(lit, 0, expected, 8, 8);
55  		var p = 0;
56  		var block = new byte[1 + 1 + 8 + 5];
57  		block[p++] = 16;                          // varint length
58  		block[p++] = (byte) ((8 - 1) << 2);       // inline literal len-1=7
59  		System.arraycopy(lit, 0, block, p, 8);
60  		p += 8;
61  		block[p++] = (byte) (0x03 | ((8 - 1) << 2)); // 4-byte-offset copy tag; len-1=7
62  		block[p++] = 8;                            // offset bytes (LE) = 8
63  		block[p++] = 0;
64  		block[p++] = 0;
65  		block[p] = 0;
66  		assertArrayEquals(expected, SnappyBlockDecompressor.decompress(block, 16));
67  	}

69  	@Test
70  	void a03_multiByteVarintLength() throws Exception {
71  		// 200-byte payload: varint length needs two bytes (0xC8 0x01).
72  		var payload = new byte[200];
73  		for (var i = 0; i < 200; i++)
74  			payload[i] = (byte) i;
75  		// literal of 200 bytes: len-1=199 >= 60 -> 1-extra-byte form... 199 fits one byte (0xC7).
76  		var p = 0;
77  		var block = new byte[2 + 2 + 200];
78  		block[p++] = (byte) 0xC8;  // varint 200 low 7 bits
79  		block[p++] = 0x01;         // varint 200 high
80  		block[p++] = (byte) (60 << 2); // literal tag, 1 extra length byte follows
81  		block[p++] = (byte) 199;   // len-1 = 199
82  		System.arraycopy(payload, 0, block, p, 200);
83  		assertArrayEquals(payload, SnappyBlockDecompressor.decompress(block, 200));
84  	}

87  	void a04_invalidUncompressedLengthThrows() {
88  		// varint declares > 256MB -> hard error.  300MB ~ 0x12C00000; encode as varint.
89  		var big = 300 * 1024 * 1024;
90  		var bytes = new byte[]{
91  			(byte) (big & 0x7F | 0x80),
92  			(byte) ((big >>> 7) & 0x7F | 0x80),
93  			(byte) ((big >>> 14) & 0x7F | 0x80),
94  			(byte) ((big >>> 21) & 0x7F | 0x80),
95  			(byte) ((big >>> 28) & 0x7F)
96  		};
97  		var ex = assertThrows(IOException.class, () -> SnappyBlockDecompressor.decompress(bytes, -1));
98  		assertTrue(ex.getMessage().contains("Invalid Snappy uncompressed length"));
99  	}
