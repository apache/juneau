439 		@Test void c16_writeAfterCloseThrows() throws Exception {
440 			var sb = new StringWriter();
441 			var w = toJsonTokens(sb);
442 			w.startArray().endArray();
443 			w.close();
444 			// Every mutating method must reject writes after close() with the closed-writer message.
445 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", w::startObject);
446 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", w::startArray);
447 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", w::endObject);
448 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", w::endArray);
449 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", () -> w.fieldName("a"));
450 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", () -> w.string("x"));
451 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", () -> w.number(1));
452 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", () -> w.bool(true));
453 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", w::nil);
454 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", () -> w.binary(new byte[]{1}));
455 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", () -> w.object(42));
456 			// close() stays idempotent and flush() after close stays a safe no-op.
457 			assertDoesNotThrow(w::close);
458 			assertDoesNotThrow(w::flush);
459 			// The pre-close content was flushed (BCT string-shape assertion on the produced output).
460 			assertString("[]", sb);
461 		}
