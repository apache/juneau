29  /**
30   * Tests JSON string escaping of carriage returns and control characters (RFC 8259).
31   */
32  class JsonControlCharEscape_Test extends TestBase {
33  
34  	private static String expectedEscape(int c) {
35  		return switch (c) {
36  			case '\b' -> "\\b";
37  			case '\t' -> "\\t";
38  			case '\n' -> "\\n";
39  			case '\f' -> "\\f";
40  			case '\r' -> "\\r";
41  			default -> f("\\u%04x", c);
42  		};
43  	}
44  
45  	// NOTE: ParserReader uses DEL (0x7f) as its internal escape-hole filler, so the parser cannot round-trip a DEL
46  	// character (pre-existing limitation; a raw DEL in input is also dropped).  DEL round-trip is therefore not asserted.
47  	private static boolean roundTrips(int c) {
48  		return c != 0x7f;
49  	}
50  
51  	private static List<Integer> controlChars() {
52  		var l = new ArrayList<Integer>();
53  		for (var c = 0; c <= 0x1f; c++)
54  			l.add(c);
55  		l.add(0x7f);
56  		return l;
57  	}
58  
59  	@Test void a01_valueEscapes_json() throws Exception {
60  		for (var c : controlChars()) {
61  			var s = "a" + (char)(int)c + "b";
62  			var out = JsonSerializer.DEFAULT.write(s);
63  			assertEquals("\"a" + expectedEscape(c) + "b\"", out, "char 0x" + Integer.toHexString(c));
64  			if (roundTrips(c))
65  				assertEquals(s, JsonParser.DEFAULT.read(out, String.class), "roundtrip 0x" + Integer.toHexString(c));
66  		}
67  	}
68  
69  	@Test void a02_valueEscapes_json5() throws Exception {
70  		for (var c : controlChars()) {
71  			var s = "a" + (char)(int)c + "b";
72  			var out = Json5Serializer.DEFAULT.write(s);
73  			assertEquals("'a" + expectedEscape(c) + "b'", out, "char 0x" + Integer.toHexString(c));
74  			if (roundTrips(c))
75  				assertEquals(s, Json5Parser.DEFAULT.read(out, String.class), "roundtrip 0x" + Integer.toHexString(c));
76  		}
77  	}
78  
79  	@Test void a03_mapKeys_json() throws Exception {
80  		for (var c : controlChars()) {
81  			var k = "k" + (char)(int)c;
82  			var m = new LinkedHashMap<String,Object>();
83  			m.put(k, 1);
84  			var out = JsonSerializer.DEFAULT.write(m);
85  			assertEquals("{\"k" + expectedEscape(c) + "\":1}", out, "char 0x" + Integer.toHexString(c));
86  			var m2 = JsonParser.DEFAULT.read(out, Map.class);
87  			assertSize(1, m2);
88  			if (roundTrips(c))
89  				assertTrue(m2.containsKey(k), "roundtrip key 0x" + Integer.toHexString(c));
90  		}
91  	}
92  
93  	@Test void a04_mapKeys_json5() throws Exception {
94  		for (var c : controlChars()) {
95  			var k = "k" + (char)(int)c;
96  			var m = new LinkedHashMap<String,Object>();
97  			m.put(k, 1);
98  			var out = Json5Serializer.DEFAULT.write(m);
99  			assertEquals("{'k" + expectedEscape(c) + "':1}", out, "char 0x" + Integer.toHexString(c));
100 			var m2 = Json5Parser.DEFAULT.read(out, Map.class);
101 			assertSize(1, m2);
102 			if (roundTrips(c))
103 				assertTrue(m2.containsKey(k), "roundtrip key 0x" + Integer.toHexString(c));
104 		}
105 	}
106 
107 	@Test void a05_crlf() throws Exception {
108 		var s = "line1\r\nline2\rline3\n";
109 		var m = new LinkedHashMap<String,Object>();
110 		m.put("a\r\nb", s);
111 		var json = "{\"a\\r\\nb\":\"line1\\r\\nline2\\rline3\\n\"}";
112 		assertEquals(json, JsonSerializer.DEFAULT.write(m));
113 		assertEquals(s, JsonParser.DEFAULT.read(json, Map.class).get("a\r\nb"));
114 		assertEquals("{'a\\r\\nb':'line1\\r\\nline2\\rline3\\n'}", Json5Serializer.DEFAULT.write(m));
115 		assertEquals(s, Json5Parser.DEFAULT.read(Json5Serializer.DEFAULT.write(m), Map.class).get("a\r\nb"));
116 	}

118 	@Test void a06_escapeSolidusStillWorks() throws Exception {
119 		var ser = JsonSerializer.create().escapeSolidus().build();
120 		assertEquals("\"a\\/\\r\\u0001\"", ser.write("a/\r\u0001"));
121 		assertEquals("\"a/\\r\\u0001\"", JsonSerializer.DEFAULT.write("a/\r\u0001"));
122 	}
