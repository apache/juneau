17  package org.apache.juneau.marshall.jena;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.net.*;
22  import java.util.concurrent.atomic.*;
23  
24  import org.apache.juneau.*;
25  import org.apache.juneau.marshall.parser.*;
26  import org.junit.jupiter.api.*;
27  
28  import com.sun.net.httpserver.*;
29  
30  /**
31   * Tests that the RDF parsers do not make network requests while parsing.
32   */
33  class RdfParser_NoNetwork_Test extends TestBase {
34  
35  	private HttpServer server;
36  	private final AtomicInteger hits = new AtomicInteger();
37  
38  	@BeforeEach void setUp() throws Exception {
39  		server = HttpServer.create(new InetSocketAddress(InetAddress.getLoopbackAddress(), 0), 0);
40  		server.createContext("/", x -> {
41  			hits.incrementAndGet();
42  			var body = "{\"@context\":{\"name\":\"http://example.com/name\"}}".getBytes();
43  			x.getResponseHeaders().add("Content-Type", "application/ld+json");
44  			x.sendResponseHeaders(200, body.length);
45  			x.getResponseBody().write(body);
46  			x.close();
47  		});
48  		server.start();
49  	}
50  
51  	@AfterEach void tearDown() {
52  		server.stop(0);
53  	}
54  
55  	private String url() {
56  		return "http://127.0.0.1:" + server.getAddress().getPort() + "/ctx.jsonld";
57  	}
58  
59  	@Test void a01_jsonLd_remoteContextNotFetchedByDefault() {
60  		var in = "{\"@context\":\"" + url() + "\",\"@id\":\"http://example.com/a\",\"name\":\"x\"}";
61  		assertThrows(Exception.class, () -> JsonLdParser.DEFAULT.read(in, Object.class));
62  		assertEquals(0, hits.get());
63  	}
64  
65  	@Test void a02_jsonLd_remoteContextFetchedWhenAllowed() {
66  		var in = "{\"@context\":\"" + url() + "\",\"@id\":\"http://example.com/a\",\"name\":\"x\"}";
67  		var p = JsonLdParser.create().allowRemoteContexts().build();
68  		try {
69  			p.read(in, Object.class);
70  		} catch (@SuppressWarnings("unused") Exception e) {
71  			// Result shape is irrelevant here; only the fetch is checked.
72  		}
73  		assertTrue(hits.get() > 0);
74  	}
75  
76  	private static final String OTHER = "http://other.example/doc.dtd";
77  
78  	private static final String RDF_OPEN = "<rdf:RDF xmlns:rdf=\"http://www.w3.org/1999/02/22-rdf-syntax-ns#\" xmlns:e=\"http://example.com/\">";
79  
80  	private static String rdf(String name) {
81  		return RDF_OPEN + "<rdf:Description rdf:about=\"http://example.com/a\"><e:name>" + name + "</e:name></rdf:Description></rdf:RDF>";
82  	}
83  
84  	private static Object readXml(String in) throws Exception {
85  		return RdfParser.create().xml().build().read(in, Object.class);
86  	}
87  
88  	@Test void a03_rdfXml_externalEntityRejected() {
89  		var in = "<?xml version=\"1.0\"?><!DOCTYPE rdf:RDF [<!ENTITY x SYSTEM \"" + url() + "\">]>" + rdf("&x;");
90  		var e = assertThrows(ParseException.class, () -> readXml(in));
91  		assertTrue(e.getMessage().contains("External DTD and entity declarations are not supported."), e.getMessage());
92  		assertEquals(0, hits.get());
93  	}
94  
95  	@Test void a04_rdfXml_externalDtdRejected() {
96  		var in = "<?xml version=\"1.0\"?><!DOCTYPE rdf:RDF SYSTEM \"" + url() + "\">" + rdf("x");
97  		var e = assertThrows(ParseException.class, () -> readXml(in));
98  		assertTrue(e.getMessage().contains("External DTD and entity declarations are not supported."), e.getMessage());
99  		assertEquals(0, hits.get());
100 	}
101 
102 	@Test void a05_rdfXml_externalParameterEntityAndPublicIdRejected() {
103 		assertThrows(ParseException.class, () -> readXml("<!DOCTYPE rdf:RDF [<!ENTITY % p SYSTEM \"" + url() + "\"> %p;]>" + rdf("x")));
104 		assertThrows(ParseException.class, () -> readXml("<!DOCTYPE rdf:RDF PUBLIC \"-//X//EN\" \"" + url() + "\">" + rdf("x")));
105 		assertEquals(0, hits.get());
106 	}
107 
108 	@Test void a06_rdfXml_internalEntityStillParses() throws Exception {
109 		var in = "<?xml version=\"1.0\"?><!DOCTYPE rdf:RDF [<!ENTITY x \"value\">]>" + rdf("&x;");
110 		assertTrue(String.valueOf(readXml(in)).contains("value"));
111 	}
112 
113 	@Test void a07_rdfXml_noDoctypeOrDoctypeTextInCommentOrCdataParses() throws Exception {
114 		assertTrue(String.valueOf(readXml(rdf("x"))).contains("x"));
115 		var comment = "<?xml version=\"1.0\"?><!-- <!DOCTYPE rdf:RDF SYSTEM \"" + url() + "\"> -->" + rdf("y");
116 		assertTrue(String.valueOf(readXml(comment)).contains("y"));
117 		var cdata = rdf("<![CDATA[<!DOCTYPE rdf:RDF SYSTEM \"" + url() + "\">]]>");
118 		assertTrue(String.valueOf(readXml(cdata)).contains("DOCTYPE"));
119 		assertEquals(0, hits.get());
120 	}
121 
122 	@Test void a08_hasExternalDoctype() {
123 		assertFalse(RdfParserSession.hasExternalDoctype(""));
124 		assertFalse(RdfParserSession.hasExternalDoctype("<a/>"));
125 		assertFalse(RdfParserSession.hasExternalDoctype("<!DOCTYPE a>"));
126 		assertFalse(RdfParserSession.hasExternalDoctype("<!DOCTYPE a [<!ENTITY x \"SYSTEM\"><!-- SYSTEM --><!ELEMENT a ANY>]><a/>"));
127 		assertFalse(RdfParserSession.hasExternalDoctype("<a><!DOCTYPE b SYSTEM \"" + OTHER + "\"></a>"));
128 		assertTrue(RdfParserSession.hasExternalDoctype("\uFEFF<?xml version=\"1.0\"?>\n<!DOCTYPE a SYSTEM \"" + OTHER + "\">"));
129 		assertTrue(RdfParserSession.hasExternalDoctype("<!DOCTYPE a PUBLIC \"p\" \"" + OTHER + "\">"));
130 		assertTrue(RdfParserSession.hasExternalDoctype("<!DOCTYPE a [<!ENTITY x \"v\"><!ENTITY y SYSTEM '" + OTHER + "'>]>"));
131 		assertTrue(RdfParserSession.hasExternalDoctype("<!DOCTYPE a [<!ENTITY y PUBLIC 'p' '" + OTHER + "'>]>"));
132 	}
