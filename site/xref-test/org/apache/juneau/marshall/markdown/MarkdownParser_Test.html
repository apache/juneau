66  	@Test void z01_malformedInlineJson5InKeyValueTableThrows() {
67  		// Backtick-wrapped cell content is embedded into the JSON5 object verbatim (unquoted); an
68  		// unbalanced brace here produces invalid JSON5 syntax, exercising the needsJson5Path(...) catch branch.
69  		var md = "| Property | Value |\n|---|---|\n| a | `{invalid` |";
70  		var ex = assertThrows(org.apache.juneau.marshall.parser.ParseException.class, ()->MarkdownParser.DEFAULT.read(md, List.class));
71  		assertTrue(ex.getMessage().contains("Could not parse key-value table"));
72  	}
