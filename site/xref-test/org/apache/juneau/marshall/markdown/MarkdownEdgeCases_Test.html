31  	//-----------------------------------------------------------------------------------------------------------------
32  	// f01 - Very long values
33  	//-----------------------------------------------------------------------------------------------------------------
34  
35  	@Test void f01_veryLongValues() {
36  		var longVal = "x".repeat(500);
37  		var bean = Map.of("key", longVal);
38  		var md = toMarkdown(bean);
39  		assertTrue(md.contains(longVal), "Expected long value in output: " + md.substring(0, Math.min(200, md.length())));
40  	}

42  	//-----------------------------------------------------------------------------------------------------------------
43  	// f02 - Multi-line strings (round-trippable via JSON5 escaping)
44  	//-----------------------------------------------------------------------------------------------------------------
45  
46  	@Test void f02_multiLineStrings() {
47  		var value = "line1\nline2\nline3";
48  		var bean = new LinkedHashMap<String,String>();
49  		bean.put("text", value);
50  		var md = toMarkdown(bean);
51  		// Multi-line strings are wrapped in JSON5 backtick syntax for round-trip correctness
52  		assertTrue(md.contains("`'") && md.contains("\\n"), "Expected JSON5 escaped newline in output: " + md);
53  		// Verify round-trip
54  		var parsed = fromMarkdown(md, Map.class);
55  		assertEquals(value, parsed.get("text"));
56  	}

165 	@Test void f11_deepNesting() {
166 		// Deeply nested structure - Maps render nested maps as inline JSON5; beans get sub-headings
167 		var s = MarkdownDocSerializer.create().title("L1").build();
168 		var nested = Map.<String,Object>of("x", "deep");
169 		var l1 = new LinkedHashMap<String,Object>();
170 		l1.put("nested", nested);
171 		var md = s.write(l1);
172 		assertTrue(md.contains("deep"), "Expected deep value: " + md);
173 		assertNotNull(md);
174 	}

204 	// Helpers keep the marshaller reference fully-qualified (the simple name Markdown is shadowed by the @Markdown annotation in this package).
205 
206 	private static String toMarkdown(Object o) {
207 		return org.apache.juneau.marshall.marshaller.Markdown.of(o);
208 	}
209 
210 	private static <T> T fromMarkdown(String input, Class<T> type) {
211 		return org.apache.juneau.marshall.marshaller.Markdown.to(input, type);
212 	}
