17  package org.apache.juneau.marshall.xml;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.apache.juneau.*;
22  import org.apache.juneau.marshall.json5.*;
23  import org.junit.jupiter.api.*;
24  
25  /**
26   * Verifies that DTD processing stays disabled even when the {@code validating} parser option is enabled.
27   *
28   * <p>
29   * DTD support is not needed for data binding, so it is kept off regardless of the {@code validating} flag:
30   * a document declaring an internal DTD with nested entity references must be rejected rather than expanding
31   * those entities, closing the input-amplification surface that would otherwise be reachable by toggling
32   * {@code validating}.
33   */
34  class XmlValidatingDtd_Test extends TestBase {
35  
36  	@Test void a01_internalDtdRejectedWhenValidatingEnabled() {
37  		var p = XmlParser.create().validating().build();
38  		var xml = "<?xml version=\"1.0\"?>"
39  			+ "<!DOCTYPE A [<!ENTITY a \"aaaaaaaaaa\"><!ENTITY b \"&a;&a;&a;&a;&a;\">]>"
40  			+ "<A>&b;</A>";
41  
42  		// The DTD is not processed, so parsing must fail rather than expanding the nested entities.
43  		var e = assertThrows(Exception.class, () -> p.read(xml, Json5Map.class));
44  
45  		// The expanded entity payload must never appear anywhere in the failure.
46  		assertFalse(e.toString().contains("aaaaaaaaaa"), "Internal DTD entity was expanded");
47  	}
48  
49  	@Test void a02_doctypeRejectedWhenValidatingEnabled() {
50  		var p = XmlParser.create().validating().build();
51  		var xml = "<?xml version=\"1.0\"?>"
52  			+ "<!DOCTYPE A [<!ELEMENT A ANY>]>"
53  			+ "<A>x</A>";
54  
55  		assertThrows(Exception.class, () -> p.read(xml, Json5Map.class));
56  	}
57  }
