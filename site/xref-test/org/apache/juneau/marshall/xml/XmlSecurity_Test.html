37  /**
38   * Exercises external-resource and expansion attacks through Juneau's configured StAX reader.
39   */
40  class XmlSecurity_Test extends TestBase {
41  
42  	private HttpServer server;
43  	private final AtomicInteger requests = new AtomicInteger();
44  	private String external;
45  
46  	@BeforeEach void startServer() throws Exception {
47  		server = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
48  		server.createContext("/external", exchange -> {
49  			requests.incrementAndGet();
50  			var body = "<!ENTITY injected 'FETCHED'>".getBytes(StandardCharsets.UTF_8);
51  			exchange.sendResponseHeaders(200, body.length);
52  			try (var stream = exchange.getResponseBody()) {
53  				stream.write(body);
54  			}
55  		});
56  		server.start();
57  		external = "http://127.0.0.1:" + server.getAddress().getPort() + "/external";
58  	}
59  
60  	@AfterEach void stopServer() {
61  		if (server != null)
62  			server.stop(0);
63  	}
64  
65  	private static String externalDocument(String vector, String uri) {
66  		return switch (vector) {
67  			case "general" -> "<!DOCTYPE A [<!ENTITY ext SYSTEM '" + uri + "'>]><A>&ext;</A>";
68  			case "parameter" -> "<!DOCTYPE A [<!ENTITY % ext SYSTEM '" + uri + "'>%ext;]><A>safe</A>";
69  			case "systemDtd" -> "<!DOCTYPE A SYSTEM '" + uri + "'><A>safe</A>";
70  			case "publicDtd" -> "<!DOCTYPE A PUBLIC '-//Juneau//DTD XXE Test//EN' '" + uri + "'><A>safe</A>";
71  			default -> throw new IllegalArgumentException(vector);
72  		};
73  	}
74  
75  	private static String expansionDocument() {
76  		// 100,000 leaf expansions exceed the JDK 17/21 default count, but stay small if protection regresses.
77  		var dtd = new StringBuilder("<!DOCTYPE A [<!ENTITY e0 'x'>");
78  		for (int i = 1; i <= 5; i++)
79  			dtd.append("<!ENTITY e").append(i).append(" '").append(("&e" + (i - 1) + ";").repeat(10)).append("'>");
80  		return dtd.append("]><A>&e5;</A>").toString();
81  	}
82  
83  	/**
84  	 * Declines every request, leaving the secure fallback responsible for blocking access.
85  	 */
86  	public static class DecliningResolver implements XMLResolver {
87  		@Override public Object resolveEntity(String publicId, String systemId, String baseUri, String namespace) {
88  			return null;
89  		}
90  	}
91  
92  	@ParameterizedTest
93  	@ValueSource(strings = {"general", "parameter", "systemDtd", "publicDtd"})
94  	void externalReferencesNeverFetch(String vector) throws Exception {
95  		for (var parser : new XmlParser[] {XmlParser.DEFAULT, XmlParser.create().resolver(DecliningResolver.class).build()}) {
96  			assertEquals("safe", parser.read("<A>safe</A>", String.class));
97  			try {
98  				var result = parser.read(externalDocument(vector, external), String.class);
99  				assertFalse(result != null && result.contains("FETCHED"));
100 			} catch (ParseException expected) {
101 				// Rejection and empty resolution are both safe, but neither may fetch the resource first.
102 			}
103 			assertEquals(0, requests.get(), vector + " fetched an external resource");
104 		}
105 	}
106 
107 	@Test void fileEntityDoesNotDiscloseContents(@TempDir Path directory) throws Exception {
108 		var secret = directory.resolve("secret.txt");
109 		Files.writeString(secret, "PRIVATE_FILE_MARKER");
110 		assertEquals("safe", XmlParser.DEFAULT.read("<A>safe</A>", String.class));
111 		try {
112 			var result = XmlParser.DEFAULT.read(externalDocument("general", secret.toUri().toString()), String.class);
113 			assertFalse(result != null && result.contains("PRIVATE_FILE_MARKER"));
114 		} catch (ParseException expected) {
115 			for (Throwable cause = expected; cause != null; cause = cause.getCause())
116 				assertFalse(cause.toString().contains("PRIVATE_FILE_MARKER"));
117 		}
118 	}
119 
120 	@Test
121 	@Timeout(10)
122 	void exponentialEntityExpansionIsRejected() {
123 		var doc = expansionDocument();
124 		assertThrows(ParseException.class, () -> XmlParser.DEFAULT.read(doc, String.class));
125 	}
126 }
