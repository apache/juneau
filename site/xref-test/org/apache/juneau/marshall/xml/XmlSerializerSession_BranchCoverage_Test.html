43  	/** Simple bean serializes to XML and round-trips back. */
44  	@Test void a01_simpleBean() {
45  		var bean = new SimpleBean();
46  		bean.name = "Alice";
47  		bean.age = 30;
48  		var xml = SER.write(bean);
49  		assertNotNull(xml);
50  		assertTrue(xml.contains("<name>Alice</name>"), "Expected name element in: " + xml);
51  		assertTrue(xml.contains("<age>30</age>"), "Expected age element in: " + xml);
52  		var parsed = PAR.read(xml, SimpleBean.class);
53  		assertEquals("Alice", parsed.name);
54  		assertEquals(30, parsed.age);
55  	}

57  	/** Bean with null field round-trips correctly — null fields are omitted by default. */
58  	@Test void a02_nullField() {
59  		var bean = new SimpleBean();

61  		bean.age = 5;
62  		var xml = SER.write(bean);
63  		assertNotNull(xml);
64  		var parsed = PAR.read(xml, SimpleBean.class);
65  		assertNull(parsed.name);
66  		assertEquals(5, parsed.age);
67  	}
