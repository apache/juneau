28  /**
29   * Coverage-focused tests for {@link XmlParserSession} targeting low-coverage paths:
30   *  - Builder property() switch (preserveRootElement, validating, alternate keys, fall-through)
31   *  - Malformed XML / unexpected events (END_DOCUMENT, end-of-stream)
32   *  - Generic-map parsing with attributes-only, mixed text+children, nested namespaces
33   *  - readIntoMap with non-string key types and duplicate keys (JsonList accumulation)
34   *  - readIntoCollection with arrays and varargs
35   *  - readAnything error / non-bean fallback / proxy / number / boolean / nil branches
36   *  - XmlFormat variants: ATTR, COLLAPSED, TEXT, XMLTEXT, MIXED, ELEMENTS
37   *  - Comments / processing instructions / CDATA inside elements
38   *  - Accessors: isPreserveRootElement, isValidating, isWhitespaceElement, getEventAllocator,
39   *    getReporter, getResolver, getXmlBeanMeta, getXmlBeanPropertyMeta, getXmlClassMeta
40   */

112 	@Test void c01_unknown_attributesOnlyAndWithText() {
113 		// Element with attrs only → MarshalledMap with attrs only.
114 		var m1 = (Map) P.read("<A b='1' c='2'/>", Object.class);
115 		assertEquals("1", m1.get("b"));
116 		assertEquals("2", m1.get("c"));
117 		// Element with attrs + text → MarshalledMap with "contents" key.
118 		var m2 = (Map) P.read("<A b='1'>hello</A>", Object.class);
119 		assertEquals("1", m2.get("b"));
120 		assertEquals("hello", m2.get("contents"));
121 	}
122 
123 	@Test void c03_unknown_cdataAndPiAndComment() {
124 		// CDATA appends text; PI and comments are skipped (PROCESSING_INSTRUCTION/COMMENT branches).
125 		assertEquals("hello", P.read("<A><![CDATA[hello]]></A>", Object.class));
126 		assertEquals("hello", P.read("<A><?pi data?><!-- comment -->hello</A>", Object.class));
127 	}
128 
129 	@Test void c05_unknown_duplicateKeysAccumulateIntoList() {
130 		// First duplicate → JsonList; third occurrence → MarshalledList.add() branch.
131 		var m = (Map) P.read("<A><x>1</x><x>2</x><x>3</x></A>", Object.class);
132 		var v = m.get("x");
133 		assertTrue(v instanceof List);
134 		assertEquals(3, ((List) v).size());
135 	}
136 
137 	@Test void c07_unknown_namedNestedElements() {
138 		// _name attribute drives the key name in the parent generic map.
139 		// The child's content goes through getUnknown which sees the _name attribute and so
140 		// builds a MarshalledMap; with text "v" it ends up as {contents: "v"}.
141 		var m = (Map) P.read("<A><x _name='custom'>v</x></A>", Object.class);
142 		assertNotNull(m.get("custom"));
143 	}
