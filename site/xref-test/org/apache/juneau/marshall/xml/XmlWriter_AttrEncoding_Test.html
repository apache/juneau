28  /**
29   * Tests that attribute values written through {@link XmlWriter} are entity-encoded by default.
30   *
31   * <p>
32   * An unencoded attribute value containing a quote character terminates the attribute early and lets the remainder of
33   * the value be interpreted as markup, so any caller placing dynamic text into an attribute would otherwise be an
34   * injection sink.  The no-encode overloads remain available for values already known to be safe attribute text.
35   */
36  @SuppressWarnings({
37  	"resource" // SimpleHtmlWriter is closed by try-with-resources; fluent oTag/attr/ceTag return this.
38  })
39  class XmlWriter_AttrEncoding_Test extends TestBase {
40  
41  	@Test void a01_hostileValueCannotBreakOutOfTheAttribute() throws Exception {
42  		try (var w = new SimpleHtmlWriter()) {
43  			w.oTag("div").attr("aria-label", "\"><script>alert(1)</script>").ceTag();
44  			var r = w.toString();
45  			assertEquals("<div aria-label='&quot;&gt;&lt;script&gt;alert(1)&lt;/script&gt;'/>", r);
46  			assertFalse(r.contains("<script>"), r);
47  		}
48  	}
49  
50  	@Test void a02_theXmlSpecialCharactersBecomeEntities() throws Exception {
51  		try (var w = new SimpleHtmlWriter()) {
52  			w.oTag("div").attr("title", "&\"<>'").ceTag();
53  			assertEquals("<div title='&amp;&quot;&lt;&gt;&apos;'/>", w.toString());
54  		}
55  	}
56  
57  	@Test void a03_ordinaryValuesArePassedThroughUnchanged() throws Exception {
58  		try (var w = new SimpleHtmlWriter()) {
59  			w.oTag("div").attr("class", "btn btn-primary").attr("data-id", "c1:more").ceTag();
60  			assertEquals("<div class='btn btn-primary' data-id='c1:more'/>", w.toString());
61  		}
62  	}
63  
64  	@Test void a04_valuesAreTakenAsRawTextSoAPreEscapedValueIsEscapedAgain() throws Exception {
65  		// Documents the caller contract: attr() owns the escaping, so callers must hand it raw text.
66  		try (var w = new SimpleHtmlWriter()) {
67  			w.oTag("div").attr("title", "a &amp; b").ceTag();
68  			assertEquals("<div title='a &amp;amp; b'/>", w.toString());
69  		}
70  	}
71  
72  	@Test void a05_theNoEncodeOverloadStillEmitsVerbatim() throws Exception {
73  		try (var w = new SimpleHtmlWriter()) {
74  			w.oTag("div").attr("data-json", "{\"a\":1}", false).ceTag();
75  			assertEquals("<div data-json='{\"a\":1}'/>", w.toString());
76  		}
77  	}
78  
79  	@Marshalled(typeName = "A")
80  	public static class B01_Bean {
81  		@Xml(format = ATTR)
82  		public String label;
83  	}
84  
85  	@Test void b01_xmlSerializerEncodesAttributeProperties() throws Exception {
86  		var a = new B01_Bean();
87  		a.label = "\"><script>alert(1)</script>";
88  		var r = XmlSerializer.DEFAULT_SQ.write(a);
89  		assertEquals("<A label='&quot;&gt;&lt;script&gt;alert(1)&lt;/script&gt;'/>", r);
90  		assertFalse(r.contains("<script>"), r);
91  	}
92  
93  	@Test void b02_encodedAttributesRoundTripBackToTheOriginalValue() throws Exception {
94  		var a = new B01_Bean();
95  		a.label = "a & b < c > d \" e ' f";
96  		var r = XmlSerializer.DEFAULT_SQ.write(a);
97  		assertEquals(a.label, XmlParser.DEFAULT.read(r, B01_Bean.class).label);
98  	}

100 	@Html(format = XML)
101 	@Marshalled(typeName = "div")
102 	public static class C01_Bean {
103 		@Xml(format = ATTR)
104 		public String label;
105 	}
106 
107 	@Test void c01_htmlSerializerEncodesAttributeProperties() throws Exception {
108 		var a = new C01_Bean();
109 		a.label = "\"><script>alert(1)</script>";
110 		var r = HtmlSerializer.DEFAULT_SQ.write(a);
111 		assertEquals("<div label='&quot;&gt;&lt;script&gt;alert(1)&lt;/script&gt;'></div>", r);
112 		assertFalse(r.contains("<script>"), r);
113 	}
