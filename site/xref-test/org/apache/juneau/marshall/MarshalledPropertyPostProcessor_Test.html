33  /**
34   * Branch-coverage closure tests for {@link MarshalledPropertyPostProcessor} dispatch paths.
35   *
36   * <p>
37   * Targets short-circuit arms in {@code applyPropertyFormats} / {@code applyClassFormats} (a typed
38   * field with {@code @MarshalledProp} / {@code @Marshalled} default = NOT_SET), edge cases in the
39   * XMLGregorianCalendar swap, the {@code temporalAccessorSwap} MILLIS branch, and other
40   * swap-internal arms the cross-serializer/parser matrix doesn't reach because it always supplies a
41   * non-NOT_SET context format and never feeds null / blank values through swap.unswap.
42   *
43   * <p>
44   * Lives in {@code org.apache.juneau.marshall} so it can use package-private MPP internals if needed.
45   */

496 	@SuppressWarnings({
497 		"unchecked"  // Unchecked cast required for generic test utility.
498 	})
499 	private static <T> T roundTripDefault(T bean, String propertyName, Object value) {
500 		// Reflectively set the property so the helper stays generic across the per-type fixture classes.
501 		try {
502 			bean.getClass().getField(propertyName).set(bean, value);
503 			var s = Json5Serializer.DEFAULT.write(bean);
504 			var b = (T) Json5Parser.DEFAULT.read(s, bean.getClass());
505 			assertNotNull(b, "parsed bean was null for " + cns(bean));
506 			return b;
507 		} catch (ReflectiveOperationException e) {
508 			throw new AssertionError(e);
509 		} catch (Exception e) {
510 			throw new AssertionError("round-trip failed for " + cns(bean), e);
511 		}
512 	}
