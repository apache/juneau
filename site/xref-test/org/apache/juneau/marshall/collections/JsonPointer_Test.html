26   * Tests for the RFC 6901 JSON-Pointer implementation {@link JsonPointer}.

28   * <p>
29   * Covers parsing/validation, the RFC 6901 §5 example vectors, {@code ~0}/{@code ~1} token escaping,

42  	@Test void a02_toStringRoundTrip() {
43  		assertEquals("/foo/0", JsonPointer.of("/foo/0").toString());
44  		assertEquals("/a~1b", JsonPointer.of("/a~1b").toString());
45  		assertEquals("/m~0n", JsonPointer.of("/m~0n").toString());
46  		assertEquals("/", JsonPointer.of("/").toString());

52  		assertThrows(IllegalArgumentException.class, () -> JsonPointer.of(null));
53  	}

59  	@Test void b01_decodeToken() {
60  		assertEquals("a/b", JsonPointer.decodeToken("a~1b"));
61  		assertEquals("m~n", JsonPointer.decodeToken("m~0n"));
62  		// ~01 decodes to ~1 (the ~0 -> ~ is applied after ~1 -> /).
63  		assertEquals("~1", JsonPointer.decodeToken("~01"));
64  		assertEquals("plain", JsonPointer.decodeToken("plain"));
65  	}
66  
67  	@Test void b02_encodeToken() {
68  		assertEquals("a~1b", JsonPointer.encodeToken("a/b"));
69  		assertEquals("m~0n", JsonPointer.encodeToken("m~n"));
70  		assertEquals("~01", JsonPointer.encodeToken("~1"));
71  		assertEquals("plain", JsonPointer.encodeToken("plain"));
72  	}
73  
74  	@Test void b03_escapeRoundTrip() {
75  		for (var raw : new String[]{"a/b", "m~n", "~", "/", "a~1b", "c/d~e"})
76  			assertEquals(raw, JsonPointer.decodeToken(JsonPointer.encodeToken(raw)));
77  	}

83  	private static JsonMap rfcDoc() {
84  		return JsonMap.of(
85  			"foo", JsonList.of("bar", "baz"),
86  			"", 0,
87  			"a/b", 1,
88  			"c%d", 2,
89  			"e^f", 3,
90  			"g|h", 4,
91  			"i\\j", 5,
92  			"k\"l", 6,
93  			" ", 7,
94  			"m~n", 8
95  		);
96  	}

164 		assertNull(JsonPointer.of("/5").eval(JsonList.of("x")));

167 	@Test void d04_typeMismatch() {
168 		// Descending into a scalar yields a miss.
169 		assertNull(JsonPointer.of("/a/b").eval(JsonMap.of("a", "scalar")));
170 		// Non-numeric token on an array yields a miss.
171 		assertNull(JsonPointer.of("/x").eval(JsonList.of(1, 2)));

178 	@Test void e01_replaceExisting() {
179 		var root = JsonMap.of("a", 1);
180 		assertSame(root, JsonPointer.of("/a").set(root, 2));
181 		assertEquals(2, JsonPointer.of("/a").eval(root));
182 	}

184 	@Test void e02_vivifyNestedObject() {
185 		var root = new JsonMap();
186 		JsonPointer.of("/a/b/c").set(root, 1);
187 		assertEquals("{\"a\":{\"b\":{\"c\":1}}}", Json.of(root));
188 	}
189 
190 	@Test void e03_vivifyNestedArray() {
191 		var root = new JsonMap();
192 		JsonPointer.of("/a/0").set(root, "x");
193 		assertEquals("{\"a\":[\"x\"]}", Json.of(root));
194 		assertInstanceOf(JsonList.class, root.get("a"));
195 	}
196 
197 	@Test void e04_appendToken() {
198 		var root = JsonMap.of("a", JsonList.of(1, 2));
199 		JsonPointer.of("/a/-").set(root, 3);
200 		assertEquals("{\"a\":[1,2,3]}", Json.of(root));
201 	}
202 
203 	@Test void e05_rootPointerReturnsValue() {
204 		assertEquals("v", JsonPointer.of("").set(JsonMap.of("a", 1), "v"));
205 	}
206 
207 	@Test void e06_setIndexOutOfRangeThrows() {
208 		// Setting at an index beyond the end of an existing list (not the append slot) is an error.
209 		var root = JsonMap.of("a", JsonList.of(1, 2));
210 		var p = JsonPointer.of("/a/5");
211 		assertThrows(IllegalArgumentException.class, () -> p.set(root, 9));
212 	}
213 
214 	@Test void e07_setReplaceAtExistingIndex() {
215 		var root = JsonMap.of("a", JsonList.of(1, 2, 3));
216 		JsonPointer.of("/a/1").set(root, 9);
217 		assertEquals("{\"a\":[1,9,3]}", Json.of(root));
218 	}
219 
220 	@Test void e08_vivifyListViaDashToken() {
221 		// A '-' next token makes the auto-vivified intermediate container a list, and appends to it.
222 		var root = new JsonMap();
223 		JsonPointer.of("/a/-").set(root, "x");
224 		assertEquals("{\"a\":[\"x\"]}", Json.of(root));
225 		assertInstanceOf(JsonList.class, root.get("a"));
226 	}
227 
228 	@Test void e09_descendThroughExistingObject() {
229 		// navigateToParent descends through an already-present intermediate map without re-vivifying it.
230 		var root = JsonMap.of("a", new JsonMap());
231 		JsonPointer.of("/a/b").set(root, 1);
232 		assertEquals("{\"a\":{\"b\":1}}", Json.of(root));
233 	}
234 
235 	@Test void e10_setNonNumericListTokenThrows() {
236 		// A non-numeric, non-'-' token on a list is an invalid array index on write.
237 		var root = JsonMap.of("a", JsonList.of(1, 2));
238 		var p = JsonPointer.of("/a/foo");
239 		assertThrows(IllegalArgumentException.class, () -> p.set(root, 9));
240 	}

247 		var root = JsonMap.of("a", 1, "b", 2);

252 	@Test void f02_removeArrayElement() {
253 		var root = JsonMap.of("a", JsonList.of("x", "y", "z"));
254 		assertEquals("y", JsonPointer.of("/a/1").remove(root));
255 		assertEquals("{\"a\":[\"x\",\"z\"]}", Json.of(root));

258 	@Test void f03_removeAbsentReturnsNull() {
259 		assertNull(JsonPointer.of("/missing").remove(JsonMap.of("a", 1)));

261 	}

263 	@Test void f04_removeRootReturnsNull() {
264 		// The root pointer addresses the whole document, which cannot be removed in place.
265 		assertNull(JsonPointer.of("").remove(JsonMap.of("a", 1)));
266 	}

268 	@Test void f05_removeMissingIntermediateReturnsNull() {

272 	@Test void f06_removeNonNumericListTokenReturnsNull() {

288 	@Test void g01_emptyTokenOnArrayMisses() {
289 		// An empty reference token is not a valid array index, so it resolves to a miss.
290 		assertNull(JsonPointer.of("/").eval(JsonList.of("x")));
291 	}

293 	@Test void g02_leadingZeroIndexRejected() {
294 		// RFC 6901 forbids leading zeros in array indices, so they resolve to a miss.
295 		assertNull(JsonPointer.of("/01").eval(JsonList.of("a", "b")));
296 		assertNull(JsonPointer.of("/00").eval(JsonList.of("a", "b")));
297 	}

299 	@Test void g03_multiDigitIndex() {
300 		var l = new JsonList();
301 		for (var i = 0; i < 12; i++)
302 			l.add(i);
303 		assertEquals(10, JsonPointer.of("/10").eval(l));
304 	}
305 
306 	@Test void g04_overflowingIndexMisses() {
307 		// An all-digit token too large to fit in an int is treated as a non-index (miss), not an error.
308 		assertNull(JsonPointer.of("/99999999999999999999").eval(JsonList.of("x")));
