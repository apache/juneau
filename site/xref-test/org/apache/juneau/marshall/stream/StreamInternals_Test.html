59  		private static String emit(Object value, PojoWalker.Options opts) throws IOException {
60  			var sw = new StringWriter();
61  			try (var w = new JsonTokenWriter(sw)) {
62  				PojoWalker.walk(w, value, opts);
63  			}
64  			return sw.toString();
65  		}

68  			return emit(value, PojoWalker.Options.DEFAULT);

524 				throw new ParseException("boom-read");
525 			}
526 			@Override public boolean isStreaming() { return true; }
527 			@Override public void close() {
528 				// No-op: this stub holds no underlying resource to release.

567 		/** Minimal TokenReader stub that overrides only the abstract methods, leaving the defaults. */
568 		static final class StubTokenReader implements TokenReader {
569 			@Override public TokenType next() { return TokenType.END_OF_STREAM; }
570 			@Override public TokenType getCurrentToken() { return TokenType.NOT_AVAILABLE; }
571 			@Override public int getDepth() { return 0; }
572 			@Override public String getFieldName() { return null; }
573 			@Override public String getString() { return null; }
574 			@Override public Number getNumber() { return null; }
575 			@Override public String getNumberLexeme() { return null; }
576 			@Override public boolean getBool() { return false; }
577 			@Override public byte[] getBinary() { return null; }
578 			@Override public void skipChildren() {
579 				// No-op: the stub never sits on a structural token, so there are no children to skip.
580 			}
581 			@Override public void close() {
582 				// No-op: this stub holds no underlying resource to release.
583 			}
584 		}

586 		/** Minimal TokenWriter stub that overrides only the abstract methods, leaving object()/write() defaults. */

589 			@Override public TokenWriter endObject() { return this; }
590 			@Override public TokenWriter startArray() { return this; }
591 			@Override public TokenWriter endArray() { return this; }
592 			@Override public TokenWriter fieldName(String name) { return this; }
593 			@Override public TokenWriter string(String value) { return this; }
594 			@Override public TokenWriter number(Number value) { return this; }
595 			@Override public TokenWriter number(long value) { return this; }
596 			@Override public TokenWriter number(double value) { return this; }
597 			@Override public TokenWriter number(BigDecimal value) { return this; }
598 			@Override public TokenWriter number(BigInteger value) { return this; }
599 			@Override public TokenWriter bool(boolean value) { return this; }
600 			@Override public TokenWriter nil() { return this; }
601 			@Override public TokenWriter binary(byte[] value) { return this; }
602 			@Override public void flush() {
603 				// No-op: this stub buffers nothing, so there is nothing to flush.

610 		@Test void f01_tokenReaderCanReadDefaultIsFalse() throws Exception {
611 			try (var r = new StubTokenReader()) {
612 				assertFalse(r.canRead());
613 				assertTrue(r.isStreaming());
614 			}
615 		}

617 		@Test void f02_tokenReaderReadDefaultsThrow() {
618 			var cm = MarshallingContext.DEFAULT.getClassMeta(Integer.class);
619 			try (var r = new StubTokenReader()) {
620 				assertThrows(UnsupportedOperationException.class, () -> r.read(Integer.class));
621 				assertThrows(UnsupportedOperationException.class, () -> r.read(cm));
622 				assertThrows(UnsupportedOperationException.class, () -> r.read((Type) Integer.class));
623 			}
624 		}

626 		@Test void f03_tokenWriterObjectDefaultThrows() {
627 			try (var w = new StubTokenWriter()) {
628 				assertThrows(UnsupportedOperationException.class, () -> w.object(42));
629 				// write(Object) default delegates to object(Object).
630 				assertThrows(UnsupportedOperationException.class, () -> w.write(42));
631 				assertTrue(w.isStreaming());
632 			}
