52  /**
53   * Parameterized contract test for the {@link RecordReader} / {@link RecordWriter} surface across
54   * every Juneau format declaring a non-{@code NONE} record-stream capability.
55   *
56   * <p>
57   * Walks a data-shape coverage matrix (scalars, beans with collection / map / array properties,
58   * multi-dimensional collections, top-level Map&lt;String,...&gt; / List&lt;Map&gt;, empty / single-element
59   * / heterogeneous payloads) against every format.  Each cell roundtrips via
60   * {@code recordWriter.write(...) → wire → recordReader.read(...)} and compares semantically.
61   * Format-specific limitations are explicit per-row {@code skips} with documented reasons.
62   *
63   * <p>
64   * FULL-format token surfaces (json, jsonl, json5, cbor, msgpack) are still covered by their
65   * own {@code *TokenStream_Test} files for the structural API; this test exercises only the
66   * record-stream entry points.
67   */
68  @SuppressWarnings({
69  	"resource" // Test fixtures use in-memory streams/writers; closing is the record adapter's responsibility, not the test's.
70  })
71  class RecordStream_Test extends TestBase {

94  		YAML       (YamlParser.DEFAULT,         YamlSerializer.DEFAULT),

103 		UON        (UonParser.DEFAULT,          UonSerializer.DEFAULT,          Skip.LIMITED_BEAN_BINDING),
104 		OAPI       (OpenApiParser.DEFAULT,      OpenApiSerializer.DEFAULT,      Skip.LIMITED_BEAN_BINDING),
105 		URLENC     (UrlEncodingParser.DEFAULT,  UrlEncodingSerializer.DEFAULT,  Skip.URLENC_LIMITATIONS),
106 		XML        (XmlParser.DEFAULT,          XmlSerializer.DEFAULT),
107 		HTML       (HtmlParser.DEFAULT,         HtmlSerializer.DEFAULT),
108 		// Writer-only formats below — read side is null.
109 		JSONSCHEMA (null,                       JsonSchemaSerializer.DEFAULT,   Skip.WRITER_ONLY_OUTPUT_ONLY),
110 		JCS        (null,                       JcsSerializer.DEFAULT,          Skip.WRITER_ONLY_OUTPUT_ONLY),
111 		SOAP       (null,                       org.apache.juneau.marshall.soap.SoapXmlSerializer.create().build(),
112 		            Skip.WRITER_ONLY_OUTPUT_ONLY);

114 		final Parser parser;
115 		final Serializer serializer;
116 		final Set<Skip> skips;
117 		final Mode mode;
118 
119 		Format(Parser p, Serializer s, Object... opts) {
120 			this.parser = p;
121 			this.serializer = s;
122 			var ss = EnumSet.noneOf(Skip.class);
123 			Mode m = Mode.TEXT;
124 			for (var o : opts) {
125 				if (o instanceof Skip k) ss.add(k);
126 				else if (o instanceof Mode mm) m = mm;
127 			}
128 			this.skips = Collections.unmodifiableSet(ss);
129 			this.mode = m;
130 		}
131 
132 		boolean has(Skip s) { return skips.contains(s); }
133 	}

135 	enum Mode { TEXT, BINARY }
136 
137 	enum Skip {
138 		/** CSV is row-based; payloads must be a List of flat beans/maps. */
139 		CSV_FLAT_TABULAR_ONLY,
140 		/** PlainText only round-trips strings via toString(). */
141 		PLAINTEXT_TOSTRING_ONLY,
142 		/** Markdown is fragment-mode (table/list); only specific shapes round-trip cleanly. */
143 		MARKDOWN_FRAGMENT_ONLY,
144 		/** SSE wraps a single SseEvent envelope, not arbitrary POJOs. */
145 		SSE_EVENT_ENVELOPE,
146 		/** Writer-only format (no parser); only the writer side is tested. */
147 		WRITER_ONLY_OUTPUT_ONLY,
148 		/** TOML/HOCON/INI require a non-scalar, non-collection root (must be a bean or Map). */
149 		NESTED_ROOT_REQUIRED,
150 		/** JSONL produces multiple records on the wire; we test it via separate multi-record assertions. */
151 		MULTI_RECORD,
152 		/** Prototext's wire format requires beans with explicit field tags; many shapes don't round-trip. */
153 		PROTO_LIMITATIONS,
154 		/** Parquet's row-group layout doesn't roundtrip arbitrary nested shapes for single-bean payloads. */
155 		PARQUET_LIMITATIONS,
156 		/** UrlEncoding handles only string-keyed top-level structures cleanly. */
157 		URLENC_LIMITATIONS,
158 		/** TOML/HOCON/UON: bean-property generic-type round-trip (Map&lt;String,Bean&gt;) is lossy. */
159 		LIMITED_BEAN_BINDING;
160 	}

216 	/** Builds a {@link Node} chain {@code depth} levels deep, null-terminated. */
217 	private static Node chain(int depth) {
218 		Node n = null;
219 		for (var i = depth; i >= 1; i--)
220 			n = new Node("n" + i, n);
221 		return n;
222 	}

421 	// =====================================================================================
422 	// Self-reference / cycle — 175ab Item 2 (depends on the 175aa PojoWalker cycle guard).
423 	// Per-format cycle handling legitimately differs: most formats (json, json5, cbor, msgpack,
424 	// hjson, uon, oapi) surface a self-reference as a *controlled* failure — a SerializeException
425 	// (the base session converts a StackOverflowError into a recursion message) or the PojoWalker
426 	// IllegalStateException on the token-streaming path — while a few (bson, yaml, xml, html) bound
427 	// the walk and complete without throwing.  The cross-format invariant verified here is that a
428 	// cycle is ALWAYS handled in bounded time and NEVER leaks a raw StackOverflowError to the caller.
429 	// =====================================================================================
430 
431 	@ParameterizedTest
432 	@EnumSource(Format.class)
433 	void g03_selfReferenceCycleIsHandled(Format fmt) throws Exception {
434 		assumeWriteCapable(fmt);
435 		assumeFalse(fmt.has(Skip.SSE_EVENT_ENVELOPE));
436 		assumeFalse(fmt.has(Skip.PLAINTEXT_TOSTRING_ONLY), "PlainText emits toString(), never recurses");
437 		assumeFalse(fmt.has(Skip.MULTI_RECORD));
438 		assumeFalse(fmt.has(Skip.WRITER_ONLY_OUTPUT_ONLY));
439 		assumeFalse(fmt.has(Skip.PROTO_LIMITATIONS));
440 		assumeFalse(fmt.has(Skip.PARQUET_LIMITATIONS));
441 		assumeFalse(fmt.has(Skip.URLENC_LIMITATIONS));
442 		assumeFalse(fmt.has(Skip.LIMITED_BEAN_BINDING));
443 		assumeFalse(fmt.has(Skip.MARKDOWN_FRAGMENT_ONLY));
444 		assumeFalse(fmt.has(Skip.CSV_FLAT_TABULAR_ONLY));
445 		assumeFalse(fmt.has(Skip.NESTED_ROOT_REQUIRED));
446 
447 		var a = new Node("a", null);
448 		a.child = a;  // self-reference
449 
450 		// Either a controlled throw (SerializeException/IOException/IllegalStateException) or a
451 		// bounded, throw-free completion is acceptable; a raw StackOverflowError is NOT.
452 		assertDoesNotThrow(() -> {
453 			try {
454 				serialize(fmt, a);
455 			} catch (@SuppressWarnings("unused") Exception expectedControlledFailure) {
456 				// Controlled failure — acceptable per-format cycle handling.
457 			}
458 		}, "cycle leaked a raw StackOverflowError for " + fmt);
459 	}

583 
584 	private static void assumeWriteCapable(Format fmt) {
585 		assumeTrue(fmt.serializer != null, "format has no serializer: " + fmt);
586 	}
587 
588 	/**
589 	 * For naturally-multi-record formats (CSV, JSONL, SSE), a top-level {@link List} payload is
590 	 * written as N {@code write(...)} calls (one per element).  For single-rooted formats the
591 	 * whole list is written as a single root value.
592 	 */
593 	private static boolean naturallyMultiRecord(Format fmt) {
594 		return fmt == Format.CSV || fmt == Format.JSONL || fmt == Format.SSE;
595 	}
596 
597 	private static String serialize(Format fmt, Object value) throws Exception {
598 		if (fmt.mode == Mode.BINARY) {
599 			var baos = new ByteArrayOutputStream();
600 			writeAll(fmt, ((RecordWritable) fmt.serializer).writeRecords(baos), value);
601 			return Base64.getEncoder().encodeToString(baos.toByteArray());
602 		}
603 		var sb = new StringWriter();
604 		writeAll(fmt, ((RecordWritable) fmt.serializer).writeRecords(sb), value);
605 		return sb.toString();
606 	}
607 
608 	private static byte[] writeBytes(Format fmt, Object value) throws Exception {
609 		var baos = new ByteArrayOutputStream();
610 		writeAll(fmt, ((RecordWritable) fmt.serializer).writeRecords(baos), value);
611 		return baos.toByteArray();
612 	}
613 
614 	private static void writeAll(Format fmt, RecordWriter w, Object value) throws IOException {
615 		try (w) {
616 			if (naturallyMultiRecord(fmt) && value instanceof List<?> list) {
617 				for (var v : list)
618 					w.write(v);
619 			} else {
620 				w.write(value);
621 			}
622 		}
623 	}
624 
625 	private static <T> T parse(Format fmt, String wire, Class<T> type) throws Exception {
626 		Object input = (fmt.mode == Mode.BINARY) ? Base64.getDecoder().decode(wire) : wire;
627 		try (var r = ((RecordReadable) fmt.parser).readRecords(input)) {
628 			return r.read(type);
629 		}
630 	}
631 
632 	private static Object parseWithElement(Format fmt, String wire, Class<?> outer, Class<?>... element) throws Exception {
633 		Object input = (fmt.mode == Mode.BINARY) ? Base64.getDecoder().decode(wire) : wire;
634 		// Use Parser.parse(Object input, Type type, Type... args) for generic type binding.
635 		var args = new java.lang.reflect.Type[element.length];
636 		System.arraycopy(element, 0, args, 0, element.length);
637 		return fmt.parser.read(input, (java.lang.reflect.Type) outer, args);
638 	}
639 
640 	private static <T> T roundTrip(Format fmt, Object value, Class<T> type) throws Exception {
641 		assertNotNull(fmt.parser, "format has no parser: " + fmt);
642 		if (fmt.mode == Mode.BINARY) {
643 			var bytes = writeBytes(fmt, value);
644 			try (var r = ((RecordReadable) fmt.parser).readRecords(bytes)) {
645 				return r.read(type);
646 			}
647 		}
648 		var wire = serialize(fmt, value);
649 		return parse(fmt, wire, type);
650 	}
