35  /**
36   * Parameterized contract test for naturally multi-record formats (JSONL, CSV, SSE).
37   *
38   * <p>
39   * For these formats:
40   * <ul>
41   *   <li>{@code writeRecords(out)} accepts N {@link RecordWriter#write(Object) write(...)} calls
42   *       and emits one record per call.
43   *   <li>{@code readRecords(in)} yields N records via {@link RecordReader#canRead() canRead()} /
44   *       {@link RecordReader#read(Class) read(...)}.
45   *   <li>{@code readArrayRecords/writeArrayRecords} is NOT defined &mdash; that surface only applies to
46   *       formats that need to unwrap a top-level wire array.
47   *   <li>The record-stream output is wire-equivalent to {@code serialize(List.of(records))}.
48   * </ul>
49   */
50  @SuppressWarnings({
51  	"resource" // Test fixtures use in-memory streams/writers; closing is the record adapter's responsibility, not the test's.
52  })
53  class MultiRecordStream_Test extends TestBase {
54  
55  	enum Format {
56  		JSONL  (JsonlParser.DEFAULT, JsonlSerializer.DEFAULT),
57  		CSV    (CsvParser.DEFAULT,   CsvSerializer.DEFAULT),
58  		SSE    (null,                SseSerializer.DEFAULT);  // SSE has no parser
59  
60  		final Parser parser;
61  		final Serializer serializer;
62  
63  		Format(Parser p, Serializer s) {
64  			this.parser = p;
65  			this.serializer = s;
66  		}
67  	}
68  
69  	public static class Bean {
70  		public String name;
71  		public int age;
72  		public Bean() {}
73  		public Bean(String name, int age) { this.name = name; this.age = age; }
74  	}
75  
76  	// =====================================================================================
77  	// Capability declared
78  	// =====================================================================================
79  
80  	@ParameterizedTest
81  	@EnumSource(Format.class)
82  	void a01_capabilityIsNonNone(Format fmt) {
83  		if (fmt.parser != null)
84  			assertInstanceOf(RecordReadable.class, fmt.parser, "reader: " + fmt);
85  		if (fmt.serializer != null)
86  			assertInstanceOf(RecordWritable.class, fmt.serializer, "writer: " + fmt);
87  	}

89  	// =====================================================================================
90  	// Multi-record round-trip
91  	// =====================================================================================
92  
93  	@ParameterizedTest
94  	@EnumSource(Format.class)
95  	void b01_threeBeanRoundTrip(Format fmt) throws Exception {
96  		assumeTrue(fmt.parser != null && fmt.serializer != null, "writer-only format covered separately");
97  		// SSE has no parser path, skip.
98  		assumeTrue(fmt != Format.SSE);
99  
100 		var beans = List.of(new Bean("a", 1), new Bean("b", 2), new Bean("c", 3));
101 		var sb = new StringWriter();
102 		try (var w = ((RecordWritable) fmt.serializer).writeRecords(sb)) {
103 			for (var b : beans)
104 				w.write(b);
105 		}
106 
107 		var got = new ArrayList<Bean>();
108 		try (var r = ((RecordReadable) fmt.parser).readRecords(sb.toString())) {
109 			while (r.canRead())
110 				got.add(r.read(Bean.class));
111 		}
112 
113 		assertEquals(3, got.size(), "format=" + fmt);
114 		assertEquals("a", got.get(0).name);
115 		assertEquals(1, got.get(0).age);
116 		assertEquals("c", got.get(2).name);
117 		assertEquals(3, got.get(2).age);
118 	}

120 	// =====================================================================================
121 	// Wire-equivalence: N writes through recordWriter == serialize(List.of(...))
122 	// =====================================================================================
123 
124 	@ParameterizedTest
125 	@EnumSource(Format.class)
126 	void c01_wireEquivalentToSerializeList(Format fmt) throws Exception {
127 		assumeTrue(fmt.serializer != null);
128 		assumeTrue(fmt != Format.SSE, "SSE event identity test in d01");
129 
130 		var beans = List.of(new Bean("a", 1), new Bean("b", 2));
131 
132 		var streamed = new StringWriter();
133 		try (var w = ((RecordWritable) fmt.serializer).writeRecords(streamed)) {
134 			for (var b : beans)
135 				w.write(b);
136 		}
137 
138 		var bulk = fmt.serializer.writeToString(beans);
139 
140 		assertEquals(bulk, streamed.toString(), "stream-write must match serialize(List) for " + fmt);
141 	}

143 	// =====================================================================================
144 	// SSE writer-only, multi-event
145 	// =====================================================================================
146 
147 	@Test
148 	void d01_sseMultipleEvents() throws Exception {
149 		var sb = new StringWriter();
150 		try (var w = SseSerializer.DEFAULT.writeRecords(sb)) {
151 			w.write(new SseEvent().setEvent("a").setData("1"));
152 			w.write(new SseEvent().setEvent("b").setData("2"));
153 		}
154 		var wire = sb.toString();
155 		assertTrue(wire.contains("event: a"));
156 		assertTrue(wire.contains("data: 1"));
157 		assertTrue(wire.contains("event: b"));
158 		assertTrue(wire.contains("data: 2"));
159 	}
160 
161 	// =====================================================================================
162 	// recordReader.read(T) returns the same as parse(input, T) for the FIRST record
163 	// =====================================================================================
164 
165 	@ParameterizedTest
166 	@EnumSource(Format.class)
167 	void e01_firstReadEqualsParse(Format fmt) throws Exception {
168 		assumeTrue(fmt.parser != null);
169 
170 		var beans = List.of(new Bean("a", 1), new Bean("b", 2));
171 		var sb = new StringWriter();
172 		try (var w = ((RecordWritable) fmt.serializer).writeRecords(sb)) {
173 			for (var b : beans)
174 				w.write(b);
175 		}
176 		var wire = sb.toString();
177 
178 		var viaParse = fmt.parser.read(wire, Bean.class);
179 		Bean viaRecord;
180 		try (var r = ((RecordReadable) fmt.parser).readRecords(wire)) {
181 			assertTrue(r.canRead());
182 			viaRecord = r.read(Bean.class);
183 		}
184 		assertEquals(viaParse.name, viaRecord.name);
185 		assertEquals(viaParse.age, viaRecord.age);
186 	}
