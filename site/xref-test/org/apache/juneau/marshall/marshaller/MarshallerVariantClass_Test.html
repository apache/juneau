49  /**
50   * Drift/presence + round-trip guard for the Feature-B variant marshaller classes.
51   *
52   * <p>
53   * Asserts, for each variant class:
54   * <ul>
55   * 	<li>It is a subclass of its format facade, and its {@link #DEFAULT} instance's serializer/parser
56   * 		identities match the mapped {@code *Serializer.DEFAULT_*}/{@code <Format>Parser.DEFAULT_*} constants.
57   * 	<li>It <b>redeclares the full static shortcut surface</b> of its base format (so none is silently
58   * 		inherited and mis-bound to the base {@code DEFAULT}).
59   * 	<li>Round-trip / wiring: {@code <Variant>.of(x)} equals the variant serializer's own output for a
60   * 		representative bean, {@code <Variant>.to(<Variant>.of(bean), …)} round-trips, and the output is in
61   * 		the expected variant form (readable/multi-line vs. compact/single-line, or otherwise distinct from
62   * 		the base {@code DEFAULT} output).
63   * </ul>
64   */
65  class MarshallerVariantClass_Test extends TestBase {

109 	@MethodSource("variants")
110 	void a01_subclassAndWiringIdentity(Class<? extends Marshaller> variant, Class<? extends Marshaller> base, Serializer s, Parser p) throws Exception {
111 		assertTrue(base.isAssignableFrom(variant), () -> variant.getSimpleName() + " must be a subclass of " + base.getSimpleName());
112 		var def = (Marshaller) variant.getField("DEFAULT").get(null);
113 		assertEquals(variant, def.getClass(), () -> variant.getSimpleName() + ".DEFAULT must be an instance of " + variant.getSimpleName());
114 		assertSame(s, def.getSerializer(), () -> variant.getSimpleName() + ".DEFAULT must be wired to the variant serializer");
115 		assertSame(p, def.getParser(), () -> variant.getSimpleName() + ".DEFAULT must be wired to the variant parser");
116 	}

118 	@ParameterizedTest
119 	@MethodSource("variants")
120 	void a02_fullStaticSurfaceRedeclared(Class<? extends Marshaller> variant, Class<? extends Marshaller> base, @SuppressWarnings("unused") Serializer s, @SuppressWarnings("unused") Parser p) {
121 		for (var m : base.getDeclaredMethods()) {
122 			if (!Modifier.isStatic(m.getModifiers()) || !Modifier.isPublic(m.getModifiers()))
123 				continue;
124 			if (!SHORTCUT_NAMES.contains(m.getName()))
125 				continue;
126 			assertDoesNotThrow(() -> variant.getDeclaredMethod(m.getName(), m.getParameterTypes()),
127 				() -> variant.getSimpleName() + " must redeclare the full static surface of " + base.getSimpleName() + " (missing " + m + ")");
128 		}
129 	}

330 	// NOTE: BsonSerializer/CborSerializer/MsgPackSerializer now honor the configured
331 	// BinaryFormat for byte[] output -- SPACED_HEX/BASE64 switch the byte[] wire representation from
332 	// each format's native binary opcode to that format's native string type containing the
333 	// spaced-hex/base64 text, so the SpacedHex/Base64 variant serializers produce visibly distinct
334 	// output from DEFAULT for a POJO with a byte[] field. These tests verify both the distinct wire
335 	// form and the round-trip back to the original bytes.
336 	//
337 	// The round-trip parse is done with an explicit Map<String,byte[]> type hint rather than the bare
338 	// Map.class used elsewhere in this file: once byte[] is on the wire as a string (SPACED_HEX/BASE64),
339 	// the string is indistinguishable from a genuine text value without a type hint telling the parser
340 	// the target is byte[] -- this is expected, since the encoding trades the format's native
341 	// self-describing binary tag for a text representation that needs the same external typing a text
342 	// serializer (e.g. JSON) would need to recover byte[] from a string.
343
