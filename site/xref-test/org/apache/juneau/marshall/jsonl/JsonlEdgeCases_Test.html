31  /**
32   * Edge case tests for {@link JsonlSerializer} and {@link JsonlParser}.
33   */
34  @SuppressWarnings({
35  	"unchecked" // Parser returns Object; cast to List<JsonMap>/List<Object> in tests
36  })
37  class JsonlEdgeCases_Test extends TestBase {
38  
39  	@Test
40  	void a01_veryLargeCollection() throws Exception {
41  		var a = IntStream.range(0, 10_000)
42  			.mapToObj(i -> JsonMap.of("id", i, "name", "item" + i))
43  			.toList();
44  		var jsonl = Jsonl.of(a);
45  		var lines = jsonl.split("\n");
46  		assertEquals(10_000, lines.length);
47  		var b = (List<JsonMap>) Jsonl.to(jsonl, List.class, JsonMap.class);
48  		assertBean(b, "0{id,name},9999{id,name}", "{0,item0},{9999,item9999}");
49  	}
50  
51  	@Test
52  	void a02_unicodeContent() throws Exception {
53  		var a = list(JsonMap.of("text", "Hello 世界 café \uD83D\uDE00"));
54  		var jsonl = Jsonl.of(a);
55  		var b = (List<JsonMap>) Jsonl.to(jsonl, List.class, JsonMap.class);
56  		assertBean(b, "0{text}", "{Hello 世界 café \uD83D\uDE00}");
57  	}
58  
59  	@Test
60  	void a03_specialCharactersInStrings() throws Exception {
61  		var a = list(JsonMap.of("s", "a\nb\tc\"d\\e"));
62  		var jsonl = Jsonl.of(a);
63  		var b = (List<JsonMap>) Jsonl.to(jsonl, List.class, JsonMap.class);
64  		assertBean(b, "0{s}", "{a\nb\tc\"d\\e}");
65  	}
66  
67  	@Test
68  	void a04_deeplyNestedObjects() throws Exception {
69  		var inner = JsonMap.of("depth", 10);
70  		for (var i = 9; i >= 1; i--) {
71  			inner = JsonMap.of("depth", i, "child", inner);
72  		}
73  		var a = list(inner);
74  		var jsonl = Jsonl.of(a);
75  		var b = (List<JsonMap>) Jsonl.to(jsonl, List.class, JsonMap.class);
76  		assertEquals(1, b.size());
77  		var c = b.get(0);
78  		for (var i = 1; i <= 10; i++) {
79  			assertEquals(i, c.getInt("depth"));
80  			c = c.get("child", JsonMap.class);
81  			if (i < 10)
82  				assertNotNull(c);
83  		}
84  	}
85  
86  	@Test
87  	void a05_emptyObjects() throws Exception {
88  		var a = list(JsonMap.of(), JsonList.of());
89  		var jsonl = Jsonl.of(a);
90  		var b = (List<Object>) Jsonl.to(jsonl, List.class, Object.class);
91  		assertEquals(2, b.size());
92  		assertTrue(b.get(0) instanceof JsonMap);
93  		assertEquals(0, ((JsonMap) b.get(0)).size());
94  		assertTrue(b.get(1) instanceof JsonList);
95  		assertEquals(0, ((JsonList) b.get(1)).size());
96  	}
97  
98  	@Test
99  	void a06_mixedLineEndings() throws Exception {
100 		var jsonl = "{\"a\":1}\r\n{\"b\":2}";
101 		var b = (List<JsonMap>) Jsonl.to(jsonl, List.class, JsonMap.class);
102 		assertBean(b, "0{a},1{b}", "{1},{2}");
103 	}
104 
105 	@Test
106 	void a07_whitespaceOnlyLines() throws Exception {
107 		var jsonl = "{\"a\":1}\n   \n\t\n{\"b\":2}";
108 		var b = (List<JsonMap>) Jsonl.to(jsonl, List.class, JsonMap.class);
109 		assertBean(b, "0{a},1{b}", "{1},{2}");
110 	}
111 
112 	@Test
113 	void a08_booleanAndNullLines() throws Exception {
114 		var jsonl = "true\nfalse\nnull";
115 		var b = (List<Object>) Jsonl.to(jsonl, List.class, Object.class);
116 		assertBean(b, "0,1,2", "true,false,<null>");
117 	}
118 
119 	@Test
120 	void a09_numericLines() throws Exception {
121 		var jsonl = "1\n2.5\n-3\n1e10";
122 		var b = (List<Object>) Jsonl.to(jsonl, List.class, Object.class);
123 		assertBean(b, "0,1,2,3", "1,2.5,-3,1.0E10");
124 	}
