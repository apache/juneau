24  
25  import org.apache.juneau.marshall.collections.*;
26  import org.apache.juneau.marshall.marshaller.*;
27  import org.junit.jupiter.api.*;

29  /**
30   * Tests for JSONL media type configuration.
31   */
32  @SuppressWarnings({
33  	"unchecked" // Parser returns Object; cast to List<JsonMap> in tests
34  })
35  class JsonlMediaType_Test {
36  
37  	@Test
38  	void a01_producesCorrectMediaType() {
39  		var ct = JsonlSerializer.DEFAULT.getResponseContentType();
40  		assertEquals("application", ct.getType());
41  		assertEquals("jsonl", ct.getSubType());
42  	}
43  
44  	@Test
45  	void a02_acceptsAllMediaTypes() {
46  		var types = new ArrayList<String>();
47  		JsonlSerializer.DEFAULT.forEachAcceptMediaType(mt -> types.add(mt.getType() + "/" + mt.getSubType()));
48  		assertTrue(types.stream().anyMatch(t -> "application/jsonl".equals(t) || t.contains("jsonl")), "Expected application/jsonl: " + types);
49  		assertTrue(types.stream().anyMatch(t -> "application/x-ndjson".equals(t) || t.contains("ndjson")), "Expected application/x-ndjson: " + types);
50  		assertTrue(types.stream().anyMatch("text/jsonl"::equals), "Expected text/jsonl: " + types);
51  	}
52  
53  	@Test
54  	void a03_consumesAllMediaTypes() {
55  		var types = JsonlParser.DEFAULT.getMediaTypes().stream()
56  			.map(mt -> mt.getType() + "/" + mt.getSubType())
57  			.toList();
58  		assertTrue(types.stream().anyMatch(t -> t.contains("jsonl")), "Expected jsonl types: " + types);
59  		assertTrue(types.stream().anyMatch(t -> t.contains("ndjson")), "Expected ndjson: " + types);
60  	}
61  
62  	@Test
63  	void a04_contentNegotiation() throws Exception {
64  		var a = list(JsonMap.of("k", "v"));
65  		var jsonl = Jsonl.of(a);
66  		assertNotNull(jsonl);
67  		var b = (List<JsonMap>) Jsonl.to(jsonl, List.class, JsonMap.class);
68  		assertBean(b, "0{k}", "{v}");
69  	}
70  }
