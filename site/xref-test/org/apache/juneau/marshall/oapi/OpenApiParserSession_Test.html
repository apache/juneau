32  /**
33   * Targeted coverage tests for {@link OpenApiParserSession} focusing on gaps not exercised by the
34   * broader round-trip suite in {@link OpenApi_Test} -- Optional unwrap/rewrap, non-default
35   * collection formats for both ARRAY and OBJECT, element/value-type fallback chains, bean/map
36   * malformed-input error paths, and toType()'s conversion-failure wrapping.
37   */
38  @SuppressWarnings({
39  	"unchecked", // Parser returns Object by default for raw target types in these tests

44  	private static final OpenApiParser DP = OpenApiParser.DEFAULT;
45  
46  	private static <T> T p(HttpPartSchema schema, String in, Class<T> c, Class<?>...args) throws Exception {
47  		return DP.read(null, schema, in, DP.getClassMeta(c, args));
48  	}

54  	@Test
55  	void a01_optionalTypeUnwrapsAndRewraps() throws Exception {
56  		// isOptional == true: the while-loop unwraps to the element type for parsing, then the
57  		// result is re-wrapped via o(t) before returning.
58  		var r = (Optional<String>) p(T_STRING, "hi", Optional.class, String.class);
59  		assertTrue(r.isPresent());
60  		assertEquals("hi", r.get());
61  	}
62  
63  	@Test
64  	void a02_doublyNestedOptionalUnwrapsBothLevels() throws Exception {
65  		// The while-loop at the top of read() unwraps EACH level of nested Optional down to the
66  		// innermost non-optional element type (only the outermost level is re-wrapped afterward).
67  		var r = (Optional<?>) p(T_STRING, "hi", Optional.class, Optional.class, String.class);
68  		assertTrue(r.isPresent());
69  		assertEquals("hi", r.get());
70  	}

76  	@Test
77  	void b01_dateFormatBlankValueIntoObjectFiltersToNull() {
78  		// sType.isObject() && f == DATE branch: isBlank(x1) == true short-circuits the
79  		// GranularZonedDateTime parse and orElse(null) fires -- schema.validateOutput() then
80  		// rejects the null result for this required-format schema, but the DATE branch itself
81  		// (and its empty-filter arm) is still exercised before that outer rejection.
82  		var ps = tNone().fDate().allowEmptyValue().build();
83  		assertThrowsWithMessage(SchemaValidationException.class, "does not match expected format",
84  			() -> DP.read(null, ps, "", Object.class));
85  	}

285 		// NO_COLLECTION_FORMAT's "starts with '@', ends with ')'" check: an input that merely
