31  class Json5l_Test extends TestBase {
32  
33  	@BeanType(properties = "name,age")
34  	public static class Person {
35  		public String name;
36  		public int age;
37  
38  		public Person() {}
39  		public Person(String name, int age) {
40  			this.name = name;
41  			this.age = age;
42  		}
43  	}
44  
45  	// =================================================================================
46  	// A. Serialization — strict-per-line default (mirrors JSONL semantics)
47  	// =================================================================================
48  
49  	@Test
50  	void a01_writeCollectionOfBeans() throws Exception {
51  		var list = list(
52  			new Person("Alice", 30),
53  			new Person("Bob", 25),
54  			new Person("Carol", 35)
55  		);
56  		var json5l = Json5l.of(list);
57  		var lines = json5l.split("\n");
58  		assertEquals(3, lines.length);
59  		assertTrue(lines[0].contains("\"name\":\"Alice\""));
60  		assertTrue(lines[0].contains("\"age\":30"));
61  		assertTrue(lines[1].contains("\"name\":\"Bob\""));
62  		assertTrue(lines[2].contains("\"name\":\"Carol\""));
63  		assertFalse(json5l.contains("["));
64  		assertFalse(json5l.contains("]"));
65  	}
66  
67  	@Test
68  	void a02_writeArray() throws Exception {
69  		var arr = new Person[]{new Person("Alice", 30), new Person("Bob", 25)};
70  		var json5l = Json5l.of(arr);
71  		var lines = json5l.split("\n");
72  		assertEquals(2, lines.length);
73  		assertTrue(lines[0].contains("\"Alice\""));
74  		assertTrue(lines[1].contains("\"Bob\""));
75  	}
76  
77  	@Test
78  	void a03_writeSingleBean() throws Exception {
79  		var p = new Person("Alice", 30);
80  		var json5l = Json5l.of(p);
81  		assertEquals(1, json5l.split("\n").length);
82  		assertTrue(json5l.contains("\"name\":\"Alice\""));
83  		assertTrue(json5l.contains("\"age\":30"));
84  	}
85  
86  	@Test
87  	void a04_writeCollectionOfStrings() throws Exception {
88  		var list = list("foo", "bar", "baz");
89  		var json5l = Json5l.of(list);
90  		var lines = json5l.split("\n");
91  		assertEquals(3, lines.length);
92  		assertTrue(lines[0].contains("\"foo\""));
93  		assertTrue(lines[1].contains("\"bar\""));
94  		assertTrue(lines[2].contains("\"baz\""));
95  	}

97  	@Test
98  	void a05_writeEmptyCollection() throws Exception {
99  		var json5l = Json5l.of(list());
100 		assertEquals("", json5l.trim());
101 	}
102 
103 	@Test
104 	void a06_writeNullValues() throws Exception {
105 		var list = list("a", null, "c");
106 		var json5l = Json5l.of(list);
107 		var lines = json5l.split("\n");
108 		assertEquals(3, lines.length);
109 		assertEquals("null", lines[1]);
110 	}

112 	@Test
113 	void a07_writeNestedObjects() throws Exception {
114 		var outer = JsonMap.of("name", "Alice", "inner", JsonMap.of("x", 1, "y", 2));
115 		var json5l = Json5l.of(outer);
116 		assertTrue(json5l.contains("\"inner\":{\"x\":1,\"y\":2}"));
117 	}

124 	void b01_defaultIsByteIdenticalToJsonl() throws Exception {
125 		var list = list(
126 			new Person("Alice", 30),

128 		);
129 		assertEquals(Jsonl.of(list), Json5l.of(list));

132 	@Test
133 	void b02_defaultProducesStrictDoubleQuotes() throws Exception {
134 		var json5l = Json5l.of(JsonMap.of("name", "Alice"));
135 		assertTrue(json5l.contains("\"name\":\"Alice\""));
136 		assertFalse(json5l.contains("'"));
137 	}

139 	// =================================================================================
140 	// C. json5Sugar opt-in
141 	// =================================================================================
142 
143 	@Test
144 	void c01_sugarUsesSingleQuotesAndUnquotedKeys() throws Exception {
145 		var s = Json5lSerializer.create().json5Sugar().build();
146 		var out = s.write(JsonMap.of("name", "Alice", "age", 30));
147 		assertTrue(out.contains("name:'Alice'"), "Expected unquoted key + single quotes: " + out);
148 		assertTrue(out.contains("age:30"), "Expected unquoted key: " + out);
149 		assertFalse(out.contains("\""), "Expected no double quotes: " + out);
150 	}
151 
152 	@Test
153 	void c02_sugarStillOneLinePerRecord() throws Exception {
154 		var s = Json5lSerializer.create().json5Sugar().build();
155 		var out = s.write(list(new Person("Alice", 30), new Person("Bob", 25)));
156 		var lines = out.split("\n");
157 		assertEquals(2, lines.length);
158 		assertTrue(lines[0].contains("name:'Alice'"));
159 		assertTrue(lines[1].contains("name:'Bob'"));
160 	}

162 	@Test
163 	void c03_sugarFlagIsReflectedOnContext() {
164 		assertFalse(Json5lSerializer.DEFAULT.isJson5Sugar());
165 		assertTrue(Json5lSerializer.create().json5Sugar().build().isJson5Sugar());
166 	}
167 
168 	@Test
169 	void c04_sugarAndStrictAreDistinctCachedInstances() {
170 		var strict = Json5lSerializer.create().build();
171 		var sugar = Json5lSerializer.create().json5Sugar().build();
172 		assertNotSame(strict, sugar);
173 		assertFalse(strict.isJson5Sugar());
174 		assertTrue(sugar.isJson5Sugar());
175 	}
176 
177 	@Test
178 	void c05_copyPreservesSugar() {
179 		var sugar = Json5lSerializer.create().json5Sugar().build();
180 		assertTrue(sugar.copy().build().isJson5Sugar());
181 	}
