231 	@Test
232 	void e01_byteArrayInListDefaultFormatUsesUtf8Fallback() throws Exception {
233 		// List<byte[]> at default BinaryFormat (NOT_SET): type.getSwap(this) returns null (no swap
234 		// configured for NOT_SET), so coerceMemberValue falls through to convertToMemberType's plain
235 		// BasicConverter default String->byte[] coercion, which is raw UTF-8 bytes -- NOT Base64 decode.
236 		// (This is a real behavioral gap called out in coerceMemberValue's own javadoc: only an
237 		// EXPLICITLY configured BinaryFormat installs a swap; the implicit default does not.)
238 		var hjson = "{data: [\"abc\"]}";
239 		var bean = HjsonParser.DEFAULT.read(hjson, BeanWithByteList.class);
240 		assertNotNull(bean);
241 		assertEquals(1, bean.data.size());
242 		assertArrayEquals("abc".getBytes(java.nio.charset.StandardCharsets.UTF_8), bean.data.get(0));
243 	}

264 	@Test
265 	void e05_nullKeyInBeanObjectFallsBackToObjectType() {
266 		// propertyType's "key != null" arm: a NULL-keyword key inside a bean-typed object -- the
267 		// beanMeta != null guard is true, but key == null short-circuits to object() rather than a
268 		// property lookup (e01-e04 above only exercise non-null keys against a non-null beanMeta).
269 		// propertyType() itself resolves cleanly (object() fallback); the resulting MarshalledMap
270 		// still fails to load into a BeanMap because of the null key downstream, which is an
271 		// orthogonal (and pre-existing) BeanMap.load() limitation, not something propertyType causes.
272 		assertThrows(Exception.class, () -> HjsonParser.DEFAULT.read("{null: 1, name: hi}", BeanWithBeanProp.class));
273 	}
