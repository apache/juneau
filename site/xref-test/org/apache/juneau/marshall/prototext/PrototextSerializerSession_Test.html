1018 		// falls through to writeAnything, which re-derives the same (still-empty) BeanMap and writes it
1019 		// via the treatAsBean path instead.
1020 		var s = PrototextSerializer.create().beanPropertiesExcludes(R02_NoVisiblePropsBean.class, "value").build();
1021 		var proto = s.write(new R02_NoVisiblePropsBean());
1022 		assertNotNull(proto);
