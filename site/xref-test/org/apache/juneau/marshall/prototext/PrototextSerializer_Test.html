31  	@Test
32  	void a01_simpleBean() {
33  		var m = new LinkedHashMap<String,Object>();
34  		m.put("host", "localhost");
35  		m.put("port", 8080);
36  		m.put("debug", true);
37  		m.put("ratio", 3.14);
38  
39  		String proto = PrototextSerializer.DEFAULT.write(m);
40  		assertNotNull(proto);
41  		assertTrue(proto.contains("host:"));
42  		assertTrue(proto.contains("localhost"));
43  		assertTrue(proto.contains("port:"));
44  		assertTrue(proto.contains("8080"));
45  		assertTrue(proto.contains("debug:"));
46  		assertTrue(proto.contains("true"));
47  		assertTrue(proto.contains("ratio:"));
48  		assertTrue(proto.contains("3.14"));
49  	}

52  	void a02_nestedBean() {
53  		var db = new LinkedHashMap<String,Object>();
54  		db.put("host", "localhost");
55  		db.put("port", 5432);
56  		var config = new LinkedHashMap<String,Object>();
57  		config.put("name", "myapp");
58  		config.put("database", db);
59  
60  		String proto = PrototextSerializer.DEFAULT.write(config);
61  		assertNotNull(proto);

65  		assertTrue(proto.contains("5432"));
