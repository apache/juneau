13  package org.apache.juneau.marshall.prototext;

15  import static org.apache.juneau.test.bct.BctAssertions.*;
16  import static org.junit.jupiter.api.Assertions.*;

18  import java.util.*;
19  
20  import org.apache.juneau.marshall.collections.*;

23  /**
24   * Round-trip tests for repeated fields (lists, arrays, nested messages) in compact and whitespace modes.
25   */
26  class PrototextRepeatedFields_Test {
27  
28  	private static final PrototextSerializer COMPACT = PrototextSerializer.DEFAULT;
29  	private static final PrototextSerializer WS = PrototextSerializer.create().ws().build();
30  	private static final PrototextParser P = PrototextParser.DEFAULT;
31  
32  	public static class Inner {
33  		public String name;
34  		public int n;
35  		public Inner() {}
36  		public Inner(String name, int n) { this.name = name; this.n = n; }
37  	}
38  
39  	public static class Scalars { public List<Integer> nums; }
40  	public static class Beans { public List<Inner> ins; }
41  	public static class BeanArray { public Inner[] ins; }
42  	public static class Maps { public Map<String,Integer> m; }
43  	public static class MapList { public List<Map<String,Object>> ms; }
44  	public static class Nested { public List<List<Integer>> grid; }
45  	public static class Empty { public List<Integer> a; public List<Inner> b; public String s; }
46  	public static class Mid { public String title; public List<Beans> parts; }
47  	public static class Untyped { public Object items; public Map<String,List<Inner>> byKey; public Inner[][] grid; }
48  	public static class Mixed { public Beans first; public List<Integer> nums; public List<Inner> ins; public String last; }
49  
50  	private static void forEachMode(java.util.function.Consumer<PrototextSerializer> c) {
51  		c.accept(COMPACT);
52  		c.accept(WS);
53  	}
54  
55  	@Test void a01_scalarList() {
56  		forEachMode(s -> {
57  			try {
58  				var b = new Scalars();
59  				b.nums = List.of(1, 2, 3);
60  				var r = P.read(s.write(b), Scalars.class);
61  				assertBean(r, "nums", "[1,2,3]");
62  			} catch (Exception e) { throw new AssertionError(e); }
63  		});
64  	}
65  
66  	@Test void a02_beanList() {
67  		forEachMode(s -> {
68  			try {
69  				var b = new Beans();
70  				b.ins = List.of(new Inner("a", 1), new Inner("b", 2));
71  				var r = P.read(s.write(b), Beans.class);
72  				assertSize(2, r.ins);
73  				assertBean(r.ins.get(0), "name,n", "a,1");
74  				assertBean(r.ins.get(1), "name,n", "b,2");
75  			} catch (Exception e) { throw new AssertionError(e); }
76  		});
77  	}
78  
79  	@Test void a03_beanListSingleElement() {
80  		forEachMode(s -> {
81  			try {
82  				var b = new Beans();

84  				var r = P.read(s.write(b), Beans.class);
85  				assertSize(1, r.ins);
86  				assertBean(r.ins.get(0), "name,n", "a,1");
87  			} catch (Exception e) { throw new AssertionError(e); }
88  		});
89  	}

91  	@Test void a04_beanArray() {
92  		forEachMode(s -> {
93  			try {
94  				var b = new BeanArray();
95  				b.ins = new Inner[]{new Inner("a", 1), new Inner("b", 2)};
96  				var r = P.read(s.write(b), BeanArray.class);
97  				assertEquals(2, r.ins.length);
98  				assertBean(r.ins[1], "name,n", "b,2");
99  			} catch (Exception e) { throw new AssertionError(e); }
100 		});
101 	}

103 	@Test void a05_map() {
104 		forEachMode(s -> {
105 			try {
106 				var b = new Maps();
107 				b.m = new LinkedHashMap<>();
108 				b.m.put("x", 1);
109 				b.m.put("y", 2);
110 				var r = P.read(s.write(b), Maps.class);
111 				assertEquals(Map.of("x", 1, "y", 2), r.m);
112 			} catch (Exception e) { throw new AssertionError(e); }
113 		});
114 	}

116 	@Test void a06_mapList() {
117 		forEachMode(s -> {
118 			try {
119 				var b = new MapList();
120 				b.ms = List.of(JsonMap.of("k", "v"), JsonMap.of("k", "w"));
121 				var r = P.read(s.write(b), MapList.class);
122 				assertSize(2, r.ms);
123 				assertEquals("v", r.ms.get(0).get("k"));
124 				assertEquals("w", r.ms.get(1).get("k"));
125 			} catch (Exception e) { throw new AssertionError(e); }
126 		});
127 	}

129 	@Test void a07_nestedLists() {
130 		forEachMode(s -> {
131 			try {
132 				var b = new Nested();
133 				b.grid = List.of(List.of(1), List.of(2, 3), List.of());
134 				var proto = s.write(b);
135 				assertFalse(proto.contains("\"["), proto);
136 				var r = P.read(proto, Nested.class);
137 				assertBean(r, "grid", "[[1],[2,3],[]]");
138 			} catch (Exception e) { throw new AssertionError(e); }
139 		});
140 	}

145 				var b = new Empty();

147 				b.b = List.of();
148 				b.s = "x";
149 				var r = P.read(s.write(b), Empty.class);
150 				assertEquals("x", r.s);
151 			} catch (Exception e) { throw new AssertionError(e); }
152 		});
153 	}

155 	@Test void a09_nestedMessagesInsideRepeatedMessages() {
156 		forEachMode(s -> {
157 			try {
158 				var b = new Mid();
159 				b.title = "t";
160 				var b1 = new Beans();
161 				b1.ins = List.of(new Inner("a", 1), new Inner("b", 2));
162 				var b2 = new Beans();
163 				b2.ins = List.of(new Inner("c", 3));
164 				b.parts = List.of(b1, b2);
165 				var r = P.read(s.write(b), Mid.class);
166 				assertEquals("t", r.title);
167 				assertSize(2, r.parts);
168 				assertSize(2, r.parts.get(0).ins);
169 				assertBean(r.parts.get(0).ins.get(1), "name,n", "b,2");
170 				assertBean(r.parts.get(1).ins.get(0), "name,n", "c,3");
171 			} catch (Exception e) { throw new AssertionError(e); }
172 		});
173 	}

175 	@Test void a10_mixedFieldOrdering() {
176 		forEachMode(s -> {
177 			try {
178 				var b = new Mixed();
179 				b.first = new Beans();
180 				b.first.ins = List.of(new Inner("z", 9));
181 				b.nums = List.of(4, 5);
182 				b.ins = List.of(new Inner("a", 1), new Inner("b", 2));
183 				b.last = "end";
184 				var r = P.read(s.write(b), Mixed.class);
185 				assertBean(r.first.ins.get(0), "name,n", "z,9");
186 				assertBean(r, "nums,last", "[4,5],end");
187 				assertSize(2, r.ins);
188 			} catch (Exception e) { throw new AssertionError(e); }
189 		});
190 	}
191 
192 	@Test void a11_erasedAndMapValueLists() {
193 		forEachMode(s -> {
194 			try {
195 				var b = new Untyped();
196 				b.items = List.of(new Inner("a", 1), new Inner("b", 2));
197 				b.byKey = new LinkedHashMap<>();
198 				b.byKey.put("k", List.of(new Inner("c", 3), new Inner("d", 4)));
199 				b.byKey.put("j", List.of(new Inner("e", 5)));
200 				var proto = s.write(b);
201 				var r = P.read(proto, Untyped.class);
202 				assertSize(2, (List<?>) r.items);
203 				assertSize(2, r.byKey.get("k"));
204 				assertBean(r.byKey.get("j").get(0), "name,n", "e,5");
205 			} catch (Exception e) { throw new AssertionError(e); }
206 		});
207 	}
208 
209 	@Test void b01_compactHasSeparatorAfterClosingBrace() throws Exception {
210 		var b = new Mixed();
211 		b.first = new Beans();
212 		b.first.ins = List.of(new Inner("z", 9));
213 		b.nums = List.of(4, 5);
214 		var proto = COMPACT.write(b);
215 		assertFalse(proto.matches("(?s).*\\}[A-Za-z\"].*"), proto);
216 	}
217 
218 	@Test void b02_exactOutput() throws Exception {
219 		var b = new Mixed();
220 		b.ins = List.of(new Inner("a", 1), new Inner("b", 2));
221 		b.nums = List.of(4);
222 		assertEquals("ins {n: 1\nname: \"a\"\n}\nins {n: 2\nname: \"b\"\n}\nnums: [4]\n", COMPACT.write(b));
223 	}
224 
225 	@Test void b03_nestedListOutput() throws Exception {
226 		var b = new Nested();
227 		b.grid = List.of(List.of(1), List.of(2, 3), List.of());
228 		assertEquals("grid {_value: [1]\n}\ngrid {_value: [2, 3]\n}\ngrid {_value: []\n}\n", COMPACT.write(b));
229 	}
