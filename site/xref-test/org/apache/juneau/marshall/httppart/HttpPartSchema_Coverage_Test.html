147 	// Fake annotations below share simple names with real Jakarta constraints so applyJakartaValidation's
148 	// switch dispatches into the same case arms, but declare a "value"/"regexp" attribute of the WRONG type
149 	// (String instead of long, etc). getAnnotationValue()'s type.isInstance() check then fails, returning null
150 	// -- exercising both its own type-mismatch branch and the caller's nn(...) guard, which real Jakarta
151 	// annotations (whose attributes are always the expected type) can never trigger.
152 	@Retention(RetentionPolicy.RUNTIME)
153 	private @interface Min {
154 		String value() default "not-a-long";
155 	}
156 
157 	@Retention(RetentionPolicy.RUNTIME)
158 	private @interface Max {
159 		String value() default "not-a-long";
160 	}
161 
162 	@Retention(RetentionPolicy.RUNTIME)
163 	private @interface Pattern {
164 		int regexp() default 0;
165 	}
166 
167 	@Retention(RetentionPolicy.RUNTIME)
168 	private @interface DecimalMin {
169 		int value() default 0;
170 	}
171 
172 	@Retention(RetentionPolicy.RUNTIME)
173 	private @interface DecimalMax {
174 		int value() default 0;
175 	}

638 	// H: applyHttpAnnotation switch dispatch + getAnnotationString/getAnnotationIntArray reflection guards
639 	// These fake annotations share simple names with the real HTTP-part annotations (Path/Response/StatusCode/...,
640 	// defined in juneau-rest-common, not a juneau-marshall dependency) purely so applyHttpAnnotation's switch on
641 	// annotationType().getSimpleName() dispatches into the same case arms.
642 	//-----------------------------------------------------------------------------------------------------------------
643 
644 	@Retention(RetentionPolicy.RUNTIME)
645 	private @interface PathRemainder {
646 		String def() default "";
647 	}
648 
649 	@Retention(RetentionPolicy.RUNTIME)
650 	private @interface Path {
651 		String name() default "";
652 		String value() default "";
653 		String def() default "";
654 	}
655 
656 	// A separate holder with its own nested "Path" annotation (shadows the outer one within this scope) that
657 	// declares no def() method at all -- triggers getAnnotationString's NoSuchMethodException guard, so def
658 	// stays null, while annotationType().getSimpleName() is still "Path" so applyHttpAnnotation's switch still
659 	// dispatches to the same case arm.
660 	static class H05Holder {
661 		@Retention(RetentionPolicy.RUNTIME)
662 		@interface Path {
663 			String name() default "";
664 		}
665 		@Path(name="userId2")
666 		public String pathNoDefAttr;
667 	}
668 
669 	@Retention(RetentionPolicy.RUNTIME)
670 	private @interface Response {}
671 
672 	@Retention(RetentionPolicy.RUNTIME)
673 	private @interface StatusCode {
674 		int[] value() default {};
675 	}
676 
677 	// A "StatusCode" annotation whose "value" attribute is the wrong return type -- triggers
678 	// getAnnotationIntArray's return-type-mismatch guard (falls through to the empty-array default).
679 	@Retention(RetentionPolicy.RUNTIME)
680 	private @interface StatusCodeWrongType {
681 		String value() default "not-an-int-array";
682 	}
683 
684 	// A "StatusCode" annotation missing the "value" attribute entirely -- triggers getAnnotationIntArray's
685 	// NoSuchMethodException guard.
686 	@Retention(RetentionPolicy.RUNTIME)
687 	private @interface StatusCodeNoValue {}
688 
689 	@Retention(RetentionPolicy.RUNTIME)
690 	private @interface UnrecognizedHttpPartAnnotation {
691 		String name() default "";
692 		String value() default "";
693 		String def() default "";
694 	}
695 
696 	static class H01Bean {
697 		@PathRemainder(def="fallback")
698 		public String pathRemainder;
699 		@Path(name="userId")
700 		public String pathNoLeadingSlash;
701 		@Path(value="/foo/*")
702 		public String pathLeadingSlash;
703 		@Path(name="userId", def="_NONE_")
704 		public String pathDefNone;
705 		@Response
706 		public String response;
707 		@StatusCode(value={200,201})
708 		public String statusCode;
709 		@StatusCodeWrongType
710 		public String statusCodeWrongType;
711 		@StatusCodeNoValue
712 		public String statusCodeNoValue;
713 		@UnrecognizedHttpPartAnnotation(name="foo", def="bar")
714 		public String unrecognized;
715 	}
