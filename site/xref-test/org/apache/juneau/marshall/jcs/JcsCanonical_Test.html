31  	@Test
32  	void d01_rfcExample1() throws Exception {
33  		// RFC 8785 Section 3.2.2: numbers, string, literals
34  		var numbers = list(333333333.33333329, 1E30, 4.50, 2e-3, 0.000000000000000000000000001);
35  		var s = Character.toString((char) 0x20AC) + "$" + Character.toString((char) 0x000F) + "\nA'B\"\\\\\"/";
36  		var literals = list((Object) null, true, false);
37  		var m = JsonMap.of("numbers", numbers, "string", s, "literals", literals);
38  		var out = JcsSerializer.DEFAULT.write(m);
39  		// Keys sorted: literals, numbers, string
40  		assertTrue(out.startsWith("{\"literals\":"));
41  		assertTrue(out.contains("\"numbers\":"));
42  		assertTrue(out.contains("\"string\":"));
43  		assertFalse(out.contains(" "));
44  	}
45  
46  	@Test
47  	void d02_rfcExample2Sorted() throws Exception {
48  		// Sorted version per RFC 3.2.3
49  		var numbers = list(333333333.33333329, 1E30, 4.50, 2e-3, 0.000000000000000000000000001);
50  		var literals = list((Object) null, true, false);
51  		var m = JsonMap.of("literals", literals, "numbers", numbers);
52  		var out = JcsSerializer.DEFAULT.write(m);
53  		assertTrue(out.startsWith("{\"literals\":[null,true,false],\"numbers\":"));
54  	}

56  	@Test
57  	void d03_noWhitespace() throws Exception {
58  		var m = JsonMap.of("a", 1, "b", 2);
59  		var s = JcsSerializer.DEFAULT.write(m);
60  		assertFalse(s.contains(" "));
61  		assertFalse(s.contains("\n"));
62  		assertFalse(s.contains("\t"));
63  	}

65  	@Test
66  	void d04_simpleBeanCanonical() throws Exception {
67  		var m = JsonMap.of("name", "Alice", "age", 30);
68  		var s = JcsSerializer.DEFAULT.write(m);
69  		assertEquals("{\"age\":30,\"name\":\"Alice\"}", s);
70  	}
71  
72  	@Test
73  	void d05_nestedBeanCanonical() throws Exception {
74  		var inner = JsonMap.of("zip", "80201", "city", "Denver");
75  		var outer = JsonMap.of("address", inner, "name", "Alice");
76  		var s = JcsSerializer.DEFAULT.write(outer);
77  		assertTrue(s.contains("\"address\":{\"city\":\"Denver\",\"zip\":\"80201\"}"));
78  		assertTrue(s.contains("\"name\":\"Alice\""));
79  	}
80  
81  	@Test
82  	void d06_mixedTypesCanonical() throws Exception {
83  		var m = JsonMap.of("n", 42, "s", "hi", "b", true, "x", (Object) null, "a", list(1, 2));
84  		var s = JcsSerializer.DEFAULT.write(m);
85  		assertFalse(s.contains(" "));
86  		// All keys present, sorted
87  		assertTrue(s.contains("\"a\":"));
88  		assertTrue(s.contains("\"b\":true"));
89  		assertTrue(s.contains("\"n\":42"));
90  		assertTrue(s.contains("\"s\":\"hi\""));
91  		assertTrue(s.contains("\"x\":null"));
92  	}

102 	@Test
103 	void d08_hashStability() throws Exception {
104 		var m = JsonMap.of("c", 3, "a", 1, "b", 2);
105 		var s1 = JcsSerializer.DEFAULT.write(m);
106 		var s2 = JcsSerializer.DEFAULT.write(m);
107 		var md = MessageDigest.getInstance("SHA-256");
108 		var bytes1 = s1.getBytes(StandardCharsets.UTF_8);
109 		var bytes2 = s2.getBytes(StandardCharsets.UTF_8);
110 		assertArrayEquals(bytes1, bytes2);
111 		assertEquals(bytesToHex(md.digest(bytes1)), bytesToHex(md.digest(bytes2)));
112 	}

120 	void d10_alphabeticalOrder() throws Exception {
121 		var m = JsonMap.of("c", 3, "a", 1, "b", 2);
122 		assertEquals("{\"a\":1,\"b\":2,\"c\":3}", JcsSerializer.DEFAULT.write(m));
123 	}

125 	private static String bytesToHex(byte[] bytes) {
126 		var sb = new StringBuilder();
127 		for (var b : bytes)
128 			sb.append(String.format("%02x", b));
129 		return sb.toString();
130 	}
