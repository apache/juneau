263 	// Regression for the tokenizer peek/skip invariant.
264 	// Newline-separated inner arrays inside an outer array must stay NESTED — the newline is an
265 	// element separator, NOT a value-concatenation join.  If skipWhitespaceAndComments wrongly read
266 	// past the closing ']' cached by readValueOrConcat's peekNoSkip and ate the following newline,
267 	// the two inner arrays would silently flatten via HOCON array-concatenation.
268 	@Test
269 	void b27_nestedArraysNewlineSeparatedStayNested() {
270 		var hocon = "outer = [\n  [1, 2]\n  [3, 4]\n]";
271 		var m = (Map<String,Object>) HoconParser.DEFAULT.read(hocon, Map.class, String.class, Object.class);
272 		var outer = (List<?>) m.get("outer");
273 		assertNotNull(outer);
274 		assertEquals(2, outer.size());
275 		var first = (List<?>) outer.get(0);
276 		var second = (List<?>) outer.get(1);
277 		assertEquals(2, first.size());
278 		assertEquals(2, second.size());
279 		assertEquals(1, ((Number) first.get(0)).intValue());
280 		assertEquals(2, ((Number) first.get(1)).intValue());
281 		assertEquals(3, ((Number) second.get(0)).intValue());
282 		assertEquals(4, ((Number) second.get(1)).intValue());
283 	}
