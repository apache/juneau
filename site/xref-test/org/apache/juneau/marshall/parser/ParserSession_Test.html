34  /**
35   * Coverage-focused tests for {@link ParserSession} targeting low-coverage paths:
36   *  - readIntoMap / readIntoCollection overload variants and error wrapping
37   *  - Parametric read(...) type paths (Type, Class<T>, ClassMeta<T>)
38   *  - readArgs (Method invocation array parsing) success and error paths
39   *  - readToBeanConsumer lifecycle (begin / acceptThrows / onError absorb / onError rethrow / complete)
40   *  - readInner BeanSupplier guard, void short-circuit
41   *  - Listener notifications (onUnknownBeanProperty, onError)
42   *  - Builder property() switch (javaMethod, outer, schema, trimStrings, alternates)
43   *  - schemaDefault no-overwrite semantics
44   *  - Accessors: getInputAsString, getLastLocation, getListener (typed/untyped),
45   *    getPosition, getSchema, isReaderParser
46   *  - Trim-strings behavior at session level
47   *
48   * Implementation notes for coverage tests:
49   *  - {@link ParserSession#doReadIntoMap}/{@code doReadIntoCollection} default impls throw
50   *    {@code UnsupportedOperationException}; this is exercised below via a parser that does not
51   *    override them (HTML-table-style parsers all do).  No JSON parser sets this path because
52   *    {@link JsonParserSession} overrides both, but the public
53  	 *    {@link ParserSession#readIntoMap}/{@link ParserSession#readIntoCollection} wrappers
54   *    re-wrap the {@code UnsupportedOperationException} as a {@link ParseException}.
55   *  - The XML follow-up bug was a {@code doReadIntoMap}/{@code doReadIntoCollection}
56   *    self-recursion where the override called {@code super.doRead(...)} instead of the
57   *    format-specific parse-anything routine.  No similar dispatch issues were observed in JSON,
58   *    UON, or CSV parser sessions; all override and call their internal {@code readAnything}.
59   */

397 	public static class CountingListener extends ParserListener {
398 		static final List<String> events = Collections.synchronizedList(new ArrayList<>());
399 		@Override public <T> void onUnknownBeanProperty(ParserSession s, String name, Class<T> beanClass, T bean) {
400 			events.add("unknown:" + name);
401 		}
402 		@Override public void onBeanSetterException(ParserSession s, Throwable t, BeanPropertyMeta p) {
403 			events.add("setter-ex:" + p.getName());
404 		}
405 		@Override public void onError(ParserSession s, Throwable t, String msg) {
406 			events.add("error:" + msg);
407 		}
408 	}
409 
410 	@Test void f01_listener_unknownProperty() {
411 		CountingListener.events.clear();
412 		var p = JsonParser.create()
413 			.ignoreUnknownBeanProperties()
414 			.listener(CountingListener.class)
415 			.build();
416 		p.read("{\"f1\":\"x\",\"f2\":1,\"unknown\":\"v\"}", Bean.class);
417 		assertTrue(CountingListener.events.stream().anyMatch(s -> s.startsWith("unknown:unknown")));
418 	}

422 		// throws a ParseException before listener.onUnknownBeanProperty is called.
423 		CountingListener.events.clear();
424 		var p = JsonParser.create().listener(CountingListener.class).build();
425 		assertThrows(ParseException.class,
426 			() -> p.read("{\"unknown\":\"v\"}", Bean.class));
