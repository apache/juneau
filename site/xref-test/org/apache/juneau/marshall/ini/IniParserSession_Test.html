31   * Coverage-focused tests for {@link IniParserSession} targeting branches not already exercised by
32   * {@link IniParser_Test} / {@link IniRoundTrip_Test}:
33   *  - {@code doRead}'s null-{@code Reader} short-circuit (real {@code r == null} arm, triggered by a null
34   *    parse input).
35   *  - {@code readIniContent}'s "reader is already a {@link BufferedReader}" reuse arm (only reachable by
36   *    calling the {@code protected} method directly with a real {@link BufferedReader}, since the public
37   *    read path always wraps input in a {@code ParserReader}).
38   *  - {@code populateBean}/{@code buildMapFromSections}'s multi-level section recursion (grandchild sections,
39   *    {@code cMeta.isMap()} nested-map-property arm, and silently-ignored unmatched nested sections -- which
40   *    is asymmetric with the root-property loop's throw-on-unknown default).
41   *  - {@code readValue}'s date/calendar/temporal/duration/period dispatch arms, and {@code '}-quoted-string
42   *    edge cases (unterminated quote, bare single quote).
43   */

162 		// Unlike the root-property loop (which throws ParseException for an unknown key -- see
163 		// IniParser_Test.b04_unknownBeanPropertyThrows), populateBean's nested-SECTION loop silently
164 		// continues past any section name that doesn't match a bean property, regardless of
165 		// ignoreUnknownBeanProperties(). Pin this asymmetry (flagged, not "fixed", per task scope).
166 		var ini = "name = John\n\n[bogus]\nx = 1";
167 		var result = IniParser.DEFAULT.read(ini, C_Employee.class);

212 	@Test void d02_grandchildOnlySectionWithNoOwnHeaderIsRecovered() {
213 		// FIXED: buildMapFromSections's isChild check used to only ever recognize a section as a "direct
214 		// child" of sectionPath by matching literal keys already present in the sections map. A grandchild
215 		// section like "employment/company" was never itself a direct child of "" (it contains
216 		// SECTION_PATH_DELIMITER), so if there's no separate literal "[employment]" header line,
217 		// "employment" was never generated as a childName candidate at the root level -- the whole
218 		// "employment/company" subtree was silently dropped from the result, even though the input
219 		// unambiguously specifies it. Any non-root section is a descendant of "" by definition, so the root
220 		// level now recognizes ALL sections as candidate children, deriving the immediate childName by
221 		// splitting on the path delimiter (matching the non-root arm's existing behavior).
222 		var ini = "name = John\n\n[employment/company]\nname = Acme";
223 		var result = (Map<String,Object>) IniParser.DEFAULT.read(ini, Map.class, String.class, Object.class);
224 		assertEquals("John", result.get("name"));
225 		var employment = (Map<String,Object>) result.get("employment");
226 		assertNotNull(employment);
227 		var company = (Map<String,Object>) employment.get("company");
228 		assertEquals("Acme", company.get("name"));
229 	}

231 	@Test void d04_grandchildSectionHeaderMissing_defaultSectionNullAndDedup() {
232 		// Three-level structure with the *middle* level's own header ("[a/b]") never literally present --
233 		// only "[a]" and two great-grandchild-ish entries "[a/b/x]" / "[a/b/y]" exist. When
234 		// buildMapFromSections recurses into the synthesized "a/b" path (derived from splitting
235 		// "a/b/x"/"a/b/y"), sections.get("a/b") is null (no such literal header), hitting defaultSection's
236 		// null arm. Then "a/b/y" resolves to the SAME childName ("b") as "a/b/x" did, already present in
237 		// result -- hitting the containsKey() dedup arm too.
238 		var ini = "[a]\n\n[a/b/x]\nval = 1\n\n[a/b/y]\nval = 2";
239 		var result = (Map<String,Object>) IniParser.DEFAULT.read(ini, Map.class, String.class, Object.class);
240 		var a = (Map<String,Object>) result.get("a");
241 		var b = (Map<String,Object>) a.get("b");
242 		var x = (Map<String,Object>) b.get("x");
243 		var y = (Map<String,Object>) b.get("y");
244 		assertEquals("1", x.get("val"));
245 		assertEquals("2", y.get("val"));
246 	}

248 	@Test void d03_duplicateChildNameDeduped() {
249 		// Two section headers that both resolve to the same immediate childName ("a") under the root --
250 		// e.g. "[a]" (processed first from its own key/values) and "[a/b]" (a grandchild under it). The
251 		// second time "a" is encountered as a candidate childName, result.containsKey("a") is already true,
252 		// so buildMapFromSections's dedup guard skips reprocessing it.
253 		var ini = "[a]\nx = 1\n\n[a/b]\ny = 2";
254 		var result = (Map<String,Object>) IniParser.DEFAULT.read(ini, Map.class, String.class, Object.class);
255 		var a = (Map<String,Object>) result.get("a");
256 		assertEquals("1", a.get("x"));
257 		var b = (Map<String,Object>) a.get("b");
258 		assertEquals("2", b.get("y"));
259 	}

262 	// e0x - Date / Calendar / Temporal / Duration / Period bean-property round trips.
263 	//
264 	// NOTE ON readValue's isDate()/isCalendar()/isTemporal()/isDuration()/isPeriod() arms: these look
265 	// reachable from populateBean's `readValue(rawValue, (ClassMeta<?>) pMeta.getBeanInfo())` call for a
266 	// Date/Calendar/Temporal/Duration/Period-typed bean property, but they are NOT -- confirmed by direct
267 	// reflection probing of BeanPropertyMeta -- because these types get a default post-processor-installed
268 	// swap, and BeanPropertyMeta#getBeanInfo() (backed by the swap-aware `typeMeta` field, distinct from the
269 	// unswapped `rawTypeMeta`) then reports the property as plain java.lang.Object rather than as
270 	// Date/Calendar/isTemporal()/etc-classified. (A plain numeric property like `int`/`double` has no such
271 	// swap, so isNumber() *is* reachable and already covered by IniParser_Test.a05.) The actual date-ish
272 	// conversion for these properties happens later via BeanMap#put()'s own type-aware coercion, which is
273 	// swap-aware -- NOT via these specific readValue lines. Since buildMapFromSections's two call sites
274 	// always pass object() too, these 10 lines are dead from every call site in this class; see the "// HTT"
275 	// markers on them. These round-trip tests are kept regardless since they're genuine regression coverage
276 	// of Date/Calendar/Temporal/Duration/Period bean properties end-to-end (exercising the real
277 	// swap-driven conversion path instead).
278 	//------------------------------------------------------------------------------------------------------------------
