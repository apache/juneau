32  /**
33   * Cross-pair round-trip coverage for {@link BinaryFormat} across every supported serializer/parser pair.
34   *
35   * <p>
36   * Sibling of {@link UuidFormat_RoundTrip_Test} — same shape, same 12 RDF tester templates, varied across
37   * every {@link BinaryFormat} value.  Round-trips representative {@code byte[]} values:
38   * <ul>
39   * 	<li>As a bean property (exercises the {@code MarshalledPropertyPostProcessor} context-format swap install
40   * 		path at {@code applyContextFormats}).
41   * 	<li>As a {@link List} element (exercises the {@code DefaultSwaps} {@link org.apache.juneau.marshall.swaps.BinarySwap}
42   * 		fallback for non-bean-property byte arrays).
43   * 	<li>As a top-level value (exercises the per-format dispatch sites in each {@code *SerializerSession} /
44   * 		{@code *ParserSession}).
45   * </ul>
46   *
47   * <p>
48   * Test combos = (tester templates) &times; ({@link BinaryFormat} values).  At the time of writing this
49   * comes to 12 &times; 5 = 60 testers per test method.
50   *
51   * <p>
52   * {@link BinaryFormat} only affects text-based serializers per the class-level "Binary serializers" note —
53   * BSON / CBOR / MsgPack / Prototext / Parquet emit native bytes regardless of the configured constant.  The
54   * variant {@code binarySwap} installed by {@code MarshalledPropertyPostProcessor} respects that by handing
55   * the raw {@code byte[]} back to {@link org.apache.juneau.marshall.serializer.OutputStreamSerializerSession}
56   * subtypes instead of the formatted wire string, so bean-property round-trips through binary serializers
57   * still resolve to the original bytes via native handling.  Top-level / {@link List}-element paths route
58   * through the default-swap dispatch ({@link org.apache.juneau.marshall.swaps.BinarySwap}) which short-circuits to
59   * raw bytes for binary sessions — same lossless round-trip via the native path.
60   */
61  class BinaryFormat_RoundTrip_Test extends TestBase {

119 	private static final List<Arguments> COMBOS = buildCombos();

121 	private static List<Arguments> buildCombos() {
122 		return Stream.of(BinaryFormat.values())
123 			.flatMap(fmt -> BUILDERS.stream().map(b -> Arguments.of(b.build(fmt), fmt)))
124 			.toList();
125 	}

127 	static Stream<Arguments> combos() {
128 		return COMBOS.stream();
129 	}

131 	/**
132 	 * Asserts that {@code expected} and {@code actual} have the same length and the same bytes in the
133 	 * same positions.  {@code byte[]} doesn't override {@link Object#equals(Object)}, so a plain
134 	 * {@code assertEquals} would compare identity.  Mirrors the {@code Float.equals} bit-pattern
135 	 * asserter pattern in {@link FloatFormat_Float_RoundTrip_Test}.
136 	 */
137 	private static void assertBytesEquals(byte[] expected, byte[] actual, String message) {
138 		if (expected == null) {
139 			assertNull(actual, message);
140 		} else {
141 			assertNotNull(actual, message);
142 			assertArrayEquals(expected, actual, message);
143 		}
144 	}

195 	//====================================================================================================
196 	// Bean with edge-case byte[] values — empty array + BASE64-padding-boundary shapes
197 	//
198 	// <p>
199 	// The 4-byte-boundary array round-trips through BASE64 without padding ({@code "EjRWeA=="} →
200 	// {@code "EjRWeA"} for URL-safe).  The 3-byte array forces 1 padding char on standard BASE64
201 	// ({@code "qrvM"}) and 0 on BASE64_URL.  The 1-byte array forces 2 padding chars on standard
202 	// ({@code "mQ=="}) and 0 on BASE64_URL.  Empty array round-trips as the empty wire string.
203 	//====================================================================================================
204 
205 	public static class A03Bean {
206 		public byte[] empty;
207 		public byte[] aligned4;
208 		public byte[] aligned3;
209 		public byte[] aligned1;
210 	}
211 
212 	@ParameterizedTest(name = "{0} | {1}")
213 	@MethodSource("combos")
214 	void a03_byteArrayProperty_edgeCases(RoundTrip_Tester t, BinaryFormat fmt) throws Exception {
215 		var x = new A03Bean();
216 		x.empty = cp(EMPTY);
217 		x.aligned4 = cp(ALIGNED_4);
218 		x.aligned3 = cp(ALIGNED_3);
219 		x.aligned1 = cp(ALIGNED_1);
220 
221 		x = t.roundTrip(x);
222 		assertBytesEquals(EMPTY, x.empty, "fmt=" + fmt + " field=empty");
223 		assertBytesEquals(ALIGNED_4, x.aligned4, "fmt=" + fmt + " field=aligned4");
224 		assertBytesEquals(ALIGNED_3, x.aligned3, "fmt=" + fmt + " field=aligned3");
225 		assertBytesEquals(ALIGNED_1, x.aligned1, "fmt=" + fmt + " field=aligned1");
226 	}

228 	//====================================================================================================
229 	// Bean with a list-of-byte[] property — collection-element dispatch
230 	//
231 	// <p>
232 	// {@code List<byte[]>} bypasses the per-property {@code binarySwap} install (that swap is keyed
233 	// on the bean property's class being {@code byte[]}, not {@code List<byte[]>}).  Collection
234 	// elements instead route through the {@code DefaultSwaps} fallback
235 	// {@link org.apache.juneau.marshall.swaps.BinarySwap}, which honours {@link BinaryFormat} for text
236 	// sessions and short-circuits to raw bytes for binary sessions.  The mixed-shape list values
237 	// below cover all three BASE64-padding boundaries plus the negative-bytes case.
238 	//====================================================================================================
239 
240 	public static class A04Bean {
241 		public List<byte[]> list;
242 	}
243 
244 	@ParameterizedTest(name = "{0} | {1}")
245 	@MethodSource("combos")
246 	void a04_byteArrayProperty_inList(RoundTrip_Tester t, BinaryFormat fmt) throws Exception {
247 		var x = new A04Bean();
248 		x.list = new ArrayList<>();
249 		x.list.add(cp(SHORT_MIXED));
250 		x.list.add(cp(ALIGNED_3));
251 		x.list.add(cp(ALIGNED_1));
252 
253 		x = t.roundTrip(x);
254 		assertNotNull(x.list, "fmt=" + fmt);
255 		assertEquals(3, x.list.size(), "fmt=" + fmt);
256 		assertBytesEquals(SHORT_MIXED, x.list.get(0), "fmt=" + fmt + " index=0");
257 		assertBytesEquals(ALIGNED_3, x.list.get(1), "fmt=" + fmt + " index=1");
258 		assertBytesEquals(ALIGNED_1, x.list.get(2), "fmt=" + fmt + " index=2");
259 	}

274 	//====================================================================================================
275 	// Standalone top-level byte[] — exercises per-format dispatch in *SerializerSession / *ParserSession
276 	//====================================================================================================
277 
278 	@ParameterizedTest(name = "{0} | {1}")
279 	@MethodSource("combos")
280 	void a06_byteArrayTopLevel(RoundTrip_Tester t, BinaryFormat fmt) throws Exception {
281 		// Validation-only testers (Json schema, Csv, Parquet) have no parser-side round-trip semantics
282 		// for a standalone byte[] value — they assert that the serializer accepts it without throwing.
283 		if (t.isValidationOnly())
284 			return;
285 
286 		var s = t.getSerializer();
287 		var p = t.getParser();
288 		if (p == null)
289 			return;
290 
291 		var x = cp(SHORT_MIXED);
292 		try {
293 			var out = t.serialize(x, s);
294 			var x2 = p.read(out, byte[].class);
295 			assertBytesEquals(SHORT_MIXED, x2, "fmt=" + fmt);
296 		} catch (Exception e) {
297 			// Mirror UuidFormat_RoundTrip_Test.a06: some serializers (URL-encoding, CSV-style) don't
298 			// support a top-level non-bean value cleanly.  Bean-property tests above already cover the
299 			// per-property dispatch path; this case is best-effort.
300 		}
