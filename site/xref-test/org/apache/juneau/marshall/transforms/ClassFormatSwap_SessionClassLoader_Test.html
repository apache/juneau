17  package org.apache.juneau.marshall.transforms;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  import static org.junit.jupiter.api.Assumptions.*;
21  
22  import java.net.*;
23  import java.nio.file.*;
24  
25  import javax.tools.*;
26  
27  import org.apache.juneau.*;
28  import org.apache.juneau.marshall.json.*;
29  import org.junit.jupiter.api.*;

31  /**
32   * Verifies that {@link org.apache.juneau.marshall.swaps.ClassFormatSwap#unswap} consults the session-installed
33   * classloader before falling back to the thread-context classloader.
34   *
35   * <p>
36   * Three scenarios are exercised:
37   * <ol>
38   *   <li>Happy path — session CL resolves a class that is invisible to the thread-context CL.</li>
39   *   <li>Fallback — no session CL is set; the thread-context CL resolves the class as before.</li>
40   *   <li>Isolation — when a session CL is set, the thread-context CL is NOT consulted as a fallback,
41   *       so a class that lives only in the thread-context CL cannot be resolved through a session CL
42   *       that does not have it.</li>
43   * </ol>
44   */
45  @SuppressWarnings({
46  	"unused" // Exception parameter intentionally unused in catch block; only the fact of the exception matters.
47  })
48  class ClassFormatSwap_SessionClassLoader_Test extends TestBase {
49  
50  	/**
51  	 * FQCN of the class compiled into the isolated temp directory.  This class does not exist on the
52  	 * test classpath and is therefore invisible to the thread-context classloader.
53  	 */
54  	private static final String ISOLATED_FQCN = "org.apache.juneau.marshall.test.isolated.IsolatedBean";
55  
56  	/** URLs pointing at the isolated class output directory — passed to the session CL. */
57  	private static URL[] isolatedUrls;
58  
59  	/** Temp directory that holds the compiled isolated class. */
60  	private static Path tempDir;
61  
62  	/**
63  	 * Compiles a tiny {@code IsolatedBean} class into a fresh temp directory using the system Java
64  	 * compiler so it is invisible to the test classpath / thread-context classloader.
65  	 */
66  	@BeforeAll
67  	static void compileIsolatedBean() throws Exception {
68  		var compiler = ToolProvider.getSystemJavaCompiler();
69  		assumeTrue(compiler != null, "System Java compiler unavailable — skipping session-CL tests");
70  		assertNotNull(compiler); // Null guard for static-analysis: assumeTrue above aborts the test if null.
71  
72  		tempDir = Files.createTempDirectory("juneau-cltest-");
73  		var srcPkg = tempDir.resolve(Path.of("src", "org", "apache", "juneau", "test", "isolated"));
74  		Files.createDirectories(srcPkg);
75  		var srcFile = srcPkg.resolve("IsolatedBean.java");
76  		Files.writeString(srcFile, "package org.apache.juneau.marshall.test.isolated; public class IsolatedBean {}");
77  
78  		var outDir = tempDir.resolve("classes");
79  		Files.createDirectories(outDir);
80  
81  		var exitCode = compiler.run(null, null, null, "-d", outDir.toString(), srcFile.toString());
82  		assertEquals(0, exitCode, "IsolatedBean compilation failed");
83  
84  		isolatedUrls = new URL[]{outDir.toUri().toURL()};
85  	}
86  
87  	@AfterAll
88  	static void cleanupTempDir() {
89  		if (tempDir != null) {
90  			try {
91  				try (var walk = Files.walk(tempDir)) {
92  					walk.sorted(java.util.Comparator.reverseOrder())
93  						.map(Path::toFile)
94  						.forEach(java.io.File::delete);
95  				}
96  			} catch (Exception ignored) { // HTT
97  			}
98  		}
99  	}
100 
101 	//====================================================================================================
102 	// a01 — happy path: session CL resolves IsolatedBean; thread-context CL cannot
103 	//====================================================================================================
104 
105 	/**
106 	 * When a session classloader is installed that contains {@code IsolatedBean}, parsing its FQCN
107 	 * succeeds and the returned {@code Class} is loaded by the session CL.
108 	 */
109 	@Test void a01_sessionCL_resolvesClassInvisibleToThreadContextCL() throws Exception {
110 		// Confirm thread-context CL cannot find IsolatedBean — our baseline.
111 		assertThrows(Exception.class,
112 			() -> JsonParser.create().build().read("\"" + ISOLATED_FQCN + "\"", Class.class),
113 			"Precondition: thread-context CL must NOT resolve IsolatedBean"
114 		);
115 
116 		// With the session CL pointing at the compiled class, parsing must succeed.
117 		try (var sessionCL = new URLClassLoader(isolatedUrls, null)) {
118 			var parser = JsonParser.create().classLoader(sessionCL).build();
119 			var result = parser.read("\"" + ISOLATED_FQCN + "\"", Class.class);
120 
121 			assertNotNull(result);
122 			assertEquals(ISOLATED_FQCN, result.getName());
123 			// The class was loaded by the session CL, not the thread-context CL.
124 			assertSame(sessionCL, result.getClassLoader(),
125 				"Class must be defined by the session CL, not the thread-context CL");
126 		}
127 	}
128 
129 	//====================================================================================================
130 	// a02 — fallback: no session CL → thread-context CL used (historical behavior preserved)
131 	//====================================================================================================
132 
133 	/**
134 	 * When no session classloader is configured, the thread-context classloader is still used as the
135 	 * fallback — preserving the historical behavior for existing callers.
136 	 */
137 	@Test void a02_noSessionCL_fallsBackToThreadContextCL() throws Exception {
138 		// JsonParser itself is on the test classpath and therefore resolvable via thread-context CL.
139 		var fqcn = JsonParser.class.getName();
140 		var parser = JsonParser.create().build();
141 
142 		var result = parser.read("\"" + fqcn + "\"", Class.class);
143 
144 		assertNotNull(result);
145 		assertEquals(fqcn, result.getName());
146 	}
147 
148 	//====================================================================================================
149 	// a03 — isolation: session CL is set but cannot see the target → no silent thread-CL fallback
150 	//====================================================================================================
151 
152 	/**
153 	 * When a session classloader is explicitly set, the thread-context classloader is NOT consulted
154 	 * as a fallback.  A class that lives only on the test classpath cannot be found through an empty
155 	 * session CL, even though the thread-context CL would resolve it.
156 	 *
157 	 * <p>
158 	 * This proves the session CL is genuinely consulted first — not the thread CL — and that there is
159 	 * no silent double-lookup that would degrade to the old behavior when the session CL comes up empty.
160 	 */
161 	@Test void a03_sessionCL_set_noFallbackToThreadContextCL() throws Exception {
162 		// JsonParser lives on the test classpath (thread-context CL sees it).
163 		// An empty URLClassLoader with null parent only delegates to the bootstrap CL
164 		// (JDK core classes only) — it cannot see non-JDK test classpath classes.
165 		var fqcn = JsonParser.class.getName();
166 
167 		try (var emptyCL = new URLClassLoader(new URL[0], null)) {
168 			var parser = JsonParser.create().classLoader(emptyCL).build();
169 
170 			// Must fail: session CL cannot find JsonParser, and there is no thread-CL fallback.
171 			assertThrows(Exception.class,
172 				() -> parser.read("\"" + fqcn + "\"", Class.class),
173 				"Should fail — session CL is set but cannot find " + fqcn
174 					+ "; thread-context CL must NOT be consulted as a fallback"
175 			);
176 		}
177 	}
