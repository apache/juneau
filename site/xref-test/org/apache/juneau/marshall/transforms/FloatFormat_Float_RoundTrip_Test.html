32  /**
33   * Cross-pair round-trip coverage for {@link FloatFormat} on {@link Float} across every supported
34   * serializer/parser pair.
35   *
36   * <p>
37   * Sibling of {@link FloatFormat_Double_RoundTrip_Test} — same shape, same 12 RDF tester templates, varied
38   * across every {@link FloatFormat} value.  Round-trips representative {@link Float} values:
39   * <ul>
40   * 	<li>As a bean property (exercises the {@code MarshalledPropertyPostProcessor} context-format swap install
41   * 		path at {@code applyContextFormats}).
42   * 	<li>As a top-level value (exercises the per-format dispatch sites in each {@code *SerializerSession} /
43   * 		{@code *ParserSession}).
44   * </ul>
45   *
46   * <p>
47   * Test combos = (tester templates) &times; ({@link FloatFormat} values).  At the time of writing this comes to
48   * 12 &times; 5 = 60 testers per test method.
49   *
50   * <p>
51   * {@link FloatFormat} only controls how non-finite values ({@code NaN}, {@code ±Infinity}) are emitted on
52   * <b>text-based</b> wire formats; finite values ride the natural bare-numeric token and binary serializers
53   * (BSON / CBOR / MsgPack / Prototext / Parquet) emit native IEEE-754 regardless of this setting per the
54   * {@link FloatFormat} class-level "Binary serializers" note.  The matrix below exercises finite values across
55   * all five {@link FloatFormat} constants for the standard tests, with a dedicated non-finite test
56   * ({@link #a04_floatProperty_nonFinite}) that skips {@link FloatFormat#NaN_AS_ERROR} (throws on swap) and
57   * canonicalises against the format-specific wire shape via {@link #expectedAfterNonFinite}.
58   */
59  @SuppressWarnings({
60  	"unused" // Parameters retained for method-signature/functional-interface consistency in test fixtures.
61  })

129 	/** Pre-computed (tester, format) combinations.  One row per (builder &times; format) pair. */
130 	private static final List<Arguments> COMBOS = buildCombos();

132 	private static List<Arguments> buildCombos() {
133 		return Stream.of(FloatFormat.values())
134 			.flatMap(fmt -> BUILDERS.stream().map(b -> Arguments.of(b.build(fmt), fmt)))
135 			.toList();

138 	static Stream<Arguments> combos() {

142 	/**
143 	 * Returns <jk>true</jk> when {@code t} is wrapped around a binary serializer (any
144 	 * {@link OutputStreamSerializer} subtype).  Binary serializers emit native IEEE-754 regardless of
145 	 * {@link FloatFormat} per the class-level "Binary serializers" note, so non-finite values round-trip
146 	 * natively without going through the format dispatch.  Covers MsgPack, CBOR, BSON, Parquet, Prototext,
147 	 * and the RDF stream variants (RdfThrift / RdfProto).
148 	 */
149 	private static boolean isBinarySerializer(RoundTrip_Tester t) {
150 		return t.getSerializer() instanceof OutputStreamSerializer;
151 	}
152 
153 	/**
154 	 * Format-aware expectation helper for finite {@link Float} values.  {@link FloatFormat} affects only
155 	 * non-finite values per the {@link FloatFormat} class-level contract; finite values ride the natural
156 	 * bare-numeric wire token on text serializers and native IEEE-754 on binary serializers.  Validation-only
157 	 * testers ({@link RoundTrip_Tester#isValidationOnly()} == {@code true}: Json schema, CSV, Parquet) return
158 	 * the original object unchanged.
159 	 */
160 	@SuppressWarnings({
161 		"java:S1172" // 'fmt' is part of the shared expectedAfter(...) helper signature used across all *Format RoundTrip tests; kept for template symmetry even where this type's expected value is format-independent.
162 	})
163 	private static Float expectedAfter(Float original, RoundTrip_Tester t, FloatFormat fmt) {
164 		if (original == null)
165 			return null;
166 		if (t.isValidationOnly())
167 			return original;
168 		return original;
169 	}
170 
171 	/**
172 	 * Format-aware expectation helper for non-finite {@link Float} values ({@code NaN}, {@code ±Infinity}).
173 	 *
174 	 * <p>
175 	 * Binary serializers ({@link OutputStreamSerializer} subtypes) emit native IEEE-754 regardless of
176 	 * {@link FloatFormat} per the class-level "Binary serializers" note — the original non-finite value
177 	 * round-trips intact.
178 	 *
179 	 * <p>
180 	 * Text serializers — only {@link FloatFormat#NaN_AS_NULL} and {@link FloatFormat#NaN_AS_STRING} are
181 	 * exercised at this matrix level (per the {@link #a04_floatProperty_nonFinite} method-doc):
182 	 * <ul>
183 	 * 	<li>{@link FloatFormat#NaN_AS_NULL} — wire emits {@code null}; round-trip resolves to {@code null}
184 	 * 		for the boxed {@link Float} property.
185 	 * 	<li>{@link FloatFormat#NaN_AS_STRING} — wire emits the quoted token; parser decodes back to the
186 	 * 		original non-finite value.
187 	 * </ul>
188 	 */
189 	private static Float expectedAfterNonFinite(Float original, RoundTrip_Tester t, FloatFormat fmt) {
190 		if (original == null)
191 			return null;
192 		if (t.isValidationOnly())
193 			return original;
194 		if (isBinarySerializer(t))
195 			return original;
196 		return switch (fmt) {
197 			case NaN_AS_NULL -> null;
198 			case NaN_AS_STRING -> original;
199 			default -> original;  // unreachable — callers skip NOT_SET / NaN_AS_NUMBER / NaN_AS_ERROR for text.
200 		};
201 	}
202 
203 	/**
204 	 * Compare two {@link Float} values using {@link Float#equals(Object)} which compares raw IEEE-754 bits.
205 	 * Critical for non-finite values: {@code Float.NaN == Float.NaN} is {@code false}, but
206 	 * {@code Float.valueOf(NaN).equals(Float.valueOf(NaN))} is {@code true} by spec.
207 	 */
208 	private static void assertFloatEquals(Float expected, Float actual, String message) {
209 		if (expected == null) {
210 			assertNull(actual, message);
211 		} else {
212 			assertNotNull(actual, message);
213 			assertEquals(expected, actual, message + " : expected " + expected + " but was " + actual);
214 		}
215 	}

283 	//====================================================================================================
284 	// Bean with non-finite Float values — NaN / +Infinity / -Infinity edges (the FloatFormat-specific path)
285 	//
286 	// <p>
287 	// The swap install logic in {@code MarshalledPropertyPostProcessor.requiresFloatSwap} only installs
288 	// the float swap when {@code format == NaN_AS_NULL || NaN_AS_STRING || NaN_AS_ERROR}.  For // NOSONAR
289 	// {@code NOT_SET} and {@code NaN_AS_NUMBER}, no swap is installed and the round-trip rides each
290 	// serializer's native non-finite handling, which varies widely across formats (JSON rejects bare
291 	// {@code NaN}; Toml/Ini have no NaN grammar; JCS rejects per RFC 8785; XML/HTML/RDF preserve
292 	// natively).  Out-of-matrix scope: those combinations are exercised at the format-specific test
293 	// level, not via this cross-pair matrix.  {@link FloatFormat#NaN_AS_ERROR} throws during swap on
294 	// non-finite values — also skipped.  This leaves the predictable contract:
295 	// <ul>
296 	// 	<li>Binary serializers — emit / parse native IEEE-754 {@code NaN} / {@code ±Infinity} regardless
297 	// 		of {@code FloatFormat} per the {@link FloatFormat} class-level "Binary serializers" note.
298 	// 	<li>Text serializers with {@link FloatFormat#NaN_AS_NULL} — emit / parse {@code null} (lossy by
299 	// 		design — round-trips to {@code null} for boxed {@link Float}).
300 	// 	<li>Text serializers with {@link FloatFormat#NaN_AS_STRING} — emit / parse the quoted
301 	// 		{@code "NaN"} / {@code "Infinity"} / {@code "-Infinity"} token (round-trip preserving).
302 	// </ul>
303 	//====================================================================================================
304 
305 	public static class A04Bean {
306 		public Float nan;
307 		public Float posInf;
308 		public Float negInf;
309 	}
310 
311 	@ParameterizedTest(name = "{0} | {1}")
312 	@MethodSource("combos")
313 	void a04_floatProperty_nonFinite(RoundTrip_Tester t, FloatFormat fmt) throws Exception {
314 		// NaN_AS_ERROR throws during swap on non-finite values — skip the entire assertion block.
315 		if (fmt == FloatFormat.NaN_AS_ERROR)
316 			return;
317 		// NOT_SET / NaN_AS_NUMBER on text serializers fall through to native handling — wildly varied
318 		// per format, out-of-matrix scope.  Binary serializers still preserve native NaN.
319 		var binary = isBinarySerializer(t);
320 		if (!binary && (fmt == FloatFormat.NOT_SET || fmt == FloatFormat.NaN_AS_NUMBER))
321 			return;
322 
323 		var x = new A04Bean();
324 		x.nan = Float.NaN;
325 		x.posInf = Float.POSITIVE_INFINITY;
326 		x.negInf = Float.NEGATIVE_INFINITY;
327 
328 		x = t.roundTrip(x);
329 		assertFloatEquals(expectedAfterNonFinite(Float.NaN, t, fmt), x.nan, "fmt=" + fmt + " field=nan");
330 		assertFloatEquals(expectedAfterNonFinite(Float.POSITIVE_INFINITY, t, fmt), x.posInf, "fmt=" + fmt + " field=posInf");
331 		assertFloatEquals(expectedAfterNonFinite(Float.NEGATIVE_INFINITY, t, fmt), x.negInf, "fmt=" + fmt + " field=negInf");
332 	}

348 	// Standalone top-level Float — exercises per-format dispatch in *SerializerSession / *ParserSession
349 	//====================================================================================================
350 
351 	@ParameterizedTest(name = "{0} | {1}")
352 	@MethodSource("combos")
353 	void a06_floatTopLevel(RoundTrip_Tester t, FloatFormat fmt) throws Exception {
354 		// Validation-only testers (Json schema, Csv, Parquet) have no parser-side round-trip semantics for a
355 		// standalone Float value — they assert that the serializer accepts it without throwing.
356 		if (t.isValidationOnly())
357 			return;
358 
359 		var s = t.getSerializer();
360 		var p = t.getParser();
361 		if (p == null)
362 			return;
363 
364 		var x = 3.14f;
365 		try {
366 			var out = t.serialize(x, s);
367 			var x2 = p.read(out, Float.class);
368 			assertFloatEquals(expectedAfter(x, t, fmt), x2, "fmt=" + fmt);
369 		} catch (Exception e) {
370 			// Mirror RoundTripDateTime_Test.a06_standaloneInstant: some serializers (URL-encoding, CSV-style)
371 			// don't support a top-level non-bean value cleanly.  Bean-property tests above already cover the
372 			// per-property dispatch path; this case is best-effort.
373 		}
