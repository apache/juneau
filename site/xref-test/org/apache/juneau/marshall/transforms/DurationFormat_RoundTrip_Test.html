32  /**
33   * Cross-pair round-trip coverage for {@link DurationFormat} across every supported serializer/parser pair.
34   *
35   * <p>
36   * This is the pilot test for the format-control round-trip pattern.  It varies the
37   * {@link DurationFormat} setting on both the serializer and parser builders for each tester template, then
38   * round-trips representative {@link Duration} values:
39   * <ul>
40   * 	<li>As a top-level value (exercises the per-format dispatch sites in each {@code *SerializerSession} /
41   * 		{@code *ParserSession}).
42   * 	<li>As a bean property (exercises the {@code MarshalledPropertyPostProcessor} context-format swap install
43   * 		path at {@code applyContextFormats}).
44   * </ul>
45   *
46   * <p>
47   * Test combos = (tester templates) &times; ({@link DurationFormat} values).  At the time of writing this comes to
48   * roughly 41 &times; 7 = 287 testers per test method.
49   *
50   * <p>
51   * If a tester is validation-only (no parser, schema-only, or CSV in serializer-only mode) the top-level
52   * standalone-Duration test is skipped — there is no Duration round-trip semantics to assert in that case.
53   */
54  class DurationFormat_RoundTrip_Test extends TestBase {
55

121 	/** Pre-computed (tester, format) combinations.  One row per (builder &times; format) pair. */
122 	private static final List<Arguments> COMBOS = buildCombos();
123 
124 	private static List<Arguments> buildCombos() {
125 		return Stream.of(DurationFormat.values())
126 			.flatMap(fmt -> BUILDERS.stream().map(b -> Arguments.of(b.build(fmt), fmt)))
127 			.toList();
128 	}
129 
130 	static Stream<Arguments> combos() {
131 		return COMBOS.stream();
132 	}

228 	//====================================================================================================
229 	// Standalone top-level Duration — exercises per-format dispatch in *SerializerSession / *ParserSession
230 	//====================================================================================================
231 
232 	@ParameterizedTest(name = "{0} | {1}")
233 	@MethodSource("combos")
234 	void a06_durationTopLevel(RoundTrip_Tester t, DurationFormat fmt) throws Exception {
235 		// Validation-only testers (Json schema, Csv, Parquet) have no parser-side round-trip semantics for a
236 		// standalone Duration value — they assert that the serializer accepts it without throwing.
237 		if (t.isValidationOnly())
238 			return;
239 
240 		var s = t.getSerializer();
241 		var p = t.getParser();
242 		if (p == null)
243 			return;
244 
245 		var x = Duration.ofHours(2).plusMinutes(15);
246 		try {
247 			var out = t.serialize(x, s);
248 			var x2 = p.read(out, Duration.class);
249 			assertEquals(x, x2, "fmt=" + fmt);
250 		} catch (Exception e) {
251 			// Mirror RoundTripDateTime_Test.a06_standaloneInstant: some serializers (URL-encoding, CSV-style)
252 			// don't support a top-level non-bean value cleanly.  Bean-property tests above already cover the
253 			// per-property dispatch path; this case is best-effort.
254 		}
255 	}
