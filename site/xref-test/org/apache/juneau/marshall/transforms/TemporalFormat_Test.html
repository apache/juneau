36  	@Test void a01_defaultPerSubtype() {
37  		assertEquals("2011-12-03T10:15:30Z", TemporalFormat.DEFAULT.format(Instant.parse("2011-12-03T10:15:30Z"), Z));
38  		assertEquals("2011-12-03T10:15:30Z", TemporalFormat.NOT_SET.format(Instant.parse("2011-12-03T10:15:30Z"), Z));
39  		assertEquals("2011-12-03", TemporalFormat.DEFAULT.format(LocalDate.parse("2011-12-03"), Z));
40  		assertEquals("2011-12-03T10:15:30", TemporalFormat.DEFAULT.format(LocalDateTime.parse("2011-12-03T10:15:30"), Z));
41  		assertEquals("10:15:30", TemporalFormat.DEFAULT.format(LocalTime.parse("10:15:30"), Z));
42  		assertEquals("2012", TemporalFormat.DEFAULT.format(Year.of(2012), Z));
43  		assertEquals("2012-12", TemporalFormat.DEFAULT.format(YearMonth.of(2012, 12), Z));
44  	}

46  	@Test void a02_isoVariants_onInstant() {
47  		var i = Instant.parse("2011-12-03T10:15:30Z");
48  		assertEquals("2011-12-03T10:15:30Z", TemporalFormat.ISO_INSTANT.format(i, Z));
49  		assertEquals("2011-12-03T10:15:30Z", TemporalFormat.ISO_OFFSET_DATE_TIME.format(i, Z));

110 	@Test void a12_lenientEpochMillis_forInstantLikeOnly() {
111 		var i = Instant.parse("2011-12-03T10:15:30Z");
112 		// Instant: lenient parse accepts a numeric string as epoch millis under DEFAULT/NOT_SET.
113 		assertEquals(i, TemporalFormat.DEFAULT.parse(Long.toString(i.toEpochMilli()), Instant.class, Z));
114 		// Year: a numeric value like "2012" must be parsed as a year, not epoch millis.
115 		assertEquals(Year.of(2012), TemporalFormat.DEFAULT.parse("2012", Year.class, Z));

168 		var i = Instant.parse("2011-12-03T10:15:30Z");
169 		var millis = i.toEpochMilli();
170 		assertEquals(i, TemporalFormat.MILLIS.parse(Long.toString(millis), Instant.class, Z));
171 		assertNotNull(TemporalFormat.MILLIS.parse(Long.toString(millis), ZonedDateTime.class, Z));
172 		assertNotNull(TemporalFormat.MILLIS.parse(Long.toString(millis), OffsetDateTime.class, Z));
173 		assertNotNull(TemporalFormat.MILLIS.parse(Long.toString(millis), LocalDateTime.class, Z));
174 		assertNotNull(TemporalFormat.MILLIS.parse(Long.toString(millis), LocalDate.class, Z));
175 		assertNotNull(TemporalFormat.MILLIS.parse(Long.toString(millis), OffsetTime.class, Z));
176 		assertNotNull(TemporalFormat.MILLIS.parse(Long.toString(millis), YearMonth.class, Z));
177 		assertNotNull(TemporalFormat.MILLIS.parse(Long.toString(millis), Year.class, Z));
