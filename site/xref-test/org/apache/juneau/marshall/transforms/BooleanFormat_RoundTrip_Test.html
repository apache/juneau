31  /**
32   * Cross-pair round-trip coverage for {@link BooleanFormat} across every supported serializer/parser pair.
33   *
34   * <p>
35   * Sibling of {@link BinaryFormat_RoundTrip_Test} — same shape, same 12 RDF tester templates, varied across
36   * every {@link BooleanFormat} value.  Round-trips representative {@link Boolean} / <code><jk>boolean</jk></code>
37   * values:
38   * <ul>
39   * 	<li>As a primitive {@code boolean} bean property (exercises the
40   * 		{@code MarshalledPropertyPostProcessor.applyContextFormats} install path for primitives).
41   * 	<li>As a boxed {@link Boolean} bean property (exercises the boxed-{@code Boolean} path including
42   * 		{@code null} handling).
43   * 	<li>As a {@link List} element (exercises the {@code DefaultSwaps} fallback for non-bean-property
44   * 		booleans — currently a no-op since {@link Boolean} has no registered swap; collection elements
45   * 		flow through each serializer's native boolean handling).
46   * 	<li>As a top-level value (exercises the per-format dispatch sites in each {@code *SerializerSession}
47   * 		/ {@code *ParserSession}).
48   * </ul>
49   *
50   * <p>
51   * Test combos = (tester templates) &times; ({@link BooleanFormat} values).  At the time of writing this
52   * comes to 12 &times; 6 = 72 testers per test method.
53   *
54   * <p>
55   * Per the {@link BooleanFormat} class-level "Binary serializers" note, the binary serializer family
56   * (BSON / CBOR / MsgPack / Prototext / Parquet) emits a native boolean wire type regardless of the
57   * configured constant.  The variant {@code booleanSwap} installed by
58   * {@link MarshalledPropertyPostProcessor} respects that by returning the raw
59   * {@link Boolean} to {@link org.apache.juneau.marshall.serializer.OutputStreamSerializerSession} subtypes
60   * instead of the formatted wire token, so bean-property round-trips through binary serializers stay
61   * lossless via native handling.  Per the class-level "Parser leniency" note, parsers accept any of
62   * the textual shapes regardless of the parser-side setting, so the unswap path on the parser side is
63   * format-agnostic.
64   */
65  class BooleanFormat_RoundTrip_Test extends TestBase {

123 	private static final List<Arguments> COMBOS = buildCombos();

125 	private static List<Arguments> buildCombos() {
126 		return Stream.of(BooleanFormat.values())
127 			.flatMap(fmt -> BUILDERS.stream().map(b -> Arguments.of(b.build(fmt), fmt)))
128 			.toList();
129 	}

131 	static Stream<Arguments> combos() {
132 		return COMBOS.stream();

209 	//====================================================================================================
210 	// Bean with a list-of-Boolean property — collection-element dispatch
211 	//
212 	// <p>
213 	// {@code List<Boolean>} bypasses the per-property {@code booleanSwap} install (that swap is keyed
214 	// on the bean property's class being {@link Boolean}, not {@code List<Boolean>}).  Collection
215 	// elements instead route through each serializer's native boolean handling.  The boxed wrapper
216 	// elements preserve their {@code true} / {@code false} identity through every format.
217 	//====================================================================================================
218 
219 	public static class A04Bean {
220 		public List<Boolean> list;
221 	}
222 
223 	@ParameterizedTest(name = "{0} | {1}")
224 	@MethodSource("combos")
225 	void a04_booleanProperty_inList(RoundTrip_Tester t, BooleanFormat fmt) throws Exception {
226 		var x = new A04Bean();
227 		x.list = new ArrayList<>();
228 		x.list.add(BOXED_TRUE);
229 		x.list.add(BOXED_FALSE);
230 		x.list.add(BOXED_TRUE);
231 
232 		x = t.roundTrip(x);
233 		assertNotNull(x.list, "fmt=" + fmt);
234 		assertEquals(3, x.list.size(), "fmt=" + fmt);
235 		assertEquals(BOXED_TRUE, x.list.get(0), "fmt=" + fmt + " index=0");
236 		assertEquals(BOXED_FALSE, x.list.get(1), "fmt=" + fmt + " index=1");
237 		assertEquals(BOXED_TRUE, x.list.get(2), "fmt=" + fmt + " index=2");
238 	}

263 	//====================================================================================================
264 	// Standalone top-level Boolean — exercises per-format dispatch in *SerializerSession / *ParserSession
265 	//====================================================================================================
266 
267 	@ParameterizedTest(name = "{0} | {1}")
268 	@MethodSource("combos")
269 	void a06_booleanTopLevel(RoundTrip_Tester t, BooleanFormat fmt) throws Exception {
270 		// Validation-only testers (Json schema, Csv, Parquet) have no parser-side round-trip semantics
271 		// for a standalone Boolean value — they assert that the serializer accepts it without throwing.
272 		if (t.isValidationOnly())
273 			return;
274 
275 		var s = t.getSerializer();
276 		var p = t.getParser();
277 		if (p == null)
278 			return;
279 
280 		var x = BOXED_TRUE;
281 		try {
282 			var out = t.serialize(x, s);
283 			var x2 = p.read(out, Boolean.class);
284 			assertEquals(x, x2, "fmt=" + fmt);
285 		} catch (Exception e) {
286 			// Mirror UuidFormat_RoundTrip_Test.a06: some serializers (URL-encoding, CSV-style) don't
287 			// support a top-level non-bean value cleanly.  Bean-property tests above already cover the
288 			// per-property dispatch path; this case is best-effort.
289 		}
