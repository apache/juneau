30  	@Test void a01_format_finiteDouble_isBoxed() {
31  		assertEquals(Double.valueOf(3.14), FloatFormat.format(3.14, FloatFormat.NaN_AS_NULL));
32  		assertEquals(Double.valueOf(-0.5), FloatFormat.format(-0.5, FloatFormat.NaN_AS_STRING));
33  		assertEquals(Double.valueOf(0.0), FloatFormat.format(0.0, FloatFormat.NaN_AS_NUMBER));
34  		assertEquals(Double.valueOf(1.0e10), FloatFormat.format(1.0e10, FloatFormat.NaN_AS_ERROR));
35  	}
36  
37  	@Test void a02_format_double_nan_asNull() {
38  		assertNull(FloatFormat.format(Double.NaN, FloatFormat.NaN_AS_NULL));
39  		assertNull(FloatFormat.format(Double.POSITIVE_INFINITY, FloatFormat.NaN_AS_NULL));
40  		assertNull(FloatFormat.format(Double.NEGATIVE_INFINITY, FloatFormat.NaN_AS_NULL));
41  	}
42  
43  	@Test void a03_format_double_nan_asNotSet() {
44  		// NOT_SET falls through to NaN_AS_NULL.
45  		assertNull(FloatFormat.format(Double.NaN, FloatFormat.NOT_SET));
46  	}

48  	@Test void a04_format_double_nan_nullFormatTreatedAsNull() {
49  		assertNull(FloatFormat.format(Double.NaN, null));
50  	}

52  	@Test void a05_format_double_nan_asString() {
53  		assertEquals("NaN", FloatFormat.format(Double.NaN, FloatFormat.NaN_AS_STRING));
54  		assertEquals("Infinity", FloatFormat.format(Double.POSITIVE_INFINITY, FloatFormat.NaN_AS_STRING));
55  		assertEquals("-Infinity", FloatFormat.format(Double.NEGATIVE_INFINITY, FloatFormat.NaN_AS_STRING));
56  	}
57  
58  	@Test void a06_format_double_nan_asNumber() {
59  		// Non-finite preserved as boxed Double.
60  		var r = FloatFormat.format(Double.NaN, FloatFormat.NaN_AS_NUMBER);
61  		assertTrue(r instanceof Double);
62  		assertTrue(Double.isNaN((Double) r));
63  		assertEquals(Double.valueOf(Double.POSITIVE_INFINITY), FloatFormat.format(Double.POSITIVE_INFINITY, FloatFormat.NaN_AS_NUMBER));
64  	}
65  
66  	@Test void a07_format_double_nan_asErrorThrows() {
67  		assertThrows(IllegalArgumentException.class, () -> FloatFormat.format(Double.NaN, FloatFormat.NaN_AS_ERROR));
68  		assertThrows(IllegalArgumentException.class, () -> FloatFormat.format(Double.POSITIVE_INFINITY, FloatFormat.NaN_AS_ERROR));
69  		assertThrows(IllegalArgumentException.class, () -> FloatFormat.format(Double.NEGATIVE_INFINITY, FloatFormat.NaN_AS_ERROR));
70  	}

128 	@Test void c04_parse_double_infinity_caseInsensitive() {
129 		assertEquals(Double.valueOf(Double.POSITIVE_INFINITY), FloatFormat.parse("Infinity", FloatFormat.NaN_AS_NUMBER, Double.class));
130 		assertEquals(Double.valueOf(Double.POSITIVE_INFINITY), FloatFormat.parse("INFINITY", FloatFormat.NaN_AS_NUMBER, Double.class));
131 		assertEquals(Double.valueOf(Double.NEGATIVE_INFINITY), FloatFormat.parse("-Infinity", FloatFormat.NaN_AS_NUMBER, Double.class));
132 		assertEquals(Double.valueOf(Double.NEGATIVE_INFINITY), FloatFormat.parse("-INF", FloatFormat.NaN_AS_NUMBER, Double.class));
133 		assertEquals(Double.valueOf(Double.POSITIVE_INFINITY), FloatFormat.parse("+Inf", FloatFormat.NaN_AS_NUMBER, Double.class));
134 	}

139 		assertTrue(Float.isNaN(r));
140 	}

146 		}

210 	@Test void e04_parse_float_specialTokens() {
211 		// Cover parseFloat's NaN / +Inf / -Inf / +Infinity / -Infinity branches.
212 		assertTrue(Float.isNaN(FloatFormat.parse("NaN", FloatFormat.NaN_AS_NUMBER, Float.class)));
213 		assertEquals(Float.valueOf(Float.POSITIVE_INFINITY), FloatFormat.parse("Infinity", FloatFormat.NaN_AS_NUMBER, Float.class));
214 		assertEquals(Float.valueOf(Float.POSITIVE_INFINITY), FloatFormat.parse("+Infinity", FloatFormat.NaN_AS_NUMBER, Float.class));
215 		assertEquals(Float.valueOf(Float.POSITIVE_INFINITY), FloatFormat.parse("Inf", FloatFormat.NaN_AS_NUMBER, Float.class));
216 		assertEquals(Float.valueOf(Float.POSITIVE_INFINITY), FloatFormat.parse("+Inf", FloatFormat.NaN_AS_NUMBER, Float.class));
217 		assertEquals(Float.valueOf(Float.NEGATIVE_INFINITY), FloatFormat.parse("-Infinity", FloatFormat.NaN_AS_NUMBER, Float.class));
218 		assertEquals(Float.valueOf(Float.NEGATIVE_INFINITY), FloatFormat.parse("-Inf", FloatFormat.NaN_AS_NUMBER, Float.class));
