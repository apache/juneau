17  package org.apache.juneau.marshall.transforms;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.apache.juneau.marshall.*;
22  import org.junit.jupiter.api.*;
23  
24  class BooleanFormat_Test {
25  
26  	@Test void a01_format_trueFalse() {
27  		assertEquals(Boolean.TRUE, BooleanFormat.format(true, BooleanFormat.TRUE_FALSE));
28  		assertEquals(Boolean.FALSE, BooleanFormat.format(false, BooleanFormat.TRUE_FALSE));
29  	}
30  
31  	@Test void a02_format_notSetFallsThroughToTrueFalse() {
32  		assertEquals(Boolean.TRUE, BooleanFormat.format(true, BooleanFormat.NOT_SET));
33  		assertEquals(Boolean.FALSE, BooleanFormat.format(false, BooleanFormat.NOT_SET));
34  	}
35  
36  	@Test void a03_format_nullFormatTreatedAsTrueFalse() {
37  		assertEquals(Boolean.TRUE, BooleanFormat.format(true, null));
38  		assertEquals(Boolean.FALSE, BooleanFormat.format(false, null));
39  	}
40  
41  	@Test void a04_format_zeroOne() {
42  		assertEquals(Integer.valueOf(1), BooleanFormat.format(true, BooleanFormat.ZERO_ONE));
43  		assertEquals(Integer.valueOf(0), BooleanFormat.format(false, BooleanFormat.ZERO_ONE));
44  	}
45  
46  	@Test void a05_format_yesNo() {
47  		assertEquals("yes", BooleanFormat.format(true, BooleanFormat.YES_NO));
48  		assertEquals("no", BooleanFormat.format(false, BooleanFormat.YES_NO));
49  	}
50  
51  	@Test void a06_format_yN() {
52  		assertEquals("Y", BooleanFormat.format(true, BooleanFormat.Y_N));
53  		assertEquals("N", BooleanFormat.format(false, BooleanFormat.Y_N));
54  	}
55  
56  	@Test void a07_format_onOff() {
57  		assertEquals("on", BooleanFormat.format(true, BooleanFormat.ON_OFF));
58  		assertEquals("off", BooleanFormat.format(false, BooleanFormat.ON_OFF));
59  	}
60  
61  	@Test void a08_isNumeric() {
62  		assertTrue(BooleanFormat.ZERO_ONE.isNumeric());
63  		assertFalse(BooleanFormat.NOT_SET.isNumeric());
64  		assertFalse(BooleanFormat.TRUE_FALSE.isNumeric());
65  		assertFalse(BooleanFormat.YES_NO.isNumeric());
66  		assertFalse(BooleanFormat.Y_N.isNumeric());
67  		assertFalse(BooleanFormat.ON_OFF.isNumeric());
68  	}
69  
70  	@Test void b01_parse_true_allShapes() {
71  		for (var f : BooleanFormat.values()) {
72  			assertEquals(Boolean.TRUE, BooleanFormat.parse("true", f), "format=" + f);
73  			assertEquals(Boolean.TRUE, BooleanFormat.parse("TRUE", f), "format=" + f);
74  			assertEquals(Boolean.TRUE, BooleanFormat.parse("1", f), "format=" + f);
75  			assertEquals(Boolean.TRUE, BooleanFormat.parse("yes", f), "format=" + f);
76  			assertEquals(Boolean.TRUE, BooleanFormat.parse("YES", f), "format=" + f);
77  			assertEquals(Boolean.TRUE, BooleanFormat.parse("y", f), "format=" + f);
78  			assertEquals(Boolean.TRUE, BooleanFormat.parse("Y", f), "format=" + f);
79  			assertEquals(Boolean.TRUE, BooleanFormat.parse("on", f), "format=" + f);
80  			assertEquals(Boolean.TRUE, BooleanFormat.parse("ON", f), "format=" + f);
81  		}
82  	}
83  
84  	@Test void b02_parse_false_allShapes() {
85  		for (var f : BooleanFormat.values()) {
86  			assertEquals(Boolean.FALSE, BooleanFormat.parse("false", f), "format=" + f);
87  			assertEquals(Boolean.FALSE, BooleanFormat.parse("FALSE", f), "format=" + f);
88  			assertEquals(Boolean.FALSE, BooleanFormat.parse("0", f), "format=" + f);
89  			assertEquals(Boolean.FALSE, BooleanFormat.parse("no", f), "format=" + f);
90  			assertEquals(Boolean.FALSE, BooleanFormat.parse("NO", f), "format=" + f);
91  			assertEquals(Boolean.FALSE, BooleanFormat.parse("n", f), "format=" + f);
92  			assertEquals(Boolean.FALSE, BooleanFormat.parse("N", f), "format=" + f);
93  			assertEquals(Boolean.FALSE, BooleanFormat.parse("off", f), "format=" + f);
94  			assertEquals(Boolean.FALSE, BooleanFormat.parse("OFF", f), "format=" + f);
95  		}
96  	}
97  
98  	@Test void b03_parse_nullAndBlank() {
99  		for (var f : BooleanFormat.values()) {
100 			var fmt = f;
101 			assertThrows(IllegalArgumentException.class, () -> BooleanFormat.parse(null, fmt), "format=" + fmt);
102 			assertThrows(IllegalArgumentException.class, () -> BooleanFormat.parse("", fmt), "format=" + fmt);
103 			assertThrows(IllegalArgumentException.class, () -> BooleanFormat.parse("   ", fmt), "format=" + fmt);
104 		}
105 	}
106 
107 	@Test void b04_parse_nullFormatHint() {
108 		assertEquals(Boolean.TRUE, BooleanFormat.parse("yes", null));
109 		assertEquals(Boolean.FALSE, BooleanFormat.parse("0", null));
110 	}
111 
112 	@Test void b05_parse_invalidThrows() {
113 		assertThrows(IllegalArgumentException.class, () -> BooleanFormat.parse("maybe", BooleanFormat.TRUE_FALSE));
114 		assertThrows(IllegalArgumentException.class, () -> BooleanFormat.parse("2", BooleanFormat.ZERO_ONE));
115 	}
116 
117 	@Test void b06_parse_trimsWhitespace() {
118 		assertEquals(Boolean.TRUE, BooleanFormat.parse("  yes  ", BooleanFormat.YES_NO));
119 	}
120 
121 	@Test void c01_roundTrip_everyFormat() {
122 		for (var f : BooleanFormat.values()) {
123 			for (var v : new boolean[] { true, false }) {
124 				var formatted = BooleanFormat.format(v, f);
125 				var s = formatted.toString();
126 				assertEquals(Boolean.valueOf(v), BooleanFormat.parse(s, f), "format=" + f + " value=" + v);
127 			}
128 		}
129 	}
130 }
