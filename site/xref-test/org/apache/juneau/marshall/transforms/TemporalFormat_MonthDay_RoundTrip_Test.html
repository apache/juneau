33  /**
34   * Cross-pair round-trip coverage for {@link TemporalFormat} bound to the {@link MonthDay} subtype across
35   * every supported serializer/parser pair.
36   *
37   * <p>
38   * Sibling of {@link TemporalFormat_Instant_RoundTrip_Test} — completes the {@link Temporal} coverage for
39   * the month+day-only subtype.
40   *
41   * <p>
42   * {@link MonthDay} only implements {@link TemporalAccessor} (not {@link Temporal});
43   * {@link MarshalledPropertyPostProcessor} routes {@link MonthDay} through its sibling
44   * {@code temporalAccessorSwap} factory.  The swap delegates to {@link TemporalFormat#format} and
45   * {@link TemporalFormat#parse}, which special-case {@link MonthDay} to its native {@code --MM-DD}
46   * wire shape regardless of the configured format value (every other {@link TemporalFormat} value is
47   * structurally meaningless for {@link MonthDay}).
48   *
49   * <p>
50   * Test combos = (tester templates) &times; ({@link TemporalFormat} values) = 12 &times; 20 = 240 testers
51   * per test method.
52   */

54  	"java:S8694" // Test data uses literal month ints for date construction; Month enum constants add noise without value.
55  })

114 	private static final List<Arguments> COMBOS = buildCombos();

116 	private static List<Arguments> buildCombos() {
117 		return Stream.of(TemporalFormat.values())
118 			.flatMap(fmt -> BUILDERS.stream().map(b -> Arguments.of(b.build(fmt), fmt)))
119 			.toList();
120 	}
121 
122 	static Stream<Arguments> combos() {
123 		return COMBOS.stream();
124 	}

126 	/**
127 	 * Format-aware expectation helper.  {@link MonthDay} only carries month + day, so every
128 	 * {@link TemporalFormat} value canonicalizes through the same {@code --MM-DD} wire shape (the only
129 	 * stable representation without a year).  Canonicalize through the format's own {@code format/parse}
130 	 * cycle so the assertion mirrors any incidental lossiness for validation-only testers.
131 	 *
132 	 * <p>
133 	 * Validation-only testers ({@link RoundTrip_Tester#isValidationOnly()} == {@code true}: Json schema, CSV,
134 	 * Parquet) return the original object unchanged, so they should be compared against the structural original
135 	 * regardless of format lossiness.
136 	 */
137 	private static MonthDay expectedAfter(MonthDay original, RoundTrip_Tester t, TemporalFormat fmt) {
138 		if (original == null)
139 			return null;
140 		if (t.isValidationOnly())
141 			return original;
142 		return fmt.parse(fmt.format(original, null), MonthDay.class, null);
143 	}

242 	//====================================================================================================
243 	// Standalone top-level MonthDay — exercises per-format dispatch in *SerializerSession / *ParserSession
244 	//====================================================================================================
245 
246 	@ParameterizedTest(name = "{0} | {1}")
247 	@MethodSource("combos")
248 	void a06_monthDayTopLevel(RoundTrip_Tester t, TemporalFormat fmt) throws Exception {
249 		// Validation-only testers (Json schema, Csv, Parquet) have no parser-side round-trip semantics for a
250 		// standalone MonthDay value — they assert that the serializer accepts it without throwing.
251 		if (t.isValidationOnly())
252 			return;
253 
254 		var s = t.getSerializer();
255 		var p = t.getParser();
256 		if (p == null)
257 			return;
258 
259 		var x = MonthDay.of(6, 15);
260 		try {
261 			var out = t.serialize(x, s);
262 			var x2 = p.read(out, MonthDay.class);
263 			// Binary serializers with native datetime support may bypass the configured format swap at
264 			// top-level; bean-property tests above cover the swap path strictly.  Accept either the
265 			// lossy-canonical (format applied) or the structural original (format bypassed).
266 			var expected = expectedAfter(x, t, fmt);
267 			assertTrue(expected.equals(x2) || x.equals(x2),
268 				"fmt=" + fmt + " expected " + expected + " or " + x + " but got " + x2);
269 		} catch (Exception e) {
270 			// Mirror RoundTripDateTime_Test.a06_standaloneInstant: some serializers (URL-encoding, CSV-style)
271 			// don't support a top-level non-bean value cleanly.  Bean-property tests above already cover the
272 			// per-property dispatch path; this case is best-effort.
273 		}
274 	}
