26  class BigNumberFormat_Test {
27  
28  	private static final BigInteger SMALL = BigInteger.valueOf(42L);
29  	private static final BigInteger JS_MAX_SAFE = BigInteger.valueOf(9007199254740991L); // 2^53 - 1
30  	private static final BigInteger JS_MAX_SAFE_PLUS_ONE = JS_MAX_SAFE.add(BigInteger.ONE);
31  	private static final BigInteger JS_MIN_SAFE = JS_MAX_SAFE.negate();
32  	private static final BigInteger JS_MIN_SAFE_MINUS_ONE = JS_MIN_SAFE.subtract(BigInteger.ONE);
33  	private static final BigInteger HUGE = new BigInteger("123456789012345678901234567890");
34  
35  	private static final BigDecimal D_SMALL = new BigDecimal("42");
36  	private static final BigDecimal D_FRAC = new BigDecimal("3.14");
37  	private static final BigDecimal D_HUGE = new BigDecimal("123456789012345678901234567890");

43  	@Test void a01_format_int_number() {
44  		assertEquals(SMALL, BigNumberFormat.format(SMALL, BigNumberFormat.NUMBER));
45  		assertEquals(HUGE, BigNumberFormat.format(HUGE, BigNumberFormat.NUMBER));
46  	}

48  	@Test void a02_format_int_notSetFallsThrough() {
49  		assertEquals(SMALL, BigNumberFormat.format(SMALL, BigNumberFormat.NOT_SET));
50  	}
51  
52  	@Test void a03_format_int_nullFormatTreatedAsNumber() {
53  		assertEquals(SMALL, BigNumberFormat.format(SMALL, null));
54  	}
55  
56  	@Test void a04_format_int_string() {
57  		assertEquals("42", BigNumberFormat.format(SMALL, BigNumberFormat.STRING));
58  		assertEquals(HUGE.toString(), BigNumberFormat.format(HUGE, BigNumberFormat.STRING));
59  	}
60  
61  	@Test void a05_format_int_auto_safe() {
62  		assertEquals(SMALL, BigNumberFormat.format(SMALL, BigNumberFormat.AUTO));
63  	}
64  
65  	@Test void a06_format_int_auto_jsMaxSafeBoundaryInclusive() {
66  		// 2^53 - 1 is INCLUDED in safe range.
67  		assertEquals(JS_MAX_SAFE, BigNumberFormat.format(JS_MAX_SAFE, BigNumberFormat.AUTO));
68  		assertEquals(JS_MIN_SAFE, BigNumberFormat.format(JS_MIN_SAFE, BigNumberFormat.AUTO));
69  	}
70  
71  	@Test void a07_format_int_auto_jsMaxSafeBoundaryExclusive() {
72  		// 2^53 (one more than max safe) flips to STRING.
73  		assertEquals(JS_MAX_SAFE_PLUS_ONE.toString(), BigNumberFormat.format(JS_MAX_SAFE_PLUS_ONE, BigNumberFormat.AUTO));
74  		assertEquals(JS_MIN_SAFE_MINUS_ONE.toString(), BigNumberFormat.format(JS_MIN_SAFE_MINUS_ONE, BigNumberFormat.AUTO));
75  	}
76  
77  	@Test void a08_format_int_auto_huge() {
78  		assertEquals(HUGE.toString(), BigNumberFormat.format(HUGE, BigNumberFormat.AUTO));
79  	}
80  
81  	@Test void a09_format_int_null() {
82  		for (var f : BigNumberFormat.values())
83  			assertNull(BigNumberFormat.format((BigInteger) null, f), "format=" + f);
84  	}

144 		for (var hint : BigNumberFormat.values()) {
145 			assertEquals(SMALL, BigNumberFormat.parse("42", hint, BigInteger.class), "hint=" + hint);
146 			assertEquals(SMALL, BigNumberFormat.parse("\"42\"", hint, BigInteger.class), "hint=" + hint);
147 		}

203 	@Test void e01_roundTrip_int() {
204 		for (var f : BigNumberFormat.values()) {
205 			var formatted = BigNumberFormat.format(HUGE, f);
206 			var s = formatted instanceof BigInteger bi ? bi.toString() : formatted.toString();
207 			assertEquals(HUGE, BigNumberFormat.parse(s, f, BigInteger.class), "format=" + f);
208 		}
209 	}
210 
211 	@Test void e02_roundTrip_dec() {
212 		for (var f : BigNumberFormat.values()) {
213 			var formatted = BigNumberFormat.format(D_HUGE, f);
214 			var s = formatted instanceof BigDecimal bd ? bd.toPlainString() : formatted.toString();
215 			assertEquals(D_HUGE, BigNumberFormat.parse(s, f, BigDecimal.class), "format=" + f);
216 		}
