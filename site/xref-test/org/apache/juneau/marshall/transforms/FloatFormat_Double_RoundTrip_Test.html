32  /**
33   * Cross-pair round-trip coverage for {@link FloatFormat} on {@link Double} across every supported
34   * serializer/parser pair.
35   *
36   * <p>
37   * Sibling of {@link FloatFormat_Float_RoundTrip_Test} — same shape, same 12 RDF tester templates, varied
38   * across every {@link FloatFormat} value.  Round-trips representative {@link Double} values:
39   * <ul>
40   * 	<li>As a bean property (exercises the {@code MarshalledPropertyPostProcessor} context-format swap install
41   * 		path at {@code applyContextFormats}).
42   * 	<li>As a top-level value (exercises the per-format dispatch sites in each {@code *SerializerSession} /
43   * 		{@code *ParserSession}).
44   * </ul>
45   *
46   * <p>
47   * Test combos = (tester templates) &times; ({@link FloatFormat} values).  At the time of writing this comes to
48   * 12 &times; 5 = 60 testers per test method.
49   *
50   * <p>
51   * {@link FloatFormat} only controls how non-finite values ({@code NaN}, {@code ±Infinity}) are emitted on
52   * <b>text-based</b> wire formats; finite values ride the natural bare-numeric token and binary serializers
53   * (BSON / CBOR / MsgPack / Prototext / Parquet) emit native IEEE-754 regardless of this setting per the
54   * {@link FloatFormat} class-level "Binary serializers" note.  The matrix below exercises finite values across
55   * all five {@link FloatFormat} constants for the standard tests, with a dedicated non-finite test
56   * ({@link #a04_doubleProperty_nonFinite}) that skips {@link FloatFormat#NaN_AS_ERROR} (throws on swap) and
57   * canonicalises against the format-specific wire shape via {@link #expectedAfterNonFinite}.
58   *
59   * <p>
60   * Unlike the {@link Float} sibling, this file has no Prototext-skip helper — the Prototext text-format tokenizer
61   * bugs that forced those skips (a neg-flag discard on the Double return path of
62   * {@code readDecimalOrFloat}, and a {@code 0.x} mis-tokenisation in {@code lexNumber}'s `0`-prefix
63   * branch) were both fixed before this file landed. The {@code a04} non-finite
64   * test still skips Prototext in the {@link Float} sibling because a {@code mightStartNumber}
65   * {@code nan}/{@code inf} asymmetry is still open — the same predicate is carried into this file for the same
66   * reason.
67   */

138 	/** Pre-computed (tester, format) combinations.  One row per (builder &times; format) pair. */
139 	private static final List<Arguments> COMBOS = buildCombos();
140 
141 	private static List<Arguments> buildCombos() {
142 		return Stream.of(FloatFormat.values())
143 			.flatMap(fmt -> BUILDERS.stream().map(b -> Arguments.of(b.build(fmt), fmt)))
144 			.toList();
145 	}
146 
147 	static Stream<Arguments> combos() {
148 		return COMBOS.stream();
149 	}

151 	/**
152 	 * Returns <jk>true</jk> when {@code t} is wrapped around a binary serializer (any
153 	 * {@link OutputStreamSerializer} subtype).  Binary serializers emit native IEEE-754 regardless of
154 	 * {@link FloatFormat} per the class-level "Binary serializers" note, so non-finite values round-trip
155 	 * natively without going through the format dispatch.  Covers MsgPack, CBOR, BSON, Parquet, Prototext,
156 	 * and the RDF stream variants (RdfThrift / RdfProto).
157 	 */
158 	private static boolean isBinarySerializer(RoundTrip_Tester t) {
159 		return t.getSerializer() instanceof OutputStreamSerializer;
160 	}
161 
162 	/**
163 	 * Format-aware expectation helper for finite {@link Double} values.  {@link FloatFormat} affects only
164 	 * non-finite values per the {@link FloatFormat} class-level contract; finite values ride the natural
165 	 * bare-numeric wire token on text serializers and native IEEE-754 on binary serializers.  Validation-only
166 	 * testers ({@link RoundTrip_Tester#isValidationOnly()} == {@code true}: Json schema, CSV, Parquet) return
167 	 * the original object unchanged.
168 	 */
169 	@SuppressWarnings({
170 		"java:S1172" // 'fmt' is part of the shared expectedAfter(...) helper signature used across all *Format RoundTrip tests; kept for template symmetry even where this type's expected value is format-independent.
171 	})
172 	private static Double expectedAfter(Double original, RoundTrip_Tester t, FloatFormat fmt) {
173 		if (original == null)
174 			return null;
175 		if (t.isValidationOnly())
176 			return original;
177 		return original;
178 	}
179 
180 	/**
181 	 * Format-aware expectation helper for non-finite {@link Double} values ({@code NaN}, {@code ±Infinity}).
182 	 *
183 	 * <p>
184 	 * Binary serializers ({@link OutputStreamSerializer} subtypes) emit native IEEE-754 regardless of
185 	 * {@link FloatFormat} per the class-level "Binary serializers" note — the original non-finite value
186 	 * round-trips intact.
187 	 *
188 	 * <p>
189 	 * Text serializers — only {@link FloatFormat#NaN_AS_NULL} and {@link FloatFormat#NaN_AS_STRING} are
190 	 * exercised at this matrix level (per the {@link #a04_doubleProperty_nonFinite} method-doc):
191 	 * <ul>
192 	 * 	<li>{@link FloatFormat#NaN_AS_NULL} — wire emits {@code null}; round-trip resolves to {@code null}
193 	 * 		for the boxed {@link Double} property.
194 	 * 	<li>{@link FloatFormat#NaN_AS_STRING} — wire emits the quoted token; parser decodes back to the
195 	 * 		original non-finite value.
196 	 * </ul>
197 	 */
198 	private static Double expectedAfterNonFinite(Double original, RoundTrip_Tester t, FloatFormat fmt) {
199 		if (original == null)
200 			return null;
201 		if (t.isValidationOnly())
202 			return original;
203 		if (isBinarySerializer(t))
204 			return original;
205 		return switch (fmt) {
206 			case NaN_AS_NULL -> null;
207 			case NaN_AS_STRING -> original;
208 			default -> original;  // unreachable — callers skip NOT_SET / NaN_AS_NUMBER / NaN_AS_ERROR for text.
209 		};
210 	}
211 
212 	/**
213 	 * Compare two {@link Double} values using {@link Double#equals(Object)} which compares raw IEEE-754 bits.
214 	 * Critical for non-finite values: {@code Double.NaN == Double.NaN} is {@code false}, but
215 	 * {@code Double.valueOf(NaN).equals(Double.valueOf(NaN))} is {@code true} by spec.
216 	 */
217 	private static void assertDoubleEquals(Double expected, Double actual, String message) {
218 		if (expected == null) {
219 			assertNull(actual, message);
220 		} else {
221 			assertNotNull(actual, message);
222 			assertEquals(expected, actual, message + " : expected " + expected + " but was " + actual);
223 		}
224 	}

292 	//====================================================================================================
293 	// Bean with non-finite Double values — NaN / +Infinity / -Infinity edges (the FloatFormat-specific path)
294 	//
295 	// <p>
296 	// The swap install logic in {@code MarshalledPropertyPostProcessor.requiresFloatSwap} only installs
297 	// the float swap when {@code format == NaN_AS_NULL || NaN_AS_STRING || NaN_AS_ERROR}.  For // NOSONAR
298 	// {@code NOT_SET} and {@code NaN_AS_NUMBER}, no swap is installed and the round-trip rides each
299 	// serializer's native non-finite handling, which varies widely across formats (JSON rejects bare
300 	// {@code NaN}; Toml/Ini have no NaN grammar; JCS rejects per RFC 8785; XML/HTML/RDF preserve
301 	// natively).  Out-of-matrix scope: those combinations are exercised at the format-specific test
302 	// level, not via this cross-pair matrix.  {@link FloatFormat#NaN_AS_ERROR} throws during swap on
303 	// non-finite values — also skipped.  This leaves the predictable contract:
304 	// <ul>
305 	// 	<li>Binary serializers — emit / parse native IEEE-754 {@code NaN} / {@code ±Infinity} regardless
306 	// 		of {@code FloatFormat} per the {@link FloatFormat} class-level "Binary serializers" note.
307 	// 	<li>Text serializers with {@link FloatFormat#NaN_AS_NULL} — emit / parse {@code null} (lossy by
308 	// 		design — round-trips to {@code null} for boxed {@link Double}).
309 	// 	<li>Text serializers with {@link FloatFormat#NaN_AS_STRING} — emit / parse the quoted
310 	// 		{@code "NaN"} / {@code "Infinity"} / {@code "-Infinity"} token (round-trip preserving).
311 	// </ul>
312 	//====================================================================================================
313 
314 	public static class A04Bean {
315 		public Double nan;
316 		public Double posInf;
317 		public Double negInf;
318 	}
319 
320 	@ParameterizedTest(name = "{0} | {1}")
321 	@MethodSource("combos")
322 	void a04_doubleProperty_nonFinite(RoundTrip_Tester t, FloatFormat fmt) throws Exception {
323 		// NaN_AS_ERROR throws during swap on non-finite values — skip the entire assertion block.
324 		if (fmt == FloatFormat.NaN_AS_ERROR)
325 			return;
326 		// NOT_SET / NaN_AS_NUMBER on text serializers fall through to native handling — wildly varied
327 		// per format, out-of-matrix scope.  Binary serializers still preserve native NaN.
328 		var binary = isBinarySerializer(t);
329 		if (!binary && (fmt == FloatFormat.NOT_SET || fmt == FloatFormat.NaN_AS_NUMBER))
330 			return;
331 
332 		var x = new A04Bean();
333 		x.nan = Double.NaN;
334 		x.posInf = Double.POSITIVE_INFINITY;
335 		x.negInf = Double.NEGATIVE_INFINITY;
336 
337 		x = t.roundTrip(x);
338 		assertDoubleEquals(expectedAfterNonFinite(Double.NaN, t, fmt), x.nan, "fmt=" + fmt + " field=nan");
339 		assertDoubleEquals(expectedAfterNonFinite(Double.POSITIVE_INFINITY, t, fmt), x.posInf, "fmt=" + fmt + " field=posInf");
340 		assertDoubleEquals(expectedAfterNonFinite(Double.NEGATIVE_INFINITY, t, fmt), x.negInf, "fmt=" + fmt + " field=negInf");
341 	}

357 	// Standalone top-level Double — exercises per-format dispatch in *SerializerSession / *ParserSession
358 	//====================================================================================================
359 
360 	@ParameterizedTest(name = "{0} | {1}")
361 	@MethodSource("combos")
362 	void a06_doubleTopLevel(RoundTrip_Tester t, FloatFormat fmt) throws Exception {
363 		// Validation-only testers (Json schema, Csv, Parquet) have no parser-side round-trip semantics for a
364 		// standalone Double value — they assert that the serializer accepts it without throwing.
365 		if (t.isValidationOnly())
366 			return;
367 
368 		var s = t.getSerializer();
369 		var p = t.getParser();
370 		if (p == null)
371 			return;
372 
373 		var x = 3.14d;
374 		try {
375 			var out = t.serialize(x, s);
376 			var x2 = p.read(out, Double.class);
377 			assertDoubleEquals(expectedAfter(x, t, fmt), x2, "fmt=" + fmt);
378 		} catch (Exception e) {
379 			// Mirror RoundTripDateTime_Test.a06_standaloneInstant: some serializers (URL-encoding, CSV-style)
380 			// don't support a top-level non-bean value cleanly.  Bean-property tests above already cover the
381 			// per-property dispatch path; this case is best-effort.
382 		}
