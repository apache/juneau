32  /**
33   * Cross-pair round-trip coverage for {@link CurrencyFormat} across every supported serializer/parser pair.
34   *
35   * <p>
36   * Sibling of {@link TimeZoneFormat_RoundTrip_Test} — same shape, same 12 RDF tester templates, varied across
37   * every {@link CurrencyFormat} value.  Round-trips representative {@link Currency} values.
38   *
39   * <p>
40   * Test combos = (tester templates) &times; ({@link CurrencyFormat} values).  At the time of writing this
41   * comes to 12 &times; 4 = 48 testers per test method.
42   *
43   * <h5 class='topic'>Locale pinning</h5>
44   *
45   * <p>
46   * {@link CurrencyFormat#SYMBOL} and {@link CurrencyFormat#DISPLAY_NAME} are locale-sensitive — the symbol
47   * {@code "$"} resolves to USD / CAD / AUD / etc. depending on locale, and {@code "US Dollar"} similarly
48   * varies.  The default JVM locale is pinned to {@link Locale#US} in {@link #pinLocale()} so the test
49   * behaviour is reproducible across CI environments, and restored in {@link #restoreLocale()}.
50   *
51   * <p>
52   * {@link Currency} uses cached singletons via {@link Currency#getInstance(String)}, so the default
53   * {@link Object#equals} comparison works.  No equals helper needed.
54   */
55  class CurrencyFormat_RoundTrip_Test extends TestBase {

138 	/**
139 	 * Format-aware expectation helper.  {@link CurrencyFormat#SYMBOL} and {@link CurrencyFormat#DISPLAY_NAME}
140 	 * are locale-sensitive and only round-trip cleanly when the parser can disambiguate the wire token
141 	 * against the pinned {@link Locale#US} default — see {@link #pinLocale()}.  The
142 	 * {@link CurrencyFormat#parse} API tries {@link Currency#getInstance(String)} first regardless of the
143 	 * configured format, so {@code ISO_CODE} wires always round-trip and {@code SYMBOL} / {@code DISPLAY_NAME}
144 	 * wires round-trip when the token is uniquely resolvable.
145 	 *
146 	 * <p>
147 	 * Validation-only testers ({@link RoundTrip_Tester#isValidationOnly()} == {@code true}: Json schema, CSV,
148 	 * Parquet) return the original object unchanged, so they should be compared against the structural
149 	 * original regardless of format lossiness.
150 	 */
151 	private static Currency expectedAfter(Currency original, RoundTrip_Tester t, CurrencyFormat fmt) {
152 		if (original == null)
153 			return null;
154 		if (t.isValidationOnly())
155 			return original;
156 		// Canonicalize through the format's own format / parse cycle.  When the parse step throws (ambiguous
157 		// symbol or unresolvable display name), fall back to the original for the assertion to fail cleanly
158 		// in the round-trip test rather than blow up here.
159 		var effective = fmt == null ? CurrencyFormat.ISO_CODE : fmt;
160 		try {
161 			return CurrencyFormat.parse(CurrencyFormat.format(original, effective, Locale.US), effective, Locale.US);
162 		} catch (@SuppressWarnings("unused") IllegalArgumentException ignored) {
163 			return original;
164 		}
165 	}

246 	}

275 	//====================================================================================================
276 	// Standalone top-level Currency — exercises per-format dispatch in *SerializerSession / *ParserSession
277 	//====================================================================================================
278 
279 	@ParameterizedTest(name = "{0} | {1}")
280 	@MethodSource("combos")
281 	void a06_currencyTopLevel(RoundTrip_Tester t, CurrencyFormat fmt) throws Exception {
282 		// Validation-only testers (Json schema, Csv, Parquet) have no parser-side round-trip semantics for a
283 		// standalone Currency value — they assert that the serializer accepts it without throwing.
284 		if (t.isValidationOnly())
285 			return;
286 
287 		var s = t.getSerializer();
288 		var p = t.getParser();
289 		if (p == null)
290 			return;
291 
292 		var x = Currency.getInstance("USD");
293 		try {
294 			var out = t.serialize(x, s);
295 			var x2 = p.read(out, Currency.class);
296 			// Accept either the lossy-canonical (format applied) or the structural original (format bypassed
297 			// by native type handling — matches the BSON-style divergence already documented for DateFormat /
298 			// CalendarFormat at top-level).
299 			var expected = expectedAfter(x, t, fmt);
300 			assertTrue(expected.equals(x2) || x.equals(x2),
301 				"fmt=" + fmt + " expected " + expected + " or " + x + " but got " + x2);
302 		} catch (Exception e) {
303 			// Mirror RoundTripDateTime_Test.a06_standaloneInstant: some serializers (URL-encoding, CSV-style)
304 			// don't support a top-level non-bean value cleanly.  Bean-property tests above already cover the
305 			// per-property dispatch path; this case is best-effort.
306 		}
307 	}
