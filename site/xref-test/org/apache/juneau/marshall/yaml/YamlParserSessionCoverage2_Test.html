17  package org.apache.juneau.marshall.yaml;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import org.apache.juneau.*;
22  import org.apache.juneau.marshall.collections.*;
23  import org.junit.jupiter.api.*;
24  
25  /**
26   * Additional coverage-focused tests for {@link YamlParserSession}, targeting branches not already exercised
27   * by {@link YamlParserSession_Test}:
28   *  - {@code readAnything}'s {@code isArray()||isArgs()} arms for both flow-sequence ('[') and block-sequence
29   *    ('-') dispatch, reached only via {@link YamlParser#readArgs}.
30   *  - {@code isWhitespace}'s tab branch inside the flow-mapping state machine.
31   *  - {@code skipToEndOfLine}'s bare-CR line terminator.
32   *  - {@code readIntoBeanMapFlow}'s bean-type-property-name skip (a {@code _type} key inside {@code {...}}).
33   */
34  class YamlParserSessionCoverage2_Test extends TestBase {
35  
36  	//------------------------------------------------------------------------------------------------------------------
37  	// a0x - readArgs(): isArgs() arms of readAnything's '[' and '-' dispatch, and the corresponding
38  	// type.getArg(argIndex++) ternary inside readFlowSequence/readBlockSequence.
39  	//------------------------------------------------------------------------------------------------------------------
40  
41  	@Test void a01_readArgs_flowSequence() throws Exception {
42  		// Top-level type IS the ARGS ClassMeta itself; '[' input drives readAnything's isArray()||isArgs()
43  		// arm (isArgs() true) which delegates to readFlowSequence, whose own type.isArgs() ternary picks
44  		// type.getArg(argIndex++) for each element in turn.
45  		var session = YamlParser.DEFAULT.getSession();
46  		var args = session.readArgs("[1, hello]", new java.lang.reflect.Type[] { int.class, String.class });
47  		assertNotNull(args);
48  		assertEquals(2, args.length);
49  		assertEquals(1, args[0]);
50  		assertEquals("hello", args[1]);
51  	}
52  
53  	@Test void a02_readArgs_blockSequence() throws Exception {
54  		// Same as a01 but via block-sequence ('-') syntax, exercising readAnything's '-' dispatch isArgs()
55  		// arm and readBlockSequence's own type.getArg(argIndex++) ternary.
56  		var session = YamlParser.DEFAULT.getSession();
57  		var args = session.readArgs("- 1\n- hello", new java.lang.reflect.Type[] { int.class, String.class });
58  		assertNotNull(args);
59  		assertEquals(2, args.length);
60  		assertEquals(1, args[0]);
61  		assertEquals("hello", args[1]);
62  	}
63  
64  	//------------------------------------------------------------------------------------------------------------------
65  	// b0x - isWhitespace: tab character inside the flow-mapping state machine (S4's "looking for value" state).
66  	//------------------------------------------------------------------------------------------------------------------
67  
68  	@Test void b01_flowMapping_tabBeforeValue() throws Exception {
69  		var m = YamlParser.DEFAULT.read("{a:\t1}", JsonMap.class);
70  		assertEquals("1", m.getString("a"));
71  	}
72  
73  	//------------------------------------------------------------------------------------------------------------------
74  	// c0x - skipToEndOfLine: a comment line terminated by a bare '\r' (old-Mac style), not '\n'.
75  	//------------------------------------------------------------------------------------------------------------------
76  
77  	@Test void c01_commentLine_bareCarriageReturnTerminator() throws Exception {
78  		var m = YamlParser.DEFAULT.read("a: 1\r# comment\rb: 2", JsonMap.class);
79  		assertEquals("1", m.getString("a"));
80  		assertEquals("2", m.getString("b"));
81  	}
82  
83  	//------------------------------------------------------------------------------------------------------------------
84  	// d0x - readIntoBeanMapFlow: a "_type" key inside a '{...}' bean mapping is skipped (not treated as an
85  	// unknown/settable property), exercising the currAttr.equals(getBeanTypePropertyName(...)) true branch.
86  	//------------------------------------------------------------------------------------------------------------------
87  
88  	public static class D_Bean {
89  		public String name;
90  	}
91  
92  	@Test void d01_flowBeanMapping_typePropertySkipped() throws Exception {
93  		var b = YamlParser.DEFAULT.read("{_type: 'org.apache.juneau.marshall.yaml.YamlParserSessionCoverage2_Test$D_Bean', name: Bob}", D_Bean.class);
94  		assertNotNull(b);
95  		assertEquals("Bob", b.name);
96  	}
97  
98  	//------------------------------------------------------------------------------------------------------------------
99  	// e0x - readIntoBeanMapBlock: unknown property with a non-null value (isNullBlockValue false), distinct
100 	// from the null-value variants already covered in YamlParserSession_Test's d0x group.
101 	//------------------------------------------------------------------------------------------------------------------
102 
103 	@Test void e01_blockBeanMapping_unknownPropertyWithValue() throws Exception {
104 		var p = YamlParser.create().ignoreUnknownBeanProperties().build();
105 		var b = p.read("name: Bob\nbogus: 1", D_Bean.class);
106 		assertNotNull(b);
107 		assertEquals("Bob", b.name);
108 	}
109 }
