33  /**
34   * Coverage-focused tests for {@link YamlSerializerSession} targeting branches not already exercised by
35   * {@link Yaml_Test} and friends:
36   *  - {@code getYamlWriter}'s reuse arm (output already a {@link YamlWriter}).
37   *  - {@code writeAnything}'s recursion-detected arm (an already-in-progress element re-encountered).
38   *  - {@code writeAnything}'s swap-to-Object rewrap arm.
39   *  - {@code writeAnything}'s direct-{@link BeanMap} dispatch (isBeanMap() as opposed to a plain Map).
40   *  - {@code writeAnything}'s {@code isUri()}/{@code pMeta.isUri()} disjuncts.
41   *  - {@code writeAnything}'s {@code Reader}/{@code InputStream} scalar arms (including the pipe-failure path).
42   *  - {@code isComplexValue}'s {@code null} and swap-present arms.
43   *  - {@code writeBeanMap}/{@code writeMap}'s complex-vs-simple value dispatch for both collection/array and
44   *    plain-complex (bean/map) values.
45   *  - {@code writeBeanMap}'s getter-exception handling.
46   */
47  @SuppressWarnings({
48  	"resource" // Reader/InputStream test fixtures wrap in-memory buffers with no OS resources to close.
49  })

69  	//------------------------------------------------------------------------------------------------------------------
70  
71  	@Test void b01_selfReferencingListElement_writesNull() throws Exception {
72  		// detectRecursions()+ignoreRecursions() is required to reach the *silent-null* recursion arm here --
73  		// without it, push() only truncates on exceeding maxDepth (a size guard, not a cycle detector), which
74  		// would produce deeply-nested output instead.
75  		var list = new ArrayList<Object>();
76  		list.add("x");
77  		list.add(list);
78  		var s = YamlSerializer.create().detectRecursions().ignoreRecursions().build();
79  		var result = s.write(list);
80  		assertTrue(result.contains("- x"), "Expected '- x' in: " + result);
81  		assertTrue(result.contains("null"), "Expected 'null' in: " + result);
82  	}

96  	@Test void c01_swapToObjectTypeRewrapsToRuntimeType() throws Exception {
97  		// Bean *property* values whose declared type has a registered swap are pre-swapped upstream, so the
98  		// swap must instead sit on a raw List *element* to reach writeAnything's own swap resolution
99  		// (mirrors the analogous finding for CborSerializerSession).
100 		var s = YamlSerializer.create().swaps(C01_ObjSwap.class).build();
101 		var result = s.write(List.of(new C01_ObjSwapTarget()));
102 		assertTrue(result.contains("swapped-1"), "Expected 'swapped-1' in: " + result);
103 	}

174 	// javac emits a synthetic bridge lambda for the `SerializerSession::handleThrown` method reference (its
175 	// generic <T extends Throwable> signature needs an IOException-typed shim to satisfy Consumer<IOException>).
176 	// That bridge only executes when pipe() itself catches an I/O error, so a failing Reader/InputStream is
177 	// needed to reach it (a happy-path pipe, as in f01/f02 above, never touches it).
178 	public static class F03_Bean {
179 		public Reader r;
180 	}
181 
182 	@Test void f03_readerThrowsDuringPipe_handledViaHandleThrown() {
183 		var a = new F03_Bean();
184 		a.r = new Reader() {
185 			@Override public int read(char[] cbuf, int off, int len) throws IOException { throw new IOException("boom"); }
186 			@Override public void close() { /* no-op */ }
187 		};
188 		assertThrowsWithMessage(SerializeException.class, "boom", () -> YamlSerializer.DEFAULT.write(a));
189 	}

207 	// A @BeanProp("*") Map field exposes per-key "dyna" properties with no static declared type
208 	// (BeanPropertyMeta.getBeanInfo() is null for them), which is the only way to reach
209 	// isComplexValue(null) -- writeBeanMap's non-dyna path always resolves a non-null actualType.
210 	public static class G01_Bean {
211 		@BeanProp("*")
212 		public Map<String,Object> extra = new LinkedHashMap<>(Map.of("dynaKey", "dynaValue"));
213 	}

216 		var result = YamlSerializer.DEFAULT.write(new G01_Bean());
217 		assertTrue(result.contains("dynaKey"), "Expected 'dynaKey' in: " + result);
218 	}

220 	// isComplexValue first checks the *unswapped* actualType's own isMapOrBean()/isCollectionOrArrayOrOptional()
221 	// (an earlier disjunct on a different line) before ever consulting the registered swap -- so the swap source
222 	// type must itself be neither bean- nor collection-shaped, or that first check short-circuits before the
223 	// swap-classification logic below is ever reached. A package-private field (no public field/getter) keeps
224 	// G02_SwapSource from being auto-detected as a bean.
225 	public static class G02_SwapSource {
226 		int x = 1;
227 	}
228 
229 	public static class G02_MapSwap extends ObjectSwap<G02_SwapSource,Map<String,Object>> {
230 		@Override public Map<String,Object> swap(MarshallingSession session, G02_SwapSource o) { return Map.of("x", o.x); }
231 	}
232 
233 	public static class G02_Bean {
234 		public G02_SwapSource inner = new G02_SwapSource();
235 	}
236 
237 	@Test void g02_beanPropertySwapToMap_hitsIsComplexValueSwapPresentArm() throws Exception {
238 		var s = YamlSerializer.create().swaps(G02_MapSwap.class).build();
239 		var result = s.write(new G02_Bean());
240 		assertTrue(result.contains("inner"), "Expected 'inner' in: " + result);
241 	}
