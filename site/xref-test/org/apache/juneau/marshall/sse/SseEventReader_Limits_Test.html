17  package org.apache.juneau.marshall.sse;
18  
19  import static org.junit.jupiter.api.Assertions.*;
20  
21  import java.io.*;
22  
23  import org.apache.juneau.*;
24  import org.junit.jupiter.api.*;
25  
26  /**
27   * Tests for the size limits of {@link SseEventReader}.
28   */
29  class SseEventReader_Limits_Test extends TestBase {
30  
31  	@Test void a01_overlongLineFails() {
32  		var r = new SseEventReader(new StringReader("data: " + "x".repeat(100) + "\n\n"), 50, 1000);
33  		assertThrows(UncheckedIOException.class, r::hasNext);
34  	}
35  
36  	@Test void a02_unterminatedLineFails() {
37  		var r = new SseEventReader(new StringReader("x".repeat(100)), 50, 1000);
38  		assertThrows(UncheckedIOException.class, r::hasNext);
39  	}
40  
41  	@Test void a03_oversizedEventDataFails() {
42  		var sb = new StringBuilder();
43  		for (var i = 0; i < 20; i++)
44  			sb.append("data: 0123456789\n");
45  		var r = new SseEventReader(new StringReader(sb.toString()), 100, 100);
46  		assertThrows(UncheckedIOException.class, r::hasNext);
47  	}
48  
49  	@Test void a04_toListSurfacesIOException() {
50  		var r = new SseEventReader(new StringReader("data: " + "x".repeat(100) + "\n\n"), 50, 1000);
51  		assertThrows(IOException.class, r::toList);
52  	}
53  
54  	@Test void a05_withinLimitsParses() throws IOException {
55  		var r = new SseEventReader(new StringReader("data: abc\n\ndata: def\n\n"), 50, 50);
56  		var l = r.toList();
57  		assertEquals(2, l.size());
58  		assertEquals("abc", l.get(0).getData());
59  	}
60  
61  	@Test void a06_limitsApplyPerEventNotCumulatively() throws IOException {
62  		var r = new SseEventReader(new StringReader("data: 0123456789\n\ndata: 0123456789\n\ndata: 0123456789\n\n"), 50, 20);
63  		assertEquals(3, r.toList().size());
64  	}
65  
66  	@Test void a07_nonPositiveLimitsRejected() {
67  		var r1 = new StringReader("");
68  		assertThrows(IllegalArgumentException.class, () -> new SseEventReader(r1, 0, 10));
69  		var r2 = new StringReader("");
70  		assertThrows(IllegalArgumentException.class, () -> new SseEventReader(r2, 10, 0));
71  	}
72  }
