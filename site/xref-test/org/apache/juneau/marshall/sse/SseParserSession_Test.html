17  package org.apache.juneau.marshall.sse;
18  
19  import static org.apache.juneau.test.bct.BctAssertions.*;
20  import static org.junit.jupiter.api.Assertions.*;
21  
22  import java.util.*;
23  
24  import org.apache.juneau.marshall.parser.*;
25  import org.junit.jupiter.api.*;
26  
27  /**
28   * Coverage-targeted tests for {@link SseParserSession#doRead}'s target-type dispatch, which is not
29   * otherwise exercised from this module (the broader SSE round-trip suite lives in
30   * juneau-integration-tests).
31   */
32  @SuppressWarnings({
33  	"unchecked" // Parser returns Object; casts to List<SseEvent> below.
34  })
35  class SseParserSession_Test {
36  
37  	private static final String WIRE = "event: progress\ndata: step 1\nid: 42\nretry: 1000\n\n";
38  
39  	@Test void a02_objectType_returnsListOfEvents() throws Exception {
40  		Object v = SseParser.DEFAULT.read(WIRE, Object.class);
41  		assertTrue(v instanceof List, "Expected List, got: " + v);
42  	}
43  
44  	@Test void a03_arrayType_returnsArrayOfEvents() throws Exception {
45  		var v = SseParser.DEFAULT.read(WIRE, SseEvent[].class);
46  		assertEquals(1, v.length);
47  		assertBean(v[0], "event,data,id,retry", "progress,step 1,42,1000");
48  	}
49  
50  	@Test void a04_singleSseEventType_returnsFirstEvent() throws Exception {
51  		var v = SseParser.DEFAULT.read(WIRE, SseEvent.class);
52  		assertBean(v, "event,data,id,retry", "progress,step 1,42,1000");
53  	}
54  
55  	@Test void a05_singleSseEventType_noEvents_returnsNull() throws Exception {
56  		var v = SseParser.DEFAULT.read("", SseEvent.class);
57  		assertNull(v);
58  	}
59  
60  	@Test void a06_listType_returnsListOfEvents() throws Exception {
61  		// List is itself a Collection subtype, so this actually takes doRead's earlier
62  		// "type.isCollectionOrArray()" branch (not the dedicated "List.class.isAssignableFrom(...)"
63  		// check further down, which is unreachable dead code -- see the HTT marker at that line).
64  		List<SseEvent> v = (List<SseEvent>) SseParser.DEFAULT.read(WIRE, List.class, SseEvent.class);
65  		assertEquals(1, v.size());
66  		assertBean(v.get(0), "event,data,id,retry", "progress,step 1,42,1000");
67  	}
68  
69  	@Test void a07_unsupportedTargetType_throwsParseException() {
70  		assertThrows(ParseException.class, () -> SseParser.DEFAULT.read(WIRE, String.class));
71  	}
72  
73  	@Test void a08_isRecordStreaming_false() {
74  		var session = SseParser.DEFAULT.createSession().build();
75  		assertFalse(session.isRecordStreaming());
76  	}
77  
78  	@Test void a09_readRecords_delegatesToArrayReader() throws Exception {
79  		var session = SseParser.DEFAULT.createSession().build();
80  		try (var records = session.readRecords(WIRE)) {
81  			assertNotNull(records);
82  		}
83  	}
84  }
