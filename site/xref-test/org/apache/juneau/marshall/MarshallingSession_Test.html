84  	@Test void b02_defaultNamerUsesCachedBeanMeta() {
85  		// Passing the session's own default namer must reuse the cached BeanMeta (same instance as no-namer path).
86  		var m = bs.toBeanMap(new A(), bs.getPropertyNamer());
87  		var cached = bc.getClassMeta(A.class).getBeanMeta();
88  		assertSame(cached, m.getMeta());
89  		assertList(m.keySet(), "fooBar", "qux");
90  	}

92  	@Test void b03_overrideNamerDoesNotPolluteCache() {
93  		// Building an alternate-namer BeanMap must not change the shared cached BeanMeta's property names.
94  		var before = bc.getClassMeta(A.class).getBeanMeta();
95  		bs.toBeanMap(new A(), PropertyNamerDLC.INSTANCE);
96  		var after = bc.getClassMeta(A.class).getBeanMeta();
97  		assertSame(before, after);
98  		assertList(after.getProperties().keySet(), "fooBar", "qux");
99  	}

159 	// newBean(Object,Class) wraps a non-BeanRuntimeException constructor failure.
160 	//====================================================================================================
161 
162 	public static class C08_ThrowingBean {
163 		public C08_ThrowingBean() {
164 			throw new IllegalStateException("boom");
165 		}
166 		public String getFoo() { return "foo"; }
167 		public void setFoo(@SuppressWarnings("unused") String v) { /* no-op */ }
168 	}
169 
170 	@Test void c08_newBeanConstructorExceptionWrappedAsBeanRuntimeException() {
171 		// The constructor's IllegalStateException is caught and wrapped as ExecutableException by
172 		// BeanMeta.newBean(), which is in turn caught by the generic `catch (Exception e)` arm here
173 		// (since it's not itself a BeanRuntimeException) and re-wrapped as a BeanRuntimeException.
174 		var ex = assertThrows(BeanRuntimeException.class, () -> bs.newBean(C08_ThrowingBean.class));
175 		assertTrue(ex.getCause() instanceof ExecutableException, () -> "Expected ExecutableException cause but got: " + ex.getCause());
176 		assertTrue(ex.getCause().getCause() instanceof IllegalStateException);
177 	}

190 	//====================================================================================================
191 	// toArray(ClassMeta,Collection): already-converted-array passthrough, nested List-of-List recursion,
192 	// and null-element-into-primitive-default.
193 	//====================================================================================================
194 
195 	@Test void c10_toArrayElementAlreadyMatchingFullArrayTypeSkipsConversion() {
196 		// `type.isInstance(x)` true-arm: because array types are covariant, an Object[] element that is
197 		// itself some other reference-array subtype (e.g. String[]) already satisfies `Object[].class
198 		// .isInstance(x)` -- note this checks against `type` (the full array type being built), NOT
199 		// `componentType`, so the shortcut only fires in this array-covariance corner case. A sibling
200 		// non-array element takes the ordinary `convertToType` arm below.
201 		var result = (Object[]) bs.toArray(bc.getClassMeta(Object[].class), List.of(new String[]{"a", "b"}, "plain"));
202 		assertArrayEquals(new String[]{"a", "b"}, (String[]) result[0]);
203 		assertEquals("plain", result[1]);
204 	}
205 
206 	@Test void c10b_toArrayNestedCollectionElementIntoArrayComponentRecurses() {
207 		// `componentType.isArray() && x instanceof Collection<?>` true-arm: a raw List<Integer> element
208 		// targeting an int[][] recurses into a nested toArray() call to build the inner int[] rather than
209 		// going through the ordinary convertToType() arm.
210 		var result = (int[][]) bs.toArray(bc.getClassMeta(int[][].class), List.of(List.of(1, 2), List.of(3, 4)));
211 		assertArrayEquals(new int[]{1, 2}, result[0]);
212 		assertArrayEquals(new int[]{3, 4}, result[1]);
213 	}
214 
215 	@Test void c11_toArrayNullElementIntoPrimitiveComponentUsesDefault() {
216 		// `x == null && componentType.isPrimitive()` true-arm: a null list element targeting a
217 		// primitive int[] component type is coerced to the primitive default (0) rather than
218 		// attempting (and failing) a null-to-int convertToType() call.
219 		var result = (int[]) bs.toArray(bc.getClassMeta(int[].class), Arrays.asList(1, null, 3));
220 		assertArrayEquals(new int[]{1, 0, 3}, result);
221 	}
