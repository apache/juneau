54  			// Bytes 0x93 0x01 0x02 0x03 encode a fixarray of length three holding the values one, two and three.
55  			var bytes = new byte[]{(byte) 0x93, 0x01, 0x02, 0x03};
56  			try (var r = MsgPackParser.DEFAULT.readTokens(bytes)) {
57  				assertEquals(TokenType.START_ARRAY, r.next());
58  				assertEquals(TokenType.VALUE_NUMBER, r.next()); assertEquals(1L, r.getNumber().longValue());
59  				assertEquals(TokenType.VALUE_NUMBER, r.next()); assertEquals(2L, r.getNumber().longValue());
60  				assertEquals(TokenType.VALUE_NUMBER, r.next()); assertEquals(3L, r.getNumber().longValue());
61  				assertEquals(TokenType.END_ARRAY, r.next());
62  			}
63  		}

147 		@Test void b07_writeAfterCloseThrows() throws Exception {
148 			var bos = new ByteArrayOutputStream();
149 			var w = MsgPackSerializer.DEFAULT.writeTokens(bos);
150 			w.startArray().endArray();
151 			w.close();
152 			// Every mutating method must reject writes after close() with the closed-writer message.
153 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", w::startObject);
154 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", w::startArray);
155 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", w::endObject);
156 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", w::endArray);
157 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", () -> w.fieldName("a"));
158 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", () -> w.string("x"));
159 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", () -> w.number(1));
160 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", () -> w.bool(true));
161 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", w::nil);
162 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", () -> w.binary(new byte[]{1}));
163 			assertThrowsWithMessage(IOException.class, "Token writer is closed.", () -> w.object(42));
164 			// close() stays idempotent and flush() after close stays a safe no-op.
165 			assertDoesNotThrow(w::close);
166 			assertDoesNotThrow(w::flush);
167 		}

204 		public static class Bean {
205 			public String name;
206 			public int age;
207 		}
208 
209 		@Test void d01_readBean() throws Exception {
210 			var b = new Bean();
211 			b.name = "alice";
212 			b.age = 30;
213 			var bytes = MsgPackSerializer.DEFAULT.write(b);
214 
215 			try (var r = MsgPackParser.DEFAULT.readTokens((byte[]) bytes)) {
216 				var got = r.read(Bean.class);
217 				assertEquals("alice", got.name);
218 				assertEquals(30, got.age);
219 			}
220 		}
221 
222 		@Test void d02_objectMatchesCanonicalSerializer() throws Exception {
223 			var b = new Bean();
224 			b.name = "alice";
225 			b.age = 30;
226 
227 			var bos = new ByteArrayOutputStream();
228 			try (var w = MsgPackSerializer.DEFAULT.writeTokens(bos)) {
229 				w.object(b);
230 			}
231 
232 			// Round-trip via MsgPackParser: the cursor walks the bean alphabetically by BeanMap, matching the
233 			// canonical MsgPackSerializer's own alphabetical ordering.  Compare the parsed Bean.
234 			var fromWalker = MsgPackParser.DEFAULT.read(bos.toByteArray(), Bean.class);
235 			assertEquals("alice", fromWalker.name);
236 			assertEquals(30, fromWalker.age);
237 		}
238 
239 		@Test void d03_streamArrayOfBeans() throws Exception {
240 			var b1 = new Bean(); b1.name = "alice"; b1.age = 30;
241 			var b2 = new Bean(); b2.name = "bob";   b2.age = 40;
242 			var list = java.util.List.of(b1, b2);
243 			var bytes = MsgPackSerializer.DEFAULT.write(list);
244 
245 			var seen = new java.util.ArrayList<Bean>();
246 			try (var r = MsgPackParser.DEFAULT.readTokens((byte[]) bytes)) {
247 				assertEquals(TokenType.START_ARRAY, r.next());
248 				while (r.canRead())
249 					seen.add(r.read(Bean.class));
250 				assertEquals(TokenType.END_ARRAY, r.next());
251 			}
252 			assertEquals(2, seen.size());
253 			assertEquals("alice", seen.get(0).name);
254 			assertEquals("bob", seen.get(1).name);
255 		}
