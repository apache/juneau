35  /**
36   * Regression tests for the Phase-2 BSON conformance fixes (175fe).
37   *
38   * <p>
39   * Each category guards one audited gap:
40   * <ul>
41   * 	<li><b>a</b> — H1: default {@code writeDatesAsDatetime=true} must not crash on local {@code java.time} types.
42   * 	<li><b>b</b> — H2: wire-declared lengths (string/binary/document) are bounds-checked against a configurable max.
43   * 	<li><b>c</b> — M1: inbound Decimal128 NaN/Infinity normalizes to {@code null} instead of crashing.
44   * 	<li><b>d</b> — M2: out-of-range {@link BigDecimal}/{@link BigInteger} fail cleanly with a {@link SerializeException}.
45   * 	<li><b>e</b> — M3: the configured bean type-property name is honored on parse (polymorphism).
46   * 	<li><b>f</b> — Guards confirming native fidelity ({@link BsonDecimal128}, int64/datetime-millis) is intact.
47   * </ul>
48   */

56  	// Helpers (byte-level construction mirrors BsonInputStream_Test)
57  	// ================================================================
58  
59  	private static BsonInputStream openIs(byte[] bytes) throws IOException {
60  		return new BsonInputStream(new ParserPipe(new ByteArrayInputStream(bytes)));
61  	}
62  
63  	private static byte[] le4(int v) {
64  		return new byte[]{(byte) v, (byte) (v >> 8), (byte) (v >> 16), (byte) (v >> 24)};
65  	}
66  
67  	private static byte[] le8(long v) {
68  		return new byte[]{(byte) v, (byte) (v >> 8), (byte) (v >> 16), (byte) (v >> 24),
69  			(byte) (v >> 32), (byte) (v >> 40), (byte) (v >> 48), (byte) (v >> 56)};
70  	}
71  
72  	private static byte[] cat(byte[]... arrs) {
73  		var total = 0;
74  		for (var a : arrs)
75  			total += a.length;
76  		var out = new byte[total];
77  		var pos = 0;
78  		for (var a : arrs) {
79  			System.arraycopy(a, 0, out, pos, a.length);
80  			pos += a.length;
81  		}
82  		return out;
83  	}
84  
85  	private static byte[] cstring(String s) {
86  		var b = s.getBytes(StandardCharsets.UTF_8);
87  		return cat(b, new byte[]{0x00});
88  	}
89  
90  	/** Builds the 16-byte little-endian Decimal128 wire form (low 64 bits, then high 64 bits). */
91  	private static byte[] decimal128(long high, long low) {
92  		return cat(le8(low), le8(high));
93  	}
94  
95  	/** Builds a single-element BSON document {@code {name: <value>}} of the given element type. */
96  	private static byte[] doc(int type, String name, byte[] value) {
97  		var body = cat(new byte[]{(byte) type}, cstring(name), value, new byte[]{0x00});
98  		return cat(le4(body.length + 4), body);
99  	}

107 		// Was: DateTimeException (Instant.from(LocalDate)) under default writeDatesAsDatetime=true.
108 		var bytes = BsonSerializer.DEFAULT.write(JsonMap.of("d", LocalDate.of(2020, Month.JANUARY, 1)));

175 	@Test
176 	void b05_configurableMaxLengthEnforcedEndToEnd() throws Exception {
177 		// A document with a 32-byte binary parses with a generous cap but fails with a tiny cap.
178 		var bytes = BsonSerializer.DEFAULT.write(JsonMap.of("data", new byte[32]));
179 		var lenient = BsonParser.create().maxLength(1024).build();
180 		var parsed = (Map<String,Object>) lenient.read(bytes, Map.class, String.class, Object.class);
181 		assertEquals(32, ((byte[]) parsed.get("data")).length);
182 
183 		var strict = BsonParser.create().maxLength(8).build();
184 		assertThrowsWithMessage(IOException.class, "exceeds maximum", () -> strict.read(bytes, Map.class, String.class, Object.class));
185 	}

187 	@Test
188 	void b06_maxLengthAffectsCacheKey() {
189 		// Different maxLength values must NOT collide in the parser cache (hashKey wiring).
190 		var p1 = BsonParser.create().maxLength(100).build();
191 		var p2 = BsonParser.create().maxLength(200).build();
192 		var p3 = BsonParser.create().maxLength(100).build();
193 		assertNotSame(p1, p2);
194 		assertSame(p1, p3);
195 		assertEquals(100, p1.getMaxLength());
196 		assertEquals(200, p2.getMaxLength());
197 	}

270 	@Marshalled(typeName="e_dog")
271 	public static class E_Dog {
272 		public String name;
273 	}
274 
275 	@Test
276 	void e01_customTypePropertyNameRoundTrips() throws Exception {
277 		var s = BsonSerializer.create().addBeanTypes().addRootType().beanDictionary(E_Dog.class).typePropertyName("_t").keepNullProperties().build();
278 		var p = BsonParser.create().beanDictionary(E_Dog.class).typePropertyName("_t").build();
279 		var a = new E_Dog();
280 		a.name = "Rex";
281 		var bytes = s.write(a);
282 		var parsed = p.read(bytes, Object.class);
283 		assertInstanceOf(E_Dog.class, parsed);
284 		assertEquals("Rex", ((E_Dog) parsed).name);
285 	}
286 
287 	@Test
288 	void e02_defaultTypePropertyNameStillRoundTrips() throws Exception {
289 		// Guard: the default discriminator name ("_type") still resolves polymorphic types.
290 		var s = BsonSerializer.create().addBeanTypes().addRootType().beanDictionary(E_Dog.class).keepNullProperties().build();
291 		var p = BsonParser.create().beanDictionary(E_Dog.class).build();
292 		var a = new E_Dog();
