46  	private static byte[] le4(int v) {
47  		return new byte[]{(byte) v, (byte) (v >> 8), (byte) (v >> 16), (byte) (v >> 24)};
48  	}
49  
50  	private static byte[] cat(byte[]... arrs) {
51  		var total = 0;
52  		for (var a : arrs)
53  			total += a.length;
54  		var out = new byte[total];
55  		var pos = 0;
56  		for (var a : arrs) {
57  			System.arraycopy(a, 0, out, pos, a.length);
58  			pos += a.length;
59  		}
60  		return out;
61  	}
62  
63  	private static byte[] cstring(String s) {
64  		var b = s.getBytes(StandardCharsets.UTF_8);
65  		return cat(b, new byte[]{0x00});
66  	}
67  
68  	/** Builds a single-element BSON document {@code {name: <value>}} of the given element type. */
69  	private static byte[] doc(int type, String name, byte[] value) {
70  		var body = cat(new byte[]{(byte) type}, cstring(name), value, new byte[]{0x00});
71  		return cat(le4(body.length + 4), body);
72  	}
