17  package org.apache.juneau.rest.client;
18  
19  import java.util.*;
20  import java.util.logging.*;
21  
22  import org.apache.juneau.commons.logging.LogRecordContext;
23  import org.apache.juneau.commons.logging.RichLogger;
24  
25  final class RestClientDebugPipeline {
26  
27  	private RestClientDebugPipeline() {}
28  
29  	static Level resolveTier(RichLogger logger) {
30  		if (logger.isLoggable(Level.FINEST))
31  			return Level.FINEST;
32  		if (logger.isLoggable(Level.FINE))
33  			return Level.FINE;
34  		if (logger.isLoggable(Level.INFO))
35  			return Level.INFO;
36  		return null;
37  	}
38  
39  	static void emit(RichLogger logger, RestClientDebugFormatter formatter, Level level, RestRequest req, RestResponse res, Throwable thrown) {
40  		if (level == null)
41  			return;
42  
43  		var sb = new StringBuilder(formatter.formatBasic(req, res));
44  		if (level.intValue() <= Level.FINE.intValue())
45  			sb.append(formatter.formatHeaders(req, res));
46  		if (level.intValue() <= Level.FINEST.intValue())
47  			sb.append(formatter.formatBody(req, res));
48  
49  		// Stable-INFO emission: the record is always stamped INFO so the basic access line survives under an
50  		// INFO handler even when a logger is raised to FINE/FINEST for diagnostics. `level` (the resolved tier)
51  		// continues to drive only the cumulative section-inclusion checks above.
52  		var logRecord = new LogRecord(Level.INFO, sb.toString());
53  		logRecord.setLoggerName(logger.getName());
54  		if (thrown != null)
55  			logRecord.setThrown(thrown);
56  
57  		// Stamp-from-field: attach the resolved correlation id as a structured field so it survives to emission
58  		// regardless of any log-context scope's lifetime (the load-bearing client fix).  Prefer the server-confirmed
59  		// echoed id, else the sent id.  Map.of rejects null, so skip the attach entirely when neither exists (e.g.
60  		// sendRequestId(false)) — a null-valued map would throw.
61  		var effectiveId = effectiveRequestId(req, res);
62  		if (effectiveId != null)
63  			LogRecordContext.attachIfAbsent(logRecord, Map.of("requestId", effectiveId));
64  
65  		logger.log(logRecord);
66  	}
67  
68  	private static String effectiveRequestId(RestRequest req, RestResponse res) {
69  		if (res != null && res.getRequestId() != null)
70  			return res.getRequestId();
71  		return req != null ? req.getRequestId() : null;
72  	}
73  }
