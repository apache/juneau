258 	/**
259 	 * Sets the request body from an arbitrary Java object.
260 	 *
261 	 * <p>
262 	 * The object is first run through the client's body converter chain. The default converters handle:
263 	 * {@link HttpBody} (passthrough), {@link InputStream}, {@code byte[]}, and {@link File}.
264 	 * Custom converters can be registered on the builder.
265 	 *
266 	 * <p>
267 	 * If no converter matches, the object is serialized with the client's explicitly-configured default serializer
268 	 * ({@link RestClient#getDefaultSerializer()}) and sent as a string body using the serializer's content type.  When
269 	 * no default serializer is configured, an {@link IllegalStateException} is raised (there is no implicit JSON
270 	 * fallback) — configure one via {@link RestClient.Builder#defaultSerializer(org.apache.juneau.marshall.serializer.Serializer)}.
271 	 *
272 	 * @param value The body object. May be <jk>null</jk> to clear the body.
273 	 * @return This object.
274 	 * @throws IOException If a converter fails or the default serializer fails.
275 	 * @throws IllegalStateException If no converter matches and no default serializer is configured on the client.
276 	 */
277 	public RestRequest body(Object value) throws IOException {
278 		if (value == null) {
279 			body = null;
280 			convertedBody = null;
281 			return this;
282 		}
283 		for (var converter : client.bodyConverters) {
284 			if (converter.canConvert(value)) {
285 				convertedBody = converter.convert(value);
286 				body = null;
287 				return this;
288 			}
289 		}
290 		// No converter matched: stream the POJO through the client's default serializer.
291 		// Mirrors classic SerializedEntity — the serializer writes straight to the transport output stream during
292 		// run() rather than being pre-materialized into a String/byte[].  Repeatable (re-serializes on resend), so a
293 		// future auto-retry can resend it; streaming bodies (InputStream/Reader) remain non-repeatable.
294 		var s = client.getDefaultSerializer().orElseThrow(() -> new IllegalStateException(
295 			"No default serializer is configured on the client.  Configure one via RestClient.Builder.defaultSerializer(...)."));
296 		var mt = s.getResponseContentType();
297 		body = SerializerBody.of(s, value, mt != null ? mt.toString() : "application/json");
298 		convertedBody = null;
299 		return this;
300 	}

351 	/**
352 	 * Sets a per-call response timeout for this request.
353 	 *
354 	 * <p>
355 	 * Threaded onto the {@link TransportRequest}.  Intended as a response timeout; connect timeouts are otherwise a client-level setting.
356 	 *
357 	 * <p>
358 	 * <b>Transport support:</b> every built-in transport honors this value, using its native per-request mechanism:
359 	 * the JDK {@code HttpRequest} timeout, an Apache HttpClient 4.5 {@code RequestConfig} (socket + connect timeout),
360 	 * an Apache HttpClient 5 request-scoped response timeout, an OkHttp per-call read timeout, the Jetty response-header
361 	 * wait, and a simulated handler deadline in the mock transport.  On expiry the transport throws a
362 	 * {@link TransportException}.  When set, it overrides the transport's own default for this request only.
363 	 *
364 	 * <p>
365 	 * A non-<jk>null</jk> value must be positive; zero and negative durations are rejected (matching the JDK
366 	 * {@code HttpRequest.timeout} contract) so every transport behaves identically.
367 	 *
368 	 * @param value The response timeout. May be <jk>null</jk> to use the transport default.
369 	 * @return This object.
370 	 * @throws IllegalArgumentException If <c>value</c> is zero or negative.
371 	 */
372 	public RestRequest timeout(Duration value) {
373 		if (value != null && (value.isZero() || value.isNegative()))
374 			throw new IllegalArgumentException("Request timeout must be positive: " + value);
375 		timeout = value;
376 		return this;
377 	}

430 		return true;
431 	}

576 	/**
577 	 * Executes the request and returns the response.
578 	 *
579 	 * <p>
580 	 * The caller is responsible for closing the returned {@link RestResponse}.
581 	 *
582 	 * @return The response. Never <jk>null</jk>.
583 	 * @throws TransportException If a network-level error occurs.
584 	 * @throws RestCallException If the response could not be processed.
585 	 */
586 	@SuppressWarnings({
587 		"java:S1130" // RestCallException retained for public-API/source stability: interceptors declare 'throws Exception', so a RestCallException they raise is caught by the precise-rethrow multi-catch below and propagated unchanged; Sonar's flow analysis misses this path. Removing the throws would break callers that catch RestCallException (a sibling checked exception of TransportException).
588 	})
589 	public RestResponse run() throws TransportException, RestCallException {
590 		var start = Instant.now();
591 		RestResponse response = null;
592 		Throwable error = null;
593 		var debugLevel = RestClientDebugPipeline.resolveTier(client.debugLogger);
594 		var debugCap = client.debugFormatter.bodyCap();
595 		// Union the client-level (builder) interceptors with any per-request (interface- then
596 		// method-level) interceptors, preserving the order builder → interface → method.
597 		var effectiveInterceptors = effectiveInterceptors();
598 		try {
599 			for (var interceptor : effectiveInterceptors)
600 				interceptor.onInit(this);
601 
602 			var transportRequest = buildTransportRequest();
603 			resolvedHeaders = transportRequest.getHeaders();
604 			if (transportRequest.getBody() == null) {
605 				cachedContentLength = 0;
606 			} else {
607 				cachedContentLength = transportRequest.getBody().getContentLength();
608 			}
609 			if (debugLevel == Level.FINEST)
610 				transportRequest = captureRequestBody(transportRequest, debugCap);
611 			if (transportRequest.isSsrfGuardActive() && ! client.transport.supportsUrlPolicy())
612 				throw new TransportException("Refusing to send a policy-covered @Remote request through a transport "
613 					+ "that cannot honor the SSRF guardrail's connect-time contract (pin-on-connect + redirect "
614 					+ "revalidation): " + cn(client.transport) + ".  Use a first-party transport that supports it, or "
615 					+ "set allowPrivateUrls(true) if this is an intentional local-dev/intranet target.");
616 			var transportResponse = client.transport.execute(transportRequest);
617 			response = new RestResponse(transportResponse, client, this, debugLevel, debugCap);
618 
619 			for (var interceptor : effectiveInterceptors)
620 				interceptor.onConnect(this, response);
621 
622 			execTime = Duration.between(start, Instant.now());
623 			response.setExecTime(execTime);
624 			return response;
625 		} catch (TransportException | RestCallException e) {
626 			error = e;
627 			throw e;
628 		} catch (Exception e) {
629 			error = e;
630 			throw new TransportException("Request failed: " + e.getMessage(), e);
631 		} finally {
632 			var elapsed = Duration.between(start, Instant.now());
633 			execTime = elapsed;
634 			exception = error;
635 			RestResponse finalResponse = response;
636 			if (error != null && finalResponse != null) {
637 				// The response was assigned but an error after that point (e.g. a throwing onConnect
638 				// interceptor) means it is never returned to the caller, so the caller can never close it
639 				// themselves — close it here to avoid leaking the underlying connection/body.
640 				try {
641 					finalResponse.setExecTime(elapsed);
642 					finalResponse.close();
643 				} catch (IOException closeError) { // HTT: close() failing on an already-broken response is not reliably reproducible
644 					// suppress — best effort, and the original error is what propagates
645 				}
646 			}
647 			for (var interceptor : effectiveInterceptors) {
648 				try {
649 					interceptor.onClose(this, finalResponse);
650 				} catch (Exception e2) { // HTT: exception in onClose; hard to test reliably
651 					// suppress interceptor close errors — best effort
652 				}
653 			}
654 			if (error != null && finalResponse == null) {
655 				var syntheticResponse = new RestResponse(TransportResponse.builder().statusCode(0).build(), client, this, debugLevel, debugCap);
656 				syntheticResponse.setExecTime(elapsed);
657 				RestClientDebugPipeline.emit(client.debugLogger, client.debugFormatter, debugLevel, this, syntheticResponse, error);
658 			}
659 		}
660 	}

662 	/**
663 	 * Executes this request and opens a closeable, cancellable cursor over the response as a
664 	 * <c>text/event-stream</c>.
665 	 *
666 	 * <p>
667 	 * This executes {@link #run()}, asserts a 2xx response status, then opens the response body as an event stream.
668 	 * Non-2xx responses throw {@link RestCallException} before the body is treated as SSE.
669 	 *
670 	 * <p>
671 	 * The response content type is not validated. This is intentional so callers can branch on response type
672 	 * (for example <c>application/json</c> vs <c>text/event-stream</c>) when protocols use both.
673 	 *
674 	 * <p>
675 	 * Adds an {@code Accept: text/event-stream} header only when neither this request nor inherited client default
676 	 * headers already include {@code Accept}.
677 	 *
678 	 * <p>
679 	 * The caller owns the returned {@link SseEventReader}; closing it also closes the underlying
680 	 * {@link RestResponse} (see {@link ResponseBody#asEventStream()}). If opening the event stream fails,
681 	 * this method best-effort closes the response before propagating.
682 	 *
683 	 * @return A cursor over the live event stream. Never <jk>null</jk>.
684 	 * @throws IOException If a transport-level error occurs (typically {@link TransportException}), if the response
685 	 * 	could not be processed (typically {@link RestCallException}), or if the response has no body to open a
686 	 * 	stream over.
687 	 */
688 	public SseEventReader openEventStream() throws IOException {
689 		if (! hasHeader("Accept"))
690 			header("Accept", "text/event-stream");
691 		var res = run();
692 		var opened = false;
693 		try {
694 			var r = res.assertOk().body().asEventStream();
695 			opened = true;
696 			return r;
697 		} finally {
698 			if (! opened)
699 				quiet(res::close);
700 		}
701 	}

703 	/** Returns the client-level interceptors followed by the per-request ones (builder → interface → method order). */
704 	private List<RestCallInterceptor> effectiveInterceptors() {
705 		if (requestInterceptors.isEmpty())
706 			return client.interceptors;
707 		var l = new ArrayList<RestCallInterceptor>(client.interceptors.size() + requestInterceptors.size());
708 		l.addAll(client.interceptors);
709 		l.addAll(requestInterceptors);
710 		return l;
711 	}

713 	@SuppressWarnings({
714 		"java:S3776" // Request-assembly branching is intentional and cohesive; refactoring would reduce readability without benefit.
715 	})
716 	private TransportRequest buildTransportRequest() {
717 		var resolvedUrl = applyPathSubstitutions(url);
718 		resolvedUri = appendQuery(resolvedUrl);
719 
720 		var builder = TransportRequest.builder()
721 			.method(method)
722 			.uri(resolvedUri)
723 			.timeout(timeout)
724 			.remoteUrlPolicy(policyEnforced, allowPrivateUrls)
725 			.allowedAddressRanges(client.allowedAddressRanges);
726 
727 		// Headers
728 		for (var h : headers) {
729 			var v = h.getValue();
730 			if (v != null)
731 				builder.header(h.getName(), v);
732 		}
733 
734 		if (client.parsers != null || client.defaultParser != null) {
735 			var hasAccept = headers.stream().anyMatch(h -> eqic("Accept", h.getName()));
736 			if (! hasAccept) {
737 				var accept = client.getDefaultAccept();
738 				if (accept != null)
739 					builder.header("Accept", accept);
740 			}
741 		}
742 
743 		// A Content-Type already present in the header list (a caller-supplied @Header param, or a constant header)
744 		// wins over the body's own content type, and is never duplicated by it (dedup invariant).
745 		var hasContentType = hasHeader("Content-Type");
746 
747 		// Pre-converted body from body(Object) takes priority
748 		if (convertedBody != null) {
749 			if (convertedBody.getContentType() != null && ! hasContentType)
750 				builder.header("Content-Type", convertedBody.getContentType());
751 			builder.body(convertedBody);
752 		} else if (!formData.isEmpty() && body == null) {
753 			// Form body when form data is present and no explicit body was set
754 			var formBody = buildFormBody();
755 			if (! hasContentType)
756 				builder.header("Content-Type", formBody.getContentType());
757 			builder.body(TransportBody.of(formBody));
758 		} else if (body != null) {
759 			if (body.getContentType() != null && ! hasContentType)
760 				builder.header("Content-Type", body.getContentType());
761 			builder.body(TransportBody.of(body));
762 		}
763 
764 		return builder.build();
765 	}

767 	private TransportRequest captureRequestBody(TransportRequest request, int cap) {
768 		var originalBody = request.getBody();
769 		if (originalBody == null) {
770 			cachedContent = null;
771 			cachedContentLength = 0;
772 			return request;
773 		}
774 
775 		var wrappedBody = new HttpBody() {
776 			@Override
777 			public String getContentType() {
778 				return originalBody.getContentType();
779 			}
780 
781 			@Override
782 			public long getContentLength() {
783 				return originalBody.getContentLength();
784 			}
785 
786 			@Override
787 			public void writeTo(OutputStream out) throws IOException {
788 				var tee = new BoundedCaptureOutputStream(out, cap);
789 				try {
790 					originalBody.writeTo(tee);
791 				} finally {
792 					cachedContent = tee.getCapturedBytes();
793 					cachedContentLength = tee.getTotalBytesWritten();
794 				}
795 			}
796 
797 			@Override
798 			public boolean isRepeatable() {
799 				return originalBody.isRepeatable();
800 			}
801 		};
802 
803 		var builder = TransportRequest.builder()
804 			.method(request.getMethod())
805 			.uri(request.getUri())
806 			.headers(request.getHeaders())
807 			.body(TransportBody.of(wrappedBody))
808 			.timeout(request.getTimeout())
809 			.remoteUrlPolicy(request.isPolicyEnforced(), request.isAllowPrivateUrls())
810 			.allowedAddressRanges(request.getAllowedAddressRanges());
811 		return builder.build();
812 	}

814 	private String applyPathSubstitutions(String template) {
815 		var result = template;
816 		Object remainder = null;
817 		for (var entry : pathData.entrySet()) {
818 			if (eq(entry.getKey(), "/*")) {
819 				remainder = entry.getValue();  // @PathRemainder — applied after named substitutions
820 				continue;
821 			}
822 			var replacement = entry.getValue() != null ? entry.getValue().toString() : "";
823 			result = result.replace("{" + entry.getKey() + "}", urlEncode(replacement));
824 		}
825 		if (remainder != null) {
826 			var r = remainder.toString();
827 			if (! r.isEmpty()) {
828 				if (result.endsWith("/*"))
829 					result = result.substring(0, result.length() - 2);
830 				if (! result.endsWith("/"))
831 					result += "/";
832 				result += urlEncodePath(r);
833 			}
834 		}
835 		return result;
836 	}

838 	/** URL-encodes a path remainder while preserving {@code /} segment separators. */
839 	private static String urlEncodePath(String value) {
840 		var segments = value.split("/", -1);
841 		var sb = new StringBuilder();
842 		for (var i = 0; i < segments.length; i++) {
843 			if (i > 0)
844 				sb.append('/');
845 			sb.append(urlEncode(segments[i]));
846 		}
847 		return sb.toString();
848 	}

850 	private URI appendQuery(String baseUrl) {
851 		if (queryData.isEmpty()) {
852 			try {
853 				return new URI(baseUrl);
854 			} catch (URISyntaxException e) {
855 				throw iaex(e, "Invalid URL: %s", baseUrl);
856 			}
857 		}
858 		var sb = new StringBuilder(baseUrl);
859 		var first = !baseUrl.contains("?");
860 		for (var part : queryData) {
861 			var v = part.getValue();
862 			if (v == null)
863 				continue;
864 			sb.append(first ? '?' : '&');
865 			sb.append(urlEncode(part.getName())).append('=').append(urlEncode(v));
866 			first = false;
867 		}
868 		try {
869 			return new URI(sb.toString());
870 		} catch (URISyntaxException e) {
871 			throw iaex(e, "Invalid URL: %s", sb);
872 		}
873 	}
874 
875 	private HttpBody buildFormBody() {
876 		var sb = new StringBuilder();
877 		var first = true;
878 		for (var part : formData) {
879 			var v = part.getValue();
880 			if (v == null)
881 				continue;
882 			if (!first)
883 				sb.append('&');
884 			sb.append(urlEncode(part.getName())).append('=').append(urlEncode(v));
885 			first = false;
886 		}
887 		final var encoded = sb.toString();
888 		return new HttpBody() {
889 			@Override public String getContentType() { return "application/x-www-form-urlencoded"; }
890 			@Override public long getContentLength() { return encoded.getBytes(StandardCharsets.UTF_8).length; }
891 			@Override public void writeTo(OutputStream out) throws IOException { out.write(encoded.getBytes(StandardCharsets.UTF_8)); }
892 			@Override public boolean isRepeatable() { return true; }
893 		};
894 	}

896 	private static String urlEncode(String value) {
897 		return URLEncoder.encode(value, StandardCharsets.UTF_8);
898 	}

900 	private static final class BoundedCaptureOutputStream extends OutputStream {
901 		private final OutputStream delegate;
902 		private final ByteArrayOutputStream capture = new ByteArrayOutputStream();
903 		private final int cap;
904 		private long totalBytesWritten;
905 
906 		BoundedCaptureOutputStream(OutputStream delegate, int cap) {
907 			this.delegate = delegate;
908 			this.cap = cap;
909 		}
910 
911 		byte[] getCapturedBytes() {
912 			return capture.toByteArray();
913 		}
914 
915 		long getTotalBytesWritten() {
916 			return totalBytesWritten;
917 		}
918 
919 		@Override
920 		public void write(int b) throws IOException {
921 			delegate.write(b);
922 			totalBytesWritten++;
923 			if (capture.size() < cap)
924 				capture.write(b);
925 		}
926 
927 		@Override
928 		public void write(byte[] b, int off, int len) throws IOException {
929 			delegate.write(b, off, len);
930 			totalBytesWritten += len;
931 			var remaining = cap - capture.size();
932 			if (remaining > 0)
933 				capture.write(b, off, Math.min(len, remaining));
934 		}
935 	}
