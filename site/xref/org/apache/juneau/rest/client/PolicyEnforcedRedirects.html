27  /**
28   * Shared Juneau-controlled redirect loop for SSRF-guard-active ({@link TransportRequest#isSsrfGuardActive()})
29   * {@code @Remote} requests.
30   *
31   * <p>
32   * Every first-party transport that supports the SSRF guardrail's connect-time contract (see
33   * {@link HttpTransport#supportsUrlPolicy()}) disables its own automatic redirect-following for policy-covered
34   * requests and instead routes each hop through {@link #execute(TransportRequest, HopExecutor)}, which re-runs the
35   * {@link RemoteUrlPolicy} pre-check &mdash; and, via the supplied {@link HopExecutor}, pin-on-connect &mdash; on
36   * every {@code Location} hop.  A public {@code @Url} that redirects to {@code http://169.254.169.254/} or
37   * {@code http://127.0.0.1/} is rejected exactly as a direct request to that target would be.
38   *
39   * <p>
40   * Follows conventional 3xx semantics: {@code 301}/{@code 302}/{@code 303} rewrite the method to {@code GET} with
41   * no body, unless the original method was already {@code GET}/{@code HEAD}; {@code 307}/{@code 308} preserve the
42   * original method and body. A non-{@linkplain TransportBody#isRepeatable() repeatable} body cannot be safely
43   * re-sent, so such a redirect is refused (fail closed) rather than risking a corrupt or partial replay.  The
44   * chain is capped at {@link RemoteUrlPolicy#MAX_REDIRECT_HOPS} hops.
45   *
46   * <p>
47   * Caller-set credential headers ({@link RedirectSecurity#stripOnCrossOrigin()}, matched case-insensitively) are
48   * carried to a hop only when it is the same origin as the <em>first</em> request in the chain and not an {@code https}
49   * &rarr; {@code http} downgrade (see {@link RedirectSecurity#shouldStripCredentials(URI, URI)}); otherwise they are
50   * dropped for that hop.  A hop that returns to the first origin gets them back, but only when its method is {@code GET} or {@code HEAD}
51   * (a {@code 307}/{@code 308} replay of a {@code POST} after leaving the first origin stays stripped).
52   *
53   * @since 10.0.0
54   */
55  public final class PolicyEnforcedRedirects {
56  
57  	private PolicyEnforcedRedirects() {}
58  
59  	/**
60  	 * Performs a single pinned HTTP exchange for one hop of a policy-covered redirect chain.
61  	 *
62  	 * <p>
63  	 * Implementations must pin-on-connect (resolve the request URI's host via
64  	 * {@link RemoteUrlPolicy#selectAllowedAddress}, then connect to that specific address) and must not themselves
65  	 * follow any redirect the target returns &mdash; {@link #execute(TransportRequest, HopExecutor)} owns the
66  	 * redirect loop.
67  	 */
68  	@FunctionalInterface
69  	public interface HopExecutor {
70  
71  		/**
72  		 * Executes one pinned hop.
73  		 *
74  		 * @param hopRequest The request to send for this hop.
75  		 * @return The response received for this hop (not yet redirect-followed).
76  		 * @throws TransportException If a network-level error occurs, or the host cannot be pinned to an allowed address.
77  		 */
78  		TransportResponse execute(TransportRequest hopRequest) throws TransportException;
79  	}
80  
81  	/**
82  	 * Runs the redirect loop for a policy-covered {@code @Remote} request.
83  	 *
84  	 * @param initial The initial request. {@link TransportRequest#isSsrfGuardActive()} is expected to be <jk>true</jk>.
85  	 * @param executor Performs one pinned hop; see {@link HopExecutor}.
86  	 * @return The final (non-redirect) response.
87  	 * @throws TransportException If a hop fails, the redirect chain exceeds {@link RemoteUrlPolicy#MAX_REDIRECT_HOPS},
88  	 * 	a hop targets a denied host, a {@code Location} is invalid or its origin cannot be compared, or a non-repeatable
89  	 * 	body would need to be replayed.
90  	 */
91  	@SuppressWarnings({
92  		"resource" // Returns the final hop's response for the caller to close; Eclipse JDT @Owning warning is by design.
93  	})
94  	public static TransportResponse execute(TransportRequest initial, HopExecutor executor) throws TransportException {
95  		var allowPrivateUrls = initial.isAllowPrivateUrls();
96  		var allowedRanges = initial.getAllowedAddressRanges();
97  		var current = initial;
98  		var left = new boolean[1];
99  		for (var hop = 0; hop < RemoteUrlPolicy.MAX_REDIRECT_HOPS; hop++) {
100 			// Hop 0's URI was already validated (or is the operator-configured client root, which the calling engine
101 			// intentionally does not re-check -- see RemoteUrlPolicy.requireAllowedUrl) before this TransportRequest
102 			// was built; only a redirect Location -- a URL discovered at runtime -- needs a fresh pre-check here.
103 			if (hop > 0)
104 				requireAllowed(current.getUri(), allowPrivateUrls, allowedRanges);
105 			var response = executor.execute(current);
106 			if (! isRedirect(response.getStatusCode()))
107 				return response;
108 			var location = requireLocation(response);
109 			closeQuietly(response);
110 			current = nextHop(initial, current, location, response.getStatusCode(), left);
111 		}
112 		throw new TransportException("Redirect limit exceeded (" + RemoteUrlPolicy.MAX_REDIRECT_HOPS
113 			+ " hops) while following a policy-covered @Remote redirect chain, starting at: " + initial.getUri());
114 	}
115 
116 	// An invalid Location, or a target whose origin cannot be compared (java.net.URI parses some host names, such as an
117 	// underscore name, with a null host), is a failed hop like any other, so it is reported as TransportException.
118 	private static TransportRequest nextHop(TransportRequest initial, TransportRequest current, String location, int statusCode, boolean[] left) throws TransportException {
119 		try {
120 			var target = RemoteUrlPolicy.resolveRedirectLocation(current.getUri(), location);
121 			if (target.getRawUserInfo() != null)
122 				throw new TransportException("Redirect Location carries user info, which is refused (it could replace the client's stored credentials): " + location);
123 			// Remember that the chain has left the first origin; a later return to it restores credentials only for GET/HEAD.
124 			var wasLeft = left[0];
125 			left[0] |= RedirectSecurity.shouldStripCredentials(initial.getUri(), current.getUri());
126 			return forRedirect(initial, current, target, statusCode, wasLeft || left[0]);
127 		} catch (IllegalArgumentException e) {
128 			throw new TransportException(e.getMessage(), e);
129 		}
130 	}
131 
132 	private static void requireAllowed(URI uri, boolean allowPrivateUrls, RemoteUrlPolicy.AllowedRanges allowedRanges) throws TransportException {
133 		try {
134 			RemoteUrlPolicy.requireAllowedUrl(uri.toString(), allowPrivateUrls, allowedRanges);
135 		} catch (IllegalArgumentException e) {
136 			throw new TransportException(e.getMessage(), e);
137 		}
138 	}
139 
140 	private static boolean isRedirect(int statusCode) {
141 		return statusCode == 301 || statusCode == 302 || statusCode == 303 || statusCode == 307 || statusCode == 308;
142 	}
143 
144 	private static String requireLocation(TransportResponse response) throws TransportException {
145 		var h = response.getFirstHeader("Location");
146 		if (h == null)
147 			throw new TransportException("Redirect response (status " + response.getStatusCode() + ") is missing a Location header");
148 		return h.value();
149 	}
150 
151 	private static void closeQuietly(TransportResponse response) {
152 		try {
153 			response.close();
154 		} catch (IOException e) {
155 			// Best-effort; the connection will be reclaimed by the pool/GC even if the close callback fails.
156 		}
157 	}
158 
159 	private static boolean safeMethod(String method) {
160 		return eqic(method, "GET") || eqic(method, "HEAD");
161 	}
162 
163 	@SuppressWarnings({
164 		"java:S3776" // Redirect method/body/header rewriting rules read best as one linear sequence.
165 	})
166 	private static TransportRequest forRedirect(TransportRequest initial, TransportRequest current, URI target, int statusCode, boolean left) throws TransportException {
167 		var preserveMethod = eqic(current.getMethod(), "GET") || eqic(current.getMethod(), "HEAD");
168 		var rewriteToGet = ! preserveMethod && (statusCode == 301 || statusCode == 302 || statusCode == 303);
169 
170 		try {
171 			var target2 = new URI(target.getScheme(), target.getRawAuthority(), target.getRawPath(), target.getRawQuery(), target.getRawFragment());
172 			var builder = TransportRequest.builder()
173 				.method(rewriteToGet ? "GET" : current.getMethod())
174 				.uri(target2)
175 				.remoteUrlPolicy(true, false)
176 				.allowedAddressRanges(current.getAllowedAddressRanges())
177 				.timeout(current.getTimeout());
178 			// Credential headers are forwarded only to the first origin (and never across an https -> http downgrade), so a hop
179 			// that returns to it gets the caller's headers back.  They are taken from the initial request, not the previous hop's.
180 			var stripCredentials = RedirectSecurity.shouldStripCredentials(initial.getUri(), target)
181 				|| (left && ! safeMethod(rewriteToGet ? "GET" : current.getMethod()));
182 			for (var h : initial.getHeaders())
183 				if (! (stripCredentials && RedactedHeaders.isSensitive(h.name(), RedirectSecurity.stripOnCrossOrigin())))
184 					builder.header(h.name(), h.value());
185 			if (! rewriteToGet) {
186 				var body = current.getBody();
187 				if (body != null) {
188 					if (! body.isRepeatable())
189 						throw new TransportException("Redirect (status " + statusCode + ") requires re-sending a "
190 							+ "non-repeatable request body, which cannot be safely replayed; refusing to follow "
191 							+ "(fail closed) for a policy-covered @Remote redirect chain: " + target);
192 					builder.body(body);
193 				}
194 			}
195 			return builder.build();
196 		} catch (URISyntaxException e) {
197 			throw new TransportException("Redirect Location did not form a valid request URI: " + target, e);
198 		}
199 	}
