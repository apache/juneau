38  /**
39   * Next-generation transport-agnostic REST client.
40   *
41   * <p>
42   * Create instances via {@link #create()} or {@link #builder()}:
43   * <p class='bjava'>
44   * 	<jc>// Minimal — uses the built-in JDK HttpClient transport</jc>
45   * 	RestClient <jv>client</jv> = RestClient.<jsm>create</jsm>();
46   *
47   * 	<jc>// Explicit transport (e.g. for tests)</jc>
48   * 	RestClient <jv>client</jv> = RestClient.<jsm>builder</jsm>()
49   * 		.transport(<jv>mockTransport</jv>)
50   * 		.header(<js>"Accept"</js>, <js>"application/json"</js>)
51   * 		.build();
52   * </p>
53   *
54   * <p>
55   * By default, {@code RestClient} uses {@link JavaHttpTransport} (backed by the JDK's
56   * {@link java.net.http.HttpClient}), which ships built into the {@code juneau-rest-client} artifact and
57   * requires no extra dependencies on Java 11+.  Pulling in one of the optional transport modules
58   * ({@code juneau-rest-client-apache-httpclient-45}, {@code -apache-httpclient-50}, {@code -okhttp},
59   * {@code -jetty}) registers a higher-priority provider via {@link ServiceLoader} that takes
60   * over automatically.  An explicit {@link Builder#transport(HttpTransport)} call always wins.
61   *
62   * <h5 class='section'>See Also:</h5><ul>
63   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/NextGenRestClient">juneau-ng REST client</a>
64   * </ul>
65   *
66   * @since 9.2.1
67   */

119 	private static HttpTransport discoverTransport() {
120 		var providers = new ArrayList<HttpTransportProvider>();
121 		for (var p : ServiceLoader.load(HttpTransportProvider.class))
122 			if (p.isAvailable()) // HTT: every HttpTransportProvider on this module's test classpath (JavaHttpTransportProvider) is unconditionally available in a standard JVM; the false branch requires a classloader/META-INF manipulation this test suite cannot safely perform without destabilizing unrelated tests.
123 				providers.add(p);
124 		if (providers.isEmpty()) // HTT: same reason as above -- at least one provider is always available on this module's own test classpath.
125 			return JavaHttpTransport.create();  // HTT: defensive fallback if META-INF/services was stripped (e.g. uber-jar shading); unreachable body of the line-124 HTT-excluded branch.
126 		providers.sort(Comparator.comparingInt(HttpTransportProvider::getPriority));
127 		return providers.get(0).create();
128 	}

247 	/**
248 	 * Returns the parser matching the given response {@code Content-Type}, or the explicitly-configured default parser.
249 	 *
250 	 * <p>
251 	 * Resolution precedence: an exact media-type match against the client's {@link ParserSet} wins; otherwise the
252 	 * explicitly-configured {@link Builder#defaultParser(Parser) default parser} is used; otherwise
253 	 * {@link Optional#empty()} is returned.  There is no implicit JSON fallback and no lone-registered-entry fallback,
254 	 * so a fully-unconfigured client resolves to empty (callers throw <c>415 Unsupported Media Type</c>).
255 	 *
256 	 * <p>
257 	 * <b>Beta — API subject to change:</b> This type is part of the next-generation REST client and HTTP stack
258 	 * ({@code org.apache.juneau.marshall.ng.*}).
259 	 * It is not API-frozen: binary- and source-incompatible changes may appear in the <b>next major</b> Juneau release
260 	 * (and possibly earlier).
261 	 *
262 	 * @param contentType The response {@code Content-Type} header value. May be <jk>null</jk>.
263 	 * @return The matching parser, or {@link Optional#empty()} if nothing matched and no default parser is configured.
264 	 */
265 	public Optional<Parser> getMatchingParser(String contentType) {
266 		if (parsers != null && contentType != null) {
267 			var p = parsers.getParser(contentType);
268 			if (p.isPresent())
269 				return p;
270 		}
271 		return o(defaultParser);
272 	}

274 	/**
275 	 * Returns the registered request serializer matching the given media type, or {@link Optional#empty()} if none matches.
276 	 *
277 	 * <p>
278 	 * Unlike {@link #getDefaultSerializer()}, this method performs media-type-driven <i>selection</i> against the
279 	 * client's {@link SerializerSet} (via {@link SerializerSet#getSerializer(String)}) and returns
280 	 * {@link Optional#empty()} on no match so the caller can apply the locked no-match fallback (use the default
281 	 * serializer but still send the overridden {@code Content-Type} label).  An explicitly-configured single default
282 	 * serializer is also consulted when no set is registered.
283 	 *
284 	 * <p>
285 	 * <b>Beta — API subject to change:</b> This type is part of the next-generation REST client and HTTP stack
286 	 * ({@code org.apache.juneau.marshall.ng.*}).
287 	 *
288 	 * @param mediaType The desired request media type. May be <jk>null</jk>/empty.
289 	 * @return The matching serializer, or {@link Optional#empty()} if no registered serializer matches.
290 	 */
291 	public Optional<Serializer> getSerializerForMediaType(String mediaType) {
292 		if (mediaType == null || mediaType.isEmpty() || serializers == null)
293 			return oe();

295 	}

318 	/**
319 	 * Returns the default {@code Accept} header value advertising what this client can read.
320 	 *
321 	 * <p>
322 	 * <b>Beta — API subject to change:</b> This type is part of the next-generation REST client and HTTP stack
323 	 * ({@code org.apache.juneau.marshall.ng.*}).
324 	 * It is not API-frozen: binary- and source-incompatible changes may appear in the <b>next major</b> Juneau release
325 	 * (and possibly earlier).
326 	 *
327 	 * @return The default {@code Accept} header value, or <jk>null</jk> if no parsers and no default parser are
328 	 * 	configured (no implicit {@code application/json} fallback).
329 	 */
330 	public String getDefaultAccept() {
331 		if (parsers != null) {
332 			var mts = parsers.getSupportedMediaTypes();
333 			if (! mts.isEmpty())
334 				return mts.stream().map(MediaType::toString).collect(Collectors.joining(", "));
335 		}
336 		if (defaultParser != null) {
337 			var mts = defaultParser.getMediaTypes();
338 			if (! mts.isEmpty())
339 				return mts.stream().map(MediaType::toString).collect(Collectors.joining(", "));
340 		}
341 		return null;
342 	}

603 		 * This is the explicit, opt-in way to restore a fallback parser.  Without it, a response whose
604 		 * {@code Content-Type} matches no registered parser fails with <c>415 Unsupported Media Type</c> rather than
605 		 * silently parsing as JSON.  Set {@code defaultParser(JsonParser.DEFAULT)} to recover the pre-10.0
606 		 * implicit-JSON behavior.
607 		 *
608 		 * @param value The default parser. May be <jk>null</jk>.
609 		 * @return This object.
610 		 */
611 		public Builder defaultParser(Parser value) {
612 			defaultParser = value;
613 			return this;
614 		}
615 
616 		/**
617 		 * Builds and returns the {@link RestClient}.
618 		 *
619 		 * @return A new instance. Never <jk>null</jk>.
620 		 */
621 		public RestClient build() {
622 			if (serializers == null && ! serializerList.isEmpty())
623 				serializers = SerializerSet.create().add(serializerList.toArray(new Serializer[0])).build();
624 			if (parsers == null && ! parserList.isEmpty())
625 				parsers = ParserSet.create().add(parserList.toArray(new Parser[0])).build();
626 			return new RestClient(this);
627 		}
628 	}
