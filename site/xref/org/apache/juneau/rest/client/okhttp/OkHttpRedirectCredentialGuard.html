20  import java.net.*;
21  import java.util.*;
22  
23  import org.apache.juneau.rest.client.*;

25  import okhttp3.*;

27  /**
28   * Network interceptor that removes caller-set credential headers before a request is sent to a different
29   * origin as the result of a {@code 3xx} redirect.
30   *
31   * <p>
32   * OkHttp's built-in redirect follower drops the {@code Authorization} header only when the target origin
33   * changes, but it keeps other credential headers (notably a caller-set {@code Cookie}).  Registered as a network
34   * interceptor, this class runs for every hop of an exchange: it compares the hop's target against the origin of the
35   * first hop and, when they differ, strips the caller's credential header values.
36   *
37   * <p>
38   * The first hop is the request as it leaves the client's application interceptors, recorded by
39   * {@link #FIRST_HOP_RECORDER} (registered as the last application interceptor).  It is not {@link Call#request()}:
40   * an application interceptor can rewrite the URL, and the rewritten URL is the origin the caller's credentials were
41   * sent to.  Without a recorded first hop, {@link Call#request()} is used.
42   *
43   * <p>
44   * The strip is sticky: once a hop of an exchange has left the first origin, every later hop of that exchange is
45   * stripped too, including a hop back to the first origin ({@code A -> B -> A}).
46   *
47   * <p>
48   * {@code Proxy-Authorization} authenticates to the proxy, not the origin, so it is the one credential header that can be
49   * kept.  It is kept on a cross-origin hop only when the first hop and this hop are both plain HTTP (not an {@code https} target, which is reached
50   * through a {@code CONNECT} tunnel) and goes through the same HTTP proxy as the first hop; stripping it there makes the
51   * proxy answer {@code 407} again, and a {@link OkHttpClient#proxyAuthenticator() proxyAuthenticator} that re-adds it loops
52   * until OkHttp's follow-up limit.  With no proxy, a different proxy, or an {@code https} target it is stripped.
53   *
54   * <p>
55   * Only values the caller sent on the first hop are stripped.  A header value OkHttp itself supplies for the new origin
56   * is kept: the target's own cookies loaded from the client's {@link CookieJar}, and credentials returned by the
57   * client's {@link Authenticator} or proxy {@link Authenticator}, as long as they differ from the value the caller sent on
58   * the first hop.
59   *
60   * <p>
61   * Once a hop has been stripped, every later hop of the same call is stripped too, including one back to the first origin.
62   * That state travels on a tag of the request, so a network {@link Interceptor} or an {@link Authenticator} that builds a
63   * request from scratch (rather than from {@code request.newBuilder()}) must preserve the request's tags, or the sticky strip
64   * is lost for a later hop back to the first origin.
65   *
66   * <p>
67   * The origin is compared on OkHttp's own {@link HttpUrl} (scheme, host and effective port), with the same semantics
68   * as {@link RedirectSecurity#shouldStripCredentials(java.net.URI, java.net.URI)}.  {@link java.net.URI} cannot be used
69   * here: it parses a host name that OkHttp accepts but RFC 2396 does not (for example {@code my_service}, a typical
70   * Docker Compose service name) with a {@code null} host, which would strip credentials from every request to that host.
71   * The credential header set is delegated to {@link RedirectSecurity#stripOnCrossOrigin()} so that all transports share it.
72   */
73  final class OkHttpRedirectCredentialGuard implements Interceptor {
74  
75  	/**
76  	 * Application interceptor that tags the request leaving the application interceptors as the first hop.
77  	 *
78  	 * <p>
79  	 * OkHttp builds every retry and redirect hop with {@link Request#newBuilder()}, which carries the tag along.
80  	 */
81  	@SuppressWarnings({
82  		"resource" // returned Response is owned and closed by OkHttp's call chain, not by this interceptor.
83  	})
84  	static final Interceptor FIRST_HOP_RECORDER = chain -> {
85  		var request = chain.request();
86  		return chain.proceed(request.newBuilder().tag(FirstHop.class, new FirstHop(request)).build());
87  	};
88  
89  	// The first hop's request, plus (set once, on the first network pass) the address of the HTTP proxy it was routed
90  	// through, or null when it was not routed through an HTTP proxy.
91  	private static final class FirstHop {
92  		final Request request;
93  		boolean routeRecorded;
94  		SocketAddress proxyAddress;
95  		// Set once any hop of the chain left the first origin; later hops (including a return to the first origin) stay stripped.
96  		boolean stripped;
97  
98  		FirstHop(Request request) {
99  			this.request = request;
100 		}
101 	}
102 
103 	@Override /* Interceptor */
104 	@SuppressWarnings({
105 		"java:S135", // Guard clauses keep the redirect credential-strip check readable.
106 		"java:S3776", // Single linear pass over the redirect hop's credential-stripping rules; splitting it would scatter them.
107 		"resource" // returned Response is owned and closed by OkHttp's call chain, not by this interceptor.
108 	})
109 	public Response intercept(Chain chain) throws IOException {
110 		var request = chain.request();
111 		var tag = request.tag(FirstHop.class);
112 		var first = tag != null ? tag.request : chain.call().request();
113 		var proxyAddress = httpProxyAddress(chain.connection());
114 		if (tag != null && ! tag.routeRecorded) {
115 			tag.routeRecorded = true;
116 			tag.proxyAddress = proxyAddress;
117 		}
118 		var crossOrigin = ! sameOrigin(first.url(), request.url());
119 		if (crossOrigin && tag != null)
120 			tag.stripped = true;
121 		if (crossOrigin || (tag != null && tag.stripped)) {
122 			// Sticky: OkHttp derives each follow-up from the pre-strip request, so a hop back to the first origin (A -> B -> A)
123 			// would otherwise be passed unmodified and re-send the caller's credentials.
124 			// Proxy-Authorization authenticates to the proxy, so it is kept only when this hop still goes to the proxy that
125 			// the first hop went through, and both hops are plain HTTP (an https target is reached through a CONNECT tunnel,
126 			// so the header would reach the origin; and a header that travelled inside a tunnel on the first hop must not be
127 			// downgraded to cleartext to the proxy).  Without a proxy, or when the proxy changed, it is stripped like the rest.
128 			var keepProxyAuth = tag != null && proxyAddress != null && ! first.isHttps() && ! request.isHttps() && proxyAddress.equals(tag.proxyAddress);
129 			Request.Builder b = null;
130 			for (var name : RedirectSecurity.stripOnCrossOrigin()) {
131 				if (keepProxyAuth && "Proxy-Authorization".equalsIgnoreCase(name))
132 					continue;
133 				var values = request.headers(name);
134 				var callerValues = first.headers(name);
135 				if (values.isEmpty() || callerValues.isEmpty())
136 					continue;
137 				var kept = new ArrayList<>(values);
138 				kept.removeAll(callerValues);
139 				if (kept.size() == values.size())
140 					continue;
141 				if (b == null)
142 					b = request.newBuilder();
143 				b.removeHeader(name);
144 				for (var v : kept)
145 					b.addHeader(name, v);
146 			}
147 			if (b != null)
148 				request = b.build();
149 		}
150 		return chain.proceed(request);
151 	}
152 
153 	private static SocketAddress httpProxyAddress(Connection connection) {
154 		if (connection == null)
155 			return null;
156 		var proxy = connection.route().proxy();
157 		return proxy.type() == Proxy.Type.HTTP ? proxy.address() : null;
158 	}
159 
160 	// HttpUrl always has a lower-case scheme (http/https), a non-null canonical (lower-case) host and an effective port
161 	// (the scheme default when none is given), so the comparison is total.  An https->http downgrade changes the scheme,
162 	// so it is never same-origin.
163 	private static boolean sameOrigin(HttpUrl from, HttpUrl to) {
164 		return from.scheme().equals(to.scheme()) && from.host().equals(to.host()) && from.port() == to.port();
165 	}
166 }
