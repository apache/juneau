17  package org.apache.juneau.rest.client.okhttp;
18  
19  import java.io.*;
20  import java.net.*;
21  
22  import org.apache.juneau.rest.client.*;
23  
24  import okhttp3.*;
25  
26  /**
27   * Network interceptor that removes caller-set credential headers before a request is sent to a different
28   * origin as the result of a {@code 3xx} redirect.
29   *
30   * <p>
31   * OkHttp's built-in redirect follower drops the {@code Authorization} header only when the target host
32   * changes, but it keeps other credential headers (notably a caller-set {@code Cookie}) and does not account
33   * for a differing port or an {@code https}&rarr;{@code http} downgrade.  Registered as a network interceptor,
34   * this class runs for every hop of an exchange: it compares the hop's target against the origin of the
35   * original call and strips the credential headers whenever they differ.
36   *
37   * <p>
38   * The forward/strip decision and the header set are delegated to {@link RedirectSecurity} so that all
39   * transports share one policy.
40   */
41  final class OkHttpRedirectCredentialGuard implements Interceptor {
42  
43  	@Override /* Interceptor */
44  	@SuppressWarnings("resource") // returned Response is owned and closed by OkHttp's call chain, not by this interceptor.
45  	public Response intercept(Chain chain) throws IOException {
46  		var request = chain.request();
47  		var from = chain.call().request().url().uri();
48  		var to = request.url().uri();
49  		if (shouldStrip(from, to)) {
50  			var b = request.newBuilder();
51  			for (var name : RedirectSecurity.stripOnCrossOrigin())
52  				b.removeHeader(name);
53  			request = b.build();
54  		}
55  		return chain.proceed(request);
56  	}
57  
58  	private static boolean shouldStrip(URI from, URI to) {
59  		try {
60  			return RedirectSecurity.shouldStripCredentials(from, to);
61  		} catch (RuntimeException e) {
62  			// If the origins cannot be compared, err on the side of not forwarding credentials.
63  			return true;
64  		}
65  	}
66  }
