17  package org.apache.juneau.rest.client.jetty;
18  
19  import java.net.*;
20  
21  import org.eclipse.jetty.client.*;
22  import org.eclipse.jetty.client.transport.*;
23  
24  /**
25   * {@link HttpRequest} that removes caller-set credential headers from the copy Jetty makes of it to follow a
26   * {@code 3xx} redirect to a different origin.
27   *
28   * <p>
29   * This is how the transport guards a client the caller supplies: that client is not a
30   * {@link CredentialGuardingHttpClient} and is not changed, but {@link HttpClient}'s default {@code copyRequest} delegates
31   * to {@link #copy(URI)}, so a request created as this class is guarded on any client.  The copy is again this class, so
32   * every hop of a redirect chain is guarded.  A client subclass that overrides {@code copyRequest} without calling the
33   * default replaces this guard with its own handling.
34   *
35   * <p>
36   * The rule and the header set are those of {@link CredentialGuardingHttpClient}.  Only the copied headers are removed:
37   * Jetty adds a {@code Cookie} from the client's cookie store and an {@code Authorization} from its authentication store
38   * to the copy afterwards, as for any request.
39   */
40  final class CredentialGuardingHttpRequest extends HttpRequest {
41  
42  	/**
43  	 * Constructor.
44  	 *
45  	 * @param client The client that sends the request.
46  	 * @param uri The request URI.
47  	 * @throws IllegalArgumentException If Jetty refuses the URI (for example a URI with a {@code null} host).
48  	 */
49  	CredentialGuardingHttpRequest(HttpClient client, URI uri) {
50  		this(client, new HttpConversation(), uri);
51  	}
52  
53  	private CredentialGuardingHttpRequest(HttpClient client, HttpConversation conversation, URI uri) {
54  		super(client, conversation, uri);
55  	}
56  
57  	@Override /* HttpRequest */
58  	public HttpRequest copy(URI newURI) {
59  		var copy = super.copy(newURI);
60  		// A null URI copies the request for its own origin (for example to replay it with authentication).
61  		return newURI == null ? copy : CredentialGuardingHttpClient.stripIfCrossOrigin(this, newURI, copy);
62  	}
63  
64  	@Override /* HttpRequest */
65  	@SuppressWarnings({
66  		"resource" // the new request is returned to the caller, which owns it.
67  	})
68  	protected HttpRequest copyInstance(URI newURI) {
69  		return new CredentialGuardingHttpRequest(getHttpClient(), getConversation(), newURI);
70  	}
71  }
