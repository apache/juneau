17  package org.apache.juneau.rest.client.jetty;
18  
19  import java.net.*;
20  import java.util.*;
21  
22  import org.apache.juneau.http.remote.*;
23  import org.eclipse.jetty.client.*;
24  import org.eclipse.jetty.util.*;
25  
26  /**
27   * {@link SocketAddressResolver} that pin-on-connects SSRF-guard-active {@code @Remote} connections: when the
28   * connection being opened belongs to a destination tagged with {@link #PINNED_TAG}, resolution is restricted to a
29   * single address selected by {@link RemoteUrlPolicy#selectAllowedAddress} &mdash; the exact address Jetty's connector
30   * then connects the socket to, while still using the original hostname (from the request's own origin) for TLS
31   * SNI/hostname verification. Connections for every other destination on the same {@code HttpClient} delegate to the
32   * wrapped resolver unchanged.
33   *
34   * <p>
35   * Jetty keys its destinations (and so its connection pools) by origin <em>including</em> the request tag, so tagging
36   * guarded requests gives them a connection pool of their own. Every connection in that pool is opened through this
37   * pin, and a guarded request can never be handed an idle connection that an unguarded request to the same origin
38   * opened without it. Keying on the destination rather than on the calling thread also keeps the pin in force when
39   * Jetty opens a connection for a queued guarded request from one of its own threads.
40   *
41   * @since 10.0.0
42   */
43  final class PolicyPinningSocketAddressResolver implements SocketAddressResolver {
44  
45  	/**
46  	 * The Jetty request tag that marks an SSRF-guard-active request, carrying the request's per-client allowed address ranges.
47  	 *
48  	 * <p>
49  	 * Value-equal on its ranges: Jetty's destination lookup (which compares tags with {@code equals()}) puts requests
50  	 * with the same ranges in one connection pool and never puts a request tagged with anything else (including a
51  	 * guarded request with different ranges) in it, so a connection opened under one allow-list is never reused under another.
52  	 */
53  	record PinTag(RemoteUrlPolicy.AllowedRanges ranges) {}
54  
55  	/** The tag for a guarded request with no per-client allowed address ranges. */
56  	static final PinTag PINNED_TAG = new PinTag(RemoteUrlPolicy.AllowedRanges.NONE);
57  
58  	static PinTag tagFor(RemoteUrlPolicy.AllowedRanges ranges) {
59  		return ranges.isEmpty() ? PINNED_TAG : new PinTag(ranges);
60  	}
61  
62  	private final SocketAddressResolver delegate;
63  
64  	PolicyPinningSocketAddressResolver(SocketAddressResolver delegate) {
65  		this.delegate = delegate;
66  	}
67  
68  	@Override /* SocketAddressResolver */
69  	public void resolve(String host, int port, Map<String, Object> context, Promise<List<InetSocketAddress>> promise) {
70  		var tag = pinTag(context);
71  		if (tag == null) {
72  			delegate.resolve(host, port, context, promise);
73  			return;
74  		}
75  		try {
76  			// A destination is tagged only for an SSRF-guard-active request, and the guard is active only when the
77  			// allowPrivateUrls opt-in is off (TransportRequest.isSsrfGuardActive()), so the deny-private checks always apply here.
78  			var pinned = RemoteUrlPolicy.selectAllowedAddress(host, false, RemoteUrlPolicy.AddressResolver.DEFAULT, tag.ranges());
79  			promise.succeeded(List.of(new InetSocketAddress(pinned, port)));
80  		} catch (UnknownHostException | RuntimeException e) {
81  			promise.failed(e);
82  		}
83  	}
84  
85  	// Jetty puts the destination the connection is being opened for into the resolve context.
86  	private static PinTag pinTag(Map<String, Object> context) {
87  		return context.get(Destination.CONTEXT_KEY) instanceof Destination d && d.getOrigin().getTag() instanceof PinTag t ? t : null;
88  	}
89  }
