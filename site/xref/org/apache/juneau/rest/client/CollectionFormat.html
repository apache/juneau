17  package org.apache.juneau.rest.client;
18  
19  import static org.apache.juneau.commons.utils.AssertionUtils.*;
20  
21  /**
22   * Specifies how collection-valued query/form parameters are serialized to a single string.
23   *
24   * <p>
25   * When a parameter value is an {@link Iterable} or array, the elements are joined using the delimiter
26   * corresponding to the selected format.  {@link #REPEATED} instead emits one name=value pair per element.
27   *
28   * <p>
29   * <b>Beta — API subject to change.</b>
30   *
31   * @since 9.2.1
32   */
33  public enum CollectionFormat {
34  
35  	/** Elements joined with {@code ,} — e.g. {@code colors=red,green,blue}. */
36  	COMMA(","),
37  
38  	/** Elements joined with {@code |} — e.g. {@code colors=red|green|blue}. */
39  	PIPE("|"),
40  
41  	/** Elements joined with a space — e.g. {@code colors=red green blue}. */
42  	SPACE(" "),
43  
44  	/** Elements joined with {@code \t} (tab) — e.g. {@code colors=red\tgreen\tblue}. */
45  	TAB("\t"),
46  
47  	/** One parameter per element — e.g. {@code colors=red&colors=green&colors=blue}. Handled specially by the request builder. */
48  	REPEATED(null);
49  
50  	private final String delimiter;
51  
52  	CollectionFormat(String delimiter) {
53  		this.delimiter = delimiter;
54  	}
55  
56  	/**
57  	 * Returns the delimiter string, or {@code null} for {@link #REPEATED}.
58  	 *
59  	 * @return The delimiter, possibly <jk>null</jk>.
60  	 */
61  	public String getDelimiter() {
62  		return delimiter;
63  	}
64  
65  	/**
66  	 * Serializes the given values to a single string using this format's delimiter.
67  	 *
68  	 * <p>
69  	 * Only valid for non-{@link #REPEATED} formats.
70  	 *
71  	 * @param values The values to join. Must not be <jk>null</jk>.
72  	 * @return The joined string. Never <jk>null</jk>.
73  	 * @throws IllegalArgumentException If called on {@link #REPEATED}.
74  	 */
75  	public String join(Iterable<?> values) {
76  		assertArg(delimiter != null, "join() is not supported for CollectionFormat.REPEATED");
77  		var sb = new StringBuilder();
78  		var first = true;
79  		for (var v : values) {
80  			if (!first)
81  				sb.append(delimiter);
82  			sb.append(v);
83  			first = false;
84  		}
85  		return sb.toString();
86  	}
87  }
