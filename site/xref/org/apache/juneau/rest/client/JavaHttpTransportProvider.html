17  package org.apache.juneau.rest.client;
18  
19  /**
20   * {@link HttpTransportProvider} implementation that supplies a {@link JavaHttpTransport}.
21   *
22   * <p>
23   * Registered via {@code META-INF/services/HttpTransportProvider} so that
24   * {@code RestClient} auto-discovers the JDK HTTP client when no higher-priority transport is present.
25   * Since the JDK transport ships in the {@code juneau-rest-client} artifact itself, it is always available
26   * unless overridden by a sibling transport module with a lower priority value.
27   *
28   * <h5 class='section'>See Also:</h5><ul>
29   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/NextGenRestClient">juneau-ng REST client</a>
30   * </ul>
31   *
32   * @since 9.2.1
33   */
34  @SuppressWarnings({
35  	"resource" // Not owned here; lifecycle is managed by the surrounding context
36  })
37  public final class JavaHttpTransportProvider implements HttpTransportProvider {
38  
39  	@Override /* HttpTransportProvider */
40  	public int getPriority() {
41  		return 80;
42  	}
43  
44  	@Override /* HttpTransportProvider */
45  	public boolean isAvailable() {
46  		try {
47  			Class.forName("java.net.http.HttpClient");
48  			return true;
49  		} catch (ClassNotFoundException e) { // HTT: java.net.http.HttpClient has shipped in every JDK since 11; unreachable on any JVM this test suite can run on.
50  			return false; // HTT: unreachable body of the line-above HTT-excluded catch block.
51  		}
52  	}
53  
54  	@Override /* HttpTransportProvider */
55  	public HttpTransport create() {
56  		return JavaHttpTransport.create();
57  	}
58  }
