33  /**
34   * Contains the meta-data about a REST proxy class.
35   *
36   * <p>
37   * Captures the information in {@link Remote @Remote} and {@link RemoteOp @RemoteOp} annotations for
38   * caching and reuse.
39   *
40   * <h5 class='section'>See Also:</h5><ul>
41   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestProxies">REST Proxy Basics</a>
42   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauRestClient">juneau-rest-client Basics</a>
43   * </ul>
44   */
45  public class RemoteMeta {

47  	private static String resolve(String s) {
48  		return VarResolver.DEFAULT.resolve(s);
49  	}

56  	private final String baseUrl;
57  	private final String accept;

59  	private final List<Map.Entry<String,String>> queryData;
60  	private final List<Map.Entry<String,String>> formData;

64  	private final boolean throwOnError;

67  	/**
68  	 * Constructor.
69  	 *
70  	 * @param c The interface class annotated with a {@link Remote @Remote} annotation (optional).
71  	 */
72  	@SuppressWarnings({
73  		"java:S3776" // Cognitive complexity acceptable for remote interface metadata construction dispatch
74  	})
75  	public RemoteMeta(Class<?> c) {
76  		var path = "";
77  
78  		var ci = ClassInfo.of(c);
79  		var remotes = rstream(ci.getAnnotations(Remote.class).toList()).map(AnnotationInfo::inner).toList();
80  
81  		var versionHeader = "Client-Version";
82  		String clientVersion = null;
83  		var headers2 = HeaderList.create().resolving();
84  
85  		var baseUrl2 = "";
86  		var accept2 = "";
87  		var contentType2 = "";
88  		var queryData2 = new ArrayList<Map.Entry<String,String>>();
89  		var formData2 = new ArrayList<Map.Entry<String,String>>();
90  		var timeout2 = "";
91  		var retries2 = 0;
92  		var retryNonIdempotent2 = false;
93  		var throwOnError2 = false;
94  		var allowPrivateUrls2 = false;
95  
96  		for (var r : remotes) {
97  			if (ine(r.path()))
98  				path = trimSlashes(resolve(r.path()));
99  			else if (ine(r.value()))
100 				path = trimSlashes(resolve(r.value()));
101 			for (var h : r.headers())
102 				headers2.append(stringHeader(resolve(h)));
103 			if (ine(r.version()))
104 				clientVersion = resolve(r.version());
105 			if (ine(r.versionHeader()))
106 				versionHeader = resolve(r.versionHeader());
107 			if (isNotVoid(r.headerList()) && HeaderList.class.isAssignableFrom(r.headerList())) {
108 				try {
109 					headers2.append(((HeaderList) r.headerList().getDeclaredConstructor().newInstance()).getAll());
110 				} catch (Exception e) {
111 					throw rex(e, "Could not instantiate HeaderSupplier class");
112 				}
113 			}
114 			if (ine(r.baseUrl()))
115 				baseUrl2 = resolve(r.baseUrl());
116 			if (ine(r.accept()))
117 				accept2 = resolve(r.accept());
118 			if (ine(r.contentType()))
119 				contentType2 = resolve(r.contentType());
120 			queryData2.addAll(RemoteProxyUtils.parseConstantParts(r.queryData(), '='));
121 			formData2.addAll(RemoteProxyUtils.parseConstantParts(r.formData(), '='));
122 			if (ine(r.timeout()))
123 				timeout2 = r.timeout();
124 			if (r.retries() > 0)
125 				retries2 = r.retries();
126 			retryNonIdempotent2 |= r.retryNonIdempotent();
127 			throwOnError2 |= r.throwOnError();
128 			allowPrivateUrls2 |= r.allowPrivateUrls();
129 			// Genuinely engine-specific: classic and NG RestCallInterceptor SPI types are nominally
130 			// incompatible, so the classic engine cannot honor interface-level interceptors.  Warn once.
131 			if (r.interceptors().length > 0)
132 				RemoteProxyUtils.warnUnsupportedMember(c, "interceptors",
133 					"classic and next-generation RestCallInterceptor SPI types are nominally incompatible.");
134 		}
135 
136 		if (nn(clientVersion))
137 			headers2.append(stringHeader(versionHeader, clientVersion));
138 
139 		Map<Method,RemoteOperationMeta> operations2 = map();
140 		var path2 = path;
141 		ci.getPublicMethods().forEach(x -> operations2.put(x.inner(), new RemoteOperationMeta(path2, x.inner(), "GET")));
142 
143 		this.operations = u(operations2);
144 		this.headers = headers2.unmodifiable();
145 		this.baseUrl = baseUrl2;
146 		this.accept = accept2;
147 		this.contentType = contentType2;
148 		this.queryData = u(queryData2);
149 		this.formData = u(formData2);
150 		this.timeout = timeout2;
151 		this.retries = retries2;
152 		this.retryNonIdempotent = retryNonIdempotent2;
153 		this.throwOnError = throwOnError2;
154 		this.allowPrivateUrls = allowPrivateUrls2;
155 	}
