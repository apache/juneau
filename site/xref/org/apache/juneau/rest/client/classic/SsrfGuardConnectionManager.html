34  /**
35   * The connection manager a {@link RestClient} installs on the client it builds, holding two separate pools.
36   *
37   * <p>
38   * Connections for SSRF-guard-active {@code @Remote} calls (requests whose context carries the {@link SsrfGuardMarker},
39   * or made while the calling thread runs a guarded call) are leased from the {@code pinned} delegate, whose connections
40   * are all opened through {@link PolicyPinningDnsResolver}.  Every other connection is leased from the {@code ordinary}
41   * delegate.  So a guarded call can never be handed an idle connection that an unguarded call opened without the pin,
42   * and the reverse.
43   *
44   * <p>
45   * {@link HttpClientConnectionManager#requestConnection(HttpRoute, Object)} is not given the request's context, so the
46   * client's {@link GuardInterceptor}, which Apache HttpClient runs on the same thread just before the lease, reads the
47   * marker and hands the decision over.  The guard then fails closed: if the interceptor did not run (an
48   * {@link HttpClientBuilder#setHttpProcessor(HttpProcessor) HttpProcessor} replaced the client's interceptors) and a
49   * guarded request was nevertheless handed an ordinary connection, {@link #connect connect} refuses to open it and the
50   * client's {@link GuardedRequestExecutor} refuses to send the request over it.
51   *
52   * <p>
53   * The pinned pool runs with the ordinary pool's settings: before each guarded lease, the pool limits, the socket and
54   * connection configuration and the validate-after-inactivity interval of the ordinary pool are copied onto it.  So a
55   * caller who configures the default connection manager (in an override of {@code createConnectionManager()} or
56   * later) configures guarded calls too.
57   *
58   * <p>
59   * Each later operation on a leased connection (connect, upgrade, release) goes to the delegate that leased it,
60   * whatever thread it runs on: a streamed response body may be released after the guarded call has returned.
61   *
62   * <p>
63   * A guarded lease on a proxy route is refused: the pin would check the proxy's address, not the target's, and the
64   * proxy then connects to the target unchecked.
65   *
66   * @since 10.0.0
67   */
68  final class SsrfGuardConnectionManager implements HttpClientConnectionManager {
69  
70  	private final HttpClientConnectionManager ordinary;
71  	private final HttpClientConnectionManager pinned;
72  
73  	// The lease decision that GuardInterceptor hands to the next requestConnection call on the same thread: set to true
74  	// for a guarded request, absent otherwise.  GuardInterceptor sets or clears it for every request and requestConnection
75  	// clears it before anything else, so a decision can only be left over when a request interceptor after GuardInterceptor
76  	// throws before the lease; the next lease on that thread is then pinned although its request is not guarded.  A
77  	// leftover can only over-guard by pinning an ordinary request, and never lets a guarded one through unpinned.
78  	private static final ThreadLocal<Boolean> NEXT_LEASE_GUARDED = new ThreadLocal<>();
79  
80  	// Leased connections of the pinned pool.  Connection classes use identity equality, so this is an identity map.
81  	// Weak keys mean a leaked (never released) connection is not kept alive by this map.
82  	private final Map<HttpClientConnection,Boolean> pinnedLeases = Collections.synchronizedMap(new WeakHashMap<>());
83  
84  	SsrfGuardConnectionManager(HttpClientConnectionManager ordinary, HttpClientConnectionManager pinned) {
85  		this.ordinary = ordinary;
86  		this.pinned = pinned;
87  	}
88  
89  	@Override /* HttpClientConnectionManager */
90  	public ConnectionRequest requestConnection(HttpRoute route, Object state) {
91  		var handedOver = NEXT_LEASE_GUARDED.get() != null;
92  		NEXT_LEASE_GUARDED.remove();
93  		if (! handedOver && ! SsrfGuardMarker.isGuarded(null, null))
94  			return ordinary.requestConnection(route, state);
95  		if (nn(route.getProxyHost()))
96  			throw new GuardRefusedException("Refusing to send a policy-covered @Remote request to '" + route.getTargetHost()
97  				+ "' through a proxy (" + route.getProxyHost() + "): the SSRF guardrail's pin-on-connect would check the "
98  				+ "proxy's address, not the target's.  Remove the proxy (proxy(...), routePlanner(...) or the proxy system "
99  				+ "properties used by useSystemProperties()), or set allowPrivateUrls(true) if this is an intentional "
100 				+ "local-dev/intranet target.");
101 		mirrorSettings(route);
102 		var request = pinned.requestConnection(route, state);
103 		return new ConnectionRequest() {
104 			@Override /* ConnectionRequest */
105 			@SuppressWarnings({
106 				"resource" // the leased connection is handed to the caller, which releases it through releaseConnection().
107 			})
108 			public HttpClientConnection get(long timeout, TimeUnit timeUnit) throws InterruptedException, ExecutionException, ConnectionPoolTimeoutException {
109 				var conn = request.get(timeout, timeUnit);
110 				pinnedLeases.put(conn, Boolean.TRUE);
111 				return conn;
112 			}
113 			@Override /* Cancellable */
114 			public boolean cancel() {
115 				return request.cancel();
116 			}
117 		};
118 	}
119 
120 	@Override /* HttpClientConnectionManager */
121 	public void releaseConnection(HttpClientConnection conn, Object newState, long validDuration, TimeUnit timeUnit) {
122 		var manager = pinnedLeases.remove(conn) != null ? pinned : ordinary;
123 		manager.releaseConnection(conn, newState, validDuration, timeUnit);
124 	}

126 	@Override /* HttpClientConnectionManager */
127 	public void connect(HttpClientConnection conn, HttpRoute route, int connectTimeout, HttpContext context) throws IOException {
128 		requirePinnedIfGuarded(conn, null, context, "to open a connection of the ordinary pool");
129 		owner(conn).connect(conn, route, connectTimeout, context);
130 	}

135 	 * wired.

137 	 * @return <jk>true</jk> if clients can be checked.

140 		return nn(CONN_MANAGER);
141 	}

143 	/**
144 	 * Returns whether the given client is wired to this connection manager: it is an Apache {@code InternalHttpClient}
145 	 * (what {@link HttpClientBuilder#build()} returns) whose connection manager is this one.
146 	 *
147 	 * @param client The client.
148 	 * @return <jk>false</jk> for any other client, including one that wraps such a client (it may send the request
149 	 * 	elsewhere), and if the client's connection manager cannot be read.
150 	 */
151 	boolean isConnectionManagerOf(HttpClient client) {
152 		if (CONN_MANAGER == null || client == null || client.getClass() != CONN_MANAGER.getDeclaringClass())
153 			return false;
154 		try {
155 			return CONN_MANAGER.get(client) == this;
156 		} catch (@SuppressWarnings("unused") IllegalAccessException e) {
157 			// HTT: unreachable, since the field was made accessible when it was looked up.
158 			return false;
159 		}
160 	}
161 
162 	// InternalHttpClient.connManager, or null if this HttpClient release has no such field (every client then counts as
163 	// not wired, so guarded calls are refused).
164 	private static final Field CONN_MANAGER = connManagerField();
165 
166 	@SuppressWarnings({
167 		"java:S3011" // The client's connection manager has no accessor, so the field is read reflectively.
168 	})
169 	private static Field connManagerField() {
170 		try {
171 			var f = Class.forName("org.apache.http.impl.client.InternalHttpClient").getDeclaredField("connManager");
172 			f.setAccessible(true);
173 			return f;
174 		} catch (@SuppressWarnings("unused") ReflectiveOperationException | RuntimeException e) {
175 			// HTT: only an HttpClient release without this field, or a JVM that denies reflective access to it, gets here.
176 			return null;
177 		}
178 	}

180 	// Refuses a guarded request on a connection of the ordinary pool, which was or would be opened without the pin.
181 	private void requirePinnedIfGuarded(HttpClientConnection conn, HttpRequest request, HttpContext context, String what) {
182 		// isGuarded first: it also refuses a guarded request whose context lacks the marker, on any connection.
183 		if (SsrfGuardMarker.isGuarded(request, context) && ! pinnedLeases.containsKey(conn))
184 			throw new GuardRefusedException("Refusing to send a policy-covered @Remote request " + what + ", which is not "
185 				+ "opened through the SSRF guardrail's pin-on-connect: the request was executed "
186 				+ "on a thread where its guard was not known when the connection was leased (the client's request "
187 				+ "interceptors were replaced, for example by httpProcessor(...)).  Remove the HttpProcessor setting, or set "
188 				+ "allowPrivateUrls(true) if this is an intentional local-dev/intranet target.");
189 	}

227 	// Copies the ordinary pool's settings onto the pinned pool (both are built by the same factory, so they are always
228 	// the same kind).  Only changed values are written: each setter takes the pool's lock.
229 	@SuppressWarnings({
230 		"java:S3776" // One flat run of compare-and-copy steps, one per pool setting.
231 	})
232 	private void mirrorSettings(HttpRoute route) {
233 		if (ordinary instanceof PoolingHttpClientConnectionManager o) {
234 			var p = (PoolingHttpClientConnectionManager)pinned;
235 			if (p.getMaxTotal() != o.getMaxTotal())
236 				p.setMaxTotal(o.getMaxTotal());
237 			if (p.getDefaultMaxPerRoute() != o.getDefaultMaxPerRoute())
238 				p.setDefaultMaxPerRoute(o.getDefaultMaxPerRoute());
239 			if (p.getMaxPerRoute(route) != o.getMaxPerRoute(route))
240 				p.setMaxPerRoute(route, o.getMaxPerRoute(route));
241 			if (p.getDefaultSocketConfig() != o.getDefaultSocketConfig())
242 				p.setDefaultSocketConfig(o.getDefaultSocketConfig());
243 			if (p.getDefaultConnectionConfig() != o.getDefaultConnectionConfig())
244 				p.setDefaultConnectionConfig(o.getDefaultConnectionConfig());
245 			// Guarded routes never have a proxy (see requestConnection), so the pool connects to the target itself.
246 			mirrorHostSettings(o, p, route.getTargetHost());
247 			if (p.getValidateAfterInactivity() != o.getValidateAfterInactivity())
248 				p.setValidateAfterInactivity(o.getValidateAfterInactivity());
249 		} else if (ordinary instanceof BasicHttpClientConnectionManager o) {
250 			var p = (BasicHttpClientConnectionManager)pinned;
251 			if (p.getSocketConfig() != o.getSocketConfig())
252 				p.setSocketConfig(o.getSocketConfig());
253 			if (p.getConnectionConfig() != o.getConnectionConfig())
254 				p.setConnectionConfig(o.getConnectionConfig());
255 		}
256 	}
257 
258 	private static void mirrorHostSettings(PoolingHttpClientConnectionManager o, PoolingHttpClientConnectionManager p, HttpHost host) {
259 		var hostSocketConfig = o.getSocketConfig(host);
260 		if (nn(hostSocketConfig) && p.getSocketConfig(host) != hostSocketConfig)
261 			p.setSocketConfig(host, hostSocketConfig);
262 		var hostConnectionConfig = o.getConnectionConfig(host);
263 		if (nn(hostConnectionConfig) && p.getConnectionConfig(host) != hostConnectionConfig)
264 			p.setConnectionConfig(host, hostConnectionConfig);
265 	}

267 	private HttpClientConnectionManager owner(HttpClientConnection conn) {
268 		return pinnedLeases.containsKey(conn) ? pinned : ordinary;
269 	}

276 		private static final long serialVersionUID = 1L;

283 	/**
284 	 * The request interceptor that reads the {@link SsrfGuardMarker} of each request and hands the lease decision to
285 	 * the connection manager (see the class description).  Apache HttpClient runs it on the thread that then leases the
286 	 * connection, as part of the same call; it is set for every request, so a decision is never left over for another.
287 	 */
288 	static final class GuardInterceptor implements HttpRequestInterceptor {
289 		@Override /* HttpRequestInterceptor */
290 		public void process(HttpRequest request, HttpContext context) {
291 			if (SsrfGuardMarker.isGuarded(request, context))
292 				NEXT_LEASE_GUARDED.set(Boolean.TRUE);
293 			else
294 				NEXT_LEASE_GUARDED.remove();
295 		}
296 	}
297 
298 	/**
299 	 * The request executor of a client wired to this connection manager: it refuses to send a guarded request over a
300 	 * connection of the ordinary pool (an already-open one is reused without {@link #connect connect}), then hands every
301 	 * request to the executor the client was configured with.
302 	 */
303 	static final class GuardedRequestExecutor extends HttpRequestExecutor {
304 		private final HttpRequestExecutor delegate;
305 		private final SsrfGuardConnectionManager manager;
306 
307 		GuardedRequestExecutor(HttpRequestExecutor delegate, SsrfGuardConnectionManager manager) {
308 			// Another client's executor (a reused HttpClientBuilder keeps the one the previous build installed) is unwrapped.
309 			this.delegate = delegate instanceof GuardedRequestExecutor g ? g.delegate : or(delegate, new HttpRequestExecutor());
310 			this.manager = manager;
311 		}
312 
313 		@Override /* HttpRequestExecutor */
314 		public HttpResponse execute(HttpRequest request, HttpClientConnection conn, HttpContext context) throws IOException, HttpException {
315 			manager.requirePinnedIfGuarded(conn, request, context, "over an open connection of the ordinary pool");
316 			return delegate.execute(request, conn, context);
317 		}
318 	}
