17  package org.apache.juneau.rest.client.classic;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import org.apache.http.*;
22  import org.apache.http.client.methods.*;
23  import org.apache.http.protocol.*;
24  import org.apache.juneau.rest.client.classic.remote.*;
25  
26  /**
27   * The per-request marker of an SSRF-guard-active {@code @Remote} call.
28   *
29   * <p>
30   * A guarded {@link RestRequest} sets the {@link #ATTRIBUTE} attribute on the {@link HttpContext} of every request it
31   * sends (each redirect hop included).  The client's connection manager, request interceptor, request executor and
32   * redirect strategy read it from there, so the guard holds on whichever thread Apache HttpClient executes the request.
33   * The calling thread's {@link RemoteUrlPolicyState} is still honoured too, so a request made on that thread while a
34   * guarded call is running stays guarded.
35   *
36   * <p>
37   * It fails closed: a request created by a guarded {@link RestRequest} whose context lacks the marker (one a custom
38   * call handler sent with a different context) is refused, not sent unguarded.
39   *
40   * @since 10.0.0
41   */
42  final class SsrfGuardMarker {
43  
44  	/** The {@link HttpContext} attribute that marks a request as SSRF-guard-active; its value is {@link Boolean#TRUE}. */
45  	static final String ATTRIBUTE = "org.apache.juneau.rest.client.classic.ssrfGuard";
46  
47  	private SsrfGuardMarker() {}
48  
49  	/**
50  	 * Marks the given context as that of an SSRF-guard-active request.
51  	 *
52  	 * @param context The context.
53  	 * @return The previous value of the attribute, so the caller can restore it.
54  	 */
55  	static Object mark(HttpContext context) {
56  		var previous = context.getAttribute(ATTRIBUTE);
57  		context.setAttribute(ATTRIBUTE, Boolean.TRUE);
58  		return previous;
59  	}
60  
61  	/**
62  	 * Returns whether the request is SSRF-guard-active: its context carries the marker, or the calling thread is
63  	 * running a guarded call.
64  	 *
65  	 * @param request The request, or <jk>null</jk> to use the one on the context.
66  	 * @param context The context, or <jk>null</jk>.
67  	 * @return <jk>true</jk> if the request is guarded.
68  	 * @throws SsrfGuardConnectionManager.GuardRefusedException If a guarded {@link RestRequest} created the request but
69  	 * 	the context lacks the marker.
70  	 */
71  	@SuppressWarnings({
72  		"resource" // RestRequest is only inspected here; its lifecycle belongs to the caller.
73  	})
74  	static boolean isGuarded(HttpRequest request, HttpContext context) {
75  		if (nn(context) && Boolean.TRUE.equals(context.getAttribute(ATTRIBUTE)))
76  			return true;
77  		var r = request;
78  		if (r == null && nn(context))
79  			r = (HttpRequest)context.getAttribute(HttpCoreContext.HTTP_REQUEST);
80  		while (r instanceof HttpRequestWrapper w)
81  			r = w.getOriginal();
82  		if (r instanceof RestRequestCreated c && nn(c.getRestRequest()) && c.getRestRequest().isSsrfGuarded())
83  			throw new SsrfGuardConnectionManager.GuardRefusedException("Refusing to send a policy-covered @Remote request "
84  				+ "whose HttpContext lacks the SSRF-guard marker: it was sent with a context other than the one its RestRequest "
85  				+ "passed on (for example by a custom RestCallHandler), so the SSRF guardrail's connect-time contract "
86  				+ "(pin-on-connect + redirect revalidation) cannot be confirmed.  Pass the context through unchanged, or set "
87  				+ "allowPrivateUrls(true) if this is an intentional local-dev/intranet target.");
88  		return RemoteUrlPolicyState.isActive();
89  	}
90  }
