17  package org.apache.juneau.rest.client.classic;
18  
19  import java.util.function.*;
20  
21  import org.apache.juneau.marshall.stream.*;
22  
23  /**
24   * Wrapper that lets a caller pass a {@code Consumer<RecordWriter>} or
25   * {@code Consumer<TokenWriter>} as the body of a {@code @RemoteOp} request.
26   *
27   * <p>
28   * The framework resolves the matching serializer for the request's {@code Content-Type},
29   * opens a writer of the appropriate kind, and hands it to the caller's consumer.  When the
30   * consumer returns, the writer is closed and the request body is dispatched.
31   *
32   * <h5 class='section'>Usage:</h5>
33   * <p class='bjava'>
34   * 	<ja>@Remote</ja>
35   * 	<jk>public interface</jk> MyApi {
36   *
37   * 		<ja>@RemotePost</ja>(<js>"/bulk-upload"</js>)
38   * 		<jk>void</jk> upload(<ja>@Content</ja> RecordStreamBody <jv>body</jv>);
39   * 	}
40   *
41   * 	<jc>// Caller</jc>
42   * 	<jv>api</jv>.upload(RecordStreamBody.<jsm>records</jsm>(<jv>w</jv> -&gt; {
43   * 		<jk>for</jk> (Bean <jv>b</jv> : <jv>source</jv>())
44   * 			<jv>w</jv>.write(<jv>b</jv>);
45   * 	}));
46   * </p>
47   */
48  public final class RecordStreamBody {
49  
50  	/**
51  	 * Creates a body that streams via a {@link RecordWriter} (whole-value record cursor).
52  	 *
53  	 * @param consumer The callback that emits records.  Must not be <jk>null</jk>.
54  	 * @return A new body wrapper.
55  	 */
56  	public static RecordStreamBody records(Consumer<RecordWriter> consumer) {
57  		return new RecordStreamBody(consumer, RecordWriter.class);
58  	}
59  
60  	/**
61  	 * Creates a body that streams via a {@link TokenWriter} (fine-grained structural cursor).
62  	 *
63  	 * @param consumer The callback that emits structural events.  Must not be <jk>null</jk>.
64  	 * @return A new body wrapper.
65  	 */
66  	public static RecordStreamBody token(Consumer<TokenWriter> consumer) {
67  		return new RecordStreamBody(consumer, TokenWriter.class);
68  	}
69  
70  	private final Consumer<?> consumer;
71  	private final Class<?> writerKind;
72  
73  	private RecordStreamBody(Consumer<?> consumer, Class<?> writerKind) {
74  		this.consumer = consumer;
75  		this.writerKind = writerKind;
76  	}
77  
78  	/** @return The caller's consumer (typed as Consumer&lt;RecordWriter&gt; or Consumer&lt;TokenWriter&gt;). */
79  	@SuppressWarnings("java:S1452") // wildcard is intentional: field holds one of two unrelated Consumer types
80  	public Consumer<?> getConsumer() { return consumer; }
81  
82  	/** @return Either {@code RecordWriter.class} or {@code TokenWriter.class}. */
83  	@SuppressWarnings("java:S1452") // wildcard is intentional: field holds one of two unrelated Class tokens
84  	public Class<?> getWriterKind() { return writerKind; }
85  }
