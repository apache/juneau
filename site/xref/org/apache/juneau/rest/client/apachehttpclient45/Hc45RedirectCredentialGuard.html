26  /**
27   * Request interceptor that removes credential headers before a request is replayed against a
28   * different origin as the result of a {@code 3xx} redirect.
29   *
30   * <p>
31   * Apache HttpClient's redirect executor copies the original request's headers &mdash; including
32   * {@code Authorization} and {@code Cookie} &mdash; onto the follow-up request aimed at the {@code Location}
33   * target.  When that target is a different scheme, host, or port (or an {@code https}&rarr;{@code http}
34   * downgrade), forwarding those headers would disclose credentials to an unrelated origin.  This interceptor
35   * runs on every outgoing request in an exchange: it records the origin of the first request and strips the
36   * credential headers from any later request in the same exchange whose origin differs from it.  A request that
37   * returns to the first origin is not stripped, so it carries the original origin's credentials again.
38   *
39   * <p>
40   * The origin is compared on HttpClient's own {@link HttpHost} (scheme, host and effective port), with the same
41   * semantics as {@link RedirectSecurity#shouldStripCredentials(java.net.URI, java.net.URI)}.  {@link java.net.URI}
42   * cannot be used here: it parses a host name that HttpClient accepts but RFC 2396 does not (for example
43   * {@code my_service}, a typical Docker Compose service name) with a {@code null} host, which would make every
44   * redirect from such a host fail.  The credential header set is delegated to {@link RedirectSecurity#stripOnCrossOrigin()}
45   * so that all transports share it.
46   *
47   * <p>
48   * This interceptor is not registered on any client.  The transport passes {@code newGuardingContext()} with every
49   * request instead, so that transport-built and caller-supplied clients behave identically; see {@code GuardingContext}.
50   * On a hop whose origin is not the first one every credential header is removed, including the cookie-store {@code Cookie} and the
51   * credentials-provider or preemptive {@code Authorization} that HttpClient adds after this interceptor runs.
52   */
53  final class Hc45RedirectCredentialGuard implements HttpRequestInterceptor {
54  
55  	private static final String ATTR_ORIGIN = Hc45RedirectCredentialGuard.class.getName() + ".origin";
56  	private static final String ATTR_LEFT = Hc45RedirectCredentialGuard.class.getName() + ".left";
57  	private static final String ATTR_STRIPPED = Hc45RedirectCredentialGuard.class.getName() + ".stripped";
58  
59  	@Override /* HttpRequestInterceptor */
60  	public void process(HttpRequest request, HttpContext context) {
61  		var target = HttpCoreContext.adapt(context).getTargetHost();
62  		if (target == null)
63  			return;
64  		var origin = (HttpHost)context.getAttribute(ATTR_ORIGIN);
65  		if (origin == null) {
66  			context.setAttribute(ATTR_ORIGIN, target);
67  			return;
68  		}
69  		// Always compared with the first origin, not the previous hop.  HttpClient copies each hop's headers from the caller's
70  		// original request (this only strips the hop's own copy), so a hop that returns to the first origin is left whole.
71  		// A hop that returns to the first origin gets the credentials back only if it is a GET or HEAD: a 307/308 replay of a
72  		// POST (or any other method) after leaving the origin would be a credentialed write the caller never asked A to see.
73  		if (sameOrigin(origin, target)) {
74  			if (context.getAttribute(ATTR_LEFT) == null || safeMethod(request)) {
75  				context.removeAttribute(ATTR_STRIPPED);
76  				return;
77  			}
78  		} else {
79  			context.setAttribute(ATTR_LEFT, Boolean.TRUE);
80  		}
81  		context.setAttribute(ATTR_STRIPPED, Boolean.TRUE);
82  		for (var name : RedirectSecurity.stripOnCrossOrigin())
83  			request.removeHeaders(name);
84  	}
85  
86  	/**
87  	 * Returns a new per-request context that applies this guard to every hop of the exchange it is used for.
88  	 *
89  	 * @return A new context. Never <jk>null</jk>.
90  	 */
91  	static HttpClientContext newGuardingContext() {
92  		return new GuardingContext();
93  	}
94  
95  	/**
96  	 * Per-request context that applies the guard without changing the client it is used with.
97  	 *
98  	 * <p>
99  	 * The transport sends every request with one of these, whether it built the client or the caller supplied it, so both
100 	 * behave identically.  On a hop that is not at the exchange's first origin (exact scheme, host and effective port), <em>every</em>
101 	 * credential header is removed from the request that goes out, including the ones HttpClient adds itself: a {@code Cookie} from the client's
102 	 * {@code CookieStore}, and an {@code Authorization} from its {@code CredentialsProvider}, its auth cache (preemptive
103 	 * auth) or the answer to a {@code 401} challenge.  The only exception is {@code Proxy-Authorization}, which is addressed
104 	 * to the proxy rather than the origin.  Headers the client adds that are not credentials, such as
105 	 * {@code setDefaultHeaders(...)}, are left alone.
106 	 *
107 	 * <p>
108 	 * A request passes through two points that matter, both visible as attributes set on the context.  HttpClient's protocol
109 	 * stage sets the target host and then the request just before it runs the client's own request interceptors; the guard
110 	 * strips the headers the redirect executor copied from the caller's original request there.  The {@code MainClientExec}
111 	 * then adds its authentication headers, and the connection sets the request one last time immediately before it is
112 	 * written to the wire; the guard strips again there, if the hop is not at the first origin.
113 	 *
114 	 * <p>
115 	 * The first origin is the one captured at the start of the exchange; each hop is compared with it, never with the previous
116 	 * hop.  A redirect chain that returns to the first origin (A&rarr;B&rarr;A, A&rarr;B&rarr;C&rarr;A) gets the original origin's credentials back
117 	 * on that hop, exactly as an unredirected request to it would: the caller's own headers (HttpClient rebuilds each hop from
118 	 * the original request), the cookie-store {@code Cookie} and the credentials-provider or preemptive {@code Authorization}.
119 	 * A returning hop gets them back only when its method is {@code GET} or {@code HEAD}: once the exchange has left the first origin, a hop back to it by any other method (a {@code 307}/{@code 308} replay of a {@code POST}) stays stripped; a hop that never left the origin is unaffected.  Every hop that is not at the first origin is stripped, so A&rarr;B&rarr;A&rarr;B strips the second B.  The comparison is exact
120 	 * and fails closed: a different port, an {@code http}/{@code https} change, or a different spelling of the host (an
121 	 * IP literal versus its name, a trailing dot, a punycode name) is a different origin.
122 	 *
123 	 * <p>
124 	 * A hop can be passed more than once: HttpClient's retry executors ({@code RetryExec} after an I/O failure,
125 	 * {@code ServiceUnavailableRetryExec} after a 503) restore the headers they captured <em>before</em> the protocol stage
126 	 * ran -- the caller's -- and run the protocol stage again on the same request object.  So the guard must run on every
127 	 * pass, not once per request object.  A protocol-stage pass is recognised by the stage setting the target host
128 	 * immediately before the request.  A client that wraps the context it is given in a child context bypasses this guard, and so does a legacy {@code AbstractHttpClient} subclass ({@code DefaultHttpClient}), which never sets the request again before it is written; the builder refuses such a client.  Proxy-Authorization is not removed at the wire stage unless the route is tunnelled (inside a tunnel the proxy never sees the request, so one would reach the origin).
129 	 *
130 	 * <p>
131 	 * On a {@code 307}/{@code 308} redirect of a non-{@code GET} request, HttpClient copies the previous hop's headers (as sent) rather than the
132 	 * original request's, so caller-set headers are not brought back on a return to the first origin; credentials HttpClient adds itself still are.  The {@code Host} header also carries over from the previous hop.  A {@code CONNECT} tunnel request is never stripped: it carries no caller headers.
133 	 */
134 	private static final class GuardingContext extends HttpClientContext {
135 
136 		private final Hc45RedirectCredentialGuard guard = new Hc45RedirectCredentialGuard();
137 		private boolean protocolPass;
138 
139 		@Override /* HttpCoreContext */
140 		@SuppressWarnings({
141 			"java:S3776" // Linear attribute-observation state machine; splitting it would scatter the pass tracking.
142 		})
143 		public void setAttribute(String id, Object obj) {
144 			super.setAttribute(id, obj);
145 			if (HttpCoreContext.HTTP_TARGET_HOST.equals(id))
146 				protocolPass = true;
147 			else if (HttpCoreContext.HTTP_REQUEST.equals(id) && obj instanceof HttpRequest request) {
148 				if (protocolPass) {
149 					protocolPass = false;
150 					guard.process(request, this);
151 				} else if (getAttribute(ATTR_STRIPPED) != null && ! "CONNECT".equalsIgnoreCase(request.getRequestLine().getMethod())) {
152 					// Proxy-Authorization is addressed to the proxy, except inside a tunnel, where the proxy never sees this request.
153 					var route = getHttpRoute();
154 					var keepProxyAuth = route == null || ! route.isTunnelled();
155 					for (var name : RedirectSecurity.stripOnCrossOrigin())
156 						if (! (keepProxyAuth && "proxy-authorization".equalsIgnoreCase(name)))
157 							request.removeHeaders(name);
158 				}
159 			}
160 		}
161 	}
162 
163 	private static boolean safeMethod(HttpRequest request) {
164 		var method = request.getRequestLine().getMethod();
165 		return "GET".equalsIgnoreCase(method) || "HEAD".equalsIgnoreCase(method);
166 	}
167 
168 	// HttpHost always has a scheme name and a host name; the port is -1 when none was given, so it is normalized to the
169 	// scheme default.  An https->http downgrade changes the scheme, so it is never same-origin.
170 	private static boolean sameOrigin(HttpHost from, HttpHost to) {
171 		return lower(from.getSchemeName()).equals(lower(to.getSchemeName()))
172 			&& lower(from.getHostName()).equals(lower(to.getHostName()))
173 			&& effectivePort(from) == effectivePort(to);
174 	}
175 
176 	// Locale.ROOT lower-casing, not equalsIgnoreCase: that folds U+0131 (dotless i), U+0130 and U+017F onto ASCII letters, so a
177 	// non-ASCII host would otherwise compare equal to an ASCII one.
178 	private static String lower(String s) {
179 		return s.toLowerCase(Locale.ROOT);
180 	}
181 
182 	private static int effectivePort(HttpHost host) {
183 		var port = host.getPort();
184 		if (port != -1)
185 			return port;
186 		return switch (host.getSchemeName().toLowerCase(Locale.ROOT)) {
187 			case "https" -> 443;
188 			case "http" -> 80;
189 			default -> -1;
190 		};
191 	}
192 }
