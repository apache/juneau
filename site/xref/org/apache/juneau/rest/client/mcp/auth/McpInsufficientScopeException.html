17  package org.apache.juneau.rest.client.mcp.auth;
18  
19  import static org.apache.juneau.commons.utils.AssertionUtils.*;
20  
21  /**
22   * Signals that an MCP call was rejected with a SEP-2350 {@code 401}/{@code 403 insufficient_scope} step-up challenge.
23   *
24   * <p>
25   * A caller-supplied {@link McpStepUpAuthorizer.ScopedCall} throws this (carrying the parsed
26   * {@link WwwAuthenticateChallenge}) so {@link McpStepUpAuthorizer} can compute the scope union, re-authorize, and retry.
27   * It is deliberately <b>not</b> an {@link McpAuthException}: the transient step-up signal must be distinguishable from
28   * the permanent authorization failure the authorizer ultimately throws when retries are exhausted.
29   *
30   * @since 10.0.0
31   */
32  public class McpInsufficientScopeException extends RuntimeException {
33  
34  	private static final long serialVersionUID = 1L;
35  
36  	// Non-transient: WwwAuthenticateChallenge is Serializable (a scheme string + immutable string map), so the
37  	// challenge survives serialization and the challenge() "Never null" guarantee holds after a round-trip.
38  	private final WwwAuthenticateChallenge challenge;
39  
40  	/**
41  	 * Constructor.
42  	 *
43  	 * @param challenge The parsed {@code WWW-Authenticate} challenge from the server.  Must not be <jk>null</jk>.
44  	 */
45  	public McpInsufficientScopeException(WwwAuthenticateChallenge challenge) {
46  		super("MCP call rejected with an insufficient_scope step-up challenge: "
47  			+ assertArgNotNull("challenge", challenge).scopes());
48  		this.challenge = challenge;
49  	}
50  
51  	/**
52  	 * Returns the parsed challenge carrying the required scopes for the current operation.
53  	 *
54  	 * @return The challenge.  Never <jk>null</jk>.
55  	 */
56  	public WwwAuthenticateChallenge challenge() {
57  		return challenge;
58  	}
59  }
