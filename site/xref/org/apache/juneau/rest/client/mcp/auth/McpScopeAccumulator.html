17  package org.apache.juneau.rest.client.mcp.auth;
18  
19  import static org.apache.juneau.commons.utils.AssertionUtils.*;
20  
21  import java.net.*;
22  import java.util.*;
23  import java.util.concurrent.*;
24  
25  /**
26   * Computes and tracks the SEP-2350 step-up scope union &mdash; the set of scopes a client re-authorizes with after a
27   * server emits a per-operation {@code insufficient_scope} challenge.
28   *
29   * <p>
30   * The <b>mandated minimum</b> union (SEP-2350 "Step-Up Authorization Flow" step 2) is
31   * <code>{previously-requested} &cup; {challenge scopes}</code>, which preserves previously-granted permissions when
32   * servers emit per-operation challenges.  {@link #union(Collection, Collection)} computes exactly that.  The broader
33   * overload {@link #union(Collection, Collection, Collection, Collection, Collection)} additionally folds in the
34   * <i>opt-in</i> contributors the spec permits a client to consult (granted-token scopes, cached PRM
35   * {@code scopes_supported}, and client-metadata scope) &mdash; supplying them widens the union; omitting them keeps the
36   * default at the mandated minimum.
37   *
38   * <p>
39   * The union is <b>order-preserving</b> ({@link LinkedHashSet}) with <b>exact-string dedup only</b> &mdash; scopes are
40   * opaque strings, so {@code repo} and {@code repo:read} are both retained (no hierarchy-aware dedup client-side;
41   * hierarchy reasoning is the <i>server's</i> MUST).
42   *
43   * <p>
44   * As an instance, this class also maintains per-<code>(resource, issuer)</code> accumulated state so a step-up driver
45   * can carry the growing union across successive challenges for the same resource/AS.  Thread-safe.
46   *
47   * @since 10.0.0
48   */
49  @SuppressWarnings({
50  	"java:S115" // Constants use UPPER_snakeCase convention (e.g., ARG_resource)
51  })
52  public class McpScopeAccumulator {
53  
54  	// Argument name constants for assertArgNotNull
55  	private static final String ARG_resource = "resource";
56  	private static final String ARG_issuer = "issuer";
57  
58  	private final ConcurrentMap<String,Set<String>> byKey = new ConcurrentHashMap<>();
59  
60  	/**
61  	 * Computes the mandated-minimum step-up union: {@code previouslyRequested ∪ challengeScopes}, order-preserving with
62  	 * exact-string dedup.
63  	 *
64  	 * @param previouslyRequested The scope set the client previously requested (may be {@code null} / empty).
65  	 * @param challengeScopes The scopes from the server's {@code insufficient_scope} challenge (may be {@code null} /
66  	 * 	empty).
67  	 * @return A new order-preserving union set.  Never {@code null}.
68  	 */
69  	public static Set<String> union(Collection<String> previouslyRequested, Collection<String> challengeScopes) {
70  		return union(previouslyRequested, challengeScopes, null, null, null);
71  	}
72  
73  	/**
74  	 * Computes the broadened step-up union, folding in the opt-in contributors the SEP-2350 Scope Selection Strategy
75  	 * permits a client to consult.  Order-preserving; exact-string dedup.  Any {@code null} contributor is treated as
76  	 * empty.
77  	 *
78  	 * @param previouslyRequested The scope set the client previously requested.
79  	 * @param challengeScopes The scopes from the server's {@code insufficient_scope} challenge.
80  	 * @param grantedTokenScopes The scopes on the currently-held token (opt-in).
81  	 * @param prmScopesSupported The cached PRM {@code scopes_supported} (opt-in).
82  	 * @param clientMetadataScope The scope declared in client metadata (opt-in).
83  	 * @return A new order-preserving union set.  Never {@code null}.
84  	 */
85  	public static Set<String> union(Collection<String> previouslyRequested, Collection<String> challengeScopes,
86  			Collection<String> grantedTokenScopes, Collection<String> prmScopesSupported, Collection<String> clientMetadataScope) {
87  		var out = new LinkedHashSet<String>();
88  		addAll(out, previouslyRequested);
89  		addAll(out, challengeScopes);
90  		addAll(out, grantedTokenScopes);
91  		addAll(out, prmScopesSupported);
92  		addAll(out, clientMetadataScope);
93  		return out;
94  	}
95  
96  	/**
97  	 * Seeds the accumulated state for a {@code (resource, issuer)} key with the client's previously-requested scopes,
98  	 * <b>unioning</b> into any existing entry for that key rather than replacing it.
99  	 *
100 	 * <p>
101 	 * Unioning (not overwriting) matters when a shared accumulator is seeded more than once for the same key &mdash;
102 	 * e.g. two {@link McpStepUpAuthorizer.Builder#build() builds} against one shared accumulator, or a re-seed after some
103 	 * challenge scopes were already {@link #accumulate accumulated} &mdash; so no previously-tracked scope is ever lost.
104 	 *
105 	 * @param resource The RFC 8707 resource indicator.  Must not be <jk>null</jk>.
106 	 * @param issuer The authorization server issuer.  Must not be <jk>null</jk>.
107 	 * @param previouslyRequested The previously-requested scopes.  May be {@code null} / empty.
108 	 * @return This object.
109 	 */
110 	public McpScopeAccumulator seed(URI resource, URI issuer, Collection<String> previouslyRequested) {
111 		assertArgNotNull(ARG_resource, resource);
112 		assertArgNotNull(ARG_issuer, issuer);
113 		byKey.merge(key(resource, issuer), union(previouslyRequested, null), McpScopeAccumulator::union);
114 		return this;
115 	}
116 
117 	/**
118 	 * Unions the given challenge scopes into the accumulated state for the {@code (resource, issuer)} key and returns
119 	 * the updated union.
120 	 *
121 	 * @param resource The RFC 8707 resource indicator.  Must not be <jk>null</jk>.
122 	 * @param issuer The authorization server issuer.  Must not be <jk>null</jk>.
123 	 * @param challengeScopes The scopes from the latest {@code insufficient_scope} challenge.  May be {@code null} /
124 	 * 	empty.
125 	 * @return A copy of the updated accumulated union for the key.  Never {@code null}.
126 	 */
127 	public Set<String> accumulate(URI resource, URI issuer, Collection<String> challengeScopes) {
128 		assertArgNotNull(ARG_resource, resource);
129 		assertArgNotNull(ARG_issuer, issuer);
130 		var updated = byKey.compute(key(resource, issuer), (k, existing) -> union(existing, challengeScopes));
131 		return new LinkedHashSet<>(updated);
132 	}
133 
134 	/**
135 	 * Returns a copy of the currently-accumulated union for the {@code (resource, issuer)} key.
136 	 *
137 	 * @param resource The RFC 8707 resource indicator.  Must not be <jk>null</jk>.
138 	 * @param issuer The authorization server issuer.  Must not be <jk>null</jk>.
139 	 * @return A copy of the current accumulated union (empty if the key was never seeded/accumulated).
140 	 */
141 	public Set<String> current(URI resource, URI issuer) {
142 		assertArgNotNull(ARG_resource, resource);
143 		assertArgNotNull(ARG_issuer, issuer);
144 		return new LinkedHashSet<>(byKey.getOrDefault(key(resource, issuer), Set.of()));
145 	}
146 
147 	private static void addAll(Set<String> out, Collection<String> in) {
148 		if (in != null)
149 			for (var s : in)
150 				if (s != null && ! s.isBlank())
151 					out.add(s);
152 	}
153 
154 	private static String key(URI resource, URI issuer) {
155 		return resource.toString() + '\n' + issuer.toString();
156 	}
157 }
