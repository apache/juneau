26  /**
27   * A parsed {@code WWW-Authenticate: Bearer ...} challenge as returned by an OAuth 2.1 / MCP resource server on a
28   * {@code 401}/{@code 403} response (RFC 6750 &sect;3, RFC 9728 &sect;5.1).
29   *
30   * <p>
31   * The MCP client uses this to recover the {@code resource_metadata} pointer to the RFC 9728 PRM document, the
32   * per-operation {@code scope} on a step-up challenge, and the {@code error} code.  Auth-param values are matched
33   * case-insensitively on the key per RFC 7235.
34   *
35   * <p>
36   * {@link Serializable} (it is just a scheme string plus an immutable string map) so it can ride along a serialized
37   * {@link McpInsufficientScopeException} without being dropped.
38   *
39   * @since 10.0.0
40   */
41  public class WwwAuthenticateChallenge implements Serializable {
42  
43  	private static final long serialVersionUID = 1L;
44  
45  	private final String scheme;
46  	private final Map<String,String> parameters;
47  
48  	/**
49  	 * Constructor.
50  	 *
51  	 * @param scheme The auth scheme (e.g. {@code "Bearer"}).  Must not be <jk>null</jk> or blank.
52  	 * @param parameters The auth-params (lower-cased keys).  Must not be <jk>null</jk>.
53  	 */
54  	protected WwwAuthenticateChallenge(String scheme, Map<String,String> parameters) {
55  		this.scheme = assertArgNotNullOrBlank("scheme", scheme);
56  		this.parameters = u(cp(assertArgNotNull("parameters", parameters)));
57  	}
58  
59  	/**
60  	 * Parses a single {@code WWW-Authenticate} header value.
61  	 *
62  	 * <p>
63  	 * Recognizes a single challenge of the form {@code <scheme> <key>=<value>, <key>="<value>", ...}.  Both quoted
64  	 * and bare token values are accepted; commas inside quoted values are preserved.
65  	 *
66  	 * @param headerValue The raw header value.  May be <jk>null</jk> or blank.
67  	 * @return The parsed challenge, or {@link Optional#empty()} if the value is null/blank or has no scheme token.
68  	 */
69  	public static Optional<WwwAuthenticateChallenge> parse(String headerValue) {
70  		if (headerValue == null)
71  			return oe();
72  		var s = headerValue.strip();
73  		if (s.isEmpty())
74  			return oe();
75  		var sp = s.indexOf(' ');
76  		if (sp < 0)
77  			return o(new WwwAuthenticateChallenge(s, Map.of()));
78  		var scheme = s.substring(0, sp);
79  		var rest = s.substring(sp + 1).strip();
80  		return o(new WwwAuthenticateChallenge(scheme, parseParams(rest)));
81  	}
82  
83  	private static Map<String,String> parseParams(String s) {
84  		var out = new LinkedHashMap<String,String>();
85  		var n = s.length();
86  		var i = 0;
87  		while ((i = skipSeparators(s, i, n)) < n) {
88  			var param = parseParam(s, i, n);
89  			i = param.nextIndex();
90  			if (param.key() != null && !param.key().isEmpty())
91  				out.putIfAbsent(param.key(), param.value());
92  		}
93  		return out;
94  	}
95  
96  	/** Advances past whitespace and separator commas, returning the index of the next non-separator char (or {@code n}). */
97  	private static int skipSeparators(String s, int i, int n) {
98  		while (i < n && (Character.isWhitespace(s.charAt(i)) || s.charAt(i) == ','))
99  			i++;
100 		return i;
101 	}
102 
103 	/**
104 	 * Parses a single {@code key=value} auth-param starting at {@code i} (which must not point at a separator).
105 	 *
106 	 * <p>
107 	 * Returns a {@code null} key when {@code i} points at a bare token with no {@code =value} (e.g. {@code token68});
108 	 * the caller skips such a param.  {@code nextIndex()} always advances past the scanned key (and value, if any),
109 	 * so the outer loop makes forward progress without a {@code break}/{@code continue}.
110 	 */
111 	private static ParsedParam parseParam(String s, int i, int n) {
112 		var keyStart = i;
113 		while (i < n && s.charAt(i) != '=' && s.charAt(i) != ',')
114 			i++;
115 		if (i >= n || s.charAt(i) == ',')
116 			return new ParsedParam(null, null, i);
117 		var key = s.substring(keyStart, i).strip().toLowerCase(Locale.ROOT);
118 		i++; // consume '='
119 		var value = parseValue(s, i, n);
120 		return new ParsedParam(key, value.value(), value.nextIndex());
121 	}
122 
123 	/** Parses a quoted-string or bare-token auth-param value starting at {@code i}. */
124 	private static ParsedValue parseValue(String s, int i, int n) {
125 		if (i < n && s.charAt(i) == '"')
126 			return parseQuotedValue(s, i, n);
127 		var valStart = i;
128 		while (i < n && s.charAt(i) != ',')
129 			i++;
130 		return new ParsedValue(s.substring(valStart, i).strip(), i);
131 	}
132 
133 	/** Parses a quoted-string value starting at the opening-quote index {@code i}, honoring {@code \}-escaped chars. */
134 	private static ParsedValue parseQuotedValue(String s, int i, int n) {
135 		i++; // consume opening quote
136 		var sb = new StringBuilder();
137 		while (i < n && s.charAt(i) != '"') {
138 			if (s.charAt(i) == '\\' && i + 1 < n) // handle escaped chars inside quoted-string
139 				i++;
140 			sb.append(s.charAt(i));
141 			i++;
142 		}
143 		if (i < n)
144 			i++; // consume closing quote
145 		return new ParsedValue(sb.toString(), i);
146 	}
147 
148 	/** A parsed auth-param; {@code key} is {@code null} for a skipped bare token with no value. */
149 	private record ParsedParam(String key, String value, int nextIndex) {}
150 
151 	/** A parsed auth-param value and the index immediately following it. */
152 	private record ParsedValue(String value, int nextIndex) {}
153 
154 	/**
155 	 * Returns the auth scheme.
156 	 *
157 	 * @return The scheme (e.g. {@code "Bearer"}).  Never {@code null}.
158 	 */
159 	public String scheme() {
160 		return scheme;
161 	}
162 
163 	/**
164 	 * Returns whether the scheme is {@code Bearer} (case-insensitive).
165 	 *
166 	 * @return <jk>true</jk> if this is a Bearer challenge.
167 	 */
168 	public boolean isBearer() {
169 		return "bearer".equalsIgnoreCase(scheme);
170 	}
171 
172 	/**
173 	 * Returns the parsed auth-params (lower-cased keys).
174 	 *
175 	 * @return An unmodifiable map.  Never {@code null}.
176 	 */
177 	public Map<String,String> parameters() {
178 		return parameters;
179 	}
180 
181 	/**
182 	 * Returns the raw value of an auth-param.
183 	 *
184 	 * @param name The param name (matched case-insensitively).  Must not be <jk>null</jk>.
185 	 * @return The value, or {@link Optional#empty()} if absent.
186 	 */
187 	public Optional<String> parameter(String name) {
188 		assertArgNotNull("name", name);
189 		return o(parameters.get(name.toLowerCase(Locale.ROOT)));
190 	}
191 
192 	/**
193 	 * Returns the {@code resource_metadata} pointer (RFC 9728 &sect;5.1) to the PRM document.
194 	 *
195 	 * @return The PRM document URI, or {@link Optional#empty()} if absent or unparseable as a URI.
196 	 */
197 	public Optional<URI> resourceMetadata() {
198 		var v = parameters.get("resource_metadata");
199 		if (v == null || v.isBlank())
200 			return oe();
201 		try {
202 			return o(URI.create(v));
203 		} catch (IllegalArgumentException e) { // defensive: a malformed resource_metadata value (e.g. unescaped space) makes URI.create(...) throw
204 			return oe();
205 		}
206 	}
207 
208 	/**
209 	 * Returns the {@code error} code (e.g. {@code "insufficient_scope"}), if present.
210 	 *
211 	 * @return The error code, or {@link Optional#empty()}.
212 	 */
213 	public Optional<String> error() {
214 		return o(parameters.get("error"));
215 	}
216 
217 	/**
218 	 * Returns the space-delimited {@code scope} param as an order-preserving set.
219 	 *
220 	 * @return The challenged scopes (empty if the param is absent or blank).  Never {@code null}.
221 	 */
222 	public Set<String> scopes() {
223 		var v = parameters.get("scope");
224 		if (v == null || v.isBlank())
225 			return Set.of();
226 		var out = new LinkedHashSet<String>();
227 		for (var tok : v.strip().split("\\s+"))
228 			if (!tok.isEmpty()) // HTT: v is already known non-blank (guarded above) and pre-stripped, and split("\\s+")'s "+" quantifier collapses runs of delimiters into one match, so it cannot yield an empty leading/trailing/internal token for any String input (verified by exhaustive scan of all BMP chars where Character.isWhitespace(...) and the \s regex disagree) -- defensive-only
229 				out.add(tok);
230 		return u(out);
231 	}
