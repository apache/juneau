37  /**
38   * Performs an RFC 7591 / OpenID Connect Dynamic Client Registration (SEP-837) round-trip against an authorization
39   * server's {@code registration_endpoint}, returning the issued credentials as a redaction-safe
40   * {@link McpClientRegistration} keyed by the AS {@code issuer} (the SEP-2352 binding key).
41   *
42   * <p>
43   * The registration declares an {@code application_type} (SEP-837 <b>MUST</b>) &mdash; {@link McpApplicationType#NATIVE}
44   * for CLI/desktop/loopback clients, {@link McpApplicationType#WEB} for remote browser apps &mdash; along with the
45   * caller-supplied redirect URIs (see {@link LoopbackRedirectUris}), grant/response types, scope, and
46   * {@code token_endpoint_auth_method} ({@code none} for a public/native client by default;
47   * {@link Builder#confidential(boolean)} switches to {@code client_secret_basic}).  A protected registration endpoint's
48   * RFC 7591 &sect;3 initial access token is attached when {@link Builder#initialAccessToken(String)} is set.
49   *
50   * <p>
51   * The DCR-issued {@code client_id}/secret then feed F1's existing flows unchanged &mdash; DCR changes only how the
52   * {@code client_id} was obtained, not the {@code resource=} (RFC 8707) or {@code iss} (RFC 9207 / SEP-2468)
53   * enforcement; see {@link McpClientRegistrations} for the bridge to {@link McpAuthorizationCodeAcquirer} /
54   * {@link McpTokenProvider}.
55   *
56   * <p>
57   * {@link #register()} is the single public round-trip entry point.  Internally it is split into a request-build step and
58   * a response-parse step (both package-private, so the module's own tests can drive them without a network) &mdash; those
59   * helpers deliberately traffic in Nimbus {@code provided}-scope types and are <b>not</b> part of the public API.  A
60   * registration failure (e.g. {@code invalid_redirect_uri}) surfaces as an {@link McpAuthException} carrying the AS error
61   * code (SEP-837 "clients MUST be prepared to handle registration failures").
62   *
63   * <h5 class='topic'>Security</h5>
64   * <p>
65   * The registration endpoint MUST originate from validated AS metadata ({@link org.apache.juneau.rest.auth.oauth.oidc.OidcDiscoveryClient}
66   * enforces the RFC 8414 / OIDC issuer-identity check) and MUST be {@code https} (loopback exempt for local testing,
67   * matching {@code McpProtectedResourceMetadataClient.requireSecure}).  Client secrets are never logged &mdash;
68   * {@link #toString()} discloses only non-secret configuration and {@link McpClientRegistration} redacts its
69   * secret-bearing fields.
70   *
71   * @since 10.0.0
72   */

292 		/**
293 		 * Builds the registrar.
294 		 *
295 		 * @return A new {@link McpDynamicClientRegistrar}.
296 		 */
297 		public McpDynamicClientRegistrar build() {
298 			if (registrationEndpoint == null)
299 				throw isex("McpDynamicClientRegistrar requires registrationEndpoint(...)");
300 			if (issuer == null)
301 				throw isex("McpDynamicClientRegistrar requires issuer(...)");
302 			if (applicationType == null)
303 				throw isex("McpDynamicClientRegistrar requires applicationType(...) (SEP-837)");
304 			if (redirectUris.isEmpty())
305 				throw isex("McpDynamicClientRegistrar requires at least one redirectUri(...)");
306 			if (requireSecure && ! isSecureOrLoopback(registrationEndpoint))
307 				throw isex("McpDynamicClientRegistrar registration endpoint must use https (loopback exempt): " + registrationEndpoint);
308 			return new McpDynamicClientRegistrar(this);
309 		}

348 	/**
349 	 * Builds the Nimbus {@link OIDCClientRegistrationRequest} that {@link #register()} would send, declaring the
350 	 * {@code application_type}, redirect URIs, grant/response types, {@code token_endpoint_auth_method}, scope, and
351 	 * (if configured) the initial access token.
352 	 *
353 	 * <p>
354 	 * Package-private: returns a Nimbus {@code provided}-scope type, so it is an internal seam for the module's tests to
355 	 * inspect the exact request without a network round-trip &mdash; not part of the public API.
356 	 *
357 	 * @return The registration request.  Never <jk>null</jk>.
358 	 */
359 	OIDCClientRegistrationRequest buildRegistrationRequest() {
360 		var md = new OIDCClientMetadata();
361 		md.setApplicationType(applicationType.toNimbus());
362 		md.setRedirectionURIs(new LinkedHashSet<>(redirectUris));
363 		md.setGrantTypes(mapGrantTypes(grantTypes));
364 		md.setResponseTypes(mapResponseTypes(responseTypes));
365 		md.setTokenEndpointAuthMethod(confidential ? ClientAuthenticationMethod.CLIENT_SECRET_BASIC : ClientAuthenticationMethod.NONE);
366 		if (! scopes.isEmpty())
367 			md.setScope(Scope.parse(scopes));
368 		if (clientName != null)
369 			md.setName(clientName);
370 		var iat = initialAccessTokenSupplier == null ? null : new BearerAccessToken(initialAccessTokenSupplier.get());
371 		return new OIDCClientRegistrationRequest(registrationEndpoint, md, iat);
372 	}

374 	/**
375 	 * Parses a (possibly canned) registration {@link HTTPResponse} into an {@link McpClientRegistration}, stamping the
376 	 * configured {@code issuer}.
377 	 *
378 	 * <p>
379 	 * On a registration error the AS error object's code is surfaced as an {@link McpAuthException} (SEP-837
380 	 * registration-failure handling).
381 	 *
382 	 * <p>
383 	 * Package-private: accepts a Nimbus {@code provided}-scope type, so it is an internal seam for the module's tests to
384 	 * map a canned response &mdash; not part of the public API.
385 	 *
386 	 * @param httpResponse The registration response.  Must not be <jk>null</jk>.
387 	 * @return The mapped registration result.  Never <jk>null</jk>.
388 	 * @throws McpAuthException If the AS returned an error or the response could not be parsed.
389 	 */
390 	McpClientRegistration parseRegistrationResponse(HTTPResponse httpResponse) {
391 		assertArgNotNull("httpResponse", httpResponse);
392 		ClientRegistrationResponse resp;
393 		try {
394 			resp = OIDCClientRegistrationResponseParser.parse(httpResponse);
395 		} catch (ParseException e) {
396 			throw new McpAuthException("Dynamic client registration response could not be parsed", e);
397 		}
398 		if (! resp.indicatesSuccess()) {
399 			var err = resp.toErrorResponse().getErrorObject();
400 			var code = (err == null || err.getCode() == null) ? "unknown_error" : err.getCode();  // HTT: Nimbus's ErrorResponse#getErrorObject() never returns null (verified: even a bodiless/non-JSON error response yields a non-null ErrorObject with a null code), so the err==null branch is unreachable
401 			String desc = err == null ? null : err.getDescription();  // HTT: same as above -- err is never null here
402 			throw new McpAuthException("Dynamic client registration failed: " + code + (desc == null ? "" : " (" + desc + ")"));
403 		}
404 		return toRegistration(resp.toSuccessResponse().getClientInformation());
405 	}

407 	/**
408 	 * Performs the live RFC 7591 registration round-trip: builds the request, applies the finite HTTP timeout and the
409 	 * caller's {@code httpRequestConfigurator}, sends it, and maps the response.
410 	 *
411 	 * @return The issued registration.  Never <jk>null</jk>.
412 	 * @throws McpAuthException If the HTTP call fails or the AS returned an error.
413 	 */
414 	public McpClientRegistration register() {
415 		var http = buildRegistrationRequest().toHTTPRequest();
416 		var ms = (int) Math.min(httpTimeout.toMillis(), Integer.MAX_VALUE);
417 		http.setConnectTimeout(ms);
418 		http.setReadTimeout(ms);
419 		if (httpRequestConfigurator != null)
420 			httpRequestConfigurator.accept(http);
421 		HTTPResponse httpResp;
422 		try {
423 			httpResp = http.send();
424 		} catch (IOException e) {
425 			throw new McpAuthException("Dynamic client registration HTTP call failed", e);
426 		}
427 		return parseRegistrationResponse(httpResp);
428 	}

430 	private McpClientRegistration toRegistration(ClientInformation info) {
431 		var secret = info.getSecret();
432 		Optional<String> clientSecret = (secret == null || secret.getValue() == null) ? oe() : o(secret.getValue());  // HTT: Nimbus's ClientCredentialsParser rejects a blank/empty client_secret while parsing (throws ParseException before a Secret with a null value could ever be constructed), so secret!=null implies secret.getValue()!=null -- verified against oauth2-oidc-sdk 11.38.2
433 		Optional<Instant> secretExpires = (secret == null || secret.getExpirationDate() == null) ? oe() : o(secret.getExpirationDate().toInstant());
434 		var rat = info.getRegistrationAccessToken();
435 		Optional<String> regAccessToken = (rat == null || rat.getValue() == null) ? oe() : o(rat.getValue());  // HTT: same guarantee as client_secret above -- Nimbus rejects a blank registration_access_token during parsing, so rat!=null implies rat.getValue()!=null
436 		Optional<URI> regClientUri = info.getRegistrationURI() == null ? oe() : o(info.getRegistrationURI());
437 		return new McpClientRegistration(
438 			info.getID().getValue(),
439 			clientSecret,
440 			secretExpires,
441 			regAccessToken,
442 			regClientUri,
443 			issuer,
444 			redirectUris,
445 			applicationType,
446 			extrasFrom(info));
447 	}

449 	/** Surfaces any non-registered custom fields the AS returned in the registration response as the registration's extras. */
450 	private static Map<String,Object> extrasFrom(ClientInformation info) {
451 		var md = info.getMetadata();
452 		var custom = md == null ? null : md.getCustomFields();  // HTT: ClientInformation's constructor does Objects.requireNonNull(metadata), so a successfully-parsed ClientInformation's getMetadata() can never return null
453 		return (custom == null || custom.isEmpty()) ? Map.of() : new LinkedHashMap<>(custom);  // HTT: ClientMetadata#getCustomFields() is documented to return an empty JSONObject, never null, so custom==null is unreachable here too
454 	}

456 	private static Set<GrantType> mapGrantTypes(Set<String> v) {
457 		var out = new LinkedHashSet<GrantType>();
458 		for (var g : v) {
459 			try {
460 				out.add(GrantType.parse(g));
461 			} catch (ParseException e) {  // HTT: catch-block entry is unreachable for the same reason as the throw below -- GrantType.parse never throws for a value that has already passed assertArgNotNullOrBlank
462 				throw iaex(e, "Invalid grant_type '%s'", g);  // HTT: GrantType.parse only throws ParseException for a null/blank value, already rejected by addGrantType's assertArgNotNullOrBlank guard
463 			}
464 		}
465 		return out;
466 	}
467 
468 	private static Set<ResponseType> mapResponseTypes(Set<String> v) {
469 		var out = new LinkedHashSet<ResponseType>();
470 		for (var r : v) {
471 			try {
472 				out.add(ResponseType.parse(r));
473 			} catch (ParseException e) {  // HTT: catch-block entry is unreachable for the same reason as the throw below -- ResponseType.parse never throws for a value that has already passed assertArgNotNullOrBlank
474 				throw iaex(e, "Invalid response_type '%s'", r);  // HTT: ResponseType.parse only throws ParseException for a null/blank value, already rejected by addResponseType's assertArgNotNullOrBlank guard
475 			}
476 		}
477 		return out;
478 	}
