35  /**
36   * Fetches and parses an RFC 9728 Protected Resource Metadata (PRM) document, and resolves the authorization server a
37   * protected MCP resource delegates to.
38   *
39   * <p>
40   * This is the client half of the OAuth 2.1 baseline PRM discovery required by the MCP {@code 2026-07-28} revision: on a
41   * {@code 401 WWW-Authenticate} challenge the client recovers the {@code resource_metadata} pointer
42   * ({@link WwwAuthenticateChallenge#resourceMetadata()}), fetches the PRM document here, selects an authorization server
43   * ({@link McpProtectedResourceMetadata#firstAuthorizationServer()}), and validates that server's issuer identity via
44   * {@link OidcDiscoveryClient} (RFC 8414 / OIDC Discovery).
45   *
46   * <p>
47   * When an expected resource identifier is supplied ({@link Builder#expectedResource(URI)} or the {@code expectedResource}
48   * argument of {@link #fetch(URI, URI)} / {@link #parse(String, URI, URI)}), the PRM document's {@code resource} field is
49   * validated against it per RFC 9728 &sect;3.3 (exact string match, mirroring the issuer-identity posture) and a mismatch
50   * is rejected &mdash; this defends against a malicious server pointing the client at a PRM document minted for a
51   * different resource.  When no expected resource is supplied the identity check is skipped.
52   *
53   * <p>
54   * By default the PRM URL and any discovered authorization-server URI must use {@code https} (loopback hosts are always
55   * permitted for local/testing use); {@link Builder#allowInsecureHttp(boolean)} disables the scheme check entirely.
56   *
57   * @since 10.0.0
58   */
59  @SuppressWarnings({
60  	"java:S115" // Constants use UPPER_snakeCase convention (e.g., ARG_value, PROP_resource)
61  })

99  		 *
100 		 * <p>
101 		 * This is the canonical URI of the MCP server the client was actually talking to.  A per-call value passed to
102 		 * {@link #fetch(URI, URI)} / {@link #parse(String, URI, URI)} overrides this.
103 		 *
104 		 * @param value The expected resource identifier.  Must not be <jk>null</jk>.
105 		 * @return This object.
106 		 */
107 		public Builder expectedResource(URI value) {
108 			expectedResource = assertArgNotNull(ARG_value, value);
109 			return this;
110 		}

112 		/**
113 		 * Sets the connect/read timeout applied to the PRM fetch and AS discovery.  Default 10 seconds.
114 		 *
115 		 * @param value The timeout.  Must not be <jk>null</jk> and must be positive.

117 		 */
118 		public Builder httpTimeout(Duration value) {
119 			assertArgNotNull(ARG_value, value);
120 			assertArg(!value.isZero() && !value.isNegative(), "httpTimeout must be positive (was %s)", value);

123 		}

174 	/**
175 	 * Fetches and parses the PRM document at the supplied URL, validating its {@code resource} field against the
176 	 * builder-configured {@link Builder#expectedResource(URI)} (if any).
177 	 *
178 	 * @param prmUrl The PRM document URL (typically the {@code resource_metadata} pointer from a {@code 401}
179 	 * 	challenge, or the well-known URL for the resource).  Must not be <jk>null</jk>.
180 	 * @return The parsed metadata.
181 	 * @throws McpAuthException If the fetch fails, the document cannot be parsed, or the resource identity check fails.
182 	 */
183 	public McpProtectedResourceMetadata fetch(URI prmUrl) {
184 		return fetch(prmUrl, expectedResource);
185 	}
186 
187 	/**
188 	 * Fetches and parses the PRM document at the supplied URL, validating its {@code resource} field against the
189 	 * supplied expected resource identifier (RFC 9728 &sect;3.3).
190 	 *
191 	 * @param prmUrl The PRM document URL.  Must not be <jk>null</jk>.
192 	 * @param expectedResource The resource identifier the client was actually talking to.  May be <jk>null</jk> to
193 	 * 	skip the identity check.
194 	 * @return The parsed metadata.
195 	 * @throws McpAuthException If the fetch fails, the document cannot be parsed, or the resource identity check fails.
196 	 */
197 	public McpProtectedResourceMetadata fetch(URI prmUrl, URI expectedResource) {
198 		assertArgNotNull("prmUrl", prmUrl);
199 		requireSecure(prmUrl, "Protected-resource-metadata URL");
200 		HTTPResponse resp;
201 		try {
202 			var req = new HTTPRequest(HTTPRequest.Method.GET, prmUrl.toURL());
203 			req.setConnectTimeout((int) httpTimeout.toMillis());
204 			req.setReadTimeout((int) httpTimeout.toMillis());
205 			req.setAccept("application/json");
206 			if (httpRequestConfigurator != null)
207 				httpRequestConfigurator.accept(req);
208 			resp = req.send();
209 		} catch (IOException e) {
210 			throw new McpAuthException("Protected-resource-metadata fetch failed for " + prmUrl, e);
211 		}
212 		if (!resp.indicatesSuccess())
213 			throw new McpAuthException("Protected-resource-metadata fetch returned HTTP " + resp.getStatusCode() + " for " + prmUrl);
214 		return parse(resp.getBody(), prmUrl, expectedResource);
215 	}

219 	 * {@link Builder#expectedResource(URI)} (if any).

222 	 * @param source The source URI, used only for error messages.  May be <jk>null</jk>.

224 	 * @throws McpAuthException If the document cannot be parsed, is missing the required {@code resource} field, or

228 		return parse(json, source, expectedResource);

231 	/**
232 	 * Parses a PRM JSON document body, validating its {@code resource} field against the supplied expected resource
233 	 * identifier (RFC 9728 &sect;3.3).
234 	 *
235 	 * @param json The JSON document body.  Must not be <jk>null</jk>.
236 	 * @param source The source URI, used only for error messages.  May be <jk>null</jk>.
237 	 * @param expectedResource The resource identifier the client was actually talking to.  May be <jk>null</jk> to
238 	 * 	skip the identity check.
239 	 * @return The parsed metadata.
240 	 * @throws McpAuthException If the document cannot be parsed, is missing the required {@code resource} field, or
241 	 * 	fails the resource identity check.
242 	 */
243 	public McpProtectedResourceMetadata parse(String json, URI source, URI expectedResource) {
244 		assertArgNotNull("json", json);
245 		JSONObject o;
246 		try {
247 			o = JSONObjectUtils.parse(json);
248 		} catch (ParseException e) {
249 			throw new McpAuthException("Protected-resource-metadata document could not be parsed for " + source, e);
250 		}
251 		McpProtectedResourceMetadata prm;
252 		try {
253 			var resource = JSONObjectUtils.getURI(o, PROP_resource);
254 			var authServers = new ArrayList<URI>();
255 			if (o.containsKey(PROP_authorizationServers))
256 				for (var v : JSONObjectUtils.getStringList(o, PROP_authorizationServers))
257 					authServers.add(URI.create(v));
258 			var scopes = new LinkedHashSet<String>();
259 			if (o.containsKey(PROP_scopesSupported))
260 				scopes.addAll(JSONObjectUtils.getStringList(o, PROP_scopesSupported));
261 			var extras = new LinkedHashMap<String,Object>();
262 			for (var e : o.entrySet())
263 				if (!STANDARD_FIELDS.contains(e.getKey()))
264 					extras.put(e.getKey(), e.getValue());
265 			prm = new McpProtectedResourceMetadata(resource, authServers, scopes, extras);
266 		} catch (ParseException | IllegalArgumentException e) {
267 			throw new McpAuthException("Protected-resource-metadata document was malformed for " + source, e);
268 		}
269 		validateResourceIdentity(prm, expectedResource);
270 		return prm;
271 	}
272 
273 	/**
274 	 * Selects the first authorization server from the PRM document and validates its issuer identity via RFC 8414 /
275 	 * OIDC discovery.
276 	 *
277 	 * @param prm The parsed PRM document.  Must not be <jk>null</jk>.
278 	 * @return The discovered authorization-server metadata (issuer validated).
279 	 * @throws McpAuthException If no authorization server is advertised or discovery/issuer validation fails.
280 	 */
281 	public OidcMetadata discoverAuthorizationServer(McpProtectedResourceMetadata prm) {
282 		assertArgNotNull("prm", prm);
283 		var as = prm.firstAuthorizationServer()
284 			.orElseThrow(() -> new McpAuthException("Protected-resource-metadata for " + prm.resource() + " advertised no authorization_servers"));
285 		requireSecure(as, "Authorization-server URL");
286 		try {
287 			return OidcDiscoveryClient.create()
288 				.issuer(as)
289 				.httpTimeout(httpTimeout)
290 				.httpRequestConfigurator(httpRequestConfigurator == null ? h -> {} : httpRequestConfigurator)
291 				.build()
292 				.discover();
293 		} catch (IOException | OidcDiscoveryException e) {
294 			throw new McpAuthException("Authorization-server discovery/issuer validation failed for " + as, e);
295 		}
296 	}
297 
298 	/**
299 	 * Validates the PRM {@code resource} field against the expected resource identifier (RFC 9728 &sect;3.3), using a
300 	 * normalization-free exact string comparison to match the issuer-identity posture.
301 	 */
302 	private static void validateResourceIdentity(McpProtectedResourceMetadata prm, URI expectedResource) {
303 		if (expectedResource == null)
304 			return;
305 		if (!expectedResource.toString().equals(prm.resource().toString()))
306 			throw new McpAuthException("Protected-resource-metadata resource identity mismatch (RFC 9728 3.3): expected '"
307 				+ expectedResource + "' but document declared '" + prm.resource() + "'");
308 	}
309 
310 	private void requireSecure(URI uri, String what) {
311 		if (allowInsecureHttp || "https".equalsIgnoreCase(uri.getScheme()) || isLoopback(uri))
312 			return;
313 		throw new McpAuthException(what + " must use https (was '" + uri + "'); enable allowInsecureHttp for non-loopback http");
314 	}
315 
316 	private static boolean isLoopback(URI uri) {
317 		var host = uri.getHost();
318 		if (host == null)
319 			return false;
320 		return "localhost".equalsIgnoreCase(host) || "127.0.0.1".equals(host) || "[::1]".equals(host) || "::1".equals(host); // HTT: java.net.URI#getHost() always brackets an IPv6 literal (verified: even the raw-host 3-arg URI constructor auto-brackets "::1" into "[::1]"), so the bare unbracketed "::1" form this last disjunct guards against is unreachable through any standard URI construction path -- defensive-only
321 	}
