17  package org.apache.juneau.rest.client.mcp;
18  
19  import static org.apache.juneau.commons.utils.AssertionUtils.*;
20  
21  import java.util.function.*;
22  
23  import org.apache.juneau.rest.client.*;
24  
25  /**
26   * A builder-configurable auth-token seam implemented as a {@link RestCallInterceptor}.
27   *
28   * <p>
29   * On {@link #onInit(RestRequest)}, sets the {@code Authorization} request header to {@code "Bearer " + token},
30   * where {@code token} comes from a caller-supplied {@link Supplier}. Sub-project F's OAuth/OIDC flows plug into
31   * this seam by supplying their own token-supplier implementation (e.g. one that refreshes an expired access
32   * token); this class implements no OAuth/OIDC flow itself.
33   *
34   * <p>
35   * If the supplied token is <jk>null</jk> <b>or blank</b> (empty or all-whitespace), no {@code Authorization}
36   * header is set for that request (the call proceeds unauthenticated) - a blank token is never sent as a
37   * credential-less {@code Authorization: Bearer } header. If the token supplier itself throws,
38   * {@link RestRequest#run()}'s documented interceptor-exception handling aborts the call before it is sent.
39   *
40   * <p>
41   * <b>Note:</b> when a token is set, {@link RestRequest#debug()} logs the full outgoing request including all
42   * headers, so enabling debug logging on a client using this interceptor will log the {@code Authorization}
43   * header (and therefore the bearer token) in plain text.
44   *
45   * @since 10.0.0
46   */
47  public class McpAuthInterceptor implements RestCallInterceptor {
48  
49  	// Argument name constants for assertArgNotNull
50  	private static final String ARG_TOKEN_SUPPLIER = "tokenSupplier";
51  	private static final String ARG_TOKEN = "token";
52  
53  	private final Supplier<String> tokenSupplier;
54  
55  	/**
56  	 * Constructor.
57  	 *
58  	 * @param tokenSupplier Supplies the bearer token for each request. Must not be <jk>null</jk>.
59  	 */
60  	public McpAuthInterceptor(Supplier<String> tokenSupplier) {
61  		this.tokenSupplier = assertArgNotNull(ARG_TOKEN_SUPPLIER, tokenSupplier);
62  	}
63  
64  	/**
65  	 * Creates an interceptor that always sends the same static bearer token.
66  	 *
67  	 * @param token The static bearer token. Must not be <jk>null</jk>.
68  	 * @return A new interceptor. Never <jk>null</jk>.
69  	 */
70  	public static McpAuthInterceptor ofStaticBearer(String token) {
71  		assertArgNotNull(ARG_TOKEN, token);
72  		return new McpAuthInterceptor(() -> token);
73  	}
74  
75  	@Override /* Overridden from RestCallInterceptor */
76  	public void onInit(RestRequest req) throws Exception {
77  		var token = tokenSupplier.get();
78  		if (token != null && ! token.isBlank())
79  			req.header("Authorization", "Bearer " + token);
80  	}
81  }
