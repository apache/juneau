17  package org.apache.juneau.rest.client.mcp.v20260728;
18  
19  import java.io.*;
20  
21  /**
22   * Closeable handle for a live {@link McpClient#listen(org.apache.juneau.bean.mcp.v20260728.SubscriptionFilter, McpSubscriptionListener)}
23   * subscription.
24   *
25   * @since 10.0.0
26   */
27  public interface McpSubscriptionHandle extends Closeable {
28  
29  	/**
30  	 * The client-generated JSON-RPC request id this subscription's opening
31  	 * {@link McpClient#listen(org.apache.juneau.bean.mcp.v20260728.SubscriptionFilter, McpSubscriptionListener)}
32  	 * call sent, and which the server echoes back as the {@code subscriptionId} on every frame of this stream
33  	 * (the acknowledged frame, each change notification, and the terminal frame). Callers can use this to
34  	 * correlate this handle against server-side logs/traces for the same subscription.
35  	 *
36  	 * @return The request id. Never <jk>null</jk> or empty.
37  	 */
38  	String id();
39  
40  	/**
41  	 * Cancels the subscription: signals the background pump to stop and closes the underlying stream. Idempotent.
42  	 *
43  	 * <p>
44  	 * Asynchronous: this method returns as soon as the stop signal is issued and the stream close is requested,
45  	 * <i>without</i> waiting for the background pump thread to actually finish running. A caller that needs to
46  	 * know the pump has fully stopped must poll {@link #isOpen()} or otherwise synchronize with it separately —
47  	 * this method does not join it.
48  	 */
49  	void cancel();
50  
51  	/**
52  	 * Alias for {@link #cancel()}, satisfying {@link Closeable} for try-with-resources use.
53  	 *
54  	 * <p>
55  	 * Deliberately declares no checked {@link IOException} (narrows {@link Closeable#close()}'s throws clause) —
56  	 * the background pump owns the stream's lifecycle and closing it is always a local, non-throwing operation.
57  	 *
58  	 * <p>
59  	 * Like {@link #cancel()}, this is asynchronous: it signals the pump to stop but does not wait for it to exit.
60  	 */
61  	@Override
62  	void close();
63  
64  	/**
65  	 * @return <jk>true</jk> if the subscription is still open (not cancelled, errored, or gracefully completed).
66  	 */
67  	boolean isOpen();
68  }
