33  /**
34   * Revision-neutral base class for Model Context Protocol (MCP) clients.
35   *
36   * <p>
37   * Wraps a single configured {@link RestClient} and executes raw JSON-RPC 2.0 request/response round trips against one
38   * endpoint in JSON response mode. A generic {@link RestCallInterceptor} passthrough is provided on the builder so
39   * per-revision facades can layer auth and other cross-cutting concerns without changing this core.
40   *
41   * <p>
42   * {@link #send(JsonRpcRequest)} returns the complete {@link JsonRpcResponse} envelope untouched, including
43   * {@code error} responses - it does not throw {@link McpException} on a JSON-RPC error. Typed facades are
44   * responsible for translating a response with a non-<jk>null</jk> {@link JsonRpcResponse#getError() error} into a
45   * thrown {@link McpException}.
46   *
47   * <p>
48   * Instances are thread-safe for concurrent reuse: {@link RestClient} and endpoint are final, and each
49   * {@link #send(JsonRpcRequest)} creates a fresh request/response scope. Builders are single-use value assemblers:
50   * calling {@code build()} twice creates two clients and both use the same explicitly configured transport instance.
51   *
52   * <p>
53   * <b>Beta - API subject to change:</b> This class builds directly on {@code org.apache.juneau.rest.client}, which
54   * is itself Beta.
55   *
56   * @since 10.0.0
57   */
58  public abstract class AbstractMcpClient implements Closeable {

66  	@SuppressWarnings("resource") // closed by this.close() (Closeable); Eclipse can't trace field-scoped ownership across construction and close().

70  	/**
71  	 * Constructor.
72  	 *
73  	 * @param builder The builder supplying this client's configuration. Must not be <jk>null</jk>.
74  	 * @throws IllegalArgumentException If {@code builder} is <jk>null</jk>, or its endpoint is <jk>null</jk>/blank
75  	 * 	or is not a syntactically valid absolute {@code http}/{@code https} URL (see
76  	 * 	{@link #validateEndpoint(String)}).
77  	 */
78  	@SuppressWarnings("resource") // the built RestClient is stored in the final restClient field and closed by close(); not leaked.
79  	protected AbstractMcpClient(Builder<?> builder) {
80  		assertArgNotNull(ARG_BUILDER, builder);
81  		this.endpoint = validateEndpoint(builder.endpoint);
82  		this.restClient = builder.restClientBuilder
83  			.defaultSerializer(JsonSerializer.DEFAULT)
84  			.defaultParser(JsonParser.DEFAULT)
85  			.build();
86  	}

88  	/**
89  	 * Validates the configured endpoint: non-<jk>null</jk>/non-blank (as before), and additionally that it
90  	 * parses as a syntactically valid, absolute URL using the {@code http} or {@code https} scheme.
91  	 *
92  	 * <p>
93  	 * Rejecting a malformed or non-http(s) endpoint here, at construction, is preferable to letting it surface
94  	 * later as an opaque transport failure on the first {@link #send(JsonRpcRequest)} call.
95  	 *
96  	 * @param value The candidate endpoint. Must not be <jk>null</jk>/blank.
97  	 * @return {@code value}, unchanged, once validated.
98  	 * @throws IllegalArgumentException If {@code value} is <jk>null</jk>/blank, is not a syntactically valid URI,
99  	 * 	has no scheme, or has a scheme other than {@code http}/{@code https} (case-insensitive).
100 	 */
101 	private static String validateEndpoint(String value) {
102 		var endpointValue = assertArgNotNullOrBlank(ARG_ENDPOINT, value);
103 		URI uri;
104 		try {
105 			uri = new URI(endpointValue);
106 		} catch (URISyntaxException e) {
107 			throw new IllegalArgumentException("Invalid MCP endpoint URL '" + endpointValue + "': " + e.getMessage(), e);
108 		}
109 		var scheme = uri.getScheme();
110 		if (scheme == null || ! (scheme.equalsIgnoreCase("http") || scheme.equalsIgnoreCase("https")))
111 			throw new IllegalArgumentException(
112 				"Invalid MCP endpoint URL '" + endpointValue + "': must be an absolute http or https URL, but scheme was '" + scheme + "'.");
113 		return endpointValue;
114 	}

116 	/**
117 	 * Sends a raw JSON-RPC request and returns the raw JSON-RPC response envelope.
118 	 *
119 	 * <p>
120 	 * A request with a <jk>null</jk> {@link JsonRpcRequest#getId() id} is a notification: this method still sends
121 	 * it over the wire, but always returns <jk>null</jk> without attempting to read a response body, per the
122 	 * JSON-RPC 2.0 notification contract (the server sends no response body for a notification).
123 	 *
124 	 * @param request The JSON-RPC request to send. Must not be <jk>null</jk>.
125 	 * @return The JSON-RPC response envelope, or <jk>null</jk> if {@code request} is a notification.
126 	 * @throws IOException If a transport-level or (de)serialization error occurs.
127 	 */
128 	public JsonRpcResponse send(JsonRpcRequest request) throws IOException {
129 		return send(request, Map.of());
130 	}

132 	/**
133 	 * Sends a raw JSON-RPC request and returns the raw JSON-RPC response envelope.
134 	 *
135 	 * <p>
136 	 * A request with a <jk>null</jk> {@link JsonRpcRequest#getId() id} is a notification: this method still sends
137 	 * it over the wire, but always returns <jk>null</jk> without attempting to read a response body, per the
138 	 * JSON-RPC 2.0 notification contract (the server sends no response body for a notification).
139 	 *
140 	 * <p>
141 	 * <b>Note:</b> for a notification, this method returns <jk>null</jk> immediately after the transport call
142 	 * completes, without ever inspecting {@link RestResponse#getStatusCode()}. A non-2xx HTTP status on a
143 	 * notification is therefore silently swallowed - the caller sees no indication that the server rejected it.
144 	 * This is intentional per the notification contract above (there is no response envelope to report an error
145 	 * through), but it does mean transport-level failures on notifications are otherwise invisible to callers.
146 	 *
147 	 * @param request The JSON-RPC request to send. Must not be <jk>null</jk>.
148 	 * @param httpHeaders Optional additional HTTP headers to add on this request. Can be <jk>null</jk>.
149 	 * @return The JSON-RPC response envelope, or <jk>null</jk> if {@code request} is a notification.
150 	 * @throws IOException If a transport-level or (de)serialization error occurs.
151 	 */
152 	public JsonRpcResponse send(JsonRpcRequest request, Map<String,String> httpHeaders) throws IOException {
153 		assertArgNotNull(ARG_REQUEST, request);
154 		var req = restClient.post(endpoint).body(request);
155 		if (httpHeaders != null) {
156 			for (var e : httpHeaders.entrySet())
157 				req.header(e.getKey(), e.getValue());
158 		}
159 		try (var res = req.run()) {
160 			if (JsonRpcResponse.notification(request.getId()))
161 				return null;
162 			var sc = res.getStatusCode();
163 			JsonRpcResponse parsed;
164 			try {
165 				parsed = res.body().as(JsonRpcResponse.class);
166 			} catch (IOException e) {
167 				if (sc < 200 || sc > 299)
168 					throw ioex(e, "MCP server returned HTTP %s and the body was not a JSON-RPC envelope.", sc);
169 				throw e;
170 			}
171 			if (parsed == null)
172 				throw ioex("No response body received for JSON-RPC request id '%s' (HTTP %s).", request.getId(), sc);
173 			return parsed;
174 		} // HTT: javac's try-with-resources close inserts a defensive null-check on 'res' before calling close(); RestRequest#run() never returns null (it always returns a fresh RestResponse or throws), so the res==null arm is unreachable
175 	}

178 	 * Opens a Server-Sent-Events stream on the configured endpoint, for revision-specific facades that support a
179 	 * duplex (server-initiated) channel alongside the request/response {@link #send} calls.
180 	 *
181 	 * @return A reader over the opened event stream. Never <jk>null</jk>. Callers are responsible for closing it.
182 	 * @throws IOException If a transport-level error occurs opening the stream.
183 	 */
184 	@SuppressWarnings("resource") // returned SseEventReader wraps the opened stream; ownership transfers to the caller, who must close it (see javadoc).
185 	protected SseEventReader openEventStream() throws IOException {
186 		return restClient.post(endpoint).openEventStream();
187 	}

189 	/**
190 	 * Opens a Server-Sent-Events stream on the configured endpoint by first POSTing {@code request} as the request
191 	 * body — for a revision-specific method (e.g. {@code subscriptions/listen}) whose held-open SSE response is
192 	 * itself the answer to one specific JSON-RPC request, as opposed to {@link #openEventStream()}'s body-less
193 	 * open used by the duplex server-push channel.
194 	 *
195 	 * @param request The JSON-RPC request to POST as the body that opens the stream. Must not be <jk>null</jk>.
196 	 * @return A reader over the opened event stream. Never <jk>null</jk>. Callers are responsible for closing it.
197 	 * @throws IOException If a transport-level or (de)serialization error occurs opening the stream.
198 	 */
199 	@SuppressWarnings("resource") // delegates to the 3-arg overload, which transfers stream ownership to the caller.
200 	protected SseEventReader openEventStream(JsonRpcRequest request) throws IOException {
201 		return openEventStream(request, Map.of());
202 	}

204 	/**
205 	 * Opens a Server-Sent-Events stream on the configured endpoint by first POSTing {@code request} as the request
206 	 * body, with additional HTTP headers on the opening request — for a revision-specific method whose held-open
207 	 * SSE response must agree with a header-based method/name contract (e.g. SEP-2243's {@code Mcp-Method}/
208 	 * {@code Mcp-Name}) the same way {@link #send(JsonRpcRequest, Map)} already does for the ordinary
209 	 * request/response path.
210 	 *
211 	 * <p>
212 	 * Unlike the body-less {@link #openEventStream()}, this opening POST is itself the answer to one specific
213 	 * JSON-RPC request, so the server may reject it with a JSON-RPC error envelope instead of opening the stream
214 	 * (e.g. a capability, {@code Accept}-header, or over-limit gate) — on any HTTP status, not just 2xx. Rather
215 	 * than handing that envelope to an {@link SseEventReader} (which would see zero events and eventually look
216 	 * like an indistinguishable transport failure), this checks the response {@code Content-Type} first: only a
217 	 * body reporting {@code text/event-stream} is opened as a stream. Anything else is parsed as a
218 	 * {@link JsonRpcResponse}; a non-<jk>null</jk> {@link JsonRpcResponse#getError() error} is thrown synchronously
219 	 * as an {@link McpException} carrying the server's code/message/data, matching the way mid-stream JSON-RPC
220 	 * errors are already surfaced to a stream's listener. A response that is neither {@code text/event-stream} nor
221 	 * a JSON-RPC error envelope still fails as a plain {@link IOException} (or the underlying
222 	 * {@link RestCallException} from a non-2xx status), never silently masquerading as an opened stream.
223 	 *
224 	 * @param request The JSON-RPC request to POST as the body that opens the stream. Must not be <jk>null</jk>.
225 	 * @param httpHeaders Optional additional HTTP headers to add on this request. Can be <jk>null</jk>.
226 	 * @return A reader over the opened event stream. Never <jk>null</jk>. Callers are responsible for closing it.
227 	 * @throws IOException If a transport-level or (de)serialization error occurs opening the stream, or the
228 	 * 	response was neither an SSE stream nor a JSON-RPC error envelope.
229 	 * @throws McpException If the server rejected {@code request} with a JSON-RPC error instead of opening the
230 	 * 	stream.
231 	 */
232 	@SuppressWarnings("resource") // 'res' is closed in the finally block via quiet(res::close) on every non-return path (Eclipse doesn't trace closes made through a method reference); the returned SseEventReader wraps 'res' and transfers ownership to the caller when opened.
233 	protected SseEventReader openEventStream(JsonRpcRequest request, Map<String,String> httpHeaders) throws IOException {
234 		assertArgNotNull(ARG_REQUEST, request);
235 		var req = restClient.post(endpoint).body(request);
236 		if (httpHeaders != null) {
237 			for (var e : httpHeaders.entrySet())
238 				req.header(e.getKey(), e.getValue());
239 		}
240 		if (! req.hasHeader("Accept"))
241 			req.header("Accept", "text/event-stream");
242 		var res = req.run();
243 		var opened = false;
244 		try {
245 			if (isEventStream(res)) {
246 				res.assertOk();
247 				var r = res.body().asEventStream();
248 				opened = true;
249 				return r;
250 			}
251 			var parsed = res.body().as(JsonRpcResponse.class);
252 			if (parsed != null && parsed.getError() != null)
253 				throw McpException.fromJsonRpcError(parsed.getError());
254 			res.assertOk(); // not a JSON-RPC error envelope either: fall back to the ordinary status diagnostic
255 			throw ioex("Expected a text/event-stream response opening %s, but got Content-Type '%s'.",
256 				request.getMethod(), contentTypeOf(res));
257 		} finally {
258 			if (! opened)
259 				quiet(res::close);
260 		}
261 	}
262 
263 	private static boolean isEventStream(RestResponse res) {
264 		var contentType = contentTypeOf(res);
265 		return contentType != null && contentType.toLowerCase(Locale.ROOT).contains("text/event-stream");
266 	}
267 
268 	private static String contentTypeOf(RestResponse res) {
269 		var h = res.getFirstHeader("Content-Type");
270 		return h == null ? null : h.value();
271 	}

273 	@Override /* Overridden from Closeable */
274 	public void close() throws IOException {
275 		restClient.close();
276 	}

282 	/**
283 	 * Fluent, self-typed builder for {@link AbstractMcpClient} subclasses.
284 	 *
285 	 * @param <SELF> The concrete builder type (CRTP self-type), allowing dated adapters to add their own builder
286 	 * 	methods while preserving fluent chaining.
287 	 * @since 10.0.0
288 	 */
289 	@SuppressWarnings({
290 		"java:S119" // 'SELF' (CRTP self-type) is intentional and clearer than a single-letter name.
291 	})
292 	public abstract static class Builder<SELF extends Builder<SELF>> {
293 
294 		String endpoint;
295 		RestClient.Builder restClientBuilder = RestClient.builder();
296 
297 		/**
298 		 * Sets the MCP server endpoint URL.
299 		 *
300 		 * @param value The absolute endpoint URL (e.g. {@code "http://localhost:8080/mcp"}). Must not be <jk>null</jk>.
301 		 * @return This object.
302 		 */
303 		public SELF endpoint(String value) {
304 			endpoint = assertArgNotNullOrBlank(ARG_ENDPOINT, value);
305 			return self();
306 		}
307 
308 		/**
309 		 * Sets the HTTP transport used by the underlying {@link RestClient}.
310 		 *
311 		 * @param value The transport. Can be <jk>null</jk> (a transport is auto-discovered via {@link java.util.ServiceLoader}).
312 		 * @return This object.
313 		 */
314 		public SELF transport(HttpTransport value) {
315 			restClientBuilder.transport(value);
316 			return self();
317 		}
318 
319 		/**
320 		 * Adds a lifecycle interceptor (e.g. the auth-token seam) to the underlying {@link RestClient}.
321 		 *
322 		 * @param value The interceptor to add. Must not be <jk>null</jk>.
323 		 * @return This object.
324 		 */
325 		public SELF interceptor(RestCallInterceptor value) {
326 			restClientBuilder.interceptors(assertArgNotNull(ARG_INTERCEPTOR, value));
327 			return self();
328 		}
329 
330 		@SuppressWarnings({
331 			"unchecked" // SELF is always the concrete builder subtype by construction (CRTP).
332 		})
333 		protected SELF self() {
334 			return (SELF) this;
335 		}
336 	}
