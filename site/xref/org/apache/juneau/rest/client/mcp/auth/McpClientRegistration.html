17  package org.apache.juneau.rest.client.mcp.auth;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import java.net.*;
22  import java.time.*;
23  import java.util.*;
24  
25  /**
26   * Immutable result of a Dynamic Client Registration (SEP-837) round-trip &mdash; the client credentials and metadata an
27   * authorization server issued, keyed by the AS {@code issuer} for SEP-2352 credential binding.
28   *
29   * <p>
30   * <b>Never log {@code clientSecret} / {@code registrationAccessToken}.</b>  {@link #toString()} redacts both
31   * secret-bearing fields (mirroring the {@code OAuthToken} / {@code KeyedSecret} redaction discipline) so credential
32   * material never reaches logs via this record's auto-generated form.
33   *
34   * @param clientId The registered {@code client_id}.  Never {@code null}.
35   * @param clientSecret The issued {@code client_secret}, if the AS issued one (absent for a public/native client).
36   * @param clientSecretExpiresAt The instant the {@code client_secret} expires, if the AS returned a non-zero
37   * 	{@code client_secret_expires_at}.
38   * @param registrationAccessToken The RFC 7592 registration access token, if the AS issued one.
39   * @param registrationClientUri The RFC 7592 registration client URI, if the AS returned one.
40   * @param issuer The authorization server {@code issuer} these credentials were registered with (the SEP-2352 key).
41   * 	Never {@code null}.
42   * @param redirectUris The redirect URIs that were registered, in registration order.  Never {@code null}.
43   * @param applicationType The declared {@code application_type}.  Never {@code null}.
44   * @param extras Any non-standard fields the AS returned.  Read-only map.  Never {@code null}.
45   * @since 10.0.0
46   */
47  public record McpClientRegistration(
48  		String clientId,
49  		Optional<String> clientSecret,
50  		Optional<Instant> clientSecretExpiresAt,
51  		Optional<String> registrationAccessToken,
52  		Optional<URI> registrationClientUri,
53  		URI issuer,
54  		List<URI> redirectUris,
55  		McpApplicationType applicationType,
56  		Map<String,Object> extras) {
57  
58  	/** Placeholder shown for an absent secret-bearing / optional field in {@link #toString()}. */
59  	private static final String NONE = "<none>";
60  
61  	/**
62  	 * Compact constructor enforcing non-null fields and defensively copying the collections.
63  	 */
64  	public McpClientRegistration {
65  		Objects.requireNonNull(clientId, "clientId");
66  		Objects.requireNonNull(clientSecret, "clientSecret");
67  		Objects.requireNonNull(clientSecretExpiresAt, "clientSecretExpiresAt");
68  		Objects.requireNonNull(registrationAccessToken, "registrationAccessToken");
69  		Objects.requireNonNull(registrationClientUri, "registrationClientUri");
70  		Objects.requireNonNull(issuer, "issuer");
71  		Objects.requireNonNull(applicationType, "applicationType");
72  		redirectUris = redirectUris == null ? List.of() : u(cp(redirectUris));
73  		extras = extras == null ? Map.of() : u(cp(extras));
74  	}
75  
76  	/**
77  	 * Whether these credentials include a client secret (a confidential client) versus a public/native client that
78  	 * authenticates with PKCE alone.
79  	 *
80  	 * @return <jk>true</jk> if a {@code client_secret} was issued.
81  	 */
82  	public boolean isConfidential() {
83  		return clientSecret.isPresent();
84  	}
85  
86  	/**
87  	 * Redacts the secret-bearing fields so credential material never reaches logs via this record's
88  	 * {@code toString()}.
89  	 *
90  	 * @return A redacted string form disclosing only non-secret metadata.
91  	 */
92  	@Override
93  	public String toString() {
94  		return "McpClientRegistration[clientId=" + clientId
95  			+ ", issuer=" + issuer
96  			+ ", applicationType=" + applicationType
97  			+ ", redirectUris=" + redirectUris
98  			+ ", clientSecret=" + (clientSecret.isPresent() ? "<redacted>" : NONE)
99  			+ ", clientSecretExpiresAt=" + clientSecretExpiresAt.map(Object::toString).orElse(NONE)
100 			+ ", registrationAccessToken=" + (registrationAccessToken.isPresent() ? "<redacted>" : NONE)
101 			+ ", registrationClientUri=" + registrationClientUri.map(Object::toString).orElse(NONE)
102 			+ ", extras=" + extras
103 			+ "]";
104 	}
105 }
