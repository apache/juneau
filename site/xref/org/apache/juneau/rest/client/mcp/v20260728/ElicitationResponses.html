17  package org.apache.juneau.rest.client.mcp.v20260728;
18  
19  import static org.apache.juneau.commons.utils.AssertionUtils.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.bean.mcp.v20260728.*;
24  import org.apache.juneau.marshall.collections.*;
25  import org.apache.juneau.marshall.marshaller.*;
26  
27  /**
28   * Static helper for encoding the client's answers to server&rarr;client MCP {@code 2026-07-28} SEP-2322
29   * elicitation requests back into the raw {@code inputResponses} wire shape a resume call expects (MRTR,
30   * the Multi-Round-Trip Requests loop, client side).
31   *
32   * <p>
33   * Paired with {@link ElicitationRequests}, which reads the server's questions out of a paused
34   * {@link McpClient#callRaw} result &mdash; the same {@code ElicitationRequests}/{@code ElicitationResponses}
35   * split the server side already uses (server {@code ElicitationRequests} builds the pause signal; server
36   * {@code ElicitationResponses} reads the resumed answers).
37   *
38   * @since 10.0.0
39   */
40  public final class ElicitationResponses {
41  
42  	private ElicitationResponses() {}
43  
44  	/**
45  	 * Encodes a single typed {@link ElicitResult} into the raw {@code inputResponses} payload shape.
46  	 *
47  	 * @param id The server-assigned id this answer responds to.  Must not be <jk>null</jk>.
48  	 * @param result The typed answer.  Must not be <jk>null</jk>.
49  	 * @return A single-entry map ready to pass to {@link #toInputResponses(Map)}'s multi-answer form, or to hand
50  	 * 	directly to a concrete request bean's {@code setInputResponses(...)} after merging with other answers.
51  	 * @throws IllegalArgumentException If {@code id} or {@code result} is <jk>null</jk>.
52  	 */
53  	public static Map<String,Object> toInputResponse(String id, ElicitResult result) {
54  		assertArgNotNull("id", id);
55  		assertArgNotNull("result", result);
56  		return toInputResponses(Map.of(id, result));
57  	}
58  
59  	/**
60  	 * Encodes several typed {@link ElicitResult}s into the raw {@code inputResponses} payload shape a resume
61  	 * call's request bean expects (e.g. {@code CallToolRequest.setInputResponses(...)}).
62  	 *
63  	 * <p>
64  	 * Unlike {@code ElicitationRequests.of(Map,Object)} (server side), an empty {@code results} map is accepted
65  	 * (returning an empty map), not rejected &mdash; deliberately asymmetric, since a caller may legitimately
66  	 * resume with zero elicitation answers if a round only carried non-elicitation MRTR pauses, whereas building
67  	 * a signal with zero questions server-side would be a pointless pause.
68  	 *
69  	 * @param results Server-assigned-id-keyed typed answers.  Must not be <jk>null</jk>, and no value may be
70  	 * 	<jk>null</jk>.  Output iteration order mirrors this map's order, so a caller wanting deterministic
71  	 * 	ordering should pass a {@link LinkedHashMap}.
72  	 * @return A keyed map of raw, wire-shaped answers.  Never <jk>null</jk> (empty if {@code results} was empty).
73  	 * @throws IllegalArgumentException If {@code results} is <jk>null</jk>, or any value in it is <jk>null</jk>.
74  	 */
75  	public static Map<String,Object> toInputResponses(Map<String,ElicitResult> results) {
76  		assertArgNotNull("results", results);
77  		Map<String,Object> out = new LinkedHashMap<>();
78  		results.forEach((id, result) -> {
79  			assertArgNotNull("results[" + id + "]", result);
80  			// Pre-marshalled to JsonMap here (rather than left as the typed ElicitResult for McpClient.call's
81  			// own toWireParams flattening) so that (a) a null-check on the encoded shape is meaningful even
82  			// when this helper is used standalone, outside McpClient.call's flow, and (b) the returned map is
83  			// immediately wire-ready for a caller who hands it straight to a concrete request bean's
84  			// setInputResponses(...) without ever going through McpClient at all.
85  			out.put(id, Json.to(Json.of(result), JsonMap.class));
86  		});
87  		return out;
88  	}
89  }
