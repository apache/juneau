19  import static org.apache.juneau.commons.utils.AssertionUtils.*;
20  
21  import java.util.*;

23  import org.apache.juneau.bean.mcp.v20260728.*;
24  import org.apache.juneau.marshall.marshaller.*;

26  /**
27   * Static helper for reading server&rarr;client MCP {@code 2026-07-28} SEP-2322 elicitation requests out of
28   * {@link McpClient#callRaw}'s raw {@code Map<String,Object>} result (MRTR, the Multi-Round-Trip Requests
29   * loop, client side).
30   *
31   * <p>
32   * Paired with {@link ElicitationResponses}, which encodes the client's answers back into the wire shape a resume
33   * call expects &mdash; the same {@code ElicitationRequests}/{@code ElicitationResponses} split the server side
34   * already uses (server {@code ElicitationRequests} builds the pause signal; server {@code ElicitationResponses}
35   * reads the resumed answers).
36   *
37   * <p>
38   * Works at the raw map/JSON level rather than exposing typed overloads per concrete request bean, because
39   * {@code CallToolRequest}/{@code GetPromptRequest}/{@code ReadResourceRequest} share no common
40   * "has-inputResponses-and-requestState" interface (each independently declares its own pair of fields; their
41   * actual common base, {@code RequestParams<T>}, carries only {@code _meta}). A caller resuming, say, a
42   * {@code CallToolRequest} calls {@link ElicitationResponses#toInputResponses(Map)} for the raw payload, then
43   * makes its own {@code .setInputResponses(...).setRequestState(...)} call on the concrete bean it already knows
44   * it holds.
45   *
46   * @since 10.0.0
47   */
48  public final class ElicitationRequests {
49  
50  	private ElicitationRequests() {}
51  
52  	/**
53  	 * Whether a {@link McpClient#callRaw} result represents a paused elicitation (or any other MRTR pause).
54  	 *
55  	 * @param raw The raw result from {@link McpClient#callRaw}.  Must not be <jk>null</jk>.
56  	 * @return <jk>true</jk> if {@code raw.get("resultType")} is {@code "input_required"}.
57  	 * @throws IllegalArgumentException If {@code raw} is <jk>null</jk>.
58  	 */
59  	public static boolean isInputRequired(Map<String,Object> raw) {
60  		assertArgNotNull("raw", raw);
61  		return "input_required".equals(raw.get("resultType"));
62  	}
63  
64  	/**
65  	 * Decodes {@code raw}'s {@code inputRequests} into typed {@link ElicitRequest}s.
66  	 *
67  	 * <p>
68  	 * Each raw value is generic JSON (a {@code Map} after wire deserialization, per {@code InputRequiredResult}'s
69  	 * lossless-map contract), converted via the same {@code Json.to(Json.of(value), type)} technique
70  	 * {@code McpMrtrResumeContext.continuationAs(Class)} demonstrates server-side. A value that is present but
71  	 * <jk>null</jk> decodes to a <jk>null</jk> entry (see the return doc); this mirrors
72  	 * {@code ElicitationResponses.all(McpMrtrResumeContext)}'s null-passthrough on the server side.
73  	 *
74  	 * <p>
75  	 * <b>Every</b> entry is decoded as an {@link ElicitRequest}, unconditionally &mdash; per
76  	 * {@code InputRequiredResult}'s Javadoc, {@code inputRequests} entries may in principle be arbitrary
77  	 * sub-request kinds (sampling/roots/elicitation), not only elicitation. A non-elicitation entry (one that
78  	 * shares no field names with {@link ElicitRequest}) simply decodes to an all-<jk>null</jk>-fields
79  	 * {@link ElicitRequest}, since the underlying {@code Json.to} conversion tolerates unknown/missing bean
80  	 * properties; it is the caller's responsibility to only call this for elicitation-flavored MRTR pauses.
81  	 *
82  	 * @param raw The raw result from {@link McpClient#callRaw}.  Must not be <jk>null</jk>.
83  	 * @return A keyed map of typed requests.  Never <jk>null</jk> (empty if {@code raw} carried none). Values
84  	 * 	may be <jk>null</jk> when the corresponding raw entry is absent/<jk>null</jk>.
85  	 * @throws IllegalArgumentException If {@code raw} is <jk>null</jk>.
86  	 * @throws ClassCastException If {@code raw.get("inputRequests")} is present but is not, at runtime, a
87  	 * 	{@code Map} &mdash; which can only happen against a malformed/non-conforming remote response, since a
88  	 * 	well-formed one always deserializes {@code inputRequests} to a {@code Map} per JSON semantics.
89  	 * @throws RuntimeException If any entry's decoded shape cannot be converted to {@link ElicitRequest}.
90  	 */
91  	@SuppressWarnings({
92  		"unchecked" // raw.get("inputRequests") deserializes to a Map<String,Object> from JSON for any well-formed remote response; a malformed one surfaces as ClassCastException here (see @throws), not a silently-wrong cast.
93  	})
94  	public static Map<String,ElicitRequest> requests(Map<String,Object> raw) {
95  		assertArgNotNull("raw", raw);
96  		var rawRequests = (Map<String,Object>) raw.get("inputRequests");
97  		Map<String,ElicitRequest> out = new LinkedHashMap<>();
98  		if (rawRequests != null)
99  			rawRequests.forEach((id, value) -> out.put(id, value == null ? null : Json.to(Json.of(value), ElicitRequest.class)));
100 		return out;
101 	}
102 
103 	/**
104 	 * Returns {@code raw}'s echoed continuation token.
105 	 *
106 	 * @param raw The raw result from {@link McpClient#callRaw}.  Must not be <jk>null</jk>.
107 	 * @return The token, or <jk>null</jk> if absent.
108 	 * @throws IllegalArgumentException If {@code raw} is <jk>null</jk>.
109 	 * @throws ClassCastException If {@code raw.get("requestState")} is present but is not, at runtime, a
110 	 * 	{@code String} &mdash; only possible against a malformed/non-conforming remote response.
111 	 */
112 	public static String requestState(Map<String,Object> raw) {
113 		assertArgNotNull("raw", raw);
114 		return (String) raw.get("requestState");
115 	}
116 }
