36  /**
37   * Interactive authorization-code + PKCE (S256) token acquisition for headless / CLI use (F1 Phase B).
38   *
39   * <p>
40   * Builds on the relocated {@link OAuthAuthorizationCodeFlow}: opens a {@link LoopbackRedirectReceiver}, generates the
41   * {@code state} and PKCE {@code code_verifier}/{@code code_challenge}, hands the authorization URL to a caller-supplied
42   * browser launcher, waits for the loopback redirect, and exchanges the returned code for a token.
43   *
44   * <p>
45   * Security checks on the callback:
46   * <ul>
47   * 	<li><b>CSRF</b> &mdash; the returned {@code state} must match the one that was issued (single-use via
48   * 		{@link EphemeralStore}).
49   * 	<li><b>SEP-2468 ({@code iss}, RFC 9207)</b> &mdash; the callback {@code iss} parameter must equal the
50   * 		discovered/expected issuer; a mismatch is rejected.  The check gates BOTH the success and error paths (a client
51   * 		MUST NOT act on an error response whose issuer doesn't match).  {@link Builder#expectedIssuer(URI)} is required
52   * 		by default (opt out with {@link Builder#skipIssuerValidation(boolean)}); a missing {@code iss} is accepted
53   * 		unless {@link Builder#requireIssuerResponseParameter(boolean)} is set.
54   * </ul>
55   *
56   * <p>
57   * {@link Builder#resource(URI)} (the RFC 8707 resource indicator) is required so the acquired token's audience is bound
58   * to the target MCP server.
59   *
60   * <p>
61   * The optional {@link Builder#offlineAccess(boolean)} adds the {@code offline_access} scope so the IdP issues a refresh
62   * token that can subsequently drive an {@link McpTokenProvider} refresh-token provider (SEP-2207).
63   *
64   * @since 10.0.0
65   */

406 	/**
407 	 * Runs the full interactive acquisition: starts the loopback listener, opens the authorization URL, waits for the
408 	 * redirect, validates it, and exchanges the code for a token.
409 	 *
410 	 * @param timeout The maximum time to wait for the user to complete authorization at the IdP.  Must not be
411 	 * 	<jk>null</jk>.
412 	 * @return The acquired token.
413 	 * @throws McpAuthException If the listener cannot start, the wait times out, or the callback fails validation.
414 	 * @throws OAuthFlowException If the token-exchange HTTP round-trip fails or the IdP returns an error.
415 	 */
416 	public OAuthToken acquire(Duration timeout) {
417 		assertArgNotNull("timeout", timeout);
418 		LoopbackRedirectReceiver receiver;
419 		try {
420 			receiver = new LoopbackRedirectReceiver(redirectPath, redirectPort, DEFAULT_CALLBACK_HTML);
421 		} catch (IOException e) {
422 			throw new McpAuthException("Failed to start the loopback redirect listener", e);
423 		}
424 		try (receiver) {
425 			var flow = buildFlow(receiver.redirectUri());
426 			var verifier = new CodeVerifier();
427 			var state = new State().getValue();
428 			var authUrl = buildAuthorizationUrl(flow, state, verifier);
429 			store.store(state, verifier.getValue(), expectedIssuer);
430 			browserLauncher.accept(authUrl);
431 			var callbackUri = receiver.awaitCallback(timeout);
432 			return handleCallback(flow, callbackUri);
433 		}
434 	}

452 	/**
453 	 * Validates a loopback callback and exchanges the authorization code for a token.
454 	 *
455 	 * <p>
456 	 * Consumes the pending {@code state} entry (single-use), performs the SEP-2468 {@code iss} check against the issuer
457 	 * recorded when the request was started, then exchanges the code.
458 	 *
459 	 * @param flow The configured flow.  Must not be <jk>null</jk>.
460 	 * @param callbackUri The full loopback callback URI.  Must not be <jk>null</jk>.
461 	 * @return The acquired token.
462 	 * @throws McpAuthException If the callback is an error response, the state is unknown/replayed, or the {@code iss}
463 	 * 	check fails.
464 	 */
465 	public OAuthToken handleCallback(OAuthAuthorizationCodeFlow flow, URI callbackUri) {
466 		assertArgNotNull("flow", flow);
467 		assertArgNotNull("callbackUri", callbackUri);
468 		var resp = parse(callbackUri);
469 		var state = resp.getState() == null ? null : resp.getState().getValue();
470 		if (state == null)
471 			throw new McpAuthException("Authorization callback is missing the state parameter");
472 		var pending = store.consume(state)
473 			.orElseThrow(() -> new McpAuthException("Authorization callback state is unknown, expired, or replayed"));
474 		var code = extractCode(resp, pending.expectedIssuer(), skipIssuerValidation, requireIssuerResponseParameter);
475 		return flow.exchange(code, new CodeVerifier(pending.codeVerifier()));
476 	}

478 	/**
479 	 * Validates an authorization-code callback and returns the authorization code, without any token exchange or
480 	 * state-store interaction.
481 	 *
482 	 * <p>
483 	 * This is the pure SEP-2468 / CSRF validation step.  The {@code iss} comparison gates BOTH success and error
484 	 * responses (per the MCP spec a client MUST NOT act on an {@code error}/{@code error_description}/{@code error_uri}
485 	 * from a response whose {@code iss} does not match), so an error response with a mismatched issuer is reported as
486 	 * an issuer mismatch rather than a denial.  A missing/mismatched {@code state} is rejected as CSRF.  When
487 	 * {@code iss} is absent it is accepted (future-proofed for the eventual "reject if absent" MUST); use
488 	 * {@link #validateAuthorizationResponse(URI, String, URI, boolean)} to require it.
489 	 *
490 	 * @param callbackUri The full callback URI.  Must not be <jk>null</jk>.
491 	 * @param expectedState The state issued when the authorization request was built.  Must not be <jk>null</jk>.
492 	 * @param expectedIssuer The discovered issuer to validate {@code iss} against.  Must not be <jk>null</jk>.
493 	 * @return The authorization code.
494 	 * @throws McpAuthException If validation fails.
495 	 */
496 	public static String validateAuthorizationResponse(URI callbackUri, String expectedState, URI expectedIssuer) {
497 		return validateAuthorizationResponse(callbackUri, expectedState, expectedIssuer, false);
498 	}

500 	/**
501 	 * Validates an authorization-code callback and returns the authorization code, without any token exchange or
502 	 * state-store interaction.
503 	 *
504 	 * @param callbackUri The full callback URI.  Must not be <jk>null</jk>.
505 	 * @param expectedState The state issued when the authorization request was built.  Must not be <jk>null</jk>.
506 	 * @param expectedIssuer The discovered issuer to validate {@code iss} against.  Must not be <jk>null</jk>.
507 	 * @param requireIssuerParameter <jk>true</jk> to reject a response that omits the {@code iss} parameter (RFC 9207
508 	 * 	{@code authorization_response_iss_parameter_supported}).
509 	 * @return The authorization code.
510 	 * @throws McpAuthException If validation fails.
511 	 */
512 	public static String validateAuthorizationResponse(URI callbackUri, String expectedState, URI expectedIssuer, boolean requireIssuerParameter) {
513 		assertArgNotNull("callbackUri", callbackUri);
514 		assertArgNotNullOrBlank("expectedState", expectedState);
515 		assertArgNotNull("expectedIssuer", expectedIssuer);
516 		var resp = parse(callbackUri);
517 		var state = resp.getState() == null ? null : resp.getState().getValue();
518 		if (!expectedState.equals(state))
519 			throw new McpAuthException("Authorization callback state mismatch (possible CSRF)");
520 		return extractCode(resp, expectedIssuer, false, requireIssuerParameter);
521 	}

523 	private static AuthorizationResponse parse(URI callbackUri) {
524 		try {
525 			return AuthorizationResponse.parse(callbackUri);
526 		} catch (ParseException e) {
527 			throw new McpAuthException("Authorization callback could not be parsed", e);
528 		}
529 	}

531 	private static String extractCode(AuthorizationResponse resp, URI expectedIssuer, boolean skipIssuerValidation, boolean requireIssuerParameter) {
532 		// SEP-2468: the iss check gates BOTH the success and error paths — a client MUST NOT act on an error response
533 		// whose issuer doesn't match, so this runs before the indicatesSuccess() branch.
534 		if (! skipIssuerValidation) {
535 			var iss = resp.getIssuer();
536 			if (iss == null) {
537 				if (requireIssuerParameter)
538 					throw new McpAuthException("Authorization response is missing the required iss parameter (SEP-2468 / RFC 9207)");
539 			} else if (expectedIssuer == null || ! iss.getValue().equals(expectedIssuer.toString())) { // HTT: expectedIssuer==null is unreachable here through either public caller — validateAuthorizationResponse(...) asserts it non-null before calling extractCode, and handleCallback(...)'s pending.expectedIssuer() is guaranteed non-null too because build() requires expectedIssuer(...) whenever skipIssuerValidation is false (the only way to reach this branch at all)
540 				throw new McpAuthException("Authorization response issuer mismatch (SEP-2468): expected '"
541 					+ expectedIssuer + "' but got '" + iss.getValue() + "'");
542 			}
543 		}
544 		if (! resp.indicatesSuccess()) {
545 			var err = resp.toErrorResponse().getErrorObject();
546 			throw new McpAuthException("Authorization request was denied: " + (err == null ? "unknown_error" : err.getCode())); // HTT: Nimbus's ErrorResponse#getErrorObject() never returns null (verified against oauth2-oidc-sdk 11.38.2, same guarantee already documented in McpDynamicClientRegistrar), so the err==null arm is unreachable
547 		}
548 		return resp.toSuccessResponse().getAuthorizationCode().getValue();
549 	}

551 	private OAuthAuthorizationCodeFlow buildFlow(URI redirectUri) {
552 		var b = OAuthAuthorizationCodeFlow.create()
553 			.authorizationEndpoint(authorizationEndpoint)
554 			.tokenEndpoint(tokenEndpoint)
555 			.clientId(clientId)
556 			.redirectUri(redirectUri)
557 			.httpTimeout(httpTimeout);
558 		if (clientSecretSupplier != null)
559 			b.clientSecretSupplier(clientSecretSupplier);
560 		scopes.forEach(b::scope);
561 		if (resource != null) // HTT: resource is always non-null here — buildFlow(...) only ever runs against a fully-built instance, and Builder#build() unconditionally rejects a null resource (see the "requires resource(...)" check above) before this constructor can even be reached
562 			b.resource(resource);
563 		if (httpRequestConfigurator != null)
564 			b.httpRequestConfigurator(httpRequestConfigurator);
565 		return b.build();
566 	}
