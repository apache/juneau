36  /**
37   * Typed <c>2026-07-28</c> Model Context Protocol (MCP) client facade.
38   *
39   * <p>
40   * Extends {@link AbstractMcpClient}, adding one typed method per <c>2026-07-28</c> JSON-RPC method
41   * (see {@link McpMethods}). Every method stamps the SEP-2243 {@code params._meta} envelope (protocol
42   * version, client capabilities, client info, and any supplied trace-context fields), builds the
43   * corresponding v2 request bean, wraps it in a {@link JsonRpcRequest} with the required
44   * {@code Mcp-Method}/{@code Mcp-Name} headers, delegates to the inherited {@link #send(JsonRpcRequest, Map)},
45   * and then either returns the {@code result} re-marshaled into its typed v2 result bean, or throws an
46   * {@link McpException} built from the response's {@code error}.
47   *
48   * <p>
49   * An optional {@link McpResponseCache} can be supplied via {@link Builder#responseCache(McpResponseCache)}
50   * to serve repeat identical calls from cache instead of the wire, keyed off the method and business params
51   * (never the per-call {@code _meta}, so a per-request trace-context supplier does not defeat the cache).
52   *
53   * <p>
54   * A server-initiated request delivered over the duplex event-stream channel is routed through the
55   * {@link McpDuplexDispatcher} registered via {@link #setServerRequestHandler(McpServerRequestHandler)}.
56   *
57   * <p>
58   * <b>Beta — API subject to change:</b> This class builds directly on {@code org.apache.juneau.rest.client.mcp},
59   * which is itself Beta.
60   *
61   * @since 10.0.0
62   */

125 	 * @throws McpException If the server returned a JSON-RPC error for {@value McpMethods#SERVER_DISCOVER}.

134 	/**
135 	 * Builds a client from a caller-configured builder and performs its one mandatory handshake call, returning a
136 	 * ready-to-use client.
137 	 *
138 	 * <p>
139 	 * If {@link #serverDiscover()} fails, the just-built client is closed before the exception propagates - a
140 	 * caller that only ever sees an exception from {@code connect(...)} never ends up owning a client (and its
141 	 * underlying transport resources) it has no reference to.
142 	 *
143 	 * @param builder The configured builder to build from. Must not be <jk>null</jk>.
144 	 * @return A new client that has already completed {@link #serverDiscover()} exactly once. Never <jk>null</jk>.
145 	 * @throws IOException If a transport-level or (de)serialization error occurs opening the connection.
146 	 * @throws McpException If the server returned a JSON-RPC error for {@value McpMethods#SERVER_DISCOVER}.
147 	 */
148 	@SuppressWarnings({
149 		"java:S1181", // Must run the close-and-rethrow cleanup below even on Error, not just checked/runtime exceptions.
150 		"resource" // client is already closed via closeQuietly(client, e) before the rethrow on the failure path below; the success-path return is @Owning - the caller owns and must close the returned client.
151 	})
152 	public static McpClient connect(Builder builder) throws IOException {
153 		assertArgNotNull("builder", builder);
154 		var client = builder.build();
155 		try {
156 			client.serverDiscover();
157 		} catch (Throwable e) {
158 			// Catches Throwable (not just IOException/RuntimeException): a client that only ever surfaces a
159 			// thrown Throwable from connect(...) must never end up leaking its transport resources just
160 			// because the handshake failed with something other than a checked/runtime exception (e.g. a
161 			// StackOverflowError decoding a pathological response). javac's "more precise rethrow" analysis
162 			// (JLS 11.2.2) sees that only IOException/RuntimeException/Error can reach this catch from the
163 			// try block, so rethrowing this effectively-final `e` still satisfies this method's declared
164 			// "throws IOException" without widening to "throws Throwable".
165 			closeQuietly(client, e);
166 			throw e;
167 		}
168 		return client;
169 	}
170 
171 	/**
172 	 * Closes {@code client}, adding any close failure as a suppressed exception on {@code primary} rather than
173 	 * letting it mask the handshake failure that is the actual reason {@link #connect(Builder)} is failing.
174 	 */
175 	@SuppressWarnings({
176 		"java:S1181" // Must record even an Error from close() as suppressed rather than let it replace the handshake failure.
177 	})
178 	private static void closeQuietly(McpClient client, Throwable primary) {
179 		try {
180 			client.close();
181 		} catch (Throwable e) {
182 			// Catches Throwable, mirroring the outer catch in connect(Builder): a caller-supplied
183 			// HttpTransport.close() throwing an unchecked failure (RuntimeException/Error) must be recorded
184 			// as suppressed, not allowed to propagate from here and replace the handshake failure (primary)
185 			// this method exists to preserve.
186 			primary.addSuppressed(e);
187 		}
188 	}

363 	 * payload without deserializing into a specific {@link Result} subtype.
364 	 *
365 	 * <p>
366 	 * Every convenience method on this class ({@link #callTool}, {@link #getPrompt}, {@link #readResource}, etc.)
367 	 * both builds its own request bean internally and deserializes strictly into one known result type, so none of
368 	 * them can represent a paused {@code input_required} response (MCP {@code 2026-07-28} SEP-2322 Multi-Round-Trip

378 	 * 	subclass-specific deserialization. Never <jk>null</jk> on a successful JSON-RPC response.
379 	 * @throws IOException On a transport failure.
380 	 * @throws McpException If the server returns a JSON-RPC error.
381 	 */
382 	@SuppressWarnings({
383 		"unchecked" // Map.class.cast(...) below always yields a raw Map from JSON deserialization.
384 	})
385 	public Map<String,Object> callRaw(String method, RequestParams<?> params) throws IOException {
386 		return call(method, params, Map.class, false);
387 	}

400 	 * <p>
401 	 * This is the ergonomic counterpart to hand-driving the resume loop with {@link #callRaw},
402 	 * {@link ElicitationRequests}, and {@link ElicitationResponses}: it detects each pause, decodes that round's
403 	 * requests, calls {@code handler} for the answers, echoes them back with the carried {@code requestState},
404 	 * and repeats until the server returns a non-{@code input_required} result, which is decoded into a typed
405 	 * {@link CallToolResult}. {@link #callRaw} remains available as the low-level escape hatch.
406 	 *
407 	 * @param name The tool name to invoke.

409 	 * @param handler The elicitation answer callback, invoked once per pause. Must not be <jk>null</jk>.
410 	 * @return The terminal call-tool result. Never <jk>null</jk> on success unless the server returns a
411 	 * 	<jk>null</jk> result.
412 	 * @throws IllegalArgumentException If {@code handler} is <jk>null</jk>, or {@code handler} returns a
413 	 * 	<jk>null</jk> result or a result map containing a <jk>null</jk> value for a requested id.
414 	 * @throws IOException If a transport-level or (de)serialization error occurs, or {@code handler} throws it.
415 	 * @throws McpException If the server returned a JSON-RPC error.
416 	 * @throws McpElicitationLimitException If the server keeps pausing past {@link #DEFAULT_MAX_ELICITATION_ROUNDS}.
417 	 */
418 	public CallToolResult callToolWithElicitation(String name, Map<String,Object> arguments, McpElicitationHandler handler) throws IOException {

442 		return decodeResult(raw, CallToolResult.class);
443 	}

529 	/**
530 	 * Shared MRTR (SEP-2322) auto-resume loop backing every {@code *WithElicitation} method.
531 	 *
532 	 * <p>
533 	 * Issues the initial {@link #callRaw} for {@code params}, then while the raw result is an
534 	 * {@code input_required} pause: decodes the round's requests, invokes {@code handler} for the answers,
535 	 * applies them plus the pause's echoed {@code requestState} onto {@code params} via {@code applyResume} (each
536 	 * concrete request bean knows its own {@code setInputResponses}/{@code setRequestState}, which have no shared
537 	 * interface — see {@link ElicitationRequests}), and re-issues. A decline/cancel answer is not short-circuited
538 	 * locally: it is echoed back like any other answer, leaving the terminal outcome of a refused elicitation to
539 	 * the server. The loop is bounded by {@code maxRounds} so a server (or handler) that never converges surfaces
540 	 * as a typed {@link McpElicitationLimitException} rather than hanging.
541 	 */
542 	private Map<String,Object> driveElicitation(String method, RequestParams<?> params, ResumeApplier applyResume, McpElicitationHandler handler, int maxRounds) throws IOException {
543 		assertArgNotNull("handler", handler);
544 		assertArg(maxRounds >= 1, "maxRounds must be >= 1 (was %s).", maxRounds);
545 		var raw = callRaw(method, params);
546 		var rounds = 0;
547 		while (ElicitationRequests.isInputRequired(raw)) {
548 			if (++rounds > maxRounds)
549 				throw new McpElicitationLimitException(maxRounds);
550 			var requests = ElicitationRequests.requests(raw);
551 			var requestState = ElicitationRequests.requestState(raw);
552 			var answers = assertArgNotNull("handler result", handler.elicit(requests));
553 			applyResume.apply(ElicitationResponses.toInputResponses(answers), requestState);
554 			raw = callRaw(method, params);
555 		}
556 		return raw;
557 	}

581 	/**
582 	 * Sends {@value McpMethods#SUBSCRIPTIONS_LISTEN}, opening a managed, held-open notification stream on a
583 	 * background thread and dispatching decoded frames to {@code listener} until the returned handle is closed,
584 	 * the server completes gracefully, or the connection drops.
585 	 *
586 	 * <p>
587 	 * Only the initial {@value McpMethods#SUBSCRIPTIONS_LISTEN} POST (opening the stream) is synchronous — a
588 	 * transport-level failure at that point surfaces here as a thrown {@link IOException}. Every frame after that
589 	 * (the mandatory {@code acknowledged} frame, zero or more change notifications, and the eventual terminal
590 	 * frame) is decoded on a dedicated background thread and delivered to {@code listener}; frame-processing
591 	 * failures surface via {@link McpSubscriptionListener#onError(Throwable)}, never as a thrown exception from
592 	 * this method. This does not reuse {@link #pumpNextServerMessage()} (single-shot, closes its stream after at
593 	 * most one event) or {@link McpDuplexDispatcher} (request/response-shaped, single registration) — {@code
594 	 * listen(...)} needs its own long-lived, multi-frame decode loop.
595 	 *
596 	 * @param filter The subscription filter (toolsListChanged/promptsListChanged/resourcesListChanged/resourceSubscriptions). Must not be <jk>null</jk>.
597 	 * @param listener The typed callback sink. Must not be <jk>null</jk>.
598 	 * @return A handle to cancel/close the subscription. Never <jk>null</jk>.
599 	 * @throws IOException If the initial listen request/stream-open fails.
600 	 */
601 	@SuppressWarnings({
602 		"resource" // reader/pump are @Owning: reader's ownership passes to pump immediately; on pump.start() failure the reader is closed via pump.closeReaderQuietly() before the rethrow; on success pump is returned to the caller as the McpSubscriptionHandle (which owns close()/cancel()), and the pump thread's run() finally closes the reader unconditionally. Warnings surface at the throw/return exit points, so suppress at method scope.
603 	})
604 	public McpSubscriptionHandle listen(SubscriptionFilter filter, McpSubscriptionListener listener) throws IOException {
605 		assertArgNotNull("filter", filter);
606 		assertArgNotNull("listener", listener);
607 		var params = new SubscriptionsListenRequest().setNotifications(filter);
608 		// params is a fresh local built one line above with no setMeta(...) call reachable in between, so
609 		// params.getMeta() is always null here and the params.getMeta() (non-null) arm of this ternary -
610 		// mirrored from #call's identical pattern, where a caller-driven multi-round elicitation loop can reuse
611 		// the same mutable params bean across rounds after the first stampMeta(...) call - is unreachable for
612 		// listen(), which never reuses params across calls.
613 		stampMeta(params.getMeta() == null ? params.setMeta(new RequestMeta()).getMeta() : params.getMeta()); // HTT: params.getMeta() is always null here; see comment above
614 		var wireParams = toWireParams(params);
615 		var req = new JsonRpcRequest()
616 			.setJsonrpc(McpProtocol.JSON_RPC_2_0)
617 			.setId(UUID.randomUUID().toString())
618 			.setMethod(McpMethods.SUBSCRIPTIONS_LISTEN)
619 			.setParams(wireParams);
620 		// SEP-2243 Mcp-Method/Mcp-Name headers, stamped exactly like every typed call(...) (see #call) - the v2
621 		// server's McpRevision.dispatch() runs validateHeaders(...) unconditionally before branching on the
622 		// method, so the opening subscriptions/listen POST is rejected with -32600 without these.
623 		var headers = Map.of(HEADER_MCP_METHOD, McpMethods.SUBSCRIPTIONS_LISTEN, HEADER_MCP_NAME, McpRoutingNames.routingName(McpMethods.SUBSCRIPTIONS_LISTEN, wireParams));
624 		var reader = openEventStream(req, headers);
625 		var pump = new SubscriptionPump(req.getId().toString(), reader, listener);
626 		try {
627 			pump.start();
628 		} catch (RuntimeException e) { // HTT: pumpThread is a fresh never-before-started Thread, so a real Thread.start() failure (IllegalThreadStateException requires start() called twice; OutOfMemoryError/SecurityException require exotic JVM-level resource exhaustion or a removed SecurityManager) is not reproducible in a unit test
629 			// Thread.start() failing (e.g. IllegalThreadStateException, or a resource-exhaustion error from
630 			// the JVM/OS) must not leak the already-opened stream: nothing else owns it yet.
631 			pump.closeReaderQuietly(); // HTT: only reachable if pump.start() throws; see catch line above
632 			throw e; // HTT: only reachable if pump.start() throws; see catch line above
633 		}
634 		return pump;
635 	}

637 	/**
638 	 * Background decode loop + closeable handle for one
639 	 * {@link #listen(SubscriptionFilter, McpSubscriptionListener)} subscription.
640 	 *
641 	 * <p>
642 	 * Every frame is decoded generically first (into a {@link JsonMap}) to tell a notification (has a
643 	 * {@code "method"} key) apart from the terminal JSON-RPC response (no {@code "method"} key) before committing
644 	 * to a strongly-typed re-parse — {@link JsonRpcRequest} and {@link JsonRpcResponse} share no common base type
645 	 * to parse into speculatively.
646 	 *
647 	 * <p>
648 	 * A named {@code "ping"} SSE event (the real transport's heartbeat, carrying no {@code data:} payload — see
649 	 * {@code SubscriptionsListenPublisher.HEARTBEAT_EVENT_NAME}) is recognized by event name and skipped
650 	 * <i>before</i> any attempt to inspect its (always <jk>null</jk>) data, so a heartbeat can never be mistaken
651 	 * for, or misreported as, a malformed JSON-RPC frame.
652 	 *
653 	 * <p>
654 	 * Package-private (rather than {@code private}) solely so tests in this package can observe the background
655 	 * thread's actual lifecycle (see {@code pumpThread}) — production code only ever sees this through the
656 	 * {@link McpSubscriptionHandle} interface {@link #listen} returns.
657 	 */
658 	static final class SubscriptionPump implements McpSubscriptionHandle, Runnable {
659 		private static final String EVENT_PING = "ping";
660 		private static final AtomicLong THREAD_SEQ = new AtomicLong();
661 
662 		private final String id;
663 		@SuppressWarnings({
664 			"resource" // Owned by this pump's lifecycle: closed via closeReaderQuietly() from cancel()/close(), and unconditionally in run()'s finally block once the decode loop ends.
665 		})
666 		private final SseEventReader reader;
667 		private final McpSubscriptionListener listener;
668 		final Thread pumpThread;
669 		private final AtomicBoolean closed = new AtomicBoolean(false);
670 		private volatile boolean open = true;
671 
672 		SubscriptionPump(String id, SseEventReader reader, McpSubscriptionListener listener) {
673 			this.id = id;
674 			this.reader = reader;
675 			this.listener = listener;
676 			this.pumpThread = new Thread(this, "mcp-subscriptions-listen-" + THREAD_SEQ.incrementAndGet());
677 			this.pumpThread.setDaemon(true);
678 		}
679 
680 		@Override
681 		public String id() {
682 			return id;
683 		}
684 
685 		void start() {
686 			pumpThread.start();
687 		}
688 
689 		@Override
690 		@SuppressWarnings({
691 			"java:S135" // Multiple break/continue are the clearest expression of this SSE-frame decode loop; splitting them out would obscure the pump's state machine.
692 		})
693 		public void run() {
694 			var reachedTerminal = false;
695 			try {
696 				while (open && reader.hasNext()) {
697 					var event = reader.next();
698 					if (EVENT_PING.equals(event.getEvent()))
699 						continue;
700 					if (event.getData() == null || event.getData().isEmpty())
701 						continue;
702 					if (! dispatch(event.getData())) {
703 						reachedTerminal = true;
704 						break;
705 					}
706 				}
707 				// The stream ended (clean EOF, reader.hasNext() == false) without ever reaching a terminal
708 				// frame, and the caller did not initiate this close (open is still true): an abrupt drop the
709 				// listener must be told about, not silent - it is otherwise indistinguishable from a hang.
710 				//
711 				// open is flipped to false BEFORE invoking the listener (guarded on the pre-read value so a
712 				// concurrent cancel() can't cause a double-fire), not after in the finally block below: the
713 				// terminal callback is very often the exact synchronization point (a latch/future countdown,
714 				// or a direct isOpen() read from inside the callback itself) a caller relies on to learn the
715 				// pump is done, and CountDownLatch/Future's happens-before is one-directional (it orders the
716 				// signal before the waiter's wakeup, never the reverse) - so this volatile write must
717 				// happen-before that signal, not after it, or a woken caller can transiently observe open
718 				// still true.
719 				if (open && ! reachedTerminal) { // HTT: reachedTerminal is only ever set true immediately after dispatch(...) returns false, and dispatch(...) itself always sets open=false first (see its own happens-before rationale below) - so open==true && reachedTerminal==true can never occur; this branch combination is structurally unreachable
720 					open = false;
721 					invokeListener(() -> listener.onError(new EOFException(
722 						"Subscription stream closed before a terminal frame was received.")));
723 				}
724 			} catch (Exception e) {
725 				if (open) {
726 					open = false;
727 					invokeListener(() -> listener.onError(e));
728 				}
729 			} finally {
730 				// Idempotent safety net: open is already false on every path above that invoked a terminal
731 				// callback, and closeReaderQuietly() must run unconditionally regardless of which path (if
732 				// any) was taken - including a plain open cancel()/close() with no callback at all.
733 				open = false;
734 				closeReaderQuietly();
735 			}
736 		}
737 
738 		/**
739 		 * Decodes one frame and invokes the matching listener callback.
740 		 *
741 		 * @return <jk>false</jk> if this was the terminal frame (the loop must stop); <jk>true</jk> to keep reading.
742 		 */
743 		private boolean dispatch(String data) {
744 			var tree = JsonParser.DEFAULT.read(data, JsonMap.class);
745 			if (tree.containsKey("method")) {
746 				dispatchNotification(JsonParser.DEFAULT.read(data, JsonRpcRequest.class));
747 				return true;
748 			}
749 			var res = JsonParser.DEFAULT.read(data, JsonRpcResponse.class);
750 			// open is flipped to false BEFORE invoking either terminal callback below, not after in run()'s
751 			// finally block - see the longer rationale on the abrupt-EOF branch in run().
752 			open = false;
753 			if (res.getError() != null) {
754 				var mcpException = McpException.fromJsonRpcError(res.getError());
755 				invokeListener(() -> listener.onError(mcpException));
756 			} else {
757 				invokeListener(listener::onComplete);
758 			}
759 			return false;
760 		}
761 
762 		private void dispatchNotification(JsonRpcRequest notif) {
763 			var method = notif.getMethod();
764 			var rawParams = notif.getParams();
765 			if (McpMethods.NOTIFICATIONS_SUBSCRIPTIONS_ACKNOWLEDGED.equals(method)) {
766 				var decoded = Json.to(Json.of(rawParams), SubscriptionsAcknowledgedNotification.class);
767 				invokeListener(() -> listener.onAcknowledged(decoded.getNotifications()));
768 			} else if (McpMethods.NOTIFICATIONS_RESOURCES_UPDATED.equals(method)) {
769 				var decoded = Json.to(Json.of(rawParams), ResourceUpdatedNotification.class);
770 				invokeListener(() -> listener.onResourceUpdated(decoded.getUri()));
771 			} else if (McpMethods.NOTIFICATIONS_RESOURCES_LIST_CHANGED.equals(method)) {
772 				invokeListener(() -> listener.onListChanged(McpListChangedKind.RESOURCES));
773 			} else if (McpMethods.NOTIFICATIONS_TOOLS_LIST_CHANGED.equals(method)) {
774 				invokeListener(() -> listener.onListChanged(McpListChangedKind.TOOLS));
775 			} else if (McpMethods.NOTIFICATIONS_PROMPTS_LIST_CHANGED.equals(method)) {
776 				invokeListener(() -> listener.onListChanged(McpListChangedKind.PROMPTS));
777 			}
778 			// Unknown method: forward-compatible no-op, not an error - a future notification kind this
779 			// client version doesn't know about must not tear down an otherwise-healthy subscription.
780 		}
781 
782 		/**
783 		 * Invokes one listener callback in isolation, so a bug in caller-supplied listener code can never be
784 		 * mistaken for (and re-routed through) a transport/decode failure, and can never itself crash the
785 		 * pump or escape this method - including a second throw from {@code onError} itself, which must not
786 		 * re-enter this same path.
787 		 *
788 		 * @param callback The single listener callback invocation to contain (e.g. {@code () -> listener.onComplete()}).
789 		 */
790 		private void invokeListener(Runnable callback) {
791 			try {
792 				callback.run();
793 			} catch (@SuppressWarnings("unused") RuntimeException e) {
794 				// Contained by design: this is the caller's own listener code, not this pump's transport/decode
795 				// path, so its exception must never surface as (or trigger) this pump's onError, and must never
796 				// tear down an otherwise-healthy subscription.
797 			}
798 		}

801 		public void cancel() {
802 			if (closed.compareAndSet(false, true)) {
803 				open = false;
804 				closeReaderQuietly();
805 			}
806 		}

809 		public void close() {
810 			cancel();
811 		}

813 		@Override
814 		public boolean isOpen() {
815 			return open;
816 		}

819 			try {
820 				reader.close();
821 			} catch (@SuppressWarnings("unused") IOException e) {
822 				// Best-effort close: the stream may already be closed by the peer/EOF.
823 			}
824 		}
825 	}

850 	/**
851 	 * Opens the duplex event-stream channel, reads at most one server-initiated request, dispatches it to the
852 	 * registered {@link McpServerRequestHandler}, and posts the result (or a JSON-RPC error) back to the server
853 	 * via {@value #DUPLEX_RETURN_METHOD}.
854 	 *
855 	 * <p>
856 	 * Package-private test/plumbing entry point: production duplex pumping is driven by the caller repeatedly
857 	 * invoking this method (or an equivalent loop) for as long as the channel should stay open.
858 	 *
859 	 * @throws IOException If a transport-level or (de)serialization error occurs.
860 	 */
861 	void pumpNextServerMessage() throws IOException {
862 		try (var r = openEventStream()) {
863 			if (! r.hasNext())
864 				return;
865 			var event = r.next();
866 			if (event.getData() == null || event.getData().isEmpty())
867 				return;
868 			var req = JsonParser.DEFAULT.read(event.getData(), JsonRpcRequest.class);
869 			try {
870 				var result = duplexDispatcher.dispatch(req, BasicBeanStore.INSTANCE);
871 				if (JsonRpcResponse.notification(req.getId()))
872 					return;
873 				postClientResult(req.getId(), JsonRpcResponse.ok(req.getId(), result));
874 			} catch (McpException e) {
875 				postClientResult(req.getId(), JsonRpcResponse.errorResponse(req.getId(), e.getCode(), e.getMessage()));
876 			}
877 		} // HTT: javac's try-with-resources close inserts a defensive null-check on 'r' before calling close(); openEventStream() never returns null (it always returns a fresh SseEventReader or throws), so the r==null arm is unreachable
878 	}

880 	// C5/C6 duplex return-channel Mcp-Name contract (TODO-327, resolved): Mcp-Name is empty by design on this
881 	// channel. Per SEP-2243, Mcp-Name is a body-mirror routing header defined only for the three forward
882 	// client->server named methods (tools/call, prompts/get, resources/read); the spec does not define it for
883 	// server->client requests or return channels. So McpRoutingNames intentionally has no case for
884 	// DUPLEX_RETURN_METHOD (mcp/clientResult) - it falls to the default empty arm - and the response envelope has
885 	// no name/uri to mirror anyway. If correlating a server-initiated request back to its originating client
886 	// request is ever needed, key it on the JSON-RPC id (SEP-2260 relatedRequestId), not on Mcp-Name.
887 	//
888 	// payload must be routed through toWireParams(...) exactly like every other outbound envelope (see #call):
889 	// AbstractMcpClient.send(...) posts through restClient, whose serializer is JsonSerializer.DEFAULT
890 	// (addBeanTypes=false) - handed the payload's typed beans directly, a polymorphic Content field (e.g.
891 	// CreateMessageResult.content holding a TextContent) would be serialized without its "type" discriminator,
892 	// since addBeanTypes is what drives whether SerializerSession writes bean-dictionary type names at all.
893 	// toWireParams(...) pre-flattens through REQUEST_SERIALIZER (addBeanTypes=true) into a generic Map/List
894 	// tree with the discriminator already baked in as literal data, so the final JsonSerializer.DEFAULT pass
895 	// re-emits it verbatim.
896 	private void postClientResult(Object id, JsonRpcResponse payload) throws IOException {
897 		var wire = toWireParams(payload);
898 		var headers = Map.of(HEADER_MCP_METHOD, DUPLEX_RETURN_METHOD, HEADER_MCP_NAME, McpRoutingNames.routingName(DUPLEX_RETURN_METHOD, wire));
899 		send(new JsonRpcRequest().setJsonrpc("2.0").setId(id).setMethod(DUPLEX_RETURN_METHOD).setParams(wire), headers);
900 	}

906 	/**
907 	 * Shared wire implementation for every typed method and {@link #callRaw}.
908 	 *
909 	 * <p>
910 	 * {@code useCache=false} (used only by {@link #callRaw}) skips the cache READ so a {@code callRaw} call never
911 	 * retrieves a differently-typed cached result (e.g. a {@link CallToolResult} primed by {@link #callTool}) under
912 	 * the same cache key, which would otherwise throw a {@link ClassCastException} on the subsequent {@code cast}.
913 	 * The cache WRITE at the end is unaffected by this flag: it is already a no-op for {@code callRaw} because a
914 	 * plain {@link Map} result is never a {@link CacheableResult}.
915 	 */
916 	private <T> T call(String method, RequestParams<?> params, Class<T> resultType, boolean useCache) throws IOException {
917 		var cacheKey = cacheKey(method, params);
918 		if (useCache) {
919 			var cached = readCache(cacheKey);
920 			if (cached != null)
921 				return resultType.cast(cached);
922 		}
923 		stampMeta(params.getMeta() == null ? params.setMeta(new RequestMeta()).getMeta() : params.getMeta());
924 		var wireParams = toWireParams(params);
925 		var headers = Map.of(HEADER_MCP_METHOD, method, HEADER_MCP_NAME, McpRoutingNames.routingName(method, wireParams));
926 		var req = new JsonRpcRequest()
927 			.setJsonrpc(McpProtocol.JSON_RPC_2_0)
928 			.setId(UUID.randomUUID().toString())
929 			.setMethod(method)
930 			.setParams(wireParams);
931 		var res = send(req, headers);
932 		if (res.getError() != null)
933 			throw McpException.fromJsonRpcError(res.getError());
934 		var result = JsonMap.of(WRAP_KEY_value, res.getResult()).get(WRAP_KEY_value, resultType);
935 		writeCache(cacheKey, result);
936 		return result;
937 	}
938 
939 	private static Object toWireParams(Object params) {
940 		return JsonParser.DEFAULT.read(REQUEST_SERIALIZER.write(params), Object.class);
941 	}
942 
943 	private void stampMeta(RequestMeta meta) {
944 		meta.setProtocolVersion(McpProtocol.VERSION_2026_07_28);
945 		meta.setClientCapabilities(clientCapabilities);
946 		if (clientInfo != null)
947 			meta.setClientInfo(clientInfo);
948 		if (traceFieldsSupplier != null) {
949 			var fields = traceFieldsSupplier.get();
950 			if (fields != null) {
951 				var carrier = new RequestMetaCarrier(meta);
952 				putIfPresent(carrier, RequestMeta.KEY_TRACEPARENT, fields.get(RequestMeta.KEY_TRACEPARENT));
953 				putIfPresent(carrier, RequestMeta.KEY_TRACESTATE, fields.get(RequestMeta.KEY_TRACESTATE));
954 				putIfPresent(carrier, RequestMeta.KEY_BAGGAGE, fields.get(RequestMeta.KEY_BAGGAGE));
955 			}
956 		}
957 	}
958 
959 	private void putIfPresent(TraceContextCarrier carrier, String key, String value) {
960 		if (value != null)
961 			carrier.set(key, value);
962 	}

964 	/**
965 	 * Computes the cache key from {@code method} and the caller-supplied business {@code params} only.
966 	 *
967 	 * <p>
968 	 * Called before {@link #stampMeta(RequestMeta)} ever touches {@code params}, so the key never reflects
969 	 * per-call {@code _meta} content (protocol version, capabilities, or trace-context fields) — otherwise a
970 	 * real per-span {@link #traceFieldsSupplier} would make every key unique and defeat caching entirely.
971 	 */
972 	private String cacheKey(String method, Object params) {
973 		return method + "|" + JsonMap.of("p", params).toString();
974 	}
975 
976 	private Object readCache(String key) {
977 		if (responseCache == null)
978 			return null;
979 		return responseCache.get(privateScopePartitionPrefix + McpCacheScope.PRIVATE.toWire(), key)
980 			.or(() -> responseCache.get(McpCacheScope.PUBLIC.toWire(), key))
981 			.orElse(null);
982 	}
983 
984 	private void writeCache(String key, Object result) {
985 		if (responseCache == null)
986 			return;
987 		if (result instanceof CacheableResult<?> c && c.getTtlMs() != null) {
988 			var scope = c.getCacheScope() == McpCacheScope.PRIVATE
989 				? privateScopePartitionPrefix + McpCacheScope.PRIVATE.toWire()
990 				: McpCacheScope.PUBLIC.toWire();
991 			responseCache.put(scope, key, result, c.getTtlMs());
992 		}
993 	}

995 	private static final class RequestMetaCarrier implements TraceContextCarrier {
996 		private final RequestMeta meta;
997 		RequestMetaCarrier(RequestMeta meta) { this.meta = meta; }
998 		@Override public String get(String key) {
999 			return switch (key) {
1000 				case RequestMeta.KEY_TRACEPARENT -> meta.getTraceparent();
1001 				case RequestMeta.KEY_TRACESTATE -> meta.getTracestate();
1002 				case RequestMeta.KEY_BAGGAGE -> meta.getBaggage();
1003 				default -> null;
1004 			};
1005 		}
1006 		@Override public Iterable<String> keys() { return List.of(RequestMeta.KEY_TRACEPARENT, RequestMeta.KEY_TRACESTATE, RequestMeta.KEY_BAGGAGE); }
1007 		@Override public void set(String key, String value) {
1008 			switch (key) { // HTT: putIfPresent(...) - this carrier's sole caller - only ever passes RequestMeta.KEY_TRACEPARENT/KEY_TRACESTATE/KEY_BAGGAGE; the default arm below is unreachable through the private class's only call path
1009 				case RequestMeta.KEY_TRACEPARENT -> meta.setTraceparent(value);
1010 				case RequestMeta.KEY_TRACESTATE -> meta.setTracestate(value);
1011 				case RequestMeta.KEY_BAGGAGE -> meta.setBaggage(value);
1012 				default -> { /* Unknown trace-context key: no corresponding RequestMeta field to set. */ }
1013 			}
1014 		}
1015 	}
