17  package org.apache.juneau.rest.client.mcp.auth;
18  
19  import java.net.*;
20  import java.util.*;
21  
22  /**
23   * SEP-2352 optional persistence SPI for Dynamic-Client-Registration / pre-registered credentials, keyed by the
24   * authorization server's {@code issuer} identifier.
25   *
26   * <p>
27   * Persistence is <b>SHOULD-level</b> (READY-312f Q8): an MCP client that performs on-demand DCR on every connection
28   * with <i>no</i> store is fully compliant.  When a store <i>is</i> used, the SEP-2352 keying MUST bites &mdash;
29   * credentials MUST be associated with the specific AS that issued them, and separate registration state MUST be kept per
30   * AS ({@link InMemoryMcpClientRegistrationStore} keys by {@code issuer.toString()}).  {@link McpClientRegistrationManager}
31   * consults a store (if present) to reuse an issuer-keyed entry and to drive re-registration on AS migration.
32   *
33   * <h5 class='topic'>Security</h5>
34   * <p>
35   * A store holds {@code clientSecret} / registration-access-token material (via {@link McpClientRegistration}).  The
36   * default {@link InMemoryMcpClientRegistrationStore} is process-scoped; a durable implementation a caller supplies is
37   * responsible for at-rest protection and MUST NOT log credential material.
38   *
39   * @since 10.0.0
40   */
41  public interface McpClientRegistrationStore {
42  
43  	/**
44  	 * Returns the registration stored for the given AS issuer, if any.
45  	 *
46  	 * @param issuer The authorization server {@code issuer} identifier.  Must not be <jk>null</jk>.
47  	 * @return The stored registration, or {@link Optional#empty()} if none is stored for that issuer.
48  	 */
49  	Optional<McpClientRegistration> find(URI issuer);
50  
51  	/**
52  	 * Stores (or replaces) the registration for the given AS issuer.
53  	 *
54  	 * @param issuer The authorization server {@code issuer} identifier.  Must not be <jk>null</jk>.
55  	 * @param registration The registration to store.  Must not be <jk>null</jk>.
56  	 */
57  	void put(URI issuer, McpClientRegistration registration);
58  
59  	/**
60  	 * Removes any registration stored for the given AS issuer.
61  	 *
62  	 * @param issuer The authorization server {@code issuer} identifier.  Must not be <jk>null</jk>.
63  	 */
64  	void remove(URI issuer);
65  }
