27  import com.sun.net.httpserver.*;

29  /**
30   * A minimal loopback ({@code http://127.0.0.1}) HTTP listener that captures the single authorization-code redirect an
31   * IdP sends back to a native/CLI OAuth client (SEP-837 loopback redirect handling).
32   *
33   * <p>
34   * Binds to the loopback interface &mdash; by default on an ephemeral port (the RFC 8252 &sect;7.3 port-agnostic strategy),
35   * or on a caller-chosen <em>fixed</em> port (the "bind-first" strategy) when constructed via
36   * {@link #LoopbackRedirectReceiver(String, int, String)} &mdash; so the effective redirect URI (see
37   * {@link #redirectUri()}) is only reachable from the same host.  The first request to the callback path completes
38   * {@link #awaitCallback(Duration)} with the full request URI (including the {@code code}/{@code state}/{@code iss} query
39   * parameters); the browser receives a small "you may close this window" page.
40   *
41   * <p>
42   * Use with try-with-resources so the underlying server is always stopped.
43   *
44   * @since 10.0.0
45   */
46  public class LoopbackRedirectReceiver implements AutoCloseable {
47  
48  	private static final String DEFAULT_PATH = "/callback";
49  	private static final String DEFAULT_HTML =
50  		"<html><body><h3>Authorization received</h3><p>You may close this window and return to the application.</p></body></html>";
51  
52  	private final HttpServer server;
53  	private final String path;
54  	private final int port;
55  	private final String host;
56  	private final CompletableFuture<URI> callback = new CompletableFuture<>();
57  
58  	/**
59  	 * Opens a receiver on an ephemeral loopback port using the default callback path {@code /callback}.
60  	 *
61  	 * @return A started receiver.
62  	 * @throws IOException If the loopback server could not be started.
63  	 */
64  	@SuppressWarnings("resource") // factory hands the started receiver (and its bound socket) to the caller, who must close it (see class javadoc).
65  	public static LoopbackRedirectReceiver open() throws IOException {
66  		return new LoopbackRedirectReceiver(DEFAULT_PATH, DEFAULT_HTML);
67  	}
68  
69  	/**
70  	 * Opens a receiver on an ephemeral loopback port (the port-agnostic strategy, RFC 8252 &sect;7.3).
71  	 *
72  	 * @param path The callback path (must start with {@code /}).  Must not be <jk>null</jk> or blank.
73  	 * @param successHtml The HTML body returned to the browser on the callback.  Must not be <jk>null</jk>.
74  	 * @throws IOException If the loopback server could not be started.
75  	 */
76  	public LoopbackRedirectReceiver(String path, String successHtml) throws IOException {
77  		this(path, 0, successHtml);
78  	}
79  
80  	/**
81  	 * Opens a receiver on a caller-chosen <em>fixed</em> loopback port (the "bind-first" strategy for strict
82  	 * exact-match authorization servers that reject port-agnostic loopback redirects).
83  	 *
84  	 * <p>
85  	 * Pass {@code port == 0} for an ephemeral port (identical to {@link #LoopbackRedirectReceiver(String, String)}); a
86  	 * non-zero port binds exactly that port so it matches a {@link LoopbackRedirectUris#forPort(int, String) forPort}
87  	 * registration.  A fixed port can fail to bind if already in use &mdash; the {@link IOException} surfaces that to the
88  	 * caller.
89  	 *
90  	 * @param path The callback path (must start with {@code /}).  Must not be <jk>null</jk> or blank.
91  	 * @param bindPort The fixed loopback port to bind, or {@code 0} for an ephemeral port.  Must not be negative.
92  	 * @param successHtml The HTML body returned to the browser on the callback.  Must not be <jk>null</jk>.
93  	 * @throws IOException If the loopback server could not be started (e.g. the fixed port is already in use).
94  	 */
95  	public LoopbackRedirectReceiver(String path, int bindPort, String successHtml) throws IOException {
96  		this.path = assertArgNotNullOrBlank("path", path);
97  		assertArg(path.startsWith("/"), "path must start with '/' (was '%s')", path);
98  		assertArg(bindPort >= 0, "port must not be negative (was %s)", bindPort);
99  		assertArgNotNull("successHtml", successHtml);
100 		var html = successHtml.getBytes(StandardCharsets.UTF_8);
101 		var svr = HttpServer.create(new InetSocketAddress(InetAddress.getLoopbackAddress(), bindPort), 0);
102 		try {
103 			port = svr.getAddress().getPort();
104 			host = hostLiteral(svr.getAddress().getAddress());
105 			svr.createContext(path, ex -> {
106 				var full = URI.create("http://" + host + ":" + port + ex.getRequestURI().toString());
107 				callback.complete(full);
108 				ex.getResponseHeaders().add("Content-Type", "text/html; charset=utf-8");
109 				ex.sendResponseHeaders(200, html.length);
110 				try (var os = ex.getResponseBody()) {
111 					os.write(html);
112 				}
113 			});
114 			svr.start();
115 		} catch (RuntimeException e) { // HTT: createContext()/start() on a freshly-created HttpServer registered with exactly one already-validated ('/'-prefixed) path never throws in practice; not reproducible without mocking/subclassing the JDK's sealed com.sun.net.httpserver implementation
116 			svr.stop(0);  // HTT: only reachable if the try-block above throws; see catch line above -- don't leak the bound socket if wiring fails after create()
117 			throw e; // HTT: only reachable if the try-block above throws; see catch line above
118 		}
119 		server = svr;
120 	}
121 
122 	/** Formats the bound loopback address as a URI host literal, bracketing IPv6 (e.g. {@code [::1]}). */
123 	private static String hostLiteral(InetAddress addr) {
124 		var ip = addr.getHostAddress();
125 		var pct = ip.indexOf('%');  // strip any IPv6 zone/scope id
126 		if (pct >= 0) // HTT: addr is always InetAddress.getLoopbackAddress() (see the sole call site above), which resolves to an IPv4 loopback address with no '%' zone-id suffix on this JVM/OS's network stack -- environment-dependent, unreachable here without a differently-configured host
127 			ip = ip.substring(0, pct); // HTT: only reachable if the zone-id is present; see the 'if' line above
128 		return addr instanceof Inet6Address ? "[" + ip + "]" : ip; // HTT: addr is always an IPv4 Inet4Address in this environment (see rationale above), so the Inet6Address bracketing arm is unreachable here
129 	}
130 
131 	/**
132 	 * Returns the loopback redirect URI the IdP should redirect the user-agent back to.
133 	 *
134 	 * @return The redirect URI (e.g. {@code http://127.0.0.1:PORT/callback}).
135 	 */
136 	public URI redirectUri() {
137 		return URI.create("http://" + host + ":" + port + path);
138 	}
139 
140 	/**
141 	 * Returns the ephemeral loopback port the receiver is listening on.
142 	 *
143 	 * @return The port.
144 	 */
145 	public int port() {
146 		return port;
147 	}
148 
149 	/**
150 	 * Blocks until the IdP redirects to the callback path, or the timeout elapses.
151 	 *
152 	 * @param timeout The maximum time to wait.  Must not be <jk>null</jk>.
153 	 * @return The full callback request URI, including query parameters.
154 	 * @throws McpAuthException If the wait times out or is interrupted.
155 	 */
156 	public URI awaitCallback(Duration timeout) {
157 		assertArgNotNull("timeout", timeout);
158 		try {
159 			return callback.get(timeout.toMillis(), TimeUnit.MILLISECONDS);
160 		} catch (TimeoutException e) {
161 			throw new McpAuthException("Timed out waiting for the loopback authorization redirect", e);
162 		} catch (InterruptedException e) {
163 			Thread.currentThread().interrupt();
164 			throw new McpAuthException("Interrupted waiting for the loopback authorization redirect", e);
165 		} catch (ExecutionException e) { // HTT: callback future is only ever completed normally by the handler
166 			throw new McpAuthException("Loopback authorization redirect failed", e); // HTT: only reachable if the future completed exceptionally; see catch line above
167 		}
168 	}
169 
170 	@Override /* Overridden from AutoCloseable */
171 	public void close() {
172 		server.stop(0);
173 	}
174 }
