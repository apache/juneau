28  /**
29   * Per-request context that removes the caller's credential headers from a redirect hop to another origin, without
30   * changing the client or its redirect handling.  The transport sends every ordinary request with one, on the client it
31   * creates itself (whose redirect strategy then allows such a hop, so it is followed without the headers) and on a
32   * caller-supplied client.
33   *
34   * <p>
35   * HttpClient's redirect executor copies the original request's headers onto each follow-up request.  The client's
36   * {@code RedirectStrategy} decides whether a redirect is followed at all: {@code DefaultRedirectStrategy} refuses a
37   * cross-origin redirect that would replay a sensitive header, but {@code LaxRedirectStrategy} or a custom strategy may
38   * not.  This context strips {@link RedirectSecurity#stripOnCrossOrigin()} from a hop whose origin (scheme, host and
39   * effective port, with the semantics of {@link RedirectSecurity#shouldStripCredentials(java.net.URI, java.net.URI)})
40   * differs from the first request's, whatever strategy made that hop, and from every later hop of the exchange: the
41   * redirect executor copies each hop from the unstripped one before it, so a redirect back to the first origin would
42   * otherwise carry the credentials again.  A hop whose origin cannot be determined counts as cross-origin.
43   *
44   * <p>
45   * HttpClient's main exec stage records each request it sends in the context ({@code setRequest}), once per send, after
46   * the protocol stage has added any {@code Authorization} from the client's {@code CredentialsProvider} and before it runs
47   * the client's own request interceptors.  So a {@code Cookie} that the client's {@code CookieStore} adds is still sent,
48   * and an {@code Authorization} from the client's {@code CredentialsProvider} on a cross-origin hop is removed with the
49   * caller's.  Every record is inspected, including a resend of the same request object after an authentication
50   * challenge, for which the protocol stage resets the headers to the unstripped copy.  A {@code CONNECT} tunnel request
51   * would not be inspected: it carries no caller headers, and HttpClient 5.6 sends it without recording it here.
52   *
53   * <p>
54   * On a proxied route that is not tunnelled (a plain {@code http} target through an HTTP proxy), {@code Proxy-Authorization}
55   * is kept on a cross-origin hop: every hop goes through the same proxy, and the header authenticates to that proxy, so
56   * removing it would make the proxy answer {@code 407} again.  On a direct or tunnelled route it is removed with the rest.
57   *
58   * <p>
59   * This relies on HttpClient's internal call order, as the HttpClient 4.5 transport's guard does; the transport's tests
60   * pin it.  A client that replaces the per-request context it is given (for example by wrapping it as the parent of a
61   * context of its own) is not guarded.  Like every {@link HttpClientContext}, an instance serves one request exchange
62   * on one thread.
63   */
64  final class CredentialGuardingContext extends HttpClientContext {
65  
66  	private boolean started;
67  	private HttpHost origin;
68  	private boolean stripping;
69  
70  	@Override /* HttpCoreContext */
71  	public void setRequest(HttpRequest request) {
72  		super.setRequest(request);
73  		if (request == null || Method.CONNECT.isSame(request.getMethod()))
74  			return;
75  		var target = originOf(request);
76  		if (! started) {
77  			started = true;
78  			origin = target;
79  			return;
80  		}
81  		if (! stripping && (origin == null || target == null || ! sameOrigin(origin, target)))
82  			stripping = true;
83  		if (stripping) {
84  			var keepProxyAuthorization = isPlainProxyRoute(getHttpRoute());
85  			for (var name : RedirectSecurity.stripOnCrossOrigin())
86  				if (! (keepProxyAuthorization && eqic(name, HttpHeaders.PROXY_AUTHORIZATION)))
87  					request.removeHeaders(name);
88  		}
89  	}
90  
91  	// The main exec stage records the route in the context just before the request.  On a proxied route that is not
92  	// tunnelled, every hop goes through the same proxy, and Proxy-Authorization authenticates to that proxy.
93  	private static boolean isPlainProxyRoute(RouteInfo route) {
94  		return route != null && route.getProxyHost() != null && ! route.isTunnelled();
95  	}
96  
97  	// The protocol stage sets the scheme and authority on every request before the main stage records it.
98  	private static HttpHost originOf(HttpRequest request) {
99  		var scheme = request.getScheme();
100 		var authority = request.getAuthority();
101 		if (scheme == null || authority == null || authority.getHostName() == null)
102 			return null;
103 		return new HttpHost(scheme, authority.getHostName(), authority.getPort());
104 	}
105 
106 	// HttpHost always has a scheme name and a host name; the port is -1 when none was given, so it is normalized to the
107 	// scheme default.  An https->http downgrade changes the scheme, so it is never same-origin.
108 	private static boolean sameOrigin(HttpHost from, HttpHost to) {
109 		return eqic(from.getSchemeName(), to.getSchemeName())
110 			&& eqic(from.getHostName(), to.getHostName())
111 			&& effectivePort(from) == effectivePort(to);
112 	}
113 
114 	private static int effectivePort(HttpHost host) {
115 		var port = host.getPort();
116 		if (port != -1)
117 			return port;
118 		return switch (host.getSchemeName().toLowerCase(Locale.ROOT)) {
119 			case "https" -> 443;
120 			case "http" -> 80;
121 			default -> -1;
122 		};
123 	}
124 }
