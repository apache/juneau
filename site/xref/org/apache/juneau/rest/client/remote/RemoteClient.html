49  /**
50   * Creates Java proxy instances for {@link Remote}-annotated interfaces, backed by an {@link RestClient}.
51   *
52   * <p>
53   * Each interface method call translates to an HTTP request via the client.
54   *
55   * <p>
56   * Obtain instances via {@link RestClient#remote(Class)}.
57   *
58   * <p class='bjava'>
59   * 	<ja>@Remote</ja>(path=<js>"/api/users"</js>)
60   * 	<jk>public interface</jk> UserService {{
61   * 		<ja>@RemoteGet</ja>(<js>"/{id}"</js>)
62   * 		String getUser(<ja>@Path</ja>(<js>"id"</js>) String <jv>id</jv>);
63   * 	}}
64   *
65   * 	UserService <jv>svc</jv> = client.remote(UserService.<jk>class</jk>);
66   * 	String <jv>user</jv> = <jv>svc</jv>.getUser(<js>"42"</js>);
67   * </p>
68   *
69   * <p>
70   * <b>Beta — API subject to change:</b> This type is part of the next-generation REST client and HTTP stack
71   * ({@code org.apache.juneau.marshall.ng.*}).
72   * It is not API-frozen: binary- and source-incompatible changes may appear in the <b>next major</b> Juneau release
73   * (and possibly earlier).
74   *
75   * <h5 class='section'>See Also:</h5><ul>
76   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/NextGenRestClient">juneau-ng REST client</a>
77   * </ul>
78   *
79   * @since 9.2.1
80   */

98  	 * Creates a Java proxy for the given {@link Remote}-annotated interface.
99  	 *
100 	 * @param <T> The interface type.
101 	 * @param iface The interface class. Must be annotated with {@link Remote}. Must not be <jk>null</jk>.
102 	 * @return A proxy instance. Never <jk>null</jk>.
103 	 * @throws IllegalArgumentException If {@code iface} is not an interface or not annotated with {@link Remote}.
104 	 */
105 	@SuppressWarnings({
106 		"unchecked" // Type erasure on reflective/generic cast; element type is verified at call site
107 	})
108 	public <T> T create(Class<T> iface) {
109 		assertArgNotNull("iface", iface);
110 		var meta = RrpcInterfaceMeta.of(iface);
111 		return (T) Proxy.newProxyInstance(
112 			iface.getClassLoader(),
113 			new Class<?>[]{iface},
114 			new RemoteInvocationHandler(client, meta)
115 		);
116 	}

172 		@Override
173 		@SuppressWarnings({
174 			"java:S3776" // Cognitive complexity acceptable for remote proxy invocation dispatch
175 		})
176 		public Object invoke(Object proxy, Method method, Object[] args) throws Throwable {
177 			// Handle Object methods directly
178 			if (method.getDeclaringClass() == Object.class)
179 				return method.invoke(this, args);
180 
181 			var methodMeta = meta.getMethodMeta(method);
182 			if (methodMeta == null)
183 				throw uoex("Method '%s' has no @Remote* annotation", method.getName());
184 
185 			// Build the full path: basePath + methodPath
186 			var basePath = meta.getBasePath();
187 			var methodPath = methodMeta.getPath();
188 			var fullPath = combinePaths(basePath, methodPath);
189 
190 			// Apply the call-time @Url parameter / declarative baseUrl override (computed per call;
191 			// never cached on the shared meta objects).  {var} substitution still runs at request time so @Path params
192 			// can fill tokens inside the resolved URL.
193 			var effectivePath = resolveEffectiveUrl(methodMeta, method, args, fullPath);
194 
195 			// throwOnError is enabled if set true at either the method or interface level.
196 			var throwOnError = meta.isThrowOnError() || methodMeta.isThrowOnError();
197 
198 			// The effective accept media type (method-level overrides interface-level) is the response
199 			// parser FALLBACK — consulted only when the response Content-Type matched no registered parser.
200 			var acceptFallback = firstNonEmpty(methodMeta.getAccept(), meta.getAccept());
201 
202 			// Execute and process the return value.  The request is rebuilt per attempt so a (gated) retry resends a
203 			// freshly-bound request.
204 			return processReturn(() -> buildRequest(methodMeta, effectivePath, method, args), methodMeta, method, throwOnError, acceptFallback);
205 		}

207 		private RestRequest buildRequest(RrpcInterfaceMethodMeta methodMeta, String path, Method method, Object[] args) throws IOException {
208 			var httpMethod = methodMeta.getHttpMethod();
209 			var req = switch (httpMethod) {
210 				case "GET" -> client.get(path);
211 				case "POST" -> client.post(path);
212 				case "PUT" -> client.put(path);
213 				case "PATCH" -> client.patch(path);
214 				case "DELETE" -> client.delete(path);
215 				// NOTE: @RemoteOp validates method against {DELETE, GET, POST, PUT, OPTIONS, HEAD, CONNECT, TRACE, PATCH}
216 				// (see RrpcInterfaceMeta#buildRemoteOpMeta), but this switch only handles 5 of those 9 -- @RemoteOp(method
217 				// = "HEAD"/"OPTIONS"/"CONNECT"/"TRACE") passes annotation validation yet always fails here with this
218 				// generic message, even though RestClient exposes head(String) and isRetryableVerb() already treats
219 				// "HEAD" as a retryable idempotent verb. Reachable/real (not HTT); flagged as a likely latent gap rather
220 				// than fixed, since fixing behavior is out of scope for this coverage pass.
221 				default -> throw iaex("Unsupported HTTP method: %s", httpMethod);
222 			};
223 
224 			// Resolve the effective content-negotiation overrides (method-level over interface-level).
225 			// contentType SELECTS the request serializer (and sets the Content-Type label); accept sets the Accept
226 			// header.  Both win over a constant header of the same name (filtered out below), but a genuinely
227 			// caller-supplied @Header param still takes effect (applied after binding).
228 			var bodyFormat = resolveBodyFormat(methodMeta);
229 			var effectiveAccept = firstNonEmpty(methodMeta.getAccept(), meta.getAccept());
230 
231 			// Apply always-applied constant parts (interface-level + method-level) to every request.
232 			// Within each part type, method-level constants take precedence over interface-level constants of the
233 			// same name; a caller-supplied parameter value (bound below) still composes with the constant.  A
234 			// constant Content-Type/Accept is dropped when the dedicated contentType/accept attribute is in effect.
235 			applyConstantHeaders(req, meta.getHeaders(), methodMeta.getConstantHeaders(), bodyFormat.isSet(), isNotEmpty(effectiveAccept));
236 			applyConstants(req::queryData, meta.getQueryData(), methodMeta.getConstantQueryData());
237 			applyConstants(req::formData, meta.getFormData(), methodMeta.getConstantFormData());
238 
239 			if (args != null) {
240 				var params = method.getParameters();
241 				for (int i = 0; i < params.length; i++)
242 					bindParam(req, methodMeta, params[i], args[i], params.length == 1, bodyFormat);
243 			}
244 
245 			// A @Multipart method assembles its body from the @Part parameters (a thin adapter onto the
246 			// existing streaming MultipartBody); otherwise apply the param-less constant body.
247 			if (methodMeta.isMultipart()) {
248 				bindMultipartBody(req, method, args);
249 			} else if (! hasBodyParam(method)) {
250 				// Param-less constant body — method-level @Content(def) with no body parameter at all.
251 				var contentDefault = methodMeta.getContentDefault();
252 				if (contentDefault != null)
253 					withContentBody(req, contentDefault, bodyFormat);
254 			}
255 
256 			// Set the Accept header from the dedicated attribute unless a caller-supplied @Header
257 			// param already provided one (caller wins per existing semantics); a single Accept header is emitted.
258 			if (isNotEmpty(effectiveAccept) && ! req.hasHeader(HEADER_ACCEPT))
259 				req.header(HEADER_ACCEPT, effectiveAccept);
260 
261 			// Apply annotation-declared interceptors + per-call timeout.
262 			applyPolicy(req, methodMeta);
263 
264 			return req;
265 		}

267 		/**
268 		 * Resolves the effective request body format from the {@code contentType} attribute
269 		 * (method-level overriding interface-level).
270 		 *
271 		 * <p>
272 		 * When a {@code contentType} media type is in effect, the matching registered request serializer is selected
273 		 * from the client's serializer set ({@link RestClient#getSerializerForMediaType(String)}).  If no registered
274 		 * serializer matches, the client's explicitly-configured default serializer is used to write the body but the
275 		 * overridden media type is still sent as the {@code Content-Type} label (supporting vendor media types such as
276 		 * {@code application/vnd.foo+json} whose bytes are really the default format).  If neither a match nor a default
277 		 * serializer is available, an {@link IllegalStateException} is raised (there is no implicit JSON fallback).
278 		 *
279 		 * @param methodMeta The method metadata.
280 		 * @return The resolved body format; {@link BodyFormat#NONE} when no {@code contentType} attribute is in effect.
281 		 * @throws IllegalStateException If no serializer matches and no default serializer is configured on the client.
282 		 */
283 		private BodyFormat resolveBodyFormat(RrpcInterfaceMethodMeta methodMeta) {
284 			var contentType = firstNonEmpty(methodMeta.getContentType(), meta.getContentType());
285 			if (isEmpty(contentType))
286 				return BodyFormat.NONE;
287 			var serializer = client.getSerializerForMediaType(contentType).or(client::getDefaultSerializer).orElseThrow(() ->
288 				isex("No serializer matched Content-Type '%s' and no default serializer is configured on the client.  Configure one via RestClient.Builder.defaultSerializer(...).", contentType));
289 			return new BodyFormat(serializer, contentType);
290 		}

292 		/**
293 		 * Computes the effective request URL for a call, applying the dynamic-URL overrides over the
294 		 * statically-computed {@code basePath + methodPath}.
295 		 *
296 		 * <p>
297 		 * Precedence (most-specific first):
298 		 * <ol>
299 		 * 	<li><b>{@code @Url} parameter</b> (call-time, Option A) — its value becomes the effective URL, replacing the
300 		 * 		whole computed path.  A value containing a scheme is absolute (and bypasses the client root URL via the
301 		 * 		existing {@code ://} seam in {@link RestClient#request(String, String)}); a scheme-less value is relative
302 		 * 		and resolved against the client root URL only (Retrofit endpoint replacement — <i>not</i> against the
303 		 * 		interface path).  {@code {var}} tokens are preserved for {@code @Path} substitution at request time.
304 		 * 	<li><b>Method-level {@code baseUrl}</b> (Option B) — substitutes the authority+root, preserving the full path.
305 		 * 	<li><b>Interface-level {@code baseUrl}</b> (Option B) — same, at lower precedence.
306 		 * 	<li><b>Client root URL</b> (default, unchanged behavior — no override applied).
307 		 * </ol>
308 		 *
309 		 * <p>
310 		 * SSRF guardrail (v1): when an override is in effect and yields a value carrying a URI scheme, the scheme must be
311 		 * {@code http} or {@code https}; any other scheme is rejected.  The default (no-override) path is left untouched
312 		 * so pre-existing absolute-{@code @RemoteOp(path)} / templating behavior does not regress.
313 		 *
314 		 * @param methodMeta The method metadata (carries the {@code @Url} param index + the method-level {@code baseUrl}).
315 		 * @param method The invoked method (for error messages).
316 		 * @param args The invocation arguments (source of the {@code @Url} value).
317 		 * @param fullPath The statically-computed {@code basePath + methodPath}.
318 		 * @return The effective URL/path to hand to {@code client.<verb>(...)}.
319 		 */
320 		private String resolveEffectiveUrl(RrpcInterfaceMethodMeta methodMeta, Method method, Object[] args, String fullPath) {
321 			// Option A: a call-time @Url parameter wins over everything.
322 			var urlParamIndex = methodMeta.getUrlParamIndex();
323 			if (urlParamIndex >= 0) {
324 				var arg = (args == null || urlParamIndex >= args.length) ? null : args[urlParamIndex]; // HTT: a JDK dynamic proxy always supplies exactly one Object per declared parameter, so args is only null for a zero-arg method (which can't have a @Url param) and urlParamIndex is always < args.length; defensive/structurally unreachable through the public proxy API.
325 				var value = arg == null ? null : arg.toString();
326 				assertArg(! isBlank(value), "@Url parameter on %s.%s must not be null or blank",
327 					method.getDeclaringClass().getName(), method.getName());
328 				return requireHttpScheme(value.trim());
329 			}
330 			// Option B: a declarative base/host override (method-level wins over interface-level).
331 			var baseUrl = firstNonEmpty(methodMeta.getBaseUrl(), meta.getBaseUrl());
332 			if (isNotEmpty(baseUrl))
333 				return requireHttpScheme(combinePaths(baseUrl, fullPath));
334 			// Default: unchanged behavior (no override, no scheme restriction — no regression).
335 			return fullPath;
336 		}

338 		/**
339 		 * Enforces the SSRF guardrail: when {@code url} carries a URI scheme it must be {@code http} or
340 		 * {@code https}; otherwise an {@link IllegalArgumentException} is thrown.  Scheme-less (relative) values pass
341 		 * through unchanged.
342 		 */
343 		private static String requireHttpScheme(String url) {
344 			var scheme = schemeOf(url);
345 			assertArg(scheme == null || scheme.equalsIgnoreCase("http") || scheme.equalsIgnoreCase("https"),
346 				"Unsupported URL scheme '%s' in @Remote URL override; only http/https are allowed: %s", scheme, url);
347 			return url;
348 		}

350 		/**
351 		 * Returns the URI scheme of a URL (the token before the first {@code :} when it precedes any {@code /}, {@code ?}
352 		 * or {@code #}), or <jk>null</jk> if the value has no scheme (e.g. a relative path or a value beginning with a
353 		 * {@code {var}} token).
354 		 */
355 		private static String schemeOf(String url) {
356 			if (url.isEmpty() || ! Character.isLetter(url.charAt(0)))
357 				return null;
358 			for (var i = 0; i < url.length(); i++) {
359 				var c = url.charAt(i);
360 				if (c == ':')
361 					return url.substring(0, i);
362 				if (c == '/' || c == '?' || c == '#')
363 					return null;
364 				if (! (Character.isLetterOrDigit(c) || c == '+' || c == '-' || c == '.'))
365 					return null;
366 			}
367 			return null;
368 		}

370 		/**
371 		 * Applies the cross-cutting call policy to a freshly-built request: annotation-declared
372 		 * interceptors (interface-level then method-level, unioned <i>after</i> the builder-level interceptors in
373 		 * {@link RestRequest#run()}) and the per-call response timeout (method-level overrides interface-level).
374 		 */
375 		private void applyPolicy(RestRequest req, RrpcInterfaceMethodMeta methodMeta) {
376 			var interceptorClasses = new ArrayList<Class<?>>();
377 			interceptorClasses.addAll(Arrays.asList(meta.getInterceptorClasses()));
378 			interceptorClasses.addAll(Arrays.asList(methodMeta.getInterceptorClasses()));
379 			if (! interceptorClasses.isEmpty()) {
380 				var resolved = new RestCallInterceptor[interceptorClasses.size()];
381 				for (var i = 0; i < resolved.length; i++)
382 					resolved[i] = resolveInterceptor(interceptorClasses.get(i));
383 				req.interceptors(resolved);
384 			}
385 			var timeout = firstNonEmpty(methodMeta.getTimeout(), meta.getTimeout());
386 			if (timeout != null && ! timeout.isEmpty()) { // HTT: firstNonEmpty(...) only ever returns null or a genuinely non-empty string, so "non-null but empty" is structurally unreachable here.
387 				var ms = getDuration(timeout);
388 				if (ms > 0)
389 					req.timeout(Duration.ofMillis(ms));
390 			}
391 		}

396 		 * public no-arg constructor.

403 		 * Returns <jk>true</jk> if the method has a parameter that produces the request body (a {@code @Content}
404 		 * parameter, or a single unannotated parameter treated as the body), in which case a method-level
405 		 * {@code @Content(def)} constant body must not be applied independently.

407 		private static boolean hasBodyParam(Method method) {
408 			var params = method.getParameters();
409 			for (var p : params)
410 				if (p.getAnnotation(Content.class) != null)
411 					return true;
412 			return params.length == 1 && ! hasPartAnnotation(params[0]);

421 				|| param.getAnnotation(Request.class) != null

423 				|| param.getAnnotation(Url.class) != null
424 				|| param.getAnnotation(Part.class) != null;

428 		 * Applies always-applied constant parts (header/query/form-data) to the request, layering method-level
429 		 * constants on top of interface-level constants.
430 		 *
431 		 * <p>
432 		 * Both scopes are merged into a single name-keyed map (preserving declaration order) so that, when the same
433 		 * name is declared at both scopes, the method-level value wins (most-specific).  The merged constants are
434 		 * emitted before parameter binding so that a caller-supplied parameter value of the same name composes with
435 		 * (rather than erases) the constant.
436 		 *
437 		 * @param adder Sink that records a single name/value part on the request.
438 		 * @param interfaceLevel Interface-level constants (applied first; lower precedence).
439 		 * @param methodLevel Method-level constants (override interface-level entries of the same name).
440 		 */
441 		private static void applyConstants(BiConsumer<String,String> adder, List<Map.Entry<String,String>> interfaceLevel, List<Map.Entry<String,String>> methodLevel) {
442 			if (interfaceLevel.isEmpty() && methodLevel.isEmpty())
443 				return;
444 			var merged = new LinkedHashMap<String,String>();
445 			for (var e : interfaceLevel)
446 				merged.put(e.getKey(), e.getValue());
447 			for (var e : methodLevel)
448 				merged.put(e.getKey(), e.getValue());
449 			merged.forEach(adder);
450 		}

452 		/**
453 		 * Applies the always-applied constant <i>headers</i> (interface-level then method-level), with the
454 		 * precedence rule that a dedicated {@code contentType}/{@code accept} attribute wins over a constant
455 		 * {@code Content-Type}/{@code Accept} header of the same name.
456 		 *
457 		 * <p>
458 		 * The two scopes are merged (method-level wins on name collisions) and emitted in declaration order, except a
459 		 * constant {@code Content-Type} is dropped when {@code skipContentType} is set (the dedicated {@code contentType}
460 		 * attribute supplies it via the serialized body) and a constant {@code Accept} is dropped when {@code skipAccept}
461 		 * is set (the dedicated {@code accept} attribute supplies it after binding).  Dropping rather than emitting the
462 		 * constant guarantees a single header with the attribute's value.
463 		 */
464 		private void applyConstantHeaders(RestRequest req, List<Map.Entry<String,String>> interfaceLevel, List<Map.Entry<String,String>> methodLevel, boolean skipContentType, boolean skipAccept) {
465 			if (interfaceLevel.isEmpty() && methodLevel.isEmpty())
466 				return;
467 			var merged = new LinkedHashMap<String,String>();
468 			for (var e : interfaceLevel)
469 				merged.put(e.getKey(), e.getValue());
470 			for (var e : methodLevel)
471 				merged.put(e.getKey(), e.getValue());
472 			merged.forEach((k, v) -> {
473 				if (skipContentType && "Content-Type".equalsIgnoreCase(k))
474 					return;
475 				if (skipAccept && HEADER_ACCEPT.equalsIgnoreCase(k))

514 			"java:S3776" // Cognitive complexity acceptable for multi-part-type parameter binding dispatch
515 		})

517 			// @Url parameters supply the request URL (handled in resolveEffectiveUrl), not a part/body.
518 			if (param.getAnnotation(Url.class) != null)

522 			if (param.getAnnotation(Part.class) != null)

598 		 * @param schema The resolved part schema (from the annotation + {@code @Schema}).
599 		 * @param annValue The annotation {@code value()} (may be blank, or {@code "*"} for dynamic expansion).
600 		 * @param annName The annotation {@code name()} (may be blank).
601 		 * @param def The parameter-level default applied when {@code arg} is <jk>null</jk>.
602 		 * @param arg The argument value.
603 		 * @param fallbackName The name to use when the annotation specifies none (param or bean-property name).
604 		 * @param serializer The custom {@link HttpPartSerializer} to use, or <jk>null</jk> for the default.
605 		 * @param adder Sink that records a single name/value part on the request.
606 		 */

608 			"java:S107" // The 9 parameters form one cohesive HTTP-part binding descriptor (type/schema/name/default/value/serializer/sink) threaded through internal dispatch; a holder object would add indirection without improving this beta reflective-proxy code.

626 			if (serialized.isEmpty() && schema != null && schema.isSkipIfEmpty())
627 				return;
628 			adder.accept(firstNonEmpty(explicit, fallbackName), serialized);
629 		}

752 		/**
753 		 * Executes the request and processes the return value, applying safe automatic retries when
754 		 * the method opts in and all safety gates pass.
755 		 *
756 		 * <p>
757 		 * <b>Retry policy.</b> When a positive retry count is in effect and the return mode + verb + body permit it,
758 		 * each attempt rebuilds the request via {@code reqSupplier} (so a fresh, re-bound request is resent) and runs
759 		 * it; a connection failure ({@link TransportException}) or a retryable status ({@code 429}/{@code 5xx}) triggers
760 		 * another attempt (after a short exponential backoff) until the cap is reached.  Safety gates (all hard):
761 		 * <ul>
762 		 * 	<li>Only idempotent verbs (GET/PUT/DELETE/HEAD) auto-retry; POST/PATCH require {@code retryNonIdempotent}.
763 		 * 	<li>A non-repeatable body ({@link RestRequest#isBodyRepeatable()} == <jk>false</jk>) is never retried.
764 		 * 	<li>Streaming return modes ({@link RemoteReturn#RESPONSE}, raw {@link InputStream}/{@link Reader}, cursors)
765 		 * 		and wrapper returns ({@link Optional}/{@link Future}/{@link CompletableFuture}) are never retried.
766 		 * </ul>
767 		 */
768 		@SuppressWarnings({
769 			"java:S3776" // Cognitive complexity acceptable for the gated retry loop (verb/body/status safety gates + backoff).
770 		})
771 		private Object processReturn(RequestSupplier reqSupplier, RrpcInterfaceMethodMeta methodMeta, Method method, boolean throwOnError, String acceptFallback) throws Exception {
772 			var returnMode = methodMeta.getReturnType();
773 			var returnType = method.getReturnType();
774 			var genericReturnType = method.getGenericReturnType();
775 
776 			var maxRetries = (isRetryableMode(returnMode, returnType) && isRetryableVerb(methodMeta)) ? effectiveRetries(methodMeta) : 0;
777 			if (maxRetries == 0)
778 				return processReturnOnce(reqSupplier.get(), returnMode, method, throwOnError, acceptFallback);
779 
780 			var attempt = 0;
781 			while (true) {
782 				var req = reqSupplier.get();
783 				// Non-repeatable body: refuse to retry (resending a consumed stream would corrupt the call).
784 				if (attempt == 0 && ! req.isBodyRepeatable())
785 					return processReturnOnce(req, returnMode, method, throwOnError, acceptFallback);
786 				var shouldRetry = false;
787 				RestResponse resp = null;
788 				try {
789 					resp = req.run();
790 				} catch (TransportException e) {
791 					if (attempt++ >= maxRetries)
792 						throw e;
793 					backoff(attempt);
794 					shouldRetry = true;
795 				}
796 				if (! shouldRetry && resp != null && attempt < maxRetries && isRetryableStatus(resp.getStatusCode())) {
797 					resp.close();
798 					attempt++;
799 					backoff(attempt);
800 					shouldRetry = true;
801 				}
802 				if (shouldRetry)
803 					continue;
804 				try (var r = resp) { // HTT - exception during close() branch
805 					return materializeResponse(r, returnMode, returnType, genericReturnType, method, throwOnError, acceptFallback);
806 				}
807 			}
808 		}

810 		/** Executes a single (non-retried) attempt, including streaming / cursor / future / RESPONSE return modes. */
811 		private Object processReturnOnce(RestRequest req, RemoteReturn returnMode, Method method, boolean throwOnError, String acceptFallback) throws Exception {
812 			var returnType = method.getReturnType();
813 			var genericReturnType = method.getGenericReturnType();
814 			return switch (returnMode) {
815 			case BODY -> processBody(req, returnType, genericReturnType, method, throwOnError, acceptFallback);
816 			case RESPONSE -> req.run(); // caller must close
817 			default -> {
818 				try (var resp = req.run()) { // HTT - exception during close() branch
819 					yield materializeResponse(resp, returnMode, returnType, genericReturnType, method, throwOnError, acceptFallback);
820 				}
821 			}
822 			};
823 		}

825 		/** Materializes the terminal value for the buffered return modes (BODY/BEAN/STATUS/NONE) from a run response. */
826 		private Object materializeResponse(RestResponse resp, RemoteReturn returnMode, Class<?> returnType, Type genericReturnType, Method method, boolean throwOnError, String acceptFallback) throws Exception {
827 			return switch (returnMode) { // HTT: the RESPONSE case is structurally unreachable here -- both call sites (the retry loop, gated by isRetryableMode's explicit RemoteReturn.RESPONSE exclusion, and processReturnOnce's switch default, which never falls through for its own explicit RESPONSE case) guarantee returnMode != RESPONSE by the time this method runs.
828 			case BODY -> materializeBufferedBody(resp, returnType, genericReturnType, method, throwOnError, acceptFallback);
829 			case BEAN -> {
830 				// HTTP-response bean (e.g. Ok / NotFound): materialize from status + body rather than parsing.
831 				throwIfError(resp, method, throwOnError);
832 				yield instantiateHttpType(returnType, resp.getBodyAsString());
833 			}
834 			case STATUS -> {
835 				var sc = resp.getStatusCode();
836 				if (returnType == int.class || returnType == Integer.class)
837 					yield sc;
838 				if (returnType == boolean.class || returnType == Boolean.class)
839 					yield sc < 400;
840 				yield sc;
841 			}
842 			case NONE -> null;
843 			case RESPONSE -> throw new IllegalStateException("RESPONSE mode is not a buffered return mode"); // HTT
844 			};
845 		}
