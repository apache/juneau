38  /**
39   * Next-generation mock REST client that routes requests in-process to a Juneau {@link RestContext}.
40   *
41   * <p>
42   * Use this class to write serverless unit tests for {@link Rest @Rest}-annotated
43   * resource classes.  Requests are dispatched directly to the {@link RestContext} without any network activity,
44   * making tests fast and self-contained.
45   *
46   * <h5 class='section'>Example:</h5>
47   * <p class='bjava'>
48   * 	<ja>@Rest</ja>
49   * 	<jk>public class</jk> MyResource {
50   * 		<ja>@RestGet</ja>(<js>"/hello"</js>)
51   * 		<jk>public</jk> String hello() { <jk>return</jk> <js>"Hello!"</js>; }
52   * 	}
53   *
54   * 	<ja>@Test</ja>
55   * 	<jk>void</jk> test() <jk>throws</jk> Exception {
56   * 		<jk>try</jk> (<jv>client</jv> = MockRestClient.<jsm>create</jsm>(MyResource.<jk>class</jk>)) {
57   * 			var <jv>body</jv> = <jv>client</jv>.get(<js>"/hello"</js>).run().asString();
58   * 			assertEquals(<js>"\"Hello!\""</js>, <jv>body</jv>);
59   * 		}
60   * 	}
61   * </p>
62   *
63   * <p>
64   * <b>Beta — API subject to change:</b> This type is part of the next-generation REST client and HTTP stack
65   * ({@code org.apache.juneau.marshall.ng.*}).
66   * It is not API-frozen: binary- and source-incompatible changes may appear in the <b>next major</b> Juneau release
67   * (and possibly earlier).
68   *
69   * <h5 class='section'>See Also:</h5><ul>
70   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/NextGenRestClient">juneau-ng REST client</a>
71   * </ul>
72   *
73   * @since 9.2.1
74   */
75  @SuppressWarnings({

188 	/**
189 	 * Fluent builder for {@link MockRestClient}.
190 	 *
191 	 * @since 9.2.1
192 	 */
193 	public static final class Builder implements BeanStoreOverridable<Builder> {
194 
195 		private final Object impl;
196 		private String contextPath;
197 		private BeanStore overridingBeanStore;
198 		private Parser defaultParser;
199 
200 		private Builder(Object impl) {
201 			this.impl = impl;
202 		}
203 
204 		/**
205 		 * Sets the servlet context path (e.g. {@code "/myapp"}).
206 		 *
207 		 * @param value The context path. May be <jk>null</jk>.
208 		 * @return This object.
209 		 */
210 		public Builder contextPath(String value) {
211 			contextPath = toValidContextPath(value);
212 			return this;
213 		}
214 
215 		/**
216 		 * Installs a {@link BeanStore} as the {@code overridingParent} of the {@link RestContext}'s
217 		 * bean store, so test-time overrides (e.g. {@code TestBeanStore}) resolve at tier 1 of the
218 		 * lookup chain &mdash; above the resource's local {@code @Bean} factory entries.
219 		 *
220 		 * <p>
221 		 * Setting this disables the per-resource {@link RestContext} cache for the duration of this
222 		 * build, so the overlay is not silently shared with a previously-cached instance.
223 		 *
224 		 * <h5 class='section'>Example:</h5>
225 		 * <p class='bjava'>
226 		 * 	<jk>var</jk> <jv>overlay</jv> = <jk>new</jk> TestBeanStore().override(MyService.<jk>class</jk>, mockSvc);
227 		 * 	<jk>try</jk> (<jv>client</jv> = MockRestClient.<jsm>builder</jsm>(MyResource.<jk>class</jk>)
228 		 * 			.overridingBeanStore(<jv>overlay</jv>)
229 		 * 			.build()) {
230 		 * 		<jv>client</jv>.get(<js>"/widgets/1"</js>).run().assertStatus().is(200);
231 		 * 	}
232 		 * </p>
233 		 *
234 		 * @param value The override layer. Can be <jk>null</jk> to clear a previously-set value.
235 		 * @return This object.
236 		 *
237 		 * @since 10.0.0
238 		 */
239 		@Override
240 		public Builder overridingBeanStore(BeanStore value) {
241 			overridingBeanStore = value;
242 			return this;
243 		}
244 
245 		/**
246 		 * Sets an explicit default parser used when the response {@code Content-Type} matches no registered parser.
247 		 *
248 		 * <p>
249 		 * Setting this also causes the client to advertise the parser's media types in the {@code Accept} header,
250 		 * so the server must be able to produce that content type.  Omit this (the default) when the target
251 		 * resource does not have a JSON serializer configured.
252 		 *
253 		 * <h5 class='section'>Example:</h5>
254 		 * <p class='bjava'>
255 		 * 	<jk>try</jk> (<jv>client</jv> = MockRestClient.<jsm>builder</jsm>(MyJsonResource.<jk>class</jk>)
256 		 * 			.defaultParser(JsonParser.<jsf>DEFAULT</jsf>)
257 		 * 			.build()) {
258 		 * 		<jk>var</jk> <jv>api</jv> = <jv>client</jv>.getClient().remote(MyApi.<jk>class</jk>);
259 		 * 		<jv>api</jv>.someMethod();
260 		 * 	}
261 		 * </p>
262 		 *
263 		 * @param value The parser. May be <jk>null</jk>.
264 		 * @return This object.
265 		 */
266 		public Builder defaultParser(Parser value) {
267 			defaultParser = value;
268 			return this;
269 		}
270 
271 		/**
272 		 * Builds and returns the {@link MockRestClient}.
273 		 *
274 		 * @return A new instance. Never <jk>null</jk>.
275 		 */
276 		@SuppressWarnings({
277 			"java:S112" // Re-wrap as RuntimeException - build() wraps arbitrary checked exceptions from reflection (newInstance) and RestContext init/postInit; no specific type fits and the fluent build() method cannot declare checked exceptions.
278 		})
279 		public MockRestClient build() {
280 			try {
281 				var c = impl instanceof Class<?> c2 ? c2 : impl.getClass();
282 				var overlay = overridingBeanStore;
283 				RestContext restContext;
284 				if (overlay == null && restContextCache.containsKey(c)) {
285 					restContext = restContextCache.get(c);
286 				} else {
287 					var isClass = impl instanceof Class<?>;
288 					var o = isClass ? ((Class<?>)impl).getDeclaredConstructor().newInstance() : impl;
289 					restContext = new RestContext(new RestContext.Args(o.getClass(), null, null, () -> o, "", bs -> {
290 						bs.addBean(Enablement.class, CONDITIONAL);
291 						bs.addBeanType(CallLogger.class, BasicTestCallLogger.class);
292 					}, overlay, null, RestContext.ContextKind.ROOT)).postInit().postInitChildFirst();
293 					if (overlay == null)
294 						restContextCache.put(c, restContext);
295 				}
296 
297 				var servletPath = toValidContextPath(restContext.getFullPath());
298 				var fullContextPath = emptyIfNull(contextPath) + emptyIfNull(servletPath);
299 
300 			var transport = new JuneauRestTransport(restContext, servletPath);
301 			var clientBuilder = RestClient.builder()
302 				.transport(transport)
303 				.rootUrl("http://localhost" + fullContextPath)
304 				.defaultSerializer(JsonSerializer.DEFAULT);
305 			if (defaultParser != null)
306 				clientBuilder.defaultParser(defaultParser);
307 			var ngClient = clientBuilder.build();
308 
309 				return new MockRestClient(ngClient);
310 			} catch (Exception e) {
311 				throw rex(e, "Could not initialize MockRestClient");
312 			}
313 		}
314 
315 		private static String emptyIfNull(String s) { return s == null ? "" : s; }
316 	}

322 	/**
323 	 * An {@link HttpTransport} that dispatches requests in-process through a Juneau {@link RestContext}.
324 	 */
325 	private static final class JuneauRestTransport implements HttpTransport {
326 
327 		private final RestContext restContext;
328 		private final Object restObject;
329 		private final String servletPath;
330 
331 		JuneauRestTransport(RestContext restContext, String servletPath) {
332 			this.restContext = restContext;
333 			this.restObject = restContext.getResource();
334 			this.servletPath = servletPath;
335 		}
336 
337 		@Override /* HttpTransport */
338 		public TransportResponse execute(TransportRequest request) throws TransportException {
339 			try {
340 				var method = request.getMethod();
341 				var uri = request.getUri().toString();
342 
343 				var servletReq = MockServletRequest.create(method, uri)
344 					.servletPath(servletPath)
345 					.contextPath("");
346 
347 				for (var h : request.getHeaders())
348 					servletReq.header(h.name(), h.value());
349 
350 				var body = request.getBody();
351 				if (body != null) {
352 					var baos = new ByteArrayOutputStream();
353 					body.writeTo(baos);
354 					servletReq.content(baos.toByteArray());
355 				}
356 
357 				var servletRes = MockServletResponse.create();
358 
359 				restContext.execute(restObject, servletReq, servletRes);
360 
361 				if (servletRes.getStatus() == 0) // HTT: would require a buggy RestContext that never sets a status
362 					throw new TransportException("REST context returned status 0 — response was never committed");
363 
364 				var responseBuilder = TransportResponse.builder()
365 					.statusCode(servletRes.getStatus())
366 					.reasonPhrase(servletRes.getMessage());
367 
368 				for (var name : servletRes.getHeaderNames())
369 					for (var value : servletRes.getHeaders(name))
370 						responseBuilder.header(name, value);
371 
372 				var content = servletRes.getContent();
373 				if (content.length > 0)
374 					responseBuilder.body(new ByteArrayInputStream(content));
375 
376 				return responseBuilder.build();
377 
378 			} catch (TransportException e) {
379 				throw e;
380 			} catch (ServletException | IOException e) {
381 				throw new TransportException("In-process dispatch failed: " + e.getMessage(), e);
382 			}
383 		}
