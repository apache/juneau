35  /**
36   * Authorization-code grant (RFC 6749 &sect;4.1) flow helper with mandatory PKCE per RFC 7636.
37   *
38   * <p>
39   * Wraps Nimbus's {@link AuthorizationRequest} (for URL construction) + {@link AuthorizationCodeGrant} +
40   * {@link TokenRequest} (for the exchange) + {@link CodeChallenge#compute(CodeChallengeMethod, CodeVerifier)
41   * PKCE S256 challenge generation}.  {@link #buildAuthorizationUrl(String, CodeChallenge)} emits a plain OAuth
42   * 2.0 authorization request; {@link #buildAuthenticationUrl(String, CodeChallenge, String, Consumer)} emits an
43   * OpenID Connect authentication request carrying a {@code nonce}.  Supports RFC 8707 resource-indicator binding
44   * via {@link Builder#resource(URI)}.
45   *
46   * <h5 class='topic'>Usage</h5>
47   *
48   * <p class='bjava'>
49   * 	<jk>var</jk> flow = OAuthAuthorizationCodeFlow.<jsm>create</jsm>()
50   * 		.authorizationEndpoint(URI.<jsm>create</jsm>(<js>"https://idp.example.com/oauth2/authorize"</js>))
51   * 		.tokenEndpoint(URI.<jsm>create</jsm>(<js>"https://idp.example.com/oauth2/token"</js>))
52   * 		.clientId(<js>"web-app"</js>)
53   * 		.redirectUri(URI.<jsm>create</jsm>(<js>"https://app.example.com/callback"</js>))
54   * 		.scope(<js>"openid"</js>, <js>"profile"</js>)
55   * 		.build();
56   *
57   * 	<jk>var</jk> verifier = <jk>new</jk> CodeVerifier();
58   * 	<jk>var</jk> challenge = CodeChallenge.<jsm>compute</jsm>(CodeChallengeMethod.<jsf>S256</jsf>, verifier);
59   * 	<jk>var</jk> authUrl = flow.buildAuthorizationUrl(<js>"some-state"</js>, challenge);
60   *
61   * 	<jc>// After the IdP redirects back with ?code=...</jc>
62   * 	<jk>var</jk> token = flow.exchange(<jv>code</jv>, verifier);
63   * </p>
64   *
65   * <p>
66   * The flow does NOT manage state / PKCE-verifier storage; that's the caller's responsibility.
67   *
68   * @since 10.0.0
69   */

263 	/**
264 	 * Builds the {@code authorize?...} URL the user-agent should be redirected to.
265 	 *
266 	 * @param state The opaque {@code state} parameter (CSRF protection).  Must not be <jk>null</jk> or
267 	 * 	blank.
268 	 * @param codeChallenge The PKCE code challenge.  Must not be <jk>null</jk>.  Always uses S256.
269 	 * @return The authorization URL.
270 	 */
271 	@SuppressWarnings({
272 		"deprecation" // Nimbus CodeChallenge overload is deprecated; no simpler alternative yet.
273 	})
274 	public URI buildAuthorizationUrl(String state, CodeChallenge codeChallenge) {
275 		assertArgNotNullOrBlank("state", state);
276 		assertArgNotNull("codeChallenge", codeChallenge);
277 		Scope nimbusScope = scopes.isEmpty() ? null : new Scope(scopes.toArray(new String[0]));
278 		var requestBuilder = new AuthorizationRequest.Builder(new ResponseType(ResponseType.Value.CODE), new ClientID(clientId))
279 			.endpointURI(authorizationEndpoint)
280 			.redirectionURI(redirectUri)
281 			.scope(nimbusScope)
282 			.state(new State(state))
283 			.codeChallenge(codeChallenge, CodeChallengeMethod.S256);
284 		if (resource != null)
285 			requestBuilder.resource(resource);
286 		return requestBuilder.build().toURI();
287 	}

289 	/**
290 	 * Builds an OpenID Connect {@code authorize?...} URL carrying an OIDC {@code nonce} parameter.
291 	 *
292 	 * <p>
293 	 * Unlike {@link #buildAuthorizationUrl(String, CodeChallenge)} (which emits a plain OAuth 2.0
294 	 * {@link AuthorizationRequest} with no {@code nonce}), this method emits an OIDC
295 	 * {@link AuthenticationRequest} so the IdP echoes the supplied {@code nonce} back inside the issued
296 	 * ID token.  An OpenID Connect relying party stores the {@code nonce} before the redirect and
297 	 * verifies the {@code nonce} claim on the returned ID token to defeat token replay.
298 	 *
299 	 * <p>
300 	 * The configured {@code scope(...)} set is used as-is; OIDC requires it to contain {@code openid}.
301 	 * The optional {@code customizer} receives the underlying {@link com.nimbusds.openid.connect.sdk.AuthenticationRequest.Builder} so
302 	 * callers can set OIDC-specific parameters (e.g. {@code prompt}, {@code max_age}, {@code acr_values})
303 	 * without this module needing to surface each one.
304 	 *
305 	 * @param state The opaque {@code state} parameter (CSRF protection).  Must not be <jk>null</jk> or
306 	 * 	blank.
307 	 * @param codeChallenge The PKCE code challenge.  Must not be <jk>null</jk>.  Always uses S256.
308 	 * @param nonce The OIDC {@code nonce} value.  Must not be <jk>null</jk> or blank.
309 	 * @param customizer Optional hook on the {@link com.nimbusds.openid.connect.sdk.AuthenticationRequest.Builder} for OIDC-specific
310 	 * 	parameters.  May be <jk>null</jk>.
311 	 * @return The authorization URL.
312 	 */
313 	@SuppressWarnings({
314 		"deprecation" // Nimbus CodeChallenge overload is deprecated; no simpler alternative yet.
315 	})
316 	public URI buildAuthenticationUrl(String state, CodeChallenge codeChallenge, String nonce, Consumer<AuthenticationRequest.Builder> customizer) {
317 		assertArgNotNullOrBlank("state", state);
318 		assertArgNotNull("codeChallenge", codeChallenge);
319 		assertArgNotNullOrBlank("nonce", nonce);
320 		Scope nimbusScope = scopes.isEmpty() ? new Scope("openid") : new Scope(scopes.toArray(new String[0]));
321 		var builder = new AuthenticationRequest.Builder(new ResponseType(ResponseType.Value.CODE), nimbusScope, new ClientID(clientId), redirectUri)
322 			.endpointURI(authorizationEndpoint)
323 			.state(new State(state))
324 			.nonce(new Nonce(nonce))
325 			.codeChallenge(codeChallenge, CodeChallengeMethod.S256);
326 		if (customizer != null)
327 			customizer.accept(builder);
328 		return builder.build().toURI();
329 	}

331 	/**
332 	 * Exchanges the authorization code returned by the IdP for an access token.
333 	 *
334 	 * @param code The {@code code} query parameter from the IdP redirect.  Must not be <jk>null</jk> or
335 	 * 	blank.
336 	 * @param codeVerifier The PKCE verifier matching the challenge passed to
337 	 * 	{@link #buildAuthorizationUrl(String, CodeChallenge)}.  Must not be <jk>null</jk>.
338 	 * @return The acquired token.
339 	 */
340 	public OAuthToken exchange(String code, CodeVerifier codeVerifier) {
341 		assertArgNotNullOrBlank("code", code);
342 		assertArgNotNull("codeVerifier", codeVerifier);
343 		var grant = new AuthorizationCodeGrant(new AuthorizationCode(code), redirectUri, codeVerifier);
344 		TokenRequest.Builder reqBuilder;
345 		if (clientSecretSupplier != null) {
346 			var clientAuth = new ClientSecretBasic(new ClientID(clientId), new Secret(clientSecretSupplier.get()));
347 			reqBuilder = new TokenRequest.Builder(tokenEndpoint, clientAuth, grant);
348 		} else {
349 			reqBuilder = new TokenRequest.Builder(tokenEndpoint, new ClientID(clientId), grant);
350 		}
351 		if (resource != null)
352 			reqBuilder.resource(resource);
353 		return Flows.send(reqBuilder.build(), httpTimeout, httpRequestConfigurator);
354 	}
