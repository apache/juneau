20  
21  import java.io.*;
22  import java.time.*;
23  import java.util.*;
24  import java.util.function.*;

26  import com.nimbusds.oauth2.sdk.*;
27  import com.nimbusds.oauth2.sdk.http.*;

29  /**
30   * Helpers shared across all flow types.
31   *
32   * <p>
33   * Centralizes the Nimbus {@code TokenRequest}-to-{@code OAuthToken} round-trip plus the standard error mapping.
34   *
35   * @since 10.0.0
36   */
37  public final class Flows {
38  
39  	/** Default connect/read timeout applied to every token-endpoint request when the caller sets none. */
40  	static final Duration DEFAULT_HTTP_TIMEOUT = Duration.ofSeconds(10);
41  
42  	private Flows() {}
43  
44  	/**
45  	 * Sends the supplied {@link TokenRequest} via Nimbus's HTTP client and maps the response to
46  	 * {@link OAuthToken}.
47  	 *
48  	 * <p>
49  	 * A finite connect/read timeout is applied BEFORE the caller's {@code httpConfigurator} runs, so an unresponsive
50  	 * IdP can never wedge the call indefinitely while still letting callers override the value.
51  	 *
52  	 * @param req The Nimbus {@code TokenRequest}.  Never {@code null}.
53  	 * @param timeout The connect/read timeout.  {@code null} selects {@link #DEFAULT_HTTP_TIMEOUT}.
54  	 * @param httpConfigurator Optional pre-send hook on the {@link HTTPRequest}.
55  	 * @return The acquired token.
56  	 * @throws OAuthFlowException If the IdP returns an error or the HTTP round-trip fails.
57  	 */
58  	public static OAuthToken send(TokenRequest req, Duration timeout, Consumer<HTTPRequest> httpConfigurator) {
59  		var http = req.toHTTPRequest();
60  		applyTimeout(http, timeout);
61  		if (httpConfigurator != null)
62  			httpConfigurator.accept(http);
63  		HTTPResponse httpResp;
64  		try {
65  			httpResp = http.send();
66  		} catch (IOException e) {
67  			throw new OAuthFlowException("OAuth token endpoint HTTP call failed", e);
68  		}
69  		TokenResponse resp;
70  		try {
71  			resp = TokenResponse.parse(httpResp);
72  		} catch (ParseException e) {
73  			throw new OAuthFlowException("OAuth token-endpoint response could not be parsed", e);
74  		}
75  		if (!resp.indicatesSuccess()) {
76  			var err = resp.toErrorResponse().getErrorObject();
77  			var code = (err == null || err.getCode() == null) ? "unknown_error" : err.getCode();
78  			throw new OAuthFlowException("OAuth token endpoint error: " + code, code);
79  		}
80  		return toOAuthToken(resp.toSuccessResponse());
81  	}
82  
83  	/**
84  	 * Applies a finite connect/read timeout to a Nimbus {@link HTTPRequest} (Nimbus defaults to infinite).
85  	 *
86  	 * @param http The request.  Never {@code null}.
87  	 * @param timeout The timeout.  {@code null} selects {@link #DEFAULT_HTTP_TIMEOUT}.
88  	 */
89  	static void applyTimeout(HTTPRequest http, Duration timeout) {
90  		var t = timeout == null ? DEFAULT_HTTP_TIMEOUT : timeout;
91  		var ms = (int) Math.min(t.toMillis(), Integer.MAX_VALUE);
92  		http.setConnectTimeout(ms);
93  		http.setReadTimeout(ms);
94  	}
95  
96  	private static OAuthToken toOAuthToken(AccessTokenResponse success) {
97  		var tokens = success.getTokens();
98  		var access = tokens.getAccessToken();
99  		String tokenType = access.getType() != null ? access.getType().getValue() : "Bearer"; // HTT: null branch unreachable; Nimbus AccessToken always has a non-null AccessTokenType
100 		Instant expiresAt = computeExpiry(access);
101 		Optional<String> refreshToken = tokens.getRefreshToken() != null
102 			? o(tokens.getRefreshToken().getValue())
103 			: oe();
104 		Optional<Set<String>> scope = access.getScope() != null
105 			? o(new LinkedHashSet<>(access.getScope().toStringList()))
106 			: oe();
107 		Optional<String> idToken = oe();
108 		var custom = success.getCustomParameters();
109 		var v = custom.get("id_token");
110 		if (v instanceof String v2)
111 			idToken = o(v2);
112 		return new OAuthToken(access.getValue(), tokenType, expiresAt, refreshToken, scope, idToken);
113 	}
114 
115 	private static Instant computeExpiry(com.nimbusds.oauth2.sdk.token.AccessToken access) {
116 		var lifetime = access.getLifetime();
117 		if (lifetime <= 0L)
118 			return Instant.MAX;
119 		return Instant.now().plusSeconds(lifetime);
120 	}
121 }
