33  /**
34   * URL and connection policy for the HTTP calls an OAuth client makes to endpoints it did not choose itself
35   * (metadata documents, token endpoints, client-registration endpoints).
36   *
37   * <p>
38   * A policy instance does three things for each {@link HTTPRequest} passed to {@link #apply(HTTPRequest)}:
39   * <ul>
40   * 	<li>Checks the target URL: an absolute <c>https</c> URL with a host, no user info, and (unless private targets are
41   * 		allowed) a host that is not a loopback, private, link-local, metadata or otherwise reserved address.
42   * 		Numeric hosts that are not in canonical dotted-quad form, hosts with a trailing dot and <c>*.localhost</c> names
43   * 		are refused, and the host name is resolved so that a name pointing at a reserved address is refused too.
44   * 	<li>Disables redirect following.
45   * 	<li>For <c>https</c> requests, installs a socket factory that checks the address the connection was actually
46   * 		established to before any TLS data is exchanged.
47   * </ul>
48   *
49   * <p>
50   * Plain <c>http</c> is accepted only when explicitly allowed, or for loopback hosts when private targets are allowed.
51   * The connection-time address check applies to <c>https</c> only; for <c>http</c> only the resolve-time check applies.
52   * Neither check is applied when the request is routed through an explicit proxy.
53   *
54   * @since 10.0.0
55   */
56  public final class OAuthHttpPolicy {
57  
58  	/** The default policy: <c>https</c> required except for literal loopback hosts; no other private or reserved targets. */
59  	public static final OAuthHttpPolicy DEFAULT = new OAuthHttpPolicy(false, false, true, RemoteUrlPolicy.AddressResolver.DEFAULT);
60  
61  	private static final Pattern NUMERIC_LABEL = Pattern.compile("(?:\\d+|0[xX][0-9a-fA-F]*)");
62  
63  	private final boolean allowPrivateUrls;
64  	private final boolean allowInsecureHttp;
65  	private final boolean loopbackExempt;
66  	private final RemoteUrlPolicy.AddressResolver resolver;
67  
68  	private OAuthHttpPolicy(boolean allowPrivateUrls, boolean allowInsecureHttp, boolean loopbackExempt, RemoteUrlPolicy.AddressResolver resolver) {
69  		this.allowPrivateUrls = allowPrivateUrls;
70  		this.allowInsecureHttp = allowInsecureHttp;
71  		this.loopbackExempt = loopbackExempt;
72  		this.resolver = resolver;
73  	}

75  	/**
76  	 * Creates a policy.
77  	 *
78  	 * @param allowPrivateUrls <jk>true</jk> to allow loopback, private, link-local and metadata targets (and plain
79  	 * 	<c>http</c> to loopback hosts).
80  	 * @param allowInsecureHttp <jk>true</jk> to allow plain <c>http</c> to any host that otherwise passes the policy.
81  	 * @return A new policy.
82  	 */
83  	public static OAuthHttpPolicy of(boolean allowPrivateUrls, boolean allowInsecureHttp) {
84  		return new OAuthHttpPolicy(allowPrivateUrls, allowInsecureHttp, false, RemoteUrlPolicy.AddressResolver.DEFAULT);
85  	}

87  	/**
88  	 * Returns a copy of this policy that resolves host names with the given resolver.
89  	 *
90  	 * @param value The resolver.  Must not be <jk>null</jk>.
91  	 * @return A new policy.
92  	 */
93  	public OAuthHttpPolicy withResolver(RemoteUrlPolicy.AddressResolver value) {
94  		return new OAuthHttpPolicy(allowPrivateUrls, allowInsecureHttp, loopbackExempt, reqnn("value", value));
95  	}

97  	/**
98  	 * Returns a copy of this policy that always accepts literal loopback hosts (<c>localhost</c>, <c>127.0.0.1</c>,
99  	 * <c>[::1]</c>) over either scheme, without resolving them.
100 	 *
101 	 * @return A new policy.
102 	 */
103 	public OAuthHttpPolicy withLoopbackExempt() {
104 		return new OAuthHttpPolicy(allowPrivateUrls, allowInsecureHttp, true, resolver);
105 	}

107 	/**
108 	 * Returns a configurator that runs the given one (if any) and then applies this policy to the request.
109 	 *
110 	 * @param user The caller's configurator.  May be <jk>null</jk>.
111 	 * @return A configurator suitable for the OAuth flow builders.
112 	 */
113 	public Consumer<HTTPRequest> wrap(Consumer<HTTPRequest> user) {
114 		return http -> {
115 			if (user != null)
116 				user.accept(http);
117 			apply(http);
118 		};
119 	}

121 	/**
122 	 * Returns whether this policy allows private and loopback targets.
123 	 *
124 	 * @return <jk>true</jk> if private targets are allowed.
125 	 */
126 	public boolean isAllowPrivateUrls() {
127 		return allowPrivateUrls;
128 	}
129 
130 	/**
131 	 * Checks a URL against this policy without resolving its host.
132 	 *
133 	 * @param uri The URL.  Must not be <jk>null</jk>.
134 	 * @return The same URL.
135 	 * @throws IllegalArgumentException If the URL is not permitted.
136 	 */
137 	public URI requireAllowed(URI uri) {
138 		reqnn("uri", uri);
139 		var host = uri.getHost();
140 		if (ie(host))
141 			throw iaex("URL has no host: %s", uri);
142 		if (uri.getRawUserInfo() != null)
143 			throw iaex("URL must not contain user info: %s", uri);
144 		if (loopbackExempt && isLoopbackHost(host))
145 			return uri;
146 		RemoteUrlPolicy.requireAllowedUrl(uri.toString(), allowPrivateUrls);
147 		if (! allowPrivateUrls) {
148 			if (host.endsWith("."))
149 				throw iaex("URL host must not end with a dot: %s", uri);
150 			if (isNonCanonicalNumericHost(host))
151 				throw iaex("URL host is a numeric form that is not a canonical IP address: %s", uri);
152 			if (host.toLowerCase().endsWith(".localhost"))
153 				throw iaex("URL targets a loopback host: %s", uri);
154 		}
155 		if (! (eqic("https", uri.getScheme()) || allowInsecureHttp || (allowPrivateUrls && isLoopbackHost(host))))
156 			throw iaex("URL must use https: %s", uri);
157 		return uri;
158 	}
159 
160 	/**
161 	 * Checks a request's URL against this policy, resolves its host, disables redirects and installs the
162 	 * connection-time address check.
163 	 *
164 	 * @param http The request to prepare.  Must not be <jk>null</jk>.
165 	 * @throws IllegalArgumentException If the URL is not permitted or its host resolves to a reserved address.
166 	 */
167 	public void apply(HTTPRequest http) {
168 		reqnn("http", http);
169 		var uri = http.getURI();
170 		requireAllowed(uri);
171 		http.setFollowRedirects(false);
172 		if (allowPrivateUrls || (loopbackExempt && isLoopbackHost(uri.getHost())) || http.getProxy() != null)
173 			return;
174 		try {
175 			RemoteUrlPolicy.selectAllowedAddress(uri.getHost(), false, resolver);
176 		} catch (@SuppressWarnings("unused") UnknownHostException e) {
177 			return; // The connection attempt reports the unresolvable host itself.
178 		}
179 		if (eqic("https", uri.getScheme())) {
180 			var base = http.getSSLSocketFactory();
181 			if (base == null)
182 				base = HTTPRequest.getDefaultSSLSocketFactory();
183 			if (base == null)
184 				base = (SSLSocketFactory)SSLSocketFactory.getDefault();
185 			http.setSSLSocketFactory(new AddressCheckingSocketFactory(base));
186 		}
187 	}
188 
189 	private static boolean isNonCanonicalNumericHost(String host) {
190 		if (host.indexOf(':') >= 0)
191 			return false; // IPv6 literal; parsed canonically by the resolver.
192 		var labels = host.split("\\.", -1);
193 		for (var l : labels)
194 			if (! NUMERIC_LABEL.matcher(l).matches())
195 				return false;
196 		if (labels.length != 4)
197 			return true;
198 		for (var l : labels) {
199 			if (l.length() > 3 || (l.length() > 1 && l.charAt(0) == '0') || l.startsWith("0x") || l.startsWith("0X") || Integer.parseInt(l) > 255)
200 				return true;
201 		}
202 		return false;
203 	}

205 	/** Socket factory that refuses a connection whose established peer address is reserved. */
206 	private static final class AddressCheckingSocketFactory extends SSLSocketFactory {
207 
208 		private final SSLSocketFactory delegate;
209 
210 		AddressCheckingSocketFactory(SSLSocketFactory delegate) {
211 			this.delegate = delegate;
212 		}
213 
214 		private static <S extends Socket> S check(S socket) throws IOException {
215 			var addr = socket.getInetAddress();
216 			if (addr != null && RemoteUrlPolicy.isDeniedAddress(addr)) {
217 				try {
218 					socket.close();
219 				} catch (@SuppressWarnings("unused") IOException e) {
220 					// Best effort; the connection is being refused either way.
221 				}
222 				throw new IOException("Connection target address is not permitted: " + addr.getHostAddress());
223 			}
224 			return socket;
225 		}
226 
227 		@Override /* Overridden from SSLSocketFactory */
228 		public String[] getDefaultCipherSuites() {
229 			return delegate.getDefaultCipherSuites();
230 		}
231 
232 		@Override /* Overridden from SSLSocketFactory */
233 		public String[] getSupportedCipherSuites() {
234 			return delegate.getSupportedCipherSuites();
235 		}
236 
237 		@Override /* Overridden from SSLSocketFactory */
238 		public Socket createSocket(Socket s, String host, int port, boolean autoClose) throws IOException {
239 			check(s);
240 			return delegate.createSocket(s, host, port, autoClose);
241 		}
242 
243 		@Override /* Overridden from SSLSocketFactory */
244 		public Socket createSocket(Socket s, InputStream consumed, boolean autoClose) throws IOException {
245 			check(s);
246 			return delegate.createSocket(s, consumed, autoClose);
247 		}
248 
249 		@Override /* Overridden from SocketFactory */
250 		public Socket createSocket() throws IOException {
251 			return delegate.createSocket();
252 		}
253 
254 		@Override /* Overridden from SocketFactory */
255 		public Socket createSocket(String host, int port) throws IOException {
256 			return check(delegate.createSocket(host, port));
257 		}
258 
259 		@Override /* Overridden from SocketFactory */
260 		public Socket createSocket(String host, int port, InetAddress localHost, int localPort) throws IOException {
261 			return check(delegate.createSocket(host, port, localHost, localPort));
262 		}
263 
264 		@Override /* Overridden from SocketFactory */
265 		public Socket createSocket(InetAddress host, int port) throws IOException {
266 			return check(delegate.createSocket(host, port));
267 		}
268 
269 		@Override /* Overridden from SocketFactory */
270 		public Socket createSocket(InetAddress address, int port, InetAddress localAddress, int localPort) throws IOException {
271 			return check(delegate.createSocket(address, port, localAddress, localPort));
272 		}
273 	}
