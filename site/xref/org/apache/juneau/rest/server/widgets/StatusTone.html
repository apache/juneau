17  package org.apache.juneau.rest.server.widgets;
18  
19  import java.util.*;
20  import java.util.stream.*;
21  
22  /**
23   * The one closed status-tone palette shared by every surface that paints a semantic status colour:
24   * {@code info} / {@code success} / {@code warning} / {@code error} / {@code neutral}.
25   *
26   * <p>
27   * Exactly five values, and no synonyms.  Server-side checks accept a {@link #wire()} token from
28   * {@link #WIRE_TOKENS} and reject anything else; the client-side painters (the pill, the QuickStats tiles and the
29   * dialog-bar badge) silently ignore an off-palette token and apply no tone class.  The same five names therefore mean
30   * the same five things in Java, in the emitted {@code is-<tone>} CSS class, and in the {@code juneau-renders.js} tone
31   * map.
32   *
33   * <p>
34   * {@link Badge#tone} uses this same enum and serializes it as the enum name.
35   *
36   * <h5 class='section'>Example:</h5>
37   * <p class='bjava'>
38   * 	<jk>boolean</jk> <jv>ok</jv> = StatusTone.<jsm>isValid</jsm>(<js>"warning"</js>);  <jc>// true</jc>
39   * 	String <jv>token</jv> = StatusTone.<jsf>WARNING</jsf>.wire();  <jc>// "warning"</jc>
40   * </p>
41   *
42   * @since 10.0.0
43   */
44  public enum StatusTone {
45  
46  	/** Informational (neither good nor bad); maps to {@code is-info}. */
47  	INFO,
48  
49  	/** Healthy / completed; maps to {@code is-success}. */
50  	SUCCESS,
51  
52  	/** Approaching a limit or needing attention; maps to {@code is-warning}. */
53  	WARNING,
54  
55  	/** Failed / over a hard limit; maps to {@code is-error}. */
56  	ERROR,
57  
58  	/** No semantic colour; inherits the surrounding text colour and emits no tone class. */
59  	NEUTRAL;
60  
61  	/**
62  	 * The five legal wire tokens, in declaration order.
63  	 *
64  	 * <p>
65  	 * The single source of truth every server-side tone check reads, so no caller can hand-roll a fourth or sixth
66  	 * value.
67  	 */
68  	@SuppressWarnings({
69  		"java:S2386" // Public for adopters and tests; the value is an unmodifiable set, so it cannot be mutated despite the Set type.
70  	})
71  	public static final Set<String> WIRE_TOKENS = wireTokens();
72  
73  	private static Set<String> wireTokens() {
74  		Set<String> s = Arrays.stream(values()).map(StatusTone::wire).collect(Collectors.toCollection(LinkedHashSet<String>::new));
75  		return Collections.unmodifiableSet(s);
76  	}
77  
78  	/**
79  	 * Returns the lowercase wire token for this tone.
80  	 *
81  	 * @return The wire token (e.g. <c>"warning"</c>).
82  	 */
83  	public String wire() {
84  		return name().toLowerCase(Locale.ROOT);
85  	}
86  
87  	/**
88  	 * Returns <jk>true</jk> if the specified token is one of the five legal tones.
89  	 *
90  	 * @param token The candidate wire token.  Can be <jk>null</jk>.
91  	 * @return <jk>true</jk> if the token is a legal tone; <jk>false</jk> for <jk>null</jk>, blank, differently-cased,
92  	 * 	or off-palette values.
93  	 */
94  	public static boolean isValid(String token) {
95  		return token != null && WIRE_TOKENS.contains(token);
96  	}
97  }
