17  package org.apache.juneau.rest.server.widgets;
18  
19  /**
20   * The comparison operator of a declarative single-field rule.
21   *
22   * <p>
23   * One vocabulary, one place.  Every rule in the toolkit that tests one row field against one value shares these
24   * constants &mdash; {@link ActionRef#enabledWhen(String,Op,Object,String) the action-bar state rule} here, and the
25   * {@code RowClassRule} row decorator in the views module, and the views module's {@code VisibilityRule}.  It lives in this module because the widgets&rarr;views
26   * dependency does not exist, so the shared type has to sit on the side both can see.
27   *
28   * <p>
29   * {@link #IN}/{@link #CONTAINS} are collection operators (the value is a list, or the tested field is) used by
30   * visibility rules only.
31   *
32   * <p>
33   * {@link #EQ}/{@link #NE}/{@link #IN}/{@link #CONTAINS} <b>require</b> a comparison value; {@link #PRESENT}/{@link #ABSENT} test only whether the
34   * field is non-null/non-empty and therefore <b>omit</b> one.  {@link #requiresValue()} is the discriminator, and
35   * every host is expected to reject the mismatched form rather than ignore the surplus value.
36   *
37   * <p>
38   * Each constant also carries its lowercase wire token, for the hosts that do serialize a rule.
39   *
40   * @since 10.0.0
41   */
42  public enum Op {
43  
44  	/** Matches when the tested field equals the rule value. */
45  	EQ("eq"),
46  
47  	/** Matches when the tested field does not equal the rule value. */
48  	NE("ne"),
49  
50  	/** Matches when the tested field is non-null/non-empty (no value needed). */
51  	PRESENT("present"),
52  
53  	/** Matches when the tested field is missing/null (no value needed). */
54  	ABSENT("absent"),
55  
56  	/** Matches when the tested field equals one of the values in the rule's value list (the value is a list). */
57  	IN("in"),
58  
59  	/** Matches when the tested field (itself a list) contains the rule's value. */
60  	CONTAINS("contains");
61  
62  	private final String wire;
63  
64  	Op(String wire) {
65  		this.wire = wire;
66  	}
67  
68  	/**
69  	 * Returns the lowercase wire token for this operator.
70  	 *
71  	 * @return The wire token (e.g. <c>"eq"</c>).
72  	 */
73  	public String wire() {
74  		return wire;
75  	}
76  
77  	/**
78  	 * Whether this operator requires a comparison value ({@link #EQ}/{@link #NE}/{@link #IN}/{@link #CONTAINS}).
79  	 *
80  	 * @return <jk>true</jk> for {@link #EQ}, {@link #NE}, {@link #IN} and {@link #CONTAINS}, <jk>false</jk> for {@link #PRESENT} and
81  	 * 	{@link #ABSENT}.
82  	 */
83  	public boolean requiresValue() {
84  		return this == EQ || this == NE || this == IN || this == CONTAINS;
85  	}
86  
87  	/**
88  	 * Whether this operator is a collection operator ({@link #IN}/{@link #CONTAINS}).
89  	 *
90  	 * <p>
91  	 * Only the visibility rule evaluator understands these; the disable-with-reason rules accept the original four.
92  	 *
93  	 * @return <jk>true</jk> for {@link #IN} and {@link #CONTAINS}.
94  	 */
95  	public boolean isCollectionOp() {
96  		return this == IN || this == CONTAINS;
97  	}
98  }
