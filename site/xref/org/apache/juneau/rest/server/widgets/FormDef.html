26  /**
27   * The declarative source of a row-action modal's input form (the form half of a {@code present=dialog} action).
28   *
29   * <h5 class='section'>A widget contract driven by a view-module runtime</h5>
30   * <p>
31   * Like {@link ModalDef}, this is a general widget contract that lives in this module while the browser runtime that
32   * paints it ships with the rich-view module.  Nothing here depends on that module.
33   *
34   * <h5 class='section'>Flat or sectioned</h5>
35   * <p>
36   * A form declares either a flat {@link #fields} list or an ordered list of {@link Section}s, never both: the flat list
37   * <i>is</i> the single-section case.  Sections render as a ribbon-format strip above one visible pane; they are a
38   * layout of one dialog, not a stack of dialogs, so they cost nothing against the runtime's modal-depth cap.
39   *
40   * <h5 class='section'>Typed inputs, not markup</h5>
41   * <p>
42   * The client paints {@link #fields} as native {@code <input>} / {@code <textarea>} / {@code <select>} /
43   * {@code <button>} elements via {@code createElement}.  Labels and option text use {@code textContent}; prefills use
44   * {@code .value} / {@code .checked}.  The runtime never assigns {@code innerHTML} from this payload.  Only the
45   * types on the allowlist ({@code text}, {@code textarea}, {@code password}, {@code checkbox}, {@code toggle}, {@code select},
46   * {@code action}) are legal &mdash; that is the XSS bound: hostile prefill cannot become an element.
47   * </p>
48   * <p>
49   * {@link #template} may name a server-side template for authors.  It is <b>not</b> a client HTML sink; the
50   * shipped {@code juneau-views.js} ignores it.
51   * </p>
52   *
53   * <h5 class='section'>Per-widget contract version (fail-loud when a form is present)</h5>
54   * <p>
55   * This bean {@link Widget#validate() validates} fail-closed and carries an instance {@link #contractVersion}.  The
56   * version defaults to {@link #CONTRACT_VERSION} from the instant the form is constructed &mdash; a raw
57   * builder no longer leaks an unversioned form regardless of whether {@link #checked()} is invoked.  The client
58   * refuses to open a form-bearing dialog whose version does not match the one its runtime bakes in.
59   * </p>
60   * <p>
61   * This constant moves in <b>lockstep</b> with {@link ModalDef#CONTRACT_VERSION} and the runtime's baked-in literal,
62   * because the runtime compares one literal against both the modal's and the form's version.  Bumping any two of the
63   * three makes every form-bearing dialog refuse to open.
64   * </p>
65   *
66   * <h5 class='section'>Example:</h5>
67   * <p class='bjava'>
68   * 	FormDef <jv>form</jv> = FormDef.<jsm>create</jsm>()
69   * 		.field(FormDef.Input.<jsm>of</jsm>(<js>"resolution"</js>, <js>"Resolution comment"</js>, <js>"textarea"</js>)
70   * 			.required()
71   * 			.value(<js>""</js>));
72   * </p>
73   *
74   * <h5 class='section'>See Also:</h5>
75   * <ul>
76   * 	<li class='jc'>{@link ModalDef}
77   * 	<li class='jc'>{@link ActionRef}
78   * </ul>
79   *
80   * @since 10.0.0
81   */
82  @BeanType(properties="contractVersion,template,fields,sections")

84  	"java:S1845" // Fluent-builder setters intentionally mirror field names (Juneau DSL convention).
85  })

324 		/**
325 		 * Fail-closed per-input validation (called from {@link FormDef#validate()}).
326 		 *
327 		 * @throws IllegalArgumentException If this input is not well-formed.
328 		 */
329 		void validate() {
330 			if (ib(name))
331 				throw iaex("FormDef.Input name must not be null or blank.");
332 			var t = resolveType();
333 			var isAction = eq(t, "action");
334 			var isTextual = eqa(t, "text", "textarea");
335 
336 			validateOptions(eq(t, "select"));
337 			validateTextualOnlyFields(isTextual);
338 			validatePattern();
339 			validatePasswordValue();
340 			validateActionField(isAction);
341 		}
342 
343 		/** Resolves and validates {@link #type} (blank defaults to {@code text}). */
344 		private String resolveType() {
345 			var t = (ib(type)) ? "text" : type;
346 			if (! INPUT_TYPES.contains(t))
347 				throw iaex("FormDef.Input '%s' type must be one of text, textarea, password, checkbox, toggle, select, action, not '%s'.", name, t);
348 			return t;
349 		}
350 
351 		/** Validates the {@code select}-only {@link #options}/{@link #value} pairing. */
352 		private void validateOptions(boolean isSelect) {
353 			if (! isSelect) {
354 				if (options != null)
355 					throw iaex("FormDef.Input '%s' options are only allowed on a select field.", name);
356 				return;
357 			}
358 			if (ie(options))
359 				throw iaex("FormDef.Input '%s' select must declare at least one option.", name);
360 			var values = new HashSet<String>();
361 			for (var o : options) {
362 				if (o == null)
363 					throw iaex("FormDef.Input '%s' select option must not be null.", name);
364 				if (o.value == null)
365 					throw iaex("FormDef.Input '%s' select option value must not be null.", name);
366 				if (o.label == null || o.label.isBlank())
367 					throw iaex("FormDef.Input '%s' select option label must not be null or blank.", name);
368 				values.add(o.value);
369 			}
370 			if (value != null && ! values.contains(value))
371 				throw iaex("FormDef.Input '%s' select value '%s' does not match any option.", name, value);
372 		}
373 
374 		/** Validates that {@link #pattern}/{@link #maxLength} are only declared on a textual field. */
375 		private void validateTextualOnlyFields(boolean isTextual) {
376 			if (isTextual)
377 				return;
378 			if (pattern != null)
379 				throw iaex("FormDef.Input '%s' pattern is only allowed on a text/textarea field.", name);
380 			if (maxLength != null)
381 				throw iaex("FormDef.Input '%s' maxLength is only allowed on a text/textarea field.", name);
382 		}
383 
384 		/** A {@code password} field is write-only: a {@link #value} would be sent to the client, so it is rejected. */
385 		private void validatePasswordValue() {
386 			if (eq(type, "password") && value != null)
387 				throw iaex("FormDef.Input '%s' value is not allowed on a password field.", name);
388 		}
389 
390 		/** Validates {@link #pattern} compiles and {@link #maxLength} is positive. */
391 		private void validatePattern() {
392 			if (pattern != null) {
393 				if (pattern.length() > PATTERN_MAX_LENGTH)
394 					throw iaex("FormDef.Input '%s' pattern must be at most %s characters.", name, PATTERN_MAX_LENGTH);
395 				try {
396 					Pattern.compile(pattern);
397 				} catch (PatternSyntaxException e) {
398 					throw iaex("FormDef.Input '%s' pattern does not compile: %s", name, e.getMessage());
399 				}
400 			}
401 			if (maxLength != null && maxLength <= 0)
402 				throw iaex("FormDef.Input '%s' maxLength must be > 0.", name);
403 		}
404 
405 		/** Validates the {@code action}-only {@link #actionId}. */
406 		private void validateActionField(boolean isAction) {
407 			if (isAction && (ib(actionId)))
408 				throw iaex("FormDef.Input '%s' action must declare an actionId.", name);
409 			if (! isAction && actionId != null)
410 				throw iaex("FormDef.Input '%s' actionId is only allowed on an action field.", name);
411 		}
412 	}

414 	/**
415 	 * One labelled group of inputs inside a sectioned form: the multi-page dialog shape.
416 	 *
417 	 * <p>
418 	 * The client paints an ordered list of these as a <b>ribbon</b>-format strip above the form body, one visible pane
419 	 * at a time.  Switching sections is visibility only &mdash; nothing is refetched and nothing is re-created, which
420 	 * is why a value typed into one section survives a trip through another, and why an inline validation error stays
421 	 * attached to the control that owns it rather than to whichever section happens to be showing.
422 	 *
423 	 * <p>
424 	 * A sectioned form is <b>not</b> a stack of dialogs.  The strip opens no layer of its own, so a sectioned dialog
425 	 * consumes exactly one entry of the runtime's modal-depth budget, the same as a flat one.
426 	 *
427 	 * @since 10.0.0
428 	 */
429 	@BeanType(properties="id,label,fields")
430 	public static class Section {
431 
432 		/** The section's stable id, used as the strip tab's identity.  Must not be blank, and must be unique in the form. */
433 		public String id;
434 
435 		/** The visible tab label, painted with {@code textContent}.  Must not be blank. */
436 		public String label;
437 
438 		/** This section's typed inputs in display order.  Must declare at least one. */
439 		public List<Input> fields;
440 
441 		/**
442 		 * Creates an empty section (add {@link #field(Input) fields}).
443 		 *
444 		 * @param id The section's stable id.  Must not be <jk>null</jk> or blank.
445 		 * @param label The visible tab label.  Must not be <jk>null</jk> or blank.
446 		 * @return A new {@link Section}.
447 		 * @throws IllegalArgumentException If {@code id} or {@code label} is <jk>null</jk> or blank.
448 		 */
449 		public static Section of(String id, String label) {
450 			if (ib(id))
451 				throw iaex("FormDef.Section id must not be null or blank.");
452 			if (ib(label))
453 				throw iaex("FormDef.Section label must not be null or blank.");
454 			var s = new Section();
455 			s.id = id;
456 			s.label = label;
457 			return s;
458 		}
459 
460 		/**
461 		 * Adds one typed input to this section.
462 		 *
463 		 * @param value The input.  Must not be <jk>null</jk>.
464 		 * @return This object.
465 		 * @throws IllegalArgumentException If {@code value} is <jk>null</jk>.
466 		 */
467 		public Section field(Input value) {
468 			if (value == null)
469 				throw iaex("FormDef.Section field must not be null.");
470 			if (fields == null)
471 				fields = l();
472 			fields.add(value);
473 			return this;
474 		}
475 
476 		/**
477 		 * Fail-closed per-section validation (called from {@link FormDef#validate()}).
478 		 *
479 		 * @throws IllegalArgumentException If this section is not well-formed.
480 		 */
481 		void validate() {
482 			if (ib(id))
483 				throw iaex("FormDef.Section id must not be null or blank.");
484 			if (ib(label))
485 				throw iaex("FormDef.Section '%s' label must not be null or blank.", id);
486 			if (ie(fields))
487 				throw iaex("FormDef.Section '%s' must declare at least one field.", id);
488 			for (var f : fields) {
489 				if (f == null)
490 					throw iaex("FormDef.Section '%s' field must not be null.", id);
491 				f.validate();
492 			}
493 		}
494 	}

583 	 */
584 	public FormDef section(Section value) {
585 		if (value == null)
586 			throw iaex("FormDef section must not be null.");
587 		if (sections == null)
588 			sections = l();
589 		sections.add(value);
590 		return this;
591 	}
592 
593 	/**
594 	 * Fail-closed bean validation.
595 	 *
596 	 * <p>
597 	 * Does <b>not</b> require {@link #contractVersion} to already be set &mdash; a raw-built form validated directly
598 	 * must not false-refuse on a null version.  A form with neither {@link #fields} nor {@link #sections} is a no-op
599 	 * (a template-only / fieldless form is a shipped shape).
600 	 *
601 	 * <p>
602 	 * Field names are unique across the <b>whole</b> form, not per section: every control shares one submit body and
603 	 * one dialog-scoped element-id namespace, so two sections declaring the same name would collide in both.
604 	 *
605 	 * @throws IllegalArgumentException If both {@link #fields} and {@link #sections} are declared, if {@link #sections}
606 	 * 	is declared but empty, if two sections share an {@code id}, or if any field is not well-formed or two fields
607 	 * 	share a {@code name}.
608 	 */
609 	@Override
610 	public void validate() {
611 		if (fields != null && sections != null)
612 			throw iaex("FormDef declares both fields and sections; they are mutually exclusive (a flat fields list is the single-section case).");
613 		if (sections != null) {
614 			validateSections();
615 			return;
616 		}
617 		validateFlatFields();
618 	}
619 
620 	/** Validates the sectioned-form shape: at least one section, unique section ids, unique field names overall. */
621 	private void validateSections() {
622 		if (sections.isEmpty())
623 			throw iaex("FormDef sections must declare at least one section.");
624 		var ids = new HashSet<String>();
625 		var names = new HashSet<String>();
626 		for (var s : sections) {
627 			if (s == null)
628 				throw iaex("FormDef section must not be null.");
629 			s.validate();
630 			if (! ids.add(s.id))
631 				throw iaex("FormDef duplicate section id '%s'.", s.id);
632 			for (var f : s.fields)
633 				if (! names.add(f.name))
634 					throw iaex("FormDef duplicate field name '%s'.", f.name);
635 		}
636 	}
637 
638 	/** Validates the flat-form shape: each field is well-formed, and field names are unique. */
639 	private void validateFlatFields() {
640 		if (fields == null)
641 			return;
642 		var names = new HashSet<String>();
643 		for (var f : fields) {
644 			if (f == null)
645 				throw iaex("FormDef field must not be null.");
646 			f.validate();
647 			if (! names.add(f.name))
648 				throw iaex("FormDef duplicate field name '%s'.", f.name);
649 		}
650 	}

652 	/**
653 	 * The serving-path hook: re-stamps {@link #CONTRACT_VERSION} (already set by construction; this
654 	 * repairs a version a caller explicitly cleared) then {@link #validate() validates}.
655 	 *
656 	 * <p>
657 	 * {@link #contractVersion} no longer depends on this being called. Structural validation
658 	 * no longer depends on it either for a form served over REST: nested inside a returned {@link ModalDef},
659 	 * this form is reached by
660 	 * {@link WidgetsMixin.WidgetValidationProcessor
661 	 * WidgetsMixin.WidgetValidationProcessor}'s call to {@link ModalDef#validate()}, which recurses into this
662 	 * bean's own {@link #validate()}, fail-closed with a 500, whether or not the producing endpoint called
663 	 * {@link ModalDef#checked()}. This method (via {@link ModalDef#checked()}) remains the only fail-closed
664 	 * structural {@link #validate() validation} a <b>non-REST</b> producer has &mdash; a direct
665 	 * {@code Json.of(...)} serialization, for instance &mdash; and calling it for a REST-bound form stays safe
666 	 * (validation is idempotent) even though it is no longer strictly required there.
667 	 *
668 	 * @return This object.
669 	 * @throws IllegalArgumentException If this form is not well-formed.
670 	 */
671 	public FormDef checked() {
672 		contractVersion = CONTRACT_VERSION;
673 		validate();
674 		return this;
675 	}
