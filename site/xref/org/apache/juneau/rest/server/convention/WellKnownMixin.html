19  import java.io.*;
20  
21  import org.apache.juneau.http.response.*;
22  import org.apache.juneau.rest.server.*;
23  import org.apache.juneau.rest.server.servlet.*;
24  
25  /**
26   * Mixin that serves <a href="https://www.rfc-editor.org/rfc/rfc8615">RFC 8615</a>
27   * {@code /.well-known/*} discovery endpoints.
28   *
29   * <p>
30   * Sibling of {@link FaviconMixin}, {@link SeoMixin}, and
31   * {@link VersionMixin}. All four classes live in the
32   * {@code org.apache.juneau.rest.server.convention} convention-endpoints mixin pack.
33   *
34   * <p>
35   * In v1 the mixin mounts a single literal path &mdash; {@code /.well-known/security.txt} per
36   * <a href="https://www.rfc-editor.org/rfc/rfc9116">RFC 9116</a>. The class is structured so that
37   * future entries (for example {@code /.well-known/openid-configuration} from a separate AuthN
38   * mixin pack) can be added without touching this source: each future entry ships its own mixin
39   * class with its own {@link Rest#paths() @Rest(paths)} declaration.
40   *
41   * <p>
42   * Compose into a host resource via
43   * {@link Rest#mixins() @Rest(mixins=WellKnownMixin.class)}; the
44   * {@code /.well-known/security.txt} URL becomes available alongside the host's own endpoints with
45   * no further wiring.
46   *
47   * <h5 class='section'>Hardcoded mount path:</h5>
48   *
49   * <p>
50   * Unlike the sibling api-docs and ops mixins (see {@link org.apache.juneau.rest.server.docs.SwaggerMixin},
51   * {@link VersionMixin}, etc.), the mount path here is <b>not</b> SVL-configurable
52   * &mdash; {@code /.well-known/security.txt} is fixed by
53   * <a href="https://www.rfc-editor.org/rfc/rfc8615">RFC 8615</a> (Well-Known URIs) and
54   * <a href="https://www.rfc-editor.org/rfc/rfc9116">RFC 9116</a> (security.txt). Both RFCs are
55   * explicit that the {@code /.well-known/} prefix is the discovery convention and that the
56   * filename suffix is the protocol-defined registry key &mdash; a runtime override here would
57   * make the endpoint undiscoverable by spec-compliant clients.
58   *
59   * <h5 class='section'>Mixin-only deployment:</h5>
60   *
61   * <p>
62   * This resource is designed for composition via {@code @Rest(mixins=...)}. The mount path is
63   * pinned at the op level by {@link RestGet @RestGet(path="/.well-known/security.txt")} on
64   * {@link #getSecurityTxt}; a class-level {@code @Rest(paths=...)} declaration would be silently
65   * ignored under the mixin pattern (see {@link Rest#paths() @Rest(paths)} Javadoc).
66   *
67   * <h5 class='figure'>Composition example:</h5>
68   *
69   * <p class='bjava'>
70   * 	<ja>@Rest</ja>(path=<js>"/api"</js>, mixins=WellKnownMixin.<jk>class</jk>)
71   * 	<jk>public class</jk> ApiResource <jk>extends</jk> RestServlet { }
72   *
73   * 	<jc>// Configure the security.txt body via a @Bean factory:</jc>
74   * 	<ja>@Bean</ja> WellKnownMixin wellKnown() {
75   * 		<jk>return</jk> WellKnownMixin.<jsm>create</jsm>()
76   * 			.securityTxt(<js>"Contact: security@example.com\nExpires: 2027-01-01T00:00:00Z\n"</js>)
77   * 			.build();
78   * 	}
79   * </p>
80   *
81   * <h5 class='section'>Defaults &amp; behavior:</h5>
82   *
83   * <ul class='spaced-list'>
84   * 	<li><b>No default body.</b> Per RFC 9116, {@code security.txt}'s mere presence is meaningful;
85   * 		shipping a placeholder body would be misleading. If the builder is not given a body via
86   * 		{@link Builder#securityTxt(String) securityTxt(String)}, the endpoint returns
87   * 		{@code 404 Not Found} (i.e. "we don't have one"). Loud documentation in topic page.
88   * 	<li><b>Content-Type.</b> {@code text/plain; charset=UTF-8} when a body is configured.
89   * 	<li><b>Excluded from generated Swagger / OpenAPI specs</b> via
90   * 		{@link OpSwagger#ignore() @OpSwagger(ignore=true)}.
91   * </ul>
92   *
93   * <h5 class='section'>See Also:</h5><ul>
94   * 	<li class='jc'>{@link FaviconMixin}
95   * 	<li class='jc'>{@link SeoMixin}
96   * 	<li class='jc'>{@link VersionMixin}
97   * 	<li class='link'><a class="doclink" href="https://www.rfc-editor.org/rfc/rfc8615">RFC 8615 &mdash; Well-Known URIs</a>
98   * 	<li class='link'><a class="doclink" href="https://www.rfc-editor.org/rfc/rfc9116">RFC 9116 &mdash; security.txt</a>
99   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestServerComposition">REST Server &mdash; Composition (mixins, paths)</a>
100  * </ul>
101  *
102  * @since 10.0.0
103  */
104 // @formatter:off
105 @Rest
106 public class WellKnownMixin extends RestMixin {
107 
108 	/**
109 	 * Creates a new builder.
110 	 *
111 	 * @return A new builder.
112 	 */
113 	public static Builder create() {
114 		return new Builder();
115 	}
116 
117 	private final String securityTxt;
118 
119 	/** No-arg constructor &mdash; results in a 404 on {@code /.well-known/security.txt}. */
120 	public WellKnownMixin() {
121 		this(create());
122 	}
123 
124 	/**
125 	 * Builder constructor.
126 	 *
127 	 * <p>
128 	 * Stashes the programmatic {@link RestBuilder}(the builder itself, carrying any {@code @Rest}-level
129 	 * overrides such as {@code path}) via {@code super(builder)} so those values take precedence over this
130 	 * mixin's own {@link Rest @Rest} annotation.
131 	 *
132 	 * @param builder The builder. Must not be <jk>null</jk>.
133 	 */
134 	protected WellKnownMixin(Builder builder) {
135 		super(builder);
136 		securityTxt = builder.securityTxt;
137 	}
138 
139 	/**
140 	 * [GET /.well-known/security.txt] &mdash; emit the configured RFC 9116 body, or 404 when
141 	 * unset.
142 	 *
143 	 * @param res The current REST response.
144 	 * @throws IOException If an I/O error occurs while writing the response.
145 	 * @throws NotFound If no {@code security.txt} body is configured.
146 	 */
147 	@RestGet(
148 		path="/.well-known/security.txt",
149 		summary="security.txt",
150 		description="RFC 9116 security.txt — disclosure / contact info for the operator.",
151 		swagger=@OpSwagger(ignore=true)
152 	)
153 	public void getSecurityTxt(RestResponse res) throws IOException {
154 		if (securityTxt == null)
155 			throw new NotFound();
156 		try (var w = res.getDirectWriter("text/plain; charset=UTF-8")) {
157 			w.write(securityTxt);
158 		}
159 	}
160 
161 	/**
162 	 * Returns the configured {@code security.txt} body (test/inspection helper).
163 	 *
164 	 * @return The body, or <jk>null</jk> if unset.
165 	 */
166 	public String getSecurityTxtBody() {
167 		return securityTxt;
168 	}
169 
170 	/**
171 	 * Builder for {@link WellKnownMixin} instances.
172 	 *
173 	 * <p>
174 	 * Extends {@link org.apache.juneau.rest.server.servlet.RestMixin.Builder} so the mixin's bespoke {@link #securityTxt(String)}
175 	 * setter chains with true covariant returns alongside the inherited {@link RestBuilder}surface (e.g.
176 	 * {@code path}, {@code roleGuard}).
177 	 */
178 	public static class Builder extends RestMixin.Builder<WellKnownMixin,Builder> {
179 
180 		private String securityTxt;
181 
182 		/** Constructor &mdash; package access for {@link WellKnownMixin#create()}. */
183 		protected Builder() {
184 			super(WellKnownMixin.class);
185 		}
186 
187 		/**
188 		 * Sets the {@code security.txt} body that will be served at
189 		 * {@code /.well-known/security.txt}.
190 		 *
191 		 * <p>
192 		 * Per RFC 9116, the body must include at least one {@code Contact:} field and an
193 		 * {@code Expires:} field; the mixin does not validate the content &mdash; that's the
194 		 * caller's responsibility.
195 		 *
196 		 * @param value The body content. Must not be <jk>null</jk> or blank.
197 		 * @return This object.
198 		 */
199 		public Builder securityTxt(String value) {
200 			securityTxt = value;
201 			return this;
202 		}
203 
204 		/**
205 		 * Builds a {@link WellKnownMixin} instance.
206 		 *
207 		 * @return A configured instance.
208 		 */
209 		@Override /* AbstractRestBuilder */
210 		public WellKnownMixin build() {
211 			return new WellKnownMixin(this);
212 		}
213 	}
214 }
