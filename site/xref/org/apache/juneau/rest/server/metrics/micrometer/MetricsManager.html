17  package org.apache.juneau.rest.server.metrics.micrometer;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import org.apache.juneau.rest.server.*;
22  
23  import io.micrometer.core.instrument.*;
24  
25  /**
26   * Shared worker for the {@code /metrics} management endpoint:  resolves a consumer-provided Micrometer
27   * {@link MeterRegistry} from the host bean store and renders a scrape payload.
28   *
29   * <p>
30   * Consistent with {@code micrometer-core} being a {@code provided} dependency and Juneau's
31   * explicit-over-magic stance, this adapter <b>strictly consumes a consumer-provided registry</b> &mdash; it
32   * never auto-registers a default.  When no {@link MeterRegistry} bean is present it degrades cleanly
33   * (returns {@code null}, which the endpoints surface as HTTP 501).
34   *
35   * <p>
36   * A Prometheus text scrape is produced when the resolved registry is a {@code PrometheusMeterRegistry}.
37   * That type lives in the separate {@code micrometer-registry-prometheus} artifact which this module does
38   * <b>not</b> depend on, so its {@code scrape()} method is invoked <b>reflectively</b> &mdash; keeping the
39   * registry choice entirely the consumer's, with no extra dependency leaked here.
40   *
41   * <h5 class='section'>See Also:</h5><ul>
42   * 	<li class='jc'>{@link MicrometerMetricsRecorder}
43   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/ManagementSurface">Management Surface</a>
44   * </ul>
45   *
46   * @since 10.0.0
47   */
48  public class MetricsManager {
49  
50  	/**
51  	 * Resolves the consumer-provided {@link MeterRegistry} from the host context's bean store.
52  	 *
53  	 * @param context The REST context whose bean store is searched.  May be <jk>null</jk>.
54  	 * @return The registry, or <jk>null</jk> if none is registered (or the context is <jk>null</jk>).
55  	 */
56  	@SuppressWarnings({
57  		"resource" // The MeterRegistry is a shared, consumer-owned bean borrowed from the bean store; this adapter must not close it.
58  	})
59  	public MeterRegistry resolveRegistry(RestContext context) {
60  		if (context == null)
61  			return null;
62  		return context.getBeanStore().getBean(MeterRegistry.class).orElse(null);
63  	}
64  
65  	/**
66  	 * Renders a Prometheus text scrape from the registry, if it is a {@code PrometheusMeterRegistry}.
67  	 *
68  	 * <p>
69  	 * Invoked reflectively to avoid a hard dependency on {@code micrometer-registry-prometheus}.  Returns
70  	 * <jk>null</jk> when the registry is <jk>null</jk>, is not a Prometheus registry, or does not expose a
71  	 * no-arg {@code scrape()} method &mdash; the endpoints surface that as HTTP 501.
72  	 *
73  	 * @param registry The resolved registry, or <jk>null</jk>.
74  	 * @return The Prometheus exposition text, or <jk>null</jk> if no scrape is available.
75  	 */
76  	public String scrape(MeterRegistry registry) {
77  		if (registry == null)
78  			return null;
79  		try {
80  			var m = registry.getClass().getMethod("scrape");
81  			var out = m.invoke(registry);
82  			return out == null ? null : out.toString();
83  		} catch (@SuppressWarnings("unused") NoSuchMethodException e) {
84  			return null;  // Not a scrapeable (Prometheus) registry — caller degrades to 501.
85  		} catch (ReflectiveOperationException e) {
86  			throw rex(e, "Failed to scrape MeterRegistry %s", registry.getClass().getName());
87  		}
88  	}
89  }
