17  package org.apache.juneau.rest.server.metrics;
18  
19  import java.time.*;
20  
21  /**
22   * Default {@link MetricsRecorder} implementation that drops every event on the floor.
23   *
24   * <p>
25   * Used whenever no consumer-provided {@code MetricsRecorder} bean is reachable from the
26   * {@code RestContext}'s bean store. Ensures the {@link MetricsRecorder} contract is always satisfied
27   * &mdash; the framework can call {@code record(...)} unconditionally without a null-check.
28   *
29   * <p>
30   * The implementation is a single static-field constant ({@link #INSTANCE}); the {@code record(...)}
31   * method does nothing, allocates nothing, and JIT-inlines to an empty method body, so the
32   * off-by-default path has no observable runtime cost.
33   *
34   * @since 10.0.0
35   */
36  @SuppressWarnings({
37  	"java:S6548" // Intentional process-wide no-op singleton; the Singleton pattern is required to satisfy the MetricsRecorder SPI contract with zero allocation.
38  })
39  public final class NoOpMetricsRecorder implements MetricsRecorder {
40  
41  	/** Process-wide singleton instance. */
42  	public static final NoOpMetricsRecorder INSTANCE = new NoOpMetricsRecorder();
43  
44  	private NoOpMetricsRecorder() {}
45  
46  	@Override /* MetricsRecorder */
47  	@SuppressWarnings({
48  		"java:S6213" // 'record' is the established MetricsRecorder SPI method name; renaming would break the public API.
49  	})
50  	public void record(String opName, String httpMethod, String uriTemplate, int statusCode, Duration elapsed, Throwable error, String metricName, String metricTags) {
51  		// Intentionally empty — the default off-by-default behaviour is to drop every event.
52  	}
53  }
