17  package org.apache.juneau.rest.server.metrics;
18  
19  import java.time.*;
20  
21  /**
22   * SPI for receiving per-request metric events from {@code juneau-rest-server}.
23   *
24   * <p>
25   * One {@code record(...)} call fires after every {@code @RestOp} handler invocation &mdash; including
26   * exception paths. Bridge implementations translate the event into the format their downstream
27   * registry expects (Micrometer {@code io.micrometer.core.instrument.Timer}, OpenTelemetry
28   * metric instruments, Dropwizard {@code MetricRegistry}, etc.).
29   *
30   * <h5 class='topic'>Off-by-default contract</h5>
31   *
32   * <p>
33   * {@code juneau-rest-server} resolves the {@code MetricsRecorder} via
34   * {@code RestContext.getBeanStore().getBean(MetricsRecorder.class)}. When no bean is supplied,
35   * {@link NoOpMetricsRecorder#INSTANCE} is used and the framework never allocates a downstream
36   * registry, never reaches Micrometer / OpenTelemetry on its own, and never adds any per-request
37   * cost beyond a single static-field read. To opt in, the consumer registers a {@code MetricsRecorder}
38   * bean &mdash; typically via {@code @Bean MetricsRecorder} on the resource or its parent.
39   *
40   * <h5 class='topic'>Usage</h5>
41   *
42   * <p class='bjava'>
43   * 	<ja>@Rest</ja>
44   * 	<jk>public class</jk> MyResource <jk>extends</jk> RestServlet {
45   *
46   * 		<ja>@Bean</ja>
47   * 		<jk>public</jk> MetricsRecorder recorder(MeterRegistry <jv>registry</jv>) {
48   * 			<jk>return new</jk> MicrometerMetricsRecorder(<jv>registry</jv>);
49   * 		}
50   * 	}
51   * </p>
52   *
53   * <h5 class='topic'>Threading</h5>
54   *
55   * <p>
56   * Implementations <b>must</b> be thread-safe &mdash; the same {@code MetricsRecorder} bean is invoked
57   * concurrently from every request thread for the lifetime of the {@code RestContext}. The framework
58   * calls {@code record(...)} synchronously inside the request thread immediately after the response
59   * status is known, so implementations should keep the call cheap (constant-time tag computation,
60   * non-blocking registry writes).
61   *
62   * <h5 class='topic'>Event ordering</h5>
63   *
64   * <p>
65   * For a given {@code @RestOp} invocation, {@code record(...)} fires exactly once after the operation
66   * completes &mdash; whether the handler returned normally, threw a {@code BasicHttpException}, or
67   * triggered an internal-server-error. The {@code elapsed} parameter measures wall-clock time spent
68   * inside the framework's per-op invocation boundary (from just before parameter resolution to just
69   * after the handler returns or throws). It does <b>not</b> include response serialization performed
70   * later in the request pipeline.
71   *
72   * <h5 class='section'>See Also:</h5><ul>
73   * 	<li class='jc'>{@link NoOpMetricsRecorder}
74   * 	<li class='jc'>{@link org.apache.juneau.rest.server.tracing.TracerHook}
75   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestServerObservability">REST Server &mdash; Observability (Micrometer + OpenTelemetry)</a>
76   * </ul>
77   *
78   * @since 10.0.0
79   */
80  @FunctionalInterface
81  public interface MetricsRecorder {
82  
83  	/**
84  	 * Records one per-request metric event.
85  	 *
86  	 * @param opName The fully-qualified {@code @RestOp} method name (e.g.
87  	 * 	{@code "com.example.OrderResource.create(java.lang.String,com.example.Order)"}).
88  	 * 	Never <jk>null</jk>; never blank.
89  	 * @param httpMethod The uppercased HTTP method of the request (e.g. {@code "GET"}, {@code "POST"}).
90  	 * 	Never <jk>null</jk>; never blank.
91  	 * @param uriTemplate The {@code @RestOp} path template (e.g. {@code "/users/{id}"}), <b>not</b> the
92  	 * 	raw concrete URI. Bridges should use this as the {@code uri} tag to keep cardinality bounded.
93  	 * 	May be empty for resource-level operations that match {@code /*}; never <jk>null</jk>.
94  	 * @param statusCode The HTTP response status code (e.g. {@code 200}, {@code 404}, {@code 500}).
95  	 * 	If the handler threw before the status was set, this is the framework's inferred status (typically
96  	 * 	{@code 500} for an unmapped exception, or the exception's status for a {@code BasicHttpException}).
97  	 * @param elapsed Wall-clock duration of the {@code @RestOp} invocation. Never <jk>null</jk>; never
98  	 * 	negative.
99  	 * @param error The exception thrown by the handler, or <jk>null</jk> if the call completed normally.
100 	 * 	Bridges typically derive an {@code exception} tag from the throwable's simple class name, or
101 	 * 	{@code "None"} when null (mirrors Spring Boot's {@code WebMvcMetricsFilter} convention).
102 	 * @param metricName Per-op metric name override from {@link org.apache.juneau.rest.server.RestOp#metricName()}.
103 	 * 	Empty string (default) means the implementation should use its own default name derivation.
104 	 * 	Never <jk>null</jk>.
105 	 * @param metricTags Per-op additional metric tags from {@link org.apache.juneau.rest.server.RestOp#metricTags()}.
106 	 * 	Format: comma-separated {@code key=value} pairs (e.g. {@code "team=payments,region=us-east"}).
107 	 * 	Empty string (default) means no additional tags. Never <jk>null</jk>.
108 	 */
109 	@SuppressWarnings({
110 		"java:S6213", // 'record' is the established SPI method name; renaming would break the public observability API.
111 		"java:S107"   // The 8 parameters form the stable per-request metric event contract; a holder object would obscure this SPI signature.
112 	})
113 	void record(String opName, String httpMethod, String uriTemplate, int statusCode, Duration elapsed, Throwable error, String metricName, String metricTags);
114 
115 	/**
116 	 * Records one custom (non-request) observation event.
117 	 *
118 	 * <p>
119 	 * Unlike {@link #record(String, String, String, int, Duration, Throwable, String, String)}, this
120 	 * entry point is <b>not</b> tied to an HTTP request &mdash; it is the substrate for the explicit
121 	 * programmatic observation API ({@link org.apache.juneau.rest.server.observation.Observations}) so
122 	 * application code can time an arbitrary block of work. Bridge implementations record a timer named
123 	 * {@code metricName} carrying {@code metricTags} (plus an {@code exception} tag derived from
124 	 * {@code error}).
125 	 *
126 	 * <p>
127 	 * The default implementation does nothing &mdash; a bridge that does not override it simply does not
128 	 * record custom observations, and the no-backend path stays zero-allocation. The shipped Micrometer
129 	 * bridge overrides it.
130 	 *
131 	 * @param metricName The timer name (e.g. {@code "order.load"}). Never <jk>null</jk>; never blank.
132 	 * @param metricTags Additional tags as comma-separated {@code key=value} pairs (e.g.
133 	 * 	{@code "team=payments,region=us-east"}). Empty string means no additional tags. Never <jk>null</jk>.
134 	 * @param elapsed Wall-clock duration of the observed block. Never <jk>null</jk>; never negative.
135 	 * @param error The exception thrown by the observed block, or <jk>null</jk> if it completed normally.
136 	 */
137 	@SuppressWarnings({
138 		"java:S6213" // 'record' is the established SPI method name; renaming would break the public observability API.
139 	})
140 	default void record(String metricName, String metricTags, Duration elapsed, Throwable error) {
141 		// Default no-op: bridges that support custom observations override this.
142 	}
143 }
