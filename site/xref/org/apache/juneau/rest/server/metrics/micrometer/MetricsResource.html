17  package org.apache.juneau.rest.server.metrics.micrometer;
18  
19  import org.apache.juneau.http.response.*;
20  import org.apache.juneau.rest.server.*;
21  import org.apache.juneau.rest.server.servlet.*;
22  
23  /**
24   * Child-resource flavor of the {@code /metrics} endpoint.
25   *
26   * <p>
27   * Mounts as a routed child via {@link Rest#children() @Rest(children=MetricsResource.class)} under the
28   * subtree {@code /metrics} and delegates to a shared {@link MetricsManager} worker &mdash; the same logic
29   * the {@link MetricsMixin mixin} flavor uses, so the two forms cannot drift.  Degrades to HTTP 501 when no
30   * scrapeable registry bean is present.
31   *
32   * <h5 class='section'>See Also:</h5><ul>
33   * 	<li class='jc'>{@link MetricsMixin}
34   * 	<li class='jc'>{@link MetricsManager}
35   * </ul>
36   *
37   * @since 10.0.0
38   */
39  @Rest(path="/metrics")
40  public class MetricsResource extends BasicRestResource {
41  
42  	private final MetricsManager manager = new MetricsManager();
43  
44  	/**
45  	 * [GET /] - Prometheus scrape of the consumer-provided MeterRegistry.
46  	 *
47  	 * @param req The HTTP request.
48  	 * @return The Prometheus exposition text.
49  	 * @throws NotImplemented If no scrapeable {@link io.micrometer.core.instrument.MeterRegistry} bean is registered.
50  	 */
51  	@RestGet(
52  		path="/*",
53  		summary="Metrics scrape",
54  		description="Renders a Prometheus text scrape from a consumer-provided Micrometer MeterRegistry bean."
55  	)
56  	public String getMetrics(RestRequest req) {
57  		// A routed child's bean store inherits from its parent, so the consumer's MeterRegistry bean is
58  		// reachable via this context's bean store.
59  		var registry = manager.resolveRegistry(req.getContext());
60  		var scrape = manager.scrape(registry);
61  		if (scrape == null)
62  			throw new NotImplemented("No scrapeable MeterRegistry is registered (a PrometheusMeterRegistry bean is required).");
63  		return scrape;
64  	}
65  }
