133 		Builder(Method method, RestContext context, Supplier<Object> resourceSupplier) {
134 
135 			this.restContext = context;
136 			this.restMethod = method;
137 			this.restResourceSupplier = resourceSupplier == null ? context::getResource : resourceSupplier;
138 
139 			var ap = context.getMarshallingContext().getAnnotationProvider();
140 			var mi = MethodInfo.of(context.getResourceClass(), method);
141 			var resourceClass = ClassInfo.of(context.getResourceClass());
142 
143 			try {
144 				var vr = context.getVarResolver();
145 				var vrs = vr.createSession();
146 
147 				// Parent-to-child merge: class-level annotations (parent class chain) applied first, then method-level
148 				// (matching-methods chain, return type, package). Method annotations thus override class annotations
149 				// for the same property. LinkedHashSet deduplicates while preserving order.
150 				//
151 				// For a mixin op (the op's context is a mixin sub-context), the host's class-level @ContextApply
152 				// config (e.g. @HtmlDocConfig, @SerializerConfig) is prepended ahead of the mixin class's own class
153 				// annotations, so the effective precedence is method > mixin-class > host-class. This is the
154 				// op-level counterpart of the context-level inheritance walk and is gated by @Rest(noInherit) on
155 				// the mixin. For non-mixin contexts the host stream is empty (no behavior change).
156 				var hostClassAnnotations = context.getInheritedHostClassAnnotations(ap);
157 				var declaringClassAnnotations = rstream(ap.find(resourceClass, SELF, PARENTS));
158 				var methodAnnotations = rstream(ap.find(mi, SELF, MATCHING_METHODS, RETURN_TYPE, PACKAGE));
159 				var allAnnotationsSet = new LinkedHashSet<AnnotationInfo<?>>();
160 				hostClassAnnotations.forEach(allAnnotationsSet::add);
161 				declaringClassAnnotations.forEach(allAnnotationsSet::add);
162 				methodAnnotations.forEach(allAnnotationsSet::add);
163 				var work = AnnotationWorkList.of(vrs, allAnnotationsSet.stream().filter(CONTEXT_APPLY_FILTER));
164 
165 				apply(work);
166 
167 			} catch (Exception e) {
168 				throw new InternalServerError(e);
169 			}
170 		}

173 		public Builder copy() {
174 			throw new NoSuchMethodError("Not implemented.");
175 		}

268 	 *
269 	 * <p>
270 	 * Walks the {@code @Rest(converters)} class chain (parent-to-child) followed by the

272 	 * {@code noInherit={"converters"}} cuts off the class-chain contribution. An
273 	 * {@code @Bean RestConverterList} bean (either as a name-anonymous bean in the bean store

482 	 * The request-guard array for this operation. <b>Cross-bucket memoizer</b> — folds three
483 	 * annotation attributes ({@code guards}, {@code roleGuard}, {@code rolesDeclared}) into a
484 	 * single effective {@link RestGuardList}.
485 	 *
486 	 * <p>
487 	 * Walk order: class {@code @Rest} chain (parent-to-child, gated by op-level
488 	 * {@code noInherit={"guards"}}), then op {@code @RestOp} chain (parent-to-child). Concretely:
489 	 * <ul>
490 	 * 	<li>Each annotation contributes its {@code guards()} classes (appended in chain order).
491 	 * 	<li>Each annotation contributes its {@code rolesDeclared()} CDL into a single role-name set.

1835 	 *
1836 	 * <p>
1837 	 * To process the call, the following must be true:
1838 	 * <ul>
1839 	 * 	<li>Path pattern must match.
1840 	 * 	<li>Matchers (if any) must match.
1841 	 * </ul>

1843 	 * @param session The call to check.
1844 	 * @return
1845 	 * 	One of the following values:
1846 	 * 	<ul>
1847 	 * 		<li><c>0</c> - Path doesn't match.

1849 	 * 		<li><c>2</c> - Matches.
1850 	 * 	</ul>
1851 	 */

1853 
1854 		var pm = matchPattern(session);
1855 
1856 		if (pm == null)
1857 			return 0;
1858 
1859 		var rm = requiredMatchers.get();
1860 		var om = optionalMatchers.get();
1861 		if (rm.length == 0 && om.length == 0) {
1862 			session.urlPathMatch(pm);  // Cache so we don't have to recalculate.
1863 			return 2;
1864 		}
1865 
1866 		try {
1867 			var req = session.getRequest();
1868 
1869 			// If the method implements matchers, test them.
1870 			for (var m : rm)
1871 				if (! m.matches(req))
1872 					return 1;
1873 			if (om.length > 0) {
1874 				var matches = false;
1875 				for (var m : om)
1876 					matches |= m.matches(req);

1879 		}
