23  import java.util.concurrent.*;

31  /**
32   * A themeable/configurable asset-serving mixin's cache of classpath-shipped assets: reads each configured
33   * resource's bytes once, computes and caches its {@link ChecksumUtils#hash8(byte[]) content hash}, and wraps
34   * bytes as a cacheable {@link HttpResource} - the read+cache+hash+cache-buster-URL+serve bundle that
35   * {@code ConsoleChromeMixin} (juneau-rest-server-console-ui) and {@code ViewsMixin} (juneau-rest-server-views)
36   * previously each hand-rolled as their own private statics and static cache maps.
37   *
38   * <h5 class='section'>Version-anchor:</h5>
39   * <p>
40   * {@link #buildVersion()} resolves {@link Package#getImplementationVersion()} from the constructor-supplied
41   * <b>anchor</b> class, not from this class's own class/package. A mixin composes one {@link ClasspathAssetCache}
42   * per mixin class (mirroring the static caches this replaces) and anchors it on <b>its own</b> class, so the
43   * resolved implementation version - and therefore every {@code ?v=<buildVersion>-<hash8>} cache-buster URL this
44   * cache builds - matches what that mixin's own jar shipped with, exactly as it did before extraction. Anchoring
45   * on this class instead would resolve {@code juneau-rest-server}'s own implementation version for every
46   * consuming mixin, silently changing every consumer's cache-buster URLs.
47   *
48   * <h5 class='section'>Thread-safety:</h5>
49   * <p>
50   * Backed by {@link ConcurrentHashMap}s; safe to share across concurrent requests. Classpath-shipped resources
51   * are assumed immutable for the lifetime of the JVM, so bytes and hashes are computed at most once per resource.
52   *
53   * @since 10.0.0
54   */
55  public class ClasspathAssetCache {
56  
57  	private final Class<?> anchor;
58  	private final ConcurrentHashMap<String,byte[]> byteCache = new ConcurrentHashMap<>();
59  	private final ConcurrentHashMap<String,String> hashCache = new ConcurrentHashMap<>();
60  
61  	/**
62  	 * Constructor.
63  	 *
64  	 * @param anchor
65  	 * 	The class that anchors both classpath resource resolution ({@link Class#getResourceAsStream(String)}) and
66  	 * 	{@link #buildVersion()}'s {@link Package#getImplementationVersion()} resolution. Typically the owning
67  	 * 	mixin's own class (see the class javadoc's version-anchor section). Must not be <jk>null</jk>.
68  	 */
69  	public ClasspathAssetCache(Class<?> anchor) {
70  		this.anchor = rnn(anchor);
71  	}
72  
73  	/** Test-only accessor for the constructor-supplied anchor class. */
74  	Class<?> anchor() {
75  		return anchor;
76  	}
77  
78  	/**
79  	 * Reads (and caches) the given classpath resource's bytes, resolved relative to this cache's anchor class.
80  	 *
81  	 * @param classpathResource A classpath-root-absolute resource path (e.g. {@code "/org/apache/juneau/foo/bar.js"}).
82  	 * 	Must not be <jk>null</jk> and must resolve to an existing resource on the anchor's classpath.
83  	 * @return The resource's bytes (served from cache on every call after the first for that resource).
84  	 */
85  	public byte[] bytes(String classpathResource) {
86  		return byteCache.computeIfAbsent(classpathResource, this::read);
87  	}
88  
89  	private byte[] read(String classpathResource) {
90  		try (var in = anchor.getResourceAsStream(classpathResource)) {
91  			if (in == null)
92  				throw new IOException("Classpath resource not found: " + classpathResource);
93  			return IoUtils.readBytes(in);
94  		} catch (IOException e) {  // HTT: unreachable - callers only pass already-validated, jar-shipped classpath resources.
95  			throw new UncheckedIOException(e);
96  		}
97  	}
98  
99  	/**
100 	 * Computes (and caches) the given classpath resource's 8-hex-char content hash.
101 	 *
102 	 * @param classpathResource A classpath-root-absolute resource path.  Same constraints as {@link #bytes(String)}.
103 	 * @return The resource's {@link ChecksumUtils#hash8(byte[])} content hash.
104 	 */
105 	public String hash(String classpathResource) {
106 		return hashCache.computeIfAbsent(classpathResource, r -> ChecksumUtils.hash8(bytes(r)));
107 	}
108 
109 	/**
110 	 * Resolves the framework build version for asset cache-busting from this cache's anchor class's package,
111 	 * falling back to {@code "dev"} when unset (e.g. running from an IDE/test classpath rather than a packaged
112 	 * jar) - see the class javadoc's version-anchor section.
113 	 *
114 	 * @return The anchor's {@link Package#getImplementationVersion()}, or {@code "dev"} if unset.
115 	 */
116 	public String buildVersion() {
117 		var v = anchor.getPackage().getImplementationVersion();
118 		return v == null ? "dev" : v;  // HTT: the non-null branch only fires when running from a packaged jar with a manifest Implementation-Version - unreachable when tests run against unpackaged target/classes.
119 	}
120 
121 	/**
122 	 * Builds the {@code ?v=<buildVersion>-<hash8>} content-sensitive cache-buster query suffix for the given
123 	 * classpath resource, suitable for appending to that resource's served URL.
124 	 *
125 	 * @param classpathResource A classpath-root-absolute resource path.  Same constraints as {@link #bytes(String)}.
126 	 * @return The {@code "?v=" + buildVersion() + "-" + hash(classpathResource)} suffix.
127 	 */
128 	public String cacheBuster(String classpathResource) {
129 		return "?v=" + buildVersion() + "-" + hash(classpathResource);
130 	}
131 
132 	/**
133 	 * Wraps pre-computed bytes as a cacheable {@link HttpResource} carrying the given content type and
134 	 * {@code Cache-Control} header.
135 	 *
136 	 * @param bytes The response body bytes.  Must not be <jk>null</jk>.
137 	 * @param contentType The {@code Content-Type} header value (e.g. {@code "text/css;charset=utf-8"}).
138 	 * @param cacheControl The {@code Cache-Control} header value (e.g. {@code "max-age=86400, public"}).
139 	 * @return The wrapped, cacheable resource.
140 	 */
141 	public HttpResource wrap(byte[] bytes, String contentType, String cacheControl) {
142 		return HttpResourceBean.of(
143 			ByteArrayBody.of(bytes, contentType),
144 			list(ContentType.of(contentType), CacheControl.of(cacheControl)));
145 	}
146 
147 	/**
148 	 * Reads (and caches) the given classpath resource and wraps it as a cacheable {@link HttpResource} - the
149 	 * {@link #bytes(String)} + {@link #wrap(byte[],String,String)} bundle every classpath-asset-serving endpoint
150 	 * needs.
151 	 *
152 	 * @param classpathResource A classpath-root-absolute resource path.  Same constraints as {@link #bytes(String)}.
153 	 * @param contentType The {@code Content-Type} header value.
154 	 * @param cacheControl The {@code Cache-Control} header value.
155 	 * @return The wrapped, cacheable resource.
156 	 */
157 	public HttpResource serve(String classpathResource, String contentType, String cacheControl) {
158 		return wrap(bytes(classpathResource), contentType, cacheControl);
159 	}
160 }
