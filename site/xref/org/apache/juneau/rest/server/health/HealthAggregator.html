29  /**
30   * Shared health-probe aggregation worker bean.
31   *
32   * <p>
33   * Holds the capability logic for evaluating {@link HealthIndicator} beans and summarizing them into a
34   * probe response. The {@link HealthServlet} (servlet flavor) and {@link HealthMixin} (mixin flavor) are
35   * independent {@code @Rest} classes that each delegate to a {@code HealthAggregator}, so the two
36   * deployment forms cannot drift.
37   *
38   * @since 10.0.0
39   */
40  @SuppressWarnings({
41  	"resource" // BeanStore is not owned here; its lifecycle is managed by RestContext.
42  })
43  public class HealthAggregator {
44  
45  	/**
46  	 * Resolves the {@link HealthIndicator} beans registered in the given context's bean store.
47  	 *
48  	 * @param context The REST context whose bean store supplies the indicators.
49  	 * @return The indicator beans keyed by bean name.
50  	 */
51  	public Map<String,HealthIndicator> indicators(RestContext context) {
52  		return context.getBeanStore().getBeansOfType(HealthIndicator.class);
53  	}
54  
55  	/**
56  	 * Evaluates the supplied indicators, summarizes their statuses, and sets the response status.
57  	 *
58  	 * @param context The REST context (supplies the {@link HealthProbeSettings} timeout).
59  	 * @param indicators The indicators to evaluate.
60  	 * @param probe The probe filter, or {@code null} to evaluate all indicators.
61  	 * @param res The response whose status code is set (200 healthy, 503 down).
62  	 * @return The aggregated health payload.
63  	 */
64  	public HealthResponse aggregate(RestContext context, Map<String,HealthIndicator> indicators, HealthProbe probe, RestResponse res) {
65  		var out = new LinkedHashMap<String,ComponentHealth>();
66  		var timeout = context.getBeanStore().getBean(HealthProbeSettings.class).map(HealthProbeSettings::getTimeout).orElse(Duration.ofSeconds(1));
67  		var timeoutMillis = Math.max(1L, timeout.toMillis());
68  
69  		for (var e : indicators.entrySet()) {
70  			var indicator = e.getValue();
71  			if (probe != null && !indicator.probes().contains(probe))
72  				continue;
73  			var componentName = firstNonEmpty(e.getKey(), cns(indicator));
74  			out.put(componentName, runIndicator(componentName, indicator, timeoutMillis));
75  		}
76  
77  		var status = summarize(out.values());
78  
79  		// Readiness gate: when shutdown has flipped the readiness state out of service, force the READY probe
80  		// down so /readyz returns 503 before the connector closes.  /livez (LIVE) and /healthz (null) are unaffected.
81  		if (probe == HealthProbe.READY && ! ReadinessState.resolve(context.getBeanStore()).isReady()) {
82  			status = HealthStatus.DOWN;
83  			out.put("readiness", new ComponentHealth(HealthStatus.DOWN, Map.of("state", "OUT_OF_SERVICE")));
84  		}
85  
86  		res.setStatus(status == HealthStatus.DOWN ? 503 : 200);
87  		return new HealthResponse(status, out);
88  	}
89  
90  	private static ComponentHealth runIndicator(String name, HealthIndicator indicator, long timeoutMillis) {
91  		var future = CompletableFuture.supplyAsync(indicator::check);
92  		try {
93  			var h = future.get(timeoutMillis, MILLISECONDS);
94  			return ComponentHealth.from(h);
95  		} catch (TimeoutException e) {
96  			future.cancel(true);
97  			return ComponentHealth.from(Health.down(name, e).detail("error", "Health check timed out after " + timeoutMillis + "ms").build());
98  		} catch (ExecutionException e) {
99  			return ComponentHealth.from(Health.down(name, e.getCause()).build());
100 		} catch (InterruptedException e) {
101 			Thread.currentThread().interrupt();
102 			return ComponentHealth.from(Health.down(name, e).build());
103 		} catch (Exception e) {
104 			return ComponentHealth.from(Health.down(name, e).build());
105 		}
106 	}
107 
108 	private static HealthStatus summarize(Collection<ComponentHealth> components) {
109 		var hasUnknown = false;
110 		for (var c : components) {
111 			if (c.status == HealthStatus.DOWN)
112 				return HealthStatus.DOWN;
113 			if (c.status == HealthStatus.UNKNOWN)
114 				hasUnknown = true;
115 		}
116 		return hasUnknown ? HealthStatus.UNKNOWN : HealthStatus.UP;
117 	}

119 	/**
120 	 * Probe response payload.
121 	 */
122 	public static class HealthResponse {
123 		private final HealthStatus status;
124 		private final Map<String,ComponentHealth> components;
125 
126 		HealthResponse(HealthStatus status, Map<String,ComponentHealth> components) {
127 			this.status = status;
128 			this.components = components;
129 		}
130 
131 		/**
132 		 * @return The aggregated probe status.
133 		 */
134 		public HealthStatus getStatus() { return status; }
135 
136 		/**
137 		 * @return The per-component health entries.
138 		 */
139 		public Map<String,ComponentHealth> getComponents() { return components; }
140 	}
141 
142 	/**
143 	 * Component payload in the response.
144 	 */
145 	public static class ComponentHealth {
146 		private final HealthStatus status;
147 		private final Map<String,Object> details;
148 
149 		ComponentHealth(HealthStatus status, Map<String,Object> details) {
150 			this.status = status;
151 			this.details = details;
152 		}
153 
154 		static ComponentHealth from(Health h) {
155 			var details = new LinkedHashMap<>(h.getDetails());
156 			if (h.getError() != null)
157 				details.putIfAbsent("error", h.getError().toString());
158 			return new ComponentHealth(h.getStatus(), details);
159 		}
160 
161 		/**
162 		 * @return The component status.
163 		 */
164 		public HealthStatus getStatus() { return status; }
165 
166 		/**
167 		 * @return The component detail map.
168 		 */
169 		public Map<String,Object> getDetails() { return details; }
170 	}
171 }
