17  package org.apache.juneau.rest.server.observation;
18  
19  /**
20   * Handle to one in-flight custom (non-request) observation.
21   *
22   * <p>
23   * Returned by {@link Observer#start(String, String)} and by the {@link Observations} facade. An
24   * observation pairs a metric timer with a tracing span around an arbitrary block of work that has no
25   * associated HTTP request &mdash; the request-path equivalent is the framework-driven boundary in
26   * {@code RestOpInvoker}.
27   *
28   * <h5 class='topic'>Usage</h5>
29   *
30   * <p>
31   * The handle is {@link AutoCloseable} and is designed for try-with-resources. On {@link #close()} it
32   * records the elapsed time to the metrics recorder and ends the span. Mark a failure with
33   * {@link #setError(Throwable)} before the block exits so both the timer's {@code exception} tag and the
34   * span's error status reflect it:
35   *
36   * <p class='bjava'>
37   * 	<jk>try</jk> (Observation <jv>o</jv> = Observations.<jsm>observe</jsm>(<js>"order.load"</js>, <js>"team=payments"</js>)) {
38   * 		<jk>return</jk> loadOrder(<jv>id</jv>);
39   * 	} <jk>catch</jk> (RuntimeException <jv>e</jv>) {
40   * 		<jv>o</jv>.setError(<jv>e</jv>);
41   * 		<jk>throw</jk> <jv>e</jv>;
42   * 	}
43   * </p>
44   *
45   * <h5 class='topic'>No-backend contract</h5>
46   *
47   * <p>
48   * When no metrics recorder / tracer is installed, {@link Observations} hands back
49   * {@link #NOOP} &mdash; a process-wide singleton whose methods do nothing and whose timing is never
50   * taken, so an observation in an unconfigured process costs nothing beyond the try-with-resources
51   * scaffolding.
52   *
53   * @since 10.0.0
54   */
55  @SuppressWarnings({
56  	"resource" // NOOP is an intentional stateless no-op singleton held in a static field; its close() is empty, so there is nothing to leak.
57  })
58  public interface Observation extends AutoCloseable {
59  
60  	/**
61  	 * Process-wide no-op observation, returned whenever no backend is installed.
62  	 *
63  	 * <p>
64  	 * Drops the error on the floor and does nothing on close.
65  	 */
66  	Observation NOOP = new Observation() {
67  		@Override /* Observation */
68  		public void setError(Throwable error) {
69  			// Intentionally empty.
70  		}
71  
72  		@Override /* Observation */
73  		public void close() {
74  			// Intentionally empty.
75  		}
76  	};
77  
78  	/**
79  	 * Marks the observed block as failed.
80  	 *
81  	 * <p>
82  	 * The throwable is surfaced as the metric timer's {@code exception} tag and recorded on the tracing
83  	 * span (status {@code ERROR} + {@code exception.type} attribute) when the observation closes. Safe
84  	 * to call at most once; a {@code null} argument is treated as "no error".
85  	 *
86  	 * @param error The exception that caused the block to fail. May be <jk>null</jk> (treated as success).
87  	 */
88  	void setError(Throwable error);
89  
90  	/**
91  	 * Closes the observation &mdash; records the elapsed timer sample and ends the span.
92  	 *
93  	 * <p>
94  	 * Always called via the try-with-resources {@code finally} contract. Implementations must not throw.
95  	 */
96  	@Override /* AutoCloseable */
97  	void close();
98  }
