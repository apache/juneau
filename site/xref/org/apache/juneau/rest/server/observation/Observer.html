17  package org.apache.juneau.rest.server.observation;
18  
19  import static org.apache.juneau.commons.utils.AssertionUtils.*;
20  import static org.apache.juneau.commons.utils.Shorts.*;
21  
22  import java.time.*;
23  
24  import org.apache.juneau.rest.server.metrics.*;
25  import org.apache.juneau.rest.server.tracing.*;
26  
27  /**
28   * Factory for custom (non-request) {@link Observation}s, composing a {@link MetricsRecorder} and a
29   * {@link TracerHook}.
30   *
31   * <p>
32   * This is the explicit programmatic observation substrate for {@code juneau-rest-server}: it reuses the
33   * exact same two SPIs the request path uses ({@link MetricsRecorder}, {@link TracerHook}) and the same
34   * {@code metricName} / {@code metricTags} comma-separated tag model, but is decoupled from
35   * {@code RestRequest} so application code can instrument an arbitrary block of work. Juneau deliberately
36   * has no method-interception / AOP substrate, so observation is explicit (a try-with-resources block)
37   * rather than annotation-driven &mdash; consistent with the framework's explicit-over-magic philosophy.
38   *
39   * <h5 class='topic'>No-backend contract</h5>
40   *
41   * <p>
42   * An {@code Observer} built from the no-op SPIs ({@link #NOOP}) never takes a timestamp and hands back
43   * {@link Observation#NOOP}, so an observation in an unconfigured process costs nothing beyond the
44   * try-with-resources scaffolding. {@link #start(String, String)} short-circuits to the no-op handle
45   * whenever <i>both</i> backends are the no-op singletons.
46   *
47   * <h5 class='topic'>Wiring</h5>
48   *
49   * <p>
50   * Typically obtained from a {@code RestContext}'s bean store &mdash; register the same
51   * {@code @Bean MetricsRecorder} / {@code @Bean TracerHook} you already register for request-path
52   * observability, then build an {@code Observer} from them (or install one as the {@link Observations}
53   * default). Construct directly for tests:
54   *
55   * <p class='bjava'>
56   * 	Observer <jv>o</jv> = <jk>new</jk> Observer(<jk>new</jk> MicrometerMetricsRecorder(<jv>registry</jv>), <jk>new</jk> OtelTracerHook(<jv>otel</jv>));
57   * </p>
58   *
59   * <h5 class='section'>See Also:</h5><ul>
60   * 	<li class='jc'>{@link Observations}
61   * 	<li class='jc'>{@link Observation}
62   * 	<li class='jc'>{@link MetricsRecorder}
63   * 	<li class='jc'>{@link TracerHook}
64   * </ul>
65   *
66   * @since 10.0.0
67   */
68  @SuppressWarnings({
69  	"resource" // start(...) returns the Observation for the caller's try-with-resources; ActiveObservation holds the tracer Scope it closes in its own close().
70  })
71  public final class Observer {
72  
73  	/**
74  	 * Observer backed by the no-op metrics recorder and tracer &mdash; produces only
75  	 * {@link Observation#NOOP}.
76  	 */
77  	public static final Observer NOOP = new Observer(NoOpMetricsRecorder.INSTANCE, NoOpTracerHook.INSTANCE);
78  
79  	// Constant name intentionally uses camelCase to match observation/metrics naming conventions.
80  	@SuppressWarnings("java:S115")
81  	private static final String ARG_recorder = "recorder";
82  	// Constant name intentionally uses camelCase to match observation/metrics naming conventions.
83  	@SuppressWarnings("java:S115")
84  	private static final String ARG_tracer = "tracer";
85  	// Constant name intentionally uses camelCase to match observation/metrics naming conventions.
86  	@SuppressWarnings("java:S115")
87  	private static final String ARG_name = "name";
88  
89  	private final MetricsRecorder recorder;
90  	private final TracerHook tracer;
91  	private final boolean active;
92  
93  	/**
94  	 * Constructor.
95  	 *
96  	 * @param recorder The metrics recorder to time observations against. Must not be <jk>null</jk>;
97  	 * 	use {@link NoOpMetricsRecorder#INSTANCE} to disable metrics.
98  	 * @param tracer The tracer to open spans against. Must not be <jk>null</jk>; use
99  	 * 	{@link NoOpTracerHook#INSTANCE} to disable tracing.
100 	 */
101 	public Observer(MetricsRecorder recorder, TracerHook tracer) {
102 		this.recorder = assertArgNotNull(ARG_recorder, recorder);
103 		this.tracer = assertArgNotNull(ARG_tracer, tracer);
104 		this.active = recorder != NoOpMetricsRecorder.INSTANCE || tracer != NoOpTracerHook.INSTANCE;
105 	}
106 
107 	/**
108 	 * Starts a new observation.
109 	 *
110 	 * <p>
111 	 * When both backends are the no-op singletons this returns {@link Observation#NOOP} without taking a
112 	 * timestamp or opening a span. Otherwise it opens a span via {@link TracerHook#startSpan(String)} and
113 	 * captures a start timestamp; the paired timer sample is recorded on {@link Observation#close()}.
114 	 *
115 	 * @param name The observation name &mdash; used as both the span name and the metric timer name.
116 	 * 	Must not be <jk>null</jk> or blank.
117 	 * @param tags Additional metric tags as comma-separated {@code key=value} pairs (e.g.
118 	 * 	{@code "team=payments,region=us-east"}), or <jk>null</jk> / empty for none.
119 	 * @return The started {@link Observation}. Never <jk>null</jk>.
120 	 */
121 	public Observation start(String name, String tags) {
122 		assertArgNotNullOrBlank(ARG_name, name);
123 		if (! active)
124 			return Observation.NOOP;
125 		var scope = tracer.startSpan(name);
126 		return new ActiveObservation(recorder, scope, name, tags == null ? "" : tags);
127 	}
128 
129 	/**
130 	 * Returns whether this observer has at least one live backend (metrics or tracing).
131 	 *
132 	 * @return <jk>true</jk> if either the metrics recorder or the tracer is non-no-op.
133 	 */
134 	public boolean isActive() {
135 		return active;
136 	}
137 
138 	private static final class ActiveObservation implements Observation {
139 
140 		private final MetricsRecorder recorder;
141 		private final Scope scope;
142 		private final String name;
143 		private final String tags;
144 		private final long startNanos;
145 		private Throwable error;
146 
147 		ActiveObservation(MetricsRecorder recorder, Scope scope, String name, String tags) {
148 			this.recorder = recorder;
149 			this.scope = scope;
150 			this.name = name;
151 			this.tags = tags;
152 			this.startNanos = System.nanoTime();
153 		}
154 
155 		@Override /* Observation */
156 		public void setError(Throwable error) {
157 			this.error = error;
158 		}
159 
160 		@Override /* Observation */
161 		public void close() {
162 			var elapsed = Duration.ofNanos(System.nanoTime() - startNanos);
163 			try {
164 				if (nn(error))
165 					scope.setError(error);
166 			} finally {
167 				try {
168 					scope.close();
169 				} finally {
170 					recorder.record(name, tags, elapsed, error);
171 				}
172 			}
173 		}
174 	}
175 }
