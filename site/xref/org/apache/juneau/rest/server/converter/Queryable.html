29  /**
30   * Converter for enabling search/view/sort/paging support on response objects returned by a
31   * <c>@RestOp</c>-annotated method, backed by the BeanQuery engine.
32   *
33   * <p>
34   * When enabled, a response that's a {@link Collection} or object array of beans, or of {@link Map Maps}, can be
35   * filtered with the BeanQuery search language via the following request parameters (see {@link BeanQueryRequest}
36   * for the per-parameter Swagger docs):
37   * <ul class='spaced-list'>
38   * 	<li><c>&amp;search=</c> Comma-separated <c>column=expression</c> clauses.
39   * 	<li><c>&amp;view=</c> Comma-separated columns to return, in order.
40   * 	<li><c>&amp;sort=</c> Comma-separated sort columns; append <js>":desc"</js> (or <js>"-"</js>) for descending.
41   * 	<li><c>&amp;position=</c> 0-based index of the first row.
42   * 	<li><c>&amp;limit=</c> Maximum rows to return.
43   * 	<li><c>&amp;opts=</c> Options as <c>key=value</c> clauses (e.g. <c>counts=true</c>).
44   * </ul>
45   *
46   * <p>
47   * A response that isn't tabular &mdash; not a {@link Collection} or object array, or one whose first non-null element
48   * is a plain string, number, date or other scalar &mdash; is returned <b>unchanged</b>; the query parameters are
49   * ignored, exactly as they were for non-tabular results before 10.0.  An empty or all-null collection is likewise
50   * returned unchanged (there's no element to derive columns from).
51   *
52   * <p>
53   * <b>Bean rows</b> get one {@link InMemoryBeanQueryContext} per element class, built once (with any
54   * {@link QueryableSettings} bean applied) and cached for the lifetime of this converter instance, so
55   * {@link QueryableSettings} is read once per element class, not on every request.  <b>{@link Map} rows</b> get one
56   * column per key of the first non-null row (type inferred from that row's value by the same rule as bean
57   * properties, see {@link QueryProperty#searchTypeOf(Class)}: a {@link Number} &rarr; {@link SearchType#NUMERIC}, a
58   * {@link Boolean} &rarr; {@link SearchType#BOOLEAN}, an enum &rarr; {@link SearchType#ENUM}, a
59   * date/calendar/temporal &rarr; {@link SearchType#TIMESTAMP}, anything else (or a <jk>null</jk>) &rarr;
60   * {@link SearchType#TEXT}); this context is built on every
61   * request, since the key set can differ per response.  Keys that appear only in later rows are not queryable.
62   *
63   * <p>
64   * <c>&amp;opts=counts=true</c> adds {@link #TOTAL_HEADER} / {@link #MATCHED_HEADER} response headers carrying the
65   * session's <c>total</c>/<c>matched</c> row counts &mdash; each header is set only when the corresponding count
66   * is present.  Without it, neither header is set.  The response body itself stays a bare list either way.
67   *
68   * <h5 class='section'>Example:</h5>
69   * <p class='bjava'>
70   * 	<ja>@Rest</ja>(converters=Queryable.<jk>class</jk>)
71   * 	<jk>public class</jk> PeopleResource <jk>extends</jk> BasicRestServlet {
72   *
73   * 		<ja>@Bean</ja>
74   * 		<jk>public</jk> QueryableSettings queryableSettings() {
75   * 			<jk>return</jk> QueryableSettings.<jsm>create</jsm>().maxLimit(500).build();
76   * 		}
77   *
78   * 		<ja>@RestGet</ja>(<js>"/people"</js>)
79   * 		<jk>public</jk> List&lt;Person&gt; getPeople() {
80   * 			<jk>return</jk> <jf>people</jf>;
81   * 		}
82   * 	}
83   * </p>
84   * <p>
85   * 	<c>GET /people?search=age=$gte(18)&amp;sort=name,age:desc&amp;view=name,age&amp;position=20&amp;limit=10</c>
86   * </p>
87   * <p>
88   * 	A bad column or option (e.g. <c>?sort=noSuchColumn</c>) throws {@link BeanQuerySyntaxException}, which
89   * 	{@code RestContext.convertThrowable} maps to an HTTP&nbsp;400 carrying an <c>X-BeanQuery-Error</c> header and,
90   * 	with problem details on, a <js>"code"</js> member.
91   * </p>
92   * <p>
93   * 	<c>GET /people?search=age=$gte(18)&amp;opts=counts=true</c> adds count headers to the (still bare-list) body:
94   * </p>
95   * <p class='bcode'>
96   * 	X-BeanQuery-Total: 42
97   * 	X-BeanQuery-Matched: 7
98   * </p>
99   *
100  * <h5 class='section'>See Also:</h5><ul>
101  * 	<li class='jc'>{@link QueryableSettings} - Per-resource limits and defaults for this converter.
102  * 	<li class='jc'>{@link BeanQueryRequest} - The request parameters this converter binds.
103  * 	<li class='jc'>{@link InMemoryBeanQueryContext} - The engine this converter drives.
104  * 	<li class='ja'>{@link RestOp#converters()} - Registering converters with REST resources.
105  * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/Converters">Converters</a>
106  * </ul>
107  *
108  * @since 10.0.0
109  */

112 })

179 	private static final List<Class<?>> SIMPLE_ELEMENT_TYPES = List.of(
180 		CharSequence.class, Number.class, Boolean.class, Character.class, Enum.class,
181 		Date.class, Calendar.class, Temporal.class
182 	);
183 
184 	private final ConcurrentHashMap<Class<?>,Optional<InMemoryBeanQueryContext<Object>>> contextCache = new ConcurrentHashMap<>();
185 
186 	@Override /* Overridden from RestConverter */
187 	public Object convert(RestRequest req, Object o) {
188 		if (o == null)
189 			return null;
190 
191 		List<Object> rows;
192 		if (o instanceof Collection<?> c)
193 			rows = nonNull(c);
194 		else if (o instanceof Object[] a)
195 			rows = nonNull(Arrays.asList(a));
196 		else
197 			return o;
198 
199 		if (rows.isEmpty())
200 			return o;
201 
202 		var first = rows.get(0);
203 		Optional<Page<?>> page;
204 		if (first instanceof Map<?,?> m)
205 			page = convertMapRows(req, rows, m);
206 		else if (isSimpleType(first.getClass()))
207 			page = Optional.empty();
208 		else
209 			page = convertBeanRows(req, rows, first.getClass());
210 
211 		return page.isPresent() ? page.get().rows() : o;
212 	}
213 
214 	/**
215 	 * Returns the cached bean context for an element class, or <jk>null</jk> if none has been built (package-private, for tests).
216 	 *
217 	 * @param elementClass The row class.
218 	 * @return The cached context, or <jk>null</jk>.
219 	 */
220 	InMemoryBeanQueryContext<Object> cachedContext(Class<?> elementClass) {
221 		return contextCache.getOrDefault(elementClass, Optional.empty()).orElse(null);
222 	}
223 
224 	private static List<Object> nonNull(Collection<?> c) {
225 		return addAllNn(new ArrayList<Object>(c.size()), c);
226 	}
227 
228 	private Optional<Page<?>> convertBeanRows(RestRequest req, List<Object> rows, Class<?> elementClass) {
229 		var ctx = contextCache.computeIfAbsent(elementClass, c -> {
230 			if (QueryModel.of(c).getReadableProperties().isEmpty())
231 				return Optional.empty();
232 			var b = InMemoryBeanQueryContext.create((Class<Object>)c);
233 			return Optional.of(settingsOf(req).applyTo(b).build());
234 		});
235 		if (ctx.isEmpty())
236 			return Optional.empty();
237 		return Optional.of(run(req, ctx.get(), rows));
238 	}
239 
240 	private Optional<Page<?>> convertMapRows(RestRequest req, List<Object> rows, Map<?,?> first) {
241 		var b = InMemoryBeanQueryContext.<Object>create();
242 		var columns = 0;
243 		for (var e : first.entrySet()) {
244 			var name = String.valueOf(e.getKey());
245 			if (! name.isBlank()) {
246 				b.column(name, e.getValue() == null ? SearchType.TEXT : QueryProperty.searchTypeOf(e.getValue().getClass()));
247 				columns++;
248 			}
249 		}
250 		if (columns == 0)
251 			return Optional.empty();
252 		return Optional.of(run(req, settingsOf(req).applyTo(b).build(), rows));
253 	}
254 
255 	private static Page<?> run(RestRequest req, InMemoryBeanQueryContext<Object> ctx, List<Object> rows) {
256 		var query = req.getRequest(BeanQueryRequest.class);
257 		try (var s = ctx.getSession(rows)) {
258 			var view = query.getView();
259 			var page = view == null || view.isBlank() ? s.find(query) : s.findValues(query);
260 			setCountHeaders(req, page);
261 			return page;
262 		}
263 	}
264 
265 	private static void setCountHeaders(RestRequest req, Page<?> page) {
266 		page.total().ifPresent(v -> req.getResponse().setHeader(TOTAL_HEADER, String.valueOf(v)));
267 		page.matched().ifPresent(v -> req.getResponse().setHeader(MATCHED_HEADER, String.valueOf(v)));
268 	}
269 
270 	private static boolean isSimpleType(Class<?> c) {
271 		for (var t : SIMPLE_ELEMENT_TYPES)
272 			if (t.isAssignableFrom(c))
273 				return true;
274 		return false;
275 	}
276 
277 	@SuppressWarnings({
278 		"resource" // The bean store is owned by the RestContext; this only borrows a bean and must not close it.
279 	})
280 	private static QueryableSettings settingsOf(RestRequest req) {
281 		return req.getContext().getBeanStore().getBean(QueryableSettings.class).orElse(QueryableSettings.DEFAULT);
282 	}
