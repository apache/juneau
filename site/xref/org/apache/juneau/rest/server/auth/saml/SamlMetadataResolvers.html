46  /**
47   * Convenience factories for building {@link MetadataResolver} instances backed by an OpenSAML implementation.
48   *
49   * <p>
50   * Juneau ships <i>convenience factories only</i> &mdash; no bundled default
51   * {@link MetadataResolver} singleton.  Applications either:
52   * <ul>
53   * 	<li>Call {@link #file(File)} for a local SAML metadata file (typical dev/staging setup).
54   * 	<li>Call {@link #url(String)} for a remote {@code /metadata} endpoint (typical production setup).
55   * 	<li>Build their own {@link MetadataResolver} directly using OpenSAML APIs (advanced).
56   * </ul>
57   *
58   * <p>
59   * Both factories return resolvers that are <b>already initialized</b> &mdash; ready to hand to
60   * {@link SamlAssertionValidator.Builder#metadataResolver(MetadataResolver)} immediately.
61   *
62   * <h5 class='section'>Notes:</h5>
63   * <p>
64   * The URL factory uses the JDK {@link HttpClient} (Java 17) to fetch the metadata blob once at construction
65   * time and wraps the parsed DOM in a {@link DOMMetadataResolver}.  No background refresh is performed; applications
66   * that need periodic refresh should construct their own {@code HTTPMetadataResolver} (from
67   * {@code opensaml-saml-impl}) with an Apache HttpClient + refresh policy.
68   *
69   * @since 10.0.0
70   */
71  public final class SamlMetadataResolvers {
72  
73  	private SamlMetadataResolvers() {}
74  
75  	/**
76  	 * Creates a {@link MetadataResolver} that loads SAML 2.0 metadata from the given file.
77  	 *
78  	 * <p>
79  	 * The file is parsed at construction time and re-parsed on disk modification by the underlying
80  	 * {@link FilesystemMetadataResolver}.
81  	 *
82  	 * @param file The metadata XML file on the local filesystem.  Must exist and be readable.
83  	 * @return An initialized {@link MetadataResolver}.
84  	 * @throws IOException If the file cannot be parsed or initialized.
85  	 */
86  	public static MetadataResolver file(File file) throws IOException {
87  		if (file == null)
88  			throw new IllegalArgumentException("file must not be null");
89  		if (!file.isFile()) // HTT: false branch (file present) requires valid SAML metadata XML; covered by integration tests
90  			throw new FileNotFoundException("SAML metadata file not found: " + file);
91  
92  		OpenSamlBootstrap.ensureInitialized();
93  		try {
94  			var parserPool = new BasicParserPool();
95  			parserPool.initialize();
96  
97  			var resolver = new FilesystemMetadataResolver(file);
98  			resolver.setId(file.getAbsolutePath());
99  			resolver.setParserPool(parserPool);
100 			resolver.setRequireValidMetadata(true);
101 			resolver.initialize();
102 			return resolver;
103 		} catch (ComponentInitializationException | ResolverException e) {
104 			throw ioex(e, "Failed to initialize FilesystemMetadataResolver for %s", file);
105 		}
106 	}
107 
108 	/**
109 	 * Creates a {@link MetadataResolver} that fetches SAML 2.0 metadata from the given URL once and serves it
110 	 * from an in-memory DOM.
111 	 *
112 	 * <p>
113 	 * Fetch happens on the calling thread with a 30-second connect/request timeout.  Applications that need
114 	 * periodic refresh should construct an OpenSAML {@code HTTPMetadataResolver} directly with an Apache
115 	 * HttpClient.
116 	 *
117 	 * <h5 class='section'>Transport:</h5>
118 	 * <p>
119 	 * The metadata blob supplies the trust-anchor signing certificate, so the transport must not be
120 	 * downgradeable by a network intermediary.  The URL must therefore use <js>"https"</js> or target a
121 	 * loopback host (<js>"localhost"</js>/<js>"127.0.0.1"</js>/<js>"::1"</js>, for local development); a
122 	 * plaintext <js>"http"</js> URL aimed at any other host is rejected.  When the IdP offers signed metadata,
123 	 * prefer {@link #url(String, X509Certificate)} so the metadata's own XML signature is verified against a
124 	 * pinned certificate independently of the transport.
125 	 *
126 	 * @param url The metadata URL.  Must use HTTPS or target a loopback host.
127 	 * @return An initialized {@link MetadataResolver}.
128 	 * @throws IOException If the URL cannot be fetched or the metadata is malformed.
129 	 */
130 	public static MetadataResolver url(String url) throws IOException {
131 		return url(url, null);
132 	}
133 
134 	/**
135 	 * Creates a {@link MetadataResolver} that fetches SAML 2.0 metadata from the given URL and verifies the
136 	 * metadata's XML signature against the supplied pinned certificate before trusting it.
137 	 *
138 	 * <p>
139 	 * This is the strongest option: the metadata document's own enveloped signature is validated against
140 	 * <jv>metadataSigningCert</jv>, so a substituted or tampered metadata blob is rejected regardless of the
141 	 * transport used to fetch it.  The same transport rule as {@link #url(String)} still applies.
142 	 *
143 	 * @param url The metadata URL.  Must use HTTPS or target a loopback host.
144 	 * @param metadataSigningCert The certificate whose public key signed the metadata document.  When
145 	 * 	<jk>null</jk>, no signature validation is performed (equivalent to {@link #url(String)}).
146 	 * @return An initialized {@link MetadataResolver}.
147 	 * @throws IOException If the URL cannot be fetched, the metadata is malformed, or its signature does not
148 	 * 	verify against the pinned certificate.
149 	 */
150 	public static MetadataResolver url(String url, X509Certificate metadataSigningCert) throws IOException {
151 		if (url == null)
152 			throw new IllegalArgumentException("url must not be null");
153 		UriUtils.assertSecureOrLoopback(URI.create(url));
154 
155 		OpenSamlBootstrap.ensureInitialized();
156 		try {
157 			var client = HttpClient.newBuilder()
158 				.connectTimeout(Duration.ofSeconds(30))
159 				.build();
160 			var req = HttpRequest.newBuilder()
161 				.uri(URI.create(url))
162 				.timeout(Duration.ofSeconds(30))
163 				.GET()
164 				.build();
165 			var resp = client.send(req, HttpResponse.BodyHandlers.ofByteArray());
166 			if (resp.statusCode() < 200 || resp.statusCode() >= 300) // HTT: false branch (2xx success) requires live SAML metadata endpoint; covered by integration tests
167 				throw ioex("Failed to fetch SAML metadata from %s (HTTP %s)", url, resp.statusCode());
168 
169 			var dbf = DocumentBuilderFactory.newInstance();
170 			dbf.setNamespaceAware(true);
171 			dbf.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);
172 			dbf.setFeature("http://xml.org/sax/features/external-general-entities", false);
173 			dbf.setFeature("http://xml.org/sax/features/external-parameter-entities", false);
174 			dbf.setXIncludeAware(false);
175 			dbf.setExpandEntityReferences(false);
176 			var doc = dbf.newDocumentBuilder().parse(new ByteArrayInputStream(resp.body()));
177 			Element root = doc.getDocumentElement();
178 
179 			var resolver = new DOMMetadataResolver(root);
180 			resolver.setId(url);
181 			resolver.setRequireValidMetadata(true);
182 			if (metadataSigningCert != null)
183 				resolver.setMetadataFilter(signatureValidationFilter(metadataSigningCert));
184 			resolver.initialize();
185 			return resolver;
186 		} catch (InterruptedException e) {
187 			Thread.currentThread().interrupt();
188 			throw ioex(e, "Interrupted while fetching SAML metadata from %s", url);
189 		} catch (ComponentInitializationException | javax.xml.parsers.ParserConfigurationException
190 				| org.xml.sax.SAXException e) {
191 			throw ioex(e, "Failed to initialize DOMMetadataResolver for %s", url);
192 		}
193 	}
194 
195 	private static MetadataFilter signatureValidationFilter(X509Certificate cert) throws ComponentInitializationException {
196 		var credential = new BasicX509Credential(cert);
197 		var trustEngine = new ExplicitKeySignatureTrustEngine(
198 			new StaticCredentialResolver(credential),
199 			DefaultSecurityConfigurationBootstrap.buildBasicInlineKeyInfoCredentialResolver());
200 		var filter = new SignatureValidationFilter(trustEngine);
201 		filter.setDefaultCriteria(new CriteriaSet(new UsageCriterion(org.opensaml.security.credential.UsageType.SIGNING)));
202 		filter.initialize();
203 		return filter;
204 	}

206 	/**
207 	 * Creates a {@link MetadataResolver} that loads SAML 2.0 metadata from the given {@link Path}.
208 	 *
209 	 * <p>
210 	 * Convenience overload of {@link #file(File)}; useful for callers that have a {@link Path} on hand.
211 	 *
212 	 * @param path The metadata XML file as a {@link Path}.
213 	 * @return An initialized {@link MetadataResolver}.
214 	 * @throws IOException If the file cannot be parsed or initialized.
215 	 */
216 	public static MetadataResolver file(Path path) throws IOException {
217 		if (path == null) // HTT: false branch (valid path) delegates to file(File); covered wherever file(File) is tested
218 			throw new IllegalArgumentException("path must not be null");
219 		return file(path.toFile());
220 	}
