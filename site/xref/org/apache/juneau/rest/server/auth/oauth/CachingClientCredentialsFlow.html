31  /**
32   * Server-side caching decorator layering an optional {@link TokenCache} over the role-neutral
33   * {@link OAuthClientCredentialsFlow}.
34   *
35   * <p>
36   * The neutral flow performs the RFC 6749 &sect;4.4 client-credentials token acquisition; this decorator adds the
37   * server-side {@code (clientId, scope)}-keyed cache so repeat callers within the cache window reuse the token rather
38   * than hitting the IdP on each call.  When no {@link Builder#tokenCache(TokenCache) cache} is configured the decorator
39   * simply delegates to the neutral flow.
40   *
41   * <h5 class='topic'>Usage</h5>
42   *
43   * <p class='bjava'>
44   * 	<jk>var</jk> token = CachingClientCredentialsFlow.<jsm>create</jsm>()
45   * 		.tokenEndpoint(URI.<jsm>create</jsm>(<js>"https://idp.example.com/oauth2/token"</js>))
46   * 		.clientId(<js>"worker-service"</js>)
47   * 		.clientSecret(<js>"..."</js>)
48   * 		.scope(<js>"read:orders"</js>)
49   * 		.tokenCache(BoundedLruTokenCache.<jsm>create</jsm>())
50   * 		.build()
51   * 		.acquire();
52   * </p>
53   *
54   * @since 10.0.0
55   */
56  public class CachingClientCredentialsFlow {
57  
58  	/**
59  	 * Static creator.
60  	 *
61  	 * @return A new builder.
62  	 */
63  	public static Builder create() {
64  		return new Builder();
65  	}

67  	/**
68  	 * Builder.
69  	 */
70  	public static class Builder {
71  		private URI tokenEndpoint;
72  		private String clientId;
73  		private Supplier<String> clientSecretSupplier;
74  		private Set<String> scopes = st();
75  		private TokenCache tokenCache;
76  		private Duration cacheSkew = Duration.ofSeconds(30);
77  		private Consumer<HTTPRequest> httpRequestConfigurator;
78  
79  		/** Constructor. */
80  		protected Builder() {}
81  
82  		/**
83  		 * Sets the token endpoint URL.  Required.
84  		 *
85  		 * @param value The endpoint URL.  Must not be <jk>null</jk>.
86  		 * @return This object.
87  		 */
88  		public Builder tokenEndpoint(URI value) {
89  			tokenEndpoint = assertArgNotNull("value", value);
90  			return this;
91  		}
92  
93  		/**
94  		 * Sets the OAuth client ID.  Required.
95  		 *
96  		 * @param value The client ID.  Must not be <jk>null</jk> or blank.
97  		 * @return This object.
98  		 */
99  		public Builder clientId(String value) {
100 			clientId = assertArgNotNullOrBlank("value", value);
101 			return this;
102 		}
103 
104 		/**
105 		 * Sets the OAuth client secret as a literal string.
106 		 *
107 		 * @param value The client secret.  Must not be <jk>null</jk> or blank.
108 		 * @return This object.
109 		 */
110 		public Builder clientSecret(String value) {
111 			assertArgNotNullOrBlank("value", value);
112 			clientSecretSupplier = () -> value;
113 			return this;
114 		}
115 
116 		/**
117 		 * Sets the OAuth client secret via a supplier.
118 		 *
119 		 * @param value The supplier.  Must not be <jk>null</jk>.
120 		 * @return This object.
121 		 */
122 		public Builder clientSecretSupplier(Supplier<String> value) {
123 			clientSecretSupplier = assertArgNotNull("value", value);
124 			return this;
125 		}
126 
127 		/**
128 		 * Adds requested OAuth scopes.
129 		 *
130 		 * @param values The scopes.
131 		 * @return This object.
132 		 */
133 		public Builder scope(String... values) {
134 			assertArgNotNull("values", values);
135 			for (var v : values) {
136 				assertArgNotNullOrBlank("scope", v);
137 				scopes.add(v);
138 			}
139 			return this;
140 		}
141 
142 		/**
143 		 * Configures the cache for the acquired token.
144 		 *
145 		 * @param value The cache.  Must not be <jk>null</jk>.
146 		 * @return This object.
147 		 */
148 		public Builder tokenCache(TokenCache value) {
149 			tokenCache = assertArgNotNull("value", value);
150 			return this;
151 		}
152 
153 		/**
154 		 * Sets the skew tolerance subtracted from the cached token's expiry.  Default 30s.
155 		 *
156 		 * @param value The skew.  Must be non-negative.
157 		 * @return This object.
158 		 */
159 		public Builder cacheSkew(Duration value) {
160 			assertArgNotNull("value", value);
161 			if (value.isNegative())
162 				throw new IllegalArgumentException("cacheSkew must be non-negative");
163 			cacheSkew = value;
164 			return this;
165 		}
166 
167 		/**
168 		 * Sets a callback that customizes the Nimbus {@link HTTPRequest}.
169 		 *
170 		 * @param value The callback.  Must not be <jk>null</jk>.
171 		 * @return This object.
172 		 */
173 		public Builder httpRequestConfigurator(Consumer<HTTPRequest> value) {
174 			httpRequestConfigurator = assertArgNotNull("value", value);
175 			return this;
176 		}
177 
178 		/**
179 		 * Builds the flow.
180 		 *
181 		 * @return A new {@link CachingClientCredentialsFlow}.
182 		 */
183 		public CachingClientCredentialsFlow build() {
184 			if (tokenEndpoint == null)
185 				throw new IllegalStateException("CachingClientCredentialsFlow requires tokenEndpoint(...)");
186 			if (clientId == null)
187 				throw new IllegalStateException("CachingClientCredentialsFlow requires clientId(...)");
188 			if (clientSecretSupplier == null)
189 				throw new IllegalStateException("CachingClientCredentialsFlow requires clientSecret(...) or clientSecretSupplier(...)");
190 			return new CachingClientCredentialsFlow(this);
191 		}
192 	}
193 
194 	private final OAuthClientCredentialsFlow delegate;
195 	private final TokenCache tokenCache;
196 	private final Duration cacheSkew;
197 	private final String cacheKey;
198 
199 	/**
200 	 * Constructor.
201 	 *
202 	 * @param b The builder.
203 	 */
204 	protected CachingClientCredentialsFlow(Builder b) {
205 		var scopes = u(cp(b.scopes));
206 		var db = OAuthClientCredentialsFlow.create()
207 			.tokenEndpoint(b.tokenEndpoint)
208 			.clientId(b.clientId)
209 			.clientSecretSupplier(b.clientSecretSupplier);
210 		if (!scopes.isEmpty())
211 			db.scope(scopes.toArray(new String[0]));
212 		if (b.httpRequestConfigurator != null)
213 			db.httpRequestConfigurator(b.httpRequestConfigurator);
214 		this.delegate = db.build();
215 		this.tokenCache = b.tokenCache;
216 		this.cacheSkew = b.cacheSkew;
217 		this.cacheKey = "cc|" + b.clientId + "|" + String.join(" ", scopes);
218 	}
219 
220 	/**
221 	 * Acquires a token, returning the cached token when a fresh one is present.
222 	 *
223 	 * @return The acquired token.
224 	 */
225 	public OAuthToken acquire() {
226 		if (tokenCache != null) {
227 			var hit = tokenCache.getToken(cacheKey, Instant.now(), cacheSkew);
228 			if (hit.isPresent())
229 				return hit.get();
230 		}
231 		var token = delegate.acquire();
232 		if (tokenCache != null)
233 			tokenCache.putToken(cacheKey, token);
234 		return token;
235 	}
236 }
