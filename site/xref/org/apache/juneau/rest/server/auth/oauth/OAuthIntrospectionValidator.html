38  /**
39   * {@link TokenValidator} that validates opaque OAuth 2.0 access tokens via the
40   * <a class="doclink" href="https://datatracker.ietf.org/doc/html/rfc7662">RFC 7662</a> Token Introspection
41   * endpoint.
42   *
43   * <p>
44   * Wraps Nimbus's {@code TokenIntrospectionRequest} / {@code TokenIntrospectionResponse} for the HTTP
45   * round-trip and the JSON parsing.  A {@link TokenCache} (default: bounded LRU cache) short-circuits
46   * the round-trip on repeated lookups within the configured TTL window.
47   *
48   * <h5 class='topic'>Usage</h5>
49   *
50   * <p class='bjava'>
51   * 	<jk>var</jk> validator = OAuthIntrospectionValidator.<jsm>create</jsm>()
52   * 		.introspectionEndpoint(URI.<jsm>create</jsm>(<js>"https://idp.example.com/oauth2/introspect"</js>))
53   * 		.clientId(<js>"api-server"</js>)
54   * 		.clientSecret(<js>"..."</js>)
55   * 		.requiredScopes(<js>"read:orders"</js>)
56   * 		.build();
57   * </p>
58   *
59   * <h5 class='topic'>Defaults</h5>
60   * <ul>
61   * 	<li>Cache TTL: 5 minutes, capped at 1 hour.
62   * 	<li>Cache size: 1000 entries.
63   * 	<li>Required scopes: none (scope enforcement lives on the validator, not the filter).
64   * </ul>
65   *
66   * <h5 class='section'>See Also:</h5>
67   * <ul>
68   * 	<li class='jc'>{@link OAuthFilter}
69   * 	<li class='jc'>{@link TokenValidator}
70   * 	<li class='link'><a class="doclink" href="https://datatracker.ietf.org/doc/html/rfc7662">RFC 7662 &mdash; OAuth 2.0 Token Introspection</a>
71   * </ul>
72   *
73   * @since 10.0.0
74   */

252 			return new OAuthIntrospectionValidator(this);

326 	@Override /* Overridden from TokenValidator */
327 	public Principal validate(String token) throws AuthenticationException {
328 		assertArgNotNullOrBlank("token", token);
329 		var cached = tokenCache.getPrincipal(token);
330 		if (cached.isPresent())
331 			return cached.get();
332 		return introspect(token);
333 	}
334 
335 	private Principal introspect(String token) throws AuthenticationException {
336 		var clientAuth = new ClientSecretBasic(new ClientID(clientId),
337 			new com.nimbusds.oauth2.sdk.auth.Secret(clientSecretSupplier.get()));
338 		AccessToken accessToken = new BearerAccessToken(token);
339 		var req = new TokenIntrospectionRequest(introspectionEndpoint, clientAuth, accessToken);
340 		var http = req.toHTTPRequest();
341 		if (httpRequestConfigurator != null)
342 			httpRequestConfigurator.accept(http);
343 
344 		com.nimbusds.oauth2.sdk.http.HTTPResponse httpResp;
345 		try {
346 			httpResp = http.send();
347 		} catch (IOException e) {
348 			throw new AuthenticationException(e, "OAuth introspection HTTP call failed")
349 				.wwwAuthenticate(bearerError("invalid_token", "introspection unreachable"));
350 		}
351 		TokenIntrospectionResponse resp;
352 		try {
353 			resp = TokenIntrospectionResponse.parse(httpResp);
354 		} catch (ParseException e) {
355 			throw new AuthenticationException(e, "OAuth introspection response could not be parsed")
356 				.wwwAuthenticate(bearerError("invalid_token", "introspection parse failed"));
357 		}
358 		if (!resp.indicatesSuccess())
359 			throw new AuthenticationException("OAuth introspection error: " + resp.toErrorResponse().getErrorObject().getCode())
360 				.wwwAuthenticate(bearerError("invalid_token", resp.toErrorResponse().getErrorObject().getCode()));
361 		var success = resp.toSuccessResponse();
362 		if (!success.isActive())
363 			throw new AuthenticationException("OAuth token inactive")
364 				.wwwAuthenticate(bearerError("invalid_token", "token inactive"));
365 		var scopes = extractScopes(success);
366 		enforceRequiredScopes(scopes);
367 		var claims = buildClaims(success, scopes);
368 		var subject = success.getSubject() != null ? success.getSubject().getValue() : "<no-sub>";
369 		var principal = new ClaimsPrincipal(subject, claims);
370 		var ttl = resolveTtl(success);
371 		tokenCache.putPrincipal(token, principal, ttl);
372 		return principal;
373 	}
374 
375 	private Set<String> extractScopes(TokenIntrospectionSuccessResponse success) {
376 		var s = success.getScope();
377 		if (s == null)
378 			return Collections.emptySet();
379 		var out = new LinkedHashSet<String>();
380 		for (var v : s.toStringList())
381 			if (v != null && !v.isBlank()) // HTT: v==null branch unreachable; Nimbus Scope.toStringList() never returns null entries
382 				out.add(v);
383 		return out;
384 	}
385 
386 	private void enforceRequiredScopes(Set<String> tokenScopes) throws AuthenticationException {
387 		for (var req : requiredScopes) {
388 			if (!tokenScopes.contains(req))
389 				throw new AuthenticationException("Required scope missing: " + req)
390 					.wwwAuthenticate(bearerError("insufficient_scope", "scope " + req + " missing"));
391 		}
392 	}
393 
394 	private Map<String,Object> buildClaims(TokenIntrospectionSuccessResponse success, Set<String> scopes) {
395 		var claims = new LinkedHashMap<String,Object>(success.toJSONObject());
396 		if (!scopes.isEmpty())
397 			claims.put("scope", new ArrayList<>(scopes));
398 		return claims;
399 	}
400 
401 	private Duration resolveTtl(TokenIntrospectionSuccessResponse success) {
402 		var ttl = cacheTtl;
403 		var exp = success.getExpirationTime();
404 		if (exp != null) {
405 			var until = Duration.between(clock.instant(), exp.toInstant());
406 			if (until.isNegative() || until.isZero())
407 				until = Duration.ofSeconds(1);
408 			if (until.compareTo(ttl) < 0)
409 				ttl = until;
410 		}
411 		return ttl;
412 	}

414 	private static String bearerError(String code, String description) {
415 		return "Bearer error=\"" + code + "\", error_description=\"" + description.replace('"', ' ') + "\"";
