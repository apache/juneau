17  package org.apache.juneau.rest.server.auth;
18  
19  import static java.lang.annotation.ElementType.*;
20  import static java.lang.annotation.RetentionPolicy.*;
21  
22  import java.lang.annotation.*;
23  import java.security.*;
24  
25  /**
26   * Marks a {@code @RestOp}-method parameter as the authenticated {@link Principal} for the current request.
27   *
28   * <p>
29   * When a request reaches an op handler annotated with {@code @Auth}, the framework's
30   * {@link AuthArg} resolver pulls the {@link Principal} previously stashed by an upstream AuthN guard
31   * (such as {@link BearerTokenGuard} or {@link ApiKeyGuard}) on the request attributes under the
32   * {@link org.apache.juneau.rest.server.RestServerConstants#PRINCIPAL_ATTR PRINCIPAL_ATTR} key.
33   *
34   * <p>
35   * If an annotated parameter is reached without a principal being stashed (i.e. no guard ran, or the
36   * request was made on an unguarded path) the resolver returns <jk>null</jk> &mdash; the guard chain is
37   * the contract that guarantees a non-null value; this annotation only requests the lookup.
38   *
39   * <h5 class='section'>Type-driven equivalent:</h5>
40   *
41   * <p>
42   * For convenience, the framework also resolves a bare {@link Principal} (or {@link ClaimsPrincipal})
43   * parameter <i>without</i> any annotation &mdash; the type alone is enough to trigger the same lookup.
44   * Use {@code @Auth} when you want the intent to be self-documenting on the parameter list or you want
45   * to disambiguate against another arg resolver that might claim the parameter first.
46   *
47   * <h5 class='section'>Example:</h5>
48   *
49   * <p class='bjava'>
50   * 	<ja>@RestGet</ja>(path=<js>"/me"</js>)
51   * 	<jk>public</jk> Profile me(<ja>@Auth</ja> Principal <jv>p</jv>) {
52   * 		<jk>return</jk> <jf>profileService</jf>.lookup(<jv>p</jv>.getName());
53   * 	}
54   *
55   * 	<ja>@RestGet</ja>(path=<js>"/claims"</js>)
56   * 	<jk>public</jk> String scope(<ja>@Auth</ja> ClaimsPrincipal <jv>p</jv>) {
57   * 		<jk>return</jk> <jv>p</jv>.getClaim(<js>"scope"</js>, String.<jk>class</jk>).orElse(<js>""</js>);
58   * 	}
59   * </p>
60   *
61   * <h5 class='section'>See Also:</h5><ul>
62   * 	<li class='jc'>{@link AuthArg}
63   * 	<li class='jc'>{@link BearerTokenGuard}
64   * 	<li class='jc'>{@link ApiKeyGuard}
65   * 	<li class='jc'>{@link ClaimsPrincipal}
66   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestServerAuthGuards">AuthN Guards</a>
67   * </ul>
68   *
69   * @since 10.0.0
70   */
71  @Target(PARAMETER)
72  @Retention(RUNTIME)
73  @Documented
74  public @interface Auth {
75  }
