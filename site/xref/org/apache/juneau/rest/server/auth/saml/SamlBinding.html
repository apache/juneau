17  package org.apache.juneau.rest.server.auth.saml;
18  
19  /**
20   * SAML 2.0 bindings supported by {@link SamlAuthFilter} for delivering a {@code <Response>} from an IdP to the SP.
21   *
22   * <p>
23   * v1 ships the two HTTP-flavored bindings.  Artifact and SOAP back-channel bindings are deferred.
24   *
25   * <h5 class='topic'>HTTP POST binding</h5>
26   * <p>
27   * The IdP returns an auto-submitting HTML form whose {@code SAMLResponse} field carries the base64-encoded
28   * {@code <Response>} document.  The browser POSTs the form to the SP's assertion-consumer URL.  This is the
29   * dominant binding for SAML 2.0 web SSO.
30   *
31   * <h5 class='topic'>HTTP Redirect binding</h5>
32   * <p>
33   * The IdP redirects the browser to the SP with the {@code SAMLResponse} carried in a query-string parameter.
34   * The value is deflated (RFC 1951) and then base64-encoded so that even mid-size assertions fit inside URL
35   * length limits.  This binding is more commonly used for {@code <AuthnRequest>} than for {@code <Response>},
36   * but some IdPs offer it for assertions too.
37   *
38   * <h5 class='section'>See Also:</h5>
39   * <ul>
40   * 	<li class='jc'>{@link SamlAuthFilter}
41   * 	<li class='link'><a class="doclink" href="https://docs.oasis-open.org/security/saml/v2.0/saml-bindings-2.0-os.pdf">SAML 2.0 Bindings (OASIS)</a>
42   * </ul>
43   *
44   * @since 10.0.0
45   */
46  public enum SamlBinding {
47  
48  	/**
49  	 * HTTP POST binding.  {@code SAMLResponse} carried as a base64-encoded form field on a POST request.
50  	 */
51  	POST,
52  
53  	/**
54  	 * HTTP Redirect binding.  {@code SAMLResponse} carried as a deflated + base64-encoded query parameter on a
55  	 * GET request.
56  	 */
57  	REDIRECT
58  }
