55  /**
56   * Validates a SAML 2.0 {@code <samlp:Response>} document and produces a {@link ClaimsPrincipal} marked
57   * with {@code issuerType=SAML}.
58   *
59   * <p>
60   * Wraps OpenSAML 5.x's parser, unmarshaller, signature validator, and (optionally) assertion decrypter.
61   * The validator enforces:
62   *
63   * <ul>
64   * 	<li><b>Mandatory assertion signature</b> &mdash; every assertion must carry its own enveloped signature.
65   * 		A response that signs only the {@code <Response>} envelope (and not the assertion) is rejected as
66   * 		unsigned; response-envelope-level signatures are not supported.
67   * 	<li><b>Strict signature-algorithm allowlist</b> &mdash; only {@code rsa-sha256} and
68   * 		{@code ecdsa-sha256} are accepted out of the box.  SHA-1 is permanently rejected.  The reference
69   * 		{@code DigestMethod} is separately allowlisted (SHA-256/384/512), so a SHA-1 digest is rejected even
70   * 		under a SHA-256 signature.
71   * 	<li><b>Response status</b> &mdash; the {@code <samlp:Status>} must report the {@code Success} status code.
72   * 	<li><b>Clock skew</b> &mdash; defaults to 60 seconds tolerance on {@code NotBefore} /
73   * 		{@code NotOnOrAfter}; capped at 300 seconds.
74   * 	<li><b>Audience restriction</b> &mdash; {@code <AudienceRestriction>} must list the configured SP entity ID.
75   * 	<li><b>Bearer subject confirmation</b> &mdash; when a {@link Builder#recipient(String) recipient} (the SP
76   * 		ACS URL) is configured, the assertion must carry a {@code bearer} {@code <SubjectConfirmation>} whose
77   * 		{@code <SubjectConfirmationData>} matches the expected {@code Recipient}, is within its
78   * 		{@code NotBefore}/{@code NotOnOrAfter} window, and satisfies the configured
79   * 		{@code InResponseTo}/{@code Address} expectations.
80   * 	<li><b>One-time use</b> &mdash; each assertion ID is recorded in a {@link ReplayCache} and a second
81   * 		presentation of the same ID is rejected.  The check is fail-closed: if the cache cannot answer, the
82   * 		assertion is rejected.
83   * 	<li><b>Encrypted assertions</b> &mdash; if the response contains
84   * 		{@code <EncryptedAssertion>}, a {@code decryptionCredential} must be configured.
85   * </ul>
86   *
87   * <h5 class='topic'>Marker claim</h5>
88   * <p>
89   * The returned {@link ClaimsPrincipal} is annotated with {@code issuerType=SAML} so downstream code
90   * can distinguish SAML-derived principals from JWT-derived principals without resorting to a subclass:
91   *
92   * <p class='bjava'>
93   * 	<jk>if</jk> (<js>"SAML"</js>.equals(<jv>principal</jv>.getClaim(<js>"issuerType"</js>, String.<jk>class</jk>).orElse(<jk>null</jk>))) {
94   * 		<jc>// SAML-issued principal</jc>
95   * 	}
96   * </p>
97   *
98   * <h5 class='topic'>Builder usage</h5>
99   *
100  * <p class='bjava'>
101  * 	<jk>var</jk> validator = SamlAssertionValidator.<jsm>create</jsm>()
102  * 		.metadataResolver(SamlMetadataResolvers.<jsm>url</jsm>(<js>"https://idp.example.com/metadata"</js>))
103  * 		.spEntityId(<js>"https://sp.example.com"</js>)
104  * 		.expectedIssuer(<js>"https://idp.example.com"</js>)
105  * 		.build();
106  * </p>
107  *
108  * <h5 class='section'>See Also:</h5>
109  * <ul>
110  * 	<li class='jc'>{@link SamlAuthFilter}
111  * 	<li class='jc'>{@link SamlMetadataResolvers}
112  * 	<li class='link'><a class="doclink" href="https://docs.oasis-open.org/security/saml/v2.0/saml-core-2.0-os.pdf">SAML 2.0 Core (OASIS)</a>
113  * </ul>
114  *
115  * @since 10.0.0
116  */

490 	/**
491 	 * Validates the supplied SAML 2.0 {@code <samlp:Response>} XML and returns the resolved principal.
492 	 *
493 	 * @param xml The full XML payload (already base64-decoded; already URL-decoded and inflated for the
494 	 * 	REDIRECT binding).  Must be a {@code <samlp:Response>} envelope.
495 	 * @return A {@link ClaimsPrincipal} carrying the IdP-supplied claims plus the {@code issuerType=SAML}
496 	 * 	marker.
497 	 * @throws AuthenticationException If the response cannot be parsed or validation fails.
498 	 */
499 	public Principal validate(String xml) throws AuthenticationException {
500 		assertArgNotNullOrBlank("xml", xml);
501 		var response = parseResponse(xml);
502 		validateStatus(response);
503 		var assertion = extractAndDecryptAssertion(response);
504 		verifySignature(assertion);
505 		var claims = buildClaims(response, assertion);
506 		validateSubjectConfirmation(assertion);
507 		recordSingleUse(assertion);
508 		var subject = nameId(assertion);
509 		return new ClaimsPrincipal(subject, claims);
510 	}
511 
512 	private void validateStatus(Response response) throws AuthenticationException {
513 		var status = response.getStatus();
514 		var code = status == null ? null : status.getStatusCode();
515 		var value = code == null ? null : code.getValue();
516 		if (! StatusCode.SUCCESS.equals(value))
517 			throw new AuthenticationException("SAML response status is not Success: " + value)
518 				.wwwAuthenticate("SAML error=\"invalid_response\"");
519 	}
520 
521 	private Response parseResponse(String xml) throws AuthenticationException {
522 		try {
523 			var dbf = DocumentBuilderFactory.newInstance();
524 			dbf.setNamespaceAware(true);
525 			dbf.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);
526 			dbf.setFeature("http://xml.org/sax/features/external-general-entities", false);
527 			dbf.setFeature("http://xml.org/sax/features/external-parameter-entities", false);
528 			dbf.setXIncludeAware(false);
529 			dbf.setExpandEntityReferences(false);
530 			var doc = dbf.newDocumentBuilder().parse(new ByteArrayInputStream(xml.getBytes(java.nio.charset.StandardCharsets.UTF_8)));
531 			Element root = doc.getDocumentElement();
532 			var registry = org.opensaml.core.xml.config.XMLObjectProviderRegistrySupport.getUnmarshallerFactory();
533 			var unmarshaller = registry.getUnmarshaller(root);
534 			if (unmarshaller == null)
535 				throw new AuthenticationException("Unknown SAML element: " + root.getNodeName())
536 					.wwwAuthenticate("SAML error=\"invalid_response\"");
537 			var obj = unmarshaller.unmarshall(root);
538 			if (!(obj instanceof Response obj2))
539 				throw new AuthenticationException("SAML envelope is not a <Response>: " + cns(obj))
540 					.wwwAuthenticate("SAML error=\"invalid_response\"");
541 			return obj2;
542 		} catch (UnmarshallingException | ParserConfigurationException | IOException | org.xml.sax.SAXException e) {
543 			throw new AuthenticationException(e, "SAML response could not be parsed")
544 				.wwwAuthenticate("SAML error=\"invalid_response\"");
545 		}
546 	}

548 	private Assertion extractAndDecryptAssertion(Response response) throws AuthenticationException {
549 		var assertions = response.getAssertions();
550 		var encrypted = response.getEncryptedAssertions();
551 		if (assertions.isEmpty() && encrypted.isEmpty())
552 			throw new AuthenticationException("SAML response contains no <Assertion>")
553 				.wwwAuthenticate("SAML error=\"invalid_response\"");
554 		if (!encrypted.isEmpty()) {
555 			if (decryptionCredential == null)
556 				throw new AuthenticationException("SAML response contains <EncryptedAssertion> but no decryption credential is configured")
557 					.wwwAuthenticate("SAML error=\"decryption_required\"");
558 			var resolver = (KeyInfoCredentialResolver) new StaticKeyInfoCredentialResolver(decryptionCredential);
559 			EncryptedKeyResolver keyResolver = new ChainingEncryptedKeyResolver(List.of(
560 				new InlineEncryptedKeyResolver(),
561 				new SimpleRetrievalMethodEncryptedKeyResolver()
562 			));
563 			var decrypter = new Decrypter(null, resolver, keyResolver);
564 			decrypter.setRootInNewDocument(true);
565 			try {
566 				return decrypter.decrypt(encrypted.get(0));
567 			} catch (org.opensaml.xmlsec.encryption.support.DecryptionException e) {
568 				throw new AuthenticationException(e, "SAML <EncryptedAssertion> could not be decrypted")
569 					.wwwAuthenticate("SAML error=\"decryption_failed\"");
570 			}
571 		}
572 		return assertions.get(0);
573 	}

575 	private void verifySignature(Assertion assertion) throws AuthenticationException {
576 		var signature = assertion.getSignature();
577 		if (signature == null)
578 			throw new AuthenticationException("SAML assertion is unsigned")
579 				.wwwAuthenticate("SAML error=\"signature_required\"");
580 
581 		var alg = signature.getSignatureAlgorithm();
582 		if (alg == null || !algorithms.contains(alg)) // HTT: alg==null unreachable; OpenSAML always populates SignatureAlgorithm after signing
583 			throw new AuthenticationException("SAML signature algorithm not allowlisted: " + alg)
584 				.wwwAuthenticate("SAML error=\"signature_algorithm_rejected\"");
585 
586 		validateDigestAlgorithms(signature);
587 
588 		// First, run the SAML signature profile validator to catch wrapping/structural attacks.
589 		try {
590 			new SAMLSignatureProfileValidator().validate(signature);
591 		} catch (SignatureException e) {
592 			throw new AuthenticationException(e, "SAML signature profile validation failed")
593 				.wwwAuthenticate("SAML error=\"signature_invalid\"");
594 		}
595 
596 		Credential credential = resolveSigningCredential();
597 		try {
598 			SignatureValidator.validate(signature, credential);
599 		} catch (SignatureException e) {
600 			throw new AuthenticationException(e, "SAML signature verification failed")
601 				.wwwAuthenticate("SAML error=\"signature_invalid\"");
602 		}
603 	}

605 	private Credential resolveSigningCredential() throws AuthenticationException {
606 		if (signingCredential != null)
607 			return signingCredential;
608 		var idp = resolveIdpDescriptor();
609 		var credential = extractSigningCredential(idp);
610 		if (credential == null)
611 			throw new AuthenticationException("No signing X.509 certificate in metadata for issuer: " + expectedIssuer)
612 				.wwwAuthenticate("SAML error=\"metadata_invalid\"");
613 		return credential;
614 	}

616 	private IDPSSODescriptor resolveIdpDescriptor() throws AuthenticationException {
617 		try {
618 			var criteria = new CriteriaSet(
619 				new EntityIdCriterion(expectedIssuer),
620 				new EntityRoleCriterion(IDPSSODescriptor.DEFAULT_ELEMENT_NAME)
621 			);
622 			var entityDescriptor = metadataResolver.resolveSingle(criteria);
623 			if (entityDescriptor == null)
624 				throw new AuthenticationException("No SAML metadata for issuer: " + expectedIssuer)
625 					.wwwAuthenticate("SAML error=\"metadata_missing\"");
626 			var idp = entityDescriptor.getIDPSSODescriptor(SAMLConstants.SAML20P_NS);
627 			if (idp == null)
628 				throw new AuthenticationException("SAML metadata for issuer has no IDPSSODescriptor")
629 					.wwwAuthenticate("SAML error=\"metadata_invalid\"");
630 			return idp;
631 		} catch (ResolverException e) {
632 			throw new AuthenticationException(e, "SAML metadata resolution failed")
633 				.wwwAuthenticate("SAML error=\"metadata_invalid\"");
634 		}
635 	}

637 	private Credential extractSigningCredential(IDPSSODescriptor idp) {
638 		for (KeyDescriptor kd : idp.getKeyDescriptors()) {
639 			var credential = credentialFromKeyDescriptor(kd);
640 			if (credential != null) // HTT: true branch (successful credential extraction) requires real X.509 metadata; covered by integration tests
641 				return credential;
642 		}
643 		return null;
644 	}

646 	private Credential credentialFromKeyDescriptor(KeyDescriptor kd) {
647 		if (!isSigningKey(kd))
648 			return null;
649 		var keyInfo = kd.getKeyInfo();
650 		if (keyInfo == null)
651 			return null;
652 		for (var x509 : keyInfo.getX509Datas()) {
653 			var credential = credentialFromX509Data(x509);
654 			if (credential != null)
655 				return credential;
656 		}
657 		return null;
658 	}
659 
660 	private Credential credentialFromX509Data(X509Data x509) {
661 		for (var cert : x509.getX509Certificates()) {
662 			var base64 = cert.getValue();
663 			if (base64 == null)
664 				continue;
665 			var credential = parseX509Credential(base64);
666 			if (credential != null) // HTT: true branch requires valid, in-window X.509 DER bytes; covered by integration tests
667 				return credential;
668 		}
669 		return null;
670 	}

672 	private static boolean isSigningKey(KeyDescriptor kd) {
673 		return kd.getUse() == null || kd.getUse() == UsageType.SIGNING || kd.getUse() == UsageType.UNSPECIFIED;
674 	}

676 	private BasicX509Credential parseX509Credential(String base64) {
677 		try {
678 			var bytes = Base64.getMimeDecoder().decode(base64);
679 			var cf = java.security.cert.CertificateFactory.getInstance("X.509");
680 			var certificate = (java.security.cert.X509Certificate) cf.generateCertificate(new ByteArrayInputStream(bytes));
681 			// The metadata signing cert is the trust anchor; reject one outside its validity window so an
682 			// expired (or not-yet-valid) cert can no longer be used to verify assertions.
683 			certificate.checkValidity(Date.from(clock.instant()));
684 			return new BasicX509Credential(certificate);
685 		} catch (java.security.cert.CertificateException | IllegalArgumentException e) {
686 			return null;  // Expired/not-yet-valid or malformed cert: caller tries the next certificate.
687 		}
688 	}

690 	private Map<String,Object> buildClaims(Response response, Assertion assertion) throws AuthenticationException {
691 		validateConditions(assertion);
692 		validateIssuer(response, assertion);
693 
694 		var claims = new LinkedHashMap<String,Object>();
695 		claims.put(ISSUER_TYPE_CLAIM, ISSUER_TYPE_SAML);
696 		claims.put("iss", expectedIssuer);
697 		claims.put("aud", spEntityId);
698 
699 		for (var stmt : assertion.getAttributeStatements()) {
700 			for (var attr : stmt.getAttributes()) {
701 				var name = attr.getName();
702 				if (name == null)
703 					continue;
704 				var values = l();
705 				for (var av : attr.getAttributeValues())
706 					values.add(extractStringValue(av));
707 				if (values.size() == 1)
708 					claims.put(name, values.get(0));
709 				else if (!values.isEmpty())
710 					claims.put(name, values);
711 			}
712 		}
713 		return claims;
714 	}

716 	private void validateConditions(Assertion assertion) throws AuthenticationException {
717 		var conditions = assertion.getConditions();
718 		if (conditions == null)
719 			throw rejectAssertion("missing <Conditions>");
720 
721 		var now = clock.instant();
722 		var nbf = conditions.getNotBefore();
723 		var noa = conditions.getNotOnOrAfter();
724 		if (nbf != null && nbf.minus(clockSkew).isAfter(now))
725 			throw rejectAssertion("assertion not yet valid");
726 		if (noa != null && !noa.plus(clockSkew).isAfter(now))
727 			throw rejectAssertion("assertion has expired");
728 
729 		boolean audienceOk = false;
730 		for (var ar : conditions.getAudienceRestrictions()) {
731 			for (var aud : ar.getAudiences()) {
732 				if (spEntityId.equals(aud.getURI())) {
733 					audienceOk = true;
734 					break;
735 				}
736 			}
737 		}
738 		if (!audienceOk)
739 			throw rejectAssertion("audience does not include SP entity ID: " + spEntityId);
740 	}

742 	private void validateIssuer(Response response, Assertion assertion) throws AuthenticationException {
743 		var assertionIssuer = assertion.getIssuer();
744 		if (assertionIssuer == null || !expectedIssuer.equals(assertionIssuer.getValue()))
745 			throw rejectAssertion("assertion <Issuer> does not match expected issuer");
746 		var respIssuer = response.getIssuer();
747 		if (respIssuer != null && !expectedIssuer.equals(respIssuer.getValue()))
748 			throw rejectAssertion("response <Issuer> does not match expected issuer");
749 	}

751 	private void validateDigestAlgorithms(org.opensaml.xmlsec.signature.Signature signature) throws AuthenticationException {
752 		var dom = signature.getDOM();
753 		if (dom == null) // HTT: null DOM unreachable; a verifiable signature is always marshalled to DOM
754 			return;
755 		var digestNodes = dom.getElementsByTagNameNS(XMLSIG_NS, "DigestMethod");
756 		for (var i = 0; i < digestNodes.getLength(); i++) {
757 			var alg = ((Element) digestNodes.item(i)).getAttribute("Algorithm");
758 			if (alg == null || alg.isEmpty() || alg.toLowerCase(Locale.ROOT).contains("sha1") || !digestAlgorithms.contains(alg))
759 				throw new AuthenticationException("SAML signature digest algorithm not allowlisted: " + alg)
760 					.wwwAuthenticate("SAML error=\"signature_algorithm_rejected\"");
761 		}
762 	}

764 	private void validateSubjectConfirmation(Assertion assertion) throws AuthenticationException {
765 		if (recipient == null)  // Bearer subject-confirmation validation is enabled by configuring recipient(...).
766 			return;
767 		var subject = assertion.getSubject();
768 		if (subject == null)
769 			throw rejectAssertion("missing <Subject> for bearer confirmation");
770 		String failure = null;
771 		var sawBearer = false;
772 		for (var sc : subject.getSubjectConfirmations()) {
773 			if (! SubjectConfirmation.METHOD_BEARER.equals(sc.getMethod()))
774 				continue;
775 			sawBearer = true;
776 			var reason = confirmationFailureReason(sc);
777 			if (reason == null)
778 				return;  // A valid bearer confirmation was found.
779 			failure = reason;
780 		}
781 		if (! sawBearer)
782 			throw rejectAssertion("missing bearer <SubjectConfirmation>");
783 		throw rejectAssertion(failure);
784 	}
785 
786 	private String confirmationFailureReason(SubjectConfirmation sc) {
787 		var data = sc.getSubjectConfirmationData();
788 		if (data == null)
789 			return "bearer <SubjectConfirmationData> is missing";
790 		if (! recipient.equals(data.getRecipient()))
791 			return "bearer confirmation <Recipient> does not match the expected ACS URL";
792 		var now = clock.instant();
793 		var noa = data.getNotOnOrAfter();
794 		if (noa == null)
795 			return "bearer confirmation is missing NotOnOrAfter";
796 		if (! noa.plus(clockSkew).isAfter(now))
797 			return "bearer confirmation has expired";
798 		var nbf = data.getNotBefore();
799 		if (nbf != null && nbf.minus(clockSkew).isAfter(now))
800 			return "bearer confirmation is not yet valid";
801 		var inResponseTo = data.getInResponseTo();
802 		if (expectedInResponseTo != null) {
803 			if (! expectedInResponseTo.equals(inResponseTo))
804 				return "bearer confirmation <InResponseTo> does not match the expected request ID";
805 		} else if (inResponseTo != null && ! inResponseTo.isEmpty()) {
806 			return "unsolicited bearer confirmation must not carry <InResponseTo>";
807 		}
808 		if (expectedAddress != null && ! expectedAddress.equals(data.getAddress()))
809 			return "bearer confirmation <Address> does not match the expected client address";
810 		return null;
811 	}

813 	private void recordSingleUse(Assertion assertion) throws AuthenticationException {
814 		var id = assertion.getID();
815 		if (id == null || id.isEmpty())
816 			throw rejectAssertion("assertion has no ID for single-use enforcement");
817 		var expiresAtMs = singleUseExpiry(assertion);
818 		var firstSeen = replayCache.checkAndRecord(id, expiresAtMs, ReplayCache.FailMode.FAIL_CLOSED,
819 			e -> LOG.log(Level.WARNING, e, () -> "SAML assertion single-use check failed; rejecting (fail-closed)."));
820 		if (! firstSeen)
821 			throw new AuthenticationException("SAML assertion has already been presented")
822 				.wwwAuthenticate("SAML error=\"assertion_replayed\"");
823 	}

825 	private long singleUseExpiry(Assertion assertion) {
826 		var conditions = assertion.getConditions();
827 		if (conditions != null && conditions.getNotOnOrAfter() != null)
828 			return conditions.getNotOnOrAfter().plus(clockSkew).toEpochMilli();
829 		return clock.instant().plus(DEFAULT_REPLAY_RETENTION).toEpochMilli();
830 	}

833 		if (av == null) // HTT: null av unreachable; OpenSAML attribute value lists never contain null entries
834 			return null;
835 		var dom = av.getDOM();
836 		if (dom != null) // HTT: DOM is non-null only after marshalling; XSString values always have DOM after Response.marshall()
837 			return dom.getTextContent();
838 		return av.toString();
