21  import java.time.*;
22  import java.util.*;

24  import org.apache.juneau.rest.server.auth.*;
25  import org.apache.juneau.rest.auth.oauth.flow.*;

27  /**
28   * Immutable record describing a logged-in OpenID Connect Relying Party session.
29   *
30   * <p>
31   * Created by {@link OidcRelyingParty#completeLogin} after a successful login and persisted via a
32   * {@link SessionStore}.  The {@link #principal()} carries the verified ID-token claims and is what the
33   * {@link OidcSessionAuthFilter} surfaces to {@code RoleBasedRestGuard} / {@code @Auth Principal}.
34   *
35   * <p>
36   * Two fields support server-side (back-channel) revocation: {@link #subject()} (the {@code sub} claim)
37   * and {@link #sid()} (the optional {@code sid} session-id claim).  A server-side-indexed
38   * {@link SessionStore} keys on these so an IdP-pushed logout token can invalidate the matching
39   * session(s).
40   *
41   * @param id The opaque server-side session id.  For {@link InMemorySessionStore} this is the cache key;
42   * 	for {@link SignedCookieSessionStore} it is a stable identifier embedded in the signed cookie.  Never
43   * 	{@code null}.
44   * @param subject The ID-token {@code sub} claim.  Never {@code null} or blank.
45   * @param sid The ID-token {@code sid} (session id) claim, if the IdP issued one.  Used for back-channel
46   * 	logout targeting a single session rather than all of a subject's sessions.
47   * @param principal The verified ID-token claims as a {@link ClaimsPrincipal}.  Never {@code null}.
48   * @param roles The roles extracted from the configured roles claim.  Never {@code null}; may be empty.
49   * @param token The tokens returned from the token endpoint (access / refresh / id), if the store retains
50   * 	them.  {@link SignedCookieSessionStore} does NOT retain tokens (stateless cookies don't carry them),
51   * 	so this is {@link Optional#empty()} for cookie-backed sessions.
52   * @param createdAt The instant the session was created.  Never {@code null}.
53   * @param expiresAt The instant the session expires (bounded independently of token lifetime).  Never
54   * 	{@code null}.
55   * @since 10.0.0
56   */
57  public record OidcSession(
58  		String id,
59  		String subject,
60  		Optional<String> sid,
61  		ClaimsPrincipal principal,
62  		Set<String> roles,
63  		Optional<OAuthToken> token,
64  		Instant createdAt,
65  		Instant expiresAt) {
66  
67  	/**
68  	 * Compact constructor enforcing non-null fields and defensively copying the role set.
69  	 */
70  	public OidcSession {
71  		Objects.requireNonNull(id, "id");
72  		Objects.requireNonNull(subject, "subject");
73  		Objects.requireNonNull(sid, "sid");
74  		Objects.requireNonNull(principal, "principal");
75  		Objects.requireNonNull(token, "token");
76  		Objects.requireNonNull(createdAt, "createdAt");
77  		Objects.requireNonNull(expiresAt, "expiresAt");
78  		roles = roles == null
79  			? Collections.emptySet()
80  			: u(cp(roles));
81  	}
82  
83  	/**
84  	 * Returns whether this session is expired relative to the supplied instant.
85  	 *
86  	 * @param now The reference instant.
87  	 * @return <jk>true</jk> if {@code now} is at or after {@link #expiresAt()}.
88  	 */
89  	public boolean isExpired(Instant now) {
90  		Objects.requireNonNull(now, "now");
91  		return !now.isBefore(expiresAt);
92  	}
93  }
