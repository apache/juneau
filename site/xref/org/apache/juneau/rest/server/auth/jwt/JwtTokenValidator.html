39  /**
40   * {@link TokenValidator} that validates signed JWTs against a JWKS-backed key source.
41   *
42   * <p>
43   * Designed to be paired with {@link BearerTokenGuard} in
44   * {@code juneau-rest-server}. It enforces a deliberately strict default policy so a misconfigured
45   * call site can't accidentally accept dangerous tokens:
46   *
47   * <ul>
48   * 	<li><b>Algorithm allowlisting</b> — defaults to {@code [RS256, ES256]}. {@code HS256} is rejected
49   * 		unless the caller explicitly opts in via
50   * 		{@link Builder#algorithms(JWSAlgorithm...) algorithms(...)}. {@code "none"} (unsigned JWTs) is
51   * 		<i>permanently</i> rejected — there is no opt-in. This defeats the well-known
52   * 		<i>algorithm confusion</i> attack where a JWT is signed with {@code HS256} using the RS256
53   * 		public key as the HMAC secret.
54   * 	<li><b>Mandatory claims</b> — {@code iss}, {@code aud}, {@code exp}, {@code nbf} are required by
55   * 		default. A token missing any of them is rejected.
56   * 	<li><b>Clock skew</b> — defaults to 60 seconds tolerance for {@code exp} and {@code nbf}.
57   * 		Capped at 300 seconds by the builder.
58   * 	<li><b>JWKS caching</b> — keys are fetched and cached for 5 minutes by default. On fetch failure
59   * 		the cache continues serving past TTL with a warning logged (see {@link JwksCache}). When a
60   * 		fresh-cache selection returns no keys (typically because an IdP rotated its signing key
61   * 		mid-TTL), the cache performs one eager out-of-band refresh before failing the request. This
62   * 		behavior is <b>enabled by default</b> ({@link Builder#jwksEagerRefreshOnKidMiss(boolean)
63   * 		jwksEagerRefreshOnKidMiss(true)}) and bounded by a per-cache cooldown (default 10 seconds)
64   * 		plus a single-in-flight guard. Opt out via
65   * 		{@link Builder#jwksEagerRefreshOnKidMiss(boolean) jwksEagerRefreshOnKidMiss(false)}.
66   * 		Applies only to {@link Builder#jwksUrl(URI) jwksUrl(...)}-backed caches; a
67   * 		caller-supplied {@link Builder#jwkSource(JWKSource) jwkSource(...)} is unaffected.
68   * </ul>
69   *
70   * <h5 class='topic'>Usage</h5>
71   *
72   * <p class='bjava'>
73   * 	<jk>var</jk> validator = JwtTokenValidator.<jsm>create</jsm>()
74   * 		.jwksUrl(URI.<jsm>create</jsm>(<js>"https://issuer.example.com/.well-known/jwks.json"</js>))
75   * 		.issuer(<js>"https://issuer.example.com/"</js>)
76   * 		.audience(<js>"https://api.example.com"</js>)
77   * 		.build();
78   *
79   * 	<jk>var</jk> guard = BearerTokenGuard.<jsm>create</jsm>()
80   * 		.realm(<js>"api"</js>)
81   * 		.validator(validator)
82   * 		.build();
83   * </p>
84   *
85   * <h5 class='topic'>Security notes</h5>
86   *
87   * <ul>
88   * 	<li>Never disable {@code iss} / {@code aud} validation. The builder enforces them.
89   * 	<li>Allowlist {@code HS256} only when both ends of the connection are services you control AND
90   * 		you have a secure secret distribution mechanism. JWKS-published JWTs should be asymmetric.
91   * 	<li>Rotate JWKS keys on a schedule. {@link JwksCache} picks up new keys after the configured TTL.
92   * </ul>
93   *
94   * <h5 class='section'>See Also:</h5>
95   * <ul>
96   * 	<li class='jc'>{@link TokenValidator}
97   * 	<li class='jc'>{@link BearerTokenGuard}
98   * 	<li class='jc'>{@link ClaimsPrincipal}
99   * 	<li class='link'><a class="doclink" href="https://datatracker.ietf.org/doc/html/rfc7519">RFC 7519 — JSON Web Token</a>
100  * 	<li class='link'><a class="doclink" href="https://datatracker.ietf.org/doc/html/rfc7517">RFC 7517 — JSON Web Key</a>
101  * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestServerAuthGuards">AuthN Guards</a>
102  * </ul>
103  *
104  * @since 10.0.0
105  */

236 		/**
237 		 * Sets the signing-algorithm allowlist.
238 		 *
239 		 * <p>
240 		 * Replaces (not adds to) the default allowlist. Callers wanting {@code HS256} support must
241 		 * include it explicitly. {@link JWSAlgorithm#NONE} is rejected at build time and at runtime
242 		 * regardless of what is passed here.
243 		 *
244 		 * @param values The allowed algorithms. Must contain at least one element; must not contain
245 		 * 	{@link JWSAlgorithm#NONE}.
246 		 * @return This object.
247 		 */
248 		public Builder algorithms(JWSAlgorithm...values) {
249 			assertArgNotNull("values", values);
250 			if (values.length == 0)
251 				throw iaex("algorithms allowlist must be non-empty");
252 			Set<JWSAlgorithm> next = st();
253 			for (var a : values) {
254 				assertArgNotNull("algorithm", a);
255 				if (Algorithm.NONE.equals(a))
256 					throw iaex("\"none\" algorithm is permanently rejected (RFC 7518 §3.6 unsafe)");
257 				next.add(a);
258 			}
259 			algorithms = next;
260 			return this;
261 		}

263 		/**
264 		 * Sets the clock-skew tolerance for {@code exp} / {@code nbf} validation.
265 		 *
266 		 * @param value The tolerance. Must be non-negative and not exceed 5 minutes.
267 		 * @return This object.
268 		 */
269 		public Builder clockSkew(Duration value) {
270 			assertArgNotNull("value", value);
271 			if (value.isNegative())
272 				throw iaex("clockSkew must be non-negative");
273 			if (value.compareTo(MAX_CLOCK_SKEW) > 0)
274 				throw iaex("clockSkew must not exceed 5 minutes (was %s)", value);
275 			clockSkew = value;

289 			jwksCacheTtl = value;

293 		/**
294 		 * Enables or disables the eager JWKS refresh that fires on a fresh-cache {@code kid} miss.
295 		 *
296 		 * <p>
297 		 * When {@code true} (the default), the first request that presents a token whose signing
298 		 * key is not in the current (still-fresh) cache triggers a single out-of-band JWKS refresh.
299 		 * This closes the rotation window where an IdP publishes a new key mid-TTL and clients would
300 		 * otherwise see up to {@code jwksCacheTtl} minutes of spurious 401 responses for perfectly
301 		 * valid tokens. The refresh is bounded by {@link #jwksEagerRefreshCooldown(Duration)} and a
302 		 * single-in-flight guard (no thundering herd).
303 		 *
304 		 * <p>
305 		 * Applies only to {@link #jwksUrl(URI) jwksUrl(...)}-backed caches. A
306 		 * caller-supplied {@link #jwkSource(JWKSource) jwkSource(...)} is unaffected.
307 		 *
308 		 * @param value {@code true} to enable (default); {@code false} to restore pre-10.0.0 behavior.
309 		 * @return This object.
310 		 */
311 		public Builder jwksEagerRefreshOnKidMiss(boolean value) {
312 			jwksEagerRefreshOnKidMiss = value;
313 			return this;
314 		}
315 
316 		/**
317 		 * Sets the minimum spacing between consecutive eager JWKS refreshes.
318 		 *
319 		 * <p>
320 		 * A burst of unknown-{@code kid} tokens (e.g. misconfigured client, replay attack) or a
321 		 * JWKS-endpoint outage will trigger at most one eager refresh per cooldown window. This
322 		 * prevents the eager-refresh path from becoming a vector for amplified JWKS-endpoint load.
323 		 *
324 		 * <p>
325 		 * Must be positive, at most 60 seconds, and at most {@link #jwksCacheTtl(Duration)
326 		 * jwksCacheTtl} (validated at {@link #build()} time).
327 		 *
328 		 * @param value The cooldown. Must not be {@code null}.
329 		 * @return This object.
330 		 */
331 		public Builder jwksEagerRefreshCooldown(Duration value) {
332 			assertArgNotNull("value", value);
333 			if (value.isZero() || value.isNegative())
334 				throw iaex("jwksEagerRefreshCooldown must be positive");
335 			if (value.compareTo(Duration.ofSeconds(60)) > 0)
336 				throw iaex("jwksEagerRefreshCooldown must not exceed 60 seconds (was %s)", value);
337 			jwksEagerRefreshCooldown = value;
338 			return this;
339 		}

341 		/**
342 		 * Overrides the {@link Clock} used for {@code exp} / {@code nbf} comparisons and JWKS cache

353 		/**
354 		 * Builds the validator.
355 		 *
356 		 * @return A new {@link JwtTokenValidator}.
357 		 */
358 		public JwtTokenValidator build() {
359 			if (issuer == null)
360 				throw isex("JwtTokenValidator requires an issuer");
361 			if (audience == null)
362 				throw isex("JwtTokenValidator requires an audience");
363 			if (jwksUrl == null && jwkSource == null)
364 				throw isex("JwtTokenValidator requires either jwksUrl(...) or jwkSource(...)");
365 			if (jwksUrl != null && jwkSource != null)
366 				throw isex("JwtTokenValidator: jwksUrl(...) and jwkSource(...) are mutually exclusive");
367 			if (jwksEagerRefreshCooldown.compareTo(jwksCacheTtl) > 0)
368 				throw isex("jwksEagerRefreshCooldown (%s) must not exceed jwksCacheTtl (%s)", jwksEagerRefreshCooldown, jwksCacheTtl);
369 			return new JwtTokenValidator(this);
370 		}

433 	@Override /* Overridden from TokenValidator */
434 	public Principal validate(String token) throws AuthenticationException {
435 		assertArgNotNull("token", token);
436 		JWT jwt;
437 		try {
438 			jwt = JWTParser.parse(token);
439 		} catch (ParseException e) {
440 			throw new AuthenticationException(e, "JWT could not be parsed")
441 				.wwwAuthenticate("Bearer error=\"invalid_token\", error_description=\"malformed JWT\"");
442 		}
443 		if (! (jwt instanceof SignedJWT jwt2))
444 			throw new AuthenticationException("JWT must be signed (alg=none and unsigned JWTs are rejected)")
445 				.wwwAuthenticate("Bearer error=\"invalid_token\", error_description=\"unsigned JWT\"");
446 		var alg = jwt2.getHeader().getAlgorithm();
447 		if (alg == null || Algorithm.NONE.equals(alg))
448 			throw new AuthenticationException("JWT alg \"none\" is not permitted")
449 				.wwwAuthenticate("Bearer error=\"invalid_token\", error_description=\"alg=none rejected\"");
450 		if (! algorithms.contains(alg))
451 			throw new AuthenticationException("JWT algorithm not allowlisted: " + alg)
452 				.wwwAuthenticate("Bearer error=\"invalid_token\", error_description=\"algorithm not allowed\"");
453 
454 		// Verify signature.
455 		var processor = new DefaultJWTProcessor<SecurityContext>();
456 		processor.setJWSKeySelector(new JWSVerificationKeySelector<>(algorithms, jwkSource));
457 		// Disable nimbus' internal claims verifier — we run our own below so we control the Clock.
458 		processor.setJWTClaimsSetVerifier((claims, ctx) -> { /* no-op */ });
459 
460 		JWTClaimsSet claims;
461 		try {
462 			claims = processor.process(jwt2, null);
463 		} catch (BadJOSEException e) {
464 			throw new AuthenticationException(e, "JWT validation failed: " + e.getMessage())
465 				.wwwAuthenticate("Bearer error=\"invalid_token\", error_description=\"" + sanitize(e.getMessage()) + "\"");
466 		} catch (JOSEException e) {
467 			throw new AuthenticationException(e, "JWT signature verification failed")
468 				.wwwAuthenticate("Bearer error=\"invalid_token\", error_description=\"signature verification failed\"");
469 		}
470 
471 		verifyClaims(claims);
472 
473 		var subject = claims.getSubject();
474 		if (isBlank(subject))
475 			subject = "<no-sub>";
476 		return new ClaimsPrincipal(subject, claims.toJSONObject());
477 	}
478 
479 	private void verifyClaims(JWTClaimsSet claims) throws AuthenticationException {
480 		// Required claims: iss, aud, exp, nbf are non-negotiable defaults.
481 		var iss = claims.getIssuer();
482 		if (isBlank(iss))
483 			throw reject("\"iss\" claim is required");
484 		if (! issuer.equals(iss))
485 			throw reject("\"iss\" claim does not match expected issuer");
486 
487 		var aud = claims.getAudience();
488 		if (aud == null || aud.isEmpty())
489 			throw reject("\"aud\" claim is required");
490 		if (! aud.contains(audience))
491 			throw reject("\"aud\" claim does not include expected audience");
492 
493 		var exp = claims.getExpirationTime();
494 		if (exp == null)
495 			throw reject("\"exp\" claim is required");
496 		var nbf = claims.getNotBeforeTime();
497 		if (nbf == null)
498 			throw reject("\"nbf\" claim is required");
499 
500 		var now = clock.instant();
501 		if (exp.toInstant().plus(clockSkew).isBefore(now))
502 			throw reject("token has expired");
503 		if (nbf.toInstant().minus(clockSkew).isAfter(now))
504 			throw reject("token not yet valid");
505 	}
506 
507 	private static AuthenticationException reject(String reason) {
508 		return new AuthenticationException("JWT validation failed: " + reason)
509 			.wwwAuthenticate("Bearer error=\"invalid_token\", error_description=\"" + sanitize(reason) + "\"");
510 	}

512 	private static JWKSource<SecurityContext> createRemoteJwkSource(URI jwksUrl) {
513 		try {
514 			return JWKSourceBuilder.create(jwksUrl.toURL()).build();
515 		} catch (MalformedURLException e) {
516 			throw iaex(e, "Invalid JWKS URL: %s", jwksUrl);
517 		}
518 	}

520 	private static String sanitize(String s) {
521 		if (s == null)
522 			return "";
523 		// Don't echo arbitrary characters into a WWW-Authenticate header value.
524 		return s.replaceAll("[\"\\r\\n]", " ");
525 	}
