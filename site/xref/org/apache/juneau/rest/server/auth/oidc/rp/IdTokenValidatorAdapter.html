39  /**
40   * Wraps Nimbus's {@link IDTokenValidator} to validate an OpenID Connect ID token's signature and claims
41   * and project the verified claims onto a Juneau {@link ClaimsPrincipal}.
42   *
43   * <p>
44   * Nimbus's {@code IDTokenValidator} performs the full OIDC ID-token validation in one call: signature
45   * verification against the IdP JWKS, exact {@code iss} match, {@code aud} contains the client id,
46   * {@code azp} when there are multiple audiences, {@code exp} (with clock skew), required {@code iat},
47   * and &mdash; when supplied &mdash; {@code nonce} match.  This adapter owns only the configuration and
48   * the projection to {@link ClaimsPrincipal}; no bespoke crypto.
49   *
50   * <p>
51   * The signing-algorithm allowlist defaults to {@code [RS256, ES256]} (both SHA-256 based).  The unsafe
52   * {@code "none"} algorithm is permanently rejected (a token signed with {@code none} has no key and so
53   * fails the key selector), and SHA-1-family algorithms are excluded by virtue of not being on the
54   * allowlist &mdash; consistent with the {@code juneau-rest-server-auth-jwt} strict-default stance.
55   *
56   * <p>
57   * For production, configure {@link Builder#jwksUri(URI)} (Nimbus builds a cached remote key source).
58   * For tests / HSM-backed signers / multi-issuer setups, inject a {@link Builder#jwkSet(JWKSet)} or
59   * {@link Builder#jwkSource(JWKSource)} directly.
60   *
61   * @since 10.0.0
62   */
63  @SuppressWarnings({
64  	"java:S1192" // Duplicate string literals are OIDC/JWT claim names (e.g. "iss", "aud", "exp"); intentional
65  })
66  public class IdTokenValidatorAdapter {
67  
68  	/**
69  	 * Static creator.
70  	 *
71  	 * @return A new builder.
72  	 */
73  	public static Builder create() {
74  		return new Builder();
75  	}

77  	/**
78  	 * Builder.
79  	 */
80  	public static class Builder {
81  		private String issuer;
82  		private String clientId;
83  		private URI jwksUri;
84  		private JWKSet jwkSet;
85  		private JWKSource<SecurityContext> jwkSource;
86  		private Set<JWSAlgorithm> algorithms = new LinkedHashSet<>(Arrays.asList(JWSAlgorithm.RS256, JWSAlgorithm.ES256));
87  		private int maxClockSkewSeconds = 60;
88  
89  		/** Constructor. */
90  		protected Builder() {}
91  
92  		/**
93  		 * Sets the expected {@code iss} (issuer).  Required.
94  		 *
95  		 * @param value The issuer.  Must not be <jk>null</jk> or blank.
96  		 * @return This object.
97  		 */
98  		public Builder issuer(String value) {
99  			issuer = assertArgNotNullOrBlank("value", value);
100 			return this;
101 		}
102 
103 		/**
104 		 * Sets the OAuth client id (expected {@code aud}).  Required.
105 		 *
106 		 * @param value The client id.  Must not be <jk>null</jk> or blank.
107 		 * @return This object.
108 		 */
109 		public Builder clientId(String value) {
110 			clientId = assertArgNotNullOrBlank("value", value);
111 			return this;
112 		}
113 
114 		/**
115 		 * Sets the JWKS endpoint URI the validator fetches signing keys from.  Mutually exclusive with
116 		 * {@link #jwkSet(JWKSet)} / {@link #jwkSource(JWKSource)}.
117 		 *
118 		 * <p>
119 		 * The URI must use <js>"https"</js> or target a loopback host; the signing-key material fetched from
120 		 * this endpoint is the trust anchor for every ID token, so a plaintext transport to a non-loopback host
121 		 * is rejected.
122 		 *
123 		 * @param value The JWKS URI.  Must not be <jk>null</jk> and must use HTTPS or target a loopback host.
124 		 * @return This object.
125 		 */
126 		public Builder jwksUri(URI value) {
127 			jwksUri = UriUtils.assertSecureOrLoopback(assertArgNotNull("value", value));
128 			return this;
129 		}
130 
131 		/**
132 		 * Sets a fixed JWK set (useful for tests / static keys).  Mutually exclusive with
133 		 * {@link #jwksUri(URI)} / {@link #jwkSource(JWKSource)}.
134 		 *
135 		 * @param value The JWK set.  Must not be <jk>null</jk>.
136 		 * @return This object.
137 		 */
138 		public Builder jwkSet(JWKSet value) {
139 			jwkSet = assertArgNotNull("value", value);
140 			return this;
141 		}
142 
143 		/**
144 		 * Sets a custom {@link JWKSource}.  Mutually exclusive with {@link #jwksUri(URI)} /
145 		 * {@link #jwkSet(JWKSet)}.
146 		 *
147 		 * @param value The JWK source.  Must not be <jk>null</jk>.
148 		 * @return This object.
149 		 */
150 		public Builder jwkSource(JWKSource<SecurityContext> value) {
151 			jwkSource = assertArgNotNull("value", value);
152 			return this;
153 		}
154 
155 		/**
156 		 * Sets the signing-algorithm allowlist.  Replaces the default {@code [RS256, ES256]}.
157 		 *
158 		 * @param values The allowed algorithms.  Must be non-empty and must not contain
159 		 * 	{@link Algorithm#NONE}.
160 		 * @return This object.
161 		 */
162 		public Builder algorithms(JWSAlgorithm...values) {
163 			assertArgNotNull("values", values);
164 			assertArg(values.length > 0, "algorithms allowlist must be non-empty");
165 			var next = new LinkedHashSet<JWSAlgorithm>();
166 			for (var a : values) {
167 				assertArgNotNull("algorithm", a);
168 				assertArg(! Algorithm.NONE.equals(a), "\"none\" algorithm is permanently rejected");
169 				next.add(a);
170 			}
171 			algorithms = next;
172 			return this;
173 		}
174 
175 		/**
176 		 * Sets the clock-skew tolerance (in seconds) for {@code exp} / {@code iat} validation.
177 		 *
178 		 * @param value The tolerance in seconds.  Must be non-negative.
179 		 * @return This object.
180 		 */
181 		public Builder maxClockSkewSeconds(int value) {
182 			assertArg(value >= 0, "maxClockSkewSeconds must be non-negative (was %s)", value);
183 			maxClockSkewSeconds = value;
184 			return this;
185 		}
186 
187 		/**
188 		 * Builds the adapter.
189 		 *
190 		 * @return A new {@link IdTokenValidatorAdapter}.
191 		 */
192 		public IdTokenValidatorAdapter build() {
193 			if (issuer == null)
194 				throw new IllegalStateException("IdTokenValidatorAdapter requires issuer(...)");
195 			if (clientId == null)
196 				throw new IllegalStateException("IdTokenValidatorAdapter requires clientId(...)");
197 			var sources = 0;
198 			if (jwksUri != null) sources++;
199 			if (jwkSet != null) sources++;
200 			if (jwkSource != null) sources++;
201 			if (sources != 1)
202 				throw new IllegalStateException("IdTokenValidatorAdapter requires exactly one of jwksUri(...), jwkSet(...), or jwkSource(...)");
203 			return new IdTokenValidatorAdapter(this);
204 		}
205 	}
206 
207 	private final IDTokenValidator validator;
208 
209 	/**
210 	 * Constructor.
211 	 *
212 	 * @param b The builder.
213 	 */
214 	protected IdTokenValidatorAdapter(Builder b) {
215 		JWKSource<SecurityContext> source = resolveSource(b);
216 		var keySelector = new JWSVerificationKeySelector<SecurityContext>(b.algorithms, source);
217 		this.validator = new IDTokenValidator(new Issuer(b.issuer), new ClientID(b.clientId), keySelector, null);
218 		this.validator.setMaxClockSkew(b.maxClockSkewSeconds);
219 	}
220 
221 	private static JWKSource<SecurityContext> resolveSource(Builder b) {
222 		if (b.jwkSource != null)
223 			return b.jwkSource;
224 		if (b.jwkSet != null)
225 			return new ImmutableJWKSet<>(b.jwkSet);
226 		try {
227 			return JWKSourceBuilder.create(b.jwksUri.toURL()).build();
228 		} catch (MalformedURLException e) {
229 			throw iaex(e, "Invalid JWKS URI: %s", b.jwksUri);
230 		}
231 	}
232 
233 	/**
234 	 * Validates an ID token and projects the verified claims onto a {@link ClaimsPrincipal}.
235 	 *
236 	 * @param idToken The raw (compact) ID token JWT.  Must not be <jk>null</jk> or blank.
237 	 * @param expectedNonce The {@code nonce} the relying party stored before the redirect; the ID
238 	 * 	token's {@code nonce} claim must match.  May be <jk>null</jk> only if the flow did not use a
239 	 * 	nonce (the RP always supplies one).
240 	 * @return The verified claims as a {@link ClaimsPrincipal} whose name is the {@code sub} claim.
241 	 * @throws AuthenticationException If parsing, signature verification, or any claim check fails.
242 	 */
243 	public ClaimsPrincipal validate(String idToken, String expectedNonce) throws AuthenticationException {
244 		assertArgNotNullOrBlank("idToken", idToken);
245 		JWT jwt;
246 		try {
247 			jwt = JWTParser.parse(idToken);
248 		} catch (ParseException e) {
249 			throw new AuthenticationException(e, "ID token could not be parsed");
250 		}
251 		IDTokenClaimsSet claims;
252 		try {
253 			claims = validator.validate(jwt, expectedNonce == null ? null : new Nonce(expectedNonce));
254 		} catch (BadJOSEException e) {
255 			throw new AuthenticationException(e, "ID token validation failed: %s", e.getMessage());
256 		} catch (JOSEException e) {
257 			throw new AuthenticationException(e, "ID token signature verification failed");
258 		}
259 		var sub = claims.getSubject().getValue();
260 		return new ClaimsPrincipal(sub, claims.toJSONObject());
261 	}
