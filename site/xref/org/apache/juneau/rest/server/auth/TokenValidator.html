17  package org.apache.juneau.rest.server.auth;
18  
19  import java.security.*;
20  
21  /**
22   * SPI for verifying an opaque bearer token and producing the associated {@link Principal}.
23   *
24   * <p>
25   * {@link BearerTokenGuard} extracts the token from the {@code Authorization: Bearer <token>} header
26   * and delegates verification to a {@code TokenValidator} the host supplies. The validator is the
27   * authoritative source of "is this token still good?" &mdash; signature, expiry, audience, issuer,
28   * revocation, etc.
29   *
30   * <p>
31   * Implementations:
32   * <ul>
33   * 	<li><b>JWT</b> &mdash; the optional {@code juneau-rest-server-auth-jwt} sub-module ships
34   * 		{@code JwtTokenValidator}, which performs JWKS-backed signature verification with full
35   * 		{@code iss} / {@code aud} / {@code exp} / {@code nbf} / algorithm-allowlist enforcement.
36   * 	<li><b>Opaque token</b> &mdash; e.g. a 32-byte random token stored in a database; the validator
37   * 		looks the token up in the store and returns the associated user record wrapped as a
38   * 		{@link Principal}.
39   * 	<li><b>Introspection</b> &mdash; e.g. RFC 7662 token introspection over HTTP against the issuing
40   * 		IdP; the validator caches successful responses for a short TTL.
41   * </ul>
42   *
43   * <h5 class='topic'>Contract</h5>
44   *
45   * <p>
46   * On success, return a non-null {@link Principal}. On failure, throw {@link AuthenticationException}
47   * with a {@code WWW-Authenticate} challenge string set via
48   * {@link AuthenticationException#wwwAuthenticate(String)}. Returning <jk>null</jk> is treated as failure
49   * by the guard but produces a generic challenge &mdash; throwing is preferred so the failure reason
50   * surfaces in the response body.
51   *
52   * <p>
53   * The supplied {@code token} string is exactly what followed {@code "Bearer "} in the {@code Authorization}
54   * header &mdash; the guard does not pre-trim or unquote. Validators should treat it as untrusted input
55   * (no logging of full tokens, no exception messages echoing the token, etc.).
56   *
57   * <h5 class='section'>See Also:</h5><ul>
58   * 	<li class='jc'>{@link BearerTokenGuard}
59   * 	<li class='jc'>{@link AuthenticationException}
60   * 	<li class='link'><a class="doclink" href="https://datatracker.ietf.org/doc/html/rfc6750">RFC 6750 &mdash; Bearer Token Usage</a>
61   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestServerAuthGuards">AuthN Guards</a>
62   * </ul>
63   *
64   * @since 10.0.0
65   */
66  @FunctionalInterface
67  public interface TokenValidator {
68  
69  	/**
70  	 * Validates the supplied bearer token and returns the associated {@link Principal}.
71  	 *
72  	 * @param token The raw token string from {@code Authorization: Bearer <token>}. Never <jk>null</jk>.
73  	 * @return The authenticated principal. Must not be <jk>null</jk> on success.
74  	 * @throws AuthenticationException If the token is invalid, expired, revoked, or otherwise unacceptable.
75  	 */
76  	Principal validate(String token) throws AuthenticationException;
77  }
