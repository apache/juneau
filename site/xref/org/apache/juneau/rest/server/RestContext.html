105 /**
106  * Defines the initial configuration of a <c>RestServlet</c> or <c>@Rest</c> annotated object.
107  *
108  * <p>
109  * An extension of the {@link ServletConfig} object used during servlet initialization.
110  *
111  * <p>
112  * Configuration is supplied declaratively through the {@link Rest @Rest} annotation on the resource class
113  * (and inherited from any parent classes), and programmatically through {@link Bean @Bean}-annotated
114  * methods/fields that contribute named beans (e.g. <c>encoders</c>, <c>parsers</c>, <c>callLogger</c>) to the REST
115  * resource's bean store. Where direct construction is needed (test rigs, mock clients, embedded usage),
116  * the public constructor takes a {@link RestContext.Args} record carrying the bootstrap state.
117  *
118  * <h5 class='section'>Example:</h5>
119  * <p class='bjava'>
120  * 	<ja>@Rest</ja>(path=<js>"/myresource"</js>, serializers=JsonSerializer.<jk>class</jk>, parsers=JsonParser.<jk>class</jk>)
121  * 	<jk>public class</jk> MyResource {
122  *
123  * 		<jc>// Programmatically contribute a bean to the resource's bean store.</jc>
124  * 		<ja>@Bean</ja>
125  * 		<jk>public</jk> CallLogger callLogger() {
126  * 			<jk>return new</jk> MyCustomCallLogger();
127  * 		}
128  * 	}
129  * </p>
130  *
131  * <p>
132  * The historical <c>public MyResource(RestContext.Builder builder)</c> constructor-injection pattern and the
133  * <c>{@link RestInit @RestInit} public void init(RestContext.Builder builder)</c> method-injection pattern were both
134  * removed in 10.0. {@code RestContext.Builder} itself is on the deletion path; new code should not depend on it.
135  * See <a class="doclink" href="https://juneau.apache.org/docs/topics/V10.0MigrationGuide">v10.0 Migration Guide</a>
136  * for replacement recipes.
137  *
138  * <h5 class='section'>Notes:</h5><ul>
139  * 	<li class='note'>This class is thread safe and reusable.
140  * </ul>
141  *
142  * <h5 class='section'>See Also:</h5><ul>
143  * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestContext">RestContext</a>
144  * </ul>
145  */

178 	 * Bootstrap arguments for {@link RestContext}.
179 	 *
180 	 * <p>
181 	 * Bundles the small, fixed set of inputs needed to construct a {@link RestContext} into a single immutable record.
182 	 *
183 	 * <p>
184 	 * All non-required values default sensibly:
185 	 * <ul>
186 	 * 	<li>{@code parentContext}, {@code servletConfig} — {@code null} (top-level resource, no servlet container)
187 	 * 	<li>{@code path} — {@code ""} (no path prefix)
188 	 * 	<li>{@code beanStoreConfigurer} — no-op (no pre-build bean-store mutation)
189 	 * 	<li>{@code overridingParent} — {@code null} (no test-time override layer)
190 	 * </ul>
191 	 *
192 	 * <p>
193 	 * The {@code beanStoreConfigurer} hook gives test fixtures and integration code a chance to register beans on
194 	 * the {@link WritableBeanStore} after the resource has been wired in but before the
195 	 * {@code findXxx()} memoizers fire.
196 	 *
197 	 * <p>
198 	 * The {@code overridingParent} hook installs a {@link BeanStore} into the {@code overridingParent} slot of the
199 	 * freshly-constructed {@link BasicBeanStore} so that test-time overrides resolve at tier 1 of the lookup chain
200 	 * (above the resource's local {@code @Bean} factory entries and the Spring bridge).  This is the
201 	 * canonical Mode INJECT wiring point for {@code juneau-junit5}'s {@code TestBeanStore}.
202 	 *
203 	 * <h5 class='section'>Example:</h5>
204 	 * <p class='bjava'>
205 	 * 	<jk>var</jk> <jv>ctx</jv> = <jk>new</jk> RestContext(<jk>new</jk> RestContext.Args(MyResource.<jk>class</jk>, () -&gt; <jk>new</jk> MyResource()));
206 	 * </p>
207 	 *
208 	 * @param resourceClass The {@link Rest @Rest}-annotated REST resource class. Must not be {@code null}.
209 	 * @param parentContext The parent {@link RestContext}, or {@code null} if this is a top-level resource.
210 	 * @param servletConfig The {@link ServletConfig} from the servlet container, or {@code null} when none is available.
211 	 * @param resource The supplier that provides the resource instance during initialization. Must not be {@code null}.
212 	 * @param path The path prefix relative to the parent. Defaults to {@code ""}.
213 	 * @param beanStoreConfigurer A pre-build hook that runs against the resolved {@link WritableBeanStore}
214 	 * 	after the resource has been wired in but before any {@code findXxx()} memoizer fires. Defaults to a no-op.
215 	 * @param overridingParent An optional test-time override layer installed into the {@code overridingParent}
216 	 * 	slot of the freshly-constructed {@link BasicBeanStore}.  May be {@code null}.  Honored only when the
217 	 * 	resource does <i>not</i> declare an {@code @Bean WritableBeanStore} factory method; user-supplied bean
218 	 * 	stores are passed through unchanged.
219 	 * @param paths An optional programmatic override for the top-level mount-paths array (the highest-priority
220 	 * 	rung in the runtime-override chain documented on {@link RestContext#getPaths()}).  {@code null} (the
221 	 * 	default) means &quot;no programmatic override&quot; &mdash; resolution falls through to the resource's
222 	 * 	{@code getPaths()} getter, then {@link Rest#paths()} (SVL-resolved per element and comma-split).
223 	 * 	An empty array explicitly clears the mount list (no top-level mounts).
224 	 * @param kind The build-time {@link ContextKind kind} of this context — {@link ContextKind#ROOT ROOT} for a
225 	 * 	top-level/host/mock resource, a {@link ContextKind.Child CHILD} carrying the {@link ResolvedChild} for a
226 	 * 	{@link Rest#children() @Rest(children)}/{@link Rest#childrenDefs() childrenDefs} sub-resource (isolated
227 	 * 	resolution, with an optional host-seeded payload), or a {@link ContextKind.Mixin MIXIN} carrying the
228 	 * 	{@link ResolvedMixin} that produced a per-mixin sub-context (parent-linked to the host so that
229 	 * 	{@link RestContext#getRestAnnotationsForProperty(String) annotation-property walks} prepend the host's
230 	 * 	{@code @Rest} chain before the mixin's own).
231 	 *
232 	 * @since 10.0.0
233 	 */

297 	 * The discovery-time normalized declaration of a single mixin.
298 	 *
299 	 * <p>
300 	 * Every mixin entry &mdash; whether a bare {@link Rest#mixins() Class&lt;?&gt;} or a rich
301 	 * {@link Rest#mixinDefs() &#64;Mixin} &mdash; is normalized to one of these immutable carriers at discovery

414 	/**
415 	 * The build-time <i>kind</i> of a {@link RestContext} — which of the three construction flavors produced it.
416 	 *
417 	 * <p>
418 	 * Replaces the former boolean {@code mixinContext} flag on {@link Args} with a typed discriminator so the
419 	 * three flavors are explicit and self-documenting at every call site, and so the {@code MIXIN} flavor can
420 	 * carry the {@link ResolvedMixin} that produced it (the single home for the host-declared override payload,
421 	 * consumed at build time when the mixin sub-context resolves its settings).
422 	 *
423 	 * <p>
424 	 * <b>Mixin vs. child asymmetry (important — do not assume symmetry):</b> a {@link Mixin} sub-context is
425 	 * parent-linked to the host and <i>inherits</i> the host's {@code @Rest} annotation chain
426 	 * ({@link #getRestAnnotationsForProperty(String)} walks host&rarr;mixin), so {@code @Mixin} overrides
427 	 * <i>layer on top of</i> an inherited chain.  A {@link Child} is <i>isolated</i> ({@link #HOST_ONLY_PROPERTIES}
428 	 * includes {@link RestServerConstants#PROPERTY_children}; children retain pre-10.0.0 isolated resolution with
429 	 * no parent walk).  A {@code @Child} seed therefore does not <i>override</i> an inherited chain the way a
430 	 * {@code @Mixin} override does — instead, {@link #computeRawRestAnnotations()} injects the seed's synthetic
431 	 * {@code @Rest} at the <b>least-derived</b> slot of the child's own (still fully isolated) annotation chain,
432 	 * between the child's real {@code @Rest} declarations and the synthesized {@link
433 	 * org.apache.juneau.rest.server.config.DefaultConfig DefaultConfig} fallback.  This ranks the seed below
434 	 * everything the child itself declares (so the child's own explicit value wins for child-wins scalars, and
435 	 * the child's own {@code noInherit} cutoff naturally cuts the seed too) but above the framework defaults.
436 	 *
437 	 * @since 10.0.0
438 	 */
439 	public sealed interface ContextKind permits ContextKind.Root, ContextKind.Child, ContextKind.Mixin {
440 
441 		/** A top-level (host) resource, mock context, or servlet-mounted resource. */
442 		record Root() implements ContextKind {}
443 
444 		/**
445 		 * A routed child resource (mounted via {@link Rest#children()}/{@link Rest#childrenDefs()}); isolated
446 		 * resolution, with an optional host-seeded {@code @Child} payload injected at the least-derived slot of
447 		 * its own annotation chain (see the class javadoc above).
448 		 *
449 		 * @param def The normalized declaration that produced this sub-context — carries the host-declared seed payload.
450 		 */
451 		record Child(ResolvedChild def) implements ContextKind {}
452 
453 		/**
454 		 * A per-mixin sub-context (composed via {@link Rest#mixins()}/{@link Rest#mixinDefs()}); inherits the host chain.
455 		 *
456 		 * @param def The normalized declaration that produced this sub-context — carries the host-declared override payload.
457 		 */
458 		record Mixin(ResolvedMixin def) implements ContextKind {}
459 
460 		/** Shared {@link Root} singleton (the common case). */
461 		ContextKind ROOT = new Root();
462 	}

721 	 * Public mount-time path resolver for use by hosting runtimes that mount {@code @Rest}-annotated resources
722 	 * before the {@link RestContext} has been constructed (e.g. {@code JettyServerComponent}'s auto-discovery
723 	 * loop, or Spring Boot's {@code ServletRegistrationBean} wiring).
724 	 *
725 	 * <p>
726 	 * Operates on the same chain as {@link RestContext#getPaths()} but skips the programmatic Builder rung
727 	 * (which is only meaningful inside the {@link RestContext} constructor):
728 	 * <ol>
729 	 * 	<li>The resource's {@code getPaths()} virtual getter (defined on {@code RestServlet} /
730 	 * 		{@code RestResource}). {@code null} return falls through; non-{@code null} (including empty array) wins.
731 	 * 	<li>{@link Rest#paths() @Rest(paths)} annotation default on the most-derived class, SVL-resolved per
732 	 * 		element and comma-split.  A {@code VarResolver} is looked up on {@code store} (when present); SVL
733 	 * 		variables like {@code $C{key}} resolve against beans visible to the bean store.
734 	 * </ol>
735 	 *
736 	 * @param resourceClass The {@code @Rest}-annotated resource class. Must not be {@code null}.
737 	 * @param resource The resource instance. May be {@code null} (in which case the getter rung is skipped).
738 	 * @param store The bean store used to look up the {@link VarResolver} for SVL substitution on annotation
739 	 * 	elements, and to back the resolver session for bean-driven SVL vars like {@code $C{key}}. May be
740 	 * 	{@code null} (SVL is then skipped; literals pass through unchanged).
741 	 * @return The resolved paths array (never {@code null}; possibly empty).
742 	 * @since 10.0.0
743 	 */
744 	public static String[] resolveTopLevelPaths(Class<?> resourceClass, Object resource, BeanStore store) {
745 		var ap = AnnotationProvider.INSTANCE;
746 		var ci = ClassInfo.of(resourceClass);
747 		var annotations = rstream(ap.find(Rest.class, ci)).toList();
748 		var vr = (store == null) ? null : store.getBean(VarResolver.class).orElse(null);
749 		// Cast for the bean-store-backed resolver session only — when store is a plain BeanStore (not
750 		// WritableBeanStore) we skip the session form and call vr.resolve(...) directly.
751 		var ws = (store instanceof WritableBeanStore store2) ? store2 : null;
752 		return resolvePathsCore(annotations, resource, vr, ws);
753 	}

755 	private static String[] resolvePathsCore(List<AnnotationInfo<Rest>> annotations, Object resource, VarResolver vr, WritableBeanStore beanStore) {
756 		// Rung 2: getPaths() virtual getter on the resource (RestServlet / RestResource / any matching class).
757 		// The getter return is normalized to a flat List<String> of raw leaves (any String / String[] /
758 		// Collection / Iterable / Stream / nested mix of those), then each leaf flows through the same
759 		// SVL-resolve + comma-split pipeline used for @Rest(paths=...) annotation elements.
760 		var raw = invokeGetPaths(resource);
761 		if (raw != null) {
762 			var leaves = normalizePaths(raw);
763 			return expandPathsElements(leaves.toArray(new String[0]), vr, beanStore);
764 		}
765 
766 		// Rung 3: @Rest(paths) annotation default — most-derived non-empty array wins, with each element
767 		// SVL-resolved then comma-split (trim each piece, drop empties).  A single template element can
768 		// therefore expand to zero, one, or many mount paths.  Annotation arrays from javac are never
769 		// null, so the only filter needed is the empty-array case.
770 		return annotations.stream()
771 			.map(ai -> ai.inner().paths())
772 			.filter(arr -> arr.length > 0)
773 			.findFirst()
774 			.map(arr -> expandPathsElements(arr, vr, beanStore))
775 			.orElseGet(() -> new String[0]);
776 	}

778 	/**
779 	 * Applies the per-element SVL + comma-split pipeline to a {@code @Rest(paths=...)} array.
780 	 *
781 	 * <p>
782 	 * Each input element is SVL-resolved (via a session of the bootstrap {@link VarResolver} that's backed
783 	 * by {@code beanStore} when one is available, so bean-driven vars like {@code $C{key}} can find the
784 	 * registered {@link Config}), then split on {@code ,}; each piece is trimmed and empty pieces are
785 	 * dropped.  This lets a single element like {@code "$C{health.paths}"} or
786 	 * {@code "$E{HEALTH_PATHS,/healthz,/readyz}"} expand to multiple mount paths.
787 	 */
788 	private static String[] expandPathsElements(String[] elements, VarResolver vr, WritableBeanStore beanStore) {
789 		// Open a single session for the whole array — reusing it across elements avoids re-walking the
790 		// Var catalog and lets ConfigVar / FileVar / etc. all share one bean-lookup context.
791 		var session = (vr == null) ? null : vr.createSession(beanStore);
792 		var out = new ArrayList<String>(elements.length);
793 		for (var e : elements) {
794 			var resolved = applySvl(e, session);
795 			splitPathsValue(resolved, out);
796 		}
797 		return out.toArray(new String[0]);
798 	}

800 	private static String applySvl(String value, VarResolverSession session) {
801 		// Callers only pass annotation-array elements, which javac guarantees non-null.  An empty string
802 		// has no SVL markers to resolve, so we short-circuit it (avoids a pointless session.resolve call).
803 		if (session == null || value.isEmpty())
804 			return value;
805 		try {
806 			return session.resolve(value);
807 		} catch (@SuppressWarnings("unused") Exception e) {
808 			return value; // SVL failure: prefer the literal over throwing during construction.
809 		}
810 	}

812 	private static void splitPathsValue(String value, List<String> out) {
813 		// Callers pass the return of applySvl, which never returns null (we control both branches above).
814 		if (value.isBlank())
815 			return;
816 		var parts = value.split(",");
817 		for (var p : parts) {
818 			var t = p.trim();
819 			if (! t.isEmpty())
820 				out.add(t);
821 		}
822 	}

825 	 * Flattens any {@link Object} returned from {@code getPaths()} (String, {@code String[]},
826 	 * {@link Collection}, {@link Iterable}, {@link Stream}, primitive array, or nested mixes of these)
827 	 * into a flat {@link List} of raw leaf strings.
828 	 *
829 	 * <p>
830 	 * Non-{@link String} leaves are coerced via {@link String#valueOf(Object)}.  The returned list
831 	 * preserves traversal order; {@code null} leaves are dropped.  Comma-split + SVL resolution are
832 	 * intentionally <i>not</i> applied here &mdash; they happen downstream in
833 	 * {@link #expandPathsElements(String[], VarResolver, WritableBeanStore)} so SVL substitution sees the
834 	 * raw element (preserving {@code $E{NAME,defaultWithCommas}}-style defaults) before the post-SVL
835 	 * comma-split fires.
836 	 */

838 		if (o == null)  // HTT: the sole call site (resolvePathsCore) only invokes this after a `raw != null` guard.
839 			return List.of();
840 		var out = new ArrayList<String>();
841 		CollectionUtils.<Object>accumulate(o).forEach(x -> {
842 			if (x != null)
843 				out.add(String.valueOf(x));
844 		});
845 		return out;
846 	}

1128 	/**
1129 	 * Walks the {@code @Rest(config=...)} annotation chain (child-first) and builds a
1130 	 * {@link ConfigPropertySource} for each distinct, non-empty config attribute.
1131 	 *
1132 	 * <p>
1133 	 * The first (most-derived) slot reuses {@code rawConfig.get()} so that any {@link Bean @Bean}
1134 	 * {@code Config} factory-method override on the resource bean is preserved &mdash; the rest of
1135 	 * the framework already sees that exact instance via {@link #getConfig()} and {@code @RestInit}
1136 	 * parameter resolution. Parent slots load via {@code Config.create().name(...)} since {@code @Bean}
1137 	 * overrides apply to the most-derived class only.
1138 	 *
1139 	 * <p>
1140 	 * Edge case: a resource with no {@code @Rest(config=...)} annotation but with a {@link Bean @Bean}
1141 	 * {@code Config} factory method has a non-null {@code rawConfig.get()} and no annotation-driven
1142 	 * slot. That single {@code rawConfig} is still registered so user-supplied {@code Config}s flow
1143 	 * through to {@code @Value} resolution.
1144 	 */
1145 	@SuppressWarnings({
1146 		"java:S3776", // Cognitive complexity acceptable for multi-source REST config property collection
1147 		"java:S135"   // Per-annotation continue guards are clearer than restructuring the config-source filter chain.
1148 	})
1149 	private List<PropertySource> collectRestConfigPropertySources() {
1150 		var bs = beanStore();
1151 		var vr = bs.getBean(VarResolver.class, PROP_bootstrapVarResolver).orElseGet(this::getBootstrapVarResolver);
1152 		var result = new ArrayList<PropertySource>();
1153 		var seen = new LinkedHashSet<String>();
1154 		// AnnotationProvider returns child-first; iterate in the same order so child wins on collision.
1155 		var anns = AnnotationProvider.INSTANCE.find(Rest.class, info(resourceClass));
1156 		for (var i = 0; i < anns.size(); i++) {
1157 			var raw = anns.get(i).inner().config();
1158 			if (isEmpty(raw))
1159 				continue;
1160 			var resolvedName = vr.resolve(raw);
1161 			if (isEmpty(resolvedName))
1162 				continue;
1163 			if (! seen.add(resolvedName))
1164 				continue;
1165 			Config cfg;
1166 			if (result.isEmpty()) {
1167 				// Most-derived child slot. rawConfig.get() captures @Bean Config override; reuse it.
1168 				cfg = rawConfig.get();
1169 			} else if ("SYSTEM_DEFAULT".equals(resolvedName)) {
1170 				cfg = Config.getSystemDefault();
1171 			} else {
1172 				cfg = Config.create().varResolver(vr).name(resolvedName).build();
1173 			}
1174 			if (cfg != null)
1175 				result.add(new ConfigPropertySource(cfg));
1176 		}
1177 		// Edge case: no @Rest(config=...) annotation produced a slot, but rawConfig.get() is non-null
1178 		// (e.g. a @Bean Config factory method without a matching @Rest(config=...) attribute, or the
1179 		// SYSTEM_DEFAULT branch fired without an annotation). Still register rawConfig as the sole source.
1180 		if (result.isEmpty()) {
1181 			var rc = rawConfig.get();
1182 			if (rc != null)
1183 				result.add(new ConfigPropertySource(rc));
1184 		}
1185 		return result;
1186 	}
