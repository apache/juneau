18  
19  import static org.apache.juneau.commons.utils.Shorts.*;

21  import java.net.*;
22  import java.util.*;
23  import java.util.regex.*;

25  /**
26   * The peer allow-list behind {@link CsrfBoundary.Builder#trustedProxies(String...)}.
27   *
28   * <p>
29   * Entries are IPv4 or IPv6 literals with an optional {@code /prefix}.  Nothing is ever resolved: a host name, a
30   * bracketed form, a zone id, an IPv4-mapped IPv6 form and an out-of-range prefix are all refused.  Host bits are
31   * masked, so {@code 10.0.0.1/8} is {@code 10.0.0.0/8}, and {@code /0} matches every address of its family.
32   *
33   * <p>
34   * Matching is per family.  A peer written as an IPv4-mapped IPv6 address ({@code ::ffff:a.b.c.d}) is unwrapped to IPv4
35   * first; other IPv4-embedded forms ({@code ::a.b.c.d}, NAT64) stay IPv6.  Anything that does not parse as a literal is
36   * not in any range.
37   */
38  final class PeerRanges {
39  
40  	private static final Pattern PREFIX = Pattern.compile("0|[1-9]\\d{0,2}");
41  	private static final Pattern IPV6_CHARS = Pattern.compile("[0-9A-Fa-f:][0-9A-Fa-f:.]*");
42  	private static final Pattern ZONE = Pattern.compile("[0-9A-Za-z_.~-]+");
43  
44  	@SuppressWarnings({
45  		"java:S6218" // Private record; never compared, hashed or printed
46  	})
47  	private record Range(byte[] network, int bits) {
48  		boolean contains(byte[] address) {
49  			if (address.length != network.length)
50  				return false;
51  			for (var i = 0; i < bits; i++) {
52  				var mask = 0x80 >> (i & 7);
53  				if ((address[i >> 3] & mask) != (network[i >> 3] & mask))
54  					return false;
55  			}
56  			return true;
57  		}
58  
59  		String text() {
60  			try {
61  				return InetAddress.getByAddress(network).getHostAddress() + '/' + bits;
62  			} catch (UnknownHostException e) {
63  				throw new IllegalStateException(e);  // Unreachable: the length is 4 or 16.
64  			}
65  		}
66  	}
67  
68  	private final List<Range> ranges;
69  	private final List<String> texts;
70  
71  	/**
72  	 * Parses the entries.
73  	 *
74  	 * @param cidrs The entries.  Must not be <jk>null</jk> or empty, and must not hold a <jk>null</jk> or invalid entry.
75  	 * @throws IllegalArgumentException If an entry is invalid or there are none.
76  	 */
77  	PeerRanges(String...cidrs) {
78  		if (cidrs == null || cidrs.length == 0)
79  			throw iaex("trustedProxies requires at least one address range.");
80  		var list = new ArrayList<Range>();
81  		var seen = new LinkedHashSet<String>();
82  		for (var cidr : cidrs) {
83  			var r = parse(cidr);
84  			if (seen.add(r.text()))
85  				list.add(r);
86  		}
87  		ranges = List.copyOf(list);
88  		texts = List.copyOf(seen);
89  	}
90  
91  	/** The normalized, masked entries in the order given, without duplicates. */
92  	List<String> texts() {
93  		return texts;
94  	}
95  
96  	/**
97  	 * Whether the peer address is in a range.
98  	 *
99  	 * @param peer The address the container reported, or <jk>null</jk> when it reported none.
100 	 * @return <jk>true</jk> if the peer is in a range.  <jk>null</jk> and anything unparseable is not.
101 	 */
102 	boolean contains(String peer) {
103 		return contains(parsePeer(peer));
104 	}
105 
106 	/**
107 	 * Whether the peer address is in a range.
108 	 *
109 	 * @param peer The address the container reported, or <jk>null</jk> when it reported none.
110 	 * @return <jk>true</jk> if the peer is in a range.  <jk>null</jk> is not.
111 	 */
112 	boolean contains(InetAddress peer) {
113 		return peer != null && contains(unwrap(peer.getAddress()));
114 	}
115 
116 	private boolean contains(byte[] address) {
117 		if (address == null)
118 			return false;
119 		for (var r : ranges)
120 			if (r.contains(address))
121 				return true;
122 		return false;
123 	}
124 
125 	// An IPv4-mapped IPv6 address is the IPv4 address it carries.  InetAddress.getByName and getByAddress(byte[]) collapse
126 	// it already; Inet6Address.getByAddress(host, bytes, scope) does not.
127 	@SuppressWarnings({
128 		"java:S1168" // A null address means "no peer address" and contains() tests for it; an empty array would be a different, invalid address.
129 	})
130 	private static byte[] unwrap(byte[] a) {
131 		if (a == null)
132 			return null;
133 		if (a.length != 16)
134 			return a;
135 		for (var i = 0; i < 10; i++)
136 			if (a[i] != 0)
137 				return a;
138 		if (a[10] != (byte)0xff || a[11] != (byte)0xff)
139 			return a;
140 		return Arrays.copyOfRange(a, 12, 16);
141 	}
142 
143 	private static Range parse(String entry) {
144 		if (entry == null)
145 			throw iaex("trustedProxies entry must not be null.");
146 		var token = entry.trim();
147 		var slash = token.indexOf('/');
148 		var literal = slash < 0 ? token : token.substring(0, slash);
149 		var address = literal.indexOf(':') < 0 ? parseIpv4(literal) : parseIpv6(literal);
150 		if (address == null || (literal.indexOf(':') >= 0 && address.length != 16))
151 			throw iaex("Invalid trustedProxies entry '%s': expected an IPv4 or IPv6 literal with an optional /prefix.", entry);
152 		var max = address.length * 8;
153 		var bits = max;
154 		if (slash >= 0) {
155 			var p = token.substring(slash + 1);
156 			if (! PREFIX.matcher(p).matches() || (bits = Integer.parseInt(p)) > max)
157 				throw iaex("Invalid trustedProxies entry '%s': the prefix length must be 0-%s.", entry, max);
158 		}
159 		var network = address.clone();
160 		for (var i = 0; i < network.length; i++) {
161 			var keep = Math.max(0, Math.min(8, bits - i * 8));
162 			network[i] &= (byte)(0xff00 >> keep);
163 		}
164 		return new Range(network, bits);
165 	}
166 
167 	// A strict dotted quad: four decimal octets 0-255, no signs and no leading zeros.  Null if it is not one.
168 	@SuppressWarnings({
169 		"java:S1168" // null signals an unparseable address; callers test for it
170 	})
171 	private static byte[] parseIpv4(String s) {
172 		var parts = s.split("\\.", -1);
173 		if (parts.length != 4)
174 			return null;
175 		var out = new byte[4];
176 		for (var i = 0; i < 4; i++) {
177 			var p = parts[i];
178 			if (! (p.equals("0") || p.matches("[1-9]\\d{0,2}")))
179 				return null;
180 			var v = Integer.parseInt(p);
181 			if (v > 255)
182 				return null;
183 			out[i] = (byte)v;
184 		}
185 		return out;
186 	}
187 
188 	// Only hex digits, colons and dots reach getByName, which then parses a literal and never asks a resolver.  Returns
189 	// null for anything else, and for a form Java collapses to IPv4 (the address length then tells the caller).
190 	@SuppressWarnings({
191 		"java:S1168" // null signals an unparseable address; callers test for it
192 	})
193 	private static byte[] parseIpv6(String s) {
194 		if (! IPV6_CHARS.matcher(s).matches() || s.indexOf(':') < 0)
195 			return null;
196 		try {
197 			return InetAddress.getByName(s).getAddress();
198 		} catch (UnknownHostException e) {
199 			return null;
200 		}
201 	}
202 
203 	// A peer as the container reports it: optional surrounding brackets and an optional zone, which is dropped.
204 	@SuppressWarnings({
205 		"java:S1168" // null signals an unparseable address; callers test for it
206 	})
207 	private static byte[] parsePeer(String peer) {
208 		if (peer == null)
209 			return null;
210 		var s = peer;
211 		if (s.length() >= 2 && s.charAt(0) == '[' && s.charAt(s.length() - 1) == ']')
212 			s = s.substring(1, s.length() - 1);
213 		var pct = s.indexOf('%');
214 		if (pct >= 0) {
215 			if (! ZONE.matcher(s.substring(pct + 1)).matches())
216 				return null;
217 			s = s.substring(0, pct);
218 		}
219 		var a = s.indexOf(':') < 0 ? parseIpv4(s) : parseIpv6(s);
220 		return unwrap(a);
221 	}
