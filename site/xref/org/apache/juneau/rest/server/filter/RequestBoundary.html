17  package org.apache.juneau.rest.server.filter;
18  
19  import jakarta.servlet.http.*;
20  
21  /**
22   * Decides whether a request came from a page this application served, and supplies the CSRF token such a page embeds.
23   *
24   * <p>
25   * {@link RequestBoundaryFilter} applies an implementation to every request and answers refusals; other callers that
26   * judge a raw request directly (a message-bus session endpoint, say) call {@link #check(HttpServletRequest)} the same
27   * way.
28   *
29   * <p>
30   * Implementations judge a request on its <b>effective</b> method, via
31   * {@link org.apache.juneau.rest.server.MethodSafety#effectiveMethod(String, String, java.util.function.UnaryOperator)},
32   * so a safe request line that claims a write through a method override takes the write checks.
33   *
34   * <p>
35   * Rejections are built only inside this package ({@link BoundaryResult}'s rejecting constructor is package-private),
36   * so an implementation outside it can only {@linkplain BoundaryResult#ALLOWED allow}.
37   *
38   * <h5 class='section'>See Also:</h5><ul>
39   * 	<li class='jc'>{@link LoopbackBoundary}
40   * 	<li class='jc'>{@link RequestBoundaryFilter}
41   * </ul>
42   *
43   * @since 10.0.0
44   */
45  public interface RequestBoundary {
46  
47  	/**
48  	 * Applies the boundary to a request.
49  	 *
50  	 * @param req The request to check.  Must not be <jk>null</jk>.
51  	 * @return {@link BoundaryResult#ALLOWED} when every applicable check passed, else the first rejection encountered.
52  	 */
53  	BoundaryResult check(HttpServletRequest req);
54  
55  	/**
56  	 * The name of the header a state-changing request must carry the CSRF token in.
57  	 *
58  	 * @return The CSRF header name.
59  	 */
60  	String csrfHeader();
61  
62  	/**
63  	 * The CSRF token value a page served in answer to {@code req} must embed.
64  	 *
65  	 * @param req The request being answered.  Must not be <jk>null</jk>.
66  	 * @return The value to embed.
67  	 */
68  	String tokenFor(HttpServletRequest req);
69  }
