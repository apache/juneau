17  package org.apache.juneau.rest.server.filter;
18  
19  /**
20   * The outcome of {@link RequestBoundary#check(jakarta.servlet.http.HttpServletRequest)}: either allowed, or a
21   * rejection carrying the reason, the HTTP status to answer with, and a message safe to return to the caller.
22   *
23   * <h5 class='section'>See Also:</h5><ul>
24   * 	<li class='jc'>{@link BoundaryReason}
25   * 	<li class='jc'>{@link RequestBoundaryFilter}
26   * </ul>
27   *
28   * @since 10.0.0
29   */
30  public final class BoundaryResult {
31  
32  	/** The outcome of a request that passed every applicable check. */
33  	public static final BoundaryResult ALLOWED = new BoundaryResult(null, 0, null);
34  
35  	private final BoundaryReason reason;
36  	private final int status;
37  	private final String message;
38  
39  	BoundaryResult(BoundaryReason reason, int status, String message) {
40  		this.reason = reason;
41  		this.status = status;
42  		this.message = message;
43  	}
44  
45  	/**
46  	 * Whether the request passed.
47  	 *
48  	 * @return <jk>true</jk> if the request may proceed.
49  	 */
50  	public boolean isAllowed() { return reason == null; }
51  
52  	/**
53  	 * Why the request was rejected.
54  	 *
55  	 * @return The rejection reason, or <jk>null</jk> when {@link #isAllowed()} is <jk>true</jk>.
56  	 */
57  	public BoundaryReason reason() { return reason; }
58  
59  	/**
60  	 * The HTTP status to answer a rejected request with.
61  	 *
62  	 * @return The status code, or {@code 0} when {@link #isAllowed()} is <jk>true</jk>.
63  	 */
64  	public int status() { return status; }
65  
66  	/**
67  	 * A message describing the rejection.
68  	 *
69  	 * <p>
70  	 * Names the header at fault and what was expected of it.  It never echoes a value the request presented, and never
71  	 * reveals the expected CSRF token.
72  	 *
73  	 * @return The message, or <jk>null</jk> when {@link #isAllowed()} is <jk>true</jk>.
74  	 */
75  	public String message() { return message; }
76  
77  	@Override /* Object */
78  	public String toString() { return isAllowed() ? "ALLOWED" : reason + "(" + status + "): " + message; }
79  }
