17  package org.apache.juneau.rest.server.filter;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import java.net.*;
22  import java.util.*;
23  
24  /**
25   * The {@code Host} values of a WebSocket handshake; over RFC 8441 with no {@code Host}, one value from the request
26   * URI's authority.
27   *
28   * <p>
29   * A handshake over HTTP/2 (RFC 8441 extended CONNECT) carries its authority as the {@code :authority} pseudo-header
30   * and no {@code Host} header, which a servlet container surfaces in the handshake's request URI.  A JSR 356
31   * {@code HandshakeRequest} exposes no protocol, so the HTTP/2 case is recognized by what an RFC 6455 handshake must
32   * carry and an RFC 8441 one does not: a {@code Sec-WebSocket-Key} or {@code Upgrade} header.  The container enforces
33   * that signal on Jetty and Tomcat (a key-less HTTP/1.1 upgrade is a 400 before any configurator runs).
34   *
35   * <p>
36   * Header names are matched ignoring case, because HTTP/2 field names arrive in lower case.
37   *
38   * @param values The {@code Host} values, in order.  Empty when there is no {@code Host} and no authority to stand in
39   * 	for it.  Never <jk>null</jk>.
40   * @param synthesised <jk>true</jk> if {@code values} is the single value built from the request URI rather than a
41   * 	real {@code Host} header.
42   * @since 10.0.0
43   */
44  public record HandshakeHosts(List<String> values, boolean synthesised) {
45  
46  	/**
47  	 * Constructor.
48  	 *
49  	 * @param values The values.  Must not be <jk>null</jk>.
50  	 * @param synthesised Whether the value was built from the request URI.
51  	 */
52  	public HandshakeHosts {
53  		reqnn("values", values);
54  		values = List.copyOf(values);
55  	}
56  
57  	/**
58  	 * The {@code Host} values of a handshake.
59  	 *
60  	 * <ol>
61  	 * 	<li>Every real {@code Host} value, across all spellings of the name, is returned as is.
62  	 * 	<li>Otherwise, if the handshake has a {@code Sec-WebSocket-Key} or {@code Upgrade} header (any value, blank
63  	 * 		included), it is an RFC 6455 handshake and the result is empty.
64  	 * 	<li>Otherwise one value is built from the request URI: its host (IPv6 brackets kept) plus <c>":"</c> and its port
65  	 * 		when one is given.  No port is invented, so the caller's scheme default applies.  Userinfo is never carried.
66  	 * 		A <jk>null</jk> URI, or one with no host, gives the empty result.
67  	 * </ol>
68  	 *
69  	 * @param headers The handshake headers.  Can be <jk>null</jk>, which is empty.
70  	 * @param requestUri The handshake's request URI.  Can be <jk>null</jk>.
71  	 * @return The hosts.  Never <jk>null</jk>.
72  	 */
73  	@SuppressWarnings({
74  		"java:S3776" // Header-case, HTTP/1.1 and HTTP/2 authority branches read best as one sequence
75  	})
76  	public static HandshakeHosts of(Map<String,List<String>> headers, URI requestUri) {
77  		var hosts = new ArrayList<String>();
78  		var rfc6455 = false;
79  		if (headers != null) {
80  			for (var e : headers.entrySet()) {
81  				var name = e.getKey();
82  				if ("Host".equalsIgnoreCase(name)) {
83  					if (e.getValue() != null)
84  						for (var v : e.getValue())
85  							if (v != null)
86  								hosts.add(v);
87  				} else if ("Sec-WebSocket-Key".equalsIgnoreCase(name) || "Upgrade".equalsIgnoreCase(name)) {
88  					rfc6455 = true;
89  				}
90  			}
91  		}
92  		if (! hosts.isEmpty())
93  			return new HandshakeHosts(hosts, false);
94  		if (rfc6455 || requestUri == null)
95  			return new HandshakeHosts(List.of(), false);
96  		// getHost(), never getAuthority(): the authority would carry userinfo into the host parser.
97  		var host = requestUri.getHost();
98  		if (ib(host))
99  			return new HandshakeHosts(List.of(), false);
100 		var port = requestUri.getPort();
101 		return new HandshakeHosts(List.of(port > 0 ? host + ":" + port : host), true);
102 	}
103 }
