25  /**
26   * Request predicates shared by the {@link RequestBoundary} implementations in this package.
27   *
28   * <p>
29   * Kept in one place so two boundaries cannot come to disagree about, say, which content types are JSON.

33  	private static final String JSON = "application/json";
34  
35  	private static final String HTTP = "http";
36  	private static final String HTTPS = "https";

38  	private BoundaryChecks() {}

40  	/**
41  	 * Whether a {@code Content-Type} header value's base type is exactly {@code application/json}.
42  	 *
43  	 * <p>
44  	 * Parameters are ignored, so {@code application/json;charset=utf-8} passes.  A suffixed type such as
45  	 * {@code application/problem+json} does not: the check exists to exclude the form-encodable types, and widening it
46  	 * to "anything ending in json" would be a per-type judgement call at a security boundary.
47  	 *
48  	 * @param contentType The header value.  Can be <jk>null</jk>, which is not JSON.
49  	 * @return <jk>true</jk> if the base type is {@code application/json}, compared case-insensitively.
50  	 */
51  	static boolean isJson(String contentType) {
52  		return eqic(baseType(contentType), JSON);
53  	}
54  
55  	/**
56  	 * Whether a {@code Content-Type} header value's base type is one a cross-origin request may send without a CORS
57  	 * preflight: {@code application/x-www-form-urlencoded}, {@code multipart/form-data} or {@code text/plain}.
58  	 *
59  	 * <p>
60  	 * These are the types a plain HTML {@code <form>} can submit, which is why they matter at a CSRF boundary.
61  	 *
62  	 * @param contentType The header value.  Can be <jk>null</jk>, which is not one of them.
63  	 * @return <jk>true</jk> if the base type is one of the three, compared case-insensitively and ignoring parameters.
64  	 */
65  	static boolean isSafelistedType(String contentType) {
66  		var base = baseType(contentType);
67  		return eqic(base, "application/x-www-form-urlencoded") || eqic(base, "multipart/form-data") || eqic(base, "text/plain");
68  	}

72  	 *
73  	 * @param host The host, lower-cased.  A bracketed IPv6 literal keeps its brackets.

78  	/**
79  	 * A parsed origin: scheme, host and port, with the port resolved to the scheme's default when absent.
80  	 *
81  	 * <p>
82  	 * Two origins are the same origin exactly when their records are equal.
83  	 *
84  	 * @param scheme The scheme, lower-cased: {@code http} or {@code https}.
85  	 * @param host The host, lower-cased.  A bracketed IPv6 literal keeps its brackets.
86  	 * @param port The port, never {@code -1}.
87  	 */
88  	record Origin(String scheme, String host, int port) {}

90  	/**
91  	 * Parses a {@code Host} header value.
92  	 *
93  	 * <p>
94  	 * The value must be a reg-name, an IPv4 address or a bracketed IPv6 literal, optionally followed by
95  	 * <c>":"</c> and a decimal port from 1 to 65535 with no sign and no leading zero.  Only ASCII is accepted.
96  	 * Whitespace, control characters, {@code @}, {@code ,}, {@code /}, {@code \} and an IPv6 zone id ({@code %})
97  	 * all make the value malformed.  Nothing is re-canonicalized: {@code 127.1}, {@code [::ffff:127.0.0.1]} and a
98  	 * trailing dot are kept as written, so they compare unequal to other spellings of the same address.
99  	 *
100 	 * @param value The header value.  Can be <jk>null</jk>.
101 	 * @return The parsed value, or <jk>null</jk> if it is <jk>null</jk> or malformed.
102 	 */
103 	@SuppressWarnings({
104 		"java:S3776" // Strict header grammar: the bracketed and plain host branches read best as one sequence
105 	})
106 	static Authority parseHost(String value) {
107 		if (ie(value))
108 			return null;
109 		String host;
110 		String rest;
111 		if (value.charAt(0) == '[') {
112 			var close = value.indexOf(']');
113 			if (close < 0)
114 				return null;
115 			host = value.substring(0, close + 1);
116 			rest = value.substring(close + 1);
117 			if (! isIpv6Literal(host))
118 				return null;
119 		} else {
120 			var colon = value.indexOf(':');
121 			host = colon < 0 ? value : value.substring(0, colon);
122 			rest = colon < 0 ? "" : value.substring(colon);
123 			if (host.isEmpty() || ! isRegName(host))
124 				return null;
125 		}
126 		var port = -1;
127 		if (! rest.isEmpty()) {
128 			if (rest.charAt(0) != ':')
129 				return null;
130 			port = parsePort(rest.substring(1));
131 			if (port < 0)
132 				return null;
133 		}
134 		return new Authority(host.toLowerCase(Locale.ROOT), port);
135 	}

137 	/**
138 	 * Parses an {@code Origin} header value.
139 	 *
140 	 * <p>
141 	 * The value must be exactly <c>scheme "://" host [":" port]</c>, where the scheme is {@code http} or
142 	 * {@code https} (case-insensitive) and the host and port follow {@link #parseHost(String)}.  A path, a trailing
143 	 * slash, userinfo, a query, a fragment or whitespace anywhere makes it unparseable, as does any other scheme
144 	 * ({@code file}, {@code chrome-extension}, ...).  The literal {@code null} is not an origin either; a caller that
145 	 * gives it its own treatment must test for it first.
146 	 *
147 	 * @param value The header value.  Can be <jk>null</jk>.
148 	 * @return The parsed origin with its port resolved, or <jk>null</jk> if it is <jk>null</jk> or does not parse.
149 	 */
150 	static Origin parseOrigin(String value) {
151 		if (value == null)
152 			return null;
153 		var sep = value.indexOf("://");
154 		if (sep <= 0)
155 			return null;
156 		var scheme = value.substring(0, sep).toLowerCase(Locale.ROOT);
157 		var port = defaultPort(scheme);
158 		if (port < 0)
159 			return null;
160 		var a = parseHost(value.substring(sep + 3));
161 		if (a == null)
162 			return null;
163 		return new Origin(scheme, a.host(), a.port() < 0 ? port : a.port());
164 	}

166 	/**
167 	 * The default port of a web scheme.
168 	 *
169 	 * @param scheme The scheme, lower-cased.  Can be <jk>null</jk>.
170 	 * @return {@code 80} for {@code http}, {@code 443} for {@code https}, else {@code -1}.
171 	 */
172 	static int defaultPort(String scheme) {
173 		if (HTTP.equals(scheme))
174 			return 80;
175 		if (HTTPS.equals(scheme))
176 			return 443;
177 		return -1;
178 	}

180 	/**
181 	 * The web scheme of a WebSocket handshake URI scheme.
182 	 *
183 	 * @param scheme The URI scheme.  Can be <jk>null</jk>.
184 	 * @return {@code http} for {@code ws} or {@code http}, {@code https} for {@code wss} or {@code https} (ignoring
185 	 * 	case), else <jk>null</jk>.
186 	 */
187 	static String handshakeScheme(String scheme) {
188 		if (scheme == null)
189 			return null;
190 		return switch (scheme.toLowerCase(Locale.ROOT)) {
191 			case "ws", "http" -> HTTP;
192 			case "wss", HTTPS -> HTTPS;
193 			default -> null;
194 		};
195 	}

197 	/**
198 	 * Every value of a request header, in order, so that a duplicate can be counted.
199 	 *
200 	 * <p>
201 	 * A container may return <jk>null</jk> from {@link HttpServletRequest#getHeaders(String)} when it will not
202 	 * enumerate headers; the single {@link HttpServletRequest#getHeader(String)} value is then all there is.
203 	 *
204 	 * @param req The request.
205 	 * @param name The header name.
206 	 * @return The values; empty when the header is absent.  Never <jk>null</jk>.
207 	 */
208 	static List<String> headerValues(HttpServletRequest req, String name) {
209 		var e = req.getHeaders(name);
210 		if (e == null) {
211 			var v = req.getHeader(name);
212 			return v == null ? List.of() : List.of(v);
213 		}
214 		var l = new ArrayList<String>();
215 		while (e.hasMoreElements())
216 			l.add(e.nextElement());
217 		return l;
218 	}

220 	/**
221 	 * The {@code Host} values of a request; on HTTP/2 with no {@code Host} header, one value synthesised from the
222 	 * container's {@code :authority} view.
223 	 *
224 	 * <p>
225 	 * A container that speaks HTTP/2 may carry the authority only as a pseudo-header, which the servlet API surfaces as
226 	 * {@link HttpServletRequest#getServerName()} and {@link HttpServletRequest#getServerPort()} and not as a
227 	 * {@code Host} header.  The rules, in order:
228 	 * <ol>
229 	 * 	<li>Any real {@code Host} header is returned unchanged, repeated values included, so they fail as they do today.
230 	 * 	<li>Otherwise, unless {@link HttpServletRequest#getProtocol()} is exactly {@code HTTP/2.0}, the list is empty.
231 	 * 		Anything else ({@code HTTP/1.x}, <jk>null</jk>, {@code HTTP/2}, a different case) stays on the HTTP/1 path.
232 	 * 	<li>Otherwise one value is built from the server name (a bare IPv6 address is bracketed, as Tomcat reports it)
233 	 * 		and, when positive, <c>":"</c> plus the server port.  A <jk>null</jk> or blank server name gives the empty list.
234 	 * </ol>
235 	 *
236 	 * @param req The request.
237 	 * @return The values; empty when there is no {@code Host} and no authority to stand in for it.  Never <jk>null</jk>.
238 	 */
239 	static List<String> hostValues(HttpServletRequest req) {
240 		var hosts = headerValues(req, "Host");
241 		if (! hosts.isEmpty())
242 			return hosts;
243 		var synthesised = synthesisedHost(req);
244 		return synthesised == null ? List.of() : List.of(synthesised);
245 	}

247 	/**
248 	 * Whether {@link #hostValues(HttpServletRequest)} would return a value synthesised from the container's
249 	 * {@code :authority} view rather than a real {@code Host} header.
250 	 *
251 	 * @param req The request.
252 	 * @return <jk>true</jk> if the request has no {@code Host} header and is HTTP/2 with a usable server name.
253 	 */
254 	static boolean isSynthesisedHost(HttpServletRequest req) {
255 		return headerValues(req, "Host").isEmpty() && synthesisedHost(req) != null;
256 	}
257 
258 	/** The {@code :authority} view of an HTTP/2 request, or <jk>null</jk> when the request is not HTTP/2 or names no server. */
259 	private static String synthesisedHost(HttpServletRequest req) {
260 		if (! "HTTP/2.0".equals(req.getProtocol()))
261 			return null;
262 		var name = req.getServerName();
263 		if (ib(name))
264 			return null;
265 		// Tomcat reports an IPv6 server name without the brackets a Host header would carry.
266 		if (name.indexOf(':') >= 0 && ! name.startsWith("["))
267 			name = "[" + name + "]";
268 		var port = req.getServerPort();
269 		return port > 0 ? name + ":" + port : name;
270 	}

272 	/**
273 	 * Whether a {@code Sec-Fetch-Site} value is exactly {@code same-origin}.
274 	 *
275 	 * <p>
276 	 * Case-sensitive and untrimmed: a browser sends this exact token, and anything else did not come from one.
277 	 *
278 	 * @param value The header value.  Can be <jk>null</jk>, which is not.
279 	 * @return <jk>true</jk> only for {@code same-origin}.
280 	 */
281 	static boolean isSameOriginFetch(String value) {
282 		return "same-origin".equals(value);
283 	}

285 	/** Whether {@code s} is all reg-name characters: ASCII letters, digits, {@code - . _ ~}. */
286 	private static boolean isRegName(String s) {
287 		for (var i = 0; i < s.length(); i++) {
288 			var c = s.charAt(i);
289 			if (! (isAsciiAlnum(c) || c == '-' || c == '.' || c == '_' || c == '~'))
290 				return false;
291 		}
292 		return true;
293 	}
294 
295 	/** Whether {@code s} is a bracketed IPv6 literal: hex digits, {@code :} and {@code .} only, with a colon, no zone id. */
296 	private static boolean isIpv6Literal(String s) {
297 		var inner = s.substring(1, s.length() - 1);
298 		if (inner.indexOf(':') < 0)
299 			return false;
300 		for (var i = 0; i < inner.length(); i++) {
301 			var c = inner.charAt(i);
302 			if (! ((c >= '0' && c <= '9') || (c >= 'a' && c <= 'f') || (c >= 'A' && c <= 'F') || c == ':' || c == '.'))
303 				return false;
304 		}
305 		return true;
306 	}
307 
308 	/** A decimal port 1-65535 with no sign and no leading zero, or {@code -1}. */
309 	private static int parsePort(String s) {
310 		if (s.isEmpty() || s.length() > 5 || s.charAt(0) == '0')
311 			return -1;
312 		var p = 0;
313 		for (var i = 0; i < s.length(); i++) {
314 			var c = s.charAt(i);
315 			if (c < '0' || c > '9')
316 				return -1;
317 			p = p * 10 + (c - '0');
318 		}
319 		return p <= 65535 ? p : -1;
320 	}
321 
322 	private static boolean isAsciiAlnum(char c) {
323 		return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9');
324 	}
325 
326 	/** The media type before any parameters, trimmed, or <jk>null</jk> for a <jk>null</jk> value. */
327 	private static String baseType(String contentType) {
328 		if (contentType == null)
329 			return null;
330 		var semi = contentType.indexOf(';');
331 		return (semi < 0 ? contentType : contentType.substring(0, semi)).trim();
332 	}
