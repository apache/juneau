28  /**
29   * Thin configuration façade over the always-on request-id correlation resolver.
30   *
31   * <p>
32   * <b>The correlation lifecycle is now owned by the always-on resolver built into {@link RestSession} at session-build
33   * time</b> &mdash; it mints or honors the {@code X-Request-Id} correlation id for <i>every</i> request (including 404 /
34   * early-error paths that never reach a {@code @RestStartCall} hook), stashes it under
35   * {@code "requestId"}, echoes it on the response, and opens the {@code requestId} log-context scope.
36   * The real tuning knobs live on {@link RequestIdSettings} (resolved via {@link RestContext#getRequestIdSettings()}).
37   *
38   * <p>
39   * This filter is retained for source compatibility.  Its {@link #apply(HttpServletRequest, HttpServletResponse) apply}
40   * method is now <b>idempotent</b>: it reads the already-resolved id back through the {@link RestSession#fromRequest
41   * session-handle seam} and re-echoes it &mdash; it never double-mints.  Its per-instance builder knobs
42   * ({@link Builder#idSupplier idSupplier}, {@link Builder#validator validator}, {@link Builder#attributeKey attributeKey})
43   * are <b>documented no-ops</b>: configure {@link RequestIdSettings} instead.
44   *
45   * <h5 class='section'>See Also:</h5><ul>
46   * 	<li class='jc'>{@link RequestIdSettings}
47   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestServerLoggingAndDebugging">Logging / Debugging</a>
48   * </ul>
49   *
50   * @since 10.0.0
51   */
52  @SuppressWarnings({
53  	"java:S1192" // Duplicate string literals are HTTP header names (e.g. X-Request-Id); intentional
54  })
55  public class RequestIdFilter {
56  
57  	/** Standard request and response header name for the request id. */
58  	public static final String HEADER_REQUEST_ID = RequestIdConstants.HEADER;
59  
60  	/**
61  	 * Legacy default validator pattern, retained for source compatibility.
62  	 *
63  	 * @deprecated The resolver now defaults to sanitize-and-accept (see {@link RequestIdSettings}); this pattern is no
64  	 * 	longer applied.
65  	 */
66  	@Deprecated(since = "10.0.0", forRemoval = false)
67  	@SuppressWarnings({
68  		"java:S1133" // Retained indefinitely for source compatibility; not scheduled for removal, so not deleted
69  	})
70  	public static final String DEFAULT_VALIDATOR_PATTERN = "^[A-Za-z0-9-_]{1,128}$";
71  
72  	/**
73  	 * Constructor.
74  	 *
75  	 * @param b The builder configuring this filter.  Must not be <jk>null</jk>.
76  	 */
77  	protected RequestIdFilter(Builder b) {
78  		reqnn("builder", b);
79  	}
80  
81  	/**
82  	 * Creates a new builder.
83  	 *
84  	 * @return A new builder.
85  	 */
86  	public static Builder create() {
87  		return new Builder();
88  	}
89  
90  	/**
91  	 * Re-echoes the correlation id already resolved by the always-on {@link RestSession} resolver.
92  	 *
93  	 * <p>
94  	 * Idempotent façade: the id was already minted/honored/echoed at session build.  This reads it back through the
95  	 * {@link RestSession#fromRequest session-handle seam} (falling back to the {@code "requestId"}
96  	 * attribute) and re-echoes it on the response &mdash; it never double-mints.
97  	 *
98  	 * @param req The servlet request.  Must not be <jk>null</jk>.
99  	 * @param res The servlet response.  Must not be <jk>null</jk>.
100 	 * @return The resolved request id, or <jk>null</jk> if none has been resolved (e.g. invoked outside a Juneau call).
101 	 */
102 	public String apply(HttpServletRequest req, HttpServletResponse res) {
103 		reqnn("req", req);
104 		reqnn("res", res);
105 		var session = RestSession.fromRequest(req);
106 		String id = session != null ? session.getRequestId() : null;
107 		if (id == null && req.getAttribute("requestId") instanceof String s && ! s.isEmpty())
108 			id = s;
109 		if (id != null)
110 			res.setHeader(HEADER_REQUEST_ID, id);
111 		return id;
112 	}
113 
114 	/**
115 	 * Builder for {@link RequestIdFilter}.
116 	 *
117 	 * <p>
118 	 * The tuning methods below are <b>documented no-ops</b> retained for source compatibility.  They still reject a
119 	 * <jk>null</jk> argument (so existing null-validation contracts hold) but no longer affect resolution &mdash;
120 	 * configure {@link RequestIdSettings} instead.
121 	 */
122 	public static class Builder {
123 
124 		/**
125 		 * Constructor.
126 		 */
127 		protected Builder() {}
128 
129 		/**
130 		 * <b>No-op</b> (retained for source compatibility).  Configure {@link RequestIdSettings.Builder#idSupplier} instead.
131 		 *
132 		 * @param value The supplier.  Must not be <jk>null</jk>.
133 		 * @return This object.
134 		 */
135 		public Builder idSupplier(Supplier<String> value) {
136 			reqnn("value", value);
137 			return this;
138 		}
139 
140 		/**
141 		 * <b>No-op</b> (retained for source compatibility).  Configure {@link RequestIdSettings.Builder#validator} instead.
142 		 *
143 		 * @param value The predicate.  Must not be <jk>null</jk>.
144 		 * @return This object.
145 		 */
146 		public Builder validator(Predicate<String> value) {
147 			reqnn("value", value);
148 			return this;
149 		}
150 
151 		/**
152 		 * <b>No-op</b> (retained for source compatibility).  Configure {@link RequestIdSettings.Builder#attributeKey} instead.
153 		 *
154 		 * @param value The attribute key.  Must not be <jk>null</jk> or blank.
155 		 * @return This object.
156 		 */
157 		public Builder attributeKey(String value) {
158 			reqnn("value", value);
159 			if (value.isBlank())
160 				throw new IllegalArgumentException("Argument 'value' must not be blank.");
161 			return this;
162 		}
163 
164 		/**
165 		 * Builds the filter.
166 		 *
167 		 * @return A new {@link RequestIdFilter}.
168 		 */
169 		public RequestIdFilter build() {
170 			return new RequestIdFilter(this);
171 		}
172 	}
173 }
