25  /**
26   * Reads the forwarded origin a trusted proxy wrote, for {@link CsrfBoundary#trustedForwarding()}.
27   *
28   * <p>
29   * One header family is read ({@code X-Forwarded-Host} plus {@code X-Forwarded-Proto}, or {@code host=} plus
30   * {@code proto=} of {@code Forwarded}); the other is never looked at.  The lines of the family are flattened in wire
31   * order into entries, the {@code n}th entry from the right is selected, and everything to its left is ignored.  Every
32   * failure is a refusal: nothing here falls back to another value.
33   */
34  @SuppressWarnings({
35  	"java:S135" // Guard clauses keep the header parsing readable
36  })
37  final class ForwardedOrigin {
38  
39  	/**
40  	 * The outcome of {@link #resolve}.
41  	 *
42  	 * @param origin The forwarded origin, or <jk>null</jk> when refused.
43  	 * @param leftHosts The hosts (lower-cased, no port) of the entries strictly left of the selected one that parse.
44  	 * @param leftUnparsed Whether an entry strictly left of the selected one carried a host that did not parse.  A container
45  	 * 	may normalize such an entry into a host of its own, so the cross-check treats it as a possible match.
46  	 * @param reason Why it was refused, or <jk>null</jk> when {@code origin} is set.
47  	 */
48  	record Resolved(Origin origin, List<String> leftHosts, boolean leftUnparsed, BoundaryReason reason) {}
49  
50  	private ForwardedOrigin() {}
51  
52  	/**
53  	 * Resolves the forwarded origin.
54  	 *
55  	 * @param source The family to read.  Must not be {@link ForwardedSource#NONE}.
56  	 * @param hops The number of trusted entries; the {@code hops}th from the right is selected.
57  	 * @param headers Every value of the named header, in wire order, or <jk>null</jk> when the container cannot say in
58  	 * 	what order they arrived (a handshake map that holds the name under more than one spelling).  An absent header is
59  	 * 	an empty list.
60  	 * @return The outcome.  Never <jk>null</jk>.
61  	 */
62  	static Resolved resolve(ForwardedSource source, int hops, Function<String,List<String>> headers) {
63  		return source == ForwardedSource.FORWARDED ? resolveForwarded(hops, headers) : resolveXForwarded(hops, headers);
64  	}
65  
66  	/**
67  	 * Whether the request claims a forwarded host in the selected family, for the peer allow-list.
68  	 *
69  	 * <p>
70  	 * A claim is any sign that a proxy, or something posing as one, spoke: a null lookup (the container cannot say), a
71  	 * header with a line that does not tokenize, or a value present at all.  The other family is never looked at.
72  	 *
73  	 * @param source The family to read.  Must not be {@link ForwardedSource#NONE}.
74  	 * @param headers Every value of the named header, in wire order, or <jk>null</jk> when the container cannot say.
75  	 * @return <jk>true</jk> if the request claims a forwarded host.
76  	 */
77  	static boolean claimsHost(ForwardedSource source, Function<String,List<String>> headers) {
78  		if (source != ForwardedSource.FORWARDED) {
79  			var lines = headers.apply("X-Forwarded-Host");
80  			return lines == null || ! lines.isEmpty();
81  		}
82  		var lines = headers.apply("Forwarded");
83  		if (lines == null)
84  			return true;
85  		var elements = new ArrayList<Element>();
86  		for (var line : lines)
87  			if (! tokenize(line == null ? "" : line, elements))
88  				return true;
89  		for (var e : elements)
90  			if (e.params().containsKey("host") || e.repeated().contains("host"))
91  				return true;
92  		return false;
93  	}
94  
95  	private static Resolved refused(BoundaryReason reason) {
96  		return new Resolved(null, List.of(), false, reason);
97  	}

102 
103 	@SuppressWarnings({
104 		"java:S3776" // Header-pair validation is a flat sequence of refusals; splitting it would scatter the order
105 	})
106 	private static Resolved resolveXForwarded(int hops, Function<String,List<String>> headers) {
107 		var hostLines = headers.apply("X-Forwarded-Host");
108 		var protoLines = headers.apply("X-Forwarded-Proto");
109 		if (hostLines == null)
110 			return refused(BoundaryReason.FORWARDED_HOST_MALFORMED);
111 		if (protoLines == null)
112 			return refused(BoundaryReason.FORWARDED_PROTO_MALFORMED);
113 		if (hostLines.isEmpty())
114 			return refused(BoundaryReason.FORWARDED_HOST_MISSING);
115 		if (protoLines.isEmpty())
116 			return refused(BoundaryReason.FORWARDED_PROTO_MISSING);
117 
118 		var hostEntries = flatten(hostLines);
119 		var protoEntries = flatten(protoLines);
120 		var k = hostEntries.size();
121 		if (k != protoEntries.size())
122 			return refused(BoundaryReason.FORWARDED_HOPS_MISMATCH);
123 		if (k < hops)
124 			return refused(BoundaryReason.FORWARDED_HOPS_INSUFFICIENT);
125 
126 		var selected = k - hops;
127 		Origin result = null;
128 		for (var i = selected; i < k; i++) {
129 			var host = BoundaryChecks.parseHost(hostEntries.get(i));
130 			if (host == null)
131 				return refused(BoundaryReason.FORWARDED_HOST_MALFORMED);
132 			var proto = proto(protoEntries.get(i));
133 			if (proto == null)
134 				return refused(BoundaryReason.FORWARDED_PROTO_MALFORMED);
135 			if (i == selected)
136 				result = origin(proto, host);
137 		}
138 
139 		var left = new ArrayList<String>();
140 		var unparsed = false;
141 		for (var i = 0; i < selected; i++) {
142 			var host = BoundaryChecks.parseHost(hostEntries.get(i));
143 			if (host != null)
144 				left.add(host.host());
145 			else
146 				unparsed = true;
147 		}
148 		return new Resolved(result, left, unparsed, null);
149 	}
150 
151 	/** The lines split on commas, with SP and HTAB trimmed around each comma, in wire order.  Empty entries are kept. */
152 	private static List<String> flatten(List<String> lines) {
153 		var entries = new ArrayList<String>();
154 		for (var line : lines) {
155 			var value = line == null ? "" : line;
156 			var start = 0;
157 			while (true) {
158 				var comma = value.indexOf(',', start);
159 				entries.add(trimOws(value.substring(start, comma < 0 ? value.length() : comma)));
160 				if (comma < 0)
161 					break;
162 				start = comma + 1;
163 			}
164 		}
165 		return entries;
166 	}

172 	/** One element of {@code Forwarded}: its parameters by lower-cased name, or a marker for a repeated name. */
173 	private record Element(boolean empty, Map<String,String> params, Set<String> repeated) {}
174 
175 	@SuppressWarnings({
176 		"java:S3776" // RFC 7239 element walk with per-hop refusals; splitting it would scatter the order
177 	})
178 	private static Resolved resolveForwarded(int hops, Function<String,List<String>> headers) {
179 		var lines = headers.apply("Forwarded");
180 		if (lines == null)
181 			return refused(BoundaryReason.FORWARDED_HOST_MALFORMED);
182 		if (lines.isEmpty())
183 			return refused(BoundaryReason.FORWARDED_HOST_MISSING);
184 
185 		// The whole header must tokenize, each line on its own, so a client's open quote cannot swallow a proxy's comma.
186 		var elements = new ArrayList<Element>();
187 		for (var line : lines)
188 			if (! tokenize(line == null ? "" : line, elements))
189 				return refused(BoundaryReason.FORWARDED_HOST_MALFORMED);
190 
191 		var k = elements.size();
192 		if (k < hops)
193 			return refused(BoundaryReason.FORWARDED_HOPS_INSUFFICIENT);
194 
195 		var selected = k - hops;
196 		Origin result = null;
197 		for (var i = selected; i < k; i++) {
198 			var e = elements.get(i);
199 			if (e.empty())
200 				return refused(BoundaryReason.FORWARDED_HOST_MALFORMED);
201 			if (e.repeated().contains("host") || e.repeated().contains("proto"))
202 				return refused(e.repeated().contains("host") ? BoundaryReason.FORWARDED_HOST_MALFORMED : BoundaryReason.FORWARDED_PROTO_MALFORMED);
203 			var hostValue = e.params().get("host");
204 			if (hostValue == null)
205 				return refused(BoundaryReason.FORWARDED_HOST_MISSING);
206 			var protoValue = e.params().get("proto");
207 			if (protoValue == null)
208 				return refused(BoundaryReason.FORWARDED_PROTO_MISSING);
209 			var host = BoundaryChecks.parseHost(hostValue);
210 			if (host == null)
211 				return refused(BoundaryReason.FORWARDED_HOST_MALFORMED);
212 			var proto = proto(protoValue);
213 			if (proto == null)
214 				return refused(BoundaryReason.FORWARDED_PROTO_MALFORMED);
215 			if (i == selected)
216 				result = origin(proto, host);
217 		}
218 
219 		var left = new ArrayList<String>();
220 		var unparsed = false;
221 		for (var i = 0; i < selected; i++) {
222 			var value = elements.get(i).params().get("host");
223 			if (value == null)
224 				continue;
225 			var host = BoundaryChecks.parseHost(value);
226 			if (host != null)
227 				left.add(host.host());
228 			else
229 				unparsed = true;
230 		}
231 		return new Resolved(result, left, unparsed, null);
232 	}
233 
234 	/**
235 	 * Tokenizes one line of {@code Forwarded} into elements, appending to {@code out}.
236 	 *
237 	 * <p>
238 	 * Elements are split at commas outside quoted-strings, parameters at semicolons.  A parameter is
239 	 * <c>token OWS "=" OWS (token / quoted-string)</c>.  A backslash, a control character or an unterminated quote
240 	 * makes the line not tokenize.
241 	 *
242 	 * @return <jk>false</jk> if the line does not tokenize.
243 	 */
244 	@SuppressWarnings({
245 		"java:S3776" // Single-pass tokenizer state machine for the Forwarded grammar
246 	})
247 	private static boolean tokenize(String s, List<Element> out) {
248 		var n = s.length();
249 		var i = 0;
250 		while (true) {
251 			var params = new LinkedHashMap<String,String>();
252 			var repeated = new HashSet<String>();
253 			var any = false;
254 			while (true) {
255 				i = skipOws(s, i);
256 				if (i < n && s.charAt(i) != ',' && s.charAt(i) != ';') {
257 					var nameStart = i;
258 					while (i < n && isTchar(s.charAt(i)))
259 						i++;
260 					if (i == nameStart)
261 						return false;
262 					var name = s.substring(nameStart, i).toLowerCase(Locale.ROOT);
263 					i = skipOws(s, i);
264 					if (i >= n || s.charAt(i) != '=')
265 						return false;
266 					i = skipOws(s, i + 1);
267 					String value;
268 					if (i < n && s.charAt(i) == '"') {
269 						var close = i + 1;
270 						while (close < n && s.charAt(close) != '"') {
271 							var c = s.charAt(close);
272 							if (c == '\\' || (c < 0x20 && c != '\t') || c == 0x7f)
273 								return false;
274 							close++;
275 						}
276 						if (close >= n)
277 							return false;
278 						value = s.substring(i + 1, close);
279 						i = close + 1;
280 					} else {
281 						var valueStart = i;
282 						while (i < n && isTchar(s.charAt(i)))
283 							i++;
284 						if (i == valueStart)
285 							return false;
286 						value = s.substring(valueStart, i);
287 					}
288 					any = true;
289 					if (params.putIfAbsent(name, value) != null)
290 						repeated.add(name);
291 					i = skipOws(s, i);
292 				}
293 				if (i < n && s.charAt(i) == ';') {
294 					i++;
295 					continue;
296 				}
297 				break;
298 			}
299 			if (i < n && s.charAt(i) != ',')
300 				return false;
301 			out.add(new Element(! any, params, repeated));
302 			if (i >= n)
303 				return true;
304 			i++;
305 		}
306 	}

312 	/** {@code http} or {@code https} in any ASCII case, lower-cased; <jk>null</jk> for anything else. */
313 	private static String proto(String value) {
314 		var p = value.toLowerCase(Locale.ROOT);
315 		return "http".equals(p) || "https".equals(p) ? p : null;
316 	}
317 
318 	private static Origin origin(String proto, Authority host) {
319 		return new Origin(proto, host.host(), host.port() >= 0 ? host.port() : BoundaryChecks.defaultPort(proto));
320 	}
321 
322 	private static String trimOws(String s) {
323 		var a = 0;
324 		var b = s.length();
325 		while (a < b && (s.charAt(a) == ' ' || s.charAt(a) == '\t'))
326 			a++;
327 		while (b > a && (s.charAt(b - 1) == ' ' || s.charAt(b - 1) == '\t'))
328 			b--;
329 		return s.substring(a, b);
330 	}
331 
332 	private static int skipOws(String s, int i) {
333 		while (i < s.length() && (s.charAt(i) == ' ' || s.charAt(i) == '\t'))
334 			i++;
335 		return i;
336 	}
337 
338 	/** RFC 7230 tchar. */
339 	private static boolean isTchar(char c) {
340 		return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') || "!#$%&'*+-.^_`|~".indexOf(c) >= 0;
341 	}
