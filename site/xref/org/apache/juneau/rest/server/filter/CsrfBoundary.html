32  /**
33   * The cross-site request forgery boundary for an application reachable from the network: decides whether a
34   * state-changing request came from this application's own page, or from a client that is not a browser, and refuses
35   * it when it came from some other site's page.
36   *
37   * <h5 class='section'>What this answers, and what it does not</h5>
38   * <p>
39   * This class answers one question:
40   * <p class='bcode'>
41   * 	Did this request come from this app's own page, or from a non-browser client?
42   * </p>
43   * <p>
44   * It does <b>not</b> answer <i>who is the caller</i> (authentication) or <i>did the user mean it</i> (intent).  A
45   * user who has the application's own page open, and is induced to click something on it, produces a request that
46   * passes every check here.  Intent needs its own, separate gate.
47   *
48   * <p>
49   * For an application bound only to the loopback interface, {@link LoopbackBoundary} pins a single known origin and is
50   * stricter.  This class is for applications whose own origin is whatever the request's {@code Host} says it is.
51   *
52   * <h5 class='section'>The checks</h5>
53   * <p>
54   * {@link #check(HttpServletRequest)} runs these steps in order, and the first failure wins.  The method judged is the
55   * {@linkplain MethodSafety#effectiveMethod(String, String, java.util.function.UnaryOperator) effective method}, so a
56   * {@code method} query parameter or an override header cannot carry a write past them.
57   * <table class='styled w800'>
58   * 	<tr><th>Step</th><th>Applies to</th><th>Refused when</th><th>Reason</th><th>Status</th></tr>
59   * 	<tr><td>Allowed hosts</td><td>Every request, when {@link Builder#allowedHosts(String...)} is set</td>
60   * 		<td>The request's own host is not listed; or its {@code Host} (on HTTP/2, its {@code :authority}) is missing or
61   * 		malformed</td>
62   * 		<td>{@code HOST_NOT_ALLOWED}; {@code HOST_MISSING}, {@code HOST_MALFORMED}</td>
63   * 		<td>Not listed: 421; missing or malformed {@code Host}: 400</td></tr>
64   * 	<tr><td>Safe method</td><td>{@code GET}, {@code HEAD}, {@code OPTIONS}, {@code TRACE}</td>
65   * 		<td>Never: allowed</td><td></td><td></td></tr>
66   * 	<tr><td>Own origin</td><td>Unsafe requests</td>
67   * 		<td>{@code Host} (on HTTP/2, {@code :authority}) is absent, repeated, or not a host with an optional port</td>
68   * 		<td>{@code HOST_MISSING}, {@code HOST_MALFORMED}</td><td>400</td></tr>
69   * 	<tr><td>Peer</td><td>Unsafe requests, and safe ones when {@code allowedHosts} is set, when
70   * 		{@link Builder#trustedProxies(String...)} is set</td>
71   * 		<td>The peer is not in the allow-list and the request carries a forwarded host in the selected family; without
72   * 		one, the request is judged on the container's own view instead</td>
73   * 		<td>{@code FORWARDED_PEER_NOT_TRUSTED}</td><td>403</td></tr>
74   * 	<tr><td>Forwarded origin</td><td>Unsafe requests, and safe ones when {@code allowedHosts} is set, when
75   * 		{@link Builder#trustedForwarding(ForwardedSource)} is on</td>
76   * 		<td>The forwarded host or proto is missing, malformed, or has fewer or unequal entries than the trusted hop count;
77   * 		or the container's view disagrees with it</td>
78   * 		<td>{@code FORWARDED_HOST_MISSING}, {@code FORWARDED_PROTO_MISSING}, {@code FORWARDED_HOST_MALFORMED},
79   * 		{@code FORWARDED_PROTO_MALFORMED}, {@code FORWARDED_HOPS_INSUFFICIENT}, {@code FORWARDED_HOPS_MISMATCH},
80   * 		{@code FORWARDED_CONTAINER_MISMATCH}</td><td>403</td></tr>
81   * 	<tr><td>{@code Origin} present</td><td>Unsafe requests</td>
82   * 		<td>The literal {@code null}; or repeated, unparseable or not this server's own origin; or
83   * 		{@code Sec-Fetch-Site} present and not exactly {@code same-origin}</td>
84   * 		<td>{@code ORIGIN_NULL}, {@code ORIGIN_MISMATCH}, {@code FETCH_SITE_NOT_SAME_ORIGIN}</td><td>403</td></tr>
85   * 	<tr><td>{@code Origin} absent</td><td>Unsafe requests</td>
86   * 		<td>{@code Sec-Fetch-Site} present and not exactly {@code same-origin}; or, with no fetch metadata either, the
87   * 		{@linkplain NonBrowser non-browser policy} refuses it</td>
88   * 		<td>{@code FETCH_SITE_NOT_SAME_ORIGIN}, {@code SIMPLE_REQUEST_WITHOUT_ORIGIN}, {@code ORIGIN_MISSING}</td>
89   * 		<td>403</td></tr>
90   * 	<tr><td>Browser request</td><td>Unsafe requests from a page</td>
91   * 		<td>{@link Builder#requireJson(boolean)} is on and the type is not JSON; or the token header is missing, or
92   * 		does not verify</td>
93   * 		<td>{@code CONTENT_TYPE_NOT_JSON} (415), {@code CSRF_TOKEN_MISSING}, {@code CSRF_TOKEN_MISMATCH}</td>
94   * 		<td>403</td></tr>
95   * </table>
96   *
97   * <p>
98   * Origins are compared as normalized {@code (scheme, host, port)} triples, never as raw text: scheme and host are
99   * lower-cased (ASCII only), an absent port means the scheme's default, and an explicit port is decimal with no sign
100  * and no leading zero.  Nothing else is re-canonicalized, so {@code 127.1}, {@code [::ffff:127.0.0.1]} and a trailing
101  * dot each fail to match any other spelling of the same address.  That is fail-closed, by design.
102  *
103  * <h5 class='section'>Non-browser clients</h5>
104  * <p>
105  * A request with neither {@code Origin} nor {@code Sec-Fetch-Site} is handled by {@link Builder#nonBrowser(NonBrowser)}.
106  * The default, {@link NonBrowser#ADMIT_NON_SIMPLE}, admits only requests a browser could not have sent cross-site
107  * without a CORS preflight, which this boundary never answers.  <b>That relies on browsers sending {@code Origin} on
108  * every non-simple request they issue; current browsers do, and the {@code Origin} check is the real control for
109  * them.</b>
110  *
111  * <h5 class='section'>The token is accepted only in the header</h5>
112  * <p>
113  * The token is read from the {@link #csrfHeader()} request header and nowhere else: never a query parameter, never a
114  * form field.  A plain HTML {@code <form>} cannot set a header, so every HTML form POST, including the application's
115  * own, is refused with {@code CSRF_TOKEN_MISSING}.  This boundary is for applications whose writes go through script
116  * that sends the header.
117  *
118  * <h5 class='section'>The request's own origin, and proxies</h5>
119  * <p>
120  * By default ({@link OriginSource#HOST_HEADER}) the request's own origin is {@link HttpServletRequest#getScheme()} plus
121  * the {@code Host} header (on HTTP/2, the {@code :authority}; see below).  {@code X-Forwarded-*} and {@code Forwarded}
122  * are never read by this class unless {@link Builder#trustedForwarding(ForwardedSource)} is set (see below).  Behind a
123  * proxy that terminates TLS, use {@link OriginSource#CONTAINER} and configure the container's own forwarded-header
124  * support; the boundary then takes the origin from {@code getScheme()}, {@code getServerName()} and
125  * {@code getServerPort()}.
126  * <p>
127  * <b>Under {@code CONTAINER} the boundary must run after the container's forwarded-header rewrite.</b>  Jetty's
128  * {@code ForwardedRequestCustomizer}, Tomcat's {@code RemoteIpValve} and Spring Boot's
129  * {@code forward-headers-strategy=native} rewrite the request before any filter runs, so order does not matter.
130  * Spring Boot's {@code forward-headers-strategy=framework} registers {@code ForwardedHeaderFilter} at
131  * {@code Ordered.HIGHEST_PRECEDENCE}, so this boundary's filter must be registered strictly later, never at
132  * {@code HIGHEST_PRECEDENCE} itself.  The boundary cannot see its own order; a misordered deployment shows up as
133  * scheme-mismatch refusals and the warning described below.
134  *
135  * <h5 class='section'>Trusting a forwarded origin</h5>
136  * <p>
137  * If your container cannot rewrite the request itself, or does not do so for WebSocket upgrades, use
138  * {@link Builder#trustedForwarding(ForwardedSource)}.  The request's own origin is then the forwarded host
139  * <b>and</b> proto of the one header family named ({@code X-Forwarded-Host} plus {@code X-Forwarded-Proto}, or
140  * {@code host=} plus {@code proto=} in one {@code Forwarded} element), on HTTP requests and WebSocket handshakes
141  * alike.  The other family and {@code X-Forwarded-Port} are never read; put a non-default port in the host value.
142  * <ul>
143  * 	<li><b>Turn it on only if every request reaches the server through your proxy, and the proxy removes or appends
144  * 		after any client-supplied copy.</b>  A server clients can reach directly lets them choose the origin, and that
145  * 		defeats {@link Builder#allowedHosts(String...)} for reads.
146  * 	<li>Set {@link Builder#trustedHops(int)} to exactly the number of proxies that <i>append</i>.  A chain of
147  * 		overwriting proxies stays at 1.  <b>Never set it higher:</b> an overstated count trusts a value the client wrote.
148  * 	<li>A missing, malformed, short or misaligned value is refused ({@code FORWARDED_*}, 403), never guessed at, and
149  * 		nothing falls back to {@code Host}.  A proxy that sets only {@code X-Forwarded-Proto} is not supported.
150  * 	<li>{@link Builder#allowedHosts(String...)} is checked against the forwarded host, so list the public names.
151  * 	<li>Under {@link OriginSource#CONTAINER} the container's rewritten origin, port included, must equal the forwarded
152  * 		one, or the request is refused ({@code FORWARDED_CONTAINER_MISMATCH}); configure both to read the same header
153  * 		family and hop.  Do not enable this where the container consumes the headers (Spring
154  * 		{@code forward-headers-strategy=framework}): every request is then refused as missing.  Under
155  * 		{@link OriginSource#HOST_HEADER} and on handshakes, a container host that equals an entry left of the selected
156  * 		one is refused too, because the container trusted a hop the client could have written.
157  * 	<li>A handshake is judged only through {@link #checkHandshake(String, HandshakeHosts, Map)}.
158  * 	<li>To believe the headers only from your proxy's own address, add {@link Builder#trustedProxies(String...)}.  A
159  * 		request from another peer that carries a forwarded host is refused ({@code FORWARDED_PEER_NOT_TRUSTED}); one
160  * 		that does not is judged on the container's view.  It needs a container that reports the TCP peer (no
161  * 		{@code RemoteIpValve}, {@code ForwardedRequestCustomizer} or PROXY protocol).  A handshake's peer comes from
162  * 		{@link #checkHandshake(String, HandshakeHosts, Map, java.net.InetAddress)}: Jetty supplies it, Tomcat does not,
163  * 		so on Tomcat a handshake that carries a forwarded host is refused while the list is set.  Only the peer is
164  * 		checked: the allow-list does not walk {@code X-Forwarded-For} or {@code for=}, so a trusted proxy that appends
165  * 		behind another untrusted tier is still believed for what it wrote.
166  * </ul>
167  *
168  * <h5 class='section'>HTTP/2 and {@code :authority}</h5>
169  * <p>
170  * An HTTP/2 request has no {@code Host} header; its {@code :authority} pseudo-header plays that part, and servlet
171  * containers surface it only through {@link HttpServletRequest#getServerName()} and
172  * {@link HttpServletRequest#getServerPort()}.  So that a deployment that speaks HTTP/2 keeps working, a request with
173  * no {@code Host} header whose {@link HttpServletRequest#getProtocol()} is exactly {@code HTTP/2.0} is judged on its
174  * server name and port in {@code Host}'s place, for both {@link OriginSource}s and for
175  * {@link Builder#allowedHosts(String...)}.  Any other request without a {@code Host} is still refused
176  * ({@code HOST_MISSING}).  A request that does carry a {@code Host} header is judged on it as before.
177  * <p>
178  * The same holds for a WebSocket handshake carried over HTTP/2 (RFC 8441, extended {@code CONNECT}): with no
179  * {@code Host}, and no {@code Sec-WebSocket-Key} or {@code Upgrade} header to mark it an RFC 6455 request, the
180  * authority of the handshake's request URI stands in for {@code Host}.  Jetty advertises RFC 8441 by default once
181  * HTTP/2 is enabled; Tomcat does not support it.
182  * <p>
183  * Because the server name is what the container's forwarded-header support rewrites, <b>under
184  * {@link OriginSource#HOST_HEADER} an HTTP/2 request is judged on the forwarded host</b> when such support is
185  * enabled (Tomcat's {@code RemoteIpValve} with a host header, Spring's {@code ForwardedHeaderFilter}, Jetty's
186  * {@code ForwardedRequestCustomizer}).  {@code HOST_HEADER} is therefore protocol-dependent: it reads the raw
187  * {@code Host} on HTTP/1.1 and the container's server name on HTTP/2.  Enable forwarded-header support only behind a
188  * proxy that overwrites {@code X-Forwarded-*}, or list the expected hosts with
189  * {@link Builder#allowedHosts(String...)}.
190  *
191  * <h5 class='section'>Non-goals</h5>
192  * <ul>
193  * 	<li><b>DNS rebinding</b>, unless {@link Builder#allowedHosts(String...)} is set: a rebound page's {@code Origin} and
194  * 		{@code Host} agree.  TLS defeats rebinding; a plain-HTTP intranet application should list its host names.
195  * 	<li><b>Cross-origin callers (CORS).</b>  Refused.  The boundary adds no {@code Access-Control-Allow-*} header; do
196  * 		not put an origin-reflecting CORS layer in front of it.
197  * 	<li><b>Authentication, intent, session management.</b>  The boundary never changes a session id and never sets
198  * 		cookie attributes.
199  * 	<li><b>Script running inside the page</b> (XSS, a malicious extension) passes every check.
200  * 	<li><b>HTML form POSTs</b> and {@code Referer} checking.
201  * 	<li><b>Telling a trustworthy proxy from a client by anything but its address.</b>  The optional allow-list of
202  * 		{@link Builder#trustedProxies(String...)} checks the peer address and nothing more: it does not authenticate
203  * 		the proxy, and a client behind a listed address (a shared NAT, a sidecar) is believed too.
204  * </ul>
205  *
206  * <h5 class='section'>What an application must do</h5>
207  * <ul>
208  * 	<li>Call {@link HttpServletRequest#changeSessionId()} on login.  A per-session token follows the session id, so
209  * 		without that a token seen before login stays valid after it.
210  * 	<li>Mark the session cookie {@code HttpOnly} and {@code SameSite}, and do not scope it to a parent {@code Domain}.
211  * 	<li>Not use URL rewriting for session ids.
212  * </ul>
213  *
214  * <h5 class='section'>Responses and logging</h5>
215  * <p>
216  * A rejection message names the header at fault and the kind of value expected.  It never echoes a value the request
217  * presented, never prints this server's own origin, and never reveals whether a session exists.  Each refusal is
218  * logged at {@code FINE} without the query string, the token header, {@code Cookie} or {@code Authorization}; the raw
219  * {@code Origin} and {@code Host} (and, for a {@code FORWARDED_*} refusal, the forwarded headers) appear only at
220  * {@code FINEST}, stripped of control characters and truncated.  Three misconfiguration patterns are logged once per process at {@code WARNING}: a scheme-only mismatch, a port-only
221  * mismatch, and repeated missing tokens from the application's own origin.
222  *
223  * <h5 class='section'>Example:</h5>
224  * <p class='bjava'>
225  * 	CsrfBoundary <jv>boundary</jv> = CsrfBoundary.<jsm>create</jsm>()
226  * 		.allowedHosts(<js>"app.example.com"</js>)
227  * 		.build();
228  * 	<jv>servletContext</jv>.addFilter(<js>"csrf"</js>, <jk>new</jk> RequestBoundaryFilter(<jv>boundary</jv>))
229  * 		.addMappingForUrlPatterns(<jk>null</jk>, <jk>false</jk>, <js>"/*"</js>);
230  * </p>
231  *

357 	/**
358 	 * Applies the boundary to a request.
359 	 *
360 	 * @param req The request to check.  Must not be <jk>null</jk>.
361 	 * @return {@link BoundaryResult#ALLOWED} when every applicable check passed, else the first rejection encountered.
362 	 */
363 	@Override /* RequestBoundary */
364 	public BoundaryResult check(HttpServletRequest req) {
365 		reqnn("req", req);
366 
367 		// Judged on the method the request claims, so an override cannot carry a write past the checks below.
368 		var method = MethodSafety.effectiveMethod(req.getMethod(), req.getQueryString(), req::getHeader);
369 
370 		var hosts = BoundaryChecks.hostValues(req);
371 		Origin self;
372 		if (trustedForwarding == ForwardedSource.NONE) {
373 			self = hosts.size() == 1 ? selfOrigin(req, BoundaryChecks.parseHost(hosts.get(0))) : null;
374 		} else {
375 			// Judged only where the own origin is needed, so a direct health probe still passes.
376 			if (MethodSafety.isSafe(method) && allowedHosts.isEmpty())
377 				return BoundaryResult.ALLOWED;
378 			if (trustedProxies != null && ! trustedProxies.contains(req.getRemoteAddr())) {
379 				// Not a proxy we trust: a forwarded host from it is refused, anything else is judged as if forwarding were off.
380 				if (ForwardedOrigin.claimsHost(trustedForwarding, name -> forwardedLines(req, name)))
381 					return refuse(req, method, BoundaryReason.FORWARDED_PEER_NOT_TRUSTED, 403,
382 						forwardedMessage(BoundaryReason.FORWARDED_PEER_NOT_TRUSTED));
383 				self = hosts.size() == 1 ? selfOrigin(req, BoundaryChecks.parseHost(hosts.get(0))) : null;
384 			} else {
385 				var forwarded = forwardedSelf(req, method, hosts);
386 				if (forwarded.refusal() != null)
387 					return forwarded.refusal();
388 				self = forwarded.origin();
389 			}
390 		}
391 
392 		// The rebinding control runs on every request: a read is as exfiltratable as a write.
393 		if (! allowedHosts.isEmpty()) {
394 			if (self == null)
395 				return hostFailure(req, method, hosts);
396 			if (! allowedHosts.contains(self.host()))
397 				return refuse(req, method, BoundaryReason.HOST_NOT_ALLOWED, 421,
398 					"Request 'Host' is not a host name this server answers to.");
399 		}
400 
401 		if (MethodSafety.isSafe(method))
402 			return BoundaryResult.ALLOWED;
403 
404 		if (self == null)
405 			return hostFailure(req, method, hosts);
406 
407 		var origins = BoundaryChecks.headerValues(req, "Origin");
408 		var fetchSites = BoundaryChecks.headerValues(req, "Sec-Fetch-Site");
409 		var originMatched = false;
410 
411 		// A blank Origin is still an Origin: a browser sent the header, so it must not reach the non-browser branch.
412 		if (! origins.isEmpty()) {
413 			if (origins.size() == 1 && "null".equalsIgnoreCase(origins.get(0)))
414 				return refuse(req, method, BoundaryReason.ORIGIN_NULL, 403,
415 					"A state-changing request must not carry an opaque 'null' value in 'Origin'.");
416 			var parsed = origins.size() == 1 ? BoundaryChecks.parseOrigin(origins.get(0)) : null;
417 			if (parsed == null || ! parsed.equals(self)) {
418 				if (parsed != null)
419 					warnNearMiss(parsed, self);
420 				return refuse(req, method, BoundaryReason.ORIGIN_MISMATCH, 403,
421 					"Request 'Origin' must be a single header naming this server's own origin.");
422 			}
423 			if (! fetchSites.isEmpty() && ! isSameOriginFetch(fetchSites))
424 				return refuseFetchSite(req, method);
425 			originMatched = true;
426 		} else if (! fetchSites.isEmpty()) {
427 			if (! isSameOriginFetch(fetchSites))
428 				return refuseFetchSite(req, method);
429 		} else {
430 			switch (nonBrowser) {
431 				case REJECT:
432 					return refuse(req, method, BoundaryReason.ORIGIN_MISSING, 403,
433 						"A state-changing request must carry an 'Origin' header.");
434 				case REQUIRE_TOKEN:
435 					break;
436 				default:
437 					if (! isNonSimple(req))
438 						return refuse(req, method, BoundaryReason.SIMPLE_REQUEST_WITHOUT_ORIGIN, 403,
439 							"A state-changing request with no 'Origin' header must use a method other than GET, HEAD or POST, "
440 							+ "or carry a single 'Content-Type' that is not a form type.");
441 					return requireJson && ! isSingleJson(req) ? refuseNotJson(req, method) : BoundaryResult.ALLOWED;
442 			}
443 		}
444 
445 		return checkBrowser(req, method, originMatched);
446 	}

467 	 * Applies this boundary to a WebSocket handshake.
468 	 *
469 	 * <p>
470 	 * A WebSocket upgrade is a {@code GET}, so {@link #check(HttpServletRequest)} lets it through untouched; this
471 	 * method is the boundary for the upgrade instead.  No token can be sent on a handshake, so the control is
472 	 * {@code Origin} and {@code Sec-Fetch-Site}:
473 	 * <ol>
474 	 * 	<li>{@code Host} must be a single well-formed value ({@code HOST_MISSING}, {@code HOST_MALFORMED}).  A handshake
475 	 * 		carried over HTTP/2 (RFC 8441) has none, and its request URI's authority stands in; see {@link HandshakeHosts}.
476 	 * 	<li>The handshake's own scheme comes from its request URI: {@code ws} is {@code http}, {@code wss} is
477 	 * 		{@code https}, and {@code http} or {@code https} are used as they are.  Any other scheme, or none, means the
478 	 * 		handshake's own origin is unknown, and it is refused ({@code ORIGIN_MISMATCH}).
479 	 * 	<li>When {@link Builder#allowedHosts(String...)} is set, the {@code Host} must be one of them
480 	 * 		({@code HOST_NOT_ALLOWED}).
481 	 * 	<li>{@code Origin} must be a single value naming the same scheme, host and port as the handshake's own
482 	 * 		scheme and {@code Host} ({@code ORIGIN_MISSING}, {@code ORIGIN_NULL}, {@code ORIGIN_MISMATCH}).
483 	 * 	<li>{@code Sec-Fetch-Site}, when present, must be a single {@code same-origin} value
484 	 * 		({@code FETCH_SITE_NOT_SAME_ORIGIN}).
485 	 * </ol>
486 	 *
487 	 * <p>
488 	 * With {@link Builder#trustedForwarding(ForwardedSource)} set, the forwarded origin replaces the {@code Host} and

490 	 * forwarded header is refused with a {@code FORWARDED_*} reason.

493 	 * The {@code Host} header (or, over HTTP/2, the request URI's authority) is the handshake's own host under either

495 	 * is set, which only {@link #checkHandshake(String, HandshakeHosts, Map)} supports.  Behind a proxy that terminates
496 	 * TLS, the scheme is right only if the
497 	 * container's own forwarded-header support rewrites the request URI it hands the WebSocket layer; where it does
498 	 * not, a {@code wss} page's handshake is refused rather than admitted on a guess.
499 	 *
500 	 * @param scheme The scheme of the handshake's request URI.  Can be <jk>null</jk>, which is refused.
501 	 * @param hosts Every {@code Host} value of the handshake.  Must not be <jk>null</jk>.
502 	 * @param origins Every {@code Origin} value of the handshake.  Must not be <jk>null</jk>.
503 	 * @param fetchSites Every {@code Sec-Fetch-Site} value of the handshake.  Must not be <jk>null</jk>.
504 	 * @return {@link BoundaryResult#ALLOWED}, or the first rejection.  A rejection message never echoes a value the
505 	 * 	handshake presented.
506 	 */

576 	/**
577 	 * Every value of a handshake header in wire order, or <jk>null</jk> when the map holds the name under more than one
578 	 * spelling (the order across spellings is lost).  Empty when absent.
579 	 */
580 	private static List<String> handshakeHeader(Map<String,List<String>> headers, String name) {
581 		var l = new ArrayList<String>();
582 		var spellings = 0;
583 		if (headers != null) {
584 			for (var e : headers.entrySet()) {
585 				if (name.equalsIgnoreCase(e.getKey()) && e.getValue() != null) {
586 					spellings++;
587 					for (var v : e.getValue())
588 						if (v != null)
589 							l.add(v);
590 				}
591 			}
592 		}
593 		if (spellings > 1) {
594 			// Origin and Sec-Fetch-Site are counted, not ordered: they stay a plain list.  Forwarded headers are refused.
595 			return name.equalsIgnoreCase("Origin") || name.equalsIgnoreCase("Sec-Fetch-Site") ? l : null;
596 		}
597 		return l;
598 	}

600 	private BoundaryResult handshake(String scheme, HandshakeHosts hosts, List<String> origins, List<String> fetchSites,
601 			Function<String,List<String>> forwardedHeaders, InetAddress peer) {
602 		var values = hosts.values();
603 		var host = values.size() == 1 ? BoundaryChecks.parseHost(values.get(0)) : null;
604 		if (host == null) {
605 			if (values.isEmpty())
606 				return refuseHandshake(BoundaryReason.HOST_MISSING, 400,
607 					"A WebSocket handshake must carry a 'Host' header (or, over HTTP/2, ':authority').", hosts, origins, forwardedHeaders, peer);
608 			return refuseHandshake(BoundaryReason.HOST_MALFORMED, 400,
609 				"Handshake 'Host' must be a single header holding a host name, IPv4 address or bracketed IPv6 address, "
610 				+ "with an optional port.", hosts, origins, forwardedHeaders, peer);
611 		}
612 
613 		var ownScheme = BoundaryChecks.handshakeScheme(scheme);
614 		if (ownScheme == null)
615 			return refuseHandshake(BoundaryReason.ORIGIN_MISMATCH, 403,
616 				"The WebSocket handshake's own scheme could not be determined, so its 'Origin' cannot be checked.", hosts, origins, forwardedHeaders, peer);
617 
618 		var self = new Origin(ownScheme, host.host(), host.port() >= 0 ? host.port() : BoundaryChecks.defaultPort(ownScheme));
619 		if (trustedForwarding != ForwardedSource.NONE) {
620 			// The list-based forms have no header access, so they refuse below before any peer logic.
621 			var untrusted = forwardedHeaders != null && trustedProxies != null && ! trustedProxies.contains(peer);
622 			if (untrusted && ForwardedOrigin.claimsHost(trustedForwarding, forwardedHeaders))
623 				return refuseHandshake(BoundaryReason.FORWARDED_PEER_NOT_TRUSTED, 403,
624 					forwardedMessage(BoundaryReason.FORWARDED_PEER_NOT_TRUSTED), hosts, origins, forwardedHeaders, peer);
625 			if (! untrusted) {
626 				// With no header access (the list-based forms) the forwarded host is missing, never guessed.
627 				var resolved = forwardedHeaders == null
628 					? new ForwardedOrigin.Resolved(null, List.of(), false, BoundaryReason.FORWARDED_HOST_MISSING)
629 					: ForwardedOrigin.resolve(trustedForwarding, trustedHops, forwardedHeaders);
630 				if (resolved.reason() != null)
631 					return refuseHandshake(resolved.reason(), 403, forwardedMessage(resolved.reason()), hosts, origins, forwardedHeaders, peer);
632 				if (containerDisagrees(self, resolved))
633 					return refuseHandshake(BoundaryReason.FORWARDED_CONTAINER_MISMATCH, 403,
634 						forwardedMessage(BoundaryReason.FORWARDED_CONTAINER_MISMATCH), hosts, origins, forwardedHeaders, peer);
635 				self = resolved.origin();
636 			}
637 		}
638 
639 		if (! allowedHosts.isEmpty() && ! allowedHosts.contains(self.host()))
640 			return refuseHandshake(BoundaryReason.HOST_NOT_ALLOWED, 421,
641 				"Handshake 'Host' is not a host name this server answers to.", hosts, origins, forwardedHeaders, peer);
642 
643 		if (origins.isEmpty())
644 			return refuseHandshake(BoundaryReason.ORIGIN_MISSING, 403, "A WebSocket handshake must carry an 'Origin' header.", hosts, origins, forwardedHeaders, peer);
645 		if (origins.size() == 1 && "null".equalsIgnoreCase(origins.get(0)))
646 			return refuseHandshake(BoundaryReason.ORIGIN_NULL, 403,
647 				"A WebSocket handshake must not carry an opaque 'null' value in 'Origin'.", hosts, origins, forwardedHeaders, peer);
648 		var parsed = origins.size() == 1 ? BoundaryChecks.parseOrigin(origins.get(0)) : null;
649 		if (parsed == null || ! parsed.equals(self)) {
650 			if (parsed != null)
651 				warnNearMiss(parsed, self);
652 			return refuseHandshake(BoundaryReason.ORIGIN_MISMATCH, 403,
653 				"Handshake 'Origin' must be a single header naming this server's own origin.", hosts, origins, forwardedHeaders, peer);
654 		}
655 
656 		if (! fetchSites.isEmpty() && ! isSameOriginFetch(fetchSites))
657 			return refuseHandshake(BoundaryReason.FETCH_SITE_NOT_SAME_ORIGIN, 403,
658 				"Handshake 'Sec-Fetch-Site' must be a single 'same-origin' value when present.", hosts, origins, forwardedHeaders, peer);
659 
660 		return BoundaryResult.ALLOWED;
661 	}

680 	 * <p>

690 	/** The fixed text for a forwarded-host refusal. Visibility relaxed for testing. */
691 	static String forwardedMessage(BoundaryReason reason) {
692 		return switch (reason) {
693 			case FORWARDED_HOST_MISSING -> "A trusted-forwarding boundary requires the forwarded host header; none was usable.";
694 			case FORWARDED_PROTO_MISSING -> "A trusted-forwarding boundary requires the forwarded proto alongside the forwarded host.";
695 			case FORWARDED_HOST_MALFORMED -> "The forwarded host header is malformed.";
696 			case FORWARDED_PROTO_MALFORMED -> "The forwarded proto must be 'http' or 'https'.";
697 			case FORWARDED_HOPS_INSUFFICIENT -> "The forwarded header holds fewer entries than the trusted hop count.";
698 			case FORWARDED_HOPS_MISMATCH -> "The forwarded host and proto headers hold different numbers of entries.";
699 			case FORWARDED_CONTAINER_MISMATCH -> "The container's view of the request disagrees with the forwarded origin.";
700 			case FORWARDED_PEER_NOT_TRUSTED -> "The forwarded header was sent by a peer that is not a trusted proxy.";
701 			default -> throw iaex("Not a forwarded-host reason: %s", reason);
702 		};

708 	/** The own origin of an HTTP request when trusted forwarding is on. */
709 	private Self forwardedSelf(HttpServletRequest req, String method, List<String> hosts) {
710 		var host = hosts.size() == 1 ? BoundaryChecks.parseHost(hosts.get(0)) : null;
711 		if (host == null)
712 			return new Self(null, hostFailure(req, method, hosts));
713 		var resolved = ForwardedOrigin.resolve(trustedForwarding, trustedHops, name -> forwardedLines(req, name));
714 		if (resolved.reason() != null)
715 			return new Self(null, refuse(req, method, resolved.reason(), 403, forwardedMessage(resolved.reason())));
716 		var c = selfOrigin(req, host);
717 		// Under CONTAINER the container is declared a forwarding authority, so its view must be exactly the forwarded one.
718 		var disagrees = originSource == OriginSource.CONTAINER ? ! resolved.origin().equals(c) : containerDisagrees(c, resolved);
719 		if (disagrees)
720 			return new Self(null, refuse(req, method, BoundaryReason.FORWARDED_CONTAINER_MISMATCH, 403,
721 				forwardedMessage(BoundaryReason.FORWARDED_CONTAINER_MISMATCH)));
722 		return new Self(resolved.origin(), null);
723 	}

774 	 *

776 	 * 	{@code getServerName()} falls back to the server's own name when the request has no {@code Host}.  On HTTP/2, which

783 		if (originSource == OriginSource.CONTAINER) {

785 			var serverName = req.getServerName();

789 			if (name == null || name.port() >= 0)
790 				return null;
791 			var port = req.getServerPort();
792 			return new Origin(scheme, name.host(), port > 0 ? port : BoundaryChecks.defaultPort(scheme));
793 		}
794 		return new Origin(scheme, host.host(), host.port() >= 0 ? host.port() : BoundaryChecks.defaultPort(scheme));
795 	}

836 	 * Logs the refusal and builds its result.
837 	 *
838 	 * <p>
839 	 * {@code message} must be a fixed text naming only headers and the kind of value expected.
840 	 */
841 	private static BoundaryResult refuse(HttpServletRequest req, String method, BoundaryReason reason, int status, String message) {
842 		if (LOG.isLoggable(Level.FINE))
843 			LOG.fine("CSRF boundary refused a request: reason=" + reason
844 				+ ", method=" + sanitize(method)
845 				+ ", requestMethod=" + sanitize(req.getMethod())
846 				+ ", uri=" + sanitize(pathOnly(req.getRequestURI()))
847 				+ ", remoteAddr=" + sanitize(req.getRemoteAddr()));
848 		if (LOG.isLoggable(Level.FINEST))
849 			LOG.finest("CSRF boundary refused a request: reason=" + reason
850 				+ ", Origin=" + sanitize(String.join(", ", BoundaryChecks.headerValues(req, "Origin")))
851 				+ ", Host=" + (BoundaryChecks.isSynthesisedHost(req) ? "(h2 :authority) " : "")
852 				+ sanitize(String.join(", ", BoundaryChecks.hostValues(req)))
853 				+ (reason.name().startsWith("FORWARDED_") ? ", " + forwardedValues(name -> forwardedLines(req, name)) : ""));
854 		return new BoundaryResult(reason, status, message);
855 	}

857 	/** One warning per process for an Origin that differs from the own origin only in the way a misconfiguration would. */
858 	private static void warnNearMiss(Origin presented, Origin self) {
859 		if (! presented.host().equals(self.host()))
860 			return;
861 		if (! presented.scheme().equals(self.scheme())) {
862 			if (SCHEME_WARNED.compareAndSet(false, true))
863 				LOG.warning("CSRF boundary: a request's 'Origin' differs from this server's own origin only in its scheme.  "
864 					+ "TLS terminated upstream?  See originSource(CONTAINER) and filter ordering.");
865 		} else if (presented.port() != self.port() && PORT_WARNED.compareAndSet(false, true)) {
866 			LOG.warning("CSRF boundary: a request's 'Origin' differs from this server's own origin only in its port.  "
867 				+ "Is a proxy or port mapping rewriting 'Host'?");
868 		}
869 	}

871 	/** The URI with any query and any {@code ;} path parameters (where a container may put a session id) cut off. */
872 	private static String pathOnly(String uri) {
873 		if (uri == null)
874 			return null;
875 		var end = uri.length();
876 		for (var c : new char[] { '?', ';', '#' }) {
877 			var i = uri.indexOf(c);
878 			if (i >= 0 && i < end)
879 				end = i;
880 		}
881 		return uri.substring(0, end);
882 	}

884 	/** A value safe for one log line: control characters removed, at most {@value #MAX_LOGGED} characters. */
885 	private static String sanitize(String value) {
886 		if (value == null)
887 			return "null";
888 		var sb = new StringBuilder(Math.min(value.length(), MAX_LOGGED));
889 		for (var i = 0; i < value.length() && sb.length() < MAX_LOGGED; i++) {
890 			var c = value.charAt(i);
891 			if (c >= 0x20 && c != 0x7f)
892 				sb.append(c);
893 		}
894 		return sb.toString();
895 	}
