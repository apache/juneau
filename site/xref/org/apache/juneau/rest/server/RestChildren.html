35  /**
36   * Implements the child resources of a {@link Rest}-annotated class.
37   *
38   * <p>
39   * Holds a registry of {@link RestContext} instances keyed by {@link RestContext#getPath() composed path}.
40   * Reads (request routing via {@link #findMatch(RestSession.Builder)} and the {@link #asMap()} accessor) are
41   * lock-free against a volatile copy-on-write snapshot; mutations
42   * ({@link #addChild(Class) addChild} / {@link #removeChild(String) removeChild}) synchronize on an internal
43   * write lock and atomically replace the snapshot. This makes runtime child management safe even while
44   * requests are in flight on other threads.
45   *
46   * <p>
47   * Supports lazy-init children registered via {@link Builder#addLazy(Class, String)} — these children have their
48   * {@link RestContext} constructed on the first matching request rather than at parent startup.  The routing entry
49   * (path prefix matcher) is always populated at startup; only the full context build is deferred.
50   *
51   * <h5 class='section'>See Also:</h5><ul>
52   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestAnnotatedClasses">@Rest-Annotated Class Basics</a>
53   * </ul>
54   */
55  public class RestChildren {

57  	private static final Logger LOGGER = Logger.getLogger(RestChildren.class.getName());

59  	/**
60  	 * Holder for a lazy-init child resource entry.
61  	 *
62  	 * <p>
63  	 * At parent startup, only the path prefix and {@link UrlPathMatcher} are populated.  The full
64  	 * {@link RestContext} is built on the first request that matches the path prefix; subsequent requests
65  	 * reuse the cached context.  Two concurrent first-requests serialize on {@link #lock} so that exactly
66  	 * one materialization runs.
67  	 *
68  	 * <p>
69  	 * If materialization throws, the exception propagates to the caller as a {@link jakarta.servlet.ServletException};
70  	 * the {@link #materialized} field is left {@code null}, so the next request will retry.
71  	 */
72  	static class LazyChildEntry {
73  
74  		final RestContext.ResolvedChild resolvedChild;
75  		final Class<?> resourceClass;
76  		final String path;
77  		final UrlPathMatcher pathMatcher;
78  		final RestContext parent;
79  		final BeanStore beanStore;
80  		final ServletConfig servletConfig;
81  
82  		/** Written once under {@link #lock}; volatile for lock-free reads. */
83  		@SuppressWarnings({
84  			"java:S3077" // Publish-once reference: assigned once under double-checked locking in materialize(); the RestContext is fully built before assignment, so volatile safe-publication is sufficient.
85  		})
86  		volatile RestContext materialized;
87  
88  		private final Object lock = new Object();
89  
90  		LazyChildEntry(RestContext.ResolvedChild resolvedChild, String path, RestContext parent, BeanStore beanStore, ServletConfig servletConfig) {
91  			this.resolvedChild = resolvedChild;
92  			this.resourceClass = resolvedChild.type();
93  			this.path = path;
94  			var p = path;
95  			if (! p.endsWith("/*"))
96  				p += "/*";
97  			this.pathMatcher = UrlPathMatcher.of(p);
98  			this.parent = parent;
99  			this.beanStore = beanStore;
100 			this.servletConfig = servletConfig;
101 		}
102 
103 		boolean isMaterialized() { return materialized != null; }
104 
105 		/**
106 		 * Returns the materialized {@link RestContext}, building it if this is the first call.
107 		 *
108 		 * <p>
109 		 * Thread-safe: at most one thread runs the construction; others block on {@link #lock}.
110 		 * If construction fails the field is left {@code null}; the next request will retry.
111 		 *
112 		 * @return The materialized context. Never {@code null} on success.
113 		 * @throws ServletException If context construction fails.
114 		 */
115 		RestContext materialize() throws ServletException {
116 			RestContext rc = materialized;
117 			if (rc != null)
118 				return rc;
119 			synchronized (lock) {
120 				rc = materialized;
121 				if (rc != null)
122 					return rc;
123 				LOGGER.info(() -> "Lazy REST child materializing: " + resourceClass.getName() + " at path '" + path + "'");
124 				try {
125 					rc = buildChildContext(parent, beanStore, servletConfig, resolvedChild, null, "");
126 					rc.postInit();
127 					rc.postInitChildFirst();
128 				} catch (Exception e) {
129 					throw new ServletException("Failed to lazily materialize child REST context for " + resourceClass.getName(), unwrapThrowable(e));
130 				}
131 				materialized = rc;
132 				LOGGER.info(() -> "Lazy REST child materialized: " + resourceClass.getName() + " at path '" + path + "'");
133 				return rc;
134 			}
135 		}

137 		/**
138 		 * Resolves the path prefix for a resource class by walking its annotation hierarchy.
139 		 *
140 		 * <p>
141 		 * Replicates the most-derived-first walk that {@link RestContext} performs during full construction,
142 		 * but does so without instantiating a {@link RestContext} — enabling startup-time path extraction
143 		 * for lazy children.
144 		 *
145 		 * @param resourceClass The resource class. Must not be {@code null}.
146 		 * @return The trimmed path (without leading slash). Empty string if no {@code @Rest(path=...)} is found.
147 		 */
148 		static String resolvePathForClass(Class<?> resourceClass) {
149 			for (Class<?> c = resourceClass; c != null; c = c.getSuperclass()) {
150 				Rest r = c.getDeclaredAnnotation(Rest.class);
151 				if (r != null && isNotEmpty(r.path()))
152 					return trimLeadingSlashes(r.path());
153 			}
154 			for (Class<?> iface : resourceClass.getInterfaces()) {
155 				Rest r = iface.getDeclaredAnnotation(Rest.class);
156 				if (r != null && isNotEmpty(r.path()))
157 					return trimLeadingSlashes(r.path());
158 			}
159 			return "";
160 		}
161 	}

216 		 * <p>
217 		 * The path is pre-resolved from the resource class so that routing works immediately at parent startup.
218 		 * The full {@link RestContext} is built on the first matching request.

227 		 * @param path The pre-resolved path prefix (without leading slash). Use {@code ""} to read from
228 		 * 	{@link Rest#path() @Rest(path)} on the resource class.
229 		 * @return This object.

235 		/**
236 		 * Registers a lazy child resource entry carrying a normalized child declaration, which may include a
237 		 * host-declared {@code @Child} seed.
238 		 *
239 		 * <p>
240 		 * The path is pre-resolved from the resource class so that routing works immediately at parent startup.
241 		 * The full {@link RestContext} is built on the first matching request, with the seed (if any) threaded
242 		 * through to {@link RestChildren#buildChildContext} at that point — so a lazily-materialized child
243 		 * receives its host-seeded settings identically to an eagerly-built one.
244 		 *
245 		 * @param resolvedChild The normalized child declaration (class plus any host-declared seed). Must not be
246 		 * 	{@code null}.
247 		 * @param path The pre-resolved path prefix (without leading slash). Use {@code ""} to read from
248 		 * 	{@link Rest#path() @Rest(path)} on the resource class.
249 		 * @return This object.
250 		 */
251 		Builder addLazy(RestContext.ResolvedChild resolvedChild, String path) {
252 			var resourceClass = resolvedChild.type();
253 			var resolvedPath = isNotEmpty(path) ? trimLeadingSlashes(path) : LazyChildEntry.resolvePathForClass(resourceClass);
254 			lazyList.add(new LazyChildEntry(resolvedChild, resolvedPath, parent, beanStore, servletConfig));
255 			return this;
256 		}

292 	/**
293 	 * Lazy-init children keyed by path prefix.
294 	 *
295 	 * <p>
296 	 * These entries are registered at parent startup for routing purposes; their {@link RestContext} bodies
297 	 * are built on first use.  Once materialized, the entry's {@link LazyChildEntry#materialized} field is set
298 	 * but the entry remains in this map so subsequent reads are lock-free (volatile field read on the entry).
299 	 * This map itself is never mutated after construction, so no copy-on-write is needed.

303 	/**
304 	 * Constructor.
305 	 *
306 	 * @param builder The builder containing the settings for this object. Must not be <jk>null</jk>.
307 	 */
308 	public RestChildren(Builder builder) {
309 		this.parent = builder.parent;
310 		this.beanStore = builder.beanStore;
311 		this.servletConfig = builder.servletConfig;
312 		var initial = new LinkedHashMap<String,RestContext>();
313 		for (var rc : builder.list)
314 			initial.put(rc.getPath(), rc);
315 		this.children = u(initial);
316 		var lazyMap = new LinkedHashMap<String,LazyChildEntry>();
317 		for (var e : builder.lazyList)
318 			lazyMap.put(e.path, e);
319 		this.lazyEntries = u(lazyMap);
320 	}

336 	/**
337 	 * Called during servlet destruction on all children to invoke all {@link RestDestroy} and {@link Servlet#destroy()} methods.
338 	 *
339 	 * <p>
340 	 * Lazy children that were never materialized are silently skipped (no context was ever built for them).
341 	 */
342 	public void destroy() {
343 		for (var r : children.values())
344 			destroyChild(r);
345 		for (var e : lazyEntries.values()) {
346 			var rc = e.materialized;  // volatile read — safe without lock
347 			if (rc != null)
348 				destroyChild(rc);
349 		}
350 	}

352 	/**
353 	 * Looks through the registered children of this object and returns the best match.
354 	 *
355 	 * <p>
356 	 * Eager children are checked first (lock-free volatile read), then lazy entries.  A matching lazy entry
357 	 * triggers {@link LazyChildEntry#materialize()} which builds the full {@link RestContext} on first call
358 	 * (blocking concurrent first-requests until construction completes) and caches the result.
359 	 *
360 	 * @param builder The HTTP call builder. Must not be <jk>null</jk>.
361 	 * @return The child that best matches the call, or an empty {@link Optional} if a match could not be made.
362 	 * @throws ServletException If a lazy child fails to materialize.
363 	 */
364 	public Optional<RestChildMatch> findMatch(RestSession.Builder builder) throws ServletException {
365 		var pi = builder.getPathInfoUndecoded();
366 		if (nn(pi) && ! pi.equals("/")) {
367 			// Check eager children first.
368 			var snapshot = children;  // single volatile read
369 			for (var rc : snapshot.values()) {
370 				var pm = rc.getPathMatcher();  // Treated as nullable per RestContext's own usage (see RestContext.getPathMatcher() null-guards).
371 				UrlPathMatch uppm = pm == null ? null : pm.match(builder.getUrlPath());
372 				if (nn(uppm))
373 					return o(RestChildMatch.create(uppm, rc));
374 			}
375 			// Check lazy entries.
376 			for (var e : lazyEntries.values()) {
377 				UrlPathMatch uppm = e.pathMatcher.match(builder.getUrlPath());
378 				if (nn(uppm)) {
379 					var rc = e.materialize();
380 					return o(RestChildMatch.create(uppm, rc));
381 				}
382 			}
383 		}
384 		return oe();
385 	}

514 	/**
515 	 * Removes the child registered at the given composed path.
516 	 *
517 	 * <p>
518 	 * The removed child has {@link RestContext#destroy()} invoked (which runs {@code @RestDestroy} hooks and
519 	 * recursively destroys any grandchildren), followed by {@link Servlet#destroy()} if the underlying resource is
520 	 * a {@link Servlet}.
521 	 *
522 	 * @param path The composed path key (as returned by {@link RestContext#getPath()}) of the child to remove.
523 	 * @return The removed and destroyed {@link RestContext}, or {@code null} if no child was registered at the
524 	 * 	given path.
525 	 */
526 	public RestContext removeChild(String path) {
527 		RestContext removed;
528 		synchronized (writeLock) {
529 			removed = children.get(path);
530 			if (removed == null)
531 				return null;
532 			var next = new LinkedHashMap<>(children);
533 			next.remove(path);
534 			children = u(next);
535 		}
536 		destroyChild(removed);
537 		return removed;
538 	}
539 
540 	/**
541 	 * Removes the first child whose resource class matches the given type.
542 	 *
543 	 * @param resourceClass The resource class to match. Matching is performed via {@link RestContext#getResourceClass()}.
544 	 * @return The removed and destroyed {@link RestContext}, or {@code null} if no matching child was found.
545 	 */
546 	public RestContext removeChild(Class<?> resourceClass) {
547 		String key = null;
548 		RestContext removed = null;
549 		synchronized (writeLock) {
550 			for (var e : children.entrySet()) {
551 				if (e.getValue().getResourceClass() == resourceClass) {
552 					key = e.getKey();
553 					removed = e.getValue();
554 					break;
555 				}
556 			}
557 			if (removed == null)
558 				return null;
559 			var next = new LinkedHashMap<>(children);
560 			next.remove(key);
561 			children = u(next);
562 		}
563 		destroyChild(removed);
564 		return removed;
565 	}

571 	/**
572 	 * Shared child-context construction recipe — used both by the eager init memoizer in {@link RestContext} and by
573 	 * the runtime {@link #addChild(Class) addChild} family of methods.
574 	 *
575 	 * <p>
576 	 * Resolves a {@link Supplier} for the resource (prefers an existing bean of the same class in the bean store,
577 	 * otherwise instantiates via {@link BeanInstantiator}), constructs the child {@link RestContext} using the
578 	 * supplied {@code parent} / {@code servletConfig} / {@code pathOverride}, and invokes any
579 	 * {@code setContext(RestContext)} method on the resource via reflection (so that {@code RestServlet}-style
580 	 * resources receive their context handle).
581 	 *
582 	 * @param parent The parent {@link RestContext}. Must not be {@code null}.
583 	 * @param beanStore The bean store to resolve instances and dependencies from. Must not be {@code null}.
584 	 * @param servletConfig The {@link ServletConfig} to pass through to the child. May be {@code null}.
585 	 * @param resolvedChild The normalized child declaration (class plus any host-declared {@code @Child} seed).
586 	 * 	Must not be {@code null}.
587 	 * @param resourceInstance A pre-supplied instance of the resource. If {@code null}, the instance is resolved
588 	 * 	from the bean store or freshly instantiated via {@link BeanInstantiator}.
589 	 * @param pathOverride An explicit path segment (relative to the parent). Use {@code ""} to read the path from
590 	 * 	{@link Rest#path() @Rest(path)} on the resource class.
591 	 * @return The newly-built child {@link RestContext}, with its {@code setContext} method (if any) invoked.
592 	 * @throws Exception If construction or reflective wiring fails.
593 	 */
594 	static RestContext buildChildContext(
595 			RestContext parent,
596 			BeanStore beanStore,
597 			ServletConfig servletConfig,
598 			RestContext.ResolvedChild resolvedChild,
599 			Object resourceInstance,
600 			String pathOverride) throws Exception {
601 		var resourceClass = resolvedChild.type();
602 		Supplier<?> so;
603 		if (resourceInstance != null) {
604 			final Object r = resourceInstance;
605 			so = () -> r;
606 		} else if (beanStore.getBean(resourceClass).isPresent()) {
607 			so = () -> beanStore.getBean(resourceClass).get();
608 		} else {
609 			Object o = BeanInstantiator.of(resourceClass, beanStore).run();
610 			so = () -> o;
611 		}
612 		var cc = new RestContext(new RestContext.Args(resourceClass, parent, servletConfig, so, pathOverride, null, null, null, new RestContext.ContextKind.Child(resolvedChild)));
613 		var mi = ClassInfo.of(so.get())
614 			.getMethod(x -> x.hasName("setContext") && x.hasParameterTypes(RestContext.class))
615 			.orElse(null);
616 		if (nn(mi))
617 			mi.accessible().invoke(so.get(), cc);
618 		return cc;
619 	}

621 	private RestContext addChildInternal(Class<?> resourceClass, Object resourceInstance, String pathOverride, boolean replace) throws ServletException {
622 		if (parent == null || beanStore == null)
623 			throw isex("Cannot add a child to a RestChildren that was not initialized with a parent RestContext.");
624 		RestContext cc;
625 		try {
626 			// The dynamic-add API has no seed — a bare ResolvedChild preserves its existing no-override behavior.
627 			cc = buildChildContext(parent, beanStore, servletConfig, RestContext.ResolvedChild.ofBare(resourceClass), resourceInstance, pathOverride);
628 		} catch (Exception e) {
629 			throw new ServletException("Failed to build child REST context for " + resourceClass.getName(), unwrapThrowable(e));
630 		}
631 		var key = cc.getPath();
632 		RestContext replaced = null;
633 		synchronized (writeLock) {
634 			if (children.containsKey(key)) {
635 				if (! replace) {
636 					destroyQuietly(cc);
637 					throw isex("Child resource already registered at path '%s'.", key);
638 				}
639 				replaced = children.get(key);
640 				var withoutExisting = new LinkedHashMap<>(children);
641 				withoutExisting.remove(key);
642 				// We don't publish the intermediate "without existing" snapshot — single atomic swap below.
643 				withoutExisting.put(key, cc);
644 				children = u(withoutExisting);
645 			} else {
646 				var next = new LinkedHashMap<>(children);
647 				next.put(key, cc);
648 				children = u(next);
649 			}
650 		}
651 		if (replaced != null)
652 			destroyChild(replaced);
653 		cc.postInit();
654 		cc.postInitChildFirst();
655 		return cc;
656 	}
657 
658 	private static void destroyChild(RestContext child) {
659 		// Servlet-backed resources delegate Servlet.destroy() through to RestContext.destroy() (see RestServlet.destroy),
660 		// so we must avoid double-destruction in that case — call Servlet.destroy() and let it transitively call
661 		// RestContext.destroy() exactly once.
662 		if (child.getResource() instanceof Servlet s)
663 			s.destroy();
664 		else
665 			child.destroy();
666 	}
667 
668 	private static void destroyQuietly(RestContext child) {
669 		try {
670 			destroyChild(child);
671 		} catch (Exception ignored) {
672 			// best-effort cleanup of a child we're about to throw away
673 		}
674 	}

676 	private static Throwable unwrapThrowable(Throwable t) {
677 		if (t instanceof InvocationTargetException t2)
678 			return t2.getTargetException();
679 		return t;
680 	}
