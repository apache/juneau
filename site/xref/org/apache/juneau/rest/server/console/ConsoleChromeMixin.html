34  /**
35   * Mixin that serves the admin-console chrome stylesheet at {@code /juneau-console/chrome.css}.
36   *
37   * <p>
38   * Compose into a host resource via {@link Rest#mixins() @Rest(mixins=ConsoleChromeMixin.class)} (see
39   * {@code DataTablesMixin} in the {@code juneau-rest-server-datatables} module for the identical asset-serving
40   * pattern this mirrors &mdash; not linked here since that module is not a dependency of this one). The served
41   * response is the static
42   * structural {@code chrome.css} (shipped in this module's classpath) with {@link Theme#OPEN}'s block, then the
43   * selected stock theme's override block if one other than {@code open} was selected, appended as
44   * {@code html:root{}} blocks.
45   *
46   * <h5 class='section'>Mount styles:</h5>
47   * <p>
48   * Both of the following arrangements serve the stylesheets at the {@link #CHROME_CSS_PATH} / {@link #THEME_CSS_DIR}
49   * URLs with no path juggling by the host resource:
50   * <ul>
51   * 	<li><b>Composed</b> &mdash; the host resource is mounted wherever the application already mounts it (e.g.
52   * 		{@code /rest/*}) and the stylesheets hang off that mount, at
53   * 		<code>&lt;host-mount&gt;/juneau-console/chrome.css</code>.
54   * 	<li><b>Standalone</b> &mdash; the host resource is registered with the servlet container at url-pattern
55   * 		{@code /juneau-console/*} so the stylesheets sit at a fixed site-root URL, independent of which page or tab
56   * 		rendered the referencing {@code <link>}.
57   * </ul>
58   * <p>
59   * The two arrangements need different operation paths, because a container mount at {@code /juneau-console/*}
60   * reports {@code servletPath="/juneau-console"} and Juneau resolves an operation's path against the request URI
61   * with {@code contextPath + servletPath} already removed &mdash; leaving only {@code /chrome.css} to match.
62   * Each operation below therefore declares <i>both</i> its prefixed path and the same path minus the
63   * {@code /juneau-console} prefix, so whichever one the arrangement leaves to be matched resolves.
64   *
65   * <h5 class='section'>Theme precedence</h5>
66   * <p>{@link Builder#theme(String)} if set, else {@link Theme#OPEN}. {@link Theme#OPEN}'s block is always appended
67   * first; a selected non-{@code open} stock theme adds exactly one more block.
68   *
69   * <h5 class='section'>Stylesheet load-order band:</h5>
70   * <p>
71   * At equal cascade specificity, whichever of {@code juneau-views.css} and this class's {@code chrome.css} is
72   * declared later in a page's {@code <head>} wins &mdash; the exact ambiguity the {@code overrideBlock} theme block
73   * below is raised to {@code html:root} to make irrelevant. Juneau's own in-tree page emitters (the examples
74   * module) don't rely on that fix alone: they declare a five-position stylesheet/script load-order band,
75   * enforced by a build-time test, so their own asset order is never left to chance in the first place:
76   * <ol>
77   * 	<li>vendor stylesheets
78   * 	<li>{@code juneau-views.css} &mdash; the views base layer
79   * 	<li>first-party widget stylesheets that build on the views layer, e.g. {@code juneau-calendar.css}
80   * 	<li>the consumer theme, e.g. this class's {@code chrome.css}
81   * 	<li>page-local {@code <style>}
82   * </ol>
83   * <p>
84   * That guard reaches only the pages Juneau itself emits. It says nothing about, and claims no control over, an
85   * arbitrary host application's own document &mdash; a consumer is free to link {@code chrome.css} and
86   * {@code juneau-views.css} in either order, which is precisely why the {@code html:root} fix exists: so the
87   * active theme wins regardless of which order the consumer's links happen to appear in.
88   *
89   * <h5 class='section'>Example:</h5>
90   * <p class='bjava'>
91   * 	<jc>// Stock chrome.css (Theme.OPEN tokens), mounted standalone at /juneau-console/*.</jc>
92   * 	<ja>@Rest</ja>(mixins=ConsoleChromeMixin.<jk>class</jk>)
93   * 	<jk>public class</jk> ConsoleAssetsRest <jk>extends</jk> BasicRestServlet {
94   * 		<ja>@Bean</ja>
95   * 		<jk>public</jk> ConsoleChromeMixin consoleChrome() {
96   * 			<jk>return</jk> ConsoleChromeMixin.<jsm>create</jsm>().build();
97   * 		}
98   * 	}
99   *
100  * 	<jc>// Bake a different stock palette into the served chrome.css.</jc>
101  * 	ConsoleChromeMixin.<jsm>create</jsm>().theme(<js>"gray"</js>).build();
102  * </p>
103  *
104  * <p>
105  * Custom palettes aren't configured here. Author them in the FreeMarker chrome with {@code <@theme>}/{@code <@token>}
106  * (see the {@link Theme} example), which emits the stock-theme {@code <link>} plus an inline override block.
107  *
108  * @since 10.0.0
109  */

162 	/**
163 	 * The inline pre-paint script that picks the page's colour scheme before first paint.
164 	 *
165 	 * <p>
166 	 * It sets {@code data-jc-scheme} on the root element from, in order: the visitor's stored choice (only when the
167 	 * root carries {@code data-jc-scheme-toggle}), the server pin ({@code data-jc-scheme-pin}), then the browser's
168 	 * {@code prefers-color-scheme}. It never throws: storage and media-query failures fall through to the next
169 	 * source, ending at {@code light}.
170 	 *
171 	 * <p>
172 	 * The text is a byte-for-byte constant so {@link #SCHEME_SCRIPT_CSP_HASH} is stable. Emit it as the first child
173 	 * of {@code <head>}, ahead of every stylesheet.
174 	 */
175 	public static final String SCHEME_SCRIPT = String.join("\n",
176 		"(function(d){var e=d.documentElement,k=\"juneau-console:scheme\",s=null;",
177 		"if(e.hasAttribute(\"data-jc-scheme-toggle\")){try{s=localStorage.getItem(k)}catch(x){}}",
178 		"if(s!==\"light\"&&s!==\"dark\"){s=e.getAttribute(\"data-jc-scheme-pin\");}",
179 		"if(s!==\"light\"&&s!==\"dark\"){try{s=matchMedia(\"(prefers-color-scheme: dark)\").matches?\"dark\":\"light\"}catch(x){s=\"light\"}}",
180 		"e.setAttribute(\"data-jc-scheme\",s);})(document);");

187 	 * script. The console shell's inline {@code <style>} (theme override blocks) still needs the request nonce or
188 	 * {@code 'unsafe-inline'} in {@code style-src}.
189 	 */
190 	public static final String SCHEME_SCRIPT_CSP_HASH = sha256Source(SCHEME_SCRIPT);

274 	/**
275 	 * The framework-authored role-token alias derivations.
276 	 *
277 	 * <p>
278 	 * {@code chrome.css} consumes <i>role</i>-named tokens ({@code --jc-header-bg}, {@code --jc-surface},
279 	 * {@code --jc-on-accent}, ...); {@link Theme#OPEN} (and every consumer theme) still defines the original
280 	 * <i>colour</i>-named leaf tokens ({@code --jc-white}, {@code --jc-border-2}, ...). This block bridges the two
281 	 * by deriving each role token <b>from</b> the legacy token it replaces, so a consumer's existing override of a
282 	 * legacy token still flows through to every derived role, and a consumer who sets a role token directly
283 	 * out-ranks the derived default by cascade order (this block is emitted inside the {@code Theme.OPEN} block,
284 	 * before any active-theme override block).
285 	 *
286 	 * <p>
287 	 * These aliases are <b>permanent</b> (no deprecation window) and are deliberately <i>not</i>
288 	 * {@link Theme#OPEN} tokens. A {@code var(--jc-*)}-valued token IS legal Theme-layer syntax &mdash;
289 	 * {@code Theme.Builder} recognizes it as a reference and resolves it to a concrete literal at {@code build()}
290 	 * time &mdash; but that resolution scope deliberately excludes these role aliases, which are appended here
291 	 * outside {@code Theme.OPEN}'s token map. Making them {@code Theme.OPEN} tokens instead would resolve each alias
292 	 * to a <i>fixed literal</i> at composition time, snapshotting the live CSS cascade (the dark-mode /
293 	 * user-agent overrides that reach these role tokens at use time) into a frozen value &mdash; so
294 	 * {@code Theme.OPEN} is kept all-literal and the aliases stay here as framework-authored literal text that is
295 	 * emitted verbatim (never routed through {@code CssValueGrammar}, which exists to validate <i>consumer</i>
296 	 * input, not the framework's own stylesheet). {@code Theme.OPEN} owns leaf values; this block owns derived
297 	 * values; no token name is declared by both.
298 	 *
299 	 * <p>
300 	 * {@code --jc-danger-text} and {@code --jc-success-text} are the text-ink roles of the danger and success
301 	 * colours; the fills keep {@code --jc-danger} and {@code --jc-success}.
302 	 *
303 	 * <p>
304 	 * {@code --jc-header-bg} (and, transitively, {@code --jc-nav-bg}) keys off {@code --jc-white}, not
305 	 * {@code --jc-chrome-bg}: in the light scheme the header and both page-nav rows (Page Tabs and Page Subtabs)
306 	 * stay white on every stock theme (the dark layer sets the header background directly), matching
307 	 * {@link Theme#OPEN}'s blue look. Theme accent still paints the selected-tab
308 	 * indicator, page-nav floor, and the rest of the page; {@code --jc-chrome-bg} remains the page-chrome
309 	 * fallback under {@code --jc-page-bg}, the hover fill, and dialog chrome. Pinning header/nav to
310 	 * {@code --jc-chrome-bg} washed those bars in light-red / light-brown / red / gray. {@code --jc-page-nav-accent}
311 	 * defaults to {@code --jc-accent} so a consumer can retint the page-nav floor / selected-section bar without
312 	 * recoloring buttons. {@code --jc-table-header-bg} also keys off {@code --jc-white} so DataTable / views-table
313 	 * column headers stay white (the original artwork gives {@code table.dataTable thead th} no background), not the page-chrome grey.
314 	 */
315 	static final String OPEN_ROLE_ALIASES = String.join("",
316 		"--jc-surface:var(--jc-white);",
317 		"--jc-header-bg:var(--jc-white);",
318 		"--jc-nav-bg:var(--jc-header-bg);",
319 		"--jc-page-nav-accent:var(--jc-accent);",
320 		"--jc-control-bg:var(--jc-surface);",
321 		"--jc-table-bg:var(--jc-surface);",
322 		"--jc-table-row-bg:var(--jc-surface);",
323 		"--jc-on-accent:var(--jc-white);",
324 		"--jc-on-btn-primary:var(--jc-on-accent);",
325 		"--jc-hover-bg:var(--jc-chrome-bg);",
326 		// DataTable / views-table headers are white (the original artwork gives `table.dataTable thead th` no background;
327 		// the white card shows through). Do not key this off --jc-chrome-bg — that page-chrome grey tints
328 		// every column header and is the wrong surface (the detail-table thead grey is a different widget).
329 		"--jc-table-header-bg:var(--jc-white);",
330 		"--jc-control-border:var(--jc-border-2);",
331 		"--jc-header-icon-text:var(--jc-header-icon-color);",
332 		"--jc-btn-primary-bg:var(--jc-btn-primary);",
333 		"--jc-btn-primary-bg-hover:var(--jc-btn-primary-hover);",
334 		// Text roles: danger/success used as ink, so the dark layer can lighten them without touching the fills.
335 		"--jc-danger-text:var(--jc-danger);",
336 		"--jc-success-text:var(--jc-success);"
337 	);

427 	/**
428 	 * [GET /juneau-console/themes/{file}] &mdash; serve one of the shipped stock-theme stylesheets.
429 	 *
430 	 * <p>
431 	 * The {@code {file}} segment is fail-closed to the known {@link #BUILTIN_THEME_NAMES} stylesheet filenames, so
432 	 * this endpoint can never read an arbitrary classpath resource. Each stylesheet is a structure-free
433 	 * {@code html:root{}} token block (one per {@link Theme} constant); the FTL {@code <@theme name="…"/>} directive

700 	/**
701 	 * Renders a theme's override block: one {@code html:root&#123;...&#125;} carrying the theme's leaf tokens, then its
702 	 * alias tokens, each in declaration order.
703 	 *
704 	 * <p>
705 	 * The two halves are escaped differently. A leaf is a validated literal and goes through
706 	 * {@link CssValueEscaper}: a grammar-accepted value can still carry a character such as {@code ;} that would end
707 	 * the declaration early. An alias is a single anchored {@code var(--jc-name)} reference
708 	 * ({@link Theme.Builder#alias(String, String)}), in which nothing dangerous is representable, so it is emitted
709 	 * verbatim. That asymmetry is the reason {@link Theme#getAliases()} stays a map.
710 	 *
711 	 * <p>
712 	 * The block uses the {@code html} type prefix. Two {@code :root} blocks in separately linked stylesheets tie at
713 	 * specificity {@code (0,0,1,0)} and are resolved by {@code <link>} order, which this framework neither sets
714 	 * nor observes. {@code html:root} scores {@code (0,0,1,1)}, so a theme override wins in either order.
715 	 *
716 	 * <p>
717 	 * A theme with no leaves and no aliases renders as the empty string. Each declared name is re-checked against
718 	 * the reserved {@code --jc-chrome-*} namespace (defense in depth; a {@link Theme} built by its builder can't
719 	 * carry one). This method is public so the FreeMarker {@code <@theme>} directive renders its inline block with
720 	 * the same code that serves {@code chrome.css}.
721 	 *
722 	 * <h5 class='section'>Example:</h5>
723 	 * <p class='bjava'>
724 	 * 	Theme <jv>theme</jv> = Theme.<jsm>create</jsm>(<js>"corporate"</js>)
725 	 * 		.token(<js>"--jc-accent"</js>, <js>"#b45309"</js>)
726 	 * 		.alias(<js>"--jc-tab-bar-bg"</js>, <js>"var(--jc-card-bg)"</js>)
727 	 * 		.build();
728 	 * 	String <jv>css</jv> = ConsoleChromeMixin.<jsm>overrideBlock</jsm>(<jv>theme</jv>);
729 	 * 	<jc>// html:root{--jc-accent:#b45309;--jc-tab-bar-bg:var(--jc-card-bg);}</jc>
730 	 * </p>
731 	 *
732 	 * @param theme The theme.  Must not be <jk>null</jk>.
733 	 * @return The override block, or an empty string if the theme declares nothing.
734 	 */
735 	public static String overrideBlock(Theme theme) {
736 		if (theme.getTokens().isEmpty() && theme.getAliases().isEmpty())
737 			return "";
738 		var sb = new StringBuilder("html:root{");
739 		for (var e : theme.getTokens().entrySet()) {
740 			Theme.rejectReservedChromeDeclaration(e.getKey(), "theme '" + theme.getName() + "' leaf token");
741 			sb.append(e.getKey()).append(':').append(CssValueEscaper.escape(e.getValue())).append(';');
742 		}
743 		for (var e : theme.getAliases().entrySet()) {
744 			Theme.rejectReservedChromeDeclaration(e.getKey(), "theme '" + theme.getName() + "' alias");
745 			sb.append(e.getKey()).append(':').append(e.getValue()).append(';');
746 		}
747 		return sb.append('}').toString();
748 	}

840 	/**
841 	 * Renders the dark-scheme override rule for app-declared dark tokens.
842 	 *
843 	 * <p>
844 	 * The rule is {@code @media not print{html:root[data-jc-scheme="dark"]{…}}}, so it wins over the light override
845 	 * block only while the page is dark and never reaches print. Leaves come first, then aliases, each in declaration
846 	 * order. A leaf must be a literal: a dark leaf that is itself a {@code var()} reference is rejected, because it
847 	 * would resolve against the light palette. A live link belongs in an alias. Names and values get the same checks
848 	 * as {@link Theme.Builder}: the name shape, the reserved {@code --jc-chrome-*} namespace, the value grammar, and
849 	 * one channel per name.
850 	 *
851 	 * <h5 class='section'>Example:</h5>
852 	 * <p class='bjava'>
853 	 * 	String <jv>css</jv> = ConsoleChromeMixin.<jsm>schemeOverrideBlock</jsm>(
854 	 * 		Map.<jsm>of</jsm>(<js>"--jc-link"</js>, <js>"#6cb4ff"</js>), Map.<jsm>of</jsm>(<js>"--jc-focus"</js>, <js>"var(--jc-accent)"</js>));
855 	 * 	<jc>// @media not print{html:root[data-jc-scheme="dark"]{--jc-link:#6cb4ff;--jc-focus:var(--jc-accent);}}</jc>
856 	 * </p>
857 	 *
858 	 * @param tokens The dark leaf tokens by name.  Must not be <jk>null</jk>.
859 	 * @param aliases The dark aliases by name, each a {@code var(--jc-name)} reference.  Must not be <jk>null</jk>.
860 	 * @return The rule, or an empty string if both maps are empty.
861 	 * @throws IllegalArgumentException If a name or value fails validation.
862 	 */
863 	public static String schemeOverrideBlock(Map<String,String> tokens, Map<String,String> aliases) {
864 		if (tokens.isEmpty() && aliases.isEmpty())
865 			return "";
866 		var b = Theme.create("dark");
867 		for (var e : tokens.entrySet()) {
868 			if (CssValueGrammar.normalize(e.getValue()).toLowerCase(Locale.ROOT).contains("var("))
869 				throw iaex("Dark token '%s' is '%s', but dark values must be literals.  Declare a live reference as an alias.", e.getKey(), e.getValue());
870 			b.token(e.getKey(), e.getValue());
871 		}
872 		for (var e : aliases.entrySet())
873 			b.alias(e.getKey(), e.getValue());
874 		var t = b.build();
875 		var sb = new StringBuilder("@media not print{html:root[data-jc-scheme=\"dark\"]{");
876 		for (var e : t.getTokens().entrySet())
877 			sb.append(e.getKey()).append(':').append(CssValueEscaper.escape(e.getValue())).append(';');
878 		for (var e : t.getAliases().entrySet())
879 			sb.append(e.getKey()).append(':').append(e.getValue()).append(';');
880 		return sb.append("}}").toString();
881 	}

900 	 * derivations ({@link #OPEN_ROLE_ALIASES}) appended inside the same block, immediately after the leaf tokens.
901 	 * The aliases are appended as trusted framework literal text (not routed through {@code CssValueEscaper},
902 	 * which is for consumer-supplied values); folding them into {@code Theme.OPEN}'s block keeps the served
903 	 * response at exactly one {@code html:root{}} block for the default theme while still placing every derived
904 	 * default before any active-theme override block. See {@link #overrideBlock(Theme)} for why the selector carries
905 	 * the {@code html} type prefix.
906 	 */
907 	private static String openRootBlock() {
908 		var sb = new StringBuilder();
909 		sb.append("html:root{");
910 		for (var e : Theme.OPEN.getTokens().entrySet())
911 			sb.append(e.getKey()).append(':').append(CssValueEscaper.escape(e.getValue())).append(';');
912 		sb.append(OPEN_ROLE_ALIASES);
913 		sb.append('}');
914 		return sb.toString();
915 	}
916 
917 	/** Returns the shipped static chrome.css text, reading (and caching, process-wide - the static file never varies) it from the classpath on first call. */
918 	private static String staticCss() throws IOException {
919 		var s = staticCss;
920 		if (s == null) {
921 			synchronized (ConsoleChromeMixin.class) {
922 				s = staticCss;
923 				if (s == null) {  // HTT: the "already set" branch is only reachable under a lock-acquisition race - unhittable single-threaded.
924 					try (var in = ConsoleChromeMixin.class.getResourceAsStream(CHROME_CSS_RESOURCE)) {
925 						if (in == null)
926 							throw new IOException("Classpath resource not found: " + CHROME_CSS_RESOURCE);
927 						s = IoUtils.read(in);
928 					}
929 					staticCss = s;
930 				}
931 			}
932 		}
933 		return s;
934 	}
