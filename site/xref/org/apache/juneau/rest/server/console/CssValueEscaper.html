17  package org.apache.juneau.rest.server.console;
18  
19  /**
20   * A declaration-boundary CSS escaper for the {@code --jc-name: <value>;} custom-property declaration
21   * {@code ConsoleChromeMixin} appends to the served {@code chrome.css}.
22   *
23   * <p>
24   * Every occurrence of the CSS/markup breakout set &mdash; <code>; {  } \ &lt; &gt;</code> &mdash; plus any C0
25   * control character is replaced by its CSS numeric escape (<code>\3B&nbsp;</code> for <code>;</code>,
26   * <code>\7B&nbsp;</code> for <code>{</code>, <code>\7D&nbsp;</code> for <code>}</code>, <code>\5C&nbsp;</code> for
27   * <code>\</code>, <code>\3C&nbsp;</code> for <code>&lt;</code>, <code>\3E&nbsp;</code> for <code>&gt;</code>,
28   * trailing-space terminator included). Everything else passes through unescaped.
29   *
30   * <p>
31   * <b>This is defense-in-depth for the {@code :root{}} declaration boundary &mdash; not the sink-safety
32   * mechanism.</b> Sink safety is {@code CssValueGrammar}'s accept-known-safe grammar plus {@code chrome.css}'s
33   * sink-property rules (color/font/length tokens never sink into a url-capable property). This escaper is
34   * deliberately <b>not</b> widened to include <code>(</code>/<code>)</code> &mdash; doing so would corrupt
35   * legitimate {@code linear-gradient(...)} / {@code rgb(...)} values on decode &mdash; and is explicitly
36   * <b>not</b> trusted to make a raw {@code url\n(} "safe": {@code CssValueGrammar} already REJECTs any value
37   * containing a C0/C1 control character before it ever reaches this class, so the C0 clause here is a leftover
38   * belt for any code path that emits a pre-validation value, not a control someone should rely on.
39   */
40  final class CssValueEscaper {
41  
42  	private CssValueEscaper() {}
43  
44  	/**
45  	 * Escapes the CSS/markup breakout set and C0 control characters in the specified value.
46  	 *
47  	 * @param value The value to escape.
48  	 * @return The escaped value, safe to place inside a {@code --jc-name: <value>;} declaration.
49  	 */
50  	static String escape(String value) {
51  		var sb = new StringBuilder(value.length());
52  		for (var i = 0; i < value.length(); i++) {
53  			var c = value.charAt(i);
54  			if (isBreakout(c)) {
55  				sb.append('\\').append(Integer.toHexString(c).toUpperCase(java.util.Locale.ROOT)).append(' ');
56  			} else {
57  				sb.append(c);
58  			}
59  		}
60  		return sb.toString();
61  	}
62  
63  	/**
64  	 * Quotes {@code value} as a CSS string literal ({@code "..."}) with every declaration-boundary breakout
65  	 * character escaped, including {@code "}. Safe to place as the entire value of {@code content}.
66  	 *
67  	 * @param value The string to quote. Must not be <jk>null</jk>.
68  	 * @return A double-quoted CSS string literal.
69  	 */
70  	static String quotedString(String value) {
71  		var sb = new StringBuilder(value.length() + 8);
72  		sb.append('"');
73  		for (var i = 0; i < value.length(); i++) {
74  			var c = value.charAt(i);
75  			if (isBreakout(c) || c == '"')
76  				sb.append('\\').append(Integer.toHexString(c).toUpperCase(java.util.Locale.ROOT)).append(' ');
77  			else
78  				sb.append(c);
79  		}
80  		sb.append('"');
81  		return sb.toString();
82  	}
83  
84  	private static boolean isBreakout(char c) {
85  		return c == ';' || c == '{' || c == '}' || c == '\\' || c == '<' || c == '>' || c <= 0x1F;
86  	}
87  }
