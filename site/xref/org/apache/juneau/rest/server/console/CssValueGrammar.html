24  /**
25   * Normalizes and validates {@link Theme.Builder#token(String, String) Theme token} values against an
26   * accept-known-safe allowlist grammar &mdash; the opposite of a {@code url(} blocklist.
27   *
28   * <p>
29   * {@code url()} is deliberately not a production in any of the seven allowed value shapes (hex color, functional
30   * color, named/global keyword color, length/number list, font-family list, gradient function, box-shadow): every spelling of
31   * {@code url(...)} REJECTs by construction, not by pattern-matching the threat.  Two ordered stages:
32   *
33   * <ol>
34   * 	<li><b>Normalize (belt, runs first):</b> trim; REJECT any C0 (<code>\x00\u2013\x1F</code>) or C1/DEL
35   * 		(<code>\x7F\u2013\x9F</code>) control character (this is what actually kills the {@code url\t(} /
36   * 		{@code url\n(} CSS-hex reconstruction vector &mdash; a control character can never survive to be hex-escaped
37   * 		by {@link CssValueEscaper} and decoded back into live whitespace); strip CSS comments
38   * 		(<code>/&#42; &#42;/</code>, dotall &mdash; kills the {@code url/**&#47;(} vector); REJECT if the
39   * 		comment-stripped remainder matches <code>(?i)url\s*\(</code> anywhere (belt-and-suspenders for {@code url (}
40   * 		and any residual whitespace-before-paren form).
41   * 	<li><b>Allowlist grammar (accept-known-safe):</b> the normalized value must match exactly one of the seven shapes
42   * 		below. Every production is a full-string {@code Pattern.matches(...)}-equivalent (anchored {@code ^...$}),
43   * 		never {@code find(...)}.
44   * </ol>
45   *
46   * <p>
47   * The gradient production is deliberately tighter than "anything shaped like a function call": nested function
48   * calls are restricted to exactly {@code rgb()}/{@code rgba()}/{@code hsl()}/{@code hsla()}/{@code calc()}
49   * (case-insensitive) &mdash; there is no general {@code ident(} production, so {@code linear-gradient(url(evil))}
50   * REJECTs at the grammar layer itself (a distinct, independent check from the normalization-stage {@code url(}
51   * reject above), not merely because the top-level belt happened to catch it first.
52   */
53  @SuppressWarnings({
54  	"java:S5843", // Pattern is one allowlist grammar production; splitting it would scatter the validated set.
55  	"java:S5998", // Patterns are matched against short, author-set theme-token values, so backtracking depth is bounded.
56  	"java:S8786" // Input is a short, author-set theme-token value, not attacker-controlled request data; rewriting this allowlist regex risks altering the validated set.
57  })
58  final class CssValueGrammar {
59  
60  	private CssValueGrammar() {}
61  
62  	//-----------------------------------------------------------------------------------------------------------------
63  	// Normalization
64  	//-----------------------------------------------------------------------------------------------------------------
65  
66  	private static final Pattern COMMENT = Pattern.compile("/\\*.*?\\*/", Pattern.DOTALL);
67  	private static final Pattern URL_REJECT = Pattern.compile("(?i)url\\s*\\(");
68  
69  	/**
70  	 * Runs the normalization belt on a token value and returns the normalized (comment-stripped, trimmed) value,
71  	 * <b>without</b> the allowlist-grammar shape check.
72  	 *
73  	 * <p>
74  	 * This is the shared belt that both {@link #normalizeAndValidate(String)} and the {@code Theme}-layer
75  	 * {@code var(--jc-name)} reference recognizer (see {@link Theme.Builder#token(String, String)}) run on, so the
76  	 * reference recognizer sees exactly the same control-character-rejected, comment-stripped, {@code url(}-rejected
77  	 * string as the grammar does &mdash; there is no second, reference-specific normalization pass to keep in sync.
78  	 * The belt does <b>not</b> recognize or accept {@code var()}: that is Theme-layer syntax handled one layer above
79  	 * this class.
80  	 *
81  	 * @param value The raw candidate value.
82  	 * @return The normalized (comment-stripped, trimmed) value.
83  	 * @throws IllegalArgumentException If the value is <jk>null</jk>, contains a control character, or contains a
84  	 * 	{@code url(} production in any spelling.
85  	 */
86  	static String normalize(String value) {
87  		if (value == null)
88  			throw iaex("Theme token value must not be null.");
89  
90  		// Scan the RAW (pre-trim) value for control characters first - String.trim() strips any character
91  		// <= U+0020 (which covers all of C0) from the boundaries, so trimming first would silently swallow a
92  		// leading/trailing C0 control character before this check ever saw it.
93  		for (var i = 0; i < value.length(); i++) {
94  			var c = value.charAt(i);
95  			if (c <= 0x1F || (c >= 0x7F && c <= 0x9F))
96  				throw iaex("Theme token value contains an illegal control character at index %d.", i);
97  		}
98  
99  		var trimmed = value.trim();
100 		var stripped = COMMENT.matcher(trimmed).replaceAll("");
101 
102 		if (URL_REJECT.matcher(stripped).find())
103 			throw iaex("Theme token value must not contain a url() production.");
104 
105 		return stripped;
106 	}
107 
108 	/**
109 	 * Normalizes and validates a token value, returning the normalized (comment-stripped, trimmed) value if it is
110 	 * accepted.
111 	 *
112 	 * @param value The raw candidate value.
113 	 * @return The normalized value.
114 	 * @throws IllegalArgumentException If the value is <jk>null</jk>, contains a control character, contains a
115 	 * 	{@code url(} production in any spelling, or does not match one of the allowed CSS value shapes.
116 	 */
117 	static String normalizeAndValidate(String value) {
118 		var stripped = normalize(value);
119 
120 		if (isAllowedShape(stripped))
121 			return stripped;
122 
123 		throw iaex("Theme token value is not one of the allowed CSS value shapes: '%s'.", value);
124 	}

126 	//-----------------------------------------------------------------------------------------------------------------
127 	// Allowlist grammar
128 	//-----------------------------------------------------------------------------------------------------------------
129 
130 	// 3/4/6/8-digit hex color.  (Not 5 or 7 - those are not legal CSS hex-color lengths.)
131 	private static final Pattern HEX_COLOR =
132 		Pattern.compile("^#(?:[0-9a-fA-F]{3}|[0-9a-fA-F]{4}|[0-9a-fA-F]{6}|[0-9a-fA-F]{8})$");
133 
134 	// rgb()/rgba()/hsl()/hsla() with a numeric/%/,//whitespace/- argument charset only.
135 	private static final Pattern FUNCTIONAL_COLOR =
136 		Pattern.compile("^(?i:rgb|rgba|hsl|hsla)\\([0-9.,%\\s-]+\\)$");
137 
138 	// <number> + allowlisted unit, one or more space-separated (unit optional, e.g. bare "0").
139 	private static final String UNIT = "px|em|rem|%|vh|vw|vmin|vmax|pt|ch|fr|deg|turn|rad|s|ms";
140 	private static final Pattern LENGTH_LIST = Pattern.compile(
141 		"^-?\\d+(?:\\.\\d+)?(?:" + UNIT + ")?(?:\\s+-?\\d+(?:\\.\\d+)?(?:" + UNIT + ")?)*$");
142 
143 	// Optional inset, 2-4 lengths, then a color.  Color is checked against hex / functional / named-color
144 	// productions (not CSS-wide keywords), so this cannot smuggle url() or an arbitrary ident(.
145 	private static final Pattern BOX_SHADOW = Pattern.compile(
146 		"^(?:inset\\s+)?-?\\d+(?:\\.\\d+)?(?:" + UNIT + ")?(?:\\s+-?\\d+(?:\\.\\d+)?(?:" + UNIT + ")?){1,3}\\s+(.+)$");
147 
148 	private static final Set<String> NOT_SHADOW_COLORS = Set.of("none", "inherit", "initial", "unset");
149 
150 	// Comma-separated font-family list: each item a quoted string (no control/quote breakout) or a bare identifier.
151 	private static final Pattern FONT_FAMILY_ITEM_SQ = Pattern.compile("^'[^'\\\\]*'$");
152 	private static final Pattern FONT_FAMILY_ITEM_DQ = Pattern.compile("^\"[^\"\\\\]*\"$");
153 	private static final Pattern FONT_FAMILY_ITEM_BARE = Pattern.compile("^[a-zA-Z][a-zA-Z0-9 -]*$");
154 
155 	// Gradient function: name, then a nested-function-call allowlist scan, then a strict argument charset.
156 	private static final Pattern GRADIENT_OUTER =
157 		Pattern.compile("^(?i:repeating-)?(?i:linear|radial|conic)-gradient\\(.*\\)$", Pattern.DOTALL);
158 	private static final Pattern FUNCTION_CALL_TOKEN = Pattern.compile("([a-zA-Z][a-zA-Z-]*)\\(");
159 	private static final Set<String> GRADIENT_ALLOWED_NESTED_FUNCTIONS = Set.of("rgb", "rgba", "hsl", "hsla", "calc");
160 	private static final Pattern GRADIENT_ARG_CHARSET = Pattern.compile("^[0-9a-zA-Z%#.,\\s()-]*$");
161 
162 	private static final Set<String> NAMED_COLORS_AND_KEYWORDS = namedColorsAndKeywords();
163 
164 	private static boolean isAllowedShape(String v) {
165 		return HEX_COLOR.matcher(v).matches()
166 			|| FUNCTIONAL_COLOR.matcher(v).matches()
167 			|| NAMED_COLORS_AND_KEYWORDS.contains(v.toLowerCase(Locale.ROOT))
168 			|| LENGTH_LIST.matcher(v).matches()
169 			|| isFontFamilyList(v)
170 			|| isGradient(v)
171 			|| isBoxShadow(v);
172 	}
173 
174 	private static boolean isBoxShadow(String v) {
175 		var m = BOX_SHADOW.matcher(v);
176 		if (! m.matches())
177 			return false;
178 		var color = m.group(1);
179 		if (HEX_COLOR.matcher(color).matches() || FUNCTIONAL_COLOR.matcher(color).matches())
180 			return true;
181 		var lower = color.toLowerCase(Locale.ROOT);
182 		return NAMED_COLORS_AND_KEYWORDS.contains(lower) && ! NOT_SHADOW_COLORS.contains(lower);
183 	}
184 
185 	private static boolean isFontFamilyList(String v) {
186 		var items = v.split(",", -1);
187 		if (items.length == 0)
188 			return false;
189 		for (var item : items) {
190 			var t = item.trim();
191 			if (t.isEmpty())
192 				return false;
193 			if (! (FONT_FAMILY_ITEM_SQ.matcher(t).matches() || FONT_FAMILY_ITEM_DQ.matcher(t).matches() || FONT_FAMILY_ITEM_BARE.matcher(t).matches()))
194 				return false;
195 		}
196 		return true;
197 	}
198 
199 	private static boolean isGradient(String v) {
200 		if (! GRADIENT_OUTER.matcher(v).matches())
201 			return false;
202 		var inner = v.substring(v.indexOf('(') + 1, v.length() - 1);
203 		var m = FUNCTION_CALL_TOKEN.matcher(inner);
204 		while (m.find()) {
205 			var fn = m.group(1).toLowerCase(Locale.ROOT);
206 			if (! GRADIENT_ALLOWED_NESTED_FUNCTIONS.contains(fn))
207 				return false;
208 		}
209 		return GRADIENT_ARG_CHARSET.matcher(inner).matches();
210 	}
