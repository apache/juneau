17  package org.apache.juneau.rest.server.console;
18  
19  import java.util.*;
20  import java.util.regex.*;
21  
22  /**
23   * Builds framework topic names so callers do not hand-format "family:key" strings.
24   *
25   * <h5 class='section'>Example:</h5>
26   * <p class='bjava'>
27   *   Topics.<jsm>selection</jsm>(<js>"changes"</js>);   <jc>// "selection:changes"</jc>
28   *   Topics.<jsm>cmd</jsm>(<js>"tasks"</js>);           <jc>// "cmd:tasks"</jc>
29   *   Topics.<jsm>job</jsm>(<jv>ref</jv>.id());          <jc>// "job:j-91"</jc>
30   * </p>
31   *
32   * @since 10.0.0
33   */
34  @SuppressWarnings({
35  	"java:S1192" // Repeated literals are contract wire keys; named constants would obscure the contract shape.
36  })
37  public final class Topics {
38  
39  	/** The framework families, in the order {@code JuneauViews.bus.FRAMEWORK_FAMILIES} lists them (pinned by a test). */
40  	public static final Set<String> FRAMEWORK_FAMILIES = Collections.unmodifiableSet(new LinkedHashSet<>(List.of(
41  		"card", "selection", "filter", "redraw", "detail", "bulk", "cmd", "probe", "job", "badge", "bridge", "data", "zoom", "legend")));
42  
43  	// The grammar.  One copy: Subscription, TopicDecl, BridgeDecl and BusWiringValidator read these, and
44  	// Topics_Test pins them against juneau-page.schema.json.
45  	private static final String CUSTOM = "[a-z][a-z0-9-]{0,31}\\.[a-z][a-z0-9-]{0,31}";
46  	private static final String KEY = "[A-Za-z0-9_.-]{1,128}";
47  	static final Pattern TOPIC = Pattern.compile("^(" + String.join("|", FRAMEWORK_FAMILIES) + "|" + CUSTOM + ")(:" + KEY + ")?$");
48  	static final Pattern TOPIC_DECL = Pattern.compile("^(" + CUSTOM + "(:(" + KEY + "|\\*))?|job:\\*|data:" + KEY + ")$");
49  	static final Pattern PUBLICATION = Pattern.compile("^(" + CUSTOM + "(:(" + KEY + "|\\*))?|(zoom|legend):" + KEY + ")$");
50  	/** {@link #PUBLICATION} without the chart state families: what BusWiringValidator treats as a custom declaration. */
51  	static final Pattern CUSTOM_PUBLICATION = Pattern.compile("^" + CUSTOM + "(:(" + KEY + "|\\*))?$");
52  	/** Framework families a declaration may name (with a concrete key): {@code data} page-level, {@code zoom}/{@code legend} on a card. */
53  	static final Set<String> DECLARABLE_FRAMEWORK = Set.of("data", "zoom", "legend");
54  	static final Pattern ROLE = Pattern.compile("^[a-z][a-z0-9-]{0,31}$");
55  	static final Pattern MAP_PATH = Pattern.compile("^[A-Za-z0-9_-]++(?:\\.[A-Za-z0-9_-]++)*+$");
56  	/** Builder-time bridge session: same-origin path (no {@code //} or {@code /\}) or a {@code servlet:} / {@code context:} URI resolved at render. */
57  	static final Pattern SESSION = Pattern.compile("^(/(?![/\\\\])|servlet:/|context:/)\\S*$");
58  	/** The card and dashboard-section id grammar ({@code $defs/id} in the page schema). */
59  	public static final Pattern ID = Pattern.compile("^[A-Za-z][A-Za-z0-9_-]{0,63}$");
60  	private static final Pattern KEY_ONLY = Pattern.compile("^" + KEY + "$");
61  
62  	static final String E40 = "invalid topic '%s': expected family[:key] (see the topic syntax)";
63  	static final String E45_FRAMEWORK = "'%s' is a framework family; framework topics are implicit and may not be declared";
64  	static final String E45_NAMESPACE = "custom topic family '%s' must be namespaced (e.g. 'app.%s')";
65  
66  	private Topics() {}
67  
68  	/** @param cardId The card id. @return {@code card:<cardId>}. */
69  	public static String card(String cardId) { return framework("card", cardId); }
70  	/** @param cardId The card id. @return {@code selection:<cardId>}. */
71  	public static String selection(String cardId) { return framework("selection", cardId); }
72  	/** @param cardId The card id. @return {@code filter:<cardId>}. */
73  	public static String filter(String cardId) { return framework("filter", cardId); }
74  	/** @param cardId The card id. @return {@code redraw:<cardId>}. */
75  	public static String redraw(String cardId) { return framework("redraw", cardId); }
76  	/** @param cardId The card id. @return {@code detail:<cardId>}. */
77  	public static String detail(String cardId) { return framework("detail", cardId); }
78  	/** @param cardId The card id. @return {@code bulk:<cardId>}. */
79  	public static String bulk(String cardId) { return framework("bulk", cardId); }
80  	/** @param cardId The card id. @return {@code cmd:<cardId>}. */
81  	public static String cmd(String cardId) { return framework("cmd", cardId); }
82  	/** @param groupId The probe group id. @return {@code probe:<groupId>}. */
83  	public static String probe(String groupId) { return framework("probe", groupId); }
84  	/** @param jobId The async job id. @return {@code job:<jobId>}. */
85  	public static String job(String jobId) { return framework("job", jobId); }
86  	/** @param badgeId The badge id. @return {@code badge:<badgeId>}. */
87  	public static String badge(String badgeId) { return framework("badge", badgeId); }
88  	/** @param bridgeId The bridge id. @return {@code bridge:<bridgeId>}. */
89  	public static String bridge(String bridgeId) { return framework("bridge", bridgeId); }
90  	/** @param source The data source name. @return {@code data:<source>}. */
91  	public static String data(String source) { return framework("data", source); }
92  	/** @param cardId The chart card id. @return {@code zoom:<cardId>}. */
93  	public static String zoom(String cardId) { return framework("zoom", cardId); }
94  	/** @param cardId The chart card id. @return {@code legend:<cardId>}. */
95  	public static String legend(String cardId) { return framework("legend", cardId); }
96  
97  	/**
98  	 * @param ns The namespace, for example {@code ssc}.
99  	 * @param name The name within it, for example {@code focus}.
100 	 * @return {@code ns.name}.
101 	 * @throws IllegalArgumentException E-45 when {@code ns} is blank; E-40 when a part is malformed.
102 	 */
103 	public static String custom(String ns, String name) {
104 		if (ns == null || ns.isBlank())
105 			throw iae(E45_NAMESPACE, name, name);
106 		var t = ns + "." + name;
107 		if (! TOPIC_DECL.matcher(t).matches())
108 			throw iae(E40, t);
109 		return t;
110 	}
111 
112 	/**
113 	 * @param ns The namespace.
114 	 * @param name The name within it.
115 	 * @param key The key, or {@code *} for a declaration pattern.
116 	 * @return {@code ns.name:key}.
117 	 * @throws IllegalArgumentException E-45 / E-40 as for {@link #custom(String, String)}, or E-40 on a bad key.
118 	 */
119 	public static String custom(String ns, String name, String key) {
120 		var t = custom(ns, name) + ":" + key;
121 		if (! TOPIC_DECL.matcher(t).matches())
122 			throw iae(E40, t);
123 		return t;
124 	}
125 
126 	/**
127 	 * Tests the topic-key grammar (the part after {@code family:}).
128 	 *
129 	 * @param key The key, or <jk>null</jk>.
130 	 * @return <jk>true</jk> if {@code key} is a valid topic key.
131 	 */
132 	public static boolean isKey(String key) {
133 		return key != null && KEY_ONLY.matcher(key).matches();
134 	}
135 
136 	private static String framework(String family, String key) {
137 		var t = family + ":" + key;
138 		if (key == null || ! KEY_ONLY.matcher(key).matches())
139 			throw iae(E40, t);
140 		return t;
141 	}
142 
143 	/** Throws E-40 unless {@code topic} is a concrete topic. */
144 	static void checkTopic(String topic) {
145 		if (topic == null || ! TOPIC.matcher(topic).matches())
146 			throw iae(E40, topic);
147 	}
148 
149 	/** @return The text before the first {@code :}, or the whole topic. */
150 	static String family(String topic) {
151 		var i = topic.indexOf(':');
152 		return i < 0 ? topic : topic.substring(0, i);
153 	}
154 
155 	static IllegalArgumentException iae(String format, Object... args) {
156 		return new IllegalArgumentException(String.format(format, args));
157 	}
158 }
