17  package org.apache.juneau.rest.server.console;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.marshall.collections.*;
22  
23  /**
24   * The built-in {@code html} card type: markup-only (JSON5 bodies are E-25), with an optional body auto-captured
25   * as a {@code <template data-card="{id}">}. A bare {@code html} card (no body) emits an empty fragment — the
26   * shell paints an empty {@code .jc-card} and nothing more.
27   *
28   * <h5 class='section'>Trust boundary for {@code src}:</h5>
29   * <p>
30   * An {@code html} card in a page contract may instead name a {@code src} endpoint. The {@code src} must be a
31   * same-origin path (checked at page build, by {@code PageCapture.addCard} for FreeMarker and {@code PageSpec} cards
32   * and by {@code CardTypeRegistry.toCard} for lifted ones; the shell checks it again before the fetch). The shell
33   * fetches it and inserts the returned HTML as-is, <b>unsanitized</b>, the same trust level as
34   * {@code footer.text}. The endpoint must therefore be a trusted first-party route, must never be derived from request
35   * data, and must escape any data it embeds in the fragment.
36   *
37   * <p>
38   * A card that cannot guarantee that can opt in with {@code "sanitize": true} (or {@code sanitize=true} on
39   * {@code <@card>}, or {@code CardSpec.sanitize(true)}). The shell then rebuilds the fetched fragment through the views
40   * attribute-allowlist sanitizer, the one behind the {@code SANITIZED_HTML} detail format, so script, {@code on*}
41   * handlers, unsafe URLs and unlisted attributes are dropped. The page must load the views toolkit; without it the
42   * card fails closed with E-JS-8 and paints nothing. The flag only affects {@code src} cards, and the default is
43   * unchanged (unsanitized).
44   *
45   * <h5 class='section'>Example:</h5>
46   * <p class='bjava'>
47   *   CardSource <jv>s</jv> = CardSource.<jsm>create</jsm>(<js>"html"</js>, <js>"welcome"</js>)
48   *     .title(<js>"Welcome"</js>)
49   *     .body(<js>"&lt;p&gt;Hello!&lt;/p&gt;"</js>)
50   *     .templateSink(<jv>page</jv>::captureTemplate)
51   *     .build();
52   *   JsonMap <jv>frag</jv> = <jk>new</jk> HtmlCardType().toFragment(<jv>s</jv>);  <jc>// {"template":"welcome"}</jc>
53   * </p>
54   *
55   * @since 10.0.0
56   */
57  public final class HtmlCardType implements CardTypeHandler {
58  
59  	@Override
60  	public String type() {
61  		return "html";
62  	}
63  
64  	/** An html card is markup: it takes only the shell's {@code refresh} / {@code params} roles. */
65  	@Override
66  	public Set<String> acceptedRoles() {
67  		return Set.of();
68  	}
69  
70  	/** An html card handles only the shell's {@code refresh} op. */
71  	@Override
72  	public Set<String> acceptedOps() {
73  		return Set.of();
74  	}
75  
76  	@Override
77  	public JsonMap toFragment(CardSource source) {
78  		if (source.hasJsonBody()) {
79  			var body = source.body();
80  			var shortBody = body != null && body.length() > 40 ? body.substring(0, 40) : body;
81  			var preview = body == null ? "" : shortBody;
82  			throw source.error("type='%s' requires a %s body; got '%s'.", "html", "markup", preview);
83  		}
84  		var frag = new JsonMap();
85  		if (source.body() != null && ! source.body().isEmpty())
86  			frag.put("template", source.captureTemplate());
87  		return frag;
88  	}
89  }
