29  /**
30   * Reflection-based adapter that bridges legacy {@code org.apache.http.HttpResponse} (Apache HttpClient 4.5)
31   * objects to the modern {@link HttpResponseMessage} contract.
32   *
33   * <p>
34   * This adapter exists so that user code returning legacy {@code org.apache.http.*} response types from
35   * {@code @RestGet} methods continues to render correctly through the response-processor chain even though
36   * {@code juneau-rest-server} no longer compiles against Apache HttpClient. The adapter uses pure
37   * reflection — no compile-time dependency on the Apache HttpClient types — so this module remains
38   * dependency-free against {@code org.apache.httpcomponents}.
39   *
40   * <p>
41   * Detection: an object is considered a "legacy HTTP response" if it implements
42   * {@code org.apache.http.HttpResponse}. Detection caches the loaded class once per VM.
43   */
44  @SuppressWarnings({
45  	"java:S1192" // Duplicate string literals are HTTP header names and MIME type strings; intentional
46  })
47  final class LegacyHttpResponseAdapter {
48  
49  	private static final Class<?> LEGACY_HTTP_RESPONSE = tryLoad("org.apache.http.HttpResponse");
50  	private static final Class<?> LEGACY_HTTP_ENTITY = tryLoad("org.apache.http.HttpEntity");
51  	private static final Class<?> LEGACY_HTTP_RESOURCE = tryLoad("org.apache.juneau.http.classic.resource.HttpResource");
52  
53  	private LegacyHttpResponseAdapter() {}
54  
55  	/**
56  	 * Returns <jk>true</jk> if the given object is an instance of the legacy
57  	 * {@code org.apache.http.HttpResponse} interface, indicating it should be adapted.
58  	 *
59  	 * @param obj The candidate object. May be <jk>null</jk>.
60  	 * @return <jk>true</jk> if adaptation is needed, otherwise <jk>false</jk>.
61  	 */
62  	static boolean isLegacyResponse(Object obj) {
63  		return obj != null && LEGACY_HTTP_RESPONSE != null && LEGACY_HTTP_RESPONSE.isInstance(obj);
64  	}
65  
66  	/**
67  	 * Returns <jk>true</jk> if the given object is an instance of the legacy
68  	 * {@code org.apache.http.HttpEntity} interface.
69  	 *
70  	 * @param obj The candidate object. May be <jk>null</jk>.
71  	 * @return <jk>true</jk> if the object is a legacy entity, otherwise <jk>false</jk>.
72  	 */
73  	static boolean isLegacyEntity(Object obj) {
74  		return obj != null && LEGACY_HTTP_ENTITY != null && LEGACY_HTTP_ENTITY.isInstance(obj);
75  	}
76  
77  	/**
78  	 * Adapts a legacy {@code org.apache.http.HttpResponse} into a {@link HttpResponseMessage}.
79  	 *
80  	 * @param legacy The legacy response. Must be an instance of {@code org.apache.http.HttpResponse}.
81  	 * @return A {@link HttpResponseMessage} view over the legacy object.
82  	 */
83  	static HttpResponseMessage adapt(Object legacy) {
84  		return new LegacyAdapter(legacy);
85  	}
86  
87  	/**
88  	 * Adapts a legacy {@code org.apache.http.HttpEntity} into an {@link HttpBody}.
89  	 *
90  	 * @param legacyEntity The legacy entity. Must be an instance of {@code org.apache.http.HttpEntity}.
91  	 * @return An {@link HttpBody} view over the legacy entity.
92  	 */
93  	static HttpBody adaptEntity(Object legacyEntity) {
94  		return new LegacyEntityBody(legacyEntity);
95  	}
96  
97  	/**
98  	 * If the given object is a legacy {@code org.apache.juneau.http.classic.resource.HttpResource}, copies
99  	 * each header onto the supplied consumer.
100 	 *
101 	 * @param legacyEntity The legacy entity. May be <jk>null</jk>.
102 	 * @param sink Receives each adapted {@link HttpHeader}. Must not be <jk>null</jk>.
103 	 */
104 	static void copyResourceHeaders(Object legacyEntity, java.util.function.Consumer<HttpHeader> sink) {
105 		if (legacyEntity == null || LEGACY_HTTP_RESOURCE == null || ! LEGACY_HTTP_RESOURCE.isInstance(legacyEntity))
106 			return;
107 		var hl = invoke(legacyEntity, "getHeaders", new Class<?>[0]);
108 		if (hl == null)
109 			return;
110 		Iterable<?> iter;
111 		if (hl instanceof Iterable<?> hl2) {
112 			iter = hl2;
113 		} else {
114 			var arr = (Object[]) invoke(hl, "getAll", new Class<?>[0]);
115 			if (arr == null)
116 				return;
117 			iter = Arrays.asList(arr);
118 		}
119 		for (var h : iter) {
120 			if (h == null)
121 				continue;
122 			var name = (String) invoke(h, "getName", new Class<?>[0]);
123 			var value = (String) invoke(h, "getValue", new Class<?>[0]);
124 			sink.accept(HttpStringHeader.of(name, value));
125 		}
126 	}
127 
128 	private static Class<?> tryLoad(String name) {
129 		try {
130 			return Class.forName(name);
131 		} catch (ClassNotFoundException e) {
132 			return null;
133 		}
134 	}
135 
136 	private static Object invoke(Object target, String method, Class<?>[] argTypes, Object...args) {
137 		try {
138 			var m = target.getClass().getMethod(method, argTypes);
139 			return m.invoke(target, args);
140 		} catch (NoSuchMethodException | IllegalAccessException | InvocationTargetException e) {
141 			throw isex(e, "Failed to invoke %s on %s", method, cn(target));
142 		}
143 	}

146 	// Inner adapter types.
147 	// -----------------------------------------------------------------------------------------------
148 
149 	private static final class LegacyAdapter implements HttpResponseMessage {
150 
151 		private final Object legacy;
152 
153 		LegacyAdapter(Object legacy) {
154 			this.legacy = legacy;
155 		}
156 
157 		@Override
158 		public HttpStatusLine getStatusLine() {
159 			var sl = invoke(legacy, "getStatusLine", new Class<?>[0]);
160 			if (sl == null)
161 				return HttpStatusLineBean.of(200, "OK");
162 			var statusCode = (int) invoke(sl, "getStatusCode", new Class<?>[0]);
163 			var reasonPhrase = (String) invoke(sl, "getReasonPhrase", new Class<?>[0]);
164 			return HttpStatusLineBean.of(statusCode, reasonPhrase);
165 		}
166 
167 		@Override
168 		public List<HttpHeader> getHeaders() {
169 			var arr = (Object[]) invoke(legacy, "getAllHeaders", new Class<?>[0]);
170 			if (arr == null || arr.length == 0)
171 				return List.of();
172 			var out = new ArrayList<HttpHeader>(arr.length);
173 			for (var h : arr) {
174 				var name = (String) invoke(h, "getName", new Class<?>[0]);
175 				var value = (String) invoke(h, "getValue", new Class<?>[0]);
176 				out.add(isUriHeader(name) ? HttpUriHeader.of(name, value) : HttpStringHeader.of(name, value));
177 			}
178 			return out;
179 		}
180 
181 		private static boolean isUriHeader(String name) {
182 			return name != null && (
183 				name.equalsIgnoreCase("Location")
184 				|| name.equalsIgnoreCase("Content-Location")
185 				|| name.equalsIgnoreCase("Referer"));
186 		}
187 
188 		@Override
189 		public HttpBody getBody() {
190 			var entity = invoke(legacy, "getEntity", new Class<?>[0]);
191 			if (entity == null)
192 				return null;
193 			return new LegacyEntityBody(entity);
194 		}
195 	}
196 
197 	private static final class LegacyEntityBody implements HttpBody {
198 
199 		private final Object entity;
200 
201 		LegacyEntityBody(Object entity) {
202 			this.entity = entity;
203 		}
204 
205 		@Override
206 		public String getContentType() {
207 			var h = invoke(entity, "getContentType", new Class<?>[0]);
208 			return h == null ? null : (String) invoke(h, "getValue", new Class<?>[0]);
209 		}
210 
211 		@Override
212 		public long getContentLength() {
213 			var v = invoke(entity, "getContentLength", new Class<?>[0]);
214 			return v == null ? -1 : ((Number) v).longValue();
215 		}
216 
217 		@Override
218 		public void writeTo(OutputStream os) throws IOException {
219 			try {
220 				var m = entity.getClass().getMethod("writeTo", OutputStream.class);
221 				m.invoke(entity, os);
222 			} catch (NoSuchMethodException | IllegalAccessException e) {
223 				throw new IOException("Cannot invoke writeTo on legacy entity", e);
224 			} catch (InvocationTargetException e) {
225 				var t = e.getCause();
226 				if (t instanceof IOException t2) throw t2;
227 				if (t instanceof RuntimeException t2) throw t2;
228 				throw new IOException(t);
229 			}
230 		}
231 	}
