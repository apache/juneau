32  /**
33   * Response processor that serializes
34   * <a class="doclink" href="https://www.rfc-editor.org/rfc/rfc7807">RFC 7807 / 9457</a> {@link Problem} payloads as
35   * {@code application/problem+json}.
36   *
37   * <p>
38   * Fires when the response content is one of the following (in order):
39   * <ul class='spaced-list'>
40   * 	<li>A {@link Problem} bean &mdash; serialized directly. Honors the client {@code Accept} header.
41   * 	<li>A {@link ProblemException} &mdash; unwrapped via {@link ProblemException#getProblem()}. Honors the
42   * 		client {@code Accept} header.
43   * 	<li>A {@link BasicHttpException} &mdash; adapted via a registered {@link ProblemMapper} (if any matches),
44   * 		else via {@link ProblemAdapters#fromException(BasicHttpException)}. This branch handles the common throw
45   * 		path where the operation handler throws a {@code BasicHttpException} and the framework sets it as the
46   * 		response content. <b>Gated by {@code isProblemDetails()}</b> on the {@link RestOpContext} (per-op
47   * 		{@code @RestGet(problemDetails)} / {@code @RestPost(problemDetails)} / etc., falling back to the
48   * 		resource-level {@code @Rest(problemDetails)} opt-in via tri-state inheritance). When opted-in, the
49   * 		{@code Accept} header is ignored on this branch (errors always emit {@code application/problem+json}; see
50   * 		RFC 7807).
51   * 	<li>Any other {@link Throwable} &mdash; only when an explicit {@link ProblemMapper} matches its class
52   * 		hierarchy. Gated by the same opt-in. Mappers that return {@code null} are skipped and the chain
53   * 		continues; if no mapper matches, the processor returns {@code NEXT}.
54   * </ul>
55   *
56   * <h5 class='section'>Mapper discovery</h5>
57   * <p>
58   * The processor pulls registered {@link ProblemMapper}s from the resource's bean store in this order:
59   * <ul class='spaced-list'>
60   * 	<li>A {@link ProblemMapperList} bean &mdash; treated as the authoritative ordered registry. Recommended
61   * 		when more than one mapper is registered, since the underlying {@code @Bean} bean-store walk pairs each
62   * 		factory method with its return type and so collapses multiple
63   * 		{@code @Bean public ProblemMapper foo()} entries on the same resource into a single slot.
64   * 	<li>A single {@link ProblemMapper} bean (via {@code BeanStore.getBean(ProblemMapper.class)}) &mdash; the
65   * 		conventional single-mapper registration pattern.
66   * </ul>
67   * <p>
68   * For a given thrown exception the processor walks the collected mappers and picks the one whose
69   * {@link ProblemMapper#getExceptionType()} sits closest to the thrown class in the hierarchy (most-specific
70   * wins; ties broken by registration order). The chosen mapper's {@link ProblemMapper#map(Throwable)} return
71   * value replaces the default adaptation; a {@code null} return falls through to the next-most-specific mapper,
72   * and ultimately to the built-in {@link ProblemAdapters#fromException(BasicHttpException)} fallback for
73   * {@link BasicHttpException} subclasses.
74   *
75   * <h5 class='section'>Localization seam</h5>
76   * <p>
77   * Before serializing, the processor consults the bean store for a {@link ProblemLocalizationStrategy} bean and
78   * passes the resolved {@link Problem} (and {@code RestRequest.getLocale()}) through it. The default behavior is
79   * {@link ProblemLocalizationStrategy#IDENTITY} (pass-through) &mdash; this is a future-work seam intentionally
80   * shipped without a built-in {@code Messages}/resource-bundle implementation. A {@code null} strategy return is
81   * treated as &quot;no opinion&quot; and the pre-localization {@link Problem} is used instead.
82   *
83   * <p>
84   * <b>Status code policy.</b> When {@link Problem#getStatus()} is non-{@code null}, the processor calls
85   * {@code RestResponse.setStatus(int)} with that value. Otherwise it leaves the existing status alone &mdash;
86   * {@code RestSession.run()} normalizes {@code 0} to {@code 200} so per-op default codes (e.g. {@code @RestPost}
87   * default of 200) still flow through.
88   *
89   * <p>
90   * <b>{@code type} field policy.</b> Serialized as-is. A {@code null} {@code type} is omitted from the JSON output
91   * rather than synthesized as {@code "type":"about:blank"} on the wire (preserves the absent-vs-explicit distinction).
92   *
93   * <h5 class='section'>See Also:</h5><ul>
94   * 	<li class='jc'>{@link Problem}
95   * 	<li class='jc'>{@link ProblemException}
96   * 	<li class='jc'>{@link ProblemMapper}
97   * 	<li class='jc'>{@link ProblemLocalizationStrategy}
98   * 	<li class='jc'>{@link ProblemAdapters}
99   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/ResponseProcessors">Response Processors</a>
100  * </ul>
101  */
102 public class ProblemDetailsProcessor implements ResponseProcessor {
103 
104 	private static final String PROBLEM_JSON = ContentType.APPLICATION_PROBLEM_JSON.getValue();
105 
106 	@Override /* Overridden from ResponseProcessor */
107 	@SuppressWarnings({
108 		"resource",  // negotiated output stream owned by the response; closed by the container/RestCall
109 		"java:S3776" // Cognitive complexity acceptable for RFC-9457 problem-details response dispatch
110 	})
111 	public int process(RestOpSession opSession) throws IOException {
112 
113 		var res = opSession.getResponse();
114 		var raw = res.getContent(Object.class);
115 
116 		if (raw == null)
117 			return NEXT;
118 
119 		Problem problem;
120 		boolean isErrorPath;
121 
122 		if (raw instanceof Problem raw2) {
123 			problem = raw2;
124 			isErrorPath = false;
125 		} else if (raw instanceof ProblemException raw2) {
126 			problem = raw2.getProblem();
127 			isErrorPath = opSession.getContext().isProblemDetails();
128 		} else if (raw instanceof BasicHttpException raw2) {
129 			if (! opSession.getContext().isProblemDetails())
130 				return NEXT;
131 			problem = mapException(opSession, raw2);
132 			isErrorPath = true;
133 		} else if (raw instanceof Throwable raw2) {
134 			if (! opSession.getContext().isProblemDetails())
135 				return NEXT;
136 			problem = mapException(opSession, raw2);
137 			if (problem == null)
138 				return NEXT;
139 			isErrorPath = true;
140 		} else {
141 			return NEXT;
142 		}
143 
144 		if (problem == null)
145 			return NEXT;
146 
147 		// Q5: error path ignores Accept (RFC 7807 §3); success path honors Accept strictly.
148 		if (! isErrorPath) {
149 			var accept = opSession.getRequest().getHeaderParam("Accept").orElse("");
150 			if (! acceptsProblemJson(accept))
151 				return NEXT;
152 		}
153 
154 		problem = localize(opSession, problem);
155 
156 		var status = problem.getStatus();
157 		if (status != null)
158 			res.setStatus(status.intValue());
159 
160 		res.setHeader(ContentType.APPLICATION_PROBLEM_JSON);
161 
162 		try {
163 			var os = res.getNegotiatedOutputStream();
164 			Json.DEFAULT.write(problem, os);
165 			os.flush();
166 			os.finish();
167 		} catch (SerializeException e) {
168 			throw new IOException("Failed to serialize Problem payload", e);
169 		}
170 
171 		return FINISHED;
172 	}
173 
174 	/**
175 	 * Resolves a {@link Problem} from a thrown exception by consulting registered {@link ProblemMapper}s in
176 	 * most-specific-first order, falling back to {@link ProblemAdapters#fromException(BasicHttpException)} for
177 	 * {@link BasicHttpException} subclasses when no mapper matches (or all matching mappers return {@code null}).
178 	 *
179 	 * @return The {@link Problem} to emit, or {@code null} when no mapper opined and the exception is not a
180 	 * 	{@link BasicHttpException}.
181 	 */
182 	@SuppressWarnings({
183 		"unchecked", // ProblemMapper#map is parameterized over the matched throwable subtype.
184 		"rawtypes"   // ProblemMapper#map is parameterized over the matched throwable subtype.
185 	})
186 	private static Problem mapException(RestOpSession opSession, Throwable thrown) {
187 		for (var mapper : resolveMappers(opSession, thrown.getClass())) {
188 			var result = ((ProblemMapper) mapper).map(thrown);
189 			if (result != null)
190 				return result;
191 		}
192 		if (thrown instanceof BasicHttpException thrown2)
193 			return ProblemAdapters.fromException(thrown2);
194 		return null;
195 	}
196 
197 	/**
198 	 * Returns all registered {@link ProblemMapper}s whose {@link ProblemMapper#getExceptionType()} is assignable
199 	 * from {@code thrownClass}, sorted most-specific-first. A {@link ProblemMapperList} bean (if present) is
200 	 * consulted first; a single {@link ProblemMapper} bean (if no list is registered) is consulted as a
201 	 * fallback. Registration order serves as the tiebreaker for mappers at the same hierarchy depth.
202 	 */
203 	@SuppressWarnings({
204 		"resource" // RestOpSession owns this bean store lifecycle.
205 	})
206 	private static List<ProblemMapper<?>> resolveMappers(RestOpSession opSession, Class<?> thrownClass) {
207 		var bs = opSession.getBeanStore();
208 		var candidates = new ArrayList<ProblemMapper<?>>();
209 		bs.getBean(ProblemMapperList.class).map(ProblemMapperList::asList).ifPresent(candidates::addAll);
210 		if (candidates.isEmpty())
211 			bs.getBean(ProblemMapper.class).ifPresent(candidates::add);
212 		if (candidates.isEmpty())
213 			return List.of();
214 		var matches = new ArrayList<ProblemMapper<?>>(candidates.size());
215 		for (var m : candidates) {
216 			if (m.getExceptionType().isAssignableFrom(thrownClass))
217 				matches.add(m);
218 		}
219 		matches.sort((a, b) -> hierarchyDepth(a.getExceptionType(), thrownClass) - hierarchyDepth(b.getExceptionType(), thrownClass));
220 		return matches;
221 	}
222 
223 	/**
224 	 * Returns the number of superclass hops from {@code thrownClass} up to {@code targetType}; lower is closer.
225 	 * A {@code 0} return means {@code targetType} equals {@code thrownClass} (exact match); larger values mean
226 	 * the mapper sits higher up the class hierarchy and is therefore less specific. Sorting ascending by this
227 	 * value puts the most-specific mapper first.
228 	 */
229 	private static int hierarchyDepth(Class<?> targetType, Class<?> thrownClass) {
230 		var depth = 0;
231 		var c = thrownClass;
232 		while (c != null && c != targetType) {
233 			c = c.getSuperclass();
234 			depth++;
235 		}
236 		return c == null ? Integer.MAX_VALUE : depth;
237 	}
238 
239 	private static Problem localize(RestOpSession opSession, Problem problem) {
240 		@SuppressWarnings({
241 			"resource" // RestOpSession owns this bean store lifecycle.
242 		})
243 		var strategy = opSession.getBeanStore().getBean(ProblemLocalizationStrategy.class).orElse(ProblemLocalizationStrategy.IDENTITY);
244 		var locale = opSession.getRequest().getLocale();
245 		var result = strategy.localize(problem, locale);
246 		return result == null ? problem : result;
247 	}
248 
249 	private static boolean acceptsProblemJson(String accept) {
250 		if (isEmpty(accept))
251 			return true;
252 		return accept.contains(PROBLEM_JSON) || accept.contains("*/*");
253 	}
