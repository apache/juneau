19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import java.io.*;
22  import java.util.*;

24  import org.apache.juneau.http.header.*;
25  import org.apache.juneau.http.response.*;

30  /**
31   * Response handler for plain-old Java objects when a serializer match is not found and they're asking for plain/text or anything.
32   *
33   * <h5 class='section'>See Also:</h5><ul>
34   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/ResponseProcessors">Response Processors</a>
35   * </ul>
36   */
37  @SuppressWarnings({
38  	"resource" // PlainTextPojoProcessor manages Closeable resources
39  })
40  public class PlainTextPojoProcessor implements ResponseProcessor {
41  
42  	@Override /* Overridden from ResponseProcessor */
43  	public int process(RestOpSession opSession) throws IOException, NotAcceptable, BasicHttpException {
44  		RestRequest req = opSession.getRequest();
45  		RestResponse res = opSession.getResponse();
46  		String accept = req.getHeaderParam("Accept").orElse("*/*");
47  
48  		if (res.getSerializerMatch().isPresent() || ! (ie(accept) || accept.startsWith("text/plain") || accept.contains("*/*")))
49  			return NEXT;
50  
51  		Object o = res.getContent(Object.class);
52  
53  		if (ie(res.getContentType()))
54  			res.setHeader(ContentType.TEXT_PLAIN);
55  
56  		FinishablePrintWriter w = res.getNegotiatedWriter();
57  		if (o == null) {
58  			w.append("null");
59  		} else if (o instanceof Map || o instanceof Collection || o.getClass().isArray()) {
60  			w.append(Json5.of(o));
61  		} else if (o instanceof Throwable && ! opSession.getRestContext().isRenderResponseStackTraces()) {
62  			// A thrown exception reached the plain-text renderer as the response content.  Emitting its
63  			// toString() (class name + message) would leak internal detail to the client, so in production
64  			// we write only the generic HTTP status text.  Structured detail remains in the 'Thrown'
65  			// response header and server-side logs; enable renderResponseStackTraces for full in-response
66  			// detail during development.
67  			var statusText = RestUtils.getHttpResponseText(res.getStatus());
68  			w.append(statusText == null ? "" : statusText);
69  		} else {
70  			w.append(req.getMarshallingSession().getClassMetaForObject(o).toString(o));
71  		}
72  		w.flush();
73  		w.finish();
74  
75  		return FINISHED;
76  	}
77  }
