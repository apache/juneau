34  /**
35   * Response processor that unwraps {@link CompletableFuture} / {@link CompletionStage} return values from
36   * {@code @RestOp}-annotated methods and bridges them to the servlet container's
37   * {@link AsyncContext asynchronous} request lifecycle.
38   *
39   * <p>
40   * When a handler returns a future, this processor is the FIRST processor in the default chain
41   * (see {@link org.apache.juneau.rest.server.config.DefaultConfig}). It detects the {@link CompletionStage}
42   * content, calls {@link HttpServletRequest#startAsync()} on the underlying servlet request, and
43   * registers a {@link CompletionStage#whenComplete(java.util.function.BiConsumer) whenComplete}
44   * callback that:
45   *
46   * <ul>
47   * 	<li>On success — installs the unwrapped value via {@link RestResponse#setContent(Object)}, re-runs
48   * 		{@link RestContext#processResponse(RestOpSession)} (this processor sees a non-future and falls
49   * 		through to {@code SerializedPojoProcessor} et al.), flushes the response, and
50   * 		calls {@link AsyncContext#complete()}.
51   * 	<li>On failure — runs the throwable through {@link RestContext#convertThrowable(Throwable)} so
52   * 		the existing error pipeline (including {@link ThrowableProcessor} and
53   * 		{@link ProblemDetailsProcessor}) handles it, then completes the {@code AsyncContext}.
54   * </ul>
55   *
56   * <h5 class='section'>Synchronous fallback</h5>
57   * <p>
58   * In environments where {@link HttpServletRequest#startAsync()} is unsupported (most notably
59   * Juneau's {@code MockServletRequest}, which returns {@code null}), this processor falls back to a
60   * blocking {@link CompletableFuture#get(long, TimeUnit) get(timeout)} on the future and then
61   * returns {@link #RESTART} so the rest of the chain runs synchronously on the unwrapped value.
62   * This keeps the unit-test surface working without requiring a real servlet container while
63   * preserving full {@code AsyncContext} semantics in production.
64   *
65   * <h5 class='section'>Timeout</h5>
66   * <p>
67   * The async timeout is configurable via {@code @Rest(asyncTimeoutMillis)} /
68   * {@code @RestOp(asyncTimeoutMillis)} (default 30s). On timeout the processor cancels the future
69   * with {@code mayInterruptIfRunning=true} and writes a {@link HttpServletResponse#SC_GATEWAY_TIMEOUT}
70   * response.
71   *
72   * <h5 class='section'>Bare {@link Future} rejection</h5>
73   * <p>
74   * Bare {@link Future} return types (anything that is a {@code Future} but not a
75   * {@link CompletionStage}) are rejected with an {@link InternalServerError} —
76   * polling a {@code Future} would block the request thread without any of the cancellation /
77   * cooperation guarantees that {@link CompletableFuture} provides. Handlers that need async dispatch
78   * must return {@link CompletableFuture} or {@link CompletionStage}.
79   *
80   * <h5 class='section'>Completion executor</h5>
81   * <p>
82   * When {@code @Rest(asyncCompletionExecutor="poolName")} or the per-op equivalent is set, this processor
83   * switches from {@code future.whenComplete(callback)} to {@code future.whenCompleteAsync(callback, executor)},
84   * routing the response-handler work through the named {@link Executor} bean. The MDC
85   * snapshot (see below) is taken on the request thread <em>before</em> the executor routing so the MDC context
86   * is always available regardless of which thread the executor picks.
87   *
88   * <h5 class='section'>Thread-local caveats</h5>
89   * <p>
90   * Anything carried via {@link ThreadLocal} (SLF4J MDC, security contexts) does NOT survive the
91   * async boundary — the {@code whenComplete} callback typically runs on a different thread than the
92   * one that produced the future. {@code RequestAttributes}, {@link org.apache.juneau.commons.svl.VarResolverSession},
93   * and {@link java.util.Locale} are request-scoped and survive correctly.
94   *
95   * <h5 class='section'>See Also:</h5><ul>
96   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestServerAsync">Async Response Handling</a>
97   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/ResponseProcessors">Response Processors</a>
98   * </ul>
99   *
100  * @since 10.0.0
101  */
102 public class AsyncResponseProcessor implements ResponseProcessor {
103 
104 	private static final Logger LOG = Logger.getLogger(AsyncResponseProcessor.class.getName());

106 	/**
107 	 * Request attribute set by this processor when the {@code AsyncContext} path is taken so
108 	 * downstream lifecycle code (specifically {@link RestOpSession#finish()} and
109 	 * {@link RestSession#finish()}) can skip the synchronous {@code flushBuffer()} / {@code req.close()}
110 	 * — the {@code AsyncContext.complete()} call inside the {@code whenComplete} callback is
111 	 * responsible for committing the response.
112 	 *
113 	 * @since 10.0.0
114 	 */
115 	public static final String ATTR_ASYNC_DISPATCH_OWNED = "org.apache.juneau.rest.server.async.dispatchOwned";
116 
117 	/** Default async timeout in milliseconds when no annotation override is supplied. */
118 	public static final long DEFAULT_ASYNC_TIMEOUT_MILLIS = 30_000L;
119 
120 	@Override /* Overridden from ResponseProcessor */
121 	@SuppressWarnings({
122 		"java:S3776", // Async dispatch logic is inherently branchy — splitting further hurts readability.
123 		"java:S1141"  // Nested try/catch cleanly separates startAsync IllegalStateException recovery from cancellation.
124 	})
125 	public int process(RestOpSession opSession) throws IOException, BasicHttpException {
126 		var res = opSession.getResponse();
127 		var content = res.getContent().orElse(null);
128 
129 		if (content == null)
130 			return NEXT;
131 
132 		if (content instanceof CompletionStage<?> stage)
133 			return processAsync(opSession, stage);
134 
135 		if (content instanceof Future<?>) {
136 			throw new InternalServerError(
137 				"Bare java.util.concurrent.Future is not supported as a return type from @RestOp methods. "
138 					+ "Return CompletableFuture or CompletionStage instead, or block on the result yourself before returning."
139 			);
140 		}
141 
142 		return NEXT;
143 	}
144 
145 	private int processAsync(RestOpSession opSession, CompletionStage<?> stage) throws BasicHttpException {
146 		var timeoutMs = resolveTimeoutMillis(opSession);
147 		var cf = stage.toCompletableFuture();
148 		var req = opSession.getRequest().getHttpServletRequest();
149 
150 		// True async path requires both isAsyncSupported() and a non-null AsyncContext from startAsync().
151 		AsyncContext asyncCtx = null;
152 		if (req.isAsyncSupported()) {
153 			try {
154 				asyncCtx = req.startAsync();
155 				if (asyncCtx != null && timeoutMs > 0)
156 					asyncCtx.setTimeout(timeoutMs);
157 			} catch (IllegalStateException e) {
158 				asyncCtx = null;  // Already committed or async not actually supported on this request.
159 			}
160 		}
161 
162 		if (asyncCtx == null)
163 			return processSyncFallback(opSession, cf, timeoutMs);
164 
165 		dispatchAsync(opSession, cf, asyncCtx, timeoutMs);
166 		return FINISHED;
167 	}
168 
169 	@SuppressWarnings({
170 		"java:S2142", // We re-set the interrupt flag immediately and surface as a 500.
171 		"java:S1166", // CancellationException is intentionally swallowed; we surface it as a 500 via convertThrowable.
172 		"java:S3516"  // Always returns RESTART by design: every outcome (success or error) re-runs the chain on the now-synchronous content.
173 	})
174 	private int processSyncFallback(RestOpSession opSession, CompletableFuture<?> cf, long timeoutMs) {
175 		var res = opSession.getResponse();
176 		try {
177 			Object value = (timeoutMs > 0) ? cf.get(timeoutMs, TimeUnit.MILLISECONDS) : cf.get();
178 			res.setContent(value);
179 			return RESTART;
180 		} catch (TimeoutException te) {
181 			cf.cancel(true);
182 			res.setStatus(SC_GATEWAY_TIMEOUT);
183 			res.setContent(opSession.getRestContext().convertThrowable(
184 				new GatewayTimeout("Async response timed out after " + timeoutMs + "ms.")
185 			));
186 			return RESTART;
187 		} catch (CancellationException ce) {
188 			res.setStatus(SC_INTERNAL_SERVER_ERROR);
189 			res.setContent(opSession.getRestContext().convertThrowable(ce));
190 			return RESTART;
191 		} catch (ExecutionException ee) {
192 			Throwable cause = or(ee.getCause(), ee);
193 			res.setStatus(SC_INTERNAL_SERVER_ERROR);
194 			res.setContent(opSession.getRestContext().convertThrowable(cause));
195 			return RESTART;
196 		} catch (InterruptedException ie) {
197 			Thread.currentThread().interrupt();
198 			cf.cancel(true);
199 			res.setStatus(SC_INTERNAL_SERVER_ERROR);
200 			res.setContent(opSession.getRestContext().convertThrowable(ie));
201 			return RESTART;
202 		}
203 	}
204 
205 	private void dispatchAsync(RestOpSession opSession, CompletableFuture<?> cf, AsyncContext asyncCtx, long timeoutMs) {
206 		var req = opSession.getRequest().getHttpServletRequest();
207 		req.setAttribute(ATTR_ASYNC_DISPATCH_OWNED, Boolean.TRUE);
208 
209 		var done = new AtomicBoolean(false);
210 
211 		asyncCtx.addListener(new AsyncListener() {
212 			@Override public void onComplete(AsyncEvent ev) { /* no-op */ }
213 			@Override public void onStartAsync(AsyncEvent ev) { /* no-op */ }
214 			@Override public void onError(AsyncEvent ev) {
215 				finalizeAsync(opSession, cf, asyncCtx, ev.getThrowable(), done, /* timeout */ false, timeoutMs);
216 			}
217 			@Override public void onTimeout(AsyncEvent ev) {
218 				finalizeAsync(opSession, cf, asyncCtx, null, done, /* timeout */ true, timeoutMs);
219 			}
220 		});
221 
222 		// MDC bridge: snapshot request-thread MDC at dispatch time, then restore it on the
223 		// completion thread so log statements inside the whenComplete callback see the same diagnostic
224 		// context (requestId, userId, etc.) that was set by upstream filters.
225 		// NOTE: the MDC wrap happens BEFORE the executor routing so the snapshot is always
226 		// available regardless of which thread the executor picks.
227 		var mdcSnapshot = opSession.getRestContext().isMdcAsyncPropagation()
228 			? MdcAsyncListener.snapshot()
229 			: null;
230 
231 		var callback = MdcAsyncListener.wrap(
232 			(value, error) -> finalizeAsync(opSession, cf, asyncCtx, error, done, /* timeout */ false, timeoutMs, value),
233 			mdcSnapshot
234 		);
235 
236 		// If a per-op or resource-level completion executor is configured, route the callback
237 		// through it via whenCompleteAsync; otherwise use the natural completion thread.
238 		var executor = opSession.getContext().getAsyncCompletionExecutor();
239 		if (executor != null)
240 			cf.whenCompleteAsync(callback, executor);
241 		else
242 			cf.whenComplete(callback);
243 	}
244 
245 	private void finalizeAsync(RestOpSession opSession, CompletableFuture<?> cf, AsyncContext asyncCtx,
246 			Throwable error, AtomicBoolean done, boolean timeout, long timeoutMs) {
247 		finalizeAsync(opSession, cf, asyncCtx, error, done, timeout, timeoutMs, null);
248 	}
249 
250 	@SuppressWarnings({
251 		"java:S3776", // Async finalization is inherently branchy.
252 		"java:S1141", // Nested try/catch separates response-processing from container cleanup.
253 		"java:S107"   // The 8 parameters are the internal async-finalization context (session, future, async ctx, error, done-latch, timeout flag, timeout ms, resolved value); a holder object would obscure this hot-path call and its sibling overload.
254 	})
255 	private void finalizeAsync(RestOpSession opSession, CompletableFuture<?> cf, AsyncContext asyncCtx,
256 			Throwable error, AtomicBoolean done, boolean timeout, long timeoutMs, Object value) {
257 		if (! done.compareAndSet(false, true))
258 			return;
259 
260 		var res = opSession.getResponse();
261 
262 		try {
263 			if (timeout) {
264 				cf.cancel(true);
265 				res.setStatus(SC_GATEWAY_TIMEOUT);
266 				res.setContent(opSession.getRestContext().convertThrowable(
267 					new GatewayTimeout("Async response timed out after " + timeoutMs + "ms.")
268 				));
269 			} else if (error != null) {
270 				Throwable cause = unwrap(error);
271 				res.setStatus(SC_INTERNAL_SERVER_ERROR);
272 				res.setContent(opSession.getRestContext().convertThrowable(cause));
273 			} else {
274 				res.setContent(value);
275 			}
276 
277 			opSession.getRestContext().processResponse(opSession);
278 			res.flushBuffer();
279 		} catch (Exception e) {
280 			LOG.log(Level.WARNING, e, () -> "Async response finalization failed: " + e.getMessage());
281 			try {
282 				if (! res.getHttpServletResponse().isCommitted())
283 					res.getHttpServletResponse().sendError(SC_INTERNAL_SERVER_ERROR);
284 			} catch (Exception inner) {
285 				LOG.log(Level.FINEST, inner, () -> "Async response error-fallback also failed: " + inner.getMessage());
286 			}
287 		} finally {
288 			try {
289 				asyncCtx.complete();
290 			} catch (IllegalStateException ise) {
291 				LOG.log(Level.FINEST, ise, () -> "AsyncContext.complete() raced with the container: " + ise.getMessage());
292 			}
293 		}
294 	}
295 
296 	private static Throwable unwrap(Throwable t) {
297 		if (t instanceof CompletionException && t.getCause() != null)
298 			return t.getCause();
299 		if (t instanceof ExecutionException && t.getCause() != null)
300 			return t.getCause();
301 		return t;
302 	}
303 
304 	private static long resolveTimeoutMillis(RestOpSession opSession) {
305 		var op = opSession.getContext().getAsyncTimeoutMillis();
306 		if (op > 0)
307 			return op;
308 		var ctx = opSession.getRestContext().getAsyncTimeoutMillis();
309 		return ctx > 0 ? ctx : DEFAULT_ASYNC_TIMEOUT_MILLIS;
310 	}

312 	/**
313 	 * Returns whether the given session has been handed off to async dispatch. Inspected by
314 	 * {@link RestOpSession#finish()} / {@link RestSession#finish()} to skip the synchronous
315 	 * {@code flushBuffer()} call when the {@code AsyncContext.complete()} path will commit the
316 	 * response itself.
317 	 *
318 	 * @param opSession The session to check.
319 	 * 	<br>Can be <jk>null</jk> (returns <jk>false</jk>).
320 	 * @return {@code true} if {@link #ATTR_ASYNC_DISPATCH_OWNED} has been set.
321 	 */
322 	public static boolean isAsyncDispatchOwned(RestOpSession opSession) {
323 		if (opSession == null)
324 			return false;
325 		return isAsyncDispatchOwned(opSession.getRequest().getHttpServletRequest());
326 	}

328 	/**
329 	 * Variant of {@link #isAsyncDispatchOwned(RestOpSession)} that operates directly on the
330 	 * underlying servlet request — used by {@link RestSession#finish()} where the
331 	 * {@link RestOpSession} may not be available.
332 	 *
333 	 * @param req The servlet request.
334 	 * 	<br>Can be <jk>null</jk> (returns <jk>false</jk>).
335 	 * @return {@code true} if {@link #ATTR_ASYNC_DISPATCH_OWNED} has been set.
336 	 */
337 	public static boolean isAsyncDispatchOwned(HttpServletRequest req) {
338 		if (req == null)
339 			return false;
340 		return isTrue(req.getAttribute(ATTR_ASYNC_DISPATCH_OWNED));
341 	}
