37  /**
38   * Shared abstract base for the {@link RestBuilder} flavor builders ({@code RestServlet.Builder},
39   * {@code RestResource.Builder}, {@code RestMixin.Builder}).
40   *
41   * <p>
42   * Each fluent {@code @Rest}-member setter is implemented <b>once</b> here and forwards into a backing
43   * {@code RestAnnotation.Builder} &mdash; the override bag.  At {@link RestContext} construction time the bag is
44   * turned into a synthetic, highest-priority {@code @Rest} annotation that is prepended to the resource's
45   * {@code @Rest} chain, so builder-supplied values <b>take precedence</b> over the class's own
46   * {@link Rest @Rest} annotation values.
47   *
48   * <h5 class='section'>Self type (CRTP):</h5>
49   *
50   * <p>
51   * {@code SELF} is the concrete builder type, left open through the flavor builders and their user subclasses so
52   * that bespoke setters on a subclass chain with true covariant returns.  Setters return {@link #self()}.
53   *
54   * @param <R> The resource type produced by {@link #build()}.
55   * @param <SELF> The concrete builder type (self type).
56   * @since 10.0.0
57   */
58  @SuppressWarnings({
59  	"unchecked", // CRTP self-type cast in self() is safe by construction.
60  	"java:S1452", // Wildcard return on getResourceType() is intentional.
61  	"java:S119" // 'SELF' (CRTP self-type) is intentional and clearer than a single-letter name.
62  })
63  public abstract class AbstractRestBuilder<R,SELF extends AbstractRestBuilder<R,SELF>> implements RestBuilder<SELF> {
64  
65  	private final Class<R> resourceType;
66  	private final RestAnnotation.Builder anno = RestAnnotation.create();
67  	private final Map<String,Object> extras = m();
68  	private Boolean mdcAsyncPropagation;
69  
70  	/**
71  	 * Constructor.
72  	 *
73  	 * @param resourceType The resource type produced by {@link #build()}.  Must not be <jk>null</jk>.
74  	 */
75  	protected AbstractRestBuilder(Class<R> resourceType) {
76  		this.resourceType = assertArgNotNull("resourceType", resourceType);
77  	}
78  
79  	/**
80  	 * Returns this builder cast to the self type.
81  	 *
82  	 * @return This object.
83  	 */
84  	protected final SELF self() {
85  		return (SELF)this;
86  	}
87  
88  	/**
89  	 * Builds the configured resource instance.
90  	 *
91  	 * @return A new resource instance configured by this builder.
92  	 */
93  	public abstract R build();
94  
95  	/**
96  	 * Reflectively instantiates the resource type, preferring a {@code (RestBuilder<?>)} constructor (constructor
97  	 * injection) and falling back to the no-arg constructor.
98  	 *
99  	 * <p>
100 	 * Flavor builder {@code build()} implementations call this then stash {@code this} on the returned instance.
101 	 *
102 	 * @return A new, uninitialized resource instance.
103 	 */
104 	protected R createResource() {
105 		var ci = ClassInfo.of(resourceType);
106 		var ctor = ci.getDeclaredConstructor(x -> x.hasParameterTypes(RestBuilder.class)).orElse(null);
107 		if (ctor != null)
108 			return ctor.accessible().newInstance(this);
109 		var noArg = ci.getNoArgConstructor(Visibility.PRIVATE)
110 			.orElseThrow(() -> isex("Resource class %s has no no-arg or RestBuilder<?> constructor.", resourceType.getName()));
111 		return noArg.accessible().newInstance();
112 	}

153 	}
