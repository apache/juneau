17  package org.apache.juneau.rest.server.view.freemarker.console;
18  
19  import static org.apache.juneau.commons.utils.CollectionUtils.*;
20  
21  import java.util.*;
22  import java.util.regex.*;
23  
24  /**
25   * Save and restore of a dashboard's state topics.
26   *
27   * <p>
28   * The listed topics are saved per browser under the namespace, 500 ms after the last change.  On load the first of
29   * these that yields a value wins: an app link decoder, then a {@code #s=} share link (with {@link #shareLink()}), then
30   * the saved state when its version matches, then each owner's default.  Bump {@link #version(int)} whenever a payload
31   * shape changes; saved state of another version is deleted.
32   * </p>
33   *
34   * <h5 class='section'>Example:</h5>
35   * <p class='bjava'>
36   * 	PersistSpec.<jsm>page</jsm>(<js>"app.orders"</js>).version(1)
37   * 		.topics(Topics.<jsm>filter</jsm>(<js>"range"</js>), Topics.<jsm>zoom</jsm>(<js>"daily"</js>))
38   * 		.shareLink();
39   * </p>
40   *
41   * @since 10.0.0
42   */
43  @SuppressWarnings({
44  	"java:S1845" // Fluent-builder setters intentionally mirror field names (Juneau DSL convention)
45  })
46  public final class PersistSpec {
47  
48  	private static final Pattern NS = Pattern.compile("^[a-z][a-z0-9.-]{0,63}$");
49  	private static final Pattern STATE_TOPIC = Pattern.compile("^(filter|zoom|legend):[A-Za-z][A-Za-z0-9_-]{0,63}$");
50  
51  	private final String ns;
52  	private int version = 1;
53  	private final List<String> topics = new ArrayList<>();
54  	private boolean shareLink;
55  	private final List<String> decoders = new ArrayList<>();
56  
57  	private PersistSpec(String ns) {
58  		this.ns = ns;
59  	}
60  
61  	/**
62  	 * Creates a per-page persistence spec.
63  	 *
64  	 * @param ns The storage namespace, matching {@code ^[a-z][a-z0-9.-]{0,63}$}.
65  	 * @return A new spec.
66  	 */
67  	public static PersistSpec page(String ns) {
68  		return new PersistSpec(ns);
69  	}
70  
71  	/**
72  	 * Sets the state version.
73  	 *
74  	 * @param value The version, at least 1.  Defaults to 1.
75  	 * @return This object.
76  	 */
77  	public PersistSpec version(int value) {
78  		version = value;
79  		return this;
80  	}
81  
82  	/**
83  	 * Appends the topics to save and restore.
84  	 *
85  	 * @param values {@code filter:}, {@code zoom:} or {@code legend:} topics owned by members of the dashboard.
86  	 * @return This object.
87  	 */
88  	public PersistSpec topics(String...values) {
89  		topics.addAll(List.of(values));
90  		return this;
91  	}
92  
93  	/**
94  	 * Enables {@code #s=} share links and the "Copy link" button.
95  	 *
96  	 * @return This object.
97  	 */
98  	public PersistSpec shareLink() {
99  		shareLink = true;
100 		return this;
101 	}
102 
103 	/**
104 	 * Appends an app link decoder, registered in JS with {@code JuneauDashboard.decoder.register(name, fn)}.
105 	 *
106 	 * <p>
107 	 * The decoder must be registered before {@code juneau-charts.js} loads (from the {@code window.JuneauConsoleDashboard}
108 	 * pre-load queue in a head script, or from a toolkit pack), because restore runs when the first chart mounts.  A page
109 	 * {@code init} script is too late.
110 	 * </p>
111 	 *
112 	 * @param name The dotted decoder name.
113 	 * @return This object.
114 	 */
115 	public PersistSpec decoder(String name) {
116 		decoders.add(name);
117 		return this;
118 	}
119 
120 	/** @return The topics, in order. */
121 	List<String> topics() {
122 		return topics;
123 	}
124 
125 	/**
126 	 * Checks this spec, reporting E-77 under {@code dash}.
127 	 *
128 	 * @param dash The dashboard id.
129 	 */
130 	void check(String dash) {
131 		if (ns == null || ! NS.matcher(ns).matches())
132 			throw Section.e77(dash, "namespace '" + ns + "' does not match " + NS.pattern());
133 		if (version < 1)
134 			throw Section.e77(dash, "persist version " + version + " must be at least 1");
135 		if (topics.isEmpty())
136 			throw Section.e77(dash, "persist needs at least one topic");
137 		var seen = new HashSet<String>();
138 		for (var t : topics) {
139 			if (t == null || ! STATE_TOPIC.matcher(t).matches())
140 				throw Section.e77(dash, "persist topic '" + t + "' is not a filter:, zoom: or legend: state topic");
141 			if (! seen.add(t))
142 				throw Section.e77(dash, "persist lists topic '" + t + "' twice");
143 		}
144 		var names = new HashSet<String>();
145 		for (var d : decoders) {
146 			if (! Section.isDottedName(d))
147 				throw Section.e77(dash, "decoder name '" + d + "' is not a dotted extension name");
148 			if (! names.add(d))
149 				throw Section.e77(dash, "decoder '" + d + "' is listed twice");
150 		}
151 	}
152 
153 	/** @return The persist JSON. */
154 	Map<String,Object> toMap() {
155 		Map<String,Object> m = map(
156 			"ns", ns,
157 			"version", version,
158 			"topics", List.copyOf(topics),
159 			"shareLink", shareLink,
160 			"decoders", List.copyOf(decoders)
161 		);
162 		return m;
163 	}
164 }
