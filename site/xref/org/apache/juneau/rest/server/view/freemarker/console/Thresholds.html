17  package org.apache.juneau.rest.server.view.freemarker.console;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.marshall.collections.*;
24  
25  /**
26   * The built-in {@code thresholds} decorator: a horizontal mark line at a y value, on a carrier series
27   * {@code juneau-thresholds-<n>}.
28   *
29   * <h5 class='section'>Example:</h5>
30   * <p class='bjava'>
31   * 	Thresholds.<jsm>at</jsm>(500).label(<js>"SLO"</js>).tone(<js>"danger"</js>);
32   * </p>
33   *
34   * @since 10.0.0
35   */
36  public final class Thresholds implements Decorator {
37  
38  	private static final Set<String> TONES = Set.of("danger", "ok", "warning");
39  
40  	private final Map<String,Object> value = new LinkedHashMap<>();
41  
42  	private Thresholds() {}
43  
44  	/** @param y The y value. @return A new decorator. */
45  	public static Thresholds at(Number y) {
46  		if (y == null)
47  			throw iaex("Thresholds y must not be null.");
48  		var x = new Thresholds();
49  		x.value.put("y", y);
50  		return x;
51  	}
52  
53  	/** @param v The line label. @return This object. */
54  	public Thresholds label(String v) { value.put("label", v); return this; }
55  
56  	/** @param v {@code danger}, {@code ok} or {@code warning}. @return This object. */
57  	public Thresholds tone(String v) {
58  		if (! TONES.contains(v))
59  			throw iaex("Thresholds tone '%s' must be 'danger', 'ok' or 'warning'.", v);
60  		value.put("tone", v);
61  		return this;
62  	}
63  
64  	@Override
65  	public JsonMap toJson() { return JsonMap.of("use", "thresholds", "values", List.of(new LinkedHashMap<>(value))); }
66  }
