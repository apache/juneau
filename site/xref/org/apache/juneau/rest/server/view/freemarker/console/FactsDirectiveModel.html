17  package org.apache.juneau.rest.server.view.freemarker.console;
18  
19  import java.io.*;
20  import java.util.*;
21  
22  import org.apache.juneau.marshall.collections.*;
23  import org.apache.juneau.marshall.json5.*;
24  
25  import freemarker.core.*;
26  import freemarker.template.*;
27  
28  /**
29   * The {@code <@facts>} FreeMarker twin of the {@code <@console facts=>} attribute: a JSON5 object body, merged
30   * last-wins with any other chrome-side facts.  Repeatable; the one strict check against Java-seeded facts happens
31   * once, at {@code </@console>}.
32   *
33   * <h5 class='section'>Example:</h5>
34   * <p class='bcode'>
35   * 	&lt;@facts&gt;{"buildVersion": "1.0"}&lt;/@facts&gt;
36   * </p>
37   *
38   * @since 10.0.0
39   */
40  public final class FactsDirectiveModel implements TemplateDirectiveModel {
41  
42  	/** The shared-variable name this directive registers under. */
43  	public static final String NAME = "facts";
44  
45  	FactsDirectiveModel() {}
46  
47  	@Override
48  	@SuppressWarnings({
49  		"unchecked" // FreeMarker's raw params Map is String-keyed by contract.
50  	})
51  	public void execute(Environment env, @SuppressWarnings("rawtypes") Map params, TemplateModel[] loopVars,
52  			TemplateDirectiveBody body) throws TemplateException, IOException {
53  		Map<String, TemplateModel> p = params;
54  		if (p.containsKey("format"))
55  			throw FtlAttrLists.reject("<@facts> has no format= attribute.");
56  		FtlAttrLists.rejectUnknown(p, NAME, Set.of());
57  
58  		var cap = PageCapture.get(env);
59  		if (cap == null || ! cap.consoleOpen)
60  			throw FtlAttrLists.reject("<@facts> must be nested inside <@console>.");
61  
62  		var markup = PageCapture.render(body);
63  		if (markup.isBlank())
64  			return;
65  		JsonMap parsed;
66  		try {
67  			parsed = Json5Parser.DEFAULT.read(markup.trim(), JsonMap.class);
68  		} catch (org.apache.juneau.marshall.parser.ParseException e) {
69  			throw FtlAttrLists.reject("<@facts> body is not valid JSON5: " + e.getMessage());
70  		}
71  		cap.mergeChromeFacts(parsed);
72  	}
73  }
