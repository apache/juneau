17  package org.apache.juneau.rest.server.view.freemarker.console;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import java.util.*;
22  
23  import freemarker.core.*;
24  import freemarker.template.*;
25  
26  /**
27   * The {@code jcHasToolkit(name)} FreeMarker method: whether the current {@code <@page toolkit=…>} lists
28   * {@code name}. Lets a chrome check which toolkits are active so it can emit toolkit-specific markup.
29   *
30   * <h5 class='section'>Example:</h5>
31   * <p class='bcode'>
32   * 	&lt;#if jcHasToolkit("views")&gt;
33   * 	  &lt;link rel="stylesheet" href="/app/views-extras.css"&gt;
34   * 	&lt;/#if&gt;
35   * </p>
36   *
37   * <p>
38   * Only {@code toolkit=} counts.  Packs that cards require (for example {@code "datatables-glue"} for a
39   * {@code type="datatables"} card) are loaded but don't make {@code jcHasToolkit} return <jk>true</jk>.
40   *
41   * @since 10.0.0
42   */
43  public final class HasToolkitMethodModel implements TemplateMethodModelEx {
44  
45  	/** The shared-variable name this method registers under. */
46  	public static final String NAME = "jcHasToolkit";
47  
48  	HasToolkitMethodModel() {}
49  
50  	@Override
51  	public Object exec(@SuppressWarnings("rawtypes") List args) throws TemplateModelException {
52  		if (args.size() != 1)
53  			throw FtlAttrLists.reject(f("%s(name) takes exactly one argument; got '%s'.", NAME, args.size()));
54  		var a = args.get(0);
55  		if (! (a instanceof TemplateScalarModel s))
56  			throw FtlAttrLists.reject(f("%s(name) takes a string; got '%s'.", NAME, a));
57  		var cap = PageCapture.get(Environment.getCurrentEnvironment());
58  		return nn(cap) && cap.toolkits().contains(s.getAsString().trim()) ? TemplateBooleanModel.TRUE : TemplateBooleanModel.FALSE;
59  	}
60  }
