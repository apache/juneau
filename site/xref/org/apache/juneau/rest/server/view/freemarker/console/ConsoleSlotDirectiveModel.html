17  package org.apache.juneau.rest.server.view.freemarker.console;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import java.io.*;
22  import java.util.*;
23  
24  import freemarker.core.*;
25  import freemarker.template.*;
26  
27  /**
28   * The capture-only console slot directives: {@code <@head>}, {@code <@scripts>}, {@code <@brand>},
29   * {@code <@actions>}, {@code <@title>}, {@code <@footer>}, and {@code <@body>}. One parameterized instance is
30   * registered per slot name (its {@link PageCapture} field), so the seven directives share this one class.
31   *
32   * <p>
33   * Each is authored inside {@code <@console>}. During the console's body pass this directive <b>captures</b> its
34   * rendered body into the matching {@link PageCapture} field and emits <b>nothing</b> at the author's position;
35   * {@code <@console>} then places each captured slot into the fixed document at its correct spot (chrome-level
36   * {@code <@head>}/{@code <@scripts>} in the head/end-of-body pass-throughs; {@code <@brand>}/{@code <@actions>}/
37   * {@code <@title>} in the header; {@code <@footer>} as the single footer; {@code <@body>} spliced onto the
38   * {@code <body>} tag). A slot authored outside a {@code <@console>} (no {@link PageCapture} in scope) is
39   * rejected fail-closed.
40   *
41   * <h5 class='section'>{@code phase=} (head/scripts only):</h5>
42   * <p>
43   * {@code <@head>} and {@code <@scripts>} accept an optional {@code phase=} that steers the capture to a distinct
44   * cascade position: {@code <@head phase="before-page-css">} lands app CSS <b>before</b> the page-local
45   * {@code <@page css=>} links (e.g. {@code a.css}), so a page {@code css=} rule still wins, and
46   * {@code <@scripts phase="after-toolkit">} lands scripts <b>between</b> the toolkit JS and the page-local
47   * {@code <@page init=>} scripts (e.g. {@code one.js}). Any other slot rejects {@code phase=}; an
48   * unrecognized value is rejected fail-closed.
49   *
50   * @since 10.0.0
51   */
52  @SuppressWarnings({
53  	"java:S1192" // Duplicated literals read more clearly inline than as constants
54  })
55  public final class ConsoleSlotDirectiveModel implements TemplateDirectiveModel {
56  
57  	/** Slot names; each is also the shared-variable name the directive registers under. */
58  	static final String HEAD = "head";
59  	static final String SCRIPTS = "scripts";
60  	static final String BRAND = "brand";
61  	static final String ACTIONS = "actions";
62  	static final String TITLE = "title";
63  	static final String FOOTER = "footer";
64  	static final String BODY = "body";
65  
66  	/** Every slot name, in registration order. */
67  	static final List<String> SLOT_NAMES = List.of(HEAD, SCRIPTS, BRAND, ACTIONS, TITLE, FOOTER, BODY);
68  
69  	/** {@code <@head phase>} value that lands the markup before the page-local {@code css=} links. */
70  	static final String PHASE_BEFORE_PAGE_CSS = "before-page-css";
71  
72  	/** {@code <@scripts phase>} value that lands the markup between the toolkit JS and {@code init=}. */
73  	static final String PHASE_AFTER_TOOLKIT = "after-toolkit";
74  
75  	private static final String PHASE = "phase";
76  	private static final String TEXT = "text";
77  
78  	private final String slot;
79  
80  	/**
81  	 * @param slot One of the slot-name constants; also the shared-variable name this directive is registered under.
82  	 */
83  	ConsoleSlotDirectiveModel(String slot) {
84  		this.slot = slot;
85  	}
86  
87  	/** @return The attributes this slot accepts (read by {@link ConsoleTemplateValidator}). */
88  	Set<String> attrs() {
89  		if (eq(HEAD, slot) || eq(SCRIPTS, slot))
90  			return Set.of(PHASE);
91  		if (eq(FOOTER, slot))
92  			return Set.of(TEXT);
93  		return Set.of();
94  	}
95  
96  	@Override
97  	@SuppressWarnings({
98  		"java:S3776", // Validation and slot dispatch read best as one straight-line method.
99  		"unchecked" // FreeMarker's raw params Map is String-keyed by contract.
100 	})
101 	public void execute(Environment env, @SuppressWarnings("rawtypes") Map params, TemplateModel[] loopVars,
102 			TemplateDirectiveBody body) throws TemplateException, IOException {
103 		Map<String, TemplateModel> p = params;
104 		if (p.containsKey("format"))
105 			throw FtlAttrLists.reject("<@" + slot + "> has no format= attribute.");
106 		FtlAttrLists.rejectUnknown(p, slot, attrs());
107 
108 		var phase = FtlAttrLists.scalar(p, PHASE);
109 		if (! phase.isEmpty()) {
110 			var ok = (eq(HEAD, slot) && eq(phase, PHASE_BEFORE_PAGE_CSS))
111 				|| (eq(SCRIPTS, slot) && eq(phase, PHASE_AFTER_TOOLKIT));
112 			if (! ok)
113 				throw FtlAttrLists.reject("<@" + slot + "> unknown phase '" + phase + "'.");
114 		}
115 
116 		var cap = PageCapture.get(env);
117 		if (n(cap) || ! cap.consoleOpen)
118 			throw FtlAttrLists.reject(f("<@%s> must be nested inside <@console>.", slot));
119 
120 		String markup;
121 		cap.slotOpen = true;
122 		try {
123 			markup = PageCapture.render(body);
124 		} finally {
125 			cap.slotOpen = false;
126 		}
127 		switch (slot) {
128 			case HEAD -> {
129 				if (phase.isEmpty())
130 					cap.head = markup;
131 				else
132 					cap.headBeforePageCss = markup;
133 			}
134 			case SCRIPTS -> {
135 				if (phase.isEmpty())
136 					cap.scripts = markup;
137 				else
138 					cap.scriptsAfterToolkit = markup;
139 			}
140 			// Captured, not merged, here: the merge needs the spec side too, so it runs once at </@console>.
141 			case BODY -> cap.ftlBodyAttrs = markup;
142 			case BRAND -> cap.addSlot("header", "brand", markup);
143 			case ACTIONS -> cap.addSlot("header", "actions", markup);
144 			case TITLE -> cap.addSlot("header", "replace", markup);
145 			case FOOTER -> footer(cap, FtlAttrLists.scalar(p, TEXT), markup);
146 			default -> throw new IllegalStateException(slot);
147 		}
148 	}
149 
150 	private static void footer(PageCapture cap, String text, String markup) throws TemplateModelException {
151 		if (! text.isEmpty()) {
152 			if (! markup.isBlank())
153 				throw FtlAttrLists.reject(f(
154 					"<@footer text='%s'> also has a body; use text= for plain text or the body for markup, not both.", text));
155 			cap.footer(f -> f.text(text));
156 		} else if (! markup.isBlank()) {
157 			cap.addSlot("footer", "content", markup);
158 		}
159 	}
160 }
