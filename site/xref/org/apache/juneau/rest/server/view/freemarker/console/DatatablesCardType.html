32  /**
33   * The built-in {@code datatables} card type, registered through {@code META-INF/services}: validates the author
34   * catalog and resolves per-column search metadata, with no shape lift — the catalog stays in author shape
35   * ({@code key}/{@code label}, not VIEW_META's {@code data}/{@code title}); {@code NS.card.liftCatalog}
36   * (juneau-views.js) does the lift at mount time.
37   *
38   * <h5 class='section'>Example:</h5>
39   * <p class='bjava'>
40   *   CardSource <jv>s</jv> = CardSource.<jsm>create</jsm>(<js>"datatables"</js>, <js>"releases"</js>)
41   *     .body(<js>"{dataUrl:'/rest/releases/data', columns:[{key:'version', label:'Version', searchType:'text'}]}"</js>)
42   *     .build();
43   *   JsonMap <jv>frag</jv> = <jk>new</jk> DatatablesCardType().toFragment(<jv>s</jv>);  <jc>// {"table": {...}}</jc>
44   * </p>
45   *
46   * @since 10.0.0
47   */
48  @SuppressWarnings({
49  	"java:S1192", // Repeated literals are contract wire keys; named constants would obscure the contract shape.
50  	"java:S3776" // Validation and rendering chains many independent guards; splitting them adds no clarity.
51  })
52  public final class DatatablesCardType implements CardTypeHandler {

85  	/**
86  	 * {@code filter:} and {@code redraw:} always; {@code selection:}, {@code detail:} and {@code bulk:}
87  	 * only when the catalog sets that key.  A {@code src}-only card's catalog arrives at runtime, so it is credited
88  	 * with all five.  The JS side applies this same rule and does not re-check once the catalog is known: a topic the
89  	 * catalog turns out not to need stays claimed and is simply never published.
90  	 */
91  	@Override
92  	public List<String> implicitTopics(JsonMap card) {
93  		var id = card.getString("id");
94  		var view = card.get("table") instanceof Map<?,?> t ? t : card;
95  		var all = view.get("dataUrl") == null;
96  		var out = new ArrayList<String>();
97  		if (all || view.get("selection") != null)
98  			out.add("selection:" + id);
99  		out.add("filter:" + id);
100 		out.add("redraw:" + id);
101 		if (all || view.get("detail") != null)
102 			out.add("detail:" + id);
103 		if (all || view.get("bulk") != null)
104 			out.add("bulk:" + id);
105 		return out;
106 	}

120 	@SuppressWarnings({
121 		"unchecked" // Narrows a JSON-shaped Map<?,?> to Map<String,Object>; its keys are always strings.
122 	})
123 	public JsonMap toFragment(CardSource source) {
124 		var hasSrc = nn(source.src()) && ! source.src().isEmpty();
125 		var hasBody = source.hasJsonBody() || (nn(source.body()) && ! source.body().isEmpty());
126 		if (hasSrc && hasBody)
127 			throw source.error("type='datatables' takes src= or a body, not both.");
128 		if (hasSrc)
129 			return new JsonMap();
130 		if (! hasBody)
131 			throw source.error(E27);
132 
133 		var catalog = source.json(); // E-25 for a non-JSON5 body (the dropped bare-URL form), E-24 for bad JSON5.
134 
135 		if (catalog.get("view") instanceof Map)
136 			throw source.error(E28);
137 
138 		var columns = catalog.getList("columns");
139 		var hasDataUrl = nn(catalog.get("dataUrl"));
140 		var hasRows = catalog.containsKey("rows");
141 		if (n(columns) || hasDataUrl == hasRows)
142 			throw source.error(E27);
143 		if (hasRows)
144 			validateRows(source, catalog);
145 
146 		checkNoContractVersion(source, catalog, "table");
147 		checkQuickStats(source, catalog);
148 
149 		// The authored body (and its lists) may be immutable, so the normalized columns go into a fresh list of copies.
150 		var normalized = new ArrayList<Object>(columns.size());
151 		var seen = new HashSet<String>();
152 		for (var i = 0; i < columns.size(); i++) {
153 			if (! (columns.get(i) instanceof Map<?,?> raw))
154 				throw source.error("columns['%s'] requires 'key'.", i);
155 			var col = new LinkedHashMap<>((Map<String,Object>) raw);
156 			normalized.add(col);
157 			var ident = columnIdent(col);
158 			if (n(ident))
159 				throw source.error("columns['%s'] requires 'key'.", i);
160 			for (var key : col.keySet())
161 				if (! KNOWN_COLUMN_KEYS.contains(key))
162 					throw source.error("column '%s' has unknown key '%s'; known: '%s'.",
163 						ident, key, String.join(", ", new TreeSet<>(KNOWN_COLUMN_KEYS)));
164 			if (! seen.add(ident))
165 				throw source.error("declares column '%s' twice.", ident);
166 			if (col.get("render") instanceof String r) {
167 				Render parsed;
168 				try {
169 					parsed = Render.parse(r);
170 				} catch (IllegalArgumentException e) {
171 					throw source.error("column '%s' render: %s", ident, e.getMessage());
172 				}
173 				var rm = new LinkedHashMap<String,Object>();
174 				rm.put("id", parsed.id);
175 				if (parsed.meta != null)
176 					rm.put("meta", parsed.meta);
177 				col.put("render", rm);
178 			} else if (n(col.get("render")) && nn(col.get("href"))) {
179 				// An href with no render implies the 'linked' renderer, as Column.toCatalogMap does.
180 				col.put("render", new LinkedHashMap<>(Map.of("id", "linked")));
181 			}
182 			resolveSearch(source, col, ident);
183 		}
184 		catalog.put("columns", normalized);
185 		projectDetailRegion(source, catalog);
186 
187 		if (catalog.get("ribbon") instanceof List<?> ribbon)
188 			for (var item : ribbon) {
189 				if (! (item instanceof Map<?,?> im))
190 					throw source.error("ribbon each item must be an object.");
191 				try {
192 					RibbonItem.validate(im);
193 				} catch (IllegalArgumentException e) {
194 					throw source.error("%s", e.getMessage());
195 				}
196 			}
197 
198 		if (nn(catalog.get("bulk")) && n(catalog.get("selection")))
199 			throw source.error("sets bulk without selection.");
200 		checkAggregateEndpoints(catalog);
201 
202 		var frag = new JsonMap();
203 		var page = catalog.remove("page");
204 		if (nn(page))
205 			frag.put("page", page);
206 		// toCard lifts a body visibleWhen to the card top level; leaving it in the catalog would emit it twice.
207 		catalog.remove("visibleWhen");
208 		frag.put("table", catalog);
209 		return frag;
210 	}

212 	/**
213 	 * Projects the detail endpoint onto the detail region, as the builder path's {@code RowDetail.toMap()} does:
214 	 * {@code type} defaults to {@code row-detail} and {@code dataUrl} to the detail endpoint, so a region that
215 	 * declares neither joins the panel's own expand GET instead of fetching again, and {@code lazy} to its effective
216 	 * per-type value ({@link RegionDef#effectiveLazy()}), which the builder always emits.  Explicit values are kept,
217 	 * and the authored (possibly immutable) maps are copied rather than written to.
218 	 */
219 	@SuppressWarnings({
220 		"unchecked" // Narrows a JSON-shaped Map<?,?> to Map<String,Object>; its keys are always strings.
221 	})
222 	private static void projectDetailRegion(CardSource source, JsonMap catalog) {
223 		if (! (catalog.get("detail") instanceof Map<?,?> detail) || ! (detail.get("region") instanceof Map<?,?> region)
224 				|| ! (detail.get("endpoint") instanceof String endpoint))
225 			return;
226 		var regionCopy = new LinkedHashMap<>((Map<String,Object>)region);
227 		regionCopy.putIfAbsent("type", RegionDef.TYPE_ROW_DETAIL);
228 		regionCopy.putIfAbsent("dataUrl", endpoint);
229 		regionCopy.putIfAbsent("lazy", RegionDef.TYPE_TAB_BODY.equals(regionCopy.get("type")));
230 		if (isJobLogRegion(region))
231 			regionCopy.put("config", jobLogConfig(source, regionCopy));
232 		var detailCopy = new LinkedHashMap<>((Map<String,Object>)detail);
233 		detailCopy.put("region", regionCopy);
234 		catalog.put("detail", detailCopy);
235 	}

238 	 * Validates a job-log region's {@code config} through {@link JobLogRegion}, the same rules as the builder path,
239 	 * and returns it in the builder's canonical key order so the two paths emit the same sidecar.
240 	 */
241 	@SuppressWarnings({
242 		"unchecked" // Narrows a JSON-shaped Map<?,?> to Map<String,Object>; its keys are always strings.
243 	})
244 	private static Map<String,Object> jobLogConfig(CardSource source, Map<String,Object> region) {
245 		var id = region.get("id") instanceof String s ? s : "";
246 		var config = region.get("config");
247 		if (! (config instanceof Map<?,?>))
248 			throw source.error("type='datatables' detail.region '%s' with populate 'job-log' requires a 'config' object.", id);
249 		try {
250 			return JobLogRegion.fromMap(id, (Map<String,Object>)config).validate().toMap();
251 		} catch (IllegalArgumentException e) {
252 			throw source.error("type='datatables' detail.region: %s", e.getMessage());
253 		}
254 	}

256 	private static boolean isJobLogRegion(Object region) {
257 		return region instanceof Map<?,?> r && JobLogRegion.POPULATOR.equals(r.get("populate"));
258 	}

269 	private static void validateRows(CardSource source, JsonMap catalog) {
270 		var rows = catalog.get("rows");
271 		if (! (rows instanceof List<?> list))
272 			throw source.error("type='datatables' 'rows' must be an array of objects.");
273 		for (var r : list)
274 			if (! (r instanceof Map<?, ?>))
275 				throw source.error("type='datatables' each 'rows' element must be an object.");
276 		if (eq("server", catalog.getString("dataMode")))
277 			throw source.error("type='datatables' inline 'rows' cannot be combined with dataMode:'server'; use 'dataUrl'.");
278 		if (nn(catalog.get("pollIntervalMs")))
279 			throw source.error("type='datatables' inline 'rows' cannot be combined with 'pollIntervalMs' (nothing to poll); use 'dataUrl'.");
280 		if (list.size() > ROWS_WARN_THRESHOLD)
281 			LOG.log(Level.WARNING, "<@card type=\"datatables\"> inline 'rows' has {0} entries; prefer 'dataUrl' for large data sets.", list.size());
282 	}

284 	private static String columnIdent(Map<String,Object> col) {
285 		var data = col.get("data");
286 		var key = col.get("key");
287 		var fallback = nn(key) ? key.toString() : null;
288 		return nn(data) ? data.toString() : fallback;
289 	}

291 	private static final Set<String> QUICK_STAT_KINDS = Set.of("tile", "meter", "segmented");
292 
293 	/** E-37: the quickStats strip is a non-empty list of uniquely-identified items, each well-formed for its kind. */
294 	private static void checkQuickStats(CardSource source, Map<String,Object> catalog) {
295 		if (! catalog.containsKey("quickStats"))
296 			return;
297 		if (! (catalog.get("quickStats") instanceof Map<?,?> qs) || ! (qs.get("items") instanceof List<?> items) || items.isEmpty())
298 			throw source.error("quickStats items must be a non-empty array.");
299 		var ids = new HashSet<String>();
300 		for (var o : items) {
301 			if (! (o instanceof Map<?,?> item))
302 				throw source.error("quickStats items must be objects.");
303 			if (! (item.get("id") instanceof String id) || id.isBlank())
304 				throw source.error("quickStats item ids must be non-blank strings.");
305 			if (! ids.add(id))
306 				throw source.error("quickStats item '%s': duplicate id.", id);
307 			checkQuickStatItem(source, id, item);
308 		}
309 	}

311 	private static void checkQuickStatItem(CardSource source, String id, Map<?,?> item) {
312 		var hasSegments = item.containsKey("segments");
313 		var hasMax = item.containsKey("max");
314 		var type = item.get("type");
315 		String kind;
316 		if (type == null) {
317 			if (hasSegments && hasMax)
318 				throw source.error("quickStats item '%s': has both segments and max; set type.", id);
319 			var unsegmented = hasMax ? "meter" : "tile";
320 			kind = hasSegments ? "segmented" : unsegmented;
321 		} else {
322 			if (! (type instanceof String t) || ! QUICK_STAT_KINDS.contains(t))
323 				throw source.error("quickStats item '%s': unknown type '%s'.", id, type);
324 			var fits = switch (t) {
325 				case "tile" -> ! hasSegments && ! hasMax;
326 				case "meter" -> hasMax && ! hasSegments;
327 				default -> hasSegments && ! hasMax;
328 			};
329 			if (! fits)
330 				throw source.error("quickStats item '%s': type '%s' does not match its shape.", id, t);
331 			kind = t;
332 		}
333 		if (! kind.equals("tile") && (! (item.get("label") instanceof String lbl) || lbl.isBlank()))
334 			throw source.error("quickStats item '%s': a %s item needs a non-blank label.", id, kind);
335 		switch (kind) {
336 			case "meter" -> {
337 				if (! (item.get("value") instanceof Number v) || ! Double.isFinite(v.doubleValue()) || v.doubleValue() < 0)
338 					throw source.error("quickStats item '%s': meter value must be a number >= 0.", id);
339 				if (! (item.get("max") instanceof Number m) || ! Double.isFinite(m.doubleValue()) || m.doubleValue() <= 0)
340 					throw source.error("quickStats item '%s': meter max must be a number > 0.", id);
341 				if (item.containsKey("text") && ! (item.get("text") instanceof String))
342 					throw source.error("quickStats item '%s': text must be a string.", id);
343 				checkQuickStatTone(source, id, item.get("tone"));
344 			}
345 			case "segmented" -> {
346 				if (item.containsKey("tone") || item.containsKey("text"))
347 					throw source.error("quickStats item '%s': a segmented item takes no tone or text; tone each segment.", id);
348 				if (! (item.get("segments") instanceof List<?> segs) || segs.isEmpty() || segs.size() > QuickStat.MAX_SEGMENTS)
349 					throw source.error("quickStats item '%s': segments must be an array of 1 to %s objects.", id, QuickStat.MAX_SEGMENTS);
350 				for (var i = 0; i < segs.size(); i++) {
351 					if (! (segs.get(i) instanceof Map<?,?> s) || ! (s.get("label") instanceof String l) || l.isBlank()
352 							|| ! (s.get("count") instanceof Number c) || ! isWholeNonNegative(c.doubleValue()))
353 						throw source.error("quickStats item '%s': segment %s must be an object with a non-blank label and a whole count >= 0.", id, i + 1);
354 					checkQuickStatTone(source, id, " segment " + (i + 1), s.get("tone"));
355 				}
356 			}
357 			default -> checkQuickStatTone(source, id, item.get("tone"));
358 		}
359 	}

361 	private static boolean isWholeNonNegative(double d) {
362 		return Double.isFinite(d) && d >= 0 && d == Math.rint(d);
363 	}

374 	/** E-31: a {@code contractVersion} key anywhere in the catalog (table, detail, quickStats, bulk) — JS stamps versions. */
375 	private static void checkNoContractVersion(CardSource source, Object node, String where) {
376 		if (node instanceof Map<?,?> m) {
377 			if (m.containsKey("contractVersion"))
378 				throw source.error("'%s' sets contractVersion; remove it, the console stamps contract versions.", where);
379 			for (var e : m.entrySet())
380 				checkNoContractVersion(source, e.getValue(), String.valueOf(e.getKey()));
381 		} else if (node instanceof List<?> l) {
382 			for (var v : l)
383 				checkNoContractVersion(source, v, where);
384 		}
385 	}

387 	@SuppressWarnings({
388 		"unchecked" // Narrows a JSON-shaped Map<?,?> to Map<String,Object>; its keys are always strings.
389 	})
390 	private static void resolveSearch(CardSource source, Map<String,Object> col, String ident) {
391 		var searchType = str(col.get("searchType"));
392 		if (ib(searchType))
393 			return; // Non-searchable column: no search block.
394 		if (n(SearchType.fromWire(searchType)))
395 			throw source.error("column '%s' searchType '%s' is unknown; known: '%s'.", ident, searchType, knownSearchTypes());
396 
397 		var customs = new LinkedHashMap<String,String>(); // name -> help
398 		if (col.get("customOperators") instanceof List<?> cs)
399 			for (var raw : cs) {
400 				if (! (raw instanceof Map<?,?>))
401 					throw source.error("column '%s' each customOperator must be an object.", ident);
402 				var cm = (Map<String,Object>) raw;
403 				customs.put(str(cm.get("name")), str(cm.get("help")));
404 			}
405 
406 		List<String> allow = null;
407 		if (col.get("searchOperators") instanceof List<?> a) {
408 			allow = new ArrayList<>();
409 			for (var n : a) {
410 				var name = str(n);
411 				if (! customs.containsKey(name) && n(SearchOperatorSet.standard().get(name)))
412 					throw source.error("column '%s' searchOperators names unknown operator '%s'; known: '%s'.",
413 						ident, name, knownOperators(customs.keySet()));
414 				allow.add(name);
415 			}
416 		}
417 
418 		var customOps = new ArrayList<Map<String,String>>();
419 		for (var e : customs.entrySet()) {
420 			var m = new LinkedHashMap<String,String>();
421 			m.put("name", e.getKey());
422 			m.put("help", e.getValue());
423 			customOps.add(m);
424 		}
425 		var search = Column.searchMeta(ident, searchType, allow, customOps);
426 		col.put("search", search);
427 		col.remove("searchType");
428 		col.remove("searchOperators");
429 		col.remove("customOperators");
430 	}

432 	private static String knownSearchTypes() {
433 		return Arrays.stream(SearchType.values()).map(SearchType::wire).collect(Collectors.joining(", "));
434 	}

436 	private static String knownOperators(Set<String> customNames) {
437 		var names = new TreeSet<String>(customNames);
438 		for (var op : SearchOperatorSet.standard().operators())
439 			names.add(op.name());
440 		return String.join(", ", names);
441 	}

443 	/**
444 	 * E-35: an {@code aggregate} bulk action's endpoint must not contain a {@code {field}} token (aggregate
445 	 * endpoints receive {@code {ids}} in the body). No {@code <@card id='%s'>} prefix — {@code BulkMutateDef}
446 	 * raises the identical message with no card in scope.
447 	 */
448 	@SuppressWarnings({
449 		"unchecked" // Narrows a JSON-shaped Map<?,?> to Map<String,Object>; its keys are always strings.
450 	})
451 	private static void checkAggregateEndpoints(JsonMap catalog) {
452 		if (! (catalog.get("bulk") instanceof Map<?,?> bulk))
453 			return;
454 		if (! (bulk.get("actions") instanceof List<?> actions))
455 			return;
456 		for (var raw : actions) {
457 			var action = (Map<String,Object>) raw;
458 			var mode = action.get(nn(action.get("mode")) ? "mode" : "bulkMode");
459 			if (neq("aggregate", mode))
460 				continue;
461 			var endpoint = str(action.get("endpoint"));
462 			if (nn(endpoint) && PLACEHOLDER.matcher(endpoint).find())
463 				throw iaex(
464 					"Bulk action '%s' has mode 'aggregate' but its endpoint '%s' contains a {field} token; "
465 					+ "aggregate endpoints receive {ids} in the body.", action.get("id"), endpoint);
466 		}
467 	}
