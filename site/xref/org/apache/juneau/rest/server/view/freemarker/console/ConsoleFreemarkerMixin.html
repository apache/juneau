32  /**
33   * A {@link FreemarkerMixin} that additionally makes the reserved-classpath admin-console chrome template
34   * ({@code org/apache/juneau/console/base.ftlh}) &mdash; and the {@code <@tag>} macro it defines &mdash; resolvable
35   * from any consumer template, without shadowing the consumer's own {@code basePath}-rooted template tree.
36   *
37   * <h5 class='section'>Used INSTEAD OF {@link FreemarkerMixin}, not beside it:</h5>
38   * <p>
39   * Register it typed as either {@link FreemarkerMixin} or {@code ConsoleFreemarkerMixin} &mdash; both declared
40   * return types work. {@code FreemarkerViewRenderer} finds the exact-type {@code FreemarkerMixin} bean when one is
41   * registered, and otherwise also looks for a bean declared under any registered subtype, which this class
42   * registers via a static initializer:
43   *
44   * <p class='bjava'>
45   * 	<ja>@Bean</ja> <jk>public</jk> FreemarkerMixin freemarker() {
46   * 		<jk>return</jk> ConsoleFreemarkerMixin.<jsm>create</jsm>().basePath(<js>"/templates/"</js>).build();
47   * 	}
48   * </p>
49   *
50   * <p>
51   * or equivalently:
52   *
53   * <p class='bjava'>
54   * 	<ja>@Bean</ja> <jk>public</jk> ConsoleFreemarkerMixin freemarker() {
55   * 		<jk>return</jk> ConsoleFreemarkerMixin.<jsm>create</jsm>().basePath(<js>"/templates/"</js>).build();
56   * 	}
57   * </p>
58   *
59   * <h5 class='section'>Consumer {@code /templates} is never shadowed:</h5>
60   * <p>
61   * The augmented {@link Configuration}'s {@code TemplateLoader} tries the consumer's own {@code basePath}-derived
62   * loader <b>first</b>; the reserved template is addressed by its full classpath-root-relative path
63   * ({@link #BASE_TEMPLATE_PATH}) on a second, classpath-<b>root</b>-rooted {@link ClassTemplateLoader}, so it cannot
64   * collide with a consumer path such as {@code /templates/base.ftlh}. Consumer templates include it with a
65   * loader-root-absolute path:
66   *
67   * <p class='bftl'>
68   * 	&lt;#include "/org/apache/juneau/console/base.ftlh"&gt;
69   * </p>
70   *
71   * <h5 class='section'>Consumer-supplied {@code Configuration} is the same instance, then fill-missing stamped:</h5>
72   * <p>
73   * When the request's {@code BeanStore} already has a {@code Configuration} bean (Spring/Spring-Boot autoconfig, or
74   * a microservice {@code BasicBeanStore.put}), {@link #resolveConfiguration} still returns that <b>same instance</b>
75   * ({@code ==}) so consumer settings win (encoding, {@code ObjectWrapper}, {@code exposeFields}, template-update
76   * delay, output format). It then fill-missing-stamps the reserved console shared variables
77   * ({@code page}, {@code card}, {@code navigation}, {@code node}, {@code theme}, {@code token}, {@code console},
78   * {@code main}, the {@code <@console>} slot names {@code head}/{@code scripts}/{@code brand}/{@code actions}/
79   * {@code title}/{@code footer}/{@code body}, and {@code jcTagHtml}) and splices the
80   * reserved-path console loader only when {@link #BASE_TEMPLATE_PATH} would not already resolve. A name the
81   * consumer already set is left alone. There is no opt-out flag on this class &mdash; opt out by registering
82   * {@link FreemarkerMixin} instead of {@code ConsoleFreemarkerMixin}.
83   *
84   * @since 10.0.0
85   */
86  @SuppressWarnings({

121 	/**
122 	 * Builder constructor.
123 	 *
124 	 * @param builder The builder. Must not be {@code null}.
125 	 * @throws IllegalArgumentException If the toolkit packs, provided packs or card requirements don't validate.
126 	 * @throws IllegalStateException E-70 if a required pack reaches a WebJar that is not on the classpath
127 	 * 	(see {@link Builder#requirePacks(String...)}).
128 	 */
129 	protected ConsoleFreemarkerMixin(Builder builder) {
130 		super(builder);
131 		this.chromeTemplate = builder.chromeTemplate;
132 		this.packs = new ToolkitPackRegistry();
133 		builder.extraPacks.forEach(packs::register);
134 		packs.provide(builder.providedPacks).validate();
135 		packs.requireKnown(builder.requiredPacks, "requirePacks");
136 		var roots = new LinkedHashSet<>(builder.requiredPacks);
137 		for (var p : builder.extraPacks)
138 			if (packs.reaches(p.name(), ToolkitPackRegistry.PACK_ECHARTS))
139 				roots.add(p.name());
140 		packs.checkWebJars(roots, builder.webJarProbe);
141 		this.cardRequirements = builder.cardRequirements.build(packs);
142 		this.cardTypes = builder.cardTypes.build();
143 		this.devMode = builder.devMode;
144 		this.adopterLoader = builder.adopterLoader;
145 		this.adopterRoot = builder.adopterRoot;
146 	}

166 	// Double-checked-locking identity cache: resolveConfiguration is called once per request, and
167 	// super.resolveConfiguration(req) returns the SAME Configuration object on every call for a given mixin
168 	// instance (consumer bean or FreemarkerDispatcher's lazy default cache). Tracking that object's identity
169 	// here means "have I already stamped THIS exact object" is answered in O(1). Presence checks on the loader
170 	// and shared-variable names still run on a cache miss so N Rest resources sharing one Spring singleton
171 	// do not nest Multi(Multi(base, console), console).
172 	@SuppressWarnings({
173 		"java:S3077" // volatile is required here for correct double-checked-locking safe-publication; the reference is publish-once (fully constructed/wrapped before assignment) and never compound-mutated.
174 	})
175 	private volatile Configuration wrappedConfiguration;
176 
177 	/**
178 	 * Resolves the active {@link Configuration} and fill-missing-stamps console shared variables onto it.
179 	 *
180 	 * <p>
181 	 * Always delegates to {@code super.resolveConfiguration(req)} first so a consumer {@code Configuration} bean
182 	 * wins for settings. Then:
183 	 * <ol class='spaced-list'>
184 	 * 	<li><b>Fill-missing names:</b> {@code page}, {@code card}, {@code navigation}, {@code node},
185 	 * 		{@code theme}, {@code token}, {@code console}, {@code main}, the {@code <@console>} slot names
186 	 * 		({@code head}, {@code scripts}, {@code brand}, {@code actions}, {@code title}, {@code footer},
187 	 * 		{@code body}), and {@code jcTagHtml} are set only when {@code getSharedVariable(name)} is unset. A
188 	 * 		consumer value is kept.
189 	 * 	<li><b>Loader splice:</b> the classpath-root console {@link ClassTemplateLoader} is wrapped in only when
190 	 * 		{@link #BASE_TEMPLATE_PATH} would not already resolve through the current loader.
191 	 * 	<li><b>Wrap-once + presence:</b> identity-cached per mixin instance so a second call does not
192 	 * 		re-wrap; presence checks so N Rest resources sharing one Spring singleton {@code Configuration} do
193 	 * 		not nest {@link MultiTemplateLoader}.
194 	 * </ol>
195 	 *
196 	 * @param req The current REST request.
197 	 * @return The active FreeMarker configuration. Never {@code null}.
198 	 */
199 	@Override
200 	public Configuration resolveConfiguration(RestRequest req) {
201 		var base = super.resolveConfiguration(req);
202 		if (base == wrappedConfiguration)
203 			return base;
204 		synchronized (this) {
205 			if (base != wrappedConfiguration) {
206 				spliceConsoleLoaderIfNeeded(base);
207 				fillMissingConsoleSharedVariables(base);
208 				wrappedConfiguration = base;
209 			}
210 		}
211 		return base;
212 	}
213 
214 	/**
215 	 * Wraps the current {@link TemplateLoader} with a classpath-root console loader only when
216 	 * {@link #BASE_TEMPLATE_PATH} would not resolve. Presence-checked so a second mixin instance sharing the
217 	 * same {@code Configuration} does not nest {@link MultiTemplateLoader}.
218 	 */
219 	private void spliceConsoleLoaderIfNeeded(Configuration cfg) {
220 		if (reservedTemplateResolves(cfg))
221 			return;
222 		var console = new ClassTemplateLoader(getClass().getClassLoader(), "");
223 		var existing = cfg.getTemplateLoader();
224 		if (n(existing))
225 			cfg.setTemplateLoader(console);
226 		else
227 			cfg.setTemplateLoader(new MultiTemplateLoader(new TemplateLoader[]{ existing, console }));
228 	}
229 
230 	/**
231 	 * Whether the reserved console chrome template is already visible on {@code cfg}'s loader. Uses
232 	 * {@link TemplateLoader#findTemplateSource(String)} so a miss is not cached as a failed
233 	 * {@link Configuration#getTemplate(String)}.
234 	 */
235 	private static boolean reservedTemplateResolves(Configuration cfg) {
236 		var loader = cfg.getTemplateLoader();
237 		if (n(loader))
238 			return false;
239 		try {
240 			var source = loader.findTemplateSource(BASE_TEMPLATE_PATH);
241 			if (n(source))
242 				return false;
243 			loader.closeTemplateSource(source);
244 			return true;
245 		} catch (@SuppressWarnings("unused") Exception e) {
246 			return false; // Loader probe failed; treat as unresolved and splice.
247 		}
248 	}

250 	/**
251 	 * Sets each reserved console shared variable only when that name is unset on {@code cfg}.
252 	 */
253 	private void fillMissingConsoleSharedVariables(Configuration cfg) {
254 		if (n(cfg.getSharedVariable(TagMethodModel.NAME)))
255 			cfg.setSharedVariable(TagMethodModel.NAME, new TagMethodModel());
256 		if (n(cfg.getSharedVariable(PageDirectiveModel.NAME))) {
257 			cfg.setSharedVariable(PageDirectiveModel.NAME, new PageDirectiveModel(chromeTemplate, packs));
258 		}
259 		if (n(cfg.getSharedVariable(CardDirectiveModel.NAME)))
260 			cfg.setSharedVariable(CardDirectiveModel.NAME, new CardDirectiveModel(cardRequirements, cardTypes));
261 		if (n(cfg.getSharedVariable(NavigationDirectiveModel.NAME)))
262 			cfg.setSharedVariable(NavigationDirectiveModel.NAME, new NavigationDirectiveModel());
263 		if (n(cfg.getSharedVariable(NodeDirectiveModel.NAME)))
264 			cfg.setSharedVariable(NodeDirectiveModel.NAME, new NodeDirectiveModel());
265 		if (n(cfg.getSharedVariable(ThemeDirectiveModel.NAME)))
266 			cfg.setSharedVariable(ThemeDirectiveModel.NAME, new ThemeDirectiveModel());
267 		if (n(cfg.getSharedVariable(TokenDirectiveModel.NAME)))
268 			cfg.setSharedVariable(TokenDirectiveModel.NAME, new TokenDirectiveModel());
269 		if (n(cfg.getSharedVariable(ConsoleDirectiveModel.NAME)))
270 			cfg.setSharedVariable(ConsoleDirectiveModel.NAME, new ConsoleDirectiveModel(devMode, cardTypes, packs));
271 		if (n(cfg.getSharedVariable(MainDirectiveModel.NAME)))
272 			cfg.setSharedVariable(MainDirectiveModel.NAME, new MainDirectiveModel());
273 		if (n(cfg.getSharedVariable(HasToolkitMethodModel.NAME)))
274 			cfg.setSharedVariable(HasToolkitMethodModel.NAME, new HasToolkitMethodModel());
275 		if (nn(adopterLoader) && n(cfg.getSharedVariable(AssetUrlMethodModel.NAME)))
276 			cfg.setSharedVariable(AssetUrlMethodModel.NAME, new AssetUrlMethodModel(adopterLoader, adopterRoot, devMode));
277 		// The seven capture-only slot directives share one parameterized class, one instance registered per slot name.
278 		for (var slot : ConsoleSlotDirectiveModel.SLOT_NAMES)
279 			if (n(cfg.getSharedVariable(slot)))
280 				cfg.setSharedVariable(slot, new ConsoleSlotDirectiveModel(slot));
281 		if (n(cfg.getSharedVariable(TopicDirectiveModel.NAME)))
282 			cfg.setSharedVariable(TopicDirectiveModel.NAME, new TopicDirectiveModel());
283 		if (n(cfg.getSharedVariable(BridgeDirectiveModel.NAME)))
284 			cfg.setSharedVariable(BridgeDirectiveModel.NAME, new BridgeDirectiveModel());
285 		if (n(cfg.getSharedVariable(BadgeDirectiveModel.NAME)))
286 			cfg.setSharedVariable(BadgeDirectiveModel.NAME, new BadgeDirectiveModel());
287 		if (n(cfg.getSharedVariable(FactsDirectiveModel.NAME)))
288 			cfg.setSharedVariable(FactsDirectiveModel.NAME, new FactsDirectiveModel());
289 	}

454 		/**
455 		 * Enables the {@code assetUrl(path)} template function that versions the adopter's own static assets.
456 		 *
457 		 * <p>
458 		 * {@code assetUrl('/js/app.js')} returns {@code /js/app.js?v=<crc32 of the bundled bytes>}, where the bytes are
459 		 * read once (and the token cached) from {@code resourceRoot + path} on {@code loader}. This is the adopter-side
460 		 * counterpart of Juneau's own {@code viewAssetUrl}/{@code consoleJsUrl}/{@code chromeCssUrl}/{@code themeAssetUrl}
461 		 * cache-busters, for files the servlet container's static handler serves. A path with a query string, a
462 		 * non-root-absolute path (e.g. an external URL), or a path naming no bundled resource is returned unchanged
463 		 * (the last also logs a warning in {@link #devMode(boolean) dev mode}). The URL path itself is never rewritten,
464 		 * so any context-path prefix must be added by the template.
465 		 *
466 		 * <h5 class='section'>Example:</h5>
467 		 * <p class='bjava'>
468 		 * 	<ja>@Bean</ja> <jk>public</jk> FreemarkerMixin freemarker() {
469 		 * 		<jk>return</jk> ConsoleFreemarkerMixin.<jsm>create</jsm>()
470 		 * 			.basePath(<js>"/templates/"</js>)
471 		 * 			.adopterAssets(getClass().getClassLoader(), <js>"static"</js>)  <jc>// /js/app.js -&gt; classpath static/js/app.js</jc>
472 		 * 			.build();
473 		 * 	}
474 		 * </p>
475 		 * <p class='bftl'>
476 		 * 	&lt;script src="${assetUrl('/js/app.js')}"&gt;&lt;/script&gt;
477 		 * </p>
478 		 *
479 		 * @param loader The class loader that holds the assets. Must not be {@code null}.
480 		 * @param resourceRoot The classpath directory the URL paths are relative to (e.g. {@code "static"}); a leading
481 		 * 	and trailing slash are optional. Must not be {@code null}.
482 		 * @return This object.
483 		 */
484 		public Builder adopterAssets(ClassLoader loader, String resourceRoot) {
485 			this.adopterLoader = rnn(loader);
486 			this.adopterRoot = rnn(resourceRoot);
487 			return this;
488 		}
