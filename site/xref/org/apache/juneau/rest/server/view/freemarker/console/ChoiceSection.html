17  package org.apache.juneau.rest.server.view.freemarker.console;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.rest.server.console.*;
22  
23  /**
24   * A rail section of checkboxes over the distinct values of one dictionary column of a data source.
25   *
26   * <p>
27   * Publishes {@code {schemaVersion:1, viewId, columns:{<dim>: [selected...]}, options:{deselected:[...]}}}, omitting
28   * {@code columns} and {@code options} when every value is selected.  The {@code deselected} list names the values the
29   * user has unticked (capped at {@link #max(int) max}), so a reload can list them again and tick them back.  The array
30   * value is a data-source extension: datatables do not apply it.
31   * </p>
32   *
33   * <p>
34   * Unticking every value publishes an empty list ({@code columns:{<dim>: []}}).  The data sources map an empty list to
35   * no restriction on the dimension, so "nothing ticked" shows every row, the same as "everything ticked".  The section
36   * shows a "Select all" button while a filter is in force.
37   * </p>
38   *
39   * <h5 class='section'>Example:</h5>
40   * <p class='bjava'>
41   * 	ChoiceSection.<jsm>fromDim</jsm>(<js>"orders"</js>, <js>"reason"</js>).label(<js>"Reason"</js>).max(50)
42   * 		.publishes(Topics.<jsm>filter</jsm>(<js>"reasons"</js>));
43   * </p>
44   *
45   * @since 10.0.0
46   */
47  @SuppressWarnings({
48  	"java:S1192" // Repeated literals are contract wire keys; named constants would obscure the contract shape.
49  })
50  public final class ChoiceSection extends Section {
51  
52  	private final String source;
53  	private final String dim;
54  	private int max = 50;
55  
56  	private ChoiceSection(String source, String dim) {
57  		this.source = source;
58  		this.dim = dim;
59  	}
60  
61  	/**
62  	 * Creates a choice section.
63  	 *
64  	 * @param source The data source id (its topic is {@code data:<source>}).
65  	 * @param dim The dictionary column whose values are listed.
66  	 * @return A new section.
67  	 */
68  	public static ChoiceSection fromDim(String source, String dim) {
69  		return new ChoiceSection(source, dim);
70  	}
71  
72  	/**
73  	 * Caps the values shown before "Show all".
74  	 *
75  	 * @param value The cap, at least 1.  Defaults to 50.
76  	 * @return This object.
77  	 */
78  	public ChoiceSection max(int value) {
79  		max = value;
80  		return this;
81  	}
82  
83  	@Override /* Section */
84  	public ChoiceSection publishes(String topic) {
85  		super.publishes(topic);
86  		return this;
87  	}
88  
89  	@Override /* Section */
90  	public ChoiceSection label(String label) {
91  		super.label(label);
92  		return this;
93  	}
94  
95  	@Override String kind() { return "choice"; }
96  
97  	/** @return The source id. */
98  	String source() {
99  		return source;
100 	}
101 
102 	@Override void check(String dash) {
103 		super.check(dash);
104 		if (! Topics.isKey(source))
105 			throw e77(dash, "choice section '" + id() + "' source '" + source + "' is not a valid topic key");
106 		if (! Topics.isKey(dim))
107 			throw e77(dash, "choice section '" + id() + "' dim '" + dim + "' is not a valid topic key");
108 		if (dim.equals("__proto__") || dim.equals("constructor") || dim.equals("prototype"))
109 			throw e77(dash, "choice section '" + id() + "' dim '" + dim + "' is a reserved name");
110 		if (max < 1)
111 			throw e77(dash, "choice section '" + id() + "' max must be at least 1");
112 	}
113 
114 	@Override void body(Map<String,Object> m) {
115 		m.put("source", source);
116 		m.put("dim", dim);
117 		m.put("max", max);
118 	}
119 }
