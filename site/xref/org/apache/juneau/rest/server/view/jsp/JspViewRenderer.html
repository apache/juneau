29  /**
30   * {@link ResponseProcessor} that detects {@link JspView}-typed return values and dispatches them
31   * to the underlying JSP engine via
32   * {@link jakarta.servlet.RequestDispatcher#forward(jakarta.servlet.ServletRequest,
33   * jakarta.servlet.ServletResponse) ServletContext.getRequestDispatcher(...).forward(...)}.
34   *
35   * <p>
36   * Auto-registered by {@link JspMixin} via
37   * {@link Rest#responseProcessors() @Rest(responseProcessors=...)}
38   * &mdash; callers who add the mixin don't need to wire up this class explicitly. Callers who want
39   * to handle {@code JspView} returns <i>without</i> adopting the mixin (e.g. mounting their own
40   * raw-JSP path) can add this class to their own
41   * {@link Rest#responseProcessors() responseProcessors} list.
42   *
43   * <h5 class='section'>Behavior:</h5>
44   *
45   * <ol class='spaced-list'>
46   * 	<li>Inspect the response content. If the value is not a {@link JspView}, return
47   * 		{@link ResponseProcessor#NEXT NEXT} so the rest of the chain runs (the standard POJO
48   * 		serializer / plain-text fallback handle non-{@code JspView} returns).
49   * 	<li>Read the active {@link JspMixin} from the {@code RestContext} bean store to
50   * 		discover the configured base path; fall back to {@code "/"} when none is registered (a
51   * 		caller using the renderer standalone without the mixin).
52   * 	<li>Copy every entry from {@link JspView#getAttributes()} onto the request as a request
53   * 		attribute via {@link jakarta.servlet.ServletRequest#setAttribute(String, Object)
54   * 		req.setAttribute(...)} so the JSP / JSTL EL pipeline can resolve them.
55   * 	<li>Apply every entry from {@link JspView#getResponseHeaders()} via
56   * 		{@link jakarta.servlet.http.HttpServletResponse#setHeader(String, String)
57   * 		res.setHeader(...)}.
58   * 	<li>Obtain the underlying {@link jakarta.servlet.ServletContext ServletContext},
59   * 		{@code getRequestDispatcher(basePath + templateName)}, and call {@code forward(...)}.
60   * 	<li>When no JSP engine is on the classpath, the dispatcher returns {@code null} (or forward
61   * 		fails with {@code ClassNotFoundException}). Surface a human-readable diagnostic naming the
62   * 		missing dependency and linking to the "Choosing a JSP engine" matrix.
63   * </ol>
64   *
65   * <h5 class='section'>Note on {@code forward()}:</h5>
66   *
67   * <p>
68   * {@link jakarta.servlet.RequestDispatcher#forward forward()} resets the response output stream.
69   * If a {@link RestPreCall @RestPreCall} hook has already
70   * written response headers/body before the renderer runs, the forward may fail or produce
71   * malformed output. The renderer runs at response-resolution time so this is rarely hit by
72   * accident; document the constraint for any user relying on {@code RestPreCall}.
73   *
74   * <h5 class='section'>See Also:</h5><ul>
75   * 	<li class='jc'>{@link JspMixin}
76   * 	<li class='jc'>{@link JspView}
77   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/ResponseProcessors">Response Processors</a>
78   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JspViewSupport">JSP View Support</a>
79   * </ul>
80   *
81   * @since 10.0.0
82   */
83  @SuppressWarnings({
84  	"resource" // Closeables here are framework-managed and not owned/closed by this class; not a real leak.
85  })
86  public class JspViewRenderer implements ViewRenderer {
87  
88  	/**
89  	 * Diagnostic message emitted when no JSP engine can be found on the classpath at forward-time.
90  	 *
91  	 * <p>
92  	 * Public so tests and consumer apps can pattern-match against the message text without
93  	 * depending on internal string literals.
94  	 */
95  	public static final String NO_ENGINE_DIAGNOSTIC =
96  		"""
97  		No JSP engine is available on the classpath. Add one of:
98  		  - org.eclipse.jetty.ee11:jetty-ee11-apache-jsp  (Jetty 12 EE11)
99  		  - org.apache.tomcat.embed:tomcat-embed-jasper  (embedded Tomcat / Spring Boot default)
100 		  - Or rely on the deployment container's bundled engine (Tomcat / JBoss / WildFly / ...)
101 		See https://juneau.apache.org/docs/topics/JspViewSupport for the full matrix.""";
102 
103 	@Override /* Overridden from ResponseProcessor */
104 	public int process(RestOpSession opSession) throws IOException, BasicHttpException {
105 		var req = opSession.getRequest();
106 		var res = opSession.getResponse();
107 
108 		var content = res.getContent(Object.class);
109 		if (! (content instanceof JspView content2))
110 			return NEXT;
111 
112 		// Resolve the base path. Prefer the registered JspMixin bean (which the mixin
113 		// instantiates), fall back to "/" when the renderer is used standalone.
114 		var basePath = req.getContext().getBeanStore()
115 			.getBean(JspMixin.class)
116 			.map(JspMixin::getBasePath)
117 			.orElse("/");
118 
119 		// Copy attributes onto the request so the JSP / JSTL EL can resolve them.
120 		content2.getAttributes().forEach(req::setAttribute);
121 
122 		// Apply caller-supplied response headers (e.g. Content-Type, Cache-Control).
123 		content2.getResponseHeaders().forEach(res::setHeader);
124 
125 		// Resolve the dispatch path: basePath + templateName, normalized for a single separator.
126 		// A path that escapes basePath (e.g. template names assembled from user input that
127 		// included ../ segments) is rejected with InternalServerError — template names are
128 		// caller-controlled, so an escape attempt indicates a server-side bug, not a request-
129 		// side attack. (User-input flowing into JSP rendering is handled by JspMixin.render
130 		// which catches IAE and surfaces it as Forbidden.)
131 		String target;
132 		try {
133 			target = joinPath(basePath, content2.getTemplateName());
134 		} catch (IllegalArgumentException ex) {
135 			throw new InternalServerError(ex, "JSP template name escapes configured base path: '%s'",
136 				content2.getTemplateName());
137 		}
138 
139 		// Find the JSP engine via the servlet context. Both forward() failure modes (missing
140 		// engine, dispatcher==null, ClassNotFoundException during forward) surface as a clear
141 		// diagnostic naming the missing dependency.
142 		try {
143 			var ctx = req.getServletContext();
144 			var rd = ctx.getRequestDispatcher(target);
145 			if (rd == null)
146 				throw new InternalServerError("Could not resolve RequestDispatcher for '%s'. %s",
147 					target, NO_ENGINE_DIAGNOSTIC);
148 			rd.forward(req.getHttpServletRequest(), res.getHttpServletResponse());
149 			return FINISHED;
150 		} catch (NoClassDefFoundError ex) {
151 			throw new InternalServerError(ex, NO_ENGINE_DIAGNOSTIC);
152 		} catch (IOException ex) {
153 			throw ex;
154 		} catch (Exception ex) {
155 			throw new InternalServerError(ex, "JSP render failed for '%s'", target);
156 		}
157 	}
158 
159 	/**
160 	 * Joins a base path and a template name with a single {@code "/"} separator, rejecting any
161 	 * resolved target that escapes {@code basePath} via {@code ..} segments.
162 	 *
163 	 * <p>
164 	 * Delegates to {@link FileUtils#resolveVirtualPathSafely(String, String)} — the canonical
165 	 * shared implementation reused by {@code JspMixin.render(...)} for the
166 	 * raw-{@code .jsp} dispatch path. Handles every {@code (basePath, template)} combination of
167 	 * trailing/leading slashes uniformly so callers can pass {@code basePath("/WEB-INF/views")} or
168 	 * {@code basePath("/WEB-INF/views/")} and {@code "hello.jsp"} or {@code "/hello.jsp"} without
169 	 * worrying about double-slash artifacts.
170 	 *
171 	 * <p>
172 	 * A {@code null} or empty {@code basePath} is normalized to {@code "/"} (the
173 	 * {@link JspMixin#DEFAULT_BASE_PATH default}) before the boundary check runs.
174 	 *
175 	 * @param basePath The base path (e.g. {@code "/WEB-INF/views/"}).
176 	 * @param template The template name (e.g. {@code "hello.jsp"}).
177 	 * @return The joined path (e.g. {@code "/WEB-INF/views/hello.jsp"}).
178 	 * @throws IllegalArgumentException If {@code template} escapes {@code basePath} via
179 	 * 	{@code ..} segments.
180 	 */
181 	static String joinPath(String basePath, String template) {
182 		var bp = isEmpty(basePath) ? "/" : basePath;
183 		return FileUtils.resolveVirtualPathSafely(bp, template);
184 	}
