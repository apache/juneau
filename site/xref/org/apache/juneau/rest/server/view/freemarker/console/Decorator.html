21  /**
22   * A chart decorator entry: one element of a chart card's {@code decorators[]}, run in the browser by the decorator
23   * registered under its {@code use} name ({@code JuneauCharts.decorator}).
24   *
25   * <h5 class='section'>Example:</h5>
26   * <p class='bjava'>
27   * 	ChartDef.<jsm>line</jsm>(<js>"t"</js>).src(<js>"/s"</js>).x(<js>"d"</js>).series(<js>"v"</js>, <js>"V"</js>)
28   * 		.decorate(Markers.<jsm>now</jsm>(<js>"Today"</js>)).decorate(Thresholds.<jsm>at</jsm>(500).label(<js>"SLO"</js>));
29   * </p>
30   *
31   * @since 10.0.0
32   */
33  public interface Decorator {
34  
35  	/**
36  	 * @return A new map {@code {"use": name, ...args}}; a string {@code topic} key is published on click, and
37  	 * 	{@link ChartDef#decorate(Decorator)} adds it to the card's {@code publishes}.
38  	 */
39  	JsonMap toJson();
40  }
