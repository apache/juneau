17  package org.apache.juneau.rest.server.view;
18  
19  import java.io.*;
20  
21  import org.apache.juneau.http.*;
22  import org.apache.juneau.rest.server.*;
23  import org.apache.juneau.rest.server.servlet.*;
24  
25  /**
26   * Abstract base for the standalone view-servlet companions of the view mixin resources.
27   *
28   * <p>
29   * Each per-engine bridge module ({@code juneau-rest-server-view-jsp},
30   * {@code juneau-rest-server-view-freemarker}, {@code juneau-rest-server-view-mustache},
31   * {@code juneau-rest-server-view-thymeleaf}) ships a thin {@code Basic*Servlet} subclass that:
32   *
33   * <ol class='spaced-list'>
34   * 	<li>Declares its own sibling top-level mount via {@link Rest#paths() @Rest(paths)} (e.g.
35   * 		{@code "/jsp/*"}), so the servlet is registered as a sibling of the host application's other
36   * 		servlets ({@code /rest/*}, {@code /jsp/*}, ...) rather than composed under a host root.
37   * 	<li>Supplies the per-engine {@link RawTemplateDispatcher} (the matching flavor-neutral
38   * 		{@code <Engine>Dispatcher} worker) via {@link #dispatcher()}.
39   * </ol>
40   *
41   * <p>
42   * The single greedy op declared here ({@code @RestGet(path="/*")}) captures the trailing remainder
43   * under the servlet's mount and delegates straight to the shared {@link RawTemplateDispatcher}.
44   * Because the mixin and standalone forms both route through the same {@code RawTemplateDispatcher}
45   * implementation, their raw file-serving behavior cannot drift.
46   *
47   * <h5 class='section'>Standalone vs mixin:</h5>
48   *
49   * <p>
50   * The mixin {@code Basic*Resource} keeps its op pinned at {@code /<token>/*} for composition into a
51   * host at {@code /}; the standalone {@code Basic*Servlet} moves the mount to the class level
52   * ({@code @Rest(paths="/<token>/*")}) and pins the op at {@code /*}, eliminating the path-doubling
53   * that a mixin-shaped op would otherwise produce when the servlet itself is mounted at
54   * {@code /<token>/*}.
55   *
56   * <h5 class='section'>See Also:</h5><ul>
57   * 	<li class='jic'>{@link RawTemplateDispatcher}
58   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestServerComposition">REST Server &mdash; Composition (mixins, paths)</a>
59   * </ul>
60   *
61   * @serial exclude
62   * @since 10.0.0
63   */
64  @Rest
65  public abstract class ViewServlet extends RestServlet {
66  
67  	private static final long serialVersionUID = 1L;
68  
69  	/**
70  	 * Returns the per-engine {@link RawTemplateDispatcher} this servlet delegates raw template
71  	 * dispatch to.
72  	 *
73  	 * <p>
74  	 * Typically the matching flavor-neutral {@code <Engine>Dispatcher} worker (which implements
75  	 * {@link RawTemplateDispatcher}).  Never {@code null}.
76  	 *
77  	 * @return The raw-template dispatcher. Never {@code null}.
78  	 */
79  	protected abstract RawTemplateDispatcher dispatcher();
80  
81  	/**
82  	 * [GET /*] &mdash; render the raw template identified by the trailing path under this servlet's
83  	 * sibling top-level mount.
84  	 *
85  	 * <p>
86  	 * The {@code @Path("/*") String path} captures the multi-segment remainder after the mount
87  	 * prefix (e.g. a request for {@code /jsp/admin/dashboard.jsp} on a servlet mounted at
88  	 * {@code /jsp/*} matches with {@code path = "admin/dashboard.jsp"}).  Delegates verbatim to the
89  	 * shared {@link RawTemplateDispatcher}.
90  	 *
91  	 * @param path The trailing path segment after the mount prefix.
92  	 * @param req The current REST request.
93  	 * @param res The current REST response.
94  	 * @throws IOException If the underlying response writer fails.
95  	 */
96  	@RestGet(
97  		path="/*",
98  		summary="View",
99  		description="Render a raw template under this servlet's sibling top-level mount.",
100 		swagger=@OpSwagger(ignore=true)
101 	)
102 	public void render(@Path("/*") String path, RestRequest req, RestResponse res) throws IOException {
103 		dispatcher().render(path, req, res);
104 	}
105 }
