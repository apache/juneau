17  package org.apache.juneau.rest.server.view;
18  
19  import java.io.*;
20  
21  import org.apache.juneau.http.*;
22  import org.apache.juneau.rest.server.*;
23  import org.apache.juneau.rest.server.servlet.*;
24  
25  /**
26   * Abstract base for the child-resource flavors of the view mixin resources.
27   *
28   * <p>
29   * Mirror of {@link ViewServlet} for routed-child deployment: each per-engine bridge module ships a thin
30   * {@code <Engine>Resource} subclass that declares its own subtree mount via
31   * {@link Rest#path() @Rest(path)} (e.g. {@code "/jsp"}) and supplies the per-engine
32   * {@link RawTemplateDispatcher} (the matching flavor-neutral {@code <Engine>Dispatcher} worker) via
33   * {@link #dispatcher()}.
34   *
35   * <p>
36   * The single greedy op declared here ({@code @RestGet(path="/*")}) captures the trailing remainder under
37   * the child's subtree mount and delegates straight to the shared {@link RawTemplateDispatcher}, so the
38   * mixin, servlet, and child forms cannot drift.
39   *
40   * <h5 class='section'>See Also:</h5><ul>
41   * 	<li class='jic'>{@link RawTemplateDispatcher}
42   * 	<li class='jc'>{@link ViewServlet}
43   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestServerComposition">REST Server &mdash; Composition (mixins, paths)</a>
44   * </ul>
45   *
46   * @since 10.0.0
47   */
48  @Rest
49  public abstract class ViewResource extends RestResource {
50  
51  	/**
52  	 * Returns the per-engine {@link RawTemplateDispatcher} this child delegates raw template dispatch
53  	 * to.
54  	 *
55  	 * <p>
56  	 * Typically the matching flavor-neutral {@code <Engine>Dispatcher} worker (which implements
57  	 * {@link RawTemplateDispatcher}).  Never {@code null}.
58  	 *
59  	 * @return The raw-template dispatcher. Never {@code null}.
60  	 */
61  	protected abstract RawTemplateDispatcher dispatcher();
62  
63  	/**
64  	 * [GET /*] &mdash; render the raw template identified by the trailing path under this child's
65  	 * subtree mount.
66  	 *
67  	 * @param path The trailing path segment after the mount prefix.
68  	 * @param req The current REST request.
69  	 * @param res The current REST response.
70  	 * @throws IOException If the underlying response writer fails.
71  	 */
72  	@RestGet(
73  		path="/*",
74  		summary="View",
75  		description="Render a raw template under this child's subtree mount.",
76  		swagger=@OpSwagger(ignore=true)
77  	)
78  	public void render(@Path("/*") String path, RestRequest req, RestResponse res) throws IOException {
79  		dispatcher().render(path, req, res);
80  	}
81  }
