29  /**
30   * The built-in {@code job-log} card type: a form that submits to an endpoint returning an {@code AsyncJobRef}, then
31   * streams the job's progress from {@code AsyncJobsMixin} into a log pane, with cancel and reload re-attach.
32   * {@code juneau-joblog.js} draws it; this handler checks the body's shape.
33   *
34   * <p>
35   * The body keys are {@code submit} ({@code endpoint}, required and same-origin; {@code method} {@code POST} or
36   * {@code PUT}; {@code envelope} {@code flat} (the default) or {@code action}; {@code action}, required with the
37   * action envelope; {@code fields}; {@code submitLabel}), {@code height} (at least {@value #MIN_HEIGHT} pixels),
38   * {@code urlParam}, {@code clearOnSubmit} (a boolean) and {@code output}.  A field takes {@code name} (required),
39   * {@code label}, {@code type} ({@code text}, {@code textarea}, {@code number} or {@code hidden}), {@code placeholder},
40   * {@code required} and {@code default}; {@code source} is rejected, since a page card has no row (use a
41   * {@link JobLogRegion}).  Unknown keys are rejected at every level, as the page schema rejects them.  In the flat
42   * envelope no field may be named {@code idempotencyKey}: the runtime adds that key, with a fresh value, to every
43   * submit body.  The action envelope sends {@code {action, targetId, idempotencyKey, fields}}, self-targeted:
44   * {@code targetId} is the key's own value.  The shape rules are {@link JobLogChecks}, shared with
45   * {@link JobLogRegion}.  There is no typed builder; build it with {@link CardSpec#of(String, String)}.
46   *
47   * <h5 class='section'>Example:</h5>
48   * <p class='bcode'>
49   * 	&lt;@card type="job-log" id="rebuild" title="Rebuild index"&gt;
50   * 	{ submit: { endpoint: '/ops/jobs/rebuild', fields: [ { name: 'shards', label: 'Shards', type: 'number', default: 4 } ] },
51   * 	  urlParam: 'job' }
52   * 	&lt;/@card&gt;
53   * </p>
54   * <p class='bjava'>
55   * 	<jv>page</jv>.card(CardSpec.<jsm>of</jsm>(<js>"job-log"</js>, <js>"rebuild"</js>).title(<js>"Rebuild index"</js>)
56   * 		.body(<js>"submit"</js>, Map.<jsm>of</jsm>(<js>"endpoint"</js>, <js>"/ops/jobs/rebuild"</js>)).body(<js>"urlParam"</js>, <js>"job"</js>));
57   * </p>
58   *
59   * @since 10.0.0
60   */
61  @SuppressWarnings({
62  	"java:S1192" // Repeated literals are contract wire keys; named constants would obscure the contract shape.
63  })
64  public final class JobLogCardType implements CardTypeHandler {
65  
66  	/** The smallest accepted {@code height}, in pixels. */
67  	public static final int MIN_HEIGHT = JobLogRegion.MIN_HEIGHT;
68  
69  	private static final Set<String> BODY_KEYS = new LinkedHashSet<>(List.of("submit", "height", "urlParam", "clearOnSubmit", "output", "visibleWhen"));
70  	private static final Pattern URL_PARAM = Pattern.compile("[A-Za-z][A-Za-z0-9_-]*");
71  
72  	/** Public no-arg constructor, required by {@link java.util.ServiceLoader}. */
73  	public JobLogCardType() {
74  		// No state; instantiated by ServiceLoader.
75  	}
76  
77  	@Override
78  	public String type() {
79  		return "job-log";
80  	}
81  
82  	/** A job-log card neither publishes nor subscribes. */
83  	@Override
84  	public Set<String> acceptedRoles() {
85  		return Set.of();
86  	}
87  
88  	/** A job-log card neither publishes nor subscribes. */
89  	@Override
90  	public Set<String> acceptedOps() {
91  		return Set.of();
92  	}
93  
94  	@Override
95  	public JsonMap toFragment(CardSource source) {
96  		if (! source.hasJsonBody())
97  			throw source.error("type='job-log' requires a JSON5 body with a 'submit' object.");
98  		var body = source.json();
99  		for (var k : body.keySet())
100 			if (! BODY_KEYS.contains(k))
101 				throw source.error("type='job-log' unknown key '%s'; allowed: '%s'.", k, String.join(", ", BODY_KEYS));
102 		if (! (body.get("submit") instanceof Map<?,?> submit) || ! (submit.get("endpoint") instanceof String endpoint) || endpoint.isBlank())
103 			throw source.error("type='job-log' requires submit.endpoint.");
104 		if (! SafePathTemplate.isSameOriginPath(endpoint))
105 			throw source.error("type='job-log' submit.endpoint must be a same-origin path; got '%s'.", endpoint);
106 		var urlParam = body.get("urlParam");
107 		try {
108 			// The shared rules (submit keys, envelope, action, fields, output) live in JobLogChecks, beside the region's.
109 			JobLogChecks.submit(submit, JobLogChecks.Mode.PAGE);
110 			JobLogChecks.height(body.get("height"), MIN_HEIGHT);
111 			if (nn(urlParam) && ! (urlParam instanceof String u && URL_PARAM.matcher(u).matches()))
112 				throw iaex("urlParam must match '%s'; got '%s'.", URL_PARAM.pattern(), urlParam);
113 			JobLogChecks.clearOnSubmit(body.get("clearOnSubmit"));
114 			if (body.containsKey("output"))
115 				JobLogChecks.output(body.get("output"));
116 		} catch (IllegalArgumentException e) {
117 			throw source.error("type='job-log' %s", e.getMessage());
118 		}
119 		var frag = new JsonMap();
120 		frag.putAll(body);
121 		frag.remove("visibleWhen");
122 		return frag;
123 	}
124 }
