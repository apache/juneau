17  package org.apache.juneau.rest.server.view.freemarker;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  import static org.apache.juneau.commons.utils.StringUtils.*;
21  
22  import java.util.*;
23  
24  import org.apache.juneau.rest.server.view.*;
25  
26  /**
27   * Immutable value class returned from {@code @RestOp}-annotated methods to ask the framework to
28   * render an Apache FreeMarker template.
29   *
30   * <p>
31   * Companion to {@link FreemarkerMixin} and {@link FreemarkerViewRenderer}: the mixin sets
32   * up the {@code /freemarker/*} mount and registers the renderer; the renderer detects
33   * {@code FreemarkerView} returns in the response-processor chain and asks the configured
34   * {@code freemarker.template.Configuration} to {@code getTemplate(templateName)} and then
35   * {@code process(dataModel, writer)} directly onto the response writer.
36   *
37   * <h5 class='figure'>Usage:</h5>
38   *
39   * <p class='bjava'>
40   * 	<ja>@RestGet</ja>(<js>"/hello/{name}"</js>)
41   * 	<jk>public</jk> View hello(<ja>@Path</ja> String <jv>name</jv>) {
42   * 		<jk>return</jk> FreemarkerView.<jsm>of</jsm>(<js>"hello.ftlh"</js>)
43   * 			.attr(<js>"name"</js>, <jv>name</jv>)
44   * 			.attr(<js>"ts"</js>, Instant.<jsm>now</jsm>());
45   * 	}
46   * </p>
47   *
48   * <p>
49   * The template name is configuration-relative &mdash; the configured
50   * {@code freemarker.template.Configuration} template loader prepends its own resource root. With
51   * the bridge's default configuration + {@code basePath("/templates/")},
52   * {@code FreemarkerView.of("hello.ftlh")} resolves to {@code /templates/hello.ftlh} on the
53   * classpath.
54   *
55   * <h5 class='section'>{@code .ftl} vs {@code .ftlh} auto-escape:</h5>
56   *
57   * <p>
58   * FreeMarker auto-selects HTML escaping by file extension: {@code .ftlh} templates emit
59   * HTML-escaped output (variable references are escaped automatically), while {@code .ftl}
60   * templates emit raw output. For HTML responses, prefer the {@code .ftlh} extension to avoid an
61   * XSS regression sneaking in via a future attribute-binding change.
62   *
63   * <h5 class='section'>Template suffix:</h5>
64   *
65   * <p>
66   * The {@link FreemarkerMixin.Builder#templateSuffix(String) templateSuffix(...)} builder
67   * knob lets the bridge append a suffix (e.g. {@code ".ftlh"}) to template names that don't
68   * already end with it, so callers can write {@code FreemarkerView.of("hello")} and have the
69   * bridge resolve it as {@code "hello.ftlh"} on the classpath (idempotent &mdash; explicit suffix
70   * is honored).
71   *
72   * <h5 class='section'>Immutability:</h5>
73   *
74   * <p>
75   * Each {@code attr(...)} / {@code attrs(...)} / {@code header(...)} call returns a <b>new</b>
76   * {@code FreemarkerView} carrying the additional binding; the original instance is unchanged.
77   * This keeps {@code FreemarkerView} a value type safe to share across requests (e.g. as a static
78   * singleton built once and returned from many handlers).
79   *
80   * <h5 class='section'>See Also:</h5><ul>
81   * 	<li class='jc'>{@link FreemarkerMixin}
82   * 	<li class='jc'>{@link FreemarkerViewRenderer}
83   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/FreemarkerViewSupport">FreeMarker View Support</a>
84   * </ul>
85   *
86   * @since 10.0.0
87   */
88  public final class FreemarkerView implements View {
89  
90  	private final String templateName;
91  	private final Map<String,Object> attributes;
92  	private final Map<String,String> responseHeaders;
93  
94  	/**
95  	 * Creates a new {@code FreemarkerView} carrying the given template name and no attributes.
96  	 *
97  	 * <p>
98  	 * The template name is configuration-relative; the configured
99  	 * {@code freemarker.template.Configuration} template loader supplies the prefix (resource
100 	 * root). With the bridge's default configuration + {@code basePath("/templates/")}, the name
101 	 * {@code "hello.ftlh"} resolves to {@code /templates/hello.ftlh} on the classpath.
102 	 *
103 	 * @param templateName The template name (configuration-relative, suffix optional when the
104 	 * 	mixin's {@link FreemarkerMixin.Builder#templateSuffix(String) templateSuffix} is
105 	 * 	configured). Must not be {@code null} or blank.
106 	 * @return A new {@code FreemarkerView} instance.
107 	 * @throws IllegalArgumentException If {@code templateName} is {@code null} or blank.
108 	 */
109 	public static FreemarkerView of(String templateName) {
110 		if (isBlank(templateName))
111 			throw iaex("templateName must not be null or blank");
112 		return new FreemarkerView(templateName, Map.of(), Map.of());
113 	}
114 
115 	private FreemarkerView(String templateName, Map<String,Object> attributes, Map<String,String> responseHeaders) {
116 		this.templateName = templateName;
117 		this.attributes = attributes;
118 		this.responseHeaders = responseHeaders;
119 	}
120 
121 	/**
122 	 * Returns a copy of this view with the given attribute added (or replaced if a binding for
123 	 * {@code key} already exists).
124 	 *
125 	 * <p>
126 	 * The full attribute map is passed to FreeMarker as the data model, so each entry is
127 	 * accessible inside the template as {@code ${key}}. {@code null} values are rejected at
128 	 * build-time because FreeMarker's default null-handling renders missing/null bindings as an
129 	 * error or empty string depending on configuration, which typically masks what is a caller
130 	 * bug; rejecting up-front surfaces the mistake at the call site (matches the
131 	 * Thymeleaf / Mustache bridge precedents).
132 	 *
133 	 * @param key The attribute key (exposed inside the template as {@code ${key}}). Must not be
134 	 * 	{@code null} or blank.
135 	 * @param value The attribute value. Must not be {@code null}.
136 	 * @return A new {@code FreemarkerView} carrying the additional attribute.
137 	 * @throws IllegalArgumentException If {@code key} is {@code null} or blank, or {@code value}
138 	 * 	is {@code null}.
139 	 */
140 	public FreemarkerView attr(String key, Object value) {
141 		if (isBlank(key))
142 			throw iaex("attribute key must not be null or blank");
143 		if (value == null)
144 			throw iaex("attribute value must not be null (attribute '%s')", key);
145 		var copy = new LinkedHashMap<>(attributes);
146 		copy.put(key, value);
147 		return new FreemarkerView(templateName, Map.copyOf(copy), responseHeaders);
148 	}
149 
150 	/**
151 	 * Returns a copy of this view with all entries from the given map added (or replaced if a
152 	 * binding for any key already exists).
153 	 *
154 	 * @param values The attributes to add. {@code null} is treated as an empty map (the result is
155 	 * 	the same instance with no entries added).
156 	 * @return A new {@code FreemarkerView} carrying the additional attributes.
157 	 * @throws IllegalArgumentException If any entry has a {@code null}/blank key or a {@code null}
158 	 * 	value.
159 	 */
160 	public FreemarkerView attrs(Map<String,?> values) {
161 		if (values == null || values.isEmpty())
162 			return this;
163 		var copy = new LinkedHashMap<>(attributes);
164 		values.forEach((k, v) -> {
165 			if (isBlank(k))
166 				throw iaex("attribute key must not be null or blank");
167 			if (v == null)
168 				throw iaex("attribute value must not be null (attribute '%s')", k);
169 			copy.put(k, v);
170 		});
171 		return new FreemarkerView(templateName, Map.copyOf(copy), responseHeaders);
172 	}
173 
174 	/**
175 	 * Returns a copy of this view with the given response header set on the rendered response.
176 	 *
177 	 * <p>
178 	 * Typical usage: {@code .header("Content-Type", "text/html; charset=UTF-8")} or
179 	 * {@code .header("Cache-Control", "no-store")}. The renderer applies these headers via
180 	 * {@link jakarta.servlet.http.HttpServletResponse#setHeader(String, String)
181 	 * HttpServletResponse.setHeader(...)} before writing the rendered output.
182 	 *
183 	 * @param name The header name. Must not be {@code null} or blank.
184 	 * @param value The header value. Must not be {@code null}.
185 	 * @return A new {@code FreemarkerView} carrying the additional response header.
186 	 * @throws IllegalArgumentException If {@code name} is {@code null} or blank, or {@code value}
187 	 * 	is {@code null}.
188 	 */
189 	public FreemarkerView header(String name, String value) {
190 		if (isBlank(name))
191 			throw iaex("header name must not be null or blank");
192 		if (value == null)
193 			throw iaex("header value must not be null (header '%s')", name);
194 		var copy = new LinkedHashMap<>(responseHeaders);
195 		copy.put(name, value);
196 		return new FreemarkerView(templateName, attributes, Map.copyOf(copy));
197 	}
198 
199 	@Override /* Overridden from View */
200 	public String getTemplateName() {
201 		return templateName;
202 	}
203 
204 	@Override /* Overridden from View */
205 	public Map<String,Object> getAttributes() {
206 		return attributes;
207 	}
208 
209 	@Override /* Overridden from View */
210 	public Map<String,String> getResponseHeaders() {
211 		return responseHeaders;
212 	}
213 
214 	@Override /* Overridden from Object */
215 	public String toString() {
216 		return "FreemarkerView[template=" + templateName + ", attrs=" + attributes.keySet()
217 			+ ", headers=" + responseHeaders.keySet() + "]";
218 	}
219 }
