31  /**
32   * Mixin that serves Juneau's first-party DataTables glue script at {@code /juneau-datatables.js}.
33   *
34   * <p>
35   * Compose into a host resource via {@link Rest#mixins() @Rest(mixins=DataTablesMixin.class)}; the
36   * {@code /juneau-datatables.js} URL then becomes available alongside the host's own endpoints, so browser pages can
37   * load the glue without the application hosting it itself.
38   *
39   * <h5 class='section'>Where the DataTables library comes from:</h5>
40   *
41   * <p>
42   * This module declares the jQuery and DataTables WebJars ({@code org.webjars:jquery},
43   * {@code org.webjars.npm:datatables.net}, {@code datatables.net-dt}, {@code datatables.net-buttons},
44   * {@code datatables.net-buttons-dt}) as dependencies; Juneau's own jars don't copy their files.
45   * {@link org.apache.juneau.rest.server.staticfile.WebJarsMixin} serves them from the classpath.  This mixin serves
46   * only the first-party {@code juneau-datatables.js} glue, which auto-initializes any server-rendered
47   * {@code <table data-juneau-datatable>} (e.g. one built by {@link DataTablesTable}) by calling
48   * {@code $(table).DataTable(opts)} once the library is present.
49   *
50   * <p>
51   * On a console page ({@code juneau-rest-server-console-ui-freemarker}), every {@code <@card type="datatables">}
52   * requires the {@code "datatables-glue"} toolkit pack, which pulls jQuery and DataTables in before the glue.  Compose
53   * this mixin and {@code WebJarsMixin} into the host; the page never loads the library by hand.  Outside a console
54   * page, wire the assets yourself as below.
55   *
56   * <h5 class='section'>Wiring the page assets (caller-side {@code @HtmlDocConfig}):</h5>
57   *
58   * <p>
59   * Juneau's {@link org.apache.juneau.marshall.html.HtmlDocConfig @HtmlDocConfig} is resolved per endpoint / per
60   * resource-class hierarchy &mdash; it is <b>not</b> contributed across a host's other endpoints by a mixin.  So a mixin
61   * cannot silently inject page assets into <i>your</i> data endpoints; instead, add the wire-points to the resource (or
62   * op) that renders the HTML table, referencing your chosen DataTables distribution and this served glue.  The URLs are
63   * SVL-resolvable, so the DataTables location can be overridden per environment without code changes.
64   *
65   * <h5 class='section'>Example:</h5>
66   * <p class='bjava'>
67   * 	<ja>@Rest</ja>(path=<js>"/releases"</js>, mixins=DataTablesMixin.<jk>class</jk>)
68   * 	<ja>@HtmlDocConfig</ja>(
69   * 		stylesheet=<js>"$S{juneau.datatables.cssUrl:https://cdn.datatables.net/2.1.8/css/dataTables.dataTables.min.css}"</js>,
70   * 		head={
71   * 			<js>"&lt;script src='$S{juneau.datatables.jqueryUrl:https://code.jquery.com/jquery-3.7.1.min.js}'&gt;&lt;/script&gt;"</js>,
72   * 			<js>"&lt;script src='$S{juneau.datatables.jsUrl:https://cdn.datatables.net/2.1.8/js/dataTables.min.js}'&gt;&lt;/script&gt;"</js>,
73   * 			<js>"&lt;script src='servlet:/juneau-datatables.js'&gt;&lt;/script&gt;"</js>
74   * 		}
75   * 	)
76   * 	<jk>public class</jk> ReleasesResource <jk>extends</jk> BasicRestServlet {
77   * 		<ja>@RestGet</ja>
78   * 		<jk>public</jk> Table getReleases() {
79   * 			<jk>return</jk> DataTablesTable.<jsm>of</jsm>(<js>"releases"</js>, <jv>releases</jv>, Release.<jk>class</jk>);
80   * 		}
81   * 	}
82   * </p>
83   *
84   * <h5 class='section'>Server-side processing ({@code JuneauDataTables.ajax}):</h5>
85   *
86   * <p>
87   * For a large table, let DataTables fetch pages from the server instead of rendering every row.  The served glue
88   * exposes <c>JuneauDataTables.ajax(url, extra?)</c>, which returns a DataTables {@code ajax} option that POSTs the
89   * server-side request as JSON to a {@code @RestOp(method="POST")} endpoint taking {@code @Content}
90   * {@link DataTablesRequest} &mdash; see {@link org.apache.juneau.rest.server.datatables.adapter.DataTablesQuery} for
91   * the endpoint side:
92   *
93   * <p class='bhtml'>
94   * 	&lt;script src="https://code.jquery.com/jquery-3.7.1.min.js"&gt;&lt;/script&gt;
95   * 	&lt;script src="https://cdn.datatables.net/2.1.8/js/dataTables.min.js"&gt;&lt;/script&gt;
96   * 	&lt;script src="/releases/juneau-datatables.js"&gt;&lt;/script&gt;
97   * 	&lt;table id="releases"&gt;&lt;/table&gt;
98   * 	&lt;script&gt;
99   * 	  new DataTable('#releases', {
100  * 	    serverSide: true,
101  * 	    ajax: JuneauDataTables.ajax('/releases/query'),
102  * 	    columns: [{data: 'name', title: 'Name'}, {data: 'version', title: 'Version'}]
103  * 	  });
104  * 	&lt;/script&gt;
105  * </p>
106  *
107  * <p>
108  * A server-rendered {@link DataTablesTable} can skip the inline {@code <script>} entirely: give it the
109  * {@link DataTablesTable#AJAX_ATTR} attribute (e.g. via a direct {@code Table} DOM edit after
110  * {@link DataTablesTable#of(String, java.util.Collection, Class) DataTablesTable.of}) and the glue wires
111  * {@code opts.ajax}/{@code opts.serverSide} automatically on load.
112  *
113  * <h5 class='section'>Mixin-only deployment:</h5>
114  *
115  * <p>
116  * This resource is designed for composition via {@code @Rest(mixins=...)}.  The mount path is pinned at the op level by
117  * {@link RestGet @RestGet(path="/juneau-datatables.js")} on {@link #getGlueScript}; a class-level
118  * {@code @Rest(paths=...)} declaration would be silently ignored under the mixin pattern.  {@code WebJarsMixin} follows the same
119  * pattern for {@code /webjars/*}.
120  *
121  * <h5 class='section'>See Also:</h5><ul>
122  * 	<li class='jc'>{@link DataTablesTable}
123  * 	<li class='jc'>{@link DataTablesColumns}
124  * 	<li class='jc'>{@link org.apache.juneau.rest.server.datatables.adapter.DataTablesQuery}
125  * 	<li class='link'><a class="doclink" href="https://datatables.net/examples/data_sources/dom">DataTables from a pre-rendered DOM table</a>
126  * </ul>
127  *
128  * @since 10.0.0
129  */
130 // @formatter:off
131 @Rest
132 public class DataTablesMixin {
133 
134 	/** The URL path at which the first-party glue script is served (relative to the host mount). */
135 	public static final String GLUE_PATH = "/juneau-datatables.js";
136 
137 	/** Classpath location of the shipped glue script (resolved against this class's package). */
138 	static final String GLUE_RESOURCE = "juneau-datatables.js";
139 
140 	/** Content type emitted for the glue script. */
141 	static final String GLUE_CONTENT_TYPE = "text/javascript;charset=utf-8";
142 
143 	/** {@code Cache-Control} header emitted for the glue script (1 day). */
144 	static final String GLUE_CACHE_CONTROL = "max-age=86400, public";
145 
146 	/** The shipped glue script bytes, read once from the classpath on first request. */
147 	@SuppressWarnings({
148 		"java:S3077" // Double-checked-locking safe publication of one whole immutable array reference, not per-element mutation; AtomicReferenceArray solves a different problem.
149 	})
150 	private static volatile byte[] glueScript;
151 
152 	/**
153 	 * [GET /juneau-datatables.js] &mdash; serve the first-party DataTables glue script.
154 	 *
155 	 * @return The glue script as a JavaScript {@link HttpResource}.
156 	 * @throws IOException If the shipped glue resource could not be read (effectively unreachable &mdash; the resource
157 	 * 	is shipped in the same jar as this class).
158 	 */
159 	@RestGet(
160 		path=GLUE_PATH,
161 		summary="Juneau DataTables glue script",
162 		description="First-party JavaScript that auto-initializes server-rendered <table data-juneau-datatable> elements.",
163 		swagger=@OpSwagger(ignore=true)
164 	)
165 	public HttpResource getGlueScript() throws IOException {
166 		return HttpResourceBean.of(
167 			ByteArrayBody.of(glueScript(), GLUE_CONTENT_TYPE),
168 			list(ContentType.of(GLUE_CONTENT_TYPE), CacheControl.of(GLUE_CACHE_CONTROL))
169 		);
170 	}
171 
172 	/** Returns the shipped glue-script bytes, reading (and caching) them from the classpath on first call. */
173 	@SuppressWarnings({
174 		"resource" // IoUtils.read(InputStream) closes the stream; JDT can't see through the call.
175 	})
176 	private static byte[] glueScript() throws IOException {
177 		var g = glueScript;
178 		if (g == null) {
179 			synchronized (DataTablesMixin.class) {
180 				g = glueScript;
181 				if (g == null) {  // HTT: the "already set" branch is only reachable if another thread wins the race between the two reads - unhittable single-threaded.
182 					g = IoUtils.read(DataTablesMixin.class.getResourceAsStream(GLUE_RESOURCE)).getBytes(StandardCharsets.UTF_8);
183 					glueScript = g;
184 				}
185 			}
186 		}
187 		return g;
188 	}
189 }
