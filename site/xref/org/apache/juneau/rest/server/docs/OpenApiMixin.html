31  /**
32   * Mixin that publishes the OpenAPI 3.1 spec endpoints for a resource.
33   *
34   * <p>
35   * Three URL paths are mounted:
36   * </p>
37   * <ul class='spaced-list'>
38   * 	<li>{@code /openapi} &mdash; content-negotiated (HTML view via {@link RedocUI} swap, plus all serializers
39   * 		registered on the host).
40   * 	<li>{@code /openapi.json} &mdash; <em>format-pinned</em>: always emits JSON regardless of the request
41   * 		{@code Accept} header. Implemented via {@link RestResponse#getDirectWriter(String)} so the pin is
42   * 		robust against {@code Accept: text/html} or any other media type.
43   * 	<li>{@code /openapi.yaml} &mdash; <em>format-pinned</em>: always emits YAML regardless of the request
44   * 		{@code Accept} header.
45   * </ul>
46   *
47   * <p>
48   * Sibling of {@link SwaggerMixin} (Swagger v2 spec at {@code /api}),
49   * {@link SwaggerUiMixin} (HTML-first mount at {@code /swagger}), and {@link RedocMixin}
50   * (HTML-first mount at {@code /redoc}). All four classes live in the
51   * {@code org.apache.juneau.rest.server.docs} api-docs mixin pack.
52   *
53   * <p>
54   * The mixin declares {@link Rest#serializers() @Rest(serializers=&#123;YamlSerializer.class&#125;)} so the
55   * {@code /openapi} content-negotiated endpoint can serve {@code application/yaml} when explicitly
56   * requested via {@code Accept}. Sub-context inheritance (per the mixin sub-context model) confines this
57   * contribution to this mixin's endpoints &mdash; the host's other endpoints retain their own serializer set.
58   *
59   * <h5 class='section'>Configurable mount path:</h5>
60   *
61   * <p>
62   * The default mount prefix {@code /openapi} can be overridden via the SVL variable
63   * {@code ${juneau.openapi.path:openapi}} &mdash; the same variable is reused for all three mounts
64   * ({@code /openapi/*}, {@code /openapi.json}, {@code /openapi.yaml}) so a single override
65   * relocates the whole surface. Set via system property
66   * ({@code -Djuneau.openapi.path=v1/openapi}), environment variable
67   * ({@code JUNEAU_OPENAPI_PATH=v1/openapi}), or {@code Config} key
68   * ({@code juneau.openapi.path = v1/openapi}) to change the runtime mount without subclassing.
69   * Resolution happens once at {@link RestContext} construction time (SVL resolution in {@code @RestOp(path)}).
70   *
71   * <p>
72   * Override accepts bare token ({@code openapi}), leading slash ({@code /openapi}), trailing slash
73   * ({@code openapi/}), or wildcard suffix ({@code /openapi/*}) &mdash; all resolve to the same mount.
74   *
75   * <h5 class='section'>Mixin-only deployment:</h5>
76   *
77   * <p>
78   * This resource is designed for composition via {@code @Rest(mixins=...)}. The three mount paths
79   * ({@code /openapi/*}, {@code /openapi.json}, {@code /openapi.yaml}) are pinned at the op level
80   * by {@link RestGet @RestGet(path=...)} on the handler methods; a class-level
81   * {@code @Rest(paths=...)} declaration would be silently ignored under the mixin pattern (see
82   * {@link Rest#paths() @Rest(paths)} Javadoc).
83   *
84   * <h5 class='figure'>Composition example:</h5>
85   *
86   * <p class='bjava'>
87   * 	<ja>@Rest</ja>(path=<js>"/myresource"</js>, mixins=OpenApiMixin.<jk>class</jk>)
88   * 	<jk>public class</jk> MyResource <jk>extends</jk> RestServlet {
89   * 		<jc>// Now also serves /myresource/openapi, /myresource/openapi.json, /myresource/openapi.yaml.</jc>
90   * 	}
91   * </p>
92   *
93   * <h5 class='section'>See Also:</h5><ul>
94   * 	<li class='jc'>{@link SwaggerMixin}
95   * 	<li class='jc'>{@link SwaggerUiMixin}
96   * 	<li class='jc'>{@link RedocMixin}
97   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestServerComposition">REST Server &mdash; Composition (mixins, paths)</a>
98   * </ul>
99   *
100  * @since 10.0.0
101  */
102 // @formatter:off
103 @Rest(
104 	serializers={YamlSerializer.class}
105 )
106 @JsonSchemaConfig(
107 	addDescriptionsTo="bean,collection,array,map,enum",
108 	addExamplesTo="bean,collection,array,map",
109 	ignoreTypes="OpenApi,org.apache.juneau.bean.html5.*",
110 	useBeanDefs="true"
111 )
112 public class OpenApiMixin {
113 
114 	/**
115 	 * [GET /openapi] &mdash; emit the OpenAPI 3.1 document for this resource.
116 	 *
117 	 * <p>
118 	 * Standard Juneau content negotiation. {@code Accept: text/html} renders the Redoc HTML view via the
119 	 * {@link RedocUI} per-media-type swap; other accept types serialize the {@link OpenApi} bean using the
120 	 * host's serializer set (plus the {@link YamlSerializer} contributed by this mixin).
121 	 *
122 	 * @param req The current REST request.
123 	 * @return The OpenAPI document.
124 	 * @throws NotFound If no OpenAPI document is available for this resource.
125 	 */
126 	@RestGet(
127 		path="/#{pathToken(${juneau.openapi.path:openapi})}/*",
128 		summary="OpenAPI 3.1 documentation",
129 		description="OpenAPI 3.1 documentation for this resource."
130 	)
131 	@HtmlDocConfig(
132 		rank=10,
133 		navlinks={
134 			"back: servlet:/",
135 			"json: servlet:/openapi.json",
136 			"yaml: servlet:/openapi.yaml"
137 		},
138 		aside="NONE"
139 	)
140 	@MarshalledConfig(
141 		swaps={
142 			RedocUI.class
143 		}
144 	)
145 	public OpenApi getOpenApi(RestRequest req) {
146 		return req.getOpenApi().orElseThrow(NotFound::new);
147 	}
148 
149 	/**
150 	 * [GET /openapi.json] &mdash; emit the OpenAPI 3.1 document as JSON, ignoring the {@code Accept} header.
151 	 *
152 	 * <p>
153 	 * Format-pinned: always returns {@code application/json}. Implemented via
154 	 * {@link RestResponse#getDirectWriter(String)} so the pin is not subject to content negotiation.
155 	 *
156 	 * @param req The current REST request.
157 	 * @param res The current REST response.
158 	 * @throws NotFound If no OpenAPI document is available for this resource.
159 	 * @throws IOException If an I/O error occurs while writing the response.
160 	 */
161 	@RestGet(
162 		path="/#{pathToken(${juneau.openapi.path:openapi})}.json",
163 		summary="OpenAPI 3.1 documentation (JSON)",
164 		description="OpenAPI 3.1 documentation for this resource as JSON."
165 	)
166 	public void getOpenApiJson(RestRequest req, RestResponse res) throws IOException {
167 		var doc = req.getOpenApi().orElseThrow(NotFound::new);
168 		try (var w = res.getDirectWriter("application/json")) {
169 			JsonSerializer.DEFAULT_READABLE.write(doc, w);
170 		}
171 	}
172 
173 	/**
174 	 * [GET /openapi.yaml] &mdash; emit the OpenAPI 3.1 document as YAML, ignoring the {@code Accept} header.
175 	 *
176 	 * <p>
177 	 * Format-pinned: always returns {@code application/yaml}.
178 	 *
179 	 * @param req The current REST request.
180 	 * @param res The current REST response.
181 	 * @throws NotFound If no OpenAPI document is available for this resource.
182 	 * @throws IOException If an I/O error occurs while writing the response.
183 	 */
184 	@RestGet(
185 		path="/#{pathToken(${juneau.openapi.path:openapi})}.yaml",
186 		summary="OpenAPI 3.1 documentation (YAML)",
187 		description="OpenAPI 3.1 documentation for this resource as YAML."
188 	)
189 	public void getOpenApiYaml(RestRequest req, RestResponse res) throws IOException {
190 		var doc = req.getOpenApi().orElseThrow(NotFound::new);
191 		try (var w = res.getDirectWriter("application/yaml")) {
192 			YamlSerializer.DEFAULT_READABLE.write(doc, w);
193 		}
194 	}
195 }
