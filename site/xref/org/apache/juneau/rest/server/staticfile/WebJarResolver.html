18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import java.io.*;
22  import java.util.*;
23  import java.util.concurrent.*;
24  
25  import org.apache.juneau.rest.server.*;
26  
27  /**
28   * Resolves a WebJar file to a cache-busted URL under the {@link WebJarsMixin} mount.
29   *
30   * <p>
31   * An asset is named {@code groupId:artifactId:path} (see {@link #asset(String, String, String)}).  The path is
32   * relative to {@code META-INF/resources/webjars/} and may contain {@link #VERSION_TOKEN}, which is replaced with the
33   * version read from the WebJar's own {@code META-INF/maven/<groupId>/<artifactId>/pom.properties}.  The URL gets a
34   * {@code ?v=<version>} cache-buster, so a version bump in the app's pom changes every URL.
35   *
36   * <p>
37   * A missing WebJar, or a missing file in a present WebJar, throws {@link IllegalStateException}, so a page never
38   * renders a tag that would 404.  Versions and file checks are cached for the life of the class loader.
39   *
40   * <h5 class='section'>Example:</h5>
41   * <p class='bjava'>
42   * 	String <jv>asset</jv> = WebJarResolver.<jsm>asset</jsm>(<js>"org.webjars"</js>, <js>"jquery"</js>, <js>"jquery/{version}/jquery.min.js"</js>);
43   * 	String <jv>url</jv> = WebJarResolver.<jsm>resolve</jsm>(<jv>req</jv>, <jv>asset</jv>);  <jc>// .../webjars/jquery/3.7.1/jquery.min.js?v=3.7.1</jc>
44   * </p>
45   *
46   * @since 10.0.0
47   */
48  public final class WebJarResolver {
49  
50  	/** Classpath root of every WebJar's files. */
51  	public static final String WEBJARS_ROOT = "META-INF/resources/webjars/";
52  
53  	/** Placeholder in an asset path that is replaced with the WebJar's version. */
54  	public static final String VERSION_TOKEN = "{version}";
55  
56  	private static final ClassLoader LOADER = WebJarResolver.class.getClassLoader();
57  	private static final Map<String,String> VERSIONS = new ConcurrentHashMap<>();
58  	private static final Map<String,Boolean> PRESENT = new ConcurrentHashMap<>();
59  
60  	private WebJarResolver() {}
61  
62  	/**
63  	 * Builds an asset id.
64  	 *
65  	 * @param groupId The WebJar group id, e.g. {@code org.webjars}.
66  	 * @param artifactId The WebJar artifact id, e.g. {@code jquery}.
67  	 * @param path The file path under {@code META-INF/resources/webjars/}, optionally containing {@link #VERSION_TOKEN}.
68  	 * @return The asset id {@code groupId:artifactId:path}.
69  	 */
70  	public static String asset(String groupId, String artifactId, String path) {
71  		return groupId + ":" + artifactId + ":" + path;
72  	}
73  
74  	/**
75  	 * Returns the version of a WebJar on the classpath.
76  	 *
77  	 * @param groupId The WebJar group id.
78  	 * @param artifactId The WebJar artifact id.
79  	 * @return The version from the WebJar's {@code pom.properties}.
80  	 * @throws IllegalStateException If the WebJar isn't on the classpath or names no version.
81  	 */
82  	public static String version(String groupId, String artifactId) {
83  		var key = groupId + ":" + artifactId;
84  		var v = VERSIONS.get(key);
85  		if (v == null) {
86  			v = readVersion(groupId, artifactId);
87  			VERSIONS.put(key, v);
88  		}
89  		return v;
90  	}
91  
92  	/**
93  	 * Resolves an asset id to its file path under {@code META-INF/resources/webjars/}, checking that the file exists.
94  	 *
95  	 * @param asset The asset id (see {@link #asset(String, String, String)}).
96  	 * @return The path with {@link #VERSION_TOKEN} filled in.
97  	 * @throws IllegalArgumentException If the asset id is malformed.
98  	 * @throws IllegalStateException If the WebJar or the file isn't on the classpath.
99  	 */
100 	public static String resolvePath(String asset) {
101 		var a = parse(asset);
102 		var version = version(a[0], a[1]);
103 		var path = a[2].replace(VERSION_TOKEN, version);
104 		if (! b(PRESENT.computeIfAbsent(path, k -> LOADER.getResource(WEBJARS_ROOT + k) != null)))
105 			throw new IllegalStateException(String.format("needs WebJar file '%s' (%s:%s:%s), which is not on the classpath.", path, a[0], a[1], version));
106 		return path;
107 	}
108 
109 	/**
110 	 * Resolves an asset id to an absolute, cache-busted URL under the {@link WebJarsMixin} mount of the request's servlet.
111 	 *
112 	 * @param req The in-flight request.
113 	 * @param asset The asset id (see {@link #asset(String, String, String)}).
114 	 * @return The URL.
115 	 */
116 	public static String resolve(RestRequest req, String asset) {
117 		var a = parse(asset);
118 		var path = resolvePath(asset);
119 		return req.getUriResolver().resolve("servlet:" + WebJarsMixin.PATH + "/" + path) + "?v=" + version(a[0], a[1]);
120 	}
121 
122 	private static String readVersion(String groupId, String artifactId) {
123 		var props = "META-INF/maven/" + groupId + "/" + artifactId + "/pom.properties";
124 		try (var in = LOADER.getResourceAsStream(props)) {
125 			if (in == null)
126 				throw new IllegalStateException(String.format("needs %s:%s on the classpath (no %s).", groupId, artifactId, props));
127 			var p = new Properties();
128 			p.load(in);
129 			var v = p.getProperty("version", "").trim();
130 			if (v.isEmpty())
131 				throw new IllegalStateException(String.format("needs %s:%s on the classpath (%s has no version).", groupId, artifactId, props));
132 			return v;
133 		} catch (IOException e) {
134 			throw new UncheckedIOException(e);
135 		}
136 	}
137 
138 	private static String[] parse(String asset) {
139 		var a = asset == null ? new String[0] : asset.split(":", 3);
140 		if (a.length != 3 || a[0].isBlank() || a[1].isBlank() || a[2].isBlank())
141 			throw new IllegalArgumentException(String.format("WebJar asset must be 'groupId:artifactId:path'; got '%s'.", asset));
142 		return a;
143 	}
144 }
