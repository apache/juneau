17  package org.apache.juneau.rest.server.staticfile;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.http.*;
22  import org.apache.juneau.http.response.*;
23  import org.apache.juneau.rest.server.*;
24  import org.apache.juneau.rest.server.servlet.*;
25  
26  /**
27   * Child-resource flavor of the static-files capability.
28   *
29   * <p>
30   * Mounts as a <b>routed child</b> via {@link Rest#children() @Rest(children=StaticFilesResource.class)}
31   * under a parent at the subtree {@code /static} and serves static files by delegating to the
32   * flavor-neutral {@link StaticFiles} worker bean resolved from the request context
33   * ({@link RestContext#getStaticFiles()}) &mdash; the same worker the {@link StaticFilesMixin mixin} and
34   * {@link StaticFilesServlet servlet} flavors use, so the three forms cannot drift.
35   *
36   * <p>
37   * Whereas the {@link StaticFilesMixin} mixin pins its op at {@code /static/*} to merge into a host's
38   * namespace, this child declares the subtree at the class level ({@link Rest#path() @Rest(path="/static")})
39   * and pins the ops at {@code /*} &mdash; the op must <b>not</b> re-pin {@code /static/*} or it would
40   * double-path to {@code /static/static/*} under the routed-subtree wiring.
41   *
42   * <h5 class='section'>See Also:</h5><ul>
43   * 	<li class='jc'>{@link StaticFilesMixin}
44   * 	<li class='jc'>{@link StaticFilesServlet}
45   * 	<li class='jc'>{@link StaticFiles}
46   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/StaticFiles">Static files</a>
47   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestServerComposition">REST Server &mdash; Mixins and Multi-Mount Paths</a>
48   * </ul>
49   *
50   * @since 10.0.0
51   */
52  // @formatter:off
53  @Rest(path="/static")
54  public class StaticFilesResource extends RestResource {
55  
56  	/**
57  	 * [GET /*] &mdash; serve a static file from the active {@link StaticFiles} worker bean.
58  	 *
59  	 * @param req The current REST request &mdash; supplies {@link RestContext#getStaticFiles()}.
60  	 * @param path The trailing remainder after the mount prefix.
61  	 * @param locale The request locale (used for localized resource lookups).
62  	 * @return The matching {@link HttpResource} (with content type + cache headers).
63  	 * @throws NotFound If no resource matches the requested path.
64  	 */
65  	@RestGet(
66  		path="/*",
67  		summary="Static files",
68  		description="Static file retrieval.",
69  		swagger=@OpSwagger(ignore=true)
70  	)
71  	public HttpResource getStaticFile(RestRequest req, @Path("/*") String path, Locale locale) {
72  		return req.getContext().getStaticFiles().resolve(path, locale).orElseThrow(NotFound::new);
73  	}
74  
75  	/**
76  	 * [HEAD /*] &mdash; return GET headers for a static file without the body.
77  	 *
78  	 * @param req The current REST request &mdash; supplies {@link RestContext#getStaticFiles()}.
79  	 * @param path The trailing remainder after the mount prefix.
80  	 * @param locale The request locale (used for localized resource lookups).
81  	 * @return The matching {@link HttpResource} (with headers; body suppressed by the processor).
82  	 * @throws NotFound If no resource matches the requested path.
83  	 */
84  	@RestOp(
85  		method="HEAD",
86  		path="/*",
87  		summary="Static files (HEAD)",
88  		description="Static file metadata retrieval.",
89  		swagger=@OpSwagger(ignore=true)
90  	)
91  	public HttpResource headStaticFile(RestRequest req, @Path("/*") String path, Locale locale) {
92  		return getStaticFile(req, path, locale);
93  	}
94  }
