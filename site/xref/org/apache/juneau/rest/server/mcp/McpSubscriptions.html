17  package org.apache.juneau.rest.server.mcp;
18  
19  /**
20   * App-facing, revision-neutral change-publish SPI for MCP {@code subscriptions/listen}.
21   *
22   * <p>
23   * Injected into tool/resource/prompt handlers via their {@link org.apache.juneau.commons.inject.BeanStore}
24   * argument (the same seam {@code RestRequest} and other request-scoped beans use), so application code can
25   * announce a change without any dependency on the v2 wire beans or transport. A {@link McpSubscriptionBroker}
26   * is the production implementation of this interface.
27   */
28  public interface McpSubscriptions {
29  
30  	/**
31  	 * Announces that a specific subscribed resource's content changed.
32  	 *
33  	 * @param uri The exact resource URI that changed. Must not be {@code null}.
34  	 */
35  	void resourceUpdated(String uri);
36  
37  	/** Announces that the tool list changed. */
38  	void toolsListChanged();
39  
40  	/** Announces that the prompt list changed. */
41  	void promptsListChanged();
42  
43  	/** Announces that the resource list changed. */
44  	void resourcesListChanged();
45  }
