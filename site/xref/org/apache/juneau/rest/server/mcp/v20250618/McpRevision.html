48  /**
49   * {@link McpRevision} implementation for MCP revision {@code 2025-06-18}.
50   *
51   * <p>
52   * Owns this revision's JSON-RPC method table and error-code table. Note the error-code table
53   * deliberately reproduces a known-wrong mapping: {@link McpErrorKind#UNKNOWN_METHOD},
54   * {@link McpErrorKind#TOOL_NOT_FOUND} and {@link McpErrorKind#PROMPT_NOT_FOUND} all report
55   * {@code -32601}, which is only actually correct for the first. Preserving that is intentional —
56   * the corrective fix for those two is tracked separately and is not part of this re-layering.
57   * {@link McpErrorKind#RESOURCE_NOT_FOUND} is not part of that known-wrong grouping: it correctly
58   * reports {@code -32002}, the {@code 2025-06-18} spec's dedicated missing-resource error code.
59   *
60   * <p>
61   * <b>Constructed per binding, not shared as a singleton (Correction C8).</b> The bound servlet or
62   * mixin builds a fresh instance on every {@code revision()} call, passing its own
63   * {@code capabilities()} hook result into the constructor — see {@link McpRestServlet} and
64   * {@link McpEndpoint}. That is how this revision learns which explicit capabilities (if any)
65   * to advertise, with no channel back into core: this class is otherwise entirely stateless, so
66   * constructing a new one per call costs nothing.
67   */

118 	/**
119 	 * Verifies that every registered tool's input <em>and</em> output schema is expressible in this
120 	 * revision's wire schema type.
121 	 *
122 	 * <p>
123 	 * The neutral {@link McpSchema} is an unconstrained JSON object carrier, but this revision's
124 	 * {@code JsonSchema} bean supports exactly six keywords and models only object schemas. Each
125 	 * schema role is therefore checked twice: its root must declare {@code "type":"object"}, and the
126 	 * complete graph reachable from that root may use only this revision's six keywords. Rather than
127 	 * silently dropping an unsupported keyword — or an incompatible root shape — on the wire, a config
128 	 * carrying one is rejected, naming the tool, the offending role ({@code inputSchema} /
129 	 * {@code outputSchema}), and the offending shape or keyword.
130 	 *
131 	 * <p>
132 	 * Before walking, each schema is passed through {@link JsonValueSafety} to bound nesting depth,
133 	 * node count, traversal time, and cyclic/shared references, so a pathological config cannot stall
134 	 * or exhaust the walk.
135 	 *
136 	 * <p>
137 	 * Note this cannot be a complete guarantee against every possible wire shape: {@code JsonSchema}
138 	 * is {@code public} and non-{@code final}, and Juneau's reflection-based marshalling will
139 	 * serialize a getter added by a subclass. No such subclass exists in this codebase, and none can
140 	 * be constructed through the public MCP API, but the check covers what a caller can express
141 	 * through {@link McpSchema}, not what reflection can reach.
142 	 *
143 	 * @param config The config to validate. Never {@code null}.
144 	 * @throws IllegalArgumentException If any tool's input or output schema has a non-object root or uses an unsupported keyword.
145 	 */
146 	public static void validateSchemas(McpServerConfig config) {
147 		config.getTools().forEach(handler -> {
148 			var spec = handler.descriptor();
149 			if (spec != null) {
150 				validateSchema(spec.getName(), "inputSchema", spec.getInputSchema());
151 				validateSchema(spec.getName(), "outputSchema", spec.getOutputSchema());
152 			}
153 		});
154 	}
155 
156 	private static void validateSchema(String toolName, String role, McpSchema schema) {
157 		if (schema == null)
158 			return;
159 		var raw = schema.toJsonMap();
160 		JsonValueSafety.check(raw, "Tool '" + toolName + "' " + role + " for MCP revision 2025-06-18");
161 		if (! "object".equals(raw.get("type")))
162 			throw iaex("Tool ''%s'' %s root type ''%s'' is not object; MCP revision 2025-06-18 requires object schemas.",
163 				toolName, role, raw.get("type"));
164 		checkKeywords(toolName, role, raw);
165 	}

167 	@SuppressWarnings({
168 		"java:S3776" // Cognitive complexity acceptable for an explicit-stack JSON-schema tree walk; splitting the per-shape branches would fragment a single cohesive traversal.
169 	})
170 	private static void checkKeywords(String toolName, String role, Object root) {
171 		var seen = Collections.newSetFromMap(new IdentityHashMap<Object,Boolean>());
172 		var stack = new ArrayDeque<SchemaValue>();
173 		stack.push(new SchemaValue(root));
174 		while (! stack.isEmpty()) {
175 			var framed = stack.pop();
176 			var value = framed.value();
177 			if ((value instanceof Map<?,?> || value instanceof Collection<?> || value != null && value.getClass().isArray())
178 				&& ! seen.add(value))
179 				continue;
180 			if (value instanceof Map<?,?> value2) {
181 				for (var entry : value2.entrySet()) {
182 					var keyword = String.valueOf(entry.getKey());
183 					if (! SUPPORTED_SCHEMA_KEYWORDS.contains(keyword))
184 						throw iaex("Tool ''%s'' %s declares JSON Schema keyword ''%s'', which MCP revision 2025-06-18 cannot represent.",
185 							toolName, role, keyword);
186 					var child = entry.getValue();
187 					if (("properties".equals(keyword) || "$defs".equals(keyword)) && child instanceof Map<?,?> child2) {
188 						for (var schema : child2.values())
189 							stack.push(new SchemaValue(schema));
190 					} else {
191 						stack.push(new SchemaValue(child));
192 					}
193 				}
194 			} else if (value instanceof Collection<?> value2)
195 				value2.forEach(child -> stack.push(new SchemaValue(child)));
196 			else if (value != null && value.getClass().isArray())
197 				for (var i = 0; i < Array.getLength(value); i++)
198 					stack.push(new SchemaValue(Array.get(value, i)));
199 		}
200 	}

214 			case RESOURCE_NOT_FOUND -> CODE_RESOURCE_NOT_FOUND;
215 			case INVALID_PARAMS -> CODE_INVALID_PARAMS;
216 			case INTERNAL_ERROR -> CODE_INTERNAL_ERROR;
217 			case PARSE_ERROR -> CODE_PARSE_ERROR;
218 		};

227 	/**
228 	 * Does the actual dispatch work. This revision never streams, so every non-notification outcome is a
229 	 * single {@link JsonRpcResponse}; {@link #dispatch} wraps it in a {@link McpResponseResult}.
230 	 */
231 	private JsonRpcResponse dispatchResponse(McpExchange exchange, McpServerConfig config, BeanStore ctx) {
232 		assertArgNotNull("exchange", exchange);
233 		assertArgNotNull("config", config);
234 		assertArgNotNull("ctx", ctx);
235 
236 		if (VALIDATED.add(config))
237 			validateSchemas(config);
238 
239 		var req = exchange.request();
240 		if (req == null)
241 			return JsonRpcResponse.errorResponse(null, errorCode(McpErrorKind.INVALID_REQUEST), "Request envelope is null");
242 
243 		var id = req.getId();
244 		var method = req.getMethod();
245 
246 		if (isEmpty(method))
247 			return JsonRpcResponse.notification(id) ? null
248 				: JsonRpcResponse.errorResponse(id, errorCode(McpErrorKind.INVALID_REQUEST), "Missing method");
249 
250 		try {
251 			var result = invoke(method, req.getParams(), config, ctx);
252 			return JsonRpcResponse.notification(id) ? null : JsonRpcResponse.ok(id, result);
253 		} catch (McpException e) {
254 			return JsonRpcResponse.notification(id) ? null : new JsonRpcResponse()
255 				.setJsonrpc(McpProtocol.JSON_RPC_2_0)
256 				.setId(id)
257 				.setError(e.toJsonRpcError());
258 		} catch (Exception e) {
259 			if (JsonRpcResponse.notification(id))
260 				return null;
261 			var message = e.getMessage() == null ? cns(e) : e.getMessage();
262 			return JsonRpcResponse.errorResponse(id, errorCode(McpErrorKind.INTERNAL_ERROR), message, JsonMap.of("type", cn(e)));
263 		}
264 	}

266 	private Object invoke(String method, Object params, McpServerConfig config, BeanStore ctx) {
267 		switch (method) {
268 			case McpMethods.INITIALIZE:
269 				return initialize(config);
270 			case McpMethods.PING:
271 				return new JsonMap();
272 			case McpMethods.TOOLS_LIST:
273 				return listTools(config, params, ctx);
274 			case McpMethods.TOOLS_CALL:
275 				return callTool(config, params, ctx);
276 			case McpMethods.PROMPTS_LIST:
277 				return listPrompts(config, params, ctx);
278 			case McpMethods.PROMPTS_GET:
279 				return getPrompt(config, params, ctx);
280 			case McpMethods.RESOURCES_LIST:
281 				return listResources(config, params, ctx);
282 			case McpMethods.RESOURCES_READ:
283 				return readResource(config, params, ctx);
284 			case McpMethods.RESOURCES_TEMPLATES_LIST:
285 				return listResourceTemplates(config, params, ctx);
286 			case McpMethods.COMPLETION_COMPLETE:
287 				return complete(config, params, ctx);
288 			default:
289 				throw new McpException(errorCode(McpErrorKind.UNKNOWN_METHOD), "Method not found: " + method);
290 		}
291 	}

293 	private InitializeResult initialize(McpServerConfig config) {
294 		var caps = this.capabilities;
295 		if (caps == null) {
296 			caps = new ServerCapabilities();
297 			if (! config.getTools().isEmpty())
298 				caps.setTools(new ToolCapability());
299 			if (! config.getPrompts().isEmpty())
300 				caps.setPrompts(new PromptCapability());
301 			if (! config.getResources().isEmpty() || ! config.getResourceTemplates().isEmpty())
302 				caps.setResources(new ResourceCapability());
303 			if (config.hasAnyCompleter())
304 				caps.setCompletions(new CompletionCapability());
305 		}
306 		return new InitializeResult()
307 			.setProtocolVersion(protocolVersion())
308 			.setCapabilities(caps)
309 			.setServerInfo(McpWire.serverInfo(config))
310 			.setInstructions(config.getInstructions());
311 	}

321 	private CallToolResult callTool(McpServerConfig config, Object params, BeanStore ctx) {
322 		var p = McpParamUtils.asMap(params);
323 		var name = McpParamUtils.strParam(p, "name");
324 		if (name == null)
325 			throw new McpException(errorCode(McpErrorKind.INVALID_PARAMS), "Missing tool name");
326 		var handler = config.getTools().stream()
327 			.filter(h -> name.equals(h.descriptor().getName()))
328 			.findFirst()
329 			.orElseThrow(() -> new McpException(errorCode(McpErrorKind.TOOL_NOT_FOUND), "Tool not found: " + name));
330 		var args = McpParamUtils.mapParam(p, PARAM_ARGUMENTS);
331 		var outcome = handler.call(args, ctx);
332 		validateStructuredOutput(outcome);
333 		return McpWire.toWire(outcome);
334 	}

336 	private static void validateStructuredOutput(McpToolOutcome outcome) {
337 		if (outcome == null || outcome.getStructuredContent() == null)
338 			return;
339 		if (! (outcome.getStructuredContent() instanceof Map<?,?>))
340 			throw new McpException(CODE_INTERNAL_ERROR,
341 				"Tool structuredContent must be a JSON object for MCP revision 2025-06-18");
342 		try {
343 			JsonValueSafety.check(outcome.getStructuredContent(), "Tool structuredContent");
344 		} catch (IllegalArgumentException e) {
345 			throw new McpException(CODE_INTERNAL_ERROR, e.getMessage());
346 		}
347 	}

378 	private ReadResourceResult readResource(McpServerConfig config, Object params, BeanStore ctx) {
379 		var p = McpParamUtils.asMap(params);
380 		var uri = McpParamUtils.strParam(p, "uri");
381 		if (uri == null)
382 			throw new McpException(errorCode(McpErrorKind.INVALID_PARAMS), "Missing resource uri");
383 		var exact = config.getResources().stream()
384 			.filter(h -> uri.equals(h.descriptor().getUri()))
385 			.findFirst();
386 		if (exact.isPresent())
387 			return McpWire.toWire(exact.get().read(uri, ctx));
388 		var match = config.resolveResourceTemplate(uri);
389 		var outcome = match == null ? null : match.handler().read(uri, match.variables(), ctx);
390 		if (outcome == null)
391 			throw new McpException(errorCode(McpErrorKind.RESOURCE_NOT_FOUND), "Resource not found: " + uri);
392 		return McpWire.toWire(outcome);
393 	}

403 	private CompleteResult complete(McpServerConfig config, Object params, BeanStore ctx) {
404 		var p = McpParamUtils.asMap(params);
405 		var ref = completionRef(McpParamUtils.mapParam(p, "ref"));
406 		var argument = McpParamUtils.mapParam(p, "argument");
407 		var argumentName = McpParamUtils.strParam(argument, "name");
408 		if (isEmpty(argumentName))
409 			throw new McpException(errorCode(McpErrorKind.INVALID_PARAMS), "Missing argument.name");
410 		var value = McpParamUtils.strictStrParam(argument, "value");
411 		if (value == null)
412 			throw new McpException(errorCode(McpErrorKind.INVALID_PARAMS), "Missing or non-string argument.value");
413 		var contextArguments = McpParamUtils.strictStrMapParam(McpParamUtils.mapParam(p, "context"), PARAM_ARGUMENTS);
414 
415 		var completer = config.completer(ref, argumentName);
416 		McpCompletionResult raw;
417 		if (completer == null) {
418 			raw = McpCompletionResult.empty();
419 		} else {
420 			var request = new McpCompletionRequest()
421 				.setRef(ref)
422 				.setArgumentName(argumentName)
423 				.setValue(value)
424 				.setContextArguments(contextArguments);
425 			raw = completer.complete(request, ctx);
426 		}
427 		return McpWire.toWire(McpCompletionResult.normalize(raw));
428 	}

430 	private McpCompletionRef completionRef(Map<String,Object> refMap) {
431 		var type = McpParamUtils.strParam(refMap, "type");
432 		if ("ref/prompt".equals(type)) {
433 			var name = McpParamUtils.strParam(refMap, "name");
434 			if (isEmpty(name))
435 				throw new McpException(errorCode(McpErrorKind.INVALID_PARAMS), "Missing ref.name for ref/prompt");
436 			return McpCompletionRef.prompt(name);
437 		}
438 		if ("ref/resource".equals(type)) {
439 			var uri = McpParamUtils.strParam(refMap, "uri");
440 			if (isEmpty(uri))
441 				throw new McpException(errorCode(McpErrorKind.INVALID_PARAMS), "Missing ref.uri for ref/resource");
442 			return McpCompletionRef.resource(uri);
443 		}
444 		throw new McpException(errorCode(McpErrorKind.INVALID_PARAMS), "Invalid or missing ref.type: " + type);
445 	}
