42  /**
43   * Mapping between the revision-neutral model and the {@code 2026-07-28} wire beans.
44   *
45   * <p>
46   * This revision maps in one direction only: neutral-to-wire. What a neutral handler returns becomes
47   * what a client receives. Unlike the {@code 2025-06-18} adapter, this revision defines no
48   * typed-handler API, so no wire-to-neutral adaptation is needed.
49   *
50   * <p>
51   * {@link #requestMeta(Object)} is the single opaque-to-typed conversion: every v2 request carries its
52   * negotiation/trace metadata nested under {@code params._meta} (never beside the JSON-RPC envelope), and this
53   * class parses that opaque {@code _meta} member into the v2 {@link RequestMeta} bean at the boundary.
54   *
55   * <p>
56   * <b>{@code null} is preserved as {@code null} throughout.</b> Juneau omits null bean properties, so
57   * normalizing a null collection to an empty one would add a property to the wire output. An empty
58   * {@code CallToolResult} must keep serializing as <c>{}</c>.
59   */
60  final class McpWire {
61  
62  	private McpWire() {}
63

66  	/**
67  	 * Parses the {@code _meta} member nested under an opaque {@code params} object into a typed
68  	 * {@link RequestMeta}.
69  	 *
70  	 * <p>
71  	 * Requires an object {@code params} containing an object {@code _meta}; the {@code 2026-07-28} schema puts
72  	 * every request's negotiation/trace metadata under {@code params._meta}, never beside the JSON-RPC
73  	 * envelope.
74  	 *
75  	 * @param params The opaque request params (the JSON-RPC {@code params} member). Can be <jk>null</jk>.
76  	 * @return The parsed metadata. Never <jk>null</jk>.
77  	 * @throws McpException If {@code params} or {@code params._meta} is absent or not an object.
78  	 */
79  	static RequestMeta requestMeta(Object params) {
80  		if (! (params instanceof Map<?,?> p))
81  			throw new McpException(McpRevision.CODE_INVALID_REQUEST, "Request params must be an object");
82  		var meta = p.get("_meta");
83  		if (! (meta instanceof Map<?,?>))
84  			throw new McpException(McpRevision.CODE_INVALID_REQUEST, "Request params._meta must be an object");
85  		return Json.to(Json.of(meta), RequestMeta.class);
86  	}
87  
88  	/**
89  	 * Returns the opaque {@code params._meta} member as a raw, string-keyed map, or an empty map if
90  	 * {@code params} or {@code params._meta} is absent or malformed.
91  	 *
92  	 * <p>
93  	 * Unlike {@link #requestMeta(Object)}, this never throws. A {@code TraceContextExtractor} runs
94  	 * before {@link McpRevision#validateMeta(Object)} enforces the per-request {@code _meta} contract
95  	 * (extraction happens once per operation invocation, valid or not), so it needs a defensive read
96  	 * rather than one that raises {@link McpException} on a shape violation.
97  	 *
98  	 * @param params The opaque request params (the JSON-RPC {@code params} member). Can be <jk>null</jk>.
99  	 * @return An unmodifiable, insertion-ordered map keyed by {@code String}. Never <jk>null</jk>;
100 	 * 	empty when {@code params._meta} is absent or not an object.
101 	 */
102 	static Map<String,Object> metaMapOrEmpty(Object params) {
103 		if (params instanceof Map<?,?> p && p.get("_meta") instanceof Map<?,?> m) {
104 			var result = new LinkedHashMap<String,Object>();
105 			m.forEach((k, v) -> result.put(String.valueOf(k), v));
106 			return Collections.unmodifiableMap(result);
107 		}
108 		return Collections.emptyMap();
109 	}

113 	static Tool toWire(McpToolSpec value) {
114 		if (value == null)
115 			return null;
116 		return new Tool()
117 			.setName(value.getName())
118 			.setDescription(value.getDescription())
119 			.setInputSchema(toWire(value.getInputSchema()))
120 			.setOutputSchema(toWire(value.getOutputSchema()));
121 	}

127 		return value == null ? null : Json.to(Json.of(value.toJsonMap()), WireJsonSchema.class);

130 	/**
131 	 * {@link JsonSchema} subclass used only for neutral-to-wire schema conversion.
132 	 *
133 	 * <p>
134 	 * {@link JsonSchema#getDefinitions()} and {@link JsonSchema#getId()} are legacy Draft-04 getters
135 	 * retained on the shared bean for backward compatibility; each falls back to reading the Draft
136 	 * 2020-12 {@code $defs}/{@code $id} field when its own field is unset "to avoid double
137 	 * serialization" in the single-field-set case the shared bean's authors anticipated. But because
138 	 * both legacy and current getters are still independently non-null once only {@code $defs}/{@code
139 	 * $id} is populated, plain bean introspection serializes both names. This adapter only ever
140 	 * populates the Draft 2020-12 fields (never the legacy ones — see the five keys asserted by
141 	 * {@code McpSchemaSafety}), so the legacy getters are pinned to {@code null} here to keep the wire
142 	 * output to the Draft 2020-12 property names alone.
143 	 */
144 	static final class WireJsonSchema extends JsonSchema<WireJsonSchema> {
145 		@SuppressWarnings({
146 			"java:S1168" // null (not {}) is required here: this pins the legacy Draft-04 getter to absent so Juneau's null-omitting serializer never emits "definitions" alongside the Draft 2020-12 "$defs" this adapter actually populates (see class Javadoc).
147 		})
148 		@Override public Map<String,JsonSchema<?>> getDefinitions() { return null; }
149 
150 		@SuppressWarnings({
151 			"java:S5738" // Intentional legacy-schema mapping: pins the deprecated-for-removal Draft-04 getter to absent for the same reason as getDefinitions() above (see class Javadoc).
152 		})
153 		@Override
154 		public URI getId() { return null; }
155 	}

157 	static CallToolResult toWire(McpToolOutcome value) {
158 		if (value == null)
159 			return null;
160 		var result = new CallToolResult().setIsError(value.getError()).setStructuredContent(value.getStructuredContent());
161 		if (value.getContent() != null)
162 			result.setContent(value.getContent().stream().map(McpWire::toWire).toList());
163 		return result;
164 	}

166 	static Content toWire(McpContentBlock x) {
167 		if (x == null)
168 			return null;
169 		return switch (x.kind()) {
170 			case TEXT -> new TextContent().setText(x.text());
171 			case IMAGE -> new ImageContent().setData(x.data()).setMimeType(x.mimeType());
172 			case RESOURCE -> new EmbeddedResourceContent().setResource(toWire(x.resource()));
173 		};
174 	}

181 			case BLOB -> new BlobResourceContents().setUri(x.uri()).setMimeType(x.mimeType()).setBlob(x.blob());
182 		};
183 	}

277 		if (config.getName() == null && config.getVersion() == null)
278 			return new Implementation()

284 	/**
285 	 * Builds the {@code server/discover} result.
286 	 *
287 	 * <p>
288 	 * Server identity is <b>not</b> set here: every successful result (including this one) receives its
289 	 * {@link ResultMeta#getServerInfo() serverInfo} uniformly from {@link McpRevision}'s common result
290 	 * finalization, so discovery does not special-case identity placement.
291 	 *
292 	 * @param capabilities The top-level capability advertisement. Can be <jk>null</jk>.
293 	 * @param protocolVersion The single supported protocol version to advertise. Must not be <jk>null</jk>.
294 	 * @param instructions Optional free-form usage instructions. Can be <jk>null</jk>.
295 	 * @return A new discovery result. Never <jk>null</jk>.
296 	 */
297 	static ServerDiscoverResult discover(ServerCapabilities capabilities, String protocolVersion, String instructions) {
298 		return new ServerDiscoverResult()
299 			.setSupportedVersions(protocolVersion)
300 			.setCapabilities(capabilities)
301 			.setInstructions(instructions);
302 	}
