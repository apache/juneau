27  /**
28   * Binding-owned OAuth 2.1 resource-server (RS) configuration for a {@code 2026-07-28} MCP endpoint &mdash; the F2
29   * authorization-hardening baseline (READY-312f).
30   *
31   * <p>
32   * <b>Off by default.</b> {@link #isEnabled()} is <jk>false</jk> unless {@link #setEnabled(boolean) enabled}, so an
33   * existing MCP endpoint that does not opt in behaves exactly as before (no bearer requirement, no well-known route).
34   * When enabled, the endpoint:
35   * <ul>
36   * 	<li>Requires a valid RFC 6750 bearer token on the MCP POST endpoint, establishing an authenticated principal.
37   * 	<li>Serves an RFC 9728 Protected Resource Metadata document from {@code .well-known/oauth-protected-resource}.
38   * 	<li>Answers a missing/invalid token with {@code 401} + a {@code WWW-Authenticate: Bearer ...} challenge carrying the
39   * 		{@code resource_metadata} pointer.
40   * 	<li>Enforces RFC 8707 audience matching against {@link #getResource() the resource identifier}.
41   * </ul>
42   *
43   * <p>
44   * <b>Mutable during setup, effectively immutable once published</b> &mdash; the same builder-less mutable-setup idiom as
45   * {@link McpMrtrConfig} / {@link McpCacheConfig}: configure it fully via the {@code createMcpOptions()} /
46   * {@code getMcpOptions()} override, then never mutate it afterward (a request-time reader sees it with no
47   * synchronization).
48   *
49   * <h5 class='section'>Token-validation mode:</h5>
50   * <p>
51   * The validation mode (RFC 7662 introspection vs JWKS/issuer JWT verification) is selected by which
52   * {@link TokenValidator} is supplied to {@link #setTokenValidator(TokenValidator)}:
53   * {@code OAuthIntrospectionValidator} (from {@code juneau-rest-server-auth-oauth}) for opaque tokens, or
54   * {@code JwtTokenValidator} (from {@code juneau-rest-server-auth-jwt}) for JWT access tokens.
55   *
56   * @since 10.0.0
57   */

304 	/**
305 	 * Registers the scopes required to invoke a specific MCP operation (SEP-2350 per-operation step-up).
306 	 *
307 	 * <p>
308 	 * Populates the default static per-operation scope map consulted by {@link #requiredScopesFor(McpOperationContext)}
309 	 * when no explicit {@link #setOperationScopeResolver(McpOperationScopeResolver) resolver} is set.  A {@code null} or
310 	 * blank {@code name} registers a method-wide entry matching every operation of that JSON-RPC {@code method} (e.g.
311 	 * every {@code tools/call} regardless of tool name); a specific {@code name} takes precedence over the method-wide
312 	 * entry.  Each supplied scope is also advertised as {@link #addScopeSupported(String) supported}.
313 	 *
314 	 * @param method The JSON-RPC method (e.g. {@code tools/call}).  Must not be <jk>null</jk> or blank.
315 	 * @param name The operation name (tool/prompt name or resource URI), or <jk>null</jk>/blank for a method-wide entry.
316 	 * @param scopes The scopes required to invoke the operation.  Must not be <jk>null</jk>, empty, or contain a
317 	 * 	<jk>null</jk>/blank element.
318 	 * @return This object (for method chaining).
319 	 * @throws IllegalArgumentException If {@code method} is <jk>null</jk>/blank or {@code scopes} is invalid.
320 	 */
321 	public McpResourceServerConfig addOperationScope(String method, String name, String... scopes) {
322 		if (method == null || method.isBlank())
323 			throw iaex("method must not be null or blank");
324 		if (scopes == null || scopes.length == 0)
325 			throw iaex("scopes must not be null or empty");
326 		// Validate the whole scopes array BEFORE mutating scopesSupported, so a throw on a later blank element never
327 		// leaves earlier scopes half-advertised.
328 		var set = new LinkedHashSet<String>();
329 		for (var s : scopes) {
330 			if (s == null || s.isBlank())
331 				throw iaex(MSG_scopeMustNotBeNullOrBlank);
332 			set.add(s);
333 		}
334 		scopesSupported.addAll(set);
335 		operationScopes.put(operationKey(method, name == null || name.isBlank() ? OPERATION_WILDCARD : name), set);
336 		return this;
337 	}

363 	/**
364 	 * Resolves the scopes required to invoke the given operation (SEP-2350).
365 	 *
366 	 * <p>
367 	 * Consults the {@link #setOperationScopeResolver(McpOperationScopeResolver) resolver} when installed; otherwise the
368 	 * static {@link #addOperationScope map}, preferring a {@link McpOperationContext#name() name}-specific entry over the
369 	 * method-wide {@value #OPERATION_WILDCARD} entry.  An empty return means "no per-operation requirement beyond the
370 	 * endpoint-wide {@link #getRequiredScopes() baseline}".
371 	 *
372 	 * @param ctx The resolved operation.  Must not be <jk>null</jk>.
373 	 * @return The required scopes (never <jk>null</jk>; possibly empty).
374 	 */
375 	public Set<String> requiredScopesFor(McpOperationContext ctx) {
376 		if (ctx == null)
377 			throw iaex("ctx must not be null");
378 		if (operationScopeResolver != null) {
379 			var s = operationScopeResolver.requiredScopesFor(ctx);
380 			return s == null ? Set.of() : s;
381 		}
382 		var name = ctx.name();
383 		if (name != null) {
384 			var s = operationScopes.get(operationKey(ctx.method(), name));
385 			if (s != null)
386 				return Collections.unmodifiableSet(s);
387 		}
388 		var s = operationScopes.get(operationKey(ctx.method(), OPERATION_WILDCARD));
389 		return s == null ? Set.of() : Collections.unmodifiableSet(s);
390 	}
391 
392 	private static String operationKey(String method, String name) {
393 		return method + '\u0000' + name;
394 	}

396 	/**
397 	 * Validates that this config is internally consistent for the {@link #isEnabled() enabled} state.
398 	 *
399 	 * <p>
400 	 * A no-op when disabled.  When enabled, requires a resource identifier (absolute, with scheme + authority) and a
401 	 * token validator.
402 	 *
403 	 * @throws IllegalStateException If enabled but misconfigured.
404 	 */
405 	public void validateEnabled() {
406 		if (!enabled)
407 			return;
408 		if (resource == null)
409 			throw isex("McpResourceServerConfig is enabled but no resource identifier was configured");
410 		if (resource.getScheme() == null || resource.getRawAuthority() == null)
411 			throw isex("McpResourceServerConfig resource %s must be an absolute URI with a scheme and authority", resource);
412 		if (tokenValidator == null)
413 			throw isex("McpResourceServerConfig is enabled but no TokenValidator was configured");
414 	}

416 	/**
417 	 * Returns this config's memoized {@link OAuthFilter} (READY-312f F2, M3).
418 	 *
419 	 * <p>
420 	 * The filter (RFC 6750 bearer extraction + delegated token validation) is built once from the
421 	 * {@link #getTokenValidator() token validator} and {@link #getRealm() realm} and reused across requests rather than
422 	 * rebuilt per request.  First use runs {@link #validateEnabled()}; under the never-mutate-after-setup contract the
423 	 * validator/realm are stable by the time a request reads this, so the memoized filter is authoritative.
424 	 *
425 	 * @return The bearer filter.  Never <jk>null</jk>.
426 	 * @throws IllegalStateException If this config is enabled but misconfigured.
427 	 */
428 	OAuthFilter oauthFilter() {
429 		var f = oauthFilter;
430 		if (f == null) {
431 			synchronized (this) {
432 				f = oauthFilter;
433 				if (f == null) {
434 					validateEnabled();
435 					f = OAuthFilter.create().validator(tokenValidator).realm(realm).build();
436 					oauthFilter = f;
437 				}
438 			}
439 		}
440 		return f;
441 	}
