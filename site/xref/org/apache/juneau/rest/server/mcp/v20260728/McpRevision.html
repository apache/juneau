63  /**
64   * {@link org.apache.juneau.rest.server.mcp.McpRevision} implementation for MCP revision {@code 2026-07-28}.
65   *
66   * <p>
67   * Owns this revision's stateless JSON-RPC method table, per-request {@code _meta} negotiation,
68   * SEP-2243 {@code Mcp-Method}/{@code Mcp-Name} header agreement checks, {@code server/discover}
69   * dispatch, and error-code table. Every request is independently negotiated from its own
70   * {@code _meta}; no handshake, session, or prior-request state exists.
71   *
72   * <p>
73   * <b>Constructed per binding, not shared as a singleton.</b> The bound servlet or mixin builds a
74   * fresh instance on every {@code revision()} call, passing its {@link McpOptions#getCapabilities()}
75   * result and its published {@link McpCacheConfig} into the constructor — see {@link McpRestServlet}
76   * and {@link McpEndpoint}. A {@code null} capabilities override auto-derives {@code server/discover}
77   * capabilities from the registered tool/prompt/resource/resource-template lists; a non-{@code null}
78   * override is advertised as-is. The cache config is binding-owned: it is supplied once at
79   * construction, is never {@code null}, and is treated as static and request-independent for the
80   * lifetime of this instance — every page of a given list method resolves to identical cache values.
81   * The instance retains no request-derived state.

271 	public McpDispatchResult dispatch(McpExchange exchange, McpServerConfig config, BeanStore ctx) {
272 		assertArgNotNull("exchange", exchange);
273 		assertArgNotNull("config", config);
274 		assertArgNotNull("ctx", ctx);
275 
276 		var req = exchange.request();
277 		if (req == null)
278 			return new McpResponseResult(reportError(ctx, null, errorCode(McpErrorKind.INVALID_REQUEST), "Request envelope is null"));
279 
280 		var id = req.getId();
281 		var method = req.getMethod();
282 
283 		if (isEmpty(method))
284 			return JsonRpcResponse.notification(id) ? null
285 				: new McpResponseResult(reportError(ctx, id, errorCode(McpErrorKind.INVALID_REQUEST), "Missing method"));
286 
287 		// SEP-2350 per-operation step-up enforcement at the POST-parse point (method now known).  Deliberately OUTSIDE
288 		// the try below: an insufficient-scope 403 (with its WWW-Authenticate step-up challenge) must propagate as a real
289 		// HTTP 403 through the response processors, not be caught and folded into a JSON-RPC error over HTTP 200.
290 		enforceStepUpScopes(method, req.getParams(), ctx);
291 
292 		try {
293 			validateHeaders(exchange, req);
294 			validateMeta(req.getParams());
295 			if (McpMethods.SUBSCRIPTIONS_LISTEN.equals(method))
296 				return JsonRpcResponse.notification(id) ? null
297 					: new McpStreamResult(dispatchSubscriptionsListen(exchange, id, req.getParams(), config, ctx));
298 			var result = finalizeResult(invoke(method, req.getParams(), config, ctx), config, ctx);
299 			return JsonRpcResponse.notification(id) ? null : new McpResponseResult(JsonRpcResponse.ok(id, result));
300 		} catch (McpException e) {
301 			if (JsonRpcResponse.notification(id))
302 				return null;
303 			recordRpcError(ctx, e.getCode(), e.getMessage());
304 			return new McpResponseResult(new JsonRpcResponse()
305 				.setJsonrpc(McpProtocol.JSON_RPC_2_0)
306 				.setId(id)
307 				.setError(e.toJsonRpcError()));
308 		} catch (Exception e) {
309 			if (JsonRpcResponse.notification(id))
310 				return null;
311 			var message = e.getMessage() == null ? cns(e) : e.getMessage();
312 			var code = errorCode(McpErrorKind.INTERNAL_ERROR);
313 			recordRpcError(ctx, code, message);
314 			return new McpResponseResult(JsonRpcResponse.errorResponse(id, code, message, JsonMap.of("type", cn(e))));
315 		}
316 	}

319 	 * Enforces SEP-2350 per-operation step-up scopes for a dispatched request (the POST-parse enforcement point).
320 	 *
321 	 * <p>
322 	 * A no-op unless an {@link McpOptions} bean is bound and its {@link McpResourceServerConfig} is
323 	 * {@link McpResourceServerConfig#isEnabled() enabled} with a per-operation scope configured for this operation.  The
324 	 * MCP method &mdash; and thus the operation identity ({@link McpRoutingNames#routingName routing name}) &mdash; is
325 	 * known here (unlike the pre-parse {@code @RestStartCall} bearer gate), so a scoped {@code 403 insufficient_scope}
326 	 * step-up challenge naming exactly this operation's required scopes can be emitted.  Delegates the actual
327 	 * granted-vs-required comparison and challenge construction to {@link McpResourceServerSupport#enforceOperationScopes}.
328 	 *
329 	 * @param method The resolved JSON-RPC method.  Never <jk>null</jk>/empty at the call site.
330 	 * @param params The raw JSON-RPC request params (used to derive the operation name and passed through verbatim as
331 	 * 	the {@link McpOperationContext#params() resolver seam}).
332 	 * @param ctx The per-request bean store (carries the bound {@link McpOptions} and {@link RestRequest}).
333 	 */
334 	private static void enforceStepUpScopes(String method, Object params, BeanStore ctx) {
335 		var opt = ctx.getBean(McpOptions.class).orElse(null);
336 		if (opt == null)
337 			return;
338 		var cfg = opt.getResourceServer();
339 		if (!cfg.isEnabled())
340 			return;
341 		var granted = McpResourceServerSupport.grantedScopes(
342 			ctx.getBean(RestRequest.class).map(RestRequest::getHttpServletRequest).orElse(null));
343 		var rn = McpRoutingNames.routingName(method, params);
344 		var name = isEmpty(rn) ? null : rn;
345 		var rawParams = params instanceof Map<?,?> m ? castParams(m) : Map.<String,Object>of();
346 		McpResourceServerSupport.enforceOperationScopes(cfg, granted, new McpOperationContext(method, name, rawParams));
347 	}

354 	/**
355 	 * Capability-gates, limit-checks, registers, and constructs the streaming publisher for {@code
356 	 * subscriptions/listen}. Bypasses {@link #finalizeResult} / {@link JsonRpcResponse#ok}: the successful
357 	 * result is a {@code Flow.Publisher<SseEvent>}, not a JSON-RPC envelope. A thrown {@link McpException}
358 	 * (for example the over-limit case) is caught by {@link #dispatch}'s existing {@code McpException}
359 	 * handler, which builds the normal JSON-RPC error envelope.
360 	 *
361 	 * @param exchange The transport-neutral request/header accessor, used to enforce the SSE {@code Accept}
362 	 * 	gate below before any broker registration happens.
363 	 * @param id The JSON-RPC request id. Never <jk>null</jk> (this method is never called for a notification).
364 	 * @param params The opaque request params. Must decode to a {@code SubscriptionsListenRequest}.
365 	 * @param config The neutral handler registry, used to auto-derive capabilities when none were set explicitly.
366 	 * @param ctx The per-request bean store, expected to carry a bound {@link McpSubscriptionBroker}
367 	 * 	(Phase 1-3 wiring) and a bound {@link McpOptions}. In production the options bean is normally present —
368 	 * 	{@code McpRestServlet#getMcpOptions()} / {@code McpEndpoint#mcpOptionsBean()} publish it into
369 	 * 	the {@code RestContext} bean store that this request-scoped {@code ctx} wraps as its parent, since the
370 	 * 	neutral core cannot add it directly ({@code McpOptions} is a v2 type). The
371 	 * 	{@code orElseGet(McpSubscriptionsConfig::new)} fallback below is defense-in-depth only (e.g. a
372 	 * 	hand-built {@code ctx} in a direct-dispatch test that omits it).
373 	 * <p>
374 	 * <b>SSE negotiation gate (checked first, before any broker registration):</b> every MCP servlet binding
375 	 * defaults an unqualified request to {@code Accept: application/json}. A {@code subscriptions/listen}
376 	 * call that never negotiated {@code text/event-stream} would still register with the broker and hand
377 	 * back a real {@code Flow.Publisher}, but {@code ReactiveResponseProcessor} would then render it in its
378 	 * BUFFER shape (unbounded {@code request(Long.MAX_VALUE)} demand, collected into a list that is only
379 	 * flushed once the publisher completes — which, for a live subscription, is never, until the connection
380 	 * itself drops). Because this dead-client cleanup is write-failure-driven (see the class javadoc), a
381 	 * BUFFER-shape stream never even attempts a write, so the broker slot, the pump thread, and the
382 	 * heartbeat executor all leak permanently, and any caller can trigger this for free. Checking {@code
383 	 * exchange.header("Accept")} here, before {@link McpSubscriptionBroker#registerIfUnder} is ever called,
384 	 * closes that hole at zero cost to a well-behaved caller (see {@code McpClient#listen} /
385 	 * {@code AbstractMcpClient#openEventStream}, which already always negotiate SSE on this call).
386 	 *
387 	 * <p>
388 	 * {@code maxConcurrentSubscriptions} is enforced as a hard cap, not an advisory one: the admission check
389 	 * and the registration are one atomic step via {@link McpSubscriptionBroker#registerIfUnder}, so
390 	 * concurrent {@code subscriptions/listen} calls against the same broker (e.g. a custom broker explicitly
391 	 * shared across multiple bindings via {@link McpOptions#setSubscriptionBroker}) cannot all observe room
392 	 * under the cap and all register, which a separate {@code activeCount()} check followed by a separate
393 	 * {@code register(...)} call would allow.
394 	 *
395 	 * <p>
396 	 * The broker registers under a fresh {@link UUID}, never under the client-supplied JSON-RPC request
397 	 * {@code id}: the broker's registry is process-wide (see {@link McpEndpoint#subscriptionBroker()}), so
398 	 * two unrelated concurrent listens that happen to share the same client id would otherwise evict each
399 	 * other's stream (the second {@code register(...)} call under a shared key closes whichever
400 	 * subscription is already registered there). The client's original {@code id} is passed to
401 	 * {@link SubscriptionsListenPublisher} unchanged and is still what every ack/notification frame on that
402 	 * client's own stream carries as its {@code _meta.subscriptionId} — this only decouples the *internal*
403 	 * registry key from the *wire* id, which stays exactly as the spec expects.
404 	 *
405 	 * @return The {@code Flow.Publisher<SseEvent>} for the newly registered subscription.
406 	 * @throws McpException If the caller never negotiated {@code text/event-stream}, the broker bean is
407 	 * 	missing, or {@code maxConcurrentSubscriptions} is reached.
408 	 */
409 	@SuppressWarnings({
410 		"resource" // subscription's ownership transfers to the returned SubscriptionsListenPublisher, which closes it in every terminal path (cancel(), terminateWithError(), checkIdle(), run()'s completion - see its own @Owning-suppressed field). Eclipse JDT @Owning warning is by design; it surfaces at the return statement, so suppress at method scope.
411 	})
412 	private Flow.Publisher<SseEvent> dispatchSubscriptionsListen(McpExchange exchange, Object id, Object params, McpServerConfig config, BeanStore ctx) {
413 		requireEventStreamAccept(exchange);
414 		var request = Json.to(Json.of(params), SubscriptionsListenRequest.class);
415 		var caps = discoverCapabilities(config);
416 		var honoredFilter = SubscriptionCapabilityGate.honor(request.getNotifications(), caps);
417 
418 		var broker = ctx.getBean(McpSubscriptionBroker.class)
419 			.orElseThrow(() -> new McpException(CODE_INTERNAL_ERROR, "McpSubscriptionBroker not bound in BeanStore"));
420 		var subscriptionsConfig = ctx.getBean(McpOptions.class).map(McpOptions::getSubscriptions).orElseGet(McpSubscriptionsConfig::new);
421 		// registerIfUnder admission-checks and registers as one atomic step (hard cap, not advisory) — a
422 		// separate activeCount() check followed by a separate register() call would leave a TOCTOU window
423 		// open for concurrent listen requests racing the same per-binding broker (see
424 		// McpOptions#resolveSubscriptionBroker()). The registry key is a server-minted UUID, deliberately
425 		// NOT String.valueOf(id): see the javadoc above.
426 		var subscription = broker.registerIfUnder(subscriptionsConfig.getMaxConcurrentSubscriptions(), UUID.randomUUID().toString(), honoredFilter)
427 			.orElseThrow(() -> new McpException(CODE_TOO_MANY_SUBSCRIPTIONS, "Maximum concurrent subscriptions exceeded"));
428 		var honoredWireFilter = SubscriptionCapabilityGate.toWireFilter(honoredFilter);
429 		return new SubscriptionsListenPublisher(id, honoredWireFilter, subscription, subscriptionsConfig.getHeartbeatIntervalMs(), subscriptionsConfig.getIdleTimeoutMs());
430 	}

434 	 * before any broker registration happens (see {@link #dispatchSubscriptionsListen}'s javadoc).
435 	 *
436 	 * <p>
437 	 * A bare {@code Accept: * / *} (every media type acceptable) is rejected exactly like an absent
438 	 * {@code Accept} header: this endpoint requires the caller to have explicitly negotiated
439 	 * {@code text/event-stream}, not merely to be willing to accept it among everything else. Without this,
440 	 * a wildcard-accepting request would fall into {@code ReactiveResponseProcessor}'s non-streaming BUFFER
441 	 * shape (the same permanent-leak hole C1 closes for a missing header), rather than the STREAM shape
442 	 * this method exists to require. A well-behaved caller is unaffected: both {@code McpClient#listen} and
443 	 * the underlying {@code AbstractMcpClient#openEventStream} already always send the explicit
444 	 * {@code text/event-stream} value on this call.
445 	 *
446 	 * @param exchange The transport-neutral request/header accessor. Must not be <jk>null</jk>.
447 	 * @throws McpException If the {@code Accept} header is absent or does not contain {@code text/event-stream}
448 	 * 	(case-insensitive, substring match — a real client's {@code Accept} header is commonly a
449 	 * 	comma-separated list of media-type ranges, not the bare value alone).
450 	 */
451 	private static void requireEventStreamAccept(McpExchange exchange) {
452 		var accept = exchange.header("Accept");
453 		if (accept == null || ! accept.toLowerCase(Locale.ROOT).contains("text/event-stream"))
454 			throw new McpException(CODE_INVALID_REQUEST, "subscriptions/listen requires Accept: text/event-stream");
455 	}

517 	/**
518 	 * Validates the opaque {@code params._meta} shape directly.
519 	 *
520 	 * <p>
521 	 * The parsed {@link RequestMeta} bean cannot distinguish a malformed {@code clientCapabilities} (a
522 	 * scalar/array) from an absent one — parsing either into {@code ClientCapabilities} raises a
523 	 * generic parse error rather than a validation failure — so metadata shape is checked against
524 	 * the opaque map to produce the exact per-field {@code -32600} messages naming the exact wire path.
525 	 *
526 	 * @param params The opaque request params (the JSON-RPC {@code params} member). Can be <jk>null</jk>.
527 	 */
528 	private static void validateMeta(Object params) {
529 		if (! (params instanceof Map<?,?> p))
530 			throw new McpException(CODE_INVALID_REQUEST, "Request params must be an object");
531 		var meta = p.get(META_KEY);
532 		if (! (meta instanceof Map<?,?> m))
533 			throw new McpException(CODE_INVALID_REQUEST, "Request params._meta must be an object");
534 		var protocolVersion = str(m.get(RequestMeta.KEY_PROTOCOL_VERSION));
535 		if (isEmpty(protocolVersion))
536 			throw new McpException(CODE_INVALID_REQUEST, "Missing required params._meta." + RequestMeta.KEY_PROTOCOL_VERSION);
537 		if (! McpProtocol.VERSION_2026_07_28.equals(protocolVersion))
538 			throw new McpException(CODE_INVALID_REQUEST, "Unsupported protocol version: " + protocolVersion);
539 		var clientInfo = m.get(RequestMeta.KEY_CLIENT_INFO);
540 		if (clientInfo != null && (! (clientInfo instanceof Map<?,?> ci) || isEmpty(str(ci.get("name"))) || isEmpty(str(ci.get("version")))))
541 			throw new McpException(CODE_INVALID_REQUEST, "Malformed params._meta." + RequestMeta.KEY_CLIENT_INFO);
542 		if (! m.containsKey(RequestMeta.KEY_CLIENT_CAPABILITIES) || m.get(RequestMeta.KEY_CLIENT_CAPABILITIES) == null)
543 			throw new McpException(CODE_INVALID_REQUEST, "Missing required params._meta." + RequestMeta.KEY_CLIENT_CAPABILITIES);
544 		if (! (m.get(RequestMeta.KEY_CLIENT_CAPABILITIES) instanceof Map<?,?>))
545 			throw new McpException(CODE_INVALID_REQUEST, "params._meta." + RequestMeta.KEY_CLIENT_CAPABILITIES + " must be an object");
546 	}

581 	 * neutral strings, read here with no OpenTelemetry import &mdash; into <c>meta</c> so a caller can

593 		ctx.getBean(RestRequest.class).ifPresent(request -> {

595 			if (isEmpty(traceparent))
596 				return;
597 			meta.setTraceparent(traceparent);
598 			var tracestate = request.getAttribute(TraceContextResponseProcessor.ATTR_TRACESTATE).as(String.class).orElse(null);
599 			if (! isEmpty(tracestate))

601 			var baggage = request.getAttribute(TraceContextResponseProcessor.ATTR_BAGGAGE).as(String.class).orElse(null);
602 			if (! isEmpty(baggage))

604 		});
605 	}

825 	/**
826 	 * The per-request MRTR resolution result: the {@link WritableBeanStore} a handler is invoked with (always
827 	 * carrying {@link McpMrtrCapabilityContext}, and on a RESUME also {@link McpMrtrResumeContext}) plus the
828 	 * current round decoded from the echoed {@code requestState} (0 on a first-round call), threaded to
829 	 * {@link #pause} so it can increment without a second unseal.
830 	 *
831 	 * <p>
832 	 * {@code store} is typed {@link WritableBeanStore} (rather than the narrower {@link BeanStore}) specifically
833 	 * so call sites can close it in a try-with-resources block: any bean a handler resolves through it is tracked
834 	 * for {@code @PreDestroy} cleanup on <i>this</i> store, not the caller-owned {@code ctx} it wraps (see
835 	 * {@link BasicBeanStore#close()}), so it must be closed once the handler invocation completes.
836 	 *
837 	 * <p>
838 	 * {@code argumentsHash} is the canonical hash of the current request's {@code arguments} (see
839 	 * {@link #argumentsHash(Map)}), computed <i>once</i> here &mdash; before the handler runs &mdash; and threaded
840 	 * to {@link #pause} so the next token seals the hash of what the client actually sent, not a hash recomputed
841 	 * from the live {@code arguments} map after a handler may have mutated it in place.
842 	 *
843 	 * <p>
844 	 * {@code target} is the operation target resolved by {@link #mrtrTarget(String, Map)} (the tool/prompt
845 	 * {@code name} or resource {@code uri}), captured at the same pre-handler point as {@code argumentsHash} and
846 	 * for the same reason: {@link #pause} seals it into the next token's {@link #aad(String, String) AAD}
847 	 * unchanged, so a handler that mutates the live {@code params} map in place cannot shift which
848 	 * tool/prompt/resource the next token is bound to.
849 	 */
850 	private record MrtrContext(WritableBeanStore store, int currentRound, String argumentsHash, String target) {}
851 
852 	/**
853 	 * Builds the {@link BeanStore} an in-scope handler receives and, on a RESUME, validates the echoed
854 	 * {@code requestState} before the handler runs.
855 	 *
856 	 * <p>
857 	 * Always wraps {@code ctx} with a {@link McpMrtrCapabilityContext} so a handler can pre-check the client's
858 	 * advertised {@code elicitation} capability (spec &sect;4). When the request carries a {@code requestState}
859 	 * (a RESUME), it is unsealed under the canonical AAD and validated for integrity, method agreement, expiry,
860 	 * the max-rounds cap, argument-hash agreement (always; see {@link #argumentsHash(Map)}), and replay (only when
861 	 * {@link McpMrtrConfig#getReplayCache()} is configured; see {@link #checkReplay(ReplayCache, String, long)})
862 	 * &mdash; any failure raises the mapped {@link McpException} <i>before</i> the handler is re-invoked &mdash;
863 	 * then the decoded continuation and the client's {@code inputResponses} are exposed via
864 	 * {@link McpMrtrResumeContext}.
865 	 *
866 	 * <p>
867 	 * <b>Check ordering is deliberate.</b> The stateless argument-hash comparison runs <i>before</i> the
868 	 * (stateful) replay {@code checkAndRecord}, so a submission carrying the wrong {@code arguments} is rejected
869 	 * without ever consuming the token's {@code jti} &mdash; an attacker holding only a leaked token (but not the
870 	 * arguments) cannot burn a victim's in-flight resume, and an honest client that sends slightly-wrong arguments
871 	 * does not destroy its own token. The argument hash is also computed <i>once</i> here, before the handler
872 	 * runs, and threaded to {@link #pause} (see {@link MrtrContext}).
873 	 *
874 	 * @param method The in-scope JSON-RPC method. Must not be <jk>null</jk>.
875 	 * @param params The request params map. Must not be <jk>null</jk>.
876 	 * @param ctx The request-scoped bean store to wrap. Must not be <jk>null</jk>.
877 	 * @return The resolved MRTR context. Never <jk>null</jk>.
878 	 */
879 	@SuppressWarnings({
880 		"resource" // wrapped (both branches) is @Owning: ownership transfers to the returned MrtrContext, which the caller closes via mrtr.store() in try-with-resources (see MrtrContext's javadoc above). 881 	})
882 	private MrtrContext resolveMrtrContext(String method, Map<String,Object> params, BeanStore ctx) {
883 		// Computed once, up front, from the CURRENT request's arguments: reused both for the RESUME-time
884 		// comparison below and (threaded through MrtrContext) for the hash pause(...) seals into the next token,
885 		// so a handler that mutates the live arguments map in place cannot make the next token seal a hash the
886 		// client can no longer reproduce. Also applies the bounded-traversal + canonicalization guards (see
887 		// argumentsHash), so structurally-hostile arguments become a -32602 here on both first-round and RESUME.
888 		var argumentsHash = argumentsHash(McpParamUtils.mapParam(params, PARAM_ARGUMENTS));
889 		// Captured once, up front, alongside argumentsHash and for the same reason: threaded through MrtrContext
890 		// to pause(...) so the AAD sealed into the next token binds the target actually resolved for THIS
891 		// request, immune to a handler mutating params in place before pause(...) runs (see MrtrContext's
892 		// javadoc above).
893 		var target = mrtrTarget(method, params);
894 		var requestState = McpParamUtils.strParam(params, "requestState");
895 		if (requestState == null) {
896 			var wrapped = new BasicBeanStore(ctx)
897 				.addBean(McpMrtrCapabilityContext.class, new McpMrtrCapabilityContext(clientElicitationSupported(params)));
898 			return new MrtrContext(wrapped, 0, argumentsHash, target);
899 		}
900 		// Validate the echoed requestState before constructing the BasicBeanStore below: every failure here
901 		// throws, and a BasicBeanStore built earlier would never reach a caller's try-with-resources to be
902 		// closed (java:S2095). Deferring construction until validation succeeds means no path leaks it.
903 		var sealed = mrtrConfig.getCodec().unseal(requestState, aad(method, target), principal(ctx))
904 			.orElseThrow(() -> new McpException(CODE_INVALID_PARAMS, "Invalid or tampered requestState"));
905 		if (! method.equals(sealed.method()))
906 			throw new McpException(CODE_INVALID_PARAMS, "requestState method mismatch");
907 		if (sealed.expiresAtMs() <= System.currentTimeMillis())
908 			throw new McpException(CODE_REQUEST_STATE_EXPIRED, "requestState has expired");
909 		if (sealed.round() >= mrtrConfig.getMaxRounds())
910 			// Client-facing message deliberately omits the configured cap value so server config is not leaked.
911 			throw new McpException(CODE_MAX_ROUNDS_EXCEEDED, "Max MRTR rounds exceeded");
912 		// Stateless argument-hash agreement BEFORE the stateful replay record (see the ordering note above). A
913 		// null sealed hash (e.g. a custom codec that dropped the field) fails closed as an ordinary mismatch
914 		// here; a malformed (non-base64url) sealed hash is a codec-contract violation and argumentsMatch itself
915 		// throws CODE_INVALID_PARAMS for it (mirroring the null-jti fail-closed check below), rather than
916 		// reaching this mismatch branch.
917 		if (! argumentsMatch(argumentsHash, sealed.argumentsHash()))
918 			throw new McpException(CODE_REQUEST_STATE_ARGUMENTS_MISMATCH, "resumed arguments do not match the original request");
919 		var replayCache = mrtrConfig.getReplayCache();
920 		if (replayCache != null) {
921 			// A null/empty jti reaching the cache is a codec/contract violation (a custom RequestStateCodec that
922 			// dropped the field, or a pre-jti in-flight token during a rolling deploy), NOT a store outage: fail
923 			// CLOSED here rather than letting a downstream NPE be swallowed by checkReplay's fail-OPEN catch,
924 			// which would silently disable replay protection.
925 			if (isEmpty(sealed.jti()))
926 				throw new McpException(CODE_INVALID_PARAMS, "Invalid or tampered requestState");
927 			if (! checkReplay(replayCache, sealed.jti(), sealed.expiresAtMs()))
928 				// Client-facing message deliberately omits any cache/backend detail, mirroring the max-rounds
929 				// message's "don't leak server config" caution above.
930 				throw new McpException(CODE_REQUEST_STATE_REPLAYED, "requestState has already been used");
931 		}
932 		var inputResponses = McpParamUtils.mapParam(params, "inputResponses");
933 		var wrapped = new BasicBeanStore(ctx)
934 			.addBean(McpMrtrCapabilityContext.class, new McpMrtrCapabilityContext(clientElicitationSupported(params)))
935 			.addBean(McpMrtrResumeContext.class, new McpMrtrResumeContext(sealed.continuation(), inputResponses));
936 		return new MrtrContext(wrapped, sealed.round(), argumentsHash, target);
937 	}

968 				"Client does not advertise the elicitation capability required for input_required");
969 		var state = new McpRequestState(signal.getContinuation(), method, currentRound + 1, System.currentTimeMillis() + mrtrConfig.getTtlMs(),

1045 	/**
1046 	 * Computes the canonical hash of a request's {@code arguments}, sealed at PAUSE time and re-verified at
1047 	 * RESUME time (see {@code McpRequestState#argumentsHash()}).
1048 	 *
1049 	 * <p>
1050 	 * First runs the module's shared bounded-traversal guard ({@link JsonValueSafety#check(Object, String)}:
1051 	 * {@code MAX_DEPTH=64}, {@code MAX_NODES}, a traversal-time deadline) so structurally-hostile {@code arguments}
1052 	 * (excessive depth or node count &mdash; reachable via {@code prompts/get}, which has no other input-shape
1053 	 * guard, and schemaless {@code tools/call}) become a deterministic {@link #CODE_INVALID_PARAMS} rejection on
1054 	 * BOTH PAUSE and RESUME, rather than a silent hash collision (the JCS canonicalizer's inherited depth limit
1055 	 * would otherwise truncate over-depth nodes to {@code null}). It then canonicalizes {@code arguments} per RFC
1056 	 * 8785 (JSON Canonicalization Scheme) using an <b>unbounded-depth</b> serializer (see {@link #JCS}), SHA-256s
1057 	 * the canonical UTF-8 bytes, and base64url-encodes the digest (mirroring {@link EphemeralKeyProvider}'s
1058 	 * existing {@code keyId} encoding convention). An absent or empty {@code arguments} map canonicalizes to the
1059 	 * empty JSON object {@code "{}"}, giving a single, consistent sentinel hash for calls that take no arguments
1060 	 * (for example {@code resources/read}'s exact-path branch), so PAUSE and RESUME agree even when there is
1061 	 * nothing to hash.
1062 	 *
1063 	 * <p>
1064 	 * By design, RFC 8785 canonical equivalences are treated as equal &mdash; for example {@code 1} and
1065 	 * {@code 1.0} canonicalize identically, so they hash the same and a resume switching between them is not a
1066 	 * mismatch. A canonicalization failure on a hostile-but-syntactically-legal value (a non-finite number such
1067 	 * as {@code 1e999}, a lone UTF-16 surrogate, or a {@code BigDecimal}/{@code BigInteger} outside IEEE-754
1068 	 * range) is mapped to {@link #CODE_INVALID_PARAMS} rather than surfacing as a generic internal error.
1069 	 *
1070 	 * <p>
1071 	 * Package-visible so tests can construct {@link McpRequestState} fixtures whose {@code argumentsHash} agrees
1072 	 * with a given (or absent) {@code arguments} map without duplicating this derivation.
1073 	 *
1074 	 * @param arguments The request's {@code arguments} map, as returned by
1075 	 * 	{@code McpParamUtils.mapParam(params, "arguments")}. Never <jk>null</jk> (empty when absent).
1076 	 * @return The canonical hash. Never <jk>null</jk>.
1077 	 * @throws McpException ({@link #CODE_INVALID_PARAMS}) if {@code arguments} violates the structural safety
1078 	 * 	limits or cannot be canonicalized.
1079 	 */
1080 	static String argumentsHash(Map<String,Object> arguments) {
1081 		try {
1082 			JsonValueSafety.check(arguments, PARAM_ARGUMENTS);
1083 		} catch (IllegalArgumentException e) {
1084 			throw new McpException(CODE_INVALID_PARAMS, e.getMessage());
1085 		}
1086 		String canonical;
1087 		try {
1088 			canonical = JCS.write(arguments);
1089 		} catch (SerializeException e) {
1090 			// Message deliberately generic (no echo of the offending value), mirroring the other MRTR "don't leak"
1091 			// messages: a hostile-but-legal input (non-finite number, lone surrogate, out-of-IEEE BigDecimal) must
1092 			// not surface as a -32603 internal error.
1093 			throw new McpException(CODE_INVALID_PARAMS, "Invalid arguments: not canonicalizable");
1094 		}
1095 		return B64URL.encodeToString(sha256(canonical.getBytes(StandardCharsets.UTF_8)));
1096 	}

1099 	 * Constant-time comparison of the current request's {@code argumentsHash} against the one sealed into the
1100 	 * echoed token. Both operands are base64url of a SHA-256 digest; they are decoded to their raw digest bytes
1101 	 * and compared with {@link MessageDigest#isEqual(byte[], byte[])}. A <jk>null</jk> sealed hash (for example a
1102 	 * custom codec that dropped the field) fails closed (returns <jk>false</jk>, surfacing as the ordinary
1103 	 * {@link #CODE_REQUEST_STATE_ARGUMENTS_MISMATCH} at the call site).
1104 	 *
1105 	 * <p>
1106 	 * A non-<jk>null</jk> sealed hash that is not valid base64url (for example a custom codec that computed
1107 	 * {@code argumentsHash} some other way) is a codec-contract violation rather than an ordinary mismatch, and
1108 	 * fails closed the same way the null-{@code jti} codec-contract violation does at the call site: a thrown
1109 	 * {@link McpException} ({@link #CODE_INVALID_PARAMS}), not a propagated {@link IllegalArgumentException}
1110 	 * from {@link Base64.Decoder#decode(String)} surfacing as a generic {@link #CODE_INTERNAL_ERROR}.
1111 	 *
1112 	 * @param currentHash The hash recomputed from the current request's {@code arguments}. Never <jk>null</jk>.
1113 	 * @param sealedHash The hash sealed into the echoed token, or <jk>null</jk>.
1114 	 * @return <jk>true</jk> if the two hashes match.
1115 	 * @throws McpException ({@link #CODE_INVALID_PARAMS}) if {@code sealedHash} is non-<jk>null</jk> but not
1116 	 * 	valid base64url.
1117 	 */
1118 	private static boolean argumentsMatch(String currentHash, String sealedHash) {
1119 		if (sealedHash == null)
1120 			return false;
1121 		byte[] sealedBytes;
1122 		try {
1123 			sealedBytes = B64URL_DECODER.decode(sealedHash);
1124 		} catch (IllegalArgumentException e) {
1125 			// Same fail-closed contract as the null-jti check below: an untrusted or misbehaving codec must
1126 			// never turn into a raw exception bubbling up as -32603 (see resolveMrtrContext's ordering note
1127 			// on why this stateless check runs before the stateful replay record).
1128 			throw new McpException(CODE_INVALID_PARAMS, "Invalid or tampered requestState");
1129 		}
1130 		return MessageDigest.isEqual(B64URL_DECODER.decode(currentHash), sealedBytes);
1131 	}

1258 		 * URI stays an attribute, never a span-name suffix), or the exact method name otherwise.
1259 		 */
1260 		private static TraceOperation buildOperation(JsonRpcRequest req) {
1261 			var method = req.getMethod();
1262 			if (isEmpty(method))
1263 				return TraceOperation.DEFAULT;
1264 			var meta = McpWire.metaMapOrEmpty(req.getParams());
1265 			var attrs = new LinkedHashMap<String,String>();
1266 			attrs.put(TraceOperation.ATTR_MCP_METHOD_NAME, method);
1267 			var protocolVersion = str(meta.get(RequestMeta.KEY_PROTOCOL_VERSION));
1268 			if (! isEmpty(protocolVersion))
1269 				attrs.put(TraceOperation.ATTR_MCP_PROTOCOL_VERSION, protocolVersion);
1270 			if (nn(req.getId()))
1271 				attrs.put(TraceOperation.ATTR_JSONRPC_REQUEST_ID, String.valueOf(req.getId()));
1272 
1273 			var spanName = method;
1274 			switch (method) {
1275 				case McpMethods.TOOLS_CALL -> {
1276 					var name = McpRoutingNames.routingName(method, req.getParams());
1277 					if (! isEmpty(name)) {
1278 						attrs.put(TraceOperation.ATTR_GEN_AI_TOOL_NAME, name);
1279 						spanName = method + " " + name;
1280 					}
1281 					attrs.put(TraceOperation.ATTR_GEN_AI_OPERATION_NAME, "execute_tool");
1282 				}
1283 				case McpMethods.PROMPTS_GET -> {
1284 					var name = McpRoutingNames.routingName(method, req.getParams());
1285 					if (! isEmpty(name)) {
1286 						attrs.put(TraceOperation.ATTR_GEN_AI_PROMPT_NAME, name);
1287 						spanName = method + " " + name;
1288 					}
1289 				}
1290 				case McpMethods.RESOURCES_READ -> {
1291 					var uri = McpRoutingNames.routingName(method, req.getParams());
1292 					if (! isEmpty(uri))
1293 						attrs.put(TraceOperation.ATTR_MCP_RESOURCE_URI, uri);
1294 				}
1295 				default -> {
1296 					// Exact method name; no method-specific attributes.
1297 				}
1298 			}
1299 			return TraceOperation.of(spanName, attrs);
1300 		}
1301 	}

1305 	 * precedence over the equivalent HTTP header, for any key &mdash; in practice the bare W3C
1306 	 * {@code traceparent} / {@code tracestate} / {@code baggage} keys, since those are the only names
1307 	 * both sides share: {@code lookup(key) = params._meta[key] if present, otherwise the HTTP header of
1308 	 * the same name}.
