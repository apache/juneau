30  /**
31   * The {@code Flow.Publisher<SseEvent>} returned by v2 {@code McpRevision.dispatch(...)} for
32   * {@code subscriptions/listen}. Emits the acknowledged frame first, then drains {@code subscription},
33   * mapping each {@code McpChangeEvent} to its wire notification, until graceful close produces a terminal
34   * {@code SubscriptionsListenResult} frame followed by {@code onComplete()}. Backpressure is one frame at a
35   * time: exactly one {@code onNext} is emitted per unit of {@code request(n)} demand. A periodic named
36   * {@code "ping"} heartbeat event (no data) is interleaved on the same demand-gated path at
37   * {@code heartbeatIntervalMs} (disabled when {@code heartbeatIntervalMs <= 0}), standing in for
38   * {@code SseHeartbeat} (see this task's design note: {@code SseHeartbeat} is {@link java.io.Writer}-based
39   * and has no integration point with a {@code Flow.Publisher}), so a dead connection's failed write surfaces
40   * identically to a failed write of any other frame.
41   *
42   * <p>
43   * A separate idle watchdog bounds how long the stream may go without any outbound activity (a delivered
44   * change event, the initial acknowledged frame, or a heartbeat) before it is terminated gracefully:
45   * {@code idleTimeoutMs} (disabled when {@code idleTimeoutMs <= 0}, which blocks {@link McpSubscription#take()}
46   * indefinitely as before). Every such activity resets the idle clock. Because a heartbeat counts as
47   * activity, configuring {@code heartbeatIntervalMs > 0} with {@code heartbeatIntervalMs < idleTimeoutMs} keeps
48   * an otherwise-quiet-but-healthy stream alive indefinitely; the idle timeout only bites once
49   * {@code heartbeatIntervalMs} is disabled (or configured larger than {@code idleTimeoutMs}), where it is the
50   * sole guard against a connection that receives no real change events. On expiry the watchdog closes
51   * {@code subscription}, which interrupts (or short-circuits) the pump's blocked {@link McpSubscription#take()}
52   * call, driving the same graceful {@code awaitDemandThenComplete()} shutdown as any other closed-subscription
53   * completion. If the pump is instead blocked waiting on {@code request(n)} demand rather than on
54   * {@code take()} at expiry, the watchdog interrupts that wait directly (see {@code Pump.checkIdle()}'s
55   * javadoc) so it cannot outlive the subscription it was waiting to drain.
56   *
57   * <p>
58   * Supports exactly one {@link #subscribe(Flow.Subscriber)} call; a second is rejected with
59   * {@code onError(IllegalStateException)} rather than spinning a second {@code Pump} over the same shared
60   * {@code subscription}.
61   */
62  final class SubscriptionsListenPublisher implements Flow.Publisher<SseEvent> {

108 	public void subscribe(Flow.Subscriber<? super SseEvent> subscriber) {
109 		if (!subscribed.compareAndSet(false, true)) {
110 			subscriber.onSubscribe(NoopSubscription.INSTANCE);

118 	/**
119 	 * Bridges {@link McpSubscription#take()}'s blocking queue to {@code Flow.Subscriber} calls on a
120 	 * single dedicated daemon thread, honoring one-at-a-time backpressure via a {@link Semaphore} of
121 	 * outstanding {@code request(n)} credits, plus a second daemon-thread {@link ScheduledExecutorService}
122 	 * that periodically contends for the same demand credits to interleave heartbeat frames. A thread per
123 	 * active subscription is acceptable: {@code maxConcurrentSubscriptions} bounds the count. All
124 	 * {@code onNext} calls (main pump and heartbeat) are serialized through {@code emitLock} so they never
125 	 * race per the {@code Flow.Subscriber} contract.
126 	 *
127 	 * <p>
128 	 * Every terminal path (graceful completion, {@code cancel()}, an invalid {@code request(n)}, or an
129 	 * unexpected {@link RuntimeException}) is idempotency-guarded by the same {@code terminated} flag,
130 	 * closes {@code subscription}, and shuts down both {@code heartbeatExecutor} and
131 	 * {@code idleWatchdogExecutor} via the shared {@code shutdownExecutors()} helper (always called under
132 	 * {@code startupLock}). {@code run()} additionally calls it directly on every one of its own exits, not
133 	 * just via {@code awaitDemandThenComplete()}/{@code terminateWithError()} — belt-and-suspenders given
134 	 * {@code shutdownNow()}'s idempotence, so no path, present or future, can leak either executor.
135 	 */
136 	private final class Pump implements Flow.Subscription {
137 		private final Flow.Subscriber<? super SseEvent> subscriber;
138 		private final Semaphore demand = new Semaphore(0);
139 		private final AtomicBoolean started = new AtomicBoolean();
140 		private final Object emitLock = new Object();
141 		/**
142 		 * Serializes {@code request(n)}'s worker/heartbeat-executor startup against {@code cancel()}'s
143 		 * teardown, so {@code cancel()} can never observe the half-initialized window between
144 		 * {@code worker.start()} and {@code heartbeatExecutor}'s field assignment (which would otherwise let
145 		 * the just-created executor escape unshut-down once {@code request(n)}'s thread resumed and
146 		 * published it after {@code cancel()} had already looked and found nothing to shut down).
147 		 */
148 		private final Object startupLock = new Object();
149 		private volatile boolean terminated;
150 		private final AtomicReference<Thread> worker = new AtomicReference<>();
151 		private final AtomicReference<ScheduledExecutorService> heartbeatExecutor = new AtomicReference<>();
152 		private final AtomicReference<ScheduledExecutorService> idleWatchdogExecutor = new AtomicReference<>();
153 		/**
154 		 * Nanosecond-clock ({@link System#nanoTime()}) activity stamp, deliberately not wall-clock
155 		 * ({@link System#currentTimeMillis()}): a wall-clock stamp is vulnerable to NTP/clock-step
156 		 * adjustments — a backward step would defer an overdue idle timeout indefinitely, and a forward
157 		 * step would reap an otherwise-healthy stream early. {@link System#nanoTime()} has no such
158 		 * relation to wall-clock time and is immune to both.
159 		 */
160 		private final AtomicLong lastActivityAtNanos = new AtomicLong();
161 		private final long idleTimeoutNanos = TimeUnit.MILLISECONDS.toNanos(idleTimeoutMs);
162 		/**
163 		 * {@code true} exactly while {@code worker} is blocked in {@code awaitDemandThenEmit}'s
164 		 * <b>unbounded</b> {@code demand.acquire()} — the one wait in this class with no timeout of its
165 		 * own. Lets {@code checkIdle()} tell that specific wait apart from {@code awaitDemandThenComplete}'s
166 		 * already-bounded {@code tryAcquire}, which needs no help from the watchdog since it always gives
167 		 * up on its own; see {@code checkIdle()}'s javadoc for why interrupting indiscriminately would be
168 		 * unsafe.
169 		 */
170 		private volatile boolean parkedOnDemand;
171 
172 		Pump(Flow.Subscriber<? super SseEvent> subscriber) {
173 			this.subscriber = subscriber;
174 		}

176 		@Override
177 		public void request(long n) {
178 			if (n <= 0) {
179 				terminateWithError(new IllegalArgumentException("request(n) must be positive, but was " + n));
180 				return;
181 			}
182 			if (terminated)
183 				return;
184 			// Clamps a single request(n) call to Integer.MAX_VALUE permits, but does not defend against
185 			// cumulative overflow across repeated request() calls (Semaphore.release() would itself throw
186 			// first if outstanding permits ever approached Integer.MAX_VALUE). Unreachable under this
187 			// stream's own driver: Pump consumes one permit per frame/heartbeat, and
188 			// ReactiveResponseProcessor's StreamingSubscriber only ever requests one at a time.
189 			demand.release(n > Integer.MAX_VALUE ? Integer.MAX_VALUE : (int) n);
190 			if (started.compareAndSet(false, true)) {
191 				synchronized (startupLock) {
192 					if (terminated)
193 						return; // cancel() already ran (holding the same lock) and found nothing to shut down
194 					lastActivityAtNanos.set(System.nanoTime());
195 					var w = new Thread(this::run, "mcp-subscriptions-listen-" + listenId);
196 					w.setDaemon(true);
197 					worker.set(w);
198 					w.start();
199 					if (heartbeatIntervalMs > 0) {
200 						var executor = Executors.newSingleThreadScheduledExecutor(r -> {
201 							var t = new Thread(r, "mcp-subscriptions-heartbeat-" + listenId);
202 							t.setDaemon(true);
203 							return t;
204 						});
205 						executor.scheduleAtFixedRate(this::sendHeartbeat, heartbeatIntervalMs, heartbeatIntervalMs, TimeUnit.MILLISECONDS);
206 						heartbeatExecutor.set(executor);
207 					}
208 					if (idleTimeoutMs > 0) {
209 						var checkPeriodMs = Math.max(idleTimeoutMs / 4, 10L);
210 						var executor = Executors.newSingleThreadScheduledExecutor(r -> {
211 							var t = new Thread(r, "mcp-subscriptions-idle-" + listenId);
212 							t.setDaemon(true);
213 							return t;
214 						});
215 						executor.scheduleAtFixedRate(this::checkIdle, checkPeriodMs, checkPeriodMs, TimeUnit.MILLISECONDS);
216 						idleWatchdogExecutor.set(executor);
217 					}
218 				}
219 			}
220 		}
221 
222 		/**
223 		 * Polled by {@code idleWatchdogExecutor}: if no activity (the initial acknowledged frame, a change
224 		 * event, or a heartbeat) has been emitted within {@code idleTimeoutMs}, closes {@code subscription}
225 		 * to drive the pump's blocked (or next) {@link McpSubscription#take()} into the existing graceful
226 		 * {@code awaitDemandThenComplete()} shutdown path. A no-op once already terminated.
227 		 *
228 		 * <p>
229 		 * {@code subscription.close()} alone only wakes {@code worker} if it is currently blocked inside
230 		 * {@link McpSubscription#take()}. It does nothing for a {@code worker} instead parked in
231 		 * {@code awaitDemandThenEmit}'s unbounded {@code demand.acquire()} (already holding a taken event,
232 		 * but never granted the {@code request(n)} needed to emit it) — that wait would otherwise block
233 		 * forever past an idle timeout. This method additionally interrupts {@code worker} directly, but
234 		 * <b>only</b> when {@code parkedOnDemand} is set, i.e. only for that specific unbounded wait.
235 		 * Deliberately never interrupts unconditionally: an interrupt landing during
236 		 * {@code awaitDemandThenComplete}'s own (already-bounded, and by then not {@code parkedOnDemand})
237 		 * {@code tryAcquire} would reintroduce exactly the lost-terminal-frame hazard documented on
238 		 * {@code run()}'s {@code catch (InterruptedException e)} block — that bounded wait needs no help
239 		 * from this watchdog, since it always gives up on its own within {@code completionAwaitTimeoutMs()}.
240 		 */
241 		private void checkIdle() {
242 			if (terminated)
243 				return;
244 			if (System.nanoTime() - lastActivityAtNanos.get() < idleTimeoutNanos)
245 				return;
246 			subscription.close();
247 			if (parkedOnDemand) {
248 				var w = worker.get();
249 				if (w != null)
250 					w.interrupt();
251 			}
252 		}
253 
254 		/**
255 		 * Shuts down both {@code heartbeatExecutor} and {@code idleWatchdogExecutor}, guarded by
256 		 * {@code startupLock} so this can never observe the half-initialized window between
257 		 * {@code worker.start()} and either executor field's assignment (see {@code startupLock}'s
258 		 * javadoc). Called from every terminal path ({@code cancel()}, {@code terminateWithError()}, and
259 		 * every {@code run()} exit via {@code awaitDemandThenComplete()} or directly) so none of them can
260 		 * leak either executor. {@link ScheduledExecutorService#shutdownNow()} on an already-shut-down
261 		 * executor is a documented no-op, so calling this more than once for the same {@code Pump} (e.g.
262 		 * once from an early {@code run()} exit and again from whichever terminal path drove it there) is
263 		 * always safe.
264 		 */
265 		private void shutdownExecutors() {
266 			synchronized (startupLock) {
267 				var executor = heartbeatExecutor.get();
268 				if (executor != null)
269 					executor.shutdownNow();
270 				var idle = idleWatchdogExecutor.get();
271 				if (idle != null)
272 					idle.shutdownNow();
273 			}
274 		}

276 		@Override
277 		public void cancel() {
278 			// shutdownExecutors() re-acquires startupLock, which is safe: Java's intrinsic locks are
279 			// reentrant, so nesting it inside this already-held block keeps the flag write and the
280 			// executor read-and-shutdown in one atomic critical section, exactly as before this method
281 			// was extracted.
282 			synchronized (startupLock) {
283 				terminated = true;
284 				shutdownExecutors();
285 			}
286 			var w = worker.get();
287 			if (w != null)
288 				w.interrupt();
289 			subscription.close();
290 		}

292 		/**
293 		 * Idempotently routes an unexpected failure (frame construction or a misbehaving
294 		 * {@code subscriber.onNext(...)}) to a single {@code onError}, then performs the same cleanup as
295 		 * every other terminal path.
296 		 *
297 	 * <p>
298 	 * The {@code worker} interrupt and {@code shutdownExecutors()} call are its own {@code startupLock}
299 	 * critical section — mirroring {@code cancel()} — so it can never observe the same half-initialized
300 	 * startup window {@code cancel()} was fixed against: without this, a failure landing between
301 	 * {@code worker.start()} and either executor field's assignment would read a still-{@code null}
302 	 * executor here, then have {@code request(n)}'s startup block publish it moments later with nothing left
303 	 * to ever shut it down. That section is deliberately NOT nested inside the {@code emitLock} section
304 	 * above (each is acquired and released independently) to avoid pairing with {@code
305 	 * request()}/{@code cancel()}'s {@code startupLock}-then-{@code emitLock} risk in the opposite order
306 	 * (neither of which actually nests {@code emitLock} today, but keeping this method's two sections
307 	 * separate keeps that true regardless).
308 	 */
309 		private void terminateWithError(Throwable t) {
310 			synchronized (emitLock) {
311 				if (terminated)
312 					return;
313 				terminated = true;
314 				subscriber.onError(t);
315 			}
316 			synchronized (startupLock) {
317 				var w = worker.get();
318 				if (w != null && w != Thread.currentThread())
319 					w.interrupt();
320 				shutdownExecutors();
321 			}
322 			subscription.close();
323 		}
324 
325 		private void sendHeartbeat() {
326 			if (terminated || !demand.tryAcquire())
327 				return;
328 			Throwable failure = null;
329 			synchronized (emitLock) {
330 				if (terminated) {
331 					demand.release();
332 					return;
333 				}
334 				try {
335 					// A fresh SseEvent per send: this is never mutated afterward, so sharing one instance
336 					// across sends would in fact be safe too, but constructing a new one keeps that
337 					// invariant from ever needing to be relied upon.
338 					subscriber.onNext(new SseEvent().setEvent(HEARTBEAT_EVENT_NAME));
339 					lastActivityAtNanos.set(System.nanoTime());
340 				} catch (RuntimeException e) {
341 					failure = e;
342 				}
343 			}
344 			// Deliberately called after releasing emitLock above: terminateWithError() takes emitLock itself,
345 			// then separately takes startupLock, and must never do so while still nested inside a caller's
346 			// own held emitLock (see terminateWithError()'s javadoc on lock ordering).
347 			if (failure != null)
348 				terminateWithError(failure);
349 		}

351 		/**
352 		 * Every exit from this method (including both {@code awaitDemandThenEmit} early returns below)
353 		 * calls {@code shutdownExecutors()} directly, in addition to whatever cleanup
354 		 * {@code awaitDemandThenComplete}/{@code terminateWithError} already perform on their own paths —
355 		 * redundant on the two in-tree {@link McpSubscription} implementations today (every real path here
356 		 * already reaches one of those two methods, or a {@code cancel()} that shut down first), but
357 		 * hardens {@code run()} itself against ever adding a future exit path that forgets to.
358 		 * {@code shutdownExecutors()}'s {@code shutdownNow()} calls are idempotent, so the redundancy is free.
359 		 */
360 		private void run() {
361 			try {
362 				if (!awaitDemandThenEmit(this::acknowledgedFrame)) {
363 					shutdownExecutors();
364 					return;
365 				}
366 				while (!terminated) {
367 					var event = takeNextEvent();
368 					if (event == null)
369 						return; // interrupted; takeNextEvent() already ran awaitDemandThenComplete() and restored the interrupt flag
370 					if (subscription.isClosed()) {
371 						awaitDemandThenComplete();
372 						return;
373 					}
374 					if (!awaitDemandThenEmit(() -> changeFrame(event))) {
375 						shutdownExecutors();
376 						return;
377 					}
378 				}
379 			} catch (RuntimeException e) {
380 				// Catches anything awaitDemandThenEmit() doesn't handle itself (e.g. an unexpected throw from
381 				// subscription.take()/isClosed()), in addition to propagating frame-construction/onNext
382 				// failures from awaitDemandThenEmit's own emitLock block once its monitor unwinds.
383 				terminateWithError(e);
384 			}
385 		}
386 
387 		/**
388 		 * Drains one event from {@code subscription}, or performs the on-interrupt shutdown sequence and
389 		 * returns <jk>null</jk> if the wait was interrupted (including by a concurrent {@code close()}).
390 		 *
391 		 * <p>
392 		 * Deliberately does NOT restore the interrupt flag before calling {@code awaitDemandThenComplete()}:
393 		 * that call still needs to perform a legitimate blocking wait on this same thread to await permission
394 		 * to emit the terminal frame, and a pre-set interrupt flag would make that wait return immediately
395 		 * regardless of real demand, silently dropping the terminal frame. The flag is restored afterward,
396 		 * just before this method returns to the (dying) {@link #run()} caller.
397 		 */
398 		private McpChangeEvent takeNextEvent() {
399 			try {
400 				return subscription.take();
401 			} catch (InterruptedException e) {
402 				awaitDemandThenComplete();
403 				Thread.currentThread().interrupt();
404 				return null;
405 			}
406 		}
407 
408 		private boolean awaitDemandThenEmit(Supplier<SseEvent> frame) {
409 			try {
410 				parkedOnDemand = true;
411 				try {
412 					demand.acquire();
413 				} finally {
414 					parkedOnDemand = false;
415 				}
416 			} catch (InterruptedException e) {
417 				Thread.currentThread().interrupt();
418 				return false;
419 			}
420 			synchronized (emitLock) {
421 				if (terminated)
422 					return false;
423 				subscriber.onNext(frame.get());
424 			}
425 			lastActivityAtNanos.set(System.nanoTime());
426 			return true;
427 		}

429 		/**
430 		 * Bounds the wait for demand to emit the terminal frame, instead of blocking forever if the client
431 		 * stalls (stops calling {@code request(n)}). On timeout (or interruption), resource release wins
432 		 * over guaranteeing the terminal frame's delivery: cleanup still runs, but the subscriber may never
433 		 * see {@code onComplete()}.
434 		 */
435 		private void awaitDemandThenComplete() {
436 			if (terminated) {
437 				// Another terminal path (cancel()/terminateWithError(), or a concurrent call into this same
438 				// method) already won the race to set `terminated`; that path is responsible for its own
439 				// shutdownExecutors() call, but this redundant one is free (see run()'s javadoc) and closes
440 				// off any future path that might reach `terminated == true` here without having done so.
441 				shutdownExecutors();
442 				return;
443 			}
444 			boolean acquired;
445 			try {
446 				acquired = demand.tryAcquire(completionAwaitTimeoutMs(), TimeUnit.MILLISECONDS);
447 			} catch (InterruptedException e) {
448 				Thread.currentThread().interrupt();
449 				acquired = false;
450 			}
451 			synchronized (emitLock) {
452 				if (terminated) {
453 					shutdownExecutors();
454 					return;
455 				}
456 				terminated = true;
457 				if (acquired) {
458 					try {
459 						subscriber.onNext(terminalFrame());
460 						subscriber.onComplete();
461 					} catch (RuntimeException e) {
462 						// Best effort only: `terminated` is already set above, and the Subscriber contract
463 						// forbids following an attempted onComplete with onError, so there is no further
464 						// signal we can deliver here - but the cleanup below must still run regardless.
465 					}
466 				}
467 				// else: a stalled client (no request(n) within the bound) or this thread was interrupted
468 				// while waiting. No Subscriber signal is sent on this path either.
469 			}
470 			// Mirrors cancel()/terminateWithError(): reading and shutting down the executors must happen
471 			// under startupLock too, not just under emitLock above, so this path can never observe the same
472 			// half-initialized window between worker.start() and heartbeatExecutor/idleWatchdogExecutor's
473 			// field assignment (see cancel()'s startupLock javadoc). Without this, a subscription closed and
474 			// immediately drained (demand already available) racing a request(n) call still inside its own
475 			// startupLock section could see both executor fields still null here, then have request(n)
476 			// publish one or both moments later with nothing left to ever shut them down.
477 			shutdownExecutors();
478 		}

480 		private long completionAwaitTimeoutMs() {
481 			return heartbeatIntervalMs > 0 ? heartbeatIntervalMs * COMPLETION_AWAIT_TIMEOUT_MULTIPLIER : DEFAULT_COMPLETION_AWAIT_TIMEOUT_MS;
482 		}

484 		/**
485 		 * Builds the initial acknowledged-subscription frame. Used only from {@link #run()}, so it lives
486 		 * here rather than on the enclosing publisher.
487 		 */
488 		private SseEvent acknowledgedFrame() {
489 			var notification = new SubscriptionsAcknowledgedNotification()
490 				.setNotifications(honoredWireFilter)
491 				.setMeta(new RequestMeta().set(RequestMeta.KEY_SUBSCRIPTION_ID, listenId));
492 			var frame = new JsonRpcRequest()
493 				.setJsonrpc(McpProtocol.JSON_RPC_2_0)
494 				.setId(null)
495 				.setMethod(McpMethods.NOTIFICATIONS_SUBSCRIPTIONS_ACKNOWLEDGED)
496 				.setParams(notification);
497 			return new SseEvent().setData(Json.of(frame));
498 		}

500 		/**
501 		 * Builds the terminal completion frame. Used only from {@link #awaitDemandThenComplete()}, so it

503 		 */
504 		private SseEvent terminalFrame() {
505 			var result = new SubscriptionsListenResult()
506 				.setMeta(new ResultMeta().set(RequestMeta.KEY_SUBSCRIPTION_ID, listenId));
507 			return new SseEvent().setData(Json.of(JsonRpcResponse.ok(listenId, result)));

510 		/**
511 		 * Maps a drained {@link McpChangeEvent} to its wire notification frame. Used only from {@link #run()},
512 		 * so it lives here rather than on the enclosing publisher.
513 		 */
514 		private SseEvent changeFrame(McpChangeEvent event) {
515 			var meta = new RequestMeta().set(RequestMeta.KEY_SUBSCRIPTION_ID, listenId);
516 			String method;
517 			Object params;
518 			switch (event.kind()) {
519 				case RESOURCE_UPDATED -> {
520 					method = McpMethods.NOTIFICATIONS_RESOURCES_UPDATED;
521 					params = new ResourceUpdatedNotification().setUri(event.resourceUri()).setMeta(meta);
522 				}
523 				case TOOLS_LIST_CHANGED -> {
524 					method = McpMethods.NOTIFICATIONS_TOOLS_LIST_CHANGED;
525 					params = new ToolsListChangedNotification().setMeta(meta);
526 				}
527 				case PROMPTS_LIST_CHANGED -> {
528 					method = McpMethods.NOTIFICATIONS_PROMPTS_LIST_CHANGED;
529 					params = new PromptsListChangedNotification().setMeta(meta);
530 				}
531 				case RESOURCES_LIST_CHANGED -> {
532 					method = McpMethods.NOTIFICATIONS_RESOURCES_LIST_CHANGED;
533 					params = new ResourcesListChangedNotification().setMeta(meta);
534 				}
535 				default -> throw new IllegalStateException("Unhandled McpChangeKind: " + event.kind());
536 			}
537 			var frame = new JsonRpcRequest().setJsonrpc(McpProtocol.JSON_RPC_2_0).setId(null).setMethod(method).setParams(params);
538 			return new SseEvent().setData(Json.of(frame));
539 		}
