17  package org.apache.juneau.rest.server.mcp;
18  
19  import java.util.concurrent.atomic.*;
20  
21  import org.apache.juneau.bean.jsonrpc.*;
22  import org.apache.juneau.commons.inject.*;
23  import org.apache.juneau.http.Content;
24  import org.apache.juneau.marshall.serializer.*;
25  import org.apache.juneau.rest.server.*;
26  import org.apache.juneau.rest.server.servlet.*;

28  /**
29   * Drop-in {@link BasicRestServlet} subclass that exposes a single MCP JSON-RPC endpoint at {@code POST /}.
30   *
31   * <p>
32   * Subclasses provide their {@link McpServerConfig} by overriding {@link #createMcpConfig()} (called once at
33   * {@link #init() init} time), or by registering an {@link McpServerConfig} bean in their {@code RestContext}
34   * bean store.
35   *
36   * <h5 class='section'>Example:</h5>
37   * <pre>
38   * @Rest(path="/mcp")
39   * public class MyMcpServlet extends org.apache.juneau.rest.server.mcp.v20250618.McpRestServlet {
40   *     @Override
41   *     protected McpServerConfig createMcpConfig() {
42   *         return new McpServerConfig()
43   *             .setName("my-server").setVersion("1.0.0")
44   *             .addTool(new MyEchoTool());
45   *     }
46   * }
47   * </pre>
48   *
49   * <p>
50   * The servlet enables {@code addBeanTypes} on its serializer so that polymorphic wire types a
51   * revision emits (content blocks, resource payloads) are tagged with their {@code type}
52   * discriminator, and disables URI resolution ({@code uriResolution="NONE"}) so that JSON-Schema
53   * JSON-Pointer {@code $ref} values (e.g. {@code "#/$defs/text"}) and absolute URIs with fragments
54   * are emitted verbatim rather than rewritten against the servlet path.
55   *
56   * @serial exclude
57   */
58  @Rest
59  @SerializerConfig(addBeanTypes = "true", uriResolution = "NONE")
60  public abstract class AbstractMcpRestServlet extends BasicRestServlet {
61  	private static final long serialVersionUID = 1L;
62  
63  	private final transient AtomicReference<McpServerConfig> config = new AtomicReference<>();
64  
65  	/**
66  	 * Returns the {@link McpServerConfig} backing this servlet.
67  	 *
68  	 * <p>
69  	 * The default implementation lazily caches the value of {@link #createMcpConfig()}; subclasses needing
70  	 * per-request reconfiguration may override.
71  	 *
72  	 * <p>
73  	 * This is invoked on every request, so the cache is published via a lock-free {@link AtomicReference#compareAndSet}
74  	 * rather than a {@code synchronized} block, avoiding monitor contention on the hot path. Under a first-access
75  	 * race, {@link #createMcpConfig()} may be invoked by more than one thread, but only one resulting instance is
76  	 * ever published to {@link #config} and returned to any caller; the others are discarded. This is safe as long
77  	 * as {@link #createMcpConfig()} is side-effect-free and idempotent (i.e. simply constructs and returns a new
78  	 * config), which is the documented contract of that method.
79  	 *
80  	 * @return The config. Never {@code null}.
81  	 */
82  	public McpServerConfig getMcpConfig() {
83  		var c = config.get();
84  		if (c == null) {
85  			var nc = createMcpConfig();
86  			if (nc == null)
87  				throw new IllegalStateException("createMcpConfig() returned null");
88  			c = config.compareAndSet(null, nc) ? nc : config.get(); // HTT: the CAS-loses branch requires a genuine concurrent first-access race; untestable deterministically.
89  		}
90  		return c;
91  	}
92  
93  	/**
94  	 * Factory method for the {@link McpServerConfig} backing this servlet.
95  	 *
96  	 * <p>
97  	 * Subclasses must override this method to declare their tools, prompts, and resources.
98  	 *
99  	 * @return A non-{@code null} server config.
100 	 */
101 	protected abstract McpServerConfig createMcpConfig();
102 
103 	/**
104 	 * The MCP protocol revision this servlet speaks.
105 	 *
106 	 * <p>
107 	 * Supplied by the revision-specific subclass a consumer extends (for example
108 	 * {@code org.apache.juneau.rest.server.mcp.v20250618.McpRestServlet}); consumers do not normally implement this themselves.
109 	 *
110 	 * @return The bound revision. Never {@code null}.
111 	 */
112 	protected abstract McpRevision revision();
113 
114 	/**
115 	 * Optional subscription broker backing this servlet's {@code subscriptions/listen} support.
116 	 *
117 	 * <p>
118 	 * Returning <jk>null</jk> (the default) means this servlet does not support subscriptions: no
119 	 * {@link McpSubscriptionBroker} or {@link McpSubscriptions} bean is added to the request-scoped
120 	 * {@link BeanStore} that {@link #handleMcp} builds, so any handler or dispatch branch that looks one up
121 	 * via {@code ctx.getBean(McpSubscriptionBroker.class)} sees {@link java.util.Optional#empty()}.
122 	 *
123 	 * <p>
124 	 * A revision binding that implements {@code subscriptions/listen} (for example
125 	 * {@code org.apache.juneau.rest.server.mcp.v20260728.McpRestServlet}) overrides this to return a
126 	 * memoized broker instance so every request sees the same live subscription registry.
127 	 *
128 	 * @return The subscription broker, or <jk>null</jk> if this servlet does not support subscriptions.
129 	 */
130 	protected McpSubscriptionBroker getSubscriptionBroker() {
131 		return null;
132 	}
133 
134 	/**
135 	 * MCP JSON-RPC endpoint.
136 	 *
137 	 * @param req The parsed JSON-RPC request envelope.
138 	 * @param restReq The HTTP request (used to surface the bean store to handlers).
139 	 * @return The JSON-RPC response, or {@code null} for notifications (which the framework renders as an
140 	 *         empty body).
141 	 */
142 	@SuppressWarnings({
143 		"resource" // Request-scoped scratch BasicBeanStore; lifetime is bounded by this handler invocation, no foreign resources are captured.
144 	})
145 	@RestPost(path = "/")
146 	public Object handleMcp(@Content JsonRpcRequest req, RestRequest restReq) {
147 		var bs = new BasicBeanStore(restReq.getContext().getBeanStore())
148 			.addBean(RestRequest.class, restReq);
149 		var broker = getSubscriptionBroker();
150 		if (broker != null)
151 			bs.addBean(McpSubscriptionBroker.class, broker).addBean(McpSubscriptions.class, broker);
152 		var exchange = new McpExchange(req, n -> restReq.getHeaderParam(n).asString().orElse(null));
153 		var result = revision().dispatch(exchange, getMcpConfig(), bs);
154 		if (result instanceof McpResponseResult r)
155 			return r.response();
156 		if (result instanceof McpStreamResult r)
157 			return r.stream();
158 		return null;
159 	}
160 }
