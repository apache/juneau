17  package org.apache.juneau.rest.server.mcp.v20260728;
18  
19  import org.apache.juneau.commons.concurrent.*;
20  
21  /**
22   * The dispatcher-owned payload sealed into an opaque {@code requestState} token by {@link RequestStateCodec}.
23   *
24   * <p>
25   * Never wire-visible directly &mdash;
26   * {@link org.apache.juneau.bean.mcp.v20260728.InputRequiredResult#getRequestState()} carries only the sealed
27   * string form. This type exists so the round counter (SEP-2322's max-rounds cap) and expiry travel *inside*
28   * the sealed token itself, per {@code McpRevision}'s documented statelessness: no server-side session exists
29   * to hold them instead.
30   *
31   * <p>
32   * <b>The sealed continuation is the only trusted per-round carrier.</b> This is what makes it security-relevant:
33   * the {@code continuation} sealed here is authenticated (AEAD) and cannot be altered by the client between
34   * rounds, whereas the request {@code arguments} a handler sees on RESUME come from the follow-up request and are
35   * fully client-controlled (see {@link McpMrtrResumeContext}). Anything a handler must be able to trust across
36   * rounds &mdash; an authenticated principal, an authorization decision, a price, a resource id &mdash; must live
37   * <i>inside</i> this continuation, never be re-derived from the per-round arguments.
38   *
39   * <p>
40   * <b>Continuation type fidelity.</b> {@code continuation} is declared as {@code Object} because it is
41   * handler-opaque, but {@link AeadRequestStateCodec#unseal} reconstructs it via generic JSON deserialization
42   * ({@code Json.to(json, McpRequestState.class)}). A handler that pauses with a {@code Map} or ordinary-bean
43   * continuation therefore gets back a <i>generic JSON</i> value on resume &mdash; a {@code JsonMap}/
44   * {@code JsonList}/boxed primitive/{@code String} &mdash; never the original Java type. A handler that needs its
45   * original type back should not reconstruct it by hand: it retrieves the resume context from the
46   * {@code BeanStore} and calls {@link McpMrtrResumeContext#continuationAs(Class)}, which converts the generic JSON
47   * value back to any Juneau-marshallable target type in one call.
48   *
49   * @param continuation The handler-opaque continuation value, exactly as thrown in
50   * 	{@code McpInputRequiredSignal#getContinuation()} (or, on a subsequent pause, exactly as the handler
51   * 	returned it a second time). Must be Juneau-marshallable (ordinary bean, {@code Map}, {@code String}, or
52   * 	similar). See the type-fidelity note above: it round-trips as generic JSON, not its original Java type.
53   * @param method The JSON-RPC method this state was sealed for ({@code tools/call}, {@code prompts/get}, or
54   * 	{@code resources/read}). Part of the AAD contract (see {@link RequestStateCodec}); also checked directly
55   * 	by the dispatcher against the follow-up request's own method as an extra sanity check.
56   * @param round The 1-based round counter: 1 on the first PAUSE, incremented on every subsequent PAUSE.
57   * @param expiresAtMs Absolute expiry timestamp in epoch milliseconds.
58   * @param jti A fresh, per-token random identifier minted on every PAUSE (see {@code McpRevision#pause}), used
59   * 	only to detect replay of THIS exact token. Opt-in: consulted by the dispatcher only when an operator has
60   * 	wired a {@link ReplayCache} via {@link McpMrtrConfig#setReplayCache(ReplayCache)}; otherwise it rides in the
61   * 	sealed plaintext unused. Not wire-visible on its own &mdash; like every other field on this record, it only
62   * 	ever travels inside the opaque, AEAD-sealed {@code requestState} string.
63   * @param argumentsHash A canonical hash of the original request's {@code arguments} (RFC 8785 JSON
64   * 	Canonicalization Scheme, then SHA-256, then base64url), captured at PAUSE time and re-verified against the
65   * 	RESUME request's own {@code arguments} on every subsequent round (see {@code McpRevision#resolveMrtrContext}).
66   * 	Always populated (unlike {@link #jti()}, this check is not opt-in): a call with absent or empty
67   * 	{@code arguments} (for example {@code resources/read}'s exact-path branch, which takes none) hashes the
68   * 	canonical empty object {@code "{}"} rather than leaving this field unset, so PAUSE and RESUME agree even when
69   * 	there is nothing to hash.
70   */
71  public record McpRequestState(Object continuation, String method, int round, long expiresAtMs, String jti, String argumentsHash) {
72  }
