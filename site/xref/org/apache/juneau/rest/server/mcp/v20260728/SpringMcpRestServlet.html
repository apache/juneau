17  package org.apache.juneau.rest.server.mcp.v20260728;
18  
19  import org.apache.juneau.rest.server.springboot.*;
20  
21  /**
22   * Convenience base class exposing a {@code 2026-07-28} MCP endpoint on a Spring Boot resource with full
23   * Spring dependency-injection support.
24   *
25   * <p>
26   * Combines {@link BasicSpringRestServlet} (Spring Boot bean-store integration via
27   * {@link org.apache.juneau.rest.server.springboot.SpringBeanStore}) with the {@link McpEndpoint} mixin
28   * (the {@code 2026-07-28} MCP handler). This is the recommended way to expose MCP from a Spring Boot
29   * resource: subclassing {@link McpRestServlet} instead does <b>not</b> work, since
30   * {@link McpRestServlet} and {@link BasicSpringRestServlet} are sibling
31   * {@code RestServlet} subclasses and a class cannot extend both — and even if it could, only the mixin
32   * path preserves the Spring {@code BeanStore} bridge that MCP tool/prompt/resource handlers rely on to
33   * resolve Spring-managed beans (for example via {@code BeanStore.getBean(MyService.class)}).
34   *
35   * <p>
36   * Subclasses supply their {@link org.apache.juneau.rest.server.mcp.McpServerConfig} by implementing
37   * {@link #getMcpConfig()}, and may optionally override {@link McpEndpoint#getMcpOptions()} to customize
38   * capabilities/instructions/cache/mrtr/subscriptions/broker behavior.
39   *
40   * <h5 class='section'>Example:</h5>
41   * <pre>
42   * @Rest(path="/mcp")
43   * public class MyMcpResource extends SpringMcpRestServlet {
44   * 	@Override
45   * 	public McpServerConfig getMcpConfig() {
46   * 		return new McpServerConfig().addTool(new MySpringAwareTool());
47   * 	}
48   * 	@Override                                        // optional
49   * 	public McpOptions getMcpOptions() {
50   * 		return new McpOptions().mrtr(m -&gt; m.setTtlMs(600_000));
51   * 	}
52   * }
53   * </pre>
54   *
55   * @serial exclude
56   */
57  @SuppressWarnings({
58  	"java:S2176", // Intentional: dated adapter binding classes are de-versioned and differentiated by package.
59  	"java:S110" // Inheritance depth is inherent to the Juneau Spring REST servlet hierarchy.
60  })
61  public abstract class SpringMcpRestServlet extends BasicSpringRestServlet implements McpEndpoint {
62  	private static final long serialVersionUID = 1L;
63  }
