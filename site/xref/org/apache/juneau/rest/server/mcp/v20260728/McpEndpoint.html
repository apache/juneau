22  import org.apache.juneau.rest.server.mcp.McpSubscriptionBroker;

24  
25  import jakarta.servlet.http.HttpServletRequest;
26  import jakarta.servlet.http.HttpServletResponse;

28  /**
29   * Mixin interface that exposes a {@code 2026-07-28} MCP endpoint at {@code POST /mcp} on any Juneau REST
30   * resource.
31   *
32   * <p>
33   * The mixin path and the servlet-subclass path ({@link McpRestServlet}) are at parity: both bind the revision
34   * at compile time and both read the same {@link McpOptions} shape for behavior configuration
35   * (capabilities/instructions/cache/mrtr/subscriptions/broker) via a single override.
36   *
37   * <h5 class='section'>Example:</h5>
38   * <pre>
39   * @Rest(path="/api")
40   * public class MyResource extends BasicRestServlet implements McpEndpoint {
41   *     @Override
42   *     public McpServerConfig getMcpConfig() {
43   *         return new McpServerConfig().addTool(new MyEchoTool());
44   *     }
45   *     @Override                                        // optional
46   *     public McpOptions getMcpOptions() {
47   *         return new McpOptions().mrtr(m -&gt; m.setTtlMs(600_000));
48   *     }
49   * }
50   * </pre>
51   */
52  public interface McpEndpoint extends org.apache.juneau.rest.server.mcp.McpEndpointMixin {
53  
54  	@Override /* McpEndpointMixin */
55  	default org.apache.juneau.rest.server.mcp.McpRevision revision() {
56  		var o = McpEndpointOptionsCache.resolve(this);
57  		return new McpRevision(o.getCapabilities(), o.getCache(), o.getInstructions(), o.getMrtr());
58  	}
59  
60  	/**
61  	 * Publishes this revision's stable {@link TraceContextExtractor} so an active {@code TracerHook}
62  	 * (via {@code RestOpInvoker}) recognizes this endpoint's resolved {@code JsonRpcRequest} argument
63  	 * and its {@code params._meta} carrier before span creation.
64  	 *
65  	 * <p>
66  	 * A no-op when no {@code TracerHook} bean is registered &mdash; see the no-tracer fast-path
67  	 * contract documented on {@code TracerHook}.
68  	 *
69  	 * @return This revision's shared {@link TraceContextExtractor}. Never <jk>null</jk>.
70  	 */
71  	@Bean
72  	default TraceContextExtractor mcpTraceContextExtractor() {
73  		return McpRevision.TRACE_CONTEXT_EXTRACTOR;
74  	}
75  
76  	/**
77  	 * This endpoint's behavior configuration &mdash; capabilities, instructions, cache hints, MRTR, subscriptions,
78  	 * and the subscription broker, all in one place.
79  	 *
80  	 * <p>
81  	 * The default returns a per-binding-fresh {@link McpOptions} (all defaults; no live-state sharing across
82  	 * endpoint instances). Override to customize any of the six concepts.
83  	 *
84  	 * <p>
85  	 * <b>This method is invoked exactly once per endpoint instance</b> and the result is memoized for that
86  	 * instance's lifetime by the framework internals that read it ({@link #revision()},
87  	 * {@link #subscriptionBroker()}, the {@code @Bean}-published accessor) &mdash; so an override may freely
88  	 * construct and configure a fresh instance on every call, exactly as shown above; it never needs to cache the
89  	 * result itself, and returning a fresh instance per call cannot reintroduce the "must return a stable
90  	 * instance" footgun the pre-consolidation per-concept hooks warned about.
91  	 *
92  	 * @return This endpoint's options. Must not be <jk>null</jk>.
93  	 */
94  	default McpOptions getMcpOptions() {
95  		return new McpOptions();
96  	}
97  
98  	/**
99  	 * Publishes this endpoint's memoized {@link McpOptions} into this resource's {@code RestContext} bean store,
100 	 * the same {@code @Bean}-method mechanism already used by {@link #mcpTraceContextExtractor()}.
101 	 *
102 	 * <p>
103 	 * This is what lets {@code McpRevision.dispatch(...)}'s {@code subscriptions/listen} branch resolve the
104 	 * real, possibly-overridden subscriptions configuration via {@code ctx.getBean(McpOptions.class)} instead of
105 	 * always falling through to that call's defensive {@code new McpSubscriptionsConfig()} default &mdash; the
106 	 * neutral core cannot add this bean itself ({@code McpOptions} is a v2 type, and the neutral module must not
107 	 * import it), so this v2-side {@code @Bean} method is the wiring's only seam.
108 	 *
109 	 * @return This endpoint's memoized options. Never <jk>null</jk>.
110 	 */
111 	@Bean
112 	default McpOptions mcpOptionsBean() {
113 		return McpEndpointOptionsCache.resolve(this);
114 	}
115 
116 	/**
117 	 * Subscription broker for this endpoint's bound revision.
118 	 *
119 	 * <p>
120 	 * Reads {@link #getMcpOptions()}'s memoized {@link McpOptions#getSubscriptionBroker()}: an explicitly
121 	 * configured broker is returned as-is; otherwise one is derived and memoized per-binding, sized from
122 	 * {@link McpOptions#getSubscriptions()}'s queue bound (see {@link McpOptions#resolveSubscriptionBroker()}).
123 	 * Because {@link McpOptions} itself is memoized once per endpoint instance ({@link McpEndpointOptionsCache}),
124 	 * two distinct endpoint instances resolve to two distinct broker registries by default &mdash; there is no
125 	 * accidental JVM-wide sharing the way the pre-consolidation mixin default had.
126 	 *
127 	 * @return The subscription broker. Never <jk>null</jk>.
128 	 */
129 	@Override
130 	default McpSubscriptionBroker subscriptionBroker() {
131 		return McpEndpointOptionsCache.resolve(this).resolveSubscriptionBroker();
132 	}
133 
134 	/**
135 	 * OAuth 2.1 resource-server bearer gate for the mixin's MCP {@code POST /mcp} endpoint (READY-312f F2).
136 	 *
137 	 * <p>
138 	 * A no-op unless {@link McpResourceServerConfig#isEnabled() RS auth is enabled}; when enabled, a request that
139 	 * <b>the router resolves</b> to the MCP {@code POST /mcp} dispatch operation must present a valid bearer token or it is
140 	 * rejected with {@code 401} / {@code 403} and a {@code WWW-Authenticate: Bearer ...} challenge.  The gate is scoped to
141 	 * the MCP dispatch <i>operation</i> (identified by resolved-operation identity, or by that operation's own router
142 	 * matchers &mdash; see {@link McpResourceServerSupport#resolvesToMcpDispatch}) so other {@code POST} operations on the
143 	 * host resource are unaffected, and it cannot drift from the router: a trailing slash, percent-encoding, a
144 	 * {@code ?method=}/{@code X-Method} override, or a host-chosen {@code @Mixin(path/paths)} re-mount cannot slip past it
145 	 * (B1/H3).  The well-known {@code GET} routes are left unauthenticated for PRM discovery.
146 	 *
147 	 * @param ctx The host {@link org.apache.juneau.rest.server.RestContext} handling this request.
148 	 * @param req The raw HTTP request.
149 	 * @param res The raw HTTP response (used to emit the challenge header on rejection).
150 	 */
151 	@RestStartCall
152 	default void mcpResourceServerAuth(org.apache.juneau.rest.server.RestContext ctx, HttpServletRequest req, HttpServletResponse res) {
153 		McpResourceServerSupport.gateMcpEndpoint(McpEndpointOptionsCache.resolve(this).getResourceServer(), ctx, req, res);
154 	}
155 
156 	/**
157 	 * Serves the RFC 9728 Protected Resource Metadata document at the root well-known location (SEP-2351 fallback).
158 	 *
159 	 * <p>
160 	 * This route (and its path-inserted sibling) is registered unconditionally, but returns {@code 404} whenever RS auth
161 	 * is {@link McpResourceServerConfig#isEnabled() disabled} &mdash; so a non-opted-in endpoint claims the well-known
162 	 * namespace but exposes no observable metadata (READY-312f F2, M7).
163 	 *
164 	 * @param req The REST request.
165 	 * @return The PRM document.
166 	 */
167 	@RestGet(path = "/.well-known/oauth-protected-resource")
168 	default McpProtectedResourceMetadata mcpProtectedResourceMetadataRoot(org.apache.juneau.rest.server.RestRequest req) {
169 		return McpResourceServerSupport.metadataForWellKnownRoot(McpEndpointOptionsCache.resolve(this).getResourceServer(), req);
170 	}
171 
172 	/**
173 	 * Serves the RFC 9728 Protected Resource Metadata document at the path-inserted well-known location (SEP-2351), but
174 	 * only when the requested suffix matches this resource's expected well-known path; otherwise {@code 404} (M5).
175 	 *
176 	 * @param req The REST request.
177 	 * @return The PRM document.
178 	 */
179 	@RestGet(path = "/.well-known/oauth-protected-resource/*")
180 	default McpProtectedResourceMetadata mcpProtectedResourceMetadataPathInserted(org.apache.juneau.rest.server.RestRequest req) {
181 		return McpResourceServerSupport.metadataForWellKnownPathInserted(McpEndpointOptionsCache.resolve(this).getResourceServer(), req);
182 	}
