41  /**
42   * Request-time glue for the F2 OAuth 2.1 resource-server baseline &mdash; bearer validation, RFC 8707 audience
43   * enforcement, {@code 401}/{@code 403} challenge emission, and RFC 9728 Protected Resource Metadata construction &mdash;
44   * driven by an {@link McpResourceServerConfig}.
45   *
46   * <p>
47   * Bearer extraction and token validation are delegated to the reusable
48   * {@link org.apache.juneau.rest.server.auth.oauth.OAuthFilter} (RFC 6750) from
49   * {@code juneau-rest-server-auth-oauth}; this class layers on the MCP-specific {@code resource_metadata} challenge
50   * parameter, RFC 8707 audience matching ({@link McpAudienceValidator}), and baseline required-scope enforcement.
51   *
52   * @since 10.0.0
53   */
54  public final class McpResourceServerSupport {

69  	/**
70  	 * Router-identity bearer gate for the {@link McpEndpoint} mixin's MCP JSON-RPC endpoint (READY-312f F2, B1/H3).
71  	 *
72  	 * <p>
73  	 * A no-op unless RS auth is {@link McpResourceServerConfig#isEnabled() enabled} and this request resolves to the MCP
74  	 * JSON-RPC dispatch operation.  See {@link #resolvesToMcpDispatch(RestContext, HttpServletRequest)} for how the target
75  	 * operation is identified without any hand-rolled path reconstruction &mdash; the key change that makes this gate
76  	 * immune to {@code @Mixin(path/paths)} re-mounts and to trailing-slash / percent-encoding / method-override drift.
77  	 *
78  	 * <p>
79  	 * Fails fast with the same origin-root {@code 500} diagnostic as the well-known route (see
80  	 * {@link #assertWellKnownMountReachable(RestRequest)}) &mdash; rather than authenticating &mdash; whenever the
81  	 * resolved MCP dispatch operation is itself mounted under a {@code @Mixin(path/paths)} re-mount prefix (H1
82  	 * unification): a {@code 401} on such a mount would advertise a {@code resource_metadata} URL that the re-mount
83  	 * has already made unreachable, so the two routes must agree rather than leaving a broken half-state.
84  	 *
85  	 * @param cfg The RS config.  Must not be <jk>null</jk>.
86  	 * @param ctx The host {@link RestContext} handling this request (injected into the {@code @RestStartCall} hook).
87  	 * 	Must not be <jk>null</jk>.
88  	 * @param req The raw HTTP request.  Must not be <jk>null</jk>.
89  	 * @param res The raw HTTP response (used to emit the challenge header on rejection).  Must not be <jk>null</jk>.
90  	 */
91  	public static void gateMcpEndpoint(McpResourceServerConfig cfg, RestContext ctx, HttpServletRequest req, HttpServletResponse res) {
92  		if (!cfg.isEnabled())
93  			return;
94  		if (!isEffectivePost(req))
95  			return;
96  		var op = resolveMcpDispatchOp(ctx, req);
97  		if (op == null)
98  			return;
99  		assertOriginRootMount(req);
100 		assertNotMixinRemounted(op.getContext().getMixinMountPrefixes());
101 		authenticate(cfg, req, res);
102 	}

105 	 * Router-aligned bearer gate for the {@link McpRestServlet}'s {@code POST /} endpoint (READY-312f F2, H3).
106 	 *
107 	 * <p>
108 	 * A no-op unless RS auth is {@link McpResourceServerConfig#isEnabled() enabled} and this request resolves to a
109 	 * {@code POST} (the dedicated MCP servlet's only writable surface).  Gating every effective {@code POST} is correct for
110 	 * a dedicated MCP servlet: there is no non-MCP writable surface to distinguish, so no per-operation identity check is
111 	 * needed.
112 	 *
113 	 * @param cfg The RS config.  Must not be <jk>null</jk>.
114 	 * @param req The raw HTTP request.  Must not be <jk>null</jk>.
115 	 * @param res The raw HTTP response (used to emit the challenge header on rejection).  Must not be <jk>null</jk>.
116 	 */
117 	public static void gateMcpServlet(McpResourceServerConfig cfg, HttpServletRequest req, HttpServletResponse res) {
118 		if (!cfg.isEnabled())
119 			return;
120 		if (!isEffectivePost(req))
121 			return;
122 		assertOriginRootMount(req);
123 		authenticate(cfg, req, res);
124 	}

126 	/**
127 	 * Returns whether this request resolves to the MCP JSON-RPC dispatch operation ({@link McpEndpointMixin#handleMcpRequest}).
128 	 *
129 	 * <p>
130 	 * <b>Why this cannot drift from the router (B1/H3):</b> instead of reconstructing the MCP endpoint's path from a
131 	 * hardcoded constant &mdash; which the previous gate did, so a {@code @Mixin(path="/api")} re-mount routed
132 	 * {@code POST /api/mcp} to the handler while the gate compared {@code ["api","mcp"]} against {@code ["mcp"]} and
133 	 * silently skipped authentication &mdash; this identifies the <i>operation</i> the router actually selected:
134 	 * <ul>
135 	 * 	<li><b>Mixin sub-context</b> (the {@code McpEndpoint} mixin's {@code @RestStartCall} fires from
136 	 * 		{@code RestSession.run()} <i>after</i> the operation is resolved): the resolved Java method is available, so we
137 	 * 		gate iff it is {@link McpEndpointMixin#handleMcpRequest} (or an override).  No path logic at all &mdash; zero drift,
138 	 * 		and re-mounts / multi-mounts are handled for free because the router already routed to the op.
139 	 * 	<li><b>Host {@code implements McpEndpoint}</b> (the host's {@code @RestStartCall} fires <i>before</i> routing, so the
140 	 * 		resolved op is not yet known): we look up the dispatch operation's OWN {@link org.apache.juneau.rest.server.util.UrlPathMatcher}
141 	 * 		objects from {@link RestContext#getRestOperations()} and match the request against them &mdash; reusing the router's
142 	 * 		own matchers, so re-mounts, {@code /*}, trailing slashes and percent-encoding are all handled exactly as the router
143 	 * 		handles them.
144 	 * </ul>
145 	 *
146 	 * @param ctx The host {@link RestContext} handling this request.
147 	 * @param req The raw HTTP request.
148 	 * @return <jk>true</jk> if the request resolves to the MCP dispatch operation.
149 	 */
150 	static boolean resolvesToMcpDispatch(RestContext ctx, HttpServletRequest req) {
151 		return resolveMcpDispatchOp(ctx, req) != null;
152 	}

154 	/**
155 	 * Returns the resolved MCP JSON-RPC dispatch operation's {@link RestOpContext} for this request, or
156 	 * <jk>null</jk> if the request does not resolve to it.
157 	 *
158 	 * <p>
159 	 * Same two-case resolution as {@link #resolvesToMcpDispatch(RestContext, HttpServletRequest)}, but returns the
160 	 * matched operation's own {@link RestOpContext} (rather than a boolean) so callers can also inspect the
161 	 * operation's owning {@link RestContext#getMixinMountPrefixes() mixin mount prefixes} &mdash; e.g. to detect a
162 	 * {@code @Mixin(path/paths)} re-mount, the same signal {@link #assertWellKnownMountReachable(RestRequest)} uses
163 	 * for the well-known route.
164 	 *
165 	 * @param ctx The host {@link RestContext} handling this request.
166 	 * @param req The raw HTTP request.
167 	 * @return The resolved MCP dispatch operation's {@link RestOpContext}, or <jk>null</jk>.
168 	 */
169 	static RestOpContext resolveMcpDispatchOp(RestContext ctx, HttpServletRequest req) {
170 		var opSession = ctx.getLocalSession().getOpSessionOrNull();
171 		if (opSession != null) {
172 			var oc = opSession.getContext();
173 			return isMcpDispatchMethod(oc.getJavaMethod()) ? oc : null;
174 		}
175 		return matchMcpDispatchOp(ctx, req);
176 	}

178 	/**
179 	 * Returns whether the specified Java method is the MCP JSON-RPC dispatch operation
180 	 * ({@link McpEndpointMixin#handleMcpRequest}, or an override of it).
181 	 *
182 	 * @param m The resolved operation's Java method.  May be <jk>null</jk>.
183 	 * @return <jk>true</jk> if it is the MCP dispatch method.
184 	 */
185 	static boolean isMcpDispatchMethod(Method m) {
186 		return m != null
187 			&& MCP_DISPATCH_METHOD.equals(m.getName())
188 			&& McpEndpointMixin.class.isAssignableFrom(m.getDeclaringClass());
189 	}

191 	/**
192 	 * Returns whether the request path matches the MCP dispatch operation's own router matchers (host
193 	 * {@code implements McpEndpoint} case, where the resolved op is not yet available).
194 	 *
195 	 * @param ctx The host {@link RestContext} handling this request.
196 	 * @param req The raw HTTP request.
197 	 * @return The MCP dispatch operation's {@link RestOpContext} if the request path matches its matchers, else
198 	 * 	<jk>null</jk>.
199 	 */
200 	static RestOpContext matchMcpDispatchOp(RestContext ctx, HttpServletRequest req) {
201 		var ops = ctx.getRestOperations();
202 		if (ops == null)
203 			return null;
204 		var urlPath = UrlPath.of(RestUtils.getPathInfoUndecoded(req));
205 		for (var oc : ops.getOpContexts())
206 			if (isMcpDispatchMethod(oc.getJavaMethod()))
207 				for (var matcher : oc.getPathMatchers())
208 					if (matcher.match(urlPath) != null)
209 						return oc;
210 		return null;
211 	}

213 	/**
214 	 * Returns whether this request resolves to a {@code POST} once method-override channels are honored (H3).
215 	 *
216 	 * <p>
217 	 * Mirrors {@code RestSession.getMethod()}: an explicit {@code POST}, or a {@code ?method=POST} query parameter, or an
218 	 * {@code X-Method: POST} header.  Because a {@code @RestStartCall} hook cannot see the host's
219 	 * {@code allowedMethodParams}/{@code allowedMethodHeaders} allow-lists, any override to {@code POST} is treated as a
220 	 * {@code POST} &mdash; this is fail-closed: the only cost is that a {@code ?method=POST} aimed at the MCP path on a
221 	 * host that does <i>not</i> allow overrides is challenged rather than 404'd, which never weakens the gate.
222 	 */
223 	static boolean isEffectivePost(HttpServletRequest req) {
224 		if ("POST".equalsIgnoreCase(req.getMethod()))
225 			return true;
226 		var mp = RestUtils.parseQuery(req.getQueryString()).get("method");
227 		if (mp != null && !mp.isEmpty() && "POST".equalsIgnoreCase(mp.get(0)))
228 			return true;
229 		return "POST".equalsIgnoreCase(req.getHeader("X-Method"));
230 	}

232 	/**
233 	 * Fails fast when this endpoint is not mounted at the origin root (READY-312f F2, H1).
234 	 *
235 	 * <p>
236 	 * RFC 9728 requires the Protected Resource Metadata document at an origin-root well-known location
237 	 * ({@code https://host/.well-known/oauth-protected-resource[/path]}), which is exactly the URL advertised in the
238 	 * {@code resource_metadata} challenge parameter.  Juneau registers the well-known {@code @RestGet} routes
239 	 * servlet-relative, so on any non-origin-root mount (a non-empty servlet path or servlet-context path) the advertised
240 	 * URL would resolve to a different location and 404.  Rather than silently advertising an unreachable document, RS
241 	 * auth refuses to operate on such a mount.
242 	 *
243 	 * <p>
244 	 * A container that reports the root context / root servlet mapping as {@code "/"} (rather than the empty string) is
245 	 * treated as origin-root &mdash; the tolerance is symmetric across {@code contextPath} and {@code servletPath} (H1 LOW).
246 	 *
247 	 * @param req The HTTP request.
248 	 * @throws InternalServerError Always, when a non-root context path or non-root servlet path is present.
249 	 */
250 	static void assertOriginRootMount(HttpServletRequest req) {
251 		var ctx = emptyIfNull(req.getContextPath());
252 		var svl = emptyIfNull(req.getServletPath());
253 		if (!isOriginRootPath(ctx) || !isOriginRootPath(svl))
254 			throw new InternalServerError(
255 				"MCP resource-server auth requires an origin-root mount so the advertised RFC 9728 well-known PRM URL "
256 				+ "resolves; found contextPath='%s' servletPath='%s'.  Mount the MCP endpoint at the origin root "
257 				+ "(no servlet path / no context path).",
258 				ctx, svl);
259 	}
260 
261 	/** Returns whether a context/servlet path denotes the origin root ({@code ""} or {@code "/"}). */
262 	private static boolean isOriginRootPath(String p) {
263 		return p.isEmpty() || p.equals("/");
264 	}

266 	/**
267 	 * Fails fast when the well-known PRM route cannot be served from the origin root (READY-312f F2, H1 + secondary MEDIUM).
268 	 *
269 	 * <p>
270 	 * Extends {@link #assertOriginRootMount(HttpServletRequest)} with the {@code @Mixin(path/paths)} re-mount case: when the
271 	 * MCP endpoint is mounted as a mixin under one or more host-chosen prefixes, the well-known {@code @RestGet} routes are
272 	 * re-mounted along with every other mixin op (to {@code /<prefix>/.well-known/...}), so the origin-root PRM URL
273 	 * advertised in the {@code resource_metadata} challenge is unreachable.  {@code contextPath}/{@code servletPath} are both
274 	 * empty in that case (the host is root-mounted), so the servlet-path check alone cannot detect it &mdash; we consult the
275 	 * resolved operation's {@link RestContext#getMixinMountPrefixes() mixin mount prefixes} instead, and fail fast with the
276 	 * same origin-root diagnostic so discovery can never silently advertise an unreachable URL.
277 	 *
278 	 * @param req The REST request (its resolved operation identifies the possibly re-mounted sub-context).
279 	 * @throws InternalServerError When a non-root mount or a {@code @Mixin(path/paths)} re-mount is detected.
280 	 */
281 	static void assertWellKnownMountReachable(RestRequest req) {
282 		assertOriginRootMount(req.getHttpServletRequest());
283 		var opCtx = req.getOpContext();
284 		assertNotMixinRemounted(opCtx == null ? List.of() : opCtx.getContext().getMixinMountPrefixes());
285 	}

288 	 * Fails fast when the resolved MCP operation is mounted under one or more host-chosen {@code @Mixin(path/paths)}
289 	 * re-mount prefixes (READY-312f F2, H1 unification) &mdash; the SAME diagnostic used by
290 	 * {@link #assertWellKnownMountReachable(RestRequest)}, shared so the {@code POST} bearer gate
291 	 * ({@link #gateMcpEndpoint}) and the well-known route never disagree on whether a given mount is supported.
292 	 *
293 	 * @param mixinMountPrefixes The resolved operation's owning {@link RestContext#getMixinMountPrefixes() mixin
294 	 * 	mount prefixes}; empty when the operation is not mixin-mounted (or not re-mounted).
295 	 * @throws InternalServerError When {@code mixinMountPrefixes} is non-empty.
296 	 */
297 	private static void assertNotMixinRemounted(List<String> mixinMountPrefixes) {
298 		if (!mixinMountPrefixes.isEmpty())
299 			throw new InternalServerError(
300 				"MCP resource-server auth requires an origin-root mount so the advertised RFC 9728 well-known PRM URL "
301 				+ "resolves; the MCP endpoint is re-mounted under host-chosen @Mixin prefix(es) %s, which moves the "
302 				+ "well-known PRM route off the origin root.  Mount the MCP mixin at the origin root (no @Mixin(path/paths)).",
303 				mixinMountPrefixes);
304 	}

322 	 * Serves this endpoint's PRM from the RFC 9728 path-inserted well-known location, but only when the requested suffix
323 	 * matches this resource's expected well-known path (M5); otherwise {@code 404}.  Fails fast on a non-root or re-mounted
324 	 * mount (H1); a {@code 404} when RS auth is disabled.
325 	 *
326 	 * @param cfg The RS config.
327 	 * @param req The REST request.
328 	 * @return The PRM document.
329 	 */
330 	public static McpProtectedResourceMetadata metadataForWellKnownPathInserted(McpResourceServerConfig cfg, RestRequest req) {
331 		if (!cfg.isEnabled())
332 			throw new NotFound();
333 		assertWellKnownMountReachable(req);
334 		var expected = McpWellKnownRouting.wellKnownRequestPath(cfg.getResource());
335 		if (!expected.equals(trimTrailingSlash(req.getHttpServletRequest().getPathInfo())))
336 			throw new NotFound();
337 		return buildMetadata(cfg);
338 	}

340 	private static String trimTrailingSlash(String p) {
341 		if (p == null)
342 			return "";
343 		while (p.length() > 1 && p.endsWith("/"))
344 			p = p.substring(0, p.length() - 1);
345 		return p;
346 	}

350 	 *
351 	 * @param cfg The RS config.  Must be {@link McpResourceServerConfig#isEnabled() enabled} and valid.
352 	 * @return The PRM bean.
353 	 */
354 	public static McpProtectedResourceMetadata buildMetadata(McpResourceServerConfig cfg) {
355 		cfg.validateEnabled();
356 		var m = new McpProtectedResourceMetadata().setResource(cfg.getResource());
357 		if (!cfg.getAuthorizationServers().isEmpty())
358 			m.setAuthorizationServers(new ArrayList<>(cfg.getAuthorizationServers()));
359 		if (!cfg.getScopesSupported().isEmpty())
360 			m.setScopesSupported(new LinkedHashSet<>(cfg.getScopesSupported()));
361 		if (!cfg.getBearerMethodsSupported().isEmpty())
362 			m.setBearerMethodsSupported(new LinkedHashSet<>(cfg.getBearerMethodsSupported()));
363 		return m;
364 	}

376 	/**
377 	 * Authenticates the request against the RS config, establishing the authenticated principal.
378 	 *
379 	 * <p>
380 	 * On success the principal is stashed under {@link RestServerConstants#PRINCIPAL_ATTR} and returned.  On failure the
381 	 * {@code WWW-Authenticate: Bearer ...} challenge (carrying the {@code resource_metadata} pointer) is written directly
382 	 * to {@code res} &mdash; because an exception thrown from a {@code @RestStartCall} hook does not route through the
383 	 * response processors that would otherwise copy an {@link AuthenticationException}'s headers &mdash; and then a
384 	 * {@code 401 Unauthorized} ({@link AuthenticationException}) or {@code 403 Forbidden} (insufficient scope) is thrown.
385 	 *
386 	 * @param cfg The RS config.  Must be {@link McpResourceServerConfig#isEnabled() enabled} and valid.
387 	 * @param req The HTTP request.  Must not be <jk>null</jk>.
388 	 * @param res The HTTP response, used to write the challenge header on failure.  Must not be <jk>null</jk>.
389 	 * @return The authenticated principal.  Never <jk>null</jk>.
390 	 */
391 	public static Principal authenticate(McpResourceServerConfig cfg, HttpServletRequest req, HttpServletResponse res) {
392 		cfg.validateEnabled();
393 		var filter = cfg.oauthFilter();
394 		Optional<AuthResult> result;
395 		try {
396 			result = filter.authenticate(req);
397 		} catch (AuthenticationException e) {
398 			res.setHeader(WWW_AUTHENTICATE, invalidTokenChallenge(cfg));
399 			throw new AuthenticationException(e, "Invalid bearer token");
400 		}
401 		if (result.isEmpty()) {
402 			res.setHeader(WWW_AUTHENTICATE, missingTokenChallenge(cfg));
403 			throw new AuthenticationException("Bearer token required");
404 		}
405 		var auth = result.get();
406 		var principal = auth.getPrincipal();
407 		if (!McpAudienceValidator.matches(principal, cfg.getAudience(), cfg.isRequireAudienceClaim())) {
408 			res.setHeader(WWW_AUTHENTICATE, invalidTokenChallenge(cfg));
409 			throw new AuthenticationException("Token audience mismatch");
410 		}
411 		if (!satisfies(auth.getRoles(), cfg.getRequiredScopes())) {
412 			res.setHeader(WWW_AUTHENTICATE, insufficientScopeChallenge(cfg));
413 			throw new Forbidden("Insufficient scope");
414 		}
415 		// M1: write the framework-standard principal attribute so @Auth Principal resolves downstream.
416 		req.setAttribute(RestServerConstants.PRINCIPAL_ATTR, principal);
417 		// SEP-2350: stash the granted scopes so the POST-parse dispatch point can enforce per-operation step-up scopes.
418 		req.setAttribute(GRANTED_SCOPES_ATTR, Set.copyOf(auth.getRoles()));
419 		return principal;
420 	}

423 	 * Returns the granted scopes stashed by {@link #authenticate} for this request (SEP-2350).
424 	 *
425 	 * @param req The HTTP request.  May be <jk>null</jk>.
426 	 * @return The granted scopes, or an empty set when none were stashed (RS auth disabled / gate not run).
427 	 */
428 	@SuppressWarnings("unchecked")
429 	public static Set<String> grantedScopes(HttpServletRequest req) {
430 		if (req == null)
431 			return Set.of();
432 		var v = req.getAttribute(GRANTED_SCOPES_ATTR);
433 		return v instanceof Set ? (Set<String>)v : Set.of();
434 	}

436 	/**
437 	 * Returns the authenticated {@link Principal} {@link #authenticate} stashed for this request (READY-312f F4).
438 	 *
439 	 * <p>
440 	 * Reads the same framework-standard {@link RestServerConstants#PRINCIPAL_ATTR} attribute {@link #authenticate}
441 	 * writes on success, so this is the single source of truth for "who is calling this MCP request".  It is the seam
442 	 * the {@code 2026-07-28} dispatcher threads into {@link RequestStateCodec#seal}/{@link RequestStateCodec#unseal}
443 	 * for principal-bound {@code requestState} AAD.  Mirrors {@link #grantedScopes(HttpServletRequest)}:
444 	 * a <jk>null</jk> request, an absent attribute, or a non-{@link Principal} value all return <jk>null</jk> &mdash; the
445 	 * anonymous / RS-auth-disabled path, which every caller must handle without an NPE.  Unlike
446 	 * {@link #grantedScopes(HttpServletRequest)}, which falls back to an empty {@link Set} in that same situation, this
447 	 * returns <jk>null</jk> rather than an empty value in that case; the divergence is intentional &mdash; a scalar
448 	 * identity has no natural "empty" representative the way a collection does.
449 	 *
450 	 * @param req The HTTP request.  May be <jk>null</jk>.
451 	 * @return The authenticated principal, or <jk>null</jk> when none was stashed (RS auth disabled / gate not run /
452 	 * 	anonymous caller).
453 	 */
454 	public static Principal principal(HttpServletRequest req) {
455 		if (req == null)
456 			return null;
457 		var v = req.getAttribute(RestServerConstants.PRINCIPAL_ATTR);
458 		return v instanceof Principal p ? p : null;
459 	}

462 	 * Enforces SEP-2350 per-operation step-up scopes at the POST-parse dispatch point.
463 	 *
464 	 * <p>
465 	 * A no-op unless RS auth is {@link McpResourceServerConfig#isEnabled() enabled} and the resolved operation has a
466 	 * non-empty {@link McpResourceServerConfig#requiredScopesFor(McpOperationContext) required-scope set}.  When the
467 	 * caller's {@code grantedScopes} do not {@link #satisfies satisfy} the operation's required scopes, throws a
468 	 * {@code 403 Forbidden} carrying a {@code WWW-Authenticate: Bearer ..., error="insufficient_scope", scope="..."}
469 	 * step-up challenge that names exactly the scopes required for this operation.
470 	 *
471 	 * <p>
472 	 * This is the enforcement point the F2 security review mandated: unlike the {@code @RestStartCall} bearer gate (which
473 	 * fires before the JSON-RPC body is parsed), this runs once the MCP method &mdash; and thus the operation &mdash; is
474 	 * known, so the challenge can be scoped to the specific operation.  It layers on top of, and never weakens, the
475 	 * baseline gate (op-identity, origin-root fail-fast, audience, and baseline required-scope enforcement all still run
476 	 * first in {@link #authenticate}).
477 	 *
478 	 * @param cfg The RS config.  Must not be <jk>null</jk>.
479 	 * @param grantedScopes The caller's granted scopes.  May be <jk>null</jk> (treated as empty).
480 	 * @param ctx The resolved operation.  Must not be <jk>null</jk>.
481 	 * @throws Forbidden {@code 403 insufficient_scope} when the granted scopes do not satisfy the operation's required
482 	 * 	scopes.
483 	 */
484 	public static void enforceOperationScopes(McpResourceServerConfig cfg, Collection<String> grantedScopes, McpOperationContext ctx) {
485 		if (!cfg.isEnabled())
486 			return;
487 		var required = cfg.requiredScopesFor(ctx);
488 		if (required.isEmpty())
489 			return;
490 		var granted = grantedScopes == null ? Set.<String>of() : grantedScopes;
491 		if (satisfies(granted, required))
492 			return;
493 		var e = new Forbidden("Insufficient scope for operation '%s'", ctx.method());
494 		e.setHeader(WWW_AUTHENTICATE, stepUpChallenge(cfg, required));
495 		throw e;
496 	}

499 	 * Builds the SEP-2350 scoped step-up {@code WWW-Authenticate} challenge naming the scopes required to invoke a
500 	 * specific operation ({@code error="insufficient_scope"}, with the operation's {@code scope} set).
501 	 *
502 	 * @param cfg The RS config.
503 	 * @param operationScopes The scopes required to invoke the operation.
504 	 * @return The challenge value.
505 	 */
506 	public static String stepUpChallenge(McpResourceServerConfig cfg, Collection<String> operationScopes) {
507 		return McpBearerChallenge.create()
508 			.realm(cfg.getRealm())
509 			.error("insufficient_scope")
510 			.scope(operationScopes)
511 			.resourceMetadata(metadataUri(cfg))
512 			.build();
513 	}

515 	/**
516 	 * Returns whether the granted scopes satisfy every required scope, honoring OAuth scope hierarchies where a broader
517 	 * granted scope implies its narrower children.
518 	 *
519 	 * <p>
520 	 * A required scope {@code r} is satisfied when some granted scope {@code g} either equals {@code r} exactly, or is a
521 	 * hierarchical <em>ancestor</em> of it &mdash; i.e. {@code r} begins with {@code g} followed by a {@code :} or
522 	 * {@code .} delimiter (so granted {@code repo} implies required {@code repo:read} / {@code repo.write}, but granted
523 	 * {@code repo:read} does <b>not</b> imply required {@code repo}).
524 	 *
525 	 * @param grantedScopes The caller's granted scopes.  Never <jk>null</jk>.
526 	 * @param requiredScopes The scopes required to invoke the operation.  Never <jk>null</jk>.
527 	 * @return <jk>true</jk> if every required scope is satisfied.
528 	 */
529 	public static boolean satisfies(Collection<String> grantedScopes, Collection<String> requiredScopes) {
530 		for (var r : requiredScopes) {
531 			var ok = false;
532 			for (var g : grantedScopes) {
533 				if (g.equals(r) || r.startsWith(g + ":") || r.startsWith(g + ".")) {
534 					ok = true;
535 					break;
536 				}
537 			}
538 			if (!ok)
539 				return false;
540 		}
541 		return true;
542 	}
