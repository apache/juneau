17  package org.apache.juneau.rest.server.mcp.v20260728;
18  
19  import static org.apache.juneau.commons.utils.AssertionUtils.*;
20  
21  import javax.crypto.SecretKey;
22  
23  /**
24   * A key paired with the identifier an operator chose for it, as returned by {@link KeyProvider#currentKey()}
25   * and resolved by {@link KeyProvider#resolveKey(String)}.
26   *
27   * <p>
28   * <b>Never log {@code key}.</b> Mirrors {@link RequestStateCodec}'s "never log" contract for the sealed token:
29   * this record holds the actual sealing key, so {@link #toString()} deliberately redacts {@code key} rather than
30   * relying on the record's auto-generated form, which would print whatever a custom/HSM {@link SecretKey} impl's
31   * own {@code toString()} legally chooses to include.
32   *
33   * @param keyId An opaque, operator-chosen short identifier for {@code key} (e.g. {@code "2026-08-a"}). Must not
34   * 	be <jk>null</jk> or blank, and is length-bounded so a sealed token's cleartext {@code keyId} wire segment
35   * 	(see {@code AeadRequestStateCodec}'s versioned wire format) stays bounded.
36   * @param key The key material. Must not be <jk>null</jk>. The AEAD codec is AES-GCM-specific by design, so
37   * 	non-AES key material fails at cipher-init time in {@code AeadRequestStateCodec}, not here.
38   * @since 10.0.0
39   */
40  public record KeyedSecret(String keyId, SecretKey key) {
41  
42  	// Bounds the base64url'd keyId wire segment so an oversized keyId cannot inflate a sealed token.
43  	private static final int MAX_KEY_ID_CHARS = 128;
44  
45  	/**
46  	 * Compact constructor &mdash; validates {@code keyId} is non-null/non-blank/length-bounded and {@code key}
47  	 * is non-null.
48  	 */
49  	public KeyedSecret {
50  		assertArgNotNullOrBlank("keyId", keyId);
51  		assertArg(keyId.length() <= MAX_KEY_ID_CHARS, "Argument 'keyId' length (%s) exceeds max of %s chars.", keyId.length(), MAX_KEY_ID_CHARS);
52  		assertArgNotNull("key", key);
53  	}
54  
55  	/**
56  	 * Redacts {@code key} so key material never reaches logs via this record's {@code toString()}.
57  	 */
58  	@Override
59  	public String toString() {
60  		return "KeyedSecret[keyId=" + keyId + ", key=<redacted>]";
61  	}
62  }
