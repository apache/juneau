31  /**
32   * Bounded, no-fetch validation of {@code tools/call} arguments against a tool's declared input schema
33   * (Resolution B2).
34   *
35   * <p>
36   * The neutral {@link McpSchema} is an unconstrained JSON object carrier, and a client can send an
37   * arbitrarily large or deep argument object. Before any validation runs, both the schema graph and the
38   * argument graph are checked against the shared {@link JsonValueSafety} depth/node/deadline bounds,
39   * so a hostile or accidentally-huge input is rejected up front rather than blowing the stack or spinning
40   * indefinitely. The two checks share a single deadline, so a slow schema graph cannot buy the argument
41   * graph extra time (or vice versa).
42   *
43   * <p>
44   * <b>No external fetches.</b> The schema is converted to a {@link JsonSchema} bean and validated as-is. No
45   * schema-resolution map is ever installed on the schema, so an external {@code $ref} is inert data:
46   * {@link JsonSchemaValidator} does not resolve references, and with no resolution map installed there is no
47   * code path that opens a network connection or a file to dereference one.
48   *
49   * <p>
50   * Validation itself runs on a fixed-size pool of daemon threads and is bounded by the same overall budget: if
51   * a pathological schema (for example a catastrophically-backtracking {@code pattern}) fails to complete in
52   * time, the task is cancelled and a {@code -32602} error is raised instead of hanging the request thread. That
53   * compute budget is measured from the moment the task actually starts running, not from the moment it is
54   * submitted: time spent waiting for a free thread in {@link #VALIDATION_POOL} (for example under heavy
55   * concurrent build/test load) is scheduling latency, not validation cost, so it is never charged against
56   * {@link #MAX_VALIDATION_MILLIS}. A separate, much more generous {@link #MAX_SCHEDULING_MILLIS} backstop still
57   * bounds the scheduling wait itself, purely so a wedged or saturated pool cannot block the caller forever.
58   */
59  final class McpSchemaSafety {
60  
61  	/** Maximum nesting depth permitted in either the schema graph or the argument graph. */
62  	static final int MAX_DEPTH = 64;
63  
64  	/** Maximum number of nodes permitted in either the schema graph or the argument graph. */
65  	static final int MAX_NODES = 10_000;
66  
67  	/**
68  	 * Maximum compute time permitted for a single schema validation, in milliseconds, measured from the
69  	 * moment the validation task actually starts running on {@link #VALIDATION_POOL} (not from submission).
70  	 * This is the DoS bound: it caps how much CPU a pathological schema can burn, and is deliberately
71  	 * unaffected by how long the task had to wait for a free thread.
72  	 */
73  	static final long MAX_VALIDATION_MILLIS = 100;
74  
75  	/**
76  	 * Maximum time permitted for a submitted validation task to begin executing on {@link #VALIDATION_POOL},
77  	 * in milliseconds. This is a generous circuit breaker against a wedged or fully-saturated pool - not part
78  	 * of the {@link #MAX_VALIDATION_MILLIS} DoS budget - so ordinary scheduling delay (queueing behind other
79  	 * validations under heavy concurrent load) never gets mistaken for an expensive schema.
80  	 */
81  	static final long MAX_SCHEDULING_MILLIS = 2_000;
82  
83  	private static final ExecutorService VALIDATION_POOL = newValidationPool();
84  
85  	private static ExecutorService newValidationPool() {
86  		var pool = new ThreadPoolExecutor(
87  			Math.max(2, Runtime.getRuntime().availableProcessors()),
88  			Math.max(2, Runtime.getRuntime().availableProcessors()),
89  			0, MILLISECONDS,
90  			new LinkedBlockingQueue<>(),
91  			r -> {
92  				var t = new Thread(r, "mcp-2026-07-28-schema-validation");
93  				t.setDaemon(true);
94  				return t;
95  			});
96  		pool.prestartAllCoreThreads(); // Warm the pool so the first validation after startup doesn't pay thread-creation cost.
97  		return pool;
98  	}
99  
100 	private McpSchemaSafety() {}
101 
102 	/**
103 	 * Validates a {@code tools/call} argument object against a tool's declared input schema, subject to the
104 	 * B2 bounds.
105 	 *
106 	 * @param schema The tool's neutral input schema, or <jk>null</jk> to skip validation entirely.
107 	 * @param args The argument object to validate. Never <jk>null</jk> (an absent argument object is an empty map).
108 	 * @throws McpException {@code -32602} if either graph exceeds {@link #MAX_DEPTH}/{@link #MAX_NODES}, if the
109 	 * 	arguments do not satisfy the schema, or if validation exceeds {@link #MAX_VALIDATION_MILLIS}.
110 	 */
111 	static void validateInput(McpSchema schema, Map<String,Object> args) {
112 		if (schema == null)
113 			return;
114 		var schemaMap = schema.toJsonMap();
115 		var deadline = JsonValueSafety.deadlineNanos();
116 		try {
117 			JsonValueSafety.check(schemaMap, "Tool input schema", deadline);
118 			JsonValueSafety.check(args, "Tool input", deadline);
119 		} catch (IllegalArgumentException e) {
120 			throw new McpException(McpRevision.CODE_INVALID_PARAMS, e.getMessage());
121 		}
122 		var jsonSchema = Json.to(Json.of(schemaMap), JsonSchema.class);
123 		validateBounded(jsonSchema, args, deadline);
124 	}
125 
126 	/**
127 	 * Runs {@link JsonSchemaValidator} on a daemon thread and enforces the remaining share of the shared
128 	 * {@link JsonValueSafety} deadline against the task's own compute time - never against however long it
129 	 * had to wait in {@link #VALIDATION_POOL} for a free thread.
130 	 */
131 	private static void validateBounded(JsonSchema<?> schema, Object value, long deadlineNanos) {
132 		var remaining = JsonValueSafety.remainingNanos(deadlineNanos);
133 		if (remaining == 0)
134 			throw validationTimeoutException();
135 		var started = new CountDownLatch(1);
136 		var startedAtNanos = new AtomicLong();
137 		var future = VALIDATION_POOL.submit(() -> {
138 			startedAtNanos.set(System.nanoTime());
139 			started.countDown();
140 			JsonSchemaValidator.of(schema).validate(value);
141 			return null;
142 		});
143 		awaitBounded(future, started, startedAtNanos, remaining);
144 	}
145 
146 	/**
147 	 * Waits for a submitted validation task to complete, charging only its own compute time - measured from
148 	 * {@code startedAtNanos}, which the task sets as its first instruction - against {@code remaining}, the
149 	 * caller's share of the shared {@link JsonValueSafety} deadline.
150 	 *
151 	 * <p>
152 	 * This waits in two phases. First, {@code started} is awaited so the calling thread learns exactly when
153 	 * the task begins running; this wait is bounded only by the generous {@link #MAX_SCHEDULING_MILLIS}
154 	 * backstop, since scheduling delay under load (queueing behind other work in {@link #VALIDATION_POOL})
155 	 * is not the thing being defended against. Second, once running, the task is given the full
156 	 * {@code remaining} share of the deadline as its own fresh compute window - anchored to its actual start
157 	 * time rather than to submission time - so a busy pool cannot eat into the budget that is meant to cap
158 	 * the validator's own work.
159 	 *
160 	 * <p>
161 	 * Package-private (rather than private) purely so this can be exercised directly against a test-local
162 	 * {@link Future}/latch pair - deterministically simulating scheduling delay - without needing to saturate
163 	 * the shared {@link #VALIDATION_POOL}.
164 	 *
165 	 * @param future The in-flight (or already-complete) validation task.
166 	 * @param started Counted down by the task as its first instruction, once it actually begins running.
167 	 * @param startedAtNanos Set by the task (before counting down {@code started}) to {@link System#nanoTime()}.
168 	 * @param remaining The caller's remaining share of the shared deadline, in nanoseconds, captured before submission.
169 	 */
170 	static void awaitBounded(Future<?> future, CountDownLatch started, AtomicLong startedAtNanos, long remaining) {
171 		try {
172 			if (! started.await(MAX_SCHEDULING_MILLIS, MILLISECONDS)) {
173 				future.cancel(true);
174 				throw new McpException(McpRevision.CODE_INVALID_PARAMS, "Tool input schema validation could not be scheduled within " + MAX_SCHEDULING_MILLIS + " ms");
175 			}
176 			var computeRemainingNanos = remaining - (System.nanoTime() - startedAtNanos.get());
177 			future.get(Math.max(0, computeRemainingNanos), NANOSECONDS);
178 		} catch (TimeoutException e) {
179 			future.cancel(true);
180 			throw validationTimeoutException();
181 		} catch (ExecutionException e) {
182 			var cause = e.getCause();
183 			if (cause instanceof McpException me)
184 				throw me;
185 			throw new McpException(McpRevision.CODE_INVALID_PARAMS, cause == null ? e.getMessage() : cause.getMessage());
186 		} catch (InterruptedException e) {
187 			future.cancel(true);
188 			Thread.currentThread().interrupt();
189 			throw new McpException(McpRevision.CODE_INVALID_PARAMS, "Tool input schema validation was interrupted");
190 		}
191 	}
192 
193 	private static McpException validationTimeoutException() {
194 		return new McpException(McpRevision.CODE_INVALID_PARAMS, "Tool input schema validation exceeded " + MAX_VALIDATION_MILLIS + " ms");
195 	}
