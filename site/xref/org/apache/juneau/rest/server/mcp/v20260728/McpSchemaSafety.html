32  /**
33   * Bounded, no-fetch validation of {@code tools/call} arguments against a tool's declared input schema.
34   *
35   * <p>
36   * The neutral {@link McpSchema} is an unconstrained JSON object carrier, and a client can send an
37   * arbitrarily large or deep argument object. Before any validation runs, both the schema graph and the
38   * argument graph are checked against the shared {@link JsonValueSafety} depth/node/deadline bounds,
39   * so a hostile or accidentally-huge input is rejected up front rather than blowing the stack or spinning
40   * indefinitely. The two checks share a single deadline, so a slow schema graph cannot buy the argument
41   * graph extra time (or vice versa).
42   *
43   * <p>
44   * <b>No external fetches.</b> The schema is converted to a {@link JsonSchema} bean and validated as-is. No
45   * schema-resolution map is ever installed on the schema, so an external {@code $ref} is inert data:
46   * {@link JsonSchemaValidator} does not resolve references, and with no resolution map installed there is no
47   * code path that opens a network connection or a file to dereference one.
48   *
49   * <p>
50   * Validation itself runs on a fixed-size pool of daemon threads and is bounded by its own independent compute
51   * budget ({@link #MAX_VALIDATION_MILLIS}), measured fresh from when the validation task actually starts running
52   * &mdash; <b>not</b> from the structural-traversal deadline above, so wall-clock time spent in the structural
53   * pre-checks or the schema-bean conversion never shrinks it (see {@link #validateBounded} for why). If a
54   * pathological schema (for example a catastrophically-backtracking {@code pattern}) fails to complete in time,
55   * the task is cancelled and a {@code -32602} error is raised instead of hanging the request thread.
56   *
57   * <p>
58   * <b>Cancellation actually reclaims the worker thread.</b> When the budget trips, {@link #awaitBounded} calls
59   * {@link Future#cancel(boolean) future.cancel(true)}, which interrupts the validating thread. A
60   * {@link java.util.regex.Matcher} does not observe {@link Thread#interrupt()} on its own, so historically a runaway
61   * regex kept burning a core in the background even though the client already had its fast {@code -32602} - a mild DoS
62   * residual where a flood of such requests could pin a core each. {@link JsonSchemaValidator} now feeds the matched
63   * string through an interruptible {@link CharSequence} (see its class notes), so an interrupt aborts a
64   * catastrophically-backtracking match promptly and the pool thread returns to idle rather than spinning. The external
65   * contract is unchanged: a genuine overrun still returns the same {@code -32602}, and the CPU-time budget with its
66   * wall-clock fallback is untouched.
67   *
68   * <p>
69   * <b>The compute budget is charged against the validating thread's actual CPU time, not wall-clock time.</b>
70   * The DoS threat being defended against is a schema that burns CPU (catastrophic backtracking, quadratic
71   * blowups); the amount of CPU a validation consumes is exactly what that budget should cap. Measuring
72   * wall-clock instead would wrongly count time the validating thread was <i>not</i> running - time it spent
73   * preempted by the OS scheduler, or queued behind sibling work - as if it were validation cost. Under heavy
74   * concurrent load (for example {@code juneau-integration-tests} running Surefire with {@code forkCount=8}) that
75   * false accounting can trip the budget on trivial input, and the same false-positive can bite a saturated
76   * production deployment. Sampling {@link ThreadMXBean#getThreadCpuTime(long) thread CPU time} on the thread
77   * that actually does the work means only real compute counts: preemption and scheduling latency no longer
78   * shrink the budget, while a genuinely expensive validation still trips it and returns the same {@code -32602}
79   * error. On a JVM where per-thread CPU timing is unavailable, {@link #awaitBounded} transparently falls back to
80   * the original wall-clock measurement so the guard still functions everywhere.
81   *
82   * <p>
83   * Independently of the compute budget, time spent waiting for a free thread in {@link #VALIDATION_POOL} is
84   * scheduling latency, not validation cost, so it is never charged against {@link #MAX_VALIDATION_MILLIS}: the
85   * budget clock (CPU or wall-clock) only starts once the task is actually running. A separate, much more
86   * generous {@link #MAX_SCHEDULING_MILLIS} backstop bounds the scheduling wait itself, purely so a wedged or
87   * saturated pool cannot block the caller forever.
88   */
89  final class McpSchemaSafety {
90  
91  	/** Maximum nesting depth permitted in either the schema graph or the argument graph. */
92  	static final int MAX_DEPTH = 64;
93  
94  	/** Maximum number of nodes permitted in either the schema graph or the argument graph. */
95  	static final int MAX_NODES = 10_000;
96  
97  	/**
98  	 * Maximum compute time permitted for a single schema validation, in milliseconds, measured as the
99  	 * validating thread's actual CPU time (see class-level notes) - not from submission, and not as wall-clock.
100 	 * This is the DoS bound: it caps how much CPU a pathological schema can burn, and is deliberately
101 	 * unaffected both by how long the task had to wait for a free thread and by any OS preemption while it runs.
102 	 */
103 	static final long MAX_VALIDATION_MILLIS = 100;
104 
105 	/**
106 	 * Maximum time permitted for a submitted validation task to begin executing on {@link #VALIDATION_POOL},
107 	 * in milliseconds. This is a generous circuit breaker against a wedged or fully-saturated pool - not part
108 	 * of the {@link #MAX_VALIDATION_MILLIS} DoS budget - so ordinary scheduling delay (queueing behind other
109 	 * validations under heavy concurrent load) never gets mistaken for an expensive schema.
110 	 */
111 	static final long MAX_SCHEDULING_MILLIS = 2_000;
112 
113 	private static final ThreadMXBean THREAD_MX = ManagementFactory.getThreadMXBean();
114 
115 	/**
116 	 * Whether per-thread CPU-time measurement is available (and enabled) on this JVM. When <jk>true</jk>, the
117 	 * compute budget is charged against the validating thread's actual CPU time; when <jk>false</jk> (a JVM that
118 	 * cannot report per-thread CPU time), {@link #awaitBounded} falls back to the original wall-clock measurement
119 	 * so the DoS guard still functions everywhere.
120 	 */
121 	private static final boolean CPU_TIME_SUPPORTED = initCpuTimeSupport();
122 
123 	/**
124 	 * How often the waiting thread re-samples the validating thread's accumulated CPU time while a validation is
125 	 * still running. Small enough that a runaway validation is stopped promptly once it burns past the budget,
126 	 * large enough that the sampling overhead is negligible.
127 	 */
128 	private static final long CPU_POLL_INTERVAL_NANOS = MILLISECONDS.toNanos(5);
129 
130 	private static boolean initCpuTimeSupport() {
131 		if (! THREAD_MX.isThreadCpuTimeSupported())
132 			return false;  // HTT: CI JVMs support per-thread CPU time; this fallback path isn't reachable there.
133 		if (! THREAD_MX.isThreadCpuTimeEnabled()) {
134 			try {
135 				THREAD_MX.setThreadCpuTimeEnabled(true);  // HTT: HotSpot enables thread CPU time by default; not reachable in CI.
136 			} catch (@SuppressWarnings("unused") SecurityException | UnsupportedOperationException e) {
137 				return false;  // HTT: enabling is rejected only under a restrictive SecurityManager; not reachable in CI.
138 			}
139 		}
140 		return true;
141 	}

143 	private static final ExecutorService VALIDATION_POOL = newValidationPool();
144 
145 	private static ExecutorService newValidationPool() {
146 		var pool = new ThreadPoolExecutor(
147 			Math.max(2, Runtime.getRuntime().availableProcessors()),
148 			Math.max(2, Runtime.getRuntime().availableProcessors()),
149 			0, MILLISECONDS,
150 			new LinkedBlockingQueue<>(),
151 			r -> {
152 				var t = new Thread(r, "mcp-2026-07-28-schema-validation");
153 				t.setDaemon(true);
154 				return t;
155 			});
156 		pool.prestartAllCoreThreads(); // Warm the pool so the first validation after startup doesn't pay thread-creation cost.
157 		return pool;
158 	}

160 	private McpSchemaSafety() {}
161 
162 	/**
163 	 * Validates a {@code tools/call} argument object against a tool's declared input schema, subject to the
164 	 * configured bounds.
165 	 *
166 	 * @param schema The tool's neutral input schema, or <jk>null</jk> to skip validation entirely.
167 	 * @param args The argument object to validate. Never <jk>null</jk> (an absent argument object is an empty map).
168 	 * @throws McpException {@code -32602} if either graph exceeds {@link #MAX_DEPTH}/{@link #MAX_NODES}, if the
169 	 * 	arguments do not satisfy the schema, or if validation exceeds {@link #MAX_VALIDATION_MILLIS}.
170 	 */
171 	static void validateInput(McpSchema schema, Map<String,Object> args) {
172 		if (schema == null)
173 			return;
174 		var schemaMap = schema.toJsonMap();
175 		var deadline = JsonValueSafety.deadlineNanos();
176 		try {
177 			JsonValueSafety.check(schemaMap, "Tool input schema", deadline);
178 			JsonValueSafety.check(args, "Tool input", deadline);
179 		} catch (IllegalArgumentException e) {
180 			throw new McpException(McpRevision.CODE_INVALID_PARAMS, e.getMessage());
181 		}
182 		var jsonSchema = Json.to(Json.of(schemaMap), JsonSchema.class);
183 		validateBounded(jsonSchema, args);
184 	}
185 
186 	/**
187 	 * Runs {@link JsonSchemaValidator} on a daemon thread and enforces a fresh {@link #MAX_VALIDATION_MILLIS}
188 	 * compute budget against the task's own compute time - deliberately <b>not</b> the caller's remaining share
189 	 * of the structural-traversal deadline computed in {@link #validateInput}. That deadline is a wall-clock
190 	 * budget that is already partially spent by the two {@link JsonValueSafety#check} calls and the
191 	 * schema-to-bean conversion above, both of which run on the (possibly preempted) request thread under load;
192 	 * deriving the validation budget from whatever wall-clock happened to remain would let purely-external
193 	 * scheduling pressure - not any property of the schema or arguments being validated - trip a false-positive
194 	 * {@code -32602} on a trivial validation. Anchoring a fresh budget here, measured from when the pool task
195 	 * actually starts (see {@link TaskStart#capture}), keeps the DoS guard (validation CPU still capped at
196 	 * {@link #MAX_VALIDATION_MILLIS}) independent of that pre-check wall-clock entirely. Also never against
197 	 * however long the task had to wait in {@link #VALIDATION_POOL} for a free thread, and (when CPU timing is
198 	 * available) never against wall-clock time it spent preempted rather than running.
199 	 */
200 	private static void validateBounded(JsonSchema<?> schema, Object value) {
201 		var started = new CountDownLatch(1);
202 		var taskStart = new AtomicReference<TaskStart>();
203 		var future = VALIDATION_POOL.submit(() -> {
204 			taskStart.set(TaskStart.capture());
205 			started.countDown();
206 			JsonSchemaValidator.of(schema).validate(value);
207 			return null;
208 		});
209 		awaitBounded(future, started, taskStart, MILLISECONDS.toNanos(MAX_VALIDATION_MILLIS));
210 	}
211 
212 	/**
213 	 * Snapshot of when a validation task actually began running, captured on the validating thread itself as its
214 	 * first instruction.
215 	 *
216 	 * @param threadId The validating thread's id, so the waiting thread can sample that same thread's CPU time.
217 	 * @param wallNanos The {@link System#nanoTime()} baseline, used by the wall-clock fallback.
218 	 * @param cpuNanos The validating thread's CPU-time baseline in nanoseconds ({@link ThreadMXBean#getThreadCpuTime(long)}),
219 	 * 	or {@code -1} if per-thread CPU timing is unavailable on this JVM.
220 	 */
221 	record TaskStart(long threadId, long wallNanos, long cpuNanos) {
222 
223 		/** Captures the current thread's start snapshot; must be called on the thread that runs the validation. */
224 		static TaskStart capture() {
225 			var id = Thread.currentThread().getId();
226 			return new TaskStart(id, System.nanoTime(), CPU_TIME_SUPPORTED ? THREAD_MX.getThreadCpuTime(id) : -1L);
227 		}
228 	}

230 	/**
231 	 * Waits for a submitted validation task to complete, charging only its own compute time against
232 	 * {@code remaining} (the caller's compute budget, a fresh interval unrelated to any other deadline the
233 	 * caller may separately be tracking) - never however long it had to wait in {@link #VALIDATION_POOL} for a
234 	 * free thread.
235 	 *
236 	 * <p>
237 	 * This waits in two phases. First, {@code started} is awaited so the calling thread learns exactly when the
238 	 * task begins running; this wait is bounded only by the generous {@link #MAX_SCHEDULING_MILLIS} backstop,
239 	 * since scheduling delay under load (queueing behind other work in {@link #VALIDATION_POOL}) is not the thing
240 	 * being defended against. Second, once the task is running, its compute is bounded: when per-thread CPU
241 	 * timing is available (the normal case) the waiting thread polls the validating thread's accumulated
242 	 * <i>CPU</i> time and trips only if that exceeds {@code remaining}, so preemption and scheduling latency
243 	 * never shrink the budget - only real CPU work does. When CPU timing is unavailable, it falls back to the
244 	 * original wall-clock window anchored to the task's actual start time.
245 	 *
246 	 * <p>
247 	 * Package-private (rather than private) purely so this can be exercised directly against a test-local
248 	 * {@link Future}/latch pair - deterministically simulating scheduling delay, a CPU burn, or a sleep that
249 	 * elapses wall-clock without consuming CPU - without needing to saturate the shared {@link #VALIDATION_POOL}.
250 	 *
251 	 * @param future The in-flight (or already-complete) validation task.
252 	 * @param started Counted down by the task as its first instruction, once it actually begins running.
253 	 * @param taskStart Set by the task (before counting down {@code started}) to its {@link TaskStart} snapshot.
254 	 * @param remaining The caller's compute budget, in nanoseconds, captured fresh before submission (see
255 	 * 	{@link #validateBounded} for why this must be independent of any pre-existing wall-clock deadline).
256 	 */
257 	static void awaitBounded(Future<?> future, CountDownLatch started, AtomicReference<TaskStart> taskStart, long remaining) {
258 		try {
259 			if (! started.await(MAX_SCHEDULING_MILLIS, MILLISECONDS)) {
260 				future.cancel(true);
261 				throw new McpException(McpRevision.CODE_INVALID_PARAMS, "Tool input schema validation could not be scheduled within " + MAX_SCHEDULING_MILLIS + " ms");
262 			}
263 			var start = taskStart.get();
264 			if (CPU_TIME_SUPPORTED && start.cpuNanos() >= 0)
265 				awaitByCpuTime(future, start.threadId(), start.cpuNanos(), remaining);
266 			else
267 				future.get(Math.max(0, remaining - (System.nanoTime() - start.wallNanos())), NANOSECONDS);  // HTT: wall-clock fallback; CI JVMs use the CPU-time path.
268 		} catch (TimeoutException e) {
269 			future.cancel(true);
270 			throw validationTimeoutException();
271 		} catch (ExecutionException e) {
272 			var cause = e.getCause();
273 			if (cause instanceof McpException me)
274 				throw me;
275 			throw new McpException(McpRevision.CODE_INVALID_PARAMS, cause == null ? e.getMessage() : cause.getMessage());
276 		} catch (InterruptedException e) {
277 			future.cancel(true);
278 			Thread.currentThread().interrupt();
279 			throw new McpException(McpRevision.CODE_INVALID_PARAMS, "Tool input schema validation was interrupted");
280 		}
281 	}

283 	/**
284 	 * Bounds a running validation by the validating thread's actual CPU time.
285 	 *
286 	 * <p>
287 	 * The waiting thread polls at {@link #CPU_POLL_INTERVAL_NANOS} intervals: each time the task hasn't finished
288 	 * yet, it re-samples thread CPU time and keeps waiting as long as the CPU consumed since {@code baseCpuNanos}
289 	 * is within {@code budgetNanos}. Wall-clock elapsed while the thread was preempted (or deliberately sleeping)
290 	 * accrues no CPU, so it never trips the budget; a schema that genuinely burns CPU does. Validation is pure
291 	 * in-memory work with no I/O or blocking, so CPU accrual is a sufficient bound and no wall-clock ceiling is
292 	 * needed here - a task that never returns would necessarily be burning CPU and will trip the budget.
293 	 *
294 	 * @param future The running validation task.
295 	 * @param threadId The validating thread's id.
296 	 * @param baseCpuNanos The validating thread's CPU-time baseline captured at task start.
297 	 * @param budgetNanos The maximum CPU time, in nanoseconds, the validation may consume.
298 	 * @throws TimeoutException If the CPU budget is exceeded while the task is still running.
299 	 */
300 	private static void awaitByCpuTime(Future<?> future, long threadId, long baseCpuNanos, long budgetNanos)
301 			throws InterruptedException, ExecutionException, TimeoutException {
302 		while (true) {
303 			try {
304 				future.get(CPU_POLL_INTERVAL_NANOS, NANOSECONDS);
305 				return;
306 			} catch (TimeoutException poll) {
307 				if (! stillWithinCpuBudget(future, threadId, baseCpuNanos, budgetNanos))
308 					throw poll;  // CPU work exceeded the DoS budget (or CPU timing was lost) while the task is still running
309 			}
310 		}
311 	}
312 
313 	/**
314 	 * Whether polling should keep waiting after a {@code future.get()} poll timed out: either the validating
315 	 * thread's CPU consumption since {@code baseCpuNanos} is still within {@code budgetNanos} (real CPU work,
316 	 * still within budget - preemption/scheduling doesn't count against it), or the task actually finished
317 	 * inside the sampling race window (the next {@code get()} will harvest its result/exception).
318 	 *
319 	 * @param future The running validation task.
320 	 * @param threadId The validating thread's id.
321 	 * @param baseCpuNanos The validating thread's CPU-time baseline captured at task start.
322 	 * @param budgetNanos The maximum CPU time, in nanoseconds, the validation may consume.
323 	 * @return <jk>true</jk> if polling should continue; <jk>false</jk> if the budget has been exceeded.
324 	 */
325 	private static boolean stillWithinCpuBudget(Future<?> future, long threadId, long baseCpuNanos, long budgetNanos) {
326 		var cpuNow = THREAD_MX.getThreadCpuTime(threadId);
327 		if (cpuNow >= 0 && cpuNow - baseCpuNanos <= budgetNanos)
328 			return true;
329 		return future.isDone();
330 	}
331 
332 	/** Whether the compute budget is charged against thread CPU time (vs. the wall-clock fallback); for tests. */
333 	static boolean cpuTimeBudgetEnabled() {
334 		return CPU_TIME_SUPPORTED;
335 	}

337 	private static McpException validationTimeoutException() {
338 		return new McpException(McpRevision.CODE_INVALID_PARAMS, "Tool input schema validation exceeded " + MAX_VALIDATION_MILLIS + " ms");
339 	}
