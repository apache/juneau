18  
19  import static org.apache.juneau.commons.utils.AssertionUtils.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.marshall.marshaller.*;
24  
25  /**
26   * Bean stashed into the {@link org.apache.juneau.commons.inject.BeanStore} passed to a handler on a RESUME call,
27   * carrying the decoded continuation and the client's collected answers.
28   *
29   * <p>
30   * Absent (not stashed) on a first-round call &mdash; a handler that only ever completes in one round never has to
31   * know this type exists. A resumable handler calls {@code ctx.getBean(McpMrtrResumeContext.class)} to detect
32   * resume and retrieve both fields.
33   *
34   * <p>
35   * <b>Per-round arguments are client-controlled; the continuation is not.</b> On RESUME the handler receives its
36   * ordinary {@code arguments} from the follow-up request &mdash; the client re-sends them each round and full
37   * schema validation runs per round, but nothing about the original round's arguments is sealed or verified, so a
38   * client can alter them between rounds and the MRTR layer will accept the resume. Only {@link #continuation()}
39   * (sealed by {@link RequestStateCodec}, AEAD-authenticated) is tamper-evident across rounds. Therefore any value
40   * a handler must trust across rounds must be carried <i>in</i> the continuation, never re-read from the per-round
41   * arguments. The client is expected to re-send the original {@code arguments} on every round; a handler that
42   * needs the exact original arguments should stash them in the continuation rather than rely on the client
43   * echoing them faithfully.
44   *
45   * <p>
46   * <b>Continuation type fidelity (round-trip contract).</b> {@link #continuation()} is whatever
47   * {@link RequestStateCodec#unseal} decoded from the client's echoed {@code requestState}: a <i>generic JSON</i>
48   * value ({@code JsonMap}/{@code JsonList}/boxed primitive/{@code String}), <b>never</b> the original Java type
49   * the handler threw in {@link McpInputRequiredSignal} (see the type-fidelity note on {@link McpRequestState}).
50   * Rather than force handler authors to parse that generic value by hand, {@link #continuationAs(Class)} converts
51   * it back to any Juneau-marshallable target type in one call, so a handler that paused with a bean recovers a
52   * bean of the same shape ergonomically.
53   *
54   * @param continuation The decoded continuation, exactly as {@link RequestStateCodec#unseal} produced it. Can be
55   * 	<jk>null</jk> if the handler paused with a <jk>null</jk> continuation. See the round-trip contract above:
56   * 	it is generic JSON, not the original Java type &mdash; use {@link #continuationAs(Class)} to recover a typed
57   * 	view.
58   * @param inputResponses The client's collected answers, echoed from the follow-up request's
59   * 	{@code inputResponses} field. Never <jk>null</jk> (an empty map when the client sent none).
60   */
61  public record McpMrtrResumeContext(Object continuation, Map<String,Object> inputResponses) {
62  
63  	/**
64  	 * Returns the continuation converted to the given type.
65  	 *
66  	 * <p>
67  	 * Convenience for the round-trip contract documented on this record: because {@link #continuation()} is a
68  	 * generic JSON value (not the handler's original Java type), a handler that paused with, say, a
69  	 * {@code ResumeState} bean would otherwise have to reconstruct it manually from a {@code JsonMap}. This
70  	 * converts the generic value to {@code type} via Juneau's JSON marshaller &mdash; the same mechanism the
71  	 * codec used to serialize it &mdash; so {@code continuationAs(ResumeState.class)} returns a fully populated
72  	 * {@code ResumeState}.
73  	 *
74  	 * <p>
75  	 * <b>Failure contract:</b> conversion is best-effort against a generic JSON value. If the continuation's
76  	 * decoded shape is incompatible with {@code type}, the underlying marshaller throws (a
77  	 * {@code RuntimeException}); a handler that lets it propagate surfaces the {@code -32603} internal-error
78  	 * fail-safe from {@code McpRevision#dispatch}. A handler that pauses and resumes with a single, consistent
79  	 * continuation type never hits this.
80  	 *
81  	 * @param <T> The target type.
82  	 * @param type The target type to convert the continuation to. Must not be <jk>null</jk>.
83  	 * @return The continuation converted to {@code type}, or <jk>null</jk> if the continuation is <jk>null</jk>.
84  	 * @throws IllegalArgumentException If {@code type} is <jk>null</jk>.
85  	 * @throws RuntimeException If the continuation's decoded shape cannot be converted to {@code type}.
86  	 */
87  	public <T> T continuationAs(Class<T> type) {
88  		assertArgNotNull("type", type);
89  		if (continuation == null)
90  			return null;
91  		return Json.to(Json.of(continuation), type);
92  	}
93  
94  	/**
95  	 * Returns the continuation as a {@link String}.
96  	 *
97  	 * <p>
98  	 * Convenience for the common case documented on {@link #continuationAs(Class)}: a handler that pauses with
99  	 * a plain {@code String} continuation can call this instead of spelling out {@code continuationAs(String.class)}
100 	 * at the headline pause/resume call site.
101 	 *
102 	 * <p>
103 	 * <b>Conversion is not lenient.</b> {@code continuationAs(String.class)} routes through the JSON marshaller,
104 	 * which parses {@code String} the same as any other target type &mdash; it requires a quoted JSON string
105 	 * literal, so a continuation that decoded to anything else (a bare number, a {@code JsonMap}, a
106 	 * {@code JsonList}, a boolean) throws below rather than being stringified. Only a continuation whose
107 	 * decoded shape is already a {@link String} converts successfully.
108 	 *
109 	 * @return The continuation as a {@link String}, or <jk>null</jk> if the continuation is <jk>null</jk>.
110 	 * @throws RuntimeException If the continuation's decoded shape is not already a {@link String}.
111 	 */
112 	public String continuationAsString() {
113 		return continuationAs(String.class);
114 	}
115 }
