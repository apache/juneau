17  package org.apache.juneau.rest.server.mcp;
18  
19  import static org.apache.juneau.commons.utils.AssertionUtils.*;
20  
21  import java.util.*;
22  import java.util.function.*;
23  
24  import org.apache.juneau.commons.inject.*;
25  
26  /**
27   * Handler for a single MCP tool.
28   *
29   * <p>
30   * Implementations declare a {@link #descriptor() descriptor} (returned from {@code tools/list}) and
31   * a {@link #call(Map, BeanStore) call} body invoked when a matching {@code tools/call} arrives. Both
32   * use the revision-neutral model, so a handler compiles against exactly one protocol revision's
33   * worth of assumptions: none.
34   *
35   * <p>
36   * The {@link BeanStore} argument is the per-request bean store, letting handlers look up additional
37   * services (or the underlying {@code RestRequest}) without this interface depending on REST runtime types.
38   *
39   * <p>
40   * This is a genuine two-method contract, not a single-method one: both {@link #descriptor()} and
41   * {@link #call(Map, BeanStore)} are abstract, so it is deliberately <b>not</b> annotated
42   * {@code @FunctionalInterface} and a bare lambda cannot implement it. Routing calls {@link #descriptor()}
43   * on every registered handler for both {@code tools/list} and {@code tools/call}, so a handler with no
44   * usable descriptor is not a valid handler; making {@link #descriptor()} abstract catches that at compile
45   * time instead of the first request to reach it at runtime. Use {@link #of(McpToolSpec, BiFunction)} for
46   * the common case of wiring both in a single expression.
47   */
48  public interface McpToolHandler {
49  
50  	/**
51  	 * Builds a handler from a fixed descriptor and a {@code call} lambda.
52  	 *
53  	 * <p>
54  	 * The sanctioned one-expression path for registering a tool: unlike a bare lambda (which cannot
55  	 * satisfy this two-method interface at all), this factory wires both {@link #descriptor()} and
56  	 * {@link #call(Map, BeanStore)} from a single call.
57  	 *
58  	 * @param descriptor The static descriptor returned from every {@link #descriptor()} call. Must not be
59  	 * 	<jk>null</jk> and must carry a non-blank {@link McpToolSpec#getName() name} (routing matches
60  	 * 	incoming {@code tools/call} requests on name, so a nameless tool would be silently unreachable).
61  	 * @param call The call body, invoked with the incoming arguments and the per-request bean store. Must not be <jk>null</jk>.
62  	 * @return A new handler wiring both. Never <jk>null</jk>.
63  	 */
64  	static McpToolHandler of(McpToolSpec descriptor, BiFunction<Map<String,Object>,BeanStore,McpToolOutcome> call) {
65  		assertArgNotNull("descriptor", descriptor);
66  		assertArgNotNullOrBlank("descriptor.getName()", descriptor.getName());
67  		assertArgNotNull("call", call);
68  		return new McpToolHandler() {
69  			@Override public McpToolSpec descriptor() {
70  				return descriptor;
71  			}
72  
73  			@Override public McpToolOutcome call(Map<String,Object> arguments, BeanStore ctx) {
74  				return call.apply(arguments, ctx);
75  			}
76  		};
77  	}
78  
79  	/**
80  	 * Returns the static descriptor for this tool.
81  	 *
82  	 * <p>
83  	 * The {@link McpToolSpec#getName() name} value is used by the bound {@link McpRevision} to route
84  	 * incoming {@code tools/call} requests, so each handler in an {@link McpServerConfig} must use a
85  	 * unique name.
86  	 *
87  	 * @return The tool descriptor. Never {@code null}.
88  	 */
89  	McpToolSpec descriptor();
90  
91  	/**
92  	 * Invokes the tool.
93  	 *
94  	 * @param arguments The arguments object passed in the JSON-RPC params (never {@code null}; empty map when omitted).
95  	 * @param ctx Per-request bean store. Never {@code null}.
96  	 * @return The call result. Never {@code null}.
97  	 */
98  	McpToolOutcome call(Map<String,Object> arguments, BeanStore ctx);
99  }
