17  package org.apache.juneau.rest.server.mcp;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.bean.jsonrpc.*;
22  
23  /**
24   * Revision-neutral, as-returned-by-the-application result of an {@link McpCompleter} invocation.
25   *
26   * <p>
27   * Supersedes the wire-level {@code CompleteResult}/{@code Completion} beans. An instance of this class
28   * is the completer's raw, unvalidated output; call {@link #normalize(McpCompletionResult)} exactly once
29   * per completion dispatch to obtain the {@link Normalized} shape both dated adapters map to their wire
30   * result.
31   *
32   * <h5 class='section'>Normalization contract:</h5>
33   * <p>
34   * Applied by {@link #normalize(McpCompletionResult)} before wire mapping:
35   * <ul>
36   * 	<li><jk>null</jk> {@link #getValues() values} normalizes to an empty list;
37   * 	<li>order is preserved (application-owned ranking) and duplicates are preserved;
38   * 	<li>a <jk>null</jk> element within {@code values} is an <b>internal handler failure</b>;
39   * 	<li>the first {@value #MAX_VALUES} values are emitted; truncation forces {@code hasMore=true};
40   * 	<li>a supplied non-negative {@link #getTotal() total} is preserved; a negative {@code total} is an
41   * 		<b>internal handler failure</b>;
42   * 	<li>without truncation, a nullable {@link #getHasMore() hasMore} is passed through unchanged; and
43   * 	<li>a completer returning <jk>null</jk> itself (rather than an {@code McpCompletionResult} with
44   * 		<jk>null</jk> fields) is also an <b>internal handler failure</b>.
45   * </ul>
46   *
47   * <h5 class='section'>Internal handler failure representation:</h5>
48   * <p>
49   * Each of the three "internal handler failure" cases above is surfaced identically: as an
50   * {@link McpException} carrying {@link #CODE_INTERNAL_ERROR} ({@code -32603}), the JSON-RPC 2.0
51   * standard "Internal error" code. That code is fixed by the JSON-RPC specification, not chosen per
52   * revision, which is why it is a local constant here rather than routed through
53   * {@code McpRevision#errorCode(McpErrorKind)} - the same reasoning {@code McpParamUtils} documents for
54   * its {@code -32602} constant. Both dated adapters' existing top-level {@code catch (McpException e)}
55   * dispatch handling (see {@code McpRevision#dispatch}) already converts any {@link McpException} thrown
56   * during {@code completion/complete} handling straight into a JSON-RPC error response with this code, so
57   * adapters do not need any additional mapping step beyond calling
58   * {@link #normalize(McpCompletionResult)} and letting a thrown {@link McpException} propagate.
59   */
60  public class McpCompletionResult {
61  
62  	/**
63  	 * The maximum number of completion values a {@link Normalized} result ever carries. Juneau performs
64  	 * no ranking, filtering, deduplication, or pagination of its own; this is strictly a truncation cap.
65  	 */
66  	public static final int MAX_VALUES = 100;
67  
68  	/**
69  	 * JSON-RPC error code raised by {@link #normalize(McpCompletionResult)} for any internal handler
70  	 * failure. See the class Javadoc "Internal handler failure representation" section.
71  	 */
72  	public static final int CODE_INTERNAL_ERROR = -32603;
73  
74  	private List<String> values;
75  	private Integer total;
76  	private Boolean hasMore;
77  
78  	/**
79  	 * Creates the neutral empty-completion result, the shape both dated adapters map to
80  	 * {@code {"completion":{"values":[]}}}.
81  	 *
82  	 * @return A new result with an empty {@link #getValues() values} list and no {@link #getTotal()
83  	 * 	total}/{@link #getHasMore() hasMore}. Never <jk>null</jk>.
84  	 */
85  	public static McpCompletionResult empty() {
86  		return new McpCompletionResult().setValues(List.of());
87  	}
88  
89  	/**
90  	 * The completion suggestions, in application-ranked order.
91  	 *
92  	 * @return The values, or <jk>null</jk> if not set. Normalized to an empty list by
93  	 * 	{@link #normalize(McpCompletionResult)}.
94  	 */
95  	public List<String> getValues() {
96  		return values;
97  	}
98  
99  	/**
100 	 * Sets the completion suggestions.
101 	 *
102 	 * @param value The new value. Can be <jk>null</jk> to unset the property.
103 	 * @return This object.
104 	 */
105 	public McpCompletionResult setValues(List<String> value) {
106 		values = value;
107 		return this;
108 	}
109 
110 	/**
111 	 * The total number of completions available, if known, independent of how many values are emitted.
112 	 *
113 	 * @return The total, or <jk>null</jk> if not set/unknown.
114 	 */
115 	public Integer getTotal() {
116 		return total;
117 	}
118 
119 	/**
120 	 * Sets the total number of completions available.
121 	 *
122 	 * @param value The new value. Must be non-negative if non-<jk>null</jk>; see
123 	 * 	{@link #normalize(McpCompletionResult)}. Can be <jk>null</jk> to unset the property.
124 	 * @return This object.
125 	 */
126 	public McpCompletionResult setTotal(Integer value) {
127 		total = value;
128 		return this;
129 	}
130 
131 	/**
132 	 * Whether more completions exist beyond {@link #getValues() values}.
133 	 *
134 	 * @return The flag, or <jk>null</jk> if not set/unknown.
135 	 */
136 	public Boolean getHasMore() {
137 		return hasMore;
138 	}
139 
140 	/**
141 	 * Sets whether more completions exist beyond {@link #getValues() values}.
142 	 *
143 	 * @param value The new value. Can be <jk>null</jk> to unset the property.
144 	 * @return This object.
145 	 */
146 	public McpCompletionResult setHasMore(Boolean value) {
147 		hasMore = value;
148 		return this;
149 	}
150 
151 	/**
152 	 * Applies the normalization contract documented on this class to a completer's raw return value.
153 	 *
154 	 * @param result The raw result returned by an {@link McpCompleter}. Can be <jk>null</jk>.
155 	 * @return The normalized shape ready for wire mapping. Never <jk>null</jk>.
156 	 * @throws McpException With {@link #CODE_INTERNAL_ERROR} if {@code result} is <jk>null</jk>, if
157 	 * 	{@code result}'s values contain a <jk>null</jk> element, or if {@code result}'s total is negative.
158 	 */
159 	public static Normalized normalize(McpCompletionResult result) {
160 		if (result == null)
161 			throw new McpException(CODE_INTERNAL_ERROR, "Completer returned a null McpCompletionResult.");
162 
163 		var raw = result.getValues();
164 		var source = raw == null ? List.<String>of() : raw;
165 		for (var v : source)
166 			if (v == null)
167 				throw new McpException(CODE_INTERNAL_ERROR, "Completer result values must not contain a null element.");
168 
169 		var total = result.getTotal();
170 		if (total != null && total < 0)
171 			throw new McpException(CODE_INTERNAL_ERROR, "Completer result total must not be negative: " + total);
172 
173 		var hasMore = result.getHasMore();
174 		List<String> values;
175 		if (source.size() > MAX_VALUES) {
176 			values = List.copyOf(source.subList(0, MAX_VALUES));
177 			hasMore = Boolean.TRUE;
178 		} else {
179 			values = List.copyOf(source);
180 		}
181 		return new Normalized(values, total, hasMore);
182 	}
183 
184 	/**
185 	 * The normalized shape of a completion result, ready for wire mapping.
186 	 *
187 	 * @param values The (possibly truncated) completion values, in order, with duplicates preserved.
188 	 * 	Immutable. Never <jk>null</jk>; empty when the completer supplied none.
189 	 * @param total The completer-supplied total, or <jk>null</jk> if unknown/unset.
190 	 * @param hasMore Whether more completions exist. <jk>true</jk> when truncation occurred; otherwise
191 	 * 	the completer-supplied value, which may be <jk>null</jk>.
192 	 */
193 	public record Normalized(List<String> values, Integer total, Boolean hasMore) {}
194 }
