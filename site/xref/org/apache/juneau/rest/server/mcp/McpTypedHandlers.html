33  /**
34   * Bridges {@link McpTypedToolHandler typed tool handlers} into raw {@link McpToolHandler}s.
35   *
36   * <p>
37   * The single public entry point is {@link #adaptTool(McpTypedToolHandler)}, which:
38   * <ul>
39   * 	<li>Derives the input and output JSON Schemas once (at adaptation time) from the typed handler's declared
40   * 		{@link McpTypedToolHandler#argumentType() argument} and {@link McpTypedToolHandler#resultType() result}
41   * 		types, caching them in a single {@link McpToolSpec descriptor} instance returned on every
42   * 		{@link McpToolHandler#descriptor()} call.
43   * 	<li>Binds incoming JSON arguments into an instance of the argument type, mapping binding failures to
44   * 		JSON-RPC {@code -32602} (invalid params).
45   * 	<li>Invokes the typed handler and canonicalizes the returned value into a generic JSON tree in a single
46   * 		serialization pass, mapping canonicalization and structural-safety failures to JSON-RPC {@code -32603}
47   * 		(internal error).
48   * 	<li>Optionally mirrors the canonical structured content as a single text content block.
49   * </ul>
50   */
51  public final class McpTypedHandlers {
52  
53  	private static final JsonSerializer CANONICAL =
54  		JsonSerializer.create().addBeanTypes(false).uriResolution(UriResolution.NONE).build();
55  
56  	private static final JsonSchemaGenerator SCHEMA_GENERATOR =
57  		JsonSchemaGenerator.create().useBeanDefs().build();
58  
59  	private McpTypedHandlers() {}
60  
61  	/**
62  	 * Adapts a typed tool handler into a raw {@link McpToolHandler}.
63  	 *
64  	 * @param <A> The tool argument type.
65  	 * @param <R> The tool result type.
66  	 * @param typed The typed handler to adapt. Must not be <jk>null</jk>.
67  	 * @return A raw handler that derives schemas, binds arguments, and canonicalizes results. Never <jk>null</jk>.
68  	 */
69  	public static <A,R> McpToolHandler adaptTool(McpTypedToolHandler<A,R> typed) {
70  		assertArgNotNull("typed", typed);
71  		var declared = assertArgNotNull("typed.descriptor()", typed.descriptor());
72  		var descriptor = new McpToolSpec()
73  			.setName(declared.getName())
74  			.setDescription(declared.getDescription())
75  			.setInputSchema(schema(typed.argumentType()))
76  			.setOutputSchema(schema(typed.resultType()));
77  		return new McpToolHandler() {
78  			@Override public McpToolSpec descriptor() {
79  				return descriptor;
80  			}
81  
82  			@Override public McpToolOutcome call(Map<String,Object> arguments, BeanStore ctx) {
83  				var bound = McpTypedHandlers.<A>bind(arguments, typed.argumentType());
84  				var result = typed.call(bound, ctx);
85  				var tree = canonicalize(result);
86  				try {
87  					JsonValueSafety.check(tree, "Tool structuredContent");
88  				} catch (IllegalArgumentException e) {
89  					throw new McpException(-32603, e.getMessage());
90  				}
91  				var outcome = new McpToolOutcome().setStructuredContent(tree);
92  				if (typed.mirrorStructuredContentAsText())
93  					outcome.setContent(List.of(McpContentBlock.text(render(tree))));
94  				return outcome;
95  			}
96  		};
97  	}
98  
99  	/**
100 	 * Builds a typed tool from a name, description, argument/result types, and a {@code call} lambda, then adapts
101 	 * it into a raw {@link McpToolHandler} via {@link #adaptTool(McpTypedToolHandler)}.
102 	 *
103 	 * <p>
104 	 * The sanctioned one-expression path for the common case: declaring a typed tool no longer requires a named
105 	 * class implementing {@link McpTypedToolHandler}. Schema derivation, argument binding, and result
106 	 * canonicalization are exactly what {@link #adaptTool(McpTypedToolHandler)} already does.
107 	 *
108 	 * @param <A> The tool argument type.
109 	 * @param <R> The tool result type.
110 	 * @param name The tool name. Must not be <jk>null</jk>.
111 	 * @param description The human-readable tool description. Can be <jk>null</jk>.
112 	 * @param argumentType The tool argument class, used for both schema derivation and argument binding. Must not be <jk>null</jk>.
113 	 * @param resultType The tool result class, used for schema derivation. Must not be <jk>null</jk>.
114 	 * @param call The call body, invoked with the bound argument and the per-request bean store. Must not be <jk>null</jk>.
115 	 * @return A raw handler that derives schemas, binds arguments, and canonicalizes results. Never <jk>null</jk>.
116 	 */
117 	public static <A,R> McpToolHandler tool(String name, String description, Class<A> argumentType, Class<R> resultType, BiFunction<A,BeanStore,R> call) {
118 		assertArgNotNull("name", name);
119 		assertArgNotNull("argumentType", argumentType);
120 		assertArgNotNull("resultType", resultType);
121 		assertArgNotNull("call", call);
122 		return adaptTool(new McpTypedToolHandler<A,R>() {
123 			@Override public McpToolSpec descriptor() {
124 				return new McpToolSpec().setName(name).setDescription(description);
125 			}
126 
127 			@Override public Type argumentType() {
128 				return argumentType;
129 			}
130 
131 			@Override public Type resultType() {
132 				return resultType;
133 			}
134 
135 			@Override public R call(A arguments, BeanStore ctx) {
136 				return call.apply(arguments, ctx);
137 			}
138 		});
139 	}
140 
141 	/**
142 	 * Derives a revision-neutral schema carrier from a Java type.
143 	 *
144 	 * <p>
145 	 * Uses a generator with bean definitions enabled, so nested bean types are emitted under {@code $defs} and
146 	 * referenced rather than inlined.
147 	 *
148 	 * @param type The type to derive a schema for.
149 	 * @return The derived schema, or <jk>null</jk> if the generator produced no schema.
150 	 */
151 	private static McpSchema schema(Type type) {
152 		try {
153 			var session = SCHEMA_GENERATOR.getSession();
154 			var raw = session.getSchema(type);
155 			if (raw != null && session.getBeanDefs() != null && ! session.getBeanDefs().isEmpty())
156 				raw.append("$defs", session.getBeanDefs());
157 			return raw == null ? null : McpSchema.of(raw);
158 		} catch (Exception e) {
159 			throw new McpException(-32603, "Failed to derive tool schema for " + type.getTypeName() + ": " + e.getMessage());
160 		}
161 	}
162 
163 	@SuppressWarnings({
164 		"unchecked" // Parser returns the value described by the authoritative runtime Type.
165 	})
166 	private static <A> A bind(Map<String,Object> arguments, Type type) {
167 		try {
168 			return (A)JsonParser.DEFAULT.read(Json.of(arguments), type);
169 		} catch (Exception e) {
170 			throw new McpException(-32602, "Failed to bind tool arguments to " + type.getTypeName() + ": " + e.getMessage());
171 		}
172 	}
173 
174 	private static Object canonicalize(Object result) {
175 		try {
176 			var json = CANONICAL.write(result);
177 			return JsonParser.DEFAULT.read(json, Object.class);
178 		} catch (Exception e) {
179 			throw new McpException(-32603, "Failed to canonicalize tool result: " + e.getMessage());
180 		}
181 	}
182 
183 	private static String render(Object tree) {
184 		try {
185 			return CANONICAL.write(tree);
186 		} catch (Exception e) {
187 			throw new McpException(-32603, "Failed to mirror tool structuredContent: " + e.getMessage());
188 		}
189 	}
