17  package org.apache.juneau.rest.server.mcp.v20260728;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import java.security.SecureRandom;
22  import java.util.Base64;
23  import java.util.Optional;
24  
25  import javax.crypto.KeyGenerator;
26  import javax.crypto.SecretKey;
27  
28  /**
29   * Built-in default {@link KeyProvider}: a random AES-256 key with a random {@code keyId}, both fixed for the
30   * lifetime of the instance.
31   *
32   * <p>
33   * <b>Not restart-durable and not shareable across process instances by design</b> &mdash; mirrors
34   * {@code AeadRequestStateCodec}'s pre-{@code KeyProvider} ephemeral-key behavior exactly, including the
35   * cross-instance-unseal-fails guarantee (see {@code AeadRequestStateCodec_Test}'s {@code a06}), just relocated
36   * one layer down. Operators who need cross-restart or multi-instance RESUME must supply {@link StaticKeyProvider}
37   * (or a custom {@link KeyProvider}) instead of relying on this default.
38   *
39   * @since 10.0.0
40   */
41  public class EphemeralKeyProvider implements KeyProvider {
42  
43  	private static final int KEY_BITS = 256;
44  	// 8 random bytes -> 11 base64url chars (unpadded): short but collision-safe for a per-process identifier.
45  	private static final int KEY_ID_BYTES = 8;
46  
47  	private static final Base64.Encoder B64URL = Base64.getUrlEncoder().withoutPadding();
48  
49  	private final KeyedSecret current;
50  
51  	/**
52  	 * Constructor. Generates a fresh, per-instance AES-256 key and a fresh, per-instance random {@code keyId}.
53  	 */
54  	public EphemeralKeyProvider() {
55  		try {
56  			var gen = KeyGenerator.getInstance("AES");
57  			gen.init(KEY_BITS);
58  			var key = gen.generateKey();
59  			var idBytes = new byte[KEY_ID_BYTES];
60  			new SecureRandom().nextBytes(idBytes);
61  			current = new KeyedSecret(B64URL.encodeToString(idBytes), key);
62  		} catch (Exception e) { // HTT every JDK guarantees AES-256 key generation via the standard JCE provider
63  			throw rex(e, "Failed to generate AES key");
64  		}
65  	}
66  
67  	@Override /* KeyProvider */
68  	public KeyedSecret currentKey() {
69  		return current;
70  	}
71  
72  	@Override /* KeyProvider */
73  	public Optional<SecretKey> resolveKey(String keyId) {
74  		return keyId.equals(current.keyId()) ? Optional.of(current.key()) : Optional.empty();
75  	}
76  }
