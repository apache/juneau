17  package org.apache.juneau.rest.server.mcp.v20260728;
18  
19  import java.security.Principal;
20  import java.util.*;
21  
22  import org.apache.juneau.commons.concurrent.*;
23  
24  /**
25   * Pluggable SPI for sealing/unsealing the opaque {@code requestState} continuation token (MCP {@code 2026-07-28}
26   * SEP-2322 Multi-Round-Trip Requests).
27   *
28   * <p>
29   * Mirrors the "pluggable SPI + built-in default" shape sub-project E established for the client-side
30   * {@code McpResponseCache} (not on this module's classpath): this interface plus {@link AeadRequestStateCodec}
31   * (the built-in default) is the server-side, seal/unseal counterpart.
32   *
33   * <p>
34   * Implementations must treat any verification failure (tamper, AAD mismatch, or an implementation-specific
35   * integrity check) as an {@link Optional#empty()} return from {@link #unseal(String, String)} &mdash; never a
36   * thrown exception. Expiry is checked by the caller ({@code McpRevision}) against
37   * {@link McpRequestState#expiresAtMs()}, not by the codec itself, since expiry is a dispatcher-level policy (see
38   * {@link McpMrtrConfig}), not a codec integrity concern.
39   *
40   * <p>
41   * <b>Trust model &mdash; {@code requestState} is a bearer token.</b> The sealed token is a replayable credential:
42   * anyone who holds it can resume the paused operation until it expires, and it carries no caller identity. A
43   * codec (and every call site) must treat it accordingly &mdash; transport only over authenticated TLS, and
44   * <b>never log</b> a {@code requestState}. <b>Multi-use within TTL by default:</b> a captured token can be
45   * replayed any number of times within its TTL unless an operator configures a {@link ReplayCache} (see
46   * {@link McpMrtrConfig#setReplayCache(ReplayCache)}) to enforce single-use &mdash; the built-in
47   * {@link InMemoryReplayCache} is per-process only, so cross-node single-use requires a shared implementation. A
48   * handler must still write idempotent resume side effects unless it knows a single-use {@link ReplayCache} is in
49   * effect; the max-rounds cap bounds chain depth, not replay count (see {@link McpMrtrConfig}).
50   *
51   * <p>
52   * <b>Canonical AAD format.</b> The dispatcher binds each token to the request that produced it by passing a
53   * single canonical AAD string: {@code method + '\u0000' + protocolVersion + '\u0000' + target} &mdash; the
54   * JSON-RPC method name, the negotiated protocol version, and the operation target (the tool {@code name} for
55   * {@code tools/call}, the prompt {@code name} for {@code prompts/get}, or the resource {@code uri} for
56   * {@code resources/read}; empty for any other method), all NUL ({@code U+0000})-separated (see
57   * {@code McpRevision#aad}). NUL is chosen because it can never appear in a method name, a protocol-version
58   * literal, or the trailing target field, so the concatenation is unambiguous. The trailing target field binds
59   * the token to the specific operation it paused against, not merely the method, so a token minted while paused
60   * on one tool/prompt/resource cannot be resumed against a different one of the same kind. An implementation
61   * only needs to treat the AAD as opaque bytes; the format is documented here so seal and unseal call sites (and
62   * any custom codec) agree on exactly what is authenticated.
63   *
64   * <p>
65   * <b>Authenticated principal (READY-312f F4).</b> Both {@link #seal(McpRequestState, String, Principal)} and
66   * {@link #unseal(String, String, Principal)} receive the caller's authenticated {@link Principal} &mdash; the same
67   * identity the F2 resource-server layer establishes (see {@code McpResourceServerSupport#principal}). This is the
68   * <i>seam</i> a hardened codec uses to fold the caller identity into its authenticated data, so a
69   * {@code requestState} minted for principal A cannot be resumed by principal B. The built-in
70   * {@link AeadRequestStateCodec} binds exactly that: it folds a canonical, deterministic {@code iss|sub} identity
71   * (see {@link AeadRequestStateCodec#principalIdentity(Principal)}) into its AEAD authenticated data, so a
72   * mismatched principal fails the GCM tag check and {@link #unseal} returns {@link Optional#empty()}. F4 is what
73   * guarantees the principal is delivered to the codec at both seal and unseal in the first place; a custom codec
74   * remains free to bind a different identity attribute (subject claim, issuer+subject, full claim set) or none at
75   * all. The principal is <b>nullable</b>: when RS auth is disabled or the caller is anonymous it is <jk>null</jk>,
76   * and every codec must seal/unseal cleanly (no NPE) in that case. The two-argument
77   * {@link #seal(McpRequestState, String)} / {@link #unseal(String, String)} convenience overloads simply pass a
78   * <jk>null</jk> (no-principal) identity.
79   */
80  public interface RequestStateCodec {
81  
82  	/**
83  	 * Seals a payload into an opaque token, exposing the authenticated caller identity to the codec.
84  	 *
85  	 * @param state The payload to seal. Must not be <jk>null</jk>.
86  	 * @param aad Additional authenticated data (the dispatcher passes the canonical
87  	 * 	{@code method + '\u0000' + protocolVersion + '\u0000' + target} form &mdash; see the class Javadoc). Must
88  	 * 	not be <jk>null</jk>.
89  	 * @param principal The authenticated caller (see the class Javadoc). May be <jk>null</jk> for an anonymous
90  	 * 	caller or when resource-server auth is disabled.
91  	 * @return The opaque token. Never <jk>null</jk>.
92  	 */
93  	String seal(McpRequestState state, String aad, Principal principal);
94  
95  	/**
96  	 * Unseals a token, verifying integrity against the supplied AAD, with the authenticated caller identity
97  	 * available to the codec.
98  	 *
99  	 * <p>
100 	 * <b>The {@code token} is arbitrary, untrusted client input.</b> It arrives verbatim from the follow-up
101 	 * request's {@code requestState} field, so an implementation must assume nothing about its length or content:
102 	 * it must not allocate buffers sized from the raw token, and it must not throw &mdash; any malformed, oversized,
103 	 * tampered, or AAD-mismatched input must return {@link Optional#empty()} (see the class Javadoc contract).
104 	 *
105 	 * @param token The token previously returned by {@link #seal(McpRequestState, String, Principal)}. Must not be
106 	 * 	<jk>null</jk>.
107 	 * @param aad The AAD to verify against. Must match the value passed to
108 	 * 	{@link #seal(McpRequestState, String, Principal)} exactly, or verification fails.
109 	 * @param principal The authenticated caller (see the class Javadoc). May be <jk>null</jk> for an anonymous
110 	 * 	caller or when resource-server auth is disabled. A codec that binds the principal must fail
111 	 * 	verification when it differs from the sealing principal.
112 	 * @return The original payload, or {@link Optional#empty()} if verification fails for any reason (tamper,
113 	 * 	AAD mismatch, malformed token, principal mismatch).
114 	 */
115 	Optional<McpRequestState> unseal(String token, String aad, Principal principal);
116 
117 	/**
118 	 * No-principal convenience overload of {@link #seal(McpRequestState, String, Principal)} (anonymous caller).
119 	 *
120 	 * @param state The payload to seal. Must not be <jk>null</jk>.
121 	 * @param aad Additional authenticated data. Must not be <jk>null</jk>.
122 	 * @return The opaque token. Never <jk>null</jk>.
123 	 */
124 	default String seal(McpRequestState state, String aad) {
125 		return seal(state, aad, null);
126 	}
127 
128 	/**
129 	 * No-principal convenience overload of {@link #unseal(String, String, Principal)} (anonymous caller).
130 	 *
131 	 * @param token The token previously returned by {@link #seal(McpRequestState, String)}. Must not be <jk>null</jk>.
132 	 * @param aad The AAD to verify against.
133 	 * @return The original payload, or {@link Optional#empty()} if verification fails for any reason.
134 	 */
135 	default Optional<McpRequestState> unseal(String token, String aad) {
136 		return unseal(token, aad, null);
137 	}
138 }
