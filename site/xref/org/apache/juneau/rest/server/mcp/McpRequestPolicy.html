17  package org.apache.juneau.rest.server.mcp;
18  
19  import java.net.*;
20  import java.util.*;
21  import java.util.concurrent.atomic.*;
22  import java.util.logging.*;
23  
24  import org.apache.juneau.http.response.*;
25  import org.apache.juneau.rest.server.*;
26  import org.apache.juneau.rest.server.auth.*;
27  
28  /**
29   * Per-request admission checks applied to an MCP POST endpoint before JSON-RPC dispatch.
30   *
31   * <p>
32   * The checks, in order:
33   * <ol>
34   * 	<li>The request <c>Content-Type</c> must be <c>application/json</c> (<c>415</c> otherwise).
35   * 	<li>A request carrying an <c>Origin</c> header must have an origin that matches its <c>Host</c> header or is listed in
36   * 		the resource's {@link Rest#allowedOrigins() allowedOrigins} (<c>403</c> otherwise).  Requests without an <c>Origin</c> header pass.
37   * 	<li>When a {@link McpServerConfig#getTokenValidator() token validator} is configured, a valid bearer token is required.
38   * 	<li>When the request is not otherwise authenticated, it is served only if the caller is a loopback address and the
39   * 		<c>Host</c> header names a loopback host or a host of an allowed origin, unless
40   * 		{@link McpServerConfig#isAllowUnauthenticated()} is set.
41   * </ol>
42   */
43  final class McpRequestPolicy {
44  
45  	private static final Logger LOG = Logger.getLogger(McpRequestPolicy.class.getName());
46  	private static final String PRINCIPAL_ATTR = "juneau.principal";
47  	private static final AtomicBoolean UNAUTHENTICATED_WARNED = new AtomicBoolean();
48  
49  	private McpRequestPolicy() {}
50  
51  	/**
52  	 * Applies the admission checks.
53  	 *
54  	 * @param config The server config.
55  	 * @param req The current request.
56  	 * @throws UnsupportedMediaType If the content type is not JSON.
57  	 * @throws Forbidden If the origin, host or caller is not admitted.
58  	 */
59  	static void check(McpServerConfig config, RestRequest req) {
60  		var ct = req.getHeader("Content-Type");
61  		if (! isJson(ct))
62  			throw new UnsupportedMediaType("Content-Type must be application/json.");
63  
64  		var allowed = req.getContext().getAllowedOrigins();
65  		var origin = req.getHeader("Origin");
66  		var host = req.getHeader("Host");
67  		if (origin != null && ! originAllowed(allowed, origin, host))
68  			throw new Forbidden("Origin not allowed.");
69  
70  		if (config.getTokenValidator() != null) {
71  			if (req.getHttpServletRequest().getAttribute(PRINCIPAL_ATTR) == null)
72  				new BearerTokenGuard(config.getTokenValidator()).guard(req, req.getResponse());
73  			return;
74  		}
75  		if (req.getHttpServletRequest().getAttribute(PRINCIPAL_ATTR) != null)
76  			return;
77  		if (config.isAllowUnauthenticated())
78  			return;
79  		if (UNAUTHENTICATED_WARNED.compareAndSet(false, true))
80  			LOG.warning(() -> "MCP endpoint is serving without authentication; only loopback callers are admitted. Configure a token validator or allowUnauthenticated() to change this.");
81  		if (! isLoopbackAddress(req.getHttpServletRequest().getRemoteAddr()))
82  			throw new Forbidden("Authentication required.");
83  		if (host != null && ! hostAllowed(allowed, host))
84  			throw new Forbidden("Host not allowed.");
85  	}
86  
87  	private static boolean isJson(String contentType) {
88  		if (contentType == null)
89  			return false;
90  		var i = contentType.indexOf(';');
91  		var mt = (i < 0 ? contentType : contentType.substring(0, i)).trim();
92  		return "application/json".equalsIgnoreCase(mt);
93  	}
94  
95  	private static boolean originAllowed(Set<String> allowed, String origin, String host) {
96  		var o = origin.endsWith("/") ? origin.substring(0, origin.length() - 1) : origin;
97  		if (allowed.contains(o))
98  			return true;
99  		if (host == null)
100 			return false;
101 		try {
102 			var u = URI.create(origin);
103 			if (u.getHost() == null || u.getScheme() == null)
104 				return false;
105 			var defaultPort = "https".equalsIgnoreCase(u.getScheme()) ? 443 : 80;
106 			var authority = u.getPort() < 0 || u.getPort() == defaultPort ? u.getHost() : u.getHost() + ":" + u.getPort();
107 			var suffix = ":" + defaultPort;
108 			var h = host.endsWith(suffix) ? host.substring(0, host.length() - suffix.length()) : host;
109 			return authority.equalsIgnoreCase(h);
110 		} catch (IllegalArgumentException e) {
111 			return false;
112 		}
113 	}
114 
115 	private static boolean hostAllowed(Set<String> allowed, String host) {
116 		var name = hostName(host);
117 		if (isLoopbackName(name))
118 			return true;
119 		for (var o : allowed) {
120 			try {
121 				var u = URI.create(o);
122 				if (u.getHost() != null && u.getHost().equalsIgnoreCase(name))
123 					return true;
124 			} catch (IllegalArgumentException e) {
125 				// Ignore malformed entry.
126 			}
127 		}
128 		return false;
129 	}
130 
131 	private static String hostName(String host) {
132 		var h = host.trim();
133 		if (h.startsWith("[")) {
134 			var e = h.indexOf(']');
135 			return e < 0 ? h : h.substring(0, e + 1);
136 		}
137 		var c = h.lastIndexOf(':');
138 		return c < 0 ? h : h.substring(0, c);
139 	}
140 
141 	private static boolean isLoopbackName(String name) {
142 		return "localhost".equalsIgnoreCase(name) || "[::1]".equals(name) || name.startsWith("127.");
143 	}
144 
145 	private static boolean isLoopbackAddress(String addr) {
146 		// An empty address means an in-process (non-network) request.
147 		if (addr == null || addr.isEmpty())
148 			return true;
149 		return addr.startsWith("127.") || "::1".equals(addr) || "0:0:0:0:0:0:0:1".equals(addr);
150 	}
151 }
