17  package org.apache.juneau.rest.server.mcp;
18  
19  /**
20   * Per-subscription queue handle minted by an {@link McpSubscriptionBroker}.
21   *
22   * <p>
23   * Deliberately named differently from the client-facing {@code McpSubscriptionHandle} (a distinct type in a
24   * separate client module/package): this is the server-side, broker-internal handle a v2 dispatch branch
25   * drains via {@link #take()} to serialize change events onto its own stream.
26   *
27   * <p>
28   * Owns its own bounded, drop-oldest queue — it does <b>not</b> reuse
29   * {@code org.apache.juneau.rest.server.sse.SseSubscription}, whose constructor and {@code offer(...)} method
30   * are package-private to a different package. Only one thread should call {@link #take()} concurrently on a
31   * given instance (matching the production single-pump-per-subscription usage): {@link #close()}'s contract
32   * of interrupting a blocked {@link #take()} tracks a single in-flight taker thread.
33   */
34  public interface McpSubscription extends AutoCloseable {
35  
36  	/**
37  	 * The subscription identifier (the originating {@code subscriptions/listen} request's JSON-RPC id).
38  	 *
39  	 * @return The identifier. Never {@code null} or empty.
40  	 */
41  	String getId();
42  
43  	/**
44  	 * The honored filter this subscription was registered with.
45  	 *
46  	 * @return The filter. Never {@code null}.
47  	 */
48  	McpSubscriptionFilter getFilter();
49  
50  	/**
51  	 * Blocks until the next matching change event is available.
52  	 *
53  	 * <p>
54  	 * Calling this on a subscription that is already closed throws {@link InterruptedException} immediately
55  	 * rather than blocking — load-bearing behavior for a pump's graceful-completion path, which relies on a
56  	 * closed-before-called {@link #take()} failing fast instead of hanging forever waiting for an event that
57  	 * will never arrive.
58  	 *
59  	 * @return The next event.
60  	 * @throws InterruptedException If the wait was interrupted, including by a concurrent {@link #close()},
61  	 * 	or if this subscription was already closed before this call.
62  	 */
63  	McpChangeEvent take() throws InterruptedException;
64  
65  	/**
66  	 * Returns whether this subscription has been closed.
67  	 *
68  	 * @return {@code true} if closed.
69  	 */
70  	boolean isClosed();
71  
72  	/**
73  	 * Closes this subscription: idempotent, deregisters from the broker that minted it, and interrupts a
74  	 * thread currently blocked in {@link #take()}.
75  	 */
76  	@Override
77  	void close();
78  }
