17  package org.apache.juneau.rest.server.mcp;
18  
19  import java.util.Optional;
20  
21  /**
22   * Revision-neutral registry + filtered fan-out publisher for MCP {@code subscriptions/listen}.
23   *
24   * <p>
25   * Extends {@link McpSubscriptions} (the publish half) and adds registration/lifecycle: a v2 dispatch branch
26   * calls {@link #registerIfUnder(int, String, McpSubscriptionFilter)} once per accepted
27   * {@code subscriptions/listen} request. Close (graceful or abrupt) does <b>not</b> flow back through this
28   * interface: the returned {@link McpSubscription#close()} instead runs a close-callback wired at
29   * registration time, which removes the entry directly from the registry the returned handle's key was
30   * minted under. {@link #unregister(String)} is a registry-completeness API for that same by-id removal —
31   * it has no current production caller, since the close-callback path never has to look the id back up.
32   * Application handlers reach this same instance only through the narrower {@link McpSubscriptions} view.
33   *
34   * <p>
35   * Plain {@link #register(String, McpSubscriptionFilter)} is uncapped — it never rejects, so a caller wanting
36   * an enforced {@code maxConcurrentSubscriptions} ceiling must use {@link #registerIfUnder(int, String,
37   * McpSubscriptionFilter)} instead, which admission-checks and mutates the registry as one atomic step. A
38   * separate "check {@link #activeCount()}, then call {@code register}" pair is exactly the TOCTOU race this
39   * capped variant exists to close: two concurrent callers can both observe a count under the cap and both
40   * register before either mutation is visible to the other.
41   */
42  public interface McpSubscriptionBroker extends McpSubscriptions {
43  
44  	/**
45  	 * Registers a new subscription and returns its queue handle. Uncapped: never rejects.
46  	 *
47  	 * @param subscriptionId The subscription identifier (the listen request's JSON-RPC id). Must not be
48  	 * 	{@code null} or empty.
49  	 * @param honoredFilter The capability-gated filter this subscription honors. Must not be {@code null}.
50  	 * @return The new subscription handle. Never {@code null}.
51  	 */
52  	McpSubscription register(String subscriptionId, McpSubscriptionFilter honoredFilter);
53  
54  	/**
55  	 * Atomically admission-checks against {@code max} and registers, closing the TOCTOU window a separate
56  	 * {@link #activeCount()} check plus {@link #register} call would leave open under concurrent callers
57  	 * (in particular against a JVM-wide shared broker instance, where unrelated callers race each other).
58  	 *
59  	 * <p>
60  	 * Re-registering an id that is already present is always allowed and never counted against {@code max},
61  	 * matching {@link #register}'s replace-on-same-id semantics (the previous subscription with that id is
62  	 * closed) — a replacement does not grow {@link #activeCount()}, so it cannot be what pushes the registry
63  	 * over the cap.
64  	 *
65  	 * @param max The maximum number of concurrently active subscriptions to admit under (exclusive: a
66  	 * 	request is admitted only while {@link #activeCount()} is strictly less than this, except for the
67  	 * 	same-id replacement case above).
68  	 * @param subscriptionId The subscription identifier (the listen request's JSON-RPC id). Must not be
69  	 * 	{@code null} or empty.
70  	 * @param honoredFilter The capability-gated filter this subscription honors. Must not be {@code null}.
71  	 * @return The new subscription handle, or {@link Optional#empty()} if the cap was reached and this was
72  	 * 	not a same-id replacement.
73  	 */
74  	Optional<McpSubscription> registerIfUnder(int max, String subscriptionId, McpSubscriptionFilter honoredFilter);
75  
76  	/**
77  	 * Deregisters and closes the subscription with the given id, if present.
78  	 *
79  	 * @param subscriptionId The subscription identifier. A missing id is a no-op.
80  	 */
81  	void unregister(String subscriptionId);
82  
83  	/**
84  	 * The number of currently registered (not yet closed) subscriptions.
85  	 *
86  	 * @return The active count.
87  	 */
88  	int activeCount();
89  }
