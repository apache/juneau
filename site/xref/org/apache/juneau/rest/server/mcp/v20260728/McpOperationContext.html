17  package org.apache.juneau.rest.server.mcp.v20260728;
18  
19  import java.util.*;
20  
21  /**
22   * The resolved MCP operation an {@link McpOperationScopeResolver} decides required scopes for (SEP-2350 per-operation
23   * step-up).
24   *
25   * <p>
26   * Carries the JSON-RPC {@code method} (e.g. {@code tools/call}), the operation {@code name} (the tool / prompt name or
27   * resource URI, as returned by the revision's routing-name derivation), and an opaque read-only view of the raw JSON-RPC
28   * request {@code params}.  A static per-operation scope map keys on {@code (method, name)}; the {@code params} view is the
29   * seam a future dynamic resolver could consult (READY-312f Q-c: not built for 10.0).
30   *
31   * <p>
32   * {@code params} is the <em>raw</em> JSON-RPC {@code params} object, not the unwrapped tool-call arguments: for a
33   * {@code tools/call} request it is {@code {"name":<tool>, "arguments":{...}}}, so a resolver reads the tool arguments via
34   * {@code ctx.params().get("arguments")} rather than treating {@code params} as the argument map itself.
35   *
36   * @param method The JSON-RPC method.  Never {@code null}.
37   * @param name The operation name (tool/prompt name or resource URI), or {@code null} for methods without one.
38   * @param params A read-only view of the raw JSON-RPC request params.  Never {@code null}.
39   * @since 10.0.0
40   */
41  public record McpOperationContext(String method, String name, Map<String,Object> params) {
42  
43  	/**
44  	 * Compact constructor enforcing a non-null method and defensively copying the params to a read-only map.
45  	 */
46  	public McpOperationContext {
47  		Objects.requireNonNull(method, "method");
48  		params = params == null ? Map.of() : Collections.unmodifiableMap(new LinkedHashMap<>(params));
49  	}
50  }
