17  package org.apache.juneau.rest.server.mcp;
18  
19  import org.apache.juneau.bean.jsonrpc.*;
20  import org.apache.juneau.commons.inject.*;
21  import org.apache.juneau.http.Content;
22  import org.apache.juneau.marshall.serializer.*;
23  import org.apache.juneau.rest.server.*;
24  
25  /**
26   * Mixin interface that exposes an MCP JSON-RPC endpoint at {@code POST /mcp} on any Juneau REST resource.
27   *
28   * <p>
29   * Implementing classes supply their {@link McpServerConfig} via {@link #getMcpConfig()} and their
30   * protocol revision via {@link #revision()}. In practice a consumer implements a revision-specific
31   * sub-interface (for example {@code org.apache.juneau.rest.server.mcp.v20250618.McpEndpoint}), which supplies {@link #revision()} for them.
32   *
33   * <h5 class='section'>Example:</h5>
34   * <pre>
35   * @Rest(path="/api")
36   * public class MyResource extends BasicRestServlet implements org.apache.juneau.rest.server.mcp.v20250618.McpEndpoint {
37   *     @Override
38   *     public McpServerConfig getMcpConfig() {
39   *         return new McpServerConfig().addTool(new MyEchoTool());
40   *     }
41   * }
42   * </pre>
43   *
44   * <p>
45   * Named {@code *Mixin} (not {@code Abstract*}) because this is an interface, not an abstract class -
46   * {@code Abstract*} is this repo's naming convention for abstract classes only. {@link McpRevision}, the
47   * other core type dated subclasses de-versioning-shadow, deliberately keeps its plain name: end-users
48   * never implement it directly (only a bound revision's own subclass does), so it never faces the
49   * same-simple-name collision this mixin interface was renamed to avoid.
50   */
51  public interface McpEndpointMixin {
52  
53  	/**
54  	 * Returns the {@link McpServerConfig} backing this endpoint.
55  	 *
56  	 * @return The config. Never {@code null}.
57  	 */
58  	McpServerConfig getMcpConfig();
59  
60  	/**
61  	 * The MCP protocol revision this endpoint speaks.
62  	 *
63  	 * @return The bound revision. Never {@code null}.
64  	 */
65  	McpRevision revision();
66  
67  	/**
68  	 * Optional subscription broker backing this endpoint's {@code subscriptions/listen} support.
69  	 *
70  	 * <p>
71  	 * Returning <jk>null</jk> (the default) means this endpoint does not support subscriptions: no
72  	 * {@link McpSubscriptionBroker} or {@link McpSubscriptions} bean is added to the request-scoped
73  	 * {@link BeanStore} that {@link #handleMcpRequest} builds.
74  	 *
75  	 * <p>
76  	 * A revision binding that implements {@code subscriptions/listen} (for example
77  	 * {@code org.apache.juneau.rest.server.mcp.v20260728.McpEndpoint}) overrides this to return a
78  	 * binding-owned broker instance (its own per-binding vs. shared-instance semantics are that
79  	 * revision's own choice — see its Javadoc).
80  	 *
81  	 * @return The subscription broker, or <jk>null</jk> if this endpoint does not support subscriptions.
82  	 */
83  	default McpSubscriptionBroker subscriptionBroker() {
84  		return null;
85  	}
86  
87  	/**
88  	 * Default MCP JSON-RPC endpoint handler.
89  	 *
90  	 * <p>
91  	 * Implementations may override this method to customize routing (path / annotations) but must
92  	 * still dispatch through {@link #revision()}.
93  	 *
94  	 * <p>
95  	 * Applies the same serializer policy as {@link AbstractMcpRestServlet} ({@code addBeanTypes} and
96  	 * {@code uriResolution="NONE"}), so the two neutral HTTP entrypoints stay at parity.
97  	 *
98  	 * @param req JSON-RPC request envelope.
99  	 * @param restReq The current REST request.
100 	 * @return The response, or {@code null} for notifications.
101 	 */
102 	@SuppressWarnings({
103 		"resource" // Request-scoped scratch BasicBeanStore; lifetime is bounded by this handler invocation, no foreign resources are captured.
104 	})
105 	@SerializerConfig(addBeanTypes = "true", uriResolution = "NONE")
106 	@RestPost(path = "/mcp")
107 	default Object handleMcpRequest(@Content JsonRpcRequest req, RestRequest restReq) {
108 		var bs = new BasicBeanStore(restReq.getContext().getBeanStore())
109 			.addBean(RestRequest.class, restReq);
110 		var broker = subscriptionBroker();
111 		if (broker != null)
112 			bs.addBean(McpSubscriptionBroker.class, broker).addBean(McpSubscriptions.class, broker);
113 		var exchange = new McpExchange(req, n -> restReq.getHeaderParam(n).asString().orElse(null));
114 		var result = revision().dispatch(exchange, getMcpConfig(), bs);
115 		if (result instanceof McpResponseResult r)
116 			return r.response();
117 		if (result instanceof McpStreamResult r)
118 			return r.stream();
119 		return null;
120 	}
121 }
