27  /**
28   * Hand-rolled, dependency-free compiler and reverse matcher for the narrow, reliably-invertible subset of
29   * <a class="doclink" href="https://www.rfc-editor.org/rfc/rfc6570">RFC 6570</a> URI templates used by MCP
30   * resource-template reads.
31   *
32   * <p>
33   * A template is {@link #compile(String) compiled} once into an immutable matcher that exposes:
34   * <ul>
35   * 	<li>{@link #isReverseMatchable()} - whether the template is one of the settled reversible forms (simple,
36   * 		reserved, fragment, path, label, or query scalar expressions, with no explode/prefix modifier and no
37   * 		ambiguous adjacency to another expression);
38   * 	<li>{@link #literalOctetCount()} - the number of normalized literal octets outside <c>{...}</c> expressions,
39   * 		for registry specificity ranking;
40   * 	<li>{@link #variableCount()} and {@link #variableNames()} - the declared template variables, in declaration
41   * 		order; and
42   * 	<li>{@link #match(String)} - reverse-matches a concrete URI against a reverse-matchable template, returning
43   * 		an immutable, insertion-ordered, percent-decoded variable map, or <jk>null</jk> if the URI does not match.
44   * </ul>
45   *
46   * <h5 class='section'>Reverse-matchable forms:</h5>
47   * <p>
48   * Non-explode, non-prefix simple scalar <c>{var}</c>, reserved scalar <c>{+var}</c>, fragment scalar
49   * <c>{#var}</c>, path segments <c>{/var}</c> / <c>{/x,y}</c>, label segments <c>{.var}</c> / <c>{.x,y}</c>, query
50   * start <c>{?x}</c> / <c>{?x,y}</c>, and query continuation <c>{&amp;x}</c> / <c>{&amp;x,y}</c>. All other legal
51   * RFC 6570 forms - exploded (<c>*</c>), prefixed (<c>:n</c>), matrix (<c>;</c>), multi-variable simple/reserved/
52   * fragment expressions, and expressions with ambiguous adjacency to a neighboring expression - parse
53   * successfully (so they remain listable and completable) but are never selected for a template-backed read;
54   * {@link #match(String)} always returns <jk>null</jk> for them.
55   *
56   * <h5 class='section'>Percent-encoding normalization:</h5>
57   * <p>
58   * Matching and literal-specificity counting follow RFC 3986 percent-normalization implemented fresh by this
59   * class: valid <c>%HH</c> triplets are hex-uppercased; triplets that decode to an unreserved octet
60   * (<c>ALPHA</c> / <c>DIGIT</c> / <c>-</c> / <c>.</c> / <c>_</c> / <c>~</c>) normalize to their literal
61   * character for comparison; triplets that decode to a reserved octet stay encoded while match boundaries are
62   * computed, so an encoded <js>"%2F"</js> is data inside a <c>{var}</c> segment rather than a path separator;
63   * captured values are then UTF-8 percent-decoded before being placed in the result map; and <c>+</c> is always
64   * a literal plus (this class never applies {@code application/x-www-form-urlencoded} plus-as-space semantics).
65   * A concrete URI with a malformed percent escape, or a capture whose decoded bytes are not valid UTF-8, never
66   * matches. This class deliberately does <b>not</b> reuse
67   * {@code org.apache.juneau.commons.utils.StringUtils} <c>urlEncode</c>/<c>urlDecode</c>/<c>fixUrl</c>/
68   * <c>urlEncodePath</c> or the SVL <c>EncodingFunctions</c>: those implement form-urlencoded semantics
69   * (<c>+</c> as space) and lack hex-case normalization, reserved-octet preservation, and malformed-escape
70   * rejection, all of which contradict the rules above.
71   *
72   * <h5 class='section'>Ambiguous-capture resolution:</h5>
73   * <p>
74   * A reversible template compiles to an alternating sequence of literal and capture pieces; because an
75   * expression with ambiguous adjacency to another expression is rejected as non-reverse-matchable (see above),
76   * a capture piece in a reverse-matchable template is always immediately followed by either a literal piece or
77   * the end of the template, never by another capture. When a reserved/fragment (unbounded) or path/label/simple
78   * (bounded-by-<c>/</c>) capture is immediately followed by a literal, {@link #match(String)} resolves the
79   * capture's length via a single longest-to-shortest search for the position at which that immediately-following
80   * literal text occurs, using {@link String#lastIndexOf(String, int)}. This deterministically implements "the
81   * longest capture that still permits the (immediately following) literal to match" without whole-suffix
82   * backtracking across multiple capture/literal boundaries, which keeps matching a bounded, linear-in-input scan
83   * per capture with no regex-style catastrophic backtracking risk, at the cost of not re-trying an earlier
84   * capture's length choice if a later, unrelated piece of the template fails to match. That scope narrowing is
85   * intentional and consistent with this matcher covering reliable inversion of a narrow subset, not a complete
86   * RFC 6570 engine.
87   *
88   * <h5 class='section'>Scope:</h5>
89   * <p>
90   * This matcher is intentionally MCP-local. It is not registered as a general-purpose RFC 3986/6570 utility;
91   * if a second real consumer outside MCP resource templates needs equivalent percent-normalization or
92   * URI-template parsing, that would justify extracting a shared helper into {@code juneau-commons} at that time,
93   * not before.
94   */

101 	/**
102 	 * Parses and compiles a single RFC 6570 URI template.
103 	 *
104 	 * <p>
105 	 * Registration syntax is validated independently of reverse-matchability: a syntactically legal template
106 	 * outside the reversible subset compiles successfully with {@link #isReverseMatchable()} returning
107 	 * <jk>false</jk>, so it remains listable and completable even though {@link #match(String)} always returns
108 	 * <jk>null</jk> for it.
109 	 *
110 	 * @param uriTemplate The URI template to compile. Must not be <jk>null</jk> or blank.
111 	 * @return A new compiled matcher.
112 	 * @throws IllegalArgumentException If the template is <jk>null</jk>/blank, has malformed braces, has a
113 	 * 	malformed percent escape, has invalid operator/varspec grammar, has a literal segment containing an
114 	 * 	invalid UTF-8 percent-escape sequence, or declares the same variable name more than once.
115 	 */
116 	@SuppressWarnings({
117 		"java:S3776" // Intentional: hand-rolled RFC 6570 reverse matcher; control-flow verified by McpUriTemplateMatcher_Test, refactor would risk matching correctness.
118 	})
119 	public static McpUriTemplateMatcher compile(String uriTemplate) {
120 		if (uriTemplate == null || uriTemplate.isBlank())
121 			throw iaex("Malformed URI template ''%s'': template must not be null or blank", uriTemplate);
122 
123 		var tokens = tokenize(uriTemplate);
124 		var declaredVars = new LinkedHashSet<String>();
125 		var pieces = new ArrayList<Piece>();
126 		var matchable = true;
127 		var literalOctetCount = 0;
128 		RawToken previous = null;
129 
130 		for (var token : tokens) {
131 			if (token instanceof RawLiteral lit) {
132 				var normalized = normalizeLiteralStrict(lit.text(), uriTemplate);
133 				literalOctetCount += countOctets(normalized);
134 				pieces.add(new LiteralPiece(normalized));
135 			} else if (token instanceof RawExpr expr) {
136 				if (previous instanceof RawExpr)
137 					matchable = false;
138 				var parsed = parseExpr(expr.body(), uriTemplate, expr.index());
139 				for (var v : parsed.vars()) {
140 					if (! declaredVars.add(v.name()))
141 						throw iaex("Malformed URI template ''%s'': duplicate variable ''%s''", uriTemplate, v.name());
142 				}
143 				if (isExprReverseMatchable(parsed))
144 					pieces.addAll(desugar(parsed));
145 				else
146 					matchable = false;
147 			}
148 			previous = token;
149 		}
150 
151 		return new McpUriTemplateMatcher(uriTemplate, matchable, literalOctetCount, List.copyOf(declaredVars),
152 			List.copyOf(pieces));
153 	}

233 	/**
234 	 * Reverse-matches a concrete URI against this template.
235 	 *
236 	 * @param uri The concrete URI to match. Can be <jk>null</jk>, in which case this method returns <jk>null</jk>.
237 	 * @return An immutable, insertion-ordered, UTF-8-decoded map of variable name to decoded value, in
238 	 * 	declaration order, or <jk>null</jk> if this template is not {@link #isReverseMatchable() reverse-matchable},
239 	 * 	the URI does not match, the URI contains a malformed percent escape, or a captured value's decoded bytes
240 	 * 	are not valid UTF-8.
241 	 */
242 	@SuppressWarnings({
243 		"java:S1168" // null means "template does not match"; an empty map would mean "matched with zero variables" - collapsing the two would turn non-matches into false matches.
244 	})
245 	public Map<String,String> match(String uri) {
246 		if (! reverseMatchable || uri == null)
247 			return null;
248 		var normalized = normalizeForMatch(uri);
249 		if (normalized == null)
250 			return null;
251 		return matchPieces(normalized);
252 	}
253 
254 	/**
255 	 * Left-to-right single-pass match against the compiled, alternating literal/capture piece list.
256 	 *
257 	 * <p>
258 	 * See the class Javadoc "Ambiguous-capture resolution" section for the boundary-resolution algorithm and its
259 	 * complexity guarantee.
260 	 */
261 	@SuppressWarnings({
262 		"java:S1168", // null means "template does not match"; an empty map would mean "matched with zero variables" - collapsing the two would turn non-matches into false matches.
263 		"java:S3776", // Intentional: hand-rolled RFC 6570 reverse matcher; control-flow verified by McpUriTemplateMatcher_Test, refactor would risk matching correctness.
264 		"java:S127", // The i++ skip of the paired literal piece after consuming a capture is intentional and documented by the class-level "Ambiguous-capture resolution" section.
265 		"java:S135" // Single continue plus the loop-counter skip above are the simplest expression of this piece-by-piece scan; splitting it out would risk matching correctness.
266 	})
267 	private Map<String,String> matchPieces(String normalized) {
268 		var captures = new LinkedHashMap<String,String>();
269 		var pos = 0;
270 		for (var i = 0; i < pieces.size(); i++) {
271 			var piece = pieces.get(i);
272 			if (piece instanceof LiteralPiece lit) {
273 				if (! normalized.startsWith(lit.text(), pos))
274 					return null;
275 				pos += lit.text().length();
276 				continue;
277 			}
278 			var capture = (CapturePiece) piece;
279 			var maxEnd = scanMaxEnd(normalized, pos, capture.stopChars());
280 			// Invariant established at compile time: a capture piece is always followed by either a literal
281 			// piece or the end of the template, never by another capture (see class Javadoc).
282 			var nextLiteral = (i + 1 < pieces.size() && pieces.get(i + 1) instanceof LiteralPiece lp) ? lp : null;
283 			String rawValue;
284 			if (nextLiteral == null) {
285 				rawValue = normalized.substring(pos, maxEnd);
286 				pos = maxEnd;
287 			} else {
288 				var literalText = nextLiteral.text();
289 				var searchFrom = Math.min(maxEnd, normalized.length() - literalText.length());
290 				var foundAt = searchFrom < pos ? -1 : normalized.lastIndexOf(literalText, searchFrom);
291 				if (foundAt < pos)
292 					return null;
293 				rawValue = normalized.substring(pos, foundAt);
294 				pos = foundAt + literalText.length();
295 				i++;
296 			}
297 			var decoded = decodeCapture(rawValue);
298 			if (decoded == null)
299 				return null;
300 			captures.put(capture.name(), decoded);
301 		}
302 		if (pos != normalized.length())
303 			return null;
304 		return Collections.unmodifiableMap(captures);
305 	}
306 
307 	/**
308 	 * Scans forward from {@code pos} while the current character is not one of {@code stopChars}, returning the
309 	 * bound at which a capture governed by that stop set must end. An empty stop set (reserved/fragment captures)
310 	 * scans unconditionally to the end of the string.
311 	 *
312 	 * <p>
313 	 * Because {@code stopChars} only ever contains reserved delimiter characters (<c>/</c>, <c>.</c>, <c>&amp;</c>,
314 	 * <c>#</c>) - none of which are <c>%</c> or a hex digit - a normalized <c>%XX</c> triplet can never be
315 	 * mistaken for one of these characters, so a plain character scan over the already-normalized string
316 	 * correctly treats an encoded delimiter as data rather than a boundary.
317 	 */
318 	private static int scanMaxEnd(String normalized, int pos, Set<Character> stopChars) {
319 		if (stopChars.isEmpty())
320 			return normalized.length();
321 		var i = pos;
322 		while (i < normalized.length() && ! stopChars.contains(normalized.charAt(i)))
323 			i++;
324 		return i;
325 	}

328 	// Tokenizing: split the raw template into literal and expression segments
329 	//-----------------------------------------------------------------------------------------------------------------
330 
331 	private sealed interface RawToken permits RawLiteral, RawExpr {}
332 
333 	private record RawLiteral(String text) implements RawToken {}
334 
335 	private record RawExpr(String body, int index) implements RawToken {}
336 
337 	@SuppressWarnings({
338 		"java:S3776" // Intentional: hand-rolled RFC 6570 reverse matcher; control-flow verified by McpUriTemplateMatcher_Test, refactor would risk matching correctness.
339 	})
340 	private static List<RawToken> tokenize(String template) {
341 		var out = new ArrayList<RawToken>();
342 		var n = template.length();
343 		var i = 0;
344 		var lit = new StringBuilder();
345 		while (i < n) {
346 			var c = template.charAt(i);
347 			if (c == '{') {
348 				if (! lit.isEmpty()) {
349 					out.add(new RawLiteral(lit.toString()));
350 					lit.setLength(0);
351 				}
352 				var close = template.indexOf('}', i + 1);
353 				if (close < 0)
354 					throw iaex("Malformed URI template ''%s'': unterminated '{' at index %s", template, i);
355 				var body = template.substring(i + 1, close);
356 				if (body.indexOf('{') >= 0)
357 					throw iaex("Malformed URI template ''%s'': nested '{' inside expression at index %s", template, i);
358 				if (body.isEmpty())
359 					throw iaex("Malformed URI template ''%s'': empty expression '{}' at index %s", template, i);
360 				out.add(new RawExpr(body, i));
361 				i = close + 1;
362 			} else if (c == '}') {
363 				throw iaex("Malformed URI template ''%s'': unmatched '}' at index %s", template, i);
364 			} else {
365 				lit.append(c);
366 				i++;
367 			}
368 		}
369 		if (! lit.isEmpty())
370 			out.add(new RawLiteral(lit.toString()));
371 		return out;
372 	}

375 	// Expression grammar: operator + comma-separated varspec list
376 	//-----------------------------------------------------------------------------------------------------------------
377 
378 	private enum Modifier { NONE, PREFIX, EXPLODE }
379 
380 	private record VarSpec(String name, Modifier modifier) {}
381 
382 	private record ParsedExpr(char operator, List<VarSpec> vars) {}
383 
384 	private static final Set<Character> OPERATORS = Set.of('+', '#', '.', '/', ';', '?', '&');
385 	private static final Set<Character> RESERVED_FOR_FUTURE_OPERATORS = Set.of('!', '=', ',', '|', '@');
386 
387 	@SuppressWarnings({
388 		"java:S5998" // Bounded by RFC 6570 varname grammar (dot-separated [A-Za-z0-9_]+ segments); real inputs are short template strings, not attacker-controlled bulk text.
389 	})
390 	private static final Pattern VARNAME = Pattern.compile("\\w+(?:\\.\\w+)*");
391 
392 	private static ParsedExpr parseExpr(String body, String template, int index) {
393 		var first = body.charAt(0);
394 		char operator;
395 		String rest;
396 		if (OPERATORS.contains(first)) {
397 			operator = first;
398 			rest = body.substring(1);
399 		} else if (RESERVED_FOR_FUTURE_OPERATORS.contains(first)) {
400 			throw iaex("Malformed URI template ''%s'': unsupported operator ''%s'' in expression at index %s",
401 				template, first, index);
402 		} else {
403 			operator = '\0';
404 			rest = body;
405 		}
406 		if (rest.isEmpty())
407 			throw iaex("Malformed URI template ''%s'': empty variable list in expression at index %s", template, index);
408 		var vars = new ArrayList<VarSpec>();
409 		for (var part : rest.split(",", -1))
410 			vars.add(parseVarSpec(part, template, index));
411 		return new ParsedExpr(operator, vars);
412 	}
413 
414 	private static VarSpec parseVarSpec(String spec, String template, int index) {
415 		if (spec.isEmpty())
416 			throw iaex("Malformed URI template ''%s'': empty variable name in expression at index %s", template, index);
417 		if (spec.endsWith("*")) {
418 			var name = spec.substring(0, spec.length() - 1);
419 			validateVarName(name, template, index);
420 			return new VarSpec(name, Modifier.EXPLODE);
421 		}
422 		var colon = spec.indexOf(':');
423 		if (colon >= 0) {
424 			var name = spec.substring(0, colon);
425 			var digits = spec.substring(colon + 1);
426 			validateVarName(name, template, index);
427 			if (digits.isEmpty() || digits.length() > 4 || ! digits.chars().allMatch(Character::isDigit))
428 				throw iaex("Malformed URI template ''%s'': invalid prefix-length modifier ''%s'' in expression at index %s",
429 					template, spec, index);
430 			return new VarSpec(name, Modifier.PREFIX);
431 		}
432 		validateVarName(spec, template, index);
433 		return new VarSpec(spec, Modifier.NONE);
434 	}
435 
436 	private static void validateVarName(String name, String template, int index) {
437 		if (! VARNAME.matcher(name).matches())
438 			throw iaex("Malformed URI template ''%s'': invalid variable name ''%s'' in expression at index %s",
439 				template, name, index);
440 	}

444 	//-----------------------------------------------------------------------------------------------------------------
445 
446 	private sealed interface Piece permits LiteralPiece, CapturePiece {}
447 
448 	private record LiteralPiece(String text) implements Piece {}
449 
450 	private record CapturePiece(String name, Set<Character> stopChars) implements Piece {}
451 
452 	private static final Set<Character> STOP_SLASH = Set.of('/');
453 	private static final Set<Character> STOP_SLASH_DOT = Set.of('/', '.');
454 	private static final Set<Character> STOP_QUERY = Set.of('&', '#');
455 	private static final Set<Character> STOP_NONE = Set.of();
456 
457 	/**
458 	 * Determines whether a single parsed expression is one of the settled reversible forms, independent of its
459 	 * neighbors (adjacency ambiguity is a whole-template concern handled by the caller).
460 	 */
461 	private static boolean isExprReverseMatchable(ParsedExpr expr) {
462 		if (expr.vars().stream().anyMatch(v -> v.modifier() != Modifier.NONE))
463 			return false;
464 		return switch (expr.operator()) {
465 			case '\0', '+', '#' -> expr.vars().size() == 1;
466 			case '/', '.', '?', '&' -> true;
467 			default -> false;
468 		};
469 	}
470 
471 	/**
472 	 * Expands a reverse-matchable expression into its literal/capture piece sequence. Every branch here ends
473 	 * with a {@link CapturePiece}, which is what guarantees the "capture is always followed by a literal or the
474 	 * end of the template" invariant documented on the class and relied on by {@link #matchPieces(String)}.
475 	 */
476 	private static List<Piece> desugar(ParsedExpr expr) {
477 		var out = new ArrayList<Piece>();
478 		switch (expr.operator()) {
479 			case '\0' -> out.add(new CapturePiece(expr.vars().get(0).name(), STOP_SLASH));
480 			case '+' -> out.add(new CapturePiece(expr.vars().get(0).name(), STOP_NONE));
481 			case '#' -> {
482 				out.add(new LiteralPiece("#"));
483 				out.add(new CapturePiece(expr.vars().get(0).name(), STOP_NONE));
484 			}
485 			case '/' -> {
486 				for (var v : expr.vars()) {
487 					out.add(new LiteralPiece("/"));
488 					out.add(new CapturePiece(v.name(), STOP_SLASH));
489 				}
490 			}
491 			case '.' -> {
492 				for (var v : expr.vars()) {
493 					out.add(new LiteralPiece("."));
494 					out.add(new CapturePiece(v.name(), STOP_SLASH_DOT));
495 				}
496 			}
497 			case '?' -> {
498 				var first = true;
499 				for (var v : expr.vars()) {
500 					out.add(new LiteralPiece((first ? "?" : "&") + v.name() + "="));
501 					out.add(new CapturePiece(v.name(), STOP_QUERY));
502 					first = false;
503 				}
504 			}
505 			case '&' -> {
506 				for (var v : expr.vars()) {
507 					out.add(new LiteralPiece("&" + v.name() + "="));
508 					out.add(new CapturePiece(v.name(), STOP_QUERY));
509 				}
510 			}
511 			default -> { /* matrix and any other non-matchable operator: never reached, never used for matching */ }
512 		}
513 		return out;
514 	}

520 	private static final char[] HEX_DIGITS = "0123456789ABCDEF".toCharArray();
521 
522 	private static boolean isUnreserved(int octet) {
523 		return (octet >= 'A' && octet <= 'Z') || (octet >= 'a' && octet <= 'z') || (octet >= '0' && octet <= '9')
524 			|| octet == '-' || octet == '.' || octet == '_' || octet == '~';
525 	}
526 
527 	private static int hexValue(char c) {
528 		if (c >= '0' && c <= '9')
529 			return c - '0';
530 		if (c >= 'A' && c <= 'F')
531 			return c - 'A' + 10;
532 		if (c >= 'a' && c <= 'f')
533 			return c - 'a' + 10;
534 		return -1;
535 	}

537 	/**
538 	 * Normalizes a template literal segment at compile time: uppercases valid percent triplets, decodes
539 	 * percent-encoded unreserved octets to their literal character, and leaves percent-encoded reserved/other
540 	 * octets encoded. Also validates that the segment's percent-escaped bytes, combined with its raw characters,
541 	 * form valid UTF-8. Throws on any malformed escape or invalid UTF-8, naming the offending template.
542 	 */
543 	private static String normalizeLiteralStrict(String raw, String template) {
544 		var sb = new StringBuilder();
545 		var bytes = new ByteArrayOutputStream();
546 		var n = raw.length();
547 		var i = 0;
548 		while (i < n) {
549 			var c = raw.charAt(i);
550 			if (c == '%') {
551 				if (i + 2 >= n || hexValue(raw.charAt(i + 1)) < 0 || hexValue(raw.charAt(i + 2)) < 0)
552 					throw iaex("Malformed URI template ''%s'': malformed percent-escape at index %s", template, i);
553 				var octet = (hexValue(raw.charAt(i + 1)) << 4) | hexValue(raw.charAt(i + 2));
554 				bytes.write(octet);
555 				appendNormalizedOctet(sb, octet);
556 				i += 3;
557 			} else {
558 				bytes.writeBytes(String.valueOf(c).getBytes(StandardCharsets.UTF_8));
559 				sb.append(c);
560 				i++;
561 			}
562 		}
563 		if (! isValidUtf8(bytes.toByteArray()))
564 			throw iaex("Malformed URI template ''%s'': invalid UTF-8 percent-escape sequence in literal text", template);
565 		return sb.toString();
566 	}

568 	/**
569 	 * Normalizes an entire concrete URI at match time, using the same rules as
570 	 * {@link #normalizeLiteralStrict(String, String)} but returning <jk>null</jk> instead of throwing when a
571 	 * percent escape is malformed, per the rule that a concrete URI with a malformed escape simply does not
572 	 * template-match.
573 	 */
574 	private static String normalizeForMatch(String raw) {
575 		var sb = new StringBuilder();
576 		var n = raw.length();
577 		var i = 0;
578 		while (i < n) {
579 			var c = raw.charAt(i);
580 			if (c == '%') {
581 				if (i + 2 >= n)
582 					return null;
583 				var h1 = hexValue(raw.charAt(i + 1));
584 				var h2 = hexValue(raw.charAt(i + 2));
585 				if (h1 < 0 || h2 < 0)
586 					return null;
587 				appendNormalizedOctet(sb, (h1 << 4) | h2);
588 				i += 3;
589 			} else {
590 				sb.append(c);
591 				i++;
592 			}
593 		}
594 		return sb.toString();
595 	}
596 
597 	private static void appendNormalizedOctet(StringBuilder sb, int octet) {
598 		if (isUnreserved(octet))
599 			sb.append((char) octet);
600 		else
601 			sb.append('%').append(HEX_DIGITS[(octet >> 4) & 0xF]).append(HEX_DIGITS[octet & 0xF]);
602 	}

604 	/**
605 	 * Counts normalized literal octets in an already-normalized string: each <c>%XX</c> triplet counts as one
606 	 * octet, and every other character counts as one octet.
607 	 */
608 	private static int countOctets(String normalized) {
609 		var count = 0;
610 		var i = 0;
611 		var n = normalized.length();
612 		while (i < n) {
613 			i += normalized.charAt(i) == '%' ? 3 : 1;
614 			count++;
615 		}
616 		return count;
617 	}

619 	/**
620 	 * Percent-decodes an already-normalized capture substring to its final UTF-8 value for placement in the
621 	 * result map, rejecting (returning <jk>null</jk> for) captures whose decoded bytes are not valid UTF-8.
622 	 */
623 	private static String decodeCapture(String normalizedSubstring) {
624 		var bytes = new ByteArrayOutputStream();
625 		var n = normalizedSubstring.length();
626 		var i = 0;
627 		while (i < n) {
628 			var c = normalizedSubstring.charAt(i);
629 			if (c == '%') {
630 				bytes.write((hexValue(normalizedSubstring.charAt(i + 1)) << 4) | hexValue(normalizedSubstring.charAt(i + 2)));
631 				i += 3;
632 			} else {
633 				bytes.writeBytes(String.valueOf(c).getBytes(StandardCharsets.UTF_8));
634 				i++;
635 			}
636 		}
637 		var data = bytes.toByteArray();
638 		if (! isValidUtf8(data))
639 			return null;
640 		return new String(data, StandardCharsets.UTF_8);
641 	}
642 
643 	private static boolean isValidUtf8(byte[] data) {
644 		try {
645 			StandardCharsets.UTF_8.newDecoder()
646 				.onMalformedInput(CodingErrorAction.REPORT)
647 				.onUnmappableCharacter(CodingErrorAction.REPORT)
648 				.decode(ByteBuffer.wrap(data));
649 			return true;
650 		} catch (CharacterCodingException e) {
651 			return false;
652 		}
653 	}
