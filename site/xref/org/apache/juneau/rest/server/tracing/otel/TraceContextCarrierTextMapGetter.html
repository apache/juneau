17  package org.apache.juneau.rest.server.tracing.otel;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.http.tracing.TraceContextCarrier;
22  
23  import io.opentelemetry.context.propagation.*;
24  
25  /**
26   * {@link TextMapGetter} implementation that reads W3C trace-context fields from a neutral
27   * {@link TraceContextCarrier} &mdash; for example an MCP request's {@code params._meta} object
28   * &mdash; instead of HTTP request headers.
29   *
30   * <p>
31   * {@link OtelTracerHook} chains an extraction pass through this getter after an ordinary HTTP-header
32   * extraction (via {@link RestRequestTextMapGetter}) so an explicit carrier value wins while an
33   * absent/invalid one falls back to whatever the HTTP-header extraction already produced. This relies
34   * entirely on the standard {@link TextMapPropagator} contract &mdash; extraction that finds nothing
35   * usable for a given concern returns the supplied {@link io.opentelemetry.context.Context} unmodified
36   * &mdash; so no bespoke precedence-merging logic lives here or in {@link OtelTracerHook}.
37   *
38   * @since 10.0.0
39   */
40  @SuppressWarnings({
41  	"java:S6548" // Singleton pattern is intentional; INSTANCE is a stateless, thread-safe TextMapGetter implementation.
42  })
43  public final class TraceContextCarrierTextMapGetter implements TextMapGetter<TraceContextCarrier> {
44  
45  	/** Process-wide singleton instance. */
46  	public static final TraceContextCarrierTextMapGetter INSTANCE = new TraceContextCarrierTextMapGetter();
47  
48  	private TraceContextCarrierTextMapGetter() {}
49  
50  	@Override /* TextMapGetter */
51  	public Iterable<String> keys(TraceContextCarrier carrier) {
52  		if (carrier == null)
53  			return List.of();
54  		var keys = carrier.keys();
55  		return keys == null ? List.of() : keys;
56  	}
57  
58  	@Override /* TextMapGetter */
59  	public String get(TraceContextCarrier carrier, String key) {
60  		if (carrier == null || key == null)
61  			return null;
62  		var value = carrier.get(key);
63  		return (value == null || value.isEmpty()) ? null : value;
64  	}
65  }
