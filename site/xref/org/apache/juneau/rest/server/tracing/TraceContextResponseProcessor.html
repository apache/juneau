18  
19  import static org.apache.juneau.commons.utils.StringUtils.*;
20  
21  import java.io.*;
22  import java.util.logging.*;
23  
24  import org.apache.juneau.http.response.*;
25  import org.apache.juneau.rest.server.*;
26  import org.apache.juneau.rest.server.processor.*;
27  
28  /**
29   * Writes the W3C Trace Context response headers ({@code traceparent} and, when present,
30   * {@code tracestate}) so a client calling this server can read the resulting trace identifiers back
31   * off the response.
32   *
33   * <h5 class='topic'>How it works</h5>
34   *
35   * <p>
36   * A {@link TracerHook} bridge (e.g. the OpenTelemetry bridge in {@code juneau-rest-server-tracing-otel})
37   * renders the W3C {@code traceparent} / {@code tracestate} values from the <i>server-started</i> span
38   * context at {@link TracerHook#startSpan(RestRequest) span-start} time &mdash; the only point where
39   * that context is reliably active &mdash; and stashes them as the request attributes
40   * {@link #ATTR_TRACEPARENT} / {@link #ATTR_TRACESTATE}. This processor runs later, during response
41   * rendering (after the span scope has already closed), reads those stashed strings, and writes them
42   * as response headers.
43   *
44   * <p>
45   * This capture-at-span-start / write-at-response-time split is deliberate: the framework opens and
46   * closes the tracing scope around the {@code @RestOp} handler in
47   * {@link RestOpInvoker#invokeOp(RestOpSession)}, which completes <i>before</i> response processors
48   * run, so {@code Span.current()} is no longer valid here. Keeping the rendered values on the request
49   * also keeps this processor free of any OpenTelemetry (or other tracer) dependency &mdash; it works
50   * with any {@code TracerHook} bridge that populates the two attributes.
51   *
52   * <h5 class='topic'>Off-by-default behavior</h5>
53   *
54   * <p>
55   * The first action in {@link #process(RestOpSession)} is a single request-attribute read. When no
56   * non-no-op {@code TracerHook} ran for the request, no {@link #ATTR_TRACEPARENT} attribute is present
57   * and the processor returns {@link #NEXT} immediately with no allocations &mdash; matching the
58   * off-by-default, zero-cost contract of the observability hooks. Registration of this processor is
59   * itself gated on {@code RestContext.responseTraceparent} (default {@code true}); setting it to
60   * {@code false} keeps the processor out of the chain entirely.
61   *
62   * <h5 class='topic'>Already-committed responses</h5>
63   *
64   * <p>
65   * If the response has already been committed (for example a streaming / SSE handler that flushed its
66   * headers before returning), the headers can no longer be set. In that case the processor logs a
67   * {@link Level#FINE FINE} message and skips the write &mdash; it never throws.
68   *
69   * <h5 class='section'>See Also:</h5><ul>
70   * 	<li class='jc'>{@link TracerHook}
71   * 	<li class='link'><a class="doclink" href="https://www.w3.org/TR/trace-context/">W3C Trace Context</a>
72   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestServerObservability">REST Server &mdash; Observability (Micrometer + OpenTelemetry)</a>
73   * </ul>
74   *
75   * @since 10.0.0
76   */
77  public class TraceContextResponseProcessor implements ResponseProcessor {
78  
79  	/**
80  	 * Request-attribute key under which a {@link TracerHook} bridge stashes the rendered W3C
81  	 * {@code traceparent} value (e.g. {@code 00-0af7651916cd43dd8448eb211c80319c-b7ad6b7169203331-01})
82  	 * for this processor to emit as a response header.
83  	 */
84  	public static final String ATTR_TRACEPARENT = "juneau.traceparent";
85  
86  	/**
87  	 * Request-attribute key under which a {@link TracerHook} bridge stashes the rendered W3C
88  	 * {@code tracestate} value (a comma-separated {@code key=value} list). Only set when the active
89  	 * trace state is non-empty.
90  	 */
91  	public static final String ATTR_TRACESTATE = "juneau.tracestate";
92  
93  	/**
94  	 * Request-attribute key under which a {@link TracerHook} bridge stashes the rendered, non-empty W3C
95  	 * {@code baggage} value.
96  	 *
97  	 * <p>
98  	 * Unlike {@link #ATTR_TRACEPARENT} / {@link #ATTR_TRACESTATE}, this processor does <b>not</b> read
99  	 * this attribute and never emits a {@code baggage} response header from it &mdash; {@code baggage} is
100 	 * stashed purely so a dated MCP adapter can echo it into a successful result's {@code _meta} (added in
101 	 * a later task). Ordinary HTTP response-header behavior for {@code traceparent} / {@code tracestate} is
102 	 * unaffected by this constant's existence.
103 	 *
104 	 * @since 10.0.0
105 	 */
106 	public static final String ATTR_BAGGAGE = "juneau.baggage";
107 
108 	/**
109 	 * Request-attribute key under which {@link RestOpInvoker} stashes the active {@link Scope} for the
110 	 * duration of one {@code @RestOp} handler invocation.
111 	 *
112 	 * <p>
113 	 * Populated only on the active-tracer path &mdash; see the no-tracer fast-path contract documented on
114 	 * {@link TracerHook} &mdash; and cleared once that invocation's synchronous or deferred observability
115 	 * lifecycle closes the scope. A dated MCP adapter running inside the handler already resolves this
116 	 * same {@link RestRequest} through its own request-scoped {@code BeanStore} entry; reading this
117 	 * attribute off that request lets it call {@link Scope#recordRpcError(int, String)} before returning a
118 	 * JSON-RPC error response &mdash; a JSON-RPC error is dispatched over HTTP {@code 200}, so it never
119 	 * otherwise reaches {@link Scope#setStatusCode(int)} with anything but the adapter's uniform success
120 	 * status.
121 	 *
122 	 * @since 10.0.0
123 	 */
124 	public static final String ATTR_SCOPE = "juneau.tracerScope";
125 
126 	private static final String HEADER_TRACEPARENT = "traceparent";
127 	private static final String HEADER_TRACESTATE = "tracestate";
128 
129 	private static final Logger LOG = Logger.getLogger(TraceContextResponseProcessor.class.getName());
130 
131 	@Override /* Overridden from ResponseProcessor */
132 	public int process(RestOpSession opSession) throws IOException, BasicHttpException {
133 		var req = opSession.getRequest();
134 
135 		// NoOp short-circuit: no active tracer stashed a trace context for this request.
136 		var traceparent = req.getAttribute(ATTR_TRACEPARENT).as(String.class).orElse(null);
137 		if (isEmpty(traceparent))
138 			return NEXT;
139 
140 		var res = opSession.getResponse();
141 
142 		// Headers already flushed (e.g. streaming / SSE response) — body was also written; chain is done.
143 		if (res.getHttpServletResponse().isCommitted()) {
144 			LOG.log(Level.FINE, () -> "Response already committed; skipping traceparent/tracestate header injection.");
145 			return FINISHED;
146 		}
147 
148 		res.setHeader(HEADER_TRACEPARENT, traceparent);
149 
150 		var tracestate = req.getAttribute(ATTR_TRACESTATE).as(String.class).orElse(null);
151 		if (tracestate != null && ! tracestate.isEmpty())
152 			res.setHeader(HEADER_TRACESTATE, tracestate);
153 
154 		// Headers only — let the downstream processors render the body.
155 		return NEXT;
156 	}
157 }
