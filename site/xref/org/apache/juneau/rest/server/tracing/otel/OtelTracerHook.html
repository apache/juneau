36  /**
37   * {@link TracerHook} bridge that opens an OpenTelemetry {@link Span} for every {@code @RestOp} call.
38   *
39   * <h5 class='topic'>Span shape</h5>
40   *
41   * <p>
42   * Each request becomes a single span with kind {@link SpanKind#SERVER}, named after the HTTP method
43   * (e.g. {@code "GET"}, {@code "POST"}) per the OpenTelemetry HTTP semantic conventions. Three OTel
44   * HTTP semantic-convention attributes are populated:
45   * <ul>
46   * 	<li><b>{@code http.request.method}</b> &mdash; uppercased HTTP method.
47   * 	<li><b>{@code http.response.status_code}</b> &mdash; HTTP response status (set right before the
48   * 		span is closed, so error paths get the framework's resolved status).
49   * 	<li><b>{@code http.route}</b> &mdash; the {@code @RestOp} path template (e.g.
50   * 		{@code "/users/{id}"}), not the raw concrete URI. Keeps trace cardinality bounded for the same
51   * 		reason the Micrometer bridge uses the template as its {@code uri} tag.
52   * </ul>
53   *
54   * <p>
55   * On the exception path, the throwable is recorded via {@link Span#recordException(Throwable)}, the
56   * span status is set to {@link StatusCode#ERROR}, and an {@code exception.type} attribute carries the
57   * exception's simple-name.
58   *
59   * <h5 class='topic'>W3C trace context propagation</h5>
60   *
61   * <p>
62   * Incoming {@code traceparent} / {@code tracestate} request headers are extracted via the configured
63   * {@link TextMapPropagator} (default: {@link io.opentelemetry.api.trace.propagation.W3CTraceContextPropagator}) so the server span continues
64   * a caller-supplied distributed trace. Downstream HTTP calls made from inside the handler will pick
65   * up the active span's context automatically when the consumer's outbound HTTP client honors the OTel
66   * {@link Context#current() current context} (the standard OTel client instrumentations do).
67   *
68   * <h5 class='topic'>Wiring</h5>
69   *
70   * <p class='bjava'>
71   * 	<ja>@Configuration</ja>
72   * 	<jk>public class</jk> ObservabilityConfig {
73   *
74   * 		<jc>// Option A: rely on the JVM-wide GlobalOpenTelemetry (recommended).</jc>
75   * 		<ja>@Bean</ja>
76   * 		<jk>public</jk> TracerHook tracerHook() {
77   * 			<jk>return new</jk> OtelTracerHook();
78   * 		}
79   *
80   * 		<jc>// Option B: pass a specific OpenTelemetry instance (e.g. for tests or multi-tenant setups).</jc>
81   * 		<ja>@Bean</ja>
82   * 		<jk>public</jk> TracerHook tracerHook(OpenTelemetry <jv>otel</jv>) {
83   * 			<jk>return new</jk> OtelTracerHook(<jv>otel</jv>);
84   * 		}
85   * 	}
86   * </p>
87   *
88   * <h5 class='section'>See Also:</h5><ul>
89   * 	<li class='jc'>{@link TracerHook}
90   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestServerObservability">REST Server &mdash; Observability (Micrometer + OpenTelemetry)</a>
91   * 	<li class='link'><a class="doclink" href="https://opentelemetry.io/docs/specs/semconv/http/http-spans/">OpenTelemetry HTTP semantic conventions</a>
92   * 	<li class='link'><a class="doclink" href="https://www.w3.org/TR/trace-context/">W3C Trace Context</a>
93   * </ul>
94   *
95   * @since 10.0.0
96   */

137 	/**
138 	 * Constructor with an explicit {@link OpenTelemetry} instance.
139 	 *
140 	 * <p>
141 	 * The instance is used to obtain both the {@link Tracer} (named
142 	 * {@value #DEFAULT_INSTRUMENTATION_NAME}) and the {@link TextMapPropagator} used for incoming /
143 	 * outgoing W3C trace-context header transfer.
144 	 *
145 	 * @param openTelemetry The {@link OpenTelemetry} instance. Must not be <jk>null</jk>.
146 	 */
147 	public OtelTracerHook(OpenTelemetry openTelemetry) {
148 		assertArgNotNull(ARG_openTelemetry, openTelemetry);
149 		this.tracer = openTelemetry.getTracer(DEFAULT_INSTRUMENTATION_NAME);
150 		this.propagator = openTelemetry.getPropagators().getTextMapPropagator();
151 	}

154 	 * Constructor with an explicit {@link Tracer} and {@link TextMapPropagator}.

159 	 *

174 	@Override /* TracerHook */
175 	public Scope startSpan(RestRequest request, TraceContextCarrier carrier, TraceOperation operation) {
176 		Context extracted = extractContext(request, carrier);
177 
178 		String httpMethod = defaultIfBlank(request.getMethod(), "UNKNOWN");
179 		String route = resolveRoute(request);
180 		String spanName = defaultIfBlank(operation.getSpanName(), httpMethod);
181 
182 		var spanBuilder = tracer.spanBuilder(spanName)
183 			.setSpanKind(SpanKind.SERVER)
184 			.setParent(extracted)
185 			.setAttribute(ATTR_HTTP_REQUEST_METHOD, httpMethod);
186 		if (! route.isEmpty())
187 			spanBuilder.setAttribute(ATTR_HTTP_ROUTE, route);
188 		operation.getAttributes().forEach((k, v) -> spanBuilder.setAttribute(AttributeKey.stringKey(k), v));
189 
190 		Span span = spanBuilder.startSpan();
191 		Context spanContext = extracted.with(span);
192 
193 		// Render the W3C trace-context headers (and baggage, when a baggage propagator is configured)
194 		// from the server-started span's context (the only point where it's reliably active) and stash
195 		// them as request attributes for TraceContextResponseProcessor / a dated MCP result to read
196 		// later. The configured propagator only injects traceparent when the span context is valid, and
197 		// tracestate/baggage only when non-empty, so both guards are handled here for free.
198 		stashTraceContext(request, spanContext);
199 
200 		io.opentelemetry.context.Scope otelScope = spanContext.makeCurrent();
201 
202 		return new OtelScope(span, otelScope);
203 	}
204 
205 	/**
206 	 * Builds the effective parent {@link Context} for a new server span, giving a non-HTTP
207 	 * {@link TraceContextCarrier} &mdash; recognized by an active {@code TraceContextExtractor} among
208 	 * the resolved operation arguments (for example an MCP request's {@code params._meta}) &mdash;
209 	 * precedence over HTTP request headers.
210 	 *
211 	 * <p>
212 	 * Extraction runs in up to two passes against the same configured {@link #propagator}: first
213 	 * against ordinary HTTP headers (via {@link RestRequestTextMapGetter}), then &mdash; only when
214 	 * <c>carrier</c> is non-<jk>null</jk> &mdash; a second pass against <c>carrier</c> (via
215 	 * {@link TraceContextCarrierTextMapGetter}), seeded with the first pass's result. Per the
216 	 * {@link TextMapPropagator} contract, extraction that finds nothing usable returns the given
217 	 * {@link Context} unmodified, so an explicit carrier value wins while an absent or invalid one
218 	 * falls back to whatever the HTTP pass already produced. This applies independently to the
219 	 * {@code traceparent}/{@code tracestate} pair (extracted together as one {@code SpanContext}) and
220 	 * to {@code baggage} (extracted separately whenever the configured propagator includes a baggage
221 	 * propagator).
222 	 *
223 	 * @param request The in-flight request; source of the HTTP-header fallback. Never <jk>null</jk>.
224 	 * @param carrier The extractor-recognized carrier, or <jk>null</jk> if none was recognized (HTTP
225 	 * 	headers only).
226 	 * @return The effective parent {@link Context}. Never <jk>null</jk>.
227 	 */
228 	private Context extractContext(RestRequest request, TraceContextCarrier carrier) {
229 		Context httpContext = propagator.extract(Context.current(), request, RestRequestTextMapGetter.INSTANCE);
230 		if (carrier == null)
231 			return httpContext;
232 		return propagator.extract(httpContext, carrier, TraceContextCarrierTextMapGetter.INSTANCE);
233 	}

235 	@Override /* TracerHook */
236 	public Scope startSpan(String spanName) {
237 		// Custom (non-request) observation: open an INTERNAL span parented to the current context
238 		// (so it nests under an active request span when one exists), with no HTTP attributes.
239 		Span span = tracer.spanBuilder(spanName)
240 			.setSpanKind(SpanKind.INTERNAL)
241 			.setParent(Context.current())
242 			.startSpan();
243 		io.opentelemetry.context.Scope otelScope = Context.current().with(span).makeCurrent();
244 		return new OtelScope(span, otelScope);
245 	}
246 
247 	private void stashTraceContext(RestRequest request, Context spanContext) {
248 		var carrier = new HashMap<String,String>(4);
249 		propagator.inject(spanContext, carrier, MAP_SETTER);
250 		var traceparent = carrier.get("traceparent");
251 		if (nn(traceparent))
252 			request.setAttribute(TraceContextResponseProcessor.ATTR_TRACEPARENT, traceparent);
253 		var tracestate = carrier.get("tracestate");
254 		if (nn(tracestate) && ! tracestate.isEmpty())
255 			request.setAttribute(TraceContextResponseProcessor.ATTR_TRACESTATE, tracestate);
256 		var baggage = carrier.get("baggage");
257 		if (nn(baggage) && ! baggage.isEmpty())
258 			request.setAttribute(TraceContextResponseProcessor.ATTR_BAGGAGE, baggage);
259 	}

261 	// Package-private (not private) so a unit test can exercise the null-carrier guard directly — TODO-185f.
262 	static final TextMapSetter<HashMap<String,String>> MAP_SETTER = (carrier, key, value) -> {
263 		if (nn(carrier))
264 			carrier.put(key, value);
265 	};

267 	@Override /* TracerHook */
268 	public void inject(TraceContextCarrier carrier) {
269 		propagator.inject(Context.current(), carrier, TraceContextCarrierTextMapSetter.INSTANCE);
270 	}

287 	private static String resolveRoute(RestRequest request) {
288 		try {
289 			var opCtx = request.getOpContext();
290 			return opCtx == null ? "" : defaultIfBlank(opCtx.getPathPattern(), "");
291 		} catch (RuntimeException e) {
292 			// Defensive: opCtx.getPathPattern() reaches into the path-matcher array; protect against NPE / array bounds in odd routing setups.
293 			return "";
294 		}
295 	}
296 
297 	private static String defaultIfBlank(String value, String fallback) {
298 		return isEmpty(value) ? fallback : value;
299 	}

301 	private static final class OtelScope implements Scope {
302 
303 		private final Span span;
304 		private final io.opentelemetry.context.Scope scope;
305 
306 		OtelScope(Span span, io.opentelemetry.context.Scope scope) {
307 			this.span = span;
308 			this.scope = scope;
309 		}
310 
311 		@Override /* Scope */
312 		public void setStatusCode(int statusCode) {
313 			span.setAttribute(ATTR_HTTP_RESPONSE_STATUS_CODE, (long) statusCode);
314 			if (statusCode >= 500)
315 				span.setStatus(StatusCode.ERROR);
316 		}
317 
318 		@Override /* Scope */
319 		public void setError(Throwable error) {
320 			if (nn(error)) {
321 				span.recordException(error);
322 				span.setAttribute(ATTR_EXCEPTION_TYPE, cns(error));
323 				span.setStatus(StatusCode.ERROR);
324 			}
325 		}
326 
327 		@Override /* Scope */
328 		public void close() {
329 			try {
330 				scope.close();
331 			} finally {
332 				span.end();
333 			}
334 		}
335 
336 		@Override /* Scope */
337 		public void recordRpcError(int code, String message) {
338 			span.setAttribute(TraceOperation.ATTR_RPC_RESPONSE_STATUS_CODE, String.valueOf(code));
339 			span.setAttribute(TraceOperation.ATTR_ERROR_TYPE, errorType(code));
340 			span.setStatus(StatusCode.ERROR, message == null ? "" : message);
341 		}
342 
343 		/**
344 		 * Maps a JSON-RPC 2.0 error code to a low-cardinality {@code error.type} category.
345 		 *
346 		 * <p>
347 		 * The five codes reserved by the JSON-RPC 2.0 specification map to their exact category name;
348 		 * any code in the specification's reserved-for-implementation-defined-server-errors range
349 		 * ({@code -32000} to {@code -32099}, inclusive) maps to the generic {@code "server_error"}
350 		 * category. An application-defined code outside every reserved range falls back to its own
351 		 * decimal string so the attribute stays populated (if not maximally low-cardinality) rather than
352 		 * silently empty.
353 		 *
354 		 * @param code The JSON-RPC error code (e.g. {@code -32601}).
355 		 * @return The {@code error.type} category. Never <jk>null</jk>.
356 		 */
357 		private static String errorType(int code) {
358 			switch (code) {
359 				case -32700: return "parse_error";
360 				case -32600: return "invalid_request";
361 				case -32601: return "method_not_found";
362 				case -32602: return "invalid_params";
363 				case -32603: return "internal_error";
364 				default:
365 					return (code <= -32000 && code >= -32099) ? "server_error" : String.valueOf(code);
366 			}
367 		}
368 	}
