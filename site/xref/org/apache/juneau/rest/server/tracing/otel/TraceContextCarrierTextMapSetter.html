19  import static org.apache.juneau.commons.utils.Shorts.*;

21  import org.apache.juneau.http.tracing.TraceContextCarrier;
22  import org.apache.juneau.rest.server.tracing.*;

24  import io.opentelemetry.context.propagation.*;

26  /**
27   * {@link TextMapSetter} implementation that writes W3C trace-context fields into a neutral
28   * {@link TraceContextCarrier} instead of an HTTP response.
29   *
30   * <p>
31   * The adapter substrate for {@link OtelTracerHook#inject(TraceContextCarrier)}: the configured
32   * {@link TextMapPropagator} renders the caller's currently-active trace context (and baggage, when a
33   * baggage propagator is configured) by calling {@link #set(TraceContextCarrier, String, String)} once
34   * per key, which this class forwards to {@link TraceContextCarrier#set(String, String)}. A propagator
35   * with nothing valid to render (no active span context, empty baggage, etc.) simply never calls
36   * {@link #set(TraceContextCarrier, String, String)}, so the carrier is left untouched &mdash; the
37   * substrate for {@link TracerHook}'s documented default no-op {@code inject(...)} contract.
38   *
39   * @since 10.0.0
40   */
41  @SuppressWarnings({
42  	"java:S6548" // Singleton pattern is intentional; INSTANCE is a stateless, thread-safe TextMapSetter implementation.
43  })
44  public final class TraceContextCarrierTextMapSetter implements TextMapSetter<TraceContextCarrier> {
45  
46  	/** Process-wide singleton instance. */
47  	public static final TraceContextCarrierTextMapSetter INSTANCE = new TraceContextCarrierTextMapSetter();
48  
49  	private TraceContextCarrierTextMapSetter() {}
50  
51  	@Override /* TextMapSetter */
52  	public void set(TraceContextCarrier carrier, String key, String value) {
53  		if (nn(carrier) && nn(key) && nn(value))
54  			carrier.set(key, value);
55  	}
56  }
