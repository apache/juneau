17  package org.apache.juneau.rest.server.tracing;
18  
19  /**
20   * Handle to one in-flight {@link TracerHook}-managed span.
21   *
22   * <p>
23   * Returned by {@link TracerHook#startSpan(org.apache.juneau.rest.server.RestRequest)}. The framework drives
24   * the {@code setStatusCode} / {@code setError} / {@code close} transitions inside a {@code finally}
25   * block so the span always closes &mdash; including on exception paths.
26   *
27   * <p>
28   * Bridge implementations typically wrap the downstream tracer's native span / scope. The OpenTelemetry
29   * bridge in {@code juneau-rest-server-tracing-otel}, for example, wraps an
30   * {@code io.opentelemetry.api.trace.Span} plus its {@code io.opentelemetry.context.Scope}: setters
31   * translate to span attribute / status writes; {@code close()} ends the span and closes the OTel
32   * scope.
33   *
34   * @since 10.0.0
35   */
36  public interface Scope extends AutoCloseable {
37  
38  	/**
39  	 * Records the HTTP response status code on the span.
40  	 *
41  	 * <p>
42  	 * Called by the framework after the {@code @RestOp} handler completes &mdash; before
43  	 * {@link #close()}, and before {@link #setError(Throwable)} if the handler also threw. Bridges
44  	 * typically translate the status into both a numeric attribute (e.g. OpenTelemetry's
45  	 * {@code http.response.status_code}) and an overall span status (HTTP {@code 5xx} maps to
46  	 * span-status {@code ERROR}; other codes map to {@code UNSET} per the OTel HTTP semantic
47  	 * conventions).
48  	 *
49  	 * @param statusCode The HTTP status code (e.g. {@code 200}, {@code 404}, {@code 500}).
50  	 */
51  	void setStatusCode(int statusCode);
52  
53  	/**
54  	 * Records the exception (if any) the {@code @RestOp} handler threw on the span.
55  	 *
56  	 * <p>
57  	 * Called by the framework only when the handler threw &mdash; immediately before {@link #close()}.
58  	 * Bridges typically record the throwable as a span event with the stack trace, set the span status
59  	 * to {@code ERROR}, and add an {@code exception.type} attribute carrying the exception's
60  	 * simple-name.
61  	 *
62  	 * @param error The exception thrown by the handler. Never <jk>null</jk> when this method is called.
63  	 */
64  	void setError(Throwable error);
65  
66  	/**
67  	 * Closes the span.
68  	 *
69  	 * <p>
70  	 * Always called by the framework &mdash; via the standard {@code finally}-block contract on
71  	 * {@link AutoCloseable} &mdash; whether the handler returned normally or threw. Implementations
72  	 * <b>must not</b> throw checked exceptions; the close path runs in a {@code finally} where masking
73  	 * the original handler exception is unsafe. Implementations that need to surface bridge-level
74  	 * errors should log + swallow instead.
75  	 */
76  	@Override
77  	void close();
78  
79  	/**
80  	 * Records a JSON-RPC-level error outcome on this span.
81  	 *
82  	 * <p>
83  	 * A dated MCP adapter dispatches every request over HTTP {@code 200}, so an application-level
84  	 * JSON-RPC error (for example an unknown tool name) never reaches {@link #setStatusCode(int)}
85  	 * with anything but {@code 200} and never throws into {@link #setError(Throwable)}. This method
86  	 * gives such a caller a neutral way to observe that outcome on the span &mdash; called from the
87  	 * request-scoped tracing observation before a JSON-RPC error response is returned, distinct from
88  	 * both the HTTP status path and the thrown-exception path.
89  	 *
90  	 * <p>
91  	 * Added as a source-compatible default so every existing {@link Scope} implementation keeps
92  	 * compiling unchanged. The default implementation is a no-op; a bridge that wants to classify
93  	 * JSON-RPC errors on the span (for example the OpenTelemetry bridge in
94  	 * {@code juneau-rest-server-tracing-otel}) overrides it.
95  	 *
96  	 * @param code The JSON-RPC error code (e.g. {@code -32601}).
97  	 * @param message The JSON-RPC error message. May be <jk>null</jk>.
98  	 */
99  	default void recordRpcError(int code, String message) {
100 		// Intentionally empty; see method-level javadoc.
101 	}
102 }
