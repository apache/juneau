33  /**
34   * Flavor-neutral operational-introspection worker bean shared by the admin {@code @Rest} flavors.
35   *
36   * <p>
37   * Holds the capability state (the name-keyed cache-flush hooks + the thread-name-prefix exclude list)
38   * and the serve logic for the four admin endpoints (thread dump, heap statistics, cache flush, and
39   * rate-limit-bucket inspection). The {@link AdminMixin} (mixin), {@link AdminResource} (child), and
40   * {@link AdminServlet} (servlet) flavors are independent {@code @Rest} classes that each hold an
41   * {@code AdminProvider} worker and delegate to it &mdash; so the three deployment forms cannot drift, and no
42   * flavor is another flavor's worker.
43   *
44   * <h5 class='section'>See Also:</h5><ul>
45   * 	<li class='jc'>{@link AdminMixin}
46   * 	<li class='jc'>{@link AdminResource}
47   * 	<li class='jc'>{@link AdminServlet}
48   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestServerComposition">REST Server &mdash; Composition (mixins, paths)</a>
49   * </ul>
50   *
51   * @since 10.0.0
52   */
53  // @formatter:off

94  	/**
95  	 * Serves the thread dump as a JSON list of currently-live threads.
96  	 *
97  	 * @param res The current REST response. Must not be {@code null}.
98  	 * @throws IOException If an I/O error occurs while writing the response.
99  	 */
100 	@SuppressWarnings({
101 		"java:S1874" // Thread.getId() is not deprecated on the Java 17 baseline; its replacement Thread.threadId() was only added in Java 19, so getId() is the correct call here.
102 	})
103 	public void serveThreads(RestResponse res) throws IOException {
104 		var allTraces = Thread.getAllStackTraces();
105 		var out = new ArrayList<Map<String,Object>>();
106 		for (var e : allTraces.entrySet()) {
107 			var t = e.getKey();
108 			if (isExcludedThread(t.getName()))
109 				continue;
110 			var entry = new LinkedHashMap<String,Object>();
111 			entry.put("name", t.getName());
112 			entry.put("id", t.getId());
113 			entry.put("state", t.getState().toString());
114 			entry.put("daemon", t.isDaemon());
115 			entry.put("priority", t.getPriority());
116 			var stack = new ArrayList<String>();
117 			for (var f : e.getValue())
118 				stack.add(f.toString());
119 			entry.put("stack", stack);
120 			out.add(entry);
121 		}
122 		writeJson(res, out);
123 	}

125 	/**
126 	 * Serves JVM heap and non-heap memory statistics.
127 	 *
128 	 * @param res The current REST response. Must not be {@code null}.
129 	 * @throws IOException If an I/O error occurs while writing the response.
130 	 */
131 	public void serveHeap(RestResponse res) throws IOException {
132 		var rt = Runtime.getRuntime();
133 		var heap = new LinkedHashMap<String,Object>();
134 		heap.put("total", rt.totalMemory());
135 		heap.put("free", rt.freeMemory());
136 		heap.put("max", rt.maxMemory());
137 		heap.put("used", rt.totalMemory() - rt.freeMemory());
138 
139 		var mx = ManagementFactory.getMemoryMXBean();
140 		var nonHeapUsage = mx.getNonHeapMemoryUsage();
141 		var nonHeap = new LinkedHashMap<String,Object>();
142 		nonHeap.put("init", nonHeapUsage.getInit());
143 		nonHeap.put("used", nonHeapUsage.getUsed());
144 		nonHeap.put("committed", nonHeapUsage.getCommitted());
145 		nonHeap.put("max", nonHeapUsage.getMax());
146 
147 		var out = new LinkedHashMap<String,Object>();
148 		out.put("heap", heap);
149 		out.put("nonHeap", nonHeap);
150 		out.put("availableProcessors", rt.availableProcessors());
151 		writeJson(res, out);
152 	}

154 	/**
155 	 * Runs all registered cache-flush hooks (or a subset).
156 	 *
157 	 * <p>
158 	 * When {@code names} is supplied, only the named hooks are invoked. Unknown names are
159 	 * silently ignored (404-on-unknown would leak the registered hook set). Hook execution is
160 	 * synchronous; long-running hooks block the request thread.
161 	 *
162 	 * @param req The current REST request &mdash; {@code names} query parameter is read off it. Must not be {@code null}.
163 	 * @param res The current REST response. Must not be {@code null}.
164 	 * @throws IOException If an I/O error occurs while writing the response.
165 	 */
166 	public void serveCacheFlush(RestRequest req, RestResponse res) throws IOException {
167 		var namesParam = req.getQueryParams().get("names").asString().orElse(null);
168 		Set<String> selected = null;
169 		if (namesParam != null && ! namesParam.isBlank()) {
170 			selected = new LinkedHashSet<>();
171 			for (var n : namesParam.split(","))
172 				if (! n.isBlank())
173 					selected.add(n.trim());
174 		}
175 		var executed = new ArrayList<String>();
176 		for (var e : cacheFlushHooks.entrySet()) {
177 			if (selected != null && ! selected.contains(e.getKey()))
178 				continue;
179 			e.getValue().run();
180 			executed.add(e.getKey());
181 		}
182 		var out = new LinkedHashMap<String,Object>();
183 		out.put("registered", new ArrayList<>(cacheFlushHooks.keySet()));
184 		out.put("executed", executed);
185 		writeJson(res, out);
186 	}

188 	/**
189 	 * Serves the registered {@link RateLimitGuard} bean configuration.
190 	 *
191 	 * <p>
192 	 * Returns {@code 404 Not Found} when no {@code RateLimitGuard} bean is registered.
193 	 *
194 	 * @param req The current REST request &mdash; supplies the bean store for guard lookup. Must not be {@code null}.
195 	 * @param res The current REST response. Must not be {@code null}.
196 	 * @throws IOException If an I/O error occurs while writing the response.
197 	 * @throws NotFound If no {@link RateLimitGuard} bean is registered on the importer's bean store.
198 	 */
199 	@SuppressWarnings({
200 		"resource" // The context's WritableBeanStore is shared, long-lived infrastructure owned by RestContext; a per-request handler must never close it.
201 	})
202 	public void serveRateLimit(RestRequest req, RestResponse res) throws IOException {
203 		var bs = req.getContext().getBeanStore();
204 		var guards = collectRateLimitGuards(bs);
205 		if (guards.isEmpty())
206 			throw new NotFound("No RateLimitGuard bean is registered.");
207 		var entries = new LinkedHashMap<String,Object>();
208 		for (var e : guards.entrySet()) {
209 			var g = e.getValue();
210 			var bucket = new LinkedHashMap<String,Object>();
211 			bucket.put("config", describeRateLimitGuard(g));
212 			bucket.put("snapshot", g.snapshot().values().stream()
213 				.sorted(Comparator.comparing(BucketState::key))
214 				.toList());
215 			entries.put(e.getKey(), bucket);
216 		}
217 		var out = new LinkedHashMap<String,Object>();
218 		out.put("guards", entries);
219 		writeJson(res, out);
220 	}

240 	private boolean isExcludedThread(String threadName) {
241 		for (var prefix : threadNamePrefixExclude)
242 			if (threadName != null && threadName.startsWith(prefix))
243 				return true;
244 		return false;
245 	}
246 
247 	private static Map<String,RateLimitGuard> collectRateLimitGuards(BeanStore bs) {
248 		// Multi-bean lookup so users running per-tier guards (free / paid / etc.) see every
249 		// registered bean keyed by its bean name. Falls through to the single-bean path when no
250 		// per-tier configuration is registered.
251 		var byName = bs.getBeansOfType(RateLimitGuard.class);
252 		if (byName != null && ! byName.isEmpty())
253 			return new LinkedHashMap<>(byName);
254 		var out = new LinkedHashMap<String,RateLimitGuard>();
255 		bs.getBean(RateLimitGuard.class).ifPresent(g -> out.put("rateLimit", g));
256 		return out;
257 	}
258 
259 	private static Map<String,Object> describeRateLimitGuard(RateLimitGuard g) {
260 		var m = new LinkedHashMap<String,Object>();
261 		m.put("class", cn(g));
262 		m.put("limit", g.getCapacity());
263 		m.put("permitsPerSecond", g.getPermitsPerSecond());
264 		m.put("xForwardedForAware", g.isXForwardedForAware());
265 		m.put("exemptPaths", g.getExemptPaths().stream().sorted().toList());
266 		return m;
267 	}

269 	private static void writeJson(RestResponse res, Object payload) throws IOException {
270 		try (var w = res.getDirectWriter("application/json")) {
271 			JsonSerializer.DEFAULT_READABLE.write(payload, w);
272 		}

275 	/**
276 	 * Builder for {@link AdminProvider} instances.
277 	 */
278 	public static class Builder {
279 
280 		private final Map<String,Runnable> cacheFlushHooks = m();
281 		private final List<String> threadNamePrefixExclude = new ArrayList<>(DEFAULT_THREAD_NAME_PREFIX_EXCLUDE);
282 
283 		/** Constructor &mdash; package access for {@link AdminProvider#create()}. */
284 		protected Builder() {}
285 
286 		/**
287 		 * Registers a cache-flush hook under a name.
288 		 *
289 		 * <p>
290 		 * The hook is invoked synchronously when {@code POST /admin/cache/flush} is called (with
291 		 * no {@code names} parameter, or with a {@code names} parameter that includes this name).
292 		 *
293 		 * @param name The hook name. Must not be {@code null} or blank.
294 		 * @param hook The hook. Must not be {@code null}.
295 		 * @return This object.
296 		 */
297 		public Builder cacheFlush(String name, Runnable hook) {
298 			if (isBlank(name))
299 				throw new IllegalArgumentException("Argument 'name' must not be null or blank");
300 			if (hook == null)
301 				throw new IllegalArgumentException("Argument 'hook' must not be null");
302 			cacheFlushHooks.put(name, hook);
303 			return this;
304 		}
305 
306 		/**
307 		 * Registers multiple cache-flush hooks at once.
308 		 *
309 		 * @param hooks The hooks, keyed by registration name. Can be {@code null} (no-op).
310 		 * @return This object.
311 		 */
312 		public Builder cacheFlushAll(Map<String,Runnable> hooks) {
313 			if (hooks != null)
314 				hooks.forEach(this::cacheFlush);
315 			return this;
316 		}
317 
318 		/**
319 		 * Replaces the thread-name-prefix exclude list.
320 		 *
321 		 * <p>
322 		 * Threads whose {@link Thread#getName() name} starts with any of the supplied prefixes are
323 		 * omitted from the {@code /admin/threads} output. Pass an empty array to disable
324 		 * filtering entirely.
325 		 *
326 		 * @param values The thread-name prefixes to exclude. Can be {@code null} (equivalent to an empty
327 		 * 	array &mdash; disables filtering); {@code null} or empty elements are skipped.
328 		 * @return This object.
329 		 */
330 		public Builder threadNamePrefixExclude(String...values) {
331 			threadNamePrefixExclude.clear();
332 			if (values != null)
333 				for (var v : values)
334 					if (v != null && ! v.isEmpty())
335 						threadNamePrefixExclude.add(v);
336 			return this;
337 		}
338 
339 		/**
340 		 * Builds an {@link AdminProvider} instance.
341 		 *
342 		 * @return A configured instance.
343 		 */
344 		public AdminProvider build() {
345 			return new AdminProvider(this);
346 		}
