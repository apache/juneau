17  package org.apache.juneau.rest.server.ops;
18  
19  import java.io.*;
20  
21  import org.apache.juneau.http.response.*;
22  import org.apache.juneau.rest.server.*;
23  import org.apache.juneau.rest.server.guard.*;
24  import org.apache.juneau.rest.server.servlet.*;
25  
26  /**
27   * Standalone servlet companion of the {@link AdminMixin} mixin.
28   *
29   * <p>
30   * Mounts as a <b>sibling top-level servlet</b> at {@code /admin/*} and serves the same operational
31   * endpoints ({@code /threads}, {@code /heap}, {@code /cache/flush}, {@code /ratelimit}) as the
32   * mixin by delegating to a shared flavor-neutral {@link AdminProvider} worker bean &mdash; so the two forms
33   * cannot drift.
34   *
35   * <p>
36   * Whereas the {@link AdminMixin} mixin pins its ops at {@code /admin/<endpoint>} for
37   * composition into a host at {@code /}, this companion declares the mount at the class level
38   * ({@link Rest#paths() @Rest(paths="/admin/*")}) and pins the ops at the bare endpoint names
39   * (e.g. {@code /threads}) &mdash; eliminating the path-doubling that mixin-shaped ops would produce
40   * when the servlet itself is mounted at {@code /admin/*}.
41   *
42   * <h5 class='section'>Default-deny security posture:</h5>
43   *
44   * <p>
45   * Like the mixin, this servlet is annotated {@link Rest#guards() @Rest(guards=DenyAllGuard.class)},
46   * so every admin path returns {@code 403 Forbidden} until the importer registers a
47   * {@link RestGuardList} factory. Pair the servlet with whatever authentication / authorization
48   * story your service uses.
49   *
50   * <h5 class='figure'>Microservice registration:</h5>
51   *
52   * <p class='bjava'>
53   * 	<ja>@Bean</ja> Servlet admin() {
54   * 		<jk>return new</jk> AdminServlet();  <jc>// auto-mounted at /admin/* by JettyServerComponent</jc>
55   * 	}
56   * </p>
57   *
58   * <h5 class='section'>See Also:</h5><ul>
59   * 	<li class='jc'>{@link AdminMixin}
60   * 	<li class='jc'>{@link AdminProvider}
61   * 	<li class='jc'>{@link DenyAllGuard}
62   * 	<li class='jc'>{@link RestGuardList}
63   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestServerComposition">REST Server &mdash; Composition (mixins, paths)</a>
64   * </ul>
65   *
66   * @serial exclude
67   * @since 10.0.0
68   */
69  // @formatter:off
70  @Rest(
71  	paths="/admin/*",
72  	guards=DenyAllGuard.class
73  )
74  public class AdminServlet extends RestServlet {
75  
76  	private static final long serialVersionUID = 1L;
77  
78  	private final transient AdminProvider worker;
79  
80  	/** No-arg constructor &mdash; uses a default {@link AdminProvider} worker. */
81  	public AdminServlet() {
82  		this(new AdminProvider());
83  	}
84  
85  	/**
86  	 * Worker constructor.
87  	 *
88  	 * @param worker The shared flavor-neutral admin worker this servlet delegates to. Must not be
89  	 * 	{@code null}.
90  	 */
91  	protected AdminServlet(AdminProvider worker) {
92  		this.worker = worker;
93  	}
94  
95  	/**
96  	 * [GET /threads] &mdash; emit a JSON list of currently-live threads.
97  	 *
98  	 * @param res The current REST response.
99  	 * @throws IOException If an I/O error occurs while writing the response.
100 	 */
101 	@RestGet(
102 		path="/threads",
103 		summary="Thread dump",
104 		description="JSON list of currently-live threads (filtered to exclude framework noise by default).",
105 		swagger=@OpSwagger(ignore=true)
106 	)
107 	public void getThreads(RestResponse res) throws IOException {
108 		worker.serveThreads(res);
109 	}
110 
111 	/**
112 	 * [GET /heap] &mdash; emit JVM heap and non-heap memory statistics.
113 	 *
114 	 * @param res The current REST response.
115 	 * @throws IOException If an I/O error occurs while writing the response.
116 	 */
117 	@RestGet(
118 		path="/heap",
119 		summary="Heap statistics",
120 		description="JVM heap + non-heap memory statistics (Runtime + MemoryMXBean).",
121 		swagger=@OpSwagger(ignore=true)
122 	)
123 	public void getHeap(RestResponse res) throws IOException {
124 		worker.serveHeap(res);
125 	}
126 
127 	/**
128 	 * [POST /cache/flush] &mdash; run all registered cache-flush hooks (or a subset).
129 	 *
130 	 * @param req The current REST request &mdash; {@code names} query parameter is read off it.
131 	 * @param res The current REST response.
132 	 * @throws IOException If an I/O error occurs while writing the response.
133 	 */
134 	@RestPost(
135 		path="/cache/flush",
136 		summary="Cache flush",
137 		description="Runs the registered cache-flush hooks (all by default; ?names=foo,bar for a subset).",
138 		swagger=@OpSwagger(ignore=true)
139 	)
140 	public void postCacheFlush(RestRequest req, RestResponse res) throws IOException {
141 		worker.serveCacheFlush(req, res);
142 	}
143 
144 	/**
145 	 * [GET /ratelimit] &mdash; emit registered {@link RateLimitGuard} bean configuration.
146 	 *
147 	 * @param req The current REST request &mdash; supplies the bean store for guard lookup.
148 	 * @param res The current REST response.
149 	 * @throws IOException If an I/O error occurs while writing the response.
150 	 * @throws NotFound If no {@link RateLimitGuard} bean is registered on the importer's bean
151 	 * 	store.
152 	 */
153 	@RestGet(
154 		path="/ratelimit",
155 		summary="Rate-limit inspection",
156 		description="JSON map of registered RateLimitGuard beans keyed by bean name.",
157 		swagger=@OpSwagger(ignore=true)
158 	)
159 	public void getRateLimit(RestRequest req, RestResponse res) throws IOException {
160 		worker.serveRateLimit(req, res);
161 	}
162 }
