21  import org.apache.juneau.http.*;
22  import org.apache.juneau.http.response.*;
23  import org.apache.juneau.rest.server.*;
24  import org.apache.juneau.rest.server.servlet.*;
25  import org.apache.juneau.rest.server.staticfile.*;

27  /**
28   * Mixin providing the residual {@code [GET /htdocs/*]} static-file endpoint.
29   *
30   * <p>
31   * Single-responsibility op-mixin carved out of the former {@code BasicRestOperations} interface. Delegates the
32   * requested path to {@link StaticFilesMixin#resolveStaticFile(RestRequest,String,Locale)} &mdash; the same
33   * resolution chokepoint used by the newer {@link StaticFilesMixin}'s configurable-mount handlers &mdash; so
34   * there is a single implementation of the static-file lookup logic even though this mixin keeps its own fixed,
35   * {@code GET}-only {@code /htdocs/*} mount for backward compatibility.
36   *
37   * <p>
38   * The greedy {@code /htdocs/*} blob handler is not API-meaningful and is excluded from the published
39   * Swagger/OpenAPI specification via {@link OpSwagger#ignore() @OpSwagger(ignore=true)}.
40   *
41   * <h5 class='section'>See Also:</h5><ul>
42   * 	<li class='jc'>{@link StaticFilesMixin}
43   * </ul>
44   *
45   * @since 10.0.0
46   */
47  @Rest
48  public class HtdocMixin extends RestMixin {
49  
50  	/**
51  	 * [GET /htdocs/*] - Retrieve static file.
52  	 *
53  	 * @param req The HTTP request.
54  	 * @param path The path to retrieve.
55  	 * @param locale The locale of the HTTP request.
56  	 * @return An HTTP resource representing the static file.
57  	 * @throws NotFound If no resource matches the requested path.
58  	 */
59  	@RestGet(
60  		path="/htdocs/*",
61  		summary="Static files",
62  		description="Static file retrieval.",
63  		swagger=@OpSwagger(ignore=true)
64  	)
65  	public HttpResource getHtdoc(RestRequest req, @Path("/*") String path, Locale locale) throws NotFound {
66  		return StaticFilesMixin.resolveStaticFile(req, path, locale);
67  	}
68  }
