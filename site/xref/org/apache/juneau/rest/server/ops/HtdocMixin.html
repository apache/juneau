17  package org.apache.juneau.rest.server.ops;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.http.*;
22  import org.apache.juneau.http.response.*;
23  import org.apache.juneau.rest.server.*;
24  import org.apache.juneau.rest.server.servlet.*;
25  
26  /**
27   * Mixin providing the residual {@code [GET /htdocs/*]} static-file endpoint.
28   *
29   * <p>
30   * Single-responsibility op-mixin carved out of the former {@code BasicRestOperations} interface. Resolves
31   * the requested path against the active {@link org.apache.juneau.rest.server.staticfile.StaticFiles} bean via
32   * {@link RestContext#getStaticFiles()}.
33   *
34   * <p>
35   * The greedy {@code /htdocs/*} blob handler is not API-meaningful and is excluded from the published
36   * Swagger/OpenAPI specification via {@link OpSwagger#ignore() @OpSwagger(ignore=true)}.
37   *
38   * <h5 class='section'>See Also:</h5><ul>
39   * 	<li class='jc'>{@link org.apache.juneau.rest.server.staticfile.StaticFilesMixin}
40   * </ul>
41   *
42   * @since 10.0.0
43   */
44  @Rest
45  public class HtdocMixin extends RestMixin {
46  
47  	/**
48  	 * [GET /htdocs/*] - Retrieve static file.
49  	 *
50  	 * @param req The HTTP request.
51  	 * @param path The path to retrieve.
52  	 * @param locale The locale of the HTTP request.
53  	 * @return An HTTP resource representing the static file.
54  	 * @throws NotFound If no resource matches the requested path.
55  	 */
56  	@RestGet(
57  		path="/htdocs/*",
58  		summary="Static files",
59  		description="Static file retrieval.",
60  		swagger=@OpSwagger(ignore=true)
61  	)
62  	public HttpResource getHtdoc(RestRequest req, @Path("/*") String path, Locale locale) throws NotFound {
63  		return req.getContext().getStaticFiles().resolve(path, locale).orElseThrow(NotFound::new);
64  	}
65  }
