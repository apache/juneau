31  /**
32   * Mixin that serves a request-echo / round-trip introspection endpoint at {@code /echo/*}
33   * (configurable).
34   *
35   * <p>
36   * Sibling of {@link AdminMixin} ({@code /admin/*}) and {@link RouteIndexMixin}
37   * ({@code /options}). All three classes live in the
38   * {@code org.apache.juneau.rest.server.ops} ops/introspection mixin pack.
39   *
40   * <p>
41   * Compose into a host resource via
42   * {@link Rest#mixins() @Rest(mixins=EchoMixin.class)}; the {@code /echo/*} URL becomes
43   * available alongside the host's own endpoints with no further wiring.
44   *
45   * <h5 class='section'>Configurable mount path:</h5>
46   *
47   * <p>
48   * The default mount {@code /echo/*} can be overridden via the SVL variable
49   * {@code ${juneau.echo.path:echo}} &mdash; set via system property
50   * ({@code -Djuneau.echo.path=introspect}), environment variable
51   * ({@code JUNEAU_ECHO_PATH=introspect}), or {@code Config} key
52   * ({@code juneau.echo.path = introspect}) to change the runtime mount without subclassing.
53   * Resolution happens once at {@link RestContext} construction time (SVL resolution in {@code @RestOp(path)}).
54   *
55   * <p>
56   * Override accepts bare token ({@code echo}), leading slash ({@code /echo}), trailing slash
57   * ({@code echo/}), or wildcard suffix ({@code /echo/*}) &mdash; all resolve to the same mount.
58   *
59   * <p>
60   * <b>Migration note (10.0.0):</b> Earlier development snapshots of this mixin mounted at both
61   * {@code /echo/*} <i>and</i> {@code /debug/echo/*} as historical aliases on a single op. That
62   * dual default has been collapsed to a single SVL-configurable mount as part of the
63   * "single path per op" principle. Deployers who relied on the
64   * {@code /debug/echo/*} alias must now either set {@code -Djuneau.echo.path=debug/echo} or
65   * compose a second instance with the override.
66   *
67   * <h5 class='section'>Mixin-only deployment:</h5>
68   *
69   * <p>
70   * This resource is designed for composition via {@code @Rest(mixins=...)}. The mount path is
71   * pinned at the op level by {@link RestOp @RestOp(path="/&#123;juneau.echo.path:echo&#125;/*")} on
72   * {@link #echo}; a class-level {@code @Rest(paths=...)} declaration would be silently ignored
73   * under the mixin pattern (see {@link Rest#paths() @Rest(paths)} Javadoc).
74   *
75   * <h5 class='figure'>Composition example:</h5>
76   *
77   * <p class='bjava'>
78   * 	<ja>@Rest</ja>(
79   * 		path=<js>"/api"</js>,
80   * 		mixins=EchoMixin.<jk>class</jk>,
81   * 		debug=<js>"conditional"</js>          <jc>// gates the echo endpoint per request</jc>
82   * 	)
83   * 	<jk>public class</jk> ApiResource <jk>extends</jk> RestServlet { }
84   *
85   * 	<jc>// Optional: tighten the body cap or adjust redacted headers via a @Bean factory:</jc>
86   * 	<ja>@Bean</ja> EchoMixin echo() {
87   * 		<jk>return</jk> EchoMixin.<jsm>create</jsm>()
88   * 			.bodyLimit(64 * 1024L)
89   * 			.redactHeader(<js>"X-Internal-Trace"</js>)
90   * 			.build();
91   * 	}
92   * </p>
93   *
94   * <h5 class='section'>Debug gating:</h5>
95   *
96   * <p>
97   * The handler is gated behind the host's
98   * {@link RestContext#getDebugEnablement() DebugEnablement} chain &mdash; the same mechanism that
99   * powers {@code @Rest(debug=...)}. When {@code Debug} resolves to {@code OFF} for the current
100  * request, the handler returns {@code 404 Not Found} (so the existence of the endpoint isn't
101  * disclosed). When debug is {@code ALWAYS}, or {@code CONDITIONAL} with the {@code Debug: true}
102  * request header, the full echo payload is returned. The recommended posture for production
103  * deployments is {@code @Rest(debug="conditional")} paired with a guard chain so only authorized
104  * operators can flip the {@code Debug} header.
105  *
106  * <h5 class='section'>Sensitive-header redaction:</h5>
107  *
108  * <p>
109  * Token-bearing headers must never be reflected back to the caller (would defeat any auth scheme
110  * in front of the endpoint). The default redacted list is &mdash; case-insensitively &mdash;
111  * {@code Authorization}, {@code Cookie}, {@code Set-Cookie}, {@code Proxy-Authorization}, and
112  * {@code X-API-Key}. Each redacted header surfaces in the echo body with the literal value
113  * {@value #REDACTED} so the caller can see the header was present without leaking its value.
114  * Override the default list via {@link Builder#redactedHeaders(String...)} (replaces) or extend
115  * via {@link Builder#redactHeader(String)} (additive).
116  *
117  * <p>
118  * {@link #REDACTED} and {@link #DEFAULT_REDACTED_HEADERS} are re-exports of
119  * {@link RedactedHeaders#REDACTED} and {@link RedactedHeaders#DEFAULT} respectively; masking is
120  * performed by delegating to {@link RedactedHeaders#redact(String, String, java.util.Collection)}.
121  *
122  * <h5 class='section'>Body capture and truncation:</h5>
123  *
124  * <p>
125  * The handler reads up to {@link Builder#bodyLimit(long) bodyLimit} bytes (default
126  * {@value #DEFAULT_BODY_LIMIT_DOC} = 1 MB) of the inbound body and emits it as a UTF-8 string in
127  * the {@code content} field of the JSON response. When the body exceeds the cap, the captured
128  * portion is truncated and the {@code truncated} flag is set to {@code true} so callers can see
129  * the response is incomplete.
130  *
131  * <h5 class='section'>Response shape:</h5>
132  *
133  * <p class='bjson'>
134  * 	{
135  * 		<jok>"method"</jok>: <jov>"POST"</jov>,
136  * 		<jok>"path"</jok>: <jov>"/echo/foo/bar"</jov>,
137  * 		<jok>"queryString"</jok>: <jov>"x=1"</jov>,
138  * 		<jok>"pathRemainder"</jok>: <jov>"foo/bar"</jov>,
139  * 		<jok>"headers"</jok>: { <jok>"User-Agent"</jok>: <jov>"curl"</jov>, <jok>"Authorization"</jok>: <jov>"[REDACTED]"</jov> },
140  * 		<jok>"queryParams"</jok>: { <jok>"x"</jok>: <jov>"1"</jov> },
141  * 		<jok>"attributes"</jok>: { },
142  * 		<jok>"contentLength"</jok>: <jov>5</jov>,
143  * 		<jok>"content"</jok>: <jov>"hello"</jov>,
144  * 		<jok>"truncated"</jok>: <jov>false</jov>
145  * 	}
146  * </p>
147  *
148  * <p>
149  * The endpoint is excluded from generated Swagger / OpenAPI specs via
150  * {@link OpSwagger#ignore() @OpSwagger(ignore=true)}.
151  *
152  * <h5 class='section'>See Also:</h5><ul>
153  * 	<li class='jc'>{@link AdminMixin}
154  * 	<li class='jc'>{@link RouteIndexMixin}
155  * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestServerComposition">REST Server &mdash; Composition (mixins, paths)</a>
156  * </ul>
157  *
158  * @since 10.0.0
159  */

194 	/**
195 	 * Builder constructor.
196 	 *
197 	 * @param builder The builder. Must not be {@code null}.
198 	 */
199 	protected EchoMixin(Builder builder) {
200 		bodyLimit = builder.bodyLimit;
201 		redactedHeaders = u(new LinkedHashSet<>(builder.redactedHeaders));
202 		var s = new LinkedHashSet<String>();
203 		for (var h : builder.redactedHeaders)
204 			s.add(h.toLowerCase(Locale.ROOT));
205 		redactedHeadersLower = u(s);
206 	}

208 	/**
209 	 * [* /echo/*] &mdash; emit an introspection echo of the inbound request.
210 	 *
211 	 * <p>
212 	 * Returns {@code 404 Not Found} when {@code Debug} is not enabled for the current request.
213 	 *
214 	 * @param req The current REST request.
215 	 * @param res The current REST response.
216 	 * @param remainder The path remainder after the mount prefix (multi-segment, may be empty).
217 	 * @throws IOException If an I/O error occurs while reading the request or writing the response.
218 	 * @throws NotFound When {@code Debug} resolves to {@code OFF} for the current request.
219 	 */
220 	@RestOp(
221 		method="*",
222 		path="/#{pathToken(${juneau.echo.path:echo})}/*",
223 		summary="Request echo",
224 		description="Round-trip introspection of the inbound request. Debug-gated.",
225 		swagger=@OpSwagger(ignore=true)
226 	)
227 	@SuppressWarnings({
228 		"resource" // sreq.getInputStream() is the servlet container's request body stream; it must not be closed by application code (lifecycle is tied to the request, not this handler).
229 	})
230 	public void echo(RestRequest req, RestResponse res, @Path("/*") String remainder) throws IOException {
231 		var ctx = req.getContext();
232 		var de = ctx.getDebugEnablement();
233 		var sreq = req.getHttpServletRequest();
234 		if (de == null || ! de.isDebug(ctx, sreq))
235 			throw new NotFound("Echo endpoint disabled (Debug not enabled).");
236 
237 		var headers = new LinkedHashMap<String,String>();
238 		var names = sreq.getHeaderNames();
239 		while (names != null && names.hasMoreElements()) {
240 			var name = names.nextElement();
241 			headers.put(name, RedactedHeaders.redact(name, sreq.getHeader(name), redactedHeaders));
242 		}
243 
244 		var queryParams = new LinkedHashMap<String,String>();
245 		for (var p : req.getQueryParams())
246 			queryParams.put(p.getName(), p.asString().orElse(null));
247 
248 		var attributes = new LinkedHashMap<String,String>();
249 		req.getAttributes().asMap().forEach((k, v) -> attributes.put(k, String.valueOf(v)));
250 
251 		var capture = readBoundedBody(sreq.getInputStream(), bodyLimit);
252 
253 		var out = new LinkedHashMap<String,Object>();
254 		out.put("method", req.getMethod());
255 		out.put("path", sreq.getRequestURI());
256 		out.put("queryString", sreq.getQueryString());
257 		out.put("pathRemainder", remainder == null ? "" : remainder);
258 		out.put("headers", headers);
259 		out.put("queryParams", queryParams);
260 		out.put("attributes", attributes);
261 		out.put("contentLength", capture.bytesRead);
262 		if (capture.bytesRead > 0)
263 			out.put("content", new String(capture.bytes, StandardCharsets.UTF_8));
264 		out.put("truncated", capture.truncated);
265 
266 		try (var w = res.getDirectWriter("application/json")) {
267 			JsonSerializer.DEFAULT_READABLE.write(out, w);
268 		}
269 	}

289 	@SuppressWarnings({
290 		"java:S135" // The two breaks are distinct truncation exits (limit-reached vs. partial-final-chunk); merging them would obscure the bounded-read logic.
291 	})
292 	private static BodyCapture readBoundedBody(InputStream in, long limit) throws IOException {
293 		if (in == null)
294 			return new BodyCapture(new byte[0], 0, false);
295 		var buf = new ByteArrayOutputStream();
296 		var chunk = new byte[8192];
297 		var truncated = false;
298 		long total = 0;
299 		int n;
300 		while ((n = in.read(chunk)) != -1) {
301 			var room = limit - total;
302 			if (room <= 0) {
303 				truncated = true;
304 				break;
305 			}
306 			if (n > room) {
307 				buf.write(chunk, 0, (int) room);
308 				total += room;
309 				truncated = true;
310 				break;
311 			}
312 			buf.write(chunk, 0, n);
313 			total += n;
314 		}
315 		return new BodyCapture(buf.toByteArray(), total, truncated);
316 	}
317 
318 	private static final class BodyCapture {
319 		final byte[] bytes;
320 		final long bytesRead;
321 		final boolean truncated;
322 
323 		BodyCapture(byte[] bytes, long bytesRead, boolean truncated) {
324 			this.bytes = bytes;
325 			this.bytesRead = bytesRead;
326 			this.truncated = truncated;
327 		}

330 	/**
331 	 * Builder for {@link EchoMixin} instances.
332 	 */
333 	public static class Builder {
334 
335 		private long bodyLimit = DEFAULT_BODY_LIMIT;
336 		private final Set<String> redactedHeaders;
337 
338 		/** Constructor &mdash; package access for {@link EchoMixin#create()}. */
339 		protected Builder() {
340 			redactedHeaders = new LinkedHashSet<>(DEFAULT_REDACTED_HEADERS);
341 		}
342 
343 		/**
344 		 * Sets the body capture cap.
345 		 *
346 		 * <p>
347 		 * Captured content is truncated when the inbound body exceeds this size; the
348 		 * {@code truncated} flag in the response is set to {@code true}.
349 		 *
350 		 * @param value Cap, in bytes. Must be {@code >= 0}.
351 		 * @return This object.
352 		 */
353 		public Builder bodyLimit(long value) {
354 			if (value < 0)
355 				throw new IllegalArgumentException("bodyLimit must be >= 0");
356 			bodyLimit = value;
357 			return this;
358 		}
359 
360 		/**
361 		 * Replaces the redacted-header set with the supplied values.
362 		 *
363 		 * <p>
364 		 * Header names are matched case-insensitively. Pass an empty array to disable redaction
365 		 * (not recommended outside of integration tests).
366 		 *
367 		 * @param values The header names to redact. Can be {@code null} (equivalent to an empty array
368 		 * 	&mdash; disables redaction); {@code null} or blank elements are skipped.
369 		 * @return This object.
370 		 */
371 		public Builder redactedHeaders(String...values) {
372 			redactedHeaders.clear();
373 			if (values != null)
374 				for (var v : values)
375 					if (v != null && ! v.isBlank())
376 						redactedHeaders.add(v);
377 			return this;
378 		}
379 
380 		/**
381 		 * Adds an additional header name to the redacted-header set.
382 		 *
383 		 * @param value The header name to redact (case-insensitive). Must not be {@code null} or
384 		 * 	blank.
385 		 * @return This object.
386 		 */
387 		public Builder redactHeader(String value) {
388 			if (isBlank(value))
389 				throw new IllegalArgumentException("Argument 'value' must not be null or blank");
390 			redactedHeaders.add(value);
391 			return this;
392 		}
393 
394 		/**
395 		 * Builds a {@link EchoMixin} instance.
396 		 *
397 		 * @return A configured instance.
398 		 */
399 		public EchoMixin build() {
400 			return new EchoMixin(this);
401 		}
