17  package org.apache.juneau.rest.server.bus;
18  
19  import java.time.*;
20  import java.util.*;
21  import java.util.concurrent.atomic.*;
22  
23  import org.apache.juneau.marshall.sse.*;
24  import org.apache.juneau.rest.server.sse.*;
25  
26  /**
27   * The SSE transport's {@link FrameSink}: an adapter over the session's {@link SseSubscription} (from
28   * {@link ServerBus#subscribeSse(BusSession)}), drained by {@link BusEventsMixin#streamBus} on the request thread.
29   *
30   * <p>
31   * Each frame becomes one {@code event: bus} {@link SseEvent}, with the frame's {@code seq} (when it has one) as its
32   * id.  The sink refuses a frame once {@code max} are pending, so the bus closes it as a slow consumer.  The
33   * subscription's queue is one larger than {@code max}, so the subscription's own drop-oldest overflow never runs and a
34   * retained update is never silently lost.
35   *
36   * <p>
37   * {@code 4429} (slow consumer), {@code 4409} (replaced) and {@link #release()} close the subscription, which drops
38   * the backlog.  After {@code 4429} the stream writes one {@code error bus:slow-consumer} frame as its last event.
39   * {@code 1001} (bus closed) and {@code 4401} (expired) let pending frames drain first.  When another stream
40   * re-subscribes the session (one connection per session), this sink ends quietly, as if replaced.
41   */
42  final class SseFrameSink implements FrameSink {
43  
44  	/** Ends a draining close.  Compared by identity and never written. */
45  	private static final SseEvent END = new SseEvent(BusFrames.EVENT, "");
46  
47  	private static final String SLOW_CONSUMER = BusFrames.error("bus:slow-consumer", "the connection fell behind", null);
48  
49  	private final SseSubscription subscription;
50  	private final int max;
51  	private final AtomicInteger pending = new AtomicInteger();
52  	private final AtomicInteger closeCode = new AtomicInteger();
53  	private boolean done;   // Consumer thread only.
54  
55  	/**
56  	 * @param subscription The session's subscription.  Its queue must hold at least {@code max + 1} events.
57  	 * @param max The most frames that may be pending ({@link BusPolicy#maxQueuedFrames()}).
58  	 */
59  	SseFrameSink(SseSubscription subscription, int max) {
60  		this.subscription = Objects.requireNonNull(subscription, "subscription");
61  		this.max = max;
62  	}
63  
64  	@Override /* FrameSink */
65  	public boolean offer(String frame) {
66  		if (closeCode.get() != 0)
67  			return false;
68  		if (subscription.isClosed()) {
69  			// Another stream re-subscribed this session: end quietly, as a replaced connection does.
70  			closeCode.compareAndSet(0, 4409);
71  			return false;
72  		}
73  		if (pending.incrementAndGet() > max) {
74  			pending.decrementAndGet();
75  			return false;
76  		}
77  		subscription.offer(toEvent(frame));
78  		return true;
79  	}
80  
81  	@Override /* FrameSink */
82  	public void close(int code, String reason) {
83  		if (! closeCode.compareAndSet(0, code))
84  			return;
85  		if (code == 1001 || code == 4401)
86  			subscription.offer(END);
87  		else
88  			subscription.close();
89  	}
90  
91  	/**
92  	 * Waits for the next event to write.
93  	 *
94  	 * @param timeout How long to wait.
95  	 * @return The event, or <jk>null</jk> on timeout or once the sink has ended (see {@link #isDone()}).  After a
96  	 * 	{@code 4429} close, the one {@code error bus:slow-consumer} event comes first.
97  	 * @throws InterruptedException If interrupted while waiting.
98  	 */
99  	SseEvent poll(Duration timeout) throws InterruptedException {
100 		if (done)
101 			return null;
102 		var e = subscription.poll(timeout);
103 		if (e != null && e != END) {
104 			pending.decrementAndGet();
105 			return e;
106 		}
107 		if (e == null && ! subscription.isClosed())
108 			return null;
109 		done = true;
110 		subscription.close();
111 		return closeCode.get() == 4429 ? toEvent(SLOW_CONSUMER) : null;
112 	}
113 
114 	/** @return <jk>true</jk> once the sink has ended and its last event has been taken. */
115 	boolean isDone() {
116 		return done;
117 	}
118 
119 	/** Ends the sink from the stream's side (the response ended).  Safe to call more than once. */
120 	void release() {
121 		closeCode.compareAndSet(0, 1000);
122 		subscription.close();
123 	}
124 
125 	private static SseEvent toEvent(String frame) {
126 		var event = new SseEvent(BusFrames.EVENT, frame);
127 		var seq = BusFrames.seq(frame);
128 		if (seq.isPresent())
129 			event.setId(String.valueOf(seq.getAsLong()));
130 		return event;
131 	}
132 }
