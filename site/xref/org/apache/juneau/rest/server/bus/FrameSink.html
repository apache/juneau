17  package org.apache.juneau.rest.server.bus;
18  
19  /**
20   * A transport's outbound side, handed to {@link ServerBus#connect(BusSession, FrameSink)} for one connection.
21   *
22   * <p>
23   * {@link #offer(String)} must not block: it queues the frame and returns at once, or returns <jk>false</jk> when the
24   * connection's queue (bounded by {@link BusPolicy#maxQueuedFrames()}) is full.  On <jk>false</jk> the bus detaches
25   * the sink and calls {@link #close(int, String)} with {@code 4429} / {@code bus:slow-consumer}.
26   *
27   * <p>
28   * {@link #close(int, String)} must not block either: the bus calls it while holding its lock.  A transport that needs a
29   * network handshake to close must hand the close off to another thread.
30   *
31   * <p>
32   * Close codes the bus uses: {@code 1001} {@code bus:closed} ({@link ServerBus#close()}), {@code 4401}
33   * {@code bus:unknown-session} (expired), {@code 4409} {@code bus:replaced} (a second connection took over the session)
34   * and {@code 4429} {@code bus:slow-consumer}.  A transport without close codes (SSE) ends its response instead.
35   *
36   * <h5 class='section'>Example:</h5>
37   * <p class='bjava'>
38   *   <jc>// A test double that records what the bus sends:</jc>
39   *   List&lt;String&gt; <jv>frames</jv> = <jk>new</jk> ArrayList&lt;&gt;();
40   *   FrameSink <jv>sink</jv> = <jk>new</jk> FrameSink() {
41   *     <ja>@Override</ja> <jk>public boolean</jk> offer(String <jv>frame</jv>) { <jk>return</jk> <jv>frames</jv>.add(<jv>frame</jv>); }
42   *     <ja>@Override</ja> <jk>public void</jk> close(<jk>int</jk> <jv>code</jv>, String <jv>reason</jv>) { }
43   *   };
44   *   <jv>bus</jv>.connect(<jv>session</jv>, <jv>sink</jv>);   <jc>// resync-begin, retained pubs, resync-end</jc>
45   * </p>
46   *
47   * @since 10.0.0
48   */
49  public interface FrameSink {
50  
51  	/**
52  	 * Queues one encoded frame for sending.  Must not block.
53  	 *
54  	 * @param frame The encoded frame (see {@link BusFrames}).
55  	 * @return <jk>false</jk> when the connection's queue is full or the sink is closed.
56  	 */
57  	boolean offer(String frame);
58  
59  	/**
60  	 * Closes the connection.  Called at most once by the bus for a given sink; must be safe to call again.
61  	 *
62  	 * @param code The close code ({@code 1001}, {@code 4401}, {@code 4409} or {@code 4429}).
63  	 * @param reason The {@code bus:*} reason.
64  	 */
65  	void close(int code, String reason);
66  }
