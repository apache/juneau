26  /**
27   * The bus wire format, version 1: one JSON object per frame, in a canonical field order shared with the
28   * browser codec and pinned by {@code bus-frames-corpus.json}.
29   *
30   * <p>
31   * The SSE transport writes each frame as {@code event: bus}, {@code id: <seq>}, {@code data: <frame>}; the WebSocket
32   * transport sends one frame per text message.
33   *
34   * <h5 class='section'>Example:</h5>
35   * <p class='bjava'>
36   *   String <jv>f</jv> = BusFrames.<jsm>pub</jsm>(<js>"ops.jobs"</js>, Json.<jsm>of</jsm>(<jv>payload</jv>), <jk>true</jk>, 7);
37   *   <jc>// {"v":1,"type":"pub","topic":"ops.jobs","payload":{...},"retained":true,"seq":7}</jc>
38   *
39   *   JsonMap <jv>up</jv> = BusFrames.<jsm>decode</jsm>(<jv>text</jv>);   <jc>// BusRefusal bus:bad-frame when malformed</jc>
40   * </p>
41   *
42   * @since 10.0.0
43   */
44  @SuppressWarnings({
45  	"java:S127", // utf8Length() and quote() skip the low surrogate of a pair inside the loop body.
46  	"java:S1192", // Frame field names and types are wire-format literals; named constants would not clarify them.
47  	"java:S3776" // Branching is inherent to decode() and isSingleObject().
48  })
49  public final class BusFrames {
50  
51  	/** The wire format version. */
52  	public static final int VERSION = 1;
53  
54  	/** The SSE event name carrying frames. */
55  	public static final String EVENT = "bus";
56  
57  	/** The frame types of version 1. */
58  	public static final Set<String> TYPES = Set.of("pub", "clear", "resync-begin", "resync-end", "ping", "error");
59  
60  	/** The largest {@code seq} both codecs represent exactly: {@code Number.MAX_SAFE_INTEGER}. */
61  	private static final long MAX_SAFE_SEQ = 9007199254740991L;
62  
63  	private static final Pattern TRAILING_SEQ = Pattern.compile("\"seq\":(\\d+)}$");
64  
65  	private BusFrames() {}

67  	/**
68  	 * A {@code pub} frame.
69  	 *
70  	 * @param topic The topic.
71  	 * @param payloadJson The payload, already serialized as JSON.
72  	 * @param retained Whether the value is retained.
73  	 * @param seq The session sequence number.
74  	 * @return The encoded frame.
75  	 */
76  	public static String pub(String topic, String payloadJson, boolean retained, long seq) {
77  		return pubRaw(topic, payloadJson, retained, seq);
78  	}

81  	 * A {@code clear} frame.
82  	 *
83  	 * @param topic The retained topic being cleared.
84  	 * @param seq The session sequence number.
85  	 * @return The encoded frame.
86  	 */
87  	public static String clear(String topic, long seq) {
88  		return head("clear").append(",\"topic\":").append(quote(topic)).append(",\"seq\":").append(seq).append('}').toString();
89  	}

102 	public static String ping() {
103 		return head("ping").append('}').toString();
104 	}

106 	/**
107 	 * An {@code error} frame.
108 	 *
109 	 * @param code The {@code bus:*} code.
110 	 * @param message Fixed server text.
111 	 * @param topic The topic concerned.  Can be <jk>null</jk>.
112 	 * @return The encoded frame.
113 	 */
114 	public static String error(String code, String message, String topic) {
115 		var sb = head("error").append(",\"code\":").append(quote(code)).append(",\"message\":").append(quote(message));
116 		if (topic != null)
117 			sb.append(",\"topic\":").append(quote(topic));
118 		return sb.append('}').toString();
119 	}

121 	/**
122 	 * Encodes a frame given as a map, in canonical field order, omitting absent optional fields.
123 	 *
124 	 * @param frame The frame ({@code type} required; {@code v} is always written as {@code 1}).
125 	 * @return The encoded frame.
126 	 * @throws IllegalArgumentException On an unknown {@code type}, or when {@code topic}, {@code code} or {@code message}
127 	 * 	is missing or not a string, or {@code retained} is not a boolean.
128 	 */
129 	public static String encode(Map<String,?> frame) {
130 		var type = String.valueOf(frame.get("type"));
131 		return switch (type) {
132 			case "pub" -> pubRaw(requireString(frame, "topic"), Json.of(frame.get("payload")), optionalBoolean(frame, "retained"),
133 				seqOf(frame.get("seq")));
134 			case "clear" -> {
135 				var sb = head("clear").append(",\"topic\":").append(quote(requireString(frame, "topic")));
136 				var seq = seqOf(frame.get("seq"));
137 				if (seq != null)
138 					sb.append(",\"seq\":").append(seq);
139 				yield sb.append('}').toString();
140 			}
141 			case "resync-begin", "resync-end" -> {
142 				var sb = head(type);
143 				var seq = seqOf(frame.get("seq"));
144 				if (seq != null)
145 					sb.append(",\"seq\":").append(seq);
146 				yield sb.append('}').toString();
147 			}
148 			case "ping" -> ping();
149 			case "error" -> error(requireString(frame, "code"), requireString(frame, "message"), optionalString(frame, "topic"));
150 			default -> throw new IllegalArgumentException("BusFrames.encode: unknown frame type '" + type + "'");
151 		};
152 	}
153 
154 	/**
155 	 * Decodes and validates one frame.  The result holds only the known fields of its type, in canonical order.
156 	 *
157 	 * @param text The frame text (strict RFC 8259 JSON).
158 	 * @return The frame.
159 	 * @throws BusRefusal {@code bus:bad-frame} when the text is not a valid version-1 frame.
160 	 */
161 	public static JsonMap decode(String text) throws BusRefusal {
162 		JsonMap m;
163 		try {
164 			m = text == null || ! isSingleObject(text) ? null : JsonMap.ofString(text, JsonParser.DEFAULT);
165 		} catch (RuntimeException e) {
166 			throw BusRefusal.badFrame("frame is not a JSON object");
167 		}
168 		if (m == null)
169 			throw BusRefusal.badFrame("frame is not a JSON object");
170 		if (! (m.get("v") instanceof Number v) || v.doubleValue() != VERSION)
171 			throw BusRefusal.badFrame("unsupported frame version");
172 		if (! (m.get("type") instanceof String type) || ! TYPES.contains(type))
173 			throw BusRefusal.badFrame("unknown frame type");
174 		var out = new JsonMap();
175 		out.put("v", VERSION);
176 		out.put("type", type);
177 		switch (type) {
178 			case "pub" -> {
179 				requireTopic(m, out);
180 				if (! m.containsKey("payload"))
181 					throw BusRefusal.badFrame("pub frame has no payload");
182 				out.put("payload", m.get("payload"));
183 				if (m.containsKey("retained")) {
184 					if (! (m.get("retained") instanceof Boolean))
185 						throw BusRefusal.badFrame("retained must be a boolean");
186 					out.put("retained", m.get("retained"));
187 				}
188 				optionalSeq(m, out);
189 			}
190 			case "clear" -> {
191 				requireTopic(m, out);
192 				optionalSeq(m, out);
193 			}
194 			case "resync-begin", "resync-end" -> optionalSeq(m, out);
195 			case "error" -> {
196 				if (! (m.get("code") instanceof String code))
197 					throw BusRefusal.badFrame("error frame has no code");
198 				if (! (m.get("message") instanceof String message))
199 					throw BusRefusal.badFrame("error frame has no message");
200 				out.put("code", code);
201 				out.put("message", message);
202 				if (m.get("topic") instanceof String topic)
203 					out.put("topic", topic);
204 			}
205 			default -> { /* ping: no fields */ }
206 		}
207 		return out;
208 	}

210 	/**
211 	 * Extracts the {@code seq} of a frame produced by this class (it is always the last field when present).
212 	 *
213 	 * @param encoded An encoded frame.
214 	 * @return The sequence number, or empty for {@code ping} and {@code error}.
215 	 */
216 	public static OptionalLong seq(String encoded) {
217 		var m = TRAILING_SEQ.matcher(encoded);
218 		return m.find() ? OptionalLong.of(Long.parseLong(m.group(1))) : OptionalLong.empty();
219 	}

221 	/**
222 	 * The UTF-8 length of a string, without allocating its bytes.
223 	 *
224 	 * @param s The string.
225 	 * @return The number of UTF-8 bytes.
226 	 */
227 	public static int utf8Length(String s) {
228 		var n = 0;
229 		var i = 0;
230 		while (i < s.length()) {
231 			var c = s.charAt(i);
232 			if (c < 0x80)
233 				n++;
234 			else if (c < 0x800)
235 				n += 2;
236 			else if (Character.isHighSurrogate(c) && i + 1 < s.length() && Character.isLowSurrogate(s.charAt(i + 1))) {
237 				n += 4;
238 				i++;
239 			} else
240 				n += 3;
241 			i++;
242 		}
243 		return n;
244 	}

246 	/**
247 	 * Quotes a string exactly as JavaScript's {@code JSON.stringify} does (lowercase {@code \\u} hex, lone surrogates
248 	 * escaped, solidus not escaped).
249 	 *
250 	 * @param s The string.
251 	 * @return The JSON string literal.
252 	 */
253 	static String quote(String s) {
254 		var sb = new StringBuilder(s.length() + 2).append('"');
255 		var i = 0;
256 		while (i < s.length()) {
257 			var c = s.charAt(i);
258 			switch (c) {
259 				case '"' -> sb.append("\\\"");
260 				case '\\' -> sb.append("\\\\");
261 				case '\n' -> sb.append("\\n");
262 				case '\r' -> sb.append("\\r");
263 				case '\t' -> sb.append("\\t");
264 				case '\b' -> sb.append("\\b");
265 				case '\f' -> sb.append("\\f");
266 				default -> {
267 					if (c < 0x20)
268 						sb.append(String.format("\\u%04x", (int) c));
269 					else if (Character.isHighSurrogate(c) && i + 1 < s.length() && Character.isLowSurrogate(s.charAt(i + 1))) {
270 						sb.append(c).append(s.charAt(i + 1));
271 						i++;
272 					} else if (Character.isSurrogate(c))
273 						sb.append(String.format("\\u%04x", (int) c));
274 					else
275 						sb.append(c);
276 				}
277 			}
278 			i++;
279 		}
280 		return sb.append('"').toString();
281 	}

283 	/**
284 	 * Whether the text is one JSON object optionally surrounded by whitespace.  {@link JsonParser} stops after the first
285 	 * value and ignores whatever follows, but a frame is the whole text.  String contents are skipped so a brace inside
286 	 * a string does not end the object early; well-formedness itself is left to the parser.
287 	 */
288 	private static boolean isSingleObject(String text) {
289 		var n = text.length();
290 		var i = 0;
291 		while (i < n && isWhitespace(text.charAt(i)))
292 			i++;
293 		if (i == n || text.charAt(i) != '{')
294 			return false;
295 		var depth = 0;
296 		for (; i < n; i++) {
297 			var c = text.charAt(i);
298 			if (c == '"') {
299 				for (i++; i < n && text.charAt(i) != '"'; i++)
300 					if (text.charAt(i) == '\\')
301 						i++;
302 			} else if (c == '{' || c == '[') {
303 				depth++;
304 			} else if ((c == '}' || c == ']') && --depth == 0) {
305 				for (i++; i < n; i++)
306 					if (! isWhitespace(text.charAt(i)))
307 						return false;
308 				return true;
309 			}
310 		}
311 		return false;
312 	}
313 
314 	private static boolean isWhitespace(char c) {
315 		return c == ' ' || c == '\t' || c == '\n' || c == '\r';
316 	}

318 	private static StringBuilder head(String type) {
319 		return new StringBuilder(64).append("{\"v\":").append(VERSION).append(",\"type\":").append(quote(type));
320 	}

322 	private static String pubRaw(String topic, String payloadJson, Boolean retained, Long seq) {
323 		var sb = head("pub").append(",\"topic\":").append(quote(topic)).append(",\"payload\":").append(payloadJson);
324 		if (retained != null)
325 			sb.append(",\"retained\":").append(retained);
326 		if (seq != null)
327 			sb.append(",\"seq\":").append(seq);
328 		return sb.append('}').toString();
329 	}
330 
331 	private static void requireTopic(JsonMap in, JsonMap out) throws BusRefusal {
332 		if (! (in.get("topic") instanceof String topic) || topic.isEmpty())
333 			throw BusRefusal.badFrame("frame has no topic");
334 		out.put("topic", topic);
335 	}
336 
337 	private static void optionalSeq(JsonMap in, JsonMap out) throws BusRefusal {
338 		if (! in.containsKey("seq"))
339 			return;
340 		var seq = seqOf(in.get("seq"));
341 		if (seq == null)
342 			throw BusRefusal.badFrame("seq must be a non-negative integer");
343 		out.put("seq", seq);
344 	}
345 
346 	private static Long seqOf(Object o) {
347 		if (! (o instanceof Number n))
348 			return null;
349 		var d = n.doubleValue();
350 		if (d < 0 || d > MAX_SAFE_SEQ || d != Math.floor(d))
351 			return null;
352 		return n.longValue();
353 	}
354 
355 	private static String requireString(Map<String,?> frame, String key) {
356 		if (! (frame.get(key) instanceof String s))
357 			throw new IllegalArgumentException("BusFrames.encode: '" + key + "' must be a string");
358 		return s;
359 	}
360 
361 	private static String optionalString(Map<String,?> frame, String key) {
362 		return frame.get(key) == null ? null : requireString(frame, key);
363 	}
364 
365 	private static Boolean optionalBoolean(Map<String,?> frame, String key) {
366 		var o = frame.get(key);
367 		if (o != null && ! (o instanceof Boolean))
368 			throw new IllegalArgumentException("BusFrames.encode: '" + key + "' must be a boolean");
369 		return (Boolean) o;
370 	}
