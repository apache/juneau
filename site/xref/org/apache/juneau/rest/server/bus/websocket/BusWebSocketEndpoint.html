17  package org.apache.juneau.rest.server.bus.websocket;
18  
19  import static java.nio.charset.StandardCharsets.*;
20  
21  import java.util.concurrent.*;
22  import java.util.concurrent.atomic.*;
23  import java.util.logging.*;
24  
25  import org.apache.juneau.rest.server.bus.*;
26  
27  import jakarta.websocket.*;
28  
29  /**
30   * The per-connection endpoint of the bus's WebSocket transport.  Not for direct use: it is public only because the
31   * container instantiates it through {@link BusWebSocketConfigurator}.
32   *
33   * <p>
34   * It closes a refused handshake at once ({@code 4401} / {@code 4403}).  Otherwise it adapts the
35   * {@link Session} to a {@link FrameSink}, connects it to the {@link ServerBus} (which sends the resync), forwards
36   * inbound text frames to {@link ServerBus#receive} (which refuses anything but a {@code pub} frame with a JSON object
37   * payload), and closes with {@code 1009} ({@code bus:frame-too-large}) on a frame over
38   * {@code BusPolicy.maxFrameBytes}.  Nothing it logs contains the session id.
39   *
40   * <p>
41   * It reads the per-connection user properties of its {@link EndpointConfig}.  The container gives each handshake its
42   * own copy of that map (WebSocket 2.1 and later; Jetty and Tomcat do), which is what keeps concurrent connections apart.
43   *
44   * <h5 class='section'>Example:</h5>
45   * <p class='bjava'>
46   *   <jc>// Do not construct this class; register the endpoint instead:</jc>
47   *   BusWebSockets.<jsm>register</jsm>(<jv>servletContext</jv>, JobsRest.<jsf>BUS</jsf>);
48   * </p>
49   *
50   * @since 10.0.0
51   */
52  public final class BusWebSocketEndpoint extends Endpoint {
53  
54  	private static final Logger LOGGER = Logger.getLogger(BusWebSocketEndpoint.class.getName());
55  
56  	private final ServerBus bus;
57  	private final Executor closer;
58  	private final AtomicReference<BusSession> busSession = new AtomicReference<>();
59  	private final AtomicReference<WsFrameSink> sink = new AtomicReference<>();
60  
61  	BusWebSocketEndpoint(ServerBus bus) {
62  		this(bus, null);
63  	}
64  
65  	/** @param closer Runs the sink's socket closes; {@code null} for the default (a daemon pool). */
66  	BusWebSocketEndpoint(ServerBus bus, Executor closer) {
67  		this.bus = bus;
68  		this.closer = closer;
69  	}
70  
71  	@Override /* Endpoint */
72  	public void onOpen(Session session, EndpointConfig config) {
73  		var props = config.getUserProperties();
74  		if (props.get(BusWebSocketConfigurator.REFUSAL_PROPERTY) instanceof BusWebSocketConfigurator.Refusal r) {
75  			WsFrameSink.closeQuietly(session, r.code(), r.reason());
76  			return;
77  		}
78  		if (! (props.get(BusWebSocketConfigurator.SESSION_PROPERTY) instanceof BusSession s)) {
79  			var r = BusWebSocketConfigurator.UNKNOWN_SESSION;
80  			WsFrameSink.closeQuietly(session, r.code(), r.reason());
81  			return;
82  		}
83  		var policy = bus.policy();
84  		var maxBytes = policy.maxFrameBytes();
85  		session.setMaxTextMessageBufferSize(maxBytes);
86  		var k = closer == null ? new WsFrameSink(session, policy.maxQueuedFrames()) : new WsFrameSink(session, policy.maxQueuedFrames(), closer);
87  		busSession.set(s);
88  		sink.set(k);
89  		try {
90  			bus.connect(s, k);
91  		} catch (RuntimeException e) {
92  			LOGGER.log(Level.WARNING, "bus websocket connect failed: {0}", e.getClass().getName());
93  			k.markClosed();
94  			try {
95  				bus.disconnect(s, k);
96  			} catch (RuntimeException e2) {
97  				// The same fault that failed the connect; the sink is already marked closed.
98  			}
99  			WsFrameSink.closeQuietly(session, 1011, "bus:connect-failed");
100 			return;
101 		}
102 		session.addMessageHandler(String.class, text -> onText(session, s, text, maxBytes));
103 	}
104 
105 	private void onText(Session session, BusSession s, String text, int maxBytes) {
106 		// A char is at least one byte, so a long enough string needs no encoding to be refused.
107 		if (text.length() > maxBytes || text.getBytes(UTF_8).length > maxBytes) {
108 			WsFrameSink.closeQuietly(session, CloseReason.CloseCodes.TOO_BIG.getCode(), "bus:frame-too-large");
109 			return;
110 		}
111 		bus.receive(s, text);
112 	}
113 
114 	@Override /* Endpoint */
115 	public void onClose(Session session, CloseReason closeReason) {
116 		var k = sink.get();
117 		if (k != null) {
118 			k.markClosed();
119 			bus.disconnect(busSession.get(), k);
120 		}
121 	}
122 
123 	@Override /* Endpoint */
124 	public void onError(Session session, Throwable thr) {
125 		// Only the class: a message or stack can carry the request path, which holds the session id.
126 		LOGGER.log(Level.FINE, "bus websocket transport error: {0}", thr == null ? null : thr.getClass().getName());
127 	}
128 }
