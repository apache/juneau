20  import static org.apache.juneau.rest.server.runreport.ReportText.*;
21  
22  import java.nio.file.*;
23  import java.util.*;
24  import java.util.stream.*;
25  
26  import org.apache.juneau.rest.server.runreport.RunEvent.*;

28  /**
29   * Reads Jest's <c>--json</c> output.
30   *
31   * <p>
32   * The framework key is <c>jest</c>.  A test file that failed to run (no assertions, a non-empty message) becomes one
33   * <c>error</c> test named <c>(suite failed to run)</c>, so a crashed file is a visible red block rather than silence.
34   * ANSI colour codes are stripped from messages.  A truncated file yields no tests and a warning.
35   */
36  public final class JestJsonReportReader implements ReportReader {
37  
38  	private static final String CRASHED = "(suite failed to run)";
39  
40  	@Override /* ReportReader */
41  	public String fw() {
42  		return "jest";
43  	}
44  
45  	@Override /* ReportReader */
46  	@SuppressWarnings({
47  		"java:S135", // Guard clauses keep the loop readable.
48  		"java:S3776" // Walking files and their assertions is inherently nested.
49  	})
50  	public ReportResult read(Path path, ReportLimits lim) {
51  		var out = new ReportBuilder();
52  		var root = readRoot(path, lim, out);
53  		if (root == null)
54  			return out.build(fw());
55  
56  		var files = new ArrayList<String>();
57  		var rows = new ArrayList<ReportTest>();
58  		var rowFiles = new ArrayList<String>();
59  		for (var tr : list(root.get("testResults"))) {
60  			var file = map(tr);
61  			if (file == null)
62  				continue;
63  			var raw = str(file.get("name")) == null ? "(unknown file)" : str(file.get("name"));
64  			if (! files.contains(raw))
65  				files.add(raw);
66  			var assertions = list(file.get("assertionResults"));
67  			var fileMsg = str(file.get("message"));
68  			if (assertions.isEmpty() && "failed".equals(str(file.get("status"))) && fileMsg != null && ! fileMsg.isBlank()) {
69  				if (rows.size() >= lim.maxTests()) {
70  					out.truncated = true;
71  					out.warnings.add("report truncated at " + lim.maxTests() + " tests");
72  					break;
73  				}
74  				var body = stripAnsi(fileMsg).strip();
75  				rows.add(new ReportTest(null, CRASHED, TestStatus.ERROR, null, clip(firstLine(body), lim.maxMsgChars()), clip(body, lim.maxTraceChars())));
76  				rowFiles.add(raw);
77  				continue;
78  			}
79  			if (! addAssertions(assertions, raw, lim, out, rows, rowFiles))
80  				break;
81  		}
82  		var display = stripCommonDir(files);
83  		for (var i = 0; i < rows.size(); i++) {
84  			var r = rows.get(i);
85  			var suite = clip(display.get(files.indexOf(rowFiles.get(i))), lim.maxNameChars());
86  			out.tests.add(new ReportTest(suite, r.name(), r.status(), r.ms(), r.msg(), r.trace()));
87  		}
88  		return out.build(fw());
89  	}
90  
91  	// Returns false when the test limit was reached.
92  	private static boolean addAssertions(List<?> assertions, String raw, ReportLimits lim, ReportBuilder out, List<ReportTest> rows, List<String> rowFiles) {
93  		for (var a : assertions) {
94  			var am = map(a);
95  			if (am == null)
96  				continue;
97  			if (rows.size() >= lim.maxTests()) {
98  				out.truncated = true;
99  				out.warnings.add("report truncated at " + lim.maxTests() + " tests");
100 				return false;
101 			}
102 			rows.add(assertion(am, lim));
103 			rowFiles.add(raw);
104 		}
105 		return true;
106 	}
107 
108 	private static ReportTest assertion(Map<String,Object> a, ReportLimits lim) {
109 		var name = str(a.get("fullName"));
110 		if (name == null || name.isEmpty()) {
111 			var parts = list(a.get("ancestorTitles")).stream().filter(String.class::isInstance).map(String.class::cast)
112 				.collect(Collectors.toCollection(ArrayList::new));
113 			var title = str(a.get("title"));
114 			if (title != null)
115 				parts.add(title);
116 			name = String.join(" ", parts);
117 		}
118 		var status = switch (String.valueOf(a.get("status"))) {
119 			case "passed" -> TestStatus.PASS;
120 			case "failed" -> TestStatus.FAIL;
121 			default -> TestStatus.SKIP;
122 		};
123 		String msg = null;
124 		String trace = null;
125 		var fm = list(a.get("failureMessages")).stream().filter(String.class::isInstance).map(String.class::cast)
126 			.map(ReportText::stripAnsi).toList();
127 		if (! fm.isEmpty()) {
128 			var first = firstLine(fm.get(0).strip());
129 			msg = first.isEmpty() ? null : clip(first, lim.maxMsgChars());
130 			trace = clip(String.join("\n", fm), lim.maxTraceChars());
131 		}
132 		return new ReportTest(null, clip(name, lim.maxNameChars()), status, ms(a.get("duration")), msg, trace);
133 	}
134 
135 	private static String firstLine(String s) {
136 		return RunViewChecks.firstLine(s);
137 	}
138 }
