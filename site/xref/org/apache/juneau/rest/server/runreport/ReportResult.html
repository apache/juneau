17  package org.apache.juneau.rest.server.runreport;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.rest.server.runreport.RunEvent.*;
24  
25  /**
26   * What a {@link ReportReader} recovered from a report.
27   *
28   * @param fw The framework key.
29   * @param tests The recovered tests, in report order.
30   * @param warnings Human-readable warnings, for example <c>report truncated: unexpected end of file after 4120 tests</c>.
31   * @param truncated Whether the file ended early or the test limit was hit.
32   * @param skippedFiles The number of files that were not read (over the size limit, unreadable).
33   */
34  public record ReportResult(String fw, List<ReportTest> tests, List<String> warnings, boolean truncated, int skippedFiles) {
35  
36  	private static final String UNNAMED = "(unnamed)";
37  
38  	/**
39  	 * Constructor.
40  	 *
41  	 * @param fw The framework key.
42  	 * @param tests The recovered tests.
43  	 * @param warnings Human-readable warnings.
44  	 * @param truncated Whether the file ended early or the test limit was hit.
45  	 * @param skippedFiles The number of files that were not read.
46  	 */
47  	public ReportResult {
48  		if (! RunViewChecks.isFw(fw))
49  			throw iaex("ReportResult fw must match ^[a-z0-9][a-z0-9._-]{0,31}$; got '%s'.", fw);
50  		tests = List.copyOf(tests);
51  		warnings = List.copyOf(warnings);
52  	}
53  
54  	/**
55  	 * Returns the events a producer appends at the end of a step.
56  	 *
57  	 * <p>
58  	 * The sequence is a <c>replace</c> for the step, one <c>test</c> per recovered test, then one <c>warn</c> note per
59  	 * warning.  When no test was recovered and there are warnings, only the notes are returned, so the step keeps its
60  	 * live <c>suite</c> placeholders.  A valid but empty report yields just the <c>replace</c>.
61  	 *
62  	 * @param step The id of the step the results belong to.
63  	 * @return The events, not yet numbered.
64  	 */
65  	public List<RunEvent> toEvents(String step) {
66  		var out = new ArrayList<RunEvent>();
67  		if (! (tests.isEmpty() && ! warnings.isEmpty()))
68  			out.add(RunEvent.replace(step));
69  		out.addAll(tests.stream().map(t -> toEvent(step, t)).toList());
70  		out.addAll(warnings.stream().map(w -> RunEvent.note(Level.WARN, clip(w, RunEvent.MAX_NOTE)).withStep(step)).toList());
71  		return out;
72  	}
73  
74  	private RunEvent toEvent(String step, ReportTest t) {
75  		var e = RunEvent.test(step, fw, name(t.suite()), name(t.name()), t.status());
76  		if (t.ms() != null && t.ms() >= 0 && t.ms() <= RunEvent.MAX_SAFE_INT)
77  			e = e.withMs(t.ms());
78  		var withMsg = t.msg() == null || t.msg().isEmpty() ? e : e.withMsg(clip(t.msg(), RunEvent.MAX_MSG));
79  		var full = t.trace() == null || t.trace().isEmpty() ? withMsg : withMsg.withTrace(clip(t.trace(), RunEvent.MAX_TRACE));
80  		// Control characters expand six-fold when serialized, so an extreme test can exceed the event size cap; shed the
81  		// bulkiest members rather than lose the test.
82  		for (var candidate : List.of(full, withMsg, e)) {
83  			try {
84  				return candidate.validate();
85  			} catch (IllegalArgumentException ex) {
86  				// try the next, smaller candidate
87  			}
88  		}
89  		return e;
90  	}
91  
92  	private static String name(String s) {
93  		return s == null || s.isEmpty() ? UNNAMED : clip(s, RunEvent.MAX_NAME);
94  	}
95  
96  	private static String clip(String s, int max) {
97  		return ReportText.clip(s, max);
98  	}
99  }
