17  package org.apache.juneau.rest.server.runreport;
18  
19  import java.nio.file.*;
20  import java.util.*;
21  import java.util.regex.*;
22  
23  /**
24   * Text helpers shared by the report readers.
25   */
26  public final class ReportText {
27  
28  	private static final Pattern ANSI = Pattern.compile("\u001b\\[[0-9;]*[A-Za-z]");
29  
30  	private ReportText() {}
31  
32  	/**
33  	 * Removes ANSI escape sequences (test runners colour their messages).
34  	 *
35  	 * @param s The text.  May be <jk>null</jk>.
36  	 * @return The text without escape sequences, or an empty string for <jk>null</jk>.
37  	 */
38  	public static String stripAnsi(String s) {
39  		return s == null ? "" : ANSI.matcher(s).replaceAll("");
40  	}
41  
42  	/**
43  	 * Truncates a string to at most {@code max} chars, ending in an ellipsis when it was cut.
44  	 *
45  	 * @param s The text.  May be <jk>null</jk>.
46  	 * @param max The maximum length, at least 1.
47  	 * @return The clipped text, or an empty string for <jk>null</jk>.
48  	 */
49  	public static String clip(String s, int max) {
50  		if (s == null)
51  			return "";
52  		return s.length() <= max ? s : s.substring(0, Math.max(0, max - 1)) + "…";
53  	}
54  
55  	/**
56  	 * Returns the file name of a path, never the directories (warnings leave the machine).
57  	 *
58  	 * @param path The path.
59  	 * @return The last segment, or an empty string.
60  	 */
61  	public static String fileName(Path path) {
62  		var n = path == null ? null : path.getFileName();
63  		return n == null ? "" : n.toString();
64  	}
65  
66  	/**
67  	 * Shortens a list of test file paths for display.
68  	 *
69  	 * <p>
70  	 * Backslashes become slashes, then the longest common <i>directory</i> prefix of all paths is removed, so
71  	 * {@code /ci/work/repo/src/a.test.js} and {@code /ci/work/repo/src/b/c.test.js} become {@code a.test.js} and
72  	 * {@code b/c.test.js}.  A list with a single path keeps its last two segments ({@code src/a.test.js}).
73  	 *
74  	 * @param paths The paths.
75  	 * @return The display paths, in the same order.
76  	 */
77  	public static List<String> stripCommonDir(List<String> paths) {
78  		var segs = paths.stream().map(p -> p.replace('\\', '/').split("/", -1)).toList();
79  		var out = new ArrayList<String>(paths.size());
80  		if (segs.isEmpty())
81  			return out;
82  		if (segs.size() == 1) {
83  			var s = segs.get(0);
84  			out.add(String.join("/", Arrays.copyOfRange(s, Math.max(0, s.length - 2), s.length)));
85  			return out;
86  		}
87  		var common = Integer.MAX_VALUE;
88  		for (var s : segs)
89  			common = Math.min(common, s.length - 1);
90  		var first = segs.get(0);
91  		var keep = 0;
92  		while (keep < common && allEqual(segs, first[keep], keep))
93  			keep++;
94  		var drop = keep;
95  		out.addAll(segs.stream().map(s -> String.join("/", Arrays.copyOfRange(s, drop, s.length))).toList());
96  		return out;
97  	}
98  
99  	private static boolean allEqual(List<String[]> segs, String value, int index) {
100 		for (var s : segs)
101 			if (! s[index].equals(value))
102 				return false;
103 		return true;
104 	}
105 }
