17  package org.apache.juneau.rest.server.reactive;
18  
19  import java.util.concurrent.*;
20  
21  /**
22   * SPI that bridges a third-party reactive return type to one of the two JDK-native shapes that the
23   * {@link ReactiveResponseProcessor} understands: a {@link CompletionStage} (single value) or a
24   * {@link java.util.concurrent.Flow.Publisher Flow.Publisher} (a stream of values).
25   *
26   * <h5 class='topic'>Why this exists</h5>
27   * <p>
28   * The opt-in {@code juneau-rest-server-reactive} module handles {@link java.util.concurrent.Flow.Publisher Flow.Publisher}
29   * (and, via the existing {@code AsyncResponseProcessor}, {@link CompletionStage}) using only JDK APIs &mdash;
30   * no external dependency. Libraries such as Project Reactor ({@code Mono} / {@code Flux}), RxJava 3
31   * ({@code Single} / {@code Maybe} / {@code Flowable} / {@code Observable}), and the
32   * Reactive-Streams {@code org.reactivestreams.Publisher} are NOT on the core classpath. The opt-in
33   * {@code juneau-rest-server-reactive-reactor} module ships {@code ReactiveStreamsAdapter} implementations that
34   * convert those types into the JDK-native shapes, so the streaming / buffering / SSE / NDJSON plumbing
35   * is implemented exactly once in the core.
36   *
37   * <h5 class='topic'>Registration</h5>
38   * <p>
39   * Adapters are discovered via the JDK {@link java.util.ServiceLoader} mechanism. A bridge module (or a
40   * consumer who wants a custom adapter) ships a
41   * {@code META-INF/services/ReactiveStreamsAdapter} file listing the
42   * implementation class names. The {@link ReactiveResponseProcessor} loads the providers once (lazily,
43   * at first use) and skips any provider whose backing library is absent from the runtime classpath
44   * (a {@link NoClassDefFoundError} / {@link java.util.ServiceConfigurationError} on instantiation is swallowed).
45   * This means a single {@code juneau-rest-server-reactive-reactor} jar can declare Reactor, RxJava, and
46   * Reactive-Streams adapters while the consumer pulls only the {@code provided}-scope libraries they
47   * actually want.
48   *
49   * @see ReactiveResponseProcessor
50   * @see Adaptation
51   * @since 10.0.0
52   */
53  public interface ReactiveStreamsAdapter {
54  
55  	/**
56  	 * Returns whether this adapter recognizes the supplied return value.
57  	 *
58  	 * <p>
59  	 * Implementations must return {@code false} (rather than throw) for values they do not handle, and
60  	 * must be cheap to call &mdash; this is invoked on the response hot path for every non-null handler
61  	 * return value once at least one adapter is registered.
62  	 *
63  	 * @param value The {@code @RestOp} handler return value. Never {@code null}.
64  	 * @return {@code true} if {@link #adapt(Object)} can convert this value.
65  	 */
66  	boolean canAdapt(Object value);
67  
68  	/**
69  	 * Converts the supplied reactive value to a JDK-native {@link Adaptation}.
70  	 *
71  	 * <p>
72  	 * Only called when {@link #canAdapt(Object)} returned {@code true} for the same value.
73  	 *
74  	 * @param value The reactive value to adapt. Never {@code null}.
75  	 * @return The adapted single-value or streaming shape. Never {@code null}.
76  	 */
77  	Adaptation adapt(Object value);
78  }
