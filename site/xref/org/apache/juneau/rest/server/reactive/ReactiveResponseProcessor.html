40  /**
41   * Response processor that bridges <a class="doclink" href="https://www.reactive-streams.org/">Reactive Streams</a>
42   * return values from {@code @RestOp} handlers to Juneau's response pipeline.
43   *
44   * <h5 class='topic'>Activation (opt-in)</h5>
45   * <p>
46   * This processor ships in the opt-in {@code juneau-rest-server-reactive} module and is <b>not</b> wired
47   * into {@code DefaultConfig}. A bare {@code juneau-rest-server} has zero reactive behavior. When the
48   * {@code juneau-rest-server-reactive} jar is on the classpath, its
49   * {@code META-INF/services/ResponseProcessor} provider file is discovered
50   * by {@code RestContext} via {@link ServiceLoader} and this processor is front-loaded ahead of
51   * {@code AsyncResponseProcessor} in the chain &mdash; no {@code @Rest(responseProcessors=...)} entry required.
52   *
53   * <h5 class='topic'>What it handles</h5>
54   * <p>
55   * This is the single, shared spine for all reactive return-type support. It natively understands the
56   * JDK type {@link java.util.concurrent.Flow.Publisher Flow.Publisher&lt;T&gt;} (no external dependency
57   * required) and, through registered {@link ReactiveStreamsAdapter} providers, any third-party reactive
58   * type the opt-in {@code juneau-rest-server-reactive-reactor} module adapts to it (Project Reactor
59   * {@code Mono} / {@code Flux}, RxJava 3 {@code Single} / {@code Maybe} / {@code Flowable} /
60   * {@code Observable}, and the Reactive-Streams {@code org.reactivestreams.Publisher}).
61   *
62   * <p>
63   * Single-value reactive types (e.g. {@code Mono}, {@code Single}) are adapted to a {@link CompletionStage}
64   * and collapse onto the existing {@link AsyncResponseProcessor} async path, inheriting its timeout,
65   * completion-executor ({@code @Rest(asyncCompletionExecutor)}), and MDC bridging behavior for free.
66   *
67   * <h5 class='topic'>Response shapes for multi-value streams</h5>
68   * <p>
69   * A streaming publisher is rendered as one of three shapes, selected by the negotiated response media
70   * type (the handler's {@link RestResponse#setContentType(String) Content-Type}, then the request
71   * {@code Accept} header):
72   * <ul>
73   * 	<li><b>SSE</b> ({@code text/event-stream}) &mdash; each element is emitted as a Server-Sent-Events
74   * 		frame. {@link SseEvent} elements are written verbatim; any other element type is JSON-encoded
75   * 		into the {@code data:} field.
76   * 	<li><b>NDJSON</b> ({@code application/x-ndjson}, {@code application/jsonl}, {@code application/json5l}) &mdash; each element is
77   * 		JSON-encoded on its own line.
78   * 	<li><b>Buffer</b> (default, any other media type) &mdash; all elements are collected into a
79   * 		{@link List List} and serialized through the normal serializer chain (e.g. a JSON
80   * 		array). The collection is wrapped in a {@link CompletableFuture} and handed to the async path,
81   * 		so a slow producer never blocks the request thread.
82   * </ul>
83   *
84   * <h5 class='topic'>Backpressure</h5>
85   * <p>
86   * Streaming subscribers request one element at a time ({@code request(1)} on subscribe and again after
87   * each frame is written and flushed). Because writing to the servlet output stream blocks until the
88   * socket accepts the bytes, this gives natural backpressure &mdash; the producer is paced by the
89   * client's drain rate and the server-side buffer does not grow without bound. Buffer-shape subscribers
90   * request {@link Long#MAX_VALUE} since the collection is bounded by the publisher's own completion.
91   *
92   * <h5 class='topic'>Threading, executors, and MDC</h5>
93   * <p>
94   * Buffer-shape responses route through {@link AsyncResponseProcessor} and therefore honor
95   * {@code @Rest(asyncCompletionExecutor)} and the SLF4J MDC bridge. Streaming-shape
96   * frame writes happen on whichever thread the publisher emits on (the Reactor / RxJava scheduler); this
97   * processor does not impose a {@code subscribeOn(...)} so it never fights the library's scheduler model.
98   * When MDC propagation is enabled, the request-thread MDC snapshot is reinstalled around each
99   * {@code onNext} / terminal callback via {@link MdcAsyncListener} so log statements emitted while writing
100  * a frame see the request's diagnostic context.
101  *
102  * <h5 class='topic'>Synchronous fallback</h5>
103  * <p>
104  * In environments where {@link HttpServletRequest#startAsync()} is unsupported (notably Juneau's
105  * {@code MockServletRequest}), streaming subscribes synchronously and blocks the request thread until
106  * the publisher terminates (bounded by the configured async timeout), writing frames as they arrive.
107  * This keeps the unit-test surface working without a real servlet container.
108  *
109  * @see ReactiveStreamsAdapter
110  * @see AsyncResponseProcessor
111  * @since 10.0.0
112  */
113 @SuppressWarnings({
114 	"resource" // Closeables here are framework-managed and not owned/closed by this class; not a real leak.
115 })

132 	@Override /* Overridden from ResponseProcessor */
133 	public int process(RestOpSession opSession) throws IOException, BasicHttpException {
134 		var res = opSession.getResponse();
135 		var content = res.getContent().orElse(null);
136 
137 		if (content == null) // HTT: true branch requires a response with no content at all; framework-level null-content suppression makes this unreachable in practice
138 			return NEXT;
139 
140 		if (content instanceof Flow.Publisher<?> content2)
141 			return dispatch(opSession, content2);
142 
143 		var a = adapters();
144 		for (var adapter : a) {
145 			if (adapter.canAdapt(content)) { // HTT: true branch requires a ReactiveStreamsAdapter on classpath (lives in juneau-rest-server-reactive-reactor, not on this module's test classpath)
146 				var adaptation = adapter.adapt(content);
147 				if (adaptation.isStream()) // HTT: both branches require a ReactiveStreamsAdapter (see above)
148 					return dispatch(opSession, adaptation.stream());
149 				res.setContent(adaptation.single());
150 				return RESTART;
151 			}
152 		}
153 
154 		return NEXT;
155 	}
156 
157 	private int dispatch(RestOpSession opSession, Flow.Publisher<?> pub) throws IOException {
158 		var shape = resolveShape(opSession);
159 		if (shape == Shape.BUFFER)
160 			return handleBuffer(opSession, pub);
161 		return handleStream(opSession, pub, shape);
162 	}

164 	// -----------------------------------------------------------------------------------------------------------------
165 	// Buffer shape — collect into a List, hand to the async path.
166 	// -----------------------------------------------------------------------------------------------------------------
167 
168 	private int handleBuffer(RestOpSession opSession, Flow.Publisher<?> pub) {
169 		var cf = new CompletableFuture<List<Object>>();
170 		pub.subscribe(new CollectingSubscriber(cf));
171 		opSession.getResponse().setContent(cf);
172 		return RESTART;
173 	}

179 	@SuppressWarnings({
180 		"java:S3776", // Streaming setup is branchy by nature (content-type prep + async-vs-sync fallback).
181 		"java:S1141"  // Nested try/catch cleanly separates startAsync recovery from the sync fallback.
182 	})
183 	private int handleStream(RestOpSession opSession, Flow.Publisher<?> pub, Shape shape) throws IOException {
184 		var res = opSession.getResponse();
185 		prepareStreamingHeaders(res, shape);
186 
187 		var writer = res.getNegotiatedWriter();
188 		FrameEncoder encoder = shape == Shape.SSE
189 			? ReactiveResponseProcessor::writeSseFrame
190 			: ReactiveResponseProcessor::writeNdjsonFrame;
191 
192 		var req = opSession.getRequest().getHttpServletRequest();
193 		var mdc = opSession.getRestContext().isMdcAsyncPropagation() // HTT: false branch requires MDC propagation enabled
194 			? MdcAsyncListener.snapshot()
195 			: null;
196 
197 		AsyncContext asyncCtx = null;
198 		if (req.isAsyncSupported()) { // HTT: true branch requires a real servlet container (MockServletRequest returns false)
199 			try {
200 				asyncCtx = req.startAsync();
201 			} catch (IllegalStateException e) {
202 				asyncCtx = null;  // Already committed, or async not actually supported.
203 			}
204 		}
205 
206 		if (asyncCtx != null) { // HTT: true branch requires a real servlet container
207 			req.setAttribute(AsyncResponseProcessor.ATTR_ASYNC_DISPATCH_OWNED, Boolean.TRUE);
208 			var ac = asyncCtx;
209 			ac.setTimeout(0);  // No artificial timeout — streams are paced by the producer / client disconnect.
210 			pub.subscribe(new StreamingSubscriber(res, writer, encoder, mdc, t -> completeAsync(ac, res, t)));
211 			return FINISHED;
212 		}
213 
214 		// Synchronous fallback (MockServletRequest et al.): block until the publisher terminates.
215 		var done = new CompletableFuture<Void>();
216 		var subscriber = new StreamingSubscriber(res, writer, encoder, mdc,
217 			t -> { if (t == null) done.complete(null); else done.completeExceptionally(t); });
218 		pub.subscribe(subscriber);
219 		if (! awaitSync(done, syncTimeoutMillis(opSession)))
220 			// Timed out or interrupted before the publisher terminated — cancel the still-live subscription so it
221 			// cannot emit late writes against a response we are about to finish.
222 			subscriber.cancel();
223 		try {
224 			writer.flush();
225 			res.flushBuffer();
226 		} catch (IOException e) {
227 			LOG.log(Level.FINEST, e, () -> "Final flush of synchronous reactive stream failed: " + e.getMessage());
228 		}
229 		return FINISHED;
230 	}

232 	private static void prepareStreamingHeaders(RestResponse res, Shape shape) {
233 		var ct = res.getContentType();
234 		if (shape == Shape.SSE) {
235 			if (ct == null || ! ct.contains("event-stream"))
236 				res.setContentType(SseSerializer.MEDIA_TYPE);
237 			res.setHeader("X-Content-Type-Options", "nosniff");
238 		} else if (ct == null) {
239 			res.setContentType("application/x-ndjson");
240 		}
241 		res.setHeader("Cache-Control", "no-cache");
242 		res.setHeader("Content-Encoding", "identity");
243 	}
244 
245 	/**
246 	 * Blocks until the synchronous stream completes, times out, or is interrupted.
247 	 *
248 	 * @return <jk>true</jk> if the publisher terminated (normally or with an error) before the timeout; <jk>false</jk>
249 	 * 	if the wait timed out or was interrupted (in which case the caller must cancel the still-live subscription).
250 	 */
251 	@SuppressWarnings({
252 		"java:S2142" // Interrupt flag restored before returning.
253 	})
254 	private static boolean awaitSync(CompletableFuture<Void> done, long timeoutMs) {
255 		try {
256 			done.get(timeoutMs, TimeUnit.MILLISECONDS);
257 		} catch (InterruptedException e) {
258 			Thread.currentThread().interrupt();
259 			return false;
260 		} catch (TimeoutException e) {
261 			LOG.log(Level.FINE, e, () -> "Synchronous reactive stream timed out: " + e.getMessage());
262 			return false;
263 		} catch (ExecutionException e) {
264 			// Publisher terminated with an error — the subscription is already finished, no orphan to cancel.
265 			LOG.log(Level.FINE, e, () -> "Synchronous reactive stream did not complete cleanly: " + e.getMessage());
266 		}
267 		return true;
268 	}
269 
270 	private static void completeAsync(AsyncContext asyncCtx, RestResponse res, Throwable error) { // HTT: only reachable from the async path (real servlet container required)
271 		try {
272 			if (error != null) // HTT: false branch only reachable from async path
273 				LOG.log(Level.FINE, error, () -> "Reactive stream terminated with error: " + error.getMessage());
274 			if (! res.getHttpServletResponse().isCommitted()) // HTT: true branch (committed=false) only via async path
275 				res.flushBuffer();
276 		} catch (IOException e) {
277 			LOG.log(Level.FINEST, e, () -> "Flush during async stream completion failed: " + e.getMessage());
278 		} finally {
279 			completeQuietly(asyncCtx);
280 		}
281 	}
282 
283 	private static void completeQuietly(AsyncContext asyncCtx) {
284 		try {
285 			asyncCtx.complete();
286 		} catch (IllegalStateException e) {
287 			LOG.log(Level.FINEST, e, () -> "AsyncContext.complete() raced with the container: " + e.getMessage());
288 		}
289 	}

295 	@SuppressWarnings({
296 		"java:S2095" // Writer is response-owned; we never close it.
297 	})
298 	private static void writeSseFrame(FinishablePrintWriter w, Object element) throws IOException, SerializeException {
299 		if (element == null)
300 			return;
301 		if (element instanceof SseEvent element2) {
302 			Sse.DEFAULT.write(element2, w);
303 			return;
304 		}
305 		var data = element instanceof CharSequence element2 ? element2.toString() : Json.of(element);
306 		Sse.DEFAULT.write(new SseEvent(null, data), w);
307 	}
308 
309 	@SuppressWarnings({
310 		"java:S2095" // Writer is response-owned; we never close it.
311 	})
312 	private static void writeNdjsonFrame(FinishablePrintWriter w, Object element) throws SerializeException {
313 		if (element == null)
314 			return;
315 		var json = element instanceof CharSequence element2 ? element2.toString() : Json.of(element);
316 		w.write(json);
317 		w.write("\n");
318 		w.flush();
319 	}

325 	private static Shape resolveShape(RestOpSession opSession) {
326 		var res = opSession.getResponse();
327 		var ct = res.getContentType();
328 		var accept = opSession.getRequest().getHttpServletRequest().getHeader("Accept");
329 		var probe = ((ct == null ? "" : ct) + "," + (accept == null ? "" : accept)).toLowerCase(Locale.ROOT);
330 		if (probe.contains("event-stream"))
331 			return Shape.SSE;
332 		if (probe.contains("ndjson") || probe.contains("jsonl") || probe.contains("json5l") || probe.contains("json-seq"))
333 			return Shape.NDJSON;
334 		return Shape.BUFFER;
335 	}
336 
337 	private static long syncTimeoutMillis(RestOpSession opSession) {
338 		var t = opSession.getContext().getAsyncTimeoutMillis();
339 		if (t <= 0)
340 			t = opSession.getRestContext().getAsyncTimeoutMillis();
341 		return t > 0 ? t : AsyncResponseProcessor.DEFAULT_ASYNC_TIMEOUT_MILLIS;
342 	}

348 	private static List<ReactiveStreamsAdapter> adapters() {
349 		var a = adapters;
350 		if (a == null) {
351 			synchronized (ReactiveResponseProcessor.class) {
352 				a = adapters;
353 				if (a == null) { // HTT: inner DCL null-check missed branch requires concurrent first-load race
354 					a = loadAdapters();
355 					adapters = a;
356 				}
357 			}
358 		}
359 		return a;
360 	}
361 
362 	private static List<ReactiveStreamsAdapter> loadAdapters() {
363 		var out = new ArrayList<ReactiveStreamsAdapter>();
364 		try {
365 			var it = ServiceLoader.load(ReactiveStreamsAdapter.class).iterator();
366 			while (it.hasNext())
367 				addNextAdapter(it, out);
368 		} catch (ServiceConfigurationError | RuntimeException e) {
369 			LOG.log(Level.FINE, e, () -> "ServiceLoader for ReactiveStreamsAdapter failed: " + e.getMessage());
370 		}
371 		return List.copyOf(out);
372 	}
373 
374 	private static void addNextAdapter(Iterator<ReactiveStreamsAdapter> it, List<ReactiveStreamsAdapter> out) {
375 		try {
376 			out.add(it.next());
377 		} catch (ServiceConfigurationError | RuntimeException e) { // HTT: catch branch requires a broken ServiceLoader registration (backing library absent at runtime)
378 			LOG.log(Level.FINE, e, () -> "Skipping reactive adapter (backing library likely absent): " + e.getMessage());
379 		}
380 	}

386 	/** Collects all emitted elements into a list and completes the supplied future (buffer shape). */
387 	private static final class CollectingSubscriber implements Flow.Subscriber<Object> {
388 		private final CompletableFuture<List<Object>> cf;
389 		private final List<Object> items = l();
390 
391 		CollectingSubscriber(CompletableFuture<List<Object>> cf) {
392 			this.cf = cf;
393 		}
394 
395 		@Override public void onSubscribe(Flow.Subscription s) { s.request(Long.MAX_VALUE); }
396 		@Override public void onNext(Object item) { items.add(item); }
397 		@Override public void onError(Throwable t) { cf.completeExceptionally(t); }
398 		@Override public void onComplete() { cf.complete(items); }
399 	}
400 
401 	/** Writes each emitted element as a wire frame (SSE / NDJSON), one element at a time (bounded backpressure). */
402 	private static final class StreamingSubscriber implements Flow.Subscriber<Object> {
403 		private final RestResponse res;
404 		private final FinishablePrintWriter writer;
405 		private final FrameEncoder encoder;
406 		private final Map<String,String> mdc;
407 		private final Consumer<Throwable> onTerminate;
408 		private final AtomicBoolean terminated = new AtomicBoolean();
409 		private volatile boolean wrote;
410 
411 		/** Published once by {@code onSubscribe} (Reactive Streams: at most one call); read by {@code onNext}/{@code cancel}. */
412 		@SuppressWarnings({
413 			"java:S3077" // Publish-once reference: assigned exactly once in onSubscribe; request()/cancel() are required by the Reactive Streams spec to support concurrent invocation, so volatile safe-publication of the reference alone is sufficient.
414 		})
415 		private volatile Flow.Subscription subscription;
416 
417 		StreamingSubscriber(RestResponse res, FinishablePrintWriter writer, FrameEncoder encoder,
418 				Map<String,String> mdc, Consumer<Throwable> onTerminate) {
419 			this.res = res;
420 			this.writer = writer;
421 			this.encoder = encoder;
422 			this.mdc = mdc;
423 			this.onTerminate = onTerminate;
424 		}
425 
426 		@Override public void onSubscribe(Flow.Subscription s) {
427 			subscription = s;
428 			s.request(1);
429 		}
430 
431 		@Override public void onNext(Object item) {
432 			try {
433 				withMdc(() -> writeFrame(item));
434 				subscription.request(1);
435 			} catch (RuntimeException e) {
436 				subscription.cancel();
437 				terminate(unwrap(e));
438 			}
439 		}
440 
441 		@Override public void onError(Throwable t) { terminate(t); }
442 		@Override public void onComplete() { terminate(null); }
443 
444 		/**
445 		 * Cancels the underlying subscription and marks this subscriber terminated so no late frames are written.
446 		 *
447 		 * <p>
448 		 * Invoked by the synchronous fallback path when the wait times out or is interrupted before the publisher
449 		 * terminates, preventing an orphaned subscription from emitting writes against a finished response.
450 		 */
451 		void cancel() {
452 			if (! terminated.compareAndSet(false, true))
453 				return;
454 			var s = subscription;
455 			if (s != null)
456 				s.cancel();
457 		}
458 
459 		private void writeFrame(Object item) {
460 			try {
461 				encoder.write(writer, item);
462 				writer.flush();
463 				// PrintWriter (which writer extends) never lets an IOException from the underlying stream escape —
464 				// it swallows it and only records it via checkError(). Without this check, a broken/closed client
465 				// connection is silently absorbed here and the pump keeps "succeeding" forever instead of detecting
466 				// the write failure and cleaning up (see SEP-2243 P4/P5 dead-client-cleanup path).
467 				if (writer.checkError())
468 					throw new IOException("Write failed: the client connection is no longer writable.");
469 				res.flushBuffer();
470 				wrote = true;
471 			} catch (IOException | SerializeException e) {
472 				throw isex(e);
473 			}
474 		}
475 
476 		private void withMdc(Runnable r) {
477 			MdcAsyncListener.wrap((BiConsumer<Object,Throwable>) (v, e) -> r.run(), mdc).accept(null, null);
478 		}
479 
480 		private void terminate(Throwable t) {
481 			if (! terminated.compareAndSet(false, true)) // HTT: false branch requires concurrent double-terminate
482 				return;
483 			withMdc(() -> {
484 				if (t != null && ! wrote) {
485 					try {
486 						if (! res.getHttpServletResponse().isCommitted()) // HTT: true branch (isCommitted=false) only via async path
487 							res.getHttpServletResponse().sendError(SC_INTERNAL_SERVER_ERROR);
488 					} catch (IOException e) {
489 						LOG.log(Level.FINEST, e, () -> "sendError on pre-write stream failure failed: " + e.getMessage());
490 					}
491 				}
492 			});
493 			onTerminate.accept(t);
494 		}
495 
496 		private static Throwable unwrap(Throwable t) { // HTT: only called when writeFrame throws RuntimeException (serializer failure)
497 			if ((t instanceof IllegalStateException || t instanceof CompletionException) && t.getCause() != null)
498 				return t.getCause();
499 			return t;
500 		}
