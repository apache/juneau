154 	 * Constructor.
155 	 */
156 	@SuppressWarnings({
157 		"java:S3776", // Cognitive complexity acceptable for response initialization
158 		"java:S112"   // throws Exception intentional - callback/lifecycle method
159 	})
160 	RestResponse(RestOpContext opContext, RestSession session, RestRequest req) throws Exception {
161 		super(session.getResponse());
162 
163 		inner = session.getResponse();
164 		request = req;
165 
166 		this.opContext = opContext;
167 		responseBeanMeta = opContext.getResponseMeta();
168 
169 		var context = session.getContext();
170 
171 		try {
172 			var passThroughHeaders = request.getHeaderParam("x-response-headers").orElse(null);
173 			if (nn(passThroughHeaders)) {
174 				var m = context.getPartParser().getPartSession().read(HEADER, null, passThroughHeaders, MarshallingContext.DEFAULT.getClassMeta(JsonMap.class));
175 				for (var e : m.entrySet())
176 					addHeader(e.getKey(), resolveUris(e.getValue()));
177 			}
178 		} catch (Exception e1) {
179 			throw new BadRequest(e1, "Invalid format for header 'x-response-headers'.  Must be in URL-encoded format.");
180 		}
181 
182 		// Find acceptable charset
183 		var h = request.getHeaderParam("accept-charset").orElse(null);
184 		Charset charset = null;
185 		if (h == null)
186 			charset = opContext.getDefaultCharset();
187 		else
188 			for (var r : StringRanges.of(h).toList()) {
189 				if (r.getQValue() > 0) {
190 					if (r.getName().equals("*"))
191 						charset = opContext.getDefaultCharset();
192 					else if (Charset.isSupported(r.getName()))
193 						charset = Charset.forName(r.getName());
194 					if (nn(charset))
195 						break;
196 				}
197 			}
198 
199 		request.getContext().getDefaultResponseHeaders().forEach(x -> addHeader(x.getName(), resolveCspNonce(maybeResolveUris(x.getName(), x.getValue()))));  // Done this way to avoid list/array copy.
200 
201 		opContext.getDefaultResponseHeaders().forEach(x -> addHeader(x.getName(), resolveCspNonce(maybeResolveUris(x.getName(), x.getValue()))));
202 
203 		if (charset == null)
204 			throw new NotAcceptable("No supported charsets in header 'Accept-Charset': '%s'", request.getHeaderParam("Accept-Charset").orElse(null));
205 		inner.setCharacterEncoding(charset.name());
206 
207 	}

209 	/**
210 	 * Adds a response header.
211 	 *
212 	 * <p>
213 	 * Any previous header values are preserved.
214 	 *
215 	 * <p>
216 	 * Value is added at the end of the headers.
217 	 *
218 	 * <p>
219 	 * If the header is a {@link HttpUriHeader}, the URI will be resolved using the {@link RestRequest#getUriResolver()} object.
220 	 *
221 	 * <p>
222 	 * Note that per <a class='doclink' href='https://www.w3.org/Protocols/rfc2616/rfc2616-sec4.html#sec4.2'>RFC2616</a>,
223 	 * only headers defined as comma-delimited lists [i.e., #(values)] should be defined as multiple message header fields.
224 	 *
225 	 * @param header The header.
226 	 * 	<br>Can be <jk>null</jk> (ignored).
227 	 * @return This object.
228 	 */
229 	public RestResponse addHeader(HttpHeader header) {
230 		if (header == null) {
231 			// Do nothing.
232 		} else if (header instanceof HttpUriHeader header2) {
233 			addHeader(header2.getName(), resolveUris(header2.getValue()));
234 		} else {
235 			addHeader(header.getName(), header.getValue());
236 		}
237 		return this;
238 	}

240 	/**
241 	 * Adds a response header with the given name and value.
242 	 *
243 	 * <p>
244 	 * This method allows response headers to have multiple values.
245 	 *
246 	 * <p>
247 	 * A no-op of either the name or value is <jk>null</jk>.
248 	 *
249 	 * <p>
250 	 * Note that per <a class='doclink' href='https://www.w3.org/Protocols/rfc2616/rfc2616-sec4.html#sec4.2'>RFC2616</a>,
251 	 * only headers defined as comma-delimited lists [i.e., #(values)] should be defined as multiple message header fields.
252 	 *
253 	 * @param name The header name.
254 	 * @param value The header value.
255 	 */
256 	@Override
257 	public void addHeader(String name, String value) {
258 		if (nn(name) && nn(value)) {
259 			if (eqic(name, HEADER_ContentType))
260 				setHeader(name, value);
261 			else {
262 				if (safeHeaders)
263 					value = stripInvalidHttpHeaderChars(value);
264 				value = abbreviate(value, maxHeaderLength);
265 				inner.addHeader(name, value);
266 			}
267 		}
268 	}

395 	/**
396 	 * Equivalent to {@link HttpServletResponse#getOutputStream()}, except wraps the output stream if an {@link Encoder}
397 	 * was found that matched the <c>Accept-Encoding</c> header.
398 	 *
399 	 * @return A negotiated output stream.
400 	 * @throws NotAcceptable If unsupported Accept-Encoding value specified.
401 	 * @throws IOException Thrown by underlying stream.
402 	 */
403 	@SuppressWarnings({
404 		"java:S3776" // Cognitive complexity acceptable for content negotiation
405 	})
406 	public FinishableServletOutputStream getNegotiatedOutputStream() throws NotAcceptable, IOException {
407 		if (os == null) {
408 			Encoder encoder = null;
409 			var encoders = request.getOpContext().getEncoders();
410 
411 			var ae = request.getHeaderParam("Accept-Encoding").orElse(null);
412 			if (ine(ae)) {
413 				var match = encoders.getEncoderMatch(ae);
414 				if (match == null) {
415 					// Identity should always match unless "identity;q=0" or "*;q=0" is specified.
416 					if (ENCODING_DISABLED_PATTERN.matcher(ae).find()) {
417 						throw new NotAcceptable("Unsupported encoding in request header 'Accept-Encoding': '%s'\n\tSupported codings: %s", ae, Json5.of(encoders.getSupportedEncodings()));
418 					}
419 				} else {
420 					encoder = match.getEncoder();
421 					var encoding = match.getEncoding();
422 
423 					// Some clients don't recognize identity as an encoding, so don't set it.
424 					if (! encoding.equals("identity"))
425 						setHeader("content-encoding", encoding);
426 				}
427 			}
428 			var sos2 = getOutputStream();
429 			os = new FinishableServletOutputStream(encoder == null ? sos2 : encoder.getOutputStream(sos2));
430 		}
431 		return os;
432 	}

611 	/**
612 	 * Returns <jk>true</jk> if <c>uri</c> is safe to pass to {@link #sendRedirect(String) sendRedirect(String)}
613 	 * without risk of redirecting to an unintended external site: either a relative path, or an absolute/
614 	 * protocol-relative URI whose host is in <c>allowedHosts</c>.
615 	 *
616 	 * @param uri The candidate redirection URL.
617 	 * @param allowedHosts The hosts (case-insensitive) an absolute or protocol-relative <c>uri</c> is allowed to target.
618 	 * @return <jk>true</jk> if the URI is safe to redirect to.
619 	 */
620 	static boolean isSafeRedirectUri(String uri, String...allowedHosts) {
621 		if (uri == null || uri.isEmpty())
622 			return true;
623 		if (uri.charAt(0) == '/' && ! uri.startsWith("//"))
624 			return true;
625 		if (uri.indexOf("://") == -1 && ! uri.startsWith("//"))
626 			return true;
627 		String host;
628 		try {
629 			host = new java.net.URI(uri.startsWith("//") ? "http:" + uri : uri).getHost();
630 		} catch (java.net.URISyntaxException e) {
631 			return false;
632 		}
633 		if (host == null || allowedHosts == null)
634 			return false;
635 		for (var h : allowedHosts)
636 			if (host.equalsIgnoreCase(h))
637 				return true;
638 		return false;
639 	}

1132 		if (value == null || ! value.contains(ContentSecurityPolicy.Builder.NONCE_PLACEHOLDER))
1133 			return value;
1134 		var nonce = request.getAttribute(RestRequest.CSP_NONCE_ATTR).as(String.class).orElse(null);
1135 		if (nonce == null) {
1136 			nonce = ContentSecurityPolicy.Builder.generateNonce();
1137 			request.setAttribute(RestRequest.CSP_NONCE_ATTR, nonce);
1138 		}
1139 		return ContentSecurityPolicy.Builder.resolveNonce(value, nonce);
1140 	}
