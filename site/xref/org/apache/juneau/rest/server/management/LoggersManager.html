17  package org.apache.juneau.rest.server.management;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.rest.server.*;
22  
23  /**
24   * Shared worker for the {@code /loggers} management endpoint:  reads and sets logger levels at runtime by
25   * delegating to the {@link LogBackend} resolved from the host context.
26   *
27   * <p>
28   * Both the {@link LoggersMixin mixin} and {@link LoggersResource resource} flavors delegate here, so the
29   * two forms cannot drift &mdash; the same pattern the health package uses with its aggregator.
30   *
31   * <h5 class='section'>Backend</h5>
32   * <p>
33   * The driven backend defaults to {@link JulLogBackend java.util.logging}.  A consumer drives a different backend
34   * (e.g. Logback / Log4j2 via {@code juneau-rest-server-management-logging}) by <i>explicitly</i> declaring it on a
35   * {@link LoggersSettings} bean via {@link LoggersSettings.Builder#backend(LogBackend)} &mdash; the endpoint never
36   * auto-detects.  The response shape (logger name &rarr; configured level, empty string = inherited) is identical
37   * across backends.
38   *
39   * <h5 class='section'>See Also:</h5><ul>
40   * 	<li class='jc'>{@link LogBackend}
41   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/ManagementSurface">Management Surface</a>
42   * </ul>
43   *
44   * @since 10.0.0
45   */
46  public class LoggersManager {
47  
48  	/**
49  	 * Returns a snapshot of all known loggers and their configured levels, for the backend resolved from the
50  	 * supplied context.
51  	 *
52  	 * @param context The REST context whose bean store supplies the {@link LoggersSettings} / {@link LogBackend}.
53  	 * 	May be <jk>null</jk> (uses the JUL default).
54  	 * @return A sorted map of logger name &rarr; configured level name (empty string = inherited).  Never <jk>null</jk>.
55  	 */
56  	public Map<String,String> getLevels(RestContext context) {
57  		return resolveSettings(context).getBackend().getLevels();
58  	}
59  
60  	/**
61  	 * Returns the configured level of a single logger, for the backend resolved from the supplied context.
62  	 *
63  	 * @param context The REST context whose bean store supplies the backend.  May be <jk>null</jk>.
64  	 * @param name The logger name ({@code "ROOT"} or empty for the root logger).
65  	 * @return The configured level name, the empty string if the level is inherited, or <jk>null</jk> if no
66  	 * 	such logger is currently registered.
67  	 */
68  	public String getLevel(RestContext context, String name) {
69  		return resolveSettings(context).getBackend().getLevel(name);
70  	}
71  
72  	/**
73  	 * Sets (or clears) the level of a single logger at runtime (process-lifetime-only), for the backend resolved
74  	 * from the supplied context.
75  	 *
76  	 * @param context The REST context whose bean store supplies the backend.  May be <jk>null</jk>.
77  	 * @param name The logger name ({@code "ROOT"} or empty for the root logger).
78  	 * @param level The backend-native level name (e.g. {@code "FINE"}/{@code "INFO"} for JUL,
79  	 * 	{@code "DEBUG"}/{@code "INFO"} for Logback/Log4j2), or <jk>null</jk>/blank to inherit.
80  	 * @throws IllegalArgumentException If {@code level} is non-blank but not a valid level name for the backend.
81  	 */
82  	public void setLevel(RestContext context, String name, String level) {
83  		resolveSettings(context).getBackend().setLevel(name, level);
84  	}
85  
86  	/**
87  	 * Resolves the {@link LoggersSettings} from the host context's bean store, falling back to the
88  	 * read-only (JUL-backed) default.
89  	 *
90  	 * @param context The REST context whose bean store is searched.  May be <jk>null</jk>.
91  	 * @return The registered settings, or {@link LoggersSettings#DEFAULT} when none is registered.
92  	 */
93  	@SuppressWarnings({
94  		"resource" // The bean store is owned by the RestContext; this adapter only borrows beans and must not close it.
95  	})
96  	public LoggersSettings resolveSettings(RestContext context) {
97  		if (context == null)
98  			return LoggersSettings.DEFAULT;
99  		return context.getBeanStore().getBean(LoggersSettings.class).orElse(LoggersSettings.DEFAULT);
100 	}
101 }
