17  package org.apache.juneau.rest.server.management;
18  
19  /**
20   * Exposure settings for the {@code /loggers} endpoint.
21   *
22   * <p>
23   * The {@code GET} (read) endpoints are always exposed; the {@code PUT}/{@code POST} <b>set-level</b> endpoints
24   * <b>mutate runtime logging</b> and are therefore <b>deny-by-default</b> on the same sensitive-by-default
25   * posture as the diagnostic dumps.  A consumer opts in <i>explicitly</i> by registering a
26   * {@code LoggersSettings} bean (built with {@link Builder#enableWrite()}) in the resource's bean store.  When
27   * no such bean is present, {@link LoggersManager} resolves the default (writes disabled) and the set-level
28   * endpoints respond {@code 403 Forbidden}.
29   *
30   * <h5 class='topic'>Backend selection (explicit)</h5>
31   *
32   * <p>
33   * The {@code /loggers} endpoint drives a {@link LogBackend}.  By default that is
34   * {@link JulLogBackend java.util.logging}.  To drive a different backend (e.g. Logback or Log4j2 via the
35   * {@code juneau-rest-server-management-logging} add-on), declare it <i>explicitly</i> with
36   * {@link Builder#backend(LogBackend)} &mdash; the endpoint never classpath-scans for, and then drives, a backend
37   * the operator did not choose.
38   *
39   * <h5 class='section'>See Also:</h5><ul>
40   * 	<li class='jc'>{@link LoggersManager}
41   * 	<li class='jc'>{@link LogBackend}
42   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/ManagementSurface">Management Surface</a>
43   * </ul>
44   *
45   * @since 10.0.0
46   */
47  public class LoggersSettings {
48  
49  	/** The default (read-only, JUL-backed) settings used when no bean is registered. */
50  	public static final LoggersSettings DEFAULT = create().build();
51  
52  	private final boolean writeEnabled;
53  	private final LogBackend backend;
54  
55  	private LoggersSettings(Builder b) {
56  		this.writeEnabled = b.writeEnabled;
57  		this.backend = b.backend == null ? JulLogBackend.INSTANCE : b.backend;
58  	}
59  
60  	/**
61  	 * Builder creator.
62  	 *
63  	 * @return A new builder (writes disabled).
64  	 */
65  	public static Builder create() {
66  		return new Builder();
67  	}
68  
69  	/**
70  	 * @return <jk>true</jk> if the {@code PUT}/{@code POST} set-level endpoints are enabled.
71  	 */
72  	public boolean isWriteEnabled() {
73  		return writeEnabled;
74  	}
75  
76  	/**
77  	 * @return The explicitly-declared {@link LogBackend} the {@code /loggers} endpoint drives, or
78  	 * 	{@link JulLogBackend#INSTANCE} when none was declared.  Never <jk>null</jk>.
79  	 */
80  	public LogBackend getBackend() {
81  		return backend;
82  	}
83  
84  	/**
85  	 * Builder for {@link LoggersSettings}.
86  	 */
87  	public static class Builder {
88  		private boolean writeEnabled;
89  		private LogBackend backend;
90  
91  		/**
92  		 * Enables the {@code PUT}/{@code POST} set-level endpoints.
93  		 *
94  		 * @return This object.
95  		 */
96  		public Builder enableWrite() {
97  			writeEnabled = true;
98  			return this;
99  		}
100 
101 		/**
102 		 * Explicitly declares the {@link LogBackend} the {@code /loggers} endpoint drives.
103 		 *
104 		 * <p>
105 		 * When not set, the endpoint drives {@link JulLogBackend java.util.logging}.  Pass a Logback / Log4j2
106 		 * backend (from {@code juneau-rest-server-management-logging}) or a custom implementation to drive a
107 		 * different backend &mdash; this is the explicit opt-in; the endpoint does not auto-detect.
108 		 *
109 		 * @param value The backend.  <jk>null</jk> resets to the JUL default.
110 		 * @return This object.
111 		 */
112 		public Builder backend(LogBackend value) {
113 			backend = value;
114 			return this;
115 		}
116 
117 		/**
118 		 * Builds the settings.
119 		 *
120 		 * @return A new {@link LoggersSettings}.
121 		 */
122 		public LoggersSettings build() {
123 			return new LoggersSettings(this);
124 		}
125 	}
126 }
