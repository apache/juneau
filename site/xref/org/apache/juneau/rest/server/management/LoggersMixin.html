17  package org.apache.juneau.rest.server.management;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.http.*;
22  import org.apache.juneau.http.response.*;
23  import org.apache.juneau.rest.server.*;
24  import org.apache.juneau.rest.server.servlet.*;
25  
26  /**
27   * Mixin flavor of the {@code /loggers} runtime log-level management endpoint.
28   *
29   * <p>
30   * Composes the logger read/set endpoints into a host resource via
31   * {@link Rest#mixins() @Rest(mixins=LoggersMixin.class)}.  Delegates to a shared {@link LoggersManager}
32   * worker so the mixin and {@link LoggersResource resource} flavors cannot drift.
33   *
34   * <p>
35   * The {@code GET} endpoints (list all / read one) are read-only, but &mdash; unlike the {@code PUT}/{@code POST}
36   * set-level endpoint, which <b>mutates runtime logging</b> and is deny-by-default (returns {@code 403} unless a
37   * {@link LoggersSettings} bean with writes enabled is registered) &mdash; they carry <b>no built-in guard of their
38   * own</b>: once a resource composes this mixin, {@code GET /loggers} and {@code GET /loggers/{name}} are reachable
39   * by anyone who can reach the resource, exposing the set of registered logger names and their configured levels
40   * (topology information, not credentials or request data). Compose this mixin only into a resource, or a subpath
41   * of one, that's already covered by whatever authentication/authorization the app applies to its management
42   * endpoints generally; this mixin does not add any of its own for the read side.
43   *
44   * <h5 class='section'>Backend:</h5><ul>
45   * 	<li>Drives {@link JulLogBackend java.util.logging} by default; declare a Logback / Log4j2 backend
46   * 		explicitly via {@link LoggersSettings.Builder#backend(LogBackend)} &mdash; see {@link LogBackend}.
47   * </ul>
48   *
49   * <h5 class='section'>See Also:</h5><ul>
50   * 	<li class='jc'>{@link LoggersResource}
51   * 	<li class='jc'>{@link LoggersManager}
52   * 	<li class='jc'>{@link LogBackend}
53   * </ul>
54   *
55   * @since 10.0.0
56   */
57  @Rest
58  public class LoggersMixin extends RestMixin {
59  
60  	private final LoggersManager manager = new LoggersManager();
61  
62  	/**
63  	 * [GET /loggers] - All loggers and their configured levels.
64  	 *
65  	 * @param req The HTTP request.
66  	 * @return A sorted map of logger name to configured level (empty string = inherited).
67  	 */
68  	@RestGet(
69  		path="/loggers",
70  		summary="Runtime logger levels",
71  		description="Lists all loggers and their configured levels (empty = inherited from ancestor)."
72  	)
73  	public Map<String,String> getLoggers(RestRequest req) {
74  		return manager.getLevels(req.getContext());
75  	}
76  
77  	/**
78  	 * [GET /loggers/{name}] - One logger's configured level.
79  	 *
80  	 * @param req The HTTP request.
81  	 * @param name The logger name ("ROOT" for the root logger).
82  	 * @return The configured level name, empty string if inherited.
83  	 * @throws NotFound If no logger with that name is registered.
84  	 */
85  	@RestGet(
86  		path="/loggers/{name}",
87  		summary="Runtime logger level",
88  		description="Returns the configured level of a single logger."
89  	)
90  	public String getLogger(RestRequest req, @Path("name") String name) {
91  		var level = manager.getLevel(req.getContext(), name);
92  		if (level == null)
93  			throw new NotFound("No logger named '%s' is registered.", name);
94  		return level;
95  	}
96  
97  	/**
98  	 * [PUT /loggers/{name}] - Set a logger's level at runtime.
99  	 *
100 	 * <p>
101 	 * <b>Sensitive (mutating) &mdash; deny-by-default:</b> responds {@code 403} unless a {@link LoggersSettings}
102 	 * bean with writes enabled is registered.
103 	 *
104 	 * @param req The HTTP request.
105 	 * @param name The logger name ("ROOT" for the root logger).
106 	 * @param level The level name (e.g. FINE, INFO, OFF), or blank to inherit from the ancestor.
107 	 * @return The newly-configured level (empty string if inherited).
108 	 * @throws Forbidden If the set-level endpoint is not enabled.
109 	 */
110 	@RestPut(
111 		path="/loggers/{name}",
112 		summary="Set logger level",
113 		description="Sets the java.util.logging level of a single logger at runtime (blank to inherit).  Disabled by default; opt in via LoggersSettings."
114 	)
115 	public String setLogger(RestRequest req, @Path("name") String name, @Content String level) {
116 		if (! manager.resolveSettings(req.getContext()).isWriteEnabled())
117 			throw new Forbidden("The /loggers set-level endpoint is disabled.  Register a LoggersSettings bean with write enabled to use it.");
118 		manager.setLevel(req.getContext(), name, level);
119 		// setLevel creates the logger on demand, so getLevel always returns non-null here (level name or "" if inherited).
120 		return manager.getLevel(req.getContext(), name);
121 	}
122 
123 	/**
124 	 * [POST /loggers/{name}] - Set a logger's level at runtime (POST alias of {@link #setLogger(RestRequest,String,String)}).
125 	 *
126 	 * @param req The HTTP request.
127 	 * @param name The logger name ("ROOT" for the root logger).
128 	 * @param level The level name, or blank to inherit.
129 	 * @return The newly-configured level (empty string if inherited).
130 	 * @throws Forbidden If the set-level endpoint is not enabled.
131 	 */
132 	@RestPost(
133 		path="/loggers/{name}",
134 		summary="Set logger level",
135 		description="Sets the java.util.logging level of a single logger at runtime (blank to inherit).  Disabled by default; opt in via LoggersSettings."
136 	)
137 	public String setLoggerPost(RestRequest req, @Path("name") String name, @Content String level) {
138 		return setLogger(req, name, level);
139 	}
140 }
