17  package org.apache.juneau.rest.server.management;
18  
19  import java.util.*;
20  
21  /**
22   * SPI behind the {@code /loggers} management endpoint: reads and sets logger levels for one concrete logging
23   * backend at runtime.
24   *
25   * <p>
26   * {@code juneau-rest-server} ships only the {@link JulLogBackend java.util.logging} implementation, keeping the
27   * core management surface dependency-free. Support for other backends is opt-in: the
28   * {@code juneau-rest-server-management-logging} add-on ships {@code LogbackLogBackend} / {@code Log4j2LogBackend}
29   * (with the backends declared {@code provided}), and any consumer can implement this interface for a backend
30   * Juneau does not cover.
31   *
32   * <h5 class='topic'>Explicit selection (no auto-drive)</h5>
33   *
34   * <p>
35   * Consistent with Juneau's explicit-over-magic stance, the backend is <b>explicitly declared</b> by registering a
36   * {@link LoggersSettings} bean built with {@link LoggersSettings.Builder#backend(LogBackend)}. When none is
37   * declared, {@link LoggersManager} drives {@link JulLogBackend} &mdash; the endpoint never silently classpath-scans
38   * for, and then drives, a backend the operator didn't choose.
39   *
40   * <h5 class='topic'>Level contract</h5>
41   *
42   * <p>
43   * Levels are exchanged as backend-native level-name strings (e.g. JUL {@code "FINE"}/{@code "INFO"}, Logback /
44   * Log4j2 {@code "DEBUG"}/{@code "INFO"}). Reads report the logger's <b>configured</b> level &mdash; the level set
45   * on the logger itself, or the empty string when it inherits from an ancestor. Sets are
46   * <b>process-lifetime-only</b> (they do not rewrite the backend's configuration file). The root logger is
47   * addressed under the key {@code "ROOT"}.
48   *
49   * <h5 class='section'>See Also:</h5><ul>
50   * 	<li class='jc'>{@link JulLogBackend}
51   * 	<li class='jc'>{@link LoggersManager}
52   * 	<li class='jc'>{@link LoggersSettings}
53   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/ManagementSurface">Management Surface</a>
54   * </ul>
55   *
56   * @since 10.0.0
57   */
58  public interface LogBackend {
59  
60  	/**
61  	 * Returns a snapshot of all known loggers and their configured levels.
62  	 *
63  	 * @return A sorted map of logger name &rarr; configured level name (empty string = inherited from an
64  	 * 	ancestor). The root logger is keyed {@code "ROOT"}. Never <jk>null</jk>.
65  	 */
66  	Map<String,String> getLevels();
67  
68  	/**
69  	 * Returns the configured level of a single logger.
70  	 *
71  	 * @param name The logger name ({@code "ROOT"} or empty for the root logger).
72  	 * @return The configured level name, the empty string if the level is inherited, or <jk>null</jk> if no such
73  	 * 	logger is currently registered.
74  	 */
75  	String getLevel(String name);
76  
77  	/**
78  	 * Sets (or clears) the level of a single logger at runtime (process-lifetime-only).
79  	 *
80  	 * @param name The logger name ({@code "ROOT"} or empty for the root logger).
81  	 * @param level The backend-native level name, or <jk>null</jk>/blank to clear the logger's own level so it
82  	 * 	inherits from its ancestor.
83  	 * @throws IllegalArgumentException If {@code level} is non-blank but not a valid level name for the backend.
84  	 */
85  	void setLevel(String name, String level);
86  }
