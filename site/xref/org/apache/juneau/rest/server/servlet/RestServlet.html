41  /**
42   * Servlet implementation of a REST resource.
43   *
44   * <p>
45   * 	The {@link RestServlet} class is the entry point for your REST resources.
46   * 	It extends directly from <l>HttpServlet</l> and is deployed like any other servlet.
47   * </p>
48   * <p>
49   * 	When the servlet <l>init()</l> method is called, it triggers the code to find and process the <l>@Rest</l>
50   * 	annotations on that class and all child classes.
51   * 	These get constructed into a {@link RestContext} object that holds all the configuration
52   * 	information about your resource in a read-only object.
53   * </p>
54   *
55   * <h5 class='section'>Notes:</h5><ul>
56   * 	<li class='note'>
57   * 		Users will typically extend from {@link BasicRestServlet} or {@link BasicRestServletGroup}
58   * 		instead of this class directly.
59   * </ul>
60   *
61   * <h5 class='section'>See Also:</h5><ul>
62   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestAnnotatedClasses">@Rest-Annotated Class Basics</a>
63   * </ul>
64   *
65   * @serial exclude
66   */

163 	 * Returns the read-only context object that contains all the configuration information about this resource.
164 	 *
165 	 * <p>

171 	 * access to this context object:
172 	 * <p class='bjava'>

174 	 * </p>
175 	 *
176 	 * @return The context information on this servlet.
177 	 */
178 	@SuppressWarnings({
179 		"java:S2886" // Reads the AtomicReference 'context' field, which already guarantees atomicity and cross-thread visibility (same as a volatile read); synchronizing this single read with the setContext()/init() lock would add hot-path contention with no correctness benefit.

183 		if (rc == null)
184 			throw new InternalServerError("RestContext object not set on resource.");
185 		return rc;

188 	/**
189 	 * Returns the path for this resource as defined by the @Rest(path) annotation or RestContext.Builder.path(String) method
190 	 * concatenated with those on all parent classes.
191 	 *
192 	 * @return The path defined on this servlet, or an empty string if not specified.
193 	 */
194 	@SuppressWarnings({
195 		"java:S2886" // Reads the AtomicReference 'context' field, which already guarantees atomicity and cross-thread visibility (same as a volatile read); synchronizing this single read with the setContext()/init() lock would add hot-path contention with no correctness benefit.
196 	})
197 	public String getPath() {
198 		var context2 = this.context.get();
199 		if (nn(context2))
200 			return context2.getFullPath();
201 		var ci = ClassInfo.of(getClass());
202 		// @formatter:off
203 		return rstream(AP.find(Rest.class, ci))
204 			.map(x -> x.inner().path())
205 			.filter(Shorts::ine)
206 			.map(StringUtils::trimSlashes)
207 			.findFirst()
208 			.orElse("");
209 		// @formatter:on
210 	}

212 	/**
213 	 * Returns the runtime-overridden top-level mount paths for this resource.
214 	 *
215 	 * <p>
216 	 * Subclasses can override this method to substitute the {@link Rest#paths() @Rest(paths)} annotation
217 	 * defaults at construction time. This is the &quot;getter&quot; rung in the runtime-override resolution
218 	 * chain documented on {@link RestContext#getPaths()}:
219 	 * <ol>
220 	 * 	<li>Programmatic ({@code RestContext.Builder.paths(String...)}) &mdash; highest precedence.
221 	 * 	<li>This {@code getPaths()} getter (when non-{@code null}).
222 	 * 	<li>{@code @Rest(paths={...})} annotation default (SVL-resolved per element, comma-split) &mdash;
223 	 * 		lowest precedence.
224 	 * </ol>
225 	 *
226 	 * <h5 class='section'>Accepted return shapes:</h5><ul>
227 	 * 	<li>{@code null} &mdash; no override; resolution falls through to the {@code @Rest(paths=...)}
228 	 * 		annotation default.
229 	 * 	<li>{@link String} &mdash; a single path or a comma-delimited list, e.g.
230 	 * 		{@code "/healthz,/readyz"}.
231 	 * 	<li>{@code String[]} &mdash; each element may itself be a comma-delimited list.
232 	 * 	<li>{@link java.util.Collection Collection}, {@link java.util.List List},
233 	 * 		{@link java.util.Set Set}, {@link Iterable}, or {@link java.util.stream.Stream Stream}
234 	 * 		of any of the above (nested mixes are recursively flattened).
235 	 * 	<li>Primitive arrays (boxed during flattening).
236 	 * </ul>
237 	 *
238 	 * <p>
239 	 * Non-{@link String} leaves are coerced via {@link String#valueOf(Object)}.  Each leaf string then runs
240 	 * through the same per-element pipeline used for {@code @Rest(paths=...)} annotation elements: SVL is
241 	 * applied via the bean-store-backed {@link org.apache.juneau.commons.svl.VarResolverSession VarResolverSession}
242 	 * (so {@code $C{key}} consults the live {@link org.apache.juneau.config.Config Config}, {@code $E{NAME,default}}
243 	 * and {@code $S{prop,default}} resolve through the bootstrap variable catalog), and the post-SVL value is
244 	 * comma-split with each piece trimmed and empty pieces dropped.
245 	 *
246 	 * <h5 class='section'>Notes:</h5><ul>
247 	 * 	<li class='note'>
248 	 * 		Returning {@code null} (the default) means &quot;inherit annotation&quot; &mdash; the next rung resolves.
249 	 * 	<li class='note'>
250 	 * 		Returning {@code new String[0]}, an empty collection, an empty string, or a blank string means
251 	 * 		&quot;explicitly clear&quot; &mdash; no top-level mounts.
252 	 * 	<li class='note'>
253 	 * 		The scalar accessor {@link #getPath()} remains orthogonal to this method &mdash; the singular
254 	 * 		{@code @Rest(path=...)} attribute has its own resolution chain and is unaffected.
255 	 * </ul>
256 	 *
257 	 * @return The runtime-overridden mount paths in any accepted shape, or {@code null} to inherit the
258 	 * 	annotation default.
259 	 * @since 10.0.0
260 	 */
261 	public Object getPaths() { return null; }

277 	@Override /* Overridden from Servlet */
278 	@SuppressWarnings({
279 		"java:S2654" // Guards the non-atomic check-then-act one-time initialization (construct + postInit()/postInitChildFirst() side effects) of the shared 'context' field against concurrent re-entry; lifecycle-only, not on the request-serving hot path.
280 	})
281 	public synchronized void init(ServletConfig servletConfig) throws ServletException {
282 		try {
283 			if (nn(context.get()))
284 				return;
285 			super.init(servletConfig);
286 			context.set(new RestContext(new RestContext.Args(this.getClass(), null, servletConfig, () -> this, "", null, null, null, RestContext.ContextKind.ROOT, restBuilder)));
287 			context.get().postInit();
288 			context.get().postInitChildFirst();
289 		} catch (ServletException e) {
290 			initException.set(e);
291 			log(SEVERE, e, MSG_servletInitError, cn(this));
292 			throw e;
293 		} catch (BasicHttpException e) {
294 			initException.set(e);
295 			log(SEVERE, e, MSG_servletInitError, cn(this));
296 		} catch (Exception e) {
297 			initException.set(new InternalServerError(e));
298 			log(SEVERE, e, MSG_servletInitError, cn(this));
299 		}
300 	}

332 	 * The main service method.
333 	 *
334 	 * <p>
335 	 * Subclasses can optionally override this method if they want to tailor the behavior of requests.
336 	 */
337 	@Override /* Overridden from Servlet */
338 	public void service(HttpServletRequest r1, HttpServletResponse r2) throws ServletException, InternalServerError, IOException {
339 		try {
340 			if (nn(initException.get()))
341 				throw initException.get();
342 			if (context.get() == null)
343 				throw new InternalServerError(
344 					"Servlet %s not initialized.  init(ServletConfig) was not called.  This can occur if you've overridden this method but didn't call super.init(RestConfig).", cn(this));
345 			getContext().execute(this, r1, r2);
346 
347 		} catch (Exception e) {
348 			r2.sendError(SC_INTERNAL_SERVER_ERROR, localizedMessage(e));
349 		}
350 	}

353 	 * Main logger method.
354 	 *
355 	 * <p>
356 	 * The default behavior logs a message to the Java logger of the class name.
357 	 *
358 	 * <p>
359 	 * Subclasses can override this method to implement their own logger handling.
360 	 *
361 	 * @param level The log level.
362 	 * @param cause Optional throwable.
363 	 * @param msg The message to log.
364 	 */
365 	protected void doLog(Level level, Throwable cause, Supplier<String> msg) {
366 		var c = context.get();
367 		var logger = c == null ? null : c.getLogger();
368 		if (logger == null)
369 			logger = Logger.getLogger(cn(this));
370 		logger.log(level, cause, msg);
371 	}

393 	 * Fluent builder for programmatically configuring a {@link RestServlet} subclass.
394 	 *
395 	 * <p>
396 	 * This is the subclassable, self-typed (CRTP) flavor builder.  Its {@code SELF} type parameter is left open so
397 	 * a user subclass's bespoke setters chain with true covariant returns alongside the inherited
398 	 * {@link RestBuilder} surface.  For the common case where the builder is not subclassed,
399 	 * use {@link RestServlet#builder(Class)} which returns the concrete {@link DefaultBuilder} leaf.
400 	 *
401 	 * @param <R> The resource type produced by {@link #build()}.
402 	 * @param <SELF> The concrete builder type (self type).
403 	 * @since 10.0.0
404 	 */
405 	@SuppressWarnings({
406 		"java:S119" // 'SELF' (CRTP self-type) is intentional and clearer than a single-letter name.
407 	})
408 	public static class Builder<R extends RestServlet,SELF extends Builder<R,SELF>> extends AbstractRestBuilder<R,SELF> {
409 
410 		/**
411 		 * Constructor.
412 		 *
413 		 * @param type The resource type produced by {@link #build()}.  Must not be <jk>null</jk>.
414 		 */
415 		protected Builder(Class<R> type) {
416 			super(type);
417 		}
418 
419 		@Override /* AbstractRestBuilder */
420 		public R build() {
421 			var r = createResource();
422 			r.restBuilder = this;
423 			return r;
