17  package org.apache.juneau.rest.server.servlet;
18  
19  import java.lang.reflect.*;
20  import java.util.*;
21  
22  import org.apache.juneau.rest.server.*;
23  
24  import jakarta.servlet.*;
25  import jakarta.servlet.annotation.*;
26  
27  /**
28   * Plain Servlet-spec {@link ServletContainerInitializer} that auto-mounts {@link RestServlet}
29   * subclasses at their self-declared top-level paths in non-Spring (e.g. {@code web.xml} / external
30   * WAR) deployments.
31   *
32   * <p>
33   * Discovered by the container via {@code META-INF/services/jakarta.servlet.ServletContainerInitializer}
34   * and (through {@link HandlesTypes @HandlesTypes(RestServlet.class)}) handed every concrete
35   * {@link RestServlet} on the webapp classpath. It mirrors the Jetty-microservice and Spring Boot
36   * auto-registration mechanisms: the mount paths are read from the top-level chain
37   * ({@link Rest#paths() @Rest(paths)} / {@code getPaths()}) via
38   * {@link RestContext#resolveTopLevelPaths(Class, Object, org.apache.juneau.commons.inject.BeanStore)
39   * RestContext.resolveTopLevelPaths(...)} &mdash; <b>not</b> {@code path()} (child-composition).
40   *
41   * <h5 class='section'>Activation:</h5>
42   *
43   * <p>
44   * Opt-in via the context init parameter {@code juneau.rest.auto-register=true} (parity with the
45   * Spring Boot {@code juneau.rest.auto-register} property). When unset or not {@code true}, the
46   * initializer does nothing, so it never collides with servlets the deployer declared manually in
47   * {@code web.xml}. Declare it in {@code web.xml}:
48   *
49   * <p class='bxml'>
50   * 	<xt>&lt;context-param&gt;</xt>
51   * 		<xt>&lt;param-name&gt;</xt>juneau.rest.auto-register<xt>&lt;/param-name&gt;</xt>
52   * 		<xt>&lt;param-value&gt;</xt>true<xt>&lt;/param-value&gt;</xt>
53   * 	<xt>&lt;/context-param&gt;</xt>
54   * </p>
55   *
56   * <h5 class='section'>Behavior:</h5>
57   *
58   * <p>
59   * For each handled class that is concrete, carries a {@link Rest @Rest} annotation, is no-arg
60   * instantiable, and resolves to at least one top-level path, an instance is created and added to the
61   * servlet context at each resolved path. Classes that resolve to no top-level paths (pure mixins /
62   * child-composed resources) or that cannot be instantiated with a no-arg constructor are skipped.
63   *
64   * <h5 class='section'>See Also:</h5><ul>
65   * 	<li class='jm'>{@link RestContext#resolveTopLevelPaths(Class, Object, org.apache.juneau.commons.inject.BeanStore)}
66   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestServerComposition">REST Server &mdash; Mixins and Multi-Mount Paths</a>
67   * </ul>
68   *
69   * @since 10.0.0
70   */
71  @HandlesTypes(RestServlet.class)
72  public class JuneauRestServletContainerInitializer implements ServletContainerInitializer {
73  
74  	/** Context init parameter that opts a webapp into auto-registration. */
75  	public static final String AUTO_REGISTER_PARAM = "juneau.rest.auto-register";
76  
77  	@Override /* ServletContainerInitializer */
78  	public void onStartup(Set<Class<?>> classes, ServletContext ctx) throws ServletException {
79  		if (! "true".equalsIgnoreCase(ctx.getInitParameter(AUTO_REGISTER_PARAM)))
80  			return;
81  		if (classes == null)
82  			return;
83  
84  		for (var c : classes)
85  			tryMount(c, ctx);
86  	}
87  
88  	private static void tryMount(Class<?> c, ServletContext ctx) {
89  		if (! isMountable(c))
90  			return;
91  		var servlet = instantiate(c);
92  		if (servlet == null)
93  			return;
94  		var resolved = RestContext.resolveTopLevelPaths(c, servlet, null);
95  		if (resolved.length == 0)
96  			return;
97  		var reg = ctx.addServlet(c.getName(), servlet);
98  		if (reg == null)
99  			return; // A servlet with this name is already registered (e.g. via web.xml).
100 		for (var path : resolved)
101 			reg.addMapping(toUrlMapping(path));
102 	}
103 
104 	private static boolean isMountable(Class<?> c) {
105 		if (c == null || c.isInterface() || Modifier.isAbstract(c.getModifiers()))
106 			return false;
107 		return c.isAnnotationPresent(Rest.class);
108 	}
109 
110 	private static RestServlet instantiate(Class<?> c) {
111 		try {
112 			return (RestServlet) c.getDeclaredConstructor().newInstance();
113 		} catch (ReflectiveOperationException | RuntimeException e) {
114 			// Best-effort: a servlet that needs constructor-injected collaborators can't be
115 			// auto-mounted by the plain Servlet-spec path and is simply skipped here.
116 			return null;
117 		}
118 	}
119 
120 	private static String toUrlMapping(String rawPath) {
121 		var p = rawPath == null ? "" : rawPath;
122 		while (p.endsWith("/") && p.length() > 1)
123 			p = p.substring(0, p.length() - 1);
124 		if (p.isEmpty())
125 			return "/";
126 		if (! p.startsWith("/"))
127 			p = "/" + p;
128 		return p;
129 	}
130 }
