17  package org.apache.juneau.rest.server;
18  
19  import java.io.*;
20  import java.nio.channels.*;
21  import java.util.*;
22  
23  /**
24   * Internal utility for recognizing failures caused by the client disconnecting before the response completed.
25   *
26   * <p>
27   * A client that closes its connection mid-response (closed browser tab, cancelled request, timed-out proxy) is routine,
28   * not a server error.  Servlet containers surface it as an {@link IOException} from the response output stream or
29   * writer, but the concrete type is container-specific.  This class detects it without a compile-time dependency on any
30   * container.
31   *
32   * <h5 class='section'>Example:</h5>
33   * <p class='bjava'>
34   * 	<jk>try</jk> {
35   * 		<jv>res</jv>.flushBuffer();
36   * 	} <jk>catch</jk> (Exception <jv>e</jv>) {
37   * 		<jk>if</jk> (ClientAborts.<jsm>isClientAbort</jsm>(<jv>e</jv>))
38   * 			<jc>// Routine: the client went away.  Don't record it as an error.</jc>
39   * 			<jk>return</jk>;
40   * 		<jk>throw</jk> <jv>e</jv>;
41   * 	}
42   * </p>
43   *
44   * <p>
45   * Not part of the public REST API; subject to change without notice.
46   *
47   * @since 10.0.0
48   */
49  public final class ClientAborts {
50  
51  	private static final int MAX_DEPTH = 32;
52  
53  	private ClientAborts() {}
54  
55  	/**
56  	 * Returns <jk>true</jk> if the specified throwable, or anything in its cause chain, indicates that the client
57  	 * disconnected.
58  	 *
59  	 * <p>
60  	 * The cause chain is walked up to a bounded depth and guarded against cycles.  A throwable matches if it is:
61  	 * <ul>
62  	 * 	<li>A class whose simple name is <c>ClientAbortException</c> (Tomcat; matched by name so there is no Tomcat dependency).
63  	 * 	<li>An {@link EOFException} (Jetty's <c>EofException</c> extends it).
64  	 * 	<li>A {@link ClosedChannelException}.
65  	 * 	<li>An {@link IOException} whose message contains <js>"broken pipe"</js> or <js>"connection reset"</js> (case-insensitive).
66  	 * </ul>
67  	 *
68  	 * @param t The throwable to test.  Can be <jk>null</jk>.
69  	 * @return <jk>true</jk> if the throwable represents a client abort.
70  	 */
71  	public static boolean isClientAbort(Throwable t) {
72  		return isClientAbort(t, false);
73  	}
74  
75  	/**
76  	 * Returns <jk>true</jk> only if the throwable, or anything in its cause chain, is a servlet-container client-abort type.
77  	 *
78  	 * <p>
79  	 * Matches only classes whose simple name is <c>ClientAbortException</c> (Tomcat, which wraps every response write/flush
80  	 * failure in it) or <c>EofException</c> (Jetty).  Unlike {@link #isClientAbort(Throwable)}, plain {@link EOFException},
81  	 * {@link ClosedChannelException} and "broken pipe"/"connection reset" {@link IOException}s do <b>not</b> match.
82  	 *
83  	 * <p>
84  	 * Use this where the failure could have come from somewhere other than the client socket (e.g. an exception escaping
85  	 * operation execution or response rendering, where a backend call might also have reset).  Use the lenient
86  	 * {@link #isClientAbort(Throwable)} only for response flush paths, where the failure is necessarily on the client socket.
87  	 *
88  	 * @param t The throwable to test.  Can be <jk>null</jk>.
89  	 * @return <jk>true</jk> if the throwable represents a container-reported client abort.
90  	 */
91  	public static boolean isContainerClientAbort(Throwable t) {
92  		return isClientAbort(t, true);
93  	}
94  
95  	private static boolean isClientAbort(Throwable t, boolean strict) {
96  		var seen = Collections.newSetFromMap(new IdentityHashMap<Throwable,Boolean>());
97  		for (var i = 0; t != null && i < MAX_DEPTH && seen.add(t); i++, t = t.getCause()) {
98  			var n = t.getClass().getSimpleName();
99  			if ("ClientAbortException".equals(n) || (strict && "EofException".equals(n)))
100 				return true;
101 			if (strict)
102 				continue;
103 			if (t instanceof EOFException || t instanceof ClosedChannelException)
104 				return true;
105 			if (t instanceof IOException && isAbortMessage(t.getMessage()))
106 				return true;
107 		}
108 		return false;
109 	}
110 
111 	private static boolean isAbortMessage(String m) {
112 		if (m == null)
113 			return false;
114 		var lc = m.toLowerCase(Locale.ROOT);
115 		return lc.contains("broken pipe") || lc.contains("connection reset");
116 	}
117 }
