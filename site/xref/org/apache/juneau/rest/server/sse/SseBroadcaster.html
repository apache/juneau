17  package org.apache.juneau.rest.server.sse;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  import static org.apache.juneau.commons.utils.StringUtils.*;
21  
22  import java.util.concurrent.*;
23  import java.util.logging.*;
24  
25  import org.apache.juneau.marshall.sse.*;
26  
27  /**
28   * In-memory server-side SSE broadcaster.
29   */
30  public class SseBroadcaster {
31  
32  	private static final Logger LOGGER = Logger.getLogger(SseBroadcaster.class.getName());
33  	private static final int DEFAULT_QUEUE_SIZE = 1024;
34  
35  	/**
36  	 * Creates a broadcaster with the default queue size.
37  	 *
38  	 * @return A new broadcaster.
39  	 */
40  	public static SseBroadcaster create() {
41  		return new SseBroadcaster(DEFAULT_QUEUE_SIZE);
42  	}
43  
44  	private final int queueSize;
45  	private final ConcurrentMap<String,SseSubscription> subscriptions;
46  
47  	/**
48  	 * Constructor.
49  	 *
50  	 * @param queueSize The per-subscriber queue size.
51  	 */
52  	public SseBroadcaster(int queueSize) {
53  		if (queueSize <= 0)
54  			throw iaex("queueSize must be greater than 0.");
55  		this.queueSize = queueSize;
56  		subscriptions = new ConcurrentHashMap<>();
57  	}
58  
59  	/**
60  	 * Creates or replaces a subscriber.
61  	 *
62  	 * @param id The subscriber identifier. Must not be <jk>null</jk> or empty.
63  	 * @return A new subscription.
64  	 */
65  	@SuppressWarnings({
66  		"resource" // Returned subscription is caller-owned and closed by the caller/framework.
67  	})
68  	public SseSubscription subscribe(String id) {
69  		if (isEmpty(id))
70  			throw iaex("id cannot be null or empty.");
71  		var subscription = new SseSubscription(id, queueSize, this::removeSubscriber);
72  		var previous = subscriptions.put(id, subscription);
73  		if (previous != null)
74  			previous.close();
75  		return subscription;
76  	}
77  
78  	/**
79  	 * Publishes an event to all active subscribers.
80  	 *
81  	 * @param event The event to publish. Can be <jk>null</jk> (ignored).
82  	 */
83  	public void publish(SseEvent event) {
84  		if (event == null)
85  			return;
86  		subscriptions.values().forEach(x -> {
87  			if (x.offer(event))
88  				LOGGER.fine(() -> "SSE queue overflow for subscriber " + x.getId() + ", dropped oldest event.");
89  		});
90  	}
91  
92  	@SuppressWarnings({
93  		"resource" // Not a leak: this callback runs from SseSubscription.close() itself (closed==true already), or from subscribe()'s replace path where the caller closes 'removed' directly above; the isClosed() guard just prevents a redundant close(), which the static analysis can't see through.
94  	})
95  	void removeSubscriber(String id) {
96  		var removed = subscriptions.remove(id);
97  		if (removed != null && ! removed.isClosed())
98  			removed.close();
99  	}
100 }
