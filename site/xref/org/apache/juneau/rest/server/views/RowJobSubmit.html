17  package org.apache.juneau.rest.server.views;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.http.response.*;
22  import org.apache.juneau.marshall.collections.*;
23  
24  /**
25   * The action-envelope submit of a row job: {@code {action, targetId, idempotencyKey, fields}}, the shape the
26   * row-action dialog and the job-log card's {@code envelope: "action"} both send.
27   *
28   * <p>
29   * {@link #parse(JsonMap)} checks the shape; {@link #verify(JobKeyLedger, String, String)} checks the action, then the
30   * key's recorded binding.  The row identity is {@link #targetId()} plus that binding &mdash; never anything in
31   * {@link #fields()}, which is user-tamperable request data.  An endpoint that also carries the row id in its path
32   * should check {@code targetId().equals(pathId)} first.
33   *
34   * <h5 class='section'>Example:</h5>
35   * <p class='bjava'>
36   * 	<jk>var</jk> <jv>submit</jv> = RowJobSubmit.<jsm>parse</jsm>(<jv>json</jv>);
37   * 	<jk>if</jk> (! <jv>submit</jv>.targetId().equals(<jv>id</jv>))
38   * 		<jk>return</jk> refuse(<jv>res</jv>, RowJobSubmit.<jsm>bindingRefusal</jsm>());
39   * 	<jk>var</jk> <jv>grant</jv> = <jv>submit</jv>.verify(<jv>jobKeys</jv>, <js>"restock-species"</js>, <jv>actor</jv>);
40   * 	<jk>if</jk> (<jv>grant</jv>.isEmpty())
41   * 		<jk>return</jk> refuse(<jv>res</jv>, RowJobSubmit.<jsm>bindingRefusal</jsm>());  <jc>// answered 409</jc>
42   * </p>
43   *
44   * <h5 class='section'>See Also:</h5>
45   * <ul>
46   * 	<li class='jc'>{@link JobKeyLedger}
47   * </ul>
48   *
49   * @param action The submitted action id.
50   * @param targetId The submitted row id.
51   * @param idempotencyKey The submitted key value.
52   * @param fields The submitted fields; empty when the body omitted them.
53   *
54   * @since 10.0.0
55   */
56  public record RowJobSubmit(String action, String targetId, String idempotencyKey, JsonMap fields) {
57  
58  	/** The HTTP status to answer {@link #bindingRefusal()} with. */
59  	public static final int REFUSAL_STATUS = 409;
60  
61  	/**
62  	 * Parses the envelope.
63  	 *
64  	 * @param body The request body.
65  	 * @return The submit.
66  	 * @throws BadRequest On a non-object body, a blank {@code action}, {@code targetId} or {@code idempotencyKey}, or
67  	 * 	a {@code fields} member that is not an object.  A missing {@code fields} reads as empty.
68  	 */
69  	public static RowJobSubmit parse(JsonMap body) {
70  		if (body == null)
71  			throw new BadRequest("Expected a JSON object body");
72  		var action = text(body, "action");
73  		var targetId = text(body, "targetId");
74  		var key = text(body, "idempotencyKey");
75  		var f = body.get("fields");
76  		JsonMap fields;
77  		if (f == null)
78  			fields = new JsonMap();
79  		else if (f instanceof Map<?,?> m)
80  			fields = new JsonMap(m);
81  		else
82  			throw new BadRequest("'fields' must be an object");
83  		return new RowJobSubmit(action, targetId, key, fields);
84  	}
85  
86  	private static String text(JsonMap body, String name) {
87  		var v = body.get(name);
88  		if (! (v instanceof CharSequence cs) || cs.toString().isBlank())
89  			throw new BadRequest("Missing '%s'", name);
90  		return cs.toString();
91  	}
92  
93  	/**
94  	 * Checks the action, then the key's binding to {@code (action, targetId)}.
95  	 *
96  	 * @param ledger The ledger the key was issued from.
97  	 * @param expectedAction The action this endpoint performs.
98  	 * @return The grant, or empty when the action differs or the key does not verify.
99  	 */
100 	public Optional<JobKeyLedger.Grant> verify(JobKeyLedger ledger, String expectedAction) {
101 		if (! Objects.equals(action, expectedAction))
102 			return Optional.empty();
103 		return ledger.verify(idempotencyKey, action, targetId);
104 	}
105 
106 	/**
107 	 * Checks the action, then the key's binding to {@code (action, targetId, principal)}.
108 	 *
109 	 * @param ledger The ledger the key was issued from.
110 	 * @param expectedAction The action this endpoint performs.
111 	 * @param principal The actor, resolved the same way the key endpoint resolved it.
112 	 * @return The grant, or empty when the action differs or the key does not verify.
113 	 */
114 	public Optional<JobKeyLedger.Grant> verify(JobKeyLedger ledger, String expectedAction, String principal) {
115 		if (! Objects.equals(action, expectedAction))
116 			return Optional.empty();
117 		return ledger.verify(idempotencyKey, action, targetId, principal);
118 	}
119 
120 	/**
121 	 * The standard refusal for a failed verify; answer it with {@link #REFUSAL_STATUS}.
122 	 *
123 	 * <p>
124 	 * One code and one message for every kind of miss (see {@link JobKeyLedger}).
125 	 *
126 	 * @return A new refusal.
127 	 */
128 	public static ActionResult bindingRefusal() {
129 		return ActionResult.refusal(JobKeyLedger.REFUSAL_CODE).message("This key is not valid for this row.");
130 	}
131 
132 	@Override /* Object */
133 	public String toString() {
134 		return "RowJobSubmit(action=" + action + ",targetId=" + targetId + ",idempotencyKey=<redacted>,fields=" + fields + ")";
135 	}
136 }
