17  package org.apache.juneau.rest.server.views;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.marshall.collections.*;
24  
25  /**
26   * Row-detail panel for a table: the per-row GET endpoint, title template, icon and the row-detail region.
27   *
28   * <p>
29   * {@link #CONTRACT_VERSION} and {@link #response(Object)} describe the JSON that endpoint returns.
30   *
31   * <h5 class='section'>Example:</h5>
32   * <p class='bjava'>
33   * 	<jv>table</jv>.detail(RowDetail.<jsm>create</jsm>(<js>"/rest/work/data/{id}"</js>)
34   * 		.title(<js>"{workId}"</js>).icon(<js>"search"</js>)
35   * 		.region(RegionDef.<jsm>create</jsm>(<js>"work-detail"</js>).populate(<js>"work-detail"</js>)
36   * 			.allowPopulators(<js>"work-detail"</js>)));
37   *
38   * 	<jc>// The matching endpoint:</jc>
39   * 	<ja>@RestGet</ja>(path=<js>"/data/{id}"</js>)
40   * 	<jk>public</jk> JsonMap detail(<ja>@Path</ja> String <jv>id</jv>) {
41   * 		<jk>return</jk> RowDetail.<jsm>response</jsm>(<jv>store</jv>.fields(<jv>id</jv>));
42   * 	}
43   * </p>
44   *
45   * @since 10.0.0
46   */
47  public final class RowDetail {
48  
49  	/** The contract version for the detail GET response (the card fragment itself carries no version). */
50  	public static final String CONTRACT_VERSION = "1";
51  
52  	private final String endpoint;
53  	private String title;
54  	private String icon;
55  	private RegionDef region;
56  
57  	private RowDetail(String endpoint) {
58  		this.endpoint = endpoint;
59  	}
60  
61  	/**
62  	 * Starts a row-detail definition for the given per-row GET endpoint.
63  	 *
64  	 * @param endpoint A same-origin path template carrying the literal {@code {id}} row-id placeholder (the client substitutes only that), e.g. {@code "/rest/work/data/{id}"}.
65  	 * 	Must not be <jk>null</jk> or blank.
66  	 * @return A new {@link RowDetail}.
67  	 * @throws IllegalArgumentException If {@code endpoint} is not a same-origin path with the literal {@code {id}} placeholder.
68  	 */
69  	public static RowDetail create(String endpoint) {
70  		if (! RegionDef.isSafeDetailEndpoint(endpoint) || ! endpoint.contains("{id}"))
71  			throw iaex("RowDetail endpoint '%s' is not a safe detail endpoint (same-origin path with an {id} placeholder).", endpoint);
72  		return new RowDetail(endpoint);
73  	}
74  
75  	/**
76  	 * Sets the detail panel's title template.
77  	 *
78  	 * @param template A row-data interpolation template, e.g. {@code "{workId}"}.  Can be <jk>null</jk> to unset.
79  	 * @return This object.
80  	 */
81  	public RowDetail title(String template) {
82  		title = template;
83  		return this;
84  	}
85  
86  	/**
87  	 * Sets the detail panel's icon.
88  	 *
89  	 * @param name A glyph name resolved by the shared icon registry.  Can be <jk>null</jk> to unset.
90  	 * @return This object.
91  	 */
92  	public RowDetail icon(String name) {
93  		icon = name;
94  		return this;
95  	}
96  
97  	/**
98  	 * Sets the detail panel's region.
99  	 *
100 	 * @param value The region.  Can be <jk>null</jk> to clear it (a header-only panel).
101 	 * 	Its {@link RegionDef#type} defaults to {@link RegionDef#TYPE_ROW_DETAIL} and its
102 	 * 	{@link RegionDef#dataUrl} defaults to this detail's own endpoint, applied at {@link #toMap()} time onto a
103 	 * 	copy; this method never mutates {@code value} itself.
104 	 * @return This object.
105 	 */
106 	public RowDetail region(RegionDef value) {
107 		region = value;
108 		return this;
109 	}
110 
111 	/**
112 	 * Builds the detail GET endpoint's own JSON response (not the card fragment).
113 	 *
114 	 * @param fields The row's detail fields, a map (or bean) in whatever shape the detail region's populator/renderer expects.
115 	 * @return {@code {contractVersion, fields}}.
116 	 */
117 	public static JsonMap response(Object fields) {
118 		var m = new JsonMap();
119 		m.put("contractVersion", CONTRACT_VERSION);
120 		m.put("fields", fields);
121 		return m;
122 	}
123 
124 	/**
125 	 * Builds this detail's card-body entry: {@code {endpoint, title?, icon?, region?}}.
126 	 *
127 	 * <p>
128 	 * The region is optional: a detail with none is a header-only panel (the client fills the {@code title}/{@code icon}
129 	 * header from the endpoint's {@code fields}) and the entry carries no {@code region} key.
130 	 *
131 	 * <p>
132 	 * When a region is set, builds from a copy: reads {@link RegionDef#toContractMap()} off the caller's {@link #region(RegionDef) region}
133 	 * into a fresh map, applies the {@code type}/{@code dataUrl} defaults onto that copy, and strips the copy's
134 	 * {@code contractVersion} key (a card fragment never nests a {@code contractVersion}, since the client stamps it
135 	 * on mount).  The caller's {@link RegionDef} instance is never written to.
136 	 *
137 	 * @return The detail entry.
138 	 */
139 	public JsonMap toMap() {
140 		var m = new JsonMap();
141 		m.put("endpoint", endpoint);
142 		if (title != null)
143 			m.put("title", title);
144 		if (icon != null)
145 			m.put("icon", icon);
146 		if (region != null) {
147 			var regionMap = new LinkedHashMap<>(region.toContractMap());
148 			regionMap.remove("contractVersion");
149 			regionMap.putIfAbsent("type", RegionDef.TYPE_ROW_DETAIL);
150 			regionMap.putIfAbsent("dataUrl", endpoint);
151 			m.put("region", regionMap);
152 		}
153 		return m;
154 	}
155 }
