17  package org.apache.juneau.rest.server.views;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import java.util.*;
22  
23  /**
24   * Serving-path allowlist of populator names that may be named on a region ({@link RegionDef#populate}).
25   *
26   * <p>
27   * This is an <b>authoring-discipline control, not a security boundary</b>, and the reason is structural, not a
28   * weaker policy choice.  {@link SinkRenderAllowlist} is real defense: a named renderer's markup still has to pass
29   * through the closed {@code fillRenderSlot} copier, so an accepted id only ever gets to write what that copier lets
30   * it write. A populator has no such copier anywhere in its path &mdash; {@code JuneauViews.regions.resolve(name)}
31   * hands the resolved function the region's container element outright, and from that point the populator owns the
32   * container completely. This class therefore controls only <i>which name is resolved</i>; it says nothing whatever
33   * about what the resolved populator then does with the container it is handed. A reader who treats this allowlist
34   * as a security boundary equivalent to {@link SinkRenderAllowlist} will over-trust it.
35   *
36   * @since 10.0.0
37   */
38  public final class PopulatorAllowlist {
39  
40  	/**
41  	 * The reserved declarative-default populator name &mdash; the only Juneau-shipped populator today.
42  	 *
43  	 * <p>
44  	 * Unlike an ordinary {@code JuneauViews.regions.register(name, fn)} entry, which is last-write-wins and freely
45  	 * overridable by a later registration under the same name, the client registry refuses to let a consumer
46  	 * override the entry named <js>"default"</js>: that name is reserved for the framework's own declarative-default
47  	 * populator.
48  	 */
49  	public static final Set<String> BUILTIN_IDS = Set.of("default");
50  
51  	private PopulatorAllowlist() {}
52  
53  	/**
54  	 * Accepts a reserved built-in name, or a name present in {@code allowedNames}.
55  	 *
56  	 * @param populatorName The populator name.  Must not be <jk>null</jk> or blank.
57  	 * @param allowedNames Opt-in populator names (e.g. {@link RegionDef#allowedPopulators}).  Can be <jk>null</jk>.
58  	 * @throws IllegalArgumentException If the name is neither a built-in nor an opted-in name.
59  	 */
60  	public static void assertAllowed(String populatorName, Collection<String> allowedNames) {
61  		if (populatorName == null || populatorName.isBlank())
62  			throw iaex("Populator name must not be null or blank.");
63  		if (BUILTIN_IDS.contains(populatorName))
64  			return;
65  		if (allowedNames != null && allowedNames.contains(populatorName))
66  			return;
67  		throw iaex("Populator name '%s' is not an allowed region populator.", populatorName);
68  	}
69  }
