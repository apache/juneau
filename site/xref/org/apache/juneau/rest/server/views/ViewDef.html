17  package org.apache.juneau.rest.server.views;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.commons.beanquery.*;
24  
25  /**
26   * A datatable view definition &mdash; an ordered set of {@link Column} definitions carrying each column's search
27   * configuration.  This is the bean the {@code VIEW_META} search block is serialized from.
28   *
29   * <p>
30   * A view may also carry a table-level {@link SearchOperatorSet}: it is inherited by every column that
31   * does not supply its own set (see {@link Column#effectiveOperators(SearchOperatorSet)} and
32   * {@link Column#searchMeta(SearchOperatorSet)}).  A column that supplies its own set entirely replaces the
33   * table-level set for that column.
34   *
35   * <h5 class='section'>Mutability</h5>
36   * <p>
37   * Deliberately mutable, like {@link Column} and {@link RegionDef}: a {@link ViewDef} is a server-side authoring
38   * bean built up once via fluent setters and {@link #column(Column)} calls while constructing a view,
39   * not an immutable value object.  Callers must not mutate one concurrently from more than one
40   * thread, and must not reuse one across requests unless every request treats it as read-only after construction.
41   *
42   * <h5 class='section'>Example:</h5>
43   * <p class='bjava'>
44   * 	<jc>// A view with a table-level $eq-only default, and one column ("status") that replaces it with $in-only.</jc>
45   * 	ViewDef <jv>view</jv> = ViewDef.<jsm>create</jsm>(<js>"releases"</js>)
46   * 		.operators(SearchOperatorSet.<jsm>of</jsm>(SearchOperatorSet.<jsm>standard</jsm>().get(<js>"$eq"</js>)))
47   * 		.column(Column.<jsm>create</jsm>(<js>"name"</js>).label(<js>"Name"</js>).searchType(SearchType.<jsf>TEXT</jsf>))
48   * 		.column(Column.<jsm>create</jsm>(<js>"status"</js>).label(<js>"Status"</js>).searchType(SearchType.<jsf>ID</jsf>)
49   * 			.operators(SearchOperatorSet.<jsm>of</jsm>(SearchOperatorSet.<jsm>standard</jsm>().get(<js>"$in"</js>))));
50   * </p>
51   *
52   * <h5 class='section'>See Also:</h5>
53   * <ul>
54   * 	<li class='jc'>{@link Column}
55   * </ul>
56   *
57   * @since 10.0.0
58   */
59  public final class ViewDef {
60  
61  	private final String id;
62  	private final Map<String,Column> columns = new LinkedHashMap<>();
63  	private SearchOperatorSet operators;
64  
65  	private ViewDef(String id) {
66  		this.id = id;
67  	}
68  
69  	/**
70  	 * Starts a new view definition.
71  	 *
72  	 * @param id The view id (matches the datatable's id on the page).  Must not be <jk>null</jk> or blank.
73  	 * @return A new, empty view definition.
74  	 * @throws IllegalArgumentException If {@code id} is <jk>null</jk> or blank.
75  	 */
76  	public static ViewDef create(String id) {
77  		req(inb(id), "ViewDef id must not be null or blank.");
78  		return new ViewDef(id);
79  	}
80  
81  	/**
82  	 * Adds a column definition, in order.
83  	 *
84  	 * @param value The column.  Must not be <jk>null</jk>.
85  	 * @return This object.
86  	 * @throws IllegalArgumentException If {@code value} is <jk>null</jk>.
87  	 */
88  	public ViewDef column(Column value) {
89  		req(value != null, "ViewDef '%s' column must not be null.", id);
90  		columns.put(value.name(), value);
91  		return this;
92  	}
93  
94  	/**
95  	 * Sets the table-level {@link SearchOperatorSet} &mdash; inherited by every column that does not
96  	 * supply its own set.
97  	 *
98  	 * @param value The operator set.  Can be <jk>null</jk> to clear it (columns without their own set then fall
99  	 * 	back to {@link SearchOperatorSet#standard()}).
100 	 * @return This object.
101 	 */
102 	public ViewDef operators(SearchOperatorSet value) {
103 		operators = value;
104 		return this;
105 	}
106 
107 	/** @return The view id. */
108 	public String id() {
109 		return id;
110 	}
111 
112 	/**
113 	 * The column definitions, in insertion order.
114 	 *
115 	 * @return An unmodifiable list of the columns.
116 	 */
117 	public List<Column> columns() {
118 		return List.copyOf(columns.values());
119 	}
120 
121 	/**
122 	 * Looks up a column by name.
123 	 *
124 	 * @param name The column name.  Can be <jk>null</jk>.
125 	 * @return The column, or <jk>null</jk> if no column has that name.
126 	 */
127 	public Column findColumn(String name) {
128 		return name == null ? null : columns.get(name);
129 	}
130 
131 	/**
132 	 * This view's table-level operator set, or <jk>null</jk> if absent (columns without their own set fall back to
133 	 * {@link SearchOperatorSet#standard()}).
134 	 *
135 	 * @return The operator set, or <jk>null</jk>.
136 	 */
137 	public SearchOperatorSet operators() {
138 		return operators;
139 	}
140 }
