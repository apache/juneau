29  /**
30   * Java description of one datatables view: data source, columns, ribbon, row styling, row actions, row detail,
31   * selection and bulk actions. It replaces hand-written {@code <@card type="datatables">} JSON5
32   * for apps that build pages in Java, and {@link #toCardBody()} serializes to exactly the card body the FTL path
33   * produces. All behavior (paging, search, rendering, actions) stays in {@code juneau-views.js}; this type never
34   * calls a card-type handler directly — see the class's rendering note on {@link #toCardBody()}.
35   *
36   * <p>
37   * {@link Column#defaultVisible(boolean) Column.defaultVisible(false)} only hides a column when the client's
38   * {@code columnConfig} (juneau-config.js) is on.
39   *
40   * <h5 class='section'>Example:</h5>
41   * <p class='bjava'>
42   * 	TableSpec <jv>t</jv> = TableSpec.<jsm>create</jsm>(<js>"slo"</js>)
43   * 		.rowType(SloPod.<jk>class</jk>)
44   * 		.dataUrl(<js>"/rest/slo/data"</js>)
45   * 		.defaultOrder(<js>"pod"</js>, TableSpec.Sort.<jsf>ASC</jsf>)
46   * 		.columns(
47   * 			Column.<jsm>create</jsm>(<js>"pod"</js>).label(<js>"Pod"</js>).className(<js>"ssc-id-cell"</js>),
48   * 			Column.<jsm>create</jsm>(<js>"incidentKey"</js>).label(<js>"Incident"</js>).render(<js>"linked"</js>).href(<js>"{incidentUrl}"</js>),
49   * 			Column.<jsm>create</jsm>(<js>"rootCause"</js>).label(<js>"Root cause"</js>).defaultVisible(<jk>false</jk>))
50   * 		.ribbon(RibbonItem.<jsm>refresh</jsm>(), RibbonItem.<jsm>export</jsm>(<js>"copy"</js>, <js>"csv"</js>))
51   * 		.rowClass(<js>"analysisState"</js>, Op.<jsf>EQ</jsf>, <js>"degraded"</js>, <js>"row-degraded"</js>)
52   * 		.rowActions(RowAction.<jsm>create</jsm>(<js>"analyze"</js>).label(<js>"Analyze pod"</js>)
53   * 			.endpoint(<js>"/rest/slo/analyze"</js>).method(RowAction.Method.<jsf>POST</jsf>))
54   * 		.detail(RowDetail.<jsm>create</jsm>(<js>"/rest/slo/data/{id}"</js>).title(<js>"{pod}"</js>)
55   * 			.region(RegionDef.<jsm>create</jsm>(<js>"slo-detail"</js>).populate(<js>"slo-detail"</js>)));
56   * 	<jv>t</jv>.validate();
57   * 	Map&lt;String,Object&gt; <jv>body</jv> = <jv>t</jv>.toCardBody();   // the unwrapped catalog
58   * </p>
59   *
60   * @since 10.0.0
61   */

144 	 * @param id The table's card id. Must not be <jk>null</jk> or blank, and must match

265 	 * Appends a value-based row-class rule ({@code juneau-views.js}'s {@code evaluateRowClassRules}):
266 	 * a row whose {@code field} is {@link Op#EQ eq}/{@link Op#NE ne} the given value gets {@code cssClass} added
267 	 * to its {@code <tr>}.
268 	 *
269 	 * @param field The row-data field to test. Must not be <jk>null</jk> or blank.
270 	 * @param op The operator. Must be {@link Op#EQ} or {@link Op#NE}.
271 	 * @param value The comparison value. Must not be <jk>null</jk>.
272 	 * @param cssClass The CSS class to add on a match (serialized as the wire key {@code class}). Must not be
273 	 * 	<jk>null</jk> or blank.
274 	 * @return This object.
275 	 * @throws IllegalArgumentException If any argument is missing, or {@code op} does not take a value.
276 	 */
277 	public TableSpec rowClass(String field, Op op, Object value, String cssClass) {
278 		checkRowClassCommon(field, op, cssClass);
279 		if (! op.requiresValue())
280 			throw iaex("TableSpec '%s' rowClass on '%s': op '%s' takes no value.", id, field, op.wire());
281 		if (value == null)
282 			throw iaex("TableSpec '%s' rowClass on '%s': op '%s' requires a value.", id, field, op.wire());
283 		var m = new JsonMap();
284 		m.put("field", field);
285 		m.put("op", op.wire());
286 		m.put("value", value);
287 		m.put("class", cssClass);
288 		rowClassRules.add(m);
289 		return this;
290 	}

296 	 * @param field The row-data field to test. Must not be <jk>null</jk> or blank.

305 		if (op.requiresValue())
306 			throw iaex("TableSpec '%s' rowClass on '%s': op '%s' requires a value.", id, field, op.wire());
307 		var m = new JsonMap();
308 		m.put("field", field);
309 		m.put("op", op.wire());
310 		m.put("class", cssClass);
311 		rowClassRules.add(m);
312 		return this;
313 	}

315 	private void checkRowClassCommon(String field, Op op, String cssClass) {
316 		if (ib(field))
317 			throw iaex("TableSpec '%s' rowClass field must not be null or blank.", id);
318 		if (op == null)
319 			throw iaex("TableSpec '%s' rowClass op must not be null.", id);
320 		if (op.isCollectionOp())
321 			throw iaex("TableSpec '%s' rowClass on '%s': op '%s' is not supported; use eq, ne, present or absent.", id, field, op.wire());
322 		if (ib(cssClass))
323 			throw iaex("TableSpec '%s' rowClass cssClass must not be null or blank.", id);
324 	}

413 	/**
414 	 * Declares URL-state directives this table's address bar carries, so their values reach the table's server requests.
415 	 *
416 	 * <p>
417 	 * Each name selects one of the built-in directives ({@code day}, {@code ids} or {@code window}); any other name is
418 	 * refused, because this method does not register codecs and the page contract only carries the built-ins.  A name
419 	 * not yet declared is emitted as an empty configuration object, which means the directive's defaults; a name already
420 	 * declared keeps its configuration (for example a {@code day} param set by {@link #urlStateDayParam(String)}).  This
421 	 * method never clears configuration.  Directives are emitted in the order they were first declared.  The call is
422 	 * atomic: every name is validated before any is declared, so if one is refused nothing is changed.
423 	 *
424 	 * @param names The directive names; each one of {@code day}, {@code ids} or {@code window}.
425 	 * @return This object.
426 	 * @throws IllegalArgumentException If a name is null, blank or not a built-in directive.
427 	 */
428 	public TableSpec urlState(String...names) {
429 		for (var n : names)
430 			checkUrlStateName(n);
431 		for (var n : names)
432 			urlState.putIfAbsent(n, new LinkedHashMap<>());
433 		return this;
434 	}

436 	/**
437 	 * Declares the {@code day} URL-state directive and names the request query parameter it sends the selected day under.
438 	 *
439 	 * <p>
440 	 * {@code urlState("day")} sends the day as a {@code day} parameter; this form is for a server whose parameter has
441 	 * another name.  The directive is emitted as {@code urlState.day.param}, in the position {@code day} was first
442 	 * declared.  Calling this again replaces the parameter name (the last call wins).
443 	 *
444 	 * @param param The query-parameter name; must match {@code ^[A-Za-z][A-Za-z0-9_-]{0,63}$}.
445 	 * @return This object.
446 	 * @throws IllegalArgumentException If {@code param} is null, blank, or malformed.
447 	 */
448 	public TableSpec urlStateDayParam(String param) {
449 		if (ib(param))
450 			throw iaex("urlState day param must not be null or blank.");
451 		checkUrlStateParam("day param", param);
452 		urlState.computeIfAbsent("day", k -> new LinkedHashMap<>()).put("param", param);
453 		return this;
454 	}

456 	/**
457 	 * Declares a URL-state directive together with its options.
458 	 *
459 	 * <p>
460 	 * This is the configurable form of {@link #urlState(String...)} for the three built-in directives; every option the
461 	 * client's built-in codecs accept can be expressed, so a page needs no hand-written map to configure one:
462 	 * <ul>
463 	 * 	<li>{@code ids} &mdash; {@code param} (request parameter name, default {@code ids}) and {@code max} (an integer
464 	 * 		&gt;= 1, default 200).
465 	 * 	<li>{@code window} &mdash; {@code params}, a map that may hold {@code start} and/or {@code end} (the request
466 	 * 		parameter names, defaults {@code start}/{@code end}).
467 	 * 	<li>{@code day} &mdash; {@code param} (request parameter name, default {@code day}); same as
468 	 * 		{@link #urlStateDayParam(String)}.
469 	 * </ul>
470 	 * Any other directive name is refused: this method does not register codecs, and the page contract only carries the
471 	 * built-ins.
472 	 *
473 	 * <p>
474 	 * Restrictions (deliberate, so the wire contract and the builder agree): every parameter name must match
475 	 * {@code ^[A-Za-z][A-Za-z0-9_-]{0,63}$} (the same grammar as {@code day.param}), and {@code max} must be an integer
476 	 * (Integer, Long, Short or Byte) of at least 1.  The client's own fallbacks for a falsy, string or fractional
477 	 * {@code max}, or for an unusual parameter name, are refused here rather than reproduced.  The result is the same
478 	 * value an FTL page writes as {@code urlState: {ids:{param:'pid',max:50}}}.
479 	 *
480 	 * <p>
481 	 * <b>Merge, not replace.</b>  The options are merged into any entry already declared for {@code name}: each option
482 	 * given here overwrites the same option declared earlier (the last call wins, as with {@link #urlStateDayParam(String)}),
483 	 * options not named here are kept, and {@code window.params} is merged key by key.  Like {@link #urlState(String...)}
484 	 * this never clears configuration, and an empty map just declares the directive.  The directive keeps the position it
485 	 * was first declared in.  The call is atomic: if any option is refused, nothing is changed.
486 	 *
487 	 * @param name The directive name; one of {@code day}, {@code ids} or {@code window}.
488 	 * @param config The directive options; deep-copied, so later changes to the map have no effect.  Must not be null.
489 	 * @return This object.
490 	 * @throws IllegalArgumentException If the name is null, blank or not a built-in directive, the config is null, or an option is unknown for a
491 	 * 	built-in directive, mistyped, or out of range (a parameter name that is malformed, a {@code max} that is not a
492 	 * 	positive integer, a {@code window.params} key other than {@code start}/{@code end}).
493 	 */
494 	@SuppressWarnings({
495 		"unchecked" // the only nested values are the Map<String,Object> copies checkUrlStateOption builds for window.params
496 	})
497 	public TableSpec urlState(String name, Map<String,Object> config) {
498 		checkUrlStateName(name);
499 		if (config == null)
500 			throw iaex("urlState '%s' config must not be null.", name);
501 		var checked = new LinkedHashMap<String,Object>();
502 		for (var e : config.entrySet()) {
503 			var key = e.getKey();
504 			if (ib(key))
505 				throw iaex("urlState '%s' option name must not be blank.", name);
506 			checked.put(key, checkUrlStateOption(name, key, e.getValue()));
507 		}
508 		var target = urlState.computeIfAbsent(name, k -> new LinkedHashMap<>());
509 		checked.forEach((k, v) -> {
510 			if ("window".equals(name) && "params".equals(k) && target.get(k) instanceof Map<?,?> old) {
511 				var merged = new LinkedHashMap<String,Object>();
512 				old.forEach((ok, ov) -> merged.put((String)ok, ov));
513 				merged.putAll((Map<String,Object>)v);
514 				target.put(k, merged);
515 			} else {
516 				target.put(k, v);
517 			}
518 		});
519 		return this;
520 	}

522 	private static void checkUrlStateName(String name) {
523 		if (ib(name))
524 			throw iaex("urlState name must not be blank.");
525 		if (! URL_STATE_BUILTINS.contains(name))
526 			throw iaex("urlState '%s' is not a built-in directive; expected day, ids or window.", name);
527 	}

529 	private static JsonMap copyUrlStateConfig(Map<String,Object> cfg) {
530 		var out = new JsonMap();
531 		cfg.forEach((k, v) -> out.put(k, v instanceof Map<?,?> m ? new JsonMap(m) : v));
532 		return out;

535 	private static Object checkUrlStateOption(String name, String key, Object value) {
536 		switch (name) {
537 			case "day":
538 				if (! "param".equals(key))
539 					throw iaex("urlState day: unknown option '%s'; expected param.", key);
540 				return checkUrlStateParam("day param", value);
541 			case "ids":
542 				if ("param".equals(key))
543 					return checkUrlStateParam("ids param", value);
544 				if ("max".equals(key)) {
545 					if (! (value instanceof Integer || value instanceof Long || value instanceof Short || value instanceof Byte) || ((Number)value).longValue() < 1)
546 						throw iaex("urlState ids max must be a positive integer but was %s.", value);
547 					return value;
548 				}
549 				throw iaex("urlState ids: unknown option '%s'; expected param or max.", key);
550 			case "window":
551 				if (! "params".equals(key))
552 					throw iaex("urlState window: unknown option '%s'; expected params.", key);
553 				if (! (value instanceof Map<?,?> m))
554 					throw iaex("urlState window params must be a map of start/end parameter names but was %s.", value);
555 				var out = new LinkedHashMap<String,Object>();
556 				for (var e : m.entrySet()) {
557 					var k = e.getKey();
558 					if (! "start".equals(k) && ! "end".equals(k))
559 						throw iaex("urlState window params: unknown key '%s'; expected start or end.", k);
560 					out.put((String)k, checkUrlStateParam("window params." + k, e.getValue()));
561 				}
562 				return out;
563 			default:
564 				throw iaex("urlState '%s' is not a built-in directive; expected day, ids or window.", name);
565 		}
566 	}
567 
568 	private static String checkUrlStateParam(String what, Object param) {
569 		if (! (param instanceof String s) || ! ID_PATTERN.matcher(s).matches())
570 			throw iaex("urlState %s '%s' must match ^[A-Za-z][A-Za-z0-9_-]{0,63}$.", what, param);
571 		return s;
572 	}

678 	/**
679 	 * Validates this table.  A duplicate column and a rowClass op/value mismatch are checked eagerly by
680 	 * {@link #columns(Column...)} and {@link #rowClass(String,Op,Object,String) rowClass}, and an unknown export button
681 	 * by {@link RibbonItem#export(String...)}, so none of those can be seen here.
682 	 *
683 	 * @throws IllegalStateException If any check fails.
684 	 */
685 	public void validate() {
686 		if (ib(dataUrl))
687 			throw isex("TableSpec '%s' requires dataUrl.", id);
688 		if (columns.isEmpty())
689 			throw isex("TableSpec '%s' requires at least one column.", id);
690 		var names = columns.stream().map(Column::name).toList();
691 		for (var o : defaultOrder)
692 			if (! names.contains(o.get("key")))
693 				throw isex("TableSpec '%s' defaultOrder column '%s' is not a column; columns: '%s'.",
694 					id, o.get("key"), String.join(", ", names));
695 		for (var item : ribbon) {
696 			var m = item.toMap();
697 			if ("dialog".equals(m.get("type")) && (m.get("form") == null || m.get("endpoint") == null))
698 				throw isex("TableSpec '%s' ribbon dialog '%s' requires form and endpoint.", id, m.get("id"));
699 		}
700 		var actionIds = new HashSet<String>();
701 		for (var a : rowActions)
702 			if (! actionIds.add(a.id))
703 				throw isex("TableSpec '%s' declares row action '%s' twice.", id, a.id);
704 		if (bulk != null && selection == null)
705 			throw isex("TableSpec '%s' sets bulk without selection.", id);
706 		if (bulk != null && (bulk.actions == null || bulk.actions.isEmpty()))
707 			throw isex("TableSpec '%s' sets bulk with no actions.", id);
708 		for (var a : rowActions)
709 			if (a.endpoint == null || a.endpoint.isBlank())
710 				throw isex("TableSpec '%s' row action '%s' requires an endpoint.", id, a.id);
711 		if (bulk != null && bulk.actions != null)
712 			for (var a : bulk.actions)
713 				if (a.endpoint == null || a.endpoint.isBlank())
714 					throw isex("TableSpec '%s' bulk action '%s' requires an endpoint.", id, a.id);
715 		for (var c : columns)
716 			if (c.searchable() && dataMode == DataMode.SERVER && c.operators() == null && operators == null)
717 				throw isex("TableSpec '%s' column '%s' sets searchType but dataMode is '%s' and the table "
718 					+ "declares no operators.", id, c.name(), dataMode.wire());
719 	}

721 	/**
722 	 * Builds this table's catalog-form card body, with no {@code contractVersion} anywhere (the client stamps every
723 	 * contract version).  This is the only place in this type that knows the datatables wire shape.  The result is
724 	 * the unwrapped catalog the {@code datatables} card type reads, and is rendered the same single way
725 	 * {@code <@card>} is, by the page mixin's {@code cardTypes.toCard(CardSource.create("datatables", id())
726 	 * .bodyMap(toCardBody()).build())}; this type never calls a card-type handler directly.
727 	 *
728 	 * <p>
729 	 * The body holds only plain data (maps, lists, strings, numbers, booleans), the same shape a parsed body has.
730 	 *
731 	 * <p>
732 	 * Each column carries its effective operator set (its own, else {@link #operators(SearchOperatorSet)}).
733 	 *
734 	 * @return The catalog, with keys in a fixed order and absent optional values omitted.
735 	 */
736 	public JsonMap toCardBody() {
737 		var t = new JsonMap();
738 		t.put("dataUrl", dataUrl);
739 		t.put("columns", plain(columns.stream().map(c -> c.toCatalogMap(operators)).toList()));
740 		if (dataMode != null)
741 			t.put("dataMode", dataMode.wire());
742 		if (! defaultOrder.isEmpty())
743 			t.put("defaultOrder", List.copyOf(defaultOrder));
744 		if (! ribbon.isEmpty())
745 			t.put("ribbon", ribbon.stream().map(RibbonItem::toMap).toList());
746 		if (rowType != null)
747 			t.put("rowType", rowType);
748 		if (pollIntervalMs != null)
749 			t.put("pollIntervalMs", pollIntervalMs);
750 		if (primary)
751 			t.put("primary", true);
752 		if (columnConfig != null)
753 			t.put("columnConfig", columnConfig);
754 		if (copyLink != null)
755 			t.put("copyLink", copyLink);
756 		if (cleanAddress != null)
757 			t.put("cleanAddress", cleanAddress);
758 		if (pausePollingWhileEditing != null)
759 			t.put("pausePollingWhileEditing", pausePollingWhileEditing);
760 		if (! urlState.isEmpty()) {
761 			var u = new JsonMap();
762 			urlState.forEach((n, cfg) -> u.put(n, copyUrlStateConfig(cfg)));
763 			t.put("urlState", u);
764 		}
765 		if (! rowClassRules.isEmpty())
766 			t.put("rowClassRules", List.copyOf(rowClassRules));
767 		if (! rowActions.isEmpty())
768 			t.put("rowActions", plain(rowActions));
769 		if (selection != null) {
770 			var s = new JsonMap();
771 			s.put("rowIdField", selection.rowIdField());
772 			s.put("selectAll", selection.selectAll());
773 			s.put("scope", selection.scope().wire());
774 			if (selection.labelField() != null)
775 				s.put("labelField", selection.labelField());
776 			if (selection.selectableWhen() != null)
777 				s.put("selectableWhen", plain(selection.selectableWhen()));
778 			t.put("selection", s);
779 		}
780 		if (bulk != null) {
781 			var b = new JsonMap();
782 			b.put("actions", plain(bulk.actions));
783 			t.put("bulk", b);
784 		}
785 		if (detail != null)
786 			t.put("detail", detail.toMap());
787 		if (quickStats != null) {
788 			var q = new JsonMap();
789 			q.put("id", quickStatsId);
790 			q.put("items", quickStats.stream().map(QuickStat::toMap).toList());
791 			t.put("quickStats", q);
792 		}
793 		return t;
794 	}
