17  package org.apache.juneau.rest.server.views;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;

21  import java.util.*;

23  /**
24   * The row-selection opt-in for a {@link ViewTable} &mdash; the first of the view table's two INDEPENDENT
25   * opt-ins.
26   *
27   * <h5 class='section'>Independent from bulk mutation, by construction</h5>
28   * <p>
29   * Declaring a {@link SelectionDef} adds a per-row checkbox column (and, when {@link #selectAll()} is set, a
30   * select-all header checkbox) to a view table &mdash; nothing else. It does <b>not</b>, by any default or
31   * convenience, surface a bulk-mutate control: {@link ViewTable}'s selection markers
32   * ({@link ViewTable#SELECT_ATTR} and siblings) have no code
33   * path that can render one. A view can therefore declare "selectable, for export" while categorically excluding
34   * bulk mutation &mdash; e.g. a PagerDuty-style incident table that wants selection for a client-side export/copy
35   * action but must never expose a multi-row mutate. A {@link BulkMutateDef} is a separate, independent type that
36   * <b>consumes</b> a {@link SelectionDef} (it requires one to be constructed at all) but a {@link SelectionDef}
37   * never requires, implies, or references one back.
38   *
39   * <h5 class='section'>Non-wire: never touches {@code VIEW_META}</h5>
40   * <p>
41   * A {@link SelectionDef} is deliberately <b>not</b> part of {@code VIEW_META} and never reaches the
42   * JSON sidecar. {@link ViewTable} instead stamps it as
43   * plain HTML attributes directly on the emitted {@code <table>} &mdash; {@link ViewTable#SELECT_ATTR},
44   * {@link ViewTable#ROW_ID_FIELD_ATTR}, {@link ViewTable#SELECT_ALL_ATTR} &mdash; mirroring the existing
45   * {@link ViewTable#CSRF_ATTR} auto-embed pattern. The {@code juneau-views.js} runtime reads them straight off the
46   * DOM, so a table with no {@link SelectionDef} carries none of these attributes and renders no selection UI at
47   * all; there is no shared, independently-versioned wire contract to keep in lockstep (contrast
48   * {@link BulkMutateDef}, whose actual bulk-action list <i>is</i> wire data and therefore carries its own,
49   * separate {@link BulkMutateDef#CONTRACT_VERSION}).
50   *
51   * <h5 class='section'>Selection identity is a stable row id, never a DOM/table index</h5>
52   * <p>
53   * {@link #rowIdField()} names the row-data key {@code juneau-views.js} reads to stamp a stable
54   * {@code data-juneau-row-id} attribute on each rendered {@code <tr>} (see {@code createdRow} in that runtime).
55   * Selection state is keyed by that value, never by a row's position in the DOM or in the DataTables draw &mdash;
56   * so a poll refresh, a sort, or a page change can never retarget a selection (or a bulk mutate driven
57   * from one) onto whichever rows now occupy those same positions. The runtime's persistence rule drops any
58   * previously-selected id that is no longer present in the newly drawn row set, so a selection whose rows have
59   * scrolled off screen or off the current page is never kept as a live, actionable target.
60   *
61   * <h5 class='section'>Example:</h5>
62   * <p class='bjava'>
63   * 	<jc>// Selectable for export, with NO bulk-mutate capability - satisfies the separability condition by construction.</jc>
64   * 	SelectionDef <jv>selection</jv> = SelectionDef.<jsm>create</jsm>(<js>"id"</js>)
65   * 		.selectAll(<jk>true</jk>)
66   * 		.selectableWhen(RowActionEnabledRule.<jsm>of</jsm>(<js>"state"</js>, Op.<jsf>EQ</jsf>, <js>"PENDING"</js>,
67   * 			<js>"Only pending changes can be aborted."</js>))
68   * 		.labelField(<js>"name"</js>);
69   * </p>
70   *
71   * <h5 class='section'>See Also:</h5>
72   * <ul>
73   * 	<li class='jc'>{@link BulkMutateDef}
74   * 	<li class='jc'>{@link ViewTable}
75   * </ul>
76   *
77   * @since 10.0.0
78   */
79  public final class SelectionDef {
80  
81  	/**
82  	 * Where a {@link SelectionDef}'s row selection lives across a view's lifetime.
83  	 *
84  	 * @since 10.0.0
85  	 */
86  	public enum Scope {
87  
88  		/** Selection survives paging, sorting, searching and refresh. */
89  		PERSISTENT("persistent"),
90  
91  		/** Selection is dropped for any row not in the current draw. */
92  		PAGE("page");
93  
94  		private final String wire;
95  
96  		Scope(String wire) {
97  			this.wire = wire;
98  		}
99  
100 		/**
101 		 * The wire-form string for this scope.
102 		 *
103 		 * @return The wire string ({@code "persistent"} or {@code "page"}).
104 		 */
105 		public String wire() {
106 			return wire;
107 		}
108 	}
109 
110 	private final String rowIdField;
111 	private boolean selectAll = true;
112 	private Scope scope = Scope.PERSISTENT;
113 	private List<RowActionEnabledRule> selectableWhen;
114 	private String labelField;
115 
116 	private SelectionDef(String rowIdField) {
117 		this.rowIdField = rowIdField;
118 	}
119 
120 	/**
121 	 * Starts a new {@link SelectionDef} bound to the specified stable row-id field.
122 	 *
123 	 * @param rowIdField The row-data key whose value is this row's stable identity (e.g. {@code "id"} or
124 	 * 	{@code "incidentId"}). Must not be <jk>null</jk> or blank.
125 	 * @return A new {@link SelectionDef}, with {@link #selectAll()} defaulting to <jk>true</jk>.
126 	 * @throws IllegalArgumentException If {@code rowIdField} is <jk>null</jk> or blank.
127 	 */
128 	public static SelectionDef create(String rowIdField) {
129 		if (ib(rowIdField))
130 			throw iaex("SelectionDef rowIdField must not be null or blank.");
131 		return new SelectionDef(rowIdField);
132 	}
133 
134 	/**
135 	 * Sets whether a select-all header checkbox is rendered alongside the per-row checkboxes.
136 	 *
137 	 * <p>
138 	 * Select-all always applies only to the rows currently on screen (the current page's draw) &mdash; it can
139 	 * never reach into rows on another page, which would be indistinguishable from the off-screen-retarget risk
140 	 * this whole opt-in exists to close.
141 	 *
142 	 * @param value The new value. Defaults to <jk>true</jk>.
143 	 * @return This object.
144 	 */
145 	public SelectionDef selectAll(boolean value) {
146 		selectAll = value;
147 		return this;
148 	}
149 
150 	/**
151 	 * The row-data key used as this table's stable row identity.
152 	 *
153 	 * @return The row-id field name.
154 	 */
155 	public String rowIdField() {
156 		return rowIdField;
157 	}
158 
159 	/**
160 	 * Whether a select-all header checkbox is rendered.
161 	 *
162 	 * @return <jk>true</jk> if select-all is enabled.
163 	 */
164 	public boolean selectAll() {
165 		return selectAll;
166 	}
167 
168 	/**
169 	 * Sets whether selection persists across paging/sorting/searching/refresh ({@link Scope#PERSISTENT}, the
170 	 * default) or is dropped for any row not in the current draw ({@link Scope#PAGE}).
171 	 *
172 	 * @param value The new scope. A <jk>null</jk> value resets to {@link Scope#PERSISTENT}.
173 	 * @return This object.
174 	 */
175 	public SelectionDef scope(Scope value) {
176 		scope = value == null ? Scope.PERSISTENT : value;
177 		return this;
178 	}
179 
180 	/**
181 	 * Sets the rules that gate which rows can be selected at all.
182 	 *
183 	 * <p>
184 	 * A row failing any rule renders its checkbox disabled, carrying the failing rule's reason as both a tooltip
185 	 * and an accessible description; select-all skips such rows entirely.
186 	 *
187 	 * @param value The gating rules. An empty or <jk>null</jk> array clears this to <jk>null</jk> (no gating).
188 	 * @return This object.
189 	 */
190 	public SelectionDef selectableWhen(RowActionEnabledRule...value) {
191 		selectableWhen = value == null || value.length == 0 ? null : l(value);
192 		return this;
193 	}
194 
195 	/**
196 	 * Sets the row-data field used as this row's human-readable label in a confirm dialog's target list.
197 	 *
198 	 * @param value The field name. May be <jk>null</jk> (falls back to the row id).
199 	 * @return This object.
200 	 */
201 	public SelectionDef labelField(String value) {
202 		labelField = value;
203 		return this;
204 	}
205 
206 	/**
207 	 * This selection's scope.
208 	 *
209 	 * @return The scope. Never <jk>null</jk>.
210 	 */
211 	public Scope scope() {
212 		return scope;
213 	}
214 
215 	/**
216 	 * This selection's gating rules.
217 	 *
218 	 * @return The rules, or <jk>null</jk> if none were set.
219 	 */
220 	public List<RowActionEnabledRule> selectableWhen() {
221 		return selectableWhen;
222 	}
223 
224 	/**
225 	 * This selection's label field.
226 	 *
227 	 * @return The field name, or <jk>null</jk> if none was set.
228 	 */
229 	public String labelField() {
230 		return labelField;
231 	}
232 }
