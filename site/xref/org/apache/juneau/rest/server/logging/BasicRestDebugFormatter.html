31  /**
32   * Default {@link RestDebugFormatter} implementation &mdash; secure by default.
33   *
34   * <p>
35   * Renders, cumulatively by tier:
36   * <ul>
37   * 	<li><b>Basic ({@code INFO})</b> &mdash; a single status line ({@code [status] HTTP method uri}), with the method and
38   * 		URI sanitized and the URI length-capped.
39   * 	<li><b>Headers ({@code FINE})</b> &mdash; request/response header blocks (<b>every</b> value of every header,
40   * 		credential-bearing values masked, all names/values sanitized and bounded), plus request/response lengths and the
41   * 		request execution time.
42   * 	<li><b>Body ({@code FINEST})</b> &mdash; request/response bodies, but <b>only if the operator has opted in</b> (see
43   * 		below); otherwise a suppression placeholder + byte count.
44   * </ul>
45   *
46   * <h5 class='section'>Secure-by-default body handling</h5>
47   * <p>
48   * Bodies are <b>never dumped by default</b>. Dumping is a deliberate operator opt-in behind a single environment-variable
49   * master gate, {@code JUNEAU_REST_DEBUG_ALLOW_DUMP_BODIES}:
50   * <ul>
51   * 	<li><b>Env-only.</b> It is an environment variable, never a system property, and has no system-property fallback
52   * 		&mdash; application code cannot flip it at runtime.
53   * 	<li><b>Truthy semantics.</b> The raw value is trimmed, then a non-empty value enables dumping except case-insensitive
54   * 		{@code false}/{@code 0}; unset/empty/all-whitespace disables. It is resolved once and cached, making it an
55   * 		emergency kill switch.
56   * 	<li><b>Gate subordinate.</b> With the gate unset, {@link #formatBody(RestRequest,RestResponse) formatBody} emits only
57   * 		the suppression placeholder + byte count &mdash; always, even if a {@link RestDebugBodyScrubber} is configured
58   * 		(the scrubber is never invoked).
59   * 	<li><b>Renderable content only.</b> Even when opted in, only renderable content (see
60   * 		{@link #isBodyRenderable(String)}, plus an identity {@code Content-Encoding}) is dumped; binary/unknown/compressed
61   * 		content yields a distinct non-renderable placeholder.
62   * 	<li><b>Fail-closed scrubber.</b> When the gate is set and a {@link RestDebugBodyScrubber} is configured, its output is
63   * 		dumped (then sanitized and capped); if it throws or returns <jk>null</jk>, the body fails closed to a placeholder
64   * 		&mdash; never the raw body.
65   * </ul>
66   * No bytes appear in any placeholder in any representation.
67   *
68   * <h5 class='section'>Header redaction and sanitization</h5>
69   * <p>
70   * Values for a widened, formatter-local credential-bearing header set are masked with {@link RedactedHeaders#REDACTED}.
71   * The set starts from {@link RedactedHeaders#DEFAULT} plus {@code X-Auth-Token}, {@code X-Authorization},
72   * {@code WWW-Authenticate}, {@code Referer}, and {@code Location}; the shared {@link RedactedHeaders#DEFAULT} is left
73   * unchanged. Matching is case-insensitive with separator folding ({@code -}/{@code _} stripped), so {@code X-Auth-Token},
74   * {@code X_Auth_Token}, and {@code XAuthToken} all match. Replace the set with {@link #redactedHeaders(Collection)} or
75   * extend it with {@link #addRedactedHeaders(Collection)}. Every emitted string (URI/method, header names/values, body
76   * text, placeholders) is sanitized against CR/LF/control-char log forging via {@link DebugTextSanitizer}, in
77   * <b>mask &rarr; escape &rarr; cap</b> order.
78   *
79   * <h5 class='section'>Bounds</h5>
80   * <p>
81   * Bodies are bounded by {@link #bodyCap()} bytes (default <b>8&nbsp;KB</b>) at capture time; headers by
82   * {@link #maxHeaders(int)}/{@link #maxHeaderScan(int)}/{@link #maxFieldLength(int)}; the URI by {@link #maxUriLength(int)}.
83   * The 8&nbsp;KB body cap is deliberately smaller than {@code EchoMixin}'s 1&nbsp;MB body cap &mdash; debug logging is
84   * intended for low-volume operator diagnostics; the two caps are independent by design.
85   *
86   * <h5 class='section'>See Also:</h5><ul>
87   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestServerLoggingAndDebugging">Logging / Debugging</a>
88   * </ul>
89   *
90   * @since 10.0.0
91   */

274 	@Override /* Overridden from RestDebugFormatter */
275 	public String statusLine(HttpServletRequest req, HttpServletResponse res) {
276 		var method = DebugTextSanitizer.sanitize(req.getMethod(), maxUriLength);
277 		var uri = DebugTextSanitizer.sanitize(req.getRequestURI(), maxUriLength);
278 		var sb = new StringBuilder();
279 		// Prepend the resolved correlation id: read the session-cached getRequestId() through the
280 		// session-handle seam — never a live public/custom-key attribute read.  Missing session/id → no prefix (today's
281 		// unprefixed line), never a throw.  Both formatBasic and the 404 render path route through here.
282 		var session = RestSession.fromRequest(req);
283 		if (session != null) {
284 			var id = session.getRequestId();
285 			if (id != null && ! id.isEmpty())
286 				sb.append("[requestId=").append(id).append("] ");
287 		}
288 		return sb.append('[').append(res.getStatus()).append("] HTTP ").append(method).append(' ').append(uri).toString();
289 	}

291 	@Override /* Overridden from RestDebugFormatter */
292 	public String formatHeaders(RestRequest req, RestResponse res) {
293 		var sreq = req.getHttpServletRequest();
294 		var sres = res.getHttpServletResponse();
295 		var sb = new StringBuilder();
296 
297 		var reqLen = req.getCachedContentLength();
298 		if (reqLen >= 0)
299 			sb.append("\n\tRequest length: ").append(reqLen).append(" bytes");
300 
301 		sb.append("\n\tResponse code: ").append(sres.getStatus());
302 
303 		var resLen = res.getCachedContentLength();
304 		if (resLen >= 0)
305 			sb.append("\n\tResponse length: ").append(resLen).append(" bytes");
306 
307 		var execTime = req.getExecTime();
308 		if (execTime != null)
309 			sb.append("\n\tExec time: ").append(execTime).append("ms");
310 
311 		var normalizedRedacted = normalizedRedactedSet();
312 		appendRequestHeaders(sb, sreq, normalizedRedacted);
313 		appendResponseHeaders(sb, sres, normalizedRedacted);
314 
315 		return sb.toString();
316 	}
317 
318 	private void appendRequestHeaders(StringBuilder sb, HttpServletRequest sreq, Set<String> normalizedRedacted) {
319 		var names = sreq.getHeaderNames();
320 		if (names == null || ! names.hasMoreElements())
321 			return;
322 		sb.append("\n---Request Headers---");
323 		var state = new HeaderScanState();
324 		while (names.hasMoreElements() && ! state.scanCapped) {
325 			var name = names.nextElement();
326 			var redact = isRedacted(name, normalizedRedacted);
327 			var sName = DebugTextSanitizer.sanitize(name, maxFieldLength);
328 			var values = sreq.getHeaders(name);
329 			appendHeaderValues(sb, sName, redact, values == null ? Collections.emptyIterator() : values.asIterator(), state);
330 		}
331 		appendOmissionMarker(sb, state.scanCapped, state.extra);
332 	}
333 
334 	private void appendResponseHeaders(StringBuilder sb, HttpServletResponse sres, Set<String> normalizedRedacted) {
335 		var names = sres.getHeaderNames();
336 		if (ie(names))
337 			return;
338 		sb.append("\n---Response Headers---");
339 		var state = new HeaderScanState();
340 		for (var name : names) {
341 			if (state.scanCapped)
342 				break;
343 			var redact = isRedacted(name, normalizedRedacted);
344 			var sName = DebugTextSanitizer.sanitize(name, maxFieldLength);
345 			appendHeaderValues(sb, sName, redact, sres.getHeaders(name).iterator(), state);
346 		}
347 		appendOmissionMarker(sb, state.scanCapped, state.extra);
348 	}

350 	/**
351 	 * Emits (or counts as omitted) every value for one header name, shared by the request- and response-header render
352 	 * loops so neither has to nest a second scan/emit/omit loop inside its own.
353 	 */
354 	private void appendHeaderValues(StringBuilder sb, String sanitizedName, boolean redact, Iterator<String> values, HeaderScanState state) {
355 		while (values.hasNext()) {
356 			if (state.scanned >= maxHeaderScan) {
357 				state.scanCapped = true;
358 				return;
359 			}
360 			state.scanned++;
361 			var v = values.next();
362 			if (state.emitted < maxHeaders) {
363 				appendHeaderLine(sb, sanitizedName, redact ? RedactedHeaders.REDACTED : v);
364 				state.emitted++;
365 			} else {
366 				state.extra++;
367 			}
368 		}
369 	}

371 	/** Mutable per-header-block emission/omission counters, threaded through {@link #appendHeaderValues}. */
372 	private static final class HeaderScanState {
373 		int emitted;
374 		long extra;
375 		int scanned;
376 		boolean scanCapped;
377 	}

379 	private void appendHeaderLine(StringBuilder sb, String sanitizedName, String value) {
380 		// Order: mask (already applied by caller) → escape → cap.
381 		sb.append("\n\t").append(sanitizedName).append(": ").append(DebugTextSanitizer.sanitize(value, maxFieldLength));
382 	}

384 	private static void appendOmissionMarker(StringBuilder sb, boolean scanCapped, long extra) {
385 		if (scanCapped)
386 			sb.append("\n\t\u2026[more headers omitted]");
387 		else if (extra > 0)
388 			sb.append("\n\t\u2026[+").append(extra).append(" more headers omitted]");
389 	}

391 	@Override /* Overridden from RestDebugFormatter */
392 	public String formatBody(RestRequest req, RestResponse res) {
393 		var sreq = req.getHttpServletRequest();
394 		var sres = res.getHttpServletResponse();
395 		var sb = new StringBuilder();
396 		appendBody(sb, "Request", req.getCachedContent(), req.getCachedContentLength(), sreq.getContentType(),
397 			sreq.getHeader("Content-Encoding"));
398 		appendBody(sb, "Response", res.getCachedContent(), res.getCachedContentLength(), sres.getContentType(),
399 			sres.getHeader("Content-Encoding"));
400 		return sb.toString();
401 	}

403 	private void appendBody(StringBuilder sb, String label, byte[] content, long totalLength, String contentType, String contentEncoding) {
404 		if (content == null || content.length == 0)
405 			return;
406 		var byteCount = totalLength >= 0 ? totalLength : content.length;
407 		var ctToken = contentTypeToken(contentType);
408 
409 		if (! isAllowDumpBodies()) {
410 			appendPlaceholder(sb, label, byteCount, ctToken,
411 				"; set " + ENV_ALLOW_DUMP_BODIES + " to enable", "body suppressed");
412 			return;
413 		}
414 
415 		if (! isRenderable(contentType, contentEncoding)) {
416 			appendPlaceholder(sb, label, byteCount, ctToken, "; binary/non-renderable content", "body not rendered");
417 			return;
418 		}
419 
420 		var raw = new String(content, UTF8);
421 		String bodyText;
422 		if (bodyScrubber != null) {
423 			String scrubbed;
424 			try {
425 				scrubbed = bodyScrubber.scrub(contentType, raw);
426 			} catch (Throwable t) {  // NOSONAR - fail closed on any scrubber failure; never re-leak via the exception.
427 				scrubbed = null;
428 			}
429 			if (scrubbed == null) {
430 				appendPlaceholder(sb, label, byteCount, ctToken, "; scrubber failed", "body suppressed");
431 				return;
432 			}
433 			bodyText = scrubbed;
434 		} else {
435 			bodyText = raw;
436 		}
437 
438 		sb.append("\n---").append(label).append(" Content---");
439 		sb.append('\n').append(DebugTextSanitizer.sanitize(bodyText, charCap()));
440 		var omitted = byteCount - content.length;
441 		if (omitted > 0)
442 			sb.append("\n\u2026[truncated ").append(omitted).append(" bytes]");
443 	}
444 
445 	private void appendPlaceholder(StringBuilder sb, String label, long byteCount, String ctToken, String reason, String verb) {
446 		sb.append("\n---").append(label).append(" Content---");
447 		sb.append("\n[").append(verb).append(": ").append(byteCount).append(" bytes, ").append(ctToken).append(reason).append(']');
448 	}

450 	private String contentTypeToken(String contentType) {
451 		if (ib(contentType))
452 			return "unknown";
453 		return DebugTextSanitizer.sanitize(contentType, maxFieldLength);
454 	}

456 	private boolean isRenderable(String contentType, String contentEncoding) {
457 		if (! isIdentityEncoding(contentEncoding))
458 			return false;
459 		return isBodyRenderable(contentType);
460 	}

462 	private static boolean isIdentityEncoding(String contentEncoding) {
463 		if (contentEncoding == null)
464 			return true;
465 		var e = contentEncoding.trim();
466 		return e.isEmpty() || eqic(e, "identity");
467 	}

469 	/**
470 	 * Returns the character-length cap for a dumped body, derived from {@link #bodyCap()}.
471 	 *
472 	 * <p>
473 	 * Worst-case escaping expands one byte to a 6-character {@code \\uXXXX} sequence, so the sanitized character cap is
474 	 * {@code bodyCap() * 6} (clamped to {@code Integer.MAX_VALUE}). Capture already bounds the raw bytes; this second
475 	 * layer bounds a scrubber that returns an oversized string.
476 	 */
477 	private int charCap() {
478 		var cap = Math.max(bodyCap, 0) * 6L;
479 		return cap > Integer.MAX_VALUE ? Integer.MAX_VALUE : (int) cap;
480 	}

482 	private Set<String> normalizedRedactedSet() {
483 		return redactedHeaders.stream().filter(Objects::nonNull).map(BasicRestDebugFormatter::normalizeHeaderName).collect(Collectors.toCollection(HashSet::new));
484 	}

486 	private static boolean isRedacted(String name, Set<String> normalizedRedacted) {
487 		return name != null && normalizedRedacted.contains(normalizeHeaderName(name));
488 	}

490 	/** Case-folds and strips {@code -}/{@code _} and whitespace so {@code X-Auth-Token}/{@code X_Auth_Token}/{@code XAuthToken} all match. */
491 	private static String normalizeHeaderName(String name) {
492 		var sb = new StringBuilder(name.length());
493 		for (var i = 0; i < name.length(); i++) {
494 			var c = name.charAt(i);
495 			if (c == '-' || c == '_' || Character.isWhitespace(c))
496 				continue;
497 			sb.append(Character.toLowerCase(c));
498 		}
499 		return sb.toString();
500 	}

502 	/**
503 	 * Returns whether body dumping is enabled by the {@code JUNEAU_REST_DEBUG_ALLOW_DUMP_BODIES} environment variable.
504 	 *
505 	 * <p>
506 	 * Resolved once from the environment and cached. Never reads a system property.
507 	 *
508 	 * @return <jk>true</jk> if body dumping is enabled.
509 	 */
510 	static boolean isAllowDumpBodies() {
511 		var v = allowDumpBodiesCache;
512 		if (v == null) {
513 			v = parseAllowDumpBodies(System.getenv(ENV_ALLOW_DUMP_BODIES));
514 			allowDumpBodiesCache = v;
515 		}
516 		return v;
517 	}

519 	/**
520 	 * Parses a raw gate value with trim-then-parse truthy semantics.
521 	 *
522 	 * <p>
523 	 * The value is trimmed first; a non-empty trimmed value enables (returns <jk>true</jk>) except case-insensitive
524 	 * {@code false} or {@code 0}; {@code null}/empty/all-whitespace disables. This is a fail-safe kill switch, so
525 	 * ambiguous values resolve to disabled.
526 	 *
527 	 * @param raw The raw environment value. Can be <jk>null</jk>.
528 	 * @return <jk>true</jk> if the value enables body dumping.
529 	 */
530 	static boolean parseAllowDumpBodies(String raw) {
531 		if (raw == null)
532 			return false;
533 		var t = raw.trim();
534 		if (t.isEmpty())
535 			return false;
536 		return ! (eqic(t, "false") || eq(t, "0"));
537 	}

547 	 * 	value is cached directly as the test gate state.

550 		allowDumpBodiesCache = override;
