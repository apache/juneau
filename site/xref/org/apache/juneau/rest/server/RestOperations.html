79  			httpMethodName = ucr(httpMethodName);
80  			map.computeIfAbsent(httpMethodName, k -> new TreeSet<>()).add(value);
81  			set.add(value);
82  			return this;
83  		}

129 	private final Map<String,List<RestOpContext>> map;
130 	private RestOpContext[] list;
131 
132 	/**
133 	 * Constructor.
134 	 *
135 	 * @param builder The builder containing the settings for this object. Must not be <jk>null</jk>.
136 	 */
137 	public RestOperations(Builder builder) {
138 		Map<String,List<RestOpContext>> m = map();
139 		for (var e : builder.map.entrySet())
140 			m.put(e.getKey(), toList(e.getValue()));
141 		this.map = m;
142 		this.list = array(builder.set, RestOpContext.class);
143 	}
144 
145 	/**
146 	 * Finds the method that should handle the specified call.
147 	 *
148 	 * @param session The HTTP call. Must not be <jk>null</jk>.
149 	 * @return The method that should handle the specified call.
150 	 * @throws MethodNotAllowed If no methods implement the requested HTTP method.
151 	 * @throws PreconditionFailed At least one method was found but it didn't match one or more matchers.
152 	 * @throws NotFound HTTP method match was found but matching path was not.
153 	 */
154 	@SuppressWarnings({
155 		"java:S3776" // Cognitive complexity acceptable for REST operation lookup/matching logic
156 	})
157 	public RestOpContext findOperation(RestSession session) throws MethodNotAllowed, PreconditionFailed, NotFound {
158 		String m = session.getMethod();
159 
160 		int rc = 0;
161 		if (map.containsKey(m)) {
162 			for (var oc : map.get(m)) {
163 				int mrc = oc.match(session);
164 				if (mrc == 2)
165 					return oc;
166 				rc = Math.max(rc, mrc);
167 			}
168 		}
169 
170 		if (map.containsKey("*")) {
171 			for (var oc : map.get("*")) {
172 				int mrc = oc.match(session);
173 				if (mrc == 2)
174 					return oc;
175 				rc = Math.max(rc, mrc);
176 			}
177 		}
178 
179 		// If no paths matched, see if the path matches any other methods.
180 		// Note that we don't want to match against "/*" patterns such as getOptions().
181 		if (rc == 0) {
182 			for (var oc : list) {
183 				if (! oc.getPathPattern().endsWith("/*")) {
184 					int orc = oc.match(session);
185 					if (orc == 2)
186 						throw new MethodNotAllowed();
187 				}
188 			}
189 		}
190 
191 		if (rc == 1)
192 			throw new PreconditionFailed("Method '%s' not found on resource on path '%s' with matching matcher.", m, session.getPathInfo());
193 
194 		throw new NotFound("Java method matching path '%s' not found on resource '%s'.", session.getPathInfo(), cn(session.getResource()));
195 	}
