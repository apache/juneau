130 	private final Map<String,List<RestOpContext>> map;
131 	private final RestOpContext[] list;
132 
133 	/**
134 	 * Constructor.
135 	 *
136 	 * @param builder The builder containing the settings for this object. Must not be <jk>null</jk>.
137 	 */
138 	public RestOperations(Builder builder) {
139 		Map<String,List<RestOpContext>> m = map();
140 		for (var e : builder.map.entrySet())
141 			m.put(e.getKey(), toList(e.getValue()));
142 		this.map = m;
143 		this.list = array(builder.set, RestOpContext.class);
144 
145 		// Fail the resource if an operation declared @Mutating is bound to a method defined as safe. Done here
146 		// because this constructor is the one path every operation table passes through, so the check cannot be
147 		// skipped by omission -- and it runs before findOperation can dispatch anything. A resource with no
148 		// @Mutating anywhere is unaffected.
149 		MethodSafety.check(getOpContexts());
150 	}

152 	/**
153 	 * Finds the method that should handle the specified call.
154 	 *
155 	 * @param session The HTTP call. Must not be <jk>null</jk>.
156 	 * @return The method that should handle the specified call.
157 	 * @throws MethodNotAllowed If no methods implement the requested HTTP method.
158 	 * @throws PreconditionFailed At least one method was found but it didn't match one or more matchers.
159 	 * @throws NotFound HTTP method match was found but matching path was not.
160 	 */
161 	@SuppressWarnings({
162 		"java:S3776" // Cognitive complexity acceptable for REST operation lookup/matching logic
163 	})
164 	public RestOpContext findOperation(RestSession session) throws MethodNotAllowed, PreconditionFailed, NotFound {
165 		String m = session.getMethod();
166 
167 		int rc = 0;
168 		if (map.containsKey(m)) {
169 			for (var oc : map.get(m)) {
170 				int mrc = oc.match(session);
171 				if (mrc == 2)
172 					return oc;
173 				rc = Math.max(rc, mrc);
174 			}
175 		}
176 
177 		if (map.containsKey("*")) {
178 			for (var oc : map.get("*")) {
179 				int mrc = oc.match(session);
180 				if (mrc == 2)
181 					return oc;
182 				rc = Math.max(rc, mrc);
183 			}
184 		}
185 
186 		// If no paths matched, see if the path matches any other methods.
187 		// Note that we don't want to match against "/*" patterns such as getOptions().
188 		if (rc == 0) {
189 			for (var oc : list) {
190 				if (! oc.getPathPattern().endsWith("/*")) {
191 					int orc = oc.match(session);
192 					if (orc == 2)
193 						throw methodNotAllowed(session);
194 				}
195 			}
196 		}
197 
198 		if (rc == 1)
199 			throw new PreconditionFailed("Method '%s' not found on resource on path '%s' with matching matcher.", m, session.getPathInfo());
200 
201 		throw new NotFound("Java method matching path '%s' not found on resource '%s'.", session.getPathInfo(), cn(session.getResource()));
202 	}

204 	/**
205 	 * Returns the value of the <c>Allow</c> header for a <c>405 Method Not Allowed</c> response to the specified call.
206 	 *
207 	 * <p>
208 	 * The methods this resource answers on the call's path: every key of the operation table, other than
209 	 * <js>"*"</js>, with at least one operation whose path pattern matches, plus the request-line method when a
210 	 * <js>"*"</js> operation's path matches (the un-overridden request is always accepted there).  The keys are what
211 	 * dispatch looks up, so a remote-proxy operation contributes <c>GET</c> and <c>POST</c>, never <c>RRPC</c>.
212 	 *
213 	 * <p>
214 	 * Only path patterns are tested.  Matchers are not run and the session's cached path match is not touched, so the
215 	 * call can be made for any request without disturbing the operation that handles it.
216 	 *
217 	 * @param session The HTTP call. Must not be <jk>null</jk>.
218 	 * @return The sorted, comma-separated method names.  Empty when nothing answers on the path.  Never <jk>null</jk>.
219 	 * @since 10.0.0
220 	 */
221 	public String allowedMethods(RestSession session) {
222 		var methods = map.entrySet().stream()
223 			.filter(e -> e.getValue().stream().anyMatch(x -> x.matchesPath(session)))
224 			.map(e -> eq(e.getKey(), "*") ? ucr(session.getRequest().getMethod()) : e.getKey())
225 			.collect(Collectors.toCollection(TreeSet::new));
226 		return String.join(", ", methods);
227 	}
228 
229 	/**
230 	 * Creates the <c>405 Method Not Allowed</c> thrown when the call's path matches only operations under other
231 	 * methods, carrying the <c>Allow</c> header.
232 	 *
233 	 * @param session The HTTP call. Must not be <jk>null</jk>.
234 	 * @return A new exception.  Never <jk>null</jk>.
235 	 */
236 	private MethodNotAllowed methodNotAllowed(RestSession session) {
237 		var e = new MethodNotAllowed();
238 		e.setHeader("Allow", allowedMethods(session));
239 		return e;
240 	}
