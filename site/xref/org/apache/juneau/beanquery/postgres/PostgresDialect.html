28  /**
29   * The PostgreSQL {@link SqlDialect}: the one dialect Juneau ships for the shared JDBC {@link SqlBeanQueryContext}.
30   *
31   * <p>
32   * It renders the built-in {@code $}-operators (design §5.1) into Postgres predicates that mirror the in-memory
33   * semantics: exact, case-sensitive {@code $eq}/{@code $in}/{@code $ne} on text/id/enum columns, with the
34   * case-insensitive {@code $eqic} (and the case-insensitive {@code $contains}/{@code $prefix}) via {@code lower(...)}
35   * and {@code ILIKE}; full-string {@code $regex} via {@code ~}/{@code ~*} with the same {@code i}/{@code m}/{@code s}
36   * flag semantics as {@code java.util.regex}; dotted-tuple version comparison and {@code $prefix}; and typed binds for
37   * numeric/boolean/timestamp comparisons.  A bare pattern maps {@code *} to {@code %} and {@code ?} to {@code _} for
38   * {@code ILIKE}, unless the literal was quoted in the search string, in which case it is matched literally.
39   *
40   * <p>
41   * Values arrive already typed (design #4 §4: {@link SearchType#parse(String)} runs during query resolution, before
42   * this dialect ever sees a value), and are always bound as {@code ?} parameters, never concatenated into SQL text.
43   *
44   * <ul class='spaced-list'>
45   * 	<li><b>ENUM columns</b> are compared as text: the column is cast ({@code col::text}) wherever this dialect applies a
46   * 		text function or operator to it, so a native Postgres {@code enum} column behaves like a {@code text} one
47   * 		(including bare {@code *}/{@code ?} wildcards, which match the whole constant name, case-insensitively).
48   * 	<li><b>VERSION columns</b> are stored as dotted text (for example {@code 10.0.1}) and compared as a tuple of integers
49   * 		({@code 10.9} sorts before {@code 10.10}), like the in-memory engine.  A stored value
50   * 		that is not a dotted sequence of integers never matches.  Known, deliberate differences from the in-memory
51   * 		parser ({@code Integer.parseInt} over {@code split("\\.")}): each part is limited to 9 digits here (so it always
52   * 		fits a Postgres {@code int}; longer parts never match, where in-memory accepts up to the {@code int} range), and a
53   * 		trailing empty part ({@code 10.}) or a signed part ({@code +1}, {@code -1}) is rejected here but accepted in-memory.
54   * 	<li><b>Blank</b> ({@code $blank}, and the blank-matching {@code $ne}) means {@code NULL} or only ASCII whitespace
55   * 		(space, tab, line breaks, vertical tab, form feed and the C0 separators), matching {@code String.isBlank()} for
56   * 		ASCII; non-ASCII Unicode spaces are not trimmed.
57   * </ul>
58   *
59   * <h5 class='section'>Example:</h5>
60   * <p class='bjava'>
61   * 	<jc>// Register Postgres as the dialect for a JDBC-backed context.</jc>
62   * 	SqlBeanQueryContext&lt;Task&gt; <jv>ctx</jv> = SqlBeanQueryContext.<jsm>create</jsm>(Task.<jk>class</jk>)
63   * 		.dialect(PostgresDialect.<jsf>INSTANCE</jsf>)
64   * 		.table(<js>"tasks"</js>)
65   * 		.column(<js>"name"</js>, SearchType.<jsf>TEXT</jsf>)
66   * 		.column(<js>"status"</js>, SearchType.<jsf>ENUM</jsf>)
67   * 		.connectionSupplier(<jv>supplier</jv>)
68   * 		.build();
69   *
70   * 	<jc>// Render a bare value directly: the value is already typed, so a NUMERIC column gets a BigDecimal bind.</jc>
71   * 	SqlFragment <jv>frag</jv> = PostgresDialect.<jsf>INSTANCE</jsf>.renderBare(<js>"\"age\""</js>, SearchType.<jsf>NUMERIC</jsf>, <jk>new</jk> BigDecimal(<js>"30"</js>), <jk>false</jk>);
72   * 	<jc>// frag.sql() == "\"age\" = ?", frag.binds() == [30]</jc>
73   * </p>
74   *
75   * @since 10.0.0
76   */
77  @SuppressWarnings({
78  	"java:S1192" // Duplicated SQL fragments read more clearly inline than as constants.
79  })

85  	private final Map<String,SearchSqlRenderer> builtins = new HashMap<>();

87  	private PostgresDialect() {
88  		builtins.put("$eq", (d, col, type, args) -> eq(col, type, args.get(0)));
89  		builtins.put("$eqic", (d, col, type, args) -> eqicFrag(col, type, args.get(0)));
90  		builtins.put("$ne", (d, col, type, args) -> ne(col, type, args));
91  		builtins.put("$in", (d, col, type, args) -> in(col, type, args));
92  		builtins.put("$contains", (d, col, type, args) -> frag(col + " ILIKE ?", like("%", text(args.get(0)), "%")));
93  		builtins.put("$prefix", (d, col, type, args) -> prefix(col, type, args.get(0)));
94  		builtins.put("$regex", (d, col, type, args) -> regex(col, args));
95  		builtins.put("$blank", (d, col, type, args) -> SqlFragment.of(blank(col)));
96  		builtins.put("$gt", (d, col, type, args) -> cmp(col, " > ", type, args.get(0)));
97  		builtins.put("$gte", (d, col, type, args) -> cmp(col, " >= ", type, args.get(0)));
98  		builtins.put("$lt", (d, col, type, args) -> cmp(col, " < ", type, args.get(0)));
99  		builtins.put("$lte", (d, col, type, args) -> cmp(col, " <= ", type, args.get(0)));
100 		builtins.put("$between", (d, col, type, args) -> between(col, type, args.get(0), args.get(1)));
101 	}

124 	@Override /* SqlDialect */
125 	public String renderLimitOffset(Long limit, Long offset) {
126 		var sb = new StringBuilder();
127 		if (limit != null)
128 			sb.append(" LIMIT ").append(limit.longValue());
129 		if (offset != null && offset != 0L)
130 			sb.append(" OFFSET ").append(offset.longValue());
131 		return sb.toString();
132 	}

134 	@Override /* SqlDialect */
135 	public SqlFragment renderBare(String columnSql, SearchType type, Object value, boolean quoted) {
136 		switch (type) {
137 			case TEXT, ID: {
138 				var token = text(value);
139 				if (quoted || ! hasWildcard(token))
140 					return frag(columnSql + " ILIKE ?", like("%", token, "%"));
141 				return frag(columnSql + " ILIKE ?", glob(token));
142 			}
143 			case ENUM: {
144 				// Same rule as the in-memory engine: a wildcard pattern is a case-insensitive whole-string glob over the
145 				// constant name; anything else (including a quoted literal) is case-insensitive equality.
146 				var token = text(value);
147 				if (quoted || ! hasWildcard(token))
148 					return eqicFrag(columnSql, type, token);
149 				return frag(textOf(columnSql, type) + " ILIKE ?", glob(token));
150 			}
151 			case VERSION:
152 				return prefix(columnSql, type, value);
153 			default:
154 				return frag(columnSql + " = ?", value);  // NUMERIC / BOOLEAN / TIMESTAMP: typed-exact; quoting is a no-op.
155 		}
156 	}

162 	/** Exact, case-sensitive equality (typed bind); used by {@code $eq}. */
163 	private static SqlFragment eq(String col, SearchType type, Object arg) {
164 		if (type == VERSION)
165 			return versionIn(col, List.of(arg));
166 		return frag(textOf(col, type) + " = ?", bind(type, arg));
167 	}
168 
169 	/**
170 	 * Case-insensitive equality; used by {@code $eqic} and the bare-token default for enum.
171 	 *
172 	 * <p>
173 	 * Named {@code eqicFrag} (rather than {@code eqic}) to make clear that it renders a SQL fragment, as opposed to
174 	 * evaluating a boolean comparison like the in-memory engine's {@code eqic} helper.
175 	 */
176 	private static SqlFragment eqicFrag(String col, SearchType type, Object arg) {
177 		return frag("lower(" + textOf(col, type) + ") = lower(?)", text(arg));
178 	}
179 
180 	private static SqlFragment ne(String col, SearchType type, List<Object> args) {
181 		if (type == VERSION) {
182 			// Mirrors InMemoryMatch: a blank cell matches; otherwise no listed version may equal the (parseable) cell.
183 			var in = versionIn(col, args);
184 			var sb = new StringBuilder(blankOr(col, col));
185 			if (in.binds().isEmpty())
186 				return SqlFragment.of(sb.append(" OR 1 = 1)").toString());
187 			sb.append(" OR NOT COALESCE(").append(in.sql()).append(", FALSE))");
188 			return SqlFragment.of(sb.toString(), in.binds());
189 		}
190 		var c = textOf(col, type);
191 		var sb = new StringBuilder(isCiText(type) ? blankOr(col, c) : "(" + col + " IS NULL");
192 		sb.append(" OR ").append(c).append(" NOT IN (");
193 		appendPlaceholders(sb, args.size());
194 		sb.append("))");
195 		return SqlFragment.of(sb.toString(), binds(type, args));
196 	}
197 
198 	private static SqlFragment in(String col, SearchType type, List<Object> args) {
199 		if (type == VERSION)
200 			return versionIn(col, args);
201 		var sb = new StringBuilder(textOf(col, type)).append(" IN (");
202 		appendPlaceholders(sb, args.size());
203 		sb.append(')');
204 		return SqlFragment.of(sb.toString(), binds(type, args));
205 	}

207 	private static SqlFragment prefix(String col, SearchType type, Object arg) {
208 		if (type == VERSION) {
209 			var v = versionArg(arg);
210 			if (v == null)
211 				return NEVER;
212 			var n = v.split("\\.").length;
213 			// Array slice: a stored version with fewer parts than the prefix yields a shorter array, which never equals it.
214 			return frag(versionArray(col) + "[1:" + n + "] = " + "string_to_array(?, '.')::int[]", v);
215 		}
216 		return frag(col + " ILIKE ?", like("", text(arg), "%"));
217 	}
218 
219 	/**
220 	 * {@code $regex}: a full-string match with the flags of {@code InMemoryMatch}/{@code java.util.regex}.
221 	 *
222 	 * <p>
223 	 * Case-insensitive unless a {@code flags=} argument is given without {@code i} (then {@code ~} instead of {@code ~*}).
224 	 * Java's {@code m} (multiline) and {@code s} (dotall) are mapped onto Postgres's embedded regex options, which bundle
225 	 * newline handling differently: neither flag &rarr; {@code (?p)} ({@code .} does not match a newline, {@code ^}/{@code $}
226 	 * only at the string ends), {@code s} &rarr; the default, {@code m} &rarr; {@code (?n)} and both &rarr; {@code (?w)}.
227 	 * The pattern is anchored with {@code \A...\Z} (string boundaries, unaffected by those options) rather than
228 	 * {@code ^...$}, so {@code m} cannot turn the full-string match into a per-line one.
229 	 *
230 	 * <p>
231 	 * The pattern is first compiled with {@link java.util.regex.Pattern} using the mapped flags; a pattern that is not a valid
232 	 * Java regex never matches (renders as an always-false predicate), exactly like the in-memory engine, and can
233 	 * therefore never break out of the surrounding {@code \A(?:...)\Z} anchors (for example {@code x)|(y}).
234 	 *
235 	 * <p>
236 	 * <b>Flavor differences:</b> this uses PostgreSQL's POSIX {@code ~}/{@code ~*} operators (Advanced Regular
237 	 * Expressions), which are not Java regex.  Java and PostgreSQL (ARE) regex syntax are similar but not identical (for example Java
238 	 * possessive quantifiers and {@code \p{L}} classes are not supported by PostgreSQL, and a Java-valid pattern may be
239 	 * rejected by the server, failing the query).  A pathological pattern can also be slow on the server, so configure a
240 	 * {@code statement_timeout} for connections that run user-supplied patterns.  A pattern that uses the {@code (?x)}
241 	 * comment flag with a trailing {@code #} comment is passed through unchanged and will be rejected by the server.
242 	 */
243 	private static SqlFragment regex(String col, List<Object> args) {
244 		var ci = true;  // Case-insensitive by default (design §5.1).
245 		var multiline = false;
246 		var dotall = false;
247 		for (var i = 1; i < args.size(); i++) {
248 			var a = text(args.get(i)).strip();
249 			if (a.startsWith("flags=")) {
250 				var f = a.substring(6);  // The last flags= argument wins, as in-memory.
251 				ci = f.indexOf('i') >= 0;
252 				multiline = f.indexOf('m') >= 0;
253 				dotall = f.indexOf('s') >= 0;
254 			}
255 		}
256 		var pattern = text(args.get(0));
257 		var javaFlags = 0;
258 		if (ci)
259 			javaFlags |= CASE_INSENSITIVE;
260 		if (multiline)
261 			javaFlags |= MULTILINE;
262 		if (dotall)
263 			javaFlags |= DOTALL;
264 		try {
265 			Pattern.compile(pattern, javaFlags);
266 		} catch (PatternSyntaxException e) {
267 			return NEVER;  // Invalid pattern = no match, like InMemoryMatch/RegexBudget; also keeps it from escaping the \A(?:...)\Z anchors.
268 		}
269 		String mods;
270 		if (multiline)
271 			mods = dotall ? "(?w)" : "(?n)";
272 		else
273 			mods = dotall ? "" : "(?p)";
274 		return frag(col + (ci ? " ~* ?" : " ~ ?"), mods + "\\A(?:" + pattern + ")\\Z");
275 	}

277 	private static SqlFragment cmp(String col, String operator, SearchType type, Object arg) {
278 		if (type == VERSION) {
279 			var v = versionArg(arg);
280 			return v == null ? NEVER : frag(versionArray(col) + operator + "string_to_array(?, '.')::int[]", v);
281 		}
282 		return frag(col + operator + "?", arg);
283 	}
284 
285 	private static SqlFragment between(String col, SearchType type, Object lo, Object hi) {
286 		if (type == VERSION) {
287 			var l = versionArg(lo);
288 			var h = versionArg(hi);
289 			return (l == null || h == null) ? NEVER : frag(versionArray(col) + " BETWEEN " + "string_to_array(?, '.')::int[]" + " AND " + "string_to_array(?, '.')::int[]", l, h);
290 		}
291 		return frag(col + " BETWEEN ? AND ?", lo, hi);
292 	}

299 	private static final SqlFragment NEVER = SqlFragment.of("(1 = 0)");

301 	/**
302 	 * The column as an {@code int[]}, or {@code NULL} when the stored text is not a dotted sequence of integers.
303 	 * Postgres compares arrays element by element and then by length, which is exactly the in-memory dotted-tuple order.
304 	 *
305 	 * <p>
306 	 * The guard literal is an {@code E'...'} string with doubled backslashes, so its meaning does not depend on
307 	 * {@code standard_conforming_strings}, and it contains no {@code ?} that pgjdbc could mistake for a placeholder.
308 	 */
309 	@SuppressWarnings({
310 		"java:S6353" // Deliberately ASCII [0-9]: Postgres \\d may match non-ASCII digits in some locales, which would then fail the int cast.
311 	})
312 	private static String versionArray(String col) {
313 		return "(CASE WHEN " + col + " ~ E'^\\\\s*[0-9]{1,9}(\\\\s*\\\\.\\\\s*[0-9]{1,9})*\\\\s*$' THEN string_to_array(" + col + ", '.')::int[] END)";
314 	}
315 
316 	/** A version argument normalized for {@link #VERSION_BIND}, or <jk>null</jk> if it is not a dotted sequence of integers. */
317 	private static String versionArg(Object arg) {
318 		// Hand-rolled scan of  ws* d{1,9} (ws* '.' ws* d{1,9})* ws*  (ASCII digits/whitespace), equivalent to that regex but
319 		// without a backtracking group.  At most 9 digits per part, so every part fits a Postgres int (no 22003 overflow).
320 		var s = text(arg);
321 		var n = s.length();
322 		var parts = new ArrayList<String>();
323 		var i = skipWs(s, 0);
324 		while (true) {
325 			var start = i;
326 			while (i < n && s.charAt(i) >= '0' && s.charAt(i) <= '9')
327 				i++;
328 			if (i == start || i - start > 9)
329 				return null;
330 			parts.add(String.valueOf(Integer.parseInt(s.substring(start, i))));
331 			i = skipWs(s, i);
332 			if (i == n)
333 				return String.join(".", parts);
334 			if (s.charAt(i) != '.')
335 				return null;
336 			i = skipWs(s, i + 1);
337 		}
338 	}
339 
340 	/** Index of the first non-whitespace char at or after {@code i}; whitespace is the regex {@code \s} set (ASCII). */
341 	private static int skipWs(String s, int i) {
342 		while (i < s.length() && " \t\n\u000B\f\r".indexOf(s.charAt(i)) >= 0)
343 			i++;
344 		return i;
345 	}

347 	/** Equality against any of the (parseable) versions; renders as never-true when none parses. */
348 	private static SqlFragment versionIn(String col, List<Object> args) {
349 		var sb = new StringBuilder();
350 		var binds = new ArrayList<>();
351 		for (var a : args) {
352 			var v = versionArg(a);
353 			if (v == null)
354 				continue;
355 			sb.append(binds.isEmpty() ? "" : " OR ").append(versionArray(col)).append(" = ").append("string_to_array(?, '.')::int[]");
356 			binds.add(v);
357 		}
358 		if (binds.isEmpty())
359 			return NEVER;
360 		return SqlFragment.of(binds.size() > 1 ? "(" + sb + ")" : sb.toString(), binds);
361 	}

367 	/** The cell is {@code NULL} or ASCII-whitespace only, like the in-memory blank test. */
368 	private static String blank(String col) {
369 		return blankOr(col, col) + ")";
370 	}
371 
372 	/** Opens a parenthesized disjunction with the blank test for {@code textExpr}; the caller appends the rest and closes it. */
373 	private static String blankOr(String col, String textExpr) {
374 		return "(" + col + " IS NULL OR btrim(" + textExpr + ", " + "E' \\t\\n\\x0B\\f\\r\\x1C\\x1D\\x1E\\x1F'" + ") = ''";
375 	}

377 	private static boolean isCiText(SearchType type) {
378 		return type == TEXT || type == ID || type == ENUM;
379 	}
380 
381 	private static boolean hasWildcard(String token) {
382 		return token.indexOf('*') >= 0 || token.indexOf('?') >= 0;
383 	}
384 
385 	/** A bare glob ({@code *} any run, {@code ?} one character) as a LIKE pattern, with LIKE metacharacters escaped first. */
386 	private static String glob(String token) {
387 		return escapeLike(token).replace("*", "%").replace("?", "_");
388 	}
389 
390 	/** The column as text: an ENUM column may be a native Postgres enum, which has no text operators of its own. */
391 	private static String textOf(String col, SearchType type) {
392 		return type == ENUM ? col + "::text" : col;
393 	}
394 
395 	/** Text-like columns bind strings; everything else binds the already-typed value. */
396 	private static Object bind(SearchType type, Object arg) {
397 		return isCiText(type) ? text(arg) : arg;
398 	}
399 
400 	private static List<Object> binds(SearchType type, List<Object> args) {
401 		return args.stream().map(a -> bind(type, a)).toList();
402 	}
403 
404 	private static void appendPlaceholders(StringBuilder sb, int count) {
405 		for (var i = 0; i < count; i++) {
406 			if (i > 0)
407 				sb.append(", ");
408 			sb.append('?');
409 		}
410 	}
411 
412 	/** Escapes LIKE/ILIKE metacharacters (backslash is the default Postgres LIKE escape). */
413 	private static String escapeLike(String s) {
414 		return s.replace("\\", "\\\\").replace("%", "\\%").replace("_", "\\_");
415 	}
416 
417 	private static String like(String pre, String s, String post) {
418 		return pre + escapeLike(s) + post;
419 	}

421 	/**
422 	 * Asserts that a value on a text-only path is a {@link String}.
423 	 *
424 	 * <p>
425 	 * Values for TEXT / ID / ENUM / VERSION columns are always {@link String}s (design #4 §4: {@link SearchType#parse}
426 	 * leaves them unchanged), so a non-string here is an engine bug, not a user error.
427 	 *
428 	 * @param o The value.
429 	 * @return The value as a string.
430 	 * @throws IllegalStateException If the value is not a {@link String}.
431 	 */
432 	private static String text(Object o) {
433 		if (o instanceof String s)
434 			return s;
435 		throw new IllegalStateException(String.format("Expected a String value for a text comparison but got '%s'.", o == null ? null : o.getClass().getName()));  // I believe we have a short for getting class name?
436 	}
437 
438 	private static SqlFragment frag(String sql, Object...binds) {
439 		return SqlFragment.of(sql, binds);
440 	}
