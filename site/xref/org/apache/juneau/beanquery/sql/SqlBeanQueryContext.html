32  /**
33   * The shared JDBC {@link BeanQueryContext}: renders validated queries into SQL for a {@link SqlDialect} and runs
34   * them on a connection from a configured source.
35   *
36   * <p>
37   * The engine is dialect-agnostic.  Identifier quoting, the built-in operator predicates and pagination come from the
38   * {@link SqlDialect} (Juneau ships {@code juneau-beanquery-postgres}).  Columns are always explicit: the class passed
39   * to {@link #create(Class)} fixes {@code T} (and is returned by {@link #getType()}); it is introspected only to
40   * choose the default row mapper, never to derive columns.
41   *
42   * <p>
43   * If no {@link Builder#rowMapper(SqlRowMapper) rowMapper} is set, {@link Builder#build()} picks a default from the
44   * row type {@code T}: a {@code Map} type that a {@link LinkedHashMap} is assignable to gets
45   * {@link SqlRowMapper#map()}, and any other type that {@link SqlRowMapper#bean(Class)} can plan (a record, or a class
46   * with a public no-arg constructor and setters) gets {@link SqlRowMapper#bean(Class)}.  {@code Object}, other
47   * {@code Map} types (such as {@code TreeMap}) and unplannable types get no default; {@code find} and {@code stream}
48   * then throw {@link IllegalStateException}, with the reason appended when there is one.  An explicit
49   * {@code rowMapper(...)} always wins, and {@code findValues}, {@code streamValues} and {@code count} never need one.
50   *
51   * <p>
52   * {@link #renderRows(ResolvedQuery)} and {@link #renderCount(ResolvedQuery, boolean)} open no connection, so the
53   * generated SQL and binds can be inspected and unit-tested.  A {@link #guard} fragment is AND-ed, in parentheses,
54   * into every statement.
55   *
56   * <h5 class='section'>Example:</h5>
57   * <p class='bjava'>
58   * 	SqlBeanQueryContext&lt;Person&gt; <jv>context</jv> = SqlBeanQueryContext
59   * 		.<jsm>create</jsm>(Person.<jk>class</jk>)
60   * 		.dialect(PostgresDialect.<jsf>INSTANCE</jsf>)
61   * 		.table(<js>"person"</js>)
62   * 		.column(<js>"name"</js>, SearchType.<jsf>TEXT</jsf>)
63   * 		.column(<js>"age"</js>, SearchType.<jsf>NUMERIC</jsf>)
64   * 		.columnSql(<js>"name"</js>, <js>"full_name"</js>)
65   * 		.guard(SqlFragment.<jsm>of</jsm>(<js>"\"deleted\" = false"</js>))
66   * 		.rowMapper((<jv>rs</jv>, <jv>cols</jv>) -&gt; <jk>new</jk> Person(<jv>rs</jv>.getString(<js>"full_name"</js>), <jv>rs</jv>.getInt(<js>"age"</js>)))
67   * 		.dataSource(<jv>dataSource</jv>)
68   * 		.queryTimeout(Duration.<jsm>ofSeconds</jsm>(5))
69   * 		.build();
70   *
71   * 	<jk>try</jk> (SqlBeanQuerySession&lt;Person&gt; <jv>session</jv> = <jv>context</jv>.getSession()) {
72   * 		Page&lt;Person&gt; <jv>page</jv> = <jv>session</jv>.find(<jv>query</jv>);
73   * 	}
74   *
75   * 	<jc>// No rowMapper(...): Person is a record, so build() defaults to SqlRowMapper.bean(Person.class).</jc>
76   * 	SqlBeanQueryContext&lt;Person&gt; <jv>defaulted</jv> = SqlBeanQueryContext.<jsm>create</jsm>(Person.<jk>class</jk>)
77   * 		.dialect(PostgresDialect.<jsf>INSTANCE</jsf>)
78   * 		.table(<js>"person"</js>)
79   * 		.column(<js>"name"</js>, SearchType.<jsf>TEXT</jsf>)
80   * 		.column(<js>"age"</js>, SearchType.<jsf>NUMERIC</jsf>)
81   * 		.dataSource(<jv>dataSource</jv>)
82   * 		.build();
83   *
84   * 	<jc>// A Map row type defaults to SqlRowMapper.map().</jc>
85   * 	SqlBeanQueryContext&lt;Map&gt; <jv>maps</jv> = SqlBeanQueryContext.<jsm>create</jsm>(Map.<jk>class</jk>)
86   * 		<jc>// ...dialect, table, columns, connection source...</jc>
87   * 		.build();
88   * </p>
89   *
90   * @param <T> The row (bean) type.
91   * @since 10.0.0
92   */
93  @SuppressWarnings({
94  	"java:S1192" // Duplicated argument-name literals read more clearly inline than as constants.

268 		/**
269 		 * {@inheritDoc}
270 		 *
271 		 * @throws IllegalStateException If the dialect, the table, the connection source or every column is missing.
272 		 */
273 		@Override /* BeanQueryContext.Builder */
274 		public SqlBeanQueryContext<T> build() {
275 			if (dialect == null)
276 				throw isex("SqlBeanQueryContext requires a dialect.");
277 			if (table == null)
278 				throw isex("SqlBeanQueryContext requires a table.");
279 			if (connectionSupplier == null)
280 				throw isex("SqlBeanQueryContext requires a connection source.");
281 			if (rowMapper != null)
282 				return new SqlBeanQueryContext<>(this, rowMapper, null);
283 			var selection = defaultRowMapper(type);
284 			return new SqlBeanQueryContext<>(this, selection.mapper(), selection.reason());
285 		}

287 		// Default row-mapper selection; runs only when no rowMapper(...) was set.
288 		@SuppressWarnings({
289 			"unchecked" // A LinkedHashMap row is assignable to every T the Map branch accepts.
290 		})
291 		private static <T> MapperSelection<T> defaultRowMapper(Class<T> type) {
292 			if (type == Object.class)
293 				return new MapperSelection<>(null, null);
294 			if (Map.class.isAssignableFrom(type)) {
295 				if (type.isAssignableFrom(LinkedHashMap.class))
296 					return new MapperSelection<>((SqlRowMapper<T>)SqlRowMapper.map(), null);
297 				return new MapperSelection<>(null, "rows are LinkedHashMap");
298 			}
299 			var plan = SqlRowMappers.tryBean(type);
300 			if (plan.reason() == null)
301 				return new MapperSelection<>(plan.mapper(), null);
302 			return new MapperSelection<>(null, format("cannot map rows to '%s': %s", type.getSimpleName(), plan.reason()));
303 		}

386 	 * Renders the row-selection statement for a validated query.
387 	 *
388 	 * @param query The validated query, from {@link #resolve(BeanQuery)}.  Must not be <jk>null</jk>.
389 	 * @return The {@code SELECT} statement.

393 	}

395 	SqlStatement renderRows(ResolvedQuery query, SqlFragment sessionGuard) {
396 		reqnn("query", query);
397 		checkOrigin(query);
398 		var cols = query.view();
399 		var sb = new StringBuilder("SELECT ");
400 		sb.append(cols.stream().map(this::columnSql).collect(Collectors.joining(", ")));
401 		var binds = new ArrayList<>();
402 		sb.append(" FROM ").append(dialect.quote(table));
403 		appendWhere(sb, binds, query, true, sessionGuard);
404 		appendOrderBy(sb, query);
405 		var limit = query.limit() == null ? null : Long.valueOf(query.limit());
406 		var offset = query.position() == 0 ? null : Long.valueOf(query.position());
407 		sb.append(dialect.renderLimitOffset(limit, offset));
408 		return new SqlStatement(sb.toString(), binds, cols, containsRegex(query.filter()));
409 	}

423 	SqlStatement renderCount(ResolvedQuery query, boolean applySearch, SqlFragment sessionGuard) {
424 		reqnn("query", query);
425 		checkOrigin(query);
426 		var sb = new StringBuilder("SELECT count(*) FROM ").append(dialect.quote(table));
427 		var binds = new ArrayList<>();
428 		appendWhere(sb, binds, query, applySearch, sessionGuard);
429 		return new SqlStatement(sb.toString(), binds, List.of(), applySearch && containsRegex(query.filter()));
430 	}
431 
432 	private void appendWhere(StringBuilder sb, List<Object> binds, ResolvedQuery query, boolean applySearch, SqlFragment sessionGuard) {
433 		var parts = new ArrayList<SqlFragment>();
434 		// guard/sessionGuard are raw, caller-supplied SQL this library cannot verify is internally safe to AND
435 		// unwrapped (e.g. it may itself be a top-level OR), so each is defensively parenthesized here.
436 		Stream.of(guard, sessionGuard)
437 			.filter(Objects::nonNull)
438 			.forEach(g -> parts.add(SqlFragment.of("(" + g.sql() + ")", g.binds())));  // Parenthesized so an OR cannot escape.
439 		if (applySearch) {
440 			var filter = query.filter();
441 			if (! (filter instanceof Filter.And and && and.items().isEmpty())) {
442 				// Unlike guard/sessionGuard, search comes from SqlSearchCompiler.compile(Filter), which already
443 				// guarantees its own top-level rendering is safe to AND with other WHERE parts (a multi-item
444 				// top-level And/Or parenthesizes itself; a top-level Not/Leaf is already one atomic, boundaried
445 				// unit) -- so wrapping it again here would just double-parenthesize it.
446 				parts.add(SqlSearchCompiler.create(dialect, this).compile(filter));
447 			}
448 		}
449 		if (parts.isEmpty())
450 			return;
451 		sb.append(" WHERE ");
452 		sb.append(parts.stream().map(SqlFragment::sql).collect(Collectors.joining(" AND ")));
453 		parts.forEach(p -> binds.addAll(p.binds()));
454 	}

456 	// True if any leaf anywhere in the filter uses the $regex operator. If filter contains a $regex leaf at all,
457 	// allowRegex was necessarily true when QueryResolver resolved it (checkRegex gates this earlier in the
458 	// pipeline), so there is nothing left to re-check here -- only to detect, for the statement-timeout fallback.
459 	private static boolean containsRegex(Filter filter) {
460 		if (filter instanceof Filter.Leaf leaf)
461 			return containsRegex(leaf.expression());
462 		if (filter instanceof Filter.Not not)
463 			return containsRegex(not.item());
464 		if (filter instanceof Filter.And and)
465 			return and.items().stream().anyMatch(SqlBeanQueryContext::containsRegex);
466 		if (filter instanceof Filter.Or or)
467 			return or.items().stream().anyMatch(SqlBeanQueryContext::containsRegex);
468 		throw new IllegalStateException("Unknown Filter type: " + filter);
469 	}
470 
471 	private static boolean containsRegex(SearchExpression node) {
472 		if (node.isLiteral())
473 			return false;
474 		if ("$regex".equals(node.operator().name()))
475 			return true;
476 		return node.args().stream().anyMatch(SqlBeanQueryContext::containsRegex);
477 	}
478 
479 	private void appendOrderBy(StringBuilder sb, ResolvedQuery query) {
480 		var keys = query.sort();
481 		if (keys.isEmpty())
482 			return;
483 		sb.append(" ORDER BY ");
484 		sb.append(keys.stream().map(k -> columnSql(k.column()) + (k.descending() ? " DESC" : " ASC")).collect(Collectors.joining(", ")));
485 	}

491 	SqlRowMapper<T> rowMapper() {
492 		if (rowMapper == null) {
493 			if (rowMapperReason == null)
494 				throw isex("SqlBeanQueryContext requires a rowMapper for find and stream.");
495 			throw isex("SqlBeanQueryContext requires a rowMapper for find and stream: %s.", rowMapperReason);
496 		}
497 		return rowMapper;
498 	}

500 	Connection acquire() {
501 		Connection c;
502 		try {
503 			c = connectionSupplier.get();
504 		} catch (BeanQueryExecutionException e) {
505 			throw e;
506 		} catch (RuntimeException e) {
507 			throw new BeanQueryExecutionException(e, "Failed to open a database connection.");
508 		}
509 		if (c == null)
510 			throw new BeanQueryExecutionException(null, "Failed to open a database connection.");
511 		return c;
512 	}
513 
514 	void release(Connection c) {
515 		try {
516 			connectionReleaser.accept(c);
517 		} catch (BeanQueryExecutionException e) {
518 			throw e;
519 		} catch (RuntimeException e) {
520 			throw new BeanQueryExecutionException(e, "Failed to release a database connection.");
521 		}
522 	}
