25  /**
26   * Compiles a resolved {@link Filter} tree into a SQL {@code WHERE} predicate for a {@link SqlDialect}.
27   *
28   * <p>
29   * This is the dialect-agnostic core of the shared JDBC engine.  The terms are already parsed, allow-listed and
30   * type-checked.  The compiler walks each tree and:
31   * <ul>
32   * 	<li>combines the terms (one per column) with {@code AND};
33   * 	<li>combines a column's {@code $and}/{@code $or}/{@code $not} sub-expressions itself;
34   * 	<li>delegates every leaf to the {@link SqlDialect} (a built-in via {@link SqlDialect#builtinRenderer(String)}, a
35   * 		bare pattern via {@link SqlDialect#renderBare}) or, for a custom operator, to that operator's
36   * 		{@link SearchOperator#extension(Class) SearchSqlRenderer extension}.
37   * 	<li>a {@link #compile(Filter)} caller gets the same per-leaf rendering across a resolved, cross-column tree.
38   * </ul>
39   *
40   * <p>
41   * <p>
42   * The SQL result must match the in-memory result, so a built-in the dialect cannot faithfully express is rejected with a
43   * {@link BeanQuerySyntaxException} whose code is {@link org.apache.juneau.commons.beanquery.BeanQuerySyntaxException.Code#BAD_VALUE BAD_VALUE} rather than
44   * rendered differently: an operator for which {@link SqlDialect#builtinRenderer(String)} returns <jk>null</jk> (for
45   * example {@code $regex} on a database without regex support), and a range/ordering operator ({@code $gt}, {@code $gte},
46   * {@code $lt}, {@code $lte}, {@code $between}) on a {@code VERSION} column when
47   * {@link SqlDialect#supportsVersionOrdering()} is <jk>false</jk>.  Equality on versions is unaffected.  A custom
48   * operator with no usable renderer still raises {@link IllegalStateException}.
49   *
50   * <h5 class='section'>Example:</h5>
51   * <p class='bjava'>
52   * 	ResolvedQuery <jv>q</jv> = <jv>context</jv>.resolve(BeanQuery.<jsm>create</jsm>().gt(<js>"age"</js>, 21).eq(<js>"name"</js>, <js>"Bob"</js>).build());
53   * 	SqlFragment <jv>where</jv> = SqlSearchCompiler.<jsm>create</jsm>(<jv>dialect</jv>, <jv>context</jv>).compile(<jv>q</jv>.filter());
54   * 	<jc>// where.sql():   "age" &gt; ? AND "name" = ?</jc>
55   * 	<jc>// where.binds(): [21, Bob]</jc>
56   * </p>
57   *
58   * @since 10.0.0
59   */
60  public final class SqlSearchCompiler {
61  
62  	/** Range/ordering built-ins; on a VERSION column these need the dialect to order dotted tuples semantically. */
63  	private static final Set<String> VERSION_ORDERING = Set.of("$gt", "$gte", "$lt", "$lte", "$between");
64  
65  	private final SqlDialect dialect;
66  	private final ColumnResolver columns;
67  
68  	private SqlSearchCompiler(SqlDialect dialect, ColumnResolver columns) {
69  		reqnn("dialect", dialect, "columns", columns);
70  		this.dialect = dialect;
71  		this.columns = columns;
72  	}
73  
74  	/**
75  	 * Creates a compiler for a dialect and column resolver.
76  	 *
77  	 * @param dialect The target dialect.  Must not be <jk>null</jk>.
78  	 * @param columns Maps validated column names to quoted SQL references.
79  	 * @return A new compiler.
80  	 */
81  	public static SqlSearchCompiler create(SqlDialect dialect, ColumnResolver columns) {
82  		return new SqlSearchCompiler(dialect, columns);
83  	}
84  
85  	/**
86  	 * Compiles an already-resolved {@link Filter} tree into a boolean {@code WHERE} predicate.
87  	 *
88  	 * <p>
89  	 * Renders the full cross-column {@code Filter.And}/{@code Filter.Or}/{@code Filter.Not} tree, so a top-level
90  	 * {@code $or} across <i>different</i> columns (for example DataTables' global search) compiles correctly.
91  	 * {@code Filter.And}/{@code Filter.Or} with no items render as always-true/always-false ({@code 1=1}/{@code 1=0})
92  	 * per their own contract &mdash; see {@code SqlBeanQueryContext.appendWhere}, which special-cases an empty
93  	 * top-level filter so an unfiltered query's {@code WHERE} clause does not pick up a spurious {@code (1=1)}.
94  	 *
95  	 * <h5 class='section'>Example:</h5>
96  	 * <p class='bjava'>
97  	 * 	SqlFragment <jv>where</jv> = SqlSearchCompiler.<jsm>create</jsm>(<jv>dialect</jv>, <jv>columns</jv>)
98  	 * 		.compile(<jv>filter</jv>);
99  	 * </p>
100 	 *
101 	 * @param filter The resolved filter, or <jk>null</jk> for no predicate.
102 	 * @return The predicate fragment, or <jk>null</jk> if {@code filter} is <jk>null</jk>.
103 	 * @throws BeanQuerySyntaxException (BAD_VALUE) If a built-in leaf is not supported by the dialect.
104 	 * @throws IllegalStateException If a custom-operator leaf cannot be rendered for the dialect.
105 	 */
106 	public SqlFragment compile(Filter filter) {
107 		return filter == null ? null : renderFilter(filter);
108 	}
109 
110 	private SqlFragment renderFilter(Filter node) {
111 		if (node instanceof Filter.Leaf leaf)
112 			return render(leaf.expression(), columns.columnSql(leaf.column()), leaf.type());
113 		if (node instanceof Filter.Not not) {
114 			var child = renderFilter(not.item());
115 			return not(child);
116 		}
117 		if (node instanceof Filter.And and) {
118 			if (and.items().isEmpty())
119 				return SqlFragment.of("1=1");
120 			return join(and.items().stream().map(this::renderFilter).toList(), " AND ");
121 		}
122 		if (node instanceof Filter.Or or) {
123 			if (or.items().isEmpty())
124 				return SqlFragment.of("1=0");
125 			return join(or.items().stream().map(this::renderFilter).toList(), " OR ");
126 		}
127 		throw new IllegalStateException("Unknown Filter type: " + node);
128 	}
129 
130 	private SqlFragment render(SearchExpression node, String columnSql, SearchType type) {
131 		if (node.isLiteral())
132 			return required(dialect.renderBare(columnSql, type, node.typedValue(), node.isQuoted()), "bare pattern", node);
133 		var op = node.operator();
134 		if (op.isCombinator())
135 			return renderCombinator(node, columnSql, type);
136 		if (op.isCustom()) {
137 			var r = op.extension(SearchSqlRenderer.class).orElse(null);
138 			if (r == null)
139 				throw new IllegalStateException("Custom operator '" + op.name() + "' has no SQL renderer; it cannot run against dialect '" + dialect.id() + "'.");
140 			return required(r.render(dialect, columnSql, type, op.typedArgs() ? node.typedArgs() : List.copyOf(node.literalArgs())), "custom operator '" + op.name() + "'", node);
141 		}
142 		if (type == SearchType.VERSION && VERSION_ORDERING.contains(op.name()) && ! dialect.supportsVersionOrdering())
143 			throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.BAD_VALUE,
144 				"Operator '%s' is not supported on version columns by the '%s' SQL dialect (version ordering is semantic in memory, so it is rejected rather than compared as text).", op.name(), dialect.id());
145 		var r = dialect.builtinRenderer(op.name());
146 		if (r == null)
147 			throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.BAD_VALUE,
148 				"Operator '%s' is not supported by the '%s' SQL dialect.", op.name(), dialect.id());
149 		return required(r.render(dialect, columnSql, type, node.typedArgs()), "operator '" + op.name() + "'", node);
150 	}
151 
152 	private SqlFragment renderCombinator(SearchExpression node, String columnSql, SearchType type) {
153 		if ("$not".equals(node.name())) {
154 			return not(render(node.args().get(0), columnSql, type));
155 		}
156 		var children = node.args().stream().map(a -> render(a, columnSql, type)).toList();
157 		return join(children, "$or".equals(node.name()) ? " OR " : " AND ");
158 	}
159 
160 	/**
161 	 * Negates a fragment with two-valued (strictly true/false) semantics.
162 	 *
163 	 * <p>
164 	 * The in-memory engine evaluates every leaf to true or false, so a negated leaf matches a row whose cell is
165 	 * {@code null}.  In SQL a predicate over a {@code NULL} cell is {@code NULL}, and {@code NOT NULL} is still
166 	 * {@code NULL}, which would drop the row.  Coalescing the inner predicate to {@code FALSE} first restores the
167 	 * in-memory result.
168 	 */
169 	private static SqlFragment not(SqlFragment child) {
170 		return SqlFragment.of("(NOT COALESCE(" + child.sql() + ", FALSE))", child.binds());
171 	}
172 
173 	/** Joins fragments with a boolean operator, concatenating binds; the whole group is always parenthesized. */
174 	private static SqlFragment join(List<SqlFragment> frags, String sep) {
175 		var sb = new StringBuilder("(");
176 		var binds = new ArrayList<>();
177 		for (var i = 0; i < frags.size(); i++) {
178 			if (i > 0)
179 				sb.append(sep);
180 			sb.append(frags.get(i).sql());
181 			binds.addAll(frags.get(i).binds());
182 		}
183 		sb.append(')');
184 		return SqlFragment.of(sb.toString(), binds);
185 	}
186 
187 	private SqlFragment required(SqlFragment f, String what, SearchExpression node) {
188 		if (f == null)
189 			throw new IllegalStateException("Dialect '" + dialect.id() + "' returned no SQL for " + what + " in '" + node + "'.");
190 		return f;
191 	}
