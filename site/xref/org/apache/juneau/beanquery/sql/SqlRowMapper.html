26  import org.apache.juneau.commons.bean.*;
27  import org.apache.juneau.commons.beanquery.*;

29  /**
30   * Turns the current {@link ResultSet} row into a bean, for {@link SqlBeanQuerySession#find(BeanQuery)} and
31   * {@link SqlBeanQuerySession#stream(BeanQuery)}.
32   *
33   * <p>
34   * The result set is positioned on the row; do not call {@link ResultSet#next()}.  {@code columns} lists the selected
35   * (logical) column names in {@code SELECT} order (the query's view, or every column), so column {@code i} of the
36   * result set is {@code columns.get(i - 1)} and a mapper can skip columns the query did not select.
37   *
38   * <p>
39   * Two built-in mappers need no dependency beyond this module:
40   * <ul>
41   * 	<li>{@link #map()} &mdash; each row as a {@link LinkedHashMap} keyed by column name in view order.
42   * 	<li>{@link #bean(Class)} &mdash; each row as a {@code record} (canonical constructor) or as a class with a public
43   * 		no-arg constructor and public setters ({@code setX}/{@code withX}, void or fluent) or public non-final fields.
44   * 		Columns are matched to Juneau bean property names by exact name; columns with no property are ignored, and
45   * 		properties with no column keep their defaults.
46   * 	<li>{@link #bean(BeanMeta)} &mdash; the same, but using existing bean metadata, so custom property namers apply and
47   * 		column names match the property names a serializer built from that metadata emits.
48   * </ul>
49   *
50   * <p>
51   * When no mapper is set on the {@link SqlBeanQueryContext.Builder}, one is chosen from the row type: a {@code Map}
52   * type gets {@link #map()}, and any other type that {@link #bean(Class)} can plan gets {@link #bean(Class)}.
53   * {@code Object}, {@code Map} types a {@link LinkedHashMap} is not assignable to (such as {@code TreeMap}) and types
54   * that cannot be planned get none, and {@code find}/{@code stream} then throw {@link IllegalStateException}.
55   *
56   * <h5 class='section'>Example:</h5>
57   * <p class='bjava'>
58   * 	<jc>// Custom mapper.</jc>
59   * 	SqlRowMapper&lt;Person&gt; <jv>mapper</jv> = (<jv>rs</jv>, <jv>columns</jv>) -&gt; <jk>new</jk> Person(
60   * 		<jv>columns</jv>.contains(<js>"name"</js>) ? <jv>rs</jv>.getString(<js>"full_name"</js>) : <jk>null</jk>,
61   * 		<jv>columns</jv>.contains(<js>"age"</js>) ? <jv>rs</jv>.getInt(<js>"age"</js>) : 0);
62   *
63   * 	<jc>// Rows as maps, or as a record / setter bean.</jc>
64   * 	SqlRowMapper&lt;Map&lt;String,Object&gt;&gt; <jv>maps</jv> = SqlRowMapper.<jsm>map</jsm>();
65   * 	SqlRowMapper&lt;Person&gt; <jv>beans</jv> = SqlRowMapper.<jsm>bean</jsm>(Person.<jk>class</jk>);
66   *
67   * 	<jc>// Default selection: no rowMapper(...) call, so Person (a record) gets bean(Person.class).</jc>
68   * 	SqlBeanQueryContext&lt;Person&gt; <jv>context</jv> = SqlBeanQueryContext.<jsm>create</jsm>(Person.<jk>class</jk>)
69   * 		<jc>// ...dialect, table, columns, connection source...</jc>
70   * 		.build();
71   * </p>
72   *
73   * @param <T> The row (bean) type.
74   * @since 10.0.0
75   */
76  @FunctionalInterface
77  public interface SqlRowMapper<T> {
78  
79  	/**
80  	 * Maps the current result-set row to a bean.
81  	 *
82  	 * @param rs The result set, positioned on the row to map.  Must not be advanced.
83  	 * @param columns The selected column names in {@code SELECT} order.
84  	 * @return The mapped bean.
85  	 * @throws SQLException If a column read fails.
86  	 */
87  	T map(ResultSet rs, List<String> columns) throws SQLException;
88  
89  	/**
90  	 * A built-in row mapper that maps each row to a new {@link LinkedHashMap}, keyed by the logical column name in
91  	 * {@code SELECT} order, with each value read via {@link ResultSet#getObject(int)} (the driver's natural type;
92  	 * <jk>null</jk> for SQL {@code NULL}).
93  	 *
94  	 * <h5 class='section'>Example:</h5>
95  	 * <p class='bjava'>
96  	 * 	SqlRowMapper&lt;Map&lt;String,Object&gt;&gt; <jv>mapper</jv> = SqlRowMapper.<jsm>map</jsm>();
97  	 * </p>
98  	 *
99  	 * @return A shared, stateless row mapper.  Always the same instance.
100 	 * @since 10.0.0
101 	 */
102 	static SqlRowMapper<Map<String,Object>> map() {
103 		return SqlRowMappers.MAP;
104 	}
105 
106 	/**
107 	 * A built-in row mapper that maps each row to a new instance of {@code type}: a {@code record} (via its
108 	 * canonical constructor) or a concrete class with a public no-arg constructor and public setters
109 	 * ({@code setX}/{@code withX} returning {@code void} or the bean type, so fluent setters work) or public non-final
110 	 * fields.
111 	 *
112 	 * <p>
113 	 * Planning happens here, once; mapping never re-inspects {@code type}.  Columns match Juneau bean property names
114 	 * exactly (setters, public fields and record components, with {@code @BeanIgnore}, {@code @BeanProp} and the
115 	 * default property namer applied; see {@link QueryModel}).  A column with no
116 	 * property is ignored; a record component with no column, or a SQL {@code NULL} for a primitive, gets the type's
117 	 * default.  Values are read with {@code ResultSet.getObject(i, type)}, except {@link Instant} (via
118 	 * {@link OffsetDateTime}), {@link java.util.Date} (via {@code getTimestamp}) and enums (by constant name).
119 	 * A failure reading or setting a column throws a {@link BeanQueryExecutionException} naming the column.  The
120 	 * returned mapper is thread-safe.
121 	 *
122 	 * <h5 class='section'>Example:</h5>
123 	 * <p class='bjava'>
124 	 * 	<jk>record</jk> Task(String name, <jk>int</jk> age) {}
125 	 * 	SqlRowMapper&lt;Task&gt; <jv>mapper</jv> = SqlRowMapper.<jsm>bean</jsm>(Task.<jk>class</jk>);
126 	 * </p>
127 	 *
128 	 * @param <T> The bean type.
129 	 * @param type The bean type.  Must not be <jk>null</jk>.
130 	 * @return A row mapper for {@code type}.
131 	 * @throws IllegalArgumentException If {@code type} cannot be planned: an interface, abstract class, enum, array or
132 	 * 	primitive type, a class with no public no-arg constructor, or a setter bean with an ambiguous (overloaded) setter.
133 	 * @since 10.0.0
134 	 */
135 	static <T> SqlRowMapper<T> bean(Class<T> type) {
136 		reqnn("type", type);
137 		var plan = SqlRowMappers.tryBean(type);
138 		if (plan.reason() != null)
139 			throw new IllegalArgumentException(format("Cannot map rows to '%s': %s.", type.getSimpleName(), plan.reason()));
140 		return plan.mapper();
141 	}
142 
143 	/**
144 	 * Same as {@link #bean(Class)}, but follows the bean rules of existing bean metadata, for example the metadata of
145 	 * the serializer that renders the rows, so column names match the property names it emits.
146 	 *
147 	 * <h5 class='section'>Example:</h5>
148 	 * <p class='bjava'>
149 	 * 	BeanConfigContext <jv>config</jv> = BeanConfigContext.<jsm>create</jsm>().propertyNamer(PropertyNamerDLC.<jsf>INSTANCE</jsf>).build();
150 	 * 	SqlRowMapper&lt;Task&gt; <jv>mapper</jv> = SqlRowMapper.<jsm>bean</jsm>(BeanMeta.<jsm>of</jsm>(Task.<jk>class</jk>, <jv>config</jv>));
151 	 * </p>
152 	 *
153 	 * @param <T> The bean type.
154 	 * @param beanMeta The bean metadata.  Must not be <jk>null</jk>.
155 	 * @return A row mapper for the bean type.
156 	 * @throws IllegalArgumentException For the same reasons as {@link #bean(Class)}.
157 	 * @since 10.0.0
158 	 */
159 	static <T> SqlRowMapper<T> bean(BeanMeta<T> beanMeta) {
160 		reqnn("beanMeta", beanMeta);
161 		var plan = SqlRowMappers.tryBean(beanMeta);
162 		if (plan.reason() != null)
163 			throw new IllegalArgumentException(format("Cannot map rows to '%s': %s.", beanMeta.getClassInfo().inner().getSimpleName(), plan.reason()));
164 		return plan.mapper();
165 	}
166 }
