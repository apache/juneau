17  package org.apache.juneau.beanquery.sql;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.commons.beanquery.*;
22  
23  /**
24   * The SQL callback an application supplies for a <b>custom</b> search operator (design §5.3) so a SQL context can
25   * render it into a {@code WHERE} clause.
26   *
27   * <p>
28   * The renderer receives the target {@link SqlDialect}, the already-quoted column reference, the column's
29   * {@link SearchType value type} (so it can bind values as the right JDBC type), and the operator's arguments (typed
30   * values when the operator asks for them, otherwise the literal strings; see {@link org.apache.juneau.commons.beanquery.SearchOperator.Builder#typedArgs(boolean) SearchOperator.Builder#typedArgs(boolean)}),
31   * and returns a {@link SqlFragment} (a boolean SQL expression plus its ordered bind values).  It is the
32   * SQL-side analog of {@link SearchPredicate} (the in-memory evaluator): built-in operators do not use it &mdash; their
33   * SQL comes from {@link SqlDialect#builtinRenderer(String)}.
34   *
35   * <p>
36   * A renderer is attached to a {@link SearchOperator} as a typed extension keyed by this interface, with
37   * {@link org.apache.juneau.commons.beanquery.SearchOperator.Builder#extension(Class, Object)
38   * extension(SearchSqlRenderer.class, renderer)}; the context retrieves it with {@link SearchOperator#extension(Class)},
39   * so commons never depends on SQL dialect types.  A
40   * renderer that does not support the supplied dialect returns <jk>null</jk>; the context then fails with a clear error
41   * rather than emitting wrong SQL.  Values must be returned as {@link SqlFragment} binds, never concatenated into the
42   * SQL text.
43   *
44   * <p>
45   * For the built-in value operators, and for a custom operator built with {@code typedArgs(true)}, each argument arrives
46   * already parsed by {@link SearchType#parse(String)} during query resolution: a {@link java.math.BigDecimal} for a
47   * {@code NUMERIC} column, a {@link Boolean} for {@code BOOLEAN}, and an {@link java.time.OffsetDateTime},
48   * {@link java.time.LocalDateTime} or {@link java.time.LocalDate} for {@code TIMESTAMP}.  Text-like types
49   * ({@code TEXT}, {@code ID}, {@code ENUM}, {@code VERSION}) are always {@link String}.  A malformed value never reaches a
50   * renderer; it fails resolution with code {@code BAD_VALUE}.
51   *
52   * <h5 class='section'>Example:</h5>
53   * <p class='bjava'>
54   * 	<jc>// A custom "$even" operator rendered as a modulo check (one bind-free fragment).</jc>
55   * 	SearchSqlRenderer <jv>renderer</jv> = (<jv>dialect</jv>, <jv>col</jv>, <jv>type</jv>, <jv>args</jv>) -&gt;
56   * 		SqlFragment.<jsm>of</jsm>(<js>"("</js> + <jv>col</jv> + <js>" % 2 = 0)"</js>);
57   *
58   * 	<jc>// A typed-args operator: args.get(0) is a BigDecimal on a NUMERIC column, bound as-is.</jc>
59   * 	SearchSqlRenderer <jv>divisibleBy</jv> = (<jv>dialect</jv>, <jv>col</jv>, <jv>type</jv>, <jv>args</jv>) -&gt;
60   * 		SqlFragment.<jsm>of</jsm>(<js>"("</js> + <jv>col</jv> + <js>" % ? = 0)"</js>, <jv>args</jv>.get(0));
61   *
62   * 	SearchOperator <jv>even</jv> = SearchOperator
63   * 		.<jsm>create</jsm>(<js>"$even"</js>, <js>"Matches even numbers. Example: $even()"</js>)
64   * 		.minArgs(0).maxArgs(0)
65   * 		.extension(SearchSqlRenderer.<jk>class</jk>, <jv>renderer</jv>)
66   * 		.build();
67   * </p>
68   *
69   * @since 10.0.0
70   */
71  @FunctionalInterface
72  public interface SearchSqlRenderer {
73  
74  	/**
75  	 * Renders this custom operator into a SQL predicate fragment.
76  	 *
77  	 * @param dialect The target SQL dialect.  Never <jk>null</jk>.
78  	 * @param columnSql The dialect-quoted column reference the predicate applies to (for example {@code "\"age\""}).
79  	 * 	Never <jk>null</jk>.
80  	 * @param type The column's value type, so the renderer can bind values as the right JDBC type.  Never <jk>null</jk>.
81  	 * @param args The operator's arguments (never <jk>null</jk>; may be empty): {@link SearchExpression#typedArgs() typed
82  	 * 	values} ({@link java.math.BigDecimal}, {@link Boolean}, {@link java.time.OffsetDateTime},
83  	 * 	{@link java.time.LocalDateTime}, {@link java.time.LocalDate}, or {@link String} for text-like types) for a built-in
84  	 * 	value operator or a custom operator with {@code typedArgs(true)}; plain {@link String} literals otherwise.
85  	 * @return The predicate fragment, or <jk>null</jk> if this renderer does not support {@code dialect}.
86  	 */
87  	SqlFragment render(SqlDialect dialect, String columnSql, SearchType type, List<Object> args);
88  }
