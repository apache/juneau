38  /**
39   * A minimal, self-contained, in-process stand-in for a real OAuth 2.1 authorization server (AS), so the
40   * {@code secured} MCP example (and its end-to-end test) run entirely offline &mdash; no external IdP, no
41   * network access, no pre-shared secrets checked into source control.
42   *
43   * <p><b>DEMO ONLY &mdash; NOT SUITABLE FOR PRODUCTION USE.</b></p>
44   *
45   * <p>
46   * This is deliberately NOT a general-purpose or spec-complete authorization server. It exists purely to make
47   * {@link SecuredExampleMcpServer}'s {@code JwtTokenValidator}-backed resource-server gate and
48   * {@link SecuredExampleClient}'s {@code McpTokenProvider}-backed client-credentials flow both have something
49   * real to talk to. Concretely, on {@link #start()} it:
50   *
51   * <ol>
52   * 	<li>Generates a fresh RSA-2048 signing key (a new key every run &mdash; nothing is persisted, so restarting
53   * 		the example invalidates any previously-issued token, which is exactly the offline-demo property we want).
54   * 	<li>Publishes only the <b>public</b> half of that key as a {@link JWKSource}, which
55   * 		{@link SecuredExampleMcpServer} feeds directly to {@code JwtTokenValidator.jwkSource(...)} &mdash; no JWKS
56   * 		HTTP endpoint is needed because the server and "AS" share this JVM.
57   * 	<li>Starts a tiny {@link HttpServer} exposing two routes:
58   * 		<ul>
59   * 			<li>{@code POST /token} &mdash; the RFC 6749 &sect;4.4 client-credentials grant against one fixed,
60   * 				randomly-generated demo client id/secret pair (see {@link #clientId()} / {@link #clientSecret()}).
61   * 				This is a REAL HTTP round trip: the client-side {@code OAuthClientCredentialsFlow} used by
62   * 				{@code McpTokenProvider} talks to it exactly as it would talk to a production IdP's token
63   * 				endpoint, just on {@code localhost}.
64   * 			<li>{@code GET /.well-known/oauth-authorization-server} &mdash; a minimal RFC 8414 Authorization
65   * 				Server Metadata document ({@code issuer} + {@code token_endpoint} only), so a real client can
66   * 				perform genuine discovery against {@link #issuerUri()} instead of being handed the token
67   * 				endpoint out of band. See {@link SecuredExampleClient} for the client half of that handshake.
68   * 		</ul>
69   * </ol>
70   *
71   * <p>
72   * <b>What this class intentionally does NOT implement</b> (called out here and in the README so nobody mistakes
73   * this demo for a template for a real deployment):
74   * <ul>
75   * 	<li>No real scope-authorization decision &mdash; {@link #handleToken} only checks the requested scope
76   * 		against a fixed allowlist ({@code mcp.read}/{@code mcp.write}); it never asks a resource owner or policy
77   * 		engine whether the client is ALLOWED that scope, which is what a real AS's consent/policy step does.
78   * 	<li>No refresh tokens, no consent screen, no user authentication of any kind &mdash; only the
79   * 		machine-to-machine client-credentials grant, which is all an MCP server-to-server OAuth demo needs.
80   * 	<li>No persistence, replay protection beyond the JWT {@code exp}/{@code nbf} window, or key rotation. The
81   * 		RFC 8414 document above is likewise minimal: no {@code jwks_uri}, no {@code scopes_supported}, no
82   * 		dynamic client registration &mdash; this AS's public key is handed directly to the validator in-process
83   * 		(see {@link #jwkSource()}) rather than published for a real JWKS fetch.
84   * </ul>
85   *
86   * <p>
87   * <b>TLS caveat:</b> this class and {@link SecuredExampleServer} talk plain {@code http} only because both
88   * ends are loopback-only ({@code 127.0.0.1}/{@code localhost}). OAuth 2.1 / RFC 9728 require {@code https} for
89   * any non-loopback resource identifier or token endpoint; a bearer token sent over plaintext to a non-loopback
90   * host is exposed to anyone on the network path.
91   *
92   * <p>
93   * A real deployment would point {@code JwtTokenValidator} at an actual IdP's JWKS URL and issuer, and would
94   * never generate or hand out client secrets like {@link #start()} does here.
95   *
96   * @since 10.0.0
97   */
98  public final class OfflineAuthorizationServer implements AutoCloseable {

100 	/** The scope minted onto a token when the client's request omits one. */
101 	public static final String DEFAULT_SCOPE = "mcp.read";
102 
103 	/**
104 	 * The fixed allowlist of scopes this offline AS will grant (H2): a client requesting anything outside this
105 	 * set is rejected with RFC 6749 &sect;5.2 {@code invalid_scope}, rather than being handed back whatever it
106 	 * asked for.
107 	 */
108 	public static final Set<String> GRANTABLE_SCOPES = Set.of(DEFAULT_SCOPE, "mcp.write");

110 	/** The RFC 6749 {@code scope} form parameter / JWT claim name, minted access tokens carry as {@code scope}. */
111 	private static final String SCOPE_PARAM = "scope";

113 	/** How long a minted access token is valid for. */
114 	private static final Duration TOKEN_TTL = Duration.ofMinutes(5);

116 	/** RFC 8414 Authorization Server Metadata well-known path. */
117 	private static final String WELL_KNOWN_AUTHORIZATION_SERVER = "/.well-known/oauth-authorization-server";

119 	// S2119: a single reused instance, not one per randomSecret() call - repeatedly constructing SecureRandom
120 	// re-seeds it from the OS entropy source every time, which is wasteful and provides no security benefit
121 	// over reusing one properly-seeded instance.
122 	private static final SecureRandom SECURE_RANDOM = new SecureRandom();

135 		this.httpServer = httpServer;
136 		this.issuer = issuer;

139 	/**
140 	 * Generates a fresh signing key and demo client credentials, and starts the {@code /token} and RFC 8414
141 	 * discovery endpoints on an OS-assigned loopback-only port.
142 	 *
143 	 * @return A running instance. Close it (or call {@link #close()}) to stop the HTTP endpoint.
144 	 * @throws Exception If key generation or the HTTP endpoint fails to start.
145 	 */
146 	@SuppressWarnings({
147 		"java:S112", // throws Exception intentional - example lifecycle method kept simple for demo readability
148 		"resource" // returned instance is owned and closed by the caller (see SecuredExampleServer.start()).
149 	})
150 	public static OfflineAuthorizationServer start() throws Exception {
151 		var signingKey = new RSAKeyGenerator(2048).keyID("demo-key-1").algorithm(JWSAlgorithm.RS256).generate();
152 		var clientId = "demo-client";
153 		var clientSecret = randomSecret();
154 
155 		var httpServer = HttpServer.create(new InetSocketAddress(InetAddress.getLoopbackAddress(), 0), 0);
156 		// H4/M10: the issuer is THIS instance's own loopback base URL, known as soon as the (possibly
157 		// OS-assigned) listen socket is bound - a constructor-supplied instance field, not a shared constant,
158 		// so two independently-started instances naturally have two different issuers (see
159 		// SecuredExampleMcpEndToEnd_Test's wrong-issuer test).
160 		var issuer = URI.create("http://127.0.0.1:" + httpServer.getAddress().getPort());
161 		var self = new OfflineAuthorizationServer(signingKey, clientId, clientSecret, httpServer, issuer);
162 		httpServer.createContext("/token", self::handleToken);
163 		httpServer.createContext(WELL_KNOWN_AUTHORIZATION_SERVER, self::handleAuthorizationServerMetadata);
164 		httpServer.start();
165 		return self;
166 	}

168 	/** Generates a random, never-checked-in demo client secret (32 bytes, URL-safe base64). */
169 	private static String randomSecret() {
170 		var bytes = new byte[32];
171 		SECURE_RANDOM.nextBytes(bytes);
172 		return Base64.getUrlEncoder().withoutPadding().encodeToString(bytes);
173 	}

241 	/**
242 	 * Implements RFC 6749 &sect;4.4 (client-credentials grant) against the fixed demo client id/secret,
243 	 * minting a signed JWT whose {@code aud} claim is exactly the RFC 8707 {@code resource} indicator the
244 	 * caller supplied &mdash; this is what lets {@code JwtTokenValidator} enforce the audience check on the
245 	 * receiving end.
246 	 */
247 	private void handleToken(HttpExchange exchange) throws IOException {
248 		// DEMO ONLY: see class javadoc - not a production AS.
249 		tokenRequestCount.incrementAndGet();
250 		try {
251 			if (! "POST".equals(exchange.getRequestMethod())) {
252 				exchange.getResponseHeaders().add("Allow", "POST");
253 				sendJson(exchange, 405, error("invalid_request", "must POST to /token"));
254 				return;
255 			}
256 			if (! authenticateClient(exchange)) {
257 				exchange.getResponseHeaders().add("WWW-Authenticate", "Basic realm=\"offline-authorization-server\"");
258 				sendJson(exchange, 401, error("invalid_client", "unknown client id or secret"));
259 				return;
260 			}
261 			Map<String,String> form;
262 			try (var body = exchange.getRequestBody()) {
263 				form = parseForm(body);
264 			}
265 			if (! "client_credentials".equals(form.get("grant_type"))) {
266 				sendJson(exchange, 400, error("unsupported_grant_type", "only client_credentials is supported"));
267 				return;
268 			}
269 			var resource = form.get("resource");
270 			if (resource == null || resource.isBlank()) {
271 				sendJson(exchange, 400, error("invalid_target", "a resource indicator (RFC 8707) is required"));
272 				return;
273 			}
274 			var scope = form.getOrDefault(SCOPE_PARAM, DEFAULT_SCOPE);
275 			if (! isGrantable(scope)) {
276 				sendJson(exchange, 400, error("invalid_scope", "requested scope must be a subset of " + GRANTABLE_SCOPES));
277 				return;
278 			}
279 			var token = mintAccessToken(resource, scope, Instant.now(), TOKEN_TTL);
280 			exchange.getResponseHeaders().add("Cache-Control", "no-store");
281 			exchange.getResponseHeaders().add("Pragma", "no-cache");
282 			sendJson(exchange, 200, Json.of(Map.<String,Object>of(
283 				"access_token", token,
284 				"token_type", "Bearer",
285 				"expires_in", TOKEN_TTL.toSeconds(),
286 				SCOPE_PARAM, scope)));
287 		} catch (RuntimeException | JOSEException e) {
288 			sendJson(exchange, 500, error("server_error", e.getMessage() == null ? "internal error" : e.getMessage()));
289 		}
290 	}

292 	/** Serves a minimal RFC 8414 Authorization Server Metadata document (M10): {@code issuer} + {@code token_endpoint} only. */
293 	private void handleAuthorizationServerMetadata(HttpExchange exchange) throws IOException {
294 		if (! "GET".equals(exchange.getRequestMethod())) {
295 			exchange.getResponseHeaders().add("Allow", "GET");
296 			exchange.sendResponseHeaders(405, -1);
297 			return;
298 		}
299 		sendJson(exchange, 200, Json.of(Map.of(
300 			"issuer", issuer.toString(),
301 			"token_endpoint", tokenEndpoint().toString())));
302 	}

304 	/** Returns whether every space-delimited token in {@code scope} is in the {@link #GRANTABLE_SCOPES} allowlist (H2). */
305 	private static boolean isGrantable(String scope) {
306 		for (var s : scope.split("\\s+"))
307 			if (! s.isBlank() && ! GRANTABLE_SCOPES.contains(s))
308 				return false;
309 		return true;
310 	}

312 	/** Validates the RFC 6749 HTTP Basic client-authentication header against the fixed demo credentials. */
313 	private boolean authenticateClient(HttpExchange exchange) {
314 		// DEMO ONLY: see class javadoc - not a production AS.
315 		var header = exchange.getRequestHeaders().getFirst("Authorization");
316 		if (header == null || ! header.startsWith("Basic "))
317 			return false;
318 		byte[] decodedBytes;
319 		try {
320 			decodedBytes = Base64.getDecoder().decode(header.substring("Basic ".length()));
321 		} catch (IllegalArgumentException e) { // L1: malformed base64 -> a clean 401, not a 500
322 			return false;
323 		}
324 		var decoded = new String(decodedBytes, StandardCharsets.UTF_8);
325 		var sep = decoded.indexOf(':');
326 		if (sep < 0)
327 			return false;
328 		var presentedId = URLDecoder.decode(decoded.substring(0, sep), StandardCharsets.UTF_8);
329 		var presentedSecret = URLDecoder.decode(decoded.substring(sep + 1), StandardCharsets.UTF_8);
330 		// M6: constant-time comparison - both MessageDigest.isEqual() calls below are unconditionally evaluated
331 		// (regardless of the '&&' below) so neither the id nor the secret check can leak timing information
332 		// about how many leading bytes matched. The boolean combinator itself has no side effects either way.
333 		var idMatches = MessageDigest.isEqual(clientId.getBytes(StandardCharsets.UTF_8), presentedId.getBytes(StandardCharsets.UTF_8));
334 		var secretMatches = MessageDigest.isEqual(clientSecret.getBytes(StandardCharsets.UTF_8), presentedSecret.getBytes(StandardCharsets.UTF_8));
335 		return idMatches && secretMatches;
336 	}

338 	/**
339 	 * Signs a fresh RS256 access token carrying the mandatory claims {@code JwtTokenValidator} requires.
340 	 *
341 	 * <p>
342 	 * Package-private test seam (H4): exposing {@code issuedAt}/{@code ttl} lets
343 	 * {@link SecuredExampleMcpEndToEnd_Test} mint an already-expired token directly, without needing a real
344 	 * clock to actually elapse.
345 	 *
346 	 * @param audience The RFC 8707 {@code resource} indicator to mint the token's {@code aud} claim for.
347 	 * @param scope The (space-delimited) {@code scope} claim.
348 	 * @param issuedAt The token's {@code iat}/{@code nbf} instant.
349 	 * @param ttl How long after {@code issuedAt} the token expires.
350 	 * @return The signed, serialized JWT.
351 	 * @throws JOSEException If signing fails.
352 	 */
353 	String mintAccessToken(String audience, String scope, Instant issuedAt, Duration ttl) throws JOSEException {
354 		var claims = new JWTClaimsSet.Builder()
355 			// M4: a client-credentials grant has no resource owner - the subject IS the client, not a
356 			// placeholder end user.
357 			.subject(clientId)
358 			.issuer(issuer.toString())
359 			.audience(audience)
360 			.claim(SCOPE_PARAM, scope)
361 			.issueTime(Date.from(issuedAt))
362 			.notBeforeTime(Date.from(issuedAt))
363 			.expirationTime(Date.from(issuedAt.plus(ttl)))
364 			.build();
365 		var header = new JWSHeader.Builder(JWSAlgorithm.RS256).keyID(signingKey.getKeyID()).build();
366 		var jwt = new SignedJWT(header, claims);
367 		jwt.sign(new RSASSASigner(signingKey));
368 		return jwt.serialize();
369 	}

371 	/** Parses an {@code application/x-www-form-urlencoded} request body into a key/value map. */
372 	private static Map<String,String> parseForm(InputStream body) throws IOException {
373 		var raw = new String(body.readAllBytes(), StandardCharsets.UTF_8);
374 		var out = new LinkedHashMap<String,String>();
375 		for (var pair : raw.split("&")) {
376 			if (pair.isEmpty())
377 				continue;
378 			var eq = pair.indexOf('=');
379 			var key = eq < 0 ? pair : pair.substring(0, eq);
380 			var value = eq < 0 ? "" : pair.substring(eq + 1);
381 			out.put(URLDecoder.decode(key, StandardCharsets.UTF_8), URLDecoder.decode(value, StandardCharsets.UTF_8));
382 		}
383 		return out;
384 	}

386 	/**
387 	 * Builds an RFC 6749 &sect;5.2 JSON error body.
388 	 *
389 	 * <p>
390 	 * M2/M3: built as a {@link Map} and serialized via {@link Json#of(Object)} (the same marshaller
391 	 * {@link SecuredExampleClient} already uses) instead of hand-concatenating strings, so a client-controlled
392 	 * {@code error_description} containing a quote or backslash can no longer reshape the JSON structure or
393 	 * produce invalid JSON.
394 	 */
395 	private static String error(String code, String description) {
396 		return Json.of(Map.of("error", code, "error_description", description));
397 	}

399 	private static void sendJson(HttpExchange exchange, int status, String json) throws IOException {
400 		var bytes = json.getBytes(StandardCharsets.UTF_8);
401 		exchange.getResponseHeaders().add("Content-Type", "application/json");
402 		exchange.sendResponseHeaders(status, bytes.length);
403 		try (var os = exchange.getResponseBody()) {
404 			os.write(bytes);
405 		}
406 	}
