17  package org.apache.juneau.examples.mcp.secured;
18  
19  import java.net.*;
20  
21  import org.apache.juneau.examples.mcp.*;
22  import org.apache.juneau.rest.server.auth.jwt.*;
23  import org.apache.juneau.rest.server.mcp.v20260728.*;
24  
25  /**
26   * The OAuth 2.1-secured variant of {@link ExampleMcpServer}: identical notes-service surface (tools, prompt,
27   * resources, resource template, subscriptions), but every {@code POST /} JSON-RPC call now requires a valid
28   * bearer token.
29   *
30   * <p>
31   * This class changes nothing about <i>what</i> the server exposes &mdash; it inherits
32   * {@link ExampleMcpServer#createMcpConfig()} unmodified, so the same {@code publishNote}/{@code deleteNote}
33   * tools, {@code summarize} prompt, and {@code note:///...} resources are still there. The only override is
34   * {@link #createMcpOptions()}, which layers an {@code McpResourceServerConfig} on top of the parent's
35   * capabilities. That single override is the entire security story:
36   *
37   * <ul>
38   * 	<li><b>{@code setEnabled(true)}</b> &mdash; turns on the RS (resource-server) gate at all. Off by default,
39   * 		so an ordinary {@link ExampleMcpServer} is completely unaffected by this class existing.
40   * 	<li><b>{@code setResource(...)}</b> &mdash; this server's own canonical URL (RFC 9728 {@code resource} /
41   * 		RFC 8707 default audience). A bearer token's {@code aud} claim must include this exact URL or it is
42   * 		rejected &mdash; the confused-deputy defense: a token minted for some OTHER resource cannot be replayed
43   * 		here even if it is otherwise perfectly valid and signed by a trusted issuer.
44   * 	<li><b>{@code setTokenValidator(...)}</b> &mdash; a {@link JwtTokenValidator} that checks the token's
45   * 		signature (against {@link OfflineAuthorizationServer#jwkSource()}, offline &mdash; no JWKS HTTP fetch),
46   * 		{@code iss} (must equal {@link OfflineAuthorizationServer#issuerUri()}), and {@code aud}/{@code exp}/
47   * 		{@code nbf} (validated against the resource URL above and wall-clock time).
48   * 	<li><b>{@code addAuthorizationServer(...)}</b> &mdash; advertised in the RFC 9728 Protected Resource
49   * 		Metadata (PRM) document a client fetches from the well-known {@code .well-known/oauth-protected-resource}
50   * 		path this framework serves automatically once RS auth is enabled. This is how a compliant client
51   * 		discovers WHICH authorization server to go get a token from, without that URL being hardcoded into the
52   * 		client at all.
53   * 	<li><b>{@code addRequiredScope(...)}</b> &mdash; the coarse, endpoint-wide baseline scope ({@link #READ_SCOPE})
54   * 		every request must carry (on top of a merely valid, correctly-audienced token) before any JSON-RPC
55   * 		method dispatches. A token missing it gets a {@code 403} with a {@code WWW-Authenticate} challenge
56   * 		naming the missing scope, not a silent failure.
57   * 	<li><b>{@code addOperationScope(...)}</b> &mdash; SEP-2350 per-operation step-up (H3): {@code publishNote}
58   * 		and {@code deleteNote} additionally require {@link #WRITE_SCOPE} on top of the baseline. A token
59   * 		carrying only {@link #READ_SCOPE} can discover the server and read resources, but a mutating tool call
60   * 		gets its own scoped {@code 403 insufficient_scope} naming {@link #WRITE_SCOPE} specifically &mdash; the
61   * 		baseline scope alone is never enough to invoke either write tool.
62   * </ul>
63   *
64   * <p>
65   * The net effect on the wire: an unauthenticated (or wrongly-audienced, or expired, or insufficiently-scoped)
66   * {@code POST /} now gets {@code 401}/{@code 403} plus a {@code WWW-Authenticate: Bearer ...} challenge
67   * instead of ever reaching {@link ExampleMcpServer}'s tool/prompt/resource handlers; a request bearing a valid
68   * bearer token dispatches exactly as it always did.
69   *
70   * <h5 class='section'>The one wrinkle this class works around:</h5>
71   * <p>
72   * {@link #getResource() The resource URL} is this server's OWN address, which is a problem when
73   * {@link SecuredExampleServer} boots on an OS-assigned ephemeral port (as the end-to-end test does): naively,
74   * that address is not knowable until AFTER Jetty has bound its socket. It is tempting to assume
75   * {@link #createMcpOptions()} (like {@code createMcpConfig()}) is called lazily on the first incoming MCP
76   * request, late enough to bind the resource URL once the port is known &mdash; but it is NOT: the REST
77   * framework eagerly walks every {@code @Bean}-annotated method declared on the resource (including this
78   * inherited {@code getMcpOptions()}) while building the servlet's {@code RestContext}, i.e. during Jetty's
79   * OWN servlet initialization, strictly before {@code Server.start()} returns. A constructor-supplied resource
80   * URL is therefore required; see {@link SecuredExampleServer#start(int)} for how it opens the Jetty connector
81   * (and thus learns the real ephemeral port) before this servlet is even constructed.
82   *
83   * @serial exclude
84   */
85  @SuppressWarnings({
86  	"java:S110" // Inheritance depth is inherent to extending the Juneau REST server hierarchy.
87  })
88  public class SecuredExampleMcpServer extends ExampleMcpServer {
89  
90  	private static final long serialVersionUID = 1L;
91  
92  	/**
93  	 * The baseline OAuth scope every request to this server must carry, on top of a valid, correctly-audienced
94  	 * token. Advertised in the PRM document's {@code scopes_supported} and required via
95  	 * {@code McpResourceServerConfig.addRequiredScope(...)}.
96  	 */
97  	public static final String READ_SCOPE = "mcp.read";
98  
99  	/**
100 	 * The additional step-up scope {@code publishNote}/{@code deleteNote} require on top of {@link #READ_SCOPE}
101 	 * (H3, SEP-2350 per-operation scoping), via {@code McpResourceServerConfig.addOperationScope(...)}.
102 	 */
103 	public static final String WRITE_SCOPE = "mcp.write";
104 
105 	@SuppressWarnings("resource") // not owned here; lifecycle is managed by the caller (SecuredExampleServer).
106 	private final transient OfflineAuthorizationServer authServer;
107 	private final transient URI resource;
108 
109 	/**
110 	 * Constructor.
111 	 *
112 	 * @param authServer The offline authorization server this instance validates bearer tokens against
113 	 * 	(its {@link OfflineAuthorizationServer#jwkSource() public key} and
114 	 * 	{@link OfflineAuthorizationServer#issuerUri() issuer}). Must not be <jk>null</jk>.
115 	 * @param resource This server's own canonical root URL (e.g. {@code http://localhost:5001/}), known and
116 	 * 	fixed before construction &mdash; see the class javadoc's "one wrinkle" section for why this cannot be
117 	 * 	supplied later. Must not be <jk>null</jk>.
118 	 */
119 	public SecuredExampleMcpServer(OfflineAuthorizationServer authServer, URI resource) {
120 		if (authServer == null)
121 			throw new IllegalArgumentException("authServer must not be null");
122 		if (resource == null)
123 			throw new IllegalArgumentException("resource must not be null");
124 		this.authServer = authServer;
125 		this.resource = resource;
126 	}
127 
128 	/**
129 	 * Returns this server's own canonical resource URL, as supplied to the constructor.
130 	 *
131 	 * @return The resource URL. Never <jk>null</jk>.
132 	 */
133 	public URI getResource() {
134 		return resource;
135 	}
136 
137 	@Override
138 	protected McpOptions createMcpOptions() {
139 		var validator = JwtTokenValidator.create()
140 			.jwkSource(authServer.jwkSource())
141 			.issuer(authServer.issuerUri().toString())
142 			.audience(resource.toString())
143 			.build();
144 		return super.createMcpOptions().resourceServer(rs -> rs
145 			.setEnabled(true)
146 			.setResource(resource)
147 			.setTokenValidator(validator)
148 			.addAuthorizationServer(authServer.issuerUri())
149 			.addRequiredScope(READ_SCOPE)
150 			// H3: publishNote/deleteNote step up to WRITE_SCOPE on top of the READ_SCOPE baseline above.
151 			.addOperationScope("tools/call", "publishNote", WRITE_SCOPE)
152 			.addOperationScope("tools/call", "deleteNote", WRITE_SCOPE));
153 	}
154 }
