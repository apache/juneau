31  /**
32   * A guided walkthrough of calling {@link SecuredExampleMcpServer}, showing the secured request/response
33   * cycle a newcomer to MCP OAuth 2.1 needs to see: a rejected unauthenticated call (twice &mdash; once at the
34   * raw wire level, once through the real {@link McpClient} SDK), the full RFC 9728 &rarr; RFC 8414 discovery
35   * chain, a successful read-only call, a scoped step-up rejection, and finally a successful write once a token
36   * carrying the write scope has been acquired.
37   *
38   * <p>
39   * Run {@link #main(String[]) main} with three arguments &mdash; the secured server's endpoint and the demo
40   * client id/secret &mdash; all three of which {@link SecuredExampleServer#main(String[])} prints on startup
41   * (they cannot be hardcoded here: the demo client secret is freshly randomly generated on every server run, by
42   * design; see {@link OfflineAuthorizationServer}). Unlike an earlier version of this walkthrough, the token
43   * endpoint is no longer a fourth argument (M10): this client discovers it itself, the same way a real client
44   * would.
45   *
46   * @serial exclude
47   */
48  @SuppressWarnings({
49  	"java:S106" // Example walkthrough intentionally prints to stdout; console output is the demo's deliverable.
50  })
51  public final class SecuredExampleClient {
52  
53  	/** The demo client's advertised {@code clientInfo.name}, shared by every {@link McpClient} connection below. */
54  	private static final String CLIENT_NAME = "juneau-secured-notes-example-client";
55  
56  	/** The demo client's advertised {@code clientInfo.version}, shared by every {@link McpClient} connection below. */
57  	private static final String CLIENT_VERSION = "1.0.0";
58  
59  	// Shared across calls rather than one HttpClient per request (S2095): HttpClient instances are heavyweight
60  	// (they own a connection pool and a selector thread), and the JDK explicitly recommends reusing a single
61  	// instance for the lifetime of the application rather than churning through short-lived ones.
62  	private static final HttpClient HTTP_CLIENT = HttpClient.newHttpClient();
63  
64  	private SecuredExampleClient() {}
65  
66  	/**
67  	 * Runs the walkthrough against a running {@link SecuredExampleServer}.
68  	 *
69  	 * @param args Three required arguments: endpoint, client id, client secret &mdash; copy these from
70  	 * 	{@link SecuredExampleServer#main(String[])}'s startup banner.
71  	 * @throws Exception If any step fails unexpectedly (a REJECTED call is expected and handled, not thrown).
72  	 */
73  	@SuppressWarnings({
74  		"java:S112" // throws Exception intentional - example main() kept simple for demo readability
75  	})
76  	public static void main(String[] args) throws Exception {
77  		if (args.length < 3) {
78  			System.out.println("Usage: SecuredExampleClient <endpoint> <clientId> <clientSecret>");
79  			System.out.println("Copy these three values from the SecuredExampleServer startup banner.");
80  			return;
81  		}
82  		run(args[0], args[1], args[2]);
83  	}
84  
85  	/**
86  	 * Executes each numbered step of the walkthrough.
87  	 *
88  	 * @param endpoint The secured MCP server's endpoint URL.
89  	 * @param clientId The demo OAuth client id.
90  	 * @param clientSecret The demo OAuth client secret.
91  	 * @throws Exception If an unexpected (non-auth-related) failure occurs.
92  	 */
93  	@SuppressWarnings({
94  		"java:S112" // throws Exception intentional - example walkthrough kept simple for demo readability
95  	})
96  	public static void run(String endpoint, String clientId, String clientSecret) throws Exception {
97  
98  		section("1. Unauthenticated call, at the raw wire level — a 401 challenge");
99  		var challenge = rawUnauthenticatedCall(endpoint);
100 
101 		section("2. The same unauthenticated call through the real McpClient SDK");
102 		mcpClientUnauthenticatedCall(endpoint);
103 
104 		section("3. RFC 9728 discovery — fetch the Protected Resource Metadata the 401 pointed at");
105 		McpProtectedResourceMetadata prm = null;
106 		// javabugs:S6416/java:S3655 fix: capture the Optional ONCE and guard/read the SAME instance. The
107 		// original code called challenge.resourceMetadata() a second time inside the if-block to call get() -
108 		// a fresh Optional the isPresent() check above never actually guarded, which is exactly the discovery-
109 		// driven "get() without a proven-present check on that instance" pattern the bug engine flagged.
110 		var resourceMetadata = challenge == null ? Optional.<URI>empty() : challenge.resourceMetadata();
111 		if (resourceMetadata.isPresent())
112 			prm = discoverProtectedResourceMetadata(endpoint, resourceMetadata.get());
113 		else
114 			System.out.println("   (no resource_metadata pointer found on the challenge; skipping)");
115 
116 		section("4. RFC 8414 discovery — resolve the authorization server's token endpoint");
117 		var tokenEndpoint = discoverTokenEndpoint(prm);
118 
119 		section("5. Acquire a read-only (mcp.read) bearer token and call again — success");
120 		readOnlyCall(endpoint, tokenEndpoint, clientId, clientSecret);
121 
122 		section("6. The SAME read-only token attempting a write — a scoped 403 step-up challenge");
123 		insufficientScopeCall(endpoint, tokenEndpoint, clientId, clientSecret);
124 
125 		section("7. Step up to mcp.read + mcp.write and publish a note — success");
126 		writeCall(endpoint, tokenEndpoint, clientId, clientSecret);
127 
128 		System.out.println("\nWalkthrough complete.");
129 	}
130 
131 	/**
132 	 * Step 1: a raw HTTP POST with no {@code Authorization} header at all, bypassing the {@link McpClient} SDK
133 	 * entirely so the actual {@code 401} status and {@code WWW-Authenticate} response header are directly
134 	 * visible. This is necessary because the resource-server gate's {@code 401}/{@code 403} rejections are
135 	 * plain-text bodies, not JSON-RPC envelopes — {@code AbstractMcpClient.send(...)} cannot parse one into a
136 	 * typed result and instead surfaces the failure as a bare {@link IOException} whose message embeds the
137 	 * status code as text. There is no way to read the status code, let alone the challenge header, through
138 	 * the SDK itself; only at the wire level.
139 	 */
140 	private static WwwAuthenticateChallenge rawUnauthenticatedCall(String endpoint) throws IOException, InterruptedException {
141 		var request = HttpRequest.newBuilder(URI.create(endpoint))
142 			.header("Content-Type", "application/json")
143 			.header("Accept", "application/json")
144 			.header("Mcp-Method", "server/discover")
145 			.header("Mcp-Name", "")
146 			.POST(HttpRequest.BodyPublishers.ofString(discoverRequestBody()))
147 			.build();
148 		var response = HTTP_CLIENT.send(request, HttpResponse.BodyHandlers.ofString());
149 		System.out.println("   HTTP status: " + response.statusCode());
150 		var header = response.headers().firstValue("WWW-Authenticate").orElse(null);
151 		System.out.println("   WWW-Authenticate: " + header);
152 		var challenge = WwwAuthenticateChallenge.parse(header).orElse(null);
153 		if (challenge != null) {
154 			System.out.println("   parsed scope hint:       " + challenge.scopes());
155 			System.out.println("   parsed resource_metadata: " + challenge.resourceMetadata().orElse(null));
156 		}
157 		return challenge;
158 	}
159 
160 	/**
161 	 * Step 2: the same unauthenticated call, but through {@link McpClient#connect(String)} — the ergonomic
162 	 * failure mode a real caller actually sees: a thrown {@link IOException} (see {@link #rawUnauthenticatedCall}
163 	 * above for why it is a bare {@link IOException} and not a typed exception) out of the mandatory
164 	 * {@code server/discover} handshake.
165 	 */
166 	private static void mcpClientUnauthenticatedCall(String endpoint) {
167 		try (var ignored = McpClient.connect(endpoint)) {
168 			throw new IllegalStateException("expected the unauthenticated connect() to fail, but it succeeded");
169 		} catch (IOException e) {
170 			System.out.println("   rejected as expected: " + e.getMessage());
171 		}
172 	}
173 
174 	/**
175 	 * Step 3: fetches the RFC 9728 Protected Resource Metadata document the challenge pointed at, showing what
176 	 * a client learns from it — notably WHICH authorization server to acquire a token from, discovered from
177 	 * the response rather than hardcoded into this client.
178 	 */
179 	private static McpProtectedResourceMetadata discoverProtectedResourceMetadata(String endpoint, URI resourceMetadataUrl) {
180 		var prm = McpProtectedResourceMetadataClient.create()
181 			.expectedResource(URI.create(endpoint))
182 			.build()
183 			.fetch(resourceMetadataUrl);
184 		System.out.println("   resource:               " + prm.resource());
185 		System.out.println("   authorization_servers:  " + prm.authorizationServers());
186 		System.out.println("   scopes_supported:       " + prm.scopesSupported());
187 		return prm;
188 	}
189 
190 	/**
191 	 * Step 4 (M10): performs real RFC 8414 discovery against the authorization server the PRM document
192 	 * advertised, via {@link McpProtectedResourceMetadataClient#discoverAuthorizationServer}, and returns the
193 	 * discovered {@code token_endpoint} — nothing here is hardcoded or passed in on the command line. This
194 	 * completes the full 401 &rarr; {@code resource_metadata} &rarr; PRM &rarr; {@code authorization_servers}
195 	 * &rarr; RFC 8414 &rarr; {@code token_endpoint} chain a compliant client walks end-to-end.
196 	 */
197 	private static URI discoverTokenEndpoint(McpProtectedResourceMetadata prm) {
198 		if (prm == null)
199 			throw new IllegalStateException("no Protected Resource Metadata was discovered in step 3");
200 		var as = McpProtectedResourceMetadataClient.create().build().discoverAuthorizationServer(prm);
201 		System.out.println("   issuer:         " + as.issuer());
202 		System.out.println("   token_endpoint: " + as.tokenEndpoint());
203 		return as.tokenEndpoint();
204 	}
205 
206 	/**
207 	 * Step 5: acquires a real bearer token scoped to {@code mcp.read} only via
208 	 * {@link McpTokenProvider#clientCredentials()} (a genuine RFC 6749 &sect;4.4 HTTP round trip to the
209 	 * discovered token endpoint), wires it into a fresh {@link McpClient} via
210 	 * {@link McpTokenProvider#interceptor()}, and shows a read succeeding transparently.
211 	 */
212 	private static void readOnlyCall(String endpoint, URI tokenEndpoint, String clientId, String clientSecret) throws IOException {
213 		var tokens = readOnlyTokenProvider(endpoint, tokenEndpoint, clientId, clientSecret);
214 		try (var client = McpClient.connect(McpClient.builder()
215 				.endpoint(endpoint)
216 				.clientInfo(new Implementation().setName(CLIENT_NAME).setVersion(CLIENT_VERSION))
217 				.interceptor(tokens.interceptor()))) {
218 			System.out.println("   server/discover succeeded: " + Json.of(client.discoveredServer().getServerInfo()));
219 			var read = client.readResource(NoteStore.SCHEME + "index");
220 			System.out.println("   " + Json.of(read.getContents()));
221 		}
222 	}
223 
224 	/**
225 	 * Step 6 (H3): the SAME read-only-scoped token from step 5's flavor, now attempting {@code publishNote} —
226 	 * a mutating tool that {@link SecuredExampleMcpServer} step-up-gates behind {@code mcp.write}. Demonstrates
227 	 * that holding a merely-valid, correctly-scoped-for-reads token is deliberately NOT enough: the call is
228 	 * rejected with a {@code 403} naming the missing {@code mcp.write} scope, not silently allowed. As with
229 	 * step 1/2 above, the 403's plain-text body surfaces as a bare {@link IOException}, not a typed exception.
230 	 */
231 	private static void insufficientScopeCall(String endpoint, URI tokenEndpoint, String clientId, String clientSecret) {
232 		var tokens = readOnlyTokenProvider(endpoint, tokenEndpoint, clientId, clientSecret);
233 		try (var client = McpClient.connect(McpClient.builder()
234 				.endpoint(endpoint)
235 				.clientInfo(new Implementation().setName(CLIENT_NAME).setVersion(CLIENT_VERSION))
236 				.interceptor(tokens.interceptor()))) {
237 			client.callToolText("publishNote", Map.of("title", "should-fail", "body", "should never be stored"));
238 			throw new IllegalStateException("expected the write with a read-only token to fail, but it succeeded");
239 		} catch (IOException e) {
240 			System.out.println("   rejected as expected: " + e.getMessage());
241 		}
242 	}
243 
244 	/**
245 	 * Step 7: acquires a SECOND token, this time carrying both {@code mcp.read} and {@code mcp.write}, and
246 	 * shows the previously-rejected {@code publishNote} call (and a confirming read) now succeeding.
247 	 */
248 	private static void writeCall(String endpoint, URI tokenEndpoint, String clientId, String clientSecret) throws IOException {
249 		var tokens = McpTokenProvider.clientCredentials()
250 			.tokenEndpoint(tokenEndpoint)
251 			.clientId(clientId)
252 			.clientSecret(clientSecret)
253 			.resource(URI.create(endpoint))
254 			.scope(SecuredExampleMcpServer.READ_SCOPE, SecuredExampleMcpServer.WRITE_SCOPE)
255 			.build();
256 
257 		try (var client = McpClient.connect(McpClient.builder()
258 				.endpoint(endpoint)
259 				.clientInfo(new Implementation().setName(CLIENT_NAME).setVersion(CLIENT_VERSION))
260 				.interceptor(tokens.interceptor()))) {
261 			System.out.println("   -> " + client.callToolText("publishNote",
262 				Map.of("title", "secured", "body", "Hello from behind OAuth 2.1")));
263 
264 			var read = client.readResource(NoteStore.uriFor("secured"));
265 			System.out.println("   " + Json.of(read.getContents()));
266 		}
267 	}
268 
269 	/** Builds a token provider scoped to {@link SecuredExampleMcpServer#READ_SCOPE} only. */
270 	private static McpTokenProvider readOnlyTokenProvider(String endpoint, URI tokenEndpoint, String clientId, String clientSecret) {
271 		return McpTokenProvider.clientCredentials()
272 			.tokenEndpoint(tokenEndpoint)
273 			.clientId(clientId)
274 			.clientSecret(clientSecret)
275 			.resource(URI.create(endpoint))
276 			.scope(SecuredExampleMcpServer.READ_SCOPE)
277 			.build();
278 	}
279 
280 	/**
281 	 * Builds a plausible (but not necessarily complete) {@code server/discover} JSON-RPC request body using
282 	 * the real wire beans, purely so step 1's raw HTTP call looks like a genuine MCP request on the wire. The
283 	 * resource-server bearer gate rejects it before the body is ever parsed, so its exact contents do not
284 	 * affect the outcome — but a reader should see real MCP traffic here, not an arbitrary placeholder.
285 	 */
286 	private static String discoverRequestBody() {
287 		var params = new RequestParamsOnly().setMeta(new RequestMeta()
288 			.setProtocolVersion(McpProtocol.VERSION_2026_07_28)
289 			.setClientInfo(new Implementation().setName(CLIENT_NAME).setVersion(CLIENT_VERSION))
290 			.setClientCapabilities(new ClientCapabilities()));
291 		var req = new JsonRpcRequest()
292 			.setJsonrpc(McpProtocol.JSON_RPC_2_0)
293 			.setId(1)
294 			.setMethod("server/discover")
295 			.setParams(params);
296 		return Json.of(req);
297 	}
