17  package org.apache.juneau.examples.mcp.secured;
18  
19  import java.net.*;
20  
21  import org.apache.juneau.commons.inject.*;
22  import org.apache.juneau.microservice.*;
23  import org.apache.juneau.microservice.jetty.*;
24  import org.eclipse.jetty.ee11.servlet.*;
25  import org.eclipse.jetty.server.*;
26  
27  import jakarta.servlet.*;
28  
29  /**
30   * Boots {@link SecuredExampleMcpServer} on an embedded Jetty server, alongside an in-process
31   * {@link OfflineAuthorizationServer} that stands in for a real OAuth 2.1 authorization server.
32   *
33   * <p>
34   * Mirrors {@link org.apache.juneau.examples.mcp.ExampleServer} (read that class first &mdash; this one only
35   * documents what is different). Run {@link #main(String[]) main} to start both servers on fixed ports and
36   * print everything a {@link SecuredExampleClient} run needs to copy: the endpoint, the offline authorization
37   * server's token endpoint, and the randomly-generated demo client id/secret. The in-process end-to-end test
38   * uses {@link #start(int) start(0)} to boot on an OS-assigned ephemeral port instead.
39   *
40   * <p>
41   * <b>The resource URL is resolved before the secured servlet is even constructed.</b> {@link SecuredExampleMcpServer}
42   * needs its own address up front (see its class javadoc), which is awkward on an OS-assigned ephemeral port
43   * because normally nothing knows that port until Jetty finishes starting. The fix: {@link ServerConnector#open()}
44   * can be, and by Jetty's own convention normally is (see {@code Server.doStart()}'s "open network connector to
45   * ensure ports are available" step), called to bind the listening socket BEFORE the rest of the server starts.
46   * Calling it ourselves, one step earlier than usual, lets us read {@link ServerConnector#getLocalPort()}
47   * immediately &mdash; before building {@link SecuredExampleMcpServer} or starting the {@link Microservice} at
48   * all. Jetty's own {@code Server.start()} then finds the connector already open and skips rebinding it.
49   */
50  @SuppressWarnings({
51  	"java:S106" // Example walkthrough intentionally prints to stdout; console output is the demo's deliverable.
52  })
53  public final class SecuredExampleServer implements AutoCloseable {
54  
55  	/** Default listen port used by {@link #main(String[])} when none is supplied. */
56  	public static final int DEFAULT_PORT = 5001;
57  
58  	private final Microservice microservice;
59  	@SuppressWarnings("resource") // closed in close(); this field's lifecycle is fully managed by this class.
60  	private final OfflineAuthorizationServer authServer;
61  	private final URI rootUrl;
62  
63  	private SecuredExampleServer(Microservice microservice, OfflineAuthorizationServer authServer, URI rootUrl) {
64  		this.microservice = microservice;
65  		this.authServer = authServer;
66  		this.rootUrl = rootUrl;
67  	}
68  
69  	/**
70  	 * Starts the secured example MCP server and its offline authorization server.
71  	 *
72  	 * @param port The TCP port the MCP server listens on, or {@code 0} to let the OS assign an ephemeral port.
73  	 * 	(The offline authorization server always uses its own separate OS-assigned ephemeral port.)
74  	 * @return A running server handle. Close it (or call {@link #close()}) to stop both servers.
75  	 * @throws Exception If either server fails to start.
76  	 */
77  	@SuppressWarnings("resource") // The bean store is handed to (and closed by) the Microservice lifecycle.
78  	public static SecuredExampleServer start(int port) throws Exception {
79  		var authServer = OfflineAuthorizationServer.start();
80  		// M9: hoisted above the try so a failure AFTER connector.open() (which has already bound a real OS
81  		// socket) can still close it in the catch below - Server.stop() alone will not close a connector that
82  		// was opened but never handed to a started Server.
83  		ServerConnector connector = null;
84  		try {
85  			var jetty = new Server();
86  			connector = new ServerConnector(jetty);
87  			connector.setPort(port);
88  			jetty.addConnector(connector);
89  			// Bind the (possibly ephemeral) listen socket now, before the servlet context (and therefore
90  			// SecuredExampleMcpServer's RestContext) is built, so the resource URL below is real and final
91  			// by the time anything asks the servlet for it. See the class javadoc for why that ordering
92  			// matters here specifically.
93  			connector.open();
94  			var rootUrl = URI.create("http://localhost:" + connector.getLocalPort() + "/");
95  
96  			var handler = new ServletContextHandler();
97  			handler.setContextPath("/");
98  			jetty.setAttribute("ServletContextHandler", handler);
99  			jetty.setHandler(handler);
100 
101 			var securedServlet = new SecuredExampleMcpServer(authServer, rootUrl);
102 
103 			var beanStore = new BasicBeanStore();
104 			beanStore.addBean(Server.class, jetty);
105 			beanStore.addBean(Servlet.class, securedServlet);
106 
107 			var microservice = Microservice.create()
108 				.beanStore(beanStore)
109 				.configurations(JettyConfiguration.class)
110 				.consoleEnabled(false)
111 				.build();
112 			microservice.start();
113 
114 			return new SecuredExampleServer(microservice, authServer, rootUrl);
115 		} catch (Exception e) {
116 			if (connector != null)
117 				connector.close();
118 			authServer.close();
119 			throw e;
120 		}
121 	}
122 
123 	/**
124 	 * Returns the root URL the secured MCP server is listening on (e.g. {@code http://localhost:5001/}).
125 	 *
126 	 * @return The root URL. Never <jk>null</jk>.
127 	 */
128 	public URI getRootUrl() {
129 		return rootUrl;
130 	}
131 
132 	/**
133 	 * Returns the offline authorization server backing this instance, so a caller (e.g. the end-to-end test,
134 	 * or {@link #main(String[])} printing a startup banner) can read its token endpoint and demo credentials.
135 	 *
136 	 * @return The offline authorization server. Never <jk>null</jk>.
137 	 */
138 	public OfflineAuthorizationServer getAuthServer() {
139 		return authServer;
140 	}
141 
142 	@Override
143 	public void close() throws Exception {
144 		try {
145 			microservice.stop();
146 		} finally {
147 			authServer.close();
148 		}
149 	}
150 
151 	/**
152 	 * Runs both servers until the process is killed, printing everything {@link SecuredExampleClient} needs.
153 	 *
154 	 * <p>
155 	 * M10: the token endpoint is no longer printed here &mdash; {@link SecuredExampleClient} now discovers it
156 	 * itself via RFC 9728 PRM + RFC 8414 discovery, exactly as a real client would, so only the endpoint and
157 	 * demo client credentials need to be copied.
158 	 *
159 	 * @param args Optional single argument: the port the MCP server listens on (defaults to {@link #DEFAULT_PORT}).
160 	 * @throws Exception If either server fails to start.
161 	 */
162 	@SuppressWarnings("resource") // example server (and its auth server) run for the JVM lifetime; closed on process exit.
163 	public static void main(String[] args) throws Exception {
164 		var port = args.length > 0 ? Integer.parseInt(args[0]) : DEFAULT_PORT;
165 		var server = start(port);
166 		var auth = server.getAuthServer();
167 		System.out.println("Juneau SECURED MCP example server is listening at " + server.getRootUrl());
168 		System.out.println("Demo client id:     " + auth.clientId());
169 		System.out.println("Demo client secret: " + auth.clientSecret());
170 		System.out.println();
171 		System.out.println("Drive it with:  SecuredExampleClient " + server.getRootUrl() + " "
172 			+ auth.clientId() + " " + auth.clientSecret());
173 		System.out.println("Press Ctrl-C to stop.");
174 		Thread.currentThread().join();
175 	}
176 }
