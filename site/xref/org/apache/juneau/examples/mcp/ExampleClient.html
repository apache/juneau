17  package org.apache.juneau.examples.mcp;
18  
19  import java.io.*;
20  import java.util.*;
21  import java.util.concurrent.*;
22  
23  import org.apache.juneau.bean.mcp.v20260728.*;
24  import org.apache.juneau.marshall.marshaller.Json;
25  import org.apache.juneau.rest.client.mcp.v20260728.*;
26  
27  /**
28   * A guided, end-to-end walkthrough of every {@link ExampleMcpServer} surface using the v2
29   * {@link McpClient}.
30   *
31   * <p>
32   * Run {@link #main(String[]) main} (optionally passing the server endpoint; defaults to
33   * {@code http://localhost:5000/}) after starting {@link ExampleServer}. Each numbered step prints
34   * what it is doing and what it got back, so the transcript reads top-to-bottom like a tutorial.
35   *
36   * <p>
37   * The client advertises the {@code elicitation} capability, which the server requires before it will
38   * pause a call for input; without it the {@code deleteNote} elicitation step would be rejected.
39   *
40   * <p>
41   * <b>Console output caveat:</b> every printed line below is produced by re-serializing the already-parsed
42   * response bean with {@code Json.of(...)}, purely for readable console output. That re-serialization drops
43   * the polymorphic content-block {@code "type"} discriminator (e.g. {@code "text"}/{@code "audio"}) the real
44   * wire format carries, so what you see here is not byte-for-byte what came off the wire.
45   */
46  @SuppressWarnings({
47  	"java:S106" // Example walkthrough intentionally prints to stdout; console output is the demo's deliverable.
48  })
49  public final class ExampleClient {
50  
51  	/** The {@code title} argument/variable name shared by the tool, prompt, and resource-template calls below. */
52  	private static final String TITLE_ARG = "title";
53  
54  	/** The sample note title used throughout this walkthrough. */
55  	private static final String GROCERIES_NOTE_TITLE = "groceries";
56  
57  	private ExampleClient() {}
58  
59  	/**
60  	 * Runs the walkthrough against a running {@link ExampleServer}.
61  	 *
62  	 * @param args Optional single argument: the server endpoint (defaults to {@code http://localhost:5000/}).
63  	 * @throws Exception If any step fails.
64  	 */
65  	@SuppressWarnings({
66  		"java:S112" // throws Exception intentional - example main() kept simple for demo readability
67  	})
68  	public static void main(String[] args) throws Exception {
69  		var endpoint = args.length > 0 ? args[0] : "http://localhost:" + ExampleServer.DEFAULT_PORT + "/";
70  		try (var client = connect(endpoint)) {
71  			run(client);
72  		}
73  	}
74  
75  	/**
76  	 * Builds and connects a client that advertises the elicitation capability.
77  	 *
78  	 * @param endpoint The MCP server endpoint URL.
79  	 * @return A connected client (its mandatory {@code server/discover} handshake already done).
80  	 * @throws IOException If the connection or handshake fails.
81  	 */
82  	@SuppressWarnings("resource") // returned client is owned and closed by the caller (see main() above).
83  	public static McpClient connect(String endpoint) throws IOException {
84  		return McpClient.connect(McpClient.builder()
85  			.endpoint(endpoint)
86  			.clientInfo(new Implementation().setName("juneau-notes-example-client").setVersion("1.0.0"))
87  			// Required for the deleteNote elicitation step: the server only pauses for input when the
88  			// client says it can answer.
89  			.clientCapabilities(new ClientCapabilities().setElicitation(new ElicitationCapability())));
90  	}
91  
92  	/**
93  	 * Executes each numbered step of the walkthrough against an already-connected client.
94  	 *
95  	 * @param client The connected MCP client.
96  	 * @throws Exception If any step fails.
97  	 */
98  	@SuppressWarnings({
99  		"java:S112" // throws Exception intentional - example walkthrough kept simple for demo readability
100 	})
101 	public static void run(McpClient client) throws Exception {
102 
103 		section("1. server/discover — who are we talking to?");
104 		var discovered = client.discoveredServer();
105 		System.out.println("   server info:  " + Json.of(discovered.getServerInfo()));
106 		System.out.println("   versions:     " + Json.of(discovered.getSupportedVersions()));
107 		System.out.println("   capabilities: " + Json.of(discovered.getCapabilities()));
108 		System.out.println("   instructions: " + discovered.getInstructions());
109 
110 		section("2. Discovery — list every advertised surface");
111 		System.out.println("   tools:             " + Json.of(client.listTools().getTools()));
112 		System.out.println("   prompts:           " + Json.of(client.listPrompts().getPrompts()));
113 		System.out.println("   resources:         " + Json.of(client.listResources().getResources()));
114 		System.out.println("   resourceTemplates: " + Json.of(client.listResourceTemplates().getResourceTemplates()));
115 
116 		section("3. tools/call publishNote — store a note (and notify subscribers)");
117 		System.out.println("   -> " + client.callToolText("publishNote", Map.of(TITLE_ARG, GROCERIES_NOTE_TITLE, "body", "Milk, eggs, bread")));
118 
119 		section("4. resources/read note:///groceries — read it back via the template");
120 		var read = client.readResource(NoteStore.uriFor(GROCERIES_NOTE_TITLE));
121 		System.out.println("   " + Json.of(read.getContents()));
122 
123 		section("5. resources/read note:///index — the fixed index resource");
124 		System.out.println("   " + Json.of(client.readResource(NoteStore.SCHEME + "index").getContents()));
125 
126 		section("6. completion/complete — complete the template's {title} variable for prefix 'gr'");
127 		var ref = new ResourceTemplateReference().setUri(NoteStore.SCHEME + "{title}");
128 		var completion = client.complete(ref, TITLE_ARG, "gr", null);
129 		System.out.println("   suggestions: " + Json.of(completion.getCompletion()));
130 
131 		section("7. prompts/get summarize — render a prompt from the stored note");
132 		var prompt = client.getPrompt("summarize", Map.of(TITLE_ARG, GROCERIES_NOTE_TITLE));
133 		System.out.println("   " + Json.of(prompt.getMessages()));
134 
135 		section("8. subscriptions/listen — receive a live change notification");
136 		runSubscriptionDemo(client);
137 
138 		section("9. tools/call deleteNote — an elicitation (confirm) round-trip, auto-answered");
139 		var deleted = client.callToolWithElicitation("deleteNote", Map.of(TITLE_ARG, GROCERIES_NOTE_TITLE), requests -> {
140 			// The server asked one or more questions; answer each with ACCEPT + confirm=true. A real client
141 			// would present these to a user (the schema in each request says how to render the control).
142 			System.out.println("   server asked: " + Json.of(requests));
143 			var answers = new LinkedHashMap<String,ElicitResult>();
144 			requests.keySet().forEach(id -> answers.put(id,
145 				new ElicitResult().setAction(ElicitAction.ACCEPT).putContent("confirm", true)));
146 			return answers;
147 		});
148 		System.out.println("   -> " + deleted.firstText());
149 
150 		section("10. resources/read note:///index — confirm the note is gone");
151 		System.out.println("   " + Json.of(client.readResource(NoteStore.SCHEME + "index").getContents()));
152 
153 		System.out.println("\nWalkthrough complete.");
154 	}
155 
156 	/**
157 	 * Subscribes for changes to a note URI, publishes that note, and prints the change frame that arrives
158 	 * over the held-open SSE stream.
159 	 */
160 	@SuppressWarnings({
161 		"java:S112" // throws Exception intentional - example walkthrough kept simple for demo readability
162 	})
163 	private static void runSubscriptionDemo(McpClient client) throws Exception {
164 		var noteUri = NoteStore.uriFor("todo");
165 		var updates = new LinkedBlockingQueue<String>();
166 		var acknowledged = new CountDownLatch(1);
167 
168 		var handle = client.listen(
169 			new SubscriptionFilter().setResourceSubscriptions(List.of(noteUri)),
170 			new McpSubscriptionListener() {
171 				@Override public void onAcknowledged(SubscriptionFilter honoredFilter) { acknowledged.countDown(); }
172 				@Override public void onResourceUpdated(String uri) { updates.add(uri); }
173 				@Override public void onError(Throwable t) { System.out.println("   subscription error: " + t); }
174 			});
175 		try {
176 			if (! acknowledged.await(10, TimeUnit.SECONDS))
177 				System.out.println("   (warning: subscription was not acknowledged in time)");
178 			else
179 				System.out.println("   subscription acknowledged; publishing '" + noteUri + "' ...");
180 
181 			client.callTool("publishNote", Map.of(TITLE_ARG, "todo", "body", "Write MCP example"));
182 
183 			var updatedUri = updates.poll(10, TimeUnit.SECONDS);
184 			if (updatedUri == null)
185 				System.out.println("   (warning: no resources/updated notification arrived in time)");
186 			else
187 				System.out.println("   -> received resources/updated for: " + updatedUri);
188 		} finally {
189 			handle.close();
190 		}
191 	}
192 
193 	private static void section(String title) {
194 		System.out.println("\n=== " + title + " ===");
195 	}
196 }
