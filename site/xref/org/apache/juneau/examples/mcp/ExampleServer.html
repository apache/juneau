17  package org.apache.juneau.examples.mcp;
18  
19  import java.net.*;
20  
21  import org.apache.juneau.commons.inject.*;
22  import org.apache.juneau.microservice.*;
23  import org.apache.juneau.microservice.jetty.*;
24  import org.eclipse.jetty.ee11.servlet.*;
25  import org.eclipse.jetty.server.*;
26  
27  import jakarta.servlet.*;
28  
29  /**
30   * Boots {@link ExampleMcpServer} on an embedded Jetty server.
31   *
32   * <p>
33   * Run {@link #main(String[]) main} to start the server on a fixed port (default {@code 5000}, or pass a
34   * port as the first argument) and leave it running so you can point {@link ExampleClient} (or any MCP
35   * client) at it. The in-process end-to-end test uses {@link #start(int) start(0)} to boot on an
36   * OS-assigned ephemeral port instead.
37   *
38   * <p>
39   * The wiring mirrors Juneau's own test fixture: a Jetty {@link Server} plus the {@link ExampleMcpServer}
40   * servlet are placed in a {@link BasicBeanStore}, and {@link JettyConfiguration} (added via
41   * {@link org.apache.juneau.microservice.Microservice.Builder#configurations(Class...)}) auto-mounts the {@link org.apache.juneau.rest.server.Rest @Rest}
42   * servlet at {@code "/"} and drives the server lifecycle. Because the {@code Server} is supplied as a bean,
43   * we control the listen port directly &mdash; no {@code jetty.xml} required.
44   */
45  @SuppressWarnings({
46  	"java:S106" // Example walkthrough intentionally prints to stdout; console output is the demo's deliverable.
47  })
48  public final class ExampleServer implements AutoCloseable {
49  
50  	/** Default listen port used by {@link #main(String[])} when none is supplied. */
51  	public static final int DEFAULT_PORT = 5000;
52  
53  	private final Microservice microservice;
54  	private final URI rootUrl;
55  
56  	private ExampleServer(Microservice microservice, URI rootUrl) {
57  		this.microservice = microservice;
58  		this.rootUrl = rootUrl;
59  	}
60  
61  	/**
62  	 * Starts the example MCP server.
63  	 *
64  	 * @param port The TCP port to listen on, or {@code 0} to let the OS assign an ephemeral port.
65  	 * @return A running server handle. Close it (or call {@link #close()}) to stop.
66  	 * @throws Exception If the server fails to start.
67  	 */
68  	@SuppressWarnings("resource") // The bean store is handed to (and closed by) the Microservice lifecycle.
69  	public static ExampleServer start(int port) throws Exception {
70  		var jetty = buildServer(port);
71  
72  		var beanStore = new BasicBeanStore();
73  		beanStore.addBean(Server.class, jetty);
74  		beanStore.addBean(Servlet.class, new ExampleMcpServer());
75  
76  		var microservice = Microservice.create()
77  			.beanStore(beanStore)
78  			.configurations(JettyConfiguration.class)
79  			// This example has no interactive commands to offer; skip the Java console entirely so
80  			// startup doesn't print "Could not create console command" for whatever default console
81  			// commands a discovered config might otherwise register.
82  			.consoleEnabled(false)
83  			.build();
84  		microservice.start();
85  
86  		return new ExampleServer(microservice, URI.create("http://localhost:" + boundPort(jetty) + "/"));
87  	}
88  
89  	/**
90  	 * Returns the root URL the server is listening on (e.g. {@code http://localhost:5000/}).
91  	 *
92  	 * @return The root URL. Never <jk>null</jk>.
93  	 */
94  	public URI getRootUrl() {
95  		return rootUrl;
96  	}
97  
98  	@Override
99  	public void close() throws Exception {
100 		microservice.stop();
101 	}
102 
103 	/**
104 	 * Runs the server until the process is killed.
105 	 *
106 	 * @param args Optional single argument: the port to listen on (defaults to {@link #DEFAULT_PORT}).
107 	 * @throws Exception If the server fails to start.
108 	 */
109 	@SuppressWarnings("resource") // example server runs for the JVM lifetime; closed on process exit.
110 	public static void main(String[] args) throws Exception {
111 		var port = args.length > 0 ? Integer.parseInt(args[0]) : DEFAULT_PORT;
112 		var server = start(port);
113 		System.out.println("Juneau MCP example server is listening at " + server.getRootUrl());
114 		System.out.println("Drive it with:  ExampleClient " + server.getRootUrl());
115 		System.out.println("Press Ctrl-C to stop.");
116 		// Microservice.join() defaults to a documented no-op, so it won't actually block here; join the
117 		// main thread directly to keep the process (and the Jetty daemon threads it started) alive.
118 		Thread.currentThread().join();
119 	}
120 
121 	/** Builds a Jetty server with a single connector on {@code port} and a root servlet context. */
122 	@SuppressWarnings("resource") // connector is added to and owned by the server; closed when the server stops.
123 	private static Server buildServer(int port) {
124 		var server = new Server();
125 		var connector = new ServerConnector(server);
126 		connector.setPort(port);
127 		server.addConnector(connector);
128 		var handler = new ServletContextHandler();
129 		handler.setContextPath("/");
130 		// JettyServerComponent discovers the handler via this attribute (same convention as jetty.xml).
131 		server.setAttribute("ServletContextHandler", handler);
132 		server.setHandler(handler);
133 		return server;
134 	}
135 
136 	/** Reads the actual bound port (meaningful even when the server was started on port 0). */
137 	private static int boundPort(Server server) {
138 		for (var connector : server.getConnectors())
139 			if (connector instanceof ServerConnector sc)
140 				return sc.getLocalPort();
141 		throw new IllegalStateException("Could not determine the bound port of the Jetty server.");
142 	}
143 }
