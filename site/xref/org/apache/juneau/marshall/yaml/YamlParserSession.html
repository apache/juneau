129 	/**
130 	 * Buffered array-element {@link RecordReader} for YAML, bound to this live session.  Calls
131 	 * {@code parse(input, List.class, Object.class)} once and iterates the result.
132 	 *
133 	 * <h5 class='section'>Streaming status:</h5>
134 	 * <p>
135 	 * A top-level YAML block sequence ({@code - elem} per line) <i>is</i> forward-readable in
136 	 * principle, so true element-at-a-time streaming is <b>feasible</b> here (unlike Parquet/BSON).
137 	 * It is left buffered for now: Juneau's YAML reader is a single hand-written indentation state
138 	 * machine ({@code readBlockSequence}/{@code readFlowSequence}) whose per-element advance and
139 	 * dedent bookkeeping are not currently exposed as a pull cursor, and extracting that safely is a
140 	 * non-trivial parser refactor with real regression surface.  This
141 	 * is demand-driven &mdash; convert when a large-YAML-array use case justifies the refactor.
142 	 *
143 	 * @param input The input.
144 	 * @return A buffered {@link RecordReader}.
145 	 * @throws IOException If a problem occurred reading the input.
146 	 */
147 	@Override /* ArrayRecordReadable */
148 	public RecordReader readArrayRecords(Object input) throws IOException {
149 		return RecordAdapter.arrayReader(this, input);
150 	}

172 	@Override /* Overridden from ReaderParserSession */
173 	protected <E> Collection<E> doReadIntoCollection(ParserPipe pipe, Collection<E> c, Type elementType) throws IOException, ParseException, ExecutableException {
174 		try (var r = pipe.getParserReader()) {
175 			skipWhitespaceAndComments(r);
176 			int ch = r.peek();
177 			if (ch == '[') {
178 				readFlowSequence(r, c, getClassMeta(elementType), null);
179 			} else {
180 				readBlockSequence(r, c, getClassMeta(elementType), null, 0);
181 			}
182 			return c;
183 		}
184 	}

193 			} else {
194 				readBlockMapping(r, m, (ClassMeta<K>)getClassMeta(keyType), (ClassMeta<V>)getClassMeta(valueType), null, 0);

196 			return m;

204 	private <T> T readAnything(ClassMeta<?> eType, ParserReader r, Object outer, BeanPropertyMeta pMeta) throws IOException, ParseException, ExecutableException {
205 
206 		if (eType == null)
207 			eType = object();
208 		var swap = (ObjectSwap<T,Object>)eType.getSwap(this);
209 		var builder = (BuilderSwap<T,Object>)eType.getBuilderSwap(this);
210 		ClassMeta<?> sType = null;
211 		if (nn(builder))
212 			sType = builder.getBuilderClassMeta(this);
213 		else if (nn(swap))
214 			sType = swap.getSwapClassMeta(this);
215 		else
216 			sType = eType;
217 
218 		if (sType.isOptional())
219 			return (T)o(readAnything(eType.getElementType(), r, outer, pMeta));
220 
221 		setCurrentClass(sType);
222 
223 		Object o = null;
224 
225 		skipWhitespaceAndComments(r);
226 		skipDocumentMarker(r);
227 		skipWhitespaceAndComments(r);
228 
229 		int contentColumn = r.getPosition().getColumn();
230 		int c = r.peek();
231 		if (c == -1) {
232 			// Let o be null.
233 		} else if (c == '{') {
234 			if (sType.isObject()) {
235 				var m2 = newGenericMap();
236 				readFlowMapping(r, m2, string(), object(), pMeta);
237 				o = cast(m2, pMeta, eType);
238 			} else if (nn(builder)) {
239 				var m = toBeanMap(builder.create(this, eType));
240 				o = builder.build(this, readIntoBeanMap(r, m).getBean(), eType);
241 			} else if (sType.canCreateNewBean(outer)) {
242 				var m = newBeanMap(outer, sType.inner());
243 				o = readIntoBeanMap(r, m).getBean();
244 			} else if (sType.isMap()) {
245 				Map m = (sType.canCreateNewInstance(outer) ? (Map)sType.newInstance(outer) : newGenericMap(sType));
246 				o = readFlowMapping(r, m, sType.getKeyType(), sType.getValueType(), pMeta);
247 			} else if (sType.isCollection()) {
248 				var m = newGenericMap();
249 				readFlowMapping(r, m, string(), object(), pMeta);
250 				o = cast(m, pMeta, eType);
251 			} else if (sType.isArray() || sType.isArgs()) {
252 				var m = newGenericMap();
253 				readFlowMapping(r, m, string(), object(), pMeta);
254 				o = cast(m, pMeta, eType);
255 			} else {
256 				Map m = newGenericMap();
257 				readFlowMapping(r, m, sType.getKeyType(), sType.getValueType(), pMeta);
258 				if (m.containsKey(getBeanTypePropertyName(eType)))
259 					o = cast((MarshalledMap)m, pMeta, eType);
260 				else if (nn(sType.getProxyInvocationHandler()))
261 					o = newBeanMap(outer, sType.inner()).load(m).getBean();
262 				else
263 					throw new ParseException(this, "Class '%s' could not be instantiated.  Reason: '%s'", cn(sType), sType.getNotABeanReason());
264 			}
265 		} else if (c == '[') {
266 			if (sType.isObject()) {
267 				o = readFlowSequence(r, newGenericList(), object(), pMeta);
268 			} else if (sType.isCollection()) {
269 				Collection l = (sType.canCreateNewInstance(outer) ? (Collection)sType.newInstance() : newGenericList());
270 				o = readFlowSequence(r, l, sType, pMeta);
271 			} else if (sType.isArray() || sType.isArgs()) {
272 				var l = (ArrayList)readFlowSequence(r, list(), sType, pMeta);
273 				o = toArray(sType, l);
274 			} else {
275 				throw new ParseException(this, "Unrecognized syntax for class type '%s', starting character '%s'", sType, (char)c);
276 			}
277 		} else if (c == '\'') {
278 			String s = readSingleQuotedString(r);
279 			o = handleQuotedScalar(s, r, sType, eType, builder, outer, pMeta, contentColumn > 0 ? contentColumn - 1 : 0);
280 		} else if (c == '"') {
281 			String s = readDoubleQuotedString(r);
282 			o = handleQuotedScalar(s, r, sType, eType, builder, outer, pMeta, contentColumn > 0 ? contentColumn - 1 : 0);
283 		} else if (c == '-') {
284 			int c2 = peekSecondChar(r);
285 			if (c2 == ' ' || c2 == '\n' || c2 == '\r') {
286 				if (sType.isObject()) {
287 					o = readBlockSequence(r, newGenericList(), object(), pMeta, 0);
288 				} else if (sType.isCollection()) {
289 					Collection l = (sType.canCreateNewInstance(outer) ? (Collection)sType.newInstance() : newGenericList());
290 					o = readBlockSequence(r, l, sType, pMeta, 0);
291 				} else if (sType.isArray() || sType.isArgs()) {
292 					var l = (ArrayList)readBlockSequence(r, list(), sType, pMeta, 0);
293 					o = toArray(sType, l);
294 				} else {
295 					throw new ParseException(this, "Unrecognized syntax for class type '%s', starting character '%s'", sType, (char)c);
296 				}
297 			} else {
298 				String s = readPlainScalar(r, 0);
299 				o = convertToType(s, sType, eType, outer, pMeta);
300 			}
301 		} else if (c == '|' || c == '>') {
302 			String s = readBlockScalar(r, (char)c);
303 			o = convertToType(s, sType, eType, outer, pMeta);
304 		} else if (c == '~') {
305 			r.read(); // consume '~'
306 			// o remains null (initialized at line 255)
307 		} else {
308 			String s = readPlainScalar(r, 0);
309 			o = handlePlainScalar(s, r, sType, eType, builder, outer, pMeta, contentColumn > 0 ? contentColumn - 1 : 0);
310 		}
311 
312 		if (nn(swap) && nn(o))
313 			o = unswap(swap, o, eType);
314 
315 		if (nn(parentBean()))
316 			setParent(eType, o, parentBean());
317 
318 		return (T)o;
319 	}

326 	private <T> Object handleQuotedScalar(String s, ParserReader r, ClassMeta<?> sType, ClassMeta<?> eType, BuilderSwap<T,Object> builder, Object outer, BeanPropertyMeta pMeta, int keyIndent) throws IOException, ParseException, ExecutableException {
327 		if (looksLikeMappingKey(r)) {
328 			r.read(); // consume ':'
329 			int cp = r.peek();
330 			if (cp == ' ')
331 				r.read();
332 
333 			if (sType.isObject()) {
334 				var m2 = newGenericMap();
335 				Object value = readAnything(object(), r, m2, pMeta);
336 				String ts = trim(s);
337 				setName(object(), value, ts);
338 				m2.put(ts, value);
339 				readBlockMappingRemainder(r, m2, string(), object(), pMeta, keyIndent);
340 				return cast(m2, pMeta, eType);
341 			} else if (nn(builder)) {
342 				var m = toBeanMap(builder.create(this, eType));
343 				readBeanProperty(r, m, s);
344 				readIntoBeanMapBlockRemainder(r, m, keyIndent);
345 				return builder.build(this, m.getBean(), eType);
346 			} else if (sType.canCreateNewBean(outer)) {
347 				var m = newBeanMap(outer, sType.inner());
348 				readBeanProperty(r, m, s);
349 				readIntoBeanMapBlockRemainder(r, m, keyIndent);
350 				return m.getBean();
351 			} else if (sType.isMap()) {
352 				Map m = (sType.canCreateNewInstance(outer) ? (Map)sType.newInstance(outer) : newGenericMap(sType));
353 				Object value = readAnything(sType.getValueType(), r, m, pMeta);
354 				Object key = convertAttrToType(m, trim(s), sType.getKeyType());
355 				setName(sType.getValueType(), value, key);
356 				m.put(key, value);
357 				readBlockMappingRemainder(r, m, sType.getKeyType(), sType.getValueType(), pMeta, keyIndent);
358 				return m;
359 			} else {
360 				var m2 = newGenericMap();
361 				Object value = readAnything(object(), r, m2, pMeta);
362 				m2.put(s, value);
363 				readBlockMappingRemainder(r, m2, string(), object(), pMeta, keyIndent);
364 				if (m2.containsKey(getBeanTypePropertyName(eType)))
365 					return cast(m2, pMeta, eType);
366 				if (nn(sType.getProxyInvocationHandler()))
367 					return newBeanMap(outer, sType.inner()).load(m2).getBean();
368 				throw new ParseException(this, "Class '%s' could not be instantiated.  Reason: '%s'", cn(sType), sType.getNotABeanReason());
369 			}
370 		}
371 		if (sType.isObject())
372 			return trim(s);
373 		return convertToType(s, sType, eType, outer, pMeta);
374 	}

377 		"java:S107", // 8 parameters required for YAML scalar handling dispatch

381 	private <T> Object handlePlainScalar(String s, ParserReader r, ClassMeta<?> sType, ClassMeta<?> eType, BuilderSwap<T,Object> builder, Object outer, BeanPropertyMeta pMeta, int keyIndent) throws IOException, ParseException, ExecutableException {
382 		if (s.isEmpty())
383 			return null;
384 
385 		if (looksLikeMappingKey(r)) {
386 			r.read(); // consume ':'
387 			int cp = r.peek();
388 			if (cp == ' ')
389 				r.read(); // consume space after ':'
390 
391 			String keyStr = isYamlNull(s) ? null : trim(s);
392 			if (sType.isObject()) {
393 				var m2 = newGenericMap();
394 				Object value = readAnything(object(), r, m2, pMeta);
395 				setName(object(), value, keyStr);
396 				m2.put(keyStr, value);
397 				readBlockMappingRemainder(r, m2, string(), object(), pMeta, keyIndent);
398 				return cast(m2, pMeta, eType);
399 			} else if (nn(builder)) {
400 				var m = toBeanMap(builder.create(this, eType));
401 				readBeanProperty(r, m, s);
402 				readIntoBeanMapBlockRemainder(r, m, keyIndent);
403 				return builder.build(this, m.getBean(), eType);
404 			} else if (sType.canCreateNewBean(outer)) {
405 				var m = newBeanMap(outer, sType.inner());
406 				readBeanProperty(r, m, s);
407 				readIntoBeanMapBlockRemainder(r, m, keyIndent);
408 				return m.getBean();
409 			} else if (sType.isMap()) {
410 				Map m = (sType.canCreateNewInstance(outer) ? (Map)sType.newInstance(outer) : newGenericMap(sType));
411 				Object value = readAnything(sType.getValueType(), r, m, pMeta);
412 				Object key = convertAttrToType(m, keyStr, sType.getKeyType());
413 				setName(sType.getValueType(), value, key);
414 				m.put(key, value);
415 				readBlockMappingRemainder(r, m, sType.getKeyType(), sType.getValueType(), pMeta, keyIndent);
416 				return m;
417 			} else {
418 				var m2 = newGenericMap();
419 				Object value = readAnything(object(), r, m2, pMeta);
420 				m2.put(s, value);
421 				readBlockMappingRemainder(r, m2, string(), object(), pMeta, keyIndent);
422 				if (m2.containsKey(getBeanTypePropertyName(eType)))
423 					return cast(m2, pMeta, eType);
424 				if (nn(sType.getProxyInvocationHandler()))
425 					return newBeanMap(outer, sType.inner()).load(m2).getBean();
426 				throw new ParseException(this, "Class '%s' could not be instantiated.  Reason: '%s'", cn(sType), sType.getNotABeanReason());
427 			}
428 		}
429 
430 		if (isYamlNull(s))
431 			return null;
432 
433 		if ("true".equals(s) || "false".equals(s)) {
434 			if (sType.isObject() || sType.isBoolean())
435 				return Boolean.valueOf(s);
436 			return convertToType(s, sType, eType, outer, pMeta);
437 		}
438 
439 		return convertToType(s, sType, eType, outer, pMeta);
440 	}

442 	@SuppressWarnings({
443 		"unused",    // eType, pMeta kept for API consistency with callers
444 		"java:S1172" // Same as above
445 	})
446 	private Object convertToType(String s, ClassMeta<?> sType, ClassMeta<?> eType, Object outer, BeanPropertyMeta pMeta) throws ParseException {
447 		if (sType.isObject()) {
448 			return resolveScalarType(trim(s));
449 		} else if (sType.isCharSequence()) {
450 			return trim(s);
451 		} else if (sType.isChar()) {
452 			return parseCharacter(s);
453 		} else if (sType.isNumber()) {
454 			return StringUtils.parseNumber(s, (Class<? extends Number>)sType.inner());
455 		} else if (sType.isBoolean()) {
456 			return b(s);
457 		} else if (sType.isDate()) {
458 			return readDate(s, sType);
459 		} else if (sType.isCalendar()) {
460 			return readCalendar(s, sType);
461 		} else if (sType.isTemporal()) {
462 			return readTemporal(s, sType);
463 		} else if (sType.isDuration()) {
464 			return readDuration(s);
465 		} else if (sType.isPeriod()) {
466 			return readPeriod(s);
467 		} else if (sType.canCreateNewInstanceFromString(outer)) {
468 			return sType.newInstanceFromString(outer, s);
469 		} else {
470 			throw new ParseException(this, "Unrecognized syntax for class type '%s', value '%s'", sType, s);
471 		}
472 	}

490 	private <T> void readBeanProperty(ParserReader r, BeanMap<T> m, String currAttr) throws IOException, ParseException, ExecutableException {
491 		var pb = swapParentBean(m.getBean(false));
492 		try {
493 		var pm = m.getPropertyMeta(currAttr);
494 		setCurrentProperty(pm);
495 		if (pm == null) {
496 			onUnknownProperty(currAttr, m, readAnything(object(), r, m.getBean(false), null));
497 		} else {
498 			var cm = (ClassMeta<?>) pm.getBeanInfo();
499 			Object value = readAnything(cm, r, m.getBean(false), pm);
500 			setName(cm, value, currAttr);
501 			try {
502 				pm.set(m, currAttr, value);
503 			} catch (BeanRuntimeException e) {
504 				onBeanSetterException(pm, e);
505 				throw e;
506 			}
507 		}
508 		} finally {
509 			swapParentBean(pb);
510 		}
511 		setCurrentProperty(null);
512 	}

514 	private static boolean isNullBlockValue(ParserReader r, int blockIndent) throws IOException {
515 		int c = r.peek();
516 		if (c == -1)
517 			return true;
518 		if (c != '\n' && c != '\r')
519 			return false;
520 		int charsRead = 0;
521 		while (true) {
522 			c = r.read();
523 			if (c == -1) {
524 				for (int j = 0; j < charsRead; j++) r.unread();
525 				return true;
526 			}
527 			charsRead++;
528 			if (c != '\n' && c != '\r' && c != ' ' && c != '\t')
529 				break;
530 		}
531 		int nextColumn = r.getPosition().getColumn() - 1;
532 		for (int j = 0; j < charsRead; j++) r.unread();
533 		return nextColumn <= blockIndent;
534 	}

566 		String currKey = null;

569 			c = r.read();

575 				else

580 				} else if (isWhitespace(c)) {

592 			} else if (state == S4) {

594 					continue;

601 			} else if (state == S5) {

608 				} else {
609 					break;
610 				}
611 			} else if (state == S6) {

613 					break;

619 					state = S3;
620 				}
621 			}
622 		}
623 		if (state == S1)
624 			throw new ParseException(this, "Expected '{' at beginning of YAML flow mapping.");

627 		if (state == S3)
628 			throw new ParseException(this, "Could not find ':' following key in YAML flow mapping.");

633 		if (state == S6)
634 			throw new ParseException(this, "Unexpected '}' found in YAML flow mapping.");
635 
636 		return null; // Unreachable.
637 	}

744

868 		// S2: Looking for attrName start.
869 		// S3: Found attrName end, looking for :.
870 		// S4: Found :, looking for valStart.
871 		// S5: Looking for , or }
872 
873 		var state = S1;

899 				} else if (state == S3) {

902 					else if (isWhitespace(c))
903 						continue;
904 				} else if (state == S4) {
905 					if (isWhitespace(c))
906 						continue;

909 						setCurrentProperty(pm);
910 						if (pm == null) {
911 							onUnknownProperty(currAttr, m, readAnything(object(), r.unread(), m.getBean(false), null));
912 							unmark();
913 						} else {

917 							setName(cm, value, currAttr);

920 							} catch (BeanRuntimeException e) {
921 								onBeanSetterException(pm, e);
922 								throw e;
923 							}
924 						}
925 						setCurrentProperty(null);
926 					}
927 					state = S5;
928 				} else if (state == S5) {
929 					if (c == ',')
930 						state = S2;

933 					else if (c == '}') {
934 						return m;
935 					}
936 				}
937 			}
938 			if (state == S1)
939 				throw new ParseException(this, "Expected '{' at beginning of YAML flow mapping.");
940 			if (state == S2)
941 				throw new ParseException(this, "Could not find attribute name in YAML flow mapping.");

944 			if (state == S4)
945 				throw new ParseException(this, "Expected value in YAML flow mapping.");
946 			if (state == S5)
947 				throw new ParseException(this, "Could not find '}' marking end of YAML flow mapping.");
948 		} finally {
949 			unmark();
950 			swapParentBean(pb);
951 		}
952 
953 		return null; // Unreachable.
954 	}

1142 					case 'b': sb.append('\b'); break;
1143 					case 'f': sb.append('\f'); break;
1144 					case 'e': sb.append('\u001b'); break;
1145 					case '/': sb.append('/'); break;
1146 					case 'x': {
1147 						String hex = r.read(2);
1148 						try {
1149 							sb.append((char)Integer.parseInt(hex, 16));
1150 						} catch (@SuppressWarnings("unused") NumberFormatException e) {
1151 							throw new ParseException(this, "Invalid \\x escape sequence in YAML string.");
1152 						}
1153 						break;
1154 					}
1155 					case 'u': {
1156 						String hex = r.read(4);
1157 						try {
1158 							sb.append((char)Integer.parseInt(hex, 16));
1159 						} catch (@SuppressWarnings("unused") NumberFormatException e) {
1160 							throw new ParseException(this, "Invalid \\u escape sequence in YAML string.");
1161 						}
1162 						break;
1163 					}
1164 					default:
1165 						throw new ParseException(this, "Invalid escape sequence '\\%s' in YAML string.", (char)c);
1166 				}
1167 				// @formatter:on
1168 			} else if (c == '"') {
1169 				return sb.toString();
1170 			} else {
1171 				sb.append((char)c);
1172 			}
1173 		}
1174 		throw new ParseException(this, "Could not find end of double-quoted YAML string.");

1180 	@SuppressWarnings({
1181 		"unused",    // indent accepted for context but not currently used in parsing logic
1182 		"java:S1172", // Same as above
1183 		"java:S3776"  // Cognitive complexity acceptable for plain scalar parsing
1184 	})
1185 	private static String readPlainScalar(ParserReader r, int indent) throws IOException {
1186 		var sb = new StringBuilder();
1187 		int c;
1188 		while ((c = r.read()) != -1) {
1189 			if (c == '\n' || c == '\r') {
1190 				r.unread();
1191 				break;
1192 			}
1193 			if (c == ':') {
1194 				int next = r.peek();
1195 				if (next == ' ' || next == '\n' || next == '\r' || next == -1) {
1196 					r.unread(); // unread ':'
1197 					break;
1198 				}
1199 				sb.append((char)c);
1200 			} else if (c == ' ') {
1201 				int next = r.peek();
1202 				if (next == '#') {
1203 					// rest of line is comment
1204 					r.unread();
1205 					break;
1206 				}
1207 				sb.append((char)c);
1208 			} else if ((c == ',' || c == '}' || c == ']') || (c == '#' && sb.isEmpty())) {
1209 				r.unread();
1210 				break;
1211 			} else {
1212 				sb.append((char)c);
1213 			}
1214 		}
1215 		String result = sb.toString();
1216 		// Trim trailing whitespace
1217 		int end = result.length();
1218 		while (end > 0 && result.charAt(end - 1) == ' ')
1219 			end--;
1220 		return result.substring(0, end);
1221 	}

1225 	// ==========================================
1226 	@SuppressWarnings({
1227 		"java:S3776", // Cognitive complexity acceptable for block scalar parsing
1228 		"java:S6541" // Brain method acceptable for block scalar state machine
1229 	})
1230 	private String readBlockScalar(ParserReader r, char indicator) throws IOException, ParseException {
1231 		r.read(); // consume '|' or '>'
1232 
1233 		// Parse indicator line for chomping
1234 		char chomping = ' '; // default = clip
1235 		int c;
1236 		while ((c = r.read()) != -1) {
1237 			if (c == '-') {
1238 				chomping = '-'; // strip
1239 			} else if (c == '+') {
1240 				chomping = '+'; // keep
1241 			} else if (c == '\n' || c == '\r') {
1242 				if (c == '\r') {
1243 					int next = r.peek();
1244 					if (next == '\n')
1245 						r.read();
1246 				}
1247 				break;
1248 			} else if (c >= '1' && c <= '9') {
1249 				// explicit indent indicator, ignore for now (auto-detect below)
1250 			} else if (c == ' ' || c == '\t') {
1251 				continue;
1252 			} else {
1253 				throw new ParseException(this, "Unexpected character '%s' in YAML block scalar indicator.", (char)c);
1254 			}
1255 		}
1256 
1257 		// Read the block content
1258 		int blockIndent = -1;
1259 		var lines = new ArrayList<String>();
1260 		int trailingNewlines = 0;
1261 
1262 		while (true) {
1263 			// Count leading spaces
1264 			int spaces = 0;
1265 			while ((c = r.read()) != -1) {
1266 				if (c == ' ')
1267 					spaces++;
1268 				else
1269 					break;
1270 			}
1271 
1272 			if (c == -1) {
1273 				break;
1274 			}
1275 
1276 			if (c == '\n' || c == '\r') {
1277 				// Empty line
1278 				trailingNewlines++;
1279 				lines.add("");
1280 				if (c == '\r' && r.peek() == '\n')
1281 					r.read();
1282 				continue;
1283 			}
1284 
1285 			if (blockIndent == -1) {
1286 				blockIndent = spaces;
1287 			}
1288 
1289 			if (spaces < blockIndent) {
1290 				// End of block, push back everything
1291 				r.unread(); // push back non-space char
1292 				for (int i = 0; i < spaces; i++)
1293 					r.unread();
1294 				break;
1295 			}
1296 
1297 			// Read rest of line
1298 			var lineSb = new StringBuilder();
1299 			// Include extra spaces beyond blockIndent
1300 			for (int i = blockIndent; i < spaces; i++)
1301 				lineSb.append(' ');
1302 			lineSb.append((char)c);
1303 			while ((c = r.read()) != -1) {
1304 				if (c == '\n' || c == '\r') {
1305 					if (c == '\r' && r.peek() == '\n')
1306 						r.read();
1307 					break;
1308 				}
1309 				lineSb.append((char)c);
1310 			}
1311 
1312 			// Account for any blank lines that preceded this content line
1313 			for (int i = 0; i < trailingNewlines; i++)
1314 				lines.add("");
1315 			trailingNewlines = 0;
1316 			lines.add(lineSb.toString());
1317 
1318 			if (c == -1)
1319 				break;
1320 		}
1321 
1322 		// Build the result based on indicator and chomping
1323 		var sb = new StringBuilder();
1324 		for (int i = 0; i < lines.size(); i++) {
1325 			String line = lines.get(i);
1326 			if (indicator == '|') {
1327 				// Literal: preserve newlines
1328 				if (i > 0)
1329 					sb.append('\n');
1330 				sb.append(line);
1331 			} else {
1332 				// Folded: single newlines -> space, multiple newlines preserved
1333 				if (i > 0) {
1334 					if (line.isEmpty() || lines.get(i - 1).isEmpty()) {
1335 						sb.append('\n');
1336 					} else {
1337 						sb.append(' ');
1338 					}
1339 				}
1340 				sb.append(line);
1341 			}
1342 		}
1343 
1344 		// Apply chomping
1345 		if (chomping == '-') {
1346 			// strip: no trailing newline
1347 		} else if (chomping == '+') {
1348 			// keep: preserve all trailing newlines
1349 			sb.append('\n');
1350 			for (int i = 0; i < trailingNewlines; i++)
1351 				sb.append('\n');
1352 		} else {
1353 			// clip (default): single trailing newline
1354 			sb.append('\n');
1355 		}
1356 
1357 		return sb.toString();
1358 	}

1445 		if (s.isEmpty())
1446 			return null;
