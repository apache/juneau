17  package org.apache.juneau.marshall.jsonl;
18  
19  import java.io.*;
20  
21  import org.apache.juneau.marshall.json.*;
22  import org.apache.juneau.marshall.parser.*;
23  import org.apache.juneau.marshall.stream.*;
24  
25  /**
26   * Reference implementation of the public {@link TokenReader} surface for the JSONL / NDJSON
27   * format.
28   *
29   * <p>
30   * JSONL is a sequence of independent JSON values separated by newlines.  There is no root
31   * container.  This cursor models that as a flat top-level sequence at {@link #getDepth() depth} 0:
32   * each line's tokens are emitted in turn, and {@link #next()} returns {@link TokenType#END_OF_STREAM}
33   * only when the underlying input is exhausted.
34   *
35   * <p>
36   * The most ergonomic way to consume a JSONL stream is via {@link #read(Class)}, which binds one
37   * line to a POJO per call:
38   * <p class='bjava'>
39   * 	<jk>try</jk> (TokenReader <jv>r</jv> = Jsonl.<jsf>DEFAULT</jsf>.readTokens(<jv>reader</jv>)) {
40   * 		<jk>while</jk> (<jv>r</jv>.canRead()) {
41   * 			MyRecord <jv>rec</jv> = <jv>r</jv>.read(MyRecord.<jk>class</jk>);
42   * 			process(<jv>rec</jv>);
43   * 		}
44   * 	}
45   * </p>
46   *
47   * <h5 class='section'>Notes:</h5><ul>
48   * 	<li class='warn'>This class is not thread safe.
49   * 	<li>The cursor is a true O(1)-memory streaming cursor ({@link #isStreaming()} == <jk>true</jk>).
50   * </ul>
51   */
52  @SuppressWarnings({
53  	"resource" // The cursor's underlying ParserPipe is owned by the caller via try-with-resources on the cursor itself; Eclipse JDT flags the inner pipe as unclosed but that's by design.
54  })
55  public class JsonlTokenReader extends JsonTokenReader {
56  
57  	/**
58  	 * Constructor with default settings.
59  	 *
60  	 * @param pipe The parser input pipe to read from.  Must not be <jk>null</jk>.
61  	 * @throws IOException If a problem occurred opening the underlying reader.
62  	 */
63  	public JsonlTokenReader(ParserPipe pipe) throws IOException {
64  		super(pipe);
65  	}
66  
67  	/**
68  	 * Constructor used by {@link JsonlParserSession#readTokens(Object)} to plumb the calling
69  	 * session through.
70  	 *
71  	 * @param pipe The parser input pipe.  Must not be <jk>null</jk>.
72  	 * @param settings The cursor-level settings.  Must not be <jk>null</jk>.
73  	 * @param session The {@link JsonlParserSession} for {@link #read(Class)} delegation, or
74  	 * 	<jk>null</jk> to disable {@code read}.
75  	 * @throws IOException If a problem occurred opening the underlying reader.
76  	 */
77  	public JsonlTokenReader(ParserPipe pipe, Settings settings, JsonlParserSession session) throws IOException {
78  		super(pipe, settings, session);
79  	}
80  
81  	@Override /* JsonTokenReader */
82  	protected void afterValue() {
83  		// At depth 0, a top-level value just completed and we want the next next() / read()
84  		// to consume the next line's value (instead of transitioning to S05_end as plain JSON
85  		// would).  Inside a container the parent's logic (S03_expectCommaOrEndObject /
86  		// S04_expectCommaOrEndArray) is still correct.
87  		if (depth == 0)
88  			state = S00_expectValue;
89  		else
90  			super.afterValue();
91  	}
92  }
