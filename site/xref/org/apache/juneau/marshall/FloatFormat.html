23  /**
24   * Supported wire formats for non-finite {@link Float} / {@link Double} values
25   * ({@link Double#NaN NaN}, {@link Double#POSITIVE_INFINITY +Infinity}, {@link Double#NEGATIVE_INFINITY -Infinity}).
26   *
27   * <p>
28   * Used by {@link MarshallingContext.Builder#floatFormat(FloatFormat)},
29   * {@link Marshalled#floatFormat()},
30   * {@link MarshalledProp#floatFormat()}, and
31   * {@link MarshalledConfig#floatFormat()} to control how non-finite
32   * floating-point values are written to text-based wire formats.  Finite values are unaffected and ride
33   * the natural bare-numeric-token wire form for the underlying serializer.
34   *
35   * <p>
36   * The default is {@link #NaN_AS_NULL} which is JSON-spec-compliant per RFC 8259 §6 (non-finite floats
37   * are not valid JSON tokens; emitting {@code null} is the closest valid representation).
38   *
39   * <h5 class='topic'>Precedence (highest to lowest)</h5>
40   * <ol>
41   * 	<li>{@link MarshalledProp#floatFormat() @MarshalledProp(floatFormat=…)} on the bean property.
42   * 	<li>{@link Marshalled#floatFormat() @Marshalled(floatFormat=…)} on the bean class.
43   * 	<li>{@link MarshalledConfig#floatFormat() @MarshalledConfig(floatFormat=…)} on
44   * 		<code><ja>@Rest</ja></code>-annotated classes / methods.
45   * 	<li>Programmatic {@link MarshallingContext.Builder#floatFormat(FloatFormat)}.
46   * 	<li>Environment variable <c>MarshallingContext.floatFormat</c>.
47   * 	<li>The default constant ({@link #NaN_AS_NULL}).
48   * </ol>
49   *
50   * <h5 class='topic'>Parser leniency</h5>
51   *
52   * <p>
53   * Parsers SHALL accept all of the following wire shapes regardless of the parser-side
54   * {@code FloatFormat} setting:
55   * <ul>
56   * 	<li>Bare numeric tokens including the non-standard {@code NaN} / {@code Infinity} / {@code -Infinity} tokens
57   * 		(Juneau's lenient parser accepts these by default).
58   * 	<li>Quoted-string forms — {@code "NaN"}, {@code "Infinity"}, {@code "-Infinity"} (case-insensitive).
59   * 	<li>{@code null} — reconstructed as {@link Double#NaN} when reading into a primitive {@code double} / {@code float};
60   * 		preserved as {@code null} when reading into a boxed {@link Double} / {@link Float}.
61   * </ul>
62   *
63   * <h5 class='topic'>JSON-spec compliance</h5>
64   *
65   * <p>
66   * {@link #NaN_AS_NUMBER} is <b>non-standard JSON</b> per RFC 8259 §6 — JSON does not permit {@code NaN} /
67   * {@code Infinity} / {@code -Infinity} tokens. Juneau's own JSON parser accepts these in lenient mode,
68   * but strict-mode consumers (e.g. some browser {@code JSON.parse} pipelines) will reject them. Use only
69   * when you control both ends of the wire.
70   *
71   * <h5 class='topic'>Binary serializers</h5>
72   *
73   * <p>
74   * Binary serializers (BSON / CBOR / MsgPack / Prototext / Parquet) emit native IEEE-754 floats regardless
75   * of this setting; native non-finite representations exist in every supported binary format (BSON
76   * {@code double} / {@code decimal128}; CBOR's tag 22 / 23 / 24 for special floats; MsgPack {@code float64}).
77   * This setting only affects text-based serializers.
78   */
79  @SuppressWarnings({
80  	"java:S115" // NaN_AS_* enum constants use the IEEE 754 abbreviation 'NaN' which includes lowercase letters; intentional
81  })
82  public enum FloatFormat {
83  
84  	/** Sentinel meaning "no value configured" — falls through to the next-higher precedence level. */
85  	NOT_SET,
86  
87  	/**
88  	 * Emit {@code null} for {@code NaN} / {@code ±Infinity} (the default).
89  	 *
90  	 * <p>
91  	 * <b>JSON-spec compliant.</b> Loses the ability to distinguish {@code NaN} from
92  	 * {@code +Infinity} from {@code -Infinity} on the wire — all three become {@code null}. Round-trips
93  	 * back to {@link Double#NaN} when reading into a primitive {@code double}; into a boxed {@link Double}
94  	 * field, {@code null} stays {@code null}.
95  	 */
96  	NaN_AS_NULL,
97  
98  	/**
99  	 * Emit the quoted string {@code "NaN"} / {@code "Infinity"} / {@code "-Infinity"}.
100 	 *
101 	 * <p>
102 	 * <b>Round-trip preserving.</b> The parser unambiguously decodes the string back to the original
103 	 * non-finite value.
104 	 */
105 	NaN_AS_STRING,
106 
107 	/**
108 	 * Emit the bare {@code NaN} / {@code Infinity} / {@code -Infinity} token.
109 	 *
110 	 * <p>
111 	 * <b>Non-standard JSON</b> per RFC 8259 §6. Accepted by Juneau's lenient parser and V8 / several
112 	 * Python libs; rejected by strict consumers. Use only when you control both ends of the wire.
113 	 */
114 	NaN_AS_NUMBER,
115 
116 	/**
117 	 * Throw {@link IllegalArgumentException} when a non-finite value is encountered.
118 	 *
119 	 * <p>
120 	 * <b>Fail-fast</b> for data-cleanliness pipelines where a {@code NaN} indicates upstream corruption.
121 	 */
122 	NaN_AS_ERROR;
123 
124 	/**
125 	 * Formats the specified {@code double} using this format.
126 	 *
127 	 * <p>
128 	 * Finite values always return the boxed {@link Double} so text serializers emit the bare numeric
129 	 * token. Non-finite values are dispatched per the constant:
130 	 * <ul>
131 	 * 	<li>{@link #NaN_AS_NULL} → {@code null}.
132 	 * 	<li>{@link #NaN_AS_STRING} → {@code "NaN"} / {@code "Infinity"} / {@code "-Infinity"}.
133 	 * 	<li>{@link #NaN_AS_NUMBER} → boxed {@link Double} (non-finite); text serializers emit the bare token.
134 	 * 	<li>{@link #NaN_AS_ERROR} → throws {@link IllegalArgumentException}.
135 	 * </ul>
136 	 *
137 	 * <p>
138 	 * {@link #NOT_SET} / {@code null} format are treated as {@link #NaN_AS_NULL}.
139 	 *
140 	 * @param value The value to format.
141 	 * @param format The configured format. Can be <jk>null</jk> (treated as {@link #NaN_AS_NULL}).
142 	 * @return The formatted value — a {@link Double} for finite or {@code NaN_AS_NUMBER}, a {@link String}
143 	 *   for {@code NaN_AS_STRING}, or {@code null} for non-finite under {@code NaN_AS_NULL}.
144 	 * @throws IllegalArgumentException If {@code format} is {@link #NaN_AS_ERROR} and the value is non-finite.
145 	 */
146 	public static Object format(double value, FloatFormat format) {
147 		if (Double.isFinite(value))
148 			return Double.valueOf(value);
149 		return formatNonFinite(value, format);
150 	}
151 
152 	/**
153 	 * Formats the specified {@code float} using this format.
154 	 *
155 	 * <p>
156 	 * Finite values return the boxed {@link Float}. Non-finite values follow the same rules as
157 	 * {@link #format(double, FloatFormat)} except that {@link #NaN_AS_NUMBER} returns the boxed
158 	 * {@link Float} (not {@link Double}).
159 	 *
160 	 * @param value The value to format.
161 	 * @param format The configured format. Can be <jk>null</jk> (treated as {@link #NaN_AS_NULL}).
162 	 * @return The formatted value — a {@link Float} for finite or {@code NaN_AS_NUMBER}, a {@link String}
163 	 *   for {@code NaN_AS_STRING}, or {@code null} for non-finite under {@code NaN_AS_NULL}.
164 	 * @throws IllegalArgumentException If {@code format} is {@link #NaN_AS_ERROR} and the value is non-finite.
165 	 */
166 	public static Object format(float value, FloatFormat format) {
167 		if (Float.isFinite(value))
168 			return Float.valueOf(value);
169 		var fmt = format == null ? NaN_AS_NULL : format;
170 		return switch (fmt) {
171 			case NOT_SET, NaN_AS_NULL -> null;
172 			case NaN_AS_STRING -> nonFiniteTokenFloat(value);
173 			case NaN_AS_NUMBER -> Float.valueOf(value);
174 			case NaN_AS_ERROR -> throw iaex("Non-finite float value %s rejected by FloatFormat.NaN_AS_ERROR", value);
175 		};
176 	}
177 
178 	private static Object formatNonFinite(double value, FloatFormat format) {
179 		var fmt = format == null ? NaN_AS_NULL : format;
180 		return switch (fmt) {
181 			case NOT_SET, NaN_AS_NULL -> null;
182 			case NaN_AS_STRING -> nonFiniteToken(value);
183 			case NaN_AS_NUMBER -> Double.valueOf(value);
184 			case NaN_AS_ERROR -> throw iaex("Non-finite double value %s rejected by FloatFormat.NaN_AS_ERROR", value);
185 		};
186 	}
187 
188 	private static String nonFiniteToken(double value) {
189 		if (Double.isNaN(value))
190 			return "NaN";
191 		return value > 0 ? "Infinity" : "-Infinity";
192 	}
193 
194 	private static String nonFiniteTokenFloat(float value) {
195 		if (Float.isNaN(value))
196 			return "NaN";
197 		return value > 0 ? "Infinity" : "-Infinity";
198 	}
199 
200 	/**
201 	 * Parses the specified wire value into a non-finite-aware {@link Double} or {@link Float}.
202 	 *
203 	 * <p>
204 	 * Lenient parsing — accepts bare numeric, quoted-string {@code "NaN"} / {@code "Infinity"} /
205 	 * {@code "-Infinity"} (case-insensitive), and unquoted {@code NaN} / {@code Infinity} / {@code -Infinity}
206 	 * tokens regardless of the {@code format} hint. The hint is informational only.
207 	 *
208 	 * @param <T> The target type ({@link Double} or {@link Float}).
209 	 * @param value The wire value. Can be <jk>null</jk> or blank (returns <jk>null</jk>).
210 	 * @param format The configured format hint (informational only — parsing is format-agnostic). Can be <jk>null</jk> (ignored; parsing is format-agnostic).
211 	 * @param targetType The desired return type — {@link Double} or {@link Float}.
212 	 * @return The parsed value, or <jk>null</jk> if {@code value} is <jk>null</jk> or blank.
213 	 * @throws IllegalArgumentException If the value cannot be parsed or {@code targetType} is unsupported.
214 	 */
215 	@SuppressWarnings({
216 		"unchecked" // Cast is safe: parameterized by caller.
217 	})
218 	public static <T extends Number> T parse(String value, FloatFormat format, Class<T> targetType) {
219 		if (value == null)
220 			return null;
221 		var s = value.trim();
222 		if (s.isEmpty())
223 			return null;
224 		// Strip surrounding quotes if present (lenient parsing — quoted-string and bare-numeric both accepted).
225 		if (s.length() >= 2 && (s.charAt(0) == '"' || s.charAt(0) == '\'') && s.charAt(s.length() - 1) == s.charAt(0))
226 			s = s.substring(1, s.length() - 1).trim();
227 		try {
228 			if (Double.class.equals(targetType) || double.class.equals(targetType))
229 				return (T) Double.valueOf(parseDouble(s));
230 			if (Float.class.equals(targetType) || float.class.equals(targetType))
231 				return (T) Float.valueOf(parseFloat(s));
232 			throw iaex("Unsupported FloatFormat target type: %s", targetType.getName());
233 		} catch (NumberFormatException e) {
234 			throw iaex("Invalid float value '%s' for format %s: %s", value, format, e.getMessage());
235 		}
236 	}
237 
238 	private static double parseDouble(String s) {
239 		var lower = s.toLowerCase(Locale.ROOT);
240 		return switch (lower) {
241 			case "nan" -> Double.NaN;
242 			case "infinity", "+infinity", "inf", "+inf" -> Double.POSITIVE_INFINITY;
243 			case "-infinity", "-inf" -> Double.NEGATIVE_INFINITY;
244 			default -> Double.parseDouble(s);
245 		};
246 	}
247 
248 	private static float parseFloat(String s) {
249 		var lower = s.toLowerCase(Locale.ROOT);
250 		return switch (lower) {
251 			case "nan" -> Float.NaN;
252 			case "infinity", "+infinity", "inf", "+inf" -> Float.POSITIVE_INFINITY;
253 			case "-infinity", "-inf" -> Float.NEGATIVE_INFINITY;
254 			default -> Float.parseFloat(s);
255 		};
256 	}
257 
258 	/**
259 	 * Returns <jk>true</jk> if this format always emits a bare numeric wire token for non-finite values.
260 	 *
261 	 * <p>
262 	 * Only {@link #NaN_AS_NUMBER} returns <jk>true</jk>. {@link #NaN_AS_NULL} may emit {@code null};
263 	 * {@link #NaN_AS_STRING} emits a quoted string; {@link #NaN_AS_ERROR} throws. Finite values are
264 	 * always emitted as bare numeric tokens regardless of this setting.
265 	 *
266 	 * @return <jk>true</jk> if this format emits a bare numeric wire token for non-finite values.
267 	 */
268 	public boolean isNumeric() {
269 		return this == NaN_AS_NUMBER;
270 	}
271 }
