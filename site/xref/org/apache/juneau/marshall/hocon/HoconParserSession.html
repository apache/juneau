92  	@Override
93  	protected <T> T doRead(ParserPipe pipe, ClassMeta<T> type) throws IOException, ParseException, ExecutableException {
94  		String inputStr;
95  		try {
96  			inputStr = pipe.asString();
97  		} catch (IOException e) {
98  			throw new ParseException(e);
99  		}
100 		if (inputStr == null)
101 			return null;
102 		var trimmed = inputStr.trim();
103 		if (trimmed.isEmpty())
104 			return null;
105 		var tokenizer = new HoconTokenizer(new StringReader(inputStr));
106 		var root = readRoot(tokenizer);
107 		if (root == null)
108 			return null;
109 		if (hoconParser.resolveSubstitutions) {
110 			var resolver = new HoconResolver(root);
111 			resolver.resolve();
112 		}
113 		var map = hoconToMap(root, type);
114 		return (T) convertToBean(map, type);
115 	}

117 	private HoconValue.HoconObject readRoot(HoconTokenizer t) throws IOException, ParseException {
118 		t.skipWhitespaceAndComments();
119 		var peek = t.peek();
120 		if (peek.type() == HoconTokenizer.TokenType.EOF)
121 			return null;
122 		if (peek.type() == HoconTokenizer.TokenType.LBRACE) {
123 			t.read();
124 			return readObject(t, null, new String[0]);
125 		}
126 		return readRootBraceless(t);
127 	}

129 	private HoconValue.HoconObject readRootBraceless(HoconTokenizer t) throws IOException, ParseException {
130 		var root = new HoconValue.HoconObject();
131 		while (t.peek().type() != HoconTokenizer.TokenType.EOF) {
132 			var path = readPath(t);
133 			if (path == null || path.length == 0) // HTT: readPath() never returns null (only EMPTY_PATH or a non-null toArray() result), so `path == null` is provably dead here
134 				break;
135 			t.skipWhitespaceAndComments();
136 			var next = t.peek();
137 			if (next.type() == HoconTokenizer.TokenType.LBRACE) {
138 				t.read();
139 				var nested = readObject(t, root, path);
140 				var existing = root.getPath(path);
141 				if (existing instanceof HoconValue.HoconObject existingObj)
142 					existingObj.merge(nested);
143 				else
144 					root.setPath(path, nested);
145 			} else if (next.type() == HoconTokenizer.TokenType.PLUS_EQUALS) {
146 				t.read();
147 				t.skipWhitespaceAndComments();
148 				var value = readValueOrConcat(t);
149 				var existing = root.getPath(path);
150 				if (existing instanceof HoconValue.HoconArray arr)
151 					arr.add(value);
152 				else {
153 					var newArr = new HoconValue.HoconArray();
154 					if (existing != null)
155 						newArr.add(existing);
156 					newArr.add(value);
157 					root.setPath(path, newArr);
158 				}
159 			} else if (next.type() == HoconTokenizer.TokenType.EQUALS || next.type() == HoconTokenizer.TokenType.COLON) {
160 				t.read();
161 				t.skipWhitespaceAndComments();
162 				var value = readValueOrConcat(t);
163 				value = resolveSelfRefConcatIfNeeded(value, path, root);
164 				root.setPath(path, value);
165 			} else {
166 				throw new ParseException(this, "Expected =, : or brace at line %s", t.getLine());
167 			}
168 			t.skipWhitespaceAndComments();
169 			if (t.peek().type() == HoconTokenizer.TokenType.NEWLINE)
170 				t.read();
171 		}
172 		return root;
173 	}

175 	private String[] readPath(HoconTokenizer t) throws IOException, ParseException {
176 		var tok = t.read();
177 		String first = switch (tok.type()) { // HTT: the EOF case is unreachable - readPath()'s only call sites (readRootBraceless, readObject) both peek and reject EOF in their while-loop condition before invoking readPath(), so `tok` here is never EOF.
178 			case EOF -> null; // HTT: unreachable -- see the HTT note on the switch line above.
179 			case UNQUOTED_STRING -> isTrimStrings() ? tok.stringValue().trim() : tok.stringValue();
180 			case QUOTED_STRING, TRIPLE_QUOTED -> isTrimStrings() ? tok.stringValue().trim() : tok.stringValue();
181 			case NULL -> null;
182 			case TRUE -> "true";
183 			case FALSE -> "false";
184 			case NUMBER -> tok.numberValue().toString();
185 			default -> throw new ParseException(this, "Expected key at line %s", t.getLine());
186 		};
187 		if (first == null)
188 			return EMPTY_PATH;
189 		var path = new ArrayList<String>();
190 		if (tok.type() == HoconTokenizer.TokenType.UNQUOTED_STRING && first.contains(".") && !first.startsWith("."))
191 			path.addAll(Arrays.asList(first.split("\\.", -1)));
192 		else {
193 			var comp = first.startsWith(".") ? first.substring(1) : first;
194 			if (!comp.isEmpty())
195 				path.add(comp);
196 			t.skipWhitespaceAndComments();
197 			while (t.peek().type() == HoconTokenizer.TokenType.UNQUOTED_STRING
198 					|| t.peek().type() == HoconTokenizer.TokenType.QUOTED_STRING
199 					|| t.peek().type() == HoconTokenizer.TokenType.TRIPLE_QUOTED) {
200 				var next = t.read().stringValue();
201 				var part = next.startsWith(".") ? next.substring(1) : next;
202 				if (!part.isEmpty())
203 					path.add(part);
204 				t.skipWhitespaceAndComments();
205 			}
206 		}
207 		return path.toArray(String[]::new);
208 	}

210 	/**
211 	 * Parses a value, supporting HOCON concatenation: adjacent values on the same line
212 	 * are concatenated (strings) or merged (objects/arrays).
213 	 */
214 	private HoconValue readValueOrConcat(HoconTokenizer t) throws IOException, ParseException {
215 		var values = new ArrayList<HoconValue>();
216 		values.add(readValue(t));
217 		while (true) {
218 			t.skipWhitespaceAndCommentsExceptNewlines();
219 			var peek = t.peekNoSkip();
220 			if (peek.type() == HoconTokenizer.TokenType.NEWLINE
221 					|| peek.type() == HoconTokenizer.TokenType.COMMA
222 					|| peek.type() == HoconTokenizer.TokenType.RBRACE
223 					|| peek.type() == HoconTokenizer.TokenType.RBRACKET
224 					|| peek.type() == HoconTokenizer.TokenType.EQUALS
225 					|| peek.type() == HoconTokenizer.TokenType.PLUS_EQUALS
226 					|| peek.type() == HoconTokenizer.TokenType.EOF)
227 				break;
228 			// HOCON: newline ends concatenation; unquoted string after array/object is next key, not concat
229 			var last = values.isEmpty() ? null : values.get(values.size() - 1); // HTT: `values` always has >= 1 element (seeded unconditionally at line 216 before this loop runs), so values.isEmpty() is provably always false
230 			if (peek.type() == HoconTokenizer.TokenType.UNQUOTED_STRING
231 					&& (last instanceof HoconValue.HoconArray || last instanceof HoconValue.HoconObject))
232 				break;
233 			values.add(readValue(t));
234 		}
235 		if (values.size() == 1)
236 			return values.get(0);
237 		return combineConcat(values);
238 	}

240 	private static HoconValue combineConcat(List<HoconValue> values) {
241 		if (values.size() == 1) // HTT: combineConcat()'s sole call site (readValueOrConcat) already returns early for values.size() == 1 before calling this method, so this is provably dead
242 			return values.get(0); // HTT: unreachable -- see the HTT note on the guard immediately above.
243 		var allArrays = values.stream().allMatch(v -> v instanceof HoconValue.HoconArray);
244 		var allObjects = values.stream().allMatch(v -> v instanceof HoconValue.HoconObject);
245 		if (allArrays) {
246 			var first = (HoconValue.HoconArray) values.get(0);
247 			for (var i = 1; i < values.size(); i++)
248 				first.concat((HoconValue.HoconArray) values.get(i));
249 			return first;
250 		}
251 		if (allObjects) {
252 			var first = (HoconValue.HoconObject) values.get(0);
253 			for (var i = 1; i < values.size(); i++)
254 				first.merge((HoconValue.HoconObject) values.get(i));
255 			return first;
256 		}
257 		var concat = new HoconValue.HoconConcat();
258 		for (var v : values)
259 			concat.add(v);
260 		return concat;
261 	}

263 	private static String[] concatenatePaths(String[] prefix, String[] path) {
264 		var result = new String[prefix.length + path.length];
265 		System.arraycopy(prefix, 0, result, 0, prefix.length);
266 		System.arraycopy(path, 0, result, prefix.length, path.length);
267 		return result;
268 	}

270 	private static HoconValue resolveSelfRefConcatIfNeeded(HoconValue value, String[] fullPath,
271 			HoconValue.HoconObject root) throws ParseException {
272 		if (!(value instanceof HoconValue.HoconConcat concat))
273 			return value;
274 		var pathStr = String.join(".", fullPath);
275 		if (!concat.referencesPath(pathStr))
276 			return value;
277 		var existing = root.getPath(fullPath);
278 		if (existing == null)
279 			return value;
280 		var resolver = new HoconResolver(root);
281 		java.util.function.Function<String,HoconValue> lookup = p -> p.equals(pathStr) ? existing : resolver.lookup(p);
282 		return resolver.resolveConcatWithLookup(concat, lookup);
283 	}

285 	private HoconValue readValue(HoconTokenizer t) throws IOException, ParseException {
286 		var tok = t.peek();
287 		return switch (tok.type()) {
288 			case UNQUOTED_STRING, QUOTED_STRING, TRIPLE_QUOTED -> {
289 				t.read();
290 				var s = isTrimStrings() ? tok.stringValue().trim() : tok.stringValue();
291 				yield new HoconValue.HoconString(s);
292 			}
293 			case NUMBER -> {
294 				t.read();
295 				yield new HoconValue.HoconNumber(tok.numberValue());
296 			}
297 			case TRUE -> {
298 				t.read();
299 				yield new HoconValue.HoconBoolean(true);
300 			}
301 			case FALSE -> {
302 				t.read();
303 				yield new HoconValue.HoconBoolean(false);
304 			}
305 			case NULL -> {
306 				t.read();
307 				yield new HoconValue.HoconNull();
308 			}
309 			case LBRACE -> {
310 				t.read();
311 				yield readObject(t, null, new String[0]);
312 			}
313 			case LBRACKET -> {
314 				t.read();
315 				yield readArray(t);
316 			}
317 			case SUBSTITUTION, OPT_SUBSTITUTION -> {
318 				var st = t.read();
319 				yield new HoconValue.HoconSubstitution(st.stringValue(), tok.type() == HoconTokenizer.TokenType.OPT_SUBSTITUTION);
320 			}
321 			case EOF -> null;
322 			default -> throw new ParseException(this, "Unexpected token %s at line %s", tok.type(), t.getLine());
323 		};
324 	}

326 	private HoconValue.HoconObject readObject(HoconTokenizer t, HoconValue.HoconObject root, String[] pathPrefix)
327 			throws IOException, ParseException {
328 		var obj = new HoconValue.HoconObject();
329 		var effectiveRoot = or(root, obj);
330 		t.skipWhitespaceAndComments();
331 		while (t.peek().type() != HoconTokenizer.TokenType.RBRACE && t.peek().type() != HoconTokenizer.TokenType.EOF) {
332 			var path = readPath(t);
333 			if (path == null || path.length == 0) // HTT: readPath() never returns null (only EMPTY_PATH or a non-null toArray() result), so `path == null` is provably dead here
334 				break;
335 			t.skipWhitespaceAndComments();
336 			var next = t.peek();
337 			if (next.type() == HoconTokenizer.TokenType.LBRACE) {
338 				t.read();
339 				var nested = readObject(t, effectiveRoot, concatenatePaths(pathPrefix, path));
340 				var existing = obj.getPath(path);
341 				if (existing instanceof HoconValue.HoconObject existingObj)
342 					existingObj.merge(nested);
343 				else
344 					obj.setPath(path, nested);
345 			} else if (next.type() == HoconTokenizer.TokenType.PLUS_EQUALS) {
346 				t.read();
347 				t.skipWhitespaceAndComments();
348 				var value = readValueOrConcat(t);
349 				var existing = obj.getPath(path);
350 				if (existing instanceof HoconValue.HoconArray arr)
351 					arr.add(value);
352 				else {
353 					var newArr = new HoconValue.HoconArray();
354 					if (existing != null)
355 						newArr.add(existing);
356 					newArr.add(value);
357 					obj.setPath(path, newArr);
358 				}
359 			} else if (next.type() == HoconTokenizer.TokenType.EQUALS || next.type() == HoconTokenizer.TokenType.COLON) {
360 				t.read();
361 				t.skipWhitespaceAndComments();
362 				var value = readValueOrConcat(t);
363 				var fullPath = concatenatePaths(pathPrefix, path);
364 				value = resolveSelfRefConcatIfNeeded(value, fullPath, effectiveRoot);
365 				obj.setPath(path, value);
366 			} else {
367 				throw new ParseException(this, "Expected =, : or brace at line %s", t.getLine());
368 			}
369 			// skipWhitespaceAndComments is a no-op while a token is cached (see HoconTokenizer), so a
370 			// closing `}` already peeked by readArrayOrConcat's concat loop is preserved here and the
371 			// loop terminates naturally via the while-condition above rather than chewing through a
372 			// newline that belongs to an enclosing scope.
373 			t.skipWhitespaceAndComments();
374 			if (t.peek().type() == HoconTokenizer.TokenType.COMMA || t.peek().type() == HoconTokenizer.TokenType.NEWLINE)
375 				t.read();
376 			t.skipWhitespaceAndComments();
377 		}
378 		if (t.peek().type() == HoconTokenizer.TokenType.RBRACE)
379 			t.read();
380 		return obj;
381 	}

383 	private HoconValue.HoconArray readArray(HoconTokenizer t) throws IOException, ParseException {
384 		var arr = new HoconValue.HoconArray();
385 		t.skipWhitespaceAndComments();
386 		while (t.peek().type() != HoconTokenizer.TokenType.RBRACKET && t.peek().type() != HoconTokenizer.TokenType.EOF) {
387 			// Elements inside [ ... ] are separated by COMMA or NEWLINE (handled below), so each
388 			// readArrayOrConcat call returns a single element.  HOCON array-concatenation
389 			// (`[a,b] [c,d]` ≡ `[a,b,c,d]`) is performed inside readArrayOrConcat when adjacent
390 			// arrays appear without a separator; the result is already a flattened HoconArray at
391 			// that point.  Here we must add the element as-is so nested arrays like
392 			// `[[1,2,3], [4,5,6]]` (with separators) stay nested rather than flattening.
393 			arr.add(readValueOrConcat(t));
394 			// readArrayOrConcat's internal concat loop calls peekNoSkip(), which eagerly consumes the
395 			// closing-bracket char from the underlying reader and stashes it as peeked=RBRACKET.
396 			// skipWhitespaceAndComments is a no-op while a token is cached (see HoconTokenizer), so it
397 			// will NOT read past the cached `]` and eat a following newline — a newline that is either
398 			// the in-array separator for the NEXT element of an enclosing array or a meaningful boundary
399 			// for an outer scope.  This prevents adjacent newline-separated inner arrays like
400 			// `[[1,2]\n[3,4]]` from being silently re-merged via HOCON array-concatenation above us.
401 			// The loop terminates on the cached `]` via the while-condition above.
402 			t.skipWhitespaceAndComments();
403 			if (t.peek().type() == HoconTokenizer.TokenType.COMMA || t.peek().type() == HoconTokenizer.TokenType.NEWLINE)
404 				t.read();
405 			t.skipWhitespaceAndComments();
406 		}
407 		if (t.peek().type() == HoconTokenizer.TokenType.RBRACKET)
408 			t.read();
409 		return arr;
410 	}

412 	private Object hoconToMap(HoconValue val, ClassMeta<?> type) throws ParseException {
413 		if (val == null) // HTT: dead -- the top-level call site (doRead) only reaches here after readRoot's own null-check, and both recursive call sites (OBJECT/ARRAY branches below) iterate over HoconObject/HoconArray members that are themselves always non-null HoconValue instances (a HOCON "null" literal is represented by a HoconNull instance, not a Java null), so val is never actually null here
414 			return null; // HTT: dead -- see comment above
415 		// STRING values targeting byte[] consult the configured BinaryFormat's variant BinarySwap
416 		// before falling back to the default String → byte[] UTF-8 coercion that the bean binder
417 		// would otherwise apply at the collection-element / top-level dispatch site.  The
418 		// bean-property path is unaffected because the per-property MPP install hides the type-level
419 		// default swap behind the per-property swap on the BeanPropertyMeta.
420 		//
421 		// At BinaryFormat.NOT_SET BinarySwap.match returns 0 and no swap fires, but
422 		// HoconSerializerSession emits byte[] at NOT_SET as a base64 string (to sidestep HOCON's
423 		// array-concatenation flattening for nested int-array wire forms).  Decode that here as the
424 		// symmetric fallback so List<byte[]> and top-level byte[] round-trip at NOT_SET.
425 		if (val.getType() == HoconValue.Type.STRING && type != null && type.inner() == byte[].class) { // HTT: type == null is unreachable - hoconToMap's every call site (doRead's top-level call, and this method's own OBJECT/ARRAY recursive calls below) always threads a non-null ClassMeta (object() at worst).
426 			var s = ((HoconValue.HoconString) val).getValue();
427 			var swap = type.getSwap(this);
428 			if (swap != null)
429 				return unswap(swap, s, type);
430 			return Base64.getDecoder().decode(s);
431 		}
432 		// Look up bean property metas so nested typed-map properties (Map<K,V> with non-String K) can have
433 		// their keys coerced via the converter's Map→Map path before the bean binds them.  Collection/Array
434 		// property types are also surfaced here so the ARRAY branch can thread the parent's element type
435 		// into recursion and the byte[] swap dispatch above can fire at the right depth.  Other property
436 		// shapes (Optional, primitive, bean) still pass through as object() so the lazy-typed conversion
437 		// path stays in place.
438 		var beanMeta = (type != null && type.isBean()) ? type.getBeanMeta() : null; // HTT: type == null is unreachable - see the HTT note on the byte[] check above; every call site threads a non-null ClassMeta.
439 		var result = switch (val.getType()) {
440 			case STRING -> ((HoconValue.HoconString) val).getValue();
441 			case NUMBER -> ((HoconValue.HoconNumber) val).getValue();
442 			case BOOLEAN -> ((HoconValue.HoconBoolean) val).getValue();
443 			case NULL -> null;
444 			case OBJECT -> {
445 				var obj = (HoconValue.HoconObject) val;
446 				var map = newGenericMap();
447 				for (var e : obj.getMembers().entrySet()) {
448 					var key = e.getKey();
449 					ClassMeta<?> childType = object();
450 					if (beanMeta != null && key != null && !key.equals(getBeanTypePropertyName(type))) { // HTT: key == null is unreachable - HoconObject member keys always come from readPath()'s non-null path components, never a Java null.
451 						var pMeta = beanMeta.getPropertyMeta(key);
452 						if (pMeta != null) {
453 							var cm = (ClassMeta<?>) pMeta.getBeanInfo();
454 							if (cm != null && (cm.isMap() || cm.isCollectionOrArray())) // HTT: cm == null is unreachable - getBeanInfo() only returns null for a BeanPropertyMeta built via the bean-modeling-only path; every BeanMeta this parser session touches is built through the marshalling-side construction path, which always populates a non-null typeMeta.
455 								childType = cm;
456 						}
457 					}
458 					map.put(key, hoconToMap(e.getValue(), childType));
459 				}
460 				yield map;
461 			}
462 			case ARRAY -> {
463 				var arr = (HoconValue.HoconArray) val;
464 				var list = newGenericList();
465 				var elType = (type != null && type.isCollectionOrArray()) ? coalesce(type.getElementType(), object()) : object(); // HTT: type == null is unreachable - see the HTT note on the byte[] check above; every call site threads a non-null ClassMeta.
466 				for (var el : arr.getElements()) {
467 					list.add(hoconToMap(el, elType));
468 				}
469 				yield list;
470 			}
471 			default -> null;
472 		};
473 		// If the enclosing target is a typed Map (e.g. Map<TestEnum,String>), coerce keys / values
474 		// via the converter's findMapConversion so the bean property binder gets a properly-keyed map.
475 		if (result instanceof Map<?,?> result2 && type != null && type.isMap() // HTT: type == null is unreachable - see the HTT note on the byte[] check above; every call site threads a non-null ClassMeta.
476 				&& type.getKeyType() != null && !type.getKeyType().isObject()) // HTT: type.getKeyType() == null is unreachable - getKeyType() on a map-typed ClassMeta always resolves to at least Object's ClassMeta, never a null reference.
477 			return convertToMemberType(null, result2, type);
478 		return result;
479 	}

481 	private Object convertToBean(Object map, ClassMeta<?> type) throws ExecutableException, ParseException {
482 		if (map == null) // HTT: dead -- the sole call site (doRead) passes hoconToMap(root, type), and root is only reached after readRoot's own null-check; hoconToMap's OBJECT branch (root is always a HoconObject) always returns a non-null MarshalledMap, so map is never actually null here
483 			return null; // HTT: dead -- see comment above
484 		if (type == null) // HTT: dead -- the sole call site (doRead) passes its own `type` parameter, which is guaranteed non-null by ParserSession.readInner's type.isVoid() check executed before doRead is ever reached
485 			type = object(); // HTT: dead -- see comment above
486 		var casted = cast((MarshalledMap) map, null, type);
487 		if (casted != map)
488 			return casted;
489 		if (!type.isObject()) {
490 			var swap = type.getSwap(this);
491 			if (nn(swap)) {
492 				var sType = swap.getSwapClassMeta(this);
493 				var swapped = convertToMemberType(null, map, sType);
494 				return unswap(swap, swapped, type);
495 			}
496 			var result = convertToMemberType(null, map, type);
497 			if (type.isBean() && result != null) // HTT: result == null is unreachable when type.isBean() - convertToMemberType() either successfully constructs and returns the bean, or throws (e.g. via a bean-instantiation/property-set failure); it never silently returns null for a non-null MarshalledMap input against a bean target.
498 				injectAnnotations((MarshalledMap) map, result);
499 			return result;
500 		}
501 		return map;
502 	}

504 	private void injectAnnotations(MarshalledMap map, Object bean) throws ExecutableException {
505 		var bm = toBeanMap(bean);
506 		for (var entry : map.entrySet()) {
507 			var key = entry.getKey();
508 			var pm = bm.getPropertyMeta(key);
509 			var val = pm == null ? null : getBeanValueSafely(bm, key);
510 			if (pm == null || val == null)
511 				continue;
512 			var cm = (ClassMeta<?>) pm.getBeanInfo();
513 			if (cm.getNameProperty() != null)
514 				setName(cm, val, key);
515 			injectParentAnnotations(cm, val, entry.getValue(), bean);
516 		}
517 	}

519 	/*
520 	 * Recursively injects @ParentProperty (and @NameProperty on map values / nested beans) into val.
521 	 * Intermediate collections and maps are transparent: a collection/map element's parent is
522 	 * parentBean (the nearest enclosing bean), skipping all intermediate containers.  This keeps HOCON
523 	 * consistent with the streaming parsers.
524 	 */
525 	private void injectParentAnnotations(ClassMeta<?> cm, Object val, Object node, Object parentBean) throws ExecutableException {
526 		if (val == null || cm == null) // HTT: cm == null is unreachable - every call site (the field loop above, and the two recursive calls below, which each null-check et/vt first) always threads a non-null ClassMeta.
527 			return;
528 		if (cm.getParentProperty() != null)
529 			setParent(cm, val, parentBean);
530 		if (cm.isCollectionOrArray()) {
531 			var et = cm.getElementType();
532 			if (et == null || et.isObject()) // HTT: et == null is unreachable - getElementType() on a collection/array-typed ClassMeta always resolves to at least Object's ClassMeta, never a null reference.
533 				return;
534 			var nodeList = node instanceof List<?> nl ? nl : null;
535 			var i = 0;
536 			for (var element : toIterable(val)) {
537 				injectParentAnnotations(et, element, nodeList != null && i < nodeList.size() ? nodeList.get(i) : null, parentBean); // HTT: nodeList == null (and by extension "i >= nodeList.size()") is unreachable - a shape-mismatched raw value (e.g. a HoconObject/MarshalledMap fed into a Collection-typed property) throws during bean-property binding before this method is ever reached in HOCON (confirmed: "children = {}" against a List<Child> property raises BeanRuntimeException rather than desyncing), so whenever we get here "node" was parsed from the very same list structure "val" was coerced from and is always a same-sized List.
538 				i++;
539 			}
540 		} else if (cm.isMap() && val instanceof Map<?,?> valMap) { // HTT: cm.isMap()==true with val not instanceof Map is unreachable - a Map-typed property either successfully coerces to a real Map instance or the parse throws before this injection pass ever runs (a shape-mismatched raw value, e.g. an array fed into a Map-typed property, throws during coercion rather than desyncing like the analogous collection case above).
541 			var vt = cm.getValueType();
542 			if (vt == null || vt.isObject()) // HTT: vt == null is unreachable - getValueType() on a map-typed ClassMeta always resolves to at least Object's ClassMeta, never a null reference.
543 				return;
544 			var nodeMap = node instanceof Map<?,?> nm ? nm : null; // HTT: the "node not instanceof Map" false-arm is unreachable - unlike collections (see the nodeList desync above), a Map-typed property's raw parsed node is always itself a Map when the value has already been successfully coerced to a Map here; any shape mismatch throws during coercion before this method is ever reached.
545 			for (Map.Entry<?,?> e : valMap.entrySet()) {
546 				if (vt.getNameProperty() != null)
547 					setName(vt, e.getValue(), e.getKey());
548 				injectParentAnnotations(vt, e.getValue(), nodeMap != null ? nodeMap.get(e.getKey()) : null, parentBean); // HTT: nodeMap == null is unreachable here - see the HTT note on the nodeMap assignment above.
549 			}
550 		} else if (cm.isBean() && !(val instanceof Map) && node instanceof MarshalledMap nodeMap) {
551 			injectAnnotations(nodeMap, val);
552 		}
553 	}

555 	private static Iterable<?> toIterable(Object val) {
556 		if (val instanceof Collection<?> c)
557 			return c;
558 		if (val instanceof Object[] a)
559 			return Arrays.asList(a);
560 		return List.of();
561 	}
