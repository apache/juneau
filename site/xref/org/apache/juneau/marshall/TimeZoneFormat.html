17  package org.apache.juneau.marshall;
18  
19  import java.time.*;
20  import java.util.*;
21  
22  /**
23   * Supported wire formats for {@link TimeZone} and {@link ZoneId} values.
24   *
25   * <p>
26   * Used by {@link MarshallingContext.Builder#timeZoneFormat(TimeZoneFormat)},
27   * {@link Marshalled#timeZoneFormat()},
28   * {@link MarshalledProp#timeZoneFormat()}, and
29   * {@link MarshalledConfig#timeZoneFormat()} to control how time-zone
30   * identifiers are written to and read from the wire.
31   *
32   * <p>
33   * The default is {@link #ID} which preserves the historical wire output produced before this enum existed
34   * ({@code TimeZone.getID()} / {@code ZoneId.getId()}).
35   */
36  public enum TimeZoneFormat {
37  
38  	/** Sentinel meaning "no value configured" — falls through to the next-higher precedence level. */
39  	NOT_SET,
40  
41  	/** The IANA zone id (the default): <js>"America/Los_Angeles"</js>, <js>"UTC"</js>, <js>"GMT-08:00"</js>. */
42  	ID,
43  
44  	/** Normalized to a {@link ZoneOffset} where possible: <js>"-08:00"</js>; falls back to {@link #ID} for region zones. */
45  	OFFSET,
46  
47  	/**
48  	 * Default long display name in {@link Locale#ROOT} (formatting only; not always round-trippable):
49  	 * <js>"Pacific Standard Time"</js>.
50  	 *
51  	 * <p>
52  	 * Parsing accepts any value that {@link ZoneId#of(String)} understands; localized display names are
53  	 * <b>not</b> reverse-parsed.
54  	 */
55  	NAME_LONG,
56  
57  	/**
58  	 * Default short display name in {@link Locale#ROOT} (formatting only; not always round-trippable):
59  	 * <js>"PST"</js>.
60  	 *
61  	 * <p>
62  	 * Parsing accepts any value that {@link ZoneId#of(String)} understands; localized display names are
63  	 * <b>not</b> reverse-parsed.
64  	 */
65  	NAME_SHORT;
66  
67  	/**
68  	 * Formats the specified {@link TimeZone} using this format.
69  	 *
70  	 * @param value The value to format. Can be <jk>null</jk> (returns <jk>null</jk>).
71  	 * @return The formatted string, or <jk>null</jk> if {@code value} is <jk>null</jk>.
72  	 */
73  	public String format(TimeZone value) {
74  		if (value == null)
75  			return null;
76  		return switch (this) {
77  			case NAME_LONG -> value.getDisplayName(false, TimeZone.LONG, Locale.ROOT);
78  			case NAME_SHORT -> value.getDisplayName(false, TimeZone.SHORT, Locale.ROOT);
79  			case OFFSET -> formatOffset(value.toZoneId());
80  			case NOT_SET, ID -> value.getID();
81  		};
82  	}
83  
84  	/**
85  	 * Formats the specified {@link ZoneId} using this format.
86  	 *
87  	 * @param value The value to format. Can be <jk>null</jk> (returns <jk>null</jk>).
88  	 * @return The formatted string, or <jk>null</jk> if {@code value} is <jk>null</jk>.
89  	 */
90  	public String format(ZoneId value) {
91  		if (value == null)
92  			return null;
93  		return switch (this) {
94  			case NAME_LONG -> TimeZone.getTimeZone(value).getDisplayName(false, TimeZone.LONG, Locale.ROOT);
95  			case NAME_SHORT -> TimeZone.getTimeZone(value).getDisplayName(false, TimeZone.SHORT, Locale.ROOT);
96  			case OFFSET -> formatOffset(value);
97  			case NOT_SET, ID -> value.getId();
98  		};
99  	}
100 
101 	/**
102 	 * Parses the specified wire value as a {@link TimeZone}.
103 	 *
104 	 * @param value The wire value. Can be <jk>null</jk> or blank (returns <jk>null</jk>).
105 	 * @return The parsed {@link TimeZone}, or <jk>null</jk> if {@code value} is <jk>null</jk> or blank.
106 	 */
107 	public static TimeZone parseTimeZone(String value) {
108 		if (value == null)
109 			return null;
110 		var s = value.trim();
111 		if (s.isEmpty())
112 			return null;
113 		return TimeZone.getTimeZone(s);
114 	}
115 
116 	/**
117 	 * Parses the specified wire value as a {@link ZoneId}.
118 	 *
119 	 * @param value The wire value. Can be <jk>null</jk> or blank (returns <jk>null</jk>).
120 	 * @return The parsed {@link ZoneId}, or <jk>null</jk> if {@code value} is <jk>null</jk> or blank.
121 	 */
122 	public static ZoneId parseZoneId(String value) {
123 		if (value == null)
124 			return null;
125 		var s = value.trim();
126 		if (s.isEmpty())
127 			return null;
128 		return ZoneId.of(s, ZoneId.SHORT_IDS);
129 	}
130 
131 	private static String formatOffset(ZoneId zoneId) {
132 		try {
133 			return ZoneOffset.from(zoneId.getRules().getOffset(Instant.now())).getId();
134 		} catch (@SuppressWarnings("unused") Exception e) {  // NOSONAR: ZoneOffset.from throws DateTimeException; fall back to ID
135 			return zoneId.getId();
136 		}
137 	}
138 }
