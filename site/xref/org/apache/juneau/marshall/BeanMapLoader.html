17  package org.apache.juneau.marshall;
18  
19  import java.io.*;
20  
21  import org.apache.juneau.commons.bean.*;
22  import org.apache.juneau.marshall.collections.*;
23  import org.apache.juneau.marshall.json5.*;
24  import org.apache.juneau.marshall.parser.*;
25  
26  /**
27   * Marshalling-side helper for loading {@link BeanMap} contents from JSON text or a reader/parser pair.
28   *
29   * <h5 class='section'>Example:</h5>
30   * <p class='bjava'>
31   * 	<jc>// Populate a bean map from a JSON string.</jc>
32   * 	BeanMap&lt;Person&gt; <jv>m</jv> = <jv>bc</jv>.newBeanMap(Person.<jk>class</jk>);
33   * 	BeanMapLoader.<jsm>load</jsm>(<jv>m</jv>, <js>"{name:'John',age:21}"</js>);
34   * </p>
35   */
36  public final class BeanMapLoader {
37  
38  	private BeanMapLoader() {}
39  
40  	/**
41  	 * Populates the supplied {@link BeanMap} with the contents of the JSON text in {@code input}.
42  	 *
43  	 * <p>
44  	 * Equivalent to the legacy {@code BeanMap.load(String)} method.
45  	 *
46  	 * @param <T> The bean type.
47  	 * @param m The bean map to populate. Must not be <jk>null</jk>.
48  	 * @param input The text that will get parsed into a map and then added to {@code m}. Can be <jk>null</jk> (nothing is added).
49  	 * @return The supplied bean map for fluent chaining.
50  	 * @throws ParseException Malformed input encountered.
51  	 */
52  	public static <T> BeanMap<T> load(BeanMap<T> m, String input) throws ParseException {
53  		m.putAll(Json5Map.ofString(input));
54  		return m;
55  	}
56  
57  	/**
58  	 * Populates the supplied {@link BeanMap} with the contents of the reader using the specified {@link ReaderParser}.
59  	 *
60  	 * <p>
61  	 * Equivalent to the legacy {@code BeanMap.load(Reader, ReaderParser)} method.
62  	 *
63  	 * @param <T> The bean type.
64  	 * @param m The bean map to populate. Must not be <jk>null</jk>.
65  	 * @param r The reader containing serialized text. Can be <jk>null</jk> (nothing is added).
66  	 * @param p The parser to use to parse the text.
67  	 * @return The supplied bean map for fluent chaining.
68  	 * @throws ParseException Malformed input encountered.
69  	 */
70  	public static <T> BeanMap<T> load(BeanMap<T> m, Reader r, ReaderParser p) throws ParseException {
71  		m.putAll(JsonMap.ofString(r, p));
72  		return m;
73  	}
74  }
