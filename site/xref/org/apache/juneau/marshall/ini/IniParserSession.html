104 	@Override
105 	protected <T> T doRead(ParserPipe pipe, ClassMeta<T> type) throws IOException, ParseException, ExecutableException {
106 		try (Reader r = pipe.getParserReader()) {
107 			if (r == null)
108 				return null;
109 			var sections = readIniContent(r);
110 			if (sections.isEmpty()) // HTT: provably dead -- readIniContent() unconditionally pre-seeds sections.computeIfAbsent("", ...) before its read loop runs, so sections.size() is always >= 1 here
111 				return type.canCreateNewBean(getOuter()) ? type.newInstance(getOuter()) : null; // HTT: unreachable -- see the isEmpty() guard above; this whole line is dead.
112 			if (!type.isBean() && !type.isMap())
113 				throw new ParseException(this, "INI format requires bean or Map<String,?> target. Got: %s", type.inner().getName());
114 			if (type.isMap()) {
115 				var result = buildMapFromSections(sections, "");
116 				return (T) convertMapToTarget(result, type);
117 			}
118 			var bm = toBeanMap(type.newInstance(getOuter()));
119 			populateBean(bm, sections, "");
120 			return type.cast(bm.getBean());
121 		} // HTT: try-with-resources close() bytecode -- ParserReader.close() (the only Reader type pipe.getParserReader() ever returns) is a documented no-op ("Input readers are closed in the ParserPipe class"), so it can never throw; the compiler-generated null-check/primary-exception/addSuppressed branches for a throwing close() are therefore all unreachable from this call site.
122 	}

124 	/**
125 	 * Parses INI content into section -> (key -> raw value) structure.
126 	 *
127 	 * @param r The reader.
128 	 * 	<br>Must not be <jk>null</jk>.
129 	 * @return Map of section name to key-value map. Default section uses "".
130 	 */
131 	protected Map<String,Map<String,String>> readIniContent(Reader r) throws IOException {
132 		var sections = new LinkedHashMap<String,Map<String,String>>();
133 		var current = sections.computeIfAbsent("", k -> m());
134 		var br = r instanceof BufferedReader r2 ? r2 : new BufferedReader(r);
135 		String line;
136 		while ((line = br.readLine()) != null) {
137 			line = line.trim();
138 			if (line.isEmpty() || line.startsWith("#"))
139 				continue;
140 			if (line.startsWith("[")) {
141 				var end = line.indexOf(']');
142 				if (end < 0)
143 					continue;
144 				var sectionName = line.substring(1, end).trim();
145 				current = sections.computeIfAbsent(sectionName, k -> m());
146 				continue;
147 			}
148 			var kv = splitKeyValue(line);
149 			if (kv != null) {
150 				var key = kv[0].trim();
151 				var value = kv[1].trim();
152 				// Strip inline comment (# ...) from value (when not inside quotes)
153 				value = stripInlineComment(value);
154 				current.put(key, value);
155 			}
156 		}
157 		return sections;
158 	}

160 	/**
161 	 * Splits an INI {@code key=value} / {@code key:value} (or configured-separator) line into its key and
162 	 * raw-value parts.
163 	 *
164 	 * <p>
165 	 * Behavior mirrors the former {@code ^([^=#\s][^=]*)\s*[=:]\s*(.*)$} regex but without its super-linear
166 	 * backtracking: when this session's {@link IniParser.Builder#kvSeparator(char) kvSeparator} is left at its
167 	 * default (<c>=</c>), an {@code '='} delimiter (if present) binds to the first {@code '='}; otherwise the
168 	 * last {@code ':'} is used (unchanged legacy behavior). When a non-default {@code kvSeparator} has been
169 	 * configured (e.g. to round-trip an {@link IniSerializer} that was configured with a matching
170 	 * {@code kvSeparator}), only that character is recognized as the delimiter. Surrounding whitespace is left
171 	 * in place (callers trim).
172 	 *
173 	 * @param line The (already-trimmed) candidate line.
174 	 * @return A two-element array of {raw-key, raw-value}, or <jk>null</jk> if the line is not a key/value pair.
175 	 */
176 	@SuppressWarnings({
177 		"java:S1168" // null is a distinct "not a key/value pair" sentinel; the caller guards on != null before indexing, so an empty array would change parse semantics.
178 	})
179 	private String[] splitKeyValue(String line) {
180 		if (line.isEmpty()) // HTT: unreachable -- the sole call site (readIniContent) already skips via `line.isEmpty() || line.startsWith("#")` before invoking this method, so line is always non-empty here.
181 			return null; // HTT: unreachable -- see above.
182 		var first = line.charAt(0);
183 		if (first == '=' || first == '#' || Character.isWhitespace(first)) // HTT: the `first == '#'` and `Character.isWhitespace(first)` disjuncts are unreachable from the sole call site -- readIniContent already filters `line.startsWith("#")`, and line was already trim()'d there, so it can never start with whitespace (a whitespace-only line would have trimmed to empty and been filtered by the isEmpty() check instead). Only `first == '='` is reachable (see f03a).
184 			return null;
185 		var sep = ctx.kvSeparator;
186 		var idx = line.indexOf(sep);
187 		if (idx < 0 && sep == '=')
188 			idx = line.lastIndexOf(':');
189 		if (idx < 1)
190 			return null;
191 		return new String[] { line.substring(0, idx), line.substring(idx + 1) };
192 	}

194 	private void populateBean(BeanMap<?> bm, Map<String,Map<String,String>> sections, String sectionPath) throws ParseException, ExecutableException {
195 		var defaultSection = sections.get(sectionPath);
196 		if (defaultSection != null) {
197 			for (Entry<String,String> e : defaultSection.entrySet()) {
198 				var key = e.getKey();
199 				var rawValue = e.getValue();
200 				var pMeta = bm.getMeta().getProperties().get(key);
201 				if (pMeta == null && isIgnoreUnknownBeanProperties())
202 					continue;
203 				if (pMeta == null)
204 					throw new ParseException(this, "Unknown property '%s'", key);
205 				var value = readValue(rawValue, (ClassMeta<?>) pMeta.getBeanInfo());
206 				bm.put(key, value);
207 			}
208 		}
209 		// Process nested sections
210 		for (var entry : sections.entrySet()) {
211 			var sectionName = entry.getKey();
212 			var sub = entry.getValue();
213 			if (sectionName.equals(sectionPath))
214 				continue;
215 			var isChild = sectionPath.isEmpty() ? !sectionName.contains(SECTION_PATH_DELIMITER)
216 				: sectionName.startsWith(sectionPath + SECTION_PATH_DELIMITER);
217 			if (!isChild)
218 				continue;
219 			var childName = sectionPath.isEmpty() ? sectionName
220 				: sectionName.substring(sectionPath.length() + SECTION_PATH_DELIMITER.length()).split(Pattern.quote(SECTION_PATH_DELIMITER))[0];
221 			var pMeta = bm.getMeta().getProperties().get(childName);
222 			if (pMeta == null && isIgnoreUnknownBeanProperties())
223 				continue;
224 			if (pMeta == null)
225 				continue;
226 			var cMeta = (ClassMeta<?>) pMeta.getBeanInfo();
227 			var childPath = sectionPath.isEmpty() ? childName : sectionPath + SECTION_PATH_DELIMITER + childName;
228 			if (cMeta.isBean()) {
229 				var child = toBeanMap(cMeta.newInstance(getOuter()));
230 				populateBean(child, sections, childPath);
231 				bm.put(childName, child.getBean());
232 			} else if (cMeta.isMap() && sub != null) { // HTT: `sub != null` is unreachable -- sub is `entry.getValue()` from iterating `sections.entrySet()`, and readIniContent's `computeIfAbsent(sectionName, k -> m())` guarantees every section entry's value is a non-null (possibly empty) Map.
233 				var valueType = cMeta.getValueType();
234 				if (valueType == null) // HTT: unreachable -- cMeta.isMap() is true here, and ClassMeta.findKeyValueTypes() always synthesizes a non-null (Object,Object) pair for any class in the MAP category, so getValueType() can never be null.
235 					valueType = object(); // HTT: unreachable -- see above.
236 				var map = new LinkedHashMap<String,Object>();
237 				for (Entry<String,String> e : sub.entrySet())
238 					map.put(e.getKey(), readValue(e.getValue(), valueType));
239 				bm.put(childName, convertMapToTarget(map, cMeta));
240 			}
241 		}
242 	}

244 	private Map<String,Object> buildMapFromSections(Map<String,Map<String,String>> sections, String sectionPath) throws ParseException, ExecutableException {
245 		var result = new LinkedHashMap<String,Object>();
246 		var defaultSection = sections.get(sectionPath);
247 		if (defaultSection != null) {
248 			for (Entry<String,String> e : defaultSection.entrySet())
249 				result.put(e.getKey(), readValue(e.getValue(), object()));
250 		}
251 		for (var entry : sections.entrySet()) {
252 			var sectionName = entry.getKey();
253 			var childSection = entry.getValue();
254 			if (sectionName.equals(sectionPath))
255 				continue;
256 			var isChild = sectionPath.isEmpty() ? !sectionName.contains(SECTION_PATH_DELIMITER)
257 				: sectionName.startsWith(sectionPath + SECTION_PATH_DELIMITER);
258 			if (!isChild)
259 				continue;
260 			var childName = sectionPath.isEmpty() ? sectionName
261 				: sectionName.substring(sectionPath.length() + SECTION_PATH_DELIMITER.length()).split(Pattern.quote(SECTION_PATH_DELIMITER))[0];
262 			if (result.containsKey(childName))
263 				continue;
264 			var childPath = sectionPath.isEmpty() ? childName : sectionPath + SECTION_PATH_DELIMITER + childName;
265 			var hasNested = sections.keySet().stream().anyMatch(s -> s.startsWith(childPath + SECTION_PATH_DELIMITER));
266 			if (hasNested) {
267 				result.put(childName, buildMapFromSections(sections, childPath));
268 			} else if (childSection != null) { // HTT: `childSection != null` is unreachable -- childSection is `entry.getValue()` from iterating `sections.entrySet()`, and readIniContent's `computeIfAbsent(sectionName, k -> m())` guarantees every section entry's value is a non-null (possibly empty) Map.
269 				var map = new LinkedHashMap<String,Object>();
270 				for (Entry<String,String> e : childSection.entrySet())
271 					map.put(e.getKey(), readValue(e.getValue(), object()));
272 				result.put(childName, map);
273 			}
274 		}
275 		return result;
276 	}

278 	private Object readValue(String raw, ClassMeta<?> targetType) throws ParseException, ExecutableException {
279 		if (raw == null) // HTT: unreachable -- all four call sites pass `e.getValue()` from a `Map<String,String>` populated exclusively by readIniContent's `current.put(key, value)`, which only ever stores non-null trimmed/comment-stripped Strings.
280 			return null; // HTT: unreachable -- see above.
281 		var trimmed = raw.trim();
282 		if (trimmed.equals("null"))
283 			return null;
284 		if (trimmed.equalsIgnoreCase("true"))
285 			return true;
286 		if (trimmed.equalsIgnoreCase("false"))
287 			return false;
288 		if (trimmed.startsWith("'") && trimmed.endsWith("'") && trimmed.length() >= 2) {
289 			var inner = trimmed.substring(1, trimmed.length() - 1).replace("''", "'");
290 			return convertToMemberType(null, inner, targetType);
291 		}
292 		if (trimmed.startsWith("[") || trimmed.startsWith("{"))
293 			return getJson5Parser().read(trimmed, targetType);
294 		if (targetType.isNumber()) {
295 			try {
296 				if (trimmed.contains(".") || trimmed.toLowerCase(Locale.ROOT).contains("e"))
297 					return Double.parseDouble(trimmed);
298 				return Long.parseLong(trimmed);
299 			} catch (@SuppressWarnings("unused") NumberFormatException e) {
300 				return convertToMemberType(null, trimmed, targetType);
301 			}
302 		}
303 		// Provably dead from every current call site (see "// HTT" markers below for the precise reason):
304 		// Date/Calendar/Temporal/Duration/Period bean properties get a default post-processor-installed
305 		// swap, so BeanPropertyMeta#getBeanInfo() (the swap-aware `typeMeta`, not the unswapped
306 		// `rawTypeMeta`) reports them as plain java.lang.Object here -- confirmed by direct reflection
307 		// probing -- and both buildMapFromSections call sites always pass object() too. The actual
308 		// conversion for these properties happens later via BeanMap#put()'s own swap-aware coercion.
309 		if (targetType.isDate()) // HTT: unreachable -- targetType is always Object here (see block comment above)
310 			return readDate(trimmed, targetType); // HTT: unreachable -- targetType is always Object here (see block comment above)
311 		if (targetType.isCalendar()) // HTT: unreachable -- targetType is always Object here (see block comment above)
312 			return readCalendar(trimmed, targetType); // HTT: unreachable -- targetType is always Object here (see block comment above)
313 		if (targetType.isTemporal()) // HTT: unreachable -- targetType is always Object here (see block comment above)
314 			return readTemporal(trimmed, targetType); // HTT: unreachable -- targetType is always Object here (see block comment above)
315 		if (targetType.isDuration()) // HTT: unreachable -- targetType is always Object here (see block comment above)
316 			return readDuration(trimmed); // HTT: unreachable -- targetType is always Object here (see block comment above)
317 		if (targetType.isPeriod()) // HTT: unreachable -- targetType is always Object here (see block comment above)
318 			return readPeriod(trimmed); // HTT: unreachable -- targetType is always Object here (see block comment above)
319 		return convertToMemberType(null, trimmed, targetType);
320 	}

322 	/**
323 	 * Returns a Json5 parser configured with this session's {@link MarshallingContext}.
324 	 *
325 	 * <p>
326 	 * Required so that format hints carried on the parent {@code IniParser} context (e.g.
327 	 * {@link LocaleFormat}, {@link TimeZoneFormat}) are honored
328 	 * when nested JSON5-encoded collection/map values are deserialized.  Using {@code Json5Parser.DEFAULT}
329 	 * here would silently drop those hints and fall back to the format-default behavior.
330 	 *
331 	 * <p>
332 	 * Memoized via {@link Memoizer} — first call constructs the parser; subsequent calls return
333 	 * the cached instance.  {@link #readValue} calls this once per JSON5-encoded value, so a single
334 	 * builder-construction + parser-instantiation is amortized across the whole INI document rather
335 	 * than repeated for every {@code […]} / {@code {…}} value.
336 	 *
337 	 * <p>
338 	 * List-of-{@link Locale} values written under {@link LocaleFormat#UNDERSCORE}
339 	 * (wire form {@code "en_US"}) were being parsed via {@code Locale.forLanguageTag("en_US")} (BCP_47 default)
340 	 * which returns {@link Locale#ROOT}, dropping the locale content.
341 	 */
342 	private JsonParser getJson5Parser() {
343 		return json5Parser.get();
344 	}

346 	private static Object convertMapToTarget(Map<String,Object> map, ClassMeta<?> type) throws ParseException, ExecutableException {
347 		if (JsonMap.class.isAssignableFrom(type.inner()))
348 			return new JsonMap(map);
349 		return map;
350 	}

352 	private static String stripInlineComment(String value) {
353 		if (value == null || !value.contains("#")) // HTT: `value == null` is unreachable -- the sole call site (readIniContent) always passes `kv[1].trim()`, a non-null String.
354 			return value;
355 		var inQuote = false;
356 		for (var i = 0; i < value.length(); i++) {
357 			var c = value.charAt(i);
358 			if (c == '\'')
359 				inQuote = !inQuote;
360 			else if (c == '#' && !inQuote)
361 				return value.substring(0, i).trim();
362 		}
363 		return value;
364 	}
