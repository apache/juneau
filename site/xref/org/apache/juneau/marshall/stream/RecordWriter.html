17  package org.apache.juneau.marshall.stream;
18  
19  import java.io.*;
20  
21  /**
22   * Public low-level whole-value push generator over a Juneau-formatted document.
23   *
24   * <p>
25   * Format-neutral cursor that emits one whole-value record per {@link #write(Object)} call.
26   * Single-rooted formats accept exactly one call (subsequent calls throw); multi-rooted formats
27   * (jsonl, csv, sse) emit one line / row / event per call.
28   *
29   * <h5 class='section'>Usage:</h5>
30   * <p class='bjava'>
31   * 	StringBuilder <jv>sb</jv> = <jk>new</jk> StringBuilder();
32   * 	<jk>try</jk> (RecordWriter <jv>w</jv> = Json.<jsf>DEFAULT</jsf>.writeRecords(<jv>sb</jv>)) {
33   * 		<jv>w</jv>.write(<jv>bean</jv>);
34   * 	}
35   * </p>
36   *
37   * <h5 class='section'>Streaming quality:</h5>
38   * <ul>
39   * 	<li>{@link #isStreaming()} == <jk>true</jk> &mdash; O(1)-memory streaming.  FULL-token formats
40   * 		return a cursor that is also a {@link TokenWriter} (the structural surface).
41   * 	<li>{@link #isStreaming()} == <jk>false</jk> &mdash; buffered whole-value cursor (backed by
42   * 		{@link RecordAdapter}); {@link #write(Object)} delegates to the format's existing
43   * 		serializer and the structural surface is not available.
44   * </ul>
45   *
46   * <h5 class='section'>Resource ownership:</h5>
47   * <p>
48   * Every {@link RecordReader} / {@code RecordWriter} produced by a factory method
49   * <b>closes what it was handed</b> &mdash; if the underlying input/output is {@link Closeable},
50   * {@link #close()} closes it.  This contract is symmetric across all factory methods.
51   *
52   * <h5 class='section'>See Also:</h5><ul>
53   * 	<li>{@link RecordReader} &mdash; the pull-side counterpart.
54   * 	<li>{@link TokenWriter} &mdash; structural extension for FULL-capability formats.
55   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/SerializersAndParsers">Serializers and Parsers</a>
56   * </ul>
57   */
58  public interface RecordWriter extends Closeable, Flushable {
59  
60  	/**
61  	 * Emits one whole-value record.
62  	 *
63  	 * @param value The value.  May be <jk>null</jk>.
64  	 * @return This object.
65  	 * @throws IOException If a problem occurred writing to the underlying stream.
66  	 */
67  	RecordWriter write(Object value) throws IOException;
68  
69  	/**
70  	 * Returns <jk>true</jk> if this cursor is an O(1)-memory streaming cursor, or <jk>false</jk> if
71  	 * it is a buffered whole-value cursor (backed by {@link RecordAdapter}).
72  	 *
73  	 * @return <jk>true</jk> if this cursor streams; <jk>false</jk> if it buffers.
74  	 */
75  	boolean isStreaming();
76  
77  	/**
78  	 * Flushes any buffered output to the underlying stream.
79  	 *
80  	 * @throws IOException If a problem occurred flushing the underlying stream.
81  	 */
82  	@Override
83  	void flush() throws IOException;
84  
85  	/**
86  	 * Closes the writer, flushing any buffered output first.
87  	 *
88  	 * <p>
89  	 * Implementations should be idempotent &mdash; closing a closed writer is a no-op.
90  	 */
91  	@Override
92  	void close() throws IOException;
93  }
