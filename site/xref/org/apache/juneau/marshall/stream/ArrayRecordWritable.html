17  package org.apache.juneau.marshall.stream;
18  
19  import java.io.*;
20  
21  /**
22   * Capability role for serializers that can emit each element of a top-level wire array as a
23   * separate record.
24   *
25   * <p>
26   * The symmetric writer-side counterpart of {@link ArrayRecordReadable}.  Capability is detected
27   * via {@code instanceof}.
28   *
29   * <h5 class='section'>See Also:</h5><ul>
30   * 	<li>{@link ArrayRecordReadable} &mdash; the symmetric reader role.
31   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/SerializersAndParsers">Serializers and Parsers</a>
32   * </ul>
33   */
34  @SuppressWarnings({
35  	"resource" // Cursor methods return Closeables owned by the caller; Eclipse JDT @Owning warning is by design.
36  })
37  public interface ArrayRecordWritable {
38  
39  	/**
40  	 * Opens a {@link RecordWriter} that wraps each {@link RecordWriter#write(Object) write(...)}
41  	 * call as one element of a top-level wire array.
42  	 *
43  	 * @param output The output.
44  	 * @return A new element-streamed {@link RecordWriter}.
45  	 * @throws IOException If a problem occurred opening the underlying output.
46  	 */
47  	RecordWriter writeArrayRecords(Object output) throws IOException;
48  
49  	/**
50  	 * Variant of {@link #writeArrayRecords(Object)} for length-prefixed binary formats that
51  	 * require the element count up front (msgpack, bson).  Other formats may ignore the count and
52  	 * behave the same as {@link #writeArrayRecords(Object)}.
53  	 *
54  	 * @param output The output.
55  	 * @param expectedCount The number of elements that will be written.  Must match the number of
56  	 * 	{@link RecordWriter#write(Object)} calls before {@link RecordWriter#close()}.
57  	 * @return A new element-streamed {@link RecordWriter}.
58  	 * @throws IOException If a problem occurred opening the underlying output.
59  	 */
60  	default RecordWriter writeArrayRecords(Object output, int expectedCount) throws IOException {
61  		return writeArrayRecords(output);
62  	}
63  
64  	/**
65  	 * Streaming-quality signal queryable <em>before</em> opening a cursor.
66  	 *
67  	 * @return <jk>true</jk> for true element streaming (json/cbor/msgpack); <jk>false</jk> for the
68  	 * 	buffered {@link RecordAdapter#arrayWriter}-backed formats (yaml, parquet, bson).
69  	 */
70  	default boolean isArrayRecordStreaming() { return true; }
71  }
