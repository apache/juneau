17  package org.apache.juneau.marshall.stream;
18  
19  import java.io.*;
20  import java.lang.reflect.*;
21  import java.util.*;
22  import java.util.stream.*;
23  
24  import org.apache.juneau.marshall.*;
25  import org.apache.juneau.marshall.parser.*;
26  
27  /**
28   * Public low-level whole-value pull cursor over a Juneau-formatted document.
29   *
30   * <p>
31   * Format-neutral cursor that yields one whole-value record per {@link #read(Class)} call.  Single-rooted
32   * formats (json, xml, bson, yaml, ...) degenerate to a one-record stream; multi-rooted formats
33   * (jsonl, csv, sse) expose each line / row / event as a separate record.
34   *
35   * <h5 class='section'>Usage:</h5>
36   * <p class='bjava'>
37   * 	<jk>try</jk> (RecordReader <jv>r</jv> = Json.<jsf>DEFAULT</jsf>.readRecords(<jv>input</jv>)) {
38   * 		<jk>while</jk> (<jv>r</jv>.canRead()) {
39   * 			Bean <jv>bean</jv> = <jv>r</jv>.read(Bean.<jk>class</jk>);
40   * 			process(<jv>bean</jv>);
41   * 		}
42   * 	}
43   * </p>
44   *
45   * <h5 class='section'>Streaming quality:</h5>
46   * <ul>
47   * 	<li>{@link #isStreaming()} == <jk>true</jk> &mdash; O(1)-memory streaming.  FULL-token formats
48   * 		return a cursor that is also a {@link TokenReader} (the structural surface).
49   * 	<li>{@link #isStreaming()} == <jk>false</jk> &mdash; buffered whole-value cursor (backed by
50   * 		{@link RecordAdapter}); {@link #read(Class)} delegates to the format's existing parser and
51   * 		the structural surface is not available.
52   * </ul>
53   *
54   * <h5 class='section'>See Also:</h5><ul>
55   * 	<li>{@link RecordWriter} &mdash; the push-side counterpart.
56   * 	<li>{@link TokenReader} &mdash; structural extension for FULL-capability formats.
57   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/SerializersAndParsers">Serializers and Parsers</a>
58   * </ul>
59   */
60  public interface RecordReader extends Closeable {
61  
62  	/**
63  	 * Returns <jk>true</jk> if {@link #read(Class)} can be called to produce another record.
64  	 *
65  	 * <p>
66  	 * Single-rooted formats return <jk>true</jk> exactly once (for the root value), then
67  	 * <jk>false</jk> after the value has been consumed.  Multi-rooted formats return <jk>true</jk>
68  	 * once per remaining record on the wire.
69  	 *
70  	 * @return <jk>true</jk> if a record is available; <jk>false</jk> at end of stream.
71  	 * @throws IOException If a problem occurred peeking at the underlying stream.
72  	 * @throws ParseException If the input is malformed for the format.
73  	 */
74  	boolean canRead() throws IOException, ParseException;
75  
76  	/**
77  	 * Reads and returns the next record bound to the requested type.
78  	 *
79  	 * @param <T> The expected return type.
80  	 * @param type The target Java type.
81  	 * @return The bound POJO.
82  	 * @throws IOException If a problem occurred reading from the underlying stream.
83  	 * @throws ParseException If the input is malformed for the format or cannot be bound to the type.
84  	 * @throws IllegalStateException If {@link #canRead()} returned <jk>false</jk>.
85  	 */
86  	<T> T read(Class<T> type) throws IOException, ParseException;
87  
88  	/**
89  	 * Same as {@link #read(Class)} but accepts a pre-resolved {@link ClassMeta}.
90  	 *
91  	 * @param <T> The expected return type.
92  	 * @param type The target type.
93  	 * @return The bound POJO.
94  	 * @throws IOException If a problem occurred reading from the underlying stream.
95  	 * @throws ParseException If the input is malformed or cannot be bound to the type.
96  	 */
97  	<T> T read(ClassMeta<T> type) throws IOException, ParseException;
98  
99  	/**
100 	 * Same as {@link #read(Class)} but accepts a parameterized {@link Type} and optional type
101 	 * arguments (the same shape as
102 	 * {@link Parser#read(Object, Type, Type...)}).
103 	 *
104 	 * @param <T> The expected return type.
105 	 * @param type The target type.
106 	 * @param args Optional type arguments for parameterized types.
107 	 * @return The bound POJO.
108 	 * @throws IOException If a problem occurred reading from the underlying stream.
109 	 * @throws ParseException If the input is malformed or cannot be bound to the type.
110 	 */
111 	<T> T read(Type type, Type... args) throws IOException, ParseException;
112 
113 	/**
114 	 * Returns an {@link Iterator} view that yields each remaining record bound to the requested
115 	 * type.
116 	 *
117 	 * <p>
118 	 * The iterator drains from this cursor; consuming both surfaces concurrently is not supported.
119 	 *
120 	 * <p>
121 	 * Checked {@link IOException} / {@link ParseException} thrown by the underlying cursor are
122 	 * wrapped in an {@link RuntimeParseException} (carrying the typed cause) so they can surface
123 	 * through the {@link Iterator} API.
124 	 *
125 	 * @param <T> The element type.
126 	 * @param type The element type.
127 	 * @return An iterator over the remaining records.
128 	 */
129 	default <T> Iterator<T> iterator(Class<T> type) {
130 		return new Iterator<>() {
131 			@Override public boolean hasNext() {
132 				try {
133 					return canRead();
134 				} catch (IOException | ParseException e) {
135 					throw new RuntimeParseException(e);
136 				}
137 			}
138 			@Override public T next() {
139 				if (! hasNext())
140 					throw new NoSuchElementException();
141 				try {
142 					return read(type);
143 				} catch (IOException | ParseException e) {
144 					throw new RuntimeParseException(e);
145 				}
146 			}
147 		};
148 	}
149 
150 	/**
151 	 * Returns a {@link Stream} view that yields each remaining record bound to the requested type.
152 	 *
153 	 * @param <T> The element type.
154 	 * @param type The element type.
155 	 * @return A stream over the remaining records.
156 	 */
157 	default <T> Stream<T> stream(Class<T> type) {
158 		var spliterator = Spliterators.spliteratorUnknownSize(iterator(type), Spliterator.ORDERED);
159 		return StreamSupport.stream(spliterator, false);
160 	}
161 
162 	/**
163 	 * Returns <jk>true</jk> if this cursor is an O(1)-memory streaming cursor, or <jk>false</jk> if
164 	 * it is a buffered whole-value cursor (backed by {@link RecordAdapter}).
165 	 *
166 	 * @return <jk>true</jk> if this cursor streams; <jk>false</jk> if it buffers.
167 	 */
168 	boolean isStreaming();
169 
170 	/**
171 	 * Closes the underlying stream.
172 	 *
173 	 * <p>
174 	 * Implementations should be idempotent &mdash; closing a closed reader is a no-op.
175 	 */
176 	@Override
177 	void close() throws IOException;
178 }
