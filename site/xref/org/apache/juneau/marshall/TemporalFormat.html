34  /**
35   * Supported wire formats for {@link TemporalAccessor} values.
36   *
37   * <p>
38   * Used by {@link MarshallingContext.Builder#temporalFormat(TemporalFormat)},
39   * {@link Marshalled#temporalFormat()},
40   * {@link MarshalledProp#temporalFormat()}, and
41   * {@link MarshalledConfig#temporalFormat()} to control how
42   * {@link TemporalAccessor} values ({@link Instant}, {@link LocalDate}, {@link LocalDateTime},
43   * {@link LocalTime}, {@link OffsetDateTime}, {@link OffsetTime}, {@link ZonedDateTime}, {@link Year},
44   * {@link YearMonth}, {@link MonthDay}, …) are written to and read from the wire.
45   *
46   * <p>
47   * The default is {@link #DEFAULT} which uses the per-subtype default formatter
48   * (e.g. {@link DateTimeFormatter#ISO_INSTANT} for {@link Instant}, {@link DateTimeFormatter#ISO_LOCAL_DATE}
49   * for {@link LocalDate}, etc.) — preserving the historical wire output produced before this enum existed.
50   *
51   * <p>
52   * Numeric formats ({@link #MILLIS}) emit native numeric wire types on binary serializers
53   * (BSON / CBOR / MsgPack) and bare numbers on text formats. Text formats emit a quoted ISO 8601 string for
54   * every other constant.
55   *
56   * <p>
57   * Most subtypes implement {@link Temporal}, but {@link MonthDay} only implements {@link TemporalAccessor}.
58   * Format/parse methods accept any {@link TemporalAccessor} so that {@link MonthDay} round-trips through the
59   * same dispatch path as the {@link Temporal} subtypes.
60   */
61  public enum TemporalFormat {
62  
63  	/** Sentinel meaning "no value configured" — falls through to the next-higher precedence level. */
64  	NOT_SET,
65  
66  	/**
67  	 * Per-subtype default formatter (the historical wire output).
68  	 *
69  	 * <p>
70  	 * Uses the ISO formatter most appropriate for each {@link Temporal} subtype:
71  	 * {@link DateTimeFormatter#ISO_INSTANT} for {@link Instant}, {@link DateTimeFormatter#ISO_LOCAL_DATE}
72  	 * for {@link LocalDate}, {@link DateTimeFormatter#ISO_OFFSET_DATE_TIME} for {@link ZonedDateTime} /
73  	 * {@link OffsetDateTime}, etc.
74  	 */
75  	DEFAULT,
76  
77  	/** {@link DateTimeFormatter#BASIC_ISO_DATE}: <js>"20111203"</js>. */
78  	BASIC_ISO_DATE,
79  
80  	/** {@link DateTimeFormatter#ISO_DATE}: <js>"2011-12-03+01:00"</js> or <js>"2011-12-03"</js>. */
81  	ISO_DATE,
82  
83  	/** {@link DateTimeFormatter#ISO_DATE_TIME}: <js>"2011-12-03T10:15:30+01:00[Europe/Paris]"</js>. */
84  	ISO_DATE_TIME,
85  
86  	/** {@link DateTimeFormatter#ISO_INSTANT}: <js>"2011-12-03T10:15:30Z"</js>. */
87  	ISO_INSTANT,
88  
89  	/** {@link DateTimeFormatter#ISO_LOCAL_DATE}: <js>"2011-12-03"</js>. */
90  	ISO_LOCAL_DATE,
91  
92  	/** {@link DateTimeFormatter#ISO_LOCAL_DATE_TIME}: <js>"2011-12-03T10:15:30"</js>. */
93  	ISO_LOCAL_DATE_TIME,
94  
95  	/** {@link DateTimeFormatter#ISO_LOCAL_TIME}: <js>"10:15:30"</js>. */
96  	ISO_LOCAL_TIME,
97  
98  	/** {@link DateTimeFormatter#ISO_OFFSET_DATE}: <js>"2011-12-03+01:00"</js>. */
99  	ISO_OFFSET_DATE,
100 
101 	/** {@link DateTimeFormatter#ISO_OFFSET_DATE_TIME}: <js>"2011-12-03T10:15:30+01:00"</js>. */
102 	ISO_OFFSET_DATE_TIME,
103 
104 	/** {@link DateTimeFormatter#ISO_OFFSET_TIME}: <js>"10:15:30+01:00"</js>. */
105 	ISO_OFFSET_TIME,
106 
107 	/** {@link DateTimeFormatter#ISO_ORDINAL_DATE}: <js>"2012-337"</js>. */
108 	ISO_ORDINAL_DATE,
109 
110 	/** {@link DateTimeFormatter#ISO_TIME}: <js>"10:15:30+01:00"</js> or <js>"10:15:30"</js>. */
111 	ISO_TIME,
112 
113 	/** {@link DateTimeFormatter#ISO_WEEK_DATE}: <js>"2012-W48-6"</js>. */
114 	ISO_WEEK_DATE,
115 
116 	/** Year only — pattern {@code "uuuu"}: <js>"2012"</js>. */
117 	ISO_YEAR,
118 
119 	/** Year and month — pattern {@code "uuuu-MM"}: <js>"2012-12"</js>. */
120 	ISO_YEAR_MONTH,
121 
122 	/** {@link DateTimeFormatter#ISO_ZONED_DATE_TIME}: <js>"2011-12-03T10:15:30+01:00[Europe/Paris]"</js>. */
123 	ISO_ZONED_DATE_TIME,
124 
125 	/** {@link DateTimeFormatter#RFC_1123_DATE_TIME}: <js>"Tue, 3 Jun 2008 11:05:30 GMT"</js>. */
126 	RFC_1123_DATE_TIME,

128 	/**
129 	 * Epoch milliseconds as a numeric value.
130 	 *
131 	 * <p>
132 	 * Emitted as a native int64 by binary serializers (BSON / CBOR / MsgPack) and as a bare integer
133 	 * literal by text serializers.
134 	 *
135 	 * <p>
136 	 * <b>Per-subtype semantics (midnight UTC for non-instant subtypes):</b>
137 	 * <ul>
138 	 *   <li>{@link Instant} — {@code instant.toEpochMilli()}.
139 	 *   <li>{@link OffsetDateTime}, {@link ZonedDateTime} — {@code value.toInstant().toEpochMilli()}.
140 	 *   <li>{@link LocalDateTime} — {@code ldt.toInstant(ZoneOffset.UTC).toEpochMilli()}.
141 	 *   <li>{@link LocalDate} — {@code ld.atStartOfDay(ZoneOffset.UTC).toInstant().toEpochMilli()} (midnight UTC).
142 	 *   <li>{@link YearMonth} — first day of the month at midnight UTC.
143 	 *   <li>{@link Year} — January 1 at midnight UTC.
144 	 *   <li>{@link LocalTime}, {@link OffsetTime}, {@link MonthDay} — fall back to the type's {@link #DEFAULT}
145 	 *       (ISO string form) because no defensible epoch-millis interpretation exists without an associated
146 	 *       date/year.  See {@link #isMillisNumeric(Class)} for the canonical decision predicate.
147 	 * </ul>
148 	 *
149 	 * <p>
150 	 * <b>The asymmetry is intentional:</b> a configured {@code MILLIS} setting silently produces a string
151 	 * for {@link LocalTime} / {@link OffsetTime} / {@link MonthDay} and a number for every other subtype.
152 	 */
153 	MILLIS;

157 	private static final DateTimeFormatter YEAR_FORMATTER = DateTimeFormatter.ofPattern("uuuu");
158 	private static final DateTimeFormatter YEAR_MONTH_FORMATTER = DateTimeFormatter.ofPattern("uuuu-MM");

160 	private static final Map<Class<?>,DateTimeFormatter> DEFAULT_FORMATTERS = Map.ofEntries(
161 		Map.entry(Instant.class, DateTimeFormatter.ISO_INSTANT),
162 		Map.entry(ZonedDateTime.class, DateTimeFormatter.ISO_OFFSET_DATE_TIME),
163 		Map.entry(OffsetDateTime.class, DateTimeFormatter.ISO_OFFSET_DATE_TIME),
164 		Map.entry(LocalDate.class, DateTimeFormatter.ISO_LOCAL_DATE),
165 		Map.entry(LocalDateTime.class, DateTimeFormatter.ISO_LOCAL_DATE_TIME),
166 		Map.entry(LocalTime.class, DateTimeFormatter.ISO_LOCAL_TIME),
167 		Map.entry(OffsetTime.class, DateTimeFormatter.ISO_OFFSET_TIME),
168 		Map.entry(Year.class, YEAR_FORMATTER),
169 		Map.entry(YearMonth.class, YEAR_MONTH_FORMATTER)
170 	);

174 	/**
175 	 * Returns <jk>true</jk> when {@link #MILLIS} produces a numeric wire string for the specified subtype.
176 	 *
177 	 * <p>
178 	 * Single source of truth for the {@code MILLIS} wire-type decision shared by {@link #format} and the
179 	 * marshalling-side {@code temporalSwap} / {@code temporalAccessorSwap} factories.  The helper exists so
180 	 * the swap-side {@code Long.valueOf(s)} coercion only fires when {@link #format} actually emitted a
181 	 * numeric string — a mismatch on either side is a {@link NumberFormatException} waiting to happen.
182 	 *
183 	 * <p>
184 	 * Returns <jk>false</jk> for {@link LocalTime}, {@link OffsetTime}, and {@link MonthDay} — these
185 	 * subtypes have no defensible epoch-millis interpretation, so {@link #format}'s {@code MILLIS} branch
186 	 * falls back to {@link #DEFAULT} (an ISO string) for them and the swap must keep the value as a string.
187 	 *
188 	 * @param c The {@link TemporalAccessor} subtype to query.
189 	 * @return <jk>true</jk> if {@link #format} emits a {@link Long}-parseable numeric string for {@code c}
190 	 *         under {@link #MILLIS}; <jk>false</jk> if the {@code MILLIS} branch falls back to ISO string.
191 	 */
192 	public static boolean isMillisNumeric(Class<? extends TemporalAccessor> c) {
193 		return ! (LocalTime.class.equals(c) || OffsetTime.class.equals(c) || MonthDay.class.equals(c));
194 	}

196 	/**
197 	 * Formats the specified value using this format.
198 	 *
199 	 * @param value The value to format. Can be <jk>null</jk> (returns <jk>null</jk>).
200 	 * @param zoneId The default zone id used for time-only / unzoned values.
201 	 * @return The formatted string, or <jk>null</jk> if {@code value} is <jk>null</jk>.
202 	 *         For {@link #MILLIS} this is a numeric string for every subtype except {@link LocalTime},
203 	 *         {@link OffsetTime}, and {@link MonthDay} (which fall back to their {@link #DEFAULT} ISO
204 	 *         string form).
205 	 */
206 	@SuppressWarnings({
207 		"java:S3776" // Per-subtype MILLIS coercion logic is intentionally explicit; flattening hurts readability
208 	})
209 	public String format(TemporalAccessor value, ZoneId zoneId) {
210 		if (value == null)
211 			return null;
212 		// MonthDay only implements TemporalAccessor (not Temporal) and has no DateTimeFormatter that
213 		// matches its --MM-DD wire form natively.  Use its own toString/parse pair regardless of the
214 		// configured format — every other TemporalFormat value is structurally meaningless without a year.
215 		if (value instanceof MonthDay value2)
216 			return value2.toString();
217 		if (this == MILLIS) {
218 			Long millis = toEpochMillis(value);
219 			if (millis != null)
220 				return Long.toString(millis);
221 			return DEFAULT.format(value, zoneId);
222 		}
223 		var zone = zoneId == null ? ZoneId.systemDefault() : zoneId;
224 		var tc = value.getClass();
225 		if (!DEFAULT_FORMATTERS.containsKey(tc) && (this == DEFAULT || this == NOT_SET))
226 			return DateTimeFormatter.ISO_INSTANT.format(toInstant(value, zone));
227 		var coerced = convertForFormatter(value, zone);
228 		return formatter(tc).format(coerced);
229 	}

231 	private static Instant toInstant(TemporalAccessor value, ZoneId zone) {
232 		if (value instanceof Instant value2)
233 			return value2;
234 		try {
235 			return Instant.from(value);
236 		} catch (Exception e) {
237 			if (value instanceof ChronoLocalDate value2)
238 				return value2.atTime(LocalTime.MIDNIGHT).atZone(zone).toInstant();
239 			if (value instanceof ChronoLocalDateTime<?> value2)
240 				return value2.atZone(zone).toInstant();
241 			if (value instanceof ChronoZonedDateTime<?> value2)
242 				return value2.toInstant();
243 			throw e;
244 		}
245 	}

247 	/**
248 	 * Parses the specified wire value using this format.
249 	 *
250 	 * @param <T> The temporal subtype.
251 	 * @param value The wire value. Can be <jk>null</jk> or blank (returns <jk>null</jk>).
252 	 * @param targetType The target temporal subtype.  Must be either a {@link Temporal} subtype or
253 	 *                   {@link MonthDay} (the only non-{@link Temporal} {@link TemporalAccessor} the JDK
254 	 *                   currently exposes).
255 	 * @param zoneId The default zone id used for time-only / unzoned values.
256 	 * @return The parsed value, or <jk>null</jk> if {@code value} is <jk>null</jk> or blank.
257 	 */
258 	@SuppressWarnings({
259 		"unchecked", // Type erasure requires unchecked cast on reflective from() result
260 		"java:S3776" // Per-subtype MILLIS dispatch is intentionally explicit; flattening hurts readability
261 	})
262 	public <T extends TemporalAccessor> T parse(String value, Class<T> targetType, ZoneId zoneId) {
263 		if (value == null)
264 			return null;
265 		var s = value.trim();
266 		if (s.isEmpty())
267 			return null;
268 		var zone = zoneId == null ? ZoneId.systemDefault() : zoneId;
269 		if (this == MILLIS) {
270 			// MonthDay reverses what format() emits at MILLIS — its native --MM-DD shape; numeric input
271 			// at MILLIS×MonthDay is unreachable on the round-trip path because format() emits a string.
272 			if (MonthDay.class.equals(targetType))
273 				return (T) MonthDay.parse(s);
274 			// LocalTime never had a defensible epoch-millis interpretation; legacy callers get DEFAULT.parse.
275 			if (LocalTime.class.equals(targetType))
276 				return DEFAULT.parse(s, targetType, zone);
277 			// OffsetTime: format() falls back to DEFAULT (ISO_OFFSET_TIME string).  Be lenient on parse —
278 			// route numeric input through fromEpochMillis (legacy lenient support) and route the
279 			// ISO_OFFSET_TIME wire form back through DEFAULT.parse so the round-trip closes.
280 			if (OffsetTime.class.equals(targetType) && ! isAllDigits(s))
281 				return DEFAULT.parse(s, targetType, zone);
282 			return fromEpochMillis(Long.parseLong(s), targetType, zone);
283 		}
284 		// MonthDay's --MM-DD wire shape is its only stable representation; route to its native parser
285 		// regardless of the configured format.
286 		if (MonthDay.class.equals(targetType))
287 			return (T) MonthDay.parse(s);
288 		// Lenient: epoch-millis numeric strings are accepted for DEFAULT/NOT_SET on instant-like subtypes only.
289 		// Date-only subtypes (Year, YearMonth, LocalDate, MonthDay, LocalTime) use ISO patterns that are themselves
290 		// numeric ("2012", "2012-12") and must not be reinterpreted as epoch millis.
291 		if ((this == DEFAULT || this == NOT_SET) && isAllDigits(s) && isInstantLike(targetType))
292 			return fromEpochMillis(Long.parseLong(s), targetType, zone);
293 		var offset = (targetType == Instant.class) ? Z : zone;
294 		var ta = new DefaultingTemporalAccessor(formatter(targetType).parse(s), offset);
295 		return (T) invokeFrom(targetType, ta);
296 	}

298 	private static boolean isInstantLike(Class<?> targetType) {
299 		return Instant.class.equals(targetType)
300 			|| ZonedDateTime.class.equals(targetType)
301 			|| OffsetDateTime.class.equals(targetType)
302 			|| LocalDateTime.class.equals(targetType);
303 	}

305 	private static boolean isAllDigits(String s) {
306 		var n = s.length();
307 		if (n == 0)
308 			return false;
309 		var i = 0;
310 		if (s.charAt(0) == '-' || s.charAt(0) == '+') {
311 			if (n == 1)
312 				return false;
313 			i = 1;
314 		}
315 		for (; i < n; i++)
316 			if (!Character.isDigit(s.charAt(i)))
317 				return false;
318 		return true;
319 	}

330 	private static DateTimeFormatter defaultFormatter(Class<?> tc) {
331 		var f = DEFAULT_FORMATTERS.get(tc);
332 		return or(f, DateTimeFormatter.ISO_INSTANT);
333 	}

335 	@SuppressWarnings({
336 		"java:S1301" // switch over enum: explicit case constants keep the per-format mapping legible
337 	})
338 	private DateTimeFormatter formatter(Class<?> tc) {
339 		return switch (this) {
340 			case NOT_SET, DEFAULT -> defaultFormatter(tc);
341 			case BASIC_ISO_DATE -> DateTimeFormatter.BASIC_ISO_DATE;
342 			case ISO_DATE -> DateTimeFormatter.ISO_DATE;
343 			case ISO_DATE_TIME -> DateTimeFormatter.ISO_DATE_TIME;
344 			case ISO_INSTANT -> DateTimeFormatter.ISO_INSTANT;
345 			case ISO_LOCAL_DATE -> DateTimeFormatter.ISO_LOCAL_DATE;
346 			case ISO_LOCAL_DATE_TIME -> DateTimeFormatter.ISO_LOCAL_DATE_TIME;
347 			case ISO_LOCAL_TIME -> DateTimeFormatter.ISO_LOCAL_TIME;
348 			case ISO_OFFSET_DATE -> DateTimeFormatter.ISO_OFFSET_DATE;
349 			case ISO_OFFSET_DATE_TIME -> DateTimeFormatter.ISO_OFFSET_DATE_TIME;
350 			case ISO_OFFSET_TIME -> DateTimeFormatter.ISO_OFFSET_TIME;
351 			case ISO_ORDINAL_DATE -> DateTimeFormatter.ISO_ORDINAL_DATE;
352 			case ISO_TIME -> DateTimeFormatter.ISO_TIME;
353 			case ISO_WEEK_DATE -> DateTimeFormatter.ISO_WEEK_DATE;
354 			case ISO_YEAR -> YEAR_FORMATTER;
355 			case ISO_YEAR_MONTH -> YEAR_MONTH_FORMATTER;
356 			case ISO_ZONED_DATE_TIME -> DateTimeFormatter.ISO_ZONED_DATE_TIME;
357 			case RFC_1123_DATE_TIME -> DateTimeFormatter.RFC_1123_DATE_TIME;
358 			case MILLIS -> throw new IllegalStateException("MILLIS does not use a DateTimeFormatter");
359 		};
360 	}

363 	 * Returns whether the configured formatter accepts a missing zone, mirroring the historical
364 	 * {@code TemporalSwap.zoneOptional()} flag.
365 	 */
366 	@SuppressWarnings({
367 		"java:S6541" // exhaustive switch keeps the per-format zone-optional table explicit
368 	})
369 	private boolean zoneOptional() {
370 		return switch (this) {
371 			case BASIC_ISO_DATE, ISO_DATE, ISO_DATE_TIME, ISO_LOCAL_DATE, ISO_LOCAL_DATE_TIME, ISO_LOCAL_TIME,
372 				ISO_ORDINAL_DATE, ISO_TIME, ISO_WEEK_DATE, ISO_YEAR, ISO_YEAR_MONTH -> true;
373 			default -> false;
374 		};
375 	}

377 	@SuppressWarnings({
378 		"java:S3776" // Per-subtype coercion table mirrors the legacy TemporalSwap.convertToSerializable path
379 	})
380 	private Temporal convertForFormatter(TemporalAccessor value, ZoneId zoneId) {
381 		if (value instanceof Temporal value2 && (this == DEFAULT || this == NOT_SET))
382 			return value2;
383 		var tc = value.getClass();
384 		// Instant always serializes in GMT.
385 		if (tc == Instant.class)
386 			return ZonedDateTime.from(new DefaultingTemporalAccessor(value, Z));
387 		// Zoned / offset variants are already serializable by any pattern.
388 		if (value instanceof Temporal value2 && (tc == ZonedDateTime.class || tc == OffsetDateTime.class))
389 			return value2;
390 		if (zoneOptional()) {
391 			if (value instanceof LocalDateTime value2)
392 				return value2;
393 			if (tc == OffsetTime.class)
394 				return ZonedDateTime.from(new DefaultingTemporalAccessor(value, zoneId));
395 			return LocalDateTime.from(new DefaultingTemporalAccessor(value, zoneId));
396 		}
397 		return ZonedDateTime.from(new DefaultingTemporalAccessor(value, zoneId));
398 	}

400 	@SuppressWarnings({
401 		"java:S3776" // Per-subtype epoch-millis coercion is intentionally explicit
402 	})
403 	private static Long toEpochMillis(TemporalAccessor value) {
404 		if (value instanceof Instant value2)
405 			return value2.toEpochMilli();
406 		if (value instanceof OffsetDateTime value2)
407 			return value2.toInstant().toEpochMilli();
408 		if (value instanceof ZonedDateTime value2)
409 			return value2.toInstant().toEpochMilli();
410 		if (value instanceof LocalDateTime value2)
411 			return value2.toInstant(ZoneOffset.UTC).toEpochMilli();
412 		if (value instanceof LocalDate value2)
413 			return value2.atStartOfDay(ZoneOffset.UTC).toInstant().toEpochMilli();
414 		if (value instanceof YearMonth value2)
415 			return value2.atDay(1).atStartOfDay(ZoneOffset.UTC).toInstant().toEpochMilli();
416 		if (value instanceof Year value2)
417 			return value2.atDay(1).atStartOfDay(ZoneOffset.UTC).toInstant().toEpochMilli();
418 		return null;
419 	}

421 	@SuppressWarnings({
422 		"unchecked" // Type erasure requires unchecked casts for per-subtype factory dispatch
423 	})
424 	private static <T extends TemporalAccessor> T fromEpochMillis(long millis, Class<T> targetType, ZoneId zoneId) {
425 		var instant = Instant.ofEpochMilli(millis);
426 		if (targetType == Instant.class) return (T) instant;
427 		if (targetType == ZonedDateTime.class) return (T) instant.atZone(zoneId);
428 		if (targetType == OffsetDateTime.class) return (T) instant.atZone(zoneId).toOffsetDateTime();
429 		if (targetType == LocalDateTime.class) return (T) instant.atZone(ZoneOffset.UTC).toLocalDateTime();
430 		if (targetType == LocalDate.class) return (T) instant.atZone(ZoneOffset.UTC).toLocalDate();
431 		if (targetType == OffsetTime.class) return (T) instant.atZone(zoneId).toOffsetDateTime().toOffsetTime();
432 		if (targetType == YearMonth.class) return (T) YearMonth.from(instant.atZone(ZoneOffset.UTC));
433 		if (targetType == Year.class) return (T) Year.from(instant.atZone(ZoneOffset.UTC));
434 		throw iaex("Cannot convert epoch millis to %s", targetType.getName());
435 	}

437 	private static Method findFromMethod(Class<? extends TemporalAccessor> tc) {
438 		// @formatter:off
439 		return FROM_METHODS.computeIfAbsent(tc, k -> info(tc).getPublicMethod(
440 			x -> x.isStatic()
441 				&& x.isNotDeprecated()
442 				&& x.hasName("from")
443 				&& x.hasReturnType(tc)
444 				&& x.hasParameterTypes(TemporalAccessor.class)
445 			)
446 			.map(MethodInfo::inner)
447 			.orElseThrow(() -> iaex("No static from(TemporalAccessor) on %s", tc.getName())));
448 		// @formatter:on
449 	}
450 
451 	private static Object invokeFrom(Class<? extends TemporalAccessor> tc, TemporalAccessor ta) {
452 		try {
453 			return findFromMethod(tc).invoke(null, ta);
454 		} catch (IllegalAccessException | InvocationTargetException e) {
455 			throw toRex(e);
456 		}
457 	}
