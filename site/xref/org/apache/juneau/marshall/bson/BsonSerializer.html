34  /**
35   * Serializes POJO models to BSON (Binary JSON).
36   *
37   * <h5 class='section'>Media types:</h5>
38   *
39   * Handles <c>Accept</c> types:  <bc>application/bson</bc>
40   * <p>
41   * Produces <c>Content-Type</c> types: <bc>application/bson</bc>
42   *
43   * <h5 class='topic'>Description</h5>
44   * <p>
45   * Java-to-BSON type mapping:
46   * <ul class='spaced-list'>
47   * 	<li>Maps and beans → BSON document (type 0x03)
48   * 	<li>Collections and arrays → BSON array (type 0x04, document with numeric string keys)
49   * 	<li>{@link String} / {@link Enum} → BSON string (type 0x02)
50   * 	<li>{@link Boolean} → BSON boolean (type 0x08)
51   * 	<li>{@code int} / {@code short} / {@code byte} → BSON int32 (type 0x10)
52   * 	<li>{@code long} → BSON int64 (type 0x12)
53   * 	<li>{@code float} / {@code double} → BSON double (type 0x01)
54   * 	<li>{@link Date} / {@link Calendar} / {@link Instant} → BSON datetime (type 0x09) when
55   * 		{@code writeDatesAsDatetime=true}; otherwise ISO 8601 string
56   * 	<li>{@link Duration} → ISO 8601 duration string
57   * 	<li>{@code byte[]} → BSON binary (type 0x05, subtype 0x00)
58   * 	<li>{@link BigDecimal} / {@link BigInteger} → BSON decimal128 (type 0x13)
59   * 	<li>{@code null} → BSON null (type 0x0A)
60   * 	<li>All other types → fallback to {@code toString()} as string
61   * </ul>
62   *
63   * <h5 class='section'>Notes:</h5><ul>
64   * 	<li class='note'>This class is thread safe and reusable.
65   * </ul>
66   *
67   * <h5 class='section'>Limitations:</h5><ul>
68   * 	<li>BSON requires a top-level document. Standalone scalars, collections, and arrays are wrapped as <c>{value: x}</c>.
69   * 	<li>Map keys must be strings. Non-string keys use <c>toString()</c>. Null keys use a configurable placeholder.
70   * 	<li>ObjectId (0x07) is not generated; parsing returns 24-character hex strings.
71   * 	<li>Deprecated types (Undefined, DBPointer, Symbol, JS with Scope) are skipped during parsing.
72   * 	<li>MongoDB-internal types (JavaScript, Timestamp, MinKey, MaxKey) are not supported.
73   * 	<li>{@link BigDecimal} / {@link BigInteger} values outside the Decimal128 range (max 34 significant digits,
74   * 		exponent -6176..6111) cannot be represented and throw a {@link SerializeException} with a clear message.
75   * 	<li>BSON datetime (0x09) is millisecond-resolution, so sub-millisecond (nanosecond) precision of
76   * 		{@link Instant}/{@link OffsetDateTime}/{@link ZonedDateTime} is truncated when
77   * 		{@code writeDatesAsDatetime=true}. Use {@code writeDatesAsDatetime=false} for lossless ISO-8601 string output.
78   * 	<li>Local {@code java.time} types ({@link LocalDate}, {@link LocalDateTime}, {@link LocalTime}, {@link Year},
79   * 		etc.) have no instant, so they are always written as ISO-8601 strings even when {@code writeDatesAsDatetime=true}.
80   * 	<li>BSON is binary; <c>SpacedHex</c> and <c>Base64</c> subclasses provide text-encodable output.
81   * </ul>
82   *
83   * <h5 class='section'>Example:</h5>
84   * <p class='bjava'>
85   * 	<jc>// Serialize a POJO to BSON</jc>
86   * 	byte[] <jv>bson</jv> = BsonSerializer.<jsf>DEFAULT</jsf>.write(<jv>someObject</jv>);
87   *
88   * 	<jc>// Create a serializer that writes dates as ISO 8601 strings instead of BSON datetime</jc>
89   * 	BsonSerializer <jv>s</jv> = BsonSerializer.<jsm>create</jsm>().writeDatesAsDatetime(<jk>false</jk>).build();
90   *
91   * 	<jc>// Or use the marshaller for convenience</jc>
92   * 	<jv>bson</jv> = Bson.<jsm>of</jsm>(<jv>someObject</jv>);
93   * </p>
94   *
95   * <h5 class='section'>See Also:</h5><ul>
96   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/Bson">BSON Basics</a>
97   * </ul>
98   */

136 		private boolean addBeanTypesBson;
137 		private String nullKeyString = DEFAULT_NULL_KEY;
138 		private boolean writeDatesAsDatetime = true;

140 		/**
141 		 * Constructor, default settings.

161 			writeDatesAsDatetime = copyFrom.writeDatesAsDatetime;
162 		}
163 
164 		/**
165 		 * Copy constructor.

167 		 * @param copyFrom The bean to copy from.
168 		 * 	<br>Cannot be <jk>null</jk>.

172 			addBeanTypesBson = copyFrom.addBeanTypesBson;
173 			nullKeyString = copyFrom.nullKeyString;
174 			writeDatesAsDatetime = copyFrom.writeDatesAsDatetime;
175 		}

177 		/**
178 		 * Add <js>"_type"</js> properties when needed.
179 		 *
180 		 * @param value The value for this setting.
181 		 * @return This object.
182 		 */
183 		public Builder addBeanTypesBson(boolean value) {
184 			addBeanTypesBson = value;
185 			return this;
186 		}
187 
188 		/**
189 		 * The string used to represent <jk>null</jk> map keys in BSON documents.
190 		 *
191 		 * <p>
192 		 * BSON document field names cannot be <jk>null</jk>. When serializing a {@link Map} with a <jk>null</jk> key,
193 		 * this string is used as the field name. When parsing, a field with this name is stored with <jk>null</jk>
194 		 * as the key.
195 		 *
196 		 * @param value The placeholder string for null keys. Default is <js>"&lt;NULL&gt;"</js>.
197 		 * @return This object.
198 		 */
199 		public Builder nullKeyString(String value) {
200 			nullKeyString = value == null ? DEFAULT_NULL_KEY : value;
201 			return this;
202 		}
203 
204 		/**
205 		 * Use BSON datetime type for dates. When <jk>false</jk>, dates are serialized as ISO-8601 strings.
206 		 *
207 		 * @param value The value for this setting.
208 		 * @return This object.
209 		 */
210 		public Builder writeDatesAsDatetime(boolean value) {
211 			writeDatesAsDatetime = value;
212 			return this;
213 		}
