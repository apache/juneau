92  	private static BsonOutputStream getBsonOutputStream(SerializerPipe out) throws IOException {
93  		Object output = out.getRawOutput();
94  		if (output instanceof BsonOutputStream output2)
95  			return output2;
96  		var os = new BsonOutputStream(out.getOutputStream());
97  		out.setOutputStream(os);
98  		return os;
99  	}

152 	private void writeElement(BsonOutputStream out, String name, Object o, ClassMeta<?> eType, BeanPropertyMeta pMeta) throws SerializeException {
153 		if (o == null) {
154 			out.writeElement(NULL.value, name);
155 			return;
156 		}
157 		if (eType == null)
158 			eType = object();
159 
160 		var aType = push2(name, o, eType);
161 		if (aType == null) {
162 			out.writeElement(NULL.value, name);
163 			pop();
164 			return;
165 		}
166 
167 		if (isOptional(aType)) {
168 			o = getOptionalValue(o);
169 			eType = getOptionalType(eType);
170 			aType = getClassMetaForObject(o, object());
171 		}
172 
173 		var sType = aType;
174 		var swap = aType.getSwap(this);
175 		if (nn(swap)) {
176 			o = swap(swap, o);
177 			sType = swap.getSwapClassMeta(this);
178 			if (sType.isObject())
179 				sType = getClassMetaForObject(o);
180 		}
181 
182 		if (o == null || (sType.isChar() && ((Character)o).charValue() == 0)) {
183 			out.writeElement(NULL.value, name);
184 		} else if (sType.isBean()) {
185 			out.writeElement(DOCUMENT.value, name);
186 			var child = out.createChild();
187 			child.startDocument();
188 			writeBeanMap(child, toBeanMap(o), getBeanTypeName(this, eType, aType, pMeta));
189 			out.writeChildDocument(child);
190 		} else if (sType.isMap()) {
191 			if (sType.isBeanMap()) {
192 				out.writeElement(DOCUMENT.value, name);
193 				var child = out.createChild();
194 				child.startDocument();
195 				writeBeanMap(child, (BeanMap)o, getBeanTypeName(this, eType, aType, pMeta));
196 				out.writeChildDocument(child);
197 			} else {
198 				out.writeElement(DOCUMENT.value, name);
199 				var child = out.createChild();
200 				child.startDocument();
201 				writeMap(child, (Map)o, eType);
202 				out.writeChildDocument(child);
203 			}
204 		} else if (sType.isByteArray()) {
205 			out.writeElement(BINARY.value, name);
206 			out.writeBinary((byte[])o);
207 		} else if (sType.isCollection() || sType.isArray()) {
208 			out.writeElement(ARRAY.value, name);
209 			var child = out.createChild();
210 			child.startDocument();
211 			writeArray(child, sType.isArray() ? toList(sType.inner(), o) : (Collection)o, eType);
212 			out.writeChildDocument(child);
213 		} else if (sType.isCharSequence() || sType.isChar() || sType.isEnum()) {
214 			out.writeElement(STRING.value, name);
215 			out.writeString(trim(o));
216 		} else if (sType.isBoolean()) {
217 			out.writeElement(BOOLEAN.value, name);
218 			out.writeBoolean((Boolean)o);
219 		} else if (sType.isNumber()) {
220 			if (sType.inner() == int.class || sType.inner() == short.class || sType.inner() == byte.class
221 				|| sType.inner() == Integer.class || sType.inner() == Short.class || sType.inner() == Byte.class) {
222 				out.writeElement(INT32.value, name);
223 				out.writeInt32(((Number)o).intValue());
224 			} else if (sType.inner() == long.class || sType.inner() == Long.class) {
225 				out.writeElement(INT64.value, name);
226 				out.writeInt64(((Number)o).longValue());
227 			} else if (sType.inner() == BigDecimal.class) {
228 				out.writeElement(DECIMAL128.value, name);
229 				out.writeDecimal128((BigDecimal)o);
230 			} else if (sType.inner() == BigInteger.class) {
231 				out.writeElement(DECIMAL128.value, name);
232 				out.writeDecimal128(new BigDecimal((BigInteger)o));
233 			} else {
234 				out.writeElement(DOUBLE.value, name);
235 				out.writeDouble(((Number)o).doubleValue());
236 			}
237 		} else if (sType.isUri() || (nn(pMeta) && pMeta.isUri())) {
238 			out.writeElement(STRING.value, name);
239 			out.writeString(resolveUri(o.toString()));
240 		} else if (sType.isDate()) {
241 			if (ctx.writeDatesAsDatetime) {
242 				out.writeElement(DATETIME.value, name);
243 				out.writeDateTime(toEpochMillis(o));
244 			} else {
245 				out.writeElement(STRING.value, name);
246 				out.writeString(writeDate((Date)o, sType));
247 			}
248 		} else if (sType.isCalendar()) {
249 			if (ctx.writeDatesAsDatetime) {
250 				out.writeElement(DATETIME.value, name);
251 				out.writeDateTime(toEpochMillis(o));
252 			} else {
253 				out.writeElement(STRING.value, name);
254 				out.writeString(writeCalendar(o, sType));
255 			}
256 		} else if (sType.isTemporal()) {
257 			// BSON datetime (0x09) is int64 millis-since-epoch, so only instant-bearing temporals can use it.
258 			// Local temporals (LocalDate/LocalDateTime/LocalTime/Year/...) cannot supply an instant; fall back to
259 			// the ISO-string path instead of crashing in Instant.from(...) (matches CBOR/MsgPack temporal handling).
260 			if (ctx.writeDatesAsDatetime && isInstantCapable(o)) {
261 				out.writeElement(DATETIME.value, name);
262 				out.writeDateTime(toEpochMillis(o));
263 			} else {
264 				out.writeElement(STRING.value, name);
265 				out.writeString(writeTemporal((TemporalAccessor)o, sType));
266 			}
267 		} else if (sType.isDuration()) {
268 			writeDuration(out, name, (Duration)o);
269 		} else if (sType.isPeriod()) {
270 			out.writeElement(STRING.value, name);
271 			out.writeString(writePeriod((Period)o));
272 		} else if (sType.isStreamable()) {
273 			out.writeElement(ARRAY.value, name);
274 			var child = out.createChild();
275 			child.startDocument();
276 			writeArray(child, toListFromStreamable(o, sType), eType);
277 			out.writeChildDocument(child);
278 		} else {
279 			out.writeElement(STRING.value, name);
280 			out.writeString(toString(o));
281 		}
282 		pop();
283 	}

285 	private void writeBeanMap(BsonOutputStream out, BeanMap<?> m, String typeName) throws SerializeException {
286 		Predicate<Object> checkNull = x -> isKeepNullProperties() || nn(x);
287 		var values = new ArrayList<BeanPropertyValue>();
288 
289 		if (nn(typeName)) {
290 			var pm = m.getMeta().getTypeProperty();
291 			if (pm != null)
292 				values.add(new BeanPropertyValue(pm, pm.getName(), typeName, null));
293 		}
294 
295 		m.forEachValue(checkNull, (pMeta, key, value, thrown) -> {
296 			if (nn(thrown)) {
297 				onBeanGetterException(pMeta, thrown);
298 				return;
299 			}
300 			var p = new BeanPropertyValue(pMeta, key, value, null);
301 			if (!isKeepNullProperties() && willRecurse(p))
302 				return;
303 			values.add(p);
304 		});
305 
306 		for (var x : values) {
307 			var pMeta = x.getMeta();
308 			if (pMeta.canRead())
309 				writeElement(out, x.getName(), x.getValue(), (ClassMeta<?>) x.getBeanInfo(), pMeta);
310 		}
311 	}

313 	@SuppressWarnings({
314 		"unchecked" // Cast from raw type is safe by checked conditional above.
315 	})
316 	private void writeArray(BsonOutputStream out, Collection c, ClassMeta<?> type) throws SerializeException {
317 		var elementType = type.getElementType();
318 		c = sort(c);
319 		var idx = 0;
320 		for (var x : c)
321 			writeElement(out, String.valueOf(idx++), x, elementType, null);
322 	}

324 	@SuppressWarnings({
325 		"unchecked" // Cast from raw type is safe by checked conditional above.
326 	})
327 	private void writeMap(BsonOutputStream out, Map m, ClassMeta<?> type) throws SerializeException {
328 		var valueType = type.getValueType();
329 		var keyType = type.getKeyType();
330 		m = sort(m);
331 		for (Map.Entry<?,?> entry : ((Map<?,?>)m).entrySet()) {
332 			var key = generalize(entry.getKey(), keyType);
333 			var name = (key == null) ? ctx.nullKeyString : trim(key);
334 			writeElement(out, name, entry.getValue(), valueType, null);
335 		}
336 	}

338 	/**
339 	 * Returns whether the given value can supply an instant for BSON datetime (0x09) encoding.
340 	 *
341 	 * <p>
342 	 * Only instant-bearing temporals (such as {@link Instant}, {@link OffsetDateTime}, and {@link ZonedDateTime})
343 	 * support {@link ChronoField#INSTANT_SECONDS}.  Local temporals ({@link LocalDate}, {@link LocalDateTime},
344 	 * {@link LocalTime}, {@link Year}, {@link YearMonth}, {@link MonthDay}) cannot, so they must fall back to the
345 	 * ISO-string path rather than crashing in {@link Instant#from(TemporalAccessor)}.
346 	 *
347 	 * @param o The value being serialized.
348 	 * @return <jk>true</jk> if the value is an instant-bearing temporal.
349 	 */
350 	private static boolean isInstantCapable(Object o) {
351 		return o instanceof TemporalAccessor o2 && o2.isSupported(ChronoField.INSTANT_SECONDS);
352 	}

354 	private static long toEpochMillis(Object o) {
355 		if (o instanceof Date o2)
356 			return o2.getTime();
357 		if (o instanceof Calendar o2)
358 			return o2.getTimeInMillis();
359 		if (o instanceof Instant o2)
360 			return o2.toEpochMilli();
361 		if (o instanceof Temporal o2)
362 			return Instant.from(o2).toEpochMilli();
363 		return 0L;
364 	}

366 	private boolean willRecurse(BeanPropertyValue v) throws SerializeException {
367 		var aType = push2(v.getName(), v.getValue(), (ClassMeta<?>) v.getBeanInfo());
368 		if (nn(aType))
369 			pop();
370 		return aType == null;
371 	}

373 	@Override
374 	protected void doWrite(SerializerPipe out, Object o) throws IOException, SerializeException {
375 		var bsonOut = getBsonOutputStream(out);
376 		bsonOut.startDocument();
377 
378 		var eType = getExpectedRootType(o);
379 		var aType = push2("root", o, eType);
380 		ClassMeta<?> sType = aType;
381 		if (nn(aType)) {
382 			var swap = aType.getSwap(this);
383 			if (nn(swap)) {
384 				o = swap(swap, o);
385 				sType = swap.getSwapClassMeta(this);
386 			}
387 		}
388 
389 		if (sType != null && sType.isBean()) {
390 			pop();
391 			writeBeanMap(bsonOut, toBeanMap(o), getBeanTypeName(this, eType, sType, null));
392 		} else if (sType != null && sType.isMap() && !sType.isBeanMap()) {
393 			pop();
394 			writeMap(bsonOut, (Map)o, eType);
395 		} else {
396 			pop();
397 			writeElement(bsonOut, "value", o, eType, null);
398 		}
399 
400 		bsonOut.writeDocumentTo(out.getOutputStream());
401 	}

403 	private void writeDuration(BsonOutputStream out, String name, Duration value) {
404 		var f = getDurationFormat();
405 		var s = writeDuration(value);
406 		if (f == DurationFormat.NANOS || f == DurationFormat.MILLIS) {
407 			out.writeElement(INT64.value, name);
408 			out.writeInt64(Long.parseLong(s));
409 		} else if (f == DurationFormat.SECONDS) {
410 			out.writeElement(DOUBLE.value, name);
411 			out.writeDouble(Double.parseDouble(s));
412 		} else {
413 			out.writeElement(STRING.value, name);
414 			out.writeString(s);
415 		}
416 	}
