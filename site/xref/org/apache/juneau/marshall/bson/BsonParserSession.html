105 	private String trimKey(String name) {
106 		return isTrimStrings() && nn(name) ? name.trim() : name; // HTT: name is always non-null here -- every call site passes BsonInputStream.readElementName()/readCString(), which never returns null (throws on premature EOF instead)
107 	}

109 	private Object readTypedValue(BsonInputStream is, int elementType, ClassMeta<?> targetType, Object outer, BeanPropertyMeta pMeta) throws IOException, ParseException, ExecutableException {
110 		Object o = switch (elementType) {
111 			case 0x01 -> is.readDouble();
112 			case 0x02 -> trim(is.readString());
113 			case 0x03 -> readDocument(is, targetType, outer, pMeta);
114 			case 0x04 -> readArray(is, targetType, outer, pMeta);
115 			case 0x05 -> is.readBinary();
116 			case 0x07 -> is.readObjectId();
117 			case 0x08 -> is.readBoolean();
118 			case 0x09 -> is.readDateTime();
119 			case 0x0A -> null;
120 			case 0x10 -> is.readInt32();
121 			case 0x12 -> is.readInt64();
122 			case 0x13 -> is.readDecimal128();
123 			default -> {
124 				is.skipValue(elementType);
125 				yield null;
126 			}
127 		};
128 		if (nn(parentBean()) && nn(o))
129 			setParent(targetType, o, parentBean());
130 		return o;
131 	}

133 	private <T> T readDocument(BsonInputStream is, ClassMeta<?> eType, Object outer, BeanPropertyMeta pMeta) throws IOException, ParseException, ExecutableException {
134 		is.readDocumentSize();
135 		// eType==null is provably unreachable -- readDocument's only two call sites are readTypedValue's
136 		// case 0x03 (targetType is always one of: sType.getValueType()/bcm/object()/string(), none of which are
137 		// ever null) and doRead's top-level `type` (which is guaranteed non-null by ParserSession.readInner's
138 		// type.isVoid() call, executed before doRead is ever reached).
139 		if (eType == null) // HTT: dead -- see comment above; targetType/type are never null at either call site
140 			eType = object(); // HTT: dead -- unreachable body; see the HTT note on the guard immediately above.
141 
142 		var swap = (ObjectSwap<T,Object>)eType.getSwap(this);
143 		var builder = (BuilderSwap<T,Object>)eType.getBuilderSwap(this);
144 		ClassMeta<?> sType;
145 		if (nn(builder))
146 			sType = builder.getBuilderClassMeta(this);
147 		else if (nn(swap))
148 			sType = swap.getSwapClassMeta(this);
149 		else
150 			sType = eType;
151 
152 		// Do not recurse for Optional here - BSON wraps non-bean/map roots as {"value":x}.
153 		// We must parse the document first, then unwrap via the map.size()==1 / "value" path below.
154 		var wrapInOptional = eType.isOptional();
155 		if (sType.isOptional())
156 			sType = eType.getElementType();
157 
158 		setCurrentClass(sType);
159 
160 		Object result;
161 		// Skip map path for Optional - we need fallback to unwrap {"value":x}
162 		if (!eType.isOptional() && sType.isMap()) {
163 			var map = sType.canCreateNewInstance(outer) ? (Map)sType.newInstance(outer) : newGenericMap(sType);
164 			// Coerce string keys to the declared key type when non-String (e.g. Map<TestEnum,String>)
165 			// so map.get(EnumConstant) works.  Matches the JSON-family readIntoMap2 pattern.
166 			var keyType = sType.getKeyType();
167 			var coerceKeys = nn(keyType) && !keyType.isObject() && !keyType.isString();
168 			while (!is.isDocumentEnd()) {
169 				var et = is.readElementType();
170 				var name = is.readElementName();
171 				var key = name.equals(nullKeyString) ? null : trimKey(name);
172 				var value = readTypedValue(is, et, sType.getValueType(), map, pMeta);
173 				setName(sType.getValueType(), value, key);
174 				var resolvedKey = coerceKeys ? convertAttrToType(map, key, keyType) : key;
175 				map.put(resolvedKey, value);
176 			}
177 			is.readDocumentTerminator();
178 			var raw = map instanceof MarshalledMap map2 ? cast(map2, pMeta, eType) : map;
179 			// Convert JsonMap to target map type (TreeMap, LinkedHashMap, etc.) when needed
180 			if (eType.isMap() && raw instanceof Map raw2 && !eType.inner().isInstance(raw))
181 				result = convertToMemberType(null, raw2, eType);
182 			else
183 				result = raw;
184 		} else if (!eType.isOptional() && (nn(builder) || sType.canCreateNewBean(outer))) {
185 			var beanMap = builder == null ? newBeanMap(outer, sType.inner()) : toBeanMap(builder.create(this, eType));
186 			var pb = swapParentBean(beanMap.getBean(false));
187 			try {
188 				while (!is.isDocumentEnd()) {
189 					var et = is.readElementType();
190 					var name = is.readElementName();
191 					var key = trimKey(name);
192 					var bpm = beanMap.getPropertyMeta(key);
193 					Object value;
194 					if (name.equals(getBeanTypePropertyName(eType))) {
195 						value = readTypedValue(is, et, string(), null, null);
196 						if (nn(value))
197 							beanMap = applyTypeProperty(beanMap, value.toString(), eType);
198 					} else if (bpm != null) {
199 						var bcm = (ClassMeta<?>) bpm.getBeanInfo();
200 						value = readTypedValue(is, et, bcm, beanMap.getBean(false), bpm);
201 						setName(bcm, value, key);
202 						try {
203 							bpm.set(beanMap, key, value);
204 						} catch (BeanRuntimeException e) {
205 							onBeanSetterException(nn(pMeta) ? pMeta : bpm, e);
206 							throw e;
207 						}
208 					} else {
209 						onUnknownProperty(key, beanMap, readTypedValue(is, et, object(), null, null));
210 					}
211 				}
212 			} finally {
213 				swapParentBean(pb);
214 			}
215 			is.readDocumentTerminator();
216 			result = builder == null ? beanMap.getBean() : builder.build(this, beanMap.getBean(), eType);
217 		} else {
218 			// Fallback: read document elements. Handles scalar, array, Optional roots (all wrapped as {"value":x}).
219 			// Do NOT call readArray here - at root the next bytes are type+name of first element, not array doc.
220 			var map = newGenericMap();
221 			while (!is.isDocumentEnd()) {
222 				var et = is.readElementType();
223 				var name = is.readElementName();
224 				var key = name.equals(nullKeyString) ? null : trimKey(name);
225 			map.put(key, readTypedValue(is, et, BSON_VALUE_KEY.equals(name) ? eType : object(), map, null));
226 		}
227 		is.readDocumentTerminator();
228 		// Unwrap {"value":x} when target is Optional or when scalar-like (BSON root wrap convention)
229 		if (map.size() == 1 && map.containsKey(BSON_VALUE_KEY)) {
230 			var wrapped = map.get(BSON_VALUE_KEY);
231 				if (eType.isOptional() || (!sType.isMap() && !sType.isBean() && !sType.isObject())
232 					|| (wrapped == null && sType.isObject())) {
233 					result = convertToMemberType(null, wrapped, eType);
234 					if (eType.isOptional())
235 						wrapInOptional = false; // convertToMemberType already wrapped
236 				} else {
237 					var castResult = cast(map, pMeta, eType);
238 					result = (eType.isMap() && castResult instanceof Map && !eType.inner().isInstance(castResult))
239 						? convertToMemberType(null, castResult, eType) : castResult;
240 				}
241 			} else {
242 				var castResult = cast(map, pMeta, eType);
243 				result = (eType.isMap() && castResult instanceof Map && !eType.inner().isInstance(castResult))
244 					? convertToMemberType(null, castResult, eType) : castResult;
245 			}
246 		}
247 		// Final fallback: when target is Optional and we have map with "value", unwrap
248 		// HTT: dead in practice -- whenever eType.isOptional(), the block above always takes one of two paths,
249 		// neither of which can leave `result` as a raw (unwrapped) size-1 {"value":x} Map: (1) if the outer
250 		// document itself was size-1/keyed "value" (229 true), disjunct 1 at line 231 (`eType.isOptional()`)
251 		// short-circuits true unconditionally, so line 233's convertToMemberType(null, wrapped, eType) always
252 		// runs with eType itself being the Optional type -- and readTypedValue's own root-wrap plumbing at
253 		// line 225 (passing `eType`, not the element type, for the "value" key) means any nested document/array
254 		// value recurses back into this same eType==Optional logic and returns already Optional-wrapped (see
255 		// the wrapInOptional/o(result) return at line 259), so `wrapped` here is never itself a naked Map in
256 		// the size-1/"value" shape; or (2) if the outer document was NOT size-1/keyed "value" (241 else), the
257 		// resulting castResult/result is -- by the exact negation of that same condition -- provably never
258 		// size-1-and-keyed-"value" either (cast() cannot introduce a "value" key that wasn't already there,
259 		// and if it resolves a bean via "_type" the result isn't a Map at all). This line is a redundant
260 		// safety net for a shape that the two paths above can never actually produce.
261 		if (eType.isOptional() && result instanceof Map result2 && result2.size() == 1 && result2.containsKey(BSON_VALUE_KEY)) { // HTT: dead -- see comment above
262 			result = convertToMemberType(null, result2.get(BSON_VALUE_KEY), eType); // HTT: dead -- unreachable body; see the HTT note on the guard immediately above.
263 			wrapInOptional = false; // HTT: dead -- unreachable body; see the HTT note on the guard immediately above.
264 		}
265 		// Convert JsonMap to target Map type (TreeMap, LinkedHashMap, etc.) when needed
266 		if (eType.isMap() && result instanceof Map m2 && !eType.inner().isInstance(result))
267 			result = convertToMemberType(null, m2, eType);
268 		// Unswap when result is in swapped form (e.g. JsonMap for bean with Map swap, String for Class).
269 		// Skip when convertToMemberType already produced normal form (e.g. Class from String).
270 		if (nn(swap) && nn(result) && swap.isSwappedObject(result))
271 			result = unswap(swap, result, eType);
272 		return (T)(wrapInOptional ? o(result) : result);
273 	}

275 	private BeanMap<?> applyTypeProperty(BeanMap<?> beanMap, String typeName, ClassMeta<?> eType) throws ExecutableException {
276 		var cm = getBeanRegistry().getClassMeta(typeName);
277 		// eType is provably never null here -- applyTypeProperty's only call site (readDocument, above) always
278 		// passes its own eType parameter, which is itself already established non-null (see the eType==null
279 		// HTT comment at the top of readDocument).
280 		if (nn(cm) && nn(eType) && eType.isAssignableFrom(cm)) // HTT: nn(eType) is dead -- see comment above
281 			return toBeanMap(newBean(cm.inner()));
282 		return beanMap;
283 	}

289 	private Object readArray(BsonInputStream is, ClassMeta<?> eType, Object outer, BeanPropertyMeta pMeta) throws IOException, ParseException, ExecutableException {
290 		is.readDocumentSize();
291 		// eType==null is provably unreachable -- readArray's only call site is readTypedValue's case 0x04,
292 		// whose targetType argument is always one of sType.getValueType()/bcm/object()/string(), none of which
293 		// are ever null (see the identical readDocument comment above for the full justification).
294 		if (eType == null) // HTT: dead -- see comment above; targetType is never null at the only call site
295 			eType = object(); // HTT: dead -- unreachable body; see the HTT note on the guard immediately above.
296 
297 		var sType = eType;
298 		// Do not recurse for Optional - parse array elements with inner type, wrap at end
299 		if (sType.isOptional())
300 			sType = eType.getElementType();
301 
302 		// Array types (String[], int[][], etc.) must use a Collection to gather elements; toArray() converts at the end.
303 		// ClassMeta.newInstance() for array types returns an empty array, not a Collection - never use it for arrays.
304 		// Only use sType.newInstance() for actual Collection types (List, Set, etc.).
305 		Collection coll;
306 		if (sType.isArray() || sType.isArgs() || sType.inner().isArray()) // HTT: the "sType.isArgs()" and "sType.inner().isArray()" disjuncts are both dead - ClassInfo.isArray() (inherited by ClassMeta) is defined as exactly "inner.isArray()", so "sType.isArray()" and "sType.inner().isArray()" are the same computation (the third disjunct can never differ from the first); and an ARGS-category ClassMeta's inner class is always synthesized as Object[].class (see the args-array constructor), so "sType.isArgs()==true" already implies "sType.isArray()==true", meaning the first disjunct always short-circuits before the second is ever meaningfully evaluated as the deciding factor.
307 			coll = newGenericList();
308 		else {
309 			var instance = sType.canCreateNewInstance(outer) ? sType.newInstance(outer) : null;
310 			coll = (instance instanceof Collection instance2) ? instance2 : newGenericList();
311 		}
312 		var elementType = coalesce(sType.getElementType(), object());
313 
314 		while (!is.isDocumentEnd()) {
315 			var et = is.readElementType();
316 			is.readElementName();
317 			coll.add(readTypedValue(is, et, elementType, coll, null));
318 		}
319 		is.readDocumentTerminator();
320 
321 		Object result = sType.isArray() || sType.isArgs() ? toArray(sType, coll) : coll; // HTT: "sType.isArgs()" is dead here for the same reason as the "isArgs()"/"inner().isArray()" disjuncts above - an ARGS-category ClassMeta's inner class is always Object[].class, so isArgs()==true already implies isArray()==true, and the first disjunct always short-circuits first.
322 		return eType.isOptional() ? o(result) : result;
323 	}

325 	@Override
326 	protected <T> T doRead(ParserPipe pipe, ClassMeta<T> type) throws IOException, ParseException, ExecutableException {
327 		try (var is = new BsonInputStream(pipe)) {
328 			is.setMaxLength(maxLength);
329 			return (T)readDocument(is, type, getOuter(), null);
330 		}
