17  package org.apache.juneau.marshall;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import org.apache.juneau.commons.bean.*;
22  import org.apache.juneau.commons.reflect.*;
23  import org.apache.juneau.commons.svl.*;
24  
25  /**
26   * Utility classes and methods for the {@link BeanConfig @BeanConfig} annotation.
27   *
28   * <p>
29   * Houses the {@link Applier} that pushes {@link BeanConfig @BeanConfig} attributes into a
30   * {@link MarshallingContext.Builder}. The annotation itself lives in
31   * {@code juneau-commons} (in {@link BeanConfig}), but
32   * the application logic lives here in {@code juneau-marshall} because it depends on
33   * marshall-layer types ({@link AnnotationApplier}, {@link MarshallingContext} builder APIs,
34   * {@link VarResolverSession}, etc.).
35   *
36   * <p>
37   * Because {@link BeanConfig @BeanConfig} cannot carry a {@link ContextApply @ContextApply}
38   * meta-annotation (that annotation lives in {@code juneau-marshall} and cannot be referenced
39   * from {@code juneau-commons}), {@link AnnotationWorkList} special-cases the
40   * {@link BeanConfig} annotation type and instantiates this {@link Applier} directly.
41   */
42  public class BeanConfigAnnotation {
43  
44  	/**
45  	 * Prevents instantiation.
46  	 */
47  	private BeanConfigAnnotation() {}
48  
49  	/**
50  	 * Applies {@link BeanConfig} annotations to a {@link MarshallingContext.Builder}.
51  	 */
52  	public static class Applier extends AnnotationApplier<BeanConfig,MarshallingContext.Builder> {
53  
54  		/**
55  		 * Constructor.
56  		 *
57  		 * @param vr The resolver for resolving values in annotations. Must not be <jk>null</jk>.
58  		 */
59  		public Applier(VarResolverSession vr) {
60  			super(BeanConfig.class, MarshallingContext.Builder.class, vr);
61  		}
62  
63  		@Override
64  		public void apply(AnnotationInfo<BeanConfig> ai, MarshallingContext.Builder b) {
65  			BeanConfig a = ai.inner();
66  
67  			// @formatter:off
68  			string(a.beanClassVisibility()).map(Visibility::valueOf).ifPresent(b::beanClassVisibility);
69  			string(a.beanConstructorVisibility()).map(Visibility::valueOf).ifPresent(b::beanConstructorVisibility);
70  			string(a.beanFieldVisibility()).map(Visibility::valueOf).ifPresent(b::beanFieldVisibility);
71  			string(a.beanMethodVisibility()).map(Visibility::valueOf).ifPresent(b::beanMethodVisibility);
72  			bool(a.beanMapPutReturnsOldValue()).ifPresent(b::beanMapPutReturnsOldValue);
73  			bool(a.beansRequireDefaultConstructor()).ifPresent(b::beansRequireDefaultConstructor);
74  			bool(a.beansRequireSerializable()).ifPresent(b::beansRequireSerializable);
75  			bool(a.beansRequireSettersForGetters()).ifPresent(b::beansRequireSettersForGetters);
76  			bool(a.disableBeansRequireSomeProperties()).ifPresent(b::disableBeansRequireSomeProperties);
77  			bool(a.findFluentSetters()).ifPresent(b::findFluentSetters);
78  			bool(a.ignoreInvocationExceptionsOnGetters()).ifPresent(b::ignoreInvocationExceptionsOnGetters);
79  			bool(a.ignoreInvocationExceptionsOnSetters()).ifPresent(b::ignoreInvocationExceptionsOnSetters);
80  			bool(a.disableIgnoreMissingSetters()).ifPresent(b::disableIgnoreMissingSetters);
81  			bool(a.disableIgnoreTransientFields()).ifPresent(b::disableIgnoreTransientFields);
82  			bool(a.ignoreUnknownBeanProperties()).ifPresent(b::ignoreUnknownBeanProperties);
83  			bool(a.validateSchema()).ifPresent(b::validateSchema);
84  			bool(a.ignoreUnknownEnumValues()).ifPresent(b::ignoreUnknownEnumValues);
85  			bool(a.disableIgnoreUnknownNullBeanProperties()).ifPresent(b::disableIgnoreUnknownNullBeanProperties);
86  			bool(a.unsortedProperties()).ifPresent(b::unsortedProperties);
87  			bool(a.disableInterfaceProxies()).ifPresent(b::disableInterfaceProxies);
88  			bool(a.useJavaBeanIntrospector()).ifPresent(b::useJavaBeanIntrospector);
89  			classes(a.notBeanClasses()).ifPresent(b::notBeanClasses);
90  			classes(a.notBeanClasses_replace()).ifPresent(x -> { b.notBeanClasses().clear(); b.notBeanClasses(x);});
91  			type(a.propertyNamer()).ifPresent(b::propertyNamer);
92  			l(a.interfaces()).stream().map(x -> BeanTypeApplyAnnotation.create(x).value(BeanTypeAnnotation.create().interfaceClass(x).build()).build()).forEach(b::annotations);
93  			strings(a.notBeanPackages()).ifPresent(b::notBeanPackages);
94  			strings(a.notBeanPackages_replace()).ifPresent(x -> {b.notBeanPackages().clear(); b.notBeanPackages(x);});
95  			// @formatter:on
96  		}
97  	}
98  }
