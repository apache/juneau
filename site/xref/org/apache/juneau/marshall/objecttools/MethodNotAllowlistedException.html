17  package org.apache.juneau.marshall.objecttools;
18  
19  import org.apache.juneau.marshall.*;
20  
21  /**
22   * Thrown by {@link ObjectIntrospector} when an attempt is made to invoke a method that has not been
23   * explicitly allow-listed.
24   *
25   * <p>
26   * As of Juneau 10.0, {@link ObjectIntrospector} is secure-by-default: reflective method dispatch is refused
27   * unless the caller has explicitly allow-listed the methods that may be invoked.  This closes off
28   * reflective-invoke-over-the-wire exposure (e.g. a REST endpoint that feeds untrusted method/argument strings
29   * straight into an introspector).
30   *
31   * <h5 class='section'>How to fix:</h5><ul>
32   * 	<li>
33   * 		Allow-list the specific method(s) you intend to expose:
34   * 		<p class='bjava'>
35   * 			ObjectIntrospector.<jsm>create</jsm>(<jv>o</jv>).allow(MyBean.<jk>class</jk>, <js>"getName"</js>).invokeMethod(...);
36   * 		</p>
37   * 	<li>
38   * 		Or, for trusted in-process callers only, restore the pre-10.0 behavior of allowing any public,
39   * 		non-deprecated method:
40   * 		<p class='bjava'>
41   * 			ObjectIntrospector.<jsm>create</jsm>(<jv>o</jv>).allowAll().invokeMethod(...);
42   * 		</p>
43   * </ul>
44   *
45   * <h5 class='section'>See Also:</h5><ul>
46   * 	<li class='jc'>{@link ObjectIntrospector}
47   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/ObjectTools">Object Tools</a>
48   * </ul>
49   *
50   * @serial exclude
51   */
52  @SuppressWarnings({
53  	"java:S110" // Deep inheritance inherent to the exception hierarchy
54  })
55  public class MethodNotAllowlistedException extends BasicRuntimeException {
56  
57  	private static final long serialVersionUID = 1L;
58  
59  	/**
60  	 * Constructor.
61  	 *
62  	 * @param message Message.
63  	 * @param args Message arguments.
64  	 */
65  	public MethodNotAllowlistedException(String message, Object...args) {
66  		super(message, args);
67  	}
68  
69  	@Override /* Overridden from BasicRuntimeException */
70  	public MethodNotAllowlistedException setMessage(String message, Object...args) {
71  		super.setMessage(message, args);
72  		return this;
73  	}
74  }
