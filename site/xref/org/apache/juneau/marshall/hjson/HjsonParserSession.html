88  	@Override
89  	protected <T> T doRead(ParserPipe pipe, ClassMeta<T> type) throws IOException, ParseException, ExecutableException {
90  		String inputStr;
91  		try {
92  			inputStr = pipe.asString();
93  		} catch (IOException e) {
94  			throw new ParseException(e);
95  		}
96  		if (inputStr == null)
97  			return null;
98  		var trimmed = inputStr.trim();
99  		if (trimmed.isEmpty())
100 			return null;
101 		var tokenizer = new HjsonTokenizer(new StringReader(inputStr));
102 		tokenizer.skipWhitespaceAndComments();
103 		var peek = tokenizer.peek();
104 		if (peek.type() == HjsonTokenizer.TokenType.EOF)
105 			return null;
106 		var peekType = peek.type();
107 		// LBRACE/LBRACKET -> readValue (object/array)
108 		if (peekType == HjsonTokenizer.TokenType.LBRACE || peekType == HjsonTokenizer.TokenType.LBRACKET) {
109 			return (T) readValue(tokenizer, type);
110 		}
111 		// QUOTELESS/STRING: distinguish braceless root "key: value" from lone value "string"
112 		if (peekType == HjsonTokenizer.TokenType.QUOTELESS || peekType == HjsonTokenizer.TokenType.STRING) {
113 			var firstToken = tokenizer.read();
114 			var firstVal = firstToken.stringValue();
115 			tokenizer.skipWhitespaceAndComments();
116 			var nextType = tokenizer.peek().type();
117 			// Target is String/number/boolean/etc - always lone value (not key:value map)
118 			if (!type.isMap() && !type.isBean())
119 				return (T) coerceMemberValue(firstVal, type);
120 			// Target is Map/Bean: COLON means key:value (braceless root)
121 			if (nextType == HjsonTokenizer.TokenType.COLON)
122 				return (T) readRootBraceless(tokenizer, type, firstVal);
123 			// No COLON - lone value (e.g. "1.0" or "true" tokenized as QUOTELESS)
124 			return (T) coerceMemberValue(firstVal, type);
125 		}
126 		// NUMBER, TRUE, FALSE, NULL, etc - lone value; use (T) not type.cast (cast fails for boxed->primitive)
127 		return (T) coerceMemberValue(readValue(tokenizer, type), type);
128 	}

130 	private Object readRootBraceless(HjsonTokenizer t, ClassMeta<?> type, String firstKey) throws IOException, ParseException, ExecutableException {
131 		var result = newGenericMap();
132 		// Same per-property type threading as readObject so braceless-root beans get typed-map keys.
133 		var beanMeta = (type != null && type.isBean()) ? type.getBeanMeta() : null; // HTT: type == null is unreachable - doRead only calls readRootBraceless() after already confirming (via type.isMap()/type.isBean()) that type is non-null.
134 		if (t.read().type() != HjsonTokenizer.TokenType.COLON) // HTT: != COLON is unreachable - doRead's sole call site only invokes readRootBraceless() after already peeking (not consuming) this exact COLON token, so the very next read() is always that same COLON.
135 			throw new ParseException(this, "Expected : after key at line %s", t.getLine()); // HTT: unreachable - guarded by the always-false condition immediately above.
136 		t.skipWhitespaceAndComments();
137 		var value = readValue(t, propertyType(beanMeta, firstKey, type));
138 		result.put(firstKey, value);
139 		t.skipWhitespaceAndComments();
140 		if (t.peek().type() == HjsonTokenizer.TokenType.NEWLINE) // HTT: NEWLINE is unreachable here - the skipWhitespaceAndComments() call immediately above already consumes any pending '\n'/'\r' (they're treated as plain whitespace, see HjsonTokenizer#skipWhitespaceAndComments), and read()/peek() themselves call skipWhitespaceAndComments() first too, so a NEWLINE token can never survive to be peeked.
141 			t.read(); // HTT: unreachable - guarded by the always-false condition immediately above.
142 		while (t.peek().type() != HjsonTokenizer.TokenType.EOF) {
143 			var key = readKey(t);
144 			t.skipWhitespaceAndComments();
145 			if (t.read().type() != HjsonTokenizer.TokenType.COLON)
146 				throw new ParseException(this, "Expected : after key at line %s", t.getLine());
147 			t.skipWhitespaceAndComments();
148 			value = readValue(t, propertyType(beanMeta, key, type));
149 			result.put(key, value);
150 			t.skipWhitespaceAndComments();
151 			if (t.peek().type() == HjsonTokenizer.TokenType.NEWLINE) // HTT: same as the first-pair NEWLINE check above - always already consumed by the preceding skipWhitespaceAndComments().
152 				t.read(); // HTT: unreachable - guarded by the always-false condition immediately above.
153 		}
154 		return convertToBean(result, type);
155 	}

157 	private Object readValue(HjsonTokenizer t, ClassMeta<?> type) throws IOException, ParseException, ExecutableException {
158 		var tok = t.read();
159 		Object raw = switch (tok.type()) {
160 			case STRING, QUOTELESS, MULTILINE -> isTrimStrings() ? tok.stringValue().trim() : tok.stringValue();
161 			case NUMBER -> tok.numberValue();
162 			case TRUE -> true;
163 			case FALSE -> false;
164 			case NULL -> null;
165 			case LBRACE -> readObject(t, type);
166 			case LBRACKET -> readArray(t, type);
167 			case EOF -> null;
168 			default -> throw new ParseException(this, "Unexpected token %s at line %s", tok.type(), t.getLine());
169 		};
170 		if (raw == null)
171 			return null;
172 		if (tok.type() == HjsonTokenizer.TokenType.LBRACE || tok.type() == HjsonTokenizer.TokenType.LBRACKET)
173 			return raw;
174 		return coerceMemberValue(raw, type);
175 	}

177 	private Object readObject(HjsonTokenizer t, ClassMeta<?> type) throws IOException, ParseException, ExecutableException {
178 		var result = newGenericMap();
179 		// When parsing a typed Bean, look up per-property ClassMetas so that nested Map<K,V> properties
180 		// see their declared key/value types and the key coercion in the inner readObject can fire
181 		// (matches JsonParserSession.readIntoBeanMap2 / readIntoMap2 pattern).
182 		var beanMeta = (type != null && type.isBean()) ? type.getBeanMeta() : null; // HTT: type == null is unreachable - readObject is only invoked (from doRead/readValue/readRootBraceless) with the never-null type threaded from doRead's own ClassMeta<T> parameter.
183 		t.skipWhitespaceAndComments();
184 		var next = t.peek();
185 		if (next.type() == HjsonTokenizer.TokenType.RBRACE) {
186 			t.read();
187 			return convertToBean(result, type);
188 		}
189 		while (true) {
190 			var key = readKey(t);
191 			t.skipWhitespaceAndComments();
192 			if (t.read().type() != HjsonTokenizer.TokenType.COLON)
193 				throw new ParseException(this, "Expected : after key at line %s", t.getLine());
194 			t.skipWhitespaceAndComments();
195 			var valueType = propertyType(beanMeta, key, type);
196 			var value = readValue(t, valueType);
197 			result.put(key, value);
198 			t.skipWhitespaceAndComments();
199 			next = t.peek();
200 			if (next.type() == HjsonTokenizer.TokenType.RBRACE) {
201 				t.read();
202 				break;
203 			}
204 			if (next.type() == HjsonTokenizer.TokenType.COMMA || next.type() == HjsonTokenizer.TokenType.NEWLINE) { // HTT: the NEWLINE half is unreachable - the skipWhitespaceAndComments() call immediately above always consumes any pending '\n'/'\r' as plain whitespace before this peek(), so next.type() can never be NEWLINE (only the COMMA half is ever reachable).
205 				t.read();
206 				t.skipWhitespaceAndComments();
207 			}
208 		}
209 		return convertToBean(result, type);
210 	}

212 	/**
213 	 * Coerces a parsed scalar value to the expected type, honoring session-aware default swaps for
214 	 * {@code byte[]} targets at the collection-element / top-level dispatch sites.
215 	 *
216 	 * <p>For all other targets this is a transparent wrapper over {@link #convertToMemberType} so the
217 	 * existing dispatch behavior is preserved.  The narrow {@code byte[]} hook lets the configured
218 	 * {@link BinaryFormat}'s variant {@link org.apache.juneau.marshall.swaps.BinarySwap} unswap the wire-form
219 	 * string back into raw bytes — these dispatch sites don't go through
220 	 * {@code MarshalledPropertyPostProcessor}'s per-property swap install, so without this they fall
221 	 * through to {@code BasicConverter}'s default {@code String → byte[]} UTF-8 coercion.
222 	 */
223 	private Object coerceMemberValue(Object raw, ClassMeta<?> type) throws ParseException {
224 		if (raw instanceof String raw2 && type != null && type.inner() == byte[].class) { // HTT: type != null is unreachable - coerceMemberValue's sole call site (readValue) always threads a non-null type (see readValue's own callers, which never pass a null ClassMeta through this class's call graph).
225 			var swap = type.getSwap(this);
226 			if (swap != null)
227 				return unswap(swap, raw2, type);
228 		}
229 		return convertToMemberType(null, raw, type);
230 	}

232 	/**
233 	 * Resolves the {@link ClassMeta} to parse a bean-property value against, falling back to {@code object()}
234 	 * when the enclosing type isn't a bean or the key isn't a known property.  Threading the property type
235 	 * lets typed-map properties (e.g. {@code Map<TestEnum,String>}) coerce their keys via the converter's
236 	 * {@code Map → Map} path.
237 	 *
238 	 * <p>
239 	 * Also surfaces {@link ClassMeta#isCollectionOrArray()} property types so the
240 	 * collection-element dispatch in {@link #readArray} threads the parent's element type into recursion —
241 	 * that's where {@link #coerceMemberValue} reads the {@code byte[]} target and invokes the configured
242 	 * {@link org.apache.juneau.marshall.swaps.BinarySwap}'s unswap on the wire-form string.  Other property shapes
243 	 * (Optional, primitive, bean) still pass through as {@code object()} so this parser's lazy-typed
244 	 * conversion path keeps working unchanged.
245 	 */
246 	private ClassMeta<?> propertyType(BeanMeta<?> beanMeta, String key, ClassMeta<?> enclosingType) {
247 		if (beanMeta != null && key != null && !key.equals(getBeanTypePropertyName(enclosingType))) {
248 			var pMeta = beanMeta.getPropertyMeta(key);
249 			if (pMeta != null) {
250 				var cm = (ClassMeta<?>) pMeta.getBeanInfo();
251 				if (cm != null && (cm.isMap() || cm.isCollectionOrArray())) // HTT: cm == null is unreachable - getBeanInfo() only returns null for a BeanPropertyMeta built via the bean-modeling-only path (see its javadoc); every BeanMeta this parser session touches is built through the marshalling-side construction path, which always populates a non-null typeMeta.
252 					return cm;
253 			}
254 		}
255 		return object();
256 	}

258 	private String readKey(HjsonTokenizer t) throws IOException {
259 		var tok = t.read();
260 		return switch (tok.type()) {
261 			case STRING, QUOTELESS, MULTILINE -> isTrimStrings() ? tok.stringValue().trim() : tok.stringValue();
262 			case NULL -> null;  // null keyword used as key -> preserve as null (matches JSON parser behavior)
263 			case TRUE -> "true";
264 			case FALSE -> "false";
265 			case NUMBER -> tok.numberValue().toString();
266 			default -> throw new ParseException(this, "Expected key at line %s", t.getLine());
267 		};
268 	}

270 	private Object readArray(HjsonTokenizer t, ClassMeta<?> type) throws IOException, ParseException, ExecutableException {
271 		if (type == null) // HTT: type == null is unreachable - readArray's call sites (readValue's LBRACKET case, itself only ever invoked with a non-null type) always thread a non-null ClassMeta through (coalesce(...,object()) at worst).
272 			type = object(); // HTT: unreachable - guarded by the always-false condition immediately above.
273 		var result = new JsonList();
274 		t.skipWhitespaceAndComments();
275 		var next = t.peek();
276 		if (next.type() == HjsonTokenizer.TokenType.RBRACKET) {
277 			t.read();
278 			return convertToCollection(result, type);
279 		}
280 		while (true) {
281 			var eType = coalesce(type.getElementType(), object());
282 			result.add(readValue(t, eType));
283 			t.skipWhitespaceAndComments();
284 			next = t.peek();
285 			if (next.type() == HjsonTokenizer.TokenType.RBRACKET) {
286 				t.read();
287 				break;
288 			}
289 			if (next.type() == HjsonTokenizer.TokenType.COMMA || next.type() == HjsonTokenizer.TokenType.NEWLINE) { // HTT: the NEWLINE half is unreachable - the skipWhitespaceAndComments() call immediately above always consumes any pending '\n'/'\r' as plain whitespace before this peek(), so next.type() can never be NEWLINE (only the COMMA half is ever reachable).
290 				t.read();
291 				t.skipWhitespaceAndComments();
292 			}
293 		}
294 		return convertToCollection(result, type);
295 	}

297 	private Object convertToBean(MarshalledMap map, ClassMeta<?> type) throws ExecutableException, ParseException {
298 		if (type == null) // HTT: type == null is unreachable - convertToBean's call sites (readObject, readRootBraceless) always thread the never-null type parameter they themselves received.
299 			type = object(); // HTT: unreachable - guarded by the always-false condition immediately above.
300 		var casted = cast(map, null, type);
301 		if (casted != map)
302 			return casted;
303 		if (!type.isObject()) {
304 			// If target type has a swap (e.g. surrogate D1→D2), parse JsonMap as swap class first, then unswap.
305 			var swap = type.getSwap(this);
306 			if (nn(swap)) {
307 				var sType = swap.getSwapClassMeta(this);
308 				var swapped = convertToMemberType(null, map, sType);
309 				return unswap(swap, swapped, type);
310 			}
311 			var result = convertToMemberType(null, map, type);
312 			if (type.isBean() && result != null) // HTT: result == null is unreachable when type.isBean() - convertToMemberType() either successfully constructs and returns the bean, or throws (e.g. via a bean-instantiation/property-set failure); it never silently returns null for a non-null MarshalledMap input against a bean target.
313 				injectAnnotations(map, result);
314 			return result;
315 		}
316 		return map;
317 	}

319 	private void injectAnnotations(MarshalledMap map, Object bean) throws ExecutableException {
320 		var bm = toBeanMap(bean);
321 		for (var entry : map.entrySet()) {
322 			var key = entry.getKey();
323 			var pm = bm.getPropertyMeta(key);
324 			var val = (pm == null) ? null : getBeanValueSafely(bm, key);
325 			if (pm == null || val == null)
326 				continue;
327 			var cm = (ClassMeta<?>) pm.getBeanInfo();
328 			if (cm.getNameProperty() != null)
329 				setName(cm, val, key);
330 			injectParentAnnotations(cm, val, entry.getValue(), bean);
331 		}
332 	}

334 	/*
335 	 * Recursively injects @ParentProperty (and @NameProperty on map values / nested beans) into val.
336 	 * Intermediate collections and maps are transparent: a collection/map element's parent is
337 	 * parentBean (the nearest enclosing bean), skipping all intermediate containers.  This keeps HJSON
338 	 * consistent with the streaming parsers.
339 	 */
340 	private void injectParentAnnotations(ClassMeta<?> cm, Object val, Object node, Object parentBean) throws ExecutableException {
341 		if (val == null || cm == null) // HTT: cm == null is unreachable - every call site (the field loop above, and the two recursive calls below, which each null-check et/vt first) always threads a non-null ClassMeta.
342 			return;
343 		if (cm.getParentProperty() != null)
344 			setParent(cm, val, parentBean);
345 		if (cm.isCollectionOrArray()) {
346 			var et = cm.getElementType();
347 			if (et == null || et.isObject()) // HTT: et == null is unreachable - getElementType() on a collection/array-typed ClassMeta always resolves to at least Object's ClassMeta, never a null reference.
348 				return;
349 			var nodeList = node instanceof List<?> nl ? nl : null;
350 			var i = 0;
351 			for (var element : toIterable(val)) {
352 				injectParentAnnotations(et, element, nodeList != null && i < nodeList.size() ? nodeList.get(i) : null, parentBean); // HTT: "nodeList != null && i >= nodeList.size()" is unreachable - whenever nodeList is non-null it was parsed from the very same raw structure that "val" was coerced from, so val's element count can never exceed nodeList's.
353 				i++;
354 			}
355 		} else if (cm.isMap() && val instanceof Map<?,?> valMap) { // HTT: cm.isMap()==true with val not instanceof Map is unreachable - a Map-typed property either successfully coerces to a real Map instance or the parse throws before this injection pass ever runs (confirmed: feeding a shape-mismatched raw value, e.g. a List, into a Map-typed property throws during coercion rather than desyncing like the analogous collection case above).
356 			var vt = cm.getValueType();
357 			if (vt == null || vt.isObject()) // HTT: vt == null is unreachable - getValueType() on a map-typed ClassMeta always resolves to at least Object's ClassMeta, never a null reference.
358 				return;
359 			var nodeMap = node instanceof Map<?,?> nm ? nm : null; // HTT: the "node not instanceof Map" false-arm is unreachable - unlike collections (see the nodeList desync above), a Map-typed property's raw parsed node is always itself a Map when the value has already been successfully coerced to a Map here; any shape mismatch throws during coercion before this method is ever reached.
360 			for (Map.Entry<?,?> e : valMap.entrySet()) {
361 				if (vt.getNameProperty() != null)
362 					setName(vt, e.getValue(), e.getKey());
363 				injectParentAnnotations(vt, e.getValue(), nodeMap != null ? nodeMap.get(e.getKey()) : null, parentBean); // HTT: nodeMap == null is unreachable here - see the HTT note on the nodeMap assignment above.
364 			}
365 		} else if (cm.isBean() && !(val instanceof Map) && node instanceof MarshalledMap nodeMap) {
366 			injectAnnotations(nodeMap, val);
367 		}
368 	}

370 	private static Iterable<?> toIterable(Object val) {
371 		if (val instanceof Collection<?> c)
372 			return c;
373 		if (val instanceof Object[] a)
374 			return Arrays.asList(a);
375 		return List.of();
376 	}

378 	private static Object getBeanValueSafely(BeanMap<?> bm, String key) {
379 		try {
380 			return bm.get(key);
381 		} catch (@SuppressWarnings("unused") Exception e) {
382 			return null;
383 		}
384 	}

386 	private Object convertToCollection(MarshalledList list, ClassMeta<?> type) throws ExecutableException {
387 		if (type == null) // HTT: type == null is unreachable - convertToCollection's sole call site (readArray) always threads a non-null type (see readArray's own null-guard just above, itself dead for the same reason).
388 			type = object(); // HTT: unreachable - guarded by the always-false condition immediately above.
389 		var eType = coalesce(type.getElementType(), object());
390 		var converted = newGenericList();
391 		for (var item : list)
392 			converted.add(convertToMemberType(null, item, eType));
393 		// For array types, convert to the appropriate array
394 		if (type.inner() != null && type.inner().isArray()) // HTT: type.inner() == null is unreachable - every ClassMeta wraps a concrete, non-null Class (this class never constructs or receives a ClassMeta lacking an inner Class).
395 			return toArray(type, converted);
396 		// For specific collection subtypes (e.g. Vector, LinkedList), convert via MarshallingSession
397 		if (!type.isObject())
398 			return convertToMemberType(null, converted, type);
399 		return converted;
400 	}
