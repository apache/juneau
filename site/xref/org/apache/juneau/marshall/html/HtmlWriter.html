17  package org.apache.juneau.marshall.html;
18  
19  import static org.apache.juneau.commons.utils.StringUtils.*;
20  
21  import java.io.*;
22  
23  import org.apache.juneau.marshall.*;
24  import org.apache.juneau.marshall.xml.*;
25  
26  /**
27   * Specialized writer for serializing HTML.
28   *
29   * <h5 class='section'>See Also:</h5><ul>
30   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/HtmlSupport">HTML Basics</a>
31  
32   * </ul>
33   */
34  @SuppressWarnings({
35  	"resource", // Writer resource managed by calling code
36  	"java:S119" // 'SELF' (CRTP self-type) is intentional and clearer than a single-letter name.
37  })
38  public abstract class HtmlWriter<SELF extends HtmlWriter<SELF>> extends XmlWriter<SELF> {
39  
40  	/**
41  	 * Copy constructor.
42  	 *
43  	 * @param w Writer being copied.
44  	 */
45  	protected HtmlWriter(HtmlWriter<?> w) {
46  		super(w);
47  	}
48  
49  	/**
50  	 * Constructor.
51  	 *
52  	 * @param out The writer being wrapped.
53  	 * @param useWhitespace If <jk>true</jk>, tabs will be used in output.
54  	 * @param maxIndent The maximum indentation level.
55  	 * @param trimStrings If <jk>true</jk>, strings should be trimmed before they're serialized.
56  	 * @param quoteChar The quote character to use (i.e. <js>'\''</js> or <js>'"'</js>)
57  	 * @param uriResolver The URI resolver for resolving URIs to absolute or root-relative form.
58  	 */
59  	protected HtmlWriter(Writer out, boolean useWhitespace, int maxIndent, boolean trimStrings, char quoteChar, UriResolver uriResolver) {
60  		super(out, useWhitespace, maxIndent, trimStrings, quoteChar, uriResolver, false, null);
61  	}
62  
63  	@Override /* Overridden from SerializerWriter */
64  	public SELF cr(int depth) {
65  		if (depth > 0)
66  			super.cr(depth);
67  		return self();
68  	}
69  
70  	@Override /* Overridden from SerializerWriter */
71  	public SELF cre(int depth) {
72  		if (depth > 0)
73  			super.cre(depth);
74  		return self();
75  	}
76  
77  	@Override /* Overridden from XmlSerializerWriter */
78  	@SuppressWarnings({
79  		"java:S3776" // Cognitive complexity acceptable for text serialization with whitespace handling
80  	})
81  	public SELF text(Object o, boolean preserveWhitespace) {
82  
83  		if (o == null) {
84  			append("<null/>");
85  			return self();
86  		}
87  		var s = o.toString();
88  		if (s.isEmpty()) {
89  			append("<sp/>");
90  			return self();
91  		}
92  
93  		for (var i = 0; i < s.length(); i++) {
94  			var test = s.charAt(i);
95  			if (test == '&')
96  				append("&amp;");
97  			else if (test == '<')
98  				append("&lt;");
99  			else if (test == '>')
100 				append("&gt;");
101 			else if (test == '\n')
102 				append(preserveWhitespace ? "\n" : "<br/>");
103 			else if (test == '\f')  // XML 1.0 doesn't support form feeds or backslashes, so we have to invent something.
104 				append(preserveWhitespace ? "\f" : "<ff/>");
105 			else if (test == '\b')
106 				append(preserveWhitespace ? "\b" : "<bs/>");
107 			else if (test == '\t')
108 				append(preserveWhitespace ? "\t" : "<sp>&#x2003;</sp>");
109 			else if ((i == 0 || i == s.length() - 1) && Character.isWhitespace(test)) {
110 				if (preserveWhitespace)
111 					w(test);
112 				else if (test == ' ')
113 					append("<sp> </sp>");
114 				else
115 					append("<sp>&#x").append(toHex4(test)).append(";</sp>");
116 			} else if (Character.isISOControl(test))
117 				append("&#" + (int)test + ";");
118 			else
119 				w(test);
120 		}
121 
122 		return self();
123 	}
124 }
