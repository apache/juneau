35  /**
36   * A lookup table for resolving bean types by name.
37   *
38   * <p>
39   * In a nutshell, provides a simple mapping of bean class objects to identifying names.
40   *
41   * <p>
42   * Class names are defined through the {@link Marshalled#typeName() @Marshalled(typeName)} annotation.
43   *
44   * <p>
45   * The dictionary is used by the framework in the following ways:
46   * <ul>
47   * 	<li>If a class type cannot be inferred through reflection during parsing, then a helper <js>"_type"</js> is added
48   * 		to the serialized output using the name defined for that class in this dictionary.  This helps determine the
49   * 		real class at parse time.
50   * 	<li>The dictionary name is used as element names when serialized to XML.
51   * </ul>
52   *
53   */
54  @SuppressWarnings({
55  	"java:S115", // Constants use UPPER_snakeCase naming convention
56  	"java:S1452"  // Wildcard required - Class<?> for bean dictionary types
57  })
58  public class BeanRegistry implements BeanRegistryLookup {
59  
60  	// Argument name constants for assertArgNotNull
61  	private static final String ARG_bc = "bc";
62  	private static final String ARG_classes = "classes";
63  
64  	private final Map<String,ClassMeta<?>> map;
65  	private final Map<Class<?>,String> reverseMap;
66  	private final MarshallingContext bc;
67  	private final AnnotationProvider ap;
68  	private final boolean isEmpty;
69  
70  	BeanRegistry(MarshallingContext bc, BeanRegistry parent, List<ClassInfo> classes) {
71  		assertArgNotNull(ARG_bc, bc);
72  		assertArgNotNull(ARG_classes, classes);
73  		this.bc = bc;
74  		this.ap = bc.getAnnotationProvider();
75  		this.map = new ConcurrentHashMap<>();
76  		this.reverseMap = new ConcurrentHashMap<>();
77  		bc.getBeanDictionary().forEach(this::addClass);
78  		if (nn(parent))
79  			parent.map.forEach(this::addToMap);
80  		classes.forEach(this::addClass);
81  		isEmpty = map.isEmpty();
82  	}
83  
84  	/**
85  	 * Gets the class metadata for the specified bean type name.
86  	 *
87  	 * @param typeName
88  	 * 	The bean type name as defined by {@link Marshalled#typeName() @Marshalled(typeName)}.
89  	 * 	Can include multi-dimensional array type names (e.g. <js>"X^^"</js>).
90  	 * 	<br>Can be <jk>null</jk> (returns <jk>null</jk>).
91  	 * @return The class metadata for the bean, or <jk>null</jk> if the registry is empty, the type name is <jk>null</jk>, or no matching entry was found.
92  	 */
93  	public ClassMeta<?> getClassMeta(String typeName) {
94  		if (isEmpty || typeName == null)
95  			return null;
96  		var cm = map.get(typeName);
97  		if (nn(cm))
98  			return cm;
99  		if (typeName.charAt(typeName.length() - 1) == '^') {
100 			cm = getClassMeta(typeName.substring(0, typeName.length() - 1));
101 			if (nn(cm)) {
102 				cm = bc.getClassMeta(Array.newInstance(cm.inner(), 1).getClass());
103 				map.put(typeName, cm);
104 			}
105 			return cm;
106 		}
107 		return null;
108 	}
109 
110 	/**
111 	 * Given the specified class, return the dictionary name for it.
112 	 *
113 	 * @param c The class to lookup in this registry. Can be <jk>null</jk> (returns <jk>null</jk>).
114 	 * @return The dictionary name for the specified class in this registry, or <jk>null</jk> if not found.
115 	 */
116 	public String getTypeName(ClassMeta<?> c) {
117 		return isEmpty || c == null ? null : reverseMap.get(c.inner());
118 	}
119 
120 	/**
121 	 * Given the specified raw class, return the dictionary name for it.
122 	 *
123 	 * <p>
124 	 * Variant of {@link #getTypeName(ClassMeta)} that takes the raw {@link Class} directly so callers in the
125 	 * bean-modeling layer (which is being decoupled from {@link ClassMeta}) can perform a polymorphic-dispatch
126 	 * lookup without holding a {@link ClassMeta} reference.
127 	 *
128 	 * @param c The class to lookup in this registry. Can be <jk>null</jk> (returns <jk>null</jk>).
129 	 * @return The dictionary name for the specified class in this registry, or <jk>null</jk> if not found.
130 	 */
131 	@Override
132 	public String getTypeName(Class<?> c) {
133 		return isEmpty || c == null ? null : reverseMap.get(c);
134 	}
135 
136 	/**
137 	 * Returns <jk>true</jk> if this dictionary has an entry for the specified type name.
138 	 *
139 	 * @param typeName The bean type name. Can be <jk>null</jk> (returns <jk>false</jk>).
140 	 * @return <jk>true</jk> if this dictionary has an entry for the specified type name.
141 	 */
142 	@Override
143 	public boolean hasName(String typeName) {
144 		return nn(getClassMeta(typeName));
145 	}

147 	protected FluentMap<String,Object> properties() {
148 		var m = filteredBeanPropertyMap();
149 		map.forEach((k, v) -> m.a(k, v.toString(true)));
150 		return m;
151 	}
152 
153 	@Override
154 	public String toString() {
155 		return r(properties());
156 	}

158 	@SuppressWarnings({
159 		"java:S3776" // Cognitive complexity acceptable for class registration with multiple conditions
160 	})
161 	private void addClass(ClassInfo ci) {
162 		try {
163 			if (nn(ci) && nn(ci.inner())) {
164 				if (ci.isAssignableTo(Collection.class)) {
165 					Collection<?> cc = BeanInstantiator.of(Collection.class).type(ci).preferZeroArgConstructor().run();
166 					cc.forEach(x -> {
167 						if (x instanceof Class<?> x2)
168 							addClass(info(x2));
169 						else
170 							throw brex("Collection class '%s' passed to BeanRegistry does not contain Class objects.", ci.getName());
171 					});
172 				} else if (ci.isAssignableTo(Map.class)) {
173 					Map<?,?> m = BeanInstantiator.of(Map.class).type(ci).preferZeroArgConstructor().run();
174 					m.forEach((k, v) -> {
175 						var typeName = s(k);
176 						ClassMeta<?> val = null;
177 						if (v instanceof Type v2)
178 							val = bc.getClassMeta(v2);
179 						else if (isArray(v))
180 							val = getTypedClassMeta(v);
181 						else
182 							throw brex("Class '%s' was passed to BeanRegistry but value of type '%s' found in map is not a Type object.", ci.getName(), cn(v));
183 						addToMap(typeName, val);
184 					});
185 				} else {
186 					// @formatter:off
187 					var typeName = ap.find(Marshalled.class, ci)
188 						.stream()
189 						.map(x -> x.inner().typeName())
190 						.filter(Shorts::ine)
191 						.findFirst()
192 						.orElseThrow(() -> brex("Class '%s' was passed to BeanRegistry but it doesn't have a @Marshalled(typeName) annotation defined.", ci.getName()));
193 					// @formatter:on
194 					addToMap(typeName, bc.getClassMeta(ci.inner()));
195 				}
196 			}
197 		} catch (BeanRuntimeException e) {
198 			throw e;
199 		} catch (Exception e) {
200 			throw brex(e);
201 		}
202 	}
203 
204 	private void addToMap(String typeName, ClassMeta<?> cm) {
205 		map.put(typeName, cm);
206 		reverseMap.put(cm.inner(), typeName);
207 	}
208 
209 	private ClassMeta<?> getTypedClassMeta(Object array) {
210 		var len = Array.getLength(array);
211 		if (len == 0)
212 			throw brex("Map entry had an empty array value.");
213 		var type = (Type)Array.get(array, 0);
214 		var args = new Type[len - 1];
215 		for (var i = 1; i < len; i++)
216 			args[i - 1] = (Type)Array.get(array, i);
217 		return bc.getClassMeta(type, args);
218 	}
