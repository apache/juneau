109 	@Override
110 	protected <T> T doRead(ParserPipe pipe, ClassMeta<T> type) throws IOException, ParseException, ExecutableException {
111 		try (Reader r = pipe.getParserReader()) {
112 			if (r == null)
113 				return null;
114 			Map<String,Object> root = readTomlDocument(new TomlTokenizer(r));
115 			if (isEmpty(root))
116 				return type.canCreateNewBean(getOuter()) ? type.newInstance(getOuter()) : null;
117 			// Root _value wrapper: document has exactly one key _value
118 			if (root.size() == 1 && root.containsKey("_value")) {
119 				Object inner = root.get("_value");
120 				return (T) convertValue(inner, type);
121 			}
122 			return convertMapToType(root, type);
123 		}
124 	}

126 	private Map<String,Object> readTomlDocument(TomlTokenizer t) throws IOException, ParseException {
127 		Map<String,Object> root = m();
128 		Map<String,Object> currentTable = root;
129 
130 		while (!t.isEof()) {
131 			t.skipWhitespaceAndComments();
132 			if (t.isEof())
133 				break;
134 
135 			int c = t.peek();
136 			if (c == '[') {
137 				t.read();
138 				if (t.peek() == '[') {
139 					t.read();
140 					String path = readTablePath(t);
141 					Object existing = getOrCreateAt(root, path);
142 					if (!(existing instanceof List)) {
143 						if (existing != null)
144 							throw t.parseException("Key " + path + " redefined");
145 						setAt(root, path, l());
146 						existing = getOrCreateAt(root, path);
147 					}
148 					Map<String,Object> newTable = m();
149 					((List)existing).add(newTable);
150 					currentTable = newTable;
151 				} else {
152 					String path = readTablePath(t);
153 					currentTable = getOrCreateTableAt(root, path);
154 				}
155 			} else if (c == '=' || Character.isLetterOrDigit(c) || c == '"' || c == '\'' || c == '_' || c == '-') {
156 				List<String> keyPath = readKey(t);
157 				if (keyPath.isEmpty())
158 					throw t.parseException("Expected key");
159 				t.skipWhitespace();
160 				if (t.read() != '=')
161 					throw t.parseException("Expected '='");
162 				t.skipWhitespace();
163 				Object value = readValue(t);
164 				setValueAt(currentTable, keyPath, value);
165 			} else {
166 				break;
167 			}
168 		}
169 		return root;
170 	}

172 	private static String readTablePath(TomlTokenizer t) throws IOException, ParseException {
173 		t.skipWhitespace();
174 		var parts = new ArrayList<String>();
175 		while (true) {
176 			String key = t.readBareKey();
177 			if (key == null)
178 				break;
179 			parts.add(key);
180 			t.skipWhitespace();
181 			if (t.peek() != '.')
182 				break;
183 			t.read();
184 			t.skipWhitespace();
185 		}
186 		if (t.peek() == ']')
187 			t.read();
188 		if (t.peek() == ']')
189 			t.read();
190 		return String.join(".", parts);
191 	}

193 	private static List<String> readKey(TomlTokenizer t) throws IOException, ParseException {
194 		var parts = new ArrayList<String>();
195 		while (true) {
196 			String key = t.readBareKey();
197 			if (key == null)
198 				break;
199 			parts.add(key);
200 			t.skipWhitespace();
201 			if (t.peek() != '.')
202 				break;
203 			t.read();
204 			t.skipWhitespace();
205 		}
206 		return parts;
207 	}

209 	private Object readValue(TomlTokenizer t) throws IOException, ParseException {
210 		int c = t.peek();
211 		if (c == '"') {
212 			if (isTripleQuote(t, '"'))
213 				return t.readMultiLineBasicString();
214 			return t.readBasicString();
215 		}
216 		if (c == '\'') {
217 			if (isTripleQuote(t, '\''))
218 				return t.readMultiLineLiteralString();
219 			return t.readLiteralString();
220 		}
221 		if (c == 't' || c == 'f')
222 			return t.readBoolean();
223 		if (c == '[') {
224 			t.read();
225 			t.skipWhitespaceAndComments();
226 			var list = l();
227 			while (t.peek() != ']') {
228 				list.add(readValue(t));
229 				t.skipWhitespaceAndComments();
230 				if (t.peek() == ',') {
231 					t.read();
232 					t.skipWhitespaceAndComments();
233 				}
234 			}
235 			t.read();
236 			return list;
237 		}
238 		if (c == '{') {
239 			t.read();
240 			t.skipWhitespace();
241 			var map = new LinkedHashMap<String,Object>();
242 			while (t.peek() != '}') {
243 				t.skipWhitespace();
244 				List<String> k = readKey(t);
245 				if (k.isEmpty())
246 					throw t.parseException("Expected key in inline table");
247 				t.skipWhitespace();
248 				if (t.read() != '=')
249 					throw t.parseException("Expected '='");
250 				t.skipWhitespace();
251 				Object v = readValue(t);
252 				map.put(k.get(0), v);
253 				t.skipWhitespace();
254 				if (t.peek() == ',') {
255 					t.read();
256 					t.skipWhitespace();
257 				}
258 			}
259 			t.read();
260 			return map;
261 		}
262 		if (Character.isDigit(c) || c == '+' || c == '-') {
263 			String s = readUntilValueEnd(t);
264 			if (isEmpty(s)) // HTT: provably dead -- both call sites of this branch (here and the 'i'/'n' branch below) only invoke readUntilValueEnd() after peek() already confirmed the current character is not a terminator, so the returned string always contains at least that one character
265 				throw t.parseException("Expected value");
266 			Number special = readSpecialFloat(s);
267 			if (special != null)
268 				return special;
269 			Number radix = readRadixInteger(s, t);
270 			if (radix != null)
271 				return radix;
272 			String noUnderscore = s.replace("_", "");
273 			if (s.length() >= 10 && s.charAt(4) == '-' && (s.contains("T") || s.contains(":") || s.contains(" ")))
274 				return TomlTokenizer.parseDateTimeString(s);
275 			// Date-like patterns kept as string for convertValue to parse as LocalDate, YearMonth, Year
276 			if (isDateLikeString(noUnderscore))
277 				return s;
278 			if (s.contains(".") || s.toLowerCase(Locale.ROOT).contains("e"))
279 				return Double.parseDouble(noUnderscore);
280 			try {
281 				return Long.parseLong(noUnderscore);
282 			} catch (@SuppressWarnings("unused") NumberFormatException e) {
283 				// May be date/time string that slipped through (e.g. LocalDate, LocalTime)
284 				if (s.contains("-") || s.contains(":"))
285 					return s;
286 				throw t.parseException(MSG_INVALID_NUMBER + s);
287 			}
288 		}
289 		if (c == 'i' || c == 'n') {
290 			String s = readUntilValueEnd(t);
291 			Number special = readSpecialFloat(s);
292 			if (special != null)
293 				return special;
294 			throw t.parseException("Expected inf or nan");
295 		}
296 		throw t.parseException("Unexpected character: " + (char)c);
297 	}

299 	private static boolean isTripleQuote(TomlTokenizer t, char q) throws IOException {
300 		if (t.peek() != q) return false; // HTT: provably dead -- both call sites (readValue's '"'/'\'' dispatch) already peek() the identical character immediately before calling, with nothing consumed in between, so peek() here always still equals q
301 		int c1 = t.read();
302 		int c2 = t.read();
303 		int c3 = t.peek();
304 		if (c2 == q && c3 == q) {
305 			t.read();
306 			return true;
307 		}
308 		if (c2 >= 0)
309 			t.unread(c2);
310 		t.unread(c1);
311 		return false;
312 	}

314 	private static boolean isDateLikeString(String s) {
315 		if (s == null || s.length() < 7) // HTT: the "s == null" half is provably dead -- the sole call site passes noUnderscore (s.replace(...)), which is never null since s (from readUntilValueEnd()) is never null (see note above)
316 			return false;
317 		// yyyy-MM-dd (LocalDate) | yyyy-MM (YearMonth) - exclude plain 4-digit (e.g. port numbers)
318 		return (s.length() == 10 && s.charAt(4) == '-' && s.charAt(7) == '-')
319 			|| (s.length() == 7 && s.charAt(4) == '-');
320 	}

322 	private static String readUntilValueEnd(TomlTokenizer t) throws IOException {
323 		var sb = new StringBuilder();
324 		int c;
325 		while ((c = t.peek()) >= 0 && c != ' ' && c != '\t' && c != '\n' && c != '\r' && c != '#' && c != ',' && c != ']' && c != '}') {
326 			sb.append((char) t.read());
327 		}
328 		return sb.isEmpty() ? null : sb.toString().trim(); // HTT: the "sb.isEmpty()" true-arm is provably dead -- both call sites first peek() a non-terminator character before calling, so at least one character is always appended (see note on the "Expected value" check above)
329 	}

331 	private static Number readSpecialFloat(String s) {
332 		if (s == null) // HTT: provably dead -- both call sites pass the (guaranteed non-null, per the note above) result of readUntilValueEnd() directly
333 			return null;
334 		if (s.equals("inf") || s.equals("+inf"))
335 			return Double.POSITIVE_INFINITY;
336 		if (s.equals("-inf"))
337 			return Double.NEGATIVE_INFINITY;
338 		if (s.equals("nan") || s.equals("+nan") || s.equals("-nan"))
339 			return Double.NaN;
340 		return null;
341 	}

343 	private static Number readRadixInteger(String s, TomlTokenizer t) throws ParseException {
344 		if (s == null || s.length() < 3 || s.charAt(0) != '0') // HTT: the "s == null" third is provably dead -- the sole call site passes the (guaranteed non-null, per the note above) result of readUntilValueEnd() directly
345 			return null;
346 		char p = s.charAt(1);
347 		int radix;
348 		if (p == 'x' || p == 'X')
349 			radix = 16;
350 		else if (p == 'o' || p == 'O')
351 			radix = 8;
352 		else if (p == 'b' || p == 'B')
353 			radix = 2;
354 		else
355 			return null;
356 		String digits = s.substring(2).replace("_", "");
357 		if (digits.isEmpty())
358 			throw t.parseException(MSG_INVALID_NUMBER + s);
359 		try {
360 			return Long.parseUnsignedLong(digits, radix);
361 		} catch (@SuppressWarnings("unused") NumberFormatException e) {
362 			throw t.parseException(MSG_INVALID_NUMBER + s);
363 		}
364 	}

367 		String[] parts = path.split("\\.");
368 		Object current = root;
369 		for (int i = 0; i < parts.length - 1; i++) {
370 			current = ((Map)current).computeIfAbsent(parts[i], k -> new LinkedHashMap<String,Object>());
371 		}
372 		return ((Map)current).get(parts[parts.length - 1]);
373 	}

394 		if (keyPath.size() == 1) {
395 			root.put(keyPath.get(0), value);

397 		}

401 		}

405 	private <T> T convertMapToType(Map<String,Object> map, ClassMeta<T> type) throws ParseException, ExecutableException {
406 		if (type.isMap()) {
407 			var keyType = type.getKeyType();
408 			var valueType = type.getValueType();
409 			Map m = type.canCreateNewInstance(getOuter()) ? (Map) type.newInstance(getOuter()) : newGenericMap(type);
410 			for (var e : map.entrySet()) {
411 				String keyStr = "null".equals(e.getKey()) ? null : e.getKey();
412 				Object key = convertAttrToType(m, keyStr, keyType);
413 				Object val = convertValue(e.getValue(), valueType);
414 				m.put(key, val);
415 			}
416 			return (T) m;
417 		}
418 		BeanMap<?> bm = toBeanMap(type.newInstance(getOuter()));
419 		populateBeanMap(bm, map);
420 		return (T) bm.getBean();
421 	}

424 		var jm = new JsonMap();
425 		for (Entry<?,?> e : map.entrySet()) {
426 			Object k = e.getKey();
427 			Object v = e.getValue();
428 			Object converted = v instanceof Map v2 ? toJsonMap(v2) : convertValue(v, object());
429 			jm.put(k == null ? "null" : k.toString(), converted); // HTT: the "k == null" true-arm is provably dead -- every map reaching toJsonMap() was built by readTomlDocument()/readValue(), whose keys always come from readKey()/readBareKey() (TOML has no null-key syntax), so k is always a non-null String
430 		}
431 		return jm;
432 	}

434 	private void populateBeanMap(BeanMap<?> bm, Map<String,Object> map) throws ParseException, ExecutableException {
435 		for (Entry<String,Object> e : map.entrySet()) {
436 			String key = e.getKey();
437 			Object val = e.getValue();
438 			BeanPropertyMeta pMeta = bm.getMeta().getProperties().get(key);
439 			if (pMeta == null && isIgnoreUnknownBeanProperties())
440 				continue;
441 			if (pMeta == null)
442 				throw new ParseException(this, "Unknown property '%s'", key);
443 			ClassMeta<?> targetType = (ClassMeta<?>) pMeta.getBeanInfo();
444 			Object converted = convertValue(val, targetType);
445 			bm.put(key, converted);
446 		}
447 	}

449 	private Object convertValue(Object val, ClassMeta<?> targetType) throws ParseException, ExecutableException {
450 		if (val == null) // HTT: provably dead -- every call site passes a value sourced from the parsed document tree (readValue()'s return value, or a container built from it), and TOML's grammar has no null literal, so val is never null here
451 			return null;
452 		String nv = ctx != null ? ctx.getNullValue() : null;
453 		if (val instanceof String val2 && nv != null && val2.equals(nv))
454 			return null;
455 		if (val instanceof Map val2) {
456 			if (JsonMap.class.isAssignableFrom(targetType.inner()))
457 				return toJsonMap(val2);
458 			if (targetType.isBean()) {
459 				BeanMap<?> child = toBeanMap(targetType.newInstance(getOuter()));
460 				populateBeanMap(child, val2);
461 				return child.getBean();
462 			}
463 		}
464 		if (val instanceof List val2 && targetType.isCollectionOrArray()) {
465 			ClassMeta<?> elType = targetType.getElementType();
466 			List result = new ArrayList();
467 			for (Object item : val2) {
468 				result.add(convertValue(item, elType));
469 			}
470 			return targetType.isArray() ? toArray(targetType, result) : result;
471 		}
472 		// Bare numeric wire literals (e.g. "2024", "8100000000000") arrive here as Number values
473 		// from the TOML tokenizer.  Route them through the format-aware parsers so the configured
474 		// MarshallingContext.get<Format>() hint (NANOS, ISO_YEAR, MILLIS, …) reaches the coercion.
475 		// Without this routing the generic Number → T coercion below would silently drop the hint
476 		// (e.g. treating Long(2024) as epoch-millis → 1970-01-01T00:00:02.024Z instead of year 2024).
477 		if (val instanceof Number val2) {
478 			if (targetType.isNumber())
479 				return convertToMemberType(null, val2, targetType);
480 			if (targetType.isDuration())
481 				return readDuration(val2.toString());
482 			if (targetType.isPeriod())
483 				return readPeriod(val2.toString());
484 			if (targetType.isDate())
485 				return readDate(val2.toString(), targetType);
486 			if (targetType.isCalendar())
487 				return readCalendar(val2.toString(), targetType);
488 			if (targetType.isTemporal())
489 				return readTemporal(val2.toString(), targetType);
490 		}
491 		if (val instanceof String val2) {
492 			if (targetType.isDate())
493 				return readDate(val2, targetType);
494 			if (targetType.isCalendar())
495 				return readCalendar(val2, targetType);
496 			if (targetType.isTemporal())
497 				return readTemporal(val2, targetType);
498 			if (targetType.isDuration())
499 				return readDuration(val2);
500 			if (targetType.isPeriod())
501 				return readPeriod(val2);
502 		}
503 		// Native TOML datetime literals (Z-zoned / offset / local) are returned by TomlTokenizer as
504 		// java.time.OffsetDateTime / LocalDateTime / LocalDate / LocalTime objects.  Re-stringify
505 		// through the temporal's own ISO toString() and route through readTemporal so the
506 		// configured TemporalFormat hint (e.g. ISO_INSTANT) is honored — otherwise the generic
507 		// Temporal → T coercion below applies the local zone and we lose the wire's UTC anchor.
508 		if (val instanceof TemporalAccessor val2 && targetType.isTemporal())
509 			return readTemporal(val2.toString(), targetType);
510 		// Route String-shaped byte[] values (collection-element + top-level dispatch sites) through
511 		// the configured BinaryFormat's variant binarySwap before falling through to the default
512 		// String → byte[] coercion.  The bean-property path is unaffected because the per-property
513 		// MPP install hides the type-level default swap behind the per-property swap.
514 		var binBytes = tryUnswapByteArray(val, targetType);
515 		if (binBytes != null)
516 			return binBytes;
517 		return convertToMemberType(null, val, targetType);
518 	}

520 	/**
521 	 * Routes a {@link String}-shaped {@code byte[]} value through the session-aware
522 	 * {@link org.apache.juneau.marshall.swaps.BinarySwap} when one is matched.
523 	 *
524 	 * <p>Collection-element and top-level dispatch sites don't go through
525 	 * {@code MarshalledPropertyPostProcessor}'s per-property swap install, so they fall back to the
526 	 * {@link org.apache.juneau.marshall.swap.DefaultSwaps} registry.  Returns {@code null} when the target isn't
527 	 * {@code byte[]}, the value isn't a {@link String}, or no swap is matched on the current session
528 	 * (e.g. {@link BinaryFormat#NOT_SET} is configured).
529 	 */
530 	@SuppressWarnings({
