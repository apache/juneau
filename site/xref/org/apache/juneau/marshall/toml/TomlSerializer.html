29  /**
30   * Serializes POJOs to TOML v1.0.0 format.
31   *
32   * <p>
33   * TOML (Tom's Obvious Minimal Language) maps unambiguously to hash tables with native typing.
34   * See <a href="https://toml.io/en/v1.0.0">TOML v1.0.0</a> specification.
35   *
36   * <h5 class='topic'>Media types</h5>
37   * <p>
38   * Produces: <bc>application/toml</bc>
39   * <br>Accepts: <bc>application/toml</bc>
40   *
41   * <h5 class='topic'>Bean-to-TOML mapping</h5>
42   * <ul class='spaced-list'>
43   * 	<li>Simple properties → key-value pairs
44   * 	<li>Nested beans → <c>[tableName]</c> sections
45   * 	<li>Collections of beans → <c>[[tableName]]</c> array of tables
46   * 	<li>Collections of simple values → <c>key = [a, b, c]</c>
47   * 	<li>Maps with string keys → <c>[mapName]</c> table
48   * 	<li>Null properties are omitted (TOML has no null type)
49   * </ul>
50   *
51   * <h5 class='section'>Example:</h5>
52   * <p class='bjava'>
53   * 	<jc>// Serialize a bean to TOML</jc>
54   * 	String <jv>toml</jv> = TomlSerializer.<jsf>DEFAULT</jsf>.write(<jv>myBean</jv>);
55   *
56   * 	<jc>// Serialize a map (e.g. config with nested sections)</jc>
57   * 	Map&lt;String, Object&gt; <jv>config</jv> = Map.of(<js>"name"</js>, <js>"myapp"</js>, <js>"port"</js>, 8080);
58   * 	<jv>toml</jv> = TomlSerializer.<jsf>DEFAULT</jsf>.write(<jv>config</jv>);
59   *
60   * 	<jc>// Use the marshaller for convenience</jc>
61   * 	<jv>toml</jv> = Toml.<jsm>of</jsm>(<jv>config</jv>);
62   *
63   * 	<jc>// Custom serializer with whitespace for readability</jc>
64   * 	TomlSerializer <jv>s</jv> = TomlSerializer.<jsm>create</jsm>().ws().build();
65   * 	<jv>toml</jv> = <jv>s</jv>.write(<jv>config</jv>);
66   * </p>
67   *
68   * <h5 class='figure'>Example output (Map of name/age):</h5>
69   * <p class='bcode'>
70   * 	name = "Alice"
71   * 	age = 30
72   * </p>
73   *
74   * <h5 class='figure'>Complex (nested table + array):</h5>
75   * <p class='bcode'>
76   * 	name = "Alice"
77   * 	age = 30
78   * 	tags = ["a", "b", "c"]
79   *
80   * 	[address]
81   * 	street = "123 Main St"
82   * 	city = "Boston"
83   * 	state = "MA"
84   * </p>
85   *
86   * <h5 class='topic'>Limitations compared to JSON</h5>
87   * <p>
88   * TOML is a configuration format with a more restricted type system than JSON.
89   * The following features supported by {@link JsonSerializer}
90   * are not available or limited in TOML:
91   * </p>
92   * <ul class='spaced-list'>
93   * 	<li><b>Null values</b> — TOML has no null type. By default, null properties are omitted.
94   * 		Use <c>keepNullProperties()</c> to write the configured <c>nullValue</c> string (e.g. <c>&lt;NULL&gt;</c>) instead.
95   * 	<li><b>Map keys</b> — TOML tables only support string keys. Maps with non-string keys
96   * 		(e.g. <c>Map&lt;Integer,String&gt;</c>) have keys converted via <c>toString()</c>; ensure results are valid TOML keys.
97   * 		Null keys are serialized as the string <c>null</c>.
98   * 	<li><b>Polymorphic types</b> — <c>addBeanTypes()</c> and <c>addRootType()</c> add <c>_type</c>
99   * 		discriminators for polymorphic parsing; use with care as TOML structure differs from JSON.
100  * 	<li><b>Root-level collections</b> — TOML expects a root table. Root arrays use array-of-tables
101  * 		(<c>[[item]]</c>) for beans/maps, or a <c>_value</c> wrapper for primitives and simple arrays.
102  * 	<li><b>Duplicate keys</b> — TOML forbids duplicate keys; structures that would produce duplicate
103  * 		keys may yield invalid TOML.
104  * </ul>
105  *
106  * <h5 class='section'>Notes:</h5><ul>
107  * 	<li class='note'>This class is thread safe and reusable.
108  * </ul>
109  */

114 public class TomlSerializer extends WriterSerializer implements RecordWritable {

116 	private static final String PROP_inlineTableThreshold = "inlineTableThreshold";
117 	private static final String PROP_useInlineTables = "useInlineTables";
118 	private static final String PROP_sortKeys = "sortKeys";
119 	private static final String PROP_nullValue = "nullValue";
120 	private static final String ARG_copyFrom = "copyFrom";
121 
122 	/**
123 	 * Builder class.
124 	 */
125 	public static class Builder extends WriterSerializer.Builder<Builder> {
126 
127 		private static final Cache<HashKey,TomlSerializer> CACHE = Cache.of(HashKey.class, TomlSerializer.class).build();
128 
129 		private int inlineTableThreshold = 3;
130 		private boolean useInlineTables = true;
131 		private boolean sortKeys = false;
132 		private String nullValue = "<NULL>";
133 
134 		protected Builder() {
135 			produces("application/toml");
136 			accept("application/toml");
137 		}
138 
139 		protected Builder(Builder copyFrom) {
140 			super(assertArgNotNull(ARG_copyFrom, copyFrom));
141 			inlineTableThreshold = copyFrom.inlineTableThreshold;
142 			useInlineTables = copyFrom.useInlineTables;
143 			sortKeys = copyFrom.sortKeys;
144 			nullValue = copyFrom.nullValue;
145 		}
146 
147 		protected Builder(TomlSerializer copyFrom) {
148 			super(assertArgNotNull(ARG_copyFrom, copyFrom));
149 			inlineTableThreshold = copyFrom.inlineTableThreshold;
150 			useInlineTables = copyFrom.useInlineTables;
151 			sortKeys = copyFrom.sortKeys;
152 			nullValue = copyFrom.nullValue;
153 		}
154 
155 		/**
156 		 * String to write for null values. Parser treats values matching this as null.
157 		 *
158 		 * <p>
159 		 * Default is {@code <NULL>} to avoid confusion with the literal string {@code "null"}.
160 		 *
161 		 * @param value The null marker string.
162 		 * @return This object.
163 		 */
164 		public Builder nullValue(String value) {
165 			nullValue = value;
166 			return this;
167 		}
168 
169 		/**
170 		 * Maximum properties for inline table format.
171 		 *
172 		 * @param value The threshold.
173 		 * @return This object.
174 		 */
175 		public Builder inlineTableThreshold(int value) {
176 			inlineTableThreshold = value;
177 			return this;
178 		}
179 
180 		/**
181 		 * Use inline tables when possible.
182 		 *
183 		 * @param value The flag.
184 		 * @return This object.
185 		 */
186 		public Builder useInlineTables(boolean value) {
187 			useInlineTables = value;
188 			return this;
189 		}
190 
191 		/**
192 		 * Sort keys alphabetically in output.
193 		 *
194 		 * @param value The flag.
195 		 * @return This object.
196 		 */
197 		public Builder sortKeys(boolean value) {
198 			sortKeys = value;
199 			return this;
200 		}
201 
202 		@Override
203 		public Builder useWhitespace() {
204 			super.useWhitespace();
205 			return this;
206 		}
207 
208 		@Override
209 		public Builder useWhitespace(boolean value) {
210 			super.useWhitespace(value);
211 			return this;
212 		}
213 
214 		@Override
215 		public Builder ws() {
216 			return useWhitespace();
217 		}
218 
219 		@Override
220 		public TomlSerializer build() {
221 			return cache(CACHE).build(TomlSerializer.class);
222 		}
223 
224 		@Override
225 		public Builder copy() {
226 			return new Builder(this);
227 		}
228 
229 		@Override
230 		public HashKey hashKey() {
231 			return HashKey.of(super.hashKey(), inlineTableThreshold, useInlineTables, sortKeys, nullValue);
232 		}
233 	}
234 
235 	/** Default serializer. */
236 	public static final TomlSerializer DEFAULT = new TomlSerializer(create());
237 
238 	/** Default serializer with blank lines between sections. */
239 	public static final TomlSerializer DEFAULT_READABLE = new TomlSerializer(create().ws());
240 
241 	public static Builder create() {
242 		return new Builder();
243 	}

245 	protected final int inlineTableThreshold;
246 	protected final boolean useInlineTables;
247 	protected final boolean sortKeys;
248 	/** String written for null values. */
249 	protected final String nullValue;
250 
251 	/**
252 	 * Constructor.
253 	 *
254 	 * @param builder The builder.
255 	 */
256 	public TomlSerializer(Builder builder) {
257 		super(builder);
258 		inlineTableThreshold = builder.inlineTableThreshold;
259 		useInlineTables = builder.useInlineTables;
260 		sortKeys = builder.sortKeys;
261 		nullValue = or(builder.nullValue, "<NULL>");
262 	}
263 
264 	/**
265 	 * Returns the string written for null values.
266 	 *
267 	 * @return The null marker.
268 	 */
269 	public String getNullValue() {
270 		return nullValue;
271 	}
272 
273 	@Override
274 	public TomlSerializerSession.Builder createSession() {
275 		return TomlSerializerSession.create(this);
276 	}
277 
278 	@Override
279 	public Builder copy() {
280 		return new Builder(this);
281 	}
282 
283 	@Override
284 	protected FluentMap<String,Object> properties() {
285 		return super.properties()
286 			.a(PROP_inlineTableThreshold, inlineTableThreshold)
287 			.a(PROP_useInlineTables, useInlineTables)
288 			.a(PROP_sortKeys, sortKeys)
289 			.a(PROP_nullValue, nullValue);
290 	}

292 	/**
293 	 * Convenience delegator that opens a {@link RecordWriter} over the output using
294 	 * <b>default session arguments</b> (mirrors {@link #write(Object)}).
295 	 *
296 	 * <p>
297 	 * The real implementation lives on {@link TomlSerializerSession#writeRecords(Object)}.
298 	 * Callers that need request-derived configuration should call {@link #createSession()} and
299 	 * invoke {@link TomlSerializerSession#writeRecords(Object)} on the built session instead.
300 	 *
301 	 * @param output The output.
302 	 * @return A new {@link RecordWriter}.
303 	 * @throws IOException If a problem occurred opening the underlying output.
304 	 */
305 	@Override /* RecordWritable */
306 	public RecordWriter writeRecords(Object output) throws IOException {
307 		return ((RecordWritable) getSession()).writeRecords(output);
308 	}
309 
310 	@Override /* RecordWritable */
311 	public boolean isRecordStreaming() {
312 		return false;
313 	}
