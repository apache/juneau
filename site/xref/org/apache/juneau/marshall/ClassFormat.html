24  /**
25   * Supported wire formats for {@link Class} values.
26   *
27   * <p>
28   * Used by {@link MarshallingContext.Builder#classFormat(ClassFormat)},
29   * {@link Marshalled#classFormat()},
30   * {@link MarshalledProp#classFormat()}, and
31   * {@link MarshalledConfig#classFormat()} to control how {@link Class} values
32   * are written to text-based wire formats.
33   *
34   * <p>
35   * The default is {@link #FQCN} which mirrors the historical {@code ClassSwap} behavior (canonical dotted
36   * name with sensible fallback to the JVM binary name).
37   *
38   * <h5 class='topic'>FQCN vs BINARY_NAME for nested classes</h5>
39   *
40   * <p>
41   * For nested-type encoding the two forms diverge:
42   * <ul>
43   * 	<li>{@link #FQCN} uses {@link Class#getCanonicalName()} when available — {@code "java.util.Map.Entry"}
44   * 		— and falls back to {@link Class#getName()} for local / anonymous classes (where
45   * 		{@code getCanonicalName()} returns {@code null}).
46   * 	<li>{@link #BINARY_NAME} always uses {@link Class#getName()} — {@code "java.util.Map$Entry"}.
47   * </ul>
48   *
49   * <p>
50   * For arrays the difference is more dramatic:
51   * <ul>
52   * 	<li>{@link #FQCN} produces {@code "int[]"} / {@code "java.lang.String[]"}.
53   * 	<li>{@link #BINARY_NAME} produces {@code "[I"} / {@code "[Ljava.lang.String;"} (JVM descriptor form).
54   * </ul>
55   *
56   * <h5 class='topic'>SIMPLE_NAME is serialize-only</h5>
57   *
58   * <p>
59   * {@link #SIMPLE_NAME} discards package and outer-class context and is <b>not round-trip safe</b>. The
60   * parser path for this constant throws {@link UnsupportedOperationException}; use {@link #FQCN} or
61   * {@link #BINARY_NAME} for round-trippable wires.
62   *
63   * <h5 class='topic'>Precedence (highest to lowest)</h5>
64   * <ol>
65   * 	<li>{@link MarshalledProp#classFormat() @MarshalledProp(classFormat=…)} on the bean property.
66   * 	<li>{@link Marshalled#classFormat() @Marshalled(classFormat=…)} on the bean class.
67   * 	<li>{@link MarshalledConfig#classFormat() @MarshalledConfig(classFormat=…)} on
68   * 		<code><ja>@Rest</ja></code>-annotated classes / methods.
69   * 	<li>Programmatic {@link MarshallingContext.Builder#classFormat(ClassFormat)}.
70   * 	<li>Environment variable <c>MarshallingContext.classFormat</c>.
71   * 	<li>The default constant ({@link #FQCN}).
72   * </ol>
73   *
74   * <h5 class='topic'>Parser leniency</h5>
75   *
76   * <p>
77   * Parsers SHALL try {@link Class#forName(String, boolean, ClassLoader)} first regardless of the
78   * parser-side {@code ClassFormat} setting — this handles {@link #BINARY_NAME} and most {@link #FQCN}
79   * inputs unambiguously. For nested-type {@link #FQCN} input ({@code "java.util.Map.Entry"}), the parser
80   * walks back through dotted segments from right to left, replacing each uppercase-prefixed dot with
81   * {@code $} and retrying — this resolves multi-level nesting like
82   * {@code "com.example.Outer.Inner.Deepest"} to {@code com.example.Outer$Inner$Deepest}.  For
83   * {@link #FQCN} array input the parser strips trailing {@code []} pairs, resolves the leaf component
84   * type (including primitives like {@code "int"}), and rebuilds the array via {@link Class#arrayType()}
85   * — so {@code "int[]"} resolves to {@code int[].class} and {@code "java.lang.String[][]"} resolves to
86   * {@code String[][].class}.
87   *
88   * <h5 class='topic'>Binary serializers</h5>
89   *
90   * <p>
91   * Binary serializers (BSON / CBOR / MsgPack / Prototext / Parquet) emit a UTF-8 string regardless of this
92   * setting — there is no native {@code Class} wire type in any supported binary format.
93   */
94  public enum ClassFormat {
95  
96  	/** Sentinel meaning "no value configured" — falls through to the next-higher precedence level. */
97  	NOT_SET,
98  
99  	/**
100 	 * Canonical fully-qualified name with dotted nested-type separators (the default).
101 	 *
102 	 * <p>
103 	 * Uses {@link Class#getCanonicalName()} when available; falls back to {@link Class#getName()} for
104 	 * local / anonymous classes where {@code getCanonicalName()} returns {@code null}.
105 	 *
106 	 * <h5 class='figure'>Example:</h5>
107 	 * <p class='bcode'>
108 	 * 	java.lang.String
109 	 * 	java.util.Map.Entry
110 	 * 	int[]
111 	 * </p>
112 	 */
113 	FQCN,
114 
115 	/**
116 	 * JVM binary name per {@link Class#getName()}.
117 	 *
118 	 * <h5 class='figure'>Example:</h5>
119 	 * <p class='bcode'>
120 	 * 	java.lang.String
121 	 * 	java.util.Map$Entry
122 	 * 	[Ljava.lang.String;
123 	 * 	[I
124 	 * </p>
125 	 *
126 	 * <p>
127 	 * Identical to {@link #FQCN} for top-level non-array reference types; diverges for nested types
128 	 * ({@code $} vs {@code .}) and arrays (JVM descriptor vs {@code Type[]} form).
129 	 */
130 	BINARY_NAME,
131 
132 	/**
133 	 * Unqualified simple name per {@link Class#getSimpleName()}.
134 	 *
135 	 * <h5 class='figure'>Example:</h5>
136 	 * <p class='bcode'>
137 	 * 	java.lang.String
138 	 * 	Entry
139 	 * 	int
140 	 * </p>
141 	 *
142 	 * <p>
143 	 * <b>Serialize-only.</b> The parser path for this constant throws
144 	 * {@link UnsupportedOperationException} — there is no defensible way to resolve {@code "Map"} back
145 	 * to a unique {@link Class} without a registry hint, and adding such a hint is out of scope.
146 	 */
147 	SIMPLE_NAME;
148 
149 	// Optional global guard vetting class names before the parser path resolves them.  null == allow all.
150 	private static final AtomicReference<Predicate<String>> classNameGuard = new AtomicReference<>();
151 
152 	/**
153 	 * Installs a global guard that vets class names before the parser path resolves them to a {@link Class}.
154 	 *
155 	 * <p>
156 	 * The guard is consulted for every candidate name (including the nested-type <c>$</c>-rewritten forms)
157 	 * before it is passed to {@link Class#forName(String, boolean, ClassLoader)}.  Names the guard rejects are
158 	 * treated as unresolvable.  This lets a deployment restrict which types may be materialized from parsed
159 	 * input to an explicit allow-list.
160 	 *
161 	 * @param value The guard predicate returning <jk>true</jk> for permitted class names, or <jk>null</jk> to
162 	 * 	allow all names (the default).
163 	 */
164 	@SuppressWarnings("java:S3066") // Public setter is the documented deployment-configuration entry point (see Javadoc above); it is also exercised directly from ClassFormat_Test in another package, so visibility cannot be narrowed without breaking that call site.
165 	public static void setClassNameGuard(Predicate<String> value) {
166 		classNameGuard.set(value);
167 	}
168 
169 	/**
170 	 * Returns whether the specified candidate class name is permitted by the installed guard.
171 	 *
172 	 * @param name The candidate class name.
173 	 * @return <jk>true</jk> if no guard is installed or the guard permits the name.
174 	 */
175 	static boolean isClassNameAllowed(String name) {
176 		var g = classNameGuard.get();
177 		return g == null || g.test(name);
178 	}
179 
180 	/**
181 	 * Formats the specified {@link Class} using this format.
182 	 *
183 	 * <p>
184 	 * {@link #NOT_SET} / {@code null} format are treated as {@link #FQCN}.
185 	 *
186 	 * @param value The value to format. Can be <jk>null</jk> (returns <jk>null</jk>).
187 	 * @param format The configured format. Can be <jk>null</jk> (treated as {@link #FQCN}).
188 	 * @return The formatted wire representation, or <jk>null</jk> if {@code value} is <jk>null</jk>.
189 	 */
190 	public static String format(Class<?> value, ClassFormat format) {
191 		if (value == null)
192 			return null;
193 		var fmt = format == null ? FQCN : format;
194 		return switch (fmt) {
195 			case NOT_SET, FQCN -> {
196 				// getCanonicalName() returns null for local / anonymous classes; fall back to binary name.
197 				var canonical = value.getCanonicalName();
198 				yield canonical != null ? canonical : value.getName();
199 			}
200 			case BINARY_NAME -> value.getName();
201 			case SIMPLE_NAME -> value.getSimpleName();
202 		};
203 	}
204 
205 	/**
206 	 * Parses the specified wire value into a {@link Class}.
207 	 *
208 	 * <p>
209 	 * Lenient parsing — handles every shape produced by {@link #format(Class, ClassFormat)} for the
210 	 * round-trippable formats ({@link #FQCN} and {@link #BINARY_NAME}), and accepts cross-form input
211 	 * as well (an {@link #FQCN} parser still resolves {@link #BINARY_NAME} input and vice versa).
212 	 *
213 	 * <p>
214 	 * Resolution steps:
215 	 * <ol>
216 	 * 	<li>Strip trailing {@code []} pairs ({@link #FQCN} array form), tracking dimension count.  Empty
217 	 * 		brackets at the leading edge (JVM descriptor form like {@code "[I"} /
218 	 * 		{@code "[Ljava.lang.String;"}) are left to {@link Class#forName(String, boolean, ClassLoader)}
219 	 * 		because that method already handles them.
220 	 * 	<li>Resolve the leaf component:
221 	 * 		<ul>
222 	 * 			<li>If the leaf names a primitive type ({@code "int"}, {@code "boolean"}, etc. — emitted
223 	 * 				by {@link #FQCN} for primitive arrays), return the corresponding {@code TYPE} class.
224 	 * 			<li>Otherwise try {@link Class#forName(String, boolean, ClassLoader)} directly.
225 	 * 			<li>On failure, walk dots from right to left, replacing each uppercase-prefixed dot with
226 	 * 				{@code $} (cumulatively — already-replaced dots stay replaced) and retrying.  This
227 	 * 				resolves multi-level nesting like {@code "com.example.Outer.Inner.Deepest"} to
228 	 * 				{@code com.example.Outer$Inner$Deepest}.  Dots whose trailing segment starts with a
229 	 * 				lowercase letter are treated as package boundaries and skipped.
230 	 * 		</ul>
231 	 * 	<li>Apply {@link Class#arrayType()} once per stripped dimension to rebuild the array class.
232 	 * </ol>
233 	 *
234 	 * <p>
235 	 * {@link #SIMPLE_NAME} is not supported on the parser side — throws
236 	 * {@link UnsupportedOperationException}.
237 	 *
238 	 * @param value The wire value. Can be <jk>null</jk> or blank (returns <jk>null</jk>).
239 	 * @param format The configured format hint. Can be <jk>null</jk> (treated as {@link #FQCN}).
240 	 * @param classLoader The class loader. Can be <jk>null</jk> (uses
241 	 *   {@link Thread#getContextClassLoader() context class loader}).
242 	 * @return The parsed {@link Class}, or <jk>null</jk> if {@code value} is <jk>null</jk> or blank.
243 	 * @throws IllegalArgumentException If the value cannot be resolved.
244 	 * @throws UnsupportedOperationException If {@code format} is {@link #SIMPLE_NAME}.
245 	 */
246 	public static Class<?> parse(String value, ClassFormat format, ClassLoader classLoader) {
247 		if (value == null)
248 			return null;
249 		var s = value.trim();
250 		if (s.isEmpty())
251 			return null;
252 		var fmt = format == null ? FQCN : format;
253 		if (fmt == SIMPLE_NAME)
254 			throw new UnsupportedOperationException("SIMPLE_NAME is serialize-only — use FQCN or BINARY_NAME for round-trippable wires");
255 		var cl = classLoader != null ? classLoader : Thread.currentThread().getContextClassLoader();
256 		try {
257 			return resolveWithArraySuffix(s, cl);
258 		} catch (ClassNotFoundException firstAttempt) {
259 			throw iaex("Could not resolve class '%s' for format %s: %s", value, format, firstAttempt.getMessage());
260 		}
261 	}
262 
263 	/**
264 	 * Strips trailing {@code []} pairs from the FQCN array form, resolves the leaf component, then
265 	 * rebuilds the array dimensions via {@link Class#arrayType()}.
266 	 *
267 	 * <p>
268 	 * Inputs that don't end in {@code []} (top-level classes and JVM-descriptor array forms like
269 	 * {@code "[I"}) pass straight through to {@link #resolveLeaf(String, ClassLoader)}.
270 	 */
271 	private static Class<?> resolveWithArraySuffix(String s, ClassLoader cl) throws ClassNotFoundException {
272 		var dims = 0;
273 		var leaf = s;
274 		while (leaf.endsWith("[]")) {
275 			dims++;
276 			leaf = leaf.substring(0, leaf.length() - 2);
277 		}
278 		if (dims == 0)
279 			return resolveLeaf(s, cl);
280 		if (leaf.isEmpty())
281 			throw new ClassNotFoundException(s);
282 		var result = resolveLeaf(leaf, cl);
283 		for (var i = 0; i < dims; i++)
284 			result = result.arrayType();
285 		return result;
286 	}
287 
288 	/**
289 	 * Resolves a non-array leaf token to a {@link Class}.
290 	 *
291 	 * <p>
292 	 * Checks the primitive-name table first ({@link #FQCN} emits {@code "int"} / {@code "boolean"} /
293 	 * etc. for primitive array leafs), then falls through to {@link Class#forName(String, boolean,
294 	 * ClassLoader)} with a multi-level nested-type fallback heuristic.
295 	 */
296 	private static Class<?> resolveLeaf(String s, ClassLoader cl) throws ClassNotFoundException {
297 		var prim = primitiveByName(s);
298 		if (prim != null)
299 			return prim;
300 		if (! isClassNameAllowed(s))
301 			throw new ClassNotFoundException(s);
302 		try {
303 			// Resolve without initializing — a parsed name must never trigger a target type's static initializer.
304 			return Class.forName(s, false, cl);
305 		} catch (ClassNotFoundException firstAttempt) {
306 			var nested = nestedTypeFallback(s, cl);
307 			if (nested != null)
308 				return nested;
309 			throw firstAttempt;
310 		}
311 	}
312 
313 	/**
314 	 * Walks dots from right to left, replacing each uppercase-prefixed dot with {@code $} and retrying
315 	 * {@link Class#forName(String, boolean, ClassLoader)}.  Replacements accumulate — once a dot has
316 	 * been replaced, it stays replaced for subsequent iterations — so multi-level nesting like
317 	 * {@code "Outer.Inner.Deepest"} resolves to {@code "Outer$Inner$Deepest"}.
318 	 *
319 	 * <p>
320 	 * Returns <jk>null</jk> if no rewrite resolves to a loadable class.  Dots whose trailing segment
321 	 * starts with a lowercase letter are treated as package boundaries and left alone.
322 	 */
323 	private static Class<?> nestedTypeFallback(String s, ClassLoader cl) {
324 		var chars = s.toCharArray();
325 		for (var i = chars.length - 1; i > 0; i--) {
326 			if (chars[i] == '.' && i + 1 < chars.length && Character.isUpperCase(chars[i + 1])) {
327 				chars[i] = '$';
328 				var candidate = new String(chars);
329 				if (! isClassNameAllowed(candidate))
330 					continue;
331 				try {
332 					return Class.forName(candidate, false, cl);
333 				} catch (@SuppressWarnings("unused") ClassNotFoundException ignored) {
334 					// Keep walking; the previous-dot replacement persists in chars so deeper nesting can resolve.
335 				}
336 			}
337 		}
338 		return null;
339 	}
340 
341 	private static Class<?> primitiveByName(String s) {
342 		return switch (s) {
343 			case "boolean" -> boolean.class;
344 			case "byte" -> byte.class;
345 			case "char" -> char.class;
346 			case "short" -> short.class;
347 			case "int" -> int.class;
348 			case "long" -> long.class;
349 			case "float" -> float.class;
350 			case "double" -> double.class;
351 			case "void" -> void.class;
352 			default -> null;
353 		};
354 	}
