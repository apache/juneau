44  /**
45   * Marshalling-side post-processor that applies {@link MarshalledProp @MarshalledProp} and {@link Swap @Swap}
46   * annotation effects to a {@link BeanPropertyMeta.Builder} after {@link BeanPropertyMeta.Builder#validate validate()}
47   * runs successfully.
48   *
49   * <p>
50   * The reads here used to live inside {@link BeanPropertyMeta.Builder#validate}.  They were lifted out during
51   * the Task-5 Step 8b-ii refactor so the bean-modeling builder no longer references marshalling-side
52   * {@link ObjectSwap}/{@link StringFormatSwap}/{@link Surrogate} types or the {@link MarshalledProp} annotation.
53   *
54   * <p>
55   * This post-processor mutates the following fields on the supplied builder when it runs:
56   * <ul>
57   * 	<li>{@code swap} — set to a {@link StringFormatSwap} (from {@link MarshalledProp#format()}) or a custom
58   * 		{@link ObjectSwap} (from {@link Swap}).
59   * 	<li>{@code dictionaryClasses} — appended with the {@link MarshalledProp#dictionary()} entries.
60   * 	<li>{@code typeMeta} — refreshed to the swap class meta when a swap is installed.
61   * </ul>
62   *
63   * <p>
64   * Only invoked from {@link BeanMeta#validateAndRegisterProperty} on the marshalling-side construction path
65   * (i.e. when {@link MarshallingContext} is non-null).  The bean-modeling-only path
66   * ({@link BeanMeta#of(Class, BeanConfigContext)}) does not run this post-processor.
67   */
68  @SuppressWarnings({
69  	"java:S6548", // Intentional stateless singleton SPI implementation wired through BeanConfigContext.
70  	"java:S6539"  // Monster Class threshold is advisory; this is a routing seam that intentionally depends on many marshalling types.
71  })
72  final class MarshalledPropertyPostProcessor implements BeanPropertyPostProcessor {

80  	 * SPI entry point — narrows the opaque marshalling context, then dispatches to
81  	 * {@link #process(MarshallingContext, BeanPropertyMeta.Builder)}.
82  	 *
83  	 * @param marshallingContext The marshalling-side context.  Must not be <jk>null</jk> on the marshalling path.
84  	 * @param builder The bean-property builder.  Must not be <jk>null</jk>.
85  	 */
86  	@Override
87  	public void process(Object marshallingContext, BeanPropertyMeta.Builder builder) {
88  		if (marshallingContext == null)
89  			return;
90  		process((MarshallingContext) marshallingContext, builder);

93  	/**
94  	 * Applies {@code @MarshalledProp}/{@code @Swap} annotation effects to {@code b} using {@code bc} for
95  	 * {@link ClassMeta} resolution.
96  	 *
97  	 * @param bc The marshalling context.  Must not be <jk>null</jk>.
98  	 * @param b The bean-property builder.  Must not be <jk>null</jk>.
99  	 */
100 	@SuppressWarnings({
101 		"java:S3776" // Annotation-discovery dispatch over innerField/getter/setter — branching is intentional; splitting hurts JIT inlining.
102 	})
103 	static void process(MarshallingContext bc, BeanPropertyMeta.Builder b) {
104 		var ap = bc.getAnnotationProvider();
105 		var bdClasses = new ArrayList<Class<?>>();
106 		Set<String> viewNames = null;
107 		var propertyClass = propertyClass(b);
108 		// Per-property null-coercion mode discovered across innerField/getter/setter (last-seen-wins, matching the
109 		// existing format-override loop semantics).  Stays Nulls.NOT_SET when no @MarshalledProp annotation
110 		// declares it, in which case only the context-level Parser.Builder#nulls(Nulls) default applies.
111 		var perPropertyNulls = Nulls.NOT_SET;
112 
113 		// XMLGregorianCalendar always uses XML format regardless of any CalendarFormat setting.
114 		if (b.swap == null && XMLGregorianCalendar.class.equals(propertyClass))
115 			b.swap = xmlGregorianCalendarSwap();
116 
117 		// innerField, getter, setter — same order as the original validate() loop.
118 		if (nn(b.innerField)) {
119 			ap.find(MarshalledProp.class, b.innerField).forEach(x -> {
120 				var mp = x.inner();
121 				if (b.swap == null)
122 					b.swap = marshalledPropSwap(x);
123 				applyPropertyFormats(b, mp, propertyClass);
124 				bdClasses.addAll(l(mp.dictionary()));
125 			});
126 			ap.find(Swap.class, b.innerField).stream().findFirst().ifPresent(x -> b.swap = swapSwap(x));
127 			b.isUri |= ap.has(Uri.class, b.innerField);
128 			for (var ai : ap.find(MarshalledProp.class, b.innerField)) {
129 				viewNames = addViews(viewNames, ai.inner().view());
130 				if (ai.inner().nulls() != Nulls.NOT_SET)
131 					perPropertyNulls = ai.inner().nulls();
132 			}
133 		}
134 
135 		if (nn(b.getter)) {
136 			ap.find(MarshalledProp.class, b.getter).forEach(x -> {
137 				var mp = x.inner();
138 				if (b.swap == null)
139 					b.swap = marshalledPropSwap(x);
140 				applyPropertyFormats(b, mp, propertyClass);
141 				bdClasses.addAll(l(mp.dictionary()));
142 			});
143 			ap.find(Swap.class, b.getter).forEach(x -> b.swap = swapSwap(x));
144 			b.isUri |= ap.has(Uri.class, b.getter);
145 			for (var ai : ap.find(MarshalledProp.class, b.getter)) {
146 				viewNames = addViews(viewNames, ai.inner().view());
147 				if (ai.inner().nulls() != Nulls.NOT_SET)
148 					perPropertyNulls = ai.inner().nulls();
149 			}
150 		}
151 
152 		if (nn(b.setter)) {
153 			ap.find(MarshalledProp.class, b.setter).forEach(x -> {
154 				var mp = x.inner();
155 				if (b.swap == null)
156 					b.swap = marshalledPropSwap(x);
157 				applyPropertyFormats(b, mp, propertyClass);
158 				bdClasses.addAll(l(mp.dictionary()));
159 			});
160 			ap.find(Swap.class, b.setter).forEach(x -> b.swap = swapSwap(x));
161 			b.isUri |= ap.has(Uri.class, b.setter);
162 			for (var ai : ap.find(MarshalledProp.class, b.setter)) {
163 				viewNames = addViews(viewNames, ai.inner().view());
164 				if (ai.inner().nulls() != Nulls.NOT_SET)
165 					perPropertyNulls = ai.inner().nulls();
166 			}
167 		}
168 
169 		if (viewNames != null)
170 			b.views = viewNames;
171 
172 		ClassInfo ownerClass = owningClass(b);
173 		if (nn(ownerClass)) {
174 			ap.find(Marshalled.class, ownerClass).stream().findFirst().ifPresent(x -> {
175 				var m = x.inner();
176 				applyClassFormats(b, m, propertyClass);
177 			});
178 		}
179 
180 		if (b.swap == null)
181 			applyContextFormats(b, bc, propertyClass);
182 
183 		if (! bdClasses.isEmpty()) {
184 			var infos = new ArrayList<ClassInfo>(b.dictionaryClasses.size() + bdClasses.size());
185 			infos.addAll(b.dictionaryClasses);
186 			bdClasses.forEach(c -> infos.add(info(c)));
187 			b.dictionaryClasses = infos;
188 		}
189 
190 		// If a swap was installed and we have a rawTypeMeta, refresh typeMeta to the swap's swap-class meta.
191 		// Matches the original behavior inside validate() which routed swap-class meta through bc.getClassMeta.
192 		// Cast to ObjectSwap because b.swap is Object-typed (BeanPropertyMeta.Builder lives in the bean-modeling
193 		// layer and cannot reference ObjectSwap directly).
194 		if (nn(b.swap) && nn(b.rawTypeMeta))
195 			b.typeMeta = bc.getClassMeta(((ObjectSwap<?,?>) b.swap).getSwapClass());
196 
197 		// Install swap-aware read/write transforms on the builder.
198 		// Previously lived as a private static helper on {@link BeanMeta}; moved here during Task-5 Step 8b-ii
199 		// so {@link BeanMeta} no longer references {@link ObjectSwap}/{@link ParseException}/{@link SerializeException}.
200 		installSwapAwareTransforms(b);
201 
202 		// Install schema-validation transforms (wraps any existing readTransform/writeTransform) when a
203 		// PropertyValidatorFactory is on the classpath and the property carries at least one @Schema annotation.
204 		installSchemaValidationTransforms(bc, b);
205 
206 		// Install null-coercion writeTransform (wraps any existing writeTransform).  Always installed so the
207 		// context-level Parser.Builder#nulls(Nulls) default can take effect at runtime even when no per-property
208 		// annotation is configured.  The wrap is a no-op for non-null values and the LEAVE policy.
209 		installNullCoercionTransform(b, perPropertyNulls, propertyClass);
210 	}

212 	/**
213 	 * Installs swap-aware read/write transforms on a {@link BeanPropertyMeta.Builder} after validation.
214 	 *
215 	 * <p>
216 	 * After {@link BeanPropertyMeta.Builder#validate validate()} succeeds, the builder's {@code swap} and
217 	 * {@code rawTypeMeta} fields describe whether the property has a configured {@link ObjectSwap} (via
218 	 * {@link MarshalledProp @MarshalledProp(format=...)} or
219 	 * {@link Swap @Swap}) and whether the property's raw type has child swaps registered
220 	 * on it.  This method packages those concerns into install-time closures so the marshalling-side swap behavior is
221 	 * established as data on the {@link BeanPropertyMeta} rather than executed by the bean-modeling
222 	 * {@link BeanPropertyMeta#get get}/{@link BeanPropertyMeta#set set} methods themselves.
223 	 *
224 	 * <p>
225 	 * If neither a property-level swap nor a child swap on the raw type meta is present, no transforms are installed
226 	 * and the property's {@code get}/{@code set} fall through to identity (raw access).
227 	 *
228 	 * @param p The builder to attach swap-aware transforms to.
229 	 */
230 	@SuppressWarnings({
231 		"java:S3776", // Centralized swap-transform wiring; branching mirrors swap/no-swap and child-swap read/write paths.
232 		"unchecked",  // Wildcard ObjectSwap captured from Object-typed builder fields for runtime polymorphic dispatch.
233 		"null"        // `sw` may be null but is guarded by `nn(sw)` before access; Eclipse doesn't recognise `nn()` as a null-check function.
234 	})
235 	static void installSwapAwareTransforms(BeanPropertyMeta.Builder p) {
236 		ObjectSwap<?,?> sw = (ObjectSwap<?,?>) p.swap;
237 		ClassMeta<?> rtm = (ClassMeta<?>) p.rawTypeMeta;
238 		if (sw == null && (rtm == null || ! rtm.hasChildSwaps()))
239 			return;
240 		if (p.readTransform == null) {
241 			p.readTransform = (session, o) -> {
242 				try {
243 					// The transform is typed against BeanSession (commons.bean SPI) but ObjectSwap.swap requires a
244 					// MarshallingSession.  The marshalling-side BeanMap always wires a MarshallingSession into its
245 					// transform call sites, so the narrowing cast is safe.
246 					var ms = (MarshallingSession) session;
247 					if (nn(sw))
248 						return ((ObjectSwap<Object,Object>) sw).swap(ms, o);
249 					if (o == null)
250 						return null;
251 					if (rtm.hasChildSwaps()) {
252 						ObjectSwap<?,?> f = rtm.getChildObjectSwapForSwap(o.getClass());
253 						if (nn(f))
254 							return ((ObjectSwap<Object,Object>) f).swap(ms, o);
255 					}
256 					return o;
257 				} catch (RuntimeException e) {
258 					throw e;
259 				} catch (Exception e) {
260 					throw new SerializeException(e);
261 				}
262 			};
263 		}
264 		if (p.writeTransform == null) {
265 			p.writeTransform = (session, o) -> {
266 				try {
267 					// See readTransform note: BeanSession → MarshallingSession is safe on the marshalling-side path.
268 					var ms = (MarshallingSession) session;
269 					if (nn(sw))
270 						return ((ObjectSwap<Object,Object>) sw).unswap(ms, o, rtm);
271 					if (o == null)
272 						return null;
273 					if (rtm.hasChildSwaps()) {
274 						ObjectSwap<?,?> f = rtm.getChildObjectSwapForUnswap(o.getClass());
275 						if (nn(f))
276 							return ((ObjectSwap<Object,Object>) f).unswap(ms, o, rtm);
277 					}
278 					return o;
279 				} catch (RuntimeException e) {
280 					throw e;
281 				} catch (Exception e) {
282 					throw new ParseException(e);
283 				}
284 			};
285 		}
286 	}

294 	 * Installs a null-coercion {@code writeTransform} on a {@link BeanPropertyMeta.Builder}.
295 	 *
296 	 * <p>
297 	 * The installed wrap is invoked on every {@code BeanPropertyMeta.set} call but is a near-zero-cost no-op when the
298 	 * post-swap value is non-null (the common case).  When the value is {@code null} (or {@link Optional#empty()} /
299 	 * {@link OptionalInt}/{@link OptionalLong}/{@link OptionalDouble} in their empty form), the wrap consults the
300 	 * effective {@link Nulls} mode (per-property &gt; session-level &gt; context-level &gt; {@link Nulls#LEAVE
301 	 * LEAVE}) and substitutes one of the documented coerced values, or returns {@link BeanPropertyMeta#SKIP_VALUE} to
302 	 * signal the bean-modeling layer to skip the set entirely.
303 	 *
304 	 * @param b The bean-property builder.
305 	 * @param perPropertyNulls The per-property {@link Nulls} mode discovered on {@code @MarshalledProp}.
306 	 * 	{@link Nulls#NOT_SET} means "no per-property override; fall through to the session/context default".
307 	 * @param propertyClass The raw property class — used to compute {@link Nulls#EMPTY} / {@link Nulls#DEFAULT}
308 	 * 	substitutes.
309 	 */
310 	@SuppressWarnings({
311 		"java:S3776" // Cognitive complexity is acceptable; switch-over-modes is the clearest expression of the contract.
312 	})
313 	static void installNullCoercionTransform(BeanPropertyMeta.Builder b, Nulls perPropertyNulls, Class<?> propertyClass) {
314 		var inner = b.writeTransform;
315 		// Capture the full property ClassMeta so the Optional empty-default is nesting-aware (e.g.
316 		// Optional<Optional<X>> → Optional.of(Optional.empty()) rather than a collapsed single-level
317 		// Optional.empty()).  The raw propertyClass alone cannot express the nesting depth.
318 		var propertyMeta = (b.rawTypeMeta instanceof ClassMeta<?> cm) ? cm : null;
319 		b.writeTransform = (session, value) -> {
320 			Object resolved = nn(inner) ? inner.apply(session, value) : value;
321 			// Fast path — non-null, non-empty-Optional values bypass the policy lookup entirely.
322 			if (resolved != null && ! isEmptyOptional(resolved))
323 				return resolved;
324 			var effective = resolveEffectiveNulls(perPropertyNulls, session);
325 			return switch (effective) {
326 				case LEAVE, NOT_SET -> coerceLeave(resolved, propertyClass, propertyMeta);
327 				case EMPTY -> emptyValueFor(propertyClass, propertyMeta);
328 				case DEFAULT -> defaultValueFor(b, propertyClass, propertyMeta, session);
329 				case SKIP -> BeanPropertyMeta.SKIP_VALUE;
330 			};
331 		};
332 	}

335 	 * Resolves the effective {@link Nulls} mode for a null-valued set.
336 	 *
337 	 * <p>
338 	 * Precedence: per-property annotation &gt; session-level override &gt; parser-context default &gt;
339 	 * {@link Nulls#LEAVE}.
340 	 *
341 	 * @param perPropertyNulls The per-property mode (may be {@link Nulls#NOT_SET}).
342 	 * @param session The bean session — narrowed to {@link ParserSession} to read the session/context defaults.
343 	 * @return The effective mode (never {@code null}; falls back to {@link Nulls#LEAVE}).
344 	 */
345 	private static Nulls resolveEffectiveNulls(Nulls perPropertyNulls, BeanSession session) {
346 		if (perPropertyNulls != null && perPropertyNulls != Nulls.NOT_SET)
347 			return perPropertyNulls;
348 		if (session instanceof ParserSession session2) {
349 			var ctx = session2.getNulls();
350 			if (ctx != null && ctx != Nulls.NOT_SET)
351 				return ctx;
352 		}
353 		return Nulls.LEAVE;
354 	}

357 	 * {@link Nulls#LEAVE} coercion — passes through unchanged for non-Optional properties, but resolves
358 	 * {@code null}/empty-Optional to {@link Optional#empty()} (or the matching primitive-Optional empty sentinel)
359 	 * for {@link Optional}-typed properties so the shared Optional contract holds.
360 	 */
361 	private static Object coerceLeave(Object resolved, Class<?> propertyClass, ClassMeta<?> propertyMeta) {
362 		var empty = emptyOptional(propertyClass, propertyMeta);
363 		return or(empty, resolved);
364 	}

366 	/**
367 	 * Returns the empty form for an {@link Optional}-flavored property, or {@code null} if the property is not an
368 	 * optional type.
369 	 *
370 	 * <p>
371 	 * For a regular {@link Optional}, this is nesting-aware: it delegates to {@link ClassMeta#getOptionalDefault()}
372 	 * so {@code Optional<Optional<X>>} resolves to {@code Optional.of(Optional.empty())} (the shared contract:
373 	 * present outer, empty inner) rather than a structure-collapsing single-level {@code Optional.empty()}.  When
374 	 * the property metadata is unavailable, falls back to a single-level {@code Optional.empty()}.
375 	 */
376 	private static Object emptyOptional(Class<?> propertyClass, ClassMeta<?> propertyMeta) {
377 		if (Optional.class.equals(propertyClass))
378 			return (propertyMeta != null && propertyMeta.isOptional()) ? propertyMeta.getOptionalDefault() : oe();
379 		if (OptionalInt.class.equals(propertyClass))
380 			return OptionalInt.empty();
381 		if (OptionalLong.class.equals(propertyClass))
382 			return OptionalLong.empty();
383 		if (OptionalDouble.class.equals(propertyClass))
384 			return OptionalDouble.empty();
385 		return null;
386 	}

388 	@SuppressWarnings({
389 		"java:S3776" // Type-dispatch chain; splitting into per-type helpers would obscure the single-purpose intent.
390 	})
391 	private static Object emptyValueFor(Class<?> propertyClass, ClassMeta<?> propertyMeta) {
392 		if (propertyClass == null || propertyClass == Object.class)
393 			return null;
394 		var emptyOpt = emptyOptional(propertyClass, propertyMeta);
395 		if (emptyOpt != null)
396 			return emptyOpt;
397 		if (String.class.equals(propertyClass) || CharSequence.class.equals(propertyClass))
398 			return "";
399 		if (propertyClass.isArray())
400 			return java.lang.reflect.Array.newInstance(propertyClass.getComponentType(), 0);
401 		if (List.class.isAssignableFrom(propertyClass) || Collection.class.equals(propertyClass)
402 				|| Iterable.class.equals(propertyClass))
403 			return l();
404 		if (Set.class.isAssignableFrom(propertyClass))
405 			return new LinkedHashSet<>();
406 		if (Map.class.isAssignableFrom(propertyClass))
407 			return m();
408 		// Primitive defaults — Java auto-unbox handles boxed-type assignment.
409 		if (propertyClass.isPrimitive()) {
410 			if (boolean.class.equals(propertyClass)) return Boolean.FALSE;
411 			if (byte.class.equals(propertyClass))    return Byte.valueOf((byte) 0);
412 			if (short.class.equals(propertyClass))   return Short.valueOf((short) 0);
413 			if (char.class.equals(propertyClass))    return Character.valueOf((char) 0);
414 			if (int.class.equals(propertyClass))     return Integer.valueOf(0);
415 			if (long.class.equals(propertyClass))    return Long.valueOf(0L);
416 			if (float.class.equals(propertyClass))   return Float.valueOf(0f);
417 			if (double.class.equals(propertyClass))  return Double.valueOf(0d);
418 		}
419 		// Other reference types — no canonical empty sentinel; fall back to null (the LEAVE behavior).
420 		return null;
421 	}

423 	/**
424 	 * {@link Nulls#DEFAULT} coercion — returns the property's value on a fresh no-arg-constructed reference
425 	 * instance of the owning bean.  When no reference instance can be built (no accessible no-arg ctor, ctor
426 	 * throws), falls back to {@link Nulls#LEAVE} behavior.
427 	 */
428 	@SuppressWarnings({
429 		"java:S1166" // RuntimeException intentionally swallowed; the documented contract is "fall back to LEAVE".
430 	})
431 	private static Object defaultValueFor(BeanPropertyMeta.Builder b, Class<?> propertyClass, ClassMeta<?> propertyMeta, BeanSession session) {
432 		if (! (session instanceof MarshallingSession session2))
433 			return coerceLeave(null, propertyClass, propertyMeta);
434 		var ownerInfo = owningClass(b);
435 		if (ownerInfo == null)
436 			return coerceLeave(null, propertyClass, propertyMeta);
437 		var ownerMeta = session2.getClassMeta(ownerInfo.inner());
438 		Optional<?> ref = ownerMeta.getProperty(NULLS_DEFAULT_REFERENCE_CACHE_KEY, cm -> {
439 			var instance = BeanInstantiator.createOrNull(cm.inner());
440 			return instance == null ? NO_DEFAULT_REFERENCE : instance;
441 		});
442 		var owner = ref.orElse(null);
443 		if (owner == null || owner == NO_DEFAULT_REFERENCE)
444 			return coerceLeave(null, propertyClass, propertyMeta);
445 		try {
446 			var beanMap = session2.toBeanMap(owner);
447 			var prop = beanMap.getMeta().getPropertyMeta(b.name);
448 			if (prop == null)
449 				return coerceLeave(null, propertyClass, propertyMeta);
450 			return prop.get(beanMap, b.name);
451 		} catch (RuntimeException e) {
452 			return coerceLeave(null, propertyClass, propertyMeta);
453 		}
454 	}

456 	private static boolean isEmptyOptional(Object o) {
457 		if (o instanceof Optional<?> o2) return o2.isEmpty();
458 		if (o instanceof OptionalInt o2)  return o2.isEmpty();
459 		if (o instanceof OptionalLong o2) return o2.isEmpty();
460 		if (o instanceof OptionalDouble o2) return o2.isEmpty();
461 		return false;
462 	}

464 	/**
465 	 * Installs {@code @Schema}-driven validation transforms on a {@link BeanPropertyMeta.Builder}.
466 	 *
467 	 * <p>
468 	 * Discovery flow:
469 	 * <ol>
470 	 * 	<li>Resolve a {@link PropertyValidatorFactory} via {@link PropertyValidators#factory()} (ServiceLoader-backed).
471 	 * 		If no factory is on the classpath, this is a silent no-op.
472 	 * 	<li>Collect every {@link Schema @Schema} annotation reachable from {@code b.innerField} / {@code b.getter} /
473 	 * 		{@code b.setter} via the {@link AnnotationProvider}.
474 	 * 	<li>If at least one is found, merge them into a single {@link JsonMap} via {@link SchemaAnnotation#asMap(Schema)}
475 	 * 		(later entries override earlier).
476 	 * 	<li>Ask the factory for a {@link PropertyValidator} keyed off the merged map.  If the factory returns
477 	 * 		{@code null} (e.g. the map carried only descriptive keywords), do nothing.
478 	 * 	<li>Wrap {@code b.readTransform} and {@code b.writeTransform} so the validator runs whenever the
479 	 * 		{@link MarshallingSession} has {@code isValidateSchema()} enabled.
480 	 * </ol>
481 	 *
482 	 * <p>
483 	 * Validation failures throw {@link SchemaValidationException}, which we wrap in {@link BeanRuntimeException} so
484 	 * the parser's outer catch block lifts them into {@link ParseException} and the serializer surfaces them through
485 	 * the existing exception flow.
486 	 *
487 	 * @param bc The marshalling context.  Must not be <jk>null</jk>.
488 	 * @param b The bean-property builder.  Must not be <jk>null</jk>.
489 	 */
490 	@SuppressWarnings({
491 		"java:S3776" // Discovery branches over innerField/getter/setter mirror the swap discovery loop above.
492 	})
493 	static void installSchemaValidationTransforms(MarshallingContext bc, BeanPropertyMeta.Builder b) {
494 		// Skip when validation is not enabled on the context.  BeanMeta is cached per-context, so contexts with
495 		// validateSchema=true get their own cache slot and we don't pay the discovery / factory cost on the
496 		// default (validateSchema=false) path.  Also avoids a recursive BeanMeta build cycle: the factory parses
497 		// JSON into org.apache.juneau.bean.jsonschema.JsonSchema (FQCN: juneau-bean-jsonschema is a runtime-only
498 		// dep, not on the juneau-marshall compile classpath) via the default JSON parser, which would otherwise
499 		// re-enter this method while the original BeanMeta is still under construction.
500 		if (! bc.isValidateSchema())
501 			return;
502 		var factory = PropertyValidators.factory();
503 		if (factory == null)
504 			return;
505 		var ap = bc.getAnnotationProvider();
506 		JsonMap merged = null;
507 		if (nn(b.innerField))
508 			for (var ai : ap.find(Schema.class, b.innerField))
509 				merged = applyToMap(merged, ai.inner());
510 		if (nn(b.getter))
511 			for (var ai : ap.find(Schema.class, b.getter))
512 				merged = applyToMap(merged, ai.inner());
513 		if (nn(b.setter))
514 			for (var ai : ap.find(Schema.class, b.setter))
515 				merged = applyToMap(merged, ai.inner());
516 		if (ie(merged))
517 			return;
518 		var validator = factory.create(merged, propertyClass(b));
519 		if (validator == null)
520 			return;
521 
522 		final var propertyName = b.name;
523 		var innerRead = b.readTransform;
524 		b.readTransform = (session, o) -> {
525 			Object v = nn(innerRead) ? innerRead.apply(session, o) : o;
526 			if (session instanceof MarshallingSession session2 && session2.isValidateSchema()) {
527 				try {
528 					validator.validate(v);
529 				} catch (SchemaValidationException e) {
530 					throw new BeanRuntimeException(e, null, "Schema validation failed on property '%s': %s", propertyName, e.getMessage());
531 				}
532 			}
533 			return v;
534 		};
535 		var innerWrite = b.writeTransform;
536 		b.writeTransform = (session, o) -> {
537 			if (session instanceof MarshallingSession session2 && session2.isValidateSchema()) {
538 				try {
539 					validator.validate(o);
540 				} catch (SchemaValidationException e) {
541 					throw new BeanRuntimeException(e, null, "Schema validation failed on property '%s': %s", propertyName, e.getMessage());
542 				}
543 			}
544 			return nn(innerWrite) ? innerWrite.apply(session, o) : o;
545 		};
546 	}

707 	private static void applyContextFormats(BeanPropertyMeta.Builder b, MarshallingContext bc, Class<?> propertyClass) {
708 		if (Duration.class.equals(propertyClass))
709 			b.swap = durationSwap(bc.getDurationFormat());
710 		else if (Period.class.equals(propertyClass))
711 			b.swap = periodSwap(bc.getPeriodFormat());
712 		else if (isCalendarType(propertyClass))
713 			b.swap = calendarSwap(bc.getCalendarFormat());
714 		else if (Date.class.equals(propertyClass))
715 			b.swap = dateSwap(bc.getDateFormat());
716 		else if (isTemporalType(propertyClass))
717 			b.swap = temporalSwap(bc.getTemporalFormat(), propertyClass);
718 		else if (isTemporalAccessorType(propertyClass))
719 			b.swap = temporalAccessorSwap(bc.getTemporalFormat(), propertyClass);
720 		else if (isTimeZoneType(propertyClass))
721 			b.swap = timeZoneSwap(bc.getTimeZoneFormat(), propertyClass);
722 		else if (Locale.class.equals(propertyClass))
723 			b.swap = localeSwap(bc.getLocaleFormat());
724 		// byte[] context default: install a per-property swap only when an explicit BinaryFormat is configured
725 		// (i.e. != NOT_SET).  The per-property swap drives both the marshalling-side serializer path AND the
726 		// {@link BeanMap} property-level read/write path.  When the context is at NOT_SET we leave byte[]
727 		// untouched so it falls back to the surrounding serializer's native array handling (e.g. JSON array).
728 		else if (byte[].class.equals(propertyClass) && bc.getBinaryFormat() != BinaryFormat.NOT_SET)
729 			b.swap = binarySwap(bc.getBinaryFormat());
730 		else if (isEnumType(propertyClass) && bc.getEnumFormat() == EnumFormat.ORDINAL)
731 			// Only install context-level swap for ORDINAL so binary serializers emit a native number.
732 			// Other formats are handled by ClassMeta.toString() / ClassMeta.newInstanceFromString() against
733 			// MarshallingContext.getEnumFormat() — installing a String-producing swap on every enum property
734 			// would mask reflection-driven enum handling for no behavioural benefit.
735 			b.swap = enumSwap(bc.getEnumFormat(), propertyClass);
736 		else if (UUID.class.equals(propertyClass) && bc.getUuidFormat() != UuidFormat.STANDARD && bc.getUuidFormat() != UuidFormat.NOT_SET)
737 			// Install context-level swap only when the configured format diverges from STANDARD (today's
738 			// default — UUID.toString()).  At STANDARD / NOT_SET, the serializer's natural String-conversion
739 			// path already produces the right wire form.
740 			b.swap = uuidSwap(bc.getUuidFormat());
741 		else if (BitSet.class.equals(propertyClass))
742 			// BitSet has no natural String round-trip (BitSet.toString() is not parseable), so install the
743 			// context-level swap unconditionally — every configured format (INDICES default included)
744 			// produces a parseable textual wire form.
745 			b.swap = bitSetSwap(bc.getBitSetFormat());
746 		else if (isBigNumberType(propertyClass) && bc.getBigNumberFormat() != BigNumberFormat.NUMBER && bc.getBigNumberFormat() != BigNumberFormat.NOT_SET)
747 			// Install context-level swap only for STRING / AUTO.  At NUMBER / NOT_SET, BigInteger /
748 			// BigDecimal are already emitted as bare numeric tokens by every text serializer.
749 			b.swap = bigNumberSwap(bc.getBigNumberFormat(), propertyClass);
750 		else if (isBooleanType(propertyClass) && bc.getBooleanFormat() != BooleanFormat.TRUE_FALSE && bc.getBooleanFormat() != BooleanFormat.NOT_SET)
751 			// Install context-level swap only when format diverges from TRUE_FALSE (today's default —
752 			// native boolean wire token).  At TRUE_FALSE / NOT_SET, text serializers' natural boolean
753 			// emit path already produces the right wire form.
754 			b.swap = booleanSwap(bc.getBooleanFormat());
755 		else if (isFloatType(propertyClass) && requiresFloatSwap(bc.getFloatFormat(), propertyClass))
756 			// Install context-level swap only when the format diverges from Juneau's native bare-numeric
757 			// emit AND the property type can meaningfully represent the transformed value.  Primitive
758 			// float/double properties cannot hold null/String results, so we skip the swap and let the
759 			// native emit handle NaN/Infinity directly.  Explicit per-property or per-class @MarshalledProp /
760 			// @Marshalled overrides still install via the applyPropertyFormats / applyClassFormats branches.
761 			b.swap = floatSwap(bc.getFloatFormat(), propertyClass);
762 		else if (Currency.class.equals(propertyClass) && bc.getCurrencyFormat() != CurrencyFormat.ISO_CODE && bc.getCurrencyFormat() != CurrencyFormat.NOT_SET)
763 			// Install context-level swap only when format diverges from ISO_CODE (today's default —
764 			// Currency.toString()).  At ISO_CODE / NOT_SET, the serializer's natural String-conversion
765 			// path already produces the right wire form.
766 			b.swap = currencySwap(bc.getCurrencyFormat());
767 		// Class<?> bean properties fall through to the {@link ClassFormatSwap}
768 		// registered in {@link DefaultSwaps} — that swap reads the context's
769 		// classFormat dynamically.  Only @MarshalledProp / @Marshalled overrides install a per-property swap
770 		// (handled in applyPropertyFormats / applyClassFormats above).
771 	}
