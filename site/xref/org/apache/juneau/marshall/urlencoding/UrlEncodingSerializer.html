29  /**
30   * Serializes POJO models to URL-encoded notation with UON-encoded values (a notation for URL-encoded query paramter values).
31   *
32   * <h5 class='section'>Media types:</h5>
33   * <p>
34   * Handles <c>Accept</c> types:  <bc>application/x-www-form-urlencoded</bc>
35   * <p>
36   * Produces <c>Content-Type</c> types:  <bc>application/x-www-form-urlencoded</bc>
37   *
38   * <h5 class='topic'>Description</h5>
39   *
40   * This serializer provides several serialization options.
41   * <br>Typically, one of the predefined DEFAULT serializers will be sufficient.
42   * <br>However, custom serializers can be constructed to fine-tune behavior.
43   *
44   * <p>
45   * The following shows a sample object defined in Javascript:
46   * <p class='bjson'>
47   * 	{
48   * 		id: 1,
49   * 		name: <js>'John Smith'</js>,
50   * 		uri: <js>'http://sample/addressBook/person/1'</js>,
51   * 		addressBookUri: <js>'http://sample/addressBook'</js>,
52   * 		birthDate: <js>'1946-08-12T00:00:00Z'</js>,
53   * 		otherIds: <jk>null</jk>,
54   * 		addresses: [
55   * 			{
56   * 				uri: <js>'http://sample/addressBook/address/1'</js>,
57   * 				personUri: <js>'http://sample/addressBook/person/1'</js>,
58   * 				id: 1,
59   * 				street: <js>'100 Main Street'</js>,
60   * 				city: <js>'Anywhereville'</js>,
61   * 				state: <js>'NY'</js>,
62   * 				zip: 12345,
63   * 				isCurrent: <jk>true</jk>,
64   * 			}
65   * 		]
66   * 	}
67   * </p>
68   *
69   * <p>
70   * Using the "strict" syntax defined in this document, the equivalent URL-encoded notation would be as follows:
71   * <p class='burlenc'>
72   * 	<ua>id</ua>=<un>1</un>
73   * 	&amp;<ua>name</ua>=<us>'John+Smith'</us>,
74   * 	&amp;<ua>uri</ua>=<us>http://sample/addressBook/person/1</us>,
75   * 	&amp;<ua>addressBookUri</ua>=<us>http://sample/addressBook</us>,
76   * 	&amp;<ua>birthDate</ua>=<us>1946-08-12T00:00:00Z</us>,
77   * 	&amp;<ua>otherIds</ua>=<uk>null</uk>,
78   * 	&amp;<ua>addresses</ua>=@(
79   * 		(
80   * 			<ua>uri</ua>=<us>http://sample/addressBook/address/1</us>,
81   * 			<ua>personUri</ua>=<us>http://sample/addressBook/person/1</us>,
82   * 			<ua>id</ua>=<un>1</un>,
83   * 			<ua>street</ua>=<us>'100+Main+Street'</us>,
84   * 			<ua>city</ua>=<us>Anywhereville</us>,
85   * 			<ua>state</ua>=<us>NY</us>,
86   * 			<ua>zip</ua>=<un>12345</un>,
87   * 			<ua>isCurrent</ua>=<uk>true</uk>
88   * 		)
89   * 	)
90   * </p>
91   *
92   * <h5 class='section'>Example:</h5>
93   * <p class='bjava'>
94   * 	<jc>// Serialize a Map</jc>
95   * 	Map <jv>map</jv> = JsonMap.<jsm>ofText</jsm>(<js>"{a:'b',c:1,d:false,e:['f',1,false],g:{h:'i'}}"</js>);
96   *
97   * 	<jc>// Serialize to value equivalent to JSON.</jc>
98   * 	<jc>// Produces "a=b&amp;c=1&amp;d=false&amp;e=@(f,1,false)&amp;g=(h=i)"</jc>
99   * 	String <jv>uenc</jv> = UrlEncodingSerializer.<jsf>DEFAULT</jsf>.write(<jv>map</jv>);
100  *
101  * 	<jc>// Serialize a bean</jc>
102  * 	<jk>public class</jk> Person {
103  * 		<jk>public</jk> Person(String <jv>name</jv>);
104  * 		<jk>public</jk> String getName();
105  * 		<jk>public int</jk> getAge();
106  * 		<jk>public</jk> Address getAddress();
107  * 		<jk>public boolean</jk> deceased;
108  * 	}
109  *
110  * 	<jk>public class</jk> Address {
111  * 		<jk>public</jk> String getStreet();
112  * 		<jk>public</jk> String getCity();
113  * 		<jk>public</jk> String getState();
114  * 		<jk>public int</jk> getZip();
115  * 	}
116  *
117  * 	Person <jv>person</jv> = <jk>new</jk> Person(<js>"John Doe"</js>, 23, <js>"123 Main St"</js>, <js>"Anywhere"</js>, <js>"NY"</js>, 12345, <jk>false</jk>);
118  *
119  * 	<jc>// Produces "name=John+Doe&amp;age=23&amp;address=(street='123+Main+St',city=Anywhere,state=NY,zip=12345)&amp;deceased=false"</jc>
120  * 	String <jv>uenc</jv> = UrlEncodingSerializer.<jsf>DEFAULT</jsf>.write(<jv>person</jv>);
121  * </p>
122  *
123  * <h5 class='section'>Notes:</h5><ul>
124  * 	<li class='note'>This class is thread safe and reusable.
125  * </ul>
126  *
127  * <h5 class='section'>See Also:</h5><ul>
128  * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/UrlEncodingSupport">URL-Encoding Basics</a>
129  * </ul>
130  */

189 		/**
190 		 * Serialize bean property collections/arrays as separate key/value pairs.
191 		 *
192 		 * <p>
193 		 * By default, serializing the array <c>[1,2,3]</c> results in <c>?key=$a(1,2,3)</c>.
194 		 * <br>When enabled, serializing the same array results in <c>?key=1&amp;key=2&amp;key=3</c>.
195 		 *
196 		 * <p>
197 		 * This option only applies to beans.
198 		 *
199 		 * <h5 class='section'>Notes:</h5><ul>
200 		 * 	<li class='note'>
201 		 * 		If parsing multi-part parameters, it's highly recommended to use <c>Collections</c> or <c>Lists</c>
202 		 * 		as bean property types instead of arrays since arrays have to be recreated from scratch every time a value
203 		 * 		is added to it.
204 		 * </ul>
205 		 *
206 		 * <h5 class='section'>Example:</h5>
207 		 * <p class='bjava'>
208 		 * 	<jc>// A sample bean.</jc>
209 		 * 	<jk>public class</jk> A {
210 		 * 		<jk>public</jk> String[] <jf>f1</jf> = {<js>"a"</js>,<js>"b"</js>};
211 		 * 		<jk>public</jk> List&lt;String&gt; <jf>f2</jf> = Arrays.<jsm>asList</jsm>(<jk>new</jk> String[]{<js>"c"</js>,<js>"d"</js>});
212 		 * 	}
213 		 *
214 		 * 	<jc>// Normal serializer.</jc>
215 		 * 	WriterSerializer <jv>serializer1</jv> = UrlEncodingSerializer.<jsf>DEFAULT</jsf>;
216 		 *
217 		 * 	<jc>// Expanded-params serializer.</jc>
218 		 * 	WriterSerializer <jv>serializer2</jv> = UrlEncodingSerializer.<jsm>create</jsm>().expandedParams().build();
219 		 *
220 		 *  <jc>// Produces "f1=(a,b)&amp;f2=(c,d)"</jc>
221 		 * 	String <jv>out1</jv> = <jv>serializer1</jv>.write(<jk>new</jk> A());
222 		 *
223 		 * 	<jc>// Produces "f1=a&amp;f1=b&amp;f2=c&amp;f2=d"</jc>
224 		 * 	String <jv>out2</jv> = <jv>serializer2</jv>.write(<jk>new</jk> A());
225 		 * </p>
226 		 *
227 		 * @return This object.
228 		 */
229 		public Builder expandedParams() {
230 			return expandedParams(true);
231 		}
232 
233 		/**
234 		 * Same as {@link #expandedParams()} but allows you to explicitly specify the value.
235 		 *
236 		 * @param value The value for this setting.
237 		 * @return This object.
238 		 */
239 		public Builder expandedParams(boolean value) {
240 			expandedParams = value;
241 			return this;
242 		}
