119 	 * Converts a Collection into an integer-indexed map.
120 	 */
121 	private static Map<Integer,Object> getCollectionMap(Collection<?> c) {
122 		var m = new TreeMap<Integer,Object>();
123 		var i = IntegerHolder.create();
124 		c.forEach(o -> m.put(i.getAndIncrement(), o));
125 		return m;
126 	}

128 	/*
129 	 * Converts an array into an integer-indexed map.
130 	 */
131 	private static Map<Integer,Object> getCollectionMap(Object array) {
132 		var m = new TreeMap<Integer,Object>();
133 		for (var i = 0; i < Array.getLength(array); i++)
134 			m.put(i, Array.get(array, i));
135 		return m;
136 	}

152 	/*
153 	 * Workhorse method. Determines the type of object, and then calls the appropriate type-specific serialization method.
154 	 */
155 	@SuppressWarnings({
156 		"java:S3776" // Cognitive complexity acceptable for URL encoding serialization routing
157 	})
158 	private SerializerWriter<?> writeAnything(UonWriter out, Object o) throws IOException, SerializeException {
159 
160 		ClassMeta<?> aType = null;			// The actual type
161 		ClassMeta<?> sType = null;			// The serialized type
162 
163 		var eType = getExpectedRootType(o);
164 		aType = push2("root", o, eType);
165 		indent--;
166 		if (aType == null)
167 			aType = object();
168 
169 		sType = aType;
170 		var typeName = getBeanTypeName(this, eType, aType, null);
171 
172 		// Swap if necessary
173 		var swap = aType.getSwap(this);
174 		if (nn(swap)) {
175 			o = swap(swap, o);
176 			sType = swap.getSwapClassMeta(this);
177 
178 			// If the getSwapClass() method returns Object, we need to figure out
179 			// the actual type now.
180 			if (sType.isObject())
181 				sType = getClassMetaForObject(o);
182 		}
183 
184 		if (sType.isBean()) {
185 			writeBeanMap(out, toBeanMap(o), typeName);
186 		} else if (sType.isMap()) {
187 			if (sType.isBeanMap())
188 				writeBeanMap(out, (BeanMap)o, typeName);
189 			else
190 				writeMap(out, (Map)o, sType);
191 		} else if (sType.isCollection() || sType.isArray()) {
192 			var m = sType.isCollection() ? getCollectionMap((Collection)o) : getCollectionMap(o);
193 			writeCollectionMap(out, m, getClassMeta(Map.class, Integer.class, Object.class));
194 		} else if (sType.isStreamable()) {
195 			writeStreamableAsCollectionMap(out, o, sType);
196 		} else if (sType.isReader()) {
197 			pipe((Reader)o, out);
198 		} else if (sType.isInputStream()) {
199 			pipe((InputStream)o, out);
200 		} else {
201 			// All other types can't be serialized as key/value pairs, so we create a
202 			// mock key/value pair with a "_value" key.
203 			out.append("_value=");
204 			pop();
205 			super.writeAnything(out, o, null, null, null);
206 			return out;
207 		}
208 
209 		pop();
210 		return out;
211 	}

214 	@SuppressWarnings({
215 		"java:S3776", // Cognitive complexity acceptable for expanded-params bean map serialization
216 		"java:S2177", // Intentional: UrlEncodingSerializerSession provides its own private writeBeanMap() with expanded-params logic
217 	})
218 	private SerializerWriter<?> writeBeanMap(UonWriter out, BeanMap<?> m, String typeName) throws SerializeException {
219 		var addAmp = Flag.create();
220 
221 		if (nn(typeName)) {
222 			var pm = m.getMeta().getTypeProperty();
223 			out.appendObject(pm.getName(), true).append('=').appendObject(typeName, false);
224 			addAmp.set();
225 		}
226 
227 		Predicate<Object> checkNull = x -> isKeepNullProperties() || nn(x);
228 		m.forEachValue(checkNull, (pMeta, key, value, thrown) -> {
229 			var cMeta = (ClassMeta<?>) pMeta.getBeanInfo();
230 			var sMeta = cMeta.getSerializedClassMeta(this);
231 
232 			if (nn(thrown))
233 				onBeanGetterException(pMeta, thrown);
234 
235 			if (canIgnoreValue(pMeta, key, value))
236 				return;
237 
238 			if (nn(value) && shouldUseExpandedParams(pMeta)) {
239 				// Transformed object array bean properties may be transformed resulting in ArrayLists,
240 				// so we need to check type if we think it's an array.
241 				if (sMeta.isCollection() || value instanceof Collection) {
242 					var c = (Collection<?>) value;
243 					if (c.isEmpty()) {
244 						// Empty expanded-params collection: emit a single "key=" segment so the
245 						// parser can distinguish "non-null empty collection" from "null property"
246 						// (both would otherwise produce no wire output, defeating round-trip).
247 						addAmp.ifSet(() -> out.cr(indent).append('&')).set();
248 						out.appendObject(key, true).append('=');
249 					} else {
250 						c.forEach(x -> {
251 							addAmp.ifSet(() -> out.cr(indent).append('&')).set();
252 							out.appendObject(key, true).append('=');
253 							super.writeAnything(out, x, cMeta.getElementType(), key, pMeta);
254 						});
255 					}
256 				} else /* array */ {
257 					var len = Array.getLength(value);
258 					if (len == 0) {
259 						// Empty expanded-params array: emit a single "key=" segment so the parser
260 						// can distinguish "non-null empty array" from "null property" (both would
261 						// otherwise produce no wire output, defeating round-trip).
262 						addAmp.ifSet(() -> out.cr(indent).append('&')).set();
263 						out.appendObject(key, true).append('=');
264 					} else {
265 						for (var i = 0; i < len; i++) {
266 							addAmp.ifSet(() -> out.cr(indent).append('&')).set();
267 							out.appendObject(key, true).append('=');
268 							super.writeAnything(out, Array.get(value, i), cMeta.getElementType(), key, pMeta);
269 						}
270 					}
271 				}
272 			} else {
273 				addAmp.ifSet(() -> out.cr(indent).append('&')).set();
274 				out.appendObject(key, true).append('=');
275 				super.writeAnything(out, value, cMeta, key, pMeta);
276 			}
277 		});
278 
279 		return out;
280 	}

282 	private SerializerWriter<?> writeCollectionMap(UonWriter out, Map<?,?> m, ClassMeta<?> type) throws SerializeException {
283 
284 		var valueType = type.getValueType();
285 
286 		var addAmp = Flag.create();
287 
288 		m.forEach((k, v) -> {
289 			addAmp.ifSet(() -> out.cr(indent).append('&')).set();
290 			out.append(k).append('=');
291 			super.writeAnything(out, v, valueType, null, null);
292 		});
293 
294 		return out;
295 	}

297 	private SerializerWriter<?> writeStreamableAsCollectionMap(UonWriter out, Object o, ClassMeta<?> sType) throws SerializeException {
298 		var addAmp = Flag.create();
299 		var i = IntegerHolder.create();
300 		forEachStreamableEntry(o, sType, v -> {
301 			addAmp.ifSet(() -> out.cr(indent).append('&')).set();
302 			out.append(i.getAndIncrement()).append('=');
303 			super.writeAnything(out, v, null, null, null);
304 		});
305 		return out;
306 	}

308 	@SuppressWarnings({
309 		"java:S2177" // Intentional: UrlEncodingSerializerSession provides its own private writeMap() with expanded-params logic
310 	})
311 	private SerializerWriter<?> writeMap(UonWriter out, Map m, ClassMeta<?> type) throws SerializeException {
312 
313 		var keyType = type.getKeyType();
314 		var valueType = type.getValueType();
315 
316 		var addAmp = Flag.create();
317 
318 		forEachEntry(m, e -> {
319 			var key = generalize(e.getKey(), keyType);
320 			var value = e.getValue();
321 
322 			if (shouldUseExpandedParams(value)) {
323 				if (value instanceof Collection value2) {
324 					value2.forEach(x -> {
325 						addAmp.ifSet(() -> out.cr(indent).append('&')).set();
326 						out.appendObject(key, true).append('=');
327 						super.writeAnything(out, x, null, s(key), null);
328 					});
329 				} else /* array */ {
330 					for (var i = 0; i < Array.getLength(value); i++) {
331 						addAmp.ifSet(() -> out.cr(indent).append('&')).set();
332 						out.appendObject(key, true).append('=');
333 						super.writeAnything(out, Array.get(value, i), null, s(key), null);
334 					}
335 				}
336 			} else {
337 				addAmp.ifSet(() -> out.cr(indent).append('&')).set();
338 				out.appendObject(key, true).append('=');
339 				super.writeAnything(out, value, valueType, (key == null ? null : key.toString()), null);
340 			}
341 		});
342 
343 		return out;
344 	}

346 	/*
347 	 * Returns <jk>true</jk> if the specified bean property should be expanded as multiple key-value pairs.
348 	 */
349 	private boolean shouldUseExpandedParams(BeanPropertyMeta pMeta) {
350 		var cm = ((ClassMeta<?>) pMeta.getBeanInfo()).getSerializedClassMeta(this);
351 		return cm.isCollectionOrArray() && (isExpandedParams() || getUrlEncodingClassMeta((ClassMeta<?>) pMeta.getBeanMeta().getBeanInfo()).isExpandedParams());
352 	}
353 
354 	/*
355 	 * Returns <jk>true</jk> if the specified value should be represented as an expanded parameter list.
356 	 */
357 	private boolean shouldUseExpandedParams(Object value) {
358 		if (value == null || ! isExpandedParams())
359 			return false;
360 		var cm = getClassMetaForObject(value).getSerializedClassMeta(this);
361 		return cm.isCollectionOrArray() && isExpandedParams();
362 	}
