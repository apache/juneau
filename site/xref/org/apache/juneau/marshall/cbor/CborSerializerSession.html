91  	/*
92  	 * Converts the specified output target object to an {@link CborOutputStream}.
93  	 */
94  	private static CborOutputStream getCborOutputStream(SerializerPipe out) throws IOException {
95  		Object output = out.getRawOutput();
96  		if (output instanceof CborOutputStream output2)
97  			return output2;
98  		var os = new CborOutputStream(out.getOutputStream());
99  		out.setOutputStream(os);
100 		return os;
101 	}

112 	/**
113 	 * Opens a low-level push generator that emits CBOR (RFC 8949) one structural event at a time,
114 	 * bound to this live session.
115 	 *
116 	 * <p>
117 	 * Containers ({@code startObject} / {@code startArray}) are emitted as CBOR
118 	 * <b>indefinite-length</b> markers (<c>0xBF</c> / <c>0x9F</c>) so the caller doesn't have to
119 	 * know element counts up front; matching {@code endXxx} calls emit the BREAK byte
120 	 * (<c>0xFF</c>).  Resulting bytes are valid RFC-8949 CBOR.
121 	 *
122 	 * <h5 class='section'>Builder properties honored:</h5>
123 	 * Databind-level settings flow through to {@link TokenWriter#object(Object) object(Object)}:
124 	 * <c>keepNullProperties</c>, <c>trimEmptyMaps</c>, <c>trimEmptyCollections</c>,
125 	 * <c>sortMaps</c>, <c>sortCollections</c>, <c>trimStrings</c>.
126 	 *
127 	 * <h5 class='section'>Builder properties NOT honored:</h5>
128 	 * <c>useWhitespace</c>, <c>maxIndent</c>, <c>quoteChar</c>, <c>escapeSolidus</c> (binary
129 	 * format &mdash; no text-formatting concepts apply); <c>uriContext</c>,
130 	 * <c>uriResolution</c>, <c>uriRelativity</c>, <c>listener</c>.
131 	 *
132 	 * @param output The output.  Accepts an {@link OutputStream}.
133 	 * @return A new {@link CborTokenWriter}.
134 	 * @throws IOException If the output type is not supported or could not be opened.
135 	 */
136 	@Override /* TokenWritable */
137 	public TokenWriter writeTokens(Object output) throws IOException {
138 		if (output == null)
139 			throw new IOException("Output cannot be null.");
140 		if (!(output instanceof OutputStream output2))
141 			throw ioex("Cannot convert object of type %s to an OutputStream.", output.getClass().getName());
142 		var walk = new PojoWalker.Options(
143 			isKeepNullProperties(),
144 			isTrimEmptyMaps(),
145 			isTrimEmptyCollections(),
146 			isSortMaps(),
147 			isSortCollections(),
148 			isTrimStrings(),
149 			getMarshallingContext());
150 		return new CborTokenWriter(output2, new CborTokenWriter.Settings(walk));
151 	}
152 
153 	/**
154 	 * Streaming array-element {@link RecordWriter} backed by {@link CborTokenWriter}.
155 	 * Memory is O(1) &mdash; CBOR's writer always uses indefinite-length arrays
156 	 * (0x9F .. 0xFF) so element count is not required up front.
157 	 *
158 	 * @param output The output.
159 	 * @return A new element-streamed {@link RecordWriter}.
160 	 * @throws IOException If a problem occurred opening the underlying output.
161 	 */
162 	@Override /* ArrayRecordWritable */
163 	public RecordWriter writeArrayRecords(Object output) throws IOException {
164 		return StreamingArrayRecord.writer(writeTokens(output));
165 	}

167 	/*
168 	 * Workhorse method.
169 	 * Determines the type of object, and then calls the appropriate type-specific serialization method.
170 	 */
171 	@SuppressWarnings({
172 		"rawtypes",   // Raw types necessary for generic type handling
173 		"java:S3776"  // Cognitive complexity acceptable for serialization dispatch logic
174 	})
175 	private CborOutputStream writeAnything(CborOutputStream out, Object o, ClassMeta<?> eType, String attrName, BeanPropertyMeta pMeta) throws SerializeException {
176 
177 		if (o == null)
178 			return out.appendNull();
179 
180 		if (eType == null)
181 			eType = object();
182 
183 		ClassMeta<?> aType = null;
184 		ClassMeta<?> sType = null;
185 
186 		aType = push2(attrName, o, eType);
187 
188 		if (aType == null)
189 			return out.appendNull();
190 
191 		if (isOptional(aType)) {
192 			o = getOptionalValue(o);
193 			eType = getOptionalType(eType);
194 			aType = getClassMetaForObject(o, object());
195 		}
196 
197 		sType = aType;
198 		String typeName = getBeanTypeName(this, eType, aType, pMeta);
199 
200 		var swap = aType.getSwap(this);
201 		if (nn(swap)) {
202 			o = swap(swap, o);
203 			sType = swap.getSwapClassMeta(this);
204 
205 			if (sType.isObject())
206 				sType = getClassMetaForObject(o);
207 		}
208 
209 		if (o == null || (sType.isChar() && ((Character)o).charValue() == 0))
210 			out.appendNull();
211 		else if (sType.isBean())
212 			writeBeanMap(out, toBeanMap(o), typeName);
213 		else if (sType.isMap()) {
214 			if (sType.isBeanMap())
215 				writeBeanMap(out, (BeanMap)o, typeName);
216 			else
217 				writeMap(out, (Map)o, eType);
218 		} else if (sType.isCollection()) {
219 			writeCollection(out, (Collection)o, eType);
220 		} else if (sType.isByteArray()) {
221 			out.appendBinary((byte[])o);
222 		} else if (sType.isArray()) {
223 			writeCollection(out, toList(sType.inner(), o), eType);
224 		} else if (sType.isBoolean()) {
225 			out.appendBoolean((Boolean)o);
226 		} else if (sType.isNumber()) {
227 			out.appendNumber((Number)o);
228 		} else if (sType.isUri() || (nn(pMeta) && pMeta.isUri())) {
229 			out.appendString(resolveUri(o.toString()));
230 		} else if (sType.isDate()) {
231 			out.appendString(writeDate((Date)o, sType));
232 		} else if (sType.isCalendar()) {
233 			out.appendString(writeCalendar(o, sType));
234 		} else if (sType.isTemporal()) {
235 			out.appendString(writeTemporal((TemporalAccessor)o, sType));
236 		} else if (sType.isDuration()) {
237 			appendDuration(out, (Duration)o);
238 		} else if (sType.isPeriod()) {
239 			out.appendString(writePeriod((Period)o));
240 		} else if (sType.isStreamable()) {
241 			writeCollection(out, toListFromStreamable(o, sType), eType);
242 		} else if (sType.isReader()) {
243 			pipe((Reader)o, out, SerializerSession::handleThrown); // HTT: javac's synthetic bridge for this method ref always throws (handleThrown re-throws any IOException as SerializeException), so JaCoCo can't register full coverage past it
244 		} else if (sType.isInputStream()) {
245 			pipe((InputStream)o, out, SerializerSession::handleThrown); // HTT: same JaCoCo limitation as the isReader() branch above
246 		} else
247 			out.appendString(toString(o));
248 
249 		pop();
250 		return out;
251 	}

253 	private void writeBeanMap(CborOutputStream out, BeanMap<?> m, String typeName) throws SerializeException {
254 
255 		Predicate<Object> checkNull = x -> isKeepNullProperties() || nn(x);
256 
257 		var values = new ArrayList<BeanPropertyValue>();
258 
259 		if (nn(typeName)) {
260 			BeanPropertyMeta pm = m.getMeta().getTypeProperty();
261 			values.add(new BeanPropertyValue(pm, pm.getName(), typeName, null));
262 		}
263 
264 		m.forEachValue(checkNull, (pMeta, key, value, thrown) -> {
265 			if (nn(thrown)) {
266 				onBeanGetterException(pMeta, thrown); // HTT: BeanPropertyMeta.getInner() only lets `thrown` reach here when isIgnoreInvocationExceptionsOnGetters() is false, in which case this call always throws; JaCoCo cannot register coverage past a call that never returns
267 				return; // HTT: unreachable -- onBeanGetterException always throws when reached (see above)
268 			}
269 			var p = new BeanPropertyValue(pMeta, key, value, null);
270 
271 			if ((! isKeepNullProperties()) && willRecurse(p)) {
272 				return;
273 			}
274 
275 			values.add(p);
276 		});
277 
278 		out.startMap(values.size());
279 
280 		values.forEach(x -> {
281 			BeanPropertyMeta pMeta = x.getMeta();
282 			if (pMeta.canRead()) { // HTT: every value here came from forEachValue's canRead()-filtered properties or the always-readable type pseudo-property, so canRead() is never false here
283 				ClassMeta<?> cMeta = (ClassMeta<?>) x.getBeanInfo();
284 				String key = x.getName();
285 				Object value = x.getValue();
286 				writeAnything(out, key, null, null, null);
287 				writeAnything(out, value, cMeta, key, pMeta);
288 			}
289 		});
290 	}

292 	@SuppressWarnings({
293 		"rawtypes",  // Raw types necessary for generic collection handling
294 		"unchecked"  // Type erasure requires unchecked operations
295 	})
296 	private void writeCollection(CborOutputStream out, Collection c, ClassMeta<?> type) throws SerializeException {
297 		var elementType = type.getElementType();
298 		List<Object> l = listOfSize(c.size());
299 		c = sort(c);
300 		l.addAll(c);
301 		out.startArray(l.size());
302 		l.forEach(x -> writeAnything(out, x, elementType, "<iterator>", null));
303 	}

305 	@SuppressWarnings({
306 		"rawtypes",  // Raw types necessary for generic map handling
307 		"unchecked"  // Type erasure requires unchecked operations
308 	})
309 	private void writeMap(CborOutputStream out, Map m, ClassMeta<?> type) throws SerializeException {
310 
311 		var keyType = type.getKeyType();
312 		var valueType = type.getValueType();
313 
314 		m = sort(m);
315 
316 		List<SimpleMapEntry> entries = listOfSize(m.size());
317 		m.forEach((k, v) -> entries.add(new SimpleMapEntry(k, v)));
318 
319 		out.startMap(entries.size());
320 
321 		entries.forEach(x -> {
322 			Object value = x.value();
323 			Object key = generalize(x.key(), keyType);
324 			writeAnything(out, key, keyType, null, null);
325 			writeAnything(out, value, valueType, null, null);
326 		});
327 	}

329 	private boolean willRecurse(BeanPropertyValue v) throws SerializeException {
330 		var aType = push2(v.getName(), v.getValue(), (ClassMeta<?>) v.getBeanInfo());
331 		if (nn(aType))
332 			pop();
333 		return aType == null;
334 	}

336 	@Override /* Overridden from SerializerSession */
337 	protected void doWrite(SerializerPipe out, Object o) throws IOException, SerializeException {
338 		writeAnything(getCborOutputStream(out), o, getExpectedRootType(o), "root", null);
339 	}

341 	private void appendDuration(CborOutputStream out, Duration value) {
342 		var f = getDurationFormat();
343 		var s = writeDuration(value);
344 		if (f == DurationFormat.NANOS || f == DurationFormat.MILLIS)
345 			out.appendNumber(Long.parseLong(s));
346 		else if (f == DurationFormat.SECONDS)
347 			out.appendNumber(Double.parseDouble(s));
348 		else
349 			out.appendString(s);
350 	}
