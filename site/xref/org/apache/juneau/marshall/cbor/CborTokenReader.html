28  /**
29   * Reference implementation of the public {@link TokenReader} surface for the CBOR format
30   * (RFC 8949).
31   *
32   * <p>
33   * Backed by the existing internal {@link CborInputStream} plumbing.  CBOR's wire format is already
34   * token-shaped (major-type encoded data items), so this cursor maps almost directly onto
35   * {@link CborInputStream#readDataType()}.
36   *
37   * <h5 class='section'>Notes:</h5><ul>
38   * 	<li class='warn'>This class is not thread safe.
39   * 	<li>The cursor is a true O(1)-memory streaming cursor ({@link #isStreaming()} == <jk>true</jk>).
40   * 	<li>The cursor is purely structural; object swaps and {@code @Schema} annotations have no
41   * 		effect.  Use the POJO databind path on {@link CborParser} for those.
42   * </ul>
43   */
44  @SuppressWarnings({
45  	"resource" // The cursor's underlying ParserPipe is owned by the caller via try-with-resources on the cursor itself; Eclipse JDT flags the inner pipe as unclosed but that's by design.
46  })
47  public class CborTokenReader implements TokenReader {
48  
49  	private static final long INDEFINITE = -1L;
50  
51  	/**
52  	 * Default maximum number of nested CBOR semantic tags (major type 6) the cursor will unwrap
53  	 * before throwing a {@link ParseException}.
54  	 *
55  	 * <p>
56  	 * CBOR allows a value to be wrapped in an arbitrary number of semantic tags
57  	 * (e.g. <c>0xC0 0xC0 ... value</c>).  Each tag is unwrapped recursively by {@link #next()}, so a
58  	 * pathologically deep chain of tags would otherwise blow the JVM call stack with a
59  	 * {@link StackOverflowError}.  This bound converts that failure mode into a clean, catchable
60  	 * {@link ParseException}.
61  	 */
62  	static final int DEFAULT_MAX_TAG_NESTING_DEPTH = 64;
63  
64  	private final CborInputStream is;
65  	private final ParserPipe pipe;
66  	private final CborParserSession session;
67  
68  	private int maxTagNestingDepth = DEFAULT_MAX_TAG_NESTING_DEPTH;
69  	// Current depth of the in-progress recursive tag unwrap (reset to 0 once a value is emitted).
70  	private int tagNestingDepth;
71  
72  	private TokenType currentToken = TokenType.NOT_AVAILABLE;
73  	private boolean ended;
74  
75  	// Container stack: parallel arrays — kind (0=array, 1=map) + remaining element count
76  	// (INDEFINITE for indefinite-length).  For maps the count covers key+value pairs at 2 each.
77  	private long[] stackRemaining = new long[16];
78  	private boolean[] stackIsMap = new boolean[16];
79  	// Per-level: at map-level, true when next emit should be a FIELD_NAME (key); false when next
80  	// emit should be a value.  Stored per-level so nested containers don't corrupt each other.
81  	private boolean[] stackAwaitingKey = new boolean[16];
82  	private int depth;
83  
84  	// Scratch state for current token's value.
85  	private String currentString;
86  	private String currentNumberLexeme;
87  	private Number currentNumber;
88  	private boolean currentBoolean;
89  	private byte[] currentBinary;

91  	// Binary-native opt-in state (175ad).  Defaults preserve the normalize-by-default contract:
92  	// nativeMode == false ⇒ tag numbers are dropped, simple values surface as VALUE_NULL with no
93  	// metadata.  When nativeMode is true, the TAG branch accumulates tag numbers into 'tags' as
94  	// it recurses, the SIMPLE branch captures the simple int into 'simpleValue', and 'nativeKind'
95  	// classifies the resulting token for format-agnostic consumers.
96  	private boolean nativeMode;
97  	private BinaryNativeKind nativeKind = BinaryNativeKind.NONE;
98  	private long[] tags = new long[4];
99  	private int tagCount;
100 	private int simpleValue;

147 	 * <p>

163 	 * Sets the maximum allowed wire-declared length (in bytes) for byte/text strings read by this cursor.
164 	 *
165 	 * <p>

169 	 * containers.  Values {@code <= 0} disable the maximum-length cap (the negative-length check still
170 	 * applies).

180 	// ==============================================================================================
181 	// State-machine summary.  CBOR is already a token-shaped wire format (RFC 8949 major types), so
182 	// next() reads the next data-type tag via CborInputStream.readDataType() and dispatches based
183 	// on three things: (1) whether we're at end-of-input, (2) whether a definite-length container
184 	// has exhausted its element count (emit END_*), or whether the data type is BREAK (also emit
185 	// END_*), and (3) within a map, whether we're at a key position or value position.
186 	//
187 	// Per-level container state lives on parallel stacks (stackRemaining + stackIsMap +
188 	// stackAwaitingKey, indexed by depth).  A stackRemaining value of -1 means indefinite-length
189 	// in that case the next() loop relies on CborInputStream returning DataType.BREAK instead of
190 	// counting elements.
191 	// ==============================================================================================
192 	@Override /* TokenReader */
193 	@SuppressWarnings({
194 		"java:S3776" // Token-reader state machine; complexity is inherent to the format decode loop.
195 	})
196 	public TokenType next() throws IOException, ParseException {
197 		if (ended) {
198 			currentToken = TokenType.END_OF_STREAM;
199 			return currentToken;
200 		}
201 		// Reset scratch.
202 		currentString = null;
203 		currentNumberLexeme = null;
204 		currentNumber = null;
205 		currentBinary = null;
206 		// Native-mode metadata: only reset at outermost entry — recursive tag unwraps preserve the
207 		// accumulated tag stack so nested tag(tag(value)) round-trips losslessly.
208 		if (tagNestingDepth == 0) {
209 			nativeKind = BinaryNativeKind.NONE;
210 			tagCount = 0;
211 		}
212 
213 		// Inside a definite-length container at element-count zero: emit END_*.
214 		if (depth > 0 && stackRemaining[depth - 1] == 0) {
215 			popContainer();
216 			return currentToken;
217 		}
218 
219 		DataType dt;
220 		try {
221 			dt = is.readDataType();
222 		} catch (IOException e) {
223 			if (depth == 0) {
224 				ended = true;
225 				currentToken = TokenType.END_OF_STREAM;
226 				return currentToken;
227 			}
228 			throw e;
229 		}
230 
231 		// Inside an indefinite-length container, BREAK closes it.
232 		if (dt == DataType.BREAK) {
233 			if (depth == 0 || stackRemaining[depth - 1] != INDEFINITE)
234 				throw new ParseException("Unexpected BREAK token outside indefinite-length container");
235 			popContainer();
236 			return currentToken;
237 		}
238 
239 		// Map key position: CBOR keys can be any data type, but the token vocabulary expects
240 		// FIELD_NAME with a string view.  Convert string keys directly; coerce other key types
241 		// to their toString() form (matches the databind path's behavior).
242 		if (depth > 0 && stackIsMap[depth - 1] && stackAwaitingKey[depth - 1]) {
243 			currentToken = TokenType.FIELD_NAME;
244 			currentString = readScalarAsString(dt);
245 			consumedOneElement();  // flips awaiting-key: key->value, decrements remaining count
246 			return currentToken;
247 		}
248 
249 		// Value position (in array, in map after key, or at root).
250 		switch (dt) {
251 			case ARRAY -> {
252 				pushContainer(false, is.isIndefinite() ? INDEFINITE : is.readLength());
253 				currentToken = TokenType.START_ARRAY;
254 			}
255 			case MAP -> {
256 				if (is.isIndefinite()) {
257 					pushContainer(true, INDEFINITE);
258 				} else {
259 					var n = is.readLength();
260 					// MAP length is pair-count; track it as 2*n element-counts.  Guard the doubling so a
261 					// malformed/overflowing count (e.g. a negative 8-byte 0xFFFFFFFFFFFFFFFF argument)
262 					// cannot wrap the element-count bookkeeping negative.
263 					if (n < 0 || n > Long.MAX_VALUE / 2)
264 						throw new ParseException("Invalid CBOR map length: %s", n);
265 					pushContainer(true, n * 2);
266 				}
267 				currentToken = TokenType.START_OBJECT;
268 				// First emit inside the map will be a key; pushContainer already set this
269 				// for map-level entries.
270 			}
271 			case UINT, NINT -> {
272 				var v = is.readSignedLong();
273 				currentNumber = v;
274 				currentNumberLexeme = Long.toString(v);
275 				currentToken = TokenType.VALUE_NUMBER;
276 				consumedOneElement();
277 			}
278 			case FLOAT -> {
279 				var v = is.readDouble();
280 				currentNumber = v;
281 				currentNumberLexeme = Double.toString(v);
282 				currentToken = TokenType.VALUE_NUMBER;
283 				consumedOneElement();
284 			}
285 			case BOOLEAN -> {
286 				currentBoolean = is.readBoolean();
287 				currentToken = TokenType.VALUE_BOOLEAN;
288 				consumedOneElement();
289 			}
290 			case NULL, UNDEFINED -> {
291 				currentToken = TokenType.VALUE_NULL;
292 				consumedOneElement();
293 			}
294 			case STRING -> {
295 				currentString = is.readString();
296 				currentToken = TokenType.VALUE_STRING;
297 				consumedOneElement();
298 			}
299 			case BINARY -> {
300 				currentBinary = is.readBinary();
301 				currentToken = TokenType.VALUE_BINARY;
302 				consumedOneElement();
303 			}
304 			case TAG, SIMPLE -> {
305 				// Q3: by default normalize to common scalars.  TAG: skip the tag and read the
306 				// next item as the wrapped value.  SIMPLE: emit as VALUE_NULL (best-effort).
307 				// In native mode (175ad): TAG numbers accumulate on the tag stack across
308 				// recursion; SIMPLE captures the simple int as metadata on the VALUE_NULL token.
309 				if (dt == DataType.TAG) {
310 					var n = is.readLength();
311 					if (nativeMode) {
312 						if (tagCount == tags.length)
313 							tags = java.util.Arrays.copyOf(tags, tags.length * 2);
314 						tags[tagCount++] = n;
315 						nativeKind = BinaryNativeKind.CBOR_TAG;
316 					}
317 					// Guard against pathologically deep tag chains blowing the JVM stack.
318 					if (tagNestingDepth >= maxTagNestingDepth)
319 						throw new ParseException(
320 							"CBOR tag nesting depth exceeded the maximum of %s.  Increase the limit via CborTokenReader.setMaxTagNestingDepth(int) if this is intentional.",
321 							maxTagNestingDepth);
322 					tagNestingDepth++;
323 					try {
324 						return next();  // consume the wrapped value
325 					} finally {
326 						tagNestingDepth--;
327 					}
328 				}
329 				if (nativeMode) {
330 					simpleValue = (int) is.readLength();
331 					nativeKind = BinaryNativeKind.CBOR_SIMPLE;
332 				}
333 				currentToken = TokenType.VALUE_NULL;
334 				consumedOneElement();
335 			}
336 			default -> throw new ParseException("Unexpected CBOR data type: %s", dt);
337 		}
338 		return currentToken;
339 	}

341 	private String readScalarAsString(DataType dt) throws IOException, ParseException {
342 		switch (dt) {
343 			case STRING:    return is.readString();
344 			case UINT, NINT: return Long.toString(is.readSignedLong());
345 			case BINARY: {
346 				var b = is.readBinary();
347 				return new String(b, java.nio.charset.StandardCharsets.UTF_8);
348 			}
349 			case BOOLEAN:   return Boolean.toString(is.readBoolean());
350 			case FLOAT:     return Double.toString(is.readDouble());
351 			default:
352 				throw new ParseException("Cannot use CBOR data type %s as a map key", dt);
353 		}
354 	}

356 	private void pushContainer(boolean isMap, long remaining) {
357 		if (depth == stackRemaining.length) {
358 			var n = depth * 2;
359 			var nr = new long[n];
360 			var nm = new boolean[n];
361 			var nk = new boolean[n];
362 			System.arraycopy(stackRemaining, 0, nr, 0, depth);
363 			System.arraycopy(stackIsMap, 0, nm, 0, depth);
364 			System.arraycopy(stackAwaitingKey, 0, nk, 0, depth);
365 			stackRemaining = nr;
366 			stackIsMap = nm;
367 			stackAwaitingKey = nk;
368 		}
369 		stackRemaining[depth] = remaining;
370 		stackIsMap[depth] = isMap;
371 		stackAwaitingKey[depth] = isMap;  // map starts awaiting a key
372 		depth++;
373 	}
374 
375 	private void popContainer() {
376 		depth--;
377 		currentToken = stackIsMap[depth] ? TokenType.END_OBJECT : TokenType.END_ARRAY;
378 		// The closed container counted as one value-emit in the parent's context.
379 		consumedOneElement();
380 	}
381 
382 	private void consumedOneElement() {
383 		if (depth == 0)
384 			return;
385 		var top = depth - 1;
386 		if (stackRemaining[top] != INDEFINITE)
387 			stackRemaining[top]--;
388 		// In a map, alternate key/value positions per element.
389 		if (stackIsMap[top])
390 			stackAwaitingKey[top] = !stackAwaitingKey[top];
391 	}

450 	@Override /* TokenReader */
451 	public boolean canRead() throws IOException, ParseException {
452 		// CBOR has no separators to advance through; the cursor is "at a value boundary" iff
453 		// (a) we're not inside a definite-length container that's exhausted, and
454 		// (b) inside an indefinite-length container, the next byte isn't BREAK, and
455 		// (c) the stream isn't ended.
456 		if (ended)
457 			return false;
458 		if (depth > 0) {
459 			var remaining = stackRemaining[depth - 1];
460 			if (remaining == 0)
461 				return false;
462 			if (remaining == INDEFINITE && is.peekBreak())
463 				return false;
464 		}
465 		return true;
466 	}

473 	@Override /* TokenReader */
474 	public <T> T read(ClassMeta<T> type) throws IOException, ParseException {
475 		var s = session();
476 		if (!canRead())
477 			throw new IllegalStateException(MSG_READ_NON_VALUE_STATE);
478 		try {
479 			T o = s.readAnything(type, is, null, null);
480 			consumedOneElement();
481 			currentToken = TokenType.NOT_AVAILABLE;
482 			return o;
483 		} catch (Exception e) {
484 			if (e instanceof IOException e2) throw e2;
485 			if (e instanceof ParseException e2) throw e2;
486 			throw new ParseException(s, "read failed: %s", e.getMessage());
487 		}
488 	}

492 		return read(session().<T>getClassMeta(type, args));
493 	}

497 			throw new UnsupportedOperationException(

503 	@Override /* TokenReader */
504 	public void skipChildren() throws IOException, ParseException {
505 		if (currentToken != TokenType.START_OBJECT && currentToken != TokenType.START_ARRAY)
506 			return;
507 		var startDepth = depth - 1;
508 		while (depth > startDepth) {
509 			var t = next();
510 			if (t == TokenType.END_OF_STREAM)
511 				throw new ParseException("Unexpected end of input while skipping children");
512 		}
513 	}
