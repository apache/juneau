17  package org.apache.juneau.marshall.swaps;
18  
19  import org.apache.juneau.marshall.*;
20  import org.apache.juneau.marshall.csv.*;
21  import org.apache.juneau.marshall.oapi.*;
22  import org.apache.juneau.marshall.parser.*;
23  import org.apache.juneau.marshall.serializer.*;
24  import org.apache.juneau.marshall.swap.*;
25  
26  /**
27   * Thin delegating swap for top-level {@code byte[]} values in textual serializers.
28   *
29   * <p>
30   * Used as a fall-back swap by {@link DefaultSwaps} for root-level / map-value /
31   * collection-element {@code byte[]} payloads that are not bean properties. Delegates to whichever
32   * {@link BinaryFormat} is configured on the
33   * {@link MarshallingContext}.
34   *
35   * <p>Disabled for:
36   * <ul>
37   * 	<li>Binary serializers / parsers with a native byte-array wire type —
38   * 		gated by {@link OutputStreamSerializerSession#hasNativeBytes()} /
39   * 		{@link InputStreamParserSession#hasNativeBytes()}.  Sessions that return {@code true}
40   * 		(MsgPack / CBOR / BSON) emit / consume native byte sequences and skip this swap.  Sessions
41   * 		that return {@code false} (Parquet, binary RDF) have no native byte-array primitive and
42   * 		still run this swap to produce / consume the configured text wire form.
43   * 	<li>CSV serializers / parsers ({@link CsvSerializerSession} / {@link CsvParserSession}) — CSV exposes its
44   * 		own per-cell {@code byteArrayFormat} setting via {@link CsvByteArrayCellFormat}.
45   * 	<li>{@code null}-session lookups (i.e. {@link ClassMeta}-level conversion plumbing) — keeps the
46   * 		String-to-{@code byte[]} fallback in {@code BasicConverter} / {@code MarshallingContext.findConversion}
47   * 		on the simple {@link String#getBytes(java.nio.charset.Charset)} path for non-binary inputs.
48   * </ul>
49   *
50   * <h5 class='section'>See Also:</h5><ul>
51   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/Swaps">Swap Basics</a>
52   * </ul>
53   */
54  public class BinarySwap extends StringSwap<byte[]> {
55  
56  	@Override /* Overridden from ObjectSwap */
57  	public int match(MarshallingSession session) {
58  		// Null sessions arrive from ClassMeta-level conversion lookup paths.  Returning 0 keeps the
59  		// default String-to-byte[] conversion (UTF-8 getBytes) intact for callers that aren't actually
60  		// trying to encode/decode a wire-format binary payload.
61  		if (session == null)
62  			return 0;
63  		// Binary serializers / parsers with a native byte-array wire type handle byte[] natively — skip
64  		// this swap so the underlying BSON / CBOR / MsgPack writer emits the raw byte sequence and the
65  		// corresponding parser consumes it.  Binary serializers without a native byte-array wire type
66  		// (Parquet, binary RDF) report hasNativeBytes()==false and still run this swap to produce /
67  		// consume the configured text wire form (HEX / BASE64 / etc.) per OQ 10 (b).
68  		if (session instanceof OutputStreamSerializerSession session2 && session2.hasNativeBytes())
69  			return 0;
70  		if (session instanceof InputStreamParserSession session2 && session2.hasNativeBytes())
71  			return 0;
72  		// CSV defers byte[] cell formatting to its own CsvByteArrayCellFormat-driven cell serializer/parser.
73  		if (session instanceof CsvSerializerSession || session instanceof CsvParserSession)
74  			return 0;
75  		// OpenAPI part serializers/parsers drive byte[] cell encoding from the {@code format} schema attribute
76  		// (BYTE / BINARY / BINARY_SPACED).  Skipping this swap leaves the raw byte[] in place so the
77  		// OpenAPI session can apply the schema-directed encoding directly.
78  		if (session instanceof OpenApiSerializerSession || session instanceof OpenApiParserSession)
79  			return 0;
80  		// NOT_SET means "no wire format configured" — leave byte[] in its native language form so the
81  		// surrounding serializer falls back to its built-in array representation (e.g. JSON array).
82  		if (session.getMarshallingContext().getBinaryFormat() == BinaryFormat.NOT_SET)
83  			return 0;
84  		return super.match(session);
85  	}
86  
87  	@Override /* Overridden from ObjectSwap */
88  	public String swap(MarshallingSession session, byte[] o) {
89  		return session.getMarshallingContext().getBinaryFormat().format(o);
90  	}
91  
92  	@Override /* Overridden from ObjectSwap */
93  	public byte[] unswap(MarshallingSession session, String o, ClassMeta<?> hint) {
94  		return session.getMarshallingContext().getBinaryFormat().parse(o);
95  	}
96  }
