137 	protected void doWrite(SerializerPipe pipe, Object o) throws IOException, SerializeException {
138 		try (var w = getMarkdownWriter(pipe)) {
139 			writeAnything(w, o, getExpectedRootType(o), null);

143 	/**
144 	 * Main serialization dispatch method.
145 	 *
146 	 * @param w The writer to write to.
147 	 * @param o The object to serialize.
148 	 * @param eType The expected type of the object.
149 	 * @param pMeta The bean property meta (may be null).
150 	 * @throws IOException Thrown by underlying stream.
151 	 * @throws SerializeException If serialization fails.
152 	 */
153 	protected void writeAnything(MarkdownWriter w, Object o, ClassMeta<?> eType, BeanPropertyMeta pMeta) throws IOException, SerializeException {
154 		if (o == null) {
155 			w.text(nullValue);
156 			return;
157 		}
158 
159 		ClassMeta cm = push2("root", o, eType);
160 		if (cm == null) {
161 			// Recursion detected
162 			pop();
163 			w.text(nullValue);
164 			return;
165 		}
166 
167 		ObjectSwap swap = cm.getSwap(this);
168 		if (swap != null) {
169 			o = swap(swap, o);
170 			cm = swap.getSwapClassMeta(this);
171 		}
172 
173 		if (cm.isOptional()) {
174 			var opt = (Optional<?>) o;
175 			if (opt.isEmpty()) {
176 				pop();
177 				w.text(nullValue);
178 			} else {
179 				writeAnything(w, opt.get(), eType, pMeta);
180 				pop();
181 			}
182 			return;
183 		}
184 
185 		if (cm.isBean()) {
186 			writeBeanMap(w, toBeanMap(o), eType, cm);
187 		} else if (cm.isMap()) {
188 			writeMap(w, (Map<?,?>) o, cm);
189 		} else if (cm.isCollectionOrArray()) {
190 			var l = cm.isArray() ? toObjectList(o) : (Collection<?>) o;
191 			writeCollection(w, l, eType, cm);
192 		} else if (cm.isStreamable()) {
193 			writeCollection(w, toListFromStreamable(o, cm), eType, cm);
194 		} else if (cm.isDate()) {
195 			w.text(writeDate((Date)o, cm));
196 		} else if (cm.isCalendar()) {
197 			w.text(writeCalendar(o, cm));
198 		} else if (cm.isTemporal()) {
199 			w.text(writeTemporal((TemporalAccessor)o, cm));
200 		} else if (cm.isDuration()) {
201 			w.text(writeDuration((Duration)o));
202 		} else if (cm.isPeriod()) {
203 			w.text(writePeriod((Period)o));
204 		} else {
205 			var s = toString(o);
206 			// Only use JSON5 wrapping for actual string/enum types where auto-detection could misinterpret
207 			// the value; non-bean objects (e.g. those with a valueOf(String) factory) use plain text so
208 			// that the parser can use convertToType() on the raw string.
209 			if ((cm.isCharSequence() || cm.isEnum()) && isAmbiguousString(s, nullValue))
210 				w.text("`'" + escapeJson5String(s) + "'`");
211 			else
212 				w.text(MarkdownWriter.escapeText(s));
213 		}
214 		
215 		pop();
216 	}

218 	/**
219 	 * Serializes a bean as a 2-column key/value Markdown table.
220 	 *
221 	 * @param w The writer.
222 	 * @param bm The bean map.
223 	 * @param eType The expected type (for type name calculation).
224 	 * @param aType The actual type (for type name calculation).
225 	 * @throws IOException Thrown by underlying stream.
226 	 * @throws SerializeException If serialization fails.
227 	 */
228 	protected void writeBeanMap(MarkdownWriter w, BeanMap<?> bm, ClassMeta<?> eType, ClassMeta<?> aType) throws IOException, SerializeException {
229 		if (showHeaders) {
230 			w.tableHeader("Property", "Value");
231 			w.tableSeparator(2);
232 		}
233 		
234 		// Add _type row if needed for polymorphic type identification
235 		// This handles both addBeanTypes (eType != aType) and addRootType (root object with dict name)
236 		String typeName = getBeanTypeName(this, eType, aType, null);
237 		if (typeName != null) {
238 			w.tableRow("_type", typeName);
239 		}
240 		
241 		for (var entry : bm.entrySet()) {
242 			if (entry instanceof BeanMapEntry entry2 && !entry2.getMeta().canRead())
243 				continue;
244 			w.tableRow(MarkdownWriter.escapeCell(entry.getKey()), writeInlineValue(entry.getValue()));
245 		}
246 	}

248 	/**
249 	 * Serializes a Map as a 2-column key/value Markdown table.
250 	 *
251 	 * @param w The writer.
252 	 * @param m The map.
253 	 * @param cm The class meta.
254 	 * @throws IOException Thrown by underlying stream.
255 	 * @throws SerializeException If serialization fails.
256 	 */
257 	protected void writeMap(MarkdownWriter w, Map<?,?> m, ClassMeta<?> cm) throws IOException, SerializeException {
258 		if (showHeaders) {
259 			w.tableHeader("Key", "Value");
260 			w.tableSeparator(2);
261 		}
262 		for (var entry : m.entrySet()) {
263 			String rawKey = entry.getKey() == null ? null : toString(entry.getKey());
264 			String keyStr;
265 			if (rawKey == null)
266 				keyStr = nullValue;
267 			else if (isAmbiguousString(rawKey, nullValue))
268 				keyStr = "`'" + escapeJson5String(rawKey) + "'`";
269 			else
270 				keyStr = MarkdownWriter.escapeCell(rawKey);
271 			w.tableRow(keyStr, writeInlineValue(entry.getValue()));
272 		}
273 	}

275 	/**
276 	 * Serializes a collection as either a multi-column table (uniform beans/maps) or a bulleted list.
277 	 *
278 	 * @param w The writer.
279 	 * @param c The collection.
280 	 * @param eType The expected type.
281 	 * @param cm The class meta.
282 	 * @throws IOException Thrown by underlying stream.
283 	 * @throws SerializeException If serialization fails.
284 	 */
285 	protected void writeCollection(MarkdownWriter w, Collection<?> c, ClassMeta<?> eType, ClassMeta<?> cm) throws IOException, SerializeException {
286 		if (c.isEmpty())
287 			return;
288 
289 		var headers = getTableHeaders(c);
290 		if (headers != null) {
291 			// Uniform collection of beans or maps → multi-column table
292 			
293 			// Determine if we need a _type column
294 			boolean needsTypeColumn = isAddBeanTypes();
295 			var elementType = eType != null && eType.isCollectionOrArray() ? eType.getElementType() : cm.getElementType();
296 
297 			// Add _type column to headers if needed
298 			String[] finalHeaders = headers;
299 			if (needsTypeColumn) {
300 				finalHeaders = new String[headers.length + 1];
301 				finalHeaders[0] = "_type";
302 				System.arraycopy(headers, 0, finalHeaders, 1, headers.length);
303 			}
304 
305 			if (showHeaders && finalHeaders.length > 0) {
306 				w.tableHeader(finalHeaders);
307 				w.tableSeparator(finalHeaders.length);
308 			}
309 
310 			for (var item : c) {
311 				if (item == null) {
312 					var row = new String[finalHeaders.length];
313 					Arrays.fill(row, nullValue);
314 					w.tableRow(row);
315 					continue;
316 				}
317 				ClassMeta itemCm = getClassMetaForObject(item);
318 				ObjectSwap itemSwap = itemCm.getSwap(this);
319 				var swapped = itemSwap != null ? swap(itemSwap, item) : item;
320 				ClassMeta swappedCm = itemSwap != null ? itemSwap.getSwapClassMeta(this) : itemCm;
321 
322 				var row = new String[finalHeaders.length];
323 				int colOffset = needsTypeColumn ? 1 : 0;
324 
325 				// Add type name if needed
326 				// For root-level collections with addRootType, treat each element as root
327 				if (needsTypeColumn) {
328 					String typeName;
329 					if (isRoot() && isAddRootType()) {
330 						// Root-level collection: add type name if the bean has a dictionary name
331 						typeName = swappedCm.getBeanDictionaryName();
332 					} else {
333 						// Nested collection: use standard logic
334 						typeName = getBeanTypeName(this, elementType, swappedCm, null);
335 					}
336 					row[0] = or(typeName, "");
337 				}
338 
339 				if (swappedCm.isBean()) {
340 					var bm = toBeanMap(swapped);
341 					for (var i = 0; i < headers.length; i++) {
342 						var val = bm.get(headers[i]);
343 						row[i + colOffset] = writeInlineValue(val);
344 					}
345 				} else if (swappedCm.isMap()) {
346 					var map = (Map<?,?>) swapped;
347 					for (var i = 0; i < headers.length; i++) {
348 						var val = map.get(headers[i]);
349 						row[i + colOffset] = writeInlineValue(val);
350 					}
351 				}
352 				w.tableRow(row);
353 			}
354 		} else {
355 			// Mixed or simple values → bulleted list
356 			for (var item : c)
357 				w.bulletItem(0, writeInlineValue(item));
358 		}
359 	}

361 	/**
362 	 * Returns true if the string could be misinterpreted during parsing when the target type is Object,
363 	 * or if the string contains characters that would be lost or corrupted in Markdown table cells.
364 	 *
365 	 * <p>Strings that are empty, equal to nullValue, look like booleans or numbers, or contain control
366 	 * characters (< 32 or = 127) are considered ambiguous and should be wrapped in JSON5 backtick syntax.
367 	 */
368 	private static boolean isAmbiguousString(String s, String nullValue) {
369 		if (s.isEmpty()) return true;
370 		if (s.equals(nullValue)) return true;
371 		if ("true".equals(s) || "false".equals(s)) return true;
372 		for (int i = 0; i < s.length(); i++) {
373 			char c = s.charAt(i);
374 			if (c < 32 || c == 127) return true;
375 		}
376 		try { Integer.parseInt(s); return true; } catch (@SuppressWarnings("unused") NumberFormatException ignored) { /* Not a number; fall through */ }
377 		try { Long.parseLong(s); return true; } catch (@SuppressWarnings("unused") NumberFormatException ignored) { /* Not a number; fall through */ }
378 		try { Double.parseDouble(s); return true; } catch (@SuppressWarnings("unused") NumberFormatException ignored) { /* Not a number; fall through */ }
379 		return false;
380 	}

382 	/**
383 	 * Escapes a string for use within a JSON5 single-quoted string literal.
384 	 */
385 	private static String escapeJson5String(String s) {
386 		if (s == null) return "";
387 		var sb = new StringBuilder(s.length() + 4);
388 		for (var i = 0; i < s.length(); i++) {
389 			var c = s.charAt(i);
390 			switch (c) {
391 				case '\'' -> sb.append("\\'");
392 				case '\\' -> sb.append("\\\\");
393 				case '\n' -> sb.append("\\n");
394 				case '\r' -> sb.append("\\r");
395 				case '\t' -> sb.append("\\t");
396 				default -> sb.append(c);
397 			}
398 		}
399 		return sb.toString();
400 	}

403 	 * Converts an array (including primitive arrays) to a List of boxed objects.
404 	 */
405 	private static List<Object> toObjectList(Object array) {
406 		int len = Array.getLength(array);
407 		var list = new ArrayList<>(len);
408 		for (int i = 0; i < len; i++)
409 			list.add(Array.get(array, i));
410 		return list;
411 	}

414 	 * Returns a Json5Serializer that shares this session's bean context (swaps, implClasses, dictionary).
415 	 * Creates a new serializer each time to ensure settings are current.
416 	 */
417 	private JsonSerializer getJson5Serializer() {
418 		var b = Json5Serializer.create().marshallingContext((MarshallingContext) getContext());
419 		if (isAddBeanTypes())
420 			b = b.addBeanTypes();
421 		if (isAddRootType())
422 			b = b.addRootType();
423 		if (isTrimStrings())
424 			b = b.trimStrings();
425 		return b.build();
426 	}

428 	/**
429 	 * Serializes a value for placement in a Markdown table cell.
430 	 *
431 	 * <p>
432 	 * Simple values (strings, numbers, booleans, enums, dates) are escaped and returned inline.
433 	 * Complex values (beans, maps, collections) are serialized as JSON5 in backticks.
434 	 *
435 	 * @param o The value to serialize.
436 	 * @return The cell string value.
437 	 * @throws SerializeException If serialization fails.
438 	 */
439 	protected String writeInlineValue(Object o) throws SerializeException {
440 		if (o == null)
441 			return nullValue;
442 
443 		ClassMeta cm = getClassMetaForObject(o);
444 		ObjectSwap swap = cm.getSwap(this);
445 		if (swap != null) {
446 			o = swap(swap, o);
447 			cm = swap.getSwapClassMeta(this);
448 		}
449 
450 		if (cm.isCharSequence() || cm.isEnum()) {
451 			var s = toString(o);
452 			if (isAmbiguousString(s, nullValue))
453 				return "`'" + escapeJson5String(s) + "'`";
454 			return MarkdownWriter.escapeCell(s);
455 		}
456 
457 		if (cm.isNumber() || cm.isBoolean())
458 			return o.toString();
459 
460 		if (cm.isDate())
461 			return MarkdownWriter.escapeCell(writeDate((Date)o, cm));
462 
463 		if (cm.isCalendar())
464 			return MarkdownWriter.escapeCell(writeCalendar(o, cm));
465 
466 		if (cm.isTemporal())
467 			return MarkdownWriter.escapeCell(writeTemporal((TemporalAccessor)o, cm));
468 
469 		if (cm.isDuration())
470 			return MarkdownWriter.escapeCell(writeDuration((Duration)o));
471 
472 		if (cm.isPeriod())
473 			return MarkdownWriter.escapeCell(writePeriod((Period)o));
474 
475 		// Complex value: serialize as JSON5 in backticks — use context-aware serializer for swaps
476 		try {
477 			var json5 = getJson5Serializer().write(o);
478 			return "`" + json5 + "`";
479 		} catch (@SuppressWarnings("unused") SerializeException e) {
480 			return MarkdownWriter.escapeCell(o.toString());
481 		}
482 	}

484 	/**
485 	 * Determines if a collection can be rendered as a multi-column table.
486 	 *
487 	 * <p>
488 	 * Returns the column header names if the collection is a uniform set of beans or maps,
489 	 * or {@code null} if it should be rendered as a bulleted list.
490 	 *
491 	 * @param c The collection to inspect.
492 	 * 	<br>Must not be <jk>null</jk>.
493 	 * @return The column header names, or {@code null} for list rendering.
494 	 * @throws SerializeException If inspection fails.
495 	 */
496 	@SuppressWarnings({
497 		"java:S1168" // null signals list rendering; empty array would incorrectly trigger table mode
498 	})
499 	protected String[] getTableHeaders(Collection<?> c) throws SerializeException {
500 		if (c.isEmpty())
501 			return null;
502 
503 		// Find first non-null element
504 		Object first = null;
505 		for (var item : c) {
506 			if (item != null) {
507 				first = item;
508 				break;
509 			}
510 		}
511 		if (first == null)
512 			return null;
513 
514 		ClassMeta cm = getClassMetaForObject(first);
515 		ObjectSwap swap = cm.getSwap(this);
516 		if (swap != null) {
517 			try {
518 				first = swap(swap, first);
519 				cm = swap.getSwapClassMeta(this);
520 			} catch (@SuppressWarnings("unused") SerializeException e) {
521 				return null;
522 			}
523 		}
524 
525 		if (!cm.isMapOrBean())
526 			return null;
527 
528 		if (cm.isBean()) {
529 			var bm = toBeanMap(first);
530 			return bm.keySet().toArray(new String[0]);
531 		}
532 
533 		// Non-bean map: collect keys from first entry
534 		if (first instanceof Map<?,?> map2) {
535 			return map2.keySet().stream().map(Object::toString).toArray(String[]::new);
536 		}
537 
538 		return null;
539 	}

542 	 * Gets or creates a {@link MarkdownWriter} from the given pipe.
543 	 *
544 	 * @param out The serializer pipe.
545 	 * 	<br>Must not be <jk>null</jk>.
546 	 * @return The writer.
547 	 */
548 	protected MarkdownWriter getMarkdownWriter(SerializerPipe out) {
549 		var output = out.getRawOutput();
550 		if (output instanceof MarkdownWriter output2)
551 			return output2;
552 		var w = new MarkdownWriter(out.getWriter(), isUseWhitespace(), getMaxIndent(), isTrimStrings(), getUriResolver());
553 		out.setWriter(w);
554 		return w;
