33  /**
34   * Parses Markdown fragment-mode output (tables and lists) back to POJOs.
35   *
36   * <h5 class='topic'>Media types</h5>
37   * <p>
38   * Handles <c>Content-Type</c> types:  <bc>text/markdown, text/x-markdown</bc>
39   *
40   * <h5 class='topic'>Description</h5>
41   * <p>
42   * Parses Markdown tables and bulleted lists produced by {@link MarkdownSerializer} back into Java objects.
43   * The following mappings apply:
44   * <ul class='spaced-list'>
45   * 	<li>2-column key/value tables are parsed as beans or {@link Map}s.
46   * 	<li>Multi-column tables are parsed as {@link List}s of beans or {@link Map}s.
47   * 	<li>Bulleted lists are parsed as {@link List}s.
48   * 	<li>Table cells containing backtick-wrapped JSON5 values are parsed via {@link Json5Parser}.
49   * </ul>
50   *
51   * <h5 class='section'>Example:</h5>
52   * <p class='bjava'>
53   * 	<jc>// Parse a Markdown table to a bean</jc>
54   * 	String <jv>md</jv> = <js>"| Property | Value |\n|---|---|\n| name | Alice |\n| age | 30 |"</js>;
55   * 	Person <jv>p</jv> = MarkdownParser.<jsf>DEFAULT</jsf>.read(<jv>md</jv>, Person.<jk>class</jk>);
56   * </p>
57   * <p class='bjava'>
58   * 	<jc>// Round-trip example</jc>
59   * 	String <jv>md</jv> = MarkdownSerializer.<jsf>DEFAULT</jsf>.write(<jv>myBean</jv>);
60   * 	MyBean <jv>parsed</jv> = MarkdownParser.<jsf>DEFAULT</jsf>.read(<jv>md</jv>, MyBean.<jk>class</jk>);
61   * </p>
62   *
63   * <h5 class='section'>Limitations:</h5><ul>
64   * 	<li class='note'>Multi-line string values cannot round-trip; newlines in cells are literal {@code <br>} text.
65   * 	<li class='note'>The configured null value string (default: {@code *null*}) is interpreted as {@code null}, not as the literal string.
66   * 	<li class='note'>Without a target type, table cell values are auto-detected as Integer, Long, Double, Boolean, or String.
67   * 	<li class='note'>Type ambiguity: without a target type, tables return {@code Map<String,Object>} (key/value) or {@code List<Map>} (multi-column).
68   * 	<li class='note'>This class is thread safe and reusable.
69   * </ul>
70   *
71   * <h5 class='section'>See Also:</h5><ul>
72   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/Markdown">Markdown Basics</a>
73   * </ul>
74   */

82  	private static final String ARG_copyFrom = "copyFrom";
83  	private static final String CONST_null = "*null*";

85  	/**
86  	 * Builder class.
87  	 */
88  	@SuppressWarnings({
89  		"java:S119" // 'SELF' (CRTP self-type) is intentional and clearer than a single-letter name.
90  	})
91  	public abstract static class Builder<SELF extends Builder<SELF>> extends ReaderParser.Builder<SELF> {
92  
93  		private static final Cache<HashKey,MarkdownParser> CACHE = Cache.of(HashKey.class, MarkdownParser.class).build();
94  
95  		String nullValue;
96  
97  		/**
98  		 * Constructor, default settings.
99  		 */
100 		protected Builder() {
101 			consumes("text/markdown,text/x-markdown");
102 			nullValue = CONST_null;
103 		}
104 
105 		/**
106 		 * Copy constructor.
107 		 *
108 		 * @param copyFrom The builder to copy from.
109 		 * 	<br>Cannot be <jk>null</jk>.
110 		 */
111 		protected Builder(Builder<?> copyFrom) {
112 			super(assertArgNotNull(ARG_copyFrom, copyFrom));
113 			nullValue = copyFrom.nullValue;
114 		}
115 
116 		/**
117 		 * Copy constructor.
118 		 *
119 		 * @param copyFrom The bean to copy from.
120 		 * 	<br>Cannot be <jk>null</jk>.
121 		 */
122 		protected Builder(MarkdownParser copyFrom) {
123 			super(assertArgNotNull(ARG_copyFrom, copyFrom));
124 			nullValue = copyFrom.nullValue;
125 		}
126 
127 		/**
128 		 * The string that is interpreted as {@code null} when parsing cell values.
129 		 *
130 		 * <p>
131 		 * Must match the value configured on the serializer. Default is {@code *null*}.
132 		 *
133 		 * @param value The null marker string.
134 		 * @return This object.
135 		 */
136 		public SELF nullValue(String value) {
137 			nullValue = value == null ? CONST_null : value;
138 			return self();
139 		}
140 
141 		@Override /* Overridden from Context.Builder<?> */
142 		public HashKey hashKey() {
143 			return HashKey.of(super.hashKey(), nullValue);
144 		}
145 
146 		@Override /* Overridden from Context.Builder<?> */
147 		public MarkdownParser build() {
148 			return cache(CACHE).build(MarkdownParser.class);
149 		}
150 
151 		@Override /* Overridden from Context.Builder<?> */
152 		public abstract SELF copy();
153 
154 
155 	}

163 
164 		DefaultBuilder(MarkdownParser copyFrom) {

167 
168 		DefaultBuilder(Builder<?> copyFrom) {
169 			super(copyFrom);
170 		}
171 
172 		@Override /* Overridden from Context.Builder<?> */
173 		public DefaultBuilder copy() {
174 			return new DefaultBuilder(this);
175 		}
176 	}

183 	/**
184 	 * Creates a new builder for this object.
185 	 *
186 	 * @return A new builder.

243 	/**
244 	 * Convenience delegator that opens a {@link RecordReader} over the input using
245 	 * <b>default session arguments</b> (mirrors {@link #read(Object, Class)}).
246 	 *
247 	 * <p>
248 	 * The real implementation lives on {@link MarkdownParserSession#readRecords(Object)}.  Callers
249 	 * that need request-derived configuration (locale, timezone, schema, swaps) should call
250 	 * {@link #createSession()} and invoke {@link MarkdownParserSession#readRecords(Object)} on the
251 	 * built session instead.
252 	 *
253 	 * @param input The input.
254 	 * @return A new {@link RecordReader} cursor.
255 	 * @throws IOException If a problem occurred opening the underlying input.
256 	 */
257 	@Override /* RecordReadable */
258 	public RecordReader readRecords(Object input) throws IOException {
259 		return getSession().readRecords(input);
260 	}
