32  /**
33   * Serializes POJO models to Markdown (fragment mode: tables and lists).
34   *
35   * <h5 class='topic'>Media types</h5>
36   * <p>
37   * Handles <c>Accept</c> types:  <bc>text/markdown, text/x-markdown</bc>
38   * <p>
39   * Produces <c>Content-Type</c> types:  <bc>text/markdown</bc>
40   *
41   * <h5 class='topic'>Description</h5>
42   * <p>
43   * The conversion is as follows:
44   * <ul class='spaced-list'>
45   * 	<li>Beans and Maps are serialized as 2-column key/value Markdown tables.
46   * 	<li>Collections and arrays of beans/maps with uniform structure are serialized as multi-column Markdown tables.
47   * 	<li>Collections and arrays of simple values are serialized as Markdown bulleted lists.
48   * 	<li>Nested beans and complex values in table cells are serialized as inline JSON5 wrapped in backticks.
49   * 	<li>Null values are rendered as <js>"*null*"</js> (italic) by default.
50   * </ul>
51   *
52   * <h5 class='section'>Example:</h5>
53   * <p class='bjava'>
54   * 	<jc>// Use the default serializer to serialize a bean</jc>
55   * 	String <jv>md</jv> = MarkdownSerializer.<jsf>DEFAULT</jsf>.write(<jv>someObject</jv>);
56   *
57   * 	<jc>// Create a custom serializer</jc>
58   * 	MarkdownSerializer <jv>serializer</jv> = MarkdownSerializer.<jsm>create</jsm>().nullValue(<js>"—"</js>).build();
59   *
60   * 	<jc>// Serialize a List of beans to a multi-column Markdown table</jc>
61   * 	String <jv>md</jv> = <jv>serializer</jv>.write(<jv>listOfBeans</jv>);
62   * </p>
63   *
64   * <h5 class='figure'>Example output (bean):</h5>
65   * <p class='bcode'>
66   * 	| Property | Value |
67   * 	|---|---|
68   * 	| name | Alice |
69   * 	| age | 30 |
70   * </p>
71   *
72   * <h5 class='section'>Limitations:</h5><ul>
73   * 	<li class='note'>Multi-line string values cannot round-trip; newlines are replaced with {@code <br>}.
74   * 	<li class='note'>The configured null value string (default: {@code *null*}) cannot be stored as a literal string value.
75   * 	<li class='note'>This serializer is thread safe and reusable.
76   * </ul>
77   *
78   * <h5 class='section'>See Also:</h5><ul>
79   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/Markdown">Markdown Basics</a>
80   * </ul>
81   */
82  @SuppressWarnings({
83  	"java:S110", // Inheritance depth acceptable for this class hierarchy

87  public class MarkdownSerializer extends WriterSerializer implements MarkdownMetaProvider, RecordWritable {

89  	private static final String ARG_copyFrom = "copyFrom";
90  	private static final String CONST_null = "*null*";
91  
92  	/**
93  	 * Builder class.
94  	 */
95  	@SuppressWarnings({
96  		"java:S119" // 'SELF' (CRTP self-type) is intentional and clearer than a single-letter name.
97  	})
98  	public abstract static class Builder<SELF extends Builder<SELF>> extends WriterSerializer.Builder<SELF> {
99  
100 		private static final Cache<HashKey,MarkdownSerializer> CACHE = Cache.of(HashKey.class, MarkdownSerializer.class).build();
101 
102 		String nullValue;
103 		boolean showHeaders;
104 
105 		/**
106 		 * Constructor, default settings.
107 		 */
108 		protected Builder() {
109 			produces("text/markdown");
110 			accept("text/markdown,text/x-markdown");
111 			nullValue = CONST_null;
112 			showHeaders = true;
113 		}
114 
115 		/**
116 		 * Copy constructor.
117 		 *
118 		 * @param copyFrom The builder to copy from.
119 		 * 	<br>Cannot be <jk>null</jk>.
120 		 */
121 		protected Builder(Builder<?> copyFrom) {
122 			super(assertArgNotNull(ARG_copyFrom, copyFrom));
123 			nullValue = copyFrom.nullValue;
124 			showHeaders = copyFrom.showHeaders;
125 		}
126 
127 		/**
128 		 * Copy constructor.
129 		 *
130 		 * @param copyFrom The bean to copy from.
131 		 * 	<br>Cannot be <jk>null</jk>.
132 		 */
133 		protected Builder(MarkdownSerializer copyFrom) {
134 			super(assertArgNotNull(ARG_copyFrom, copyFrom));
135 			nullValue = copyFrom.nullValue;
136 			showHeaders = copyFrom.showHeaders;
137 		}
138 
139 		/**
140 		 * The string to render for null values.
141 		 *
142 		 * <p>
143 		 * Default is {@code *null*} (rendered as Markdown italic).
144 		 *
145 		 * @param value The null marker string.
146 		 * @return This object.
147 		 */
148 		public SELF nullValue(String value) {
149 			nullValue = value == null ? CONST_null : value;
150 			return self();
151 		}
152 
153 		/**
154 		 * Whether to show table header rows.
155 		 *
156 		 * <p>
157 		 * Default is <jk>true</jk>.
158 		 *
159 		 * @param value Whether to show headers.
160 		 * @return This object.
161 		 */
162 		public SELF showHeaders(boolean value) {
163 			showHeaders = value;
164 			return self();
165 		}
166 
167 		@Override /* Overridden from Context.Builder<?> */
168 		public HashKey hashKey() {
169 			return HashKey.of(super.hashKey(), nullValue, showHeaders);
170 		}
171 
172 		@Override /* Overridden from Context.Builder<?> */
173 		public MarkdownSerializer build() {
174 			return cache(CACHE).build(MarkdownSerializer.class);
175 		}
176 
177 		@Override /* Overridden from Context.Builder<?> */
178 		public abstract SELF copy();
179 
180 
181 	}

221 	private final Map<ClassMeta<?>,MarkdownClassMeta> markdownClassMetas = new ConcurrentHashMap<>();

223 	final String nullValue;

265 	@Override /* Overridden from MarkdownMetaProvider */
266 	public MarkdownBeanPropertyMeta getMarkdownBeanPropertyMeta(BeanPropertyMeta bpm) {
267 		if (bpm == null)
268 			return MarkdownBeanPropertyMeta.DEFAULT;
269 		return markdownBeanPropertyMetas.computeIfAbsent(bpm, k -> new MarkdownBeanPropertyMeta(k.getDelegateFor(), this));
270 	}
271 
272 	@Override /* Overridden from MarkdownMetaProvider */
273 	public MarkdownClassMeta getMarkdownClassMeta(ClassMeta<?> cm) {
274 		return markdownClassMetas.computeIfAbsent(cm, k -> new MarkdownClassMeta(k, this));
275 	}

281 	 * Convenience delegator that opens a {@link RecordWriter} over the output using

284 	 * <p>
285 	 * The real implementation lives on {@link MarkdownSerializerSession#writeRecords(Object)}.
286 	 * Callers that need request-derived configuration should call {@link #createSession()} and
287 	 * invoke {@link MarkdownSerializerSession#writeRecords(Object)} on the built session instead.
288 	 *
289 	 * @param output The output.
290 	 * @return A new {@link RecordWriter}.
291 	 * @throws IOException If a problem occurred opening the underlying output.
292 	 */
293 	@Override /* RecordWritable */
294 	public RecordWriter writeRecords(Object output) throws IOException {
295 		return getSession().writeRecords(output);
296 	}
