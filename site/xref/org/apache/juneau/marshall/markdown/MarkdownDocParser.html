17  package org.apache.juneau.marshall.markdown;
18  
19  import static org.apache.juneau.commons.utils.AssertionUtils.*;
20  
21  import org.apache.juneau.commons.collections.*;
22  
23  /**
24   * Parses Markdown document-mode output (headings + tables) back to POJOs.
25   *
26   * <h5 class='topic'>Media types</h5>
27   * <p>
28   * Handles <c>Content-Type</c> types:  <bc>text/markdown, text/x-markdown</bc>
29   *
30   * <h5 class='topic'>Description</h5>
31   * <p>
32   * Extends {@link MarkdownParser} to parse Markdown documents produced by {@link MarkdownDocSerializer}.
33   * Heading sections are mapped back to nested bean properties based on heading text matching property names.
34   *
35   * <h5 class='section'>Example:</h5>
36   * <p class='bjava'>
37   * 	<jc>// Parse a Markdown document back to a bean</jc>
38   * 	String <jv>md</jv> = <js>"# Person\n\n| Property | Value |\n|---|---|\n| name | Alice |\n\n## address\n\n| Property | Value |\n|---|---|\n| city | Boston |"</js>;
39   * 	Person <jv>p</jv> = MarkdownDocParser.<jsf>DEFAULT</jsf>.read(<jv>md</jv>, Person.<jk>class</jk>);
40   * </p>
41   * <p class='bjava'>
42   * 	<jc>// Round-trip example</jc>
43   * 	String <jv>md</jv> = MarkdownDocSerializer.<jsf>DEFAULT</jsf>.write(<jv>myBean</jv>);
44   * 	MyBean <jv>parsed</jv> = MarkdownDocParser.<jsf>DEFAULT</jsf>.read(<jv>md</jv>, MyBean.<jk>class</jk>);
45   * </p>
46   *
47   * <h5 class='section'>Limitations:</h5><ul>
48   * 	<li class='note'>Heading text must match bean property names exactly (case-sensitive).
49   * 	<li class='note'>Heading levels must be consistent with serializer output; manually edited heading levels may not parse correctly.
50   * 	<li class='note'>A heading followed by a bulleted list is treated as a {@code List&lt;String&gt;} property.
51   * 	<li class='note'>A heading followed by a multi-column table is treated as a {@code List&lt;Bean&gt;} property.
52   * 	<li class='note'>The top-level heading (title) is ignored; only sub-headings map to properties.
53   * 	<li class='note'>This class is thread safe and reusable.
54   * </ul>
55   *
56   * <h5 class='section'>See Also:</h5><ul>
57   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/Markdown">Markdown Basics</a>
58   * </ul>
59   */
60  @SuppressWarnings({
61  	"java:S110", // Inheritance depth acceptable for this class hierarchy
62  	"java:S115", // Constants use UPPER_snakeCase naming convention
63  })
64  public class MarkdownDocParser extends MarkdownParser {
65  
66  	private static final String ARG_copyFrom = "copyFrom";
67  
68  	/**
69  	 * Builder class.
70  	 */
71  	public static class Builder extends MarkdownParser.Builder<Builder> {
72  
73  		private static final Cache<HashKey,MarkdownDocParser> CACHE = Cache.of(HashKey.class, MarkdownDocParser.class).build();
74  
75  		int headingLevel;
76  
77  		/**
78  		 * Constructor, default settings.
79  		 */
80  		protected Builder() {
81  			headingLevel = 1;
82  			type(MarkdownDocParser.class);
83  		}
84  
85  		/**
86  		 * Copy constructor.
87  		 *
88  		 * @param copyFrom The builder to copy from.
89  		 * 	<br>Cannot be <jk>null</jk>.
90  		 */
91  		protected Builder(Builder copyFrom) {
92  			super(assertArgNotNull(ARG_copyFrom, copyFrom));
93  			headingLevel = copyFrom.headingLevel;
94  		}
95  
96  		/**
97  		 * Copy constructor.
98  		 *
99  		 * @param copyFrom The bean to copy from.
100 		 * 	<br>Cannot be <jk>null</jk>.
101 		 */
102 		protected Builder(MarkdownDocParser copyFrom) {
103 			super(assertArgNotNull(ARG_copyFrom, copyFrom));
104 			headingLevel = copyFrom.headingLevel;
105 		}
106 
107 		/**
108 		 * The expected starting heading level for the document.
109 		 *
110 		 * <p>
111 		 * Default is 1 ({@code # Heading}). Must match the level used by the serializer.
112 		 *
113 		 * @param value The heading level.
114 		 * @return This object.
115 		 */
116 		public Builder headingLevel(int value) {
117 			headingLevel = Math.min(6, Math.max(1, value));
118 			return this;
119 		}
120 
121 		@Override /* Overridden from Context.Builder<?> */
122 		public HashKey hashKey() {
123 			return HashKey.of(super.hashKey(), headingLevel);
124 		}
125 
126 		@Override /* Overridden from Context.Builder<?> */
127 		public MarkdownDocParser build() {
128 			return cache(CACHE).build(MarkdownDocParser.class);
129 		}
130 
131 		@Override /* Overridden from Context.Builder<?> */
132 		public Builder copy() {
133 			return new Builder(this);
134 		}
135 
136 	}
137 
138 	/**
139 	 * Default parser, all default settings.
140 	 */
141 	public static final MarkdownDocParser DEFAULT = new MarkdownDocParser(create());
142 
143 	/**
144 	 * Creates a new builder for this object.
145 	 *
146 	 * @return A new builder.
147 	 */
148 	public static Builder create() {
149 		return new Builder();
150 	}
151 
152 	final int headingLevel;
153 
154 	/**
155 	 * Constructor.
156 	 *
157 	 * @param builder The builder for this object.
158 	 */
159 	public MarkdownDocParser(Builder builder) {
160 		super(builder);
161 		headingLevel = builder.headingLevel;
162 	}
163 
164 	@Override /* Overridden from Context */
165 	public Builder copy() {
166 		return new Builder(this);
167 	}
168 
169 	@Override /* Overridden from Context */
170 	public MarkdownDocParserSession.Builder createSession() {
171 		return MarkdownDocParserSession.create(this);
172 	}
173 
174 	@Override /* Overridden from Context */
175 	public MarkdownDocParserSession getSession() { return createSession().build(); }
176 }
