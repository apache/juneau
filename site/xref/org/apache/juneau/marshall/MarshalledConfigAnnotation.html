17  package org.apache.juneau.marshall;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.commons.http.*;
22  import org.apache.juneau.commons.reflect.*;
23  import org.apache.juneau.commons.svl.*;
24  
25  /**
26   * Utility classes and methods for the {@link MarshalledConfig @MarshalledConfig} annotation.
27   *
28   */
29  public class MarshalledConfigAnnotation {
30  
31  	/**
32  	 * Prevents instantiation.
33  	 */
34  	private MarshalledConfigAnnotation() {}
35  
36  	/**
37  	 * Applies {@link MarshalledConfig} annotations to a {@link MarshallingContext.Builder}.
38  	 */
39  	public static class Applier extends AnnotationApplier<MarshalledConfig,MarshallingContext.Builder> {
40  
41  		/**
42  		 * Constructor.
43  		 *
44  		 * @param vr The resolver for resolving values in annotations. Must not be <jk>null</jk>.
45  		 */
46  		public Applier(VarResolverSession vr) {
47  			super(MarshalledConfig.class, MarshallingContext.Builder.class, vr);
48  		}
49  
50  		@Override
51  		@SuppressWarnings({
52  			"java:S3776" // Cognitive complexity acceptable for the flat per-attribute config-applier sequence.
53  		})
54  		public void apply(AnnotationInfo<MarshalledConfig> ai, MarshallingContext.Builder b) {
55  			MarshalledConfig a = ai.inner();
56  
57  			// @formatter:off
58  			bool(a.debug()).ifPresent(b::debug);
59  			string(a.typePropertyName()).ifPresent(b::typePropertyName);
60  			string(a.locale()).map(Locale::forLanguageTag).ifPresent(b::locale);
61  			string(a.mediaType()).map(MediaType::of).ifPresent(b::mediaType);
62  			string(a.timeZone()).map(TimeZone::getTimeZone).ifPresent(b::timeZone);
63  			if (a.durationFormat() != DurationFormat.NOT_SET) b.durationFormat(a.durationFormat());
64  			if (a.periodFormat() != PeriodFormat.NOT_SET) b.periodFormat(a.periodFormat());
65  			if (a.calendarFormat() != CalendarFormat.NOT_SET) b.calendarFormat(a.calendarFormat());
66  			if (a.dateFormat() != DateFormat.NOT_SET) b.dateFormat(a.dateFormat());
67  			if (a.temporalFormat() != TemporalFormat.NOT_SET) b.temporalFormat(a.temporalFormat());
68  			if (a.timeZoneFormat() != TimeZoneFormat.NOT_SET) b.timeZoneFormat(a.timeZoneFormat());
69  			if (a.localeFormat() != LocaleFormat.NOT_SET) b.localeFormat(a.localeFormat());
70  			if (a.binaryFormat() != BinaryFormat.NOT_SET) b.binaryFormat(a.binaryFormat());
71  			if (a.enumFormat() != EnumFormat.NOT_SET) b.enumFormat(a.enumFormat());
72  			if (a.uuidFormat() != UuidFormat.NOT_SET) b.uuidFormat(a.uuidFormat());
73  			if (a.bitSetFormat() != BitSetFormat.NOT_SET) b.bitSetFormat(a.bitSetFormat());
74  			if (a.bigNumberFormat() != BigNumberFormat.NOT_SET) b.bigNumberFormat(a.bigNumberFormat());
75  			if (a.booleanFormat() != BooleanFormat.NOT_SET) b.booleanFormat(a.booleanFormat());
76  			if (a.floatFormat() != FloatFormat.NOT_SET) b.floatFormat(a.floatFormat());
77  			if (a.currencyFormat() != CurrencyFormat.NOT_SET) b.currencyFormat(a.currencyFormat());
78  			if (a.classFormat() != ClassFormat.NOT_SET) b.classFormat(a.classFormat());
79  			classes(a.dictionary()).ifPresent(b::beanDictionary);
80  			classes(a.dictionary_replace()).ifPresent(x -> { b.beanDictionary().clear(); b.beanDictionary(x);});
81  			classes(a.swaps()).ifPresent(b::swaps);
82  			classes(a.swaps_replace()).ifPresent(x -> { b.swaps().clear(); b.swaps(x);});
83  			// @formatter:on
84  		}
85  	}
86  }
