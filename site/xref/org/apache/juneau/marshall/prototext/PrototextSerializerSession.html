110 	@Override
111 	protected void doWrite(SerializerPipe out, Object o) throws IOException, SerializeException {
112 		if (o == null)
113 			return;
114 		if (debugTrace) traceLog.add("[Prototext] doWrite: o=" + cn(o) + ", getExpectedRootType=" + getExpectedRootType(o).inner().getName());
115 		var w = getPrototextWriter(out);
116 		var eType = getExpectedRootType(o);
117 		var cm = getClassMetaForObject(o);
118 		// For root object: if it's a bean (or toBeanMap works), serialize as bean directly to avoid
119 		// writeAnything swap/optional complexity that can produce empty output for some POJOs
120 		if (!cm.isMap() && !cm.isCollection() && !cm.isArray() && !cm.isStreamable()) {
121 			try {
122 				var bm = toBeanMap(o);
123 			if (!isEmpty(bm)) {
124 				writeBeanMap(w, bm, getBeanTypeName(this, eType, cm, null));
125 					return;
126 				}
127 			} catch (@SuppressWarnings("unused") Exception e) {
128 				// Not a bean, fall through to writeAnything
129 			}
130 		}
131 		writeAnything(w, o, eType, null, null);
132 	}

134 	protected final PrototextWriter getPrototextWriter(SerializerPipe out) {
135 		var output = out.getRawOutput();
136 		if (output instanceof PrototextWriter output2)
137 			return output2;
138 		var w = new PrototextWriter(out.getWriter(), isUseWhitespace(), getMaxIndent(), isTrimStrings(), getUriResolver());
139 		out.setWriter(w);
140 		return w;
141 	}
142 
143 	// NOTE: this method has a single call site -- doWrite() always invokes it with fieldName=null and
144 	// pMeta=null (root dispatch only; nested bean/map/collection properties are rendered by the dedicated
145 	// writeBeanMap/writeMap/writeCollection/writeStreamable helpers below, which never call back into
146 	// writeAnything). As a result, every "nn(fieldName)" check and the fieldName-wrapped branch it guards
147 	// is permanently dead below (fieldName is provably always null), and so is the "isRecursion" slot
148 	// (fixed once from the root push2() call, which cannot be a repeat visit on the very first push).
149 	private void writeAnything(PrototextWriter out, Object o, ClassMeta<?> eType, String fieldName, BeanPropertyMeta pMeta) throws SerializeException {
150 		if (o == null) // HTT: provably dead -- writeAnything's sole call site (doWrite) already returns early on a null root, so o is always non-null here
151 			return;
152 		if (eType == null) // HTT: provably dead -- eType is always getExpectedRootType(o), which never returns null (see SerializerSession#getExpectedRootType)
153 			eType = object();
154 
155 		var aType = push2(fieldName, o, eType);
156 		var isRecursion = aType == null;
157 		if (debugTrace && fieldName == null) {
158 			var aTypeName = aType == null ? "null" : aType.inner().getName();
159 			var isBean = aType != null ? aType.isBean() : "n/a";
160 			var isMap = aType != null ? aType.isMap() : "n/a";
161 			traceLog.add("[Prototext] writeAnything(root): push2 aType=" + aTypeName
162 				+ ", isBean=" + isBean + ", isMap=" + isMap + ", isRecursion=" + isRecursion);
163 		}
164 		if (aType == null) {
165 			pop();
166 			return;
167 		}
168 
169 		if (isOptional(aType)) {
170 			o = getOptionalValue(o);
171 			eType = getOptionalType(eType);
172 			aType = getClassMetaForObject(o, object());
173 		}
174 
175 		var swap = aType.getSwap(this);
176 		if (nn(swap)) {
177 			o = swap(swap, o);
178 			aType = swap.getSwapClassMeta(this);
179 			if (aType.isObject())
180 				aType = getClassMetaForObject(o);
181 		}
182 
183 		if (o == null)
184 			return;
185 
186 		var sType = aType;
187 		var typeName = getBeanTypeName(this, eType, aType, pMeta);
188 
189 		if (debugTrace && fieldName == null) {
190 			String path;
191 			if (sType.isBean())
192 				path = "bean";
193 			else if (sType.isMap())
194 				path = "map";
195 			else if (sType.isCollection() || sType.isArray())
196 				path = "collection";
197 			else
198 				path = "scalar";
199 			traceLog.add("[Prototext] writeAnything(root): sType=" + sType.inner().getName() + ", isBean=" + sType.isBean() + ", isMap=" + sType.isMap()
200 				+ ", path=" + path);
201 		}
202 		// Workaround: when isBean() is false but toBeanMap works (e.g. POJO with public fields),
203 		// treat as bean so we serialize fields instead of Object.toString()
204 		var treatAsBean = sType.isBean();
205 		if (!treatAsBean && fieldName == null && !sType.isMap() && !sType.isCollection() && !sType.isArray() && !sType.isStreamable()) {
206 			try {
207 				var bm = toBeanMap(o);
208 				treatAsBean = !isEmpty(bm); // HTT: provably dead -- toBeanMap(o) uses the same getBeanMeta() lookup as sType.isBean() (just checked false above for this exact o/sType), so it always throws before reaching this line; never observed to succeed
209 			} catch (@SuppressWarnings("unused") Exception e) {
210 				// Not a bean, will use scalar path
211 			}
212 		}
213 		if (treatAsBean) {
214 			var bm = toBeanMap(o);
215 			if (nn(fieldName)) { // HTT: provably dead true-arm -- fieldName is always null here (see class-level note above writeAnything)
216 				out.messageStart(fieldName, ctx.useColonForMessages); // HTT: unreachable body of dead true-arm above
217 				indent++; // HTT: unreachable body of dead true-arm above
218 				writeBeanMap(out, bm, typeName); // HTT: unreachable body of dead true-arm above
219 				indent--; // HTT: unreachable body of dead true-arm above
220 				out.messageEnd(indent); // HTT: unreachable body of dead true-arm above
221 			} else {
222 				writeBeanMap(out, bm, typeName);
223 			}
224 		} else if (sType.isMap()) {
225 			if (nn(fieldName)) { // HTT: provably dead true-arm -- fieldName is always null here (see class-level note above writeAnything)
226 				out.messageStart(fieldName, ctx.useColonForMessages); // HTT: unreachable body of dead true-arm above
227 				indent++; // HTT: unreachable body of dead true-arm above
228 				if (sType.isBeanMap()) // HTT: unreachable body of dead true-arm above
229 					writeBeanMap(out, (BeanMap) o, typeName); // HTT: unreachable body of dead true-arm above
230 				else
231 					writeMap(out, (Map) o, sType); // HTT: unreachable body of dead true-arm above
232 				indent--; // HTT: unreachable body of dead true-arm above
233 				out.messageEnd(indent); // HTT: unreachable body of dead true-arm above
234 			} else {
235 				if (sType.isBeanMap())
236 					writeBeanMap(out, (BeanMap) o, typeName);
237 				else
238 					writeMap(out, (Map) o, sType);
239 			}
240 		} else if (sType.isCollection()) {
241 			writeCollection(out, (Collection) o, sType, nn(fieldName) ? fieldName : CONST_value); // HTT: fieldName is always null here, so this ternary always resolves to CONST_value (see class-level note above writeAnything)
242 		} else if (sType.isArray()) {
243 			writeCollection(out, toList(sType.inner(), o), sType, nn(fieldName) ? fieldName : CONST_value); // HTT: fieldName is always null here, so this ternary always resolves to CONST_value (see class-level note above writeAnything)
244 		} else if (sType.isStreamable()) {
245 			writeStreamable(out, o, sType, nn(fieldName) ? fieldName : CONST_value); // HTT: fieldName is always null here, so this ternary always resolves to CONST_value (see class-level note above writeAnything)
246 		} else {
247 			var name = nn(fieldName) ? fieldName : CONST_value; // HTT: fieldName is always null here, so this ternary always resolves to CONST_value (see class-level note above writeAnything)
248 			out.scalarField(name);
249 			writeScalarValue(out, o, sType);
250 			out.w('\n');
251 		}
252 
253 		if (!isRecursion) // HTT: provably dead false-arm -- isRecursion is fixed once from the root push2() call above, and we already returned early at "aType == null" if it were true, so by this point it is always false
254 			pop();
255 	}

257 	private void writeBeanMap(PrototextWriter out, BeanMap<?> m, String typeName) throws SerializeException {
258 		if (isAddBeanTypes() && nn(typeName)) {
259 			out.cr(indent);
260 			out.scalarField("_type");
261 			out.stringValue(typeName);
262 			out.w('\n');
263 		}
264 		Predicate<Object> checkNull = x -> isKeepNullProperties() || nn(x);
265 		m.forEachValue(checkNull, (pMeta, key, value, thrown) -> {
266 			if (nn(thrown))
267 				onBeanGetterException(pMeta, thrown);
268 			if (value == null)
269 				return;
270 			var cMeta = (ClassMeta<?>) pMeta.getBeanInfo();
271 			if (canIgnoreValue(pMeta, key, value))
272 				return;
273 			var protoPMeta = ctx.getPrototextBeanPropertyMeta(pMeta);
274 			if (nn(protoPMeta) && !protoPMeta.getComment().isEmpty())
275 				out.comment(protoPMeta.getComment());
276 			out.cr(indent);
277 			var aType = getClassMetaForObject(value, cMeta);
278 			if (aType.isBean() || aType.isMap()) {
279 				out.messageStart(key, ctx.useColonForMessages);
280 				indent++;
281 				if (aType.isBean())
282 					writeBeanMap(out, toBeanMap(value), getBeanTypeName(this, cMeta, aType, pMeta));
283 				else
284 					writeMap(out, (Map) value, aType);
285 				indent--;
286 				out.messageEnd(indent);
287 			} else if (aType.isCollection() || aType.isArray()) {
288 				var c = aType.isArray() ? toList(aType.inner(), value) : (Collection<?>) value;
289 				var elType = aType.getElementType();
290 				if (elType.isBean() || elType.isMap()) {
291 					if (ctx.useListSyntaxForBeans) {
292 						out.scalarField(key);
293 						out.listStart();
294 						var first = true;
295 						for (Object item : c) {
296 							if (!first)
297 								out.w(", ");
298 							first = false;
299 							out.w("{\n");
300 							indent++;
301 							writeBeanOrMapItem(out, item);
302 							indent--;
303 							out.i(indent).w('}');
304 						}
305 						out.listEnd().w('\n');
306 					} else {
307 						for (var item : c) {
308 							out.messageStart(key, ctx.useColonForMessages);
309 							indent++;
310 							writeBeanOrMapItem(out, item);
311 							indent--;
312 							out.messageEnd(indent);
313 						}
314 					}
315 				} else {
316 					out.scalarField(key);
317 					out.listStart();
318 					var first = true;
319 					for (var el : c) {
320 						if (!first)
321 							out.w(", ");
322 						first = false;
323 						writeScalarValue(out, el, elType);
324 					}
325 					out.listEnd().w('\n');
326 				}
327 			} else {
328 				out.scalarField(key);
329 				writeScalarValue(out, value, aType);
330 				out.w('\n');
331 			}
332 		});
333 	}

335 	private void writeMap(PrototextWriter out, Map<?,?> map, ClassMeta<?> type) throws SerializeException {
336 		forEachEntry(map, e -> {
337 			var k = toString(e.getKey());
338 			var v = e.getValue();
339 			if (v == null)
340 				return;
341 			out.cr(indent);
342 			var valueType = type.getValueType();
343 			var aType = getClassMetaForObject(v, valueType);
344 			if (aType.isBean() || aType.isMap()) {
345 				out.messageStart(k, ctx.useColonForMessages);
346 				indent++;
347 				if (aType.isBean())
348 					writeBeanMap(out, toBeanMap(v), null);
349 				else
350 					writeMap(out, (Map) v, aType);
351 				indent--;
352 				out.messageEnd(indent);
353 			} else if (aType.isCollection() || aType.isArray()) {
354 				var c = aType.isArray() ? toList(aType.inner(), v) : (Collection<?>) v;
355 				writeCollection(out, c, aType, k);
356 			} else {
357 				out.scalarField(k);
358 				writeScalarValue(out, v, aType);
359 				out.w('\n');
360 			}
361 		});
362 	}

364 	private void writeCollectionItem(PrototextWriter out, Object item) throws SerializeException {
365 		var aType = getClassMetaForObject(item);
366 		if (aType.isBean())
367 			writeBeanMap(out, toBeanMap(item), null);
368 		else if (aType.isMap())
369 			writeMap(out, (Map) item, aType);
370 		else
371 			writeScalarValue(out, item, aType);
372 	}

374 	private void writeBeanOrMapItem(PrototextWriter out, Object item) throws SerializeException {
375 		var aType = getClassMetaForObject(item);
376 		if (aType.isMap())
377 			writeMap(out, (Map) item, aType);
378 		else
379 			writeBeanMap(out, toBeanMap(item), null);
380 	}

382 	private void writeCollection(PrototextWriter out, Collection<?> c, ClassMeta<?> type, String fieldName) throws SerializeException {
383 		var elType = type.getElementType();
384 		// When element type is Object, resolve actual type from first element for beans/maps
385 		var elementIsBeanOrMap = elType.isBean() || elType.isMap();
386 		if (!elementIsBeanOrMap && elType.isObject() && !c.isEmpty()) {
387 			var first = c.iterator().next();
388 			elementIsBeanOrMap = getClassMetaForObject(first).isBean() || getClassMetaForObject(first).isMap();
389 		}
390 		if (elementIsBeanOrMap) {
391 			if (ctx.useListSyntaxForBeans && nn(fieldName)) {
392 				out.scalarField(fieldName);
393 				out.listStart();
394 				var first = true;
395 				for (var item : c) {
396 					if (!first)
397 						out.w(", ");
398 					first = false;
399 					out.w("{\n");
400 					indent++;
401 					writeCollectionItem(out, item);
402 					indent--;
403 					out.i(indent).w('}');
404 				}
405 				out.listEnd().w('\n');
406 			} else if (nn(fieldName)) { // HTT: provably dead false-arm -- every writeCollection() call site (writeAnything's CONST_value fallback, writeMap's map key) always passes a non-null fieldName, so the implicit "else, write nothing" is never reachable
407 				for (var item : c) {
408 					out.messageStart(fieldName, ctx.useColonForMessages);
409 					indent++;
410 					writeCollectionItem(out, item);
411 					indent--;
412 					out.messageEnd(indent);
413 				}
414 			}
415 		} else if (nn(fieldName)) { // HTT: provably dead false-arm -- every writeCollection() call site always passes a non-null fieldName (see note above), so the implicit "else, write nothing" is never reachable
416 			out.scalarField(fieldName);
417 			out.listStart();
418 			var first = true;
419 			for (var el : c) {
420 				if (!first)
421 					out.w(", ");
422 				first = false;
423 				var actualElType = elType.isObject() ? getClassMetaForObject(el, elType) : elType;
424 				writeScalarValue(out, el, actualElType);
425 			}
426 			out.listEnd().w('\n');
427 		}
428 	}

430 	private void writeStreamable(PrototextWriter out, Object o, ClassMeta<?> sType, String fieldName) throws SerializeException {
431 		var elType = sType.getElementType();
432 		if (elType.isBean() || elType.isMap()) {
433 			forEachStreamableEntry(o, sType, item -> {
434 				out.messageStart(fieldName, ctx.useColonForMessages);
435 				indent++;
436 				writeCollectionItem(out, item);
437 				indent--;
438 				out.messageEnd(indent);
439 			});
440 		} else if (nn(fieldName)) { // HTT: provably dead false-arm -- writeStreamable's sole call site (writeAnything's CONST_value fallback) always passes a non-null fieldName, so the implicit "else, write nothing" is never reachable
441 			out.scalarField(fieldName);
442 			out.listStart();
443 			var first = Flag.create();
444 			forEachStreamableEntry(o, sType, el -> {
445 				first.ifSet(() -> out.w(", ")).set();
446 				writeScalarValue(out, el, elType);
447 			});
448 			out.listEnd().w('\n');
449 		}
450 	}

452 	private void writeScalarValue(PrototextWriter out, Object value, ClassMeta<?> type) throws SerializeException {
453 		if (value == null)
454 			return;
455 		// Resolve to the runtime type and apply any registered swap.  Required for collection/array
456 		// elements that arrive here without having gone through writeAnything's swap step (e.g.
457 		// List<Locale> elements where the bean-property swap is on the List, not on Locale).
458 		// Without this, Locale list-elements wire as Locale.toString() ("en_US") regardless
459 		// of the configured LocaleFormat, breaking BCP_47 round-trips on the parser side.
460 		var rType = getClassMetaForObject(value, type);
461 		var swap = rType.getSwap(this);
462 		if (nn(swap)) {
463 			value = swap(swap, value);
464 			rType = swap.getSwapClassMeta(this);
465 			if (rType.isObject())
466 				rType = getClassMetaForObject(value);
467 			type = rType;
468 		}
469 		if (value == null)
470 			return;
471 		if (type.isNumber()) {
472 			if (value instanceof Float || value instanceof Double)
473 				out.floatValue(((Number) value).doubleValue());
474 			else
475 				out.integerValue(((Number) value).longValue());
476 		} else if (type.isBoolean()) {
477 			out.booleanValue((Boolean) value);
478 		} else if (type.isEnum()) {
479 			out.enumValue(((Enum<?>) value).name());
480 		} else if (type.isDate()) {
481 			out.stringValue(writeDate((Date)value, type));
482 		} else if (type.isCalendar()) {
483 			out.stringValue(writeCalendar(value, type));
484 		} else if (type.isTemporal()) {
485 			out.stringValue(writeTemporal((TemporalAccessor)value, type));
486 		} else if (type.isDuration()) {
487 			out.stringValue(writeDuration((Duration)value));
488 		} else if (type.isPeriod()) {
489 			out.stringValue(writePeriod((Period)value));
490 		} else if (value instanceof byte[] value2) {
491 			out.bytesValue(value2);
492 		} else {
493 			out.stringValue(toString(value));
494 		}
495 	}
