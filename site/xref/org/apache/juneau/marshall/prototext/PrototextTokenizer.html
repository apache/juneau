24  /**
25   * Internal tokenizer for Protobuf Text Format parsing.
26   *
27   * <p>
28   * Character-by-character reader that handles the full Protobuf Text Format grammar:
29   * identifiers, strings (with C-style escaping), numbers (decimal, hex, octal, float),
30   * booleans, comments, and structural tokens.
31   *
32   * <h5 class='section'>Notes:</h5><ul>
33   * 	<li class='note'>This class is package-private and not intended for external use.
34   * </ul>
35   *
36   * <h5 class='section'>See Also:</h5><ul>
37   * 	<li class='link'><a class="doclink" href="https://protobuf.dev/reference/protobuf/textformat-spec">Protobuf Text Format Specification</a>
38   * </ul>
39   */
40  @SuppressWarnings({
41  	"java:S115",  // CONST_ prefix follows framework convention
42  	"java:S135",  // Multiple break/continue necessary for tokenizer state machine loops
43  	"java:S3776", // Cognitive complexity acceptable for protobuf grammar
44  	"java:S6541", // Brain method acceptable for tokenizer
45  	"resource" // Reader field is owned by the caller; the tokenizer reads from it but does not own its lifecycle.
46  })
47  class PrototextTokenizer {
48  
49  	private static final String CONST_integerOverflow = "Integer overflow: ";
50  	private static final String CONST_invalidFloat = "Invalid float: ";
51  
52  	private final Reader reader;
53  	private final Deque<Integer> pushback = new ArrayDeque<>();
54  	private int line = 1;
55  	private int column;
56  	private PrototextToken nextToken;
57  
58  	PrototextTokenizer(Reader reader) {
59  		this.reader = reader == null ? new StringReader("") : reader;
60  	}
61  
62  	int peekChar() throws IOException {
63  		if (!pushback.isEmpty())
64  			return pushback.peekLast();
65  		int c = reader.read();
66  		if (c >= 0)
67  			pushback.addLast(c);
68  		return c;
69  	}
70  
71  	int readChar() throws IOException {
72  		if (!pushback.isEmpty()) {
73  			int c = pushback.removeLast();
74  			advancePosition(c);
75  			return c;
76  		}
77  		int c = reader.read();
78  		advancePosition(c);
79  		return c;
80  	}
81  
82  	private void advancePosition(int c) {
83  		if (c == '\n') {
84  			line++;
85  			column = 0;
86  		} else if (c >= 0) {
87  			column++;
88  		}
89  	}
90  
91  	void unread(int c) {
92  		if (c >= 0)
93  			pushback.addLast(c);
94  	}
95  
96  	void skipWhitespaceAndComments() throws IOException {
97  		while (true) {
98  			int c = peekChar();
99  			if (c < 0)
100 				break;
101 			if (c == ' ' || c == '\t' || c == '\n' || c == '\r' || c == 0x0B || c == 0x0C) {
102 				readChar();
103 				continue;
104 			}
105 			if (c == '#') {
106 				readChar();
107 			while ((c = readChar()) >= 0 && c != '\n' && c != '\r')
108 				{ /* consume comment characters until end of line */ }
109 				if (c == '\r' && peekChar() == '\n')
110 					readChar();
111 				continue;
112 			}
113 			break;
114 		}
115 	}
116 
117 	PrototextToken peek() throws IOException, ParseException {
118 		if (nextToken == null)
119 			nextToken = lex();
120 		return nextToken;
121 	}
122 
123 	PrototextToken read() throws IOException, ParseException {
124 		var t = peek();
125 		nextToken = null;
126 		return t;
127 	}

129 	private PrototextToken lex() throws IOException, ParseException {
130 		skipWhitespaceAndComments();
131 		int c = peekChar();
132 		if (c < 0)
133 			return new PrototextToken(PrototextToken.TokenType.EOF, null);
134 
135 		switch (c) {
136 			case '{' -> { readChar(); return new PrototextToken(PrototextToken.TokenType.LBRACE, null); }
137 			case '}' -> { readChar(); return new PrototextToken(PrototextToken.TokenType.RBRACE, null); }
138 			case '<' -> { readChar(); return new PrototextToken(PrototextToken.TokenType.LANGLE, null); }
139 			case '>' -> { readChar(); return new PrototextToken(PrototextToken.TokenType.RANGLE, null); }
140 			case '[' -> { readChar(); return new PrototextToken(PrototextToken.TokenType.LBRACKET, null); }
141 			case ']' -> { readChar(); return new PrototextToken(PrototextToken.TokenType.RBRACKET, null); }
142 			case ':' -> { readChar(); return new PrototextToken(PrototextToken.TokenType.COLON, null); }
143 			case ',' -> { readChar(); return new PrototextToken(PrototextToken.TokenType.COMMA, null); }
144 			case ';' -> { readChar(); return new PrototextToken(PrototextToken.TokenType.SEMICOLON, null); }
145 			case '"' -> { return new PrototextToken(PrototextToken.TokenType.STRING, readDoubleQuotedString()); }
146 			case '\'' -> { return new PrototextToken(PrototextToken.TokenType.STRING, readSingleQuotedString()); }
147 			default -> { /* Fall through to number/identifier handling or throw below */ }
148 		}
149 
150 		if (mightStartNumber(c))
151 			return lexNumber();
152 		if (isLetterOrUnderscore(c))
153 			return new PrototextToken(PrototextToken.TokenType.IDENT, readIdentifier());
154 		throw parseException("Unexpected character: " + (char) c);
155 	}

157 	private boolean mightStartNumber(int c) throws IOException {
158 		if (Character.isDigit(c) || c == '.')
159 			return true;
160 		if (c == '+' || c == '-') {
161 			readChar(); // consume sign to look ahead
162 			int next = peekChar();
163 			unread(c); // restore sign for lexNumber
164 			return Character.isDigit(next) || next == '.' || next == 'i' || next == 'I' || next == 'n' || next == 'N';
165 		}
166 		// 'n'/'i' might start nan / inf / infinity at value position, but it might also
167 		// start a regular identifier-shaped field name (nano_seconds, inf, index, ...).  Multi-char
168 		// lookahead disambiguates: peek past the leading letter to see if the assembled token is
169 		// nan / inf / infinity (case-insensitive) AND is followed by a value-terminator (whitespace,
170 		// comma, semicolon, ], }, or EOF).  ':' is deliberately NOT a terminator — a field whose
171 		// name happens to be {@code inf} (or {@code nan}) is legal proto and shows up as
172 		// {@code inf: value}, so the colon-immediately-after case must stay on the IDENT path.
173 		if (c == 'i' || c == 'I' || c == 'n' || c == 'N')
174 			return startsSpecialFloatLiteral();
175 		return false;
176 	}
177 
178 	/**
179 	 * Multi-char lookahead helper for {@link #mightStartNumber}.  Reads identifier-shaped chars
180 	 * ({@link #isIdentChar}) up to 8 deep starting at the current position, peeks the next char as
181 	 * the candidate terminator, then unreads everything and restores the line/column counters so
182 	 * the lexer can reparse the same chars freshly.  Returns <jk>true</jk> only when the assembled
183 	 * token (case-folded) is one of {@code nan} / {@code inf} / {@code infinity} AND the next char
184 	 * is a value-terminator (whitespace / {@code ,} / {@code ;} / {@code ]} / {@code }} / EOF).
185 	 */
186 	private boolean startsSpecialFloatLiteral() throws IOException {
187 		var savedLine = line;
188 		var savedCol = column;
189 		var buf = new StringBuilder();
190 		while (buf.length() <= 8) {
191 			int next = peekChar();
192 			if (next < 0 || !isIdentChar(next))
193 				break;
194 			buf.append((char) readChar());
195 		}
196 		int term = peekChar();
197 		var isTerminator = term < 0
198 			|| term == ' ' || term == '\t' || term == '\n' || term == '\r' || term == 0x0B || term == 0x0C
199 			|| term == ',' || term == ';' || term == ']' || term == '}';
200 		var tok = buf.toString().toLowerCase(Locale.ROOT);
201 		var isSpecial = isTerminator && (tok.equals("nan") || tok.equals("inf") || tok.equals("infinity"));
202 		// Restore consumed chars in reverse order (pushback is a LIFO stack — to make the next read
203 		// return buf[0], buf[0] must be on top, so we push buf[N-1] ... buf[0]).
204 		for (var i = buf.length() - 1; i >= 0; i--)
205 			unread(buf.charAt(i));
206 		// readChar() advances line/column unconditionally, while unread() does NOT decrement, so we
207 		// must explicitly restore the pre-lookahead position counters.  Without this every multi-char
208 		// peek would inflate column by buf.length() before the actual lex consumed the same chars.
209 		line = savedLine;
210 		column = savedCol;
211 		return isSpecial;
212 	}

214 	private PrototextToken lexNumber() throws IOException, ParseException {
215 		int c = peekChar();
216 		boolean neg = false;
217 		if (c == '-') {
218 			readChar();
219 			neg = true;
220 			c = peekChar();
221 		} else if (c == '+') {
222 			readChar();
223 			c = peekChar();
224 		}
225 		if (c == '0') {
226 			readChar();
227 			int next = peekChar();
228 			if (next == 'x' || next == 'X') {
229 				readChar();
230 				long val = readHexInteger();
231 				return new PrototextToken(PrototextToken.TokenType.HEX_INT, neg ? -val : val);
232 			}
233 			if (next >= '0' && next <= '7') {
234 				long val = readOctalInteger();
235 				return new PrototextToken(PrototextToken.TokenType.OCT_INT, neg ? -val : val);
236 			}
237 			if (next == '.' || next == 'e' || next == 'E' || next == 'f' || next == 'F') {
238 				double val = readFloatLiteral(neg, "0");
239 				return new PrototextToken(PrototextToken.TokenType.FLOAT, val);
240 			}
241 			return new PrototextToken(PrototextToken.TokenType.DEC_INT, 0L);
242 		}
243 		if (c == '.') {
244 			double val = readFloatLiteral(neg, "");
245 			return new PrototextToken(PrototextToken.TokenType.FLOAT, val);
246 		}
247 		if (c == 'i' || c == 'I' || c == 'n' || c == 'N')
248 			return readSpecialFloatOrIdent(neg);
249 		var result = readDecimalOrFloat(neg);
250 		if (result instanceof Double result2)
251 			return new PrototextToken(PrototextToken.TokenType.FLOAT, result2);
252 		return new PrototextToken(PrototextToken.TokenType.DEC_INT, result);
253 	}

255 	private long readHexInteger() throws IOException, ParseException {
256 		var sb = new StringBuilder();
257 		while (isHexDigit(peekChar()))
258 			sb.append((char) readChar());
259 		if (sb.isEmpty())
260 			throw parseException("Invalid hex integer");
261 		try {
262 			return Long.parseLong(sb.toString(), 16);
263 		} catch (@SuppressWarnings("unused") NumberFormatException e) {
264 			throw parseException(CONST_integerOverflow + sb);
265 		}
266 	}
267 
268 	private long readOctalInteger() throws IOException, ParseException {
269 		var sb = new StringBuilder("0");
270 		while (peekChar() >= '0' && peekChar() <= '7')
271 			sb.append((char) readChar());
272 		try {
273 			return Long.parseLong(sb.toString(), 8);
274 		} catch (@SuppressWarnings("unused") NumberFormatException e) {
275 			throw parseException(CONST_integerOverflow + sb);
276 		}
277 	}

279 	private Object readDecimalOrFloat(boolean neg) throws IOException, ParseException {
280 		var sb = new StringBuilder();
281 		while (Character.isDigit(peekChar()) || peekChar() == '_') {
282 			int ch = readChar();
283 			if (ch != '_')
284 				sb.append((char) ch);
285 		}
286 		int c = peekChar();
287 		if (c == '.' || c == 'e' || c == 'E') {
288 			if (c == '.') {
289 				sb.append((char) readChar());
290 				while (Character.isDigit(peekChar()) || peekChar() == '_') {
291 					int ch = readChar();
292 					if (ch != '_')
293 						sb.append((char) ch);
294 				}
295 			}
296 			if (peekChar() == 'e' || peekChar() == 'E') {
297 				sb.append((char) readChar());
298 				if (peekChar() == '+' || peekChar() == '-')
299 					sb.append((char) readChar());
300 				while (Character.isDigit(peekChar()) || peekChar() == '_') {
301 					int ch = readChar();
302 					if (ch != '_')
303 						sb.append((char) ch);
304 				}
305 			}
306 			if (peekChar() == 'f' || peekChar() == 'F')
307 				sb.append((char) readChar());
308 			var s = sb.toString();
309 			try {
310 				double val = Double.parseDouble(s);
311 				return neg ? -val : val;
312 			} catch (@SuppressWarnings("unused") NumberFormatException e) {
313 				throw parseException(CONST_invalidFloat + s);
314 			}
315 		}
316 		if (peekChar() == 'f' || peekChar() == 'F') {
317 			readChar();
318 			var s = sb.toString();
319 			try {
320 				double val = Double.parseDouble(s);
321 				return neg ? -val : val;
322 			} catch (@SuppressWarnings("unused") NumberFormatException e) {
323 				throw parseException(CONST_invalidFloat + s);
324 			}
325 		}
326 		if (sb.isEmpty())
327 			throw parseException("Expected number");
328 		var s = sb.toString();
329 		try {
330 			return neg ? -Long.parseLong(s) : Long.parseLong(s);
331 		} catch (@SuppressWarnings("unused") NumberFormatException e) {
332 			throw parseException(CONST_integerOverflow + s);
333 		}
334 	}

336 	/**
337 	 * Reads an identifier-shaped token starting with 'n'/'N'/'i'/'I' and dispatches to a FLOAT
338 	 * token when the lowercased form is one of the proto special-float literals (nan / inf /
339 	 * infinity), or to an IDENT token otherwise.
340 	 */
341 	private PrototextToken readSpecialFloatOrIdent(boolean neg) throws IOException, ParseException {
342 		var sb = new StringBuilder();
343 		while (isLetterOrUnderscore(peekChar()) || Character.isDigit(peekChar()))
344 			sb.append((char) readChar());
345 		var s = sb.toString();
346 		var lower = s.toLowerCase(Locale.ROOT);
347 		if (lower.equals("inf") || lower.equals("infinity"))
348 			return new PrototextToken(PrototextToken.TokenType.FLOAT, neg ? Double.NEGATIVE_INFINITY : Double.POSITIVE_INFINITY);
349 		if (lower.equals("nan"))
350 			return new PrototextToken(PrototextToken.TokenType.FLOAT, Double.NaN);
351 		if (neg)
352 			throw parseException("Invalid float literal: -" + s);
353 		return new PrototextToken(PrototextToken.TokenType.IDENT, s);
354 	}

356 	/**
357 	 * Reads the tail of a float literal.  The integer part (if any) is pre-supplied as
358 	 * {@code prefix} (e.g. {@code "0"} for the leading-zero float branch, {@code ""} for a bare
359 	 * {@code .5}-style literal).  The reader optionally consumes {@code .digits}, an exponent
360 	 * ({@code eDigits} / {@code Edigits} with optional sign), and a trailing {@code f}/{@code F}
361 	 * suffix.
362 	 *
363 	 * @param neg <jk>true</jk> if a leading minus sign was consumed by the caller.
364 	 * @param prefix The integer part already consumed by the caller (without sign).
365 	 * @return The parsed double value.
366 	 * @throws IOException If the reader cannot be read.
367 	 * @throws ParseException If the assembled literal is not a valid float.
368 	 */
369 	private double readFloatLiteral(boolean neg, String prefix) throws IOException, ParseException {
370 		var sb = new StringBuilder(neg ? "-" : "");
371 		sb.append(prefix);
372 		if (peekChar() == '.') {
373 			sb.append((char) readChar());
374 			while (Character.isDigit(peekChar()) || peekChar() == '_') {
375 				int ch = readChar();
376 				if (ch != '_')
377 					sb.append((char) ch);
378 			}
379 		}
380 		if (peekChar() == 'e' || peekChar() == 'E') {
381 			sb.append((char) readChar());
382 			if (peekChar() == '+' || peekChar() == '-')
383 				sb.append((char) readChar());
384 			while (Character.isDigit(peekChar()) || peekChar() == '_') {
385 				int ch = readChar();
386 				if (ch != '_')
387 					sb.append((char) ch);
388 			}
389 		}
390 		if (peekChar() == 'f' || peekChar() == 'F')
391 			readChar();
392 		var s = sb.toString();
393 		try {
394 			return Double.parseDouble(s);
395 		} catch (@SuppressWarnings("unused") NumberFormatException e) {
396 			throw parseException(CONST_invalidFloat + s);
397 		}
398 	}

400 	String readIdentifier() throws IOException, ParseException {
401 		var sb = new StringBuilder();
402 		int c = peekChar();
403 		if (!isLetterOrUnderscore(c))
404 			throw parseException("Expected identifier");
405 		while (isIdentChar(peekChar()))
406 			sb.append((char) readChar());
407 		return sb.toString();
408 	}

410 	String readString() throws IOException, ParseException {
411 		int c = peekChar();
412 		if (c == '"')
413 			return readDoubleQuotedString();
414 		if (c == '\'')
415 			return readSingleQuotedString();
416 		throw parseException("Expected quoted string");
417 	}
418 
419 	private String readDoubleQuotedString() throws IOException, ParseException {
420 		if (readChar() != '"')
421 			throw parseException("Expected '\"'");
422 		return readStringContent('"');
423 	}
424 
425 	private String readSingleQuotedString() throws IOException, ParseException {
426 		if (readChar() != '\'')
427 			throw parseException("Expected \"'\"");
428 		return readStringContent('\'');
429 	}

431 	private String readStringContent(char quote) throws IOException, ParseException {
432 		var sb = new StringBuilder();
433 		int c;
434 		while ((c = readChar()) >= 0 && c != quote) {
435 			if (c == '\\') {
436 				int esc = readChar();
437 				if (esc < 0)
438 					throw parseException("Unexpected end of string");
439 				sb.append(processEscape(esc));
440 			} else {
441 				sb.append((char) c);
442 			}
443 		}
444 		if (c < 0)
445 			throw parseException("Unterminated string");
446 		return sb.toString();
447 	}

449 	private char processEscape(int c) throws IOException, ParseException {
450 		return switch (c) {
451 			case 'a' -> '\u0007';
452 			case 'b' -> '\b';
453 			case 'f' -> '\f';
454 			case 'n' -> '\n';
455 			case 'r' -> '\r';
456 			case 't' -> '\t';
457 			case 'v' -> '\u000B';
458 			case '?' -> '?';
459 			case '\\' -> '\\';
460 			case '\'' -> '\'';
461 			case '"' -> '"';
462 			case 'x', 'X' -> (char) readHexEscape(2);
463 			case '0', '1', '2', '3', '4', '5', '6', '7' -> {
464 				unread(c);
465 				yield (char) readOctalEscape();
466 			}
467 			case 'u' -> readUnicodeEscape(4);
468 			case 'U' -> readUnicodeEscape(8);
469 			default -> throw parseException("Invalid escape sequence: \\" + (char) c);
470 		};
471 	}

473 	private int readHexEscape(int maxLen) throws IOException, ParseException {
474 		var hex = new StringBuilder();
475 		for (int i = 0; i < maxLen && isHexDigit(peekChar()); i++)
476 			hex.append((char) readChar());
477 		if (hex.isEmpty())
478 			throw parseException("Invalid hex escape");
479 		return Integer.parseInt(hex.toString(), 16);
480 	}
481 
482 	private int readOctalEscape() throws IOException, ParseException {
483 		var oct = new StringBuilder();
484 		for (int i = 0; i < 3 && peekChar() >= '0' && peekChar() <= '7'; i++)
485 			oct.append((char) readChar());
486 		if (oct.isEmpty())
487 			throw parseException("Invalid octal escape");
488 		return Integer.parseInt(oct.toString(), 8);
489 	}
490 
491 	private char readUnicodeEscape(int len) throws IOException, ParseException {
492 		var hex = new StringBuilder();
493 		for (int i = 0; i < len; i++) {
494 			int c = readChar();
495 			if (c < 0 || !isHexDigit(c))
496 				throw parseException("Invalid unicode escape");
497 			hex.append((char) c);
498 		}
499 		int cp = Integer.parseInt(hex.toString(), 16);
500 		if (Character.isValidCodePoint(cp))
501 			return (char) cp;
502 		throw parseException("Invalid unicode code point: " + hex);
503 	}

505 	Number readInteger() throws IOException, ParseException {
506 		var t = read();
507 		return switch (t.type()) {
508 			case DEC_INT, OCT_INT, HEX_INT -> t.numberValue();
509 			default -> throw parseException("Expected integer, got " + t.type());
510 		};
511 	}

513 	double readFloat() throws IOException, ParseException {
514 		var t = read();
515 		if (t.type() == PrototextToken.TokenType.FLOAT)
516 			return t.numberValue().doubleValue();
517 		if (t.type() == PrototextToken.TokenType.DEC_INT || t.type() == PrototextToken.TokenType.HEX_INT || t.type() == PrototextToken.TokenType.OCT_INT)
518 			return t.numberValue().doubleValue();
519 		throw parseException("Expected float, got " + t.type());
520 	}
521 
522 	boolean readBoolean() throws IOException, ParseException {
523 		var t = read();
524 		if (t.type() == PrototextToken.TokenType.IDENT) {
525 			var s = t.stringValue().toLowerCase(Locale.ROOT);
526 			if (s.equals("true") || s.equals("t") || s.equals("1"))
527 				return true;
528 			if (s.equals("false") || s.equals("f") || s.equals("0"))
529 				return false;
530 		}
531 		if (t.type() == PrototextToken.TokenType.DEC_INT) {
532 			long v = t.numberValue().longValue();
533 			if (v == 0) return false;
534 			if (v == 1) return true;
535 		}
536 		throw parseException("Expected boolean, got " + t.type());
537 	}

547 	ParseException parseException(String message) {
548 		// Pass the raw message as a %s arg so any literal '%' in echoed input can't be interpreted as a printf directive.
549 		return new ParseException("%s at line %s, column %s", message, line, column);
550 	}

552 	private static boolean isLetterOrUnderscore(int c) {
553 		return (c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z') || c == '_';
554 	}
555 
556 	private static boolean isIdentChar(int c) {
557 		return (c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') || c == '_';
558 	}
