35  /**
36   * Serializes POJO models to RFC 8259 JSON.
37   *
38   * <p>
39   * Produces strict JSON with double-quoted strings by default. For JSON5-style output (single quotes,
40   * etc.), use {@link Json5Serializer}.
41   *
42   * <h5 class='topic'>Media types</h5>
43   * <p>
44   * Handles <c>Accept</c> types:  <bc>application/json, text/json</bc>
45   * <p>
46   * Produces <c>Content-Type</c> types:  <bc>application/json</bc>
47   *
48   * <h5 class='topic'>Description</h5>
49   * <p>
50   * The conversion is as follows...
51   * <ul class='spaced-list'>
52   * 	<li>
53   * 		Maps (e.g. {@link HashMap HashMaps}, {@link TreeMap TreeMaps}) are converted to JSON objects.
54   * 	<li>
55   * 		Collections (e.g. {@link HashSet HashSets}, {@link LinkedList LinkedLists}) and Java arrays are converted to
56   * 		JSON arrays.
57   * 	<li>
58   * 		{@link String Strings} are converted to JSON strings.
59   * 	<li>
60   * 		{@link Number Numbers} (e.g. {@link Integer}, {@link Long}, {@link Double}) are converted to JSON numbers.
61   * 	<li>
62   * 		{@link Boolean Booleans} are converted to JSON booleans.
63   * 	<li>
64   * 		{@code nulls} are converted to JSON nulls.
65   * 	<li>
66   * 		{@code arrays} are converted to JSON arrays.
67   * 	<li>
68   * 		{@code beans} are converted to JSON objects.
69   * </ul>
70   *
71   * <p>
72   * The types above are considered "JSON-primitive" object types.
73   * Any non-JSON-primitive object types are transformed into JSON-primitive object types through
74   * {@link ObjectSwap ObjectSwaps} associated through the
75   * {@link org.apache.juneau.marshall.MarshallingContext.Builder#swaps(Class...)} method.
76   * Several default transforms are provided for transforming Dates, Enums, Iterators, etc...
77   *
78   * <p>
79   * This serializer provides several serialization options.
80   * Typically, one of the predefined DEFAULT serializers will be sufficient.
81   * However, custom serializers can be constructed to fine-tune behavior.
82   *
83   * <h5 class='topic'>Behavior-specific subclasses</h5>
84   * <p>
85   * The following direct subclasses are provided for convenience:
86   * <ul class='spaced-list'>
87   * 	<li>
88   * 		{@link Json5Serializer} - Default serializer, single quotes, simple mode.
89   * 	<li>
90   * 		{@link org.apache.juneau.marshall.json5.Json5Serializer.Readable} - Default serializer, single quotes, simple mode, with whitespace.
91   * </ul>
92   *
93   * <h5 class='section'>Example:</h5>
94   * <p class='bjava'>
95   * 	<jc>// Use one of the default serializers to serialize a POJO</jc>
96   * 	String <jv>json</jv> = JsonSerializer.<jsf>DEFAULT</jsf>.write(<jv>someObject</jv>);
97   *
98   * 	<jc>// Create a custom serializer for lax syntax using single quote characters</jc>
99   * 	JsonSerializer <jv>serializer</jv> = JsonSerializer.<jsm>create</jsm>().simple().sq().build();
100  *
101  * 	<jc>// Clone an existing serializer and modify it to use single-quotes</jc>
102  * 	<jv>serializer</jv> = JsonSerializer.<jsf>DEFAULT</jsf>.copy().sq().build();
103  *
104  * 	<jc>// Serialize a POJO to JSON</jc>
105  * 	String <jv>json</jv> = <jv>serializer</jv>.write(<jv>someObject</jv>);
106  * </p>
107  *
108  * <h5 class='figure'>Example output (Map of name/age):</h5>
109  * <p class='bjson'>
110  * 	{<js>"name"</js>:<js>"Alice"</js>,<js>"age"</js>:30}
111  * </p>
112  *
113  * <h5 class='figure'>Complex (nested object + array):</h5>
114  * <p class='bjson'>
115  * 	{<js>"name"</js>:<js>"Alice"</js>,<js>"age"</js>:30,<js>"address"</js>:{<js>"street"</js>:<js>"123 Main St"</js>,<js>"city"</js>:<js>"Boston"</js>,<js>"state"</js>:<js>"MA"</js>},<js>"tags"</js>:[<js>"a"</js>,<js>"b"</js>,<js>"c"</js>]}
116  * </p>
117  *
118  * <h5 class='section'>Notes:</h5><ul>
119  * 	<li class='note'>This class is thread safe and reusable.
120  * </ul>
121  *
122  * <h5 class='section'>See Also:</h5><ul>
123  * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JsonSupport">JSON Basics</a>
124  * 	<li class='link'>{@link Json5Serializer} - For JSON5-style output (single quotes, etc.)
125  * </ul>
126  */

188 		/**
189 		 * Add <js>"_type"</js> properties when needed.
190 		 *
191 		 * <p>
192 		 * If <jk>true</jk>, then <js>"_type"</js> properties will be added to beans if their type cannot be inferred
193 		 * through reflection.
194 		 *
195 		 * <p>
196 		 * When present, this value overrides the {@link org.apache.juneau.marshall.serializer.Serializer.Builder#addBeanTypes()} setting and is
197 		 * provided to customize the behavior of specific serializers in a {@link SerializerSet}.
198 		 *
199 		 * <p>
200 		 * <b>Token streams:</b> the cursor's raw structural methods do not emit <c>"_type"</c>;
201 		 * however this setting IS honored by
202 		 * {@link TokenWriter#object(Object) TokenWriter.object(Object)}
203 		 * when walking a bean.
204 		 *
205 		 * @return This object.
206 		 */
207 		public SELF addBeanTypesJson() {
208 			return addBeanTypesJson(true);
209 		}

213 		 *
214 		 * @param value The value for this setting.
215 		 * @return This object.
216 		 */
217 		public SELF addBeanTypesJson(boolean value) {
218 			addBeanTypesJson = value;
219 			return self();
220 		}

230 		/**
231 		 * Prefix solidus <js>'/'</js> characters with escapes.
232 		 *
233 		 * <p>
234 		 * If enabled, solidus (e.g. slash) characters should be escaped.
235 		 *
236 		 * <p>
237 		 * The JSON specification allows for either format.
238 		 * <br>However, if you're embedding JSON in an HTML script tag, this setting prevents confusion when trying to serialize
239 		 * <xt>&lt;\/script&gt;</xt>.
240 		 *
241 		 * <h5 class='section'>Example:</h5>
242 		 * <p class='bjava'>
243 		 * 	<jc>// Create a JSON serializer that escapes solidus characters.</jc>
244 		 * 	WriterSerializer <jv>serializer</jv> = JsonSerializer
245 		 * 		.<jsm>create</jsm>()
246 		 * 		.simple()
247 		 * 		.escapeSolidus()
248 		 * 		.build();
249 		 *
250 		 * 	<jc>// Produces: "{foo:'&lt;\/bar&gt;'"</jc>
251 		 * 	String <jv>json</jv> = <jv>serializer</jv>.write(JsonMap.<jsm>of</jsm>(<js>"foo"</js>, <js>"&lt;/bar&gt;"</js>);
252 		 * </p>
253 		 *
254 		 * @return This object.
255 		 */
256 		public SELF escapeSolidus() {
257 			return escapeSolidus(true);
258 		}
259 
260 		/**
261 		 * Same as {@link #escapeSolidus()} but allows you to explicitly specify the value.
262 		 *
263 		 * @param value The value for this setting.
264 		 * @return This object.
265 		 */
266 		public SELF escapeSolidus(boolean value) {
267 			escapeSolidus = value;
268 			return self();
269 		}

381 	/**
382 	 * Returns the schema serializer based on the settings of this serializer.
383 	 *
384 	 * <p>
385 	 * Note that this method creates a builder initialized to all default settings, whereas {@link #copy()} copies
386 	 * the settings of the object called on.
387 	 *
388 	 * @return The schema serializer.
389 	 */
390 	public JsonSchemaSerializer getSchemaSerializer() {
391 		JsonSchemaSerializer result = schemaSerializer.get();
392 		if (result == null) {
393 			result = JsonSchemaSerializer.create().marshallingContext(getMarshallingContext()).build();
394 			if (! schemaSerializer.compareAndSet(null, result)) {
395 				result = schemaSerializer.get();
396 			}
397 		}
398 		return result;
399 	}
