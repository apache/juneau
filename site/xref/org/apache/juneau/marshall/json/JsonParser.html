34  /**
35   * Parses any valid RFC 8259 JSON text into a POJO model.
36   *
37   * <p>
38   * This parser strictly enforces RFC 8259 JSON syntax. Double-quoted strings only; single-quoted strings,
39   * comments, trailing commas, and unquoted keys are rejected. For JSON5 syntax (single quotes, comments,
40   * etc.), use {@link Json5Parser}.
41   *
42   * <h5 class='topic'>Media types</h5>
43   * <p>
44   * Handles <c>Content-Type</c> types:  <bc>application/json, text/json, application/jcs+json</bc>
45   *
46   * <h5 class='topic'>Description</h5>
47   * <p>
48   * This parser uses a state machine, which makes it very fast and efficient.  It parses JSON in about 70% of the
49   * time that it takes the built-in Java DOM parsers to parse equivalent XML.
50   *
51   * <p>
52   * This parser handles all valid RFC 8259 JSON syntax.
53   *
54   * <p>
55   * Also handles negative, decimal, hexadecimal, octal, and double numbers, including exponential notation.
56   *
57   * <p>
58   * This parser handles the following input, and automatically returns the corresponding Java class.
59   * <ul class='spaced-list'>
60   * 	<li>
61   * 		JSON objects (<js>"{...}"</js>) are converted to {@link JsonMap JsonMaps}.
62   * 		<b>Note:</b>  If a <code><xa>_type</xa>=<xs>'xxx'</xs></code> attribute is specified on the object, then an
63   * 		attempt is made to convert the object to an instance of the specified Java bean class.
64   * 		See the {@link org.apache.juneau.marshall.MarshallingContext.Builder#typePropertyName(String)} setting for more information about parsing
65   * 		beans from JSON.
66   * 	<li>
67   * 		JSON arrays (<js>"[...]"</js>) are converted to {@link JsonList JsonLists}.
68   * 	<li>
69   * 		JSON string literals (<js>"\"xyz\""</js>) are converted to {@link String Strings}.
70   * 	<li>
71   * 		JSON numbers (<js>"123"</js>, including octal/hexadecimal/exponential notation) are converted to
72   * 		{@link Integer Integers}, {@link Long Longs}, {@link Float Floats}, or {@link Double Doubles} depending on
73   * 		whether the number is decimal, and the size of the number.
74   * 	<li>
75   * 		JSON booleans (<js>"false"</js>) are converted to {@link Boolean Booleans}.
76   * 	<li>
77   * 		JSON nulls (<js>"null"</js>) are converted to <jk>null</jk>.
78   * </ul>
79   *
80   * <p>
81   * Input can be any of the following:
82   * <ul class='spaced-list'>
83   * 	<li>
84   * 		<js>"{...}"</js> - Converted to an {@link JsonMap} or an instance of a Java bean if a <xa>_type</xa>
85   * 		attribute is present.
86   * 	<li>
87   * 		<js>"[...]"</js> - Converted to an {@link JsonList}.
88   * 	<li>
89   * 		<js>"123..."</js> - Converted to a {@link Number} (either {@link Integer}, {@link Long}, {@link Float},
90   * 		or {@link Double}).
91   * 	<li>
92   * 		<js>"true"</js>/<js>"false"</js> - Converted to a {@link Boolean}.
93   * 	<li>
94   * 		<js>"null"</js> - Returns <jk>null</jk>.
95   * 	<li>
96   * 		<js>"\"xxx\""</js> - Converted to a {@link String} (double quotes only).
97   * </ul>
98   *
99   * <p>
100  * TIP:  If you know you're parsing a JSON object or array, it can be easier to parse it using the
101  * {@link JsonMap#JsonMap(CharSequence) JsonMap(CharSequence)} or {@link JsonList#JsonList(CharSequence)
102  * JsonList(CharSequence)} constructors instead of using this class.
103  * The end result should be the same.
104  *
105  * <h5 class='figure'>Example input (Map of name/age):</h5>
106  * <p class='bjson'>
107  * 	{<js>"name"</js>:<js>"Alice"</js>,<js>"age"</js>:30}
108  * </p>
109  *
110  * <h5 class='figure'>Complex (nested object + array):</h5>
111  * <p class='bjson'>
112  * 	{<js>"name"</js>:<js>"Alice"</js>,<js>"age"</js>:30,<js>"address"</js>:{<js>"street"</js>:<js>"123 Main St"</js>,<js>"city"</js>:<js>"Boston"</js>,<js>"state"</js>:<js>"MA"</js>},<js>"tags"</js>:[<js>"a"</js>,<js>"b"</js>,<js>"c"</js>]}
113  * </p>
114  *
115  * <h5 class='section'>Notes:</h5><ul>
116  * 	<li class='note'>This class is thread safe and reusable.
117  * </ul>
118  *
119  * <h5 class='section'>See Also:</h5><ul>
120  * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JsonSupport">JSON Basics</a>
121  * 	<li class='link'>{@link Json5Parser} - For JSON5 syntax (single quotes, comments, etc.)
122  * </ul>
123  */
124 @SuppressWarnings({
125 	"java:S110", // Inheritance depth acceptable
126 	"java:S115", // Constants use UPPER_snakeCase convention
127 	"resource"   // readTokens(...) returns a Closeable owned by the caller; Eclipse JDT @Owning warning is by design.
128 })
129 public class JsonParser extends ReaderParser implements JsonMetaProvider, TokenReadable, ArrayRecordReadable {
130 
131 	// Argument name constants for assertArgNotNull
132 	private static final String ARG_copyFrom = "copyFrom";
133 
134 	/**
135 	 * Builder class.
136 	 *
137 	 * <p>
138 	 * Unlike the abstract context builders, this builder is concrete <i>and</i> generic: it terminates the CRTP
139 	 * recursion via the leaf-free {@code new Builder<>()} pattern (see {@link JsonParser#create()} /
140 	 * {@link JsonParser#copy()}) rather than via a separate concrete {@code DefaultBuilder} leaf.
141 	 */
142 	@SuppressWarnings({
143 		"java:S119" // 'SELF' (CRTP self-type) is intentional and clearer than a single-letter name.
144 	})
145 	public static class Builder<SELF extends Builder<SELF>> extends ReaderParser.Builder<SELF> {
146 
147 		private static final Cache<HashKey,JsonParser> CACHE = Cache.of(HashKey.class, JsonParser.class).build();
148 
149 		private boolean validateEnd;
150 
151 		/**
152 		 * Constructor, default settings.
153 		 */
154 		protected Builder() {
155 			consumes("application/json,text/json,application/jcs+json");
156 			validateEnd = env("JsonParser.validateEnd", false);
157 		}
158 
159 		/**
160 		 * Copy constructor.
161 		 *
162 		 * @param copyFrom The builder to copy from.
163 		 * 	<br>Cannot be <jk>null</jk>.
164 		 */
165 		protected Builder(Builder<?> copyFrom) {
166 			super(assertArgNotNull(ARG_copyFrom, copyFrom));
167 			validateEnd = copyFrom.validateEnd;
168 		}
169 
170 		/**
171 		 * Copy constructor.
172 		 *
173 		 * @param copyFrom The bean to copy from.
174 		 * 	<br>Cannot be <jk>null</jk>.
175 		 */
176 		protected Builder(JsonParser copyFrom) {
177 			super(assertArgNotNull(ARG_copyFrom, copyFrom));
178 			validateEnd = copyFrom.validateEnd;
179 		}
180 
181 		@Override /* Overridden from Context.Builder<?> */
182 		public JsonParser build() {
183 			return cache(CACHE).build(JsonParser.class);
184 		}
185 
186 		@Override /* Overridden from Context.Builder<?> */
187 		@SuppressWarnings({
188 			"unchecked" // CRTP self-type cast on the leaf-free Builder copy is safe by construction.
189 		})
190 		public SELF copy() {
191 			return (SELF) new Builder<>(this);
192 		}
193 
194 		@Override /* Overridden from Context.Builder<?> */
195 		public HashKey hashKey() {
196 			return HashKey.of(super.hashKey(), validateEnd);
197 		}
198 
199 
200 		/**
201 		 * Validate end.
202 		 *
203 		 * <p>
204 		 * When enabled, after parsing a POJO from the input, verifies that the remaining input in
205 		 * the stream consists of only comments or whitespace.
206 		 *
207 		 * <h5 class='section'>Example:</h5>
208 		 * <p class='bjava'>
209 		 * 	<jc>// Create a parser that validates that there's no garbage at the end of the input.</jc>
210 		 * 	ReaderParser <jv>parser</jv> = JsonParser.
211 		 * 		.<jsm>create</jsm>()
212 		 * 		.validateEnd()
213 		 * 		.build();
214 		 *
215 		 * 	<jc>// Should fail because input has multiple POJOs.</jc>
216 		 * 	String <jv>json</jv> = <js>"{foo:'bar'}{baz:'qux'}"</js>;
217 		 * 	MyBean <jv>myBean</jv> =<jv>parser</jv>.read(<jv>json</jv>, MyBean.<jk>class</jk>);
218 		 * </p>
219 		 *
220 		 * <p>
221 		 * <b>Token streams:</b> this setting is not honored by {@link #readTokens(Object) readTokens}
222 		 * &mdash; the cursor leaves end-of-input checking to the caller.
223 		 *
224 		 * @return This object.
225 		 */
226 		public SELF validateEnd() {
227 			return validateEnd(true);
228 		}
229 
230 		/**
231 		 * Same as {@link #validateEnd()} but allows you to explicitly specify the value.
232 		 *
233 		 * @param value The value for this setting.
234 		 * @return This object.
235 		 */
236 		public SELF validateEnd(boolean value) {
237 			validateEnd = value;
238 			return self();
239 		}
240 	}
241 
242 	/** Default parser, all default settings.*/
243 	public static final JsonParser DEFAULT = new JsonParser(create());
244 
245 	/**
246 	 * Creates a new builder for this object.
247 	 *
248 	 * @return A new builder.
249 	 */
250 	@SuppressWarnings({
251 		"java:S1452" // Self-typed builder: Builder<?> is the only non-raw, leaf-free return; chaining + build() unaffected.
252 	})
253 	public static Builder<?> create() {
254 		return new Builder<>();
255 	}
256 
257 	protected final boolean validateEnd;
258 
259 	private final Map<BeanPropertyMeta,JsonBeanPropertyMeta> jsonBeanPropertyMetas = new ConcurrentHashMap<>();
260 	private final Map<ClassMeta<?>,JsonClassMeta> jsonClassMetas = new ConcurrentHashMap<>();
261 	
262 
263 	/**
264 	 * Constructor.
265 	 *
266 	 * @param builder The builder for this object.
267 	 */
268 	public JsonParser(Builder<?> builder) {
269 		super(builder);
270 		validateEnd = builder.validateEnd;
271 	}
272 
273 	@Override /* Overridden from Context */
274 	public Builder<?> copy() {
275 		return new Builder<>(this);
276 	}
277 
278 	@Override /* Overridden from Context */
279 	public JsonParserSession.Builder<?> createSession() {
280 		return JsonParserSession.create(this);
281 	}
282 
283 	@Override /* Overridden from JsonMetaProvider */
284 	public JsonBeanPropertyMeta getJsonBeanPropertyMeta(BeanPropertyMeta bpm) {
285 		if (bpm == null)
286 			return JsonBeanPropertyMeta.DEFAULT;
287 		return jsonBeanPropertyMetas.computeIfAbsent(bpm, k -> new JsonBeanPropertyMeta(k.getDelegateFor(), this));
288 	}
289 
290 	@Override /* Overridden from JsonMetaProvider */
291 	public JsonClassMeta getJsonClassMeta(ClassMeta<?> cm) {
292 		return jsonClassMetas.computeIfAbsent(cm, k -> new JsonClassMeta(k, this));
293 	}
294 
295 	@Override /* Overridden from Context */
296 	public JsonParserSession getSession() { return createSession().build(); }
297 
298 	/**
299 	 * Convenience delegator that opens a {@link JsonTokenReader} over the input using
300 	 * <b>default session arguments</b> (mirrors {@link #read(Object, Class)}).
301 	 *
302 	 * <p>
303 	 * The real implementation lives on {@link JsonParserSession#readTokens(Object)}.  Callers
304 	 * that need request-derived configuration (locale, timezone, schema, swaps) should call
305 	 * {@link #createSession()} and invoke {@link JsonParserSession#readTokens(Object)} on the
306 	 * built session instead.
307 	 *
308 	 * @param input The input.  Accepts {@link Reader}, {@link CharSequence}, {@link InputStream},
309 	 * 	<code><jk>byte</jk>[]</code>, or {@link File}.
310 	 * @return A new {@link JsonTokenReader}.
311 	 * @throws IOException If a problem occurred opening the underlying input.
312 	 */
313 	@Override /* TokenReadable */
314 	public TokenReader readTokens(Object input) throws IOException {
315 		return getSession().readTokens(input);
316 	}
317 
318 	/**
319 	 * Validate end.
320 	 *
321 	 * @see Builder#validateEnd()
322 	 * @return
323 	 * 	<jk>true</jk> if after parsing a POJO from the input, verifies that the remaining input in
324 	 * 	the stream consists of only comments or whitespace.
325 	 */
326 	protected final boolean isValidateEnd() { return validateEnd; }
327 
328 	/**
329 	 * Convenience delegator for the streaming array-element {@link RecordReader} (uses default
330 	 * session args; see {@link #readTokens(Object)}).  Real impl on
331 	 * {@link JsonParserSession#readArrayRecords(Object)}.
332 	 *
333 	 * @param input The input.
334 	 * @return A new element-streamed {@link RecordReader}.
335 	 * @throws IOException If a problem occurred opening the underlying input.
336 	 */
337 	@Override /* ArrayRecordReadable */
338 	public RecordReader readArrayRecords(Object input) throws IOException {
339 		return getSession().readArrayRecords(input);
340 	}
