153 	@SuppressWarnings({
154 		"java:S1452" // Returns the passed-in writer whose concrete CRTP self-type is not nameable here.
155 	})
156 	protected SerializerWriter<?> writeBeanMap(JsonWriter<?> out, BeanMap<?> m, String typeName) throws SerializeException {
157 		int i = indent;
158 		out.w('{');
159 
160 		var addComma = Flag.create();
161 		Predicate<Object> checkNull = x -> isKeepNullProperties() || nn(x);
162 
163 		if (nn(typeName)) {
164 			BeanPropertyMeta pm = m.getMeta().getTypeProperty();
165 			out.cr(i).attr(pm.getName()).w(':').s(i).stringValue(typeName);
166 			addComma.set();
167 		}
168 
169 		m.forEachValue(checkNull, (pMeta, key, value, thrown) -> {
170 			var cMeta = (ClassMeta<?>) pMeta.getBeanInfo();
171 			if (nn(thrown))
172 				onBeanGetterException(pMeta, thrown);
173 
174 			if (canIgnoreValue(pMeta, key, value))
175 				return;
176 
177 			addComma.ifSet(() -> out.append(',').smi(i)).set();
178 
179 			out.cr(i).attr(key).w(':').s(i);
180 
181 			writeAnything(out, value, cMeta, key, pMeta);
182 		});
183 
184 		out.cre(i - 1).w('}');
185 		return out;
186 	}

189 	private SerializerWriter<?> writeCollection(JsonWriter<?> out, Collection c, ClassMeta<?> type) throws SerializeException {
190 
191 		var elementType = type.getElementType();
192 
193 		out.w('[');
194 		var addComma = Flag.create();
195 		forEachEntry(c, x -> {
196 			addComma.ifSet(() -> out.w(',').smi(indent)).set();
197 			out.cr(indent);
198 			writeAnything(out, x, elementType, "<iterator>", null);
199 		});
200 
201 		out.cre(indent - 1).w(']');
202 		return out;
203 	}

205 	private SerializerWriter<?> writeStreamable(JsonWriter<?> out, Object o, ClassMeta<?> sType, ClassMeta<?> type) throws SerializeException {
206 		var elementType = type.getElementType();
207 
208 		out.w('[');
209 		var addComma = Flag.create();
210 		forEachStreamableEntry(o, sType, x -> {
211 			addComma.ifSet(() -> out.w(',').smi(indent)).set();
212 			out.cr(indent);
213 			writeAnything(out, x, elementType, "<iterator>", null);
214 		});
215 
216 		out.cre(indent - 1).w(']');
217 		return out;
218 	}

221 	@SuppressWarnings({
222 		"java:S1452" // Returns the passed-in writer whose concrete CRTP self-type is not nameable here.
223 	})
224 	protected SerializerWriter<?> writeMap(JsonWriter<?> out, Map m, ClassMeta<?> type) throws SerializeException {
225 
226 		var keyType = type.getKeyType();
227 		var valueType = type.getValueType();
228 
229 		int i = indent;
230 		out.w('{');
231 
232 		var addComma = Flag.create();
233 		forEachEntry(m, x -> {
234 			addComma.ifSet(() -> out.w(',').smi(i)).set();
235 			Object value = x.getValue();
236 			Object key = generalize(x.getKey(), keyType);
237 			out.cr(i).attr(toString(key)).w(':').s(i);
238 			writeAnything(out, value, valueType, (key == null ? null : toString(key)), null);
239 		});
240 
241 		out.cre(i - 1).w('}');
242 
243 		return out;
244 	}

263 	 *
264 	 * @param out The output target object.
265 	 * 	<br>Must not be <jk>null</jk>.
266 	 * @return The output target object wrapped in an {@link JsonWriter}.
267 	 */
268 	@SuppressWarnings({
269 		"java:S1452" // Raw output may be any JsonWriter CRTP leaf; the concrete self-type is not nameable here.
270 	})
271 	protected JsonWriter<?> getJsonWriter(SerializerPipe out) {
272 		var output = out.getRawOutput();
273 		if (output instanceof JsonWriter<?> output2)
274 			return output2;
275 		var w = new BasicJsonWriter(out.getWriter(), isUseWhitespace(), getMaxIndent(), isEscapeSolidus(), getQuoteChar(), false, isTrimStrings(), getUriResolver());
276 		out.setWriter(w);
277 		return w;
278 	}

299 	/**
300 	 * Opens a low-level push generator that emits JSON one structural event at a time, bound to
301 	 * this session's live config.
302 	 *
303 	 * <p>
304 	 * The writer is purely structural &mdash; object swaps and {@code @Schema} annotations are not
305 	 * applied at this layer.  See {@link TokenWriter} for the contract.  The higher-level
306 	 * {@link TokenWriter#object(Object)} bridge walks an arbitrary POJO via {@link PojoWalker} and
307 	 * DOES honor the databind settings (<c>keepNullProperties</c>, <c>trimEmptyMaps</c>,
308 	 * <c>trimEmptyCollections</c>, <c>sortMaps</c>, <c>sortCollections</c>, <c>trimStrings</c>).
309 	 *
310 	 * @param output The output.  Accepts {@link Writer} or {@link OutputStream}.
311 	 * @return A new {@link JsonTokenWriter}.
312 	 * @throws IOException If the output type is not supported or could not be opened.
313 	 */
314 	@Override /* TokenWritable */
315 	public TokenWriter writeTokens(Object output) throws IOException {
316 		var walk = new PojoWalker.Options(
317 			isKeepNullProperties(),
318 			isTrimEmptyMaps(),
319 			isTrimEmptyCollections(),
320 			isSortMaps(),
321 			isSortCollections(),
322 			isTrimStrings(),
323 			getMarshallingContext());
324 		var settings = new JsonTokenWriter.Settings(
325 			isUseWhitespace(),
326 			getMaxIndent(),
327 			getQuoteChar(),
328 			isEscapeSolidus(),
329 			isTrimStrings(),
330 			isSimpleAttrs(),
331 			walk,
332 			false /* disableObject */);
333 		return JsonTokenWriter.forOutput(output, settings);
334 	}

338 	 * Memory is O(1) in the array length &mdash; elements are emitted as they arrive.

349 	/**
350 	 * Workhorse method.
351 	 * Determines the type of object, and then calls the appropriate type-specific serialization method.
352 	 *
353 	 * @param out The output writer.
354 	 * 	<br>Must not be <jk>null</jk>.
355 	 * @param o The object to serialize.
356 	 * 	<br>Can be <jk>null</jk> (written as the JSON literal <js>"null"</js>).
357 	 * @param eType The expected type.
358 	 * 	<br>Can be <jk>null</jk> (defaults to {@code Object}).
359 	 * @param attrName The attribute name.
360 	 * @param pMeta The bean property currently being serialized.
361 	 * 	<br>Can be <jk>null</jk> (the bean-property-level dictionary lookup and URI detection are skipped).
362 	 * @return The same writer passed in.
363 	 * @throws SerializeException General serialization error occurred.
364 	 */
365 	
366 	@SuppressWarnings({
367 		"java:S3776", // Cognitive complexity acceptable for this specific logic
368 		"java:S1452" // Returns the passed-in writer whose concrete CRTP self-type is not nameable here.
369 	})
370 	protected JsonWriter<?> writeAnything(JsonWriter<?> out, Object o, ClassMeta<?> eType, String attrName, BeanPropertyMeta pMeta) throws SerializeException {
371 
372 		if (o == null) {
373 			out.append("null");
374 			return out;
375 		}
376 
377 		if (eType == null)
378 			eType = object();
379 
380 		ClassMeta<?> aType = null;			// The actual type
381 		ClassMeta<?> sType = null;			// The serialized type
382 
383 		aType = push2(attrName, o, eType);
384 		boolean isRecursion = aType == null;
385 
386 		// Handle recursion
387 		if (aType == null) {
388 			o = null;
389 			aType = object();
390 		}
391 
392 		// Handle Optional<X>
393 		if (isOptional(aType)) {
394 			o = getOptionalValue(o);
395 			eType = getOptionalType(eType);
396 			aType = getClassMetaForObject(o, object());
397 		}
398 
399 		sType = aType;
400 		String typeName = getBeanTypeName(this, eType, aType, pMeta);
401 
402 		// Swap if necessary
403 		var swap = aType.getSwap(this);
404 		if (nn(swap)) {
405 			o = swap(swap, o);
406 			sType = swap.getSwapClassMeta(this);
407 
408 			// If the getSwapClass() method returns Object, we need to figure out
409 			// the actual type now.
410 			if (sType.isObject())
411 				sType = getClassMetaForObject(o);
412 		}
413 
414 		String wrapperAttr = getJsonClassMeta(sType).getWrapperAttr();
415 		if (nn(wrapperAttr)) {
416 			out.w('{').cr(indent).attr(wrapperAttr).w(':').s(indent);
417 			indent++;
418 		}
419 
420 		// '\0' characters are considered null.
421 		if (o == null || (sType.isChar() && ((Character)o).charValue() == 0)) {
422 			out.append("null");
423 		} else if (sType.isBean()) {
424 			writeBeanMap(out, toBeanMap(o), typeName);
425 		} else if (sType.isMap()) {
426 			if (sType.isBeanMap())
427 				writeBeanMap(out, (BeanMap)o, typeName);
428 			else
429 				writeMap(out, (Map)o, eType);
430 		} else if (sType.isCollection()) {
431 			writeCollection(out, (Collection)o, eType);
432 		} else if (sType.isArray()) {
433 			writeCollection(out, toList(sType.inner(), o), eType);
434 		} else if (sType.isNumber() || sType.isBoolean()) {
435 			out.append(o);
436 		} else if (sType.isUri() || (nn(pMeta) && pMeta.isUri())) {
437 			out.uriValue(o);
438 		} else if (sType.isDate()) {
439 			out.stringValue(writeDate((Date)o, sType));
440 		} else if (sType.isCalendar()) {
441 			out.stringValue(writeCalendar(o, sType));
442 		} else if (sType.isTemporal()) {
443 			out.stringValue(writeTemporal((TemporalAccessor)o, sType));
444 		} else if (sType.isDuration()) {
445 			var value = writeDuration((Duration)o);
446 			if (getDurationFormat().isNumeric())
447 				out.append(value);
448 			else
449 				out.stringValue(value);
450 		} else if (sType.isPeriod()) {
451 			out.stringValue(writePeriod((Period)o));
452 		} else if (sType.isStreamable()) {
453 			writeStreamable(out, o, sType, eType);
454 		} else if (sType.isReader()) {
455 			pipe((Reader)o, out, SerializerSession::handleThrown);
456 		} else if (sType.isInputStream()) {
457 			pipe((InputStream)o, out, SerializerSession::handleThrown);
458 		} else {
459 			out.stringValue(toString(o));
460 		}
461 
462 		if (nn(wrapperAttr)) {
463 			indent--;
464 			out.cre(indent - 1).w('}');
465 		}
466 
467 		if (! isRecursion)
468 			pop();
469 		return out;
470 	}

472 	/**
473 	 * Serializes an object to a JSON string using this session's configuration.
474 	 *
475 	 * <p>
476 	 * Protected extension hook that subclasses can call or override when they need to serialize a
477 	 * value to JSON (e.g. for embedding a nested JSON fragment).
478 	 *
479 	 * @param o The object to serialize.
480 	 * @return The serialized object.
481 	 * @throws Exception Error occurred.
482 	 */
483 	@SuppressWarnings({
484 		"java:S112" // throws Exception intentional - callback/lifecycle method for serialization hooks
485 	})
486 	protected String writeJson(Object o) throws Exception {
487 		var sw = new StringWriter();
488 		writeAnything(getJsonWriter(createPipe(sw)).i(getInitialDepth()), o, getExpectedRootType(o), "root", null);
489 		return sw.toString();
490 	}
