29  /**
30   * Reference implementation of the public {@link TokenReader} surface for the JSON format.
31   *
32   * <p>
33   * Backed by the existing internal {@link ParserReader} plumbing.  Operates as a hand-rolled state
34   * machine over the JSON grammar &mdash; structural tokens (<c>{</c>, <c>}</c>, <c>[</c>, <c>]</c>,
35   * <c>,</c>, <c>:</c>) and the four scalar shapes (string, number, boolean, null).
36   *
37   * <p>
38   * Pipe-level settings (charset, auto-close, unbuffered) are configured on the
39   * {@link ParserPipe} the caller hands in.  Cursor-level behavior is driven by a small
40   * {@link Settings} bundle &mdash; currently just <c>trimStrings</c>.
41   *
42   * <h5 class='section'>Notes:</h5><ul>
43   * 	<li class='warn'>This class is not thread safe.
44   * 	<li>Whitespace and JSON comments are tolerated &mdash; comments are silently skipped to mirror
45   * 		Juneau's existing JSON parser behavior.
46   * 	<li>The cursor is purely structural; object swaps and {@code @Schema} annotations have no
47   * 		effect.  Use the POJO databind path on {@link JsonParser} for those.
48   * </ul>
49   */
50  @SuppressWarnings({
51  	"java:S3776", // Cognitive complexity acceptable for JSON tokenizer state machine
52  	"java:S6541", // Brain method acceptable for tokenizer dispatch
53  	"java:S115",  // Message-key constant uses UPPER_camelCase convention.
54  	"resource"    // The cursor's underlying ParserPipe is owned by the caller via try-with-resources on the cursor itself; Eclipse JDT flags the inner pipe as unclosed but that's by design.
55  })
56  public class JsonTokenReader implements TokenReader {

78  	//
79  	// The fields and methods below are part of JsonTokenReader's documented subclass-extension
80  	// contract.  Cross-package subclasses (Json5TokenReader in marshall.json5, JsonlTokenReader in
81  	// marshall.jsonl) need access to the cursor's internal state machine and per-token scratch
82  	// buffers in order to implement format dialects (single-quoted strings, bare identifiers,

88  	// indirection without improving safety.  In return, subclasses MUST treat them as

218 	// State-machine summary.  next() reads the next non-whitespace, non-comment character from the
219 	// underlying ParserReader and dispatches to one of the readXxx() methods below based on the
220 	// cursor's current state.  Each readXxx() method is overridable by subclasses to add
221 	// dialect-specific behavior.
222 	//
223 	//   S00_expectValue                 - At root, after [, after :, after , inside [.
224 	//                                        Expecting a JSON value: { [ " digit t f n.
225 	//                                        Dispatches to readValueStart.
226 	//   S01_expectFieldName             - After { or after , inside {.
227 	//                                        Expecting a quoted string field name (or }).
228 	//                                        Dispatches to readFieldNameOrEndObject.
229 	//   S02_expectColon                 - Just emitted a FIELD_NAME.  Expecting :.
230 	//                                        Handled inline by readColonThenValue.
231 	//   S03_expectCommaOrEndObject      - Just emitted a value inside {.  Expecting , or }.
232 	//                                        Dispatches to readCommaOrEndObject.
233 	//   S04_expectCommaOrEndArray       - Just emitted a value inside [.  Expecting , or ].
234 	//                                        Dispatches to readCommaOrEndArray.
235 	//   S05_end                         - Root document fully consumed.  next() returns
236 	//                                        END_OF_STREAM.
237 	//
238 	// State transitions are performed by afterValue() (sets the post-value state based on the
239 	// enclosing container) and by the dispatch methods themselves.
240 	// ==============================================================================================
241 	@Override /* Overridden from TokenReader */
242 	public TokenType next() throws IOException, ParseException {
243 		if (state == S05_end) {
244 			currentToken = TokenType.END_OF_STREAM;
245 			return currentToken;
246 		}
247 
248 		// Reset per-token scratch.
249 		currentFieldName = null;
250 		currentString = null;
251 		currentNumberLexeme = null;
252 		currentNumber = null;
253 
254 		int c = readSkipWsAndComments();
255 		if (c == -1) {
256 			if (state != S00_expectValue || depth != 0)
257 				throw parseException("Unexpected end of input");
258 			state = S05_end;
259 			currentToken = TokenType.END_OF_STREAM;
260 			return currentToken;
261 		}
262 
263 		switch (state) {
264 			case S00_expectValue                 -> readValueStart(c);
265 			case S01_expectFieldName             -> readFieldNameOrEndObject(c);
266 			case S02_expectColon                 -> readColonThenValue(c);
267 			case S03_expectCommaOrEndObject      -> readCommaOrEndObject(c);
268 			case S04_expectCommaOrEndArray       -> readCommaOrEndArray(c);
269 			default                              -> throw parseException("Internal tokenizer state error");
270 		}
271 		return currentToken;
272 	}

291 	@Override /* Overridden from TokenReader */
292 	public String getString() {
293 		switch (currentToken) {
294 			case VALUE_STRING:    return currentString;
295 			case FIELD_NAME:      return currentFieldName;
296 			case VALUE_NUMBER:    return currentNumberLexeme;
297 			case VALUE_BOOLEAN:   return Boolean.toString(currentBoolean);
298 			case VALUE_NULL:      return null;
299 			default: throw new IllegalStateException(
300 				"Current token does not have a string view (was " + currentToken + ")");
301 		}
302 	}

335 	@Override /* Overridden from TokenReader */
336 	public boolean canRead() throws IOException, ParseException {
337 		// Auto-advance through structural separators that don't themselves emit value tokens.
338 		// This lets users stream array elements with a canRead()/read() loop without manually
339 		// consuming commas.
340 		advanceToValueState();
341 		if (state != S00_expectValue)
342 			return false;
343 		// At S00_expectValue the next non-whitespace char might still be a closer (empty
344 		// container).  Peek to disambiguate.
345 		var c = readSkipWsAndComments();
346 		if (c == -1)
347 			return false;
348 		r.unread();
349 		return c != ']' && c != '}';
350 	}

352 	/**
353 	 * Helper for {@link #canRead()} and {@link #read(ClassMeta)}: if the cursor is
354 	 * sitting on a structural separator (comma between array elements, colon after a field name),
355 	 * consume it so the cursor lands at S00_expectValue.  No-op if already at a value
356 	 * boundary or at end-of-container.
357 	 */
358 	private void advanceToValueState() throws IOException, ParseException {
359 		while (true) {
360 			switch (state) {
361 				case S04_expectCommaOrEndArray: {
362 					var c = readSkipWsAndComments();
363 					if (c == -1 || c == ']') {
364 						if (c != -1) r.unread();
365 						return;
366 					}
367 					if (c != ',')
368 						throw parseException("Expected , or ] but got '%s'", (char) c);
369 					// If a trailing-comma case follows (next non-ws is ']'), keep the comma-or-end
370 					// state so a subsequent next() closes the container correctly.  Format-specific
371 					// hook: plain JSON rejects trailing commas, JSON5 accepts.
372 					var c2 = readSkipWsAndComments();
373 					if (c2 == ']') {
374 						r.unread();
375 						if (!allowsTrailingComma())
376 							throw parseException("Trailing comma not allowed in this format");
377 						return;  // state stays S04_expectCommaOrEndArray
378 					}
379 					if (c2 != -1) r.unread();
380 					state = S00_expectValue;
381 					break;
382 				}
383 				case S03_expectCommaOrEndObject: {
384 					var c = readSkipWsAndComments();
385 					if (c == -1 || c == '}') {
386 						if (c != -1) r.unread();
387 						return;
388 					}
389 					if (c != ',')
390 						throw parseException("Expected , or } but got '%s'", (char) c);
391 					var c2 = readSkipWsAndComments();
392 					if (c2 == '}') {
393 						r.unread();
394 						if (!allowsTrailingComma())
395 							throw parseException("Trailing comma not allowed in this format");
396 						return;  // state stays S03_expectCommaOrEndObject
397 					}
398 					if (c2 != -1) r.unread();
399 					// After a comma inside an object, the next thing is a field name, not a value.
400 					state = S01_expectFieldName;
401 					return;
402 				}
403 				case S02_expectColon: {
404 					var c = readSkipWsAndComments();
405 					if (c == -1)
406 						return;
407 					if (c != ':')
408 						throw parseException("Expected : after field name but got '%s'", (char) c);
409 					state = S00_expectValue;
410 					break;
411 				}
412 				default:
413 					return;
414 			}
415 		}
416 	}

438 	@Override /* Overridden from TokenReader */
439 	public <T> T read(ClassMeta<T> type) throws IOException, ParseException {
440 		var s = session();
441 		advanceToValueState();
442 		assertReadValueState();
443 		try {
444 			T o = s.readAnything(type, r, null, null);
445 			markValueConsumed();
446 			return o;
447 		} catch (Exception e) {
448 			if (e instanceof IOException e2) throw e2;
449 			if (e instanceof ParseException e2) throw e2;
450 			throw new ParseException(s, "read failed: %s", e.getMessage());
451 		}
452 	}

454 	@Override /* Overridden from TokenReader */
455 	public <T> T read(Type type, Type... args) throws IOException, ParseException {
456 		return read(session().<T>getClassMeta(type, args));
457 	}

468 	 * Validates that the cursor is positioned at a value boundary (i.e. the next emitted token
469 	 * would be value-shaped).
470 	 */
471 	private void assertReadValueState() {
472 		if (state != S00_expectValue)
473 			throw new IllegalStateException(MSG_READ_NON_VALUE_STATE);
474 	}

476 	/**
477 	 * Resyncs the cursor's state machine after {@code parseAnything} consumed a value.  The cursor's
478 	 * depth does not change (the value was contained at the current depth); only the post-value
479 	 * separator-expectation state needs to update.
480 	 */
481 	private void markValueConsumed() {
482 		// We did not surface an explicit token to the caller, but functionally this is "a value
483 		// just got emitted at the current depth," so the same post-value transition that next()
484 		// would have performed applies.
485 		afterValue();
486 		// The current token slot is meaningless after read (the caller has the bound POJO).
487 		// Set a sentinel so accidental getStringValue() / getCurrentToken() usage doesn't return
488 		// a stale prior token's value.
489 		currentToken = TokenType.NOT_AVAILABLE;
490 	}

492 	@Override /* Overridden from TokenReader */
493 	public void skipChildren() throws IOException, ParseException {
494 		if (currentToken != TokenType.START_OBJECT && currentToken != TokenType.START_ARRAY)
495 			return;
496 		var startDepth = depth - 1;
497 		while (depth > startDepth) {
498 			var t = next();
499 			if (t == TokenType.END_OF_STREAM)
500 				throw parseException("Unexpected end of input while skipping children");
501 		}
502 	}

511 	/**
512 	 * State-machine dispatch: handles the next character at {@link #S00_expectValue}.
513 	 *
514 	 * <p>
515 	 * Subclasses (e.g. Json5TokenReader) override this to add dialect-specific value shapes
516 	 * (single-quoted strings, bare identifiers, missing-value handling).
517 	 *
518 	 * @param c The next non-whitespace, non-comment character read from the input.
519 	 * @throws IOException If a problem occurred reading from the underlying stream.
520 	 * @throws ParseException If the input is malformed at this position.
521 	 */
522 	protected void readValueStart(int c) throws IOException, ParseException {
523 		switch (c) {
524 			case '{':
525 				pushContainer(false);
526 				currentToken = TokenType.START_OBJECT;
527 				state = S01_expectFieldName;
528 				return;
529 			case '[':
530 				pushContainer(true);
531 				currentToken = TokenType.START_ARRAY;
532 				state = S00_expectValue;
533 				return;
534 			case '"':
535 				r.unread();
536 				currentString = maybeTrim(readString());
537 				currentToken = TokenType.VALUE_STRING;
538 				afterValue();
539 				return;
540 			case 't', 'f', 'n':
541 				r.unread();
542 				readKeywordValue(c);
543 				afterValue();
544 				return;
545 			case ']':
546 				if (depth > 0 && isInArray()) {
547 					popContainer();
548 					currentToken = TokenType.END_ARRAY;
549 					afterValue();
550 					return;
551 				}
552 				throw parseException(MSG_unexpectedChar, (char) c);
553 			default:
554 				if (isNumberStart(c)) {
555 					r.unread();
556 					readNumberValue();
557 					afterValue();
558 					return;
559 				}
560 				throw parseException(MSG_unexpectedChar, (char) c);
561 		}
562 	}

564 	/**
565 	 * State-machine dispatch: handles the next character at {@link #S01_expectFieldName}.
566 	 *
567 	 * <p>
568 	 * Subclasses (e.g. Json5TokenReader) override this to accept bare-identifier field names and
569 	 * single-quoted field names.
570 	 *
571 	 * @param c The next non-whitespace character read from the input.
572 	 * @throws IOException If a problem occurred reading from the underlying stream.
573 	 * @throws ParseException If the input is malformed at this position.
574 	 */
575 	protected void readFieldNameOrEndObject(int c) throws IOException, ParseException {
576 		if (c == '}') {
577 			popContainer();
578 			currentToken = TokenType.END_OBJECT;
579 			afterValue();
580 			return;
581 		}
582 		if (c != '"')
583 			throw parseException("Expected \" for field name but got '%s'", (char) c);
584 		r.unread();
585 		currentFieldName = maybeTrim(readString());
586 		currentToken = TokenType.FIELD_NAME;
587 		state = S02_expectColon;
588 	}

590 	private String maybeTrim(String s) {
591 		if (!settings.trimStrings || s == null)
592 			return s;
593 		return s.trim();
594 	}

596 	private void readColonThenValue(int c) throws IOException, ParseException {
597 		if (c != ':')
598 			throw parseException("Expected : after field name but got '%s'", (char) c);
599 		var c2 = readSkipWsAndComments();
600 		if (c2 == -1)
601 			throw parseException("Unexpected end of input after ':'");
602 		state = S00_expectValue;
603 		readValueStart(c2);
604 	}

606 	/**
607 	 * State-machine dispatch: handles the next character at
608 	 * {@link #S03_expectCommaOrEndObject}.
609 	 *
610 	 * <p>
611 	 * Subclasses (e.g. Json5TokenReader) override this to accept trailing commas before <c>}</c>.
612 	 *
613 	 * @param c The next non-whitespace character read from the input.
614 	 * @throws IOException If a problem occurred reading from the underlying stream.
615 	 * @throws ParseException If the input is malformed at this position.
616 	 */
617 	protected void readCommaOrEndObject(int c) throws IOException, ParseException {
618 		if (c == '}') {
619 			popContainer();
620 			currentToken = TokenType.END_OBJECT;
621 			afterValue();
622 			return;
623 		}
624 		if (c != ',')
625 			throw parseException("Expected , or } but got '%s'", (char) c);
626 		var c2 = readSkipWsAndComments();
627 		if (c2 == -1)
628 			throw parseException("Unexpected end of input after ','");
629 		// After a comma inside an object the next thing must be a field name (or '}' is illegal here).
630 		state = S01_expectFieldName;
631 		readFieldNameOrEndObject(c2);
632 	}

634 	/**
635 	 * State-machine dispatch: handles the next character at
636 	 * {@link #S04_expectCommaOrEndArray}.
637 	 *
638 	 * <p>
639 	 * Subclasses (e.g. Json5TokenReader) override this to accept trailing commas before <c>]</c>.
640 	 *
641 	 * @param c The next non-whitespace character read from the input.
642 	 * @throws IOException If a problem occurred reading from the underlying stream.
643 	 * @throws ParseException If the input is malformed at this position.
644 	 */
645 	protected void readCommaOrEndArray(int c) throws IOException, ParseException {
646 		if (c == ']') {
647 			popContainer();
648 			currentToken = TokenType.END_ARRAY;
649 			afterValue();
650 			return;
651 		}
652 		if (c != ',')
653 			throw parseException("Expected , or ] but got '%s'", (char) c);
654 		var c2 = readSkipWsAndComments();
655 		if (c2 == -1)
656 			throw parseException("Unexpected end of input after ','");
657 		state = S00_expectValue;
658 		readValueStart(c2);
659 	}

661 	/**
662 	 * Sets the post-value state based on the enclosing container.
663 	 *
664 	 * <p>
665 	 * Subclasses (e.g. JsonlTokenReader) override this when the format treats consecutive
666 	 * top-level values differently &mdash; JSONL transitions back to {@link #S00_expectValue}
667 	 * at depth 0 instead of {@link #S05_end} so the next line is read.
668 	 */
669 	protected void afterValue() {
670 		if (depth == 0)
671 			state = S05_end;
672 		else if (isInArray())
673 			state = S04_expectCommaOrEndArray;
674 		else
675 			state = S03_expectCommaOrEndObject;
676 	}

680 	/**
681 	 * Reads a double-quoted JSON string from the current input position.
682 	 *
683 	 * <p>
684 	 * Handles escape sequences (<c>\\n</c>, <c>\\t</c>, <c>\\u<i>NNNN</i></c>, etc.) inline via
685 	 * {@link ParserReader#replace(char)}.  Subclasses (e.g. Json5TokenReader) override this to
686 	 * also accept single-quoted strings.
687 	 *
688 	 * @return The parsed string (without surrounding quotes).
689 	 * @throws IOException If a problem occurred reading from the underlying stream.
690 	 * @throws ParseException If the input is malformed at this position.
691 	 */
692 	protected String readString() throws IOException, ParseException {
693 		r.mark();
694 		var qc = r.read();
695 		if (qc != '"')
696 			throw parseException("Did not find quote character marking beginning of string");
697 		String s = null;
698 		var inEscape = false;
699 		var c = 0;
700 		while (c != -1) {
701 			c = r.read();
702 			if (c <= 0x1F && c != -1)
703 				throw parseException("Unescaped control character encountered: '0x%s'", String.format("%04X", c));
704 			if (inEscape) {
705 				switch (c) {
706 					case 'n': r.replace('\n'); break;
707 					case 'r': r.replace('\r'); break;
708 					case 't': r.replace('\t'); break;
709 					case 'f': r.replace('\f'); break;
710 					case 'b': r.replace('\b'); break;
711 					case '\\': r.replace('\\'); break;
712 					case '/': r.replace('/'); break;
713 					case '\'': r.replace('\''); break;
714 					case '"': r.replace('"'); break;
715 					case 'u': {
716 						var hex = r.read(4);
717 						try {
718 							r.replace(Integer.parseInt(hex, 16), 6);
719 						} catch (@SuppressWarnings("unused") NumberFormatException e) {
720 							throw parseException("Invalid Unicode escape sequence in string");
721 						}
722 						break;
723 					}
724 					default:
725 						throw parseException("Invalid escape sequence in string");
726 				}
727 				inEscape = false;
728 			} else {
729 				if (c == '\\') {
730 					inEscape = true;
731 					r.delete();
732 				} else if (c == qc) {
733 					s = r.getMarked(1, -1);
734 					break;
735 				}
736 			}
737 		}
738 		if (s == null)
739 			throw parseException("Unterminated string");
740 		return s;
741 	}

743 	/**
744 	 * Reads one of the JSON keyword literals (<c>true</c>, <c>false</c>, <c>null</c>) starting
745 	 * with the given lookahead character.  Sets {@link #currentToken} and {@link #currentBoolean}
746 	 * accordingly.
747 	 *
748 	 * @param c The first character of the keyword (one of <c>'t'</c>, <c>'f'</c>, <c>'n'</c>).
749 	 * @throws IOException If a problem occurred reading from the underlying stream.
750 	 * @throws ParseException If the keyword is malformed.
751 	 */
752 	protected void readKeywordValue(int c) throws IOException, ParseException {
753 		switch (c) {
754 			case 't':
755 				expectLiteral("true");
756 				currentBoolean = true;
757 				currentToken = TokenType.VALUE_BOOLEAN;
758 				return;
759 			case 'f':
760 				expectLiteral("false");
761 				currentBoolean = false;
762 				currentToken = TokenType.VALUE_BOOLEAN;
763 				return;
764 			case 'n':
765 				expectLiteral("null");
766 				currentToken = TokenType.VALUE_NULL;
767 				return;
768 			default:
769 				throw parseException(MSG_unexpectedChar, (char) c);
770 		}
771 	}

773 	private void expectLiteral(String literal) throws IOException, ParseException {
774 		var s = r.read(literal.length());
775 		if (!literal.equals(s))
776 			throw parseException("Unrecognized syntax.  Expected='%s', Actual='%s'", literal, s);
777 	}

779 	/**
780 	 * Reads a JSON number from the current input position.  Captures the lexeme via
781 	 * {@link ParserReader#readNumberString()} and validates it; defers actual numeric parsing
782 	 * to {@link #getNumber()} (lazy).  Sets {@link #currentNumberLexeme} and {@link #currentToken}.
783 	 *
784 	 * @throws IOException If a problem occurred reading from the underlying stream.
785 	 * @throws ParseException If the input is not a valid JSON number.
786 	 */
787 	protected void readNumberValue() throws IOException, ParseException {
788 		// Mirror JsonParserSession#parseNumber's lexeme-collection: collect "number chars" then
789 		// hand off to StringUtils.parseNumber.  The lexeme is preserved as currentNumberLexeme so
790 		// precision-sensitive consumers can recover the original textual form (resolved Q6).
791 		var s = r.readNumberString();
792 		validateNumberLexeme(s);
793 		currentNumberLexeme = s;
794 		currentToken = TokenType.VALUE_NUMBER;
795 		// Number itself is parsed lazily on first getNumberValue() call.
796 	}

798 	private void validateNumberLexeme(String s) throws ParseException {
799 		if (s.isEmpty())
800 			throw parseException(MSG_invalidNumber, s);
801 		var isNegative = false;
802 		var c = s.charAt(0);
803 		if (c == '-') {
804 			isNegative = true;
805 			c = (s.length() == 1 ? 'x' : s.charAt(1));
806 		}
807 		if (c == '.')
808 			throw parseException(MSG_invalidNumber, s);
809 		if (c == '0' && s.length() > (isNegative ? 2 : 1)) {
810 			var c2 = s.charAt(isNegative ? 2 : 1);
811 			if (c2 != '.' && c2 != 'e' && c2 != 'E')
812 				throw parseException(MSG_invalidNumber, s);
813 		}
814 		var i = s.indexOf('.');
815 		if (i != -1 && (s.length() == (i + 1) || !isDigit(s.charAt(i + 1))))
816 			throw parseException(MSG_invalidNumber, s);
817 	}

821 	/**
822 	 * Reads the next non-whitespace, non-comment character or -1 at end of stream.
823 	 *
824 	 * @return The next significant character, or <c>-1</c> at end of input.
825 	 * @throws IOException If a problem occurred reading from the underlying stream.
826 	 * @throws ParseException If a JSON comment is malformed.
827 	 */
828 	protected int readSkipWsAndComments() throws IOException, ParseException {
829 		while (true) {
830 			var c = r.read();
831 			if (c == -1)
832 				return -1;
833 			if (c == '/')
834 				skipComment();
835 			else if (!Character.isWhitespace(c))
836 				return c;
837 		}
838 	}

840 	@SuppressWarnings({
841 		"java:S2677" // read() return intentionally discarded: consuming the already-peeked '/' that closes a block comment.
842 	})
843 	private void skipComment() throws IOException, ParseException {
844 		var c = r.read();
845 		if (c == '*') {
846 			while (true) {
847 				c = r.read();
848 				if (c == -1)
849 					throw parseException("Unterminated block comment");
850 				if (c == '*' && r.peek() == '/') {
851 					r.read();
852 					return;
853 				}
854 			}
855 		}
856 		if (c == '/') {
857 			while (true) {
858 				c = r.read();
859 				if (c == -1 || c == '\n')
860 					return;
861 			}
862 		}
863 		throw parseException("Open ended comment");
864 	}

888 	/**
889 	 * Pushes a new container onto the cursor's depth stack.
890 	 *
891 	 * @param isArray <jk>true</jk> if the new container is an array; <jk>false</jk> for an object.
892 	 */
893 	protected void pushContainer(boolean isArray) {
894 		if (depth < 64) {
895 			if (isArray)
896 				containerBits |= (1L << depth);
897 			else
898 				containerBits &= ~(1L << depth);
899 		} else {
900 			if (containerOverflow == null || depth - 64 >= containerOverflow.length)
901 				growOverflow();
902 			containerOverflow[depth - 64] = isArray;
903 		}
904 		depth++;
905 	}
906 
907 	/**
908 	 * Pops the top container off the cursor's depth stack.
909 	 *
910 	 * @throws ParseException If the stack is already empty (defensive only; well-formed JSON
911 	 * 	cannot trigger this).
912 	 */
913 	protected void popContainer() throws ParseException {
914 		if (depth == 0)
915 			throw parseException("Container stack underflow");
916 		depth--;
917 	}
918 
919 	/**
920 	 * Returns whether the cursor's current container (the one most recently pushed) is an array.
921 	 *
922 	 * @return <jk>true</jk> if currently inside an array; <jk>false</jk> if inside an object.
923 	 */
924 	protected boolean isInArray() {
925 		var top = depth - 1;
926 		if (top < 64)
927 			return (containerBits & (1L << top)) != 0;
928 		return containerOverflow[top - 64];
929 	}
930 
931 	private void growOverflow() {
932 		var n = containerOverflow == null ? 16 : containerOverflow.length << 1;
933 		var bigger = new boolean[n];
934 		if (containerOverflow != null)
935 			System.arraycopy(containerOverflow, 0, bigger, 0, containerOverflow.length);
936 		containerOverflow = bigger;
937 	}
