81  	/**
82  	 * Formats a date/time or Duration value to its ISO 8601 string representation.
83  	 *
84  	 * <p>
85  	 * Convenience dispatcher that uses default per-type wire formats; callers that need format-aware
86  	 * output should use the per-type helpers directly:
87  	 * {@link #formatDate}, {@link #formatCalendar}, {@link #formatTemporal},
88  	 * {@link #formatDuration}, {@link #formatPeriod}.
89  	 *
90  	 * @param value The value to format.  <jk>null</jk> returns <jk>null</jk> (consistent with the per-type helpers).
91  	 * @param type The class metadata for the value (used for selecting the appropriate formatter for Temporal types).
92  	 * @param timeZone The session time zone (used when the value lacks zone info).
93  	 * @return The ISO 8601 string representation, or <jk>null</jk> if {@code value} is <jk>null</jk>.
94  	 */
95  	@SuppressWarnings({
96  		"java:S1172" // type kept for API compatibility; future callers may use it for per-type dispatch hints
97  	})
98  	public static String format(Object value, ClassMeta<?> type, TimeZone timeZone) {
99  		if (value == null)
100 			return null;
101 		if (value instanceof Duration value2)
102 			return formatDuration(value2, DurationFormat.ISO_8601_WITH_DAYS);
103 		if (value instanceof Period value2)
104 			return formatPeriod(value2, PeriodFormat.ISO_8601);
105 		if (value instanceof Calendar || value instanceof XMLGregorianCalendar)
106 			return formatCalendar(value, type, CalendarFormat.ISO_OFFSET_DATE_TIME, timeZone);
107 		if (value instanceof Date value2)
108 			return formatDate(value2, type, DateFormat.ISO_LOCAL_DATE_TIME, timeZone);
109 		if (value instanceof TemporalAccessor value2)
110 			return formatTemporal(value2, type, TemporalFormat.DEFAULT, timeZone);
111 		return value.toString();
112 	}

211 	 * Formats a date/time value as an ISO date (date-only, for OpenAPI 'date' format).
212 	 *
213 	 * @param value The value to format.  <jk>null</jk> returns <jk>null</jk>.
214 	 * @param type The class metadata.
215 	 * @param timeZone The session time zone.
216 	 * @return The ISO date string, or <jk>null</jk> if {@code value} is <jk>null</jk>.
217 	 */
218 	@SuppressWarnings({
219 		"java:S1172" // type kept for API compatibility; callers pass ClassMeta context for potential future use
220 	})
221 	public static String formatAsDate(Object value, ClassMeta<?> type, TimeZone timeZone) {
222 		if (value == null)
223 			return null;
224 		ZoneId zoneId = timeZone != null ? timeZone.toZoneId() : ZoneId.systemDefault();
225 		if (value instanceof Calendar value2) {
226 			ZonedDateTime zdt = (value2 instanceof GregorianCalendar value3) ? value3.toZonedDateTime() : value2.toInstant().atZone(value2.getTimeZone().toZoneId());
227 			return DateTimeFormatter.ISO_DATE.format(zdt);
228 		}
229 		if (value instanceof Date value2)
230 			return DateTimeFormatter.ISO_DATE.format(value2.toInstant().atZone(zoneId));
231 		if (value instanceof Temporal value2)
232 			return DateTimeFormatter.ISO_DATE.format(ZonedDateTime.from(new DefaultingTemporalAccessor(value2, zoneId)));
233 		return value.toString();
234 	}

268 	/**
269 	 * Parses an ISO 8601 string into the specified date/time target type.
270 	 *
271 	 * <p>
272 	 * Dispatches to the appropriate per-type parser based on the target's {@link ClassMeta} category:
273 	 * {@link #parseDate parseDate}, {@link #parseCalendar parseCalendar}, {@link #parseTemporal parseTemporal},
274 	 * {@link #parseDuration parseDuration}, or {@link #parsePeriod parsePeriod} — each using its default
275 	 * wire format.
276 	 *
277 	 * @param <T> The target type.
278 	 * @param iso8601 The ISO 8601 string to parse. <jk>null</jk> returns <jk>null</jk>.
279 	 * @param targetType The target class metadata.
280 	 * @param timeZone The session time zone (used for types that need a default zone).
281 	 * @return The parsed object, or <jk>null</jk> if {@code iso8601} is <jk>null</jk> or the target type is
282 	 * 	not a recognized date/time/duration/period type.
283 	 */
284 	@SuppressWarnings({
285 		"unchecked" // Type erasure requires cast for date/time type dispatch
286 	})
287 	public static <T> T parse(String iso8601, ClassMeta<T> targetType, TimeZone timeZone) {
288 		if (iso8601 == null)
289 			return null;
290 		if (targetType.isDate())
291 			return parseDate(iso8601, targetType, DateFormat.ISO_LOCAL_DATE_TIME, timeZone);
292 		if (targetType.isCalendar())
293 			return parseCalendar(iso8601, targetType, CalendarFormat.ISO_OFFSET_DATE_TIME, timeZone);
294 		if (targetType.isTemporal())
295 			return (T) parseTemporal(iso8601, (ClassMeta<? extends Temporal>)(ClassMeta<?>)targetType, TemporalFormat.DEFAULT, timeZone);
296 		if (targetType.isDuration())
297 			return (T) parseDuration(iso8601, DurationFormat.MILLIS);
298 		if (targetType.isPeriod())
299 			return (T) parsePeriod(iso8601, PeriodFormat.ISO_8601);
300 		return null;
301 	}

303 	/**
304 	 * Parses an ISO 8601 / numeric wire value into a {@link Duration}.
305 	 *
306 	 * <p>
307 	 * Handles ISO 8601 forms ({@code "PT1H"}, {@code "-PT30M"}, {@code "PT-6H"}, {@code "PT1.5S"}, …) as well
308 	 * as the bare-number wire forms produced by {@link DurationFormat#MILLIS}, {@link DurationFormat#SECONDS},
309 	 * {@link DurationFormat#NANOS}, and {@link DurationFormat#HOCON} (e.g. {@code "500ms"}, {@code "1.5s"}).
310 	 *
311 	 * <p>
312 	 * The {@code formatHint} disambiguates bare integer strings that could be milliseconds, nanos, or seconds —
313 	 * for all other numeric or ISO forms the input is self-describing and the hint is ignored.
314 	 *
315 	 * @param iso8601 The wire value to parse. <jk>null</jk> or empty returns <jk>null</jk>.
316 	 * @param formatHint The duration wire format used to disambiguate bare integer inputs. May be <jk>null</jk>
317 	 * 	(defaults to {@link DurationFormat#MILLIS}).
318 	 * @return The parsed {@link Duration}, or <jk>null</jk> if {@code iso8601} is <jk>null</jk> or empty.
319 	 */
320 	public static Duration parseDuration(String iso8601, DurationFormat formatHint) {
321 		if (iso8601 == null || iso8601.isEmpty())
322 			return null;
323 		String s = iso8601.trim();
324 		// Strip optional surrounding quotes
325 		if (s.length() >= 2 && s.startsWith("\"") && s.endsWith("\""))
326 			s = s.substring(1, s.length() - 1).trim();
327 		if (s.startsWith("PT-"))
328 			s = "-PT" + s.substring(3);
329 		// Number sniffing and explicit format hints.
330 		if (isIntegerLiteral(s))
331 			return (formatHint == null ? DurationFormat.MILLIS : formatHint).parse(s);
332 		if (isDecimalLiteral(s))
333 			return DurationFormat.SECONDS.parse(s);
334 		if (isHoconLiteral(s))
335 			return DurationFormat.HOCON.parse(s);
336 		// Use manual parser for full control (handles PTnH, PTnM, PTn.nS, -PTnH, PT-6H, etc.)
337 		Duration d = parseDurationManual(s);
338 		if (d != null)
339 			return d;
340 		// Fallback to standard parser
341 		try {
342 			return Duration.parse(s);
343 		} catch (Exception e) {
344 			throw toRex(e);
345 		}
346 	}

350 	 *
351 	 * <p>
352 	 * Handles standard ISO 8601 period strings ({@code "P1Y2M3D"}, …) as well as the bare-integer wire form

358 	 * @return The parsed {@link Period}, or <jk>null</jk> if {@code value} is <jk>null</jk> or empty.

369 	/**
370 	 * Parses an ISO 8601 wire value into a {@link Calendar} or {@link XMLGregorianCalendar}.
371 	 *
372 	 * <p>
373 	 * Dispatches by target type:
374 	 * <ul>
375 	 * 	<li>{@link XMLGregorianCalendar} targets always use {@link DatatypeFactory#newXMLGregorianCalendar(String)}
376 	 * 		and ignore {@code formatHint}.
377 	 * 	<li>{@link Calendar} / {@link GregorianCalendar} targets honor {@code formatHint} when set to anything
378 	 * 		other than {@link CalendarFormat#NOT_SET} or {@link CalendarFormat#ISO_OFFSET_DATE_TIME}; the default
379 	 * 		path uses content-based formatter autodetection.
380 	 * </ul>
381 	 *
382 	 * @param <T> The target type.
383 	 * @param iso8601 The wire value to parse. <jk>null</jk> returns <jk>null</jk>.
384 	 * @param targetType The target class metadata. Must represent a {@link Calendar} or {@link XMLGregorianCalendar}.
385 	 * @param formatHint The calendar wire format. May be <jk>null</jk> (treated as {@link CalendarFormat#NOT_SET}).
386 	 * @param timeZone The session time zone (used as the default zone when the wire value lacks zone info).
387 	 * @return The parsed value, or <jk>null</jk> if {@code iso8601} is <jk>null</jk> or the target is not a
388 	 * 	recognized calendar type.
389 	 */
390 	@SuppressWarnings({
391 		"unchecked" // Type erasure requires cast for Calendar/XMLGregorianCalendar dispatch
392 	})
393 	public static <T> T parseCalendar(String iso8601, ClassMeta<T> targetType, CalendarFormat formatHint, TimeZone timeZone) {
394 		if (iso8601 == null)
395 			return null;
396 		Class<T> tc = targetType.inner();
397 		if (XMLGregorianCalendar.class.isAssignableFrom(tc))
398 			return (T) datatypeFactory.newXMLGregorianCalendar(iso8601);
399 		if (! Calendar.class.isAssignableFrom(tc))
400 			return null;
401 		ZoneId zoneId = timeZone != null ? timeZone.toZoneId() : ZoneId.systemDefault();
402 		if (formatHint != null && formatHint != CalendarFormat.NOT_SET && formatHint != CalendarFormat.ISO_OFFSET_DATE_TIME)
403 			return (T) formatHint.parse(iso8601, zoneId);
404 		return (T) parseCalendarDefault(iso8601, zoneId);
405 	}

464 		if (! Temporal.class.isAssignableFrom(tc))
465 			return null;

470 		return (T) parseTemporalDefault(iso8601, ttc, zoneId);

473 	/**
474 	 * Hand-written state machine that parses the {@code PT}-prefixed body as a strict, gapless sequence of
475 	 * {@code number[whitespace]unit} components ({@code H}/{@code M}/{@code S}, case-insensitive).
476 	 *
477 	 * <p>
478 	 * Whitespace is permitted between a number and its unit and between/around components, but every
479 	 * non-whitespace character must belong to a valid component. The parser rejects (by returning <jk>null</jk>,
480 	 * the same signal used for any input it cannot handle) when it encounters stray characters, a malformed
481 	 * number (e.g. more than one decimal point, or a trailing dot with no fraction), or a number without a
482 	 * trailing unit. On rejection {@link #parseDuration} falls through to {@link Duration#parse(CharSequence)},
483 	 * which throws for genuinely invalid values. Numeric conversion uses the floating-point expression
484 	 * {@code (long)(value * 1_000_000_000)} to preserve historical rounding for well-formed inputs.
485 	 *
486 	 * @param s The candidate duration string (already trimmed/dequoted/PT-normalized by {@link #parseDuration}).
487 	 * @return The parsed {@link Duration}, or <jk>null</jk> if the string is too short, lacks a {@code PT}/{@code pt}
488 	 * 	prefix, contains no component, or contains any stray/malformed content.
489 	 */
490 	@SuppressWarnings({
491 		"java:S3776" // Hand-written ISO-8601 duration lexer/state-machine; decomposing the char-by-char scan would obscure the parse invariants without changing behavior.
492 	})
493 	private static Duration parseDurationManual(String s) {
494 		if (s == null || s.length() < 3)
495 			return null;
496 		boolean neg = s.startsWith("-");
497 		String s2 = neg ? s.substring(1) : s;
498 		if (!s2.startsWith("PT") && !s2.startsWith("pt"))
499 			return null;
500 		s2 = s2.substring(2);
501 		long totalNanos = 0;
502 		boolean found = false;
503 		int len = s2.length();
504 		int i = 0;
505 		while (i < len) {
506 			// Whitespace between/around components is skipped; any other non-component character is rejected.
507 			if (isRegexWhitespace(s2.charAt(i))) {
508 				i++;
509 				continue;
510 			}
511 			// A component must begin with a digit; any other character is stray junk → reject.
512 			if (!isAsciiDigit(s2.charAt(i)))
513 				return null;
514 			int start = i;
515 			while (i < len && isAsciiDigit(s2.charAt(i)))  // Integer part.
516 				i++;
517 			// Optional fraction: a single '.' that MUST be followed by at least one digit.
518 			if (i < len && s2.charAt(i) == '.') {
519 				i++;
520 				if (i >= len || !isAsciiDigit(s2.charAt(i)))
521 					return null;  // Trailing dot with no fraction → malformed.
522 				while (i < len && isAsciiDigit(s2.charAt(i)))
523 					i++;
524 			}
525 			int numEnd = i;
526 			while (i < len && isRegexWhitespace(s2.charAt(i)))  // Optional whitespace between number and unit.
527 				i++;
528 			// A unit designator is required; a number without one (or followed by stray chars) is malformed.
529 			if (i >= len || !isDurationUnit(s2.charAt(i)))
530 				return null;
531 			found = true;
532 			double v = Double.parseDouble(s2.substring(start, numEnd));
533 			long nanos = (long)(v * 1_000_000_000);
534 			char unit = Character.toUpperCase(s2.charAt(i));
535 			switch (unit) {
536 				case 'H': totalNanos += nanos * 3600; break;
537 				case 'M': totalNanos += nanos * 60; break;
538 				case 'S': totalNanos += nanos; break;
539 				default: break;  // HTT: isDurationUnit guarantees H/M/S
540 			}
541 			i++;  // Consume the unit.
542 		}
543 		if (!found)
544 			return null;
545 		Duration d = Duration.ofNanos(totalNanos);
546 		return neg ? d.negated() : d;
547 	}

549 	private static boolean isAsciiDigit(char c) {
550 		return c >= '0' && c <= '9';
551 	}

553 	/**
554 	 * Matches the regex {@code \s} character class (without the {@code UNICODE_CHARACTER_CLASS} flag).
555 	 */
556 	private static boolean isRegexWhitespace(char c) {
557 		return c == ' ' || c == '\t' || c == '\n' || c == '\u000B' || c == '\f' || c == '\r';
558 	}

560 	private static boolean isDurationUnit(char c) {
561 		return c == 'H' || c == 'h' || c == 'M' || c == 'm' || c == 'S' || c == 's';
562 	}

564 	/**
565 	 * State-machine equivalent of {@code ^[+-]?\d++$} — an optional sign followed by one or more ASCII digits.
566 	 *
567 	 * @param s The string to test.
568 	 * @return <jk>true</jk> if the entire string is an optionally-signed integer literal.
569 	 */
570 	private static boolean isIntegerLiteral(String s) {
571 		int len = s.length();
572 		int i = 0;
573 		if (i < len && (s.charAt(i) == '+' || s.charAt(i) == '-'))
574 			i++;
575 		if (i == len)
576 			return false;
577 		for (int k = i; k < len; k++)
578 			if (!isAsciiDigit(s.charAt(k)))
579 				return false;
580 		return true;
581 	}
582 
583 	/**
584 	 * State-machine equivalent of {@code ^[+-]?\d++\.\d++$} — an optionally-signed decimal with digits on both
585 	 * sides of a single dot.
586 	 *
587 	 * @param s The string to test.
588 	 * @return <jk>true</jk> if the entire string is an optionally-signed decimal literal.
589 	 */
590 	private static boolean isDecimalLiteral(String s) {
591 		int len = s.length();
592 		int i = 0;
593 		if (i < len && (s.charAt(i) == '+' || s.charAt(i) == '-'))
594 			i++;
595 		int intStart = i;
596 		while (i < len && isAsciiDigit(s.charAt(i)))
597 			i++;
598 		if (i == intStart)
599 			return false;
600 		if (i >= len || s.charAt(i) != '.')
601 			return false;
602 		i++;
603 		int fracStart = i;
604 		while (i < len && isAsciiDigit(s.charAt(i)))
605 			i++;
606 		if (i == fracStart)
607 			return false;
608 		return i == len;
609 	}

611 	/**
612 	 * State-machine equivalent of {@code ^[+-]?\d++(?:\.\d++)?(?:ns|us|ms|s|m|h|d)$} — an optionally-signed number
613 	 * (with optional fraction) followed by a HOCON time unit.
614 	 *
615 	 * @param s The string to test.
616 	 * @return <jk>true</jk> if the entire string is an optionally-signed HOCON duration literal.
617 	 */
618 	private static boolean isHoconLiteral(String s) {
619 		int len = s.length();
620 		int i = 0;
621 		if (i < len && (s.charAt(i) == '+' || s.charAt(i) == '-'))
622 			i++;
623 		int intStart = i;
624 		while (i < len && isAsciiDigit(s.charAt(i)))
625 			i++;
626 		if (i == intStart)
627 			return false;
628 		// Optional fraction consumed only when '.' is followed by at least one digit; otherwise left for the unit.
629 		if (i < len && s.charAt(i) == '.' && i + 1 < len && isAsciiDigit(s.charAt(i + 1))) {
630 			i++;
631 			while (i < len && isAsciiDigit(s.charAt(i)))
632 				i++;
633 		}
634 		String unit = s.substring(i);
635 		return unit.equals("ns") || unit.equals("us") || unit.equals("ms")
636 			|| unit.equals("s") || unit.equals("m") || unit.equals("h") || unit.equals("d");
637 	}

641 		var ta = new DefaultingTemporalAccessor(formatter.parse(iso8601), zoneId);
642 		return GregorianCalendar.from(ZonedDateTime.from(ta));

651 	@SuppressWarnings({
652 		"unchecked" // Type erasure requires cast for Temporal.from() reflection
653 	})
654 	private static <T extends Temporal> T parseTemporalDefault(String iso8601, Class<T> tc, ZoneId zoneId) {
655 		ZoneId offset = (tc == Instant.class) ? Z : zoneId;
656 		var formatter = getFormatterForType(tc);
657 		var ta = new DefaultingTemporalAccessor(formatter.parse(iso8601), offset);
658 
659 		try {
660 			var parseMethod = info(tc).getPublicMethod(
661 				x -> x.isStatic()
662 					&& x.isNotDeprecated()
663 					&& x.hasName("from")
664 					&& x.hasReturnType(tc)
665 					&& x.hasParameterTypes(TemporalAccessor.class)
666 			).map(MethodInfo::inner).orElse(null);
667 
668 			if (parseMethod != null)
669 				return (T) parseMethod.invoke(null, ta);
670 		} catch (IllegalAccessException | InvocationTargetException e) {
671 			throw toRex(e);
672 		}
673 
674 		throw iaex("Cannot parse ISO 8601 string into type: %s", tc.getName());
675 	}

677 	private static DateTimeFormatter getFormatterForType(Class<?> tc) {
678 		var f = DEFAULT_FORMATTERS.get(tc);
679 		return or(f, DateTimeFormatter.ISO_INSTANT);
680 	}

682 	/**
683 	 * Auto-detects the appropriate parser formatter based on the string content.
684 	 */
685 	private static DateTimeFormatter selectParserFormatter(String iso8601) {
686 		boolean hasTime = iso8601.contains("T");
687 		boolean hasZone = iso8601.endsWith("Z") || iso8601.contains("+")
688 			|| (iso8601.length() > 10 && iso8601.lastIndexOf('-') > 10);
689 		if (hasTime && hasZone)
690 			return DateTimeFormatter.ISO_OFFSET_DATE_TIME;
691 		if (hasTime)
692 			return DateTimeFormatter.ISO_LOCAL_DATE_TIME;
693 		if (hasZone)
694 			return DateTimeFormatter.ISO_DATE;
695 		return DateTimeFormatter.ISO_LOCAL_DATE;
696 	}

698 	/**
699 	 * Converts epoch milliseconds to the target date/time type.
700 	 *
701 	 * @param <T> The target type.
702 	 * @param epochMillis The epoch milliseconds value.
703 	 * @param targetType The target class metadata.
704 	 * @param timeZone The session time zone.
705 	 * @return The converted date/time object, or <jk>null</jk> if the target type is not a recognized date/time type.
706 	 */
707 	@SuppressWarnings({
708 		"unchecked" // Type erasure requires cast for date/time type conversion
709 	})
710 	public static <T> T fromEpochMillis(long epochMillis, ClassMeta<T> targetType, TimeZone timeZone) {
711 		Class<T> tc = targetType.inner();
712 		ZoneId zoneId = timeZone != null ? timeZone.toZoneId() : ZoneId.systemDefault();
713 		Instant instant = Instant.ofEpochMilli(epochMillis);
714 
715 		if (tc == Instant.class) return (T) instant;
716 		if (tc == ZonedDateTime.class) return (T) instant.atZone(zoneId);
717 		if (tc == OffsetDateTime.class) return (T) instant.atZone(zoneId).toOffsetDateTime();
718 		if (tc == LocalDateTime.class) return (T) instant.atZone(zoneId).toLocalDateTime();
719 		if (tc == LocalDate.class) return (T) instant.atZone(zoneId).toLocalDate();
720 		if (tc == LocalTime.class) return (T) instant.atZone(zoneId).toLocalTime();
721 		if (tc == OffsetTime.class) return (T) instant.atZone(zoneId).toOffsetDateTime().toOffsetTime();
722 		if (tc == Date.class) return (T) Date.from(instant);
723 		if (Calendar.class.isAssignableFrom(tc)) {
724 			var cal = GregorianCalendar.from(instant.atZone(zoneId));
725 			return (T) cal;
726 		}
727 		if (XMLGregorianCalendar.class.isAssignableFrom(tc)) {
728 			var gc = GregorianCalendar.from(instant.atZone(zoneId));
729 			return (T) datatypeFactory.newXMLGregorianCalendar(gc);
730 		}
731 
732 		return null;
733 	}
