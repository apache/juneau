31  /**
32   * Maintain the list of default swaps used by all serializers and parsers.
33   *
34   * <h5 class='section'>See Also:</h5><ul>
35   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/Swaps">Swap Basics</a>
36   * </ul>
37   */
38  public class DefaultSwaps {
39  
40  	/**
41  	 * Prevents instantiation.
42  	 */
43  	private DefaultSwaps() {}
44  
45  	private static final Map<Class<?>,ObjectSwap<?,?>> SWAPS = new ConcurrentHashMap<>();
46  	static {
47  		SWAPS.put(Locale.class, new StringSwap<Locale>() {
48  			@Override public String swap(MarshallingSession s, Locale o) {
49  				return s.getMarshallingContext().getLocaleFormat().format(o);
50  			}
51  			@Override public Locale unswap(MarshallingSession s, String o, ClassMeta<?> hint) {
52  				return s.getMarshallingContext().getLocaleFormat().parse(o);
53  			}
54  		});
55  		SWAPS.put(StackTraceElement.class, new StackTraceElementSwap());
56  		SWAPS.put(TimeZone.class, new StringSwap<TimeZone>() {
57  			@Override public String swap(MarshallingSession s, TimeZone o) {
58  				return s.getMarshallingContext().getTimeZoneFormat().format(o);
59  			}
60  			@Override public TimeZone unswap(MarshallingSession s, String o, ClassMeta<?> hint) {
61  				return TimeZoneFormat.parseTimeZone(o);
62  			}
63  		});
64  		SWAPS.put(ZoneId.class, new StringSwap<ZoneId>() {
65  			@Override public String swap(MarshallingSession s, ZoneId o) {
66  				return s.getMarshallingContext().getTimeZoneFormat().format(o);
67  			}
68  			@Override public ZoneId unswap(MarshallingSession s, String o, ClassMeta<?> hint) {
69  				return TimeZoneFormat.parseZoneId(o);
70  			}
71  		});
72  		SWAPS.put(Currency.class, new StringSwap<Currency>() {
73  			@Override public String swap(MarshallingSession s, Currency o) {
74  				return CurrencyFormat.format(o, s.getMarshallingContext().getCurrencyFormat(), s.getLocale());
75  			}
76  			@Override public Currency unswap(MarshallingSession s, String o, ClassMeta<?> hint) {
77  				return CurrencyFormat.parse(o, s.getMarshallingContext().getCurrencyFormat(), s.getLocale());
78  			}
79  		});
80  		SWAPS.put(MatchResult.class, new MatchResultSwap());
81  		SWAPS.put(URL.class, new UrlSwap());
82  		SWAPS.put(byte[].class, new BinarySwap());
83  		SWAPS.put(Class.class, new ClassFormatSwap());
84  	}
85  
86  	/**
87  	 * Find the default ObjectSwap for the specified class.
88  	 *
89  	 * @param ci The class to find the swap for. Must not be <jk>null</jk>.
90  	 * @return The matched swap, or <jk>null</jk> if it couldn't be found.
91  	 */
92  	@SuppressWarnings({
93  		"java:S1452"  // Wildcard required - ObjectSwap<?,?> for heterogeneous default swap types
94  	})
95  	public static ObjectSwap<?,?> find(ClassInfo ci) {
96  		var ci2 = ci.getAllParents().stream().filter(x -> nn(SWAPS.get(x.inner()))).findFirst().orElse(null);
97  		return ci2 == null ? null : SWAPS.get(ci2.inner());
98  	}
99  }
