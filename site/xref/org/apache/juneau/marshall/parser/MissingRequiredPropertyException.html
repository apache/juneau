17  package org.apache.juneau.marshall.parser;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.commons.bean.*;
22  
23  /**
24   * Thrown when parser input (or the source map of a {@code Map}-to-bean conversion) does not contain one or more
25   * properties marked {@link BeanProp#required() @BeanProp(required=true)}.
26   *
27   * <p>
28   * The check runs once per bean, after all input properties have been read and before the bean is constructed, so
29   * records and constructor-argument beans get this exception instead of a constructor failure.  An explicit
30   * <jk>null</jk> value counts as present.
31   *
32   * <p>
33   * This is a {@link ParseException}, so REST {@code @Content} parameters map it to <c>400 Bad Request</c>.
34   *
35   * <h5 class='section'>Example:</h5>
36   * <p class='bjava'>
37   * 	<jk>public record</jk> Person(<ja>@BeanProp</ja>(required=<jk>true</jk>) String <jv>name</jv>, Integer <jv>age</jv>) {}
38   *
39   * 	<jk>try</jk> {
40   * 		JsonParser.<jsf>DEFAULT</jsf>.read(<js>"{\"age\":30}"</js>, Person.<jk>class</jk>);
41   * 	} <jk>catch</jk> (MissingRequiredPropertyException <jv>e</jv>) {
42   * 		<jv>e</jv>.getBeanClass();       <jc>// Person.class</jc>
43   * 		<jv>e</jv>.getPropertyNames();   <jc>// [name]</jc>
44   * 	}
45   * </p>
46   *
47   * <h5 class='section'>See Also:</h5><ul>
48   * 	<li class='ja'>{@link BeanProp#required()}
49   * </ul>
50   *
51   * @serial exclude
52   */
53  @SuppressWarnings({
54  	"java:S110", // Inherited hierarchy depth from ParseException; acceptable for a specific parse failure.
55  	"java:S2166" // ParseException extends BasicRuntimeException, so the class is a runtime exception and the name is accurate.
56  })
57  public class MissingRequiredPropertyException extends ParseException {
58  
59  	private static final long serialVersionUID = 1L;
60  
61  	/**
62  	 * Throws this exception if the bean map is missing required properties.
63  	 *
64  	 * <p>
65  	 * Parser sessions call this through {@code ParserSession.checkRequired(BeanMap)}.  Conversion code that has no
66  	 * parser session passes <jk>null</jk>, which omits the input position from the message.
67  	 *
68  	 * @param <T> The bean type.
69  	 * @param session The current parser session, or <jk>null</jk> if not parsing.
70  	 * @param m The bean map to check.  Can be <jk>null</jk>.
71  	 * @return The same bean map, for chaining.
72  	 * @throws MissingRequiredPropertyException If a required property was never written.
73  	 */
74  	@SuppressWarnings({
75  		"java:S1168" // Pass-through for chaining: returns the argument itself, so null in gives null out
76  	})
77  	public static <T> BeanMap<T> check(ParserSession session, BeanMap<T> m) {
78  		if (m == null)
79  			return null;
80  		var missing = m.getMissingRequiredProperties();
81  		if (! missing.isEmpty())
82  			throw new MissingRequiredPropertyException(session, m.getMeta().getClassInfo().inner(), missing);
83  		return m;
84  	}
85  
86  	/**
87  	 * Finds a {@link MissingRequiredPropertyException} in the cause chain of a throwable.
88  	 *
89  	 * <p>
90  	 * Bean property setters wrap failures from nested conversions, so conversion code uses this to re-surface the
91  	 * original exception (which is a {@link ParseException} and maps to HTTP 400).
92  	 *
93  	 * @param t The throwable to inspect.  Can be <jk>null</jk>.
94  	 * @return The first exception of this type in the chain (including <c>t</c> itself), or <jk>null</jk> if none.
95  	 */
96  	public static MissingRequiredPropertyException find(Throwable t) {
97  		for (var i = 0; t != null && i < 32; i++, t = t.getCause())
98  			if (t instanceof MissingRequiredPropertyException e)
99  				return e;
100 		return null;
101 	}
102 
103 	private final Class<?> beanClass;
104 	private final String[] propertyNames;  // Array (not List) so the exception stays serializable.
105 
106 	/**
107 	 * Constructor.
108 	 *
109 	 * @param session The parser session, or <jk>null</jk>.  Used to add the input position to the message.
110 	 * @param beanClass The bean class.
111 	 * @param propertyNames The missing property names.
112 	 */
113 	public MissingRequiredPropertyException(ParserSession session, Class<?> beanClass, List<String> propertyNames) {
114 		super(session, "Missing required properties on bean class '%s': %s", beanClass.getSimpleName(), propertyNames);
115 		this.beanClass = beanClass;
116 		this.propertyNames = propertyNames.toArray(String[]::new);
117 	}
118 
119 	/**
120 	 * Returns the bean class whose required properties were missing.
121 	 *
122 	 * @return The bean class.  Never <jk>null</jk>.
123 	 */
124 	public Class<?> getBeanClass() { return beanClass; }
125 
126 	/**
127 	 * Returns the names of the missing required properties, in bean property order.
128 	 *
129 	 * @return An unmodifiable list.  Never <jk>null</jk> or empty.
130 	 */
131 	public List<String> getPropertyNames() { return List.of(propertyNames); }
132 }
