17  package org.apache.juneau.marshall.csv;
18  
19  import java.io.*;
20  
21  import org.apache.juneau.marshall.*;
22  import org.apache.juneau.marshall.serializer.*;
23  
24  /**
25   * Specialized writer for serializing CSV.
26   *
27   * <h5 class='section'>Notes:</h5><ul>
28   * 	<li class='note'>
29   * 		This class is not intended for external use.
30   * </ul>
31   */
32  @SuppressWarnings({
33  	"resource",  // Writer resource managed by calling code
34  	"java:S3776" // Cognitive complexity acceptable for CSV cell quoting/escaping logic
35  })
36  public class CsvWriter extends SerializerWriter<CsvWriter> {
37  
38  	/**
39  	 * Constructor.
40  	 *
41  	 * @param out The writer being wrapped.
42  	 * @param useWhitespace If <jk>true</jk>, tabs and spaces will be used in output.
43  	 * @param maxIndent The maximum indentation level.
44  	 * @param quoteChar The quote character to use (i.e. <js>'\''</js> or <js>'"'</js>)
45  	 * @param trimStrings If <jk>true</jk>, strings will be trimmed before being serialized.
46  	 * @param uriResolver The URI resolver for resolving URIs to absolute or root-relative form.
47  	 */
48  	protected CsvWriter(Writer out, boolean useWhitespace, int maxIndent, char quoteChar, boolean trimStrings, UriResolver uriResolver) {
49  		super(out, useWhitespace, maxIndent, trimStrings, quoteChar, uriResolver);
50  	}
51  
52  	/**
53  	 * Writes an entry to the writer.
54  	 *
55  	 * <p>
56  	 * Follows RFC 4180 quoting rules:
57  	 * <ul>
58  	 *   <li>Values containing commas, double-quote characters, or newlines are enclosed in double quotes.
59  	 *   <li>Double-quote characters within a quoted field are escaped by preceding them with another double-quote.
60  	 *   <li>The literal string {@code null} is written unquoted; the string value {@code "null"} is quoted.
61  	 * </ul>
62  	 *
63  	 * @param value The value to write.
64  	 * 	<br>Can be <jk>null</jk> (written as the literal <js>"null"</js>).
65  	 */
66  	public void writeEntry(Object value) {
67  		if (value == null) {
68  			w("null");
69  		} else {
70  			var s = value.toString();
71  			var mustQuote = false;
72  			for (var i = 0; i < s.length() && !mustQuote; i++) {
73  				var c = s.charAt(i);
74  				if (c == ',' || c == '"' || c == '\r' || c == '\n')
75  					mustQuote = true;
76  			}
77  			if (mustQuote) {
78  				w('"');
79  				for (var i = 0; i < s.length(); i++) {
80  					var c = s.charAt(i);
81  					if (c == '"')
82  						w('"');  // RFC 4180: escape embedded quote by doubling it
83  					w(c);
84  				}
85  				w('"');
86  			} else {
87  				w(s);
88  			}
89  		}
90  	}
91  }
