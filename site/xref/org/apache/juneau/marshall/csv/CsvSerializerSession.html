189 	/**
190 	 * Applies any registered object swap to the specified value.
191 	 *
192 	 * <p>
193 	 * If a swap is registered for the value's type, the value is transformed using the swap's
194 	 * {@code swap()} method before being serialized.
195 	 *
196 	 * @param value The value to potentially swap.
197 	 * @param type The class metadata of the value's type.
198 	 * @return The swapped value, or the original value if no swap is registered.
199 	 */
200 	@SuppressWarnings({
201 		"rawtypes" // Raw types necessary for ObjectSwap handling
202 	})
203 	private Object applySwap(Object value, ClassMeta<?> type) {
204 		try {
205 			if (value == null || type == null) // HTT: `type == null` (with value != null) is unreachable -- every call site passes `getClassMetaForObject(v)` (single-arg), which is null only when v itself is null, and value == null already short-circuits this condition.
206 				return value;
207 
208 			ObjectSwap swap = type.getSwap(this);
209 			if (nn(swap)) {
210 				return swap(swap, value);
211 			}
212 			if (type.isDate())
213 				return writeDate((Date)value, type);
214 			if (type.isCalendar())
215 				return writeCalendar(value, type);
216 			if (type.isTemporal())
217 				return writeTemporal((TemporalAccessor)value, type);
218 			if (type.isDuration())
219 				return writeDuration((Duration)value);
220 			if (type.isPeriod())
221 				return writePeriod((Period)value);
222 			return value;
223 		} catch (SerializeException e) {
224 			throw rex(e);
225 		}
226 	}

229 		"rawtypes", // Raw types necessary for generic type handling
230 		"unchecked" // Type erasure requires unchecked casts

232 	@Override /* Overridden from SerializerSession */
233 	protected void doWrite(SerializerPipe pipe, Object o) throws IOException, SerializeException {
234 		var cm = push2("root", o, getClassMetaForObject(o));
235 		if (cm == null)
236 			return;
237 		try {
238 			try (var w = getCsvWriter(pipe)) {
239 				Collection<?> l = null;
240 				if (cm.isArray()) {
241 				// Primitive arrays and byte[] serialize as single row (value = [1;2;3] or base64)
242 				if (o.getClass().getComponentType().isPrimitive())
243 					l = Collections.singletonList(o);
244 				else
245 					l = l((Object[])o);
246 			} else if (cm.isCollection()) {
247 				l = (Collection<?>)o;
248 			} else if (cm.isStreamable()) {
249 				// CSV must inspect first element for column headers, so materialization is unavoidable.
250 				l = toListFromStreamable(o, cm);
251 			} else {
252 				l = Collections.singleton(o);
253 			}
254 
255 			if (ine(l)) {
256 				var firstOpt = first(l);
257 				if (!firstOpt.isPresent())
258 					return;
259 
260 				// Apply any registered swap to the first element to determine column structure.
261 				var firstRaw = firstOpt.get();
262 				var firstEntry = applySwap(firstRaw, getClassMetaForObject(firstRaw));
263 				var entryType = getClassMetaForObject(firstEntry);
264 				// If swapped type is not a bean (e.g. interface proxy), use raw type for strategy
265 				if (! entryType.isBean() && firstRaw != null) { // HTT: `firstRaw != null` is unreachable when false -- `first(l)` (CollectionUtils) wraps `iterator.next()` via Optional.ofNullable, so `firstOpt.isPresent()` (checked just above) being true already guarantees firstRaw is non-null.
266 					var rawType = getClassMetaForObject(firstRaw);
267 					if (rawType.isBean())
268 						entryType = rawType;
269 				}
270 
271 				// Expected type for each element (for type discriminator)
272 				var eType = cm.isArray() || cm.isCollection() || cm.isStreamable()
273 					? cm.getElementType()
274 					: getExpectedRootType(firstRaw);
275 
276 				// CSV is flat; when addBeanTypes or addRootType is set, add _type column to all rows
277 				var addTypeColumn = isAddBeanTypes() || isAddRootType();
278 				var typeColName = addTypeColumn ? getBeanTypePropertyName(entryType) : null;
279 
280 				// Determine the best representation strategy.
281 				// Use the BeanMeta from the entry type for bean serialization.
282 				var bm = entryType.isBean() ? entryType.getBeanMeta() : null;
283 
284 				if (bm != null) {
285 					// Bean or DynaBean path: header row = property names + optional _type
286 					var addComma = Flag.create();
287 					bm.getProperties().values().stream().filter(BeanPropertyMeta::canRead).forEach(x -> {
288 						addComma.ifSet(() -> w.w(',')).set();
289 						w.writeEntry(x.getName());
290 					});
291 					// Always append _type when addBeanTypes or addRootType to support polymorphic parsing
292 					if (addTypeColumn && ine(typeColName)) { // HTT: `ine(typeColName)` is unreachable as false -- getBeanTypePropertyName(...) is documented to never return null, and every code path that can produce it (default "_type" or an annotation-supplied name filtered via Shorts::ine) is non-empty, so whenever addTypeColumn is true, typeColName is always non-empty.
293 						w.w(',');
294 						w.writeEntry(typeColName);
295 					}
296 					w.append('\n');
297 					var readableProps = bm.getProperties().values().stream().filter(BeanPropertyMeta::canRead).toList();
298 					l.forEach(x -> {
299 						var addComma2 = Flag.create();
300 						if (x == null) {
301 							// Null entry: write null marker for each column
302 							readableProps.forEach(y -> {
303 								addComma2.ifSet(() -> w.w(',')).set();
304 								w.writeEntry(nullValue);
305 							});
306 							if (addTypeColumn && ine(typeColName)) { // HTT: `ine(typeColName)` is unreachable as false -- see the HTT note above on the header-row check of the same expression.
307 								w.w(',');
308 								w.writeEntry("");
309 							}
310 						} else {
311 							// Apply swap before extracting bean properties (e.g. surrogate swaps)
312 							var swapped = applySwap(x, getClassMetaForObject(x));
313 							var aType = getClassMetaForObject(swapped);
314 							// When swap yields non-bean (e.g. String), use raw object for property extraction
315 							var objForBean = aType.isBean() ? swapped : x;
316 							BeanMap<?> bean = toBeanMap(objForBean);
317 							readableProps.forEach(y -> {
318 								addComma2.ifSet(() -> w.w(',')).set();
319 								var value = y.get(bean, y.getName());
320 								value = formatIfDateOrDuration(value);
321 								value = formatForCsvCell(value);
322 								// Use toString() to respect trimStrings setting on String values
323 								if (value instanceof String value2) value = toString(value2);
324 								w.writeEntry(or(value, nullValue));
325 							});
326 							if (addTypeColumn && ine(typeColName)) { // HTT: `ine(typeColName)` is unreachable as false -- see the HTT note above on the header-row check of the same expression.
327 								var typeName = getBeanTypeName(this, eType, aType, null);
328 								w.w(',');
329 								w.writeEntry(or(typeName, ""));
330 							}
331 						}
332 						w.w('\n');
333 					});
334 				} else if (entryType.isMap()) {
335 					// Map path: header row = map keys from the first entry + optional _type
336 					var addComma = Flag.create();
337 					var first = (Map) firstEntry;
338 					first.keySet().forEach(x -> {
339 						addComma.ifSet(() -> w.w(',')).set();
340 						// Apply trimStrings to map keys as well
341 						Object keyVal;
342 						if (x == null)
343 							keyVal = nullValue;
344 						else if (x instanceof String x2)
345 							keyVal = toString(x2);
346 						else
347 							keyVal = x;
348 						w.writeEntry(keyVal);
349 					});
350 					if (addTypeColumn && ine(typeColName)) { // HTT: `ine(typeColName)` is unreachable as false -- see the HTT note above on the bean path's header-row check of the same expression.
351 						w.w(',');
352 						w.writeEntry(typeColName);
353 					}
354 					w.append('\n');
355 					l.forEach(x -> {
356 						var addComma2 = Flag.create();
357 						var swapped = applySwap(x, getClassMetaForObject(x));
358 						var aType = getClassMetaForObject(swapped);
359 						var map = (Map) swapped;
360 						map.values().forEach(y -> {
361 							addComma2.ifSet(() -> w.w(',')).set();
362 							var value = applySwap(y, getClassMetaForObject(y));
363 							value = formatForCsvCell(value);
364 							// Apply trimStrings to map values
365 							if (value instanceof String value2) value = toString(value2);
366 							w.writeEntry(or(value, nullValue));
367 						});
368 						if (addTypeColumn && ine(typeColName)) { // HTT: `ine(typeColName)` is unreachable as false -- see the HTT note above on the bean path's header-row check of the same expression.
369 							var typeName = getBeanTypeName(this, eType, aType, null);
370 							w.w(',');
371 							w.writeEntry(or(typeName, ""));
372 						}
373 						w.w('\n');
374 					});
375 				} else {
376 					// Simple value path: single "value" column + optional _type
377 					w.writeEntry("value");
378 					if (addTypeColumn && ine(typeColName)) { // HTT: `ine(typeColName)` is unreachable as false -- see the HTT note above on the bean path's header-row check of the same expression.
379 						w.w(',');
380 						w.writeEntry(typeColName);
381 					}
382 					w.append('\n');
383 					l.forEach(x -> {
384 						var value = applySwap(x, getClassMetaForObject(x));
385 						value = formatForCsvCell(value);
386 						// Use toString() to respect trimStrings setting
387 						w.writeEntry(value != null ? toString(value) : nullValue);
388 						if (addTypeColumn && ine(typeColName)) { // HTT: `ine(typeColName)` is unreachable as false -- see the HTT note above on the bean path's header-row check of the same expression.
389 							if (x != null) {
390 								var aType = getClassMetaForObject(x);
391 								var typeName = getBeanTypeName(this, eType, aType, null);
392 								w.w(',');
393 								w.writeEntry(or(typeName, ""));
394 							} else {
395 								w.w(',');
396 								w.writeEntry("");
397 							}
398 						}
399 						w.w('\n');
400 					});
401 				}
402 			}
403 			} // HTT: try-with-resources close() bytecode -- getCsvWriter(pipe) always returns a writer chain ultimately backed by SerializerPipe.getWriter()'s NoCloseWriter (documented: "does not close the underlying writer"), so close() here can never throw; the compiler-generated null-check/primary-exception/addSuppressed branches for a throwing close() are therefore all unreachable from this call site.
404 		} finally {
405 			pop();
406 		}
407 	}

433 	 */
434 	private Object formatForCsvCell(Object value) {
435 		if (value == null)
436 			return null;
437 		if (allowNestedStructures) {
438 			var type = getClassMetaForObject(value);
439 			if (value instanceof Map || value instanceof Collection || value instanceof Object[]
440 					|| (type.isBean() && !(value instanceof Map))) // HTT: the `value instanceof Map == true` outcome of the trailing `!(value instanceof Map)` is unreachable here -- this disjunct is only evaluated once the first `value instanceof Map` check earlier in the same expression has already short-circuited to false for this same `value`, so it can never be true when re-checked.
441 				return new CsvCellSerializer(byteArrayFormat, nullValue).write(value, this);
442 		}
443 		if (value instanceof byte[] value2) {
444 			return byteArrayFormat == CsvByteArrayCellFormat.SEMICOLON_DELIMITED
445 				? formatByteArraySemicolon(value2)
446 				: base64Encode(value2);
447 		}
448 		if (value instanceof int[] value2) {
449 			var sb = new StringBuilder();
450 			sb.append('[');
451 			for (var i = 0; i < value2.length; i++) {
452 				if (i > 0) sb.append(';');
453 				sb.append(value2[i]);
454 			}
455 			sb.append(']');
456 			return sb.toString();
457 		}
458 		if (value instanceof long[] value2) {
459 			var sb = new StringBuilder();
460 			sb.append('[');
461 			for (var i = 0; i < value2.length; i++) {
462 				if (i > 0) sb.append(';');
463 				sb.append(value2[i]);
464 			}
465 			sb.append(']');
466 			return sb.toString();
467 		}
468 		if (value instanceof double[] value2) {
469 			var sb = new StringBuilder();
470 			sb.append('[');
471 			for (var i = 0; i < value2.length; i++) {
472 				if (i > 0) sb.append(';');
473 				sb.append(value2[i]);
474 			}
475 			sb.append(']');
476 			return sb.toString();
477 		}
478 		if (value instanceof float[] value2) {
479 			var sb = new StringBuilder();
480 			sb.append('[');
481 			for (var i = 0; i < value2.length; i++) {
482 				if (i > 0) sb.append(';');
483 				sb.append(value2[i]);
484 			}
485 			sb.append(']');
486 			return sb.toString();
487 		}
488 		if (value instanceof short[] value2) {
489 			var sb = new StringBuilder();
490 			sb.append('[');

497 		}

506 			return sb.toString();
507 		}
508 		if (value instanceof char[] value2) {

531 	 * Prepares a value for inline cell serialization (bean→Map, date format, etc.).
532 	 * Used by {@link CsvCellSerializer}.
533 	 */
534 	Object prepareForInlineValue(Object value) {
535 		if (value == null) // HTT: unreachable -- the only call site, CsvCellSerializer.writeValue(), already short-circuits on `value == null` (returning the null marker) before ever calling this method.
536 			return null; // HTT: unreachable -- see above.
