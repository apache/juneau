100 	private static ProtobufWriter getProtobufWriter(SerializerPipe out) throws IOException {
101 		var output = out.getRawOutput();
102 		if (output instanceof ProtobufWriter output2)
103 			return output2;
104 		var w = new ProtobufWriter(out.getOutputStream());
105 		out.setOutputStream(w);
106 		return w;
107 	}

109 	@Override /* Overridden from SerializerSession */
110 	protected void doWrite(SerializerPipe out, Object o) throws IOException, SerializeException {
111 		var w = getProtobufWriter(out);
112 		if (o == null)
113 			return;
114 		var cm = getClassMetaForObject(o);
115 		if (cm != null && (cm.isMap() && ! cm.isBean()) && ! cm.isBeanMap())
116 			throw new SerializeException("Protobuf binary serialization requires a bean root type, not a raw Map.");
117 		writeBean(w, toBeanMap(o), getClassMetaForObject(o));
118 		w.flush();
119 	}

121 	private void writeBean(ProtobufWriter out, BeanMap<?> m, ClassMeta<?> cm) throws SerializeException {
122 		var pcm = ctx.getProtobufClassMeta(cm);
123 
124 		// Collect non-null property values keyed by name (presence model:  null => omit).
125 		var values = new HashMap<String,Object>();
126 		m.forEachValue(Shorts::nn, (pMeta, key, value, thrown) -> {
127 			if (nn(thrown))
128 				onBeanGetterException(pMeta, thrown);
129 			else
130 				values.put(key, value);
131 		});
132 
133 		// Emit in field-number order for deterministic output.
134 		for (var entry : pcm.entries()) {
135 			var value = values.get(entry.name());
136 			if (value == null)
137 				continue;
138 			writeField(out, entry, value);
139 		}
140 	}

142 	private void writeField(ProtobufWriter out, ProtobufFieldEntry entry, Object value) throws SerializeException {
143 		var fn = entry.fieldNumber();
144 		switch (entry.kind()) {
145 			case SCALAR, MESSAGE -> writeSingle(out, fn, entry.scalarType(), value, entry.propertyType());
146 			case PACKED_REPEATED -> writePackedField(out, fn, entry, value);
147 			case TAGGED_REPEATED -> writeTaggedRepeatedField(out, fn, entry, value);
148 			case MAP -> writeMapField(out, fn, (Map)value, entry.propertyType());
149 		}
150 	}

152 	/**
153 	 * Serializes a single scalar or message value at the given field number, honoring any session-level
154 	 * {@link ObjectSwap} and deriving the on-wire shape (message vs. scalar) from the value's <b>runtime</b> type.
155 	 *
156 	 * <p>
157 	 * The protobuf field table ({@link ProtobufClassMeta}) is built once per bean from declared/raw property types and
158 	 * is session-independent, so the precomputed {@link ProtobufFieldEntry#kind() kind}/{@link
159 	 * ProtobufFieldEntry#scalarType() scalarType} cannot account for session-registered swaps.  This method therefore
160 	 * re-derives the shape at dispatch time, mirroring {@code CborSerializerSession.writeAnything}'s
161 	 * {@link ClassMeta#getSwap(MarshallingSession) getSwap(this)} step.
162 	 *
163 	 * <p>
164 	 * Resolving the swap from the <b>runtime</b> type (not the declared type) is what keeps this correct in both
165 	 * directions:
166 	 * <ul>
167 	 * 	<li>Bean <i>property</i> values whose declared type has a registered swap are already pre-swapped by the bean
168 	 * 		layer ({@code MarshalledPropertyPostProcessor.installSwapAwareTransforms} installs a read transform when the
169 	 * 		property's raw type {@code hasChildSwaps()}), so the value arriving here is the swap-class instance whose
170 	 * 		runtime {@code getSwap()} is <jk>null</jk> &mdash; no double swap, and the shape follows the swapped value.
171 	 * 	<li>Raw {@code List}/{@code Map} elements are <b>not</b> pre-swapped (the transform installs on the element type,
172 	 * 		not on the container property), so their runtime {@code getSwap()} fires here exactly once.
173 	 * </ul>
174 	 *
175 	 * <p>
176 	 * The no-swap path stays byte-identical to the golden vectors:  a non-bean/non-map runtime type is emitted as a
177 	 * scalar using the precomputed {@code declaredScalar} (preserving {@code @Protobuf(type=…)} overrides), and a
178 	 * bean/map runtime type is emitted as a length-delimited sub-message.
179 	 *
180 	 * @param out The protobuf writer.
181 	 * @param fn The protobuf field number.
182 	 * @param declaredScalar The precomputed scalar type for the no-swap scalar path, or {@link ProtobufScalarType#AUTO}.
183 	 * @param value The non-null value to serialize.
184 	 * @param declaredType The declared property (or element/key/value) type, used only as a runtime-type hint.
185 	 */
186 	private void writeSingle(ProtobufWriter out, int fn, ProtobufScalarType declaredScalar, Object value, ClassMeta<?> declaredType) throws SerializeException {
187 		var rType = getClassMetaForObject(value, declaredType);
188 		var swap = rType.getSwap(this);
189 		var v = value;
190 		var st = declaredScalar;
191 		if (swap != null) {
192 			v = swap(swap, value);
193 			if (v == null)
194 				return;  // Swapped to null => omit the field entirely (presence model).
195 			rType = swap.getSwapClassMeta(this);
196 			if (rType.isObject())
197 				rType = getClassMetaForObject(v);
198 			st = ProtobufClassMeta.defaultScalarType(rType);
199 		}
200 		if (rType.isBean() || rType.isMap())
201 			writeMessageField(out, fn, v, rType, declaredType);
202 		else
203 			writeScalarField(out, fn, st == ProtobufScalarType.AUTO ? ProtobufClassMeta.defaultScalarType(rType) : st, v, rType);
204 	}

206 	/**
207 	 * Writes a bean or Map value as a length-delimited sub-message.  Bean values continue through
208 	 * {@link #writeBean} as before; Map values (which {@link #toBeanMap} cannot handle -- it returns
209 	 * <jk>null</jk> for a non-bean Map, leading to an NPE) are instead written as a key(1)/value(2) sequence via
210 	 * {@link #writeMapMessage}, mirroring how {@code ProtobufParserSession.readMapMessage} reads them back.
211 	 *
212 	 * @param out The protobuf writer.
213 	 * @param fn The protobuf field number.
214 	 * @param value The non-null bean or Map value to serialize.
215 	 * @param rType The value's resolved runtime type (post-swap).
216 	 * @param declaredType The declared property (or element/key/value) type, preferred over {@code rType} for
217 	 * 	key/value type resolution when it is itself Map-shaped, since {@code rType} (derived purely from the
218 	 * 	runtime class) carries no generic key/value type information.
219 	 */
220 	private void writeMessageField(ProtobufWriter out, int fn, Object value, ClassMeta<?> rType, ClassMeta<?> declaredType) throws SerializeException {
221 		var aType = push2("field", value, rType);
222 		if (aType == null) {  // Recursion detected.
223 			pop();
224 			return;
225 		}
226 		try {
227 			if (rType.isMap()) {
228 				var mapType = declaredType != null && declaredType.isMap() ? declaredType : rType;
229 				writeMapMessage(out, fn, (Map)value, mapType);
230 			} else {
231 				var block = new ByteArrayOutputStream();
232 				writeBean(new ProtobufWriter(block), toBeanMap(value), aType);
233 				out.writeTag(fn, WireType.LEN);
234 				out.writeLenDelimited(block.toByteArray());
235 			}
236 		} finally {
237 			pop();
238 		}
239 	}

241 	/**
242 	 * Writes a raw (non-bean) Map value as a length-delimited sub-message containing key(1)/value(2) tag pairs,
243 	 * one pair per entry.  Used for Map targets reached as list elements, map values, or swap targets -- none of
244 	 * which have a bean shape for {@link #writeBean}/{@link #toBeanMap} to serialize.
245 	 *
246 	 * @param out The protobuf writer.
247 	 * @param fn The protobuf field number.
248 	 * @param map The non-null map to serialize.
249 	 * @param mapType The map's resolved type, used to determine the key/value scalar types.
250 	 */
251 	private void writeMapMessage(ProtobufWriter out, int fn, Map map, ClassMeta<?> mapType) throws SerializeException {
252 		var keyType = mapType.getKeyType();
253 		var valueType = mapType.getValueType();
254 		var keySt = ProtobufClassMeta.defaultScalarType(keyType);
255 		var valSt = ProtobufClassMeta.defaultScalarType(valueType);
256 		var block = new ByteArrayOutputStream();
257 		var bw = new ProtobufWriter(block);
258 		forEachEntry(map, e -> {
259 			var k = e.getKey();
260 			var v = e.getValue();
261 			if (k == null || v == null)
262 				return;
263 			writeSingle(bw, 1, keySt, k, keyType);
264 			writeSingle(bw, 2, valSt, v, valueType);
265 		});
266 		out.writeTag(fn, WireType.LEN);
267 		out.writeLenDelimited(block.toByteArray());
268 	}

270 	private void writePackedField(ProtobufWriter out, int fn, ProtobufFieldEntry entry, Object value) throws SerializeException {
271 		var elements = toElementList(value);
272 		if (elements.isEmpty())
273 			return;
274 		var st = entry.scalarType();
275 		var elType = entry.propertyType().getElementType();
276 		var block = new ByteArrayOutputStream();
277 		var bw = new ProtobufWriter(block);
278 		for (var el : elements) {
279 			if (el == null)
280 				continue;
281 			encodeScalarValue(bw, st, el, elType);
282 		}
283 		out.writeTag(fn, WireType.LEN);
284 		out.writeLenDelimited(block.toByteArray());
285 	}
286 
287 	private void writeTaggedRepeatedField(ProtobufWriter out, int fn, ProtobufFieldEntry entry, Object value) throws SerializeException {
288 		var elements = toElementList(value);
289 		var elType = entry.propertyType().getElementType();
290 		for (var el : elements) {
291 			if (el == null)
292 				continue;
293 			writeSingle(out, fn, entry.scalarType(), el, elType);
294 		}
295 	}

297 	private void writeMapField(ProtobufWriter out, int fn, Map map, ClassMeta<?> mapType) throws SerializeException {
298 		var keyType = mapType.getKeyType();
299 		var valueType = mapType.getValueType();
300 		var keySt = ProtobufClassMeta.defaultScalarType(keyType);
301 		var valSt = ProtobufClassMeta.defaultScalarType(valueType);
302 		forEachEntry(map, e -> {
303 			var k = e.getKey();
304 			var v = e.getValue();
305 			if (k == null || v == null)
306 				return;
307 			var block = new ByteArrayOutputStream();
308 			var bw = new ProtobufWriter(block);
309 			// Entry message:  key=field 1, value=field 2.  Keys and values honor any registered swap.
310 			writeSingle(bw, 1, keySt, k, keyType);
311 			writeSingle(bw, 2, valSt, v, valueType);
312 			out.writeTag(fn, WireType.LEN);
313 			out.writeLenDelimited(block.toByteArray());
314 		});
315 	}

317 	private void writeScalarField(ProtobufWriter out, int fn, ProtobufScalarType scalarType, Object value, ClassMeta<?> cm) throws SerializeException {
318 		var st = scalarType;
319 		out.writeTag(fn, st.wireType());
320 		encodeScalarValue(out, st, value, cm);
321 	}
322 
323 	private void encodeScalarValue(ProtobufWriter out, ProtobufScalarType st, Object value, ClassMeta<?> cm) throws SerializeException {
324 		switch (st) {
325 			case BOOL -> out.writeVarint(toBoolean(value) ? 1 : 0);
326 			case INT32 -> out.writeVarint(((Number)value).intValue());
327 			case INT64 -> out.writeVarint(((Number)value).longValue());
328 			case UINT32 -> out.writeVarint(((Number)value).intValue() & 0xFFFFFFFFL);
329 			case UINT64 -> out.writeVarint(toUint64Bits(value));
330 			case SINT32 -> out.writeZigZag32(((Number)value).intValue());
331 			case SINT64 -> out.writeZigZag64(((Number)value).longValue());
332 			case FIXED32, SFIXED32 -> out.writeFixed32(((Number)value).intValue());
333 			case FIXED64, SFIXED64 -> out.writeFixed64(((Number)value).longValue());
334 			case FLOAT -> out.writeFixed32(Float.floatToRawIntBits(((Number)value).floatValue()));
335 			case DOUBLE -> out.writeFixed64(Double.doubleToRawLongBits(((Number)value).doubleValue()));
336 			case STRING -> out.writeString(scalarToString(value, cm));
337 			case BYTES -> out.writeLenDelimited((byte[])value);
338 			case ENUM_INT -> out.writeVarint(((Enum<?>)value).ordinal());
339 			case ENUM_STRING -> out.writeString(((Enum<?>)value).name());
340 			default -> out.writeString(toString(value));  // HTT: AUTO is resolved to a concrete type before reaching here.
341 		}
342 	}

344 	private static List<Object> toElementList(Object value) {
345 		if (value instanceof Collection<?> value2)
346 			return new ArrayList<>(value2);
347 		if (value != null && value.getClass().isArray())
348 			return toList(value.getClass(), value);
349 		return l();
350 	}

352 	private static boolean toBoolean(Object value) {
353 		if (value instanceof Boolean value2)
354 			return value2;
355 		return Boolean.parseBoolean(value.toString());
356 	}

358 	private static long toUint64Bits(Object value) {
359 		if (value instanceof BigInteger value2)
360 			return value2.longValue();
361 		return ((Number)value).longValue();
362 	}

364 	private String scalarToString(Object value, ClassMeta<?> cm) throws SerializeException {
365 		if (cm.isDate())
366 			return writeDate((Date)value, cm);
367 		if (cm.isCalendar())
368 			return writeCalendar(value, cm);
369 		if (cm.isTemporal())
370 			return writeTemporal((TemporalAccessor)value, cm);
371 		if (cm.isDuration())
372 			return writeDuration((Duration)value);
373 		if (cm.isPeriod())
374 			return writePeriod((Period)value);
375 		return toString(value);
376 	}
