104 	private static ProtobufWriter getProtobufWriter(SerializerPipe out) throws IOException {
105 		var output = out.getRawOutput();
106 		if (output instanceof ProtobufWriter output2)
107 			return output2;
108 		var w = new ProtobufWriter(out.getOutputStream());
109 		out.setOutputStream(w);
110 		return w;
111 	}

113 	@Override /* Overridden from SerializerSession */
114 	protected void doWrite(SerializerPipe out, Object o) throws IOException, SerializeException {
115 		var w = getProtobufWriter(out);
116 		if (o == null)
117 			return;
118 		var cm = getClassMetaForObject(o);
119 		if (cm != null && (cm.isMap() && ! cm.isBean()) && ! cm.isBeanMap())  // HTT: two branches dead. `cm != null` -- getClassMetaForObject(o) with o guaranteed non-null (checked above) always resolves via getClassMeta(o.getClass()), which never returns null. `! cm.isBean()`'s false outcome -- ClassMeta.findBeanMeta() only computes a non-null BeanMeta when cat.isUnknown(), so cm.isMap()==true (a known category) structurally guarantees cm.isBean()==false; the two can never both be true.
120 			throw new SerializeException("Protobuf binary serialization requires a bean root type, not a raw Map.");
121 		writeBean(w, toBeanMap(o), getClassMetaForObject(o));
122 		w.flush();
123 	}

125 	private void writeBean(ProtobufWriter out, BeanMap<?> m, ClassMeta<?> cm) throws SerializeException {
126 		var pcm = ctx.getProtobufClassMeta(cm);
127 
128 		// Collect non-null property values keyed by name (presence model:  null => omit).
129 		var values = new HashMap<String,Object>();
130 		m.forEachValue(Shorts::nn, (pMeta, key, value, thrown) -> {
131 			if (nn(thrown))
132 				onBeanGetterException(pMeta, thrown);  // HTT: JaCoCo cannot register this line as covered -- onBeanGetterException is only ever called here with a non-null `thrown` when isIgnoreInvocationExceptionsOnGetters() is false (see BeanPropertyMeta#getRaw, which otherwise swallows the getter exception before it ever reaches this callback), and under that same flag onBeanGetterException unconditionally throws SerializeException. The call is exercised at runtime (verified via debug logging) but JaCoCo's post-invoke probe for this line is never reached because control never returns normally from the call.
133 			else
134 				values.put(key, value);
135 		});
136 
137 		// Emit in field-number order for deterministic output.
138 		for (var entry : pcm.entries()) {
139 			var value = values.get(entry.name());
140 			if (value == null)
141 				continue;
142 			writeField(out, entry, value);
143 		}
144 	}

146 	private void writeField(ProtobufWriter out, ProtobufFieldEntry entry, Object value) throws SerializeException {
147 		var fn = entry.fieldNumber();
148 		switch (entry.kind()) {  // HTT: the synthetic default arm (no case matched) is dead -- Kind has exactly 5 constants, all listed as case labels below.
149 			case SCALAR, MESSAGE -> writeSingle(out, fn, entry.scalarType(), value, entry.propertyType());
150 			case PACKED_REPEATED -> writePackedField(out, fn, entry, value);
151 			case TAGGED_REPEATED -> writeTaggedRepeatedField(out, fn, entry, value);
152 			case MAP -> writeMapField(out, fn, (Map)value, entry.propertyType());
153 		}
154 	}

156 	/**
157 	 * Serializes a single scalar or message value at the given field number, honoring any session-level
158 	 * {@link ObjectSwap} and deriving the on-wire shape (message vs. scalar) from the value's <b>runtime</b> type.
159 	 *
160 	 * <p>
161 	 * The protobuf field table ({@link ProtobufClassMeta}) is built once per bean from declared/raw property types and
162 	 * is session-independent, so the precomputed {@link ProtobufFieldEntry#kind() kind}/{@link
163 	 * ProtobufFieldEntry#scalarType() scalarType} cannot account for session-registered swaps.  This method therefore
164 	 * re-derives the shape at dispatch time, mirroring {@code CborSerializerSession.writeAnything}'s
165 	 * {@link ClassMeta#getSwap(MarshallingSession) getSwap(this)} step.
166 	 *
167 	 * <p>
168 	 * Resolving the swap from the <b>runtime</b> type (not the declared type) is what keeps this correct in both
169 	 * directions:
170 	 * <ul>
171 	 * 	<li>Bean <i>property</i> values whose declared type has a registered swap are already pre-swapped by the bean
172 	 * 		layer ({@code MarshalledPropertyPostProcessor.installSwapAwareTransforms} installs a read transform when the
173 	 * 		property's raw type {@code hasChildSwaps()}), so the value arriving here is the swap-class instance whose
174 	 * 		runtime {@code getSwap()} is <jk>null</jk> &mdash; no double swap, and the shape follows the swapped value.
175 	 * 	<li>Raw {@code List}/{@code Map} elements are <b>not</b> pre-swapped (the transform installs on the element type,
176 	 * 		not on the container property), so their runtime {@code getSwap()} fires here exactly once.
177 	 * </ul>
178 	 *
179 	 * <p>
180 	 * The no-swap path stays byte-identical to the golden vectors:  a non-bean/non-map runtime type is emitted as a
181 	 * scalar using the precomputed {@code declaredScalar} (preserving {@code @Protobuf(type=…)} overrides), and a
182 	 * bean/map runtime type is emitted as a length-delimited sub-message.
183 	 *
184 	 * @param out The protobuf writer.
185 	 * @param fn The protobuf field number.
186 	 * @param declaredScalar The precomputed scalar type for the no-swap scalar path, or {@link ProtobufScalarType#AUTO}.
187 	 * @param value The non-null value to serialize.
188 	 * @param declaredType The declared property (or element/key/value) type, used only as a runtime-type hint.
189 	 */
190 	private void writeSingle(ProtobufWriter out, int fn, ProtobufScalarType declaredScalar, Object value, ClassMeta<?> declaredType) throws SerializeException {
191 		var rType = getClassMetaForObject(value, declaredType);
192 		var swap = rType.getSwap(this);
193 		var v = value;
194 		var st = declaredScalar;
195 		if (swap != null) {
196 			v = swap(swap, value);
197 			if (v == null)
198 				return;  // Swapped to null => omit the field entirely (presence model).
199 			rType = swap.getSwapClassMeta(this);
200 			if (rType.isObject())
201 				rType = getClassMetaForObject(v);
202 			st = ProtobufClassMeta.defaultScalarType(rType);
203 		}
204 		if (rType.isBean() || rType.isMap())
205 			writeMessageField(out, fn, v);
206 		else
207 			writeScalarField(out, fn, st == ProtobufScalarType.AUTO ? ProtobufClassMeta.defaultScalarType(rType) : st, v, rType);
208 	}

210 	private void writeMessageField(ProtobufWriter out, int fn, Object value) throws SerializeException {
211 		var aType = push2("field", value, getClassMetaForObject(value));
212 		if (aType == null) {  // Recursion detected.
213 			pop();
214 			return;
215 		}
216 		try {
217 			var block = new ByteArrayOutputStream();
218 			writeBean(new ProtobufWriter(block), toBeanMap(value), aType);
219 			out.writeTag(fn, WireType.LEN);
220 			out.writeLenDelimited(block.toByteArray());
221 		} finally {
222 			pop();
223 		}
224 	}

226 	private void writePackedField(ProtobufWriter out, int fn, ProtobufFieldEntry entry, Object value) throws SerializeException {
227 		var elements = toElementList(value);
228 		if (elements.isEmpty())
229 			return;
230 		var st = entry.scalarType();
231 		var elType = entry.propertyType().getElementType();
232 		var block = new ByteArrayOutputStream();
233 		var bw = new ProtobufWriter(block);
234 		for (var el : elements) {
235 			if (el == null)
236 				continue;
237 			encodeScalarValue(bw, st, el, elType);
238 		}
239 		out.writeTag(fn, WireType.LEN);
240 		out.writeLenDelimited(block.toByteArray());
241 	}
242 
243 	private void writeTaggedRepeatedField(ProtobufWriter out, int fn, ProtobufFieldEntry entry, Object value) throws SerializeException {
244 		var elements = toElementList(value);
245 		var elType = entry.propertyType().getElementType();
246 		for (var el : elements) {
247 			if (el == null)
248 				continue;
249 			writeSingle(out, fn, entry.scalarType(), el, elType);
250 		}
251 	}

253 	private void writeMapField(ProtobufWriter out, int fn, Map map, ClassMeta<?> mapType) throws SerializeException {
254 		var keyType = mapType.getKeyType();
255 		var valueType = mapType.getValueType();
256 		var keySt = ProtobufClassMeta.defaultScalarType(keyType);
257 		var valSt = ProtobufClassMeta.defaultScalarType(valueType);
258 		forEachEntry(map, e -> {
259 			var k = e.getKey();
260 			var v = e.getValue();
261 			if (k == null || v == null)
262 				return;
263 			var block = new ByteArrayOutputStream();
264 			var bw = new ProtobufWriter(block);
265 			// Entry message:  key=field 1, value=field 2.  Keys and values honor any registered swap.
266 			writeSingle(bw, 1, keySt, k, keyType);
267 			writeSingle(bw, 2, valSt, v, valueType);
268 			out.writeTag(fn, WireType.LEN);
269 			out.writeLenDelimited(block.toByteArray());
270 		});
271 	}

273 	private void writeScalarField(ProtobufWriter out, int fn, ProtobufScalarType scalarType, Object value, ClassMeta<?> cm) throws SerializeException {
274 		var st = scalarType;
275 		out.writeTag(fn, st.wireType());
276 		encodeScalarValue(out, st, value, cm);
277 	}

279 	private void encodeScalarValue(ProtobufWriter out, ProtobufScalarType st, Object value, ClassMeta<?> cm) throws SerializeException {
280 		switch (st) {  // HTT: the default arm (AUTO) is dead -- AUTO is resolved to a concrete scalar type before encodeScalarValue is ever invoked (see writeSingle/writeScalarField).
281 			case BOOL -> out.writeVarint(toBoolean(value) ? 1 : 0);
282 			case INT32 -> out.writeVarint(((Number)value).intValue());
283 			case INT64 -> out.writeVarint(((Number)value).longValue());
284 			case UINT32 -> out.writeVarint(((Number)value).intValue() & 0xFFFFFFFFL);
285 			case UINT64 -> out.writeVarint(toUint64Bits(value));
286 			case SINT32 -> out.writeZigZag32(((Number)value).intValue());
287 			case SINT64 -> out.writeZigZag64(((Number)value).longValue());
288 			case FIXED32, SFIXED32 -> out.writeFixed32(((Number)value).intValue());
289 			case FIXED64, SFIXED64 -> out.writeFixed64(((Number)value).longValue());
290 			case FLOAT -> out.writeFixed32(Float.floatToRawIntBits(((Number)value).floatValue()));
291 			case DOUBLE -> out.writeFixed64(Double.doubleToRawLongBits(((Number)value).doubleValue()));
292 			case STRING -> out.writeString(scalarToString(value, cm));
293 			case BYTES -> out.writeLenDelimited((byte[])value);
294 			case ENUM_INT -> out.writeVarint(((Enum<?>)value).ordinal());
295 			case ENUM_STRING -> out.writeString(((Enum<?>)value).name());
296 			default -> out.writeString(toString(value));  // HTT: AUTO is resolved to a concrete type before reaching here.
297 		}
298 	}

300 	private static List<Object> toElementList(Object value) {
301 		if (value instanceof Collection<?> value2)
302 			return new ArrayList<>(value2);
303 		if (value != null && value.getClass().isArray())  // HTT: both conjuncts are dead -- `value != null` per the caller invariant above, and `isArray()`'s false outcome would require a repeated-field value that is neither a Collection (already excluded above) nor an array, contradicting the same PACKED_REPEATED/TAGGED_REPEATED invariant noted below.
304 			return toList(value.getClass(), value);
305 		return l();  // HTT: dead -- PACKED_REPEATED/TAGGED_REPEATED entries are only assigned by ProtobufClassMeta.kindOf for properties whose declared type is a Collection or array (`cm.isCollection() || cm.isArray()`), so the runtime value here is always one or the other.
306 	}

314 	private static long toUint64Bits(Object value) {
315 		if (value instanceof BigInteger value2)
316 			return value2.longValue();
317 		return ((Number)value).longValue();

320 	private String scalarToString(Object value, ClassMeta<?> cm) throws SerializeException {
321 		if (cm.isDate())
322 			return writeDate((Date)value, cm);
323 		if (cm.isCalendar())
324 			return writeCalendar(value, cm);
325 		if (cm.isTemporal())
326 			return writeTemporal((TemporalAccessor)value, cm);
327 		if (cm.isDuration())
328 			return writeDuration((Duration)value);
329 		if (cm.isPeriod())
330 			return writePeriod((Period)value);
331 		return toString(value);
332 	}
