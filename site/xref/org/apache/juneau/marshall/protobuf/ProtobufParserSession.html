102 	@Override /* Overridden from ParserSession */
103 	protected <T> T doRead(ParserPipe pipe, ClassMeta<T> type) throws IOException, ParseException, ExecutableException {
104 		var is = new ProtobufReader(pipe.getInputStream());
105 		is.setMaxLength(ctx.getMaxLength());
106 		if (! type.isBean())
107 			throw new ParseException(this, "Protobuf binary parsing requires a bean target type, but got '%s'", cn(type.inner()));
108 		return (T)readMessage(type, is, getOuter());
109 	}

111 	@SuppressWarnings({
112 		"java:S3776", // Cognitive complexity acceptable for the tag-loop dispatch
113 		"java:S6541"  // Brain method acceptable for the parse workhorse
114 	})
115 	private Object readMessage(ClassMeta<?> type, ProtobufReader is, Object outer) throws IOException, ParseException, ExecutableException {
116 		var pcm = ctx.getProtobufClassMeta(type);
117 		var m = newBeanMap(outer, type.inner());
118 
119 		var repeated = new LinkedHashMap<Integer,List<Object>>();
120 		var maps = new LinkedHashMap<Integer,Map<Object,Object>>();
121 
122 		long tag;
123 		while ((tag = is.readTag()) != ProtobufReader.EOF) {
124 			var fn = ProtobufReader.fieldNumber(tag);
125 			var wt = ProtobufReader.wireType(tag);
126 			var entry = pcm.entryFor(fn);
127 			if (entry == null) {
128 				is.skipField(wt);
129 				continue;
130 			}
131 			switch (entry.kind()) { // HTT: the synthetic default arm (no case matched) is dead -- Kind has exactly 5 constants, all listed as case labels above.
132 				case SCALAR -> setProperty(m, entry, decodeSingle(entry.scalarType(), entry.propertyType(), is, m.getBean(false)));
133 				case MESSAGE -> setProperty(m, entry, decodeSingle(ProtobufScalarType.AUTO, entry.propertyType(), is, m.getBean(false)));
134 				case PACKED_REPEATED -> {
135 					var list = repeated.computeIfAbsent(fn, k -> l());
136 					var elType = entry.propertyType().getElementType();
137 					if (wt == WireType.LEN) {
138 						var sub = is.readLenDelimitedReader();
139 						while (sub.available() > 0)
140 							list.add(decodeScalar(entry.scalarType(), elType, sub));
141 					} else {
142 						// Interop tolerance:  accept unpacked repeated scalars too.
143 						list.add(decodeScalar(entry.scalarType(), elType, is));
144 					}
145 				}
146 				case TAGGED_REPEATED -> {
147 					var list = repeated.computeIfAbsent(fn, k -> l());
148 					var elType = entry.propertyType().getElementType();
149 					var elSwap = elType == null ? null : elType.getSwap(this); // HTT: elType is never null here -- ProtobufClassMeta.kindOf only assigns TAGGED_REPEATED after calling cm.getElementType() unconditionally, so a null element type would already have failed field-table construction.
150 					if (elSwap != null)
151 						list.add(decodeSwapped(elSwap, elType, is, m.getBean(false)));
152 					else if (elType.isBean() || elType.isMap())
153 						list.add(readMessage(elType, is.readLenDelimitedReader(), m.getBean(false)));
154 					else
155 						list.add(decodeScalar(entry.scalarType(), elType, is));
156 				}
157 				case MAP -> {
158 					var map = maps.computeIfAbsent(fn, k -> m());
159 					readMapEntry(entry.propertyType(), is.readLenDelimitedReader(), map);
160 				}
161 			}
162 		}
163 
164 		repeated.forEach((fn, list) -> {
165 			var entry = pcm.entryFor(fn);
166 			setProperty(m, entry, convertToType(list, entry.propertyType()));
167 		});
168 		maps.forEach((fn, map) -> {
169 			var entry = pcm.entryFor(fn);
170 			setProperty(m, entry, convertToType(map, entry.propertyType()));
171 		});
172 
173 		return m.getBean();
174 	}

176 	/**
177 	 * Decodes a single scalar or message value, honoring any session-level {@link ObjectSwap} registered on the
178 	 * declared type.
179 	 *
180 	 * <p>
181 	 * Mirrors the serializer's {@code trySerializeSwapped} step:  the protobuf field table is built from declared/raw
182 	 * types and is session-independent, so per-property swaps are resolved here at parse time via
183 	 * {@link ClassMeta#getSwap(MarshallingSession) getSwap(this)}.  When a swap is present
184 	 * the wire value is decoded according to the <i>swapped</i> type's shape (message vs. scalar) and then run back
185 	 * through {@link #unswap(ObjectSwap, Object, ClassMeta) unswap}.  When absent the original behavior is preserved:
186 	 * an {@link ProtobufScalarType#AUTO} hint indicates a {@code MESSAGE} field (length-delimited sub-message), any
187 	 * other hint a plain scalar.
188 	 *
189 	 * <p>
190 	 * Property-level swaps already installed on the bean property are not re-applied here (the declared type is then
191 	 * the swapped type, whose {@code getSwap()} is <jk>null</jk>); {@link #setProperty(BeanMap, ProtobufFieldEntry, Object)}'s
192 	 * setter performs that unswap.  Only class/session-registered swaps surface here.
193 	 *
194 	 * @param declaredScalar The precomputed scalar type, or {@link ProtobufScalarType#AUTO} for a message field.
195 	 * @param declaredType The declared property type.
196 	 * @param is The protobuf reader positioned at the value.
197 	 * @param outer The outer bean (for nested-message parent wiring).
198 	 * @return The decoded (and unswapped, if applicable) value.
199 	 */
200 	private Object decodeSingle(ProtobufScalarType declaredScalar, ClassMeta<?> declaredType, ProtobufReader is, Object outer) throws IOException, ParseException, ExecutableException {
201 		var swap = declaredType == null ? null : declaredType.getSwap(this); // HTT: declaredType is never null here -- always called with entry.propertyType(), which ProtobufFieldEntry always populates from the real bean property's ClassMeta.
202 		if (swap != null)
203 			return decodeSwapped(swap, declaredType, is, outer);
204 		if (declaredScalar == ProtobufScalarType.AUTO)
205 			return readMessage(declaredType, is.readLenDelimitedReader(), outer);
206 		return decodeScalar(declaredScalar, declaredType, is);
207 	}
208 
209 	/**
210 	 * Decodes a value whose declared type has a registered {@link ObjectSwap}, then unswaps it back to the declared
211 	 * type.  The on-wire shape is determined by the swap's swap-class:  a bean/map swap-class is read as a
212 	 * length-delimited sub-message, anything else as a scalar of the swap-class's default scalar type.
213 	 *
214 	 * @param swap The resolved object swap.
215 	 * @param declaredType The declared (unswapped) type, used as the unswap hint.
216 	 * @param is The protobuf reader positioned at the value.
217 	 * @param outer The outer bean (for nested-message parent wiring).
218 	 * @return The unswapped value.
219 	 */
220 	private Object decodeSwapped(ObjectSwap<?,?> swap, ClassMeta<?> declaredType, ProtobufReader is, Object outer) throws IOException, ParseException, ExecutableException {
221 		var sType = swap.getSwapClassMeta(this);
222 		Object sval;
223 		if (sType.isBean() || sType.isMap())
224 			sval = readMessage(sType, is.readLenDelimitedReader(), outer);
225 		else
226 			sval = decodeScalar(ProtobufClassMeta.defaultScalarType(sType), sType, is);
227 		return unswap(swap, sval, declaredType);
228 	}

230 	private void setProperty(BeanMap<?> m, ProtobufFieldEntry entry, Object value) {
231 		var cm = entry.propertyType();
232 		try {
233 			entry.property().set(m, entry.name(), convertToType(value, cm));
234 		} catch (BeanRuntimeException e) {
235 			onBeanSetterException(entry.property(), e);
236 			throw e;
237 		}
238 	}

240 	private Object decodeScalar(ProtobufScalarType st, ClassMeta<?> cm, ProtobufReader is) throws IOException, ParseException {
241 		switch (st) { // HTT: the synthetic default arm's dispatch is dead for the same reason as the default body below (L259) -- AUTO is resolved to a concrete type before decodeScalar is ever invoked.
242 			case BOOL: return is.readVarint() != 0;
243 			case INT32: return is.readVarint();
244 			case INT64: return is.readVarint();
245 			case UINT32: return is.readVarint() & 0xFFFFFFFFL;
246 			case UINT64: return decodeUint64(cm, is);
247 			case SINT32: return (long)is.readZigZag32();
248 			case SINT64: return is.readZigZag64();
249 			case FIXED32: return is.readFixed32() & 0xFFFFFFFFL;
250 			case SFIXED32: return (long)is.readFixed32();
251 			case FIXED64: return is.readFixed64();
252 			case SFIXED64: return is.readFixed64();
253 			case FLOAT: return Float.intBitsToFloat(is.readFixed32());
254 			case DOUBLE: return Double.longBitsToDouble(is.readFixed64());
255 			case STRING: return decodeString(cm, is.readString());
256 			case BYTES: return is.readLenDelimited();
257 			case ENUM_INT: return decodeEnumOrdinal(cm, (int)is.readVarint());
258 			case ENUM_STRING: return is.readString();
259 			default: return is.readString();  // HTT: AUTO is resolved to a concrete type before reaching here.
260 		}
261 	}

263 	private static Object decodeUint64(ClassMeta<?> cm, ProtobufReader is) throws IOException {
264 		var raw = is.readVarint();
265 		if (ProtobufClassMeta.isBigInteger(cm))
266 			return new BigInteger(Long.toUnsignedString(raw));
267 		return raw;
268 	}
269 
270 	private static Object decodeEnumOrdinal(ClassMeta<?> cm, int ordinal) {
271 		var constants = cm.inner().getEnumConstants();
272 		if (constants != null && ordinal >= 0 && ordinal < constants.length)
273 			return constants[ordinal];
274 		return ordinal;
275 	}

277 	private Object decodeString(ClassMeta<?> cm, String s) throws ParseException {
278 		if (cm.isDate())
279 			return readDate(s, cm);
280 		if (cm.isCalendar())
281 			return readCalendar(s, cm);
282 		if (cm.isTemporal())
283 			return readTemporal(s, cm);
284 		if (cm.isDuration())
285 			return readDuration(s);
286 		if (cm.isPeriod())
287 			return readPeriod(s);
288 		return s;
289 	}

291 	private void readMapEntry(ClassMeta<?> mapType, ProtobufReader is, Map map) throws IOException, ParseException, ExecutableException {
292 		var keyType = mapType.getKeyType();
293 		var valueType = mapType.getValueType();
294 		Object key = null;
295 		Object value = null;
296 		long tag;
297 		while ((tag = is.readTag()) != ProtobufReader.EOF) {
298 			var fn = ProtobufReader.fieldNumber(tag);
299 			var wt = ProtobufReader.wireType(tag);
300 			if (fn == 1)
301 				key = decodeMapComponent(keyType, is);
302 			else if (fn == 2)
303 				value = decodeMapComponent(valueType, is);
304 			else
305 				is.skipField(wt);
306 		}
307 		map.put(convertToType(key, keyType), convertToType(value, valueType));
308 	}
309 
310 	private Object decodeMapComponent(ClassMeta<?> cm, ProtobufReader is) throws IOException, ParseException, ExecutableException {
311 		var swap = cm == null ? null : cm.getSwap(this); // HTT: cm is never null here -- ClassMeta.getKeyType()/getValueType() always default to the Object.class ClassMeta for a raw Map, never null, for any type where isMap() is true (which readMapEntry's callers guarantee).
312 		if (swap != null)
313 			return decodeSwapped(swap, cm, is, null);
314 		if (cm.isBean() || cm.isMap())
315 			return readMessage(cm, is.readLenDelimitedReader(), null);
316 		return decodeScalar(ProtobufClassMeta.defaultScalarType(cm), cm, is);
317 	}
