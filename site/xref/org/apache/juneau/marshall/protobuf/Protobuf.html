17  package org.apache.juneau.marshall.protobuf;
18  
19  import static java.lang.annotation.ElementType.*;
20  import static java.lang.annotation.RetentionPolicy.*;
21  
22  import java.lang.annotation.*;
23  
24  /**
25   * Annotation that can be applied to bean properties to control how they map onto the protobuf binary wire format
26   * via {@link ProtobufSerializer} and {@link ProtobufParser}.
27   *
28   * <p>
29   * This is the binary-format counterpart to the text-format {@link org.apache.juneau.marshall.prototext.Prototext @Prototext}
30   * annotation, and is kept deliberately separate from it.
31   *
32   * <p>
33   * Can be used in the following locations:
34   * <ul>
35   * 	<li>Marshalled classes/methods/fields.
36   * </ul>
37   *
38   * <h5 class='section'>See Also:</h5><ul>
39   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/Protobuf">Protobuf Binary Format Basics</a>
40   * </ul>
41   */
42  @Documented
43  @Target({ TYPE, FIELD, METHOD })
44  @Retention(RUNTIME)
45  @Inherited
46  public @interface Protobuf {
47  
48  	/**
49  	 * The explicit protobuf field number for this property.
50  	 *
51  	 * <p>
52  	 * When set to a value greater than zero, this field number is used on the wire instead of the auto-assigned
53  	 * value.  Explicitly-assigned numbers are reserved first, then remaining properties are auto-numbered into the
54  	 * gaps in alphabetical order.  Setting explicit field numbers is what enables true external <c>protoc</c> interop.
55  	 *
56  	 * @return The annotation value.  Defaults to <c>0</c> (auto-assign).
57  	 */
58  	int fieldNumber() default 0;
59  
60  	/**
61  	 * The explicit protobuf scalar type for this property.
62  	 *
63  	 * <p>
64  	 * When set to a value other than {@link ProtobufScalarType#AUTO}, this overrides the default scalar mapping
65  	 * derived from the Java type (e.g. to select <c>sint32</c> zigzag encoding or a <c>fixed64</c> form).  For
66  	 * repeated properties, this applies to the elements.
67  	 *
68  	 * @return The annotation value.  Defaults to {@link ProtobufScalarType#AUTO}.
69  	 */
70  	ProtobufScalarType type() default ProtobufScalarType.AUTO;
71  
72  	/**
73  	 * Optional description for the exposed API.
74  	 *
75  	 * @return The annotation value.
76  	 */
77  	String[] description() default {};
78  }
