17  package org.apache.juneau.marshall.protobuf;
18  
19  import static org.apache.juneau.commons.utils.IoUtils.*;
20  
21  import java.io.*;
22  
23  import org.apache.juneau.marshall.serializer.*;
24  
25  /**
26   * Low-level byte emitter for the protobuf binary wire format.
27   *
28   * <p>
29   * Wraps an {@link OutputStream} and provides the primitive encodings that make up a protobuf message:
30   * varints, zigzag varints, little-endian fixed32/fixed64, field tags, and length-delimited blocks.
31   *
32   * <p>
33   * Length-delimited sub-messages and packed repeated fields require the length to precede the payload, so
34   * callers encode the payload into a scratch buffer (e.g. a {@link ByteArrayOutputStream} wrapped in its own
35   * {@code ProtobufWriter}) and then emit it via {@link #writeLenDelimited(byte[])}.
36   *
37   * <h5 class='section'>Notes:</h5><ul>
38   * 	<li class='note'>This class is not intended for external use.
39   * 	<li class='note'>This class is not thread-safe; each serialization session uses its own instance.
40   * </ul>
41   *
42   * <h5 class='section'>See Also:</h5><ul>
43   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/Protobuf">Protobuf Binary Format Basics</a>
44   * 	<li class='link'><a class="doclink" href="https://protobuf.dev/programming-guides/encoding/">Protocol Buffers Encoding</a>
45   * </ul>
46   */
47  @SuppressWarnings({
48  	"resource" // The wrapped OutputStream is owned by the caller/session; the fluent writeXxx() methods return 'this' (a Closeable) by design, so Eclipse JDT's resource-leak/@Owning analysis flags them, but there is no resource for this writer to close.
49  })
50  public class ProtobufWriter extends OutputStream {
51  
52  	private final OutputStream os;
53  
54  	/**
55  	 * Constructor.
56  	 *
57  	 * @param os The output stream being wrapped.  Must not be <jk>null</jk>.
58  	 */
59  	public ProtobufWriter(OutputStream os) {
60  		this.os = os;
61  	}
62  
63  	@Override /* Overridden from OutputStream */
64  	public void write(int b) {
65  		try {
66  			os.write(b);
67  		} catch (IOException e) {
68  			throw new SerializeException(e);
69  		}
70  	}
71  
72  	@Override /* Overridden from OutputStream */
73  	public void write(byte[] b, int off, int len) {
74  		try {
75  			os.write(b, off, len);
76  		} catch (IOException e) {
77  			throw new SerializeException(e);
78  		}
79  	}
80  
81  	@Override /* Overridden from OutputStream */
82  	public void flush() {
83  		try {
84  			os.flush();
85  		} catch (IOException e) {
86  			throw new SerializeException(e);
87  		}
88  	}
89  
90  	/**
91  	 * Writes a base-128 varint (LEB128, unsigned interpretation of the bits).
92  	 *
93  	 * <p>
94  	 * The value's bits are treated as unsigned, so a negative {@code long} (e.g. a sign-extended int32
95  	 * {@code -1}) produces the canonical 10-byte form.
96  	 *
97  	 * @param value The value whose 64 bits are encoded.
98  	 * @return This object.
99  	 */
100 	public ProtobufWriter writeVarint(long value) {
101 		while ((value & ~0x7FL) != 0) {
102 			write((int)((value & 0x7F) | 0x80));
103 			value >>>= 7;
104 		}
105 		write((int)(value & 0x7F));
106 		return this;
107 	}
108 
109 	/**
110 	 * Writes a 32-bit signed integer using zigzag encoding (maps small-magnitude negatives to small varints).
111 	 *
112 	 * @param value The value to encode.
113 	 * @return This object.
114 	 */
115 	public ProtobufWriter writeZigZag32(int value) {
116 		return writeVarint(((value << 1) ^ (value >> 31)) & 0xFFFFFFFFL);
117 	}
118 
119 	/**
120 	 * Writes a 64-bit signed integer using zigzag encoding.
121 	 *
122 	 * @param value The value to encode.
123 	 * @return This object.
124 	 */
125 	public ProtobufWriter writeZigZag64(long value) {
126 		return writeVarint((value << 1) ^ (value >> 63));
127 	}
128 
129 	/**
130 	 * Writes a 32-bit value as 4 little-endian bytes (fixed32/sfixed32/float).
131 	 *
132 	 * @param value The 32 bits to write.
133 	 * @return This object.
134 	 */
135 	public ProtobufWriter writeFixed32(int value) {
136 		write(value & 0xFF);
137 		write((value >> 8) & 0xFF);
138 		write((value >> 16) & 0xFF);
139 		write((value >> 24) & 0xFF);
140 		return this;
141 	}
142 
143 	/**
144 	 * Writes a 64-bit value as 8 little-endian bytes (fixed64/sfixed64/double).
145 	 *
146 	 * @param value The 64 bits to write.
147 	 * @return This object.
148 	 */
149 	public ProtobufWriter writeFixed64(long value) {
150 		write((int)(value & 0xFF));
151 		write((int)((value >> 8) & 0xFF));
152 		write((int)((value >> 16) & 0xFF));
153 		write((int)((value >> 24) & 0xFF));
154 		write((int)((value >> 32) & 0xFF));
155 		write((int)((value >> 40) & 0xFF));
156 		write((int)((value >> 48) & 0xFF));
157 		write((int)((value >> 56) & 0xFF));
158 		return this;
159 	}
160 
161 	/**
162 	 * Writes a field tag (<c>(fieldNumber &lt;&lt; 3) | wireType</c>) as a varint.
163 	 *
164 	 * @param fieldNumber The protobuf field number.
165 	 * @param wireType The wire type of the value that follows.
166 	 * @return This object.
167 	 */
168 	public ProtobufWriter writeTag(int fieldNumber, WireType wireType) {
169 		return writeVarint(((long)fieldNumber << 3) | wireType.code());
170 	}
171 
172 	/**
173 	 * Writes a length-delimited block:  a varint length followed by the raw bytes.
174 	 *
175 	 * @param value The bytes to write.
176 	 * @return This object.
177 	 */
178 	public ProtobufWriter writeLenDelimited(byte[] value) {
179 		writeVarint(value.length);
180 		write(value, 0, value.length);
181 		return this;
182 	}
183 
184 	/**
185 	 * Writes a length-delimited UTF-8 string.
186 	 *
187 	 * @param value The string to write.
188 	 * @return This object.
189 	 */
190 	public ProtobufWriter writeString(String value) {
191 		return writeLenDelimited(value.getBytes(UTF8));
192 	}
193 }
