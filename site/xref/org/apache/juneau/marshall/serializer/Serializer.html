38  /**
39   * Parent class for all Juneau serializers.
40   *
41   * <h5 class='topic'>Description</h5>
42   * <p>
43   * Base serializer class that serves as the parent class for all serializers.
44   *
45   * <p>
46   * The purpose of this class is:
47   * <ul>
48   * 	<li>Maintain a read-only configuration state of a serializer.
49   * 	<li>Create session objects used for serializing POJOs (i.e. {@link SerializerSession}).
50   * 	<li>Provide convenience methods for serializing POJOs without having to construct session objects.
51   * </ul>
52   *
53   * <p>
54   * Subclasses should (but are not required to) extend directly from {@link OutputStreamSerializer} or {@link WriterSerializer} depending on
55   * whether it's a stream or character based serializer.
56   *
57   * <p>
58   * Subclasses must implement serialization via one of the following methods:
59   * <ul class='javatree'>
60   * 	<li class='jmp'>{@link #doWrite(SerializerSession, SerializerPipe, Object)}
61   * 	<li class='jmp'>{@link SerializerSession#doWrite(SerializerPipe, Object)}
62   * </ul>
63   * <br>
64   *
65   * <h5 class='section'>Notes:</h5><ul>
66   * 	<li class='note'>This class is thread safe and reusable.
67   * </ul>
68   *
69   * <h5 class='section'>See Also:</h5><ul>
70   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/SerializersAndParsers">Serializers and Parsers</a>
71  
72   * </ul>
73   */

221 		 * Add <js>"_type"</js> properties when needed.
222 		 *
223 		 * <p>
224 		 * When enabled, <js>"_type"</js> properties will be added to beans if their type cannot be inferred
225 		 * through reflection.
226 		 *
227 		 * <p>
228 		 * This is used to recreate the correct objects during parsing if the object types cannot be inferred.
229 		 * <br>For example, when serializing a <c>Map&lt;String,Object&gt;</c> field where the bean class cannot be determined from
230 		 * the type of the values.
231 		 *
232 		 * <p>
233 		 * Note the differences between the following settings:
234 		 * <ul class='javatree'>
235 		 * 	<li class='jf'>{@link #addRootType()} - Affects whether <js>'_type'</js> is added to root node.
236 		 * 	<li class='jf'>{@link #addBeanTypes()} - Affects whether <js>'_type'</js> is added to any nodes.
237 		 * </ul>
238 		 *
239 		 * <h5 class='section'>Example:</h5>
240 		 * <p class='bjava'>
241 		 * 	<jc>// Create a serializer that adds _type to nodes.</jc>
242 		 * 	WriterSerializer <jv>serializer</jv> = JsonSerializer
243 		 * 		.<jsm>create</jsm>()
244 		 * 		.addBeanTypes()
245 		 * 		.build();
246 		 *
247 		 * 	<jc>// Our map of beans to serialize.</jc>
248 		 * 	<ja>@Marshalled</ja>(typeName=<js>"mybean"</js>)
249 		 * 	<jk>public class</jk> MyBean {
250 		 * 		<jk>public</jk> String <jf>foo</jf> = <js>"bar"</js>;
251 		 * 	}
252 		 * 	JsonMap <jv>myMap</jv> = JsonMap.of(<js>"foo"</js>, <jk>new</jk> MyBean());
253 		 *
254 		 * 	<jc>// Will contain:  {"foo":{"_type":"mybean","foo":"bar"}}</jc>
255 		 * 	String <jv>json</jv> = <jv>serializer</jv>.write(<jv>myMap</jv>);
256 		 * </p>
257 		 *
258 		 * <p>
259 		 * <b>Token streams:</b> the cursor's raw structural methods do not emit <c>"_type"</c>;
260 		 * however this setting IS honored by
261 		 * {@link TokenWriter#object(Object) TokenWriter.object(Object)}

439 		/**
440 		 * Don't serialize bean properties whose values equal the default for their type.
441 		 *
442 		 * <p>
443 		 * When enabled, bean properties whose current value equals the type's natural default are omitted from
444 		 * the serialized output. This is the Juneau analog of Jackson's <c>JsonInclude.Include.NON_DEFAULT</c>
445 		 * inclusion mode.
446 		 *
447 		 * <p>
448 		 * The notion of "default" depends on the property type:
449 		 * <ul class='spaced-list'>
450 		 * 	<li>Primitives and primitive wrappers (<code><jk>int</jk></code>, <code>Integer</code>,
451 		 * 		<code><jk>boolean</jk></code>, <code>Boolean</code>, etc.) use the Java type default
452 		 * 		(<c>0</c>, <c>false</c>, <code>'\0'</code>, <jk>null</jk>).
453 		 * 	<li>Bean properties on bean-typed property values use a <i>reference instance</i> of the bean type —
454 		 * 		instantiated once via the no-arg constructor and cached on the {@link ClassMeta} — and the
455 		 * 		property is omitted when its current value equals that reference's value.
456 		 * 	<li>When no reference instance can be built (no accessible no-arg constructor, or instantiation
457 		 * 		throws), <c>nonDefault</c> is silently skipped for that bean — the property is emitted normally
458 		 * 		rather than failing the serialize.
459 		 * </ul>
460 		 *
461 		 * <p>
462 		 * <b>Equality semantics:</b> value-equals-default uses {@link Objects#equals}, with numeric values
463 		 * compared by mathematical value (BigDecimal.compareTo-style) so <c>1.0</c> equals <c>1</c>,
464 		 * <c>0.0</c> equals <c>-0.0</c>, and <code>BigDecimal(<js>"1.0"</js>)</code> equals
465 		 * <code>BigDecimal(<js>"1"</js>)</code>.
466 		 *
467 		 * <p>
468 		 * <b>Precedence:</b> the serialize-side inclusion knobs short-circuit in a fixed order — first "omit" wins:
469 		 * <ol>
470 		 * 	<li><jk>null</jk> values (including {@link Optional#empty()}) are governed by
471 		 * 		{@link #keepNullProperties()}.
472 		 * 	<li>Non-<jk>null</jk> string values are normalized by {@link #trimStrings()}.
473 		 * 	<li>Empty collections / arrays / maps are omitted by {@link #trimEmptyCollections()} /
474 		 * 		{@link #trimEmptyMaps()}.
475 		 * 	<li>Surviving values that equal their type's default are omitted by <c>nonDefault</c>.
476 		 * </ol>
477 		 * <jk>null</jk> is resolved by <c>keepNullProperties</c>, not treated as a <c>nonDefault</c> case.
478 		 *
479 		 * <h5 class='section'>Example:</h5>
480 		 * <p class='bjava'>
481 		 * 	<jc>// Create a serializer that omits default-valued properties.</jc>
482 		 * 	WriterSerializer <jv>serializer</jv> = JsonSerializer
483 		 * 		.<jsm>create</jsm>()
484 		 * 		.nonDefault()
485 		 * 		.build();
486 		 *
487 		 * 	<jc>// Our bean to serialize.</jc>
488 		 * 	<jk>public class</jk> MyBean {
489 		 * 		<jk>public int</jk> <jf>count</jf>;            <jc>// Default 0; omitted unless set to non-zero.</jc>
490 		 * 		<jk>public</jk> String <jf>name</jf> = <js>"x"</js>; <jc>// Bean-default is "x"; only omitted when value equals "x".</jc>
491 		 * 	}
492 		 *
493 		 * 	<jc>// Produces "{}" because both fields equal their defaults.</jc>
494 		 * 	String <jv>json</jv> = <jv>serializer</jv>.write(<jk>new</jk> MyBean());
495 		 * </p>
496 		 *
497 		 * <h5 class='section'>See Also:</h5><ul>
498 		 * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/InclusionAndNullHandling">Inclusion and null handling</a>
499 		 * </ul>
500 		 *
501 		 * @return This object.
502 		 * @since 10.0.0
503 		 */

506 		}

522 		 *
523 		 * <p>
524 		 * Class used to listen for errors and warnings that occur during serialization.
525 		 *
526 		 * <h5 class='section'>Example:</h5>
527 		 * <p class='bjava'>
528 		 * 	<jc>// Define our serializer listener.</jc>
529 		 * 	<jc>// Simply captures all errors.</jc>
530 		 * 	<jk>public class</jk> MySerializerListener <jk>extends</jk> SerializerListener {
531 		 *
532 		 * 		<jc>// A simple property to store our events.</jc>
533 		 * 		<jk>public</jk> List&lt;String&gt; <jf>events</jf> = <jk>new</jk> LinkedList&lt;&gt;();
534 		 *
535 		 * 		<ja>@Override</ja>
536 		 * 		<jk>public</jk> &lt;T&gt; <jk>void</jk> onError(SerializerSession <jv>session</jv>, Throwable <jv>throwable</jv>, String <jv>msg</jv>) {
537 		 * 			<jf>events</jf>.add(<jv>session</jv>.getLastLocation() + <js>","</js> + <jv>msg</jv> + <js>","</js> + <jv>throwable</jv>);
538 		 * 		}
539 		 * 	}
540 		 *
541 		 * 	<jc>// Create a serializer using our listener.</jc>
