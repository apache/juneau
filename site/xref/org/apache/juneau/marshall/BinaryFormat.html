19  import static org.apache.juneau.commons.utils.Shorts.*;
20  import static org.apache.juneau.commons.utils.StringUtils.*;

22  import java.util.*;

24  /**
25   * Supported wire formats for <code><jk>byte</jk>[]</code> values and stream-based serialization output.
26   *
27   * <p>
28   * Used by {@link MarshallingContext.Builder#binaryFormat(BinaryFormat)},
29   * {@link Marshalled#binaryFormat()},
30   * {@link MarshalledProp#binaryFormat()}, and
31   * {@link MarshalledConfig#binaryFormat()} to control how
32   * <code><jk>byte</jk>[]</code> values are written to and read from the wire on text formats, as well as
33   * the encoding used by stream-based serializers when converting their byte output to a string.
34   *
35   * <p>
36   * The default is {@link #NOT_SET} — leaves {@code byte[]} handling to the surrounding serializer / parser
37   * (e.g. textual serializers fall back to the natural language-level array representation, and binary
38   * serializers emit native bytes).  Callers wanting a textual encoding on the wire should set this to
39   * {@link #HEX}, {@link #BASE64}, {@link #SPACED_HEX}, or {@link #BASE64_URL}.
40   *
41   * <p>
42   * Binary serializers (BSON / CBOR / MsgPack / Prototext) emit native bytes regardless of the configured
43   * {@code BinaryFormat}; this setting only affects text-based serializers and the
44   * {@code OutputStreamSerializerSession#writeToString(Object)} stream-to-string convenience.
45   */
46  public enum BinaryFormat {
47  
48  	/** Sentinel meaning "no value configured" — falls through to the next-higher precedence level. */
49  	NOT_SET,
50  
51  	/**
52  	 * Spaced-hex.
53  	 *
54  	 * <h5 class='figure'>Example:</h5>
55  	 * <p class='bcode'>
56  	 * 	12 34 56 78 90 AB CD EF
57  	 * </p>
58  	 */
59  	SPACED_HEX,
60  
61  	/**
62  	 * Hex.
63  	 *
64  	 * <h5 class='figure'>Example:</h5>
65  	 * <p class='bcode'>
66  	 * 	1234567890ABCDEF
67  	 * </p>
68  	 */
69  	HEX,
70  
71  	/** Standard {@link Base64} encoding. */
72  	BASE64,
73  
74  	/**
75  	 * URL-safe {@link Base64} encoding (RFC 4648 §5).
76  	 *
77  	 * <p>
78  	 * Uses {@code -} and {@code _} in place of {@code +} and {@code /}. Padding ({@code =}) is omitted on
79  	 * the wire but accepted on input.
80  	 */
81  	BASE64_URL;
82  
83  	private static final byte[] EMPTY_BYTES = new byte[0];
84  
85  	/**
86  	 * Formats the specified value using this format.
87  	 *
88  	 * @param value The value to format.  Can be <jk>null</jk> (returns <jk>null</jk>).
89  	 * @return The formatted string, or <jk>null</jk> if {@code value} is <jk>null</jk>.
90  	 */
91  	public String format(byte[] value) {
92  		if (value == null)
93  			return null;
94  		return switch (this) {
95  			case NOT_SET, BASE64 -> base64Encode(value);
96  			case BASE64_URL -> Base64.getUrlEncoder().withoutPadding().encodeToString(value);
97  			case HEX -> toHex(value);
98  			case SPACED_HEX -> toSpacedHex(value);
99  		};
100 	}
101 
102 	/**
103 	 * Parses the specified wire value using this format.
104 	 *
105 	 * <p>
106 	 * The parser accepts any of the formats produced by {@link #format(byte[])} regardless of the
107 	 * configured constant, mirroring the format-agnostic parser convention used by other {@code <type>Format}
108 	 * enums in this package.  The configured format is consulted only as a hint for ambiguous input.
109 	 *
110 	 * @param value The wire value.  Can be <jk>null</jk> or blank (returns an empty array).
111 	 * @return The parsed bytes, or an empty array if {@code value} is <jk>null</jk> or blank.
112 	 */
113 	public byte[] parse(String value) {
114 		if (value == null)
115 			return EMPTY_BYTES;
116 		var s = value.trim();
117 		if (s.isEmpty())
118 			return EMPTY_BYTES;
119 		// Honor the configured format constant for BASE64_URL: the URL-safe encoder emits without padding,
120 		// so a non-3-aligned payload that happens not to contain `-`/`_` (because the encoded chars all fall
121 		// in the base64-alphabet intersection) would otherwise be misrouted to the standard decoder, which
122 		// rejects missing padding.  The URL-safe decoder accepts both with-padding and without-padding input.
123 		if (this == BASE64_URL) {
124 			try {
125 				return Base64.getUrlDecoder().decode(s);
126 			} catch (IllegalArgumentException e) {
127 				throw iaex("Invalid binary value '%s' for format %s: %s", s, this, e.getMessage());
128 			}
129 		}
130 		// Format-agnostic parsing: sniff the wire shape and decode accordingly, regardless of the
131 		// configured format on `this`.  A space anywhere implies spaced-hex; a `-`/`_` implies url-safe
132 		// base64; otherwise pure hex shape (even-length, hex-only chars) implies hex.  Anything left
133 		// is standard base64.
134 		if (s.indexOf(' ') >= 0)
135 			return fromSpacedHex(s);
136 		if (s.indexOf('-') >= 0 || s.indexOf('_') >= 0)
137 			return Base64.getUrlDecoder().decode(s);
138 		if (isHex(s))
139 			return fromHex(s);
140 		try {
141 			return base64Decode(s);
142 		} catch (IllegalArgumentException e) {
143 			throw iaex("Invalid binary value '%s' for format %s: %s", s, this, e.getMessage());
144 		}
145 	}
146 
147 	private static boolean isHex(String s) {
148 		if (s.isEmpty() || (s.length() & 1) == 1)
149 			return false;
150 		for (var i = 0; i < s.length(); i++) {
151 			var c = s.charAt(i);
152 			if (! ((c >= '0' && c <= '9') || (c >= 'a' && c <= 'f') || (c >= 'A' && c <= 'F')))
153 				return false;
154 		}
155 		return true;
156 	}
157 }
