17  package org.apache.juneau.marshall.parquet;
18  
19  import java.io.*;
20  import java.nio.charset.*;
21  
22  /**
23   * Writes column values using PLAIN encoding for Parquet data pages.
24   *
25   * <p>
26   * Values are written in little-endian format per the
27   * <a class="doclink" href="https://parquet.apache.org/docs/file-format/data-pages/encodings/">Parquet encodings</a>
28   * specification. Tracks value count and byte sizes for page header creation.
29   */
30  final class ParquetColumnWriter {
31  
32  	private static final int PAGE_TYPE_DATA = 0;
33  	private static final int ENCODING_PLAIN = 0;
34  	private static final int ENCODING_RLE = 3;
35  
36  	private final ParquetSchemaElement schema;
37  	private final ByteArrayOutputStream out = new ByteArrayOutputStream();
38  	private int valueCount;
39  	private int boolBits;
40  	private int boolBitCount;
41  
42  	ParquetColumnWriter(ParquetSchemaElement schema) {
43  		this.schema = schema;
44  		this.valueCount = 0;
45  	}
46  
47  	/**
48  	 * Writes a boolean using PLAIN encoding: 1 bit per value, bit-packed LSB-first into successive bytes
49  	 * (GAP-4).  Only present (non-null) booleans are written; the partial trailing byte is flushed by
50  	 * {@link #finalizePage()}.
51  	 */
52  	void writeBoolean(boolean value) {
53  		if (value)
54  			boolBits |= 1 << boolBitCount;
55  		boolBitCount++;
56  		if (boolBitCount == 8) {
57  			out.write(boolBits & 0xFF);
58  			boolBits = 0;
59  			boolBitCount = 0;
60  		}
61  		valueCount++;
62  	}
63  
64  	void writeInt32(int value) {
65  		out.write(value & 0xFF);
66  		out.write((value >> 8) & 0xFF);
67  		out.write((value >> 16) & 0xFF);
68  		out.write((value >> 24) & 0xFF);
69  		valueCount++;
70  	}
71  
72  	void writeInt64(long value) {
73  		out.write((int)(value & 0xFF));
74  		out.write((int)((value >> 8) & 0xFF));
75  		out.write((int)((value >> 16) & 0xFF));
76  		out.write((int)((value >> 24) & 0xFF));
77  		out.write((int)((value >> 32) & 0xFF));
78  		out.write((int)((value >> 40) & 0xFF));
79  		out.write((int)((value >> 48) & 0xFF));
80  		out.write((int)((value >> 56) & 0xFF));
81  		valueCount++;
82  	}
83  
84  	void writeFloat(float value) {
85  		writeInt32(Float.floatToIntBits(value));
86  	}
87  
88  	void writeDouble(double value) {
89  		writeInt64(Double.doubleToLongBits(value));
90  	}
91  
92  	void writeByteArray(byte[] value) throws IOException {
93  		if (value == null)
94  			value = new byte[0];
95  		writeInt32(value.length);
96  		if (value.length > 0)
97  			out.write(value);
98  		valueCount++;
99  	}
100 
101 	void writeByteArray(String value) throws IOException {
102 		byte[] bytes = value == null ? new byte[0] : value.getBytes(StandardCharsets.UTF_8);
103 		writeByteArray(bytes);
104 	}
105 
106 	void writeFixedLenByteArray(byte[] value) {
107 		if (value == null || schema.typeLength == null)
108 			return;
109 		int len = Math.min(value.length, schema.typeLength);
110 		for (int i = 0; i < schema.typeLength; i++)
111 			out.write(i < len ? value[i] : 0);
112 		valueCount++;
113 	}
114 
115 	void writeNull() {
116 		valueCount++;
117 	}
118 
119 	int getValueCount() {
120 		return valueCount;
121 	}
122 
123 	/**
124 	 * Finalizes the current page and returns the raw (uncompressed) value bytes.
125 	 * Resets the writer for the next page.
126 	 */
127 	byte[] finalizePage() {
128 		if (boolBitCount > 0) {
129 			out.write(boolBits & 0xFF);
130 			boolBits = 0;
131 			boolBitCount = 0;
132 		}
133 		var result = out.toByteArray();
134 		out.reset();
135 		valueCount = 0;
136 		return result;
137 	}
138 
139 	/**
140 	 * Creates a PageHeader (Thrift) for a data page with the given metrics.
141 	 */
142 	static byte[] createPageHeader(int numValues, int uncompressedSize, int compressedSize) throws IOException {
143 		var baos = new ByteArrayOutputStream();
144 		var enc = new ThriftCompactEncoder(baos);
145 		enc.writeStructBegin();
146 		enc.writeFieldBegin(ThriftCompactEncoder.I32, 1);
147 		enc.writeI32(PAGE_TYPE_DATA);
148 		enc.writeFieldBegin(ThriftCompactEncoder.I32, 2);
149 		enc.writeI32(uncompressedSize);
150 		enc.writeFieldBegin(ThriftCompactEncoder.I32, 3);
151 		enc.writeI32(compressedSize);
152 		enc.writeFieldBegin(ThriftCompactEncoder.STRUCT, 5);
153 		enc.writeStructBegin();
154 		enc.writeFieldBegin(ThriftCompactEncoder.I32, 1);
155 		enc.writeI32(numValues);
156 		enc.writeFieldBegin(ThriftCompactEncoder.I32, 2);
157 		enc.writeI32(ENCODING_PLAIN);
158 		enc.writeFieldBegin(ThriftCompactEncoder.I32, 3);
159 		enc.writeI32(ENCODING_RLE);
160 		enc.writeFieldBegin(ThriftCompactEncoder.I32, 4);
161 		enc.writeI32(ENCODING_RLE);
162 		enc.writeStructEnd();
163 		enc.writeStructEnd();
164 		return baos.toByteArray();
165 	}
166 }
