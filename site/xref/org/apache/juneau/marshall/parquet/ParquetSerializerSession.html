138 	@Override
139 	protected void doWrite(SerializerPipe pipe, Object o) throws IOException, SerializeException {
140 		var out = pipe.getOutputStream();
141 		var rows = collectBeans(o);
142 		if (rows.isEmpty()) {
143 			writeEmptyFile(out);
144 			return;
145 		}
146 		var first = rows.get(0);
147 		List<ParquetSchemaElement> schema;
148 		// collectBeans() always returns rows whose first element is a BeanMap (bean / collection / array /
149 		// single-value roots, all BeanMap-wrapped) or a Map (string-keyed and non-string-keyed map roots),
150 		// so those are the only two cases.
151 		if (first instanceof BeanMap<?> first2) {
152 			var elementType = getClassMetaForObject(first2.getBean());
153 			// For collection root: use element type for flat row schema (root.name, root.age).
154 			// Each collection element becomes one Parquet row. Avoids list column format issues.
155 			schema = new ParquetSchemaBuilder(ctx.getMarshallingContext(), ctx.writeDatesAsTimestamp, ctx.cycleHandling, ctx.maxRecursionDepth, ctx.nativeLogicalTypes).buildSchema(elementType, first);
156 		} else {
157 			var mapFirst = (Map<?,?>) first;
158 			if (!(((Map<?,?>) o).keySet().iterator().next() instanceof String)) {
159 				schema = new ParquetSchemaBuilder(ctx.getMarshallingContext(), ctx.writeDatesAsTimestamp, ctx.cycleHandling, ctx.maxRecursionDepth, ctx.nativeLogicalTypes)
160 					.buildSchemaForKeyValuePairs(mapFirst.get("key"), mapFirst.get("value"));
161 			} else {
162 				schema = new ParquetSchemaBuilder(ctx.getMarshallingContext(), ctx.writeDatesAsTimestamp, ctx.cycleHandling, ctx.maxRecursionDepth, ctx.nativeLogicalTypes).buildSchemaFromMap(mapFirst);
163 			}
164 		}
165 		var leafColumns = ParquetSchemaBuilder.getLeafColumns(schema);
166 		writeMagic(out);
167 		long dataStart = 4;
168 		var rowGroups = new ArrayList<RowGroupMeta>();
169 		long fileOffset = dataStart;
170 		// Multi-row-group write (GAP-2 round-trip): split rows into batches of rowsPerRowGroup and emit one
171 		// row group per batch.  Defaults keep the historical single-row-group output (the cap is huge), but a
172 		// small rowGroupSize produces multiple groups for interop/round-trip coverage.
173 		int rowsPerRowGroup = rowsPerRowGroup(leafColumns.size());
174 		for (int start = 0; start < rows.size(); start += rowsPerRowGroup) {
175 			var batch = rows.subList(start, Math.min(start + rowsPerRowGroup, rows.size()));
176 			long groupStart = fileOffset;
177 			var columnChunks = new ArrayList<ColumnChunkMeta>();
178 			for (var col : leafColumns) {
179 				var chunk = writeColumnChunk(out, batch, col, fileOffset);
180 				columnChunks.add(chunk);
181 				fileOffset += chunk.totalCompressedSize;
182 			}
183 			long totalByteSize = 0;
184 			long totalCompressedSize = 0;
185 			for (var c : columnChunks) {
186 				totalByteSize += c.totalUncompressedSize;
187 				totalCompressedSize += c.totalCompressedSize;
188 			}
189 			rowGroups.add(new RowGroupMeta(columnChunks, totalByteSize, batch.size(), groupStart, totalCompressedSize));
190 		}
191 		var footerBytes = writeFileMetaData(schema, rows.size(), rowGroups);
192 		out.write(footerBytes);
193 		writeLe4(out, footerBytes.length);
194 		out.write(MAGIC);
195 	}

197 	/**
198 	 * Rows per row group, derived from the {@code rowGroupSize} byte budget via a coarse per-row estimate.
199 	 *
200 	 * <p>
201 	 * Real writers flush a group once its accumulated byte size crosses the budget; Juneau's column-oriented
202 	 * writer materializes whole columns at once, so we instead translate the byte budget into a row count
203 	 * using a nominal {@code BYTES_PER_ROW_ESTIMATE} per leaf column.  With the default 128&nbsp;MB budget this
204 	 * yields an effectively unbounded count (single group, unchanged output); a small {@code rowGroupSize}
205 	 * (e.g. set in a test) produces several groups.
206 	 */
207 	private int rowsPerRowGroup(int leafColumnCount) {
208 		long perRow = (long) Math.max(1, leafColumnCount) * BYTES_PER_ROW_ESTIMATE;
209 		long n = Math.max(1, ctx.rowGroupSize / perRow);
210 		return (int) Math.min(n, Integer.MAX_VALUE);
211 	}

229 	private static void writeLe4(OutputStream out, int v) throws IOException {
230 		out.write(v & 0xFF);
231 		out.write((v >> 8) & 0xFF);
232 		out.write((v >> 16) & 0xFF);
233 		out.write((v >> 24) & 0xFF);

236 	private List<?> collectBeans(Object o) throws SerializeException {
237 		if (o == null)
238 			return List.of();
239 		// Unwrap Optional at root so we serialize the inner value, not Optional as a bean
240 		if (o instanceof Optional<?> o2) {
241 			var inner = o2.orElse(null);
242 			if (inner == null)
243 				return List.of();
244 			return collectBeans(inner);
245 		}
246 		var sType = getExpectedRootType(o);
247 		if (sType.isMap()) {
248 			var m = (Map<?,?>) o;
249 			if (m.isEmpty())
250 				return List.of();
251 			var firstKey = m.keySet().iterator().next();
252 			if (firstKey instanceof String || firstKey == null) {
253 				// Apply POJO swaps to map values so types like A (with swap to JsonMap) use their swapped schema/encoding.
254 				// Replace null keys with the null-key sentinel so they survive as a named column in the schema.
255 				var swappedMap = m();
256 				for (var e : m.entrySet()) {
257 					var key = e.getKey() == null ? ctx.nullKeyString : trim(e.getKey());
258 					var val = e.getValue();
259 					if (val != null)
260 						val = applySwap(val, getClassMetaForObject(val));
261 					swappedMap.put(key, val);
262 				}
263 				return List.of(swappedMap);
264 			}
265 			var result = new ArrayList<Map<String,Object>>(m.size());
266 			for (var e : m.entrySet()) {
267 				var row = new LinkedHashMap<String,Object>();
268 				row.put("key", e.getKey());
269 				row.put("value", e.getValue());
270 				result.add(row);
271 			}
272 			return result;
273 		}
274 		if (sType.isCollection() || sType.isArray()) {
275 			Collection<?> list = sType.isArray() ? toList(sType.inner(), o) : (Collection<?>)o;
276 			if (cn(list).startsWith("java.util.ImmutableCollections"))
277 				list = new ArrayList<>(list);
278 			var result = new ArrayList<BeanMap<?>>(list.size());
279 			for (var e : list) {
280 				if (e == null) {
281 					result.add(toBeanMap(createWrapperBean(null)));
282 					continue;
283 				}
284 				var swapped = applySwap(e, getClassMetaForObject(e));
285 				var swappedType = getClassMetaForObject(swapped);
286 				var origType = getClassMetaForObject(e);
287 				Object toUse;
288 				if (swappedType.isBean())
289 					toUse = swapped;
290 				else if (origType.isBean())
291 					toUse = e;
292 				else
293 					toUse = swapped;
294 				var useType = getClassMetaForObject(toUse);
295 				if (useType.isBean())
296 					result.add(toBeanMap(toUse));
297 				else
298 					result.add(toBeanMap(createWrapperBean(toUse)));
299 			}
300 			return result;
301 		}
302 		if (sType.isStreamable()) {
303 			return collectBeans(toListFromStreamable(o, sType));
304 		}
305 		var swapped = applySwap(o, sType);
306 		var aType = getClassMetaForObject(swapped);
307 		if (aType.isBean())
308 			return List.of(toBeanMap(swapped));
309 		return List.of(toBeanMap(createWrapperBean(swapped)));
310 	}

334 	/**
335 	 * Applies any class-level swap registered via {@link org.apache.juneau.marshall.swap.DefaultSwaps}
336 	 * (or context-registered swaps) to a leaf value just before it is written to a Parquet column.
337 	 *
338 	 * <p>
339 	 * Necessary because bean-property values pass through {@link BeanMap#get(Object)}, which only
340 	 * applies the per-property {@code readTransform} installed by
341 	 * {@code MarshalledPropertyPostProcessor}.  At {@code NOT_SET} / unconfigured formats no
342 	 * per-property swap is installed for {@code Class<?>} (and other types whose default swap lives
343 	 * in {@code DefaultSwaps}); without this hook the raw value would fall through to
344 	 * {@link String#valueOf(Object)} and produce wires like {@code "class java.lang.String"}.
345 	 *
346 	 * <p>
347 	 * Mirrors the {@code aType.getSwap(session)} layer applied by every other serializer
348 	 * ({@link org.apache.juneau.marshall.json.JsonSerializerSession},
349 	 * {@link org.apache.juneau.marshall.msgpack.MsgPackSerializerSession}, etc.) in their {@code
350 	 * writeAnything} dispatcher.  No-op for values whose runtime type has no registered swap.
351 	 */
352 	private Object applyDefaultSwap(Object value) throws SerializeException {
353 		var cm = getClassMetaForObject(value);
354 		var swap = cm.getSwap(this);
355 		if (nn(swap))
356 			return swap(swap, value);
357 		return value;
358 	}

360 	private ColumnChunkMeta writeColumnChunk(OutputStream out, List<?> rows, ParquetSchemaElement col, long chunkStart) throws IOException, SerializeException {
361 		var path = rowRelativePath(col.path);
362 		if (path.contains(".list.element"))
363 			return writeListColumnChunk(out, rows, col, path, chunkStart);
364 		if (isMapKeyValueColumnPath(path))
365 			return writeMapColumnChunk(out, rows, col, path, chunkStart);
366 		// Definition-level handling for non-list/non-map columns.
367 		// A leaf nested under N OPTIONAL groups needs a max def level of N (GAP-14) so the reader can tell
368 		// a null intermediate group apart from a present group with a null leaf.  Single-segment columns
369 		// (maxDef==1) use the original 1-bit scheme unchanged.
370 		int maxDef = path.split("\\.").length;
371 		int numRows = rows.size();
372 
373 		// Multi-page write (GAP-1 round-trip): split the column's rows into page batches of rowsPerPage.
374 		// Defaults keep a single page (huge cap); a small pageSize produces several data pages in one chunk,
375 		// which the page-loop reader concatenates.
376 		int rowsPerPage = rowsPerPage();
377 		var pathParts = List.of(col.path.split("\\."));
378 		long totalUncompressed = 0;
379 		long totalStored = 0;
380 		int firstHeaderSize = 0;
381 		boolean firstPage = true;
382 		for (int start = 0; start < numRows; start += rowsPerPage) {
383 			var pageRows = rows.subList(start, Math.min(start + rowsPerPage, numRows));
384 			byte[] pageData = encodeColumnPage(pageRows, col, path, maxDef);
385 			int uncompressedSize = pageData.length;
386 			byte[] compressedData = ctx.compressionCodec.compress(pageData);
387 			int compressedSize = compressedData.length;
388 			var pageHeader = ParquetColumnWriter.createPageHeader(pageRows.size(), uncompressedSize, compressedSize);
389 			out.write(pageHeader);
390 			out.write(compressedData);
391 			totalUncompressed += uncompressedSize;
392 			totalStored += compressedSize + (long) pageHeader.length;
393 			if (firstPage) {
394 				firstHeaderSize = pageHeader.length;
395 				firstPage = false;
396 			}
397 		}
398 		return new ColumnChunkMeta(col.path, pathParts, col.type, numRows, totalUncompressed, totalStored, chunkStart, firstHeaderSize);
399 	}

401 	private int rowsPerPage() {
402 		long n = Math.max(1, ctx.pageSize / BYTES_PER_ROW_ESTIMATE);
403 		return (int) Math.min(n, Integer.MAX_VALUE);
404 	}

406 	/** Encodes one data page (def levels + PLAIN values) for a batch of rows of a non-list/non-map column. */
407 	private byte[] encodeColumnPage(List<?> rows, ParquetSchemaElement col, String path, int maxDef) throws IOException, SerializeException {
408 		var writer = new ParquetColumnWriter(col);
409 		byte[] defLevelBytes;
410 		if (maxDef >= 2) {
411 			var defLevels = new RleBitPackingEncoder(defRepBitWidth(maxDef));
412 			for (var row : rows) {
413 				var dv = computeDefValue(row, path, maxDef);
414 				defLevels.writeInt(dv.def());
415 				if (dv.def() == maxDef)
416 					writeValue(writer, col, dv.value());
417 			}
418 			defLevelBytes = defLevels.toByteArrayWithLength();
419 		} else {
420 			var values = extractColumnValues(rows, path);
421 			// Root key_value format uses root.key/root.value - convert keys to stored strings (ISO8601, etc.)
422 			boolean isRootKeyColumn = "root.key".equals(col.path);
423 			if (isRootKeyColumn) {
424 				var converted = new ArrayList<>(values.size());
425 				for (var val : values) {
426 					converted.add(val == null ? null : mapKeyToStoredString(unwrapOptional(val)));
427 				}
428 				values = converted;
429 			}
430 			var defLevels = new RleBitPackingEncoder(1);
431 			for (var i = 0; i < values.size(); i++) {
432 				var v = unwrapOptional(values.get(i));
433 				defLevels.writeInt(v == null ? 0 : 1);
434 				if (v != null)
435 					writeValue(writer, col, v);
436 			}
437 			defLevelBytes = defLevels.toByteArrayWithLength();
438 		}
439 		byte[] valueBytes = writer.finalizePage();
440 		byte[] pageData = new byte[defLevelBytes.length + valueBytes.length];
441 		System.arraycopy(defLevelBytes, 0, pageData, 0, defLevelBytes.length);
442 		System.arraycopy(valueBytes, 0, pageData, defLevelBytes.length, valueBytes.length);
443 		return pageData;
444 	}

448 	/**
449 	 * Walks {@code path} from {@code row}, returning the definition level (count of leading present OPTIONAL
450 	 * nodes) and the leaf value.  Used by multi-level OPTIONAL columns (GAP-14): {@code def == maxDef} means
451 	 * the leaf is present; {@code def == maxDef-1} means the deepest group is present but the leaf is null;
452 	 * {@code def < maxDef-1} means an intermediate group is null at level {@code def}.
453 	 */
454 	private DefValue computeDefValue(Object row, String path, int maxDef) throws SerializeException {
455 		var parts = path.split("\\.");
456 		var seen = Collections.newSetFromMap(new IdentityHashMap<Object,Boolean>());
457 		seen.add(row);
458 		seen.add(((BeanMap<?>) row).getBean());
459 		Object current = row;
460 		int def = 0;
461 		for (var i = 0; i < parts.length; i++) {
462 			var part = parts[i];
463 			if (current instanceof Optional<?> current2)
464 				current = current2.orElse(null);
465 			else if (current instanceof BeanMap<?> current2)
466 				current = current2.get(part);
467 			else
468 				current = toBeanMap(current).get(part);
469 			if (current != null && seen.contains(current))
470 				current = handleCycle(path);
471 			if (current == null)
472 				return new DefValue(def, null);
473 			seen.add(current);
474 			def = i + 1;
475 		}
476 		return new DefValue(maxDef, unwrapOptional(current));
477 	}

495 		int depth = 0;

509 	private ColumnChunkMeta writeListColumnChunk(OutputStream out, List<?> rows, ParquetSchemaElement col, String path, long chunkStart) throws IOException, SerializeException {
510 		var flattened = extractFlattenedListValues(rows, path);
511 		int maxDef = maxDefLevelForListPath(path);
512 		int maxRep = listDepth(path);
513 		int defBitWidth = defRepBitWidth(maxDef);
514 		int repBitWidth = defRepBitWidth(maxRep);
515 		var defLevels = new RleBitPackingEncoder(defBitWidth);
516 		var repLevels = new RleBitPackingEncoder(repBitWidth);
517 		var writer = new ParquetColumnWriter(col);
518 		for (var e : flattened) {
519 			repLevels.writeInt(e.repLevel);
520 			defLevels.writeInt(e.defLevel);
521 			var v = unwrapOptional(e.value);
522 			if (e.defLevel >= maxDef)
523 				writeValue(writer, col, v);
524 		}
525 		byte[] valueBytes = writer.finalizePage();
526 		byte[] defLevelBytes = defLevels.toByteArrayWithLength();
527 		byte[] repLevelBytes = repLevels.toByteArrayWithLength();
528 		byte[] pageData = new byte[repLevelBytes.length + defLevelBytes.length + valueBytes.length];
529 		int off = 0;
530 		System.arraycopy(repLevelBytes, 0, pageData, off, repLevelBytes.length);
531 		off += repLevelBytes.length;
532 		System.arraycopy(defLevelBytes, 0, pageData, off, defLevelBytes.length);
533 		off += defLevelBytes.length;
534 		System.arraycopy(valueBytes, 0, pageData, off, valueBytes.length);

668 	@SuppressWarnings({
669 		"java:S107" // Recursive list flattening requires path/index/def/rep context; refactor would obscure
670 	})
671 	private void flattenListValuesImpl(Object obj, String path, int partIndex, boolean firstInRow, int currentRep, int listDepth, int maxDef, List<FlattenedEntry> out, Set<Object> seen) throws SerializeException {
672 		var parts = path.split("\\.");
673 		if (partIndex >= parts.length) {
674 			var val = unwrapOptional(obj);
675 			out.add(new FlattenedEntry(val, val != null ? maxDef : Math.max(0, maxDef - 1), currentRep));
676 			return;
677 		}
678 		if (partIndex + 1 < parts.length && "list".equals(parts[partIndex]) && "element".equals(parts[partIndex + 1])) {
679 			Collection<?> list = null;
680 			if (obj != null) {
681 			if (obj instanceof Collection<?> obj2)
682 				list = obj2;
683 			else if (obj instanceof Object[] obj2) {
684 				list = Arrays.asList(obj2);
685 			} else if (obj.getClass().isArray()) {
686 				// Primitive arrays (boolean[], int[], char[], etc.) — box each element
687 				int len = Array.getLength(obj);
688 				var al = new ArrayList<>(len);
689 				for (int ai = 0; ai < len; ai++)
690 					al.add(Array.get(obj, ai));
691 				list = al;
692 			} else
693 				list = Collections.singletonList(obj);
694 			}
695 		// Count how many "list" segments have appeared up to and including partIndex.
696 		// This gives the true nesting depth regardless of non-list path segments (e.g. "value" for Optional).
697 		int listDepthHere = 0;
698 		for (int di = 0; di <= partIndex; di++)
699 			if ("list".equals(parts[di]))
700 				listDepthHere++;
701 		// def levels for null/empty: 2*(depth-1) and 2*(depth-1)+1, so depth=1 → 0,1; depth=2 → 2,3.
702 		int nullDef  = 2 * (listDepthHere - 1);
703 		int emptyDef = 2 * (listDepthHere - 1) + 1;
704 		if (list == null) {
705 			out.add(new FlattenedEntry(null, nullDef, firstInRow ? 0 : currentRep));
706 			return;
707 		}
708 		if (list.isEmpty()) {
709 			out.add(new FlattenedEntry(null, emptyDef, firstInRow ? 0 : currentRep));
710 			return;
711 		}
712 		int level = partIndex / 2 + 1;
713 		// Use currentRep for the first-element rep level so nested lists carry the correct rep from their parent.
714 		int repLevel = firstInRow ? 0 : currentRep;
715 			int idx = 0;
716 			for (var item : list) {
717 				int r = idx == 0 ? repLevel : level;
718 				var val = unwrapOptional(item);
719 				int def = val == null ? Math.max(0, maxDef - 1) : maxDef;
720 				if (partIndex + 2 >= parts.length) {
721 					out.add(new FlattenedEntry(val, def, r));
722 				} else {
723 					flattenListValues(item, path, partIndex + 2, firstInRow, r, listDepth, maxDef, out, seen);
724 				}
725 				firstInRow = false;
726 				idx++;
727 			}
728 			return;
729 		}
730 		Object next = obj;
731 		if (next != null) {
732 			if (next instanceof Optional<?> next2)
733 				next = next2.orElse(null);
734 			else if (next instanceof BeanMap<?> next2)
735 				next = next2.get(parts[partIndex]);
736 			else if (next instanceof Map<?,?> next2)
737 				next = next2.get(parts[partIndex]);
738 			else
739 				next = toBeanMap(next).get(parts[partIndex]);
740 		}
741 		flattenListValues(next, path, partIndex + 1, firstInRow, currentRep, listDepth, maxDef, out, seen);
742 	}
