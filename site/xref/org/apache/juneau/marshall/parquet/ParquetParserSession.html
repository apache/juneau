167 	/**
168 	 * Buffered array-element {@link RecordReader} for Parquet, bound to this live session.  Calls
169 	 * {@code parse(input, List.class, Object.class)} once and iterates the result.
170 	 *
171 	 * <h5 class='section'>Why this stays buffered (not O(1) streaming):</h5>
172 	 * <p>
173 	 * Although a Parquet file is logically row-shaped, it is <b>not</b> a forward, element-at-a-time
174 	 * wire format.  Its metadata footer (schema + row-group/column-chunk offsets) lives at the
175 	 * <b>end</b> of the file, so a reader must seek to the tail before it can interpret any row, and
176 	 * values are stored <b>column-major</b> &mdash; reconstructing one logical record requires
177 	 * reading one page from every column chunk.  Both properties force whole-file (or at minimum
178 	 * whole-row-group) buffering; a true O(1)-in-rows cursor is not achievable over the current
179 	 * single-row-group in-memory decoder.  Left buffered by design.
180 	 *
181 	 * @param input The input.
182 	 * @return A buffered {@link RecordReader}.
183 	 * @throws IOException If a problem occurred reading the input.
184 	 */
185 	@Override /* ArrayRecordReadable */
186 	public RecordReader readArrayRecords(Object input) throws IOException {
187 		return RecordAdapter.arrayReader(this, input);
188 	}

219 	@Override
220 	@SuppressWarnings({
221 		"unchecked" // Generic (T) casts required due to type erasure in doRead
222 	})
223 	protected <T> T doRead(ParserPipe pipe, ClassMeta<T> type) throws IOException, ParseException {
224 		var bytes = readAllBytes(pipe, effectiveMaxInputLength());
225 		if (bytes.length < 12)
226 			throw new ParseException("Parquet file too small");
227 		if (!startsWith(bytes, MAGIC, 0) || !startsWith(bytes, MAGIC, bytes.length - 4))
228 			throw new ParseException("Invalid Parquet magic");
229 		int footerLen = readLe4(bytes, bytes.length - 8);
230 		int footerStart = bytes.length - 8 - footerLen;
231 		if (footerStart < 4)
232 			throw new ParseException("Invalid footer length");
233 		var footer = Arrays.copyOfRange(bytes, footerStart, bytes.length - 8);
234 		var meta = readFileMetaData(footer);
235 		if (meta.numRows() < 0 || meta.numRows() > effectiveMaxCount())
236 			throw new ParseException("Invalid numRows: %s", meta.numRows());
237 		ClassMeta<?> effectiveType = type;
238 		if (type.isOptional())
239 			effectiveType = type.getElementType();
240 		var elementType = effectiveType.isCollection() ? effectiveType.getElementType() : effectiveType;
241 		if (elementType == null || elementType.isObject())
242 			elementType = ctx.getMarshallingContext().getClassMeta(Map.class);
243 		// For scalar types, use Map so reassembleRows keeps raw map rows for unwrap logic
244 		if (!effectiveType.isMap() && !effectiveType.isCollection() && !effectiveType.isArray()) {
245 			var inner = effectiveType.inner();
246 			if (inner != null && (inner.isPrimitive() || inner == String.class || inner == Boolean.class
247 				|| inner == Character.class || Number.class.isAssignableFrom(inner)))
248 				elementType = ctx.getMarshallingContext().getClassMeta(Map.class);
249 		}
250 		var rows = readAllRows(bytes, meta, elementType, meta.schemaRepetition(), meta.rawByteArrayPaths(), meta.uuidPaths(), meta.columnLogical());
251 		// Unwrap ValueHolder {value: X} when single row has "value" key and target expects scalar (not Map/List<Map>)
252 		boolean targetWantsScalar = !type.isMap()
253 			&& !(type.isCollection() && type.getElementType() != null && type.getElementType().isMap())
254 			&& !(type.isArray() && type.getElementType() != null && type.getElementType().isMap());
255 		if (rows.size() == 1 && rows.get(0) instanceof Map<?,?> m && m.containsKey("value") && targetWantsScalar) {
256 			var inner = m.get("value");
257 			if (type.isOptional())
258 				return (T)o(convertToType(inner, type.getElementType()));
259 			if (type.isArray()) {
260 				var list = new ArrayList<>(List.of(inner));
261 				return (T)convertToType(list, type);
262 			}
263 			if (type.isCollection()) {
264 				Object coll = type.newInstance();
265 				if (coll == null)
266 					coll = l();
267 				((Collection<Object>)coll).add(convertToType(prepareMapForBean(inner, type.getElementType()), type.getElementType()));
268 				return (T)coll;
269 			}
270 			return (T)convertToType(prepareMapForBean(inner, type), type);
271 		}
272 		if (type.isMap()) {
273 			// Unwrap ValueHolder {value: Map} when addRootType wrapped root Map (2.2)
274 			if (rows.size() == 1 && rows.get(0) instanceof Map<?,?> m && m.containsKey("value") && m.size() == 1) {
275 				var inner = m.get("value");
276 				if (inner instanceof Map<?,?>)
277 					return (T)convertToType(inner, type);
278 			}
279 			var keyType = type.getKeyType();
280 			if (keyType != null && !keyType.isAssignableTo(String.class) && isKeyValuePairFormat(rows)) {
281 				var valueType = type.getValueType();
282 				if (valueType == null) valueType = ctx.getMarshallingContext().getClassMeta(Object.class);
283 				var result = m();
284 				for (var row : rows) {
285 					var rowMap = (Map<?,?>)row;
286 					var rawKey = rowMap.get("key");
287 					var k = (rawKey != null && ctx.nullKeyString.equals(String.valueOf(rawKey)))
288 						? null
289 							: convertToType(rawKey, keyType);
290 					var v = convertToType(rowMap.get("value"), valueType);
291 					result.put(k, v);
292 				}
293 				return (T)convertToType(result, type);
294 			}
295 			if (rows.size() == 1 && rows.get(0) instanceof Map<?,?> m) {
296 				// Unwrap {root: {...}} from buildSchemaFromMap flat format
297 				if (m.size() == 1 && m.containsKey("root") && m.get("root") instanceof Map<?,?> inner)
298 					return (T)convertToType(prepareMapForBean(inner, type), type);
299 				return (T)convertToType(prepareMapForBean(m, type), type);
300 			}
301 		}
302 		if (type.isOptional()) {
303 			if (rows.isEmpty())
304 				return (T)oe();
305 			var inner = unwrapValueHolder(rows.get(0), type.getElementType());
306 			return (T)o(inner);
307 		}
308 		if (type.isArray()) {
309 			var componentMeta = type.getElementType();
310 			var unwrapped = l();
311 			for (var row : rows) {
312 				var v = unwrapValueHolder(row, componentMeta);
313 				unwrapped.add(v);
314 			}
315 			return (T)toArray(unwrapped, componentMeta);
316 		}
317 		if (type.isCollection()) {
318 			var collElemType = type.getElementType();
319 			var unwrapped = l();
320 			for (var row : rows) {
321 				var v = unwrapValueHolder(row, collElemType);
322 				unwrapped.add(v);
323 			}
324 			return (T)toCollection(unwrapped, type);
325 		}
326 		if (rows.size() == 1) {
327 			var row0 = rows.get(0);
328 			var unwrapped = unwrapValueHolder(row0, type);
329 			return (T)unwrapped;
330 		}
331 		if (rows.isEmpty())
332 			return null;
333 		return (T)rows;
334 	}

336 	private static byte[] readAllBytes(ParserPipe pipe, long maxInputLength) throws IOException, ParseException {
337 		var cap = maxInputLength <= 0 ? Long.MAX_VALUE : maxInputLength;
338 		try (var is = pipe.getInputStream()) {
339 			var baos = new ByteArrayOutputStream();
340 			var buf = new byte[8192];
341 			int n;
342 			long total = 0;
343 			while ((n = is.read(buf)) >= 0) {
344 				total += n;
345 				if (total > cap)
346 					throw new ParseException("Parquet input length exceeds maximum allowed %s bytes", cap);
347 				baos.write(buf, 0, n);
348 			}
349 			return baos.toByteArray();
350 		}
351 	}

450 	/**
451 	 * Reads a single Parquet {@code PageHeader} Thrift struct starting at {@code off} in {@code fileBytes}.
452 	 *
453 	 * @param fileBytes The whole file.
454 	 * @param off Offset of the page header.
455 	 * @param columnPath Column path (for error messages only).
456 	 * @param maxLength The configured maximum page byte size (see {@link ParquetParser.Builder#maxLength(int)}).
457 	 * @param maxCount The configured maximum count, applied to the page's own {@code num_values} sub-header field
458 	 * 	(see {@link ParquetParser.Builder#maxCount(int)}).
459 	 * @return The decoded header plus the offset where the page body begins.
460 	 * @throws ParseException If the header is malformed or declares out-of-range sizes.
461 	 * @throws IOException If the underlying Thrift decode fails.
462 	 */
463 	@SuppressWarnings({
464 		"resource" // bais is an in-memory ByteArrayInputStream; no OS resource to close.
465 	})
466 	private static PageHeaderInfo readPageHeader(byte[] fileBytes, int off, String columnPath, int maxLength, long maxCount) throws ParseException, IOException {
467 		var bais = new ByteArrayInputStream(fileBytes, off, fileBytes.length - off);
468 		var dec = new ThriftCompactDecoder(bais);
469 		dec.readStructBegin();
470 		int pageType = PAGE_DATA;
471 		int uncompressedSize = 0;
472 		int compressedSize = 0;
473 		int numValues = 0;
474 		int encoding = ENCODING_PLAIN;
475 		ThriftCompactDecoder.FieldHeader fh;
476 		while (!(fh = dec.readFieldHeader()).isStop) {
477 			switch (fh.fieldId) {
478 				case 1 -> pageType = dec.readI32();
479 				case 2 -> uncompressedSize = dec.readI32();
480 				case 3 -> compressedSize = dec.readI32();
481 				case 5, 7 -> { // DataPageHeader (5) or DictionaryPageHeader (7): both carry num_values(1)+encoding(2).
482 					dec.readStructBegin();
483 					ThriftCompactDecoder.FieldHeader sub;
484 					while (!(sub = dec.readFieldHeader()).isStop) {
485 						if (sub.fieldId == 1)
486 							numValues = dec.readI32();
487 						else if (sub.fieldId == 2)
488 							encoding = dec.readI32();
489 						else
490 							dec.skipField(sub.type);
491 					}
492 					dec.readStructEnd();
493 				}
494 				default -> dec.skipField(fh.type);
495 			}
496 		}
497 		dec.readStructEnd();
498 		int headerConsumed = (fileBytes.length - off) - bais.available();
499 		int bodyStart = off + headerConsumed;
500 		if (uncompressedSize < 0 || uncompressedSize > maxLength
501 			|| compressedSize < 0 || compressedSize > maxLength
502 			|| bodyStart + compressedSize > fileBytes.length)
503 			throw new ParseException("Invalid page header for column '%s': uncompressed='%s', compressed='%s'",
504 				columnPath, uncompressedSize, compressedSize);
505 		if (numValues < 0 || numValues > maxCount)
506 			throw new ParseException("Invalid page num_values for column '%s': %s", columnPath, numValues);
507 		return new PageHeaderInfo(pageType, uncompressedSize, compressedSize, numValues, encoding, bodyStart);
508 	}

582 	private static SchemaReadResult readSchema(ThriftCompactDecoder dec) throws IOException {
583 		var lh = dec.readListHeader();
584 		var pathStack = new ArrayList<SchemaStackFrame>();
585 		var result = new LinkedHashMap<String,Integer>();
586 		var rawByteArrayPaths = new LinkedHashSet<String>();
587 		var uuidPaths = new LinkedHashSet<String>();
588 		var columnLogical = new LinkedHashMap<String,ColumnLogical>();
589 		for (int i = 0; i < lh.size; i++) {
590 			dec.readStructBegin();
591 			Integer type = null;
592 			Integer repetitionType = null;
593 			String name = null;
594 			Integer numChildren = null;
595 			Integer convertedType = null;
596 			Integer logicalType = null;
597 			int scale = 0;
598 			int precision = 0;
599 			ThriftCompactDecoder.FieldHeader fh;
600 			while (!(fh = dec.readFieldHeader()).isStop) {
601 				switch (fh.fieldId) {
602 					case 1 -> type = dec.readI32();
603 					case 2 -> dec.readI32();
604 					case 3 -> repetitionType = dec.readI32();
605 					case 4 -> name = dec.readString();
606 					case 5 -> numChildren = dec.readI32();
607 					case 6 -> convertedType = dec.readI32();
608 					case 7 -> scale = dec.readI32();      // SchemaElement.scale (DECIMAL)
609 					case 8 -> precision = dec.readI32();  // SchemaElement.precision (DECIMAL)
610 					case 10 -> logicalType = readLogicalTypeUnion(dec);
611 					default -> dec.skipField(fh.type);
612 				}
613 			}
614 			dec.readStructEnd();
615 			if (name == null)
616 				continue;
617 			var parts = new ArrayList<String>();
618 			for (var f : pathStack)
619 				parts.add(f.name);
620 			parts.add(name);
621 			var path = String.join(".", parts);
622 			if (numChildren != null && numChildren > 0) {
623 				pathStack.add(new SchemaStackFrame(name, numChildren));
624 			} else if (type != null) {
625 				// Leaf element: record it and decrement parent's child count
626 				int rep = or(repetitionType, OPTIONAL);
627 				result.put(path, rep);
628 				// Capture logical-type metadata for native-type decode (GAP-9/10).  Only the DECIMAL and
629 				// temporal converted types matter to the decoder; the UUID/string paths are handled separately.
630 				if (convertedType != null)
631 					columnLogical.put(path, new ColumnLogical(convertedType, scale, precision));
632 				// A TYPE_BYTE_ARRAY column with no convertedType uniquely identifies a
633 				// raw byte[] column emitted by ParquetSchemaBuilder.addLeafSchema's cm.isByteArray()
634 				// branch.  Every other writer site that emits TYPE_BYTE_ARRAY sets a discriminator
635 				// (CONVERTED_UTF8 for strings/dates/durations/decimals, CONVERTED_ENUM for enums, or
636 				// CONVERTED_TIMESTAMP_MILLIS).  Mirrors the UUID parse-back assumption
637 				// (TYPE_FIXED_LEN_BYTE_ARRAY uniquely identifies UUIDs) at a different physical type.
638 				if (type == TYPE_BYTE_ARRAY && convertedType == null)
639 					rawByteArrayPaths.add(path);
640 				// Prefer the explicit LogicalType discriminant (field 10) when present; otherwise fall back to
641 				// the legacy "FIXED_LEN_BYTE_ARRAY == UUID" physical-type heuristic for backward compatibility
642 				// with files written without the opt-in discriminator.
643 				if (type == TYPE_FIXED_LEN_BYTE_ARRAY)
644 					uuidPaths.add(path);
645 				if (parquetDebug())
646 					parquetDebugLog("readSchema leaf: path=" + path + " name=" + name + " rep=" + rep
647 						+ " convertedType=" + convertedType + " logicalType=" + logicalType + " pathStack=" + pathStack);
648 				boolean moreToDecrement = true;
649 				while (!pathStack.isEmpty() && moreToDecrement) {
650 					var top = pathStack.get(pathStack.size() - 1);
651 					var newRemaining = top.remaining - 1;
652 					if (newRemaining <= 0)
653 						pathStack.remove(pathStack.size() - 1);
654 					else {
655 						pathStack.set(pathStack.size() - 1, new SchemaStackFrame(top.name, newRemaining));
656 						moreToDecrement = false;
657 					}
658 				}
659 			}
660 		}
661 		if (parquetDebug())
662 			parquetDebugLog("readSchema result: " + result + " rawByteArrayPaths=" + rawByteArrayPaths + " uuidPaths=" + uuidPaths);
663 		return new SchemaReadResult(result, rawByteArrayPaths, uuidPaths, columnLogical);
664 	}

771 	private List<?> readAllRows(byte[] fileBytes, FileMeta meta, ClassMeta<?> elementType, Map<String,Integer> schemaRepetition, Set<String> rawByteArrayPaths, Set<String> uuidPaths, Map<String,ColumnLogical> columnLogical) throws ParseException {
772 		if (meta.numRows() == 0 || meta.rowGroups().isEmpty())
773 			return List.of();
774 		// Multi-row-group (GAP-2): read every row group and concatenate its reassembled rows in order.
775 		// Each group declares its own num_rows; fall back to the file total only for a lone group whose
776 		// per-group count wasn't recorded (older Juneau-written files).
777 		var maxCount = effectiveMaxCount();
778 		var allRows = new ArrayList<Object>((int)Math.min(meta.numRows(), maxCount));
779 		// Each per-row-group num_rows is untrusted and drives ArrayList pre-allocation and reassembly loops
780 		// downstream.  Bound every group count (and the running total) against the same configured ceiling
781 		// used for the file-level total above, before it is used to size anything, so a tiny footer
782 		// declaring a huge per-group count cannot force a giant allocation.
783 		long totalGroupRows = 0;
784 		for (var group : meta.rowGroups()) {
785 			if (group.numRows() < 0 || group.numRows() > maxCount)
786 				throw new ParseException("Invalid row group numRows: %s", group.numRows());
787 			totalGroupRows += group.numRows();
788 			if (totalGroupRows > maxCount)
789 				throw new ParseException("Total row group numRows %s exceeds maximum allowed %s", totalGroupRows, maxCount);
790 			int groupRows = (int)group.numRows();
791 			allRows.addAll(readRowGroupRows(fileBytes, group, groupRows, elementType, schemaRepetition, rawByteArrayPaths, uuidPaths, columnLogical));
792 		}
793 		return allRows;
794 	}

855 	private static List<Object> readListColumnChunk(byte[] fileBytes, ColumnChunkMeta cc, int numRows, boolean trimStrings, int maxLength, long maxCount) throws ParseException {
856 		try {
857 			var path = String.join(".", cc.pathInSchema());
858 			var rowRelPath = rowRelativePath(path);
859 			int listDepth = listDepth(rowRelPath);
860 			int maxDef = 3 * listDepth;
861 			int maxRep = listDepth;
862 			int defBitWidth = 32 - Integer.numberOfLeadingZeros(maxDef);
863 			int repBitWidth = 32 - Integer.numberOfLeadingZeros(maxRep);
864 
865 			if (cc.numValues() < 0 || cc.numValues() > maxCount)
866 				throw new ParseException("Invalid numValues for column '%s': %s", String.join(".", cc.pathInSchema()), cc.numValues());
867 			var codec = CompressionCodec.fromThrift(cc.codec());
868 			// Skip a leading dictionary page if present, then read the single data page.  Repeated
869 			// (list/map) columns carry rep+def levels with cross-page state; Juneau emits them as a single
870 			// data page, and dictionary-encoded repeated columns are not produced here, so a single-data-page
871 			// decode (via the shared page-header reader for Snappy + validation) is sufficient.
872 			var chunkPath = String.join(".", cc.pathInSchema());
873 			int dataPageOff = skipToDataPage(fileBytes, (int)cc.dataPageOffset(), chunkPath, maxLength, maxCount);
874 			var ph = readPageHeader(fileBytes, dataPageOff, chunkPath, maxLength, maxCount);
875 			var compressedData = Arrays.copyOfRange(fileBytes, ph.bodyStart(), ph.bodyStart() + ph.compressedSize());
876 			var decompressed = codec.decompress(compressedData, ph.uncompressedSize());
877 
878 			int numValues = (int)cc.numValues();
879 			// The rep-level and def-level length prefixes are untrusted LE4 values read from the (decompressed)
880 			// page body; bound each against the bytes actually available before it is used to size an
881 			// Arrays.copyOfRange allocation, so a tiny page cannot declare a huge length and force a giant
882 			// buffer.
883 			if (decompressed.length < 4)
884 				throw new ParseException("Invalid Parquet list page for column '%s': too small for rep-level length", chunkPath);
885 			int off2 = 0;
886 			int repLen = checkPageLength(readLe4(decompressed, off2), decompressed.length - 4, "list rep-level", chunkPath);
887 			off2 += 4 + repLen;
888 			int defStart = off2;
889 			if (defStart + 4 > decompressed.length)
890 				throw new ParseException("Invalid Parquet list page for column '%s': truncated def-level length", chunkPath);
891 			int defLen = checkPageLength(readLe4(decompressed, defStart), decompressed.length - defStart - 4, "list def-level", chunkPath);
892 			off2 += 4 + defLen;
893 			var valueBytes = Arrays.copyOfRange(decompressed, off2, decompressed.length);
894 			var repBytes = Arrays.copyOfRange(decompressed, 4, 4 + repLen);
895 			var defBytes = Arrays.copyOfRange(decompressed, defStart + 4, defStart + 4 + defLen);
896 			var repDecoder = new RleBitPackingDecoder(repBytes, repBitWidth);
897 			var defDecoder = new RleBitPackingDecoder(defBytes, defBitWidth);
898 			var valueReader = new ParquetColumnReader(valueBytes, numValues, 0);
899 
900 			var flattened = new ArrayList<Object[]>();
901 			for (int i = 0; i < numValues; i++) {
902 				int rep = repDecoder.readInt();
903 				int def = defDecoder.readInt();
904 				Object val = def >= maxDef ? readValue(valueReader, cc.type(), trimStrings) : null;
905 				flattened.add(new Object[] { val, rep, def });
906 			}
907 
908 			if (listDepth == 1)
909 				return reconstructRowsFromListColumn(flattened, numRows, maxDef, maxCount);
910 			return reconstructNestedListColumn(flattened, numRows, listDepth, maxDef, maxCount);
911 		} catch (IOException e) {
912 			throw new ParseException(e);
913 		}
914 	}

916 	/**
917 	 * Reconstructs depth-N nested list columns (depth &ge; 2), replacing the former depth-2-only method.
918 	 *
919 	 * <p>This unified algorithm supports arbitrarily deep nesting (e.g. {@code boolean[][]},
920 	 * {@code List<boolean[]>}, {@code List<Long>[][]}, {@code List<Long[][]>}).
921 	 *
922 	 * <h5>Rep level semantics (depth=N, maxRep=N):</h5>
923 	 * <ul>
924 	 *   <li>rep=0 — new row; flush the entire stack and start fresh
925 	 *   <li>rep=k (0 &lt; k &lt; N) — new element at nesting level k; flush inner levels k+1..N
926 	 *   <li>rep=N — continue the innermost list (append another element)
927 	 * </ul>
928 	 *
929 	 * <h5>Def level semantics (maxDef = 3*N):</h5>
930 	 * <ul>
931 	 *   <li>def=2*(k-1) — null list at level k (1 &le; k &le; N)
932 	 *   <li>def=2*(k-1)+1 — empty list at level k (1 &le; k &le; N)
933 	 *   <li>def=maxDef-1 — leaf element is null
934 	 *   <li>def=maxDef — leaf element is present
935 	 * </ul>
936 	 *
937 	 * <p>The "stacks" array tracks the currently open list at each nesting level.
938 	 * {@code stacks[0]} is the outermost list (what will be placed in the row),
939 	 * {@code stacks[N-1]} is the innermost list being appended to.
940 	 */
941 	@SuppressWarnings({
942 		"unchecked" // Generic array creation for stacks.
943 	})
944 	private static List<Object> reconstructNestedListColumn(List<Object[]> flattened, int numRows, int depth, int maxDef, long maxCount) {
945 		var result = new ArrayList<>(clampCapacity(numRows, maxCount));
946 		var stacks = new ArrayList[depth];
947 
948 		for (var e : flattened) {
949 			Object val = e[0];
950 			int rep = (Integer)e[1];
951 			int def = (Integer)e[2];
952 
953 			// Flush: close inner lists from the deepest level down to `rep`, adding each to its parent.
954 			// For rep=0 this flushes everything including stacks[0] into result.
955 			for (int k = depth - 1; k >= rep; k--) {
956 				if (stacks[k] != null) {
957 					if (k > 0)
958 						stacks[k - 1].add(stacks[k]);
959 					else
960 						result.add(stacks[k]);
961 					stacks[k] = null;
962 				}
963 			}
964 
965 			// Null at level 1 is the only case where def==0 AND rep==0; it was flushed above.
966 			if (rep == 0 && def == 0) {
967 				result.add(null);   // whole field is null
968 			} else if (def <= 2 * (depth - 1) + 1) {
969 				// Null or empty list at nesting level k (1-indexed): def=2*(k-1) → null; def=2*(k-1)+1 → empty.
970 				int level = def / 2 + 1;
971 				boolean isNull = (def % 2 == 0);
972 				// Ensure stacks[start..level-2] exist (parents of the null/empty list).
973 				int startK = (rep == 0) ? 0 : rep;
974 				for (int k = startK; k < level - 1; k++)
975 					if (stacks[k] == null) stacks[k] = new ArrayList<>();
976 				if (level == 1) {
977 					// Level-1 null is handled above (def==0); level-1 empty reaches here (def==1).
978 					result.add(l());
979 				} else if (isNull) {
980 					stacks[level - 2].add(null);   // null list at level k → parent gets null slot
981 				} else {
982 					stacks[level - 2].add(l());  // empty list at level k
983 				}
984 			} else {
985 				// Element (null if def < maxDef, present if def == maxDef).
986 				if (rep < depth) {
987 					// Initialize stacks from the first missing level down to the innermost.
988 					int startK = (rep == 0) ? 0 : rep;
989 					for (int k = startK; k < depth; k++)
990 						if (stacks[k] == null) stacks[k] = new ArrayList<>();
991 				} else {
992 					// rep == depth: continue innermost list (defensive init in case it's missing).
993 					if (stacks[depth - 1] == null) stacks[depth - 1] = new ArrayList<>();
994 				}
995 				stacks[depth - 1].add(def >= maxDef ? val : null);
996 			}
997 		}
998 
999 		// Flush the final row's stacks.
1000 		for (int k = depth - 1; k >= 0; k--) {
1001 			if (stacks[k] != null) {
1002 				if (k > 0)
1003 					stacks[k - 1].add(stacks[k]);
1004 				else
1005 					result.add(stacks[k]);
1006 				stacks[k] = null;
1007 			}
1008 		}
1009 		while (result.size() < numRows)
1010 			result.add(l());
1011 		return result;
1012 	}

1396 	private static UUID uuidFromFixedLenBytes(byte[] b) {
1397 		if (b == null || b.length != 16)
1398 			return new UUID(0L, 0L);
1399 		long msb = 0L;
1400 		long lsb = 0L;
