17  package org.apache.juneau.marshall.parquet;
18  
19  import java.io.*;
20  
21  /**
22   * RLE/Bit-Packing Hybrid encoder for definition/repetition levels and booleans.
23   *
24   * <p>
25   * Implements the encoding from the
26   * <a class="doclink" href="https://parquet.apache.org/docs/file-format/data-pages/encodings/">Parquet encodings</a>
27   * specification. Values are packed in groups of 8 (bit-packed run) or encoded as repeated runs (RLE run).
28   */
29  final class RleBitPackingEncoder {
30  
31  	private final int bitWidth;
32  	private final ByteArrayOutputStream out = new ByteArrayOutputStream();
33  	private int lastValue = -1;
34  	private int runLength;
35  	private final int[] bitPackedBuffer = new int[8];
36  	private int bitPackedCount;
37  
38  	RleBitPackingEncoder(int bitWidth) {
39  		if (bitWidth < 1 || bitWidth > 32)
40  			throw new IllegalArgumentException("bitWidth must be 1-32");
41  		this.bitWidth = bitWidth;
42  	}
43  
44  	void writeInt(int value) {
45  		if (value == lastValue) {
46  			runLength++;
47  			if (runLength >= 8) {
48  				flushRun();
49  			}
50  		} else {
51  			if (runLength > 0) {
52  				flushRun();
53  			}
54  			lastValue = value;
55  			runLength = 1;
56  		}
57  	}
58  
59  	private void flushRun() {
60  		if (runLength >= 8) {
61  			writeRleRun(lastValue, runLength);
62  		} else {
63  			for (int i = 0; i < runLength; i++) {
64  				bitPackedBuffer[bitPackedCount++] = lastValue;
65  				if (bitPackedCount == 8) {
66  					writeBitPackedRun(bitPackedBuffer, 8);
67  					bitPackedCount = 0;
68  				}
69  			}
70  		}
71  		runLength = 0;
72  		lastValue = -1;
73  	}
74  
75  	private void writeRleRun(int value, int count) {
76  		writeVarint((long) count << 1);
77  		var bytesPerValue = (bitWidth + 7) / 8;
78  		for (int i = 0; i < bytesPerValue; i++)
79  			out.write((value >> (i * 8)) & 0xFF);
80  	}
81  
82  	/**
83  	 * Packs 8 values into bytes. Bits are packed from LSB of each value to MSB, and from LSB of each byte.
84  	 * Per Parquet spec: values packed from LSB of each byte to MSB; bits within each value stay MSB to LSB.
85  	 */
86  	private void writeBitPackedRun(int[] values, int count) {
87  		var scaledLen = count / 8;
88  		writeVarint((long) scaledLen << 1 | 1);
89  		var totalBits = bitWidth * 8;
90  		var totalBytes = (totalBits + 7) / 8;
91  		for (int byteIdx = 0; byteIdx < totalBytes; byteIdx++) {
92  			var acc = 0;
93  			for (int bitIdx = 0; bitIdx < 8; bitIdx++) {
94  				var globalBit = byteIdx * 8 + bitIdx;
95  				var valIdx = globalBit / bitWidth;
96  				var valBit = bitWidth - 1 - (globalBit % bitWidth);
97  				var val = values[valIdx] & ((1 << bitWidth) - 1);
98  				if ((val & (1 << valBit)) != 0)
99  					acc |= 1 << bitIdx;
100 			}
101 			out.write(acc & 0xFF);
102 		}
103 	}
104 
105 	private void writeVarint(long value) {
106 		out.write((int)(value & 0x7F));
107 	}
108 
109 	/**
110 	 * Finalizes and returns the encoded bytes.
111 	 */
112 	byte[] toByteArray() {
113 		if (runLength > 0) {
114 			for (int i = 0; i < runLength; i++)
115 				bitPackedBuffer[bitPackedCount++] = lastValue;
116 			runLength = 0;
117 			while (bitPackedCount < 8)
118 				bitPackedBuffer[bitPackedCount++] = 0;
119 			writeBitPackedRun(bitPackedBuffer, 8);
120 		}
121 		return out.toByteArray();
122 	}
123 
124 	/**
125 	 * Returns encoded bytes with a 4-byte little-endian length prefix (for Data Page v1 definition/repetition levels).
126 	 */
127 	byte[] toByteArrayWithLength() {
128 		var data = toByteArray();
129 		var result = new byte[4 + data.length];
130 		result[0] = (byte)(data.length & 0xFF);
131 		result[1] = (byte)((data.length >> 8) & 0xFF);
132 		result[2] = (byte)((data.length >> 16) & 0xFF);
133 		result[3] = (byte)((data.length >> 24) & 0xFF);
134 		System.arraycopy(data, 0, result, 4, data.length);
135 		return result;
136 	}
137 }
