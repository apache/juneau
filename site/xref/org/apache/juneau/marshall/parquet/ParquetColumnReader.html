17  package org.apache.juneau.marshall.parquet;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import java.io.*;
22  import java.nio.charset.*;
23  import java.time.*;
24  
25  /**
26   * Reads column values from Parquet page data (PLAIN encoding).
27   *
28   * <p>
29   * Page data layout: 4-byte LE definition levels length, RLE-encoded definition levels, then PLAIN-encoded values.
30   * For REQUIRED columns there are no definition levels.
31   */
32  @SuppressWarnings({
33  	"resource" // valueStream is an in-memory ByteArrayInputStream over page bytes; there is no OS resource to close.
34  })
35  final class ParquetColumnReader {
36  
37  	private final RleBitPackingDecoder defLevelDecoder;
38  	private final InputStream valueStream;
39  	private final int numValues;
40  	private final int maxDefLevel;
41  	private int readCount;
42  	private boolean nextNull;
43  	private int defLevel;
44  	private int boolByte;
45  	private int boolBitPos = 8;
46  
47  	ParquetColumnReader(byte[] pageData, int numValues, int maxDefLevel) {
48  		this(pageData, numValues, maxDefLevel, 1);
49  	}
50  
51  	/**
52  	 * @param pageData The (decompressed) page bytes.
53  	 * @param numValues Number of level/value entries in the page.
54  	 * @param maxDefLevel Maximum definition level; the leaf is present iff its def level equals this.
55  	 * @param defBitWidth Bit width of the RLE-encoded definition levels.  For multi-level OPTIONAL nesting
56  	 * 	(GAP-14) this is wider than 1 so intermediate-null vs leaf-null can be distinguished.
57  	 */
58  	ParquetColumnReader(byte[] pageData, int numValues, int maxDefLevel, int defBitWidth) {
59  		this.numValues = numValues;
60  		this.maxDefLevel = maxDefLevel;
61  		this.readCount = 0;
62  		this.nextNull = false;
63  		int off = 0;
64  		if (maxDefLevel > 0 && pageData.length >= 4) {
65  			var defLen = (pageData[off] & 0xFF) | ((pageData[off + 1] & 0xFF) << 8)
66  				| ((pageData[off + 2] & 0xFF) << 16) | ((pageData[off + 3] & 0xFF) << 24);
67  			var hasDefLevels = defLen >= 0 && defLen <= pageData.length - 4;
68  			if (hasDefLevels) {
69  				off += 4;
70  				this.defLevelDecoder = new RleBitPackingDecoder(pageData, off, defLen, defBitWidth);
71  				off += defLen;
72  			} else {
73  				this.defLevelDecoder = null;
74  			}
75  		} else {
76  			this.defLevelDecoder = null;
77  		}
78  		this.valueStream = new ByteArrayInputStream(pageData, off, pageData.length - off);
79  	}
80  
81  	boolean isNull() {
82  		return nextNull;
83  	}
84  
85  	/**
86  	 * Returns the definition level read by the most recent {@link #advance()} call.
87  	 *
88  	 * <p>
89  	 * Used by multi-level OPTIONAL reconstruction (GAP-14) to tell a null intermediate group
90  	 * (def &lt; maxDefLevel-1) apart from a present group with a null leaf (def == maxDefLevel-1).
91  	 *
92  	 * @return The last definition level (0 when the column carries no definition levels).
93  	 */
94  	int getDefLevel() {
95  		return defLevel;
96  	}
97  
98  	boolean hasNext() {
99  		return readCount < numValues;
100 	}
101 
102 	/**
103 	 * Advances to the next value (reads definition level and increments count).
104 	 * Must be called once per value before reading or checking null.
105 	 */
106 	void advance() throws IOException {
107 		advanceDefLevel();
108 	}
109 
110 	boolean readBoolean() throws IOException {
111 		if (nextNull)
112 			return false;
113 		// PLAIN BOOLEAN is bit-packed 1 bit/value, LSB-first (GAP-4).  Pull a fresh byte every 8 present
114 		// booleans; null slots consume no bit (matches the writer, which only packs present values).
115 		if (boolBitPos == 8) {
116 			boolByte = valueStream.read();
117 			boolBitPos = 0;
118 		}
119 		var bit = (boolByte >> boolBitPos) & 1;
120 		boolBitPos++;
121 		return bit == 1;
122 	}
123 
124 	int readInt32() throws IOException {
125 		if (nextNull)
126 			return 0;
127 		var b0 = valueStream.read() & 0xFF;
128 		var b1 = valueStream.read() & 0xFF;
129 		var b2 = valueStream.read() & 0xFF;
130 		var b3 = valueStream.read() & 0xFF;
131 		return b0 | (b1 << 8) | (b2 << 16) | (b3 << 24);
132 	}
133 
134 	long readInt64() throws IOException {
135 		if (nextNull)
136 			return 0;
137 		var b0 = (long)(valueStream.read() & 0xFF);
138 		var b1 = (long)(valueStream.read() & 0xFF);
139 		var b2 = (long)(valueStream.read() & 0xFF);
140 		var b3 = (long)(valueStream.read() & 0xFF);
141 		var b4 = (long)(valueStream.read() & 0xFF);
142 		var b5 = (long)(valueStream.read() & 0xFF);
143 		var b6 = (long)(valueStream.read() & 0xFF);
144 		var b7 = (long)(valueStream.read() & 0xFF);
145 		return b0 | (b1 << 8) | (b2 << 16) | (b3 << 24) | (b4 << 32) | (b5 << 40) | (b6 << 48) | (b7 << 56);
146 	}
147 
148 	/**
149 	 * Reads a legacy INT96 value (GAP-8) — 12 bytes: an 8-byte little-endian nanoseconds-of-day followed
150 	 * by a 4-byte little-endian Julian day number — and converts it to an {@link Instant}.  Used by
151 	 * Impala/Hive/older-Spark timestamp columns.  Returns <jk>null</jk> for a null slot.
152 	 */
153 	Instant readInt96AsInstant() throws IOException {
154 		if (nextNull) {
155 			// Still consume the 12 bytes so the stream stays aligned.
156 			for (var i = 0; i < 12; i++)
157 				valueStream.read();
158 			return null;
159 		}
160 		long nanosOfDay = readInt64();
161 		int julianDay = readInt32();
162 		// Julian day 2440588 == 1970-01-01.  Days since epoch * 86400s + nanos-of-day.
163 		long epochDay = julianDay - 2440588L;
164 		long epochSecond = epochDay * 86_400L + nanosOfDay / 1_000_000_000L;
165 		long nanoAdjust = nanosOfDay % 1_000_000_000L;
166 		return Instant.ofEpochSecond(epochSecond, nanoAdjust);
167 	}
168 
169 	float readFloat() throws IOException {
170 		return Float.intBitsToFloat(readInt32());
171 	}
172 
173 	double readDouble() throws IOException {
174 		return Double.longBitsToDouble(readInt64());
175 	}
176 
177 	private static final int MAX_BYTE_ARRAY_LEN = 64 * 1024 * 1024;
178 
179 	byte[] readByteArray() throws IOException {
180 		if (nextNull)
181 			return new byte[0];
182 		var len = readInt32Raw();
183 		if (len <= 0)
184 			return new byte[0];
185 		if (len > MAX_BYTE_ARRAY_LEN)
186 			throw ioex("Byte array length %s exceeds maximum %s", len, MAX_BYTE_ARRAY_LEN);
187 		var b = new byte[len];
188 		var n = 0;
189 		while (n < len) {
190 			var r = valueStream.read(b, n, len - n);
191 			if (r <= 0)
192 				break;
193 			n += r;
194 		}
195 		return b;
196 	}
197 
198 	String readByteArrayAsString() throws IOException {
199 		return new String(readByteArray(), StandardCharsets.UTF_8);
200 	}
201 
202 	byte[] readFixedLenByteArray(int length) throws IOException {
203 		if (nextNull)
204 			return new byte[0];
205 		var b = new byte[length];
206 		var n = 0;
207 		while (n < length) {
208 			var r = valueStream.read(b, n, length - n);
209 			if (r <= 0)
210 				break;
211 			n += r;
212 		}
213 		return b;
214 	}
215 
216 	private void advanceDefLevel() throws IOException {
217 		if (readCount >= numValues)
218 			return;
219 		if (maxDefLevel > 0 && defLevelDecoder != null) {
220 			defLevel = defLevelDecoder.readInt();
221 			nextNull = defLevel < maxDefLevel;
222 		} else {
223 			defLevel = maxDefLevel;
224 			nextNull = false;
225 		}
226 		readCount++;
227 	}
228 
229 	private int readInt32Raw() throws IOException {
230 		var b0 = valueStream.read() & 0xFF;
231 		var b1 = valueStream.read() & 0xFF;
232 		var b2 = valueStream.read() & 0xFF;
233 		var b3 = valueStream.read() & 0xFF;
234 		return b0 | (b1 << 8) | (b2 << 16) | (b3 << 24);
235 	}
