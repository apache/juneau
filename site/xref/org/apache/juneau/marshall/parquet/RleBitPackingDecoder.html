17  package org.apache.juneau.marshall.parquet;
18  
19  import java.io.*;
20  
21  /**
22   * RLE/Bit-Packing Hybrid decoder for definition/repetition levels and booleans.
23   */
24  @SuppressWarnings({
25  	"resource" // in is an in-memory ByteArrayInputStream over level bytes; there is no OS resource to close.
26  })
27  final class RleBitPackingDecoder {
28  
29  	private final InputStream in;
30  	private final int bitWidth;
31  	private int bitPackedCount;
32  	private int bitPackedRead;
33  	private final int[] bitPackedBuffer = new int[8];
34  	private int rleValue;
35  	private int rleCount;
36  	RleBitPackingDecoder(byte[] data, int bitWidth) {
37  		this(new ByteArrayInputStream(data), bitWidth);
38  	}
39  
40  	RleBitPackingDecoder(byte[] data, int offset, int length, int bitWidth) {
41  		this(new ByteArrayInputStream(data, offset, length), bitWidth);
42  	}
43  
44  	RleBitPackingDecoder(InputStream in, int bitWidth) {
45  		this.in = in;
46  		this.bitWidth = bitWidth;
47  		this.bitPackedCount = 0;
48  		this.bitPackedRead = 0;
49  		this.rleCount = 0;
50  	}
51  
52  	int readInt() throws IOException {
53  		if (rleCount > 0) {
54  			rleCount--;
55  			return rleValue;
56  		}
57  		if (bitPackedRead < bitPackedCount) {
58  			return bitPackedBuffer[bitPackedRead++];
59  		}
60  		var header = readVarint();
61  		if ((header & 1) == 1) {
62  			readBitPackedGroup();
63  			bitPackedRead = 1;
64  			return bitPackedBuffer[0];
65  		}
66  		var count = (int)(header >>> 1);
67  		var bytesPerValue = (bitWidth + 7) / 8;
68  		rleValue = 0;
69  		for (int i = 0; i < bytesPerValue; i++) {
70  			var x = in.read();
71  			if (x >= 0)
72  				rleValue |= (x & 0xFF) << (i * 8);
73  		}
74  		rleValue &= (1 << bitWidth) - 1;
75  		rleCount = count - 1;
76  		return rleValue;
77  	}
78  
79  	private void readBitPackedGroup() throws IOException {
80  		var totalBits = bitWidth * 8;
81  		var totalBytes = (totalBits + 7) / 8;
82  		var mask = (1 << bitWidth) - 1;
83  		for (int i = 0; i < 8; i++)
84  			bitPackedBuffer[i] = 0;
85  		for (int byteIdx = 0; byteIdx < totalBytes; byteIdx++) {
86  			var b = in.read();
87  			if (b < 0)
88  				break;
89  			for (int bitIdx = 0; bitIdx < 8; bitIdx++) {
90  				var globalBit = byteIdx * 8 + bitIdx;
91  				var valIdx = globalBit / bitWidth;
92  				var valBit = bitWidth - 1 - (globalBit % bitWidth);
93  				if (((b >> bitIdx) & 1) != 0)
94  					bitPackedBuffer[valIdx] |= 1 << valBit;
95  			}
96  		}
97  		for (int i = 0; i < 8; i++)
98  			bitPackedBuffer[i] &= mask;
99  		bitPackedCount = 8;
100 		bitPackedRead = 0;
101 	}
102 
103 	private long readVarint() throws IOException {
104 		var b = in.read();
105 		if (b < 0)
106 			return 0;
107 		return b & 0x7FL;
108 	}
109 
110 	boolean hasNext() throws IOException {
111 		return rleCount > 0 || bitPackedRead < bitPackedCount || in.available() > 0;
112 	}
113 }
