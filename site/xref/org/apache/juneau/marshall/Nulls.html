17  package org.apache.juneau.marshall;
18  
19  import java.util.*;
20  
21  /**
22   * Per-property null-coercion modes consulted by parsers when an explicit {@code null} (or absent
23   * {@link Optional}) is encountered on the wire.
24   *
25   * <p>
26   * Juneau's analog of Jackson's {@code com.fasterxml.jackson.annotation.Nulls}.  The selected mode
27   * decides what the parser does <em>at the moment of setting</em> the bean property — it does not
28   * affect serialization.
29   *
30   * <p>
31   * Precedence:
32   * <ol>
33   * 	<li>Per-property {@link MarshalledProp#nulls() @MarshalledProp(nulls=…)}.
34   * 	<li>Context-level default on {@link org.apache.juneau.marshall.parser.Parser.Builder#nulls(Nulls)}.
35   * 	<li>{@link #LEAVE} when neither is set.
36   * </ol>
37   *
38   * <h5 class='section'>Optional contract:</h5>
39   * <p>
40   * For an {@link Optional}-typed property, {@link #EMPTY} and {@link #DEFAULT} both resolve to
41   * {@link Optional#empty()} — never a bare {@code null} inside the {@link Optional}.  The same rule
42   * applies to {@link OptionalInt}, {@link OptionalLong}, and {@link OptionalDouble}.
43   *
44   * @since 10.0.0
45   */
46  public enum Nulls {
47  
48  	/** Sentinel meaning "no value configured" — falls through to the next-higher precedence level. */
49  	NOT_SET,
50  
51  	/**
52  	 * Pass the {@code null} value through to the setter unchanged.
53  	 *
54  	 * <p>
55  	 * For an {@link Optional}-typed property this resolves to {@link Optional#empty()} per the shared
56  	 * Optional contract — never a bare {@code null} inside the {@link Optional}.
57  	 *
58  	 * <p>
59  	 * This is the default behavior when no policy is configured.
60  	 */
61  	LEAVE,
62  
63  	/**
64  	 * Substitute the type's "empty" value:
65  	 * <ul>
66  	 * 	<li>{@link String} → {@code ""}
67  	 * 	<li>{@link Collection} → empty collection (preserves declared type when possible)
68  	 * 	<li>{@link Map} → empty map
69  	 * 	<li>Array → zero-length array of the declared element type
70  	 * 	<li>{@link Optional} / {@link OptionalInt} / {@link OptionalLong} / {@link OptionalDouble} →
71  	 * 		the corresponding {@code empty()} sentinel
72  	 * 	<li>Primitive — the Java primitive default ({@code 0}, {@code false}, {@code (char)0})
73  	 * 	<li>Other reference types — falls back to {@link #LEAVE}
74  	 * </ul>
75  	 */
76  	EMPTY,
77  
78  	/**
79  	 * Substitute the bean-constructed default for this property — the value the property holds on a
80  	 * fresh no-arg-constructed instance of the bean.
81  	 *
82  	 * <p>
83  	 * The reference instance is built once per bean class (and cached on its {@code ClassMeta}).  When
84  	 * no reference instance can be built (no accessible no-arg ctor / constructor throws), this mode
85  	 * silently falls back to {@link #LEAVE}.
86  	 */
87  	DEFAULT,
88  
89  	/**
90  	 * Do not call the setter at all.
91  	 *
92  	 * <p>
93  	 * Any value already present on the bean (typically the bean's own field-initializer value) is
94  	 * preserved.  This is useful for properties whose absence on the wire should be treated as "no
95  	 * change" rather than "reset to null".
96  	 */
97  	SKIP;
98  }
