331 	private void addNamespace(Namespace ns) {
332 		if (ns == defaultNamespace)
333 			return;
334 
335 		for (var n : namespaces)
336 			if (n == ns)
337 				return;
338 
339 		if (nn(defaultNamespace) && (ns.uri.equals(defaultNamespace.uri) || ns.name.equals(defaultNamespace.name)))
340 			defaultNamespace = ns;
341 		else
342 			namespaces = addAll(namespaces, ns);
343 	}

346 		if (n == null)
347 			return null;
348 		if (nn(n.name) && nn(n.uri))
349 			return n;
350 		if (n.uri == null) {
351 			for (var n2 : getNamespaces())
352 				if (n2.name.equals(n.name))

356 			for (var n2 : getNamespaces())
357 				if (n2.uri.equals(n.uri))
358 					return n2;
359 		}

366 	private static boolean isTextNode(Object o) {
367 		if (o == null)
368 			return false;
369 		var c = o.getClass();
370 		// Text nodes are strings and primitives (not beans, collections, arrays, or other complex types)
371 		return CharSequence.class.isAssignableFrom(c) || Number.class.isAssignableFrom(c) || Boolean.class.isAssignableFrom(c) || c.isPrimitive();
372 	}

398 	private ContentResult writeBeanMap(XmlWriter<?> out, BeanMap<?> m, Namespace elementNs, boolean isCollapsed, boolean isMixedOrText) throws SerializeException {
399 		boolean hasChildren = false;
400 		var bm = m.getMeta();
401 
402 		var lp = new ArrayList<BeanPropertyValue>();
403 
404 		var checkNull = (Predicate<Object>)(x -> isKeepNullProperties() || nn(x));
405 		m.forEachValue(checkNull, (pMeta, key, value, thrown) -> lp.add(new BeanPropertyValue(pMeta, key, value, thrown)));
406 
407 		var xbm = getXmlBeanMeta(bm);
408 
409 		// @formatter:off
410 		Set<String> attrs = xbm.getAttrPropertyNames();
411 		Set<String> elements = xbm.getElementPropertyNames();
412 		Set<String> collapsedElements = xbm.getCollapsedPropertyNames();
413 		var attrsProperty = xbm.getAttrsPropertyName();
414 		var contentProperty = xbm.getContentPropertyName();
415 		// @formatter:on
416 
417 		XmlFormat cf = null;
418 
419 		Object content = null;
420 		ClassMeta<?> contentType = null;
421 		for (var p : lp) {
422 			var n = p.getName();
423 			if (attrs.contains(n) || attrs.contains("*") || n.equals(attrsProperty)) {
424 				var pMeta = p.getMeta();
425 				if (pMeta.canRead()) {
426 					var kv = getPropertyKeyValueIfNotIgnored(p, pMeta);
427 					if (kv.isEmpty())
428 						continue;
429 					var key = kv.get().getKey();
430 					var value = kv.get().getValue();
431 					var bpXml = getXmlBeanPropertyMeta(pMeta);
432 					var ns = (isEnableNamespaces() && bpXml.getNamespace() != elementNs ? bpXml.getNamespace() : null);
433 
434 					if (pMeta.isUri()) {
435 						out.attrUri(ns, key, value);
436 					} else if (n.equals(attrsProperty)) {
437 						if (value instanceof BeanMap<?> value2) {
438 							value2.forEachValue(x -> true, (pMeta2, key2, value3, thrown2) -> {
439 								if (nn(thrown2))
440 									onBeanGetterException(pMeta, thrown2);
441 								out.attr(ns, key2, value3);
442 							});
443 						} else /* Map */ {
444 							var m2 = (Map)value;
445 							if (nn(m2))
446 								m2.forEach((k, v) -> out.attr(ns, s(k), v));
447 						}
448 					} else {
449 						out.attr(ns, key, value);
450 					}
451 				}
452 			}
453 		}
454 
455 		boolean hasContent = false;
456 		boolean preserveWhitespace = false;
457 		boolean isVoidElement = xbm.getContentFormat() == VOID;
458 
459 		for (var p : lp) {
460 			BeanPropertyMeta pMeta = p.getMeta();
461 			if (pMeta.canRead()) {
462 				ClassMeta<?> cMeta = (ClassMeta<?>) p.getBeanInfo();
463 
464 				var n = p.getName();
465 				if (n.equals(contentProperty)) {
466 					content = p.getValue();
467 					contentType = (ClassMeta<?>) p.getBeanInfo();
468 					hasContent = true;
469 					cf = xbm.getContentFormat();
470 					if (cf.isOneOf(MIXED, MIXED_PWS, TEXT, TEXT_PWS, XMLTEXT))
471 						isMixedOrText = true;
472 					if (cf.isOneOf(MIXED_PWS, TEXT_PWS))
473 						preserveWhitespace = true;
474 					if ((contentType.isCollection() && ((Collection)content).isEmpty()) || (contentType.isArray() && Array.getLength(content) == 0))
475 						hasContent = false;
476 				} else if (elements.contains(n) || collapsedElements.contains(n) || elements.contains("*") || collapsedElements.contains("*")) {
477 					var kv = getPropertyKeyValueIfNotIgnored(p, pMeta);
478 					if (kv.isEmpty())
479 						continue;
480 					var key = kv.get().getKey();
481 					var value = kv.get().getValue();
482 					if (! hasChildren) {
483 						hasChildren = true;
484 						out.appendIf(! isCollapsed, '>').nlIf(! isMixedOrText, indent);
485 					}
486 
487 					var bpXml = getXmlBeanPropertyMeta(pMeta);
488 					writeAnything(out, value, cMeta, key, null, bpXml.getNamespace(), false, bpXml.getXmlFormat(), isMixedOrText, false, pMeta);
489 				}
490 			}
491 		}
492 		if (contentProperty == null && ! hasContent) {
493 			if (hasChildren) {
494 				return CR_ELEMENTS;
495 			} else if (isVoidElement) {
496 				return CR_VOID;
497 			} else {
498 				return CR_EMPTY;
499 			}
500 		}
501 
502 		// Serialize XML content.
503 		if (nn(content)) {
504 			out.w('>').nlIf(! isMixedOrText, indent);
505 			if (contentType == null) {
506 				// No content type specified - skip content serialization
507 			} else if (contentType.isCollection()) {
508 				var c = (Collection)content;
509 				boolean previousWasTextNode = false;
510 				for (var value : c) {
511 					boolean currentIsTextNode = isTextNode(value);
512 					// Insert delimiter between consecutive text nodes
513 					if (previousWasTextNode && currentIsTextNode && nn(textNodeDelimiter) && ! textNodeDelimiter.isEmpty()) {
514 						out.append(textNodeDelimiter);
515 					}
516 					writeAnything(out, value, contentType.getElementType(), null, null, null, false, cf, isMixedOrText, preserveWhitespace, null);
517 					previousWasTextNode = currentIsTextNode;
518 				}
519 			} else if (contentType.isArray()) {
520 				var c = toList(Object[].class, content);
521 				boolean previousWasTextNode = false;
522 				for (var value : c) {
523 					boolean currentIsTextNode = isTextNode(value);
524 					// Insert delimiter between consecutive text nodes
525 					if (previousWasTextNode && currentIsTextNode && nn(textNodeDelimiter) && ! textNodeDelimiter.isEmpty()) {
526 						out.append(textNodeDelimiter);
527 					}
528 					writeAnything(out, value, contentType.getElementType(), null, null, null, false, cf, isMixedOrText, preserveWhitespace, null);
529 					previousWasTextNode = currentIsTextNode;
530 				}
531 			} else {
532 				writeAnything(out, content, contentType, null, null, null, false, cf, isMixedOrText, preserveWhitespace, null);
533 			}
534 		} else {
535 			if (isAddJsonTags())
536 				out.attr("nil", "true");
537 			out.w('>').nlIf(! isMixedOrText, indent);
538 		}
539 		return isMixedOrText ? CR_MIXED : CR_ELEMENTS;
540 	}

601 		});

606 	private ContentResult writeMap(XmlWriter<?> out, Map m, ClassMeta<?> sType, ClassMeta<?> eKeyType, ClassMeta<?> eValueType, boolean isMixed) throws SerializeException {
607 
608 		var keyType = eKeyType == null ? sType.getKeyType() : eKeyType;
609 		var valueType = eValueType == null ? sType.getValueType() : eValueType;
610 
611 		var hasChildren = Flag.create();
612 		forEachEntry(m, e -> {
613 
614 			var k = e.getKey();
615 			if (k == null) {
616 				k = "\u0000";
617 			} else {
618 				k = generalize(k, keyType);
619 				if (isTrimStrings() && k instanceof String)
620 					k = k.toString().trim();
621 			}

627 		});

665 	/**
666 	 * Recursively searches for the XML namespaces on the specified POJO and adds them to the serializer context object.
667 	 *
668 	 * @param o The POJO to check.
669 	 * @throws SerializeException Thrown if bean recursion occurred.
670 	 */
671 	@SuppressWarnings({
672 		"null", // Null handling verified by context or framework
673 		"java:S3776", // Cognitive complexity acceptable for this specific logic
674 	})
675 	protected final void findNsfMappings(Object o) throws SerializeException {
676 		ClassMeta<?> aType = null;						// The actual type
677 
678 		try {
679 			aType = push(null, o, null);
680 		} catch (MarshallingRecursionException e) {
681 			throw new SerializeException(e);
682 		}
683 
684 		if (nn(aType)) {
685 			var ns = getXmlClassMeta(aType).getNamespace();
686 			if (nn(ns) && nn(ns.uri))
687 				addNamespace(ns);
688 		}
689 
690 		// Handle recursion
691 		if (nn(aType) && ! aType.isPrimitive()) {
692 
693 			BeanMap<?> bm = null;
694 			if (aType.isBeanMap()) {
695 				bm = (BeanMap<?>)o;
696 			} else if (aType.isBean()) {
697 				bm = toBeanMap(o);
698 			} else if (aType.isDelegate()) {
699 				var innerType = (ClassMeta<?>)((Delegate<?>)o).getBeanInfo();
700 				var ns = Holder.of(getXmlClassMeta(innerType).getNamespace());
701 				if (ns.isPresent()) {
702 					if (nn(ns.get().uri))
703 						addNamespace(ns.get());
704 					else
705 						ns.getAndUnset();
706 				}
707 
708 				if (innerType.isBean()) {
709 					innerType.getBeanMeta().getProperties().values().stream().filter(BeanPropertyMeta::canRead).forEach(x -> {
710 						ns.set(getXmlBeanPropertyMeta(x).getNamespace());
711 						if (ns.isPresent() && nn(ns.get().uri))
712 							addNamespace(ns.get());
713 					});
714 				} else if (innerType.isMap()) {
715 					((Map<?,?>)o).forEach((k, v) -> findNsfMappings(v));
716 				} else if (innerType.isCollection()) {
717 					((Collection<?>)o).forEach(this::findNsfMappings);
718 				}
719 
720 			} else if (aType.isMap()) {
721 				((Map<?,?>)o).forEach((k, v) -> findNsfMappings(v));
722 			} else if (aType.isCollection()) {
723 				((Collection<?>)o).forEach(this::findNsfMappings);
724 			} else if (aType.isArray() && ! aType.getElementType().isPrimitive()) {
725 				for (var o2 : ((Object[])o))
726 					findNsfMappings(o2);
727 			}
728 			if (nn(bm)) {
729 				var checkNull = (Predicate<Object>)(x -> isKeepNullProperties() || nn(x));
730 				bm.forEachValue(checkNull, (pMeta, key, value, thrown) -> {
731 					var ns = getXmlBeanPropertyMeta(pMeta).getNamespace();
732 					if (nn(ns) && nn(ns.uri))
733 						addNamespace(ns);
734 
735 					try {
736 						findNsfMappings(value);
737 					} catch (@SuppressWarnings("unused") Exception x) {
738 						// Ignore
739 					}
740 				});
741 			}
742 		}
743 
744 		pop();
745 	}

817 	/**
818 	 * Workhorse method.
819 	 *
820 	 * @param out The writer to send the output to.
821 	 * 	<br>Must not be <jk>null</jk>.
822 	 * @param o The object to serialize.
823 	 * 	<br>Can be <jk>null</jk> (written as the XML null representation, e.g. <c>type='null'</c>).
824 	 * @param eType The expected type if this is a bean property value being serialized.
825 	 * 	<br>Can be <jk>null</jk> (treated as the generic {@link Object} type).
826 	 * @param keyName The property name or map key name.
827 	 * 	<br>Can be <jk>null</jk> (the element is written without a separate property/key name).
828 	 * @param elementName The root element name.
829 	 * 	<br>Can be <jk>null</jk> (falls back to the bean dictionary name, then <c>keyName</c>, then the JSON type name).
830 	 * @param elementNamespace The namespace of the element.
831 	 * 	<br>Can be <jk>null</jk> (resolved from the serialized type's, then the actual type's, then the session's default namespace).
832 	 * @param addNamespaceUris Flag indicating that namespace URIs need to be added.
833 	 * @param format The format to serialize the output to.
834 	 * 	<br>Must not be <jk>null</jk>.
835 	 * @param isMixedOrText We're serializing mixed content, so don't use whitespace.
836 	 * @param preserveWhitespace
837 	 * 	<jk>true</jk> if we're serializing {@link XmlFormat#MIXED_PWS} or {@link XmlFormat#TEXT_PWS}.
838 	 * @param pMeta The bean property metadata if this is a bean property being serialized.
839 	 * 	<br>Can be <jk>null</jk> (no bean-property-level URI/child-name/namespace metadata is applied).
840 	 * @return The same writer passed in so that calls to the writer can be chained.
841 	 * @throws SerializeException General serialization error occurred.
842 	 */
843 	@SuppressWarnings({
844 		"null", // Null handling verified by context or framework
845 		"java:S3776", // Cognitive complexity acceptable for this specific logic
846 		"java:S6541", // Single-threaded session contexts do not require synchronization
847 		"java:S107", // Method has many parameters; acceptable for builder/configuration methods
848 	})
849 	protected ContentResult writeAnything(XmlWriter<?> out, Object o, ClassMeta<?> eType, String keyName, String elementName, Namespace elementNamespace, boolean addNamespaceUris, XmlFormat format,
850 		boolean isMixedOrText, boolean preserveWhitespace, BeanPropertyMeta pMeta) throws SerializeException {
851 
852 		JsonType type = null;              // The type string (e.g. <type> or <x x='type'>
853 		int i = isMixedOrText ? 0 : indent;       // Current indentation
854 		ClassMeta<?> aType = null;     // The actual type
855 		ClassMeta<?> wType = null;     // The wrapped type (delegate)
856 		ClassMeta<?> sType; // The serialized type
857 
858 		aType = push2(keyName, o, eType);
859 
860 		if (eType == null)
861 			eType = object();
862 
863 		// Handle recursion
864 		if (aType == null) {
865 			o = null;
866 			aType = object();
867 		}
868 
869 		// Handle Optional<X>
870 		if (isOptional(aType)) {
871 			o = getOptionalValue(o);
872 			eType = getOptionalType(eType);
873 			aType = getClassMetaForObject(o, object());
874 		}
875 
876 		if (nn(o)) {
877 
878 			if (aType.isDelegate()) {
879 				wType = aType;
880 				eType = aType = (ClassMeta<?>)((Delegate<?>)o).getBeanInfo();
881 			}
882 
883 			sType = aType;
884 
885 			// Swap if necessary
886 			var swap = aType.getSwap(this);
887 			if (nn(swap)) {
888 				o = swap(swap, o);
889 				sType = swap.getSwapClassMeta(this);
890 
891 				// If the getSwapClass() method returns Object, we need to figure out
892 				// the actual type now.
893 				if (sType.isObject())
894 					sType = getClassMetaForObject(o);
895 			}
896 		} else {
897 			sType = eType.getSerializedClassMeta(this);
898 		}
899 
900 		// Does the actual type match the expected type?
901 		boolean isExpectedType = true;
902 		if (o == null || ! eType.same(aType)) {
903 			if (eType.isNumber())
904 				isExpectedType = aType.isNumber();
905 			else if (eType.isMap())
906 				isExpectedType = aType.isMap();
907 			else if (eType.isCollectionOrArray() || eType.isStreamable())
908 				isExpectedType = aType.isCollectionOrArray() || aType.isStreamable();
909 			else
910 				isExpectedType = false;
911 		}
912 
913 		var resolvedDictionaryName = isExpectedType ? null : aType.getBeanDictionaryName();
914 
915 		// Note that the dictionary name may be specified on the actual type or the serialized type.
916 		// HTML templates will have them defined on the serialized type.
917 		var dictionaryName = aType.getBeanDictionaryName();
918 		if (dictionaryName == null)
919 			dictionaryName = sType.getBeanDictionaryName();
920 
921 		// char '\0' is interpreted as null.
922 		if (nn(o) && sType.isChar() && ((Character)o).charValue() == 0)
923 			o = null;
924 
925 		boolean isCollapsed = false;		// If 'true', this is a collection and we're not rendering the outer element.
926 		boolean isRaw = (sType.isReader() || sType.isInputStream()) && nn(o);
927 
928 		// Get the JSON type string.
929 		if (o == null) {
930 			type = NULL;
931 		} else if (sType.isCharSequence() || sType.isChar()) {
932 			type = STRING;
933 		} else if (sType.isNumber()) {
934 			type = NUMBER;
935 		} else if (sType.isBoolean()) {
936 			type = BOOLEAN;
937 		} else if (sType.isMapOrBean()) {
938 			isCollapsed = getXmlClassMeta(sType).getFormat() == COLLAPSED;
939 			type = OBJECT;
940 		} else if (sType.isCollectionOrArray() || sType.isStreamable()) {
941 			isCollapsed = (format == COLLAPSED && ! addNamespaceUris);
942 			type = ARRAY;
943 		} else {
944 			type = STRING;
945 		}
946 
947 		if (format.isOneOf(MIXED, MIXED_PWS, TEXT, TEXT_PWS, XMLTEXT) && type.isOneOf(NULL, STRING, NUMBER, BOOLEAN))
948 			isCollapsed = true;
949 
950 		// Is there a name associated with this bean?
951 
952 		var name = keyName;
953 		if (elementName == null && nn(dictionaryName)) {
954 			elementName = dictionaryName;
955 			isExpectedType = nn(o);  // preserve type='null' when it's null.
956 		}
957 
958 		if (elementName == null) {
959 			elementName = name;
960 			name = null;
961 		}
962 
963 		if (eq(name, elementName))
964 			name = null;
965 
966 		if (isEnableNamespaces()) {
967 			if (elementNamespace == null)
968 				elementNamespace = getXmlClassMeta(sType).getNamespace();
969 			if (elementNamespace == null)
970 				elementNamespace = getXmlClassMeta(aType).getNamespace();
971 			if (nn(elementNamespace) && elementNamespace.uri == null)
972 				elementNamespace = null;
973 			if (elementNamespace == null)
974 				elementNamespace = defaultNamespace;
975 		} else {
976 			elementNamespace = null;
977 		}
978 
979 		// Do we need a carriage return after the start tag?
980 		boolean cr = nn(o) && (sType.isMapOrBean() || sType.isCollectionOrArray()) && ! isMixedOrText;
981 
982 		var en = elementName;
983 		if (en == null && ! isRaw && isAddJsonTags()) {
984 			en = type.toString();
985 			type = null;
986 		}
987 
988 		boolean encodeEn = nn(elementName);
989 		var ns = (elementNamespace == null ? null : elementNamespace.name);
990 		String dns = null;
991 		String elementNs = null;
992 		if (isEnableNamespaces()) {
993 			dns = elementName == null && nn(defaultNamespace) ? defaultNamespace.name : null;
994 			elementNs = elementName == null ? dns : ns;
995 			if (elementName == null)
996 				elementNamespace = null;
997 		}
998 
999 		// Render the start tag.
1000 		if (! isCollapsed) {
1001 			if (nn(en)) {
1002 				out.oTag(i, elementNs, en, encodeEn);
1003 				if (addNamespaceUris) {
1004 					out.attr((String)null, "xmlns", defaultNamespace.getUri());
1005 
1006 					for (var n : namespaces)
1007 						out.attr("xmlns", n.getName(), n.getUri());
1008 				}
1009 				if (! isExpectedType) {
1010 					if (nn(resolvedDictionaryName))
1011 						out.attr(dns, getBeanTypePropertyName(eType), resolvedDictionaryName);
1012 					else if (nn(type) && type != STRING)
1013 						out.attr(dns, getBeanTypePropertyName(eType), type);
1014 				}
1015 				if (nn(name))
1016 					out.attr(MarshallingSession.NAME_PROPERTY_NAME, name);
1017 			} else {
1018 				out.i(i);
1019 			}
1020 			if (o == null && (sType.isBoolean() || sType.isNumber()) && ! sType.isNullable())
1021 				o = sType.getPrimitiveDefault();
1022 
1023 			if (nn(o) && ! (sType.isMapOrBean() || en == null))
1024 				out.w('>');
1025 
1026 			if (cr && ! (sType.isMapOrBean()))
1027 				out.nl(i + 1);
1028 		}
1029 
1030 		var rc = CR_ELEMENTS;
1031 
1032 		// Render the tag contents.
1033 		if (nn(o)) {
1034 			if (sType.isBean()) {
1035 				rc = writeBeanMap(out, toBeanMap(o), elementNamespace, isCollapsed, isMixedOrText);
1036 			} else if (sType.isMap() || (nn(wType) && wType.isMap())) {
1037 				if (o instanceof BeanMap o2)
1038 					rc = writeBeanMap(out, o2, elementNamespace, isCollapsed, isMixedOrText);
1039 				else
1040 					rc = writeMap(out, (Map)o, sType, eType.getKeyType(), eType.getValueType(), isMixedOrText);
1041 			} else if (sType.isCollection() || sType.isArray() || (nn(wType) && (wType.isCollection() || wType.isArray()))) {
1042 				if (isCollapsed)
1043 					indent--;
1044 				writeCollection(out, o, sType, eType, pMeta, isMixedOrText);
1045 				if (isCollapsed)
1046 					indent++;
1047 			} else if (sType.isUri() || (nn(pMeta) && pMeta.isUri())) {
1048 				// Must come before isCharSequence/isChar: @Uri-annotated String properties (where sType is String but pMeta.isUri() is true) need to route here for URI text emission.
1049 				out.textUri(o);
1050 			} else if (sType.isCharSequence() || sType.isChar()) {
1051 				if (isXmlText(format, sType))
1052 					out.append(o);
1053 				else
1054 					out.text(o, preserveWhitespace);
1055 			} else if (sType.isNumber() || sType.isBoolean()) {
1056 				out.append(o);
1057 			} else if (sType.isDate()) {
1058 				out.text(writeDate((Date)o, sType));
1059 			} else if (sType.isCalendar()) {
1060 				out.text(writeCalendar(o, sType));
1061 			} else if (sType.isTemporal()) {
1062 				out.text(writeTemporal((TemporalAccessor)o, sType));
1063 			} else if (sType.isDuration()) {
1064 				out.text(writeDuration((Duration)o));
1065 			} else if (sType.isPeriod()) {
1066 				out.text(writePeriod((Period)o));
1067 			} else if (sType.isStreamable()) {
1068 				if (isCollapsed)
1069 					indent--;
1070 				writeStreamable(out, o, sType, eType, pMeta, isMixedOrText);
1071 				if (isCollapsed)
1072 					indent++;
1073 			} else if (sType.isReader()) {
1074 				pipe((Reader)o, out, SerializerSession::handleThrown);
1075 			} else if (sType.isInputStream()) {
1076 				pipe((InputStream)o, out, SerializerSession::handleThrown);
1077 			} else {
1078 				if (isXmlText(format, sType))
1079 					out.append(toString(o));
1080 				else
1081 					out.text(toString(o));
1082 			}
1083 		}
1084 
1085 		pop();
1086 
1087 		// Render the end tag.
1088 		if (! isCollapsed) {
1089 			if (nn(en)) {
1090 				if (rc == CR_EMPTY) {
1091 					if (isHtmlMode())
1092 						out.w('>').eTag(elementNs, en, encodeEn);
1093 					else
1094 						out.w('/').w('>');
1095 				} else if (rc == CR_VOID || o == null) {
1096 					out.w('/').w('>');
1097 				} else
1098 					out.ie(cr && rc != CR_MIXED ? i : 0).eTag(elementNs, en, encodeEn);
1099 			}
1100 			if (! isMixedOrText)
1101 				out.nl(i);
1102 		}
1103 
1104 		return rc;
1105 	}
