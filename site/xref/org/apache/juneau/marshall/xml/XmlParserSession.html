167 	private static int getJsonType(String s) {
168 		if (s == null)
169 			return UNKNOWN;
170 		var c = s.charAt(0);
171 		return switch (c) {
172 			case 'o' -> (s.equals("object") ? OBJECT : UNKNOWN);
173 			case 'a' -> (s.equals("array") ? ARRAY : UNKNOWN);
174 			case 's' -> (s.equals("string") ? STRING : UNKNOWN);
175 			case 'b' -> (s.equals("boolean") ? BOOLEAN : UNKNOWN);
176 			case 'n' -> {
177 				c = s.charAt(2);
178 				yield switch (c) {
179 					case 'm' -> (s.equals("number") ? NUMBER : UNKNOWN);
180 					case 'l' -> (s.equals("null") ? NULL : UNKNOWN);
181 					default -> NUMBER;
182 				};
183 			}
184 			default -> UNKNOWN;
185 		};
186 	}

223 	 * Takes the element being read from the XML stream reader and reconstructs it as XML.
224 	 * Used when reconstructing bean properties of type {@link XmlFormat#XMLTEXT}.
225 	 */
226 	private String getElementAsString(XmlReader r) {
227 		int t = r.getEventType();
228 		if (t > 2)
229 			throw rex("Invalid event type on stream reader for elementToString() method: '%s'", XmlUtils.toReadableEvent(r));
230 		rsb.setLength(0);
231 		rsb.append("<").append(t == 1 ? "" : "/").append(r.getLocalName());
232 		if (t == 1)
233 			for (var i = 0; i < r.getAttributeCount(); i++)
234 				rsb.append(' ').append(r.getAttributeName(i)).append('=').append('\'').append(r.getAttributeValue(i)).append('\'');
235 		rsb.append('>');
236 		return rsb.toString();
237 	}

281 	private Object getUnknown(XmlReader r) throws IOException, ParseException, ExecutableException, XMLStreamException {
282 		if (r.getEventType() != START_ELEMENT) {
283 			throw new ParseException(this, "Parser must be on START_ELEMENT to read next text.");
284 		}
285 		MarshalledMap m = null;
286 
287 		// If this element has attributes, then it's always a MarshalledMap.
288 		if (r.getAttributeCount() > 0) {
289 			m = newGenericMap();
290 			for (var i = 0; i < r.getAttributeCount(); i++) {
291 				var key = getAttributeName(r, i);
292 				var val = r.getAttributeValue(i);
293 				if (! isSpecialAttr(key))
294 					m.put(key, val);
295 			}
296 		}
297 		int eventType = r.next();
298 		var sb = getStringBuilder();
299 		while (eventType != END_ELEMENT) {
300 			if (eventType == CHARACTERS || eventType == CDATA || eventType == XMLStreamConstants.SPACE || eventType == ENTITY_REFERENCE) {
301 				sb.append(r.getText());
302 			} else if (eventType == PROCESSING_INSTRUCTION || eventType == COMMENT) {
303 				// skipping
304 			} else if (eventType == END_DOCUMENT) {
305 				throw new ParseException(this, "Unexpected end of document when reading element text content");
306 			} else if (eventType == START_ELEMENT) {
307 				// Oops...this has an element in it.
308 				// Parse it as a map.
309 				if (m == null)
310 					m = newGenericMap();
311 				int depth = 0;
312 				do {
313 					int event = (eventType == -1 ? r.nextTag() : eventType);
314 					String currAttr;
315 					if (event == START_ELEMENT) {
316 						depth++;
317 						currAttr = getNameProperty(r);
318 						if (currAttr == null)
319 							currAttr = getElementName(r);
320 						String key = convertAttrToType(null, currAttr, string());
321 						var value = readAnything(object(), currAttr, r, null, false, null);
322 						if (m.containsKey(key)) {
323 							var o = m.get(key);
324 							if (o instanceof MarshalledList o2)
325 								o2.add(value);
326 							else
327 								m.put(key, new JsonList(o, value).setBeanSession(this));
328 						} else {
329 							m.put(key, value);
330 						}
331 
332 					} else if (event == END_ELEMENT) {
333 						break;
334 					}
335 					eventType = -1;
336 				} while (depth > 0);
337 				break;
338 			} else {
339 				throw new ParseException(this, "Unexpected event type '%s'", eventType);
340 			}
341 			eventType = r.next();
342 		}
343 		var s = sb.toString().trim();
344 		returnStringBuilder(sb);
345 		s = decodeString(s);
346 		if (nn(m)) {
347 			if (! s.isEmpty())
348 				m.put("contents", s);
349 			return m;
350 		}
351 		return s;
352 	}

363 	private <T> BeanMap<T> readIntoBean(XmlReader r, BeanMap<T> m, boolean isNil) throws IOException, ParseException, ExecutableException, XMLStreamException {
364 		var bMeta = m.getMeta();
365 		var xmlMeta = getXmlBeanMeta(bMeta);
366 
367 		for (var i = 0; i < r.getAttributeCount(); i++) {
368 			String key = getAttributeName(r, i);
369 			if (! ("nil".equals(key) || isSpecialAttr(key))) {
370 				var val = r.getAttributeValue(i);
371 				var ns = r.getAttributeNamespace(i);
372 				var bpm = xmlMeta.getPropertyMeta(key);
373 				if (bpm == null) {
374 					if (nn(xmlMeta.getAttrsProperty())) {
375 						xmlMeta.getAttrsProperty().add(m, key, key, val);
376 					} else if (ns == null) {
377 						onUnknownProperty(key, m, val);
378 					}
379 				} else {
380 					try {
381 						bpm.set(m, key, val);
382 					} catch (BeanRuntimeException e) {
383 						onBeanSetterException(bpm, e);
384 						throw e;
385 					}
386 				}
387 			}
388 		}
389 
390 		var cp = xmlMeta.getContentProperty();
391 		var cpf = xmlMeta.getContentFormat();
392 		var trim = cp == null || ! cpf.isOneOf(MIXED_PWS, TEXT_PWS);
393 		ClassMeta<?> cpcm = (cp == null ? object() : (ClassMeta<?>) cp.getBeanInfo());
394 		StringBuilder sb = null;
395 		var breg = cp == null ? null : cp.getBeanRegistry();
396 		List<Object> l = null;
397 
398 		int depth = 0;
399 		do {
400 			var event = r.next();
401 			String currAttr;
402 			// We only care about text in MIXED mode.
403 			// Ignore if in ELEMENTS mode.
404 			if (event == CHARACTERS) {
405 				if (nn(cp) && cpf.isOneOf(MIXED, MIXED_PWS)) {
406 					if (cpcm.isCollectionOrArray()) {
407 						if (l == null)
408 							l = ll();
409 						l.add(getText(r, false));
410 					} else {
411 						cp.set(m, null, getText(r, trim));
412 					}
413 				} else if (cpf != ELEMENTS) {
414 					var s = getText(r, trim);
415 					if (nn(s)) {
416 						if (sb == null)
417 							sb = getStringBuilder();
418 						sb.append(s);
419 					}
420 				} else {
421 					// Do nothing...we're in ELEMENTS mode.
422 				}
423 			} else if (event == START_ELEMENT) {
424 				if (nn(cp) && cpf.isOneOf(TEXT, TEXT_PWS)) {
425 					var s = readText(r);
426 					if (nn(s)) {
427 						if (sb == null)
428 							sb = getStringBuilder();
429 						sb.append(s);
430 					}
431 					depth--;
432 				} else if (cpf == XMLTEXT) {
433 					if (sb == null)
434 						sb = getStringBuilder();
435 					sb.append(getElementAsString(r));
436 					depth++;
437 				} else if (nn(cp) && cpf.isOneOf(MIXED, MIXED_PWS)) {
438 					if (isWhitespaceElement(r) && (breg == null || ! breg.hasName(r.getLocalName()))) {
439 						if (cpcm.isCollectionOrArray()) {
440 							if (l == null)
441 								l = ll();
442 							l.add(readWhitespaceElement(r));
443 						} else {
444 							cp.set(m, null, readWhitespaceElement(r));
445 						}
446 					} else {
447 						if (cpcm.isCollectionOrArray()) {
448 							if (l == null)
449 								l = ll();
450 							l.add(readAnything(cpcm.getElementType(), cp.getName(), r, m.getBean(false), false, cp));
451 						} else {
452 							cp.set(m, null, readAnything(cpcm, cp.getName(), r, m.getBean(false), false, cp));
453 						}
454 					}
455 				} else if (nn(cp) && cpf == ELEMENTS) {
456 					cp.add(m, null, readAnything(cpcm.getElementType(), cp.getName(), r, m.getBean(false), false, cp));
457 				} else {
458 					currAttr = getNameProperty(r);
459 					if (currAttr == null)
460 						currAttr = getElementName(r);
461 					var pMeta = xmlMeta.getPropertyMeta(currAttr);
462 					if (pMeta == null) {
463 						var value = readAnything(object(), currAttr, r, m.getBean(false), false, null);
464 						onUnknownProperty(currAttr, m, value);
465 					} else {
466 						setCurrentProperty(pMeta);
467 						var xf = getXmlBeanPropertyMeta(pMeta).getXmlFormat();
468 						if (xf == COLLAPSED) {
469 							var et = ((ClassMeta<?>) pMeta.getBeanInfo()).getElementType();
470 							var value = readAnything(et, currAttr, r, m.getBean(false), false, pMeta);
471 							setName(et, value, currAttr);
472 							pMeta.add(m, currAttr, value);
473 						} else if (xf == ATTR) {
474 							pMeta.set(m, currAttr, getAttributeValue(r, 0));
475 							r.nextTag();
476 						} else {
477 							var cm = (ClassMeta<?>) pMeta.getBeanInfo();
478 							var value = readAnything(cm, currAttr, r, m.getBean(false), false, pMeta);
479 							setName(cm, value, currAttr);
480 							pMeta.set(m, currAttr, value);
481 						}
482 						setCurrentProperty(null);
483 					}
484 				}
485 			} else if (event == END_ELEMENT) {
486 				if (depth > 0) {
487 					if (cpf == XMLTEXT) {
488 						if (sb == null)
489 							sb = getStringBuilder();
490 						sb.append(getElementAsString(r));
491 					} else
492 						throw new ParseException("End element found where one was not expected.  %s", XmlUtils.toReadableEvent(r));
493 				}
494 				depth--;
495 			} else if (event == COMMENT) {
496 				// Ignore comments.
497 			} else {
498 				throw new ParseException("Unexpected event type: %s", XmlUtils.toReadableEvent(r));
499 			}
500 		} while (depth >= 0);
501 
502 		if (nn(cp) && ! isNil) {
503 			if (nn(sb))
504 				cp.set(m, null, sb.toString());
505 			else if (nn(l))
506 				cp.set(m, null, XmlUtils.collapseTextNodes(l));
507 			else if (cpcm.isCollectionOrArray()) {
508 				var o = cp.get(m, null);
509 				if (o == null)
510 					cp.set(m, cp.getName(), list());
511 			}
512 		}
513 
514 		returnStringBuilder(sb);
515 		return m;
516 	}

518 	private <E> Collection<E> readIntoCollection(XmlReader r, Collection<E> l, ClassMeta<?> type, BeanPropertyMeta pMeta) throws IOException, ParseException, ExecutableException, XMLStreamException {
519 		int depth = 0;
520 		int argIndex = 0;
521 		do {
522 			var event = r.nextTag();
523 			if (event == START_ELEMENT) {
524 				depth++;
525 				ClassMeta<?> elementType;
526 				if (type == null) {
527 					elementType = object();
528 				} else if (type.isArgs()) {
529 					elementType = type.getArg(argIndex++);
530 				} else {
531 					elementType = type.getElementType();
532 				}
533 				E value = (E)readAnything(elementType, null, r, l, false, pMeta);
534 				l.add(value);
535 			} else if (event == END_ELEMENT) {
536 				return l;
537 			}
538 		} while (depth > 0);
539 		return l;
540 	}

543 		"java:S3776" // Cognitive complexity acceptable for parser state machine
544 	})
545 	private <K,V> Map<K,V> readIntoMap(XmlReader r, Map<K,V> m, ClassMeta<K> keyType, ClassMeta<V> valueType, BeanPropertyMeta pMeta)
546 		throws IOException, ParseException, ExecutableException, XMLStreamException {
547 		int depth = 0;
548 		for (var i = 0; i < r.getAttributeCount(); i++) {
549 			var a = r.getAttributeLocalName(i);
550 			if (! isSpecialAttr(a)) {
551 				K key = trim(convertAttrToType(m, a, keyType));
552 				V value = trim(convertAttrToType(m, r.getAttributeValue(i), valueType));
553 				setName(valueType, value, key);
554 				m.put(key, value);
555 			}
556 		}
557 		do {
558 			var event = r.nextTag();
559 			String currAttr;
560 			if (event == START_ELEMENT) {
561 				depth++;
562 				currAttr = getNameProperty(r);
563 				if (currAttr == null)
564 					currAttr = getElementName(r);
565 				K key = convertAttrToType(m, currAttr, keyType);
566 				V value = readAnything(valueType, currAttr, r, m, false, pMeta);
567 				setName(valueType, value, currAttr);
568 				if (valueType.isObject() && m.containsKey(key)) {
569 					var o = m.get(key);
570 					if (o instanceof List o2)
571 						o2.add(value);
572 					else
573 						m.put(key, (V)new JsonList(o, value).setBeanSession(this));
574 				} else {
575 					m.put(key, value);
576 				}
577 			} else if (event == END_ELEMENT) {
578 				return m;
579 			}
580 		} while (depth > 0);
581 		return m;
582 	}

658 	/**
659 	 * Buffered array-element {@link RecordReader} that honors the caller-specified root element
660 	 * name.
661 	 *
662 	 * <p>
663 	 * The document is parsed once into a generic structure and the records to yield are selected by
664 	 * the supplied {@code rootElementName}:
665 	 * <ul>
666 	 * 	<li>If the document root is a <b>wrapper element</b> (parses to a {@link Map}) that contains a
667 	 * 		child named {@code rootElementName}, that child's value supplies the records (a repeated
668 	 * 		child element parses to a {@link List}; a single occurrence yields one record).  Sibling
669 	 * 		elements with other names are ignored.
670 	 * 	<li>If the document root is an <b>array wrapper</b> (parses to a {@link List}, e.g. Juneau's
671 	 * 		{@code <array>} envelope), its already-unwrapped elements are the records and the hint is a
672 	 * 		no-op.
673 	 * 	<li>A scalar root yields a single record.
674 	 * </ul>
675 	 *
676 	 * <p>
677 	 * A <jk>null</jk> {@code rootElementName} is equivalent to {@link #readArrayRecords(Object)}.
678 	 * The cursor is buffered ({@link RecordReader#isStreaming()} == <jk>false</jk>); see
679 	 * {@code 175ab} Item 3 for why true element-at-a-time XML streaming is left for a
680 	 * demand-driven StAX cursor.
681 	 *
682 	 * @param input The input.
683 	 * @param rootElementName The name of the wrapping/child element whose occurrences are the
684 	 * 	records.  <jk>null</jk> defers to {@link #readArrayRecords(Object)}.
685 	 * @return A buffered {@link RecordReader}.
686 	 * @throws IOException If a problem occurred reading the input.
687 	 */
688 	@Override /* ArrayRecordReadable */
689 	public RecordReader readArrayRecords(Object input, String rootElementName) throws IOException {
690 		if (rootElementName == null)
691 			return readArrayRecords(input);
692 		Object parsed;
693 		try {
694 			parsed = read(input, Object.class);
695 		} catch (ParseException e) {
696 			throw new IOException(e);
697 		}
698 		var records = selectRootElements(parsed, rootElementName);
699 		var iter = records.iterator();
700 		return new RecordReader() {
701 			@Override public boolean canRead() { return iter.hasNext(); }
702 			@Override public <T> T read(Class<T> type) { return convertToType(nextElement(), type); }
703 			@Override public <T> T read(ClassMeta<T> type) { return convertToType(nextElement(), type); }
704 			@Override public <T> T read(Type type, Type... args) { return XmlParserSession.this.<T>convertToType(nextElement(), type, args); }
705 			@Override public boolean isStreaming() { return false; }
706 			@Override public void close() throws IOException {
707 				if (input instanceof Closeable input2)
708 					input2.close();
709 			}
710 			private Object nextElement() {
711 				if (! iter.hasNext())
712 					throw new IllegalStateException("Array stream is exhausted.");
713 				return iter.next();
714 			}
715 		};
716 	}

718 	/** Selects the records to yield from a generically-parsed XML document, honoring the root-element hint. */
719 	private static List<Object> selectRootElements(Object parsed, String rootElementName) {
720 		if (parsed == null)
721 			return Collections.emptyList();
722 		if (parsed instanceof Map<?,?> parsed2) {
723 			var v = parsed2.get(rootElementName);
724 			if (v == null)
725 				return Collections.emptyList();
726 			if (v instanceof List<?> v2)
727 				return new ArrayList<>(v2);
728 			return new ArrayList<>(Collections.singletonList(v));
729 		}
730 		if (parsed instanceof List<?> parsed2)
731 			return new ArrayList<>(parsed2);
732 		return new ArrayList<>(Collections.singletonList(parsed));
733 	}

877 	 * @param <T> The expected type of object.
878 	 * @param eType The expected type of object.
879 	 * 	<br>Can be <jk>null</jk>, in which case the generic {@link Object} type is used and the actual type is auto-detected from the XML.
880 	 * @param currAttr The current bean property name.

886 	 * @param isRoot If <jk>true</jk>, then we're serializing a root element in the document.

889 	 * @return The parsed object, or <jk>null</jk> if the element represents a <jk>null</jk> value.
890 	 * @throws IOException Thrown by underlying stream.
891 	 * @throws ParseException Malformed input encountered.
892 	 * @throws ExecutableException Exception occurred on invoked constructor/method/field.
893 	 * @throws XMLStreamException Malformed XML encountered.
894 	 */
895 	@SuppressWarnings({
896 		"null",        // Null handling verified by context or framework
897 		"java:S3776",  // Cognitive complexity acceptable for this specific logic
898 		"java:S6541",  // Brain Method — complex XML parsing logic; structural refactoring would reduce readability without meaningful benefit.
899 	})
900 	protected <T> T readAnything(ClassMeta<T> eType, String currAttr, XmlReader r, Object outer, boolean isRoot, BeanPropertyMeta pMeta)
901 		throws IOException, ParseException, ExecutableException, XMLStreamException {
902 
903 		if (eType == null)
904 			eType = (ClassMeta<T>)object();
905 		var swap = (ObjectSwap<T,Object>)eType.getSwap(this);
906 		var builder = (BuilderSwap<T,Object>)eType.getBuilderSwap(this);
907 		ClassMeta<?> sType = null;
908 		if (nn(builder))
909 			sType = builder.getBuilderClassMeta(this);
910 		else if (nn(swap))
911 			sType = swap.getSwapClassMeta(this);
912 		else
913 			sType = eType;
914 
915 		if (sType.isOptional())
916 			return (T)o(readAnything(eType.getElementType(), currAttr, r, outer, isRoot, pMeta));
917 
918 		setCurrentClass(sType);
919 
920 		var wrapperAttr = (isRoot && isPreserveRootElement()) ? r.getName().getLocalPart() : null;
921 		var typeAttr = r.getAttributeValue(null, getBeanTypePropertyName(eType));
922 		var isNil = "true".equals(r.getAttributeValue(null, "nil"));
923 		var jsonType = getJsonType(typeAttr);
924 		var elementName = getElementName(r);
925 		if (jsonType == 0) {
926 			if (elementName == null || elementName.equals(currAttr))
927 				jsonType = UNKNOWN;
928 			else {
929 				typeAttr = elementName;
930 				jsonType = getJsonType(elementName);
931 			}
932 		}
933 
934 		ClassMeta tcm = getClassMeta(typeAttr, pMeta, eType);
935 		if (tcm == null && nn(elementName) && ! elementName.equals(currAttr))
936 			tcm = getClassMeta(elementName, pMeta, eType);
937 		if (nn(tcm))
938 			sType = eType = tcm;
939 
940 		Object o = null;
941 
942 		if (jsonType == NULL) {
943 			r.nextTag();	// Discard end tag
944 			return null;
945 		}
946 
947 		if (sType.isObject()) {
948 			if (jsonType == OBJECT) {
949 				var m = newGenericMap();
950 				readIntoMap(r, m, string(), object(), pMeta);
951 				if (nn(wrapperAttr))
952 					m = newGenericMap().append(wrapperAttr, m);
953 				o = cast(m, pMeta, eType);
954 			} else if (jsonType == ARRAY)

956 			else if (jsonType == STRING) {
957 				o = getElementText(r);
958 				if (sType.isChar())

960 			} else if (jsonType == NUMBER)
961 				o = parseNumber(getElementText(r), null);
962 			else if (jsonType == BOOLEAN)
963 				o = Boolean.parseBoolean(getElementText(r));
964 			else if (jsonType == UNKNOWN)
965 				o = getUnknown(r);
966 		} else if (nn(builder) || sType.canCreateNewBean(outer)) {
967 			if (getXmlClassMeta(sType).getFormat() == COLLAPSED) {
968 				var fieldName = r.getLocalName();
969 				var m = nn(builder) ? toBeanMap(builder.create(this, eType)) : newBeanMap(outer, sType.inner());
970 				var bpm = getXmlBeanMeta(m.getMeta()).getPropertyMeta(fieldName);
971 				var cm = (ClassMeta<?>) m.getMeta().getBeanInfo();
972 				var pb = swapParentBean(m.getBean(false));
973 				try {
974 					Object value = readAnything(cm, currAttr, r, m.getBean(false), false, null);
975 					setName(cm, value, currAttr);

979 				}

981 			} else {

983 				var pb = swapParentBean(m.getBean(false));
984 				try {
985 					m = readIntoBean(r, m, isNil);
986 				} finally {

988 				}

1062 		while (true) {
1063 			var et = r.getEventType();

1066 				depth++;

1071 				depth--;

1073 					break;
1074 			}
1075 			r.next();
1076 		}

1078 		returnStringBuilder(sb2);
