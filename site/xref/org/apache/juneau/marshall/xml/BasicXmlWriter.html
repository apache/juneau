17  package org.apache.juneau.marshall.xml;
18  
19  import java.io.*;
20  
21  import org.apache.juneau.marshall.*;
22  
23  /**
24   * Concrete {@link XmlWriter} leaf for standard XML output.
25   *
26   * <p>
27   * This is the directly-instantiable form of {@link XmlWriter}, which is itself an abstract CRTP self-type base so it
28   * can also be extended (e.g. by {@link org.apache.juneau.marshall.html.HtmlWriter}).
29   *
30   * <h5 class='section'>Notes:</h5><ul>
31   * 	<li class='note'>
32   * 		This class is not intended for external use.
33   * </ul>
34   */
35  @SuppressWarnings({
36  	"resource" // Writer resource managed by calling code
37  })
38  public final class BasicXmlWriter extends XmlWriter<BasicXmlWriter> {
39  
40  	/**
41  	 * Constructor.
42  	 *
43  	 * @param out The wrapped writer.
44  	 * @param useWhitespace If <jk>true</jk> XML elements will be indented.
45  	 * @param maxIndent The maximum indentation level.
46  	 * @param trimStrings If <jk>true</jk>, strings should be trimmed before they're serialized.
47  	 * @param quoteChar The quote character to use for attributes.  Should be <js>'\''</js> or <js>'"'</js>.
48  	 * @param uriResolver The URI resolver for resolving URIs to absolute or root-relative form.
49  	 * @param enableNs Flag to indicate if XML namespaces are enabled.
50  	 * @param defaultNamespace The default namespace if XML namespaces are enabled.
51  	 */
52  	@SuppressWarnings({
53  		"java:S107" // Constructor requires 8 parameters for XML writer configuration
54  	})
55  	public BasicXmlWriter(Writer out, boolean useWhitespace, int maxIndent, boolean trimStrings, char quoteChar, UriResolver uriResolver, boolean enableNs, Namespace defaultNamespace) {
56  		super(out, useWhitespace, maxIndent, trimStrings, quoteChar, uriResolver, enableNs, defaultNamespace);
57  	}
58  }
