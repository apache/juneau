32  /**
33   * Serializes POJOs to plain text using just the <c>toString()</c> method on the serialized object.
34   *
35   * <h5 class='topic'>Media types</h5>
36   *
37   * Handles <c>Accept</c> types:  <bc>text/plain</bc>
38   * <p>
39   * Produces <c>Content-Type</c> types:  <bc>text/plain</bc>
40   *
41   * <h5 class='topic'>Description</h5>
42   *
43   * Essentially converts POJOs to plain text using the <c>toString()</c> method.
44   *
45   * <p>
46   * Also serializes objects using a transform if the object class has an {@link ObjectSwap ObjectSwap&lt;?,String&gt;}
47   * transform defined on it.
48   *
49   * <h5 class='section'>Notes:</h5><ul>
50   * 	<li class='note'>This class is thread safe and reusable.
51   * </ul>
52   *
53   */
54  @SuppressWarnings({
55  	"java:S110", // Inheritance depth acceptable for this class hierarchy
56  	"java:S115", // Constants use UPPER_snakeCase naming convention
57  	"resource" // Closeable resources are owned by the caller's serializer session; Eclipse JDT @Owning warning is by design.
58  })
59  public class PlainTextSerializer extends WriterSerializer implements PlainTextMetaProvider, RecordWritable {
60  
61  	// Argument name constants for assertArgNotNull
62  	private static final String ARG_copyFrom = "copyFrom";
63  
64  	/**
65  	 * Builder class.
66  	 */
67  	public static class Builder extends WriterSerializer.Builder<Builder> {
68  
69  		private static final Cache<HashKey,PlainTextSerializer> CACHE = Cache.of(HashKey.class, PlainTextSerializer.class).build();
70  
71  		/**
72  		 * Constructor, default settings.
73  		 */
74  		protected Builder() {
75  			produces("text/plain");
76  		}
77  
78  		/**
79  		 * Copy constructor.
80  		 *
81  		 * @param copyFrom The builder to copy from.
82  		 * 	<br>Cannot be <jk>null</jk>.
83  		 */
84  		protected Builder(Builder copyFrom) {
85  			super(assertArgNotNull(ARG_copyFrom, copyFrom));
86  		}
87  
88  		/**
89  		 * Copy constructor.
90  		 *
91  		 * @param copyFrom The bean to copy from.
92  		 * 	<br>Cannot be <jk>null</jk>.
93  		 */
94  		protected Builder(PlainTextSerializer copyFrom) {
95  			super(assertArgNotNull(ARG_copyFrom, copyFrom));
96  		}
97  
98  		@Override /* Overridden from Context.Builder<?> */
99  		public PlainTextSerializer build() {
100 			return cache(CACHE).build(PlainTextSerializer.class);
101 		}
102 
103 		@Override /* Overridden from Context.Builder<?> */
104 		public Builder copy() {
105 			return new Builder(this);
106 		}
107 
108 
109 	}
110 
111 	/** Default serializer, all default settings.*/
112 	public static final PlainTextSerializer DEFAULT = new PlainTextSerializer(create());
113 
114 	/**
115 	 * Creates a new builder for this object.
116 	 *
117 	 * @return A new builder.
118 	 */
119 	public static Builder create() {
120 		return new Builder();
121 	}
122 
123 	private final Map<ClassMeta<?>,PlainTextClassMeta> plainTextClassMetas = new ConcurrentHashMap<>();
124 	private final Map<BeanPropertyMeta,PlainTextBeanPropertyMeta> plainTextBeanPropertyMetas = new ConcurrentHashMap<>();
125 
126 	/**
127 	 * Constructor.
128 	 *
129 	 * @param builder
130 	 * 	The builder for this object.
131 	 */
132 	public PlainTextSerializer(Builder builder) {
133 		super(builder);
134 	}
135 
136 	@Override /* Overridden from Context */
137 	public Builder copy() {
138 		return new Builder(this);
139 	}
140 
141 	@Override /* Overridden from Context */
142 	public PlainTextSerializerSession.Builder createSession() {
143 		return PlainTextSerializerSession.create(this);
144 	}
145 
146 	@Override /* Overridden from PlainTextMetaProvider */
147 	public PlainTextBeanPropertyMeta getPlainTextBeanPropertyMeta(BeanPropertyMeta bpm) {
148 		if (bpm == null)
149 			return PlainTextBeanPropertyMeta.DEFAULT;
150 		return plainTextBeanPropertyMetas.computeIfAbsent(bpm, k -> new PlainTextBeanPropertyMeta(k.getDelegateFor(), this));
151 	}
152 
153 	@Override /* Overridden from PlainTextMetaProvider */
154 	public PlainTextClassMeta getPlainTextClassMeta(ClassMeta<?> cm) {
155 		return plainTextClassMetas.computeIfAbsent(cm, k -> new PlainTextClassMeta(k, this));
156 	}
157 
158 	@Override /* Overridden from Context */
159 	public PlainTextSerializerSession getSession() { return createSession().build(); }
160 
161 	/**
162 	 * Convenience delegator that opens a {@link RecordWriter} over the output using
163 	 * <b>default session arguments</b> (mirrors {@link #write(Object)}).
164 	 *
165 	 * <p>
166 	 * The real implementation lives on {@link PlainTextSerializerSession#writeRecords(Object)}.
167 	 * Callers that need request-derived configuration should call {@link #createSession()} and
168 	 * invoke {@link PlainTextSerializerSession#writeRecords(Object)} on the built session instead.
169 	 *
170 	 * @param output The output.
171 	 * @return A new {@link RecordWriter}.
172 	 * @throws IOException If a problem occurred opening the underlying output.
173 	 */
174 	@Override /* RecordWritable */
175 	public RecordWriter writeRecords(Object output) throws IOException {
176 		return getSession().writeRecords(output);
177 	}
178 
179 	@Override /* RecordWritable */
180 	public boolean isRecordStreaming() {
181 		return false;
182 	}
183 }
