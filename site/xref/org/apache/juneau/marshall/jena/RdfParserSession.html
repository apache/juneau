123 		var langName = ctx.getLanguage();
124 		lang = toLang(langName);
125 		if (lang == null)
126 			throw isex("Unknown RDF language: %s", langName);

129 	private static Lang toLang(String langName) {
130 		var lang = RDFLanguages.nameToLang(langName);
131 		if (lang != null)
132 			return lang;
133 		if ("RDF/PROTO".equals(langName)) // HTT - not registered in Jena's RDFLanguages
134 			return Lang.RDFPROTO;
135 		return null;
136 	}

142 	/*
143 	 * Decodes the specified string.
144 	 * If {@link RdfParser#RDF_trimWhitespace} is <jk>true</jk>, the resulting string is trimmed before decoding.
145 	 * If {@link #isTrimStrings()} is <jk>true</jk>, the resulting string is trimmed after decoding.
146 	 */
147 	private String decodeString(Object o) {
148 		if (o == null)
149 			return null;
150 		var s = o.toString();
151 		if (s.isEmpty())
152 			return s;
153 		if (isTrimWhitespace())
154 			s = s.trim();
155 		s = XmlUtils.decode(s, null);
156 		if (isTrimStrings())
157 			s = s.trim();
158 		return s;
159 	}

161 	/*
162 	 * Finds the roots in the model using either the "root" property to identify it,
163 	 * 	or by resorting to scanning the model for all nodes with no incoming predicates.
164 	 */
165 	private List<Resource> getRoots(Model m) {
166 		var l = new LinkedList<Resource>();
167 
168 		// First try to find the root using the "http://www.apache.org/juneau/root" property.
169 		var root = m.createProperty(getJuneauNs().getUri(), RDF_juneauNs_ROOT);
170 		for (ResIterator i = m.listResourcesWithProperty(root); i.hasNext();)
171 			l.add(i.next());
172 
173 		if (! l.isEmpty())
174 			return l;
175 
176 		// Otherwise, we need to find all resources that aren't objects.
177 		// We want to explicitly ignore statements where the subject
178 		// and object are the same node.
179 		var objects = new HashSet<RDFNode>();
180 		for (StmtIterator i = m.listStatements(); i.hasNext();) {
181 			var st = i.next();
182 			var subject = st.getSubject();
183 			var object = st.getObject();
184 			if (object.isResource() && ! object.equals(subject))
185 				objects.add(object);
186 		}
187 		for (ResIterator i = m.listSubjects(); i.hasNext();) {
188 			var r = i.next();
189 			if (! objects.contains(r))
190 				l.add(r);
191 		}
192 		return l;
193 	}

195 	private Object getValue(RDFNode n, Object outer) throws ParseException {
196 		if (n.isLiteral())
197 			return n.asLiteral().getValue();
198 		if (n.isResource()) {
199 			var st = n.asResource().getProperty(pValue);
200 			if (nn(st)) {
201 				n = st.getObject();
202 				if (n.isLiteral())
203 					return n.asLiteral().getValue();
204 				return readAnything(object(), st.getObject(), outer, null);
205 			}
206 		}
207 		throw new ParseException(this, "Unknown value type for node '%s'", n);
208 	}

210 	private boolean isBag(RDFNode n) {
211 		if (n.isResource()) {
212 			var st = n.asResource().getProperty(pRdfType);
213 			if (nn(st))
214 				return RDF_BAG.equals(st.getResource().getURI());
215 		}
216 		return false;
217 	}

220 		var bpRdf = getRdfBeanPropertyMeta(pMeta);
221 
222 		if (bpRdf.getCollectionFormat() != RdfCollectionFormat.DEFAULT)
223 			return bpRdf.getCollectionFormat() == RdfCollectionFormat.MULTI_VALUED;
224 
225 		return getCollectionFormat() == RdfCollectionFormat.MULTI_VALUED;

237 	private <T> T readAnything(ClassMeta<?> eType, RDFNode n, Object outer, BeanPropertyMeta pMeta) throws ParseException {
238 
239 		if (eType == null)
240 			eType = object();
241 		var swap = (ObjectSwap<T,Object>)eType.getSwap(this);
242 		var builder = (BuilderSwap<T,Object>)eType.getBuilderSwap(this);
243 		ClassMeta<?> sType;
244 		if (nn(builder))
245 			sType = builder.getBuilderClassMeta(this);
246 		else if (nn(swap))
247 			sType = swap.getSwapClassMeta(this);
248 		else
249 			sType = eType;
250 
251 		if (sType.isOptional())
252 			return (T)o(readAnything(eType.getElementType(), n, outer, pMeta));
253 
254 		setCurrentClass(sType);
255 
256 		if (! sType.canCreateNewInstance(outer) && n.isResource()) {
257 			var st = n.asResource().getProperty(pType);
258 			if (nn(st)) {
259 				var c = st.getLiteral().getString();
260 				var tcm = getClassMeta(c, pMeta, eType);
261 				if (nn(tcm))
262 					sType = eType = tcm;
263 			}
264 		}
265 
266 		Object o = null;
267 		if (n.isResource() && nn(n.asResource().getURI()) && n.asResource().getURI().equals(RDF_NIL)) {
268 			// Do nothing.  Leave o == null.
269 		} else if (sType.isObject()) {
270 			if (n.isLiteral()) {
271 				o = n.asLiteral().getValue();
272 				if (o instanceof String o2) {
273 					o = decodeString(o2);
274 				}
275 			} else if (n.isResource()) {
276 				var r = n.asResource();
277 				if (! urisVisited.add(r))
278 					o = r.getURI();
279 				else if (nn(r.getProperty(pValue))) {
280 					o = readAnything(object(), n.asResource().getProperty(pValue).getObject(), outer, null);
281 				} else if (isSeq(r)) {
282 					o = new JsonList(this);
283 					readIntoCollection(r.as(Seq.class), (Collection)o, sType, pMeta);
284 				} else if (isBag(r)) {
285 					o = new JsonList(this);
286 					readIntoCollection(r.as(Bag.class), (Collection)o, sType, pMeta);
287 				} else if (r.canAs(RDFList.class)) {
288 					o = new JsonList(this);
289 					readIntoCollection(r.as(RDFList.class), (Collection)o, sType, pMeta);
290 				} else {
291 					// If it has a URI and no child properties, we interpret this as an
292 					// external resource, and convert it to just a URL.
293 					var uri = r.getURI();
294 					if (nn(uri) && ! r.listProperties().hasNext()) {
295 						o = r.getURI();
296 					} else {
297 						var m2 = new JsonMap(this);
298 						readIntoMap(r, m2, null, null, pMeta);
299 						o = cast(m2, pMeta, eType);
300 					}
301 				}
302 			} else {
303 				throw new ParseException(this, "Unrecognized node type '%s' for object", n);
304 			}
305 		} else if (nn(builder)) {
306 			var r = n.asResource();
307 			if (! urisVisited.add(r))
308 				return null;
309 			var bm = toBeanMap(builder.create(this, eType));
310 			o = builder.build(this, readIntoBeanMap(r, bm).getBean(), eType);
311 		} else if (sType.canCreateNewBean(outer)) {
312 			var r = n.asResource();
313 			if (! urisVisited.add(r))
314 				return null;
315 			var bm = newBeanMap(outer, sType.inner());
316 			o = readIntoBeanMap(r, bm).getBean();
317 		} else if (sType.isMap()) {
318 			var r = n.asResource();
319 			if (! urisVisited.add(r))
320 				return null;
321 			var m = (sType.canCreateNewInstance(outer) ? (Map)sType.newInstance(outer) : newGenericMap(sType));
322 			o = readIntoMap(r, m, eType.getKeyType(), eType.getValueType(), pMeta);
323 		} else if (sType.isCollectionOrArray() || sType.isArgs()) {
324 			if (sType.isArray() || sType.isArgs())
325 				o = list();
326 			else
327 				o = (sType.canCreateNewInstance(outer) ? (Collection<?>)sType.newInstance(outer) : new JsonList(this));
328 			var r = n.asResource();
329 			if (! urisVisited.add(r))
330 				return null;
331 			if (isSeq(r)) {
332 				readIntoCollection(r.as(Seq.class), (Collection)o, sType, pMeta);
333 			} else if (isBag(r)) {
334 				readIntoCollection(r.as(Bag.class), (Collection)o, sType, pMeta);
335 			} else if (r.canAs(RDFList.class)) {
336 				readIntoCollection(r.as(RDFList.class), (Collection)o, sType, pMeta);
337 			} else {
338 				throw new ParseException(this, "Unrecognized node type '%s' for collection", n);
339 			}
340 			if (sType.isArray() || sType.isArgs())
341 				o = toArray(sType, (Collection)o);
342 		} else if (sType.isCharSequence()) {
343 			o = decodeString(getValue(n, outer));
344 		} else if (sType.isChar()) {
345 			o = parseCharacter(decodeString(getValue(n, outer)));
346 		} else if (sType.isNumber()) {
347 			o = parseNumber(getValue(n, outer).toString(), (Class<? extends Number>)sType.inner());
348 		} else if (sType.isBoolean()) {
349 			o = convertToType(getValue(n, outer), boolean.class);
350 		} else if (sType.isUri() && n.isResource()) {
351 			// RDF URIs must be reconstructed from the resource's URI string (not from a literal),
352 			// so this branch is kept here to fire before the temporal/scalar coercion fallbacks.
353 			o = sType.newInstanceFromString(outer, decodeString(n.asResource().getURI()));
354 		} else if (sType.isDate()) {
355 			o = readDate(getValue(n, outer).toString(), sType);
356 		} else if (sType.isCalendar()) {
357 			o = readCalendar(getValue(n, outer).toString(), sType);
358 		} else if (sType.isTemporal()) {
359 			o = readTemporal(getValue(n, outer).toString(), sType);
360 		} else if (sType.isDuration()) {
361 			o = readDuration(getValue(n, outer).toString());
362 		} else if (sType.isPeriod()) {
363 			o = readPeriod(getValue(n, outer).toString());
364 		} else if (sType.canCreateNewInstanceFromString(outer)) {
365 			o = sType.newInstanceFromString(outer, decodeString(getValue(n, outer)));
366 		} else if (n.isResource()) {
367 			var r = n.asResource();
368 			var m = newGenericMap(sType);
369 			readIntoMap(r, m, sType.getKeyType(), sType.getValueType(), pMeta);
370 			if (m.containsKey(getBeanTypePropertyName(eType)))
371 				o = cast((MarshalledMap)m, pMeta, eType);
372 			else if (nn(sType.getProxyInvocationHandler()))
373 				o = newBeanMap(outer, sType.inner()).load(m).getBean();
374 			else
375 				throw new ParseException(this, "Class '%s' could not be instantiated.  Reason: '%s'", cn(sType), sType.getNotABeanReason());
376 		} else {
377 			throw new ParseException(this, "Class '%s' could not be instantiated.  Reason: '%s'", cn(sType), sType.getNotABeanReason());
378 		}
379 
380 		if (nn(swap) && nn(o))
381 			o = unswap(swap, o, eType);
382 
383 		if (nn(outer))
384 			setParent(eType, o, outer);
385 
386 		return (T)o;
387 	}

389 	private <T> BeanMap<T> readIntoBeanMap(Resource r2, BeanMap<T> m) throws ParseException {
390 		var bm = m.getMeta();
391 		var rbm = getRdfBeanMeta(bm);
392 		if (rbm.hasBeanUri() && nn(r2.getURI()))
393 			rbm.getBeanUriProperty().set(m, null, r2.getURI());
394 		for (StmtIterator i = r2.listProperties(); i.hasNext();) {
395 			var st = i.next();
396 			var p = st.getPredicate();
397 			var key = decodeString(p.getLocalName());
398 			var pMeta = m.getPropertyMeta(key);
399 			setCurrentProperty(pMeta);
400 			if (nn(pMeta)) {
401 				var o = st.getObject();
402 				var cm = (ClassMeta<?>) pMeta.getBeanInfo();
403 				if (cm.isCollectionOrArray() && isMultiValuedCollections(pMeta)) {
404 					var et = cm.getElementType();
405 					var value = readAnything(et, o, m.getBean(false), pMeta);
406 					setName(et, value, key);
407 					try {
408 						pMeta.add(m, key, value);
409 					} catch (BeanRuntimeException e) {
410 						onBeanSetterException(pMeta, e);
411 						throw e;
412 					}
413 				} else {
414 					var value = readAnything(cm, o, m.getBean(false), pMeta);
415 					setName(cm, value, key);
416 					try {
417 						pMeta.set(m, key, value);
418 					} catch (BeanRuntimeException e) {
419 						onBeanSetterException(pMeta, e);
420 						throw e;
421 					}
422 				}
423 			} else if (! (p.equals(pRoot) || p.equals(pType))) {
424 				var o = st.getObject();
425 				var value = readAnything(object(), o, m.getBean(false), null);
426 				onUnknownProperty(key, m, value);
427 			}
428 			setCurrentProperty(null);
429 		}
430 		return m;
431 	}

433 	private <E> Collection<E> readIntoCollection(Container c, Collection<E> l, ClassMeta<?> type, BeanPropertyMeta pMeta) throws ParseException {
434 		int argIndex = 0;
435 		for (NodeIterator ni = c.iterator(); ni.hasNext();) {
436 			// HTT: isArgs() requires method-argument list ClassMeta; not reachable through public parse() API.
437 			E e = (E)readAnything(type.isArgs() ? type.getArg(argIndex++) : type.getElementType(), ni.next(), l, pMeta);
438 			l.add(e);
439 		}
440 		return l;
441 	}

443 	private <E> Collection<E> readIntoCollection(RDFList list, Collection<E> l, ClassMeta<?> type, BeanPropertyMeta pMeta) throws ParseException {
444 		int argIndex = 0;
445 		for (ExtendedIterator<RDFNode> ni = list.iterator(); ni.hasNext();) {
446 			// HTT: isArgs() requires method-argument list ClassMeta; not reachable through public parse() API.
447 			E e = (E)readAnything(type.isArgs() ? type.getArg(argIndex++) : type.getElementType(), ni.next(), l, pMeta);
448 			l.add(e);
449 		}
450 		return l;
451 	}

453 	private <K,V> Map<K,V> readIntoMap(Resource r, Map<K,V> m, ClassMeta<K> keyType, ClassMeta<V> valueType, BeanPropertyMeta pMeta) throws ParseException {
454 		// Add URI as "uri" to generic maps.
455 		if (nn(r.getURI())) {
456 			var uri = convertAttrToType(m, "uri", keyType);
457 			var value = convertAttrToType(m, r.getURI(), valueType);
458 			m.put(uri, value);
459 		}
460 		for (StmtIterator i = r.listProperties(); i.hasNext();) {
461 			var st = i.next();
462 			var p = st.getPredicate();
463 			var key = p.getLocalName();
464 			if (! (key.equals("root") && p.getURI().equals(getJuneauNs().getUri()))) {
465 				key = decodeString(key);
466 				var o = st.getObject();
467 				var key2 = convertAttrToType(m, key, keyType);
468 				V value = readAnything(valueType, o, m, pMeta);
469 				setName(valueType, value, key);
470 				m.put(key2, value);
471 			}
472 
473 		}
474 		return m;
475 	}

477 	@Override /* Overridden from ReaderParserSession */
478 	protected <T> T doRead(ParserPipe pipe, ClassMeta<T> type) throws IOException, ParseException, ExecutableException {
479 
480 		// RDFDataMgr.read accepts StringReader; use RDFParser for Reader input
481 		var content = pipe.asString();
482 		RDFParser.fromString(content, lang)
483 			.base("http://unknown/")
484 			.parse(model);
485 
486 		var roots = getRoots(model);
487 
488 		// Special case where we're parsing a loose collection of resources.
489 		if (isLooseCollections() && type.isCollectionOrArray()) {
490 			Collection c;
491 			// HTT: isArgs() requires method-argument list ClassMeta; not reachable through public parse() API.
492 			if (type.isArray() || type.isArgs())
493 				c = list();
494 			else
495 				c = (type.canCreateNewInstance(getOuter()) ? (Collection<?>)type.newInstance(getOuter()) : new JsonList(this));
496 
497 			var argIndex = new AtomicInteger(0);
498 			var c2 = c;
499 			// HTT: isArgs() guard — same constraint as above.
500 			roots.forEach(x -> c2.add(readAnything(type.isArgs() ? type.getArg(argIndex.getAndIncrement()) : type.getElementType(), x, getOuter(), null)));
501 
502 			// HTT: isArgs() guard — same constraint as above.
503 			if (type.isArray() || type.isArgs())
504 				return (T)toArray(type, c);
505 			return (T)c;
506 		}
507 
508 		if (roots.isEmpty())
509 			return type.isOptional() ? (T)oe() : null;
510 
511 		if (roots.size() > 1)
512 			throw new ParseException(this, "Too many root nodes found in model:  %s", roots.size());
513 		var resource = roots.get(0);
514 
515 		return readAnything(type, resource, getOuter(), null);
516 	}
