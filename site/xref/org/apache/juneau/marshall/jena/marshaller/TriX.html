17  package org.apache.juneau.marshall.jena.marshaller;
18  
19  import org.apache.juneau.marshall.jena.*;
20  import org.apache.juneau.marshall.marshaller.*;
21  import java.lang.reflect.*;
22  import org.apache.juneau.marshall.parser.*;
23  import org.apache.juneau.marshall.serializer.*;
24  
25  /**
26   * A pairing of a {@link TriXSerializer} and {@link TriXParser} into a single class with convenience to/of methods.
27   *
28   * <h5 class='figure'>Examples:</h5>
29   * <p class='bjava'>
30   * 	<jc>// Serialize a bean or map to TriX (RDF/XML-based)</jc>
31   * 	String <jv>trix</jv> = TriX.<jsm>of</jsm>(<jv>myBean</jv>);
32   *
33   * 	<jc>// Parse TriX into a bean or map</jc>
34   * 	MyPojo <jv>parsed</jv> = TriX.<jsm>to</jsm>(<jv>trix</jv>, MyPojo.<jk>class</jk>);
35   * </p>
36   * <p class='bjava'>
37   * 	<jc>// Using instance methods</jc>
38   * 	TriX <jv>m</jv> = TriX.<jsf>DEFAULT</jsf>;
39   * 	<jv>trix</jv> = <jv>m</jv>.write(<jv>myBean</jv>);
40   * 	<jv>parsed</jv> = <jv>m</jv>.read(<jv>trix</jv>, MyPojo.<jk>class</jk>);
41   * </p>
42   *
43   * <h5 class='figure'>Example output (bean with name/age):</h5>
44   * <p class='bxml'>
45   * 	<xt>&lt;TriX&gt;</xt>
46   * 		<xt>&lt;graph&gt;</xt>
47   * 			<xt>&lt;uri&gt;</xt>...<xt>&lt;/uri&gt;</xt>
48   * 			<xt>&lt;triple&gt;</xt>
49   * 				<xt>&lt;id&gt;</xt>...<xt>&lt;/id&gt;</xt>
50   * 				<xt>&lt;name&gt;</xt>Alice<xt>&lt;/name&gt;</xt>
51   * 				<xt>&lt;age&gt;</xt>30<xt>&lt;/age&gt;</xt>
52   * 			<xt>&lt;/triple&gt;</xt>
53   * 		<xt>&lt;/graph&gt;</xt>
54   * 	<xt>&lt;/TriX&gt;</xt>
55   * </p>
56   *
57   * <h5 class='figure'>Complex (nested address + array):</h5>
58   * <p class='bxml'>
59   * 	<xt>&lt;TriX&gt;</xt>
60   * 		<xt>&lt;graph&gt;</xt>
61   * 			<xt>&lt;triple&gt;</xt>...subject, predicate, object for name, age, address ref, tags...
62   * 			<xt>&lt;triple&gt;</xt>...address subject, street, city, state...
63   * 			<xt>&lt;triple&gt;</xt>...tags list members...
64   * 		<xt>&lt;/graph&gt;</xt>
65   * 	<xt>&lt;/TriX&gt;</xt>
66   * </p>
67   *
68   * <p class='bjava'>
69   *	<jc>// Using static shortcuts.</jc>
70   * 	MyPojo <jv>myPojo</jv> = TriX.<jsm>to</jsm>(<jv>string</jv>, MyPojo.<jk>class</jk>);
71   * 	String <jv>string</jv> = TriX.<jsm>of</jsm>(<jv>myPojo</jv>);
72   * </p>
73   *
74   * <h5 class='section'>See Also:</h5><ul>
75   * 	<li class='link'>{doc jmr.RdfDetails}
76   * 	<li class='link'>{doc jm.Marshallers}
77   * </ul>
78   */
79  public class TriX extends CharMarshaller {
80  
81  	/** Default reusable instance.*/
82  	public static final TriX DEFAULT = new TriX();
83  
84  	/**
85  	 * Serializes a POJO to a <c>String</c> using the {@link #DEFAULT} marshaller.
86  	 *
87  	 * <p>
88  	 * A shortcut for calling <c><jsf>DEFAULT</jsf>.write(<jv>object</jv>)</c>.
89  	 *
90  	 * @param object The object to serialize.
91  	 * @return The serialized object.
92  	 * @throws SerializeException If a problem occurred trying to convert the output.
93  	 */
94  	public static String of(Object object) throws SerializeException {
95  		return DEFAULT.write(object);
96  	}
97  
98  	/**
99  	 * Parses an input into the specified object type using the {@link #DEFAULT} marshaller.
100 	 *
101 	 * <p>
102 	 * A shortcut for calling <c><jsf>DEFAULT</jsf>.read(<jv>input</jv>, <jv>type</jv>)</c>.
103 	 *
104 	 * @param <T> The class type of the object being created.
105 	 * @param input The input.
106 	 * @param type The object type to create.
107 	 * @return The parsed object.
108 	 * @throws ParseException Malformed input encountered.
109 	 */
110 	public static <T> T to(String input, Class<T> type) throws ParseException {
111 		return DEFAULT.read(input, type);
112 	}
113 
114 	/**
115 	 * Parses an input into the specified parameterized object type using the {@link #DEFAULT} marshaller.
116 	 *
117 	 * <p>
118 	 * A shortcut for calling <c><jsf>DEFAULT</jsf>.read(<jv>input</jv>, <jv>type</jv>, <jv>args</jv>)</c>.
119 	 *
120 	 * @param <T> The class type of the object to create.
121 	 * @param input The input.
122 	 * @param type The object type to create.
123 	 * @param args The type arguments of the class if it's a collection or map.
124 	 * @return The parsed object.
125 	 * @throws ParseException Malformed input encountered.
126 	 */
127 	public static <T> T to(String input, Type type, Type... args) throws ParseException {
128 		return DEFAULT.read(input, type, args);
129 	}
130 
131 	/** Constructor using defaults.*/
132 	public TriX() {
133 		this(TriXSerializer.DEFAULT, TriXParser.DEFAULT);
134 	}
135 
136 	/**
137 	 * Constructor with serializer and parser.
138 	 *
139 	 * @param s The serializer to use.
140 	 * 	<br>Must not be <jk>null</jk>.
141 	 * @param p The parser to use.
142 	 * 	<br>Must not be <jk>null</jk>.
143 	 */
144 	public TriX(TriXSerializer s, TriXParser p) {
145 		super(s, p);
146 	}
147 }
