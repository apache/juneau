223 		/**
224 		 * RDF language.
225 		 *
226 		 * <p>
227 		 * Can be any of the following:
228 		 * <ul class='spaced-list'>
229 		 * 	<li>
230 		 * 		<js>"RDF/XML"</js>
231 		 * 	<li>
232 		 * 		<js>"RDF/XML-ABBREV"</js> (default)
233 		 * 	<li>
234 		 * 		<js>"N-TRIPLE"</js>
235 		 * 	<li>
236 		 * 		<js>"N3"</js> - General name for the N3 writer.
237 		 * 		Will make a decision on exactly which writer to use (pretty writer, plain writer or simple writer) when
238 		 * 		created.
239 		 * 		Default is the pretty writer but can be overridden with system property
240 		 * 		<c>org.apache.jena.n3.N3JenaWriter.writer</c>.
241 		 * 	<li>
242 		 * 		<js>"N3-PP"</js> - Name of the N3 pretty writer.
243 		 * 		The pretty writer uses a frame-like layout, with prefixing, clustering like properties and embedding
244 		 * 		one-referenced bNodes.
245 		 * 	<li>
246 		 * 		<js>"N3-PLAIN"</js> - Name of the N3 plain writer.
247 		 * 		The plain writer writes records by subject.
248 		 * 	<li>
249 		 * 		<js>"N3-TRIPLES"</js> - Name of the N3 triples writer.
250 		 * 		This writer writes one line per statement, like N-Triples, but does N3-style prefixing.
251 		 * 	<li>
252 		 * 		<js>"TURTLE"</js> -  Turtle writer.
253 		 * 		http://www.dajobe.org/2004/01/turtle/
254 		 * </ul>
255 		 *
256 		 * <h5 class='section'>See Also:</h5><ul>
257 		 * 	<li class='jm'>{@link RdfParser.Builder#n3()}
258 		 * 	<li class='jm'>{@link RdfParser.Builder#ntriple()}
259 		 * 	<li class='jm'>{@link RdfParser.Builder#turtle()}
260 		 * 	<li class='jm'>{@link RdfParser.Builder#xml()}
261 		 * </ul>
262 		 *
263 		 * @param value The new value for this property.
264 		 * 	<br>Cannot be <jk>null</jk>.
265 		 * @return This object.
266 		 */

272 		/**
273 		 * Collections should be serialized and parsed as loose collections.
274 		 *
275 		 * <p>
276 		 * When specified, collections of resources are handled as loose collections of resources in RDF instead of
277 		 * resources that are children of an RDF collection (e.g. Sequence, Bag).
278 		 *
279 		 * <p>
280 		 * Note that this setting is specialized for RDF syntax, and is incompatible with the concept of
281 		 * losslessly representing POJO models, since the tree structure of these POJO models are lost
282 		 * when serialized as loose collections.
283 		 *
284 		 * <p>
285 		 * This setting is typically only useful if the beans being parsed into do not have a bean property
286 		 * annotated with {@link Rdf#beanUri @Rdf(beanUri=true)}.
287 		 *
288 		 * <h5 class='section'>Example:</h5>
289 		 * <p class='bjava'>
290 		 * 	WriterSerializer <jv>serializer</jv> = RdfSerializer
291 		 * 		.<jsm>create</jsm>()
292 		 * 		.xmlabbrev()
293 		 * 		.looseCollections()
294 		 * 		.build();
295 		 *
296 		 * 	ReaderParser <jv>parser</jv> = RdfParser
297 		 * 		.<jsm>create</jsm>()
298 		 * 		.xml()
299 		 * 		.looseCollections()
300 		 * 		.build();
301 		 *
302 		 * 	List&lt;MyBean&gt; <jv>list</jv> = <jsm>createListOfMyBeans</jsm>();
303 		 *
304 		 * 	<jc>// Serialize to RDF/XML as loose resources</jc>
305 		 * 	String <jv>rdfXml</jv> = <jv>serializer</jv>.write(<jv>list</jv>);
306 		 *
307 		 * 	<jc>// Parse back into a Java collection</jc>
308 		 * 	<jv>list</jv> = <jv>parser</jv>.read(<jv>rdfXml</jv>, LinkedList.<jk>class</jk>, MyBean.<jk>class</jk>);
309 		 *
310 		 * 	MyBean[] <jv>bean</jv> = <jsm>createArrayOfMyBeans</jsm>();
311 		 *
312 		 * 	<jc>// Serialize to RDF/XML as loose resources</jc>
313 		 * 	<jv>rdfXml</jv> = <jv>serializer</jv>.write(<jv>beans</jv>);
314 		 *
315 		 * 	<jc>// Parse back into a bean array</jc>
316 		 * 	<jv>beans</jv> = <jv>parser</jv>.read(<jv>rdfXml</jv>, MyBean[].<jk>class</jk>);
317 		 * </p>
318 		 *
319 		 * @return This object.
320 		 */
321 		public SELF looseCollections() {
322 			return looseCollections(true);
323 		}
