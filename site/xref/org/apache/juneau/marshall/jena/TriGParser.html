17  package org.apache.juneau.marshall.jena;
18  
19  /**
20   * Subclass of {@link RdfParser} for parsing TriG into POJOs.
21   *
22   * <p>
23   * Accepts RDF datasets in TriG format (Turtle with named graphs) and converts them to Java beans,
24   * maps, collections, and primitive types. Date/time literals are parsed via ISO 8601.
25   *
26   * <h5 class='figure'>Examples:</h5>
27   * <p class='bjava'>
28   * 	<jc>// Parse TriG string into a bean.</jc>
29   * 	String <jv>triG</jv> = ...;  <jc>// TriG content</jc>
30   * 	Person <jv>person</jv> = TriGParser.<jsf>DEFAULT</jsf>.read(<jv>triG</jv>, Person.<jk>class</jk>);
31   * </p>
32   * <p class='bjava'>
33   * 	<jc>// Or use the TriG marshaller for convenience.</jc>
34   * 	Person <jv>person</jv> = TriG.<jsm>to</jsm>(<jv>triG</jv>, Person.<jk>class</jk>);
35   * </p>
36   * <p class='bjava'>
37   * 	<jc>// Custom parser with swaps.</jc>
38   * 	TriGParser <jv>p</jv> = TriGParser.create().swaps(DateSwap.<jk>class</jk>).build();
39   * 	MyBean <jv>bean</jv> = <jv>p</jv>.read(<jv>triG</jv>, MyBean.<jk>class</jk>);
40   * </p>
41   *
42   * <h5 class='figure'>Example input (bean with name/age):</h5>
43   * <p class='bcode'>
44   * 	@prefix j: &lt;...&gt; .
45   * 	&lt;...&gt; { &lt;...&gt; a j:Person ; j:name "Alice" ; j:age 30 . }
46   * </p>
47   *
48   * <h5 class='figure'>Complex (nested address + array):</h5>
49   * <p class='bcode'>
50   * 	@prefix j: &lt;...&gt; .
51   * 	&lt;...&gt; { &lt;...&gt; a j:Person ; j:name "Alice" ; j:age 30 ;
52   * 		j:address &lt;.../address&gt; . }
53   * 	&lt;...&gt; { &lt;.../address&gt; a j:Address ; j:street "123 Main St" ;
54   * 		j:city "Boston" ; j:state "MA" . }
55   * </p>
56   *
57   * <h5 class='section'>Notes:</h5><ul>
58   * 	<li class='note'>This class is thread safe and reusable.
59   * </ul>
60   *
61   * <h5 class='section'>See Also:</h5><ul>
62   * 	<li class='link'>{doc jmr.RdfDetails}
63   * </ul>
64   */
65  @SuppressWarnings({
66  	"java:S110" // Deep inheritance inherent to the RDF parser hierarchy.
67  })
68  public class TriGParser extends RdfParser {
69  
70  	/** Default TriG parser, all default settings.*/
71  	public static final TriGParser DEFAULT = new TriGParser(create());
72  
73  	/**
74  	 * Creates a new builder for this object.
75  	 *
76  	 * @return A new builder.
77  	 */
78  	@SuppressWarnings({
79  		"java:S1452" // Builder<?> wildcard return intentional; callers chain via fluent API without needing the concrete type
80  	})
81  	public static RdfParser.Builder<?> create() {
82  		return RdfParser.create().triG();
83  	}
84  
85  	/**
86  	 * Constructor.
87  	 *
88  	 * @param builder The builder for this object.
89  	 * 	<br>Cannot be <jk>null</jk>.
90  	 */
91  	public TriGParser(RdfParser.Builder<?> builder) {
92  		super(builder.triG().consumes("application/trig"));
93  	}
94  }
