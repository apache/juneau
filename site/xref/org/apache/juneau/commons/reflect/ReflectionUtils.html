26  /**
27   * Utility class providing convenient static methods for creating reflection info objects.
28   *
29   * <p>
30   * This class provides static factory methods that convert standard Java reflection objects
31   * ({@link Class}, {@link Method}, {@link Field}, {@link Constructor}) to their corresponding
32   * info wrapper objects ({@link ClassInfo}, {@link MethodInfo}, {@link FieldInfo}, {@link ConstructorInfo}).
33   *
34   * <h5 class='section'>Features:</h5>
35   * <ul class='spaced-list'>
36   * 	<li>Convenient factory methods - convert reflection objects to info wrappers
37   * 	<li>Null-safe - methods handle <jk>null</jk> inputs gracefully
38   * 	<li>Unified API - consistent method naming across all reflection types
39   * </ul>
40   *
41   * <h5 class='section'>Use Cases:</h5>
42   * <ul class='spaced-list'>
43   * 	<li>Converting reflection objects to info wrappers in a consistent way
44   * 	<li>Simplifying code that works with both reflection and info objects
45   * 	<li>Providing a centralized location for reflection-to-info conversions
46   * </ul>
47   *
48   * <h5 class='section'>Usage:</h5>
49   * <p class='bjava'>
50   * 	<jc>// Convert Class to ClassInfo</jc>
51   * 	ClassInfo <jv>ci</jv> = ReflectionUtils.<jsm>info</jsm>(MyClass.<jk>class</jk>);
52   *
53   * 	<jc>// Convert Method to MethodInfo</jc>
54   * 	Method <jv>m</jv> = MyClass.<jk>class</jk>.getMethod(<js>"myMethod"</js>);
55   * 	MethodInfo <jv>mi</jv> = ReflectionUtils.<jsm>info</jsm>(<jv>m</jv>);
56   *
57   * 	<jc>// Convert Field to FieldInfo</jc>
58   * 	Field <jv>f</jv> = MyClass.<jk>class</jk>.getField(<js>"myField"</js>);
59   * 	FieldInfo <jv>fi</jv> = ReflectionUtils.<jsm>info</jsm>(<jv>f</jv>);
60   *
61   * 	<jc>// Convert Constructor to ConstructorInfo</jc>
62   * 	Constructor&lt;?&gt; <jv>c</jv> = MyClass.<jk>class</jk>.getConstructor();
63   * 	ConstructorInfo <jv>ci2</jv> = ReflectionUtils.<jsm>info</jsm>(<jv>c</jv>);
64   * </p>
65   *
66   * <h5 class='section'>Null Handling:</h5>
67   * <p>
68   * All methods in this class handle <jk>null</jk> inputs gracefully, returning <jk>null</jk> if the
69   * input is <jk>null</jk>. This makes it safe to use in scenarios where reflection objects may be <jk>null</jk>.
70   *
71   * <h5 class='section'>See Also:</h5><ul>
72   * 	<li class='jc'>{@link ClassInfo} - Class introspection wrapper
73   * 	<li class='jc'>{@link MethodInfo} - Method introspection wrapper
74   * 	<li class='jc'>{@link FieldInfo} - Field introspection wrapper
75   * 	<li class='jc'>{@link ConstructorInfo} - Constructor introspection wrapper
76   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauCommonsReflection">Reflection Package</a>
77   * </ul>
78   */

88  	 *

91  	 * @return The {@link ClassInfoTyped} wrapper, or <jk>null</jk> if the input is <jk>null</jk>.

137 	/**
138 	 * Converts a class to an enum class, suppressing rawtypes warnings.
139 	 *
140 	 * <p>
141 	 * This utility method wraps {@link Class#asSubclass(Class)} for enum classes and suppresses
142 	 * the rawtypes warning that occurs when converting to {@code Class<? extends Enum>} for use
143 	 * with methods like {@link EnumSet#allOf(Class)}.
144 	 *
145 	 * <h5 class='section'>Example:</h5>
146 	 * <p class='bjava'>
147 	 * 	<jc>// Instead of:</jc>
148 	 * 	<ja>@SuppressWarnings</ja>(<js>"rawtypes"</js>)
149 	 * 	Class&lt;? <jk>extends</jk> Enum&gt; <jv>enumClass</jv> = <jv>clazz</jv>.asSubclass(Enum.<jk>class</jk>);
150 	 *
151 	 * 	<jc>// Use:</jc>
152 	 * 	Class&lt;? <jk>extends</jk> Enum&gt; <jv>enumClass</jv> = ReflectionUtils.<jsm>asEnumClass</jsm>(<jv>clazz</jv>);
153 	 * </p>
154 	 *
155 	 * @param clazz The class to convert. Must be an enum class. Must not be <jk>null</jk>.
156 	 * @return The enum class as a raw type suitable for use with {@link EnumSet#allOf(Class)}.
157 	 * @throws ClassCastException if the class is not an enum class.
158 	 */
159 	@SuppressWarnings({
160 		"rawtypes" // Raw types necessary for enum class return
161 	})
162 	public static Class<? extends Enum> asEnumClass(Class<?> clazz) {
163 		return clazz.asSubclass(Enum.class);
164 	}

166 	/**
167 	 * Resolves and creates a collection, array, or map containing all beans of the element type from the bean store.
168 	 *
169 	 * <p>
170 	 * This is a helper method used internally by {@link ParameterInfo} and {@link FieldInfo} for resolving collection values.
171 	 *
172 	 * @param elementType The element type (for arrays/collections) or value type (for maps).
173 	 * @param beanStore The bean store to retrieve beans from.
174 	 * @param targetType The target collection type (must be a collection type).
175 	 * @return The collection, array, or map containing all beans of the element type, or <jk>null</jk> if not a supported collection type.
176 	 */
177 	static Object resolveCollectionValue(Class<?> elementType, BeanStore beanStore, ClassInfo targetType) {
178 		if (elementType == null || ! targetType.isInjectCollectionType())
179 			return null;
180 
181 		// Get all beans of the element type
182 		var beans = beanStore.getBeansOfType(elementType);
183 
184 		// Handle Map<String,T> - getBeansOfType already returns Map<String,T>
185 		var inner = o(targetType.inner()).orElse(Object.class);
186 		if (Map.class.isAssignableFrom(inner)) {
187 			// Verify it's Map<String,T> by checking the parameterized type
188 			Type parameterizedType = targetType.innerType();
189 			if (parameterizedType instanceof ParameterizedType parameterizedType2) {
190 				var typeArgs = parameterizedType2.getActualTypeArguments();
191 				if (typeArgs.length >= 2 && typeArgs[0] == String.class) {
192 					return new LinkedHashMap<>(beans);
193 				}
194 			}
195 			return null;
196 		}
197 
198 		var values = new ArrayList<>(beans.values());
199 
200 		// Convert to appropriate collection/array type
201 		if (targetType.isArray()) {
202 			var array = Array.newInstance(elementType, values.size());
203 			for (var i = 0; i < values.size(); i++)
204 				Array.set(array, i, values.get(i));
205 			return array;
206 		}
207 
208 		if (eq(inner, Set.class)) {
209 			return new LinkedHashSet<>(values);
210 		} else if (eq(inner, List.class)) {
211 			return values;
212 		}
213 
214 		// Defensive check: return null if inner is null or not a supported collection type.
215 		return null;
216 	}
