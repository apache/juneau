17  package org.apache.juneau.commons.settings;
18  
19  import static org.apache.juneau.commons.utils.AssertionUtils.*;
20  import static org.apache.juneau.commons.utils.Shorts.*;
21  import static org.apache.juneau.commons.utils.ThrowableUtils.*;
22  
23  import java.util.function.*;
24  
25  import org.apache.juneau.commons.function.*;
26  
27  /**
28   * A writable {@link PropertyStore} implementation created from functional interfaces.
29   */
30  @SuppressWarnings({
31  	"java:S115" // Constants use ARG_lowerCamel convention to match the corresponding constructor parameter name (e.g., ARG_reader → reader).
32  })
33  public class FunctionalPropertyStore implements PropertyStore {
34  
35  	private static final String ARG_reader = "reader";
36  	private static final String ARG_writer = "writer";
37  	private static final String ARG_unsetter = "unsetter";
38  	private static final String ARG_clearer = "clearer";
39  
40  	private final UnaryOperator<String> reader;
41  	private final BiConsumer<String,String> writer;
42  	private final Consumer<String> unsetter;
43  	private final Snippet clearer;
44  
45  	/**
46  	 * Constructor.
47  	 *
48  	 * @param reader The function used to read property values by name.  Must not be <jk>null</jk>.
49  	 * @param writer The function used to write property values.  Must not be <jk>null</jk>.
50  	 * @param unsetter The function used to remove a property by name.  Must not be <jk>null</jk>.
51  	 * @param clearer The function used to clear all properties.  Must not be <jk>null</jk>.
52  	 */
53  	public FunctionalPropertyStore(
54  		UnaryOperator<String> reader,
55  		BiConsumer<String,String> writer,
56  		Consumer<String> unsetter,
57  		Snippet clearer
58  	) {
59  		assertArgNotNull(ARG_reader, reader);
60  		assertArgNotNull(ARG_writer, writer);
61  		assertArgNotNull(ARG_unsetter, unsetter);
62  		assertArgNotNull(ARG_clearer, clearer);
63  		this.reader = reader;
64  		this.writer = writer;
65  		this.unsetter = unsetter;
66  		this.clearer = clearer;
67  	}
68  
69  	@Override
70  	public PropertyLookupResult get(String name) {
71  		var v = reader.apply(name);
72  		return v == null ? PropertyLookupResult.missing() : PropertyLookupResult.present(o(v));
73  	}
74  
75  	@Override
76  	public void set(String name, String value) {
77  		writer.accept(name, value);
78  	}
79  
80  	@Override
81  	public void unset(String name) {
82  		unsetter.accept(name);
83  	}
84  
85  	@Override
86  	public void clear() {
87  		safe(clearer::run);
88  	}
89  
90  	/**
91  	 * Creates a new store from the specified functions.
92  	 *
93  	 * @param reader The function used to read property values by name.  Must not be <jk>null</jk>.
94  	 * @param writer The function used to write property values.  Must not be <jk>null</jk>.
95  	 * @param unsetter The function used to remove a property by name.  Must not be <jk>null</jk>.
96  	 * @param clearer The function used to clear all properties.  Must not be <jk>null</jk>.
97  	 * @return A new store.
98  	 */
99  	public static FunctionalPropertyStore of(
100 		UnaryOperator<String> reader,
101 		BiConsumer<String,String> writer,
102 		Consumer<String> unsetter,
103 		Snippet clearer
104 	) {
105 		return new FunctionalPropertyStore(reader, writer, unsetter, clearer);
106 	}
107 }
