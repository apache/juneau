32  /**
33   * Encapsulates Java system properties with support for global and per-thread overrides for unit testing.
34   *
35   * <p>
36   * This class provides a thread-safe way to access system properties that can be overridden at both the global level
37   * and per-thread level, making it useful for unit tests that need to temporarily change system property
38   * values without affecting other tests or threads.
39   *
40   * <p>
41   * Settings instances are created using the {@link Builder} pattern. Use {@link #create()} to create a new builder,
42   * or {@link #get()} to get the singleton instance (which is created using default builder settings).
43   *
44   * <h5 class='section'>Lookup Order:</h5>
45   * <p>
46   * When retrieving a property value, the lookup order is:
47   * <ol>
48   * 	<li>Per-thread store (if set via {@link #setLocal(String, String)})
49   * 	<li>Global store (if set via {@link #setGlobal(String, String)})
50   * 	<li>Sources in reverse order (last source added via {@link Builder#addSource(PropertySource)} is checked first)
51   * 	<li>System property source (default, always second-to-last)
52   * 	<li>System environment variable source (default, always last)
53   * </ol>
54   *
55   * <h5 class='section'>Sources vs Stores:</h5>
56   * <ul class='spaced-list'>
57   * 	<li><b>Sources</b> ({@link PropertySource}) - Provide read-only access to property values. Examples: {@link FunctionalPropertySource}
58   * 	<li><b>Stores</b> ({@link PropertyStore}) - Provide read/write access to property values. Examples: {@link MapStore}, {@link FunctionalPropertyStore}
59   * 	<li>Stores can be used as sources (they extend {@link PropertySource}), so you can add stores via {@link Builder#addSource(PropertySource)}
60   * </ul>
61   *
62   * <h5 class='section'>Features:</h5>
63   * <ul class='spaced-list'>
64   * 	<li>System property access - read Java system properties with type conversion via {@link StringSetting}
65   * 	<li>Global overrides - override system properties globally for all threads (stored in a {@link PropertyStore})
66   * 	<li>Per-thread overrides - override system properties for specific threads (stored in a per-thread {@link PropertyStore})
67   * 	<li>Custom sources - add arbitrary property sources (e.g., Spring properties, environment variables, config files) via the {@link Builder}
68   * 	<li>Disable override support - system property to prevent new global overrides from being set
69   * 	<li>Type-safe accessors - type conversion methods on {@link StringSetting} for common types: Integer, Long, Boolean, Double, Float, File, Path, URI, Charset
70   * 	<li>Resettable suppliers - settings are returned as {@link StringSetting} instances that can be reset to force recomputation
71   * </ul>
72   *
73   * <h5 class='section'>Usage Examples:</h5>
74   * <p class='bjava'>
75   * 	<jc>// Get a system property as a StringSetting (using singleton instance)</jc>
76   * 	StringSetting <jv>setting</jv> = Settings.<jsf>get</jsf>().get(<js>"my.property"</js>);
77   * 	String <jv>value</jv> = <jv>setting</jv>.get();  <jc>// Get the string value</jc>
78   *
79   * 	<jc>// Get with type conversion using StringSetting methods</jc>
80   * 	Setting&lt;Integer&gt; <jv>intSetting</jv> = Settings.<jsf>get</jsf>().get(<js>"my.int.property"</js>).asInteger();
81   * 	Setting&lt;Long&gt; <jv>longSetting</jv> = Settings.<jsf>get</jsf>().get(<js>"my.long.property"</js>).asLong();
82   * 	Setting&lt;Boolean&gt; <jv>boolSetting</jv> = Settings.<jsf>get</jsf>().get(<js>"my.bool.property"</js>).asBoolean();
83   * 	Setting&lt;Double&gt; <jv>doubleSetting</jv> = Settings.<jsf>get</jsf>().get(<js>"my.double.property"</js>).asDouble();
84   * 	Setting&lt;Float&gt; <jv>floatSetting</jv> = Settings.<jsf>get</jsf>().get(<js>"my.float.property"</js>).asFloat();
85   * 	Setting&lt;File&gt; <jv>fileSetting</jv> = Settings.<jsf>get</jsf>().get(<js>"my.file.property"</js>).asFile();
86   * 	Setting&lt;Path&gt; <jv>pathSetting</jv> = Settings.<jsf>get</jsf>().get(<js>"my.path.property"</js>).asPath();
87   * 	Setting&lt;URI&gt; <jv>uriSetting</jv> = Settings.<jsf>get</jsf>().get(<js>"my.uri.property"</js>).asURI();
88   * 	Setting&lt;Charset&gt; <jv>charsetSetting</jv> = Settings.<jsf>get</jsf>().get(<js>"my.charset.property"</js>).asCharset();
89   *
90   * 	<jc>// Use custom type conversion</jc>
91   * 	Setting&lt;MyCustomType&gt; <jv>customSetting</jv> = Settings.<jsf>get</jsf>().get(<js>"my.custom.property"</js>).asType(MyCustomType.<jk>class</jk>);
92   *
93   * 	<jc>// Reset a setting to force recomputation</jc>
94   * 	<jv>setting</jv>.reset();
95   *
96   * 	<jc>// Override for current thread (useful in unit tests)</jc>
97   * 	Settings.<jsf>get</jsf>().setLocal(<js>"my.property"</js>, <js>"test-value"</js>);
98   * 	<jc>// ... test code that uses the override ...</jc>
99   * 	Settings.<jsf>get</jsf>().unsetLocal(<js>"my.property"</js>);  <jc>// Remove specific override</jc>
100  * 	<jc>// OR</jc>
101  * 	Settings.<jsf>get</jsf>().clearLocal();  <jc>// Clear all thread-local overrides</jc>
102  *
103  * 	<jc>// Set global override (applies to all threads)</jc>
104  * 	Settings.<jsf>get</jsf>().setGlobal(<js>"my.property"</js>, <js>"global-value"</js>);
105  * 	<jc>// ... code that uses the global override ...</jc>
106  * 	Settings.<jsf>get</jsf>().unsetGlobal(<js>"my.property"</js>);  <jc>// Remove specific override</jc>
107  * 	<jc>// OR</jc>
108  * 	Settings.<jsf>get</jsf>().clearGlobal();  <jc>// Clear all global overrides</jc>
109  *
110  * 	<jc>// Create a custom Settings instance with custom sources (e.g., Spring properties)</jc>
111  * 	MapStore <jv>springSource</jv> = <jk>new</jk> MapStore();
112  * 	<jv>springSource</jv>.set(<js>"spring.datasource.url"</js>, <js>"jdbc:postgresql://localhost/db"</js>);
113  * 	Settings <jv>custom</jv> = Settings.<jsf>create</jsf>()
114  * 		.addSource(<jv>springSource</jv>)
115  * 		.addSource(FunctionalPropertySource.<jsf>of</jsf>(System::getProperty))
116  * 		.build();
117  * </p>
118  *
119  * <h5 class='section'>System Properties:</h5>
120  * <ul class='spaced-list'>
121  * 	<li><c>juneau.settings.disableGlobal</c> (system property) or <c>JUNEAU_SETTINGS_DISABLEGLOBAL</c> (system env) - If set to <c>true</c>, prevents new global overrides
122  * 		from being set via {@link #setGlobal(String, String)}. Existing global overrides will still be
123  * 		returned by {@link #get(String)} until explicitly removed.
124  * 		<p class='bnote'>
125  * 			Note: This property is read once at class initialization time when creating the singleton instance
126  * 			and cannot be changed at runtime. Changing the system property after the class has been loaded will
127  * 			have no effect on the singleton instance. However, you can create custom Settings instances using
128  * 			{@link #create()} that ignore this property.
129  * 		</p>
130  * </ul>
131  */

180 	 * Returns properties for this Settings object itself.
181 	 * Note that these are initialized at startup and not changeable through System.setProperty().
182 	 */
183 	private static final Optional<String> initProperty(String property) {
184 		var v = SYSTEM_PROPERTY_SOURCE.get(property);
185 		if (v.isPresent())
186 			return v.value();  // Not testable
187 		v = SYSTEM_ENV_SOURCE.get(ucr(property.replace('.', '_')));
188 		if (v.isPresent())
189 			return v.value();  // Not testable
190 		return oe();
191 	}

287 		 * Discovers and registers {@link PropertySourceProvider} instances via {@link ServiceLoader}.
288 		 *
289 		 * <p>
290 		 * Providers are sorted by {@link PropertySourceProvider#order()} ascending.
291 		 *
292 		 * @return This builder for method chaining.
293 		 */
294 		public Builder useServiceLoader() {
295 			ServiceLoader.load(PropertySourceProvider.class)
296 				.stream()
297 				.map(ServiceLoader.Provider::get)
298 				.sorted(Comparator.comparingInt(PropertySourceProvider::order))
299 				.map(PropertySourceProvider::create)
300 				.filter(Objects::nonNull)
301 				.forEach(this::addSource);
302 			return this;
303 		}

410 	/**
411 	 * Removes a previously-added property source.
412 	 *
413 	 * <p>
414 	 * The match is by identity ({@code ==}), not equality.  Callers that mutate the singleton
415 	 * {@link #get() Settings.get()} from a test should capture the exact reference returned to/passed
416 	 * into {@link #addSource(PropertySource)} and pass it back here in their teardown to avoid
417 	 * cross-test bleed.
418 	 *
419 	 * @param source The property source to remove. Must not be <jk>null</jk>.
420 	 * @return <jk>true</jk> if the source was present and removed, <jk>false</jk> otherwise.
421 	 */
422 	public boolean removeSource(PropertySource source) {
423 		assertArgNotNull(ARG_source, source);
424 		// Identity-based removal so that PropertySource impls without a sensible equals() still
425 		// round-trip cleanly through add/remove pairs.
426 		for (var i = 0; i < sources.size(); i++) {
427 			if (sources.get(i) == source) {
428 				sources.remove(i);
429 				return true;
430 			}
431 		}
432 		return false;
433 	}

435 	/**
436 	 * Returns a {@link StringSetting} for the specified system property.
437 	 *
438 	 * <p>
439 	 * The returned {@link StringSetting} is a resettable supplier that caches the lookup result.
440 	 * Use the {@link StringSetting#asInteger()}, {@link StringSetting#asBoolean()}, etc. methods
441 	 * to convert to different types.
442 	 *
443 	 * <p>
444 	 * The lookup order is:
445 	 * <ol>
446 	 * 	<li>Per-thread override (if set via {@link #setLocal(String, String)})
447 	 * 	<li>Global override (if set via {@link #setGlobal(String, String)})
448 	 * 	<li>Sources in reverse order (last source added via {@link Builder#addSource(PropertySource)} is checked first)
449 	 * 	<li>System property source (default, always second-to-last)
450 	 * 	<li>System environment variable source (default, always last)
451 	 * </ol>
452 	 *
453 	 * @param name The property name. Must not be <jk>null</jk>.
454 	 * @return A {@link StringSetting} that provides the resolved property value.
455 	 */
456 	public StringSetting get(String name) {
457 		assertArgNotNull(ARG_name, name);
458 		return new StringSetting(this, () -> {
459 			// 1. Check thread-local override
460 			var v = localStore.get().get(name);
461 			if (v.isPresent())
462 				return v.value().orElse(null); // Present result: Optional.empty() means explicit null override.
463 
464 			// 2. Check global override
465 			v = globalStore.get().get(name);
466 			if (v.isPresent())
467 				return v.value().orElse(null); // Present result: Optional.empty() means explicit null override.
468 
469 			// 3. Check sources in reverse order (last added first)
470 			for (int i = sources.size() - 1; i >= 0; i--) {
471 				var source = sources.get(i);
472 				var result = source.get(name);
473 				if (result.isPresent())
474 					return result.value().orElse(null);
475 			}
476 
477 			return null;
478 		});
479 	}

481 	/**
482 	 * Returns the override value for the specified property, consulting only the per-thread
483 	 * (local) and global stores &mdash; <i>not</i> the sources list.
484 	 *
485 	 * <p>
486 	 * Intended for callers (notably the {@code @Value} resolution path) that interleave a
487 	 * caller-scoped property source between the Settings override stores and the sources list.
488 	 * Using this method, the caller can honor the "{@link #setLocal(String, String) local}
489 	 * / {@link #setGlobal(String, String) global} override wins" contract before consulting
490 	 * its own scoped sources, and still fall through to {@link #get(String)} (which checks
491 	 * the sources list) afterwards without the override stores accidentally being consulted twice.
492 	 *
493 	 * <p>
494 	 * Lookup order honored by this method:
495 	 * <ol>
496 	 * 	<li>Per-thread (local) store
497 	 * 	<li>Global store
498 	 * </ol>
499 	 *
500 	 * <p>
501 	 * If a store is present with an empty value (i.e. an explicit null override), the returned
502 	 * {@link Optional} is {@code Optional.empty()} but {@link #isOverridden(String)} would
503 	 * return {@code true}. Callers needing to distinguish "absent" from "explicit null" should
504 	 * use {@link #isOverridden(String)} alongside this method.
505 	 *
506 	 * @param name The property name. Must not be <jk>null</jk>.
507 	 * @return The override value, or {@link Optional#empty()} if neither store carries an override.
508 	 * @see #isOverridden(String)
509 	 * @see #get(String)
510 	 */
511 	public Optional<String> getOverride(String name) {
512 		assertArgNotNull(ARG_name, name);
513 		var v = localStore.get().get(name);
514 		if (v.isPresent())
515 			return v.value();
516 		v = globalStore.get().get(name);
517 		if (v.isPresent())
518 			return v.value();
519 		return oe();
520 	}

539 	/**
540 	 * Looks up a system property, returning a default value if not found.
541 	 *
542 	 * <p>
543 	 * This method searches for a value using the same lookup order as {@link #get(String)}.
544 	 * If a value is found, it is converted to the type of the default value using {@link #toType(String, Class)}.
545 	 * Supported types include any type that has a static method with signature <c>public static &lt;T&gt; T anyName(String arg)</c>
546 	 * or a public constructor with signature <c>public T(String arg)</c>, such as {@link Boolean}, {@link Integer}, {@link Charset}, {@link java.io.File}, etc.
547 	 *
548 	 * <h5 class='section'>Example:</h5>
549 	 * <p class='bjava'>
550 	 * 	<jc>// System property: -Dmy.property=true</jc>
551 	 * 	Boolean <jv>flag</jv> = get(<js>"my.property"</js>, <jk>false</jk>);  <jc>// true</jc>
552 	 *
553 	 * 	<jc>// Environment variable: MY_PROPERTY=UTF-8</jc>
554 	 * 	Charset <jv>charset</jv> = get(<js>"my.property"</js>, Charset.defaultCharset());  <jc>// UTF-8</jc>
555 	 *
556 	 * 	<jc>// Not found, returns default</jc>
557 	 * 	String <jv>value</jv> = get(<js>"nonexistent"</js>, <js>"default"</js>);  <jc>// "default"</jc>
558 	 * </p>
559 	 *
560 	 * @param <T> The type to convert the value to.
561 	 * @param name The property name. Must not be <jk>null</jk>.
562 	 * @param def The default value to return if not found. Must not be <jk>null</jk>.
563 	 * @return The found value (converted to type T), or the default value if not found.
564 	 * @see #get(String)
565 	 * @see #toType(String, Class)
566 	 */
567 	@SuppressWarnings({
568 		"unchecked" // Type erasure requires unchecked cast
569 	})
570 	public <T> T get(String name, T def) {
571 		assertArgNotNull(ARG_def, def);
572 		return get(name).asType((Class<T>)def.getClass()).orElse(def);
573 	}

712 	/**
713 	 * Converts a string to the specified type using reflection to find conversion methods or constructors.
714 	 *
715 	 * <p>
716 	 * This method attempts to convert a string to the specified type using the following lookup order:
717 	 * <ol>
718 	 * 	<li>Custom type functions registered via {@link Builder#addTypeFunction(Class, Function)}</li>
719 	 * 	<li>Special handling for {@link String} (returns the string as-is)</li>
720 	 * 	<li>Special handling for {@link Enum} types (uses {@link Enum#valueOf(Class, String)})</li>
721 	 * 	<li>Reflection lookup for static methods with signature <c>public static &lt;T&gt; T anyName(String arg)</c>
722 	 * 		on the target class</li>
723 	 * 	<li>Reflection lookup for static methods on the superclass (for abstract classes like {@link Charset}
724 	 * 		where concrete implementations need to use the abstract class's static method)</li>
725 	 * 	<li>Reflection lookup for public constructors with signature <c>public T(String arg)</c></li>
726 	 * </ol>
727 	 *
728 	 * <p>
729 	 * When a conversion method or constructor is found via reflection, it is cached in the
730 	 * <c>toTypeFunctions</c> map for future use.
731 	 *
732 	 * <h5 class='section'>Examples:</h5>
733 	 * <ul class='spaced-list'>
734 	 * 	<li><c>Boolean</c> - Uses <c>Boolean.valueOf(String)</c> static method
735 	 * 	<li><c>Integer</c> - Uses <c>Integer.valueOf(String)</c> static method
736 	 * 	<li><c>Charset</c> - Uses <c>Charset.forName(String)</c> static method (even for concrete implementations)
737 	 * 	<li><c>File</c> - Uses <c>File(String)</c> constructor
738 	 * </ul>
739 	 *
740 	 * @param <T> The target type.
741 	 * @param s The string to convert. Must not be <jk>null</jk>.
742 	 * @param c The target class. Must not be <jk>null</jk>.
743 	 * @return The converted value.
744 	 * @throws RuntimeException If the type is not supported for conversion (no static method or constructor found).
745 	 */
746 	@SuppressWarnings({
747 		"unchecked", // Type erasure requires unchecked cast
748 		"rawtypes" // Raw types necessary for generic type handling
749 	})
750 	public <T> T toType(String s, Class<T> c) {
751 		assertArgNotNull(ARG_s, s);
752 		assertArgNotNull(ARG_c, c);
753 		var f = (Function<String,T>)toTypeFunctions.get(c);
754 		if (f == null) {
755 			if (c == String.class)
756 				return (T)s;
757 			if (c.isEnum())
758 				return (T)Enum.valueOf((Class<? extends Enum>)c, s);
759 			ClassInfoTyped<T> ci = info(c);
760 			f = ci.getDeclaredMethod(x -> x.isStatic() && x.hasParameterTypes(String.class) && x.hasReturnType(c) && FROM_STRING_METHOD_NAMES.contains(x.getName())).map(x -> (Function<String,T>)s2 -> x.invoke(null, s2)).orElse(null);
761 			// Also accept `static T parse(CharSequence)` etc. — common for java.time types (Instant,
762 			// LocalDate, Duration) whose static factories declare the wider CharSequence parameter.
763 			if (f == null)
764 				f = ci.getDeclaredMethod(x -> x.isStatic() && x.hasParameterTypes(CharSequence.class) && x.hasReturnType(c) && FROM_STRING_METHOD_NAMES.contains(x.getName())).map(x -> (Function<String,T>)s2 -> x.invoke(null, s2)).orElse(null);
765 			if (f == null)
766 				f = ci.getPublicConstructor(x -> x.hasParameterTypes(String.class)).map(x -> (Function<String,T>)x::newInstance).orElse(null);
767 			if (f != null)
768 				toTypeFunctions.putIfAbsent(c, f);
769 		}
770 		if (f == null)
771 			throw rex("Invalid env type: %s", c);
772 		return f.apply(s);
773 	}
