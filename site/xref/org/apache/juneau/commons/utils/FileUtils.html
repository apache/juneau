56  	 * Creates a file if it doesn't already exist using {@link File#createNewFile()}.
57  	 *
58  	 * <p>
59  	 * Throws a {@link RuntimeException} if the file could not be created.
60  	 *
61  	 * @param f The file to create.  Must not be <jk>null</jk>.
62  	 */
63  	public static void create(File f) {
64  		if (f.exists())
65  			return;
66  		safe(() -> optional(f.createNewFile()).filter(x -> x).orElseThrow(() -> rex("Could not create file '%s'", f.getAbsolutePath())));
67  	}

69  	/**
70  	 * Create a temporary file with the specified name.
71  	 *
72  	 * <p>
73  	 * The name is broken into file name and suffix, and the parts are passed to
74  	 * {@link File#createTempFile(String, String)}.
75  	 *
76  	 * <p>
77  	 * {@link File#deleteOnExit()} is called on the resulting file before being returned by this method.
78  	 *
79  	 * @param name The file name.  Must not be <jk>null</jk> (unguarded — a <jk>null</jk> name throws {@link NullPointerException}).
80  	 * @return A newly-created temporary file.
81  	 * @throws IOException Thrown by underlying stream.
82  	 */
83  	@SuppressWarnings({
84  		"java:S5443" // Controlled internal helper: uses JDK default temp dir, deleteOnExit; explicit POSIX perms aren't portable to Windows.
85  	})
86  	public static File createTempFile(String name) throws IOException {
87  		var parts = name.split("\\.");
88  		var f = File.createTempFile(parts[0], "." + parts[1]);
89  		f.deleteOnExit();
90  		return f;
91  	}

93  	/**
94  	 * Create a temporary file with the specified name and specified contents.
95  	 *
96  	 * <p>
97  	 * The name is broken into file name and suffix, and the parts are passed to
98  	 * {@link File#createTempFile(String, String)}.
99  	 *
100 	 * <p>
101 	 * {@link File#deleteOnExit()} is called on the resulting file before being returned by this method.
102 	 *
103 	 * @param name The file name.  Must not be <jk>null</jk> (unguarded — a <jk>null</jk> name throws {@link NullPointerException}).
104 	 * @param contents The file contents.  Can be <jk>null</jk> (creates an empty file).
105 	 * @return A newly-created temporary file.
106 	 * @throws IOException Thrown by underlying stream.
107 	 */
108 	public static File createTempFile(String name, String contents) throws IOException {
109 		var f = createTempFile(name);
110 		if (contents != null) {
111 			try (var r = new StringReader(contents); Writer w = new FileWriter(f)) {
112 				pipe(r, w);
113 				w.flush();
114 			}
115 		}
116 		// If contents is null, create an empty file
117 		return f;
118 	}

120 	/**
121 	 * Recursively deletes a file or directory.
122 	 *
123 	 * @param f The file or directory to delete.  Can be <jk>null</jk> (returns <jk>true</jk>).
124 	 * @return <jk>true</jk> if file or directory was successfully deleted.
125 	 */
126 	public static boolean deleteFile(File f) {
127 		if (f == null)
128 			return true;
129 		if (f.isDirectory()) {
130 			var cf = f.listFiles();
131 			if (isNotNull(cf))
132 				for (var c : cf)
133 					deleteFile(c);
134 		}
135 		try {
136 			Files.delete(f.toPath());
137 			return true;
138 		} catch (@SuppressWarnings("unused") IOException e) {
139 			return false;
140 		}
141 	}

157 	 * Strips the extension from a file name.
158 	 *
159 	 * @param name The file name.
160 	 * @return The file name without the extension, or <jk>null</jk> if name was <jk>null</jk>.
161 	 */
162 	public static String getBaseName(String name) {
163 		if (name == null)
164 			return null;
165 		var i = name.lastIndexOf('.');
166 		if (i == -1)
167 			return name;
168 		return name.substring(0, i);
169 	}

172 	 * Returns the extension from a file name.
173 	 *
174 	 * @param name The file name.
175 	 * @return The the extension, or <jk>null</jk> if name was <jk>null</jk>.
176 	 */
177 	public static String getFileExtension(String name) {
178 		if (name == null)
179 			return null;
180 		var i = name.lastIndexOf('.');
181 		if (i == -1)
182 			return "";
183 		return name.substring(i + 1);
184 	}

186 	/**
187 	 * Given an arbitrary path, returns the file name portion of that path.
188 	 *
189 	 * @param path The path to check.  Can be <jk>null</jk> or empty (returns <jk>null</jk>).
190 	 * @return The file name, or <jk>null</jk> if the path was <jk>null</jk>, empty, or contained only slashes.
191 	 */
192 	public static String getFileName(String path) {
193 		if (isEmpty(path))
194 			return null;
195 		path = trimTrailingSlashes(path);
196 		if (isEmpty(path))
197 			return null;  // Path contained only slashes
198 		// Handle both forward slashes (Unix) and backslashes (Windows)
199 		var i = Math.max(path.lastIndexOf('/'), path.lastIndexOf('\\'));
200 		return i == -1 ? path : path.substring(i + 1);
201 	}

211 		if (name == null || ext == null)
212 			return false;
213 		return ext.equals(getFileExtension(name));

217 	 * Same as {@link File#mkdirs()} except throws a RuntimeExeption if directory could not be created.
218 	 *
219 	 * @param f The directory to create.  Must not be <jk>null</jk>.
220 	 * @param clean If <jk>true</jk>, deletes the contents of the directory if it already exists.
221 	 * @return The same file.
222 	 * @throws RuntimeException if directory could not be created.
223 	 */
224 	public static File mkdirs(File f, boolean clean) {
225 		assertArgNotNull(ARG_f, f);
226 		if (f.exists()) {
227 			if (clean) {
228 				optional(deleteFile(f)).filter(x -> x).orElseThrow(() -> rex("Could not clean directory '%s'", f.getAbsolutePath()));
229 			} else {
230 				return f;
231 			}
232 		}
233 		optional(f.mkdirs()).filter(x -> x).orElseThrow(() -> rex("Could not create directory '%s'", f.getAbsolutePath()));
234 		return f;
235 	}

249 	/**
250 	 * Resolves a user-supplied path string against a root directory with a strict
251 	 * filesystem-boundary check &mdash; rejects any path that escapes the root (via
252 	 * {@code ../}, absolute-path injection, symlinks pointing outside, etc.).
253 	 *
254 	 * <p>
255 	 * Resolves both {@code rootDir} and the resolved target via
256 	 * {@link Path#toRealPath Path.toRealPath()} + {@link Path#normalize Path.normalize()}
257 	 * to handle symlinks deterministically, then asserts {@code target.startsWith(root)}
258 	 * before returning the resolved {@link File}.
259 	 *
260 	 * <p>
261 	 * Symlinks <i>inside</i> {@code rootDir} that resolve <i>inside</i> {@code rootDir} are
262 	 * followed silently. Symlinks that resolve <i>outside</i> {@code rootDir} are rejected with
263 	 * {@link IllegalArgumentException} (callers should map to a {@code 403 Forbidden}).
264 	 *
265 	 * <p>
266 	 * Pre-existence semantics: when the resolved target does not exist on the filesystem, the
267 	 * helper returns {@link Optional#empty()} so the caller can map to a {@code 404 Not Found}.
268 	 * Boundary violation always wins over pre-existence &mdash; even a non-existent path that
269 	 * escapes {@code rootDir} (after {@code ../} normalization) is rejected with
270 	 * {@link IllegalArgumentException} rather than {@code Optional.empty()}.
271 	 *
272 	 * <p>
273 	 * Used by {@code DirectoryResource}, {@code LogsResource}, and any other resource that
274 	 * resolves an HTTP-supplied path under a configured root directory.
275 	 *
276 	 * @param rootDir The configured root directory. Must not be <jk>null</jk>.
277 	 * @param userPath The user-supplied path, relative to {@code rootDir}. May be
278 	 * 	<jk>null</jk> or empty &mdash; in which case the helper returns the resolved root
279 	 * 	itself.
280 	 * @return An {@link Optional} containing the resolved {@link File} if it exists inside
281 	 * 	{@code rootDir}, or {@link Optional#empty()} if the target does not exist.
282 	 * @throws IllegalArgumentException If the resolved target escapes {@code rootDir}, or if
283 	 * 	the supplied {@code userPath} is not a valid path string. Callers should map to
284 	 * 	{@code 403 Forbidden}.
285 	 */
286 	public static Optional<File> resolveSafely(File rootDir, String userPath) {
287 		assertArgNotNull(ARG_rootDir, rootDir);
288 		var root = canonicalizeRoot(rootDir.toPath());
289 		if (userPath == null || userPath.isEmpty())
290 			return optional(root.toFile());
291 		Path target;
292 		try {
293 			target = root.resolve(userPath).normalize();
294 		} catch (@SuppressWarnings("unused") InvalidPathException e) {
295 			throw iaex(MSG_pathEscape);
296 		}
297 		if (! target.startsWith(root))
298 			throw iaex(MSG_pathEscape);
299 		var f = target.toFile();
300 		if (! f.exists())
301 			return emptyOptional();
302 		try {
303 			if (! target.toRealPath().startsWith(root))
304 				throw iaex(MSG_pathEscape);
305 		} catch (@SuppressWarnings("unused") NoSuchFileException e) {
306 			return emptyOptional();
307 		} catch (IOException e) {
308 			throw rex(e, "Could not canonicalize '%s'", target);
309 		}
310 		return optional(f);
311 	}

313 	/**
314 	 * Virtual-path variant of {@link #resolveSafely(File, String)} for use with servlet-context
315 	 * paths, classpath resource lookups, or other virtual-path resolvers where no filesystem
316 	 * operation is desired.
317 	 *
318 	 * <p>
319 	 * Operates purely on the virtual path string &mdash; no {@code toRealPath()} call, no
320 	 * filesystem touch, no symlink resolution. Safe to use against base paths whose resources
321 	 * do not exist on the local filesystem at all (e.g. {@code /WEB-INF/views/} inside a
322 	 * Spring Boot fat jar, or a classpath URL).
323 	 *
324 	 * <p>
325 	 * Normalizes the combined path by walking segments and applying {@code ./} (drop) and
326 	 * {@code ../} (pop) semantics, then asserts the result stays under the normalized
327 	 * {@code basePath}. {@code ../} segments that would pop above {@code basePath} are
328 	 * rejected.
329 	 *
330 	 * <p>
331 	 * URL-encoding is the caller's responsibility &mdash; this helper treats the literal
332 	 * {@code %2e%2e} as a normal path segment (not as {@code ..}). Callers that receive
333 	 * user input from URLs should URL-decode before calling.
334 	 *
335 	 * @param basePath The configured base path (virtual; e.g. {@code "/WEB-INF/views/"}).
336 	 * 	Must not be <jk>null</jk> or blank.
337 	 * @param userPath The user-supplied path, relative to {@code basePath}. May be
338 	 * 	<jk>null</jk> or empty &mdash; in which case the helper returns the normalized
339 	 * 	{@code basePath}.
340 	 * @return The resolved virtual path string (always starting with {@code basePath}).
341 	 * @throws IllegalArgumentException If the resolved target escapes {@code basePath}, or if
342 	 * 	{@code basePath} is <jk>null</jk> or blank. Callers should map to
343 	 * 	{@code 403 Forbidden}.
344 	 */
345 	public static String resolveVirtualPathSafely(String basePath, String userPath) {
346 		assertArgNotNull(ARG_basePath, basePath);
347 		if (basePath.isBlank())
348 			throw iaex("basePath must not be blank.");
349 		var bp = normalizeVirtualPath(basePath);
350 		if (bp == null)
351 			throw iaex("basePath escapes its own root.");
352 		if (! bp.endsWith("/"))
353 			bp = bp + "/";
354 		if (userPath == null || userPath.isEmpty())
355 			return bp;
356 		var up = userPath.startsWith("/") ? userPath.substring(1) : userPath;
357 		var combined = normalizeVirtualPath(bp + up);
358 		if (combined == null || ! combined.startsWith(bp))
359 			throw iaex(MSG_pathEscape);
360 		return combined;
361 	}

363 	/**
364 	 * Resolves {@code rootDir} via {@link Path#toRealPath()} so symlinks at the root itself
365 	 * are normalized once per call. Falls back to {@link Path#toAbsolutePath()} +
366 	 * {@link Path#normalize()} when the directory does not exist at the moment of the call
367 	 * (rare; legacy contract tolerated this).
368 	 */
369 	private static Path canonicalizeRoot(Path rootDir) {
370 		try {
371 			return rootDir.toRealPath();
372 		} catch (@SuppressWarnings("unused") IOException e) {
373 			return rootDir.toAbsolutePath().normalize();
374 		}
375 	}

377 	/**
378 	 * String-level path normalization for forward-slash separated virtual paths.
379 	 * Returns {@code null} if a {@code ..} segment would pop above the root.
380 	 */
381 	@SuppressWarnings({
382 		"java:S3776", // Cognitive complexity: small loop with three early-exit cases; splitting hurts readability.
383 		"java:S135" // Multiple continue statements are intrinsic to this path-segment scan/normalization loop.
384 	})
385 	private static String normalizeVirtualPath(String path) {
386 		if (path == null)
387 			return null;
388 		var startsWithSlash = path.startsWith("/");
389 		var endsWithSlash = path.length() > 1 && path.endsWith("/");
390 		var segments = new ArrayList<String>();
391 		for (var s : path.split("/")) {
392 			if (s.isEmpty() || ".".equals(s))
393 				continue;
394 			if ("..".equals(s)) {
395 				if (segments.isEmpty())
396 					return null;
397 				segments.remove(segments.size() - 1);
398 				continue;
399 			}
400 			segments.add(s);
401 		}
402 		var sb = new StringBuilder();
403 		if (startsWithSlash)
404 			sb.append('/');
405 		sb.append(String.join("/", segments));
406 		if (endsWithSlash && (sb.isEmpty() || sb.charAt(sb.length() - 1) != '/'))
407 			sb.append('/');
408 		return sb.toString();
409 	}

411 	/**
412 	 * Updates the modified timestamp on the specified file.
413 	 *
414 	 * <p>
415 	 * Method ensures that the timestamp changes even if it's been modified within the past millisecond.
416 	 *
417 	 * @param f The file to modify the modified timestamp on.  Must not be <jk>null</jk>.
418 	 */
419 	public static void modifyTimestamp(File f) {
420 		var lm = f.lastModified();
421 		var l = System.currentTimeMillis();
422 		if (lm == l)
423 			l++;
424 		optional(f.setLastModified(l)).filter(x -> x).orElseThrow(() -> rex("Could not modify timestamp on file '%s'", f.getAbsolutePath()));
425 
426 		// Linux only gives 1s precision, so set the date 1s into the future.
427 		if (lm == f.lastModified())
428 			optional(f.setLastModified(l + 1000)).filter(x -> x).orElseThrow(() -> rex("Could not modify timestamp on file '%s'", f.getAbsolutePath()));
429 	}
