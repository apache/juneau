17  package org.apache.juneau.commons.utils;
18  
19  import java.security.*;
20  import java.time.*;
21  import java.util.*;
22  
23  /**
24   * Generator for
25   * <a class="doclink" href="https://www.rfc-editor.org/rfc/rfc9562.html#section-5.7">RFC&nbsp;9562 §5.7</a> version-7 UUIDs.
26   *
27   * <p>
28   * A v7 UUID is time-ordered: a 48-bit big-endian Unix-milliseconds timestamp in the most-significant bits, followed by
29   * the RFC&nbsp;9562 version (7) and variant nibbles and {@link SecureRandom}-filled random bits.  No JDK API produces v7
30   * natively as of the Java&nbsp;17 baseline this repo targets.
31   *
32   * <p>
33   * The {@link SecureRandom} used to fill the random bits is cached per thread ({@link ThreadLocal}) so minting is not a
34   * fresh {@link SecureRandom} construction (and potential entropy syscall) on the request path.
35   *
36   * <h5 class='section'>See Also:</h5><ul>
37   * 	<li class='extlink'><a class="doclink" href="https://www.rfc-editor.org/rfc/rfc9562.html">RFC 9562 — Universally Unique IDentifiers (UUIDs)</a>
38   * </ul>
39   */
40  public final class Uuid7 {
41  
42  	@SuppressWarnings({
43  		"java:S5164" // Process-lifetime cached SecureRandom per thread is intentional; there is no scope boundary at which remove() would apply.
44  	})
45  	private static final ThreadLocal<SecureRandom> RANDOM = ThreadLocal.withInitial(SecureRandom::new);
46  
47  	private static final Clock SYSTEM_CLOCK = Clock.systemUTC();
48  
49  	private static final long VERSION_7 = 0x7000L;      // version nibble in the low 16 bits of the MSB long
50  	private static final long RAND_A_MASK = 0x0FFFL;    // 12 random bits alongside the version nibble
51  	private static final long VARIANT_IETF = 0x2L << 62;// binary 10 in the top 2 bits of the LSB long
52  	private static final long RAND_B_MASK = 0x3FFFFFFFFFFFFFFFL; // 62 random bits
53  
54  	private Uuid7() {}
55  
56  	/**
57  	 * Generates a new version-7 UUID using the current wall-clock time.
58  	 *
59  	 * @return A new v7 UUID.  Never <jk>null</jk>.
60  	 */
61  	public static UUID create() {
62  		return create(SYSTEM_CLOCK);
63  	}
64  
65  	/**
66  	 * Generates a new version-7 UUID whose timestamp is read from the specified clock.
67  	 *
68  	 * <p>
69  	 * Lets tests pin the encoded timestamp with {@link Clock#fixed(Instant, ZoneId)}.  The random bits are still
70  	 * {@link SecureRandom}-filled, so two UUIDs minted against the same fixed clock differ.
71  	 *
72  	 * @param clock The clock supplying the Unix-milliseconds timestamp.  Must not be <jk>null</jk>.
73  	 * @return A new v7 UUID.  Never <jk>null</jk>.
74  	 */
75  	public static UUID create(Clock clock) {
76  		var random = RANDOM.get();
77  		var ts = clock.millis() & 0xFFFFFFFFFFFFL;
78  		var randA = random.nextInt() & RAND_A_MASK;
79  		var msb = (ts << 16) | VERSION_7 | randA;
80  		var randB = random.nextLong() & RAND_B_MASK;
81  		var lsb = VARIANT_IETF | randB;
82  		return new UUID(msb, lsb);
83  	}
84  
85  	/**
86  	 * Generates a new version-7 UUID and returns its canonical string form.
87  	 *
88  	 * @return The string form of a new v7 UUID.  Never <jk>null</jk>.
89  	 */
90  	public static String createString() {
91  		return create().toString();
92  	}
93  
94  	/**
95  	 * Extracts the 48-bit Unix-milliseconds timestamp encoded in a v7 UUID's most-significant bits.
96  	 *
97  	 * @param uuid The UUID.  Must not be <jk>null</jk>.
98  	 * @return The encoded Unix-milliseconds timestamp.
99  	 */
100 	public static long timestamp(UUID uuid) {
101 		return uuid.getMostSignificantBits() >>> 16;
102 	}
103 
104 	/**
105 	 * White-box accessor: the {@link SecureRandom} instance cached for the current thread.  For tests only.
106 	 *
107 	 * @return The current thread's cached {@link SecureRandom}.
108 	 */
109 	static SecureRandom currentRandom() {
110 		return RANDOM.get();
111 	}
112 }
