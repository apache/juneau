65  	/** Null/array/annotation-aware equality. */
66  	public static <T> boolean equal(T o1, T o2) {
67  		if (o1 instanceof java.lang.annotation.Annotation o1a && o2 instanceof java.lang.annotation.Annotation o2a)
68  			return AnnotationUtils.equals(o1a, o2a);
69  		if (ClassUtils.isArray(o1) && ClassUtils.isArray(o2)) {
70  			var l1 = Array.getLength(o1);
71  			var l2 = Array.getLength(o2);
72  			if (l1 != l2) return false;
73  			for (var i = 0; i < l1; i++)
74  				if (notEqual(Array.get(o1, i), Array.get(o2, i))) return false;
75  			return true;
76  		}
77  		return Objects.equals(o1, o2);
78  	}

86  	public static <T,U> boolean equal(T o1, U o2, BiPredicate<T,U> test) {
87  		if (o1 == null) return o2 == null;
88  		if (o2 == null) return false;
89  		if (o1 == o2) return true;
90  		return test.test(o1, o2);

94  	@SafeVarargs
95  	public static <T> boolean equalsAny(T o1, T...o2) {
96  		if (o2 == null || o2.length == 0) return false;

108 	public static <T,U> boolean notEqual(T o1, U o2, BiPredicate<T,U> test) {
109 		if (o1 == null) return isNotNull(o2);

111 		if (o1 == o2) return false;
112 		return ! test.test(o1, o2);

115 	/** Null-tolerant natural-order compare (0 if types differ / not Comparable). */
116 	@SuppressWarnings({
117 		"unchecked",   // Type erasure requires unchecked casts.
118 		"java:S3740"   // Raw Comparable; parameterizing breaks compareTo.
119 	})
120 	public static int compare(Object o1, Object o2) {
121 		if (o1 == null) return o2 == null ? 0 : -1;
122 		if (o2 == null) return 1;
123 		if (equal(o1.getClass(), o2.getClass()) && o1 instanceof Comparable o12)
124 			return o12.compareTo(o2);
125 		return 0;
126 	}

185 	 * Returns the minimum of two comparable values.
186 	 *
187 	 * @param <T> The comparable type.
188 	 * @param o1 Value 1.  Can be <jk>null</jk>.
189 	 * @param o2 Value 2.  Can be <jk>null</jk>.
190 	 * @return The minimum value.  If one argument is <jk>null</jk>, the other is returned (which may itself be <jk>null</jk> if both are <jk>null</jk>).
191 	 */
192 	public static <T extends Comparable<T>> T min(T o1, T o2) {
193 		if (o1 == null) return o2;
194 		if (o2 == null) return o1;
195 		return o1.compareTo(o2) <= 0 ? o1 : o2;

203 	 * @param o2 Value 2.  Can be <jk>null</jk>.
204 	 * @return The maximum value.  If one argument is <jk>null</jk>, the other is returned (which may itself be <jk>null</jk> if both are <jk>null</jk>).

207 		if (o1 == null) return o2;

268 	/** Absolute value across Number subtypes.  <br>Returns <jk>null</jk> if the argument is <jk>null</jk>. */
269 	@SuppressWarnings({ "unchecked" // Type erasure requires unchecked cast for Number types.
270 	})
271 	public static <T extends Number> T abs(T value) {
272 		if (value == null) return null;
273 		if (value instanceof Integer) return (T)Integer.valueOf(Math.abs(value.intValue()));
274 		if (value instanceof Long) return (T)Long.valueOf(Math.abs(value.longValue()));
275 		if (value instanceof Double) return (T)Double.valueOf(Math.abs(value.doubleValue()));
276 		if (value instanceof Float) return (T)Float.valueOf(Math.abs(value.floatValue()));
277 		if (value instanceof Short) return (T)Short.valueOf((short)Math.abs(value.shortValue()));
278 		if (value instanceof Byte) return (T)Byte.valueOf((byte)Math.abs(value.byteValue()));
279 		return (T)Double.valueOf(Math.abs(value.doubleValue()));
280 	}
