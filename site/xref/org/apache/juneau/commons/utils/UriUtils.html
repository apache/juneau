17  package org.apache.juneau.commons.utils;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import java.net.*;
22  
23  /**
24   * Various {@link URI}-related utility methods.
25   *
26   * @since 10.0.0
27   */
28  public class UriUtils {
29  
30  	/**
31  	 * Returns <jk>true</jk> if the specified URI either uses the <js>"https"</js> scheme or targets a loopback
32  	 * host.
33  	 *
34  	 * <p>
35  	 * A loopback host is one of <js>"localhost"</js> (case-insensitive), <js>"127.0.0.1"</js>, <js>"::1"</js>, or
36  	 * <js>"[::1]"</js>.  A plaintext (<js>"http"</js>) URI aimed at any other host returns <jk>false</jk>, as does
37  	 * a <jk>null</jk> or scheme-less (malformed) URI.  This is the check to make before transmitting anything
38  	 * sensitive to a caller-configured endpoint so that a plaintext transport is only tolerated for local
39  	 * development.
40  	 *
41  	 * @param uri The URI to check.  Can be <jk>null</jk> (returns <jk>false</jk>).
42  	 * @return <jk>true</jk> if the URI is <js>"https"</js> or targets a loopback host.
43  	 */
44  	public static boolean isSecureOrLoopback(URI uri) {
45  		if (uri == null)
46  			return false;
47  		if (eqic("https", uri.getScheme()))
48  			return true;
49  		return isLoopbackHost(uri.getHost());
50  	}
51  
52  	/**
53  	 * Returns <jk>true</jk> if the specified host name is a loopback host.
54  	 *
55  	 * <p>
56  	 * A loopback host is one of <js>"localhost"</js> (case-insensitive), <js>"127.0.0.1"</js>, <js>"::1"</js>, or
57  	 * <js>"[::1]"</js>.  Other addresses in the <c>127.0.0.0/8</c> range and other spellings of the same addresses
58  	 * are not matched.
59  	 *
60  	 * @param host The host name.  Can be <jk>null</jk> (returns <jk>false</jk>).
61  	 * @return <jk>true</jk> if the host is a loopback host.
62  	 */
63  	public static boolean isLoopbackHost(String host) {
64  		return host != null && (eqic(host, "localhost") || eqa(host, "127.0.0.1", "[::1]", "::1"));
65  	}
66  
67  	/**
68  	 * Parses a strict dotted-quad IPv4 literal.
69  	 *
70  	 * <p>
71  	 * The value must be exactly four decimal octets in the range 0-255 separated by dots, with no sign, no leading
72  	 * zeros and no surrounding whitespace.  No name lookup is done.
73  	 *
74  	 * <h4 class='section'>Example:</h4>
75  	 * <p class='bjava'>
76  	 * 	parseIpv4(<js>"10.0.0.1"</js>);  <jc>// {10, 0, 0, 1}</jc>
77  	 * 	parseIpv4(<js>"10.0.0.01"</js>); <jc>// null</jc>
78  	 * </p>
79  	 *
80  	 * @param s The string to parse.  Can be <jk>null</jk> (returns <jk>null</jk>).
81  	 * @return The four address bytes, or <jk>null</jk> if the string is not a strict dotted-quad literal.
82  	 */
83  	@SuppressWarnings({
84  		"java:S1168" // null signals an unparseable address; callers test for it
85  	})
86  	public static byte[] parseIpv4(String s) {
87  		if (s == null)
88  			return null;
89  		var parts = s.split("\\.", -1);
90  		if (parts.length != 4)
91  			return null;
92  		var out = new byte[4];
93  		for (var i = 0; i < 4; i++) {
94  			var v = parseOctet(parts[i]);
95  			if (v < 0)
96  				return null;
97  			out[i] = (byte)v;
98  		}
99  		return out;
100 	}
101 
102 	// Returns the value of a 1-3 digit decimal octet with no leading zero, or -1 if it isn't one or exceeds 255.
103 	private static int parseOctet(String p) {
104 		var n = p.length();
105 		if (n == 0 || n > 3 || (n > 1 && p.charAt(0) == '0'))
106 			return -1;
107 		var v = 0;
108 		for (var i = 0; i < n; i++) {
109 			var c = p.charAt(i);
110 			if (c < '0' || c > '9')
111 				return -1;
112 			v = v * 10 + (c - '0');
113 		}
114 		return v > 255 ? -1 : v;
115 	}
116 
117 	/**
118 	 * Throwing variant of {@link #isSecureOrLoopback(URI)}.
119 	 *
120 	 * @param uri The URI to validate.  Can be <jk>null</jk> (rejected).
121 	 * @return The validated URI (for fluent use).
122 	 * @throws IllegalArgumentException If the URI is not <js>"https"</js> and does not target a loopback host.
123 	 */
124 	public static URI assertSecureOrLoopback(URI uri) {
125 		if (! isSecureOrLoopback(uri))
126 			throw iaex("URI must use https or target a loopback host: %s", uri);
127 		return uri;
128 	}

130 	/**
131 	 * Constructor.
132 	 */
133 	protected UriUtils() {}
134 }
