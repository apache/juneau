17  package org.apache.juneau.commons.utils;
18  
19  import java.util.zip.*;
20  
21  /**
22   * Content-checksum utility methods.
23   */
24  public class ChecksumUtils {
25  
26  	/** Prevents instantiation. */
27  	private ChecksumUtils() {}
28  
29  	/**
30  	 * Formats a CRC32 checksum of the specified bytes as a zero-padded, lower-case, 8-hex-char content hash.
31  	 *
32  	 * <p>
33  	 * Suitable as a compact, content-sensitive cache-buster for classpath-shipped assets (JS/CSS/images, etc.):
34  	 * the hash changes the instant the underlying bytes do, without relying on a build/version string alone
35  	 * (which stays stable across {@code -SNAPSHOT} dev rebuilds and so cannot bust a browser's cache on its own).
36  	 *
37  	 * @param bytes The bytes to checksum.  Must not be <jk>null</jk>.
38  	 * @return The zero-padded 8-hex-char CRC32 checksum (e.g. {@code "1a2b3c4d"}).
39  	 */
40  	public static String hash8(byte[] bytes) {
41  		var crc = new CRC32();
42  		crc.update(bytes);
43  		return String.format("%08x", crc.getValue());
44  	}
45  }
