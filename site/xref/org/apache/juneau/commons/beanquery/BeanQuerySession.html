26  /**
27   * One unit of query work against a {@link BeanQueryContext}: the rows (or connection) to query, plus optional
28   * per-request narrowing of the context's settings.
29   *
30   * <p>
31   * Open a session per request and close it with try-with-resources.  A session is not thread-safe.  Its public methods
32   * are {@code final}: each validates the {@link BeanQuery} into a {@link ResolvedQuery} (throwing
33   * {@link BeanQuerySyntaxException} for bad input) and then calls the engine's {@code doXxx} hook.  Engine failures
34   * throw {@link BeanQueryExecutionException}.
35   *
36   * <p>
37   * {@link Builder#restrictColumns(String...)} only narrows the context's columns.  The limit, cap and count-policy
38   * overrides replace the context's values for this session (so they can also raise or remove a cap); they are for
39   * trusted server-side callers only, never request data.
40   *
41   * <h5 class='section'>Example:</h5>
42   * <p class='bjava'>
43   * 	<jc>// Shortcut: the context's settings as-is.</jc>
44   * 	<jk>try</jk> (InMemoryBeanQuerySession&lt;Person&gt; <jv>session</jv> = <jv>context</jv>.getSession(<jv>rows</jv>)) {
45   * 		Page&lt;Person&gt; <jv>page</jv> = <jv>session</jv>.find(<jv>query</jv>);
46   * 	}
47   *
48   * 	<jc>// Narrowed for a guest user.</jc>
49   * 	<jk>try</jk> (InMemoryBeanQuerySession&lt;Person&gt; <jv>session</jv> = <jv>context</jv>.createSession()
50   * 			.rows(<jv>rows</jv>)
51   * 			.restrictColumns(<js>"name"</js>, <js>"city"</js>)
52   * 			.defaultLimit(20)
53   * 			.maxLimit(20)
54   * 			.build()) {
55   * 		List&lt;Map&lt;String,Object&gt;&gt; <jv>values</jv> = <jv>session</jv>.findValues(<jv>query</jv>).rows();
56   * 	}
57   * </p>
58   *
59   * @param <T> The row (bean) type.
60   * @since 10.0.0
61   */
62  public abstract class BeanQuerySession<T> implements AutoCloseable {
63  
64  	/**
65  	 * The settings every engine's session builder shares.  Each starts from the context's value.
66  	 *
67  	 * @param <T> The row type.
68  	 * @param <SELF> The concrete builder type, returned by every setter.
69  	 */
70  	@SuppressWarnings({
71  		"java:S119" // 'SELF' (CRTP self-type) is intentional and clearer than a single-letter name.
72  	})
73  	public abstract static class Builder<T,SELF extends Builder<T,SELF>> {
74  
75  		private final BeanQueryContext<T> context;
76  		final QuerySettings settings;
77  		private boolean defaultLimitExplicit;  // True once defaultLimit(int) is called on THIS builder.
78  
79  		/**
80  		 * Constructor.
81  		 *
82  		 * @param context The context the session runs against.  Must not be <jk>null</jk>.
83  		 */
84  		protected Builder(BeanQueryContext<T> context) {
85  			this.context = reqnn("context", context);
86  			settings = this.context.settings.copy();
87  		}
88  
89  		/**
90  		 * Returns this builder as its concrete type.
91  		 *
92  		 * @return This object.
93  		 */
94  		@SuppressWarnings({
95  			"unchecked" // CRTP self-type cast is safe by construction.
96  		})
97  		protected final SELF self() {
98  			return (SELF)this;
99  		}

101 		/**
102 		 * Keeps only the specified columns for this session.  The context's column order is kept.
103 		 *
104 		 * @param names The column names.  Each must be a column of the context (or of an earlier restriction).
105 		 * @return This object.
106 		 * @throws IllegalArgumentException If a name is not a current column.
107 		 */
108 		public SELF restrictColumns(String...names) {
109 			var keep = new HashSet<String>();
110 			for (var n : reqnns("names", names)) {
111 				if (! settings.columns.containsKey(n))
112 					throw iaex("Column '%s' is not a column of this context; a session can only narrow the columns.", n);
113 				keep.add(n);
114 			}
115 			settings.columns.keySet().retainAll(keep);
116 			settings.columnOperators.keySet().retainAll(keep);
117 			return self();
118 		}
119 
120 		/**
121 		 * Overrides {@link BeanQueryContext.Builder#regexTimeout(Duration)} for this session.
122 		 *
123 		 * @param value The budget.  Must be positive.
124 		 * @return This object.
125 		 */
126 		public SELF regexTimeout(Duration value) {
127 			settings.regexTimeout = QuerySettings.positive("regexTimeout", value);
128 			return self();
129 		}
130 
131 		/**
132 		 * Overrides {@link BeanQueryContext.Builder#countPolicy(CountPolicy)} for this session.
133 		 *
134 		 * @param value The policy.  Must not be <jk>null</jk>.
135 		 * @return This object.
136 		 */
137 		public SELF countPolicy(CountPolicy value) {
138 			settings.countPolicy = reqnn("value", value);
139 			return self();
140 		}
141 
142 		/**
143 		 * Overrides {@link BeanQueryContext.Builder#defaultLimit(int)} for this session.
144 		 *
145 		 * @param value The limit.  Must be positive.
146 		 * @return This object.
147 		 */
148 		public SELF defaultLimit(int value) {
149 			settings.defaultLimit = QuerySettings.positive("defaultLimit", value);
150 			defaultLimitExplicit = true;
151 			return self();
152 		}
153 
154 		/**
155 		 * Overrides {@link BeanQueryContext.Builder#maxLimit(Integer)} for this session.
156 		 *
157 		 * <p>
158 		 * If this session's {@code defaultLimit} was never explicitly set on this builder (it is still the value
159 		 * inherited from the context), narrowing {@code maxLimit} below it auto-narrows {@code defaultLimit} to match,
160 		 * rather than leaving the two in conflict. An explicitly-chosen {@code defaultLimit} is never
161 		 * silently changed by this setter — a resulting conflict is left for {@link #build()} to reject.
162 		 *
163 		 * @param value The cap, or <jk>null</jk> for no cap.  Must be positive if set.
164 		 * @return This object.
165 		 */
166 		public SELF maxLimit(Integer value) {
167 			settings.maxLimit = QuerySettings.positiveOrNull("maxLimit", value);
168 			if (! defaultLimitExplicit && settings.maxLimit != null && settings.defaultLimit > settings.maxLimit)
169 				settings.defaultLimit = settings.maxLimit;
170 			return self();
171 		}
172 
173 		/**
174 		 * Overrides {@link BeanQueryContext.Builder#maxSearchClauses(int)} for this session.
175 		 *
176 		 * @param value The cap.  Must be positive.
177 		 * @return This object.
178 		 */
179 		public SELF maxSearchClauses(int value) {
180 			settings.maxSearchClauses = QuerySettings.positive("maxSearchClauses", value);
181 			return self();
182 		}
183 
184 		/**
185 		 * Overrides {@link BeanQueryContext.Builder#maxSortKeys(int)} for this session.
186 		 *
187 		 * @param value The cap.  Must be positive.
188 		 * @return This object.
189 		 */
190 		public SELF maxSortKeys(int value) {
191 			settings.maxSortKeys = QuerySettings.positive("maxSortKeys", value);
192 			return self();
193 		}
194 
195 		/**
196 		 * Overrides {@link BeanQueryContext.Builder#maxSearchLength(int)} for this session.
197 		 *
198 		 * @param value The cap.  Must be positive.
199 		 * @return This object.
200 		 */
201 		public SELF maxSearchLength(int value) {
202 			settings.maxSearchLength = QuerySettings.positive("maxSearchLength", value);
203 			return self();
204 		}

207 		 * Overrides {@link BeanQueryContext.Builder#maxExpressionDepth(int)} for this session.
208 		 *
209 		 * @param value The cap.  Must be positive.
210 		 * @return This object.

212 		public SELF maxExpressionDepth(int value) {

217 		/**
218 		 * Builds the session.
219 		 *
220 		 * @return A new session.  Close it when done.
221 		 * @throws IllegalStateException If a required setting is missing or two settings conflict.
222 		 */
223 		public abstract BeanQuerySession<T> build();
224 	}

229 	/**
230 	 * Constructor.  Takes a private copy of the builder's settings and validates it.

233 	 * @throws IllegalStateException If the settings are invalid.
234 	 */
235 	protected BeanQuerySession(Builder<T,?> builder) {
236 		context = reqnn("builder", builder).context;
237 		settings = builder.settings.copy();

286 	 * Validates a query against this session's rules without running it.
287 	 *
288 	 * @param query The query.  Must not be <jk>null</jk>.
289 	 * @return The resolved query.
290 	 * @throws BeanQuerySyntaxException If the query breaks a rule.

292 	public final ResolvedQuery resolve(BeanQuery query) {
293 		return QueryResolver.resolve(reqnn("query", query), settings, context);

296 	/**
297 	 * Returns one page of matching rows, plus the counts the {@link CountPolicy} selects.
298 	 *
299 	 * @param query The query.  Must not be <jk>null</jk>.
300 	 * @return The page.
301 	 * @throws BeanQuerySyntaxException If the query breaks a rule.
302 	 * @throws BeanQueryExecutionException If the engine fails.
303 	 */
304 	public final Page<T> find(BeanQuery query) {
305 		return doFind(resolve(query));
306 	}

309 	 * Like {@link #find(BeanQuery)}, but each row is a map of the viewed columns, in view order.

396 	/**
397 	 * Builds a page, computing only the counts the query's {@link ResolvedQuery#counts() policy} selects.
398 	 *
399 	 * @param <R> The row type.
400 	 * @param rows The page's rows.
401 	 * @param query The validated query.
402 	 * @param total Computes the total count.  Called only under {@link CountPolicy#BOTH}.
403 	 * @param matched Computes the matched count.  Called only under {@link CountPolicy#MATCHED} or {@link CountPolicy#BOTH}.
404 	 * @return The page.
405 	 */
406 	protected static <R> Page<R> page(List<R> rows, ResolvedQuery query, LongSupplier total, LongSupplier matched) {
407 		switch (query.counts()) {
408 			case BOTH:    return Page.of(rows, total.getAsLong(), matched.getAsLong());
409 			case MATCHED: return Page.ofMatched(rows, matched.getAsLong());
410 			default:      return Page.of(rows);
411 		}
412 	}
