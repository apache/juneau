20  
21  import java.lang.reflect.*;

25  import org.apache.juneau.commons.bean.*;

27  /**
28   * One bean property as BeanQuery sees it: its column name, value type and inferred {@link SearchType}, and how to read
29   * and write it.
30   *
31   * <p>
32   * Instances come from {@link QueryModel}.  The name is the Juneau bean property name, after {@link BeanProp @BeanProp}
33   * and the configured {@link PropertyNamer}, so it is the name serializers emit.  Reads use the getter (or record
34   * accessor), else the field.  Writes use the setter, else a non-final field.
35   *
36   * <h5 class='section'>Example:</h5>
37   * <p class='bjava'>
38   * 	QueryProperty <jv>age</jv> = QueryModel.<jsm>of</jsm>(Person.<jk>class</jk>).getReadableProperty(<js>"age"</js>);
39   * 	SearchType <jv>type</jv> = <jv>age</jv>.getSearchType();  <jc>// NUMERIC</jc>
40   * 	Object <jv>value</jv> = <jv>age</jv>.read(<jv>person</jv>);
41   * </p>
42   *
43   * @since 10.0.0
44   */
45  public final class QueryProperty {
46  
47  	private final String name;
48  	private final Method getter;
49  	private final Method setter;
50  	private final Field field;
51  	private final Class<?> type;
52  	private final Class<?> writeType;
53  	private final SearchType searchType;
54  	private final boolean writeOnly;  // @BeanProp(wo) / writeOnlyProperties.
55  
56  	QueryProperty(BeanPropertyMeta p) {
57  		name = p.getName();
58  		writeOnly = p.isWriteOnly();
59  		getter = p.getGetter() == null ? null : p.getGetter().inner();
60  		setter = p.getSetter() == null ? null : p.getSetter().inner();
61  		field = p.getField() == null ? null : p.getField().inner();
62  		if (nn(getter))
63  			type = getter.getReturnType();
64  		else if (nn(field))
65  			type = field.getType();
66  		else if (nn(setter))
67  			type = setter.getParameterTypes()[0];
68  		else
69  			throw isex("Property '%s' has no getter, setter or field.", name);
70  		writeType = nn(setter) ? setter.getParameterTypes()[0] : type;
71  		searchType = searchTypeOf(type);
72  	}
73  
74  	/**
75  	 * Infers a column's {@link SearchType} from a property's declared type.
76  	 *
77  	 * <p>
78  	 * Booleans are {@link SearchType#BOOLEAN}; characters are {@link SearchType#TEXT}; primitives and {@link Number}s
79  	 * are {@link SearchType#NUMERIC}; enums are {@link SearchType#ENUM}; dates, calendars and temporals are
80  	 * {@link SearchType#TIMESTAMP}; everything else is {@link SearchType#TEXT}.
81  	 *
82  	 * <h5 class='section'>Example:</h5>
83  	 * <p class='bjava'>
84  	 * 	QueryProperty.<jsm>searchTypeOf</jsm>(<jk>int</jk>.<jk>class</jk>);        <jc>// NUMERIC</jc>
85  	 * 	QueryProperty.<jsm>searchTypeOf</jsm>(Instant.<jk>class</jk>);     <jc>// TIMESTAMP</jc>
86  	 * 	QueryProperty.<jsm>searchTypeOf</jsm>(String.<jk>class</jk>);      <jc>// TEXT</jc>
87  	 * </p>
88  	 *
89  	 * @param t The property type.  Must not be <jk>null</jk>.
90  	 * @return The inferred type (never <jk>null</jk>).
91  	 */
92  	public static SearchType searchTypeOf(Class<?> t) {
93  		if (t == boolean.class || t == Boolean.class)
94  			return SearchType.BOOLEAN;
95  		if (t == char.class || t == Character.class)
96  			return SearchType.TEXT;
97  		if (t.isPrimitive() || Number.class.isAssignableFrom(t))
98  			return SearchType.NUMERIC;
99  		if (t.isEnum())
100 			return SearchType.ENUM;
101 		if (Date.class.isAssignableFrom(t) || Calendar.class.isAssignableFrom(t) || TemporalAccessor.class.isAssignableFrom(t))
102 			return SearchType.TIMESTAMP;
103 		return SearchType.TEXT;
104 	}
105 
106 	/**
107 	 * Returns the property (column) name.
108 	 *
109 	 * @return The name.  Never <jk>null</jk>.
110 	 */
111 	public String getName() { return name; }
112 
113 	/**
114 	 * Returns the declared read type: the getter's return type, else the field type, else the setter's parameter type.
115 	 *
116 	 * @return The type.  Never <jk>null</jk>.
117 	 */
118 	public Class<?> getType() { return type; }
119 
120 	/**
121 	 * Returns the type {@link #write(Object, Object)} accepts: the setter's parameter type, else {@link #getType()}.
122 	 *
123 	 * @return The type.  Never <jk>null</jk>.
124 	 */
125 	public Class<?> getWriteType() { return writeType; }
126 
127 	/**
128 	 * Returns the search type inferred from {@link #getType()}.
129 	 *
130 	 * @return The search type.  Never <jk>null</jk>.
131 	 */
132 	public SearchType getSearchType() { return searchType; }
133 
134 	/**
135 	 * Returns <jk>true</jk> if the property has a getter or a field and isn't write-only.
136 	 *
137 	 * @return <jk>true</jk> if {@link #read(Object)} is supported.
138 	 */
139 	public boolean canRead() { return ! writeOnly && (nn(getter) || nn(field)); }
140 
141 	/**
142 	 * Returns <jk>true</jk> if the property has a setter or a non-final field.
143 	 *
144 	 * <p>
145 	 * {@link BeanProp#ro() @BeanProp(ro)} is deliberately ignored: in Juneau it means "parsers must not set this
146 	 * property", and loading a database row is not parsing.
147 	 *
148 	 * @return <jk>true</jk> if {@link #write(Object, Object)} is supported.
149 	 */
150 	public boolean canWrite() { return (nn(setter) || (nn(field) && ! Modifier.isFinal(field.getModifiers()))); }
151 
152 	/**
153 	 * Reads the property from a bean.
154 	 *
155 	 * @param bean The bean.  Must not be <jk>null</jk>.
156 	 * @return The value.  Can be <jk>null</jk>.
157 	 * @throws BeanQueryExecutionException If the getter throws or the value can't be read.  The cause is the getter's
158 	 * 	own exception.
159 	 * @throws IllegalArgumentException If <c>bean</c> is <jk>null</jk>.
160 	 * @throws IllegalStateException If the property is not readable.
161 	 */
162 	public Object read(Object bean) {
163 		reqnn("bean", bean);
164 		if (! canRead())
165 			throw isex("Property '%s' is not readable.", name);
166 		try {
167 			return nn(getter) ? getter.invoke(bean) : field.get(bean);
168 		} catch (InvocationTargetException e) {
169 			throw new BeanQueryExecutionException(e.getCause(), "Failed to read column '%s'.", name);
170 		} catch (IllegalAccessException | IllegalArgumentException e) {
171 			throw new BeanQueryExecutionException(e, "Failed to read column '%s'.", name);
172 		}
173 	}
174 
175 	/**
176 	 * Writes the property on a bean.  A <jk>null</jk> for a primitive property is ignored, so the bean keeps its
177 	 * default.
178 	 *
179 	 * @param bean The bean.  Must not be <jk>null</jk>.
180 	 * @param value The value, assignable to {@link #getWriteType()}.  Can be <jk>null</jk>.
181 	 * @throws BeanQueryExecutionException If the setter throws or the value can't be written.  The cause is the
182 	 * 	setter's own exception.
183 	 * @throws IllegalArgumentException If <c>bean</c> is <jk>null</jk>.
184 	 * @throws IllegalStateException If the property is not writable.
185 	 */
186 	@SuppressWarnings({
187 		"java:S3011" // Field.set on a public field; no accessibility override, reflection is the point of this class.
188 	})
189 	public void write(Object bean, Object value) {
190 		reqnn("bean", bean);
191 		if (! canWrite())
192 			throw isex("Property '%s' is not writable.", name);
193 		if (value == null && writeType.isPrimitive())
194 			return;
195 		try {
196 			if (nn(setter))
197 				setter.invoke(bean, value);
198 			else
199 				field.set(bean, value);
200 		} catch (InvocationTargetException e) {
201 			throw new BeanQueryExecutionException(e.getCause(), "Failed to write column '%s'.", name);
202 		} catch (IllegalAccessException | IllegalArgumentException e) {
203 			throw new BeanQueryExecutionException(e, "Failed to write column '%s'.", name);
204 		}
205 	}
206 
207 	@Override /* Object */
208 	public String toString() {
209 		return name + ":" + searchType;
210 	}
211 }
