17  package org.apache.juneau.commons.beanquery;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import java.util.*;
22  
23  /**
24   * One page of query results, plus the optional counts selected by the session's {@link CountPolicy}.
25   *
26   * <ul>
27   * 	<li>{@link #total()}: rows that pass the guards (before the search).
28   * 	<li>{@link #matched()}: rows that pass the guards <b>and</b> the search.
29   * </ul>
30   *
31   * <p>
32   * Under {@link CountPolicy#NONE} neither count is present.  Under {@link CountPolicy#MATCHED} only
33   * {@link #matched()} is.  Under {@link CountPolicy#BOTH} both are.  Counts are opt-in and independent of each
34   * other: a caller that did not ask for a count gets an empty {@link OptionalLong} back rather than an exception, so
35   * "was a count computed" and "read the count" are the same call.
36   *
37   * <h5 class='section'>Example:</h5>
38   * <p class='bjava'>
39   * 	Page&lt;Person&gt; <jv>page</jv> = <jv>session</jv>.find(BeanQuery.<jsm>create</jsm>().counts(CountRequest.<jsf>BOTH</jsf>).build());
40   * 	<jk>for</jk> (Person <jv>p</jv> : <jv>page</jv>.rows())
41   * 		<jsm>render</jsm>(<jv>p</jv>);
42   * 	<jv>page</jv>.total().ifPresent(<jv>total</jv> -&gt; <jsm>renderFooter</jsm>(<jv>page</jv>.matched().getAsLong() + <js>" of "</js> + <jv>total</jv>));
43   * </p>
44   *
45   * @param <T> The row type.
46   * @since 10.0.0
47   */
48  public final class Page<T> {
49  
50  	private final List<T> rows;
51  	private final Long total;
52  	private final Long matched;
53  
54  	private Page(List<T> rows, Long total, Long matched) {
55  		this.rows = copyRows(rows);
56  		this.total = total;
57  		this.matched = matched;
58  	}
59  
60  	// List.copyOf throws a bare, message-less NullPointerException on a null element; give it a clear cause instead.
61  	private static <T> List<T> copyRows(List<T> rows) {
62  		for (var i = 0; i < rows.size(); i++)
63  			if (rows.get(i) == null)
64  				throw new BeanQueryExecutionException(null, "Row %s is null.", i);
65  		return List.copyOf(rows);
66  	}
67  
68  	/**
69  	 * Creates a page without counts.
70  	 *
71  	 * @param <T> The row type.
72  	 * @param rows The rows.  Must not be <jk>null</jk> or contain <jk>null</jk>.
73  	 * @return A new page.
74  	 */
75  	public static <T> Page<T> of(List<T> rows) {
76  		return new Page<>(rows, null, null);
77  	}
78  
79  	/**
80  	 * Creates a page with both counts.
81  	 *
82  	 * @param <T> The row type.
83  	 * @param rows The rows.  Must not be <jk>null</jk> or contain <jk>null</jk>.
84  	 * @param total The number of rows that pass the guards.
85  	 * @param matched The number of rows that pass the guards and the search.
86  	 * @return A new page.
87  	 */
88  	public static <T> Page<T> of(List<T> rows, long total, long matched) {
89  		return new Page<>(rows, total, matched);
90  	}
91  
92  	/**
93  	 * Creates a page with only the matched count.
94  	 *
95  	 * @param <T> The row type.
96  	 * @param rows The rows.  Must not be <jk>null</jk> or contain <jk>null</jk>.
97  	 * @param matched The number of rows that pass the guards and the search.
98  	 * @return A new page.
99  	 */
100 	public static <T> Page<T> ofMatched(List<T> rows, long matched) {
101 		return new Page<>(rows, null, matched);
102 	}
103 
104 	/**
105 	 * Returns the rows of this page.
106 	 *
107 	 * @return An unmodifiable list (never <jk>null</jk>).
108 	 */
109 	public List<T> rows() {
110 		return rows;
111 	}
112 
113 	/**
114 	 * The number of rows in scope before the query's search (after any server-only guard).
115 	 *
116 	 * @return The total row count, or empty if it was not requested.
117 	 */
118 	public OptionalLong total() {
119 		return total == null ? OptionalLong.empty() : OptionalLong.of(total);
120 	}
121 
122 	/**
123 	 * The number of rows that survive the search, before any page limit.
124 	 *
125 	 * @return The matched row count, or empty if it was not requested.
126 	 */
127 	public OptionalLong matched() {
128 		return matched == null ? OptionalLong.empty() : OptionalLong.of(matched);
129 	}
130 
131 	@Override /* Object */
132 	public boolean equals(Object o) {
133 		return this == o || (o instanceof Page<?> p && eq(rows, p.rows) && eq(total, p.total) && eq(matched, p.matched));
134 	}
135 
136 	@Override /* Object */
137 	public int hashCode() {
138 		return h(rows, total, matched);
139 	}
140 
141 	@Override /* Object */
142 	public String toString() {
143 		var sb = new StringBuilder("Page[rows=").append(rows.size());
144 		if (total != null)
145 			sb.append(", total=").append(total);
146 		if (matched != null)
147 			sb.append(", matched=").append(matched);
148 		return sb.append(']').toString();
149 	}
150 }
