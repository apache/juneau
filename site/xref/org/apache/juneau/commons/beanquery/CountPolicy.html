17  package org.apache.juneau.commons.beanquery;
18  
19  /**
20   * Which counts a {@link BeanQuerySession#find(BeanQuery) find} / {@link BeanQuerySession#findValues(BeanQuery)
21   * findValues} call computes alongside its page of rows.
22   *
23   * <p>
24   * Set on {@link BeanQueryContext.Builder#countPolicy(CountPolicy)} and overridable per session with
25   * {@link BeanQuerySession.Builder#countPolicy(CountPolicy)}.  Counts can cost extra queries (two {@code COUNT(*)}
26   * statements on a SQL engine), so the default {@link #IF_REQUESTED} lets the caller opt in while the server keeps
27   * the final say.  {@link BeanQuerySession#count(BeanQuery)} always computes both counts regardless of the policy.
28   *
29   * <h5 class='section'>Example:</h5>
30   * <p class='bjava'>
31   * 	<jc>// Always return the matched count, whatever the caller asks for.</jc>
32   * 	InMemoryBeanQueryContext&lt;Person&gt; <jv>context</jv> = InMemoryBeanQueryContext
33   * 		.<jsm>create</jsm>(Person.<jk>class</jk>)
34   * 		.countPolicy(CountPolicy.<jsf>MATCHED</jsf>)
35   * 		.build();
36   * 	<jk>try</jk> (InMemoryBeanQuerySession&lt;Person&gt; <jv>session</jv> = <jv>context</jv>.getSession(<jv>people</jv>)) {
37   * 		<jk>long</jk> <jv>matched</jv> = <jv>session</jv>.find(<jv>query</jv>).matched().getAsLong();
38   * 	}
39   * </p>
40   *
41   * @since 10.0.0
42   */
43  public enum CountPolicy {
44  
45  	/** Never compute counts; {@link Page#total()} and {@link Page#matched()} are empty. */
46  	NONE,
47  
48  	/** Compute only the matched count (rows that pass the guards and the search). */
49  	MATCHED,
50  
51  	/** Compute both the total count (rows that pass the guards) and the matched count. */
52  	BOTH,
53  
54  	/**
55  	 * Let the query decide through its {@code opts}: {@code counts=matched} &rarr; {@link #MATCHED};
56  	 * {@code counts=both} or {@code counts=true} &rarr; {@link #BOTH}; anything else &rarr; {@link #NONE}.
57  	 *
58  	 * <p>
59  	 * A {@link ResolvedQuery#counts() resolved query} never carries this value; it is always replaced by one of the
60  	 * other three.
61  	 */
62  	IF_REQUESTED
63  }
