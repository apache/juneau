17  package org.apache.juneau.commons.beanquery;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import java.time.*;
22  import java.util.*;
23  
24  /**
25   * The mutable settings shared by {@link BeanQueryContext.Builder} and {@link BeanQuerySession.Builder}.
26   *
27   * <p>
28   * Builders mutate their own copy.  A built context or session holds a private copy that is never mutated again, so
29   * the built objects are immutable.  Package-private: only the base classes and {@link QueryResolver} read the fields.
30   */
31  final class QuerySettings {
32  
33  	/** Longest {@code $regex} pattern accepted (spec D7). */
34  	static final int MAX_REGEX_LENGTH = 256;
35  
36  	/** Declared columns in declaration order, each with its value type.  The allow-list for search, sort and view. */
37  	final LinkedHashMap<String,SearchType> columns = new LinkedHashMap<>();
38  
39  	/** Per-column operator sets that replace {@link #defaultOperators} for that column. */
40  	final LinkedHashMap<String,SearchOperatorSet> columnOperators = new LinkedHashMap<>();
41  
42  	SearchOperatorSet defaultOperators = SearchOperatorSet.standard();
43  	boolean allowRegex = true;
44  	Duration regexTimeout = Duration.ofMillis(50);
45  	CountPolicy countPolicy = CountPolicy.IF_REQUESTED;
46  	int defaultLimit = 100;
47  	Integer maxLimit = 1000;
48  	int maxSearchClauses = 64;
49  	int maxSortKeys = 8;
50  	int maxSearchLength = 4096;
51  	int maxExpressionDepth = 16;
52  
53  	/**
54  	 * Returns an independent copy (maps copied; operator sets are immutable and shared).
55  	 *
56  	 * @return A new settings object.
57  	 */
58  	QuerySettings copy() {
59  		var s = new QuerySettings();
60  		s.columns.putAll(columns);
61  		s.columnOperators.putAll(columnOperators);
62  		s.defaultOperators = defaultOperators;
63  		s.allowRegex = allowRegex;
64  		s.regexTimeout = regexTimeout;
65  		s.countPolicy = countPolicy;
66  		s.defaultLimit = defaultLimit;
67  		s.maxLimit = maxLimit;
68  		s.maxSearchClauses = maxSearchClauses;
69  		s.maxSortKeys = maxSortKeys;
70  		s.maxSearchLength = maxSearchLength;
71  		s.maxExpressionDepth = maxExpressionDepth;
72  		return s;
73  	}
74  
75  	/**
76  	 * Returns the operator set used to parse the specified column's expressions (including {@code $regex}; the regex
77  	 * gate is a separate §5 check so it can report its own message).
78  	 *
79  	 * @param column The column name.
80  	 * @return The column's set, else the default set.
81  	 */
82  	SearchOperatorSet operators(String column) {
83  		return columnOperators.getOrDefault(column, defaultOperators);
84  	}
85  
86  	/**
87  	 * Checks the cross-setting rules that single setters cannot check.
88  	 *
89  	 * @throws IllegalStateException If no column is declared, or {@code maxLimit} is below {@code defaultLimit}.
90  	 */
91  	void validate() {
92  		if (columns.isEmpty())
93  			throw isex("At least one column must be declared.");
94  		if (maxLimit != null && maxLimit < defaultLimit)
95  			throw isex("maxLimit (%s) must not be less than defaultLimit (%s).", maxLimit, defaultLimit);
96  	}
97  
98  	static int positive(String name, int value) {
99  		req(value > 0, "%s must be positive: %s", name, value);
100 		return value;
101 	}
102 
103 	static Integer positiveOrNull(String name, Integer value) {
104 		req(value == null || value > 0, "%s must be positive or null: %s", name, value);
105 		return value;
106 	}
107 
108 	static Duration positive(String name, Duration value) {
109 		reqnn(name, value);
110 		req(! value.isNegative() && ! value.isZero(), "%s must be positive: %s", name, value);
111 		return value;
112 	}
113 }
