17  package org.apache.juneau.commons.beanquery;
18  
19  import java.util.*;
20  
21  /**
22   * A resolved, engine-neutral search tree (design §7) &mdash; the {@code ResolvedQuery.filter()} replacement for
23   * the earlier flat {@code Term}/{@code terms()} list.
24   *
25   * <p>
26   * {@link QueryResolver} builds this tree from a request's raw {@code search} string: it parses the string with
27   * {@link SearchParser} into a {@link SearchItem} tree, then resolves every leaf's column (against the context's
28   * allow-list) and expression (with {@link SearchExpressionParser}, using that column's operator set), carrying the
29   * column's {@link SearchType} along for the engine to use. An empty search resolves to an empty {@link Filter.And}
30   * (design §7) &mdash; not <jk>null</jk> &mdash; so every engine can treat "no search" as just another tree to
31   * evaluate, one that matches every row.
32   *
33   * <p>
34   * Both engines walk this same shape: {@link InMemoryBeanQuerySession} evaluates it recursively against one row;
35   * {@code SqlSearchCompiler} (not linkable from here &mdash; it lives in the separate {@code juneau-beanquery-sql}
36   * module, which this one does not depend on) renders it to a parenthesized SQL boolean expression. Neither engine
37   * ever sees a raw request string or an unresolved column name.
38   *
39   * <h5 class='section'>Example:</h5>
40   * <p class='bjava'>
41   * 	<jc>// ageExpr/nameExpr/emailExpr are already-resolved SearchExpressions, e.g. from SearchExpressionParser.parse()</jc>
42   * 	<jc>// (age &gt; 21) AND (name CONTAINS "bob" OR email CONTAINS "bob")</jc>
43   * 	Filter <jv>f</jv> = <jk>new</jk> Filter.And(List.<jsm>of</jsm>(
44   * 		<jk>new</jk> Filter.Leaf(<js>"age"</js>, ageExpr, SearchType.<jsf>NUMERIC</jsf>),
45   * 		<jk>new</jk> Filter.Or(List.<jsm>of</jsm>(
46   * 			<jk>new</jk> Filter.Leaf(<js>"name"</js>, nameExpr, SearchType.<jsf>TEXT</jsf>),
47   * 			<jk>new</jk> Filter.Leaf(<js>"email"</js>, emailExpr, SearchType.<jsf>TEXT</jsf>)))));
48   * </p>
49   *
50   * <p>
51   * Unlike the package-private {@link SearchItem} it mirrors, {@code Filter} is public: {@code SqlSearchCompiler}
52   * (above) lives in {@code juneau-beanquery-sql}, a separate module downstream of this one, and must be able to
53   * pattern-match over this tree directly.
54   *
55   * @since 10.0.0
56   */
57  public sealed interface Filter {
58  
59  	/**
60  	 * One resolved {@code column=expression} leaf, with the column's {@link SearchType} carried alongside for the
61  	 * engine.
62  	 *
63  	 * <p>
64  	 * Note: {@link SearchExpression} has no value-based {@code equals}/{@code hashCode} of its own, so two
65  	 * {@code Leaf} instances built from separately-parsed-but-identical expression text are <b>not</b>
66  	 * {@link #equals(Object) equal} unless they share the same {@code expression} instance.
67  	 */
68  	record Leaf(String column, SearchExpression expression, SearchType type) implements Filter {}
69  
70  	/** An AND of two or more (or, at the top level, zero) sub-filters.  An empty {@code And} matches every row. */
71  	record And(List<Filter> items) implements Filter {
72  		public And(List<Filter> items) {
73  			this.items = List.copyOf(items);
74  		}
75  	}
76  
77  	/** An OR of two or more sub-filters. */
78  	record Or(List<Filter> items) implements Filter {
79  		public Or(List<Filter> items) {
80  			this.items = List.copyOf(items);
81  		}
82  	}
83  
84  	/** The negation of exactly one sub-filter. */
85  	record Not(Filter item) implements Filter {}
86  }
