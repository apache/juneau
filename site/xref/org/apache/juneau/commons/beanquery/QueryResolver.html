24  /**
25   * The validation pipeline (spec §5): turns a raw {@link BeanQuery} into a {@link ResolvedQuery}, or throws
26   * {@link BeanQuerySyntaxException} with a caller-facing message.  Checks run in the spec's order.
27   */
28  final class QueryResolver {
29  
30  	private QueryResolver() {}
31  
32  	/** Built-in leaf operators whose arguments are typed by column type (design #4 section 5). */
33  	private static final Set<String> VALUE_OPS = Set.of("$eq", "$ne", "$in", "$gt", "$gte", "$lt", "$lte", "$between");
34  
35  	static ResolvedQuery resolve(BeanQuery query, QuerySettings s, BeanQueryContext<?> origin) {
36  		return resolve(query, s, origin, Clock.systemUTC());
37  	}
38  
39  	// The clock is read exactly once, here; everything below takes the captured Instant so every relative-duration
40  	// literal in one query (e.g. two leaves, or both ends of $between(-7d,-1d)) resolves against the identical instant.
41  	static ResolvedQuery resolve(BeanQuery query, QuerySettings s, BeanQueryContext<?> origin, Clock clock) {
42  		reqnn("query", query);
43  		reqnn("clock", clock);
44  		checkLength(BeanQuerySyntaxException.Code.SEARCH_TOO_LONG, "Search", query.getSearch(), s.maxSearchLength);
45  		checkLength(BeanQuerySyntaxException.Code.SEARCH_TOO_LONG, "Sort", query.getSort(), s.maxSearchLength);
46  		checkLength(BeanQuerySyntaxException.Code.SEARCH_TOO_LONG, "View", query.getView(), s.maxSearchLength);
47  		checkLength(BeanQuerySyntaxException.Code.OPTS_TOO_LONG, "Options", query.getOpts(), s.maxSearchLength);
48  		var requestTime = clock.instant();
49  		var filter = filter(query.getSearch(), s, requestTime);
50  		var sort = sort(query.getSort(), s);
51  		var view = view(query.getView(), s);
52  		var position = query.getPosition();
53  		if (position != null && position < 0)
54  			throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.NEGATIVE_POSITION, "Position must not be negative.");
55  		return new ResolvedQuery(filter, sort, view, position == null ? 0 : position, limit(query.getLimit(), s), counts(query.getOpts(), s), origin);
56  	}
57  
58  	private static void checkLength(BeanQuerySyntaxException.Code code, String what, String value, int max) {
59  		if (value != null && value.length() > max)
60  			throw new BeanQuerySyntaxException(code, "%s string exceeds %s characters.", what, max);
61  	}

63  	// Rows 2-6: clause count, allowed column, parse + depth, regex gate, operator/type.
64  	private static Filter filter(String search, QuerySettings s, Instant requestTime) {
65  		var items = SearchParser.parse(search);
66  		if (FilterResolver.countLeaves(items) > s.maxSearchClauses)
67  			throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.TOO_MANY_CLAUSES, "Too many search clauses (max %s).", s.maxSearchClauses);
68  		var groupDepths = new ArrayList<Integer>();
69  		collectGroupDepths(items, 0, groupDepths);
70  		var leafIndex = new int[] {0};
71  		return FilterResolver.resolve(items, (column, expression) -> {
72  			var type = s.columns.get(column);
73  			if (type == null)
74  				throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.UNKNOWN_COLUMN, "Unknown column '%s' in search.", column);
75  			var depthIdx = leafIndex[0]++;
76  			if (expression.isBlank())
77  				return null;
78  			var tree = SearchExpressionParser.parse(expression, s.operators(column));
79  			var totalDepth = groupDepths.get(depthIdx) + depth(tree);
80  			if (totalDepth > s.maxExpressionDepth)
81  				throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.TOO_DEEP, "Search expression nested too deeply (max %s).", s.maxExpressionDepth);
82  			checkRegex(tree, s.allowRegex);
83  			checkTypes(tree, type, column);
84  			if (type == SearchType.NUMERIC || type == SearchType.BOOLEAN || type == SearchType.TIMESTAMP)
85  				tree = tree.isLiteral() ? typeLiteral(tree, type, column, requestTime) : typeTree(tree, type, column, requestTime);
86  			return new Filter.Leaf(column, tree, type);
87  		});
88  	}

90  	// Depth = group nesting + expression nesting (design §7 row 4). Walks the raw SearchItem tree, in the same
91  	// left-to-right pre-order FilterResolver.resolve visits leaves, recording each leaf's group-nesting depth so the
92  	// leafResolver callback above can look its own leaf's depth up by position once it has parsed the expression.
93  	private static void collectGroupDepths(List<SearchItem> items, int groupDepth, List<Integer> out) {
94  		for (var item : items) {
95  			if (item instanceof SearchItem.Leaf)
96  				out.add(groupDepth);
97  			else if (item instanceof SearchItem.And and)
98  				collectGroupDepths(and.items(), groupDepth + 1, out);
99  			else if (item instanceof SearchItem.Or or)
100 				collectGroupDepths(or.items(), groupDepth + 1, out);
101 			else if (item instanceof SearchItem.Not not)
102 				collectGroupDepths(List.of(not.item()), groupDepth + 1, out);
103 		}
104 	}

106 	private static int depth(SearchExpression node) {
107 		if (node.isLiteral())
108 			return 0;
109 		var d = 0;
110 		for (var a : node.args())
111 			d = Math.max(d, depth(a));
112 		return d + 1;
113 	}
114 
115 	private static void checkRegex(SearchExpression node, boolean allowRegex) {
116 		if (node.isLiteral())
117 			return;
118 		if ("$regex".equals(node.name())) {
119 			if (! allowRegex)
120 				throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.REGEX_DISABLED, "Regex search is not enabled for this list.");
121 			if (node.literalArgs().get(0).length() > QuerySettings.MAX_REGEX_LENGTH)
122 				throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.REGEX_TOO_LONG, "Regex pattern exceeds %s characters.", QuerySettings.MAX_REGEX_LENGTH);
123 		}
124 		for (var a : node.args())
125 			checkRegex(a, allowRegex);
126 	}
127 
128 	private static void checkTypes(SearchExpression node, SearchType type, String column) {
129 		if (node.isLiteral())
130 			return;
131 		var op = node.operator();
132 		if (! op.appliesTo(type))
133 			throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.OPERATOR_TYPE, "Operator %s does not apply to %s column '%s'.", op.name(), type.wire(), column);
134 		for (var a : node.args())
135 			checkTypes(a, type, column);
136 	}
137 
138 	// Types the leaves design #4 section 5 says to type: bare literals (called directly on a literal tree) and every
139 	// literal argument of a value operator or a typedArgs(true) custom operator.  Runs after checkTypes so
140 	// OPERATOR_TYPE wins over BAD_VALUE.  Everything else ($contains/$prefix/$regex/$eqic/$blank arguments, a custom
141 	// operator that did not opt in) is returned unchanged and stays text.
142 	private static SearchExpression typeTree(SearchExpression node, SearchType type, String column, Instant requestTime) {
143 		var op = node.operator();
144 		if (op.isCombinator()) {
145 			var newArgs = node.args().stream().map(a -> a.isLiteral() ? typeLiteral(a, type, column, requestTime) : typeTree(a, type, column, requestTime)).toList();
146 			return SearchExpression.func(op, newArgs);
147 		}
148 		if (VALUE_OPS.contains(op.name()) || (op.isCustom() && op.typedArgs())) {
149 			var newArgs = node.args().stream().map(a -> a.isLiteral() ? typeLiteral(a, type, column, requestTime) : a).toList();
150 			return SearchExpression.func(op, newArgs);
151 		}
152 		return node;
153 	}
154 
155 	private static SearchExpression typeLiteral(SearchExpression literal, SearchType type, String column, Instant requestTime) {
156 		try {
157 			return literal.withTypedValue(type.parse(literal.value(), requestTime));
158 		} catch (BeanQuerySyntaxException e) {
159 			throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.BAD_VALUE, "Value '%s' is not a valid %s for column '%s'.", literal.value(), type.wire(), column);
160 		}
161 	}

163 	// Row 7.
164 	private static List<ResolvedQuery.SortKey> sort(String sort, QuerySettings s) {
165 		var keys = new ArrayList<ResolvedQuery.SortKey>();
166 		if (ib(sort))
167 			return keys;
168 		for (var raw : sort.split(",")) {
169 			var t = raw.strip();  // An empty key passes through the suffix checks unchanged and is skipped below.
170 			var desc = false;
171 			if (t.regionMatches(true, t.length() - 5, ":desc", 0, 5)) {
172 				desc = true;
173 				t = t.substring(0, t.length() - 5).strip();
174 			} else if (t.regionMatches(true, t.length() - 4, ":asc", 0, 4)) {
175 				t = t.substring(0, t.length() - 4).strip();
176 			} else if (t.endsWith("-")) {
177 				desc = true;
178 				t = t.substring(0, t.length() - 1).strip();
179 			} else if (t.endsWith("+")) {
180 				t = t.substring(0, t.length() - 1).strip();
181 			}
182 			if (t.isEmpty())
183 				continue;
184 			if (keys.size() == s.maxSortKeys)
185 				throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.TOO_MANY_SORT_KEYS, "Too many sort keys (max %s).", s.maxSortKeys);
186 			var type = s.columns.get(t);
187 			if (type == null)
188 				throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.UNKNOWN_COLUMN, "Unknown sort column '%s'.", t);
189 			keys.add(new ResolvedQuery.SortKey(t, type, desc));
190 		}
191 		return keys;
192 	}

194 	// Row 8.
195 	private static List<String> view(String view, QuerySettings s) {
196 		var cols = new LinkedHashSet<String>();
197 		if (inb(view)) {
198 			for (var raw : view.split(",")) {
199 				var t = raw.strip();
200 				if (t.isEmpty())
201 					continue;
202 				if (! s.columns.containsKey(t))
203 					throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.UNKNOWN_COLUMN, "Unknown view column '%s'.", t);
204 				cols.add(t);
205 			}
206 		}
207 		return List.copyOf(cols.isEmpty() ? s.columns.keySet() : cols);
208 	}
209 
210 	// Row 9 (limit half): missing -> default; negative -> cap; above cap -> cap; null only when uncapped.
211 	private static Integer limit(Integer limit, QuerySettings s) {
212 		if (limit == null)
213 			return s.defaultLimit;
214 		if (limit < 0)
215 			return s.maxLimit;
216 		return s.maxLimit == null ? limit : Math.min(limit, s.maxLimit);
217 	}
218 
219 	// Row 10.
220 	private static CountPolicy counts(String opts, QuerySettings s) {
221 		if (s.countPolicy != CountPolicy.IF_REQUESTED)
222 			return s.countPolicy;
223 		var v = ClauseParser.parse(opts).get("counts");
224 		if (v != null)
225 			v = v.strip();
226 		if (eqic(v, "matched"))
227 			return CountPolicy.MATCHED;
228 		if (eqic(v, "both") || eqic(v, "true"))
229 			return CountPolicy.BOTH;
230 		return CountPolicy.NONE;
231 	}
