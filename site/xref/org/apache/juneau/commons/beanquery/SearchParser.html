17  package org.apache.juneau.commons.beanquery;
18  
19  import java.util.*;
20  import java.util.stream.*;
21  
22  /**
23   * Parses and renders the {@link BeanQuery#getSearch() search} string grammar (design §4):
24   *
25   * <pre>
26   *   search = item ("," item)*
27   *   item   = group | leaf
28   *   group  = "$or(" item ("," item)+ ")" | "$and(" item ("," item)+ ")" | "$not(" item ")"
29   *   leaf   = key "=" expression
30   *   key    = backslash-unescaped column name (a backslash before any character is accepted on parse;
31   *            {@link ClauseParser#escape(String)} is the narrower inverse used on render, only ever escaping
32   *            "\", ",", "=", and a leading "$")
33   * </pre>
34   *
35   * <p>
36   * A leaf's {@code expression} is left unparsed here — it is the per-column {@code $}-expression grammar, unchanged
37   * by this design; {@code SearchExpressionParser} parses it later, once the column and its operator set are known.
38   * Group-vs-leaf disambiguation is an exact text match on {@code $or(}/{@code $and(}/{@code $not(} at the start of
39   * an item. Structural errors ({@code $or}/{@code $and} needing 2+ items, {@code $not} needing exactly 1, unbalanced
40   * group parens) are caught here; per-expression errors (unterminated quote, unknown operator, ...) surface later
41   * from {@code SearchExpressionParser} / {@code QueryResolver}. A blank leaf ({@code key=} with an empty expression)
42   * is preserved here, not dropped, as a {@link SearchItem.Leaf} with an empty expression &mdash; so {@code
43   * QueryResolver} can validate its column first (an unknown column still throws, even with a blank value) and only
44   * then drop the blank leaf during resolution (design §7), via {@code FilterResolver}'s null-as-drop-signal
45   * convention. A group left with no items after that drop is dropped in turn.
46   *
47   * <p>
48   * {@link #parse(String)} also rejects a column repeated among the returned <em>top-level</em> items (design D2) —
49   * the same column repeated inside a group, or once at top level and again inside a top-level group, is allowed.
50   * Both {@code QueryResolver} (resolving a request's search) and {@code BeanQuery.copy()} (re-parsing an existing
51   * query's search into a builder) go through this one check.
52   */
53  final class SearchParser {
54  
55  	private final String s;
56  	private final int len;
57  	private int pos;
58  
59  	private SearchParser(String s) {
60  		this.s = s;
61  		this.len = s.length();
62  	}
63  
64  	/**
65  	 * Parses a whole search string into its top-level items, in order.
66  	 *
67  	 * @param search The raw search string.  <jk>null</jk> or blank yields an empty list (matches every row).
68  	 * @return The top-level items (AND-ed together), never <jk>null</jk>.
69  	 * @throws BeanQuerySyntaxException On unbalanced group parens, a group arity violation, or a column repeated
70  	 * 	among the top-level items.
71  	 */
72  	static List<SearchItem> parse(String search) {
73  		if (search == null || search.isBlank())
74  			return List.of();
75  		var p = new SearchParser(search);
76  		List<SearchItem> items = new ArrayList<>();
77  		p.skipWs();
78  		while (! p.atEnd()) {
79  			SearchItem item = p.parseItem();
80  			if (item != null)
81  				items.add(item);
82  			p.skipWs();
83  			if (! p.atEnd()) {
84  				p.expect(',');
85  				p.skipWs();
86  			}
87  		}
88  		checkNoDuplicateTopLevelColumns(items);
89  		return items;
90  	}
91  
92  	private static void checkNoDuplicateTopLevelColumns(List<SearchItem> items) {
93  		Set<String> seen = new HashSet<>();
94  		for (SearchItem item : items)
95  			if (item instanceof SearchItem.Leaf leaf && ! seen.add(leaf.column()))
96  				throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.DUPLICATE_COLUMN, "Column '%s' appears more than once in search; combine with $and(...).", leaf.column());
97  	}
98  
99  	/** Renders a top-level item list back to its wire search string (used by {@code BeanQuery.copy()}/{@code build()}). */
100 	static String render(List<SearchItem> items) {
101 		return items.stream().map(SearchParser::renderItem).collect(Collectors.joining(","));
102 	}
103 
104 	private static String renderItem(SearchItem item) {
105 		if (item instanceof SearchItem.Leaf l)
106 			return ClauseParser.escape(l.column()) + "=" + l.expression();
107 		if (item instanceof SearchItem.And a)
108 			return AND_OPEN + a.items().stream().map(SearchParser::renderItem).collect(Collectors.joining(",")) + ")";
109 		if (item instanceof SearchItem.Or o)
110 			return OR_OPEN + o.items().stream().map(SearchParser::renderItem).collect(Collectors.joining(",")) + ")";
111 		if (item instanceof SearchItem.Not n)
112 			return NOT_OPEN + renderItem(n.item()) + ")";
113 		throw new IllegalStateException("Unreachable: " + item.getClass());
114 	}
115 
116 	// -- parsing ------------------------------------------------------------------------------------------------
117 
118 	private static final String AND_OPEN = "$and(";
119 	private static final String OR_OPEN = "$or(";
120 	private static final String NOT_OPEN = "$not(";
121 
122 	private enum GroupKind { AND, OR, NOT }
123 
124 	/** Parses one item slot. Returns <jk>null</jk> only for a syntactically-missing slot (e.g. a bare comma with no
125 	 * {@code key=} at all); a blank-valued leaf still comes back as a {@link SearchItem.Leaf} with an empty
126 	 * expression. */
127 	private SearchItem parseItem() {
128 		skipWs();
129 		if (atEnd() || peek() == ',' || peek() == ')')
130 			return null;
131 		if (startsWithAt(OR_OPEN))
132 			return parseGroup(OR_OPEN, GroupKind.OR);
133 		if (startsWithAt(AND_OPEN))
134 			return parseGroup(AND_OPEN, GroupKind.AND);
135 		if (startsWithAt(NOT_OPEN))
136 			return parseGroup(NOT_OPEN, GroupKind.NOT);
137 		return parseLeaf();
138 	}
139 
140 	private SearchItem parseGroup(String prefix, GroupKind kind) {
141 		pos += prefix.length();
142 		skipWs();
143 		List<SearchItem> children = new ArrayList<>();
144 		int rawCount = 0;
145 		if (! atEnd() && peek() == ')') {
146 			pos++;
147 			validateArity(kind, 0);
148 			return null;
149 		}
150 		var closed = false;
151 		while (! closed) {
152 			rawCount++;
153 			SearchItem child = parseItem();
154 			if (child != null)
155 				children.add(child);
156 			skipWs();
157 			if (atEnd())
158 				throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.UNBALANCED, "Unbalanced group in search.");
159 			char c = peek();
160 			if (c != ')' && c != ',')
161 				throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.UNBALANCED, "Unbalanced group in search.");
162 			pos++;
163 			closed = c == ')';
164 			if (! closed)
165 				skipWs();
166 		}
167 		validateArity(kind, rawCount);
168 		if (children.isEmpty())
169 			return null;
170 		return switch (kind) {
171 			case AND -> new SearchItem.And(children);
172 			case OR -> new SearchItem.Or(children);
173 			case NOT -> new SearchItem.Not(children.get(0));
174 		};
175 	}
176 
177 	private void validateArity(GroupKind kind, int rawCount) {
178 		switch (kind) {
179 			case AND -> { if (rawCount < 2) throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.GROUP_ARITY, "$and takes at least two items."); }
180 			case OR -> { if (rawCount < 2) throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.GROUP_ARITY, "$or takes at least two items."); }
181 			case NOT -> { if (rawCount != 1) throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.GROUP_ARITY, "$not takes exactly one item."); }
182 		}
183 	}
184 
185 	private SearchItem parseLeaf() {
186 		String column = parseKey();
187 		String expression = scanBalancedUntilDelimiter();
188 		return new SearchItem.Leaf(column, expression);
189 	}
190 
191 	/** Reads a backslash-escaped key up to (and consuming) the first unescaped {@code '='}. */
192 	private String parseKey() {
193 		var sb = new StringBuilder();
194 		while (! atEnd()) {
195 			char c = s.charAt(pos);
196 			if (c == '\\' && pos + 1 < len) {
197 				sb.append(s.charAt(pos + 1));
198 				pos += 2;
199 			} else if (c == '=') {
200 				pos++;
201 				return sb.toString();
202 			} else {
203 				sb.append(c);
204 				pos++;
205 			}
206 		}
207 		throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.UNBALANCED, "Missing '=' after column '%s' in search.", sb);
208 	}
209 
210 	/**
211 	 * Reads raw expression text up to (not consuming) the next depth-0, non-quoted {@code ','} or {@code ')'}, or
212 	 * end of string — tracking paren depth and quote state (with design D5 doubled-quote-is-literal handling) only
213 	 * well enough to find this leaf's boundary.  It does not itself validate that the expression's own quotes and
214 	 * parens balance; an unterminated quote inside one leaf's expression is caught later by
215 	 * {@code SearchExpressionParser} when the expression is actually parsed.
216 	 */
217 	@SuppressWarnings({
218 		"java:S3776" // Straight-line character scanner; splitting it would obscure the single quote/paren state machine.
219 	})
220 	private String scanBalancedUntilDelimiter() {
221 		int start = pos;
222 		int depth = 0;
223 		Character quote = null;
224 		while (! atEnd()) {
225 			char c = s.charAt(pos);
226 			if (quote == null && depth == 0 && (c == ')' || c == ','))
227 				break;
228 			if (quote != null) {
229 				if (c != quote)
230 					pos++;
231 				else if (pos + 1 < len && s.charAt(pos + 1) == quote)
232 					pos += 2;
233 				else {
234 					quote = null;
235 					pos++;
236 				}
237 			} else if (c == '"' || c == '\'') {
238 				quote = c;
239 				pos++;
240 			} else {
241 				if (c == '(')
242 					depth++;
243 				else if (c == ')')
244 					depth--;
245 				pos++;
246 			}
247 		}
248 		return s.substring(start, pos).strip();
249 	}
250 
251 	// -- cursor helpers -------------------------------------------------------------------------------------------
252 
253 	private boolean atEnd() {
254 		return pos >= len;
255 	}
256 
257 	private char peek() {
258 		return s.charAt(pos);
259 	}
260 
261 	private void skipWs() {
262 		while (! atEnd() && Character.isWhitespace(peek()))
263 			pos++;
264 	}
265 
266 	private void expect(char c) {
267 		if (atEnd())
268 			throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.UNBALANCED, "Unexpected end of search string; expected '%s'.", c);
269 		if (peek() != c)
270 			throw new BeanQuerySyntaxException(BeanQuerySyntaxException.Code.UNBALANCED, "Expected '%s' in search but found '%s'.", c, peek());
271 		pos++;
272 	}
273 
274 	private boolean startsWithAt(String literal) {
275 		return s.regionMatches(pos, literal, 0, literal.length());
276 	}
277 }
