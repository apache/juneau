26  /**
27   * The in-memory evaluation of a parsed {@link SearchExpression} against a cell value, plus the type-aware value
28   * coercion and ordered comparison the in-memory context also uses for sorting.
29   *
30   * <p>
31   * This holds the leaf semantics of the {@code $}-language (design §5.1) for the built-in operators; a custom operator
32   * is evaluated through its {@link SearchOperator#predicate() predicate}.  It is the behavior reference the
33   * {@code juneau-search.js} client engine mirrors.
34   */
35  final class InMemoryMatch {
36  
37  	private InMemoryMatch() {}
38  
39  	// -----------------------------------------------------------------------------------------------------------------
40  	// Evaluation
41  	// -----------------------------------------------------------------------------------------------------------------
42  
43  	static boolean matches(SearchExpression node, Object cell, SearchType type, RegexBudget budget) {
44  		if (node.isLiteral())
45  			return matchBare(node.typedValue(), node.isQuoted(), cell, type);
46  		var op = node.operator();
47  		if (op.isCombinator())
48  			return matchCombinator(node, cell, type, budget);
49  		if (op.isCustom()) {
50  			var p = op.predicate();
51  			if (p == null)
52  				throw iaex("Custom operator '%s' has no in-memory predicate; it cannot be evaluated in memory.", op.name());
53  			return p.test(cell, node.literalArgs());
54  		}
55  		return matchLeaf(node, cell, type, budget);
56  	}
57  
58  	/**
59  	 * Convenience overload for callers that don't already have a {@link RegexBudget} for the call (notably tests):
60  	 * builds one with the default {@code $regex} timeout.  Production row-matching goes through
61  	 * {@link #matches(SearchExpression, Object, SearchType, RegexBudget)} via {@link FilterEvaluator}, reusing one
62  	 * budget across every row/column in the call.
63  	 */
64  	static boolean matches(SearchExpression node, Object cell, SearchType type) {
65  		return matches(node, cell, type, new RegexBudget(new QuerySettings().regexTimeout));
66  	}
67  
68  	private static boolean matchCombinator(SearchExpression node, Object cell, SearchType type, RegexBudget budget) {
69  		var name = node.name();
70  		if ("$and".equals(name))
71  			return node.args().stream().allMatch(a -> matches(a, cell, type, budget));
72  		if ("$or".equals(name))
73  			return node.args().stream().anyMatch(a -> matches(a, cell, type, budget));
74  		return ! matches(node.args().get(0), cell, type, budget);  // $not — arity guaranteed 1 by the parser.
75  	}
76  
77  	private static boolean matchLeaf(SearchExpression node, Object cell, SearchType type, RegexBudget budget) {
78  		var args = node.typedArgs();  // Value operators read typed values; text operators get the original strings (never typed).
79  		switch (node.operator().name()) {
80  			case "$eq":       return eqExact(cell, args.get(0), type);
81  			case "$eqic":     return eqCi(cell, args.get(0), type);
82  			case "$ne":       return isBlankCell(cell) || args.stream().noneMatch(a -> eqExact(cell, a, type));
83  			case "$in":       return args.stream().anyMatch(a -> eqExact(cell, a, type));
84  			case "$contains": return contains(cell, str(args.get(0)));
85  			case "$prefix":   return prefix(cell, str(args.get(0)), type);
86  			case "$regex":    { var v = str(cell); return v != null && budget.matches(node, v); }
87  			case "$blank":    return isBlankCell(cell);
88  			case "$gt":       { var c = cmp(cell, args.get(0), type); return c != null && c > 0; }
89  			case "$gte":      { var c = cmp(cell, args.get(0), type); return c != null && c >= 0; }
90  			case "$lt":       { var c = cmp(cell, args.get(0), type); return c != null && c < 0; }
91  			case "$lte":      { var c = cmp(cell, args.get(0), type); return c != null && c <= 0; }
92  			case "$between":  return between(cell, args, type);
93  			// SearchOperator.Builder#build() now rejects renaming a copy() of a built-in, so that route can no
94  			// longer reach this switch under an unrecognized name. The only way left to construct a custom=false
95  			// operator this switch doesn't recognize is the package-private SearchOperator#builtin(...) factory
96  			// used directly (see InMemoryMatch_Test's i01/i02) - fail clearly rather than silently misinterpreting
97  			// it as $between.
98  			default:          throw iaex("Unsupported operator '%s' for in-memory evaluation.", node.operator().name());
99  		}
100 	}

102 	/** A bare pattern token.  Quoted bare literals never get wildcard interpretation (§3 "Quoting rules"). */
103 	private static boolean matchBare(Object value, boolean quoted, Object cell, SearchType type) {
104 		var token = str(value);
105 		switch (type) {
106 			case TEXT, ID: return (quoted || ! hasWildcard(token)) ? contains(cell, token) : wildcardMatches(cell, token);
107 			case ENUM:    return (quoted || ! hasWildcard(token)) ? eqCi(cell, token, type) : wildcardMatches(cell, token);
108 			case VERSION: return prefix(cell, token, type);
109 			default:      return eqCi(cell, value, type);  // numeric / timestamp / boolean — not wildcards, quoting is a no-op.
110 		}
111 	}

117 	/** Case-insensitive typed equality (bare-token default and {@code $eqic}). */
118 	private static boolean eqCi(Object cell, Object arg, SearchType type) {
119 		switch (type) {
120 			case TEXT, ID, ENUM: { var v = str(cell); return v != null && eqic(v, str(arg)); }
121 			case BOOLEAN:  return boolEquals(cell, arg);
122 			default:      { var c = cmp(cell, arg, type); return c != null && c == 0; }  // numeric / timestamp / version
123 		}
124 	}
125 
126 	/** Exact (case-sensitive) typed equality ({@code $eq} / {@code $ne} / {@code $in}, D2). */
127 	private static boolean eqExact(Object cell, Object arg, SearchType type) {
128 		switch (type) {
129 			case TEXT, ID, ENUM: { var v = str(cell); return v != null && eq(v, str(arg)); }
130 			case BOOLEAN:  return boolEquals(cell, arg);
131 			default:      { var c = cmp(cell, arg, type); return c != null && c == 0; }  // numeric / timestamp / version
132 		}
133 	}

135 	private static boolean contains(Object cell, String arg) {
136 		var v = str(cell);
137 		return v != null && v.toLowerCase(Locale.ROOT).contains(arg.toLowerCase(Locale.ROOT));
138 	}

140 	@SuppressWarnings({
141 		"javabugs:S2259" // Only caller is matchBare(), whose literal node value comes from the parser's decodeLiteral() and is never null.
142 	})
143 	private static boolean hasWildcard(String token) {
144 		return token.indexOf('*') >= 0 || token.indexOf('?') >= 0;
145 	}
146 
147 	/** A bare text/id/enum token with a wildcard: a case-insensitive whole-string glob match. */
148 	private static boolean wildcardMatches(Object cell, String token) {
149 		var v = str(cell);
150 		if (v == null)
151 			return false;
152 		return globMatches(v.toLowerCase(Locale.ROOT), token.toLowerCase(Locale.ROOT));
153 	}
154 
155 	/**
156 	 * Whole-string glob match (D1): {@code *} matches any sequence of characters (including none); {@code ?} matches
157 	 * exactly one character; every other character must match literally.  Case sensitivity is the caller's
158 	 * responsibility (fold both arguments first).
159 	 *
160 	 * <p>
161 	 * This is the classic greedy two-pointer glob algorithm, worst case {@code O(text.length() * pattern.length())}
162 	 * with no backtracking blowup — unlike a regex built from the same pattern (a run of {@code a*} segments can
163 	 * backtrack exponentially via {@link Pattern}), this always terminates in bounded time regardless of input.
164 	 */
165 	private static boolean globMatches(String text, String pattern) {
166 		var ti = 0;
167 		var pi = 0;
168 		var starIdx = -1;
169 		var starTi = 0;
170 		while (ti < text.length()) {
171 			if (pi < pattern.length() && pattern.charAt(pi) == '*') {
172 				starIdx = pi++;
173 				starTi = ti;
174 			} else if (pi < pattern.length() && (pattern.charAt(pi) == '?' || pattern.charAt(pi) == text.charAt(ti))) {
175 				pi++;
176 				ti++;
177 			} else if (starIdx >= 0) {
178 				pi = starIdx + 1;
179 				ti = ++starTi;
180 			} else {
181 				return false;
182 			}
183 		}
184 		while (pi < pattern.length() && pattern.charAt(pi) == '*')
185 			pi++;
186 		return pi == pattern.length();
187 	}

189 	private static boolean prefix(Object cell, String arg, SearchType type) {
190 		if (type == SearchType.VERSION)
191 			return versionPrefix(str(cell), arg);
192 		var v = str(cell);
193 		return v != null && v.toLowerCase(Locale.ROOT).startsWith(arg.toLowerCase(Locale.ROOT));
194 	}
195 
196 	private static boolean between(Object cell, List<Object> args, SearchType type) {
197 		var lo = cmp(cell, args.get(0), type);
198 		var hi = cmp(cell, args.get(1), type);
199 		return lo != null && hi != null && lo >= 0 && hi <= 0;
200 	}
201 
202 	/** Ordered comparison of {@code cell} vs {@code arg}, or <jk>null</jk> when either side is not comparable. */
203 	private static Integer cmp(Object cell, Object arg, SearchType type) {
204 		switch (type) {
205 			case NUMERIC: {
206 				var a = decimal(cell);
207 				var b = decimal(arg);
208 				return (a == null || b == null) ? null : a.compareTo(b);
209 			}
210 			case TIMESTAMP: {
211 				var a = millis(cell);
212 				var b = millis(arg);
213 				return (a == null || b == null) ? null : Long.compare(a, b);
214 			}
215 			case VERSION:
216 				return cmpVersion(str(cell), str(arg));
217 			default:
218 				return null;  // text / id / enum / boolean are not ordered.
219 		}
220 	}

226 	/**
227 	 * Compares two cell values for sorting under a value type: numeric/timestamp/version compare by their coerced
228 	 * value, everything else by case-insensitive string.  Nulls sort last.
229 	 *
230 	 * <p>
231 	 * For the coercible types (NUMERIC/TIMESTAMP/VERSION), a cell that fails to coerce is never compared as a raw
232 	 * string against one that does — doing so can contradict the value-based order and break the comparator's
233 	 * transitivity contract.  Instead, a non-coercible side unconditionally sorts <em>after</em> every
234 	 * coercible value for that type; the case-insensitive string fallback below is reached only when <em>both</em>
235 	 * sides fail to coerce (comparing two already-opaque strings is always safe, since there is no coerced value on
236 	 * either side for a lexical order to contradict).
237 	 */
238 	@SuppressWarnings({
239 		"java:S3776" // Per-type coercion branches are inherent to the sort comparator; splitting would obscure the transitivity reasoning.
240 	})
241 	static int compareCells(Object a, Object b, SearchType type) {
242 		if (a == null && b == null)
243 			return 0;
244 		if (a == null)
245 			return 1;
246 		if (b == null)
247 			return -1;
248 		switch (type) {
249 			case NUMERIC: {
250 				var x = decimal(a);
251 				var y = decimal(b);
252 				if (x != null && y != null)
253 					return x.compareTo(y);
254 				if (x != null || y != null)  // Exactly one coerces: it sorts before the non-coercible side.
255 					return x != null ? -1 : 1;
256 				break;
257 			}
258 			case TIMESTAMP: {
259 				var x = millis(a);
260 				var y = millis(b);
261 				if (x != null && y != null)
262 					return Long.compare(x, y);
263 				if (x != null || y != null)  // Exactly one coerces: it sorts before the non-coercible side.
264 					return x != null ? -1 : 1;
265 				break;
266 			}
267 			case VERSION: {
268 				var x = version(str(a));
269 				var y = version(str(b));
270 				if (x != null && y != null)
271 					return cmpVersionParts(x, y);
272 				if (x != null || y != null)  // Exactly one coerces: it sorts before the non-coercible side.
273 					return x != null ? -1 : 1;
274 				break;
275 			}
276 			default:
277 				break;
278 		}
279 		return String.valueOf(a).compareToIgnoreCase(String.valueOf(b));
280 	}

291 		var v = str(cell);

295 	/** Both sides coerce to a boolean and are equal; a side that is not a boolean never matches. */
296 	private static boolean boolEquals(Object cell, Object arg) {
297 		var a = bool(cell);
298 		return a.isPresent() && eq(a, bool(arg));
299 	}
300 
301 	private static Optional<Boolean> bool(Object o) {
302 		if (o instanceof Boolean b)
303 			return Optional.of(b);
304 		var v = str(o);
305 		if (v == null)
306 			return Optional.empty();
307 		v = v.strip();
308 		if (eqic(v, "true"))
309 			return Optional.of(Boolean.TRUE);
310 		if (eqic(v, "false"))
311 			return Optional.of(Boolean.FALSE);
312 		return Optional.empty();
313 	}

315 	private static BigDecimal decimal(Object o) {
316 		if (o instanceof Number n) {
317 			if (n instanceof Double d && (d.isNaN() || d.isInfinite()))
318 				return null;  // NaN/Infinity have no BigDecimal form: an unparseable cell.
319 			if (n instanceof Float f && (f.isNaN() || f.isInfinite()))
320 				return null;
321 			return new BigDecimal(n.toString());
322 		}
323 		var v = str(o);
324 		if (ib(v))
325 			return null;
326 		try {
327 			return new BigDecimal(v.strip());
328 		} catch (NumberFormatException e) {
329 			return null;
330 		}
331 	}
332 
333 	private static Long millis(Object o) {
334 		if (o instanceof Number n)
335 			return n.longValue();
336 		if (o instanceof Date d)
337 			return d.getTime();
338 		if (o instanceof Instant i)
339 			return i.toEpochMilli();
340 		if (o instanceof OffsetDateTime t)
341 			return t.toInstant().toEpochMilli();
342 		if (o instanceof ZonedDateTime t)
343 			return t.toInstant().toEpochMilli();
344 		if (o instanceof LocalDateTime t)
345 			return t.toInstant(ZoneOffset.UTC).toEpochMilli();
346 		if (o instanceof LocalDate d)
347 			return d.atStartOfDay(ZoneOffset.UTC).toInstant().toEpochMilli();
348 		var v = str(o);
349 		if (ib(v))
350 			return null;
351 		// Cells use the absolute-only parser (a stored "-24h" is never "now minus a day"); a String that does not parse is an unparseable cell.
352 		try {
353 			return millis(ValueParse.parseAbsolute(SearchType.TIMESTAMP, v));
354 		} catch (BeanQuerySyntaxException e) {
355 			return null;
356 		}
357 	}

359 	@SuppressWarnings({
360 		"java:S1168", // Private sentinel: null means "not a version", distinct from any parsed (possibly zero-length) value; hot sort path.
361 		"javabugs:S2259" // The ib(v) guard returns for a null or blank v before v.strip().
362 	})
363 	private static int[] version(String v) {
364 		if (ib(v))
365 			return null;
366 		var parts = v.strip().split("\\.");
367 		var out = new int[parts.length];
368 		for (var i = 0; i < parts.length; i++) {
369 			try {
370 				out[i] = Integer.parseInt(parts[i].strip());
371 			} catch (NumberFormatException e) {
372 				return null;
373 			}
374 		}
375 		return out;
376 	}
377 
378 	private static Integer cmpVersion(String a, String b) {
379 		var x = version(a);
380 		var y = version(b);
381 		if (x == null || y == null)
382 			return null;
383 		return cmpVersionParts(x, y);
384 	}
385 
386 	private static int cmpVersionParts(int[] x, int[] y) {
387 		var n = Math.min(x.length, y.length);
388 		for (var i = 0; i < n; i++)
389 			if (x[i] != y[i])
390 				return Integer.compare(x[i], y[i]);
391 		return Integer.compare(x.length, y.length);
392 	}
393 
394 	private static boolean versionPrefix(String cell, String arg) {
395 		var c = version(cell);
396 		var a = version(arg);
397 		if (c == null || a == null || a.length > c.length)
398 			return false;
399 		for (var i = 0; i < a.length; i++)
400 			if (a[i] != c[i])
401 				return false;
402 		return true;
403 	}
