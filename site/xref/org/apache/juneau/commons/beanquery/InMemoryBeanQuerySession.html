21  import java.util.*;

25  /**
26   * A {@link BeanQuerySession} over one in-memory collection.
27   *
28   * <p>
29   * The rows are filtered once, when the session is built, through the context guard <b>and</b> the session guard.
30   * Every call then searches, sorts and pages that scoped list.  All {@code $regex} matching in one call shares one
31   * deadline ({@link BeanQueryContext.Builder#regexTimeout(java.time.Duration) regexTimeout}).  {@link #close()} is a
32   * no-op, but use try-with-resources anyway so the code does not change if the engine does.
33   *
34   * <h5 class='section'>Example:</h5>
35   * <p class='bjava'>
36   * 	<jk>try</jk> (InMemoryBeanQuerySession&lt;Person&gt; <jv>session</jv> = <jv>context</jv>.createSession()
37   * 			.rows(<jv>people</jv>)
38   * 			.guard(<jv>p</jv> -&gt; <jv>p</jv>.getTenant().equals(<jv>tenant</jv>))   <jc>// This request's tenant only.</jc>
39   * 			.build()) {
40   * 		Page&lt;Person&gt; <jv>page</jv> = <jv>session</jv>.find(<jv>query</jv>);
41   * 		Counts <jv>counts</jv> = <jv>session</jv>.count(<jv>query</jv>);
42   * 	}
43   * </p>
44   *
45   * @param <T> The row (bean) type.
46   * @since 10.0.0
47   */
48  public final class InMemoryBeanQuerySession<T> extends BeanQuerySession<T> {
49  
50  	/**
51  	 * Builder for {@link InMemoryBeanQuerySession}.  See the class Javadoc for an example.
52  	 *
53  	 * @param <T> The row type.
54  	 */
55  	public static final class Builder<T> extends BeanQuerySession.Builder<T,Builder<T>> {
56  
57  		private final InMemoryBeanQueryContext<T> context;
58  		private Collection<? extends T> rows;
59  		private Predicate<? super T> guard;
60  
61  		Builder(InMemoryBeanQueryContext<T> context) {
62  			super(context);
63  			this.context = context;
64  		}
65  
66  		/**
67  		 * Sets the rows to query.  Required.
68  		 *
69  		 * @param value The rows.  Must not be <jk>null</jk>.
70  		 * @return This object.
71  		 */
72  		public Builder<T> rows(Collection<? extends T> value) {
73  			rows = reqnn("value", value);
74  			return this;
75  		}
76  
77  		/**
78  		 * Sets a per-request filter, applied together with the context guard.
79  		 *
80  		 * @param value The guard, or <jk>null</jk> for none.
81  		 * @return This object.
82  		 */
83  		public Builder<T> guard(Predicate<? super T> value) {
84  			guard = value;
85  			return this;
86  		}
87  
88  		/**
89  		 * {@inheritDoc}
90  		 *
91  		 * @throws IllegalStateException If no rows were set.
92  		 */
93  		@Override /* BeanQuerySession.Builder */
94  		@SuppressWarnings({
95  			"resource" // Caller takes ownership of the returned session
96  		})
97  		public InMemoryBeanQuerySession<T> build() {
98  			if (rows == null)
99  				throw isex("InMemoryBeanQuerySession requires rows.");
100 			return new InMemoryBeanQuerySession<>(this);
101 		}
102 	}
103 
104 	private final InMemoryBeanQueryContext<T> context;
105 	private final List<T> scoped;
106 
107 	@SuppressWarnings({
108 		"java:S9391" // Per-row try/catch that re-wraps guard failures; a stream lambda would obscure it.
109 	})
110 	InMemoryBeanQuerySession(Builder<T> builder) {
111 		super(builder);
112 		context = builder.context;
113 		var g = builder.guard;
114 		var l = new ArrayList<T>();
115 		for (T r : builder.rows) {
116 			try {
117 				if (context.inScope(r) && (g == null || g.test(r)))
118 					l.add(r);
119 			} catch (BeanQueryExecutionException | BeanQuerySyntaxException e) {
120 				throw e;
121 			} catch (RuntimeException e) {
122 				throw new BeanQueryExecutionException(e, "Failed to apply guard.");
123 			}
124 		}
125 		scoped = Collections.unmodifiableList(l);
126 	}
127 
128 	/**
129 	 * {@inheritDoc}
130 	 *
131 	 * @return The context that created this session.
132 	 */
133 	@Override /* BeanQuerySession */
134 	public InMemoryBeanQueryContext<T> getContext() {
135 		return context;
136 	}
137 
138 	@Override /* BeanQuerySession */
139 	protected Page<T> doFind(ResolvedQuery query) {
140 		var m = matched(query);
141 		return page(slice(sorted(m, query), query), query, scoped::size, m::size);
142 	}
143 
144 	@Override /* BeanQuerySession */
145 	protected Page<Map<String,Object>> doFindValues(ResolvedQuery query) {
146 		var m = matched(query);
147 		return page(values(slice(sorted(m, query), query), query), query, scoped::size, m::size);
148 	}
149 
150 	@Override /* BeanQuerySession */
151 	protected Counts doCount(ResolvedQuery query) {
152 		return Counts.of(scoped.size(), matched(query).size());
153 	}
154 
155 	@Override /* BeanQuerySession */
156 	protected Stream<T> doStream(ResolvedQuery query) {
157 		return slice(sorted(matched(query), query), query).stream();
158 	}
159 
160 	@Override /* BeanQuerySession */
161 	protected Stream<Map<String,Object>> doStreamValues(ResolvedQuery query) {
162 		return values(slice(sorted(matched(query), query), query), query).stream();
163 	}
164 
165 	@Override /* BeanQuerySession */
166 	public void close() {
167 		// Nothing to release: the rows belong to the caller.
168 	}
169 
170 	private List<T> matched(ResolvedQuery query) {
171 		var filter = query.filter();
172 		if (filter instanceof Filter.And and && and.items().isEmpty())
173 			return scoped;  // No search: every scoped row matches, without constructing a RegexBudget for nothing.
174 		var budget = new RegexBudget(settings.regexTimeout);  // One deadline for the whole call.
175 		return scoped.stream().filter(r -> FilterEvaluator.matches(filter, col -> context.cell(r, col), budget)).toList();
176 	}
177 
178 	private List<T> sorted(List<T> rows, ResolvedQuery query) {
179 		Comparator<T> c = null;
180 		for (var k : query.sort()) {
181 			Comparator<T> next = (a, b) -> InMemoryMatch.compareCells(context.cell(a, k.column()), context.cell(b, k.column()), k.type());
182 			if (k.descending())
183 				next = next.reversed();
184 			c = c == null ? next : c.thenComparing(next);
185 		}
186 		if (c == null)
187 			return rows;
188 		var l = new ArrayList<>(rows);
189 		l.sort(c);
190 		return l;
191 	}
192 
193 	private static <R> List<R> slice(List<R> rows, ResolvedQuery query) {
194 		var from = Math.min(query.position(), rows.size());
195 		var limit = query.limit();
196 		var to = limit == null ? rows.size() : (int)Math.min((long)from + limit, rows.size());
197 		return rows.subList(from, to);
198 	}
199 
200 	private List<Map<String,Object>> values(List<T> rows, ResolvedQuery query) {
201 		var l = new ArrayList<Map<String,Object>>(rows.size());
202 		for (var r : rows) {
203 			var m = new LinkedHashMap<String,Object>();
204 			for (var c : query.view())
205 				m.put(c, context.cell(r, c));
206 			l.add(Collections.unmodifiableMap(m));
207 		}
208 		return l;
209 	}
210 }
