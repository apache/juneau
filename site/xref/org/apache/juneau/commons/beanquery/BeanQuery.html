27  /**
28   * A protocol-agnostic query request against a collection of beans: what to search, which columns to return, how to
29   * sort, and which page to return (design §5.1, §5.3).
30   *
31   * <p>
32   * Every field is a plain <b>raw value</b> &mdash; there are no HTTP-binding annotations, no {@link java.util.Optional},
33   * no search map, no page string, and no parsed operators on this bean.  A <jk>null</jk> field means <b>absent</b>.  The
34   * grammar for {@link #getSearch() search} and {@link #getOpts() opts} lives <b>inside</b> those single strings and is
35   * decoded by a {@link BeanQuerySession session} when the query runs, not by this bean.
36   *
37   * <h5 class='section'>Fields</h5>
38   * <ul>
39   * 	<li><b>search</b> ({@link String}) &mdash; the one search string.  Top-level commas (outside parentheses) separate
40   * 		clauses; the first unescaped {@code =} separates a column from its raw {@code $}-expression
41   * 		({@link SearchExpressionParser}); backslash escapes {@code \,} / {@code \=} / {@code \\} are literals.  Columns
42   * 		combine with {@code and}.  See {@link ClauseParser}.
43   * 	<li><b>view</b> ({@link String}) &mdash; a comma-separated list of column names, in output order.
44   * 	<li><b>sort</b> ({@link String}) &mdash; a comma-separated list of {@code column} (ascending) or {@code column:desc}
45   * 		tokens (a trailing {@code -}/{@code +} is also accepted).
46   * 	<li><b>position</b> ({@link Integer}) &mdash; the 0-based first row of the page, or <jk>null</jk> for the start.
47   * 	<li><b>limit</b> ({@link Integer}) &mdash; the page size, or <jk>null</jk> for all rows.
48   * 	<li><b>opts</b> ({@link String}) &mdash; an open bag of {@code key=value} clauses (same grammar as {@code search}).
49   * 		The session reads {@code counts} and ignores unknown keys.
50   * </ul>
51   *
52   * <p>
53   * Construct with {@link #create()}, which returns a {@link Builder} that assembles these raw strings fluently:
54   *
55   * <h5 class='section'>Example:</h5>
56   * <p class='bjava'>
57   * 	BeanQuery <jv>q</jv> = BeanQuery.<jsm>create</jsm>()
58   * 		.eq(<js>"status"</js>, <js>"active"</js>)
59   * 		.or(<jv>g</jv> -&gt; <jv>g</jv>.contains(<js>"name"</js>, <js>"bob"</js>).contains(<js>"email"</js>, <js>"bob"</js>))
60   * 		.sort(<js>"name"</js>)
61   * 		.limit(50)
62   * 		.build();
63   * </p>
64   *
65   * <p>
66   * The HTTP-binding {@code BeanQueryRequest} ({@code juneau-rest-common}) extends this class and adds the
67   * {@code @Query} bindings.
68   *
69   * @since 10.0.0
70   */

212 	/**
213 	 * Creates a {@link Builder} pre-populated with this query's current state, so individual settings can be
214 	 * changed or added without disturbing the rest. {@link #getSearch()} is re-parsed into the builder's item
215 	 * tree, so a new condition on an existing top-level column merges through <c>$and(...)</c> rather than being
216 	 * dropped or duplicated.
217 	 *
218 	 * @return A new builder seeded from this query.
219 	 */
220 	public Builder copy() {
221 		var b = new Builder();
222 		if (search != null)
223 			b.group.items.seed(SearchParser.parse(search));
224 		if (view != null)
225 			b.view.addAll(splitNonBlank(view));
226 		if (sort != null)
227 			b.sort.addAll(splitNonBlank(sort));
228 		b.position = position;
229 		b.limit = limit;
230 		if (opts != null)
231 			b.opts.putAll(ClauseParser.parse(opts));
232 		return b;
233 	}

235 	/** Splits a comma-delimited list, trimming each token and dropping blank ones. */
236 	private static List<String> splitNonBlank(String s) {
237 		var l = new ArrayList<>(StringUtils.split(s));
238 		l.removeIf(String::isEmpty);
239 		return l;
240 	}

475 		/**
476 		 * Adds a cross-column <c>$or(...)</c> group. A block that adds no items adds nothing; a block that adds
477 		 * exactly one item emits that item unwrapped (no <c>$or(...)</c> wrapper).
478 		 *
479 		 * @param block Populates the group's conditions on a fresh {@link GroupBuilder}.
480 		 * @return This object.
481 		 */

483 			items.addGroup(group(block, SearchItem.Or::new));
484 			return this;
485 		}

487 		/**
488 		 * Adds a cross-column <c>$and(...)</c> group. A block that adds no items adds nothing; a block that adds
489 		 * exactly one item emits that item unwrapped (no <c>$and(...)</c> wrapper).
490 		 *
491 		 * @param block Populates the group's conditions on a fresh {@link GroupBuilder}.
492 		 * @return This object.
493 		 */
494 		public GroupBuilder and(Consumer<GroupBuilder> block) {
495 			items.addGroup(group(block, SearchItem.And::new));
496 			return this;
497 		}
498 
499 		/**
500 		 * Adds a <c>$not(...)</c> group. The block must add exactly one item (a leaf or a nested group); any
501 		 * other count throws {@link IllegalArgumentException}.
502 		 *
503 		 * @param block Populates the single negated condition on a fresh {@link GroupBuilder}.
504 		 * @return This object.
505 		 * @throws IllegalArgumentException If the block adds a number of items other than exactly one.
506 		 */
507 		public GroupBuilder not(Consumer<GroupBuilder> block) {
508 			reqnn("block", block);
509 			var inner = new GroupBuilder();
510 			block.accept(inner);
511 			var innerItems = inner.items.items();
512 			req(innerItems.size() == 1,
513 				"BeanQuery.GroupBuilder.not(...) requires exactly one condition, found %s.", innerItems.size());
514 			items.addGroup(new SearchItem.Not(innerItems.get(0)));
515 			return this;
516 		}
517 
518 		private static SearchItem group(Consumer<GroupBuilder> block, Function<List<SearchItem>,SearchItem> wrap) {
519 			reqnn("block", block);
520 			var inner = new GroupBuilder();
521 			block.accept(inner);
522 			var innerItems = inner.items.items();
523 			if (innerItems.isEmpty())
524 				return null;
525 			if (innerItems.size() == 1)
526 				return innerItems.get(0);
527 			return wrap.apply(innerItems);
528 		}
529 
530 		private GroupBuilder emit(String column, String name, Object...values) {
531 			reqnb("column", column);
532 			req(values != null && values.length > 0,
533 				"Operator '%s' on column '%s' requires at least one value.", name, column);
534 			var body = Arrays.stream(values).map(GroupBuilder::quote).collect(Collectors.joining(","));
535 			items.addLeaf(column, name + "(" + body + ")");
536 			return this;
537 		}
538 
539 		private static String quote(Object value) {
540 			var s = ValueFormat.format(value);
541 			// Escape a literal '$' by doubling it, in addition to (not instead of) the quote-escaping below:
542 			// decodeLiteral unconditionally collapses $$ -> $ as its last decode step, so an un-escaped
543 			// literal "$$" here would wrongly round-trip to "$". (The two replace() calls operate on
544 			// disjoint alphabets, so the order between them doesn't matter - only that both happen.)
545 			return "\"" + s.replace("$", "$$").replace("\"", "\"\"") + "\"";
546 		}

722 		 * Sets the {@code counts} option from a {@link CountRequest}. {@link CountRequest#NONE} removes the

724 		 *

733 			}

737 		/**
738 		 * Builds a new, immutable {@link BeanQuery} from this builder's current state. The builder remains usable
739 		 * afterward; later calls do not affect the {@link BeanQuery} already built.
740 		 *
741 		 * @return A new {@link BeanQuery}.
742 		 */
743 		public BeanQuery build() {
744 			var q = new BeanQuery();
745 			var items = group.items.items();
746 			if (!items.isEmpty())
747 				q.setSearch(SearchParser.render(items));
748 			if (!view.isEmpty())
749 				q.setView(String.join(",", view));
750 			if (!sort.isEmpty())
751 				q.setSort(String.join(",", sort));
752 			q.setPosition(position);
753 			q.setLimit(limit);
754 			if (!opts.isEmpty())
755 				q.setOpts(opts.entrySet().stream()
756 					.map(e -> ClauseParser.escape(e.getKey()) + "=" + ClauseParser.escape(e.getValue()))
757 					.collect(Collectors.joining(",")));
758 			return q;
759 		}
