19  import static org.apache.juneau.commons.utils.Shorts.*;

21  import java.time.*;

23  /**
24   * The per-column value type that governs which search operators apply to a column and how a bare (no leading
25   * {@code $}) pattern token is interpreted.
26   *
27   * <p>
28   * A column's value type selects both the operator set offered for that column
29   * ({@link SearchOperatorSet#forType(SearchType)}) and the leaf-comparison semantics a context applies.  The lowercase
30   * {@link #wire() wire token} is the stable external spelling.  {@link #bareHelp()} is a fixed, one-sentence
31   * explanation of bare-token behavior intended for display in a search-bar popover.
32   *
33   * <h5 class='section'>Bare-token behavior per type:</h5>
34   * <ul>
35   * 	<li>{@link #TEXT} / {@link #ID} &mdash; case-insensitive substring (same idea as {@code $contains}).
36   * 	<li>{@link #VERSION} &mdash; dotted-tuple prefix.
37   * 	<li>{@link #NUMERIC} / {@link #TIMESTAMP} / {@link #ENUM} / {@link #BOOLEAN} &mdash; exact equality.
38   * </ul>
39   *
40   * <h5 class='section'>Example:</h5>
41   * <p class='bjava'>
42   * 	String <jv>help</jv> = SearchType.VERSION.<jsm>bareHelp</jsm>();
43   * 	<jc>// help == "Matches any version starting with this prefix, e.g. 250 matches 250.1 and 250.2.3."</jc>
44   * </p>
45   *
46   * <h5 class='section'>Example (typing a literal):</h5>
47   * <p class='bjava'>
48   * 	<jc>// Type a literal the same way for every engine, so a bad value is BAD_VALUE (400) everywhere.</jc>
49   * 	Object <jv>v</jv> = SearchType.<jsf>NUMERIC</jsf>.<jsm>parse</jsm>(<js>"42"</js>);  <jc>// BigDecimal("42")</jc>
50   * </p>
51   *
52   * @since 10.0.0
53   */
54  public enum SearchType {
55  
56  	/** Free text.  Bare token is a case-insensitive substring match. */
57  	TEXT("text", "Matches any value containing this text (case-insensitive). Use $eq(...) for an exact, case-sensitive match."),
58  
59  	/** An id-like string (identifiers, keys).  Bare token is a case-insensitive substring match, same as {@link #TEXT}. */
60  	ID("id", "Matches any value containing this text (case-insensitive). Use $eq(...) for an exact, case-sensitive match."),
61  
62  	/** A number.  Bare token is exact equality; supports the ordered operators. */
63  	NUMERIC("numeric", "Matches a number exactly. Use $gt / $gte / $lt / $lte / $between for a range, or a duration such as -24h or -PT24H (= -86400000 ms)."),
64  
65  	/** A dotted version tuple (e.g. {@code 10.0.0}).  Bare token is a dotted-tuple prefix; supports the ordered operators and {@code $prefix}. */
66  	VERSION("version", "Matches any version starting with this prefix, e.g. 250 matches 250.1 and 250.2.3."),
67  
68  	/**
69  	 * An instant/timestamp.  Bare token is exact equality; supports the ordered operators.
70  	 *
71  	 * <p>
72  	 * <b>Offset advice:</b> a value with an explicit offset ({@code 2026-09-30T12:00:00Z},
73  	 * {@code 2026-09-30T12:00:00-04:00}) always denotes one absolute instant.  The local forms (a date-time without an
74  	 * offset, or a bare date) have no zone: in-memory matching treats them as UTC, while a SQL engine hands them to the
75  	 * database, which interprets them in the <i>session</i> time zone.  The two can therefore disagree, so prefer the
76  	 * offset forms (or epoch milliseconds, which are always UTC) in any search that must behave identically everywhere.
77  	 * See {@link #parse(String)}.
78  	 */
79  	TIMESTAMP("timestamp", "Matches an exact instant; a bare date means UTC midnight exactly, so for a whole day use $between(d, d+1). Also accepts a duration such as -24h or -PT24H, relative to the request time."),
80  
81  	/** A one-of enumerated value.  Bare token is a case-insensitive exact match. */
82  	ENUM("enum", "Matches one value exactly, case-insensitive."),
83  
84  	/** A boolean.  Bare token is exact equality; supports only {@code $eq}/{@code $ne}. */
85  	BOOLEAN("boolean", "Matches true or false exactly.");
86  
87  	private final String wire;
88  	private final String bareHelp;
89  
90  	SearchType(String wire, String bareHelp) {
91  		this.wire = wire;
92  		this.bareHelp = bareHelp;
93  	}
94  
95  	/**
96  	 * Returns the lowercase wire token for this type (e.g. {@code "text"}).
97  	 *
98  	 * @return The wire token.
99  	 */
100 	public String wire() {
101 		return wire;
102 	}
103 
104 	/**
105 	 * Returns a fixed, one-sentence explanation of what a bare (no leading {@code $}) token means for a column of
106 	 * this type, intended for display in a search-bar popover.
107 	 *
108 	 * @return The help string.  Never <jk>null</jk> or blank.
109 	 */
110 	public String bareHelp() {
111 		return bareHelp;
112 	}
113 
114 	/**
115 	 * Resolves a wire token back to its {@link SearchType}.
116 	 *
117 	 * @param wire The lowercase wire token (case-insensitive).  Can be <jk>null</jk>.
118 	 * @return The matching type, or <jk>null</jk> if {@code wire} is <jk>null</jk>, blank, or unrecognized.
119 	 */
120 	public static SearchType fromWire(String wire) {
121 		if (wire == null)
122 			return null;
123 		var w = wire.trim();
124 		for (var t : values())
125 			if (eqic(t.wire, w))
126 				return t;
127 		return null;
128 	}
129 
130 	/**
131 	 * Parses a decoded literal (quotes stripped, escapes undone) into this type's Java representation.
132 	 *
133 	 * <p>
134 	 * {@link #NUMERIC} parses to {@link java.math.BigDecimal}; {@link #BOOLEAN} to {@link Boolean}
135 	 * (case-insensitive {@code true}/{@code false}); {@link #TIMESTAMP} to {@link java.time.OffsetDateTime} (has an
136 	 * offset), {@link java.time.LocalDateTime} (no offset), {@link java.time.LocalDate} (date only), or, for a bare
137 	 * integer, an {@link java.time.OffsetDateTime} at UTC built from epoch milliseconds (optional leading {@code -}; no
138 	 * {@code +}).  The local forms carry no zone - see the offset advice on {@link #TIMESTAMP}.  {@link #TEXT}, {@link #ID},
139 	 * {@link #ENUM} and {@link #VERSION} are returned unchanged (never <jk>null</jk>; a <jk>null</jk> {@code value}
140 	 * becomes {@code ""}).
141 	 *
142 	 * <h5 class='section'>Example:</h5>
143 	 * <p class='bjava'>
144 	 * 	BigDecimal <jv>n</jv> = (BigDecimal)SearchType.<jsf>NUMERIC</jsf>.<jsm>parse</jsm>(<js>"-3.5"</js>);
145 	 * </p>
146 	 *
147 	 * @param value The decoded literal text, or <jk>null</jk> (treated as empty).
148 	 * @return The parsed value.
149 	 * @throws BeanQuerySyntaxException (code {@link BeanQuerySyntaxException.Code#BAD_VALUE BAD_VALUE}) If
150 	 * 	{@code value} does not parse for this type ({@link #NUMERIC}, {@link #BOOLEAN} or {@link #TIMESTAMP} only).
151 	 * @since 10.0.0
152 	 */
153 	public Object parse(String value) {
154 		return ValueParse.parse(this, value);
155 	}
156 
157 	/**
158 	 * Parses a decoded literal like {@link #parse(String)}, resolving a relative-duration literal against
159 	 * {@code requestTime} instead of the current instant.
160 	 *
161 	 * <p>
162 	 * A relative-duration literal is {@code [+-]<n>(ms|s|m|h|d)} (lowercase units only), e.g. {@code -24h}, or a signed ISO-8601 duration ({@code PT24H}, {@code -PT24H},
163 	 * {@code +PT30M}, {@code P1D}) with identical semantics (only one leading sign; {@code PT-1H}, years, months and
164 	 * weeks are {@code BAD_VALUE}).  For
165 	 * {@link #NUMERIC} it parses to the signed amount in milliseconds (a {@link java.math.BigDecimal}; "now" is never
166 	 * involved).  For {@link #TIMESTAMP} it parses to {@code requestTime} plus the signed amount, as an
167 	 * {@link java.time.OffsetDateTime} at UTC.  A bare signed number such as {@code -5} is never a duration, and a total that overflows a {@code long} (for
168 	 * {@link #TIMESTAMP}, after adding {@code requestTime}'s epoch milliseconds) is {@code BAD_VALUE}.  On
169 	 * {@link #BOOLEAN} a duration is {@code BAD_VALUE}; {@link #TEXT}, {@link #ID}, {@link #ENUM} and {@link #VERSION}
170 	 * are untyped and keep the text literally.  Every
171 	 * other literal shape parses exactly as in {@link #parse(String)}; {@link #parse(String)} itself delegates here
172 	 * with {@link Instant#now()}.
173 	 *
174 	 * <h5 class='section'>Example:</h5>
175 	 * <p class='bjava'>
176 	 * 	Instant <jv>now</jv> = Instant.<jsm>parse</jsm>(<js>"2026-09-30T12:00:00Z"</js>);
177 	 * 	Object <jv>yesterday</jv> = SearchType.<jsf>TIMESTAMP</jsf>.<jsm>parse</jsm>(<js>"-24h"</js>, <jv>now</jv>);  <jc>// 2026-09-29T12:00Z</jc>
178 	 * 	Object <jv>ms</jv> = SearchType.<jsf>NUMERIC</jsf>.<jsm>parse</jsm>(<js>"-24h"</js>, <jv>now</jv>);  <jc>// BigDecimal("-86400000")</jc>
179 	 * </p>
180 	 *
181 	 * @param value The decoded literal text, or <jk>null</jk> (treated as empty).
182 	 * @param requestTime The instant a relative-duration {@link #TIMESTAMP} literal resolves against.  Must not be <jk>null</jk>.
183 	 * @return The parsed value.
184 	 * @throws BeanQuerySyntaxException (code {@link BeanQuerySyntaxException.Code#BAD_VALUE BAD_VALUE}) If
185 	 * 	{@code value} does not parse for this type ({@link #NUMERIC}, {@link #BOOLEAN} or {@link #TIMESTAMP} only).
186 	 * @since 10.0.0
187 	 */
188 	public Object parse(String value, Instant requestTime) {
189 		return ValueParse.parse(this, value, reqnn("requestTime", requestTime));
190 	}
191 }
