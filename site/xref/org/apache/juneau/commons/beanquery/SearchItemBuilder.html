17  package org.apache.juneau.commons.beanquery;
18  
19  import java.util.*;
20  
21  /**
22   * The same-column {@code $and} merging logic (design D2) shared by {@link BeanQuery.Builder} and
23   * {@link BeanQuery.GroupBuilder}: an ordered list of {@link SearchItem}s at one nesting level (top level, or inside
24   * one group), where adding a leaf on a column already present at that level replaces it with both conditions
25   * combined by {@code $and(...)} &mdash; appending into an existing {@code $and} rather than nesting a new one
26   * inside it &mdash; while a group is always appended as a new, distinct sibling (groups never merge with each
27   * other or with a leaf).
28   */
29  final class SearchItemBuilder {
30  
31  	private final List<SearchItem> items = new ArrayList<>();
32  	private final Map<String,Integer> leafIndex = new HashMap<>();
33  
34  	/** Adds a leaf condition on {@code column}, merging via {@code $and(...)} if that column is already present. */
35  	void addLeaf(String column, String expression) {
36  		var idx = leafIndex.get(column);
37  		if (idx == null) {
38  			leafIndex.put(column, items.size());
39  			items.add(new SearchItem.Leaf(column, expression));
40  		} else {
41  			var existing = (SearchItem.Leaf) items.get(idx);
42  			items.set(idx, new SearchItem.Leaf(column, mergeExpression(existing.expression(), expression)));
43  		}
44  	}
45  
46  	/**
47  	 * Appends a group as a new sibling.  A <jk>null</jk> group (an empty or fully-blank lambda) adds nothing.
48  	 *
49  	 * <p>
50  	 * Groups are never registered in {@link #leafIndex}: a column named inside a nested group is invisible to this
51  	 * level's same-column merge, exactly like {@link SearchParser}'s own duplicate-column check only looking at
52  	 * top-level leaves.
53  	 */
54  	void addGroup(SearchItem group) {
55  		if (group != null)
56  			items.add(group);
57  	}
58  
59  	/** Seeds this builder's state from an already-parsed top-level item list (used by {@code BeanQuery.copy()}). */
60  	void seed(List<SearchItem> parsed) {
61  		for (var item : parsed) {
62  			if (item instanceof SearchItem.Leaf leaf)
63  				addLeaf(leaf.column(), leaf.expression());
64  			else
65  				addGroup(item);
66  		}
67  	}
68  
69  	/** The accumulated items at this level, in insertion order. */
70  	List<SearchItem> items() {
71  		return List.copyOf(items);
72  	}
73  
74  	// Splices raw expression text, not a parsed tree: a Leaf's expression is the opaque, unparsed per-column
75  	// $-expression (see SearchItem.Leaf / SearchParser), so string-level $and(...) detection is the only option here.
76  	private static String mergeExpression(String existing, String next) {
77  		if (existing.startsWith("$and(") && existing.endsWith(")"))
78  			return existing.substring(0, existing.length() - 1) + "," + next + ")";
79  		return "$and(" + existing + "," + next + ")";
80  	}
81  }
