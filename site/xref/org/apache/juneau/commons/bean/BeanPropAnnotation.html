24  /**
25   * Utility classes and methods for the {@link BeanProp @BeanProp} annotation.
26   *
27   * <p>
28   * Provides a {@link Builder} that constructs a synthetic {@link BeanProp @BeanProp} annotation instance
29   * programmatically without requiring it to be declared on a program element at compile time.
30   *
31   * <p>
32   * This builder is pure-data: it captures the annotation attribute
33   * values and produces an annotation instance that is functionally equivalent to a declared one, but
34   * does <i>not</i> apply itself to a marshalling context. The application logic remains in
35   * <c>juneau-marshall</c> (see {@code MarshalledPropAnnotation} and {@code BeanPropApplyAnnotation})
36   * until later phases move that machinery down into <c>juneau-commons</c>.
37   *
38   * <h5 class='section'>See Also:</h5><ul>
39   * 	<li class='ja'>{@link BeanProp}
40   * 	<li class='jc'>{@link AnnotationObject}
41   * </ul>
42   */
43  @SuppressWarnings({
44  	"rawtypes" // Raw types required for reflective annotation application.
45  })
46  public class BeanPropAnnotation {
47  
48  	/**
49  	 * Prevents instantiation.
50  	 */
51  	private BeanPropAnnotation() {}
52  
53  	/**
54  	 * Builder class.
55  	 */
56  	public static class Builder extends AnnotationObject.Builder {
57  
58  		private String[] description = {};
59  		private Class<?> type = void.class;
60  		private Class<?> elementType = void.class;
61  		private Class<?>[] params = new Class[0];
62  		private Class<? extends BeanFactory> factory = BeanFactory.Void.class;
63  		private String name = "";
64  		private String ro = "";
65  		private String summary = "";
66  		private String value = "";
67  		private String wo = "";

70  		 * Constructor.

72  		protected Builder() {

74  		}
75  
76  		/**
77  		 * Instantiates a new {@link BeanProp @BeanProp} object initialized with this builder.
78  		 *
79  		 * @return A new {@link BeanProp @BeanProp} object.
80  		 */
81  		public BeanProp build() {
82  			return new Object(this);
83  		}

85  		/**

90  		 */
91  		public Builder description(String...value) {

198 	@SuppressWarnings({
199 		"java:S2160", // equals() inherited from AnnotationObject compares all annotation interface methods; subclass fields are accessed via those methods
200 		"annotationSuperInterface" // Eclipse JDT: intentional concrete implementation of annotation interface for runtime-built annotation instances
201 	})
202 	private static class Object extends AnnotationObject implements BeanProp {
203 
204 		private final String[] description;
205 		private final Class<?> type;
206 		private final Class<?> elementType;
207 		private final Class<?>[] params;
208 		private final Class<? extends BeanFactory> factory;
209 		private final String name;
210 		private final String value;
211 		private final String ro;
212 		private final String summary;
213 		private final String wo;
214 
215 		Object(BeanPropAnnotation.Builder b) {
216 			super(b);
217 			description = cp(b.description);
218 			elementType = b.elementType;
219 			factory = b.factory;
220 			name = b.name;
221 			params = cp(b.params);
222 			ro = b.ro;
223 			summary = b.summary;
224 			type = b.type;
225 			value = b.value;
226 			wo = b.wo;
227 		}
228 
229 		@Override /* Overridden from BeanProp */
230 		public String[] description() {
231 			return cp(description);
232 		}
233 
234 		@Override /* Overridden from BeanProp */
235 		public Class<?> elementType() {
236 			return elementType;
237 		}
238 
239 		@Override /* Overridden from BeanProp */
240 		public Class<? extends BeanFactory> factory() {
241 			return factory;
242 		}
243 
244 		@Override /* Overridden from BeanProp */
245 		public String name() {
246 			return name;
247 		}
248 
249 		@Override /* Overridden from BeanProp */
250 		public Class<?>[] params() {
251 			return cp(params);
252 		}
253 
254 		@Override /* Overridden from BeanProp */
255 		public String ro() {
256 			return ro;
257 		}
258 
259 		@Override /* Overridden from BeanProp */
260 		public String summary() {
261 			return summary;
262 		}
263 
264 		@Override /* Overridden from BeanProp */
265 		public Class<?> type() {
266 			return type;
267 		}
268 
269 		@Override /* Overridden from BeanProp */
270 		public String value() {
271 			return value;
272 		}
273 
274 		@Override /* Overridden from BeanProp */
275 		public String wo() {
276 			return wo;
277 		}
278 	}
279 
280 	/** Default value */
281 	public static final BeanProp DEFAULT = create().build();

283 	/**
284 	 * Instantiates a new builder for this class.
285 	 *
286 	 * @return A new builder object.
287 	 */

289 		return new Builder();
290 	}
