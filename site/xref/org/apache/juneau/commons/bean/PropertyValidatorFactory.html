17  package org.apache.juneau.commons.bean;
18  
19  import java.util.*;
20  
21  /**
22   * Builds {@link PropertyValidator} instances from a JSON-Schema-shaped map of constraints.
23   *
24   * <p>
25   * The factory is resolved at runtime via {@link ServiceLoader ServiceLoader} through
26   * {@link PropertyValidators#factory()}.  If no factory is on the classpath (i.e. {@code juneau-bean-jsonschema} is
27   * not present), schema-validation mode silently becomes a no-op.
28   *
29   * <p>
30   * The marshalling side calls the factory once per bean property at {@link BeanMeta} construction time and installs
31   * the resulting validator into the property's read/write transform slots.
32   *
33   * <h5 class='topic'>Thread safety</h5>
34   * Factory implementations must be safe to invoke from multiple threads concurrently.
35   *
36   * @see PropertyValidator
37   * @see PropertyValidators
38   * @since 10.0.0
39   */
40  @FunctionalInterface
41  public interface PropertyValidatorFactory {
42  
43  	/**
44  	 * Builds a validator from a JSON-Schema-shaped map of constraints.
45  	 *
46  	 * @param schemaMap
47  	 * 	A map mirroring the JSON Schema constraint shape (the same shape produced by
48  	 * 	{@code SchemaAnnotation.asMap()}).  May be empty, in which case implementations should return <jk>null</jk>.
49  	 * @param propertyType
50  	 * 	The Java type of the property being validated.  May be used by the factory to choose narrower numeric or
51  	 * 	collection validators.  May be <jk>null</jk>.
52  	 * @return A validator, or <jk>null</jk> if the map carries no actionable constraints.
53  	 */
54  	PropertyValidator create(Map<String,Object> schemaMap, Class<?> propertyType);
55  }
