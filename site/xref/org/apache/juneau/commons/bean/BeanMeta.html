45  /**
46   * Encapsulates all access to the properties of a bean class (like a souped-up {@link BeanInfo}).
47   *
48   * <h5 class='topic'>Description</h5>
49   *
50   * Uses introspection to find all the properties associated with this class.  If the
51   * {@link BeanType @BeanType} annotation is present on the class (or the marshalling-side
52   * {@code @Marshalled} sibling), that information is used to determine the properties on the class.
53   * Otherwise, the {@code BeanInfo} functionality in Java is used to determine the properties on the class.
54   *
55   * <h5 class='topic'>Bean property ordering</h5>
56   *
57   * The order of the properties are as follows:
58   * <ul class='spaced-list'>
59   * 	<li>
60   * 		If {@link BeanType#properties() @BeanType(properties)} is specified on the class, the order matches the
61   * 		list of properties in the annotation.
62   * 	<li>
63   * 		Otherwise, the order is based on the following:
64   * 		<ul>
65   * 			<li>Public fields (same order as {@code Class.getFields()}).
66   * 			<li>Properties returned by {@code BeanInfo.getPropertyDescriptors()}.
67   * 			<li>Non-standard getters/setters with {@link BeanProp @BeanProp} annotation defined on them.
68   * 		</ul>
69   * </ul>
70   *
71   * <h5 class='topic'>Thread safety</h5>
72   *
73   * Instances are effectively immutable after construction and are safe for concurrent read access.
74   * Construction is not thread-safe and should be completed before sharing an instance across threads.
75   *
76   *
77   * @param <T> The class type that this metadata applies to.
78   */

130 		/*
131 		 * Returns true if this method matches the class type of the specified property.
132 		 * Only meant to be used for setters.
133 		 */
134 		private boolean matchesPropertyType(BeanPropertyMeta.Builder b) {
135 			if (b == null)  // HTT: unreachable - the sole caller (the setter pass in the constructor) only looks up a name that the earlier all-bms pass already computeIfAbsent-registered a builder for, so normalProps.get(x.propertyName) is never null here.  Retained as a defensive guard against future refactors that decouple the two passes.
136 				return false;
137 
138 			// Don't do further validation if this is the "*" bean property.
139 			if ("*".equals(b.name))
140 				return true;
141 
142 			// Get the bean property type from the getter/field.
143 			Class<?> pt = null;  // Uses raw Class<?> here because assignment checks are performed against Class.
144 			if (nn(b.getter))
145 				pt = b.getter.getReturnType().inner();
146 			else if (nn(b.field))
147 				pt = b.field.inner().getType();
148 
149 			// Matches if only a setter is defined.
150 			if (pt == null)
151 				return true;
152 
153 			// Doesn't match if not same type or super type as getter/field.
154 			if (! type.isAssignableFrom(pt))
155 				return false;
156 
157 			// If a setter was previously set, only use this setter if it's a closer
158 			// match (e.g. prev type is a superclass of this type).
159 			if (b.setter == null)
160 				return true;
161 
162 			return type.isStrictChildOf(b.setter.getParameterTypes().get(0).inner());
163 		}
164 	}

175 	 * Creates a {@link BeanMeta} instance for the specified bean type info.
176 	 *
177 	 * <p>
178 	 * This is the marshalling-side factory entry point.  It attempts to create bean metadata for a class; if the class is
179 	 * determined to be a bean, the returned {@link BeanMetaValue} carries the {@link BeanMeta} instance and a
180 	 * <jk>null</jk> reason.  Otherwise it carries <jk>null</jk> for the bean metadata and a non-empty string explaining
181 	 * why the class is not a bean.
182 	 *
183 	 * <p>
184 	 * The bean filter is resolved internally from the {@code @Marshalled} / {@link BeanType @BeanType} annotations on the
185 	 * class via {@link BeanMetaInitializer#buildBeanFilter(BeanInfo)}.  See {@link #of(Class, BeanConfigContext)} for
186 	 * the pure bean-modeling-side entry point that does not require a {@link BeanInfo}.
187 	 *
188 	 * <h5 class='section'>Exception handling:</h5>
189 	 * <p>
190 	 * If a {@link RuntimeException} is thrown during bean metadata creation, it is caught and the exception message
191 	 * is returned as the {@code notABeanReason} with <jk>null</jk> for the bean metadata.
192 	 *
193 	 * <h5 class='section'>Example:</h5>
194 	 * <p class='bjava'>
195 	 * 	<jc>// Create bean metadata for a class via the marshalling-side context.</jc>
196 	 * 	BeanInfo&lt;Person&gt; <jv>cm</jv> = <jv>marshallingContext</jv>.getClassMeta(Person.<jk>class</jk>);
197 	 * 	BeanMetaHolder&lt;Person&gt; <jv>result</jv> = BeanMeta.<jsm>create</jsm>(<jv>cm</jv>, <jk>null</jk>);
198 	 *
199 	 * 	<jc>// Check if it's a bean</jc>
200 	 * 	<jk>if</jk> (<jv>result</jv>.beanMeta() != <jk>null</jk>) {
201 	 * 		BeanMeta&lt;Person&gt; <jv>bm</jv> = <jv>result</jv>.beanMeta();
202 	 * 		<jc>// Use the bean metadata...</jc>
203 	 * 	} <jk>else</jk> {
204 	 * 		String <jv>reason</jv> = <jv>result</jv>.notABeanReason();
205 	 * 		<jc>// Handle the case where it's not a bean...</jc>
206 	 * 	}
207 	 * </p>
208 	 *
209 	 * @param <T> The class type.
210 	 * @param cm The bean type info for the class to create bean metadata for.  Must not be <jk>null</jk>.
211 	 * @param implClass Optional implementation class info to use when looking for a no-arg constructor.  Can be <jk>null</jk> (the bean class's own no-arg constructor is used instead).
212 	 * @return A {@link BeanMetaValue} containing the bean metadata (if successful) or a reason why it's not a bean.
213 	 */
214 	public static <T> BeanMetaValue<T> create(BeanInfo<T> cm, ClassInfo implClass) {
215 		return create(cm, implClass, null);
216 	}

218 	/**
219 	 * Same as {@link #create(BeanInfo, ClassInfo)} but allows an explicit {@link PropertyNamer} to override the one
220 	 * that would otherwise be derived from the bean filter or {@link BeanConfigContext#getPropertyNamer()}.
221 	 *
222 	 * <p>
223 	 * The returned {@link BeanMeta} is <b>not</b> cached anywhere — callers that need a namer-specific view (for
224 	 * example, {@code MarshallingSession#toBeanMap(Object, PropertyNamer)}) build one on demand so the shared,
225 	 * {@code Class}-keyed default {@link BeanMeta} cache is never polluted with alternate-namer property keys.
226 	 *
227 	 * @param <T> The class type.
228 	 * @param cm The bean type info for the class to create bean metadata for.  Must not be <jk>null</jk>.
229 	 * @param implClass Optional implementation class info to use when looking for a no-arg constructor.  Can be <jk>null</jk> (the bean class's own no-arg constructor is used instead).
230 	 * @param propertyNamer Optional property namer to use for deriving property names.  If <jk>null</jk>, the namer is
231 	 * 	resolved the normal way (bean filter, then {@link BeanConfigContext#getPropertyNamer()}).
232 	 * @return A {@link BeanMetaValue} containing the bean metadata (if successful) or a reason why it's not a bean.
233 	 */
234 	public static <T> BeanMetaValue<T> create(BeanInfo<T> cm, ClassInfo implClass, PropertyNamer propertyNamer) {
235 		try {
236 			var cfg = cm.getBeanConfigContext();
237 			var ap = cfg.getAnnotationProvider();
238 			var bmi = cfg.getBeanMetaInitializer();
239 
240 			// Sanity checks first.
241 			if (cfg.isNotABean(cm))
242 				return notABean("Class matches exclude-class list");
243 
244 			if (cfg.isBeansRequireSerializable() && ! cm.isAssignableTo(Serializable.class) && ! bmi.hasBeanRegistrationAnnotation(cfg, cm))
245 				return notABean("Class is not serializable");
246 
247 			if (ap.has(BeanIgnore.class, cm))
248 				return notABean("Class is annotated with @BeanIgnore");
249 
250 		if ((! cfg.getBeanClassVisibility().isVisible(cm.getModifiers()) || cm.isAnonymousClass()) && ! bmi.hasBeanRegistrationAnnotation(cfg, cm))
251 			return notABean("Class is not public");
252 
253 			var bm = new BeanMeta<>(cm, bmi.buildBeanFilter(cm), null, implClass, propertyNamer);
254 
255 			if (nn(bm.notABeanReason))
256 				return notABean(bm.notABeanReason);
257 
258 			return new BeanMetaValue<>(bm, null);
259 		} catch (RuntimeException e) {
260 			return new BeanMetaValue<>(null, e.getMessage());
261 		}
262 	}

280 	 * Extracts the property name from {@link BeanProp @BeanProp} or {@link Name @Name} annotations.
281 	 *
282 	 * <p>
283 	 * If {@link Name @Name} annotations are present, returns the value from the last one.
284 	 * Otherwise, searches through {@link BeanProp @BeanProp} annotations and returns the first non-empty
285 	 * {@link BeanProp#value() value()} or {@link BeanProp#name() name()} found.
286 	 *
287 	 * @param p List of {@link BeanProp @BeanProp} annotations.
288 	 * @param n List of {@link Name @Name} annotations.
289 	 * @return The property name, or <jk>null</jk> if no name is found.
290 	 */
291 	private static String bpName(List<BeanProp> p, List<Name> n) {
292 		if (p.isEmpty() && n.isEmpty())  // HTT: n.isEmpty() is always true here (see below) - the p.isEmpty()==false/n.isEmpty()==false combo can't occur
293 			return null;
294 		if (! n.isEmpty())  // HTT: n is always empty - @Name is @Target(PARAMETER)-only, so the sole call site's method-level ap.find(Name.class, m) can never find one; retained for symmetry with name(AnnotationInfo) in case a future caller scans parameter-level annotations instead
295 			return last(n).value();
296 
297 		var name = Holder.of(p.isEmpty() ? null : "");
298 		p.forEach(x -> {
299 			if (! x.value().isEmpty())
300 				name.set(x.value());
301 			if (! x.name().isEmpty())
302 				name.set(x.name());
303 		});
304 
305 		return name.orElse(null);
306 	}

308 	/*
309 	 * Extracts the property name from a single {@link BeanProp @BeanProp} or {@link Name @Name} annotation.
310 	 *
311 	 * <p>
312 	 * For {@link BeanProp @BeanProp} annotations, returns the first non-empty value found in the following order:
313 	 * <ol>
314 	 * 	<li>{@link BeanProp#name() name()}
315 	 * 	<li>{@link BeanProp#value() value()}
316 	 * </ol>
317 	 *
318 	 * <p>
319 	 * For {@link Name @Name} annotations, returns the {@link Name#value() value()}.
320 	 *
321 	 * <p>
322 	 * This method is used to extract property names from individual annotations, typically when processing
323 	 * annotation lists in stream operations.
324 	 *
325 	 * @param ai The annotation info containing either a {@link BeanProp @BeanProp} or {@link Name @Name} annotation.
326 	 * @return The property name extracted from the annotation, or <jk>null</jk> if no name is found.
327 	 * @see #bpName(List, List)
328 	 */
329 	private static String name(AnnotationInfo<?> ai) {
330 		if (ai.isType(BeanProp.class)) {
331 			var p = ai.cast(BeanProp.class).inner();
332 			if (ine(p.name()))
333 				return p.name();
334 			if (ine(p.value()))
335 				return p.value();
336 		} else {  // HTT: unreachable at both call sites (field-level annotation scans) - @Name is @Target(PARAMETER)-only, so a field can never carry one; the upstream filter's isType(Name.class) check never matches.  Retained for symmetry with bpName(List, List) in case a future caller scans parameter-level annotations instead.
337 			var n = ai.cast(Name.class).inner();
338 			if (ine(n.value()))
339 				return n.value();
340 		}
341 		return null;
342 	}

1225 			if (rcon != null)
1226 				return new BeanConstructor(o(rcon.accessible()), components.stream().map(RecordComponent::getName).toList());
1227 		}

1231 
1232 		var con = ci.getNoArgConstructor(config.getBeanMetaInitializer().hasBeanRegistrationAnnotation(config, classInfo) ? Visibility.PRIVATE : vis).orElse(null);
1233 		if (con != null)
1234 			return new BeanConstructor(o(con.accessible()), emptyList());

1236 		return new BeanConstructor(oe(), emptyList());

1489 	 * Builds a list of all classes in the class hierarchy for this bean.
1490 	 *
1491 	 * <p>
1492 	 * Traverses the complete inheritance hierarchy (classes and interfaces) starting from the bean class (or the
1493 	 * interface class specified in the bean filter) and collects all classes up to (but not including) the stop class.
1494 	 *
1495 	 * <p>
1496 	 * The traversal order follows a depth-first approach:
1497 	 * <ol>
1498 	 * 	<li>First, recursively traverses the superclass hierarchy
1499 	 * 	<li>Then, recursively traverses all implemented interfaces
1500 	 * 	<li>Finally, adds the current class itself
1501 	 * </ol>
1502 	 *
1503 	 * <p>
1504 	 * If a {@link BeanFilter#getInterfaceClass() bean filter interface class} is specified, the traversal starts
1505 	 * from that interface class instead of the bean class itself. This allows beans to use properties defined on
1506 	 * a parent interface rather than the concrete implementation class.
1507 	 *
1508 	 * <p>
1509 	 * The resulting list is used to find bean properties, methods, and fields across the entire class hierarchy.
1510 	 *
1511 	 * @return An unmodifiable list of {@link ClassInfo} objects representing all classes in the hierarchy, in
1512 	 * 	traversal order (superclasses first, then interfaces, then the class itself).
1513 	 */
1514 	private List<ClassInfo> findClassHierarchy() {
1515 		var result = new LinkedList<ClassInfo>();
1516 		// If @BeanType(interfaceClass) is specified on the parent class, then we want
1517 		// to use the properties defined on that class, not the subclass.
1518 		var c2 = (nn(beanFilter) && nn(beanFilter.getInterfaceClass()) ? beanFilter.getInterfaceClass() : classInfo);
1519 		findClassHierarchy(c2, stopClass, result::add);
1520 		return u(result);
1521 	}

1524 	 * Recursively traverses the class hierarchy and invokes the consumer for each class found.
1525 	 *
1526 	 * <p>
1527 	 * This is a helper method that performs a depth-first traversal of the class hierarchy:
1528 	 * <ol>
1529 	 * 	<li>Recursively processes the superclass (if present and not the stop class)
1530 	 * 	<li>Recursively processes all implemented interfaces
1531 	 * 	<li>Invokes the consumer with the current class
1532 	 * </ol>
1533 	 *
1534 	 * <p>
1535 	 * The traversal stops when it reaches the stop class (which is not included in the traversal).
1536 	 *
1537 	 * @param c The class to start traversal from.
1538 	 * @param stopClass The class to stop traversal at (exclusive). Traversal will not proceed beyond this class.
1539 	 * @param consumer The consumer to invoke for each class in the hierarchy.
1540 	 */
1541 	private void findClassHierarchy(ClassInfo c, ClassInfo stopClass, Consumer<ClassInfo> consumer) {
1542 		var sc = c.getSuperclass();
1543 		if (nn(sc) && ! sc.is(stopClass.inner()))
1544 			findClassHierarchy(sc, stopClass, consumer);
1545 		c.getInterfaces().forEach(x -> findClassHierarchy(x, stopClass, consumer));
1546 		consumer.accept(c);
1547 	}

1549 	/*
1550 	 * Finds the dictionary name (type name) for this bean class.
1551 	 *
1552 	 * <p>
1553 	 * The dictionary name is used in serialized forms to identify the specific type of a bean instance, enabling
1554 	 * polymorphic serialization and deserialization. This is especially important when serializing/parsing beans
1555 	 * that are part of an inheritance hierarchy.
1556 	 *
1557 	 * <p>
1558 	 * The dictionary name is determined by searching in the following order of precedence:
1559 	 * <ol>
1560 	 * 	<li><b>Bean filter type name:</b> If a bean filter is present and has a type name specified via
1561 	 * 		{@link BeanFilter#getTypeName()}, that value is used.
1562 	 * 	<li><b>Bean registry lookup:</b> If a bean registry exists for this bean, it is queried for the type name
1563 	 * 		of this class.
1564 	 * 	<li><b>Parent class registry lookup:</b> Searches through parent classes and interfaces (starting from the
1565 	 * 		second one, skipping the class itself) and checks if any of their bean registries contain a type name
1566 	 * 		for this class.
1567 	 * 	<li><b>{@code @Marshalled} annotation:</b> If the class has a {@code @Marshalled} annotation with a non-empty
1568 	 * 		{@code typeName()}, that value is used.
1569 	 * 	<li><b>No dictionary name:</b> Returns <jk>null</jk> if no dictionary name is found.
1570 	 * </ol>
1571 	 *
1572 	 * <p>
1573 	 * If a dictionary name is found, it will be used in serialized output (typically as a special property like
1574 	 * <js>"_type"</js>) so that parsers can determine the correct class to instantiate when deserializing.
1575 	 *
1576 	 * @return The dictionary name for this bean, or <jk>null</jk> if no dictionary name is defined.
1577 	 */
1578 	private String findDictionaryName() {
1579 		if (nn(beanFilter) && nn(beanFilter.getTypeName()))
1580 			return beanFilter.getTypeName();
1581 
1582 		// Pure-reflection class identity for BeanRegistry.getTypeName(...) — this method now uses
1583 		// surviving `this.classMeta` references inside this method to `classInfo.inner()` so the dictionary
1584 		// lookup does not require a ClassMeta for the bean's own class.  BeanRegistry still lives in
1585 		// juneau-marshall; it just exposes a raw-Class overload for callers that have a ClassInfo.
1586 		var rawClass = classInfo.inner();
1587 
1588 		var br = getBeanRegistry();
1589 		if (nn(br)) {
1590 			String s = br.getTypeName(rawClass);
1591 			if (nn(s))
1592 				return s;
1593 		}
1594 
1595 		// Parent-class BeanRegistry lookup is a marshalling-side concern: it walks parents and interfaces to
1596 		// see if any of THEIR ClassMeta-backed BeanRegistries declares a typeName for our raw class.  Lifted
1597 		// out to the BeanMetaInitializer SPI carried on BeanConfigContext.  Returns null on the commons-side path.
1598 		var bmi = config.getBeanMetaInitializer();
1599 		var n = bmi.findTypeNameInParents(marshallingContext, classInfo, rawClass);
1600 		if (n != null)
1601 			return n;
1602 
1603 		return bmi.findMarshalledTypeName(config, classInfo);
1604 	}
