32  /**
33   * Immutable runtime configuration for the bean-modeling layer.
34   *
35   * <p>
36   * {@code BeanConfigContext} is the runtime sibling of the {@link BeanConfig @BeanConfig} annotation:
37   * it carries the resolved values that drive how Java types are introspected as beans, independent
38   * of any marshalling concern.  Settings include visibility thresholds, required-property toggles,
39   * fluent-setter detection, property naming, not-bean exclusions, the active {@link BeanStore} and
40   * {@link AnnotationProvider}, and a few related hooks.
41   *
42   * <p>
43   * This type is the bean-modeling counterpart of the marshalling-layer context that lives in
44   * {@code juneau-marshall}.  It exists so the bean-modeling runtime (in {@code commons.bean}) can
45   * be used independently of the marshalling stack, and so the marshalling layer can compose a
46   * {@code BeanConfigContext} snapshot to feed bean-modeling code without exposing marshalling-only
47   * APIs.
48   *
49   * <h5 class='section'>Construction:</h5>
50   * <p class='bjava'>
51   * 	<jc>// All defaults.</jc>
52   * 	BeanConfigContext <jv>defaults</jv> = BeanConfigContext.<jsf>DEFAULT</jsf>;
53   *
54   * 	<jc>// Customized.</jc>
55   * 	BeanConfigContext <jv>ctx</jv> = BeanConfigContext.<jsm>create</jsm>()
56   * 		.beanClassVisibility(Visibility.<jsf>PROTECTED</jsf>)
57   * 		.findFluentSetters(<jk>true</jk>)
58   * 		.propertyNamer(<jk>new</jk> PropertyNamerDLC())
59   * 		.build();
60   *
61   * 	<jc>// Copy and tweak.</jc>
62   * 	BeanConfigContext <jv>ctx2</jv> = <jv>ctx</jv>.copy()
63   * 		.unsortedProperties(<jk>true</jk>)
64   * 		.build();
65   * </p>
66   *
67   * <h5 class='topic'>Thread safety</h5>
68   *
69   * {@link BeanConfigContext} instances are immutable and thread-safe.
70   * {@link Builder} instances are mutable and not thread-safe.
71   *
72   * <h5 class='section'>See Also:</h5><ul>
73   * 	<li class='ja'>{@link BeanConfig @BeanConfig} — the annotation form.
74   * </ul>
75   */

398 	/**
399 	 * Returns <jk>true</jk> if the specified class is excluded from bean detection.
400 	 *
401 	 * <p>
402 	 * The default implementation checks {@link #getNotBeanClasses()}, {@link #getNotBeanPackageNames()}, and
403 	 * {@link #getNotBeanPackagePrefixes()}, and rejects arrays/primitives/enums/annotations.  A custom
404 	 * predicate (set via {@link Builder#notABeanPredicate(Predicate)}) overrides this behavior entirely.
405 	 *
406 	 * @param ci The class info being tested.  Must not be <jk>null</jk>.
407 	 * @return <jk>true</jk> if the class is excluded from bean detection.
408 	 */
409 	@SuppressWarnings({
410 		"java:S3776" // Cognitive complexity acceptable for bean-exclusion predicate dispatch
411 	})
412 	public boolean isNotABean(ClassInfo ci) {
413 		assertArgNotNull("ci", ci);
414 		if (notABeanPredicate != null)
415 			return notABeanPredicate.test(ci);
416 		if (ci.isArray() || ci.isPrimitive() || ci.isEnum() || ci.isAnnotation())
417 			return true;
418 		var p = ci.getPackage();
419 		if (nn(p)) {
420 			var pn = p.getName();
421 			for (var p2 : notBeanPackageNames)
422 				if (pn.equals(p2))
423 					return true;
424 			for (var p2 : notBeanPackagePrefixes)
425 				if (pn.startsWith(p2))
426 					return true;
427 		}
428 		for (var exclude : notBeanClasses)
429 			if (ci.isAssignableTo(exclude))
430 				return true;
431 		return false;
432 	}
