17  package org.apache.juneau.commons.bean;
18  
19  import java.util.*;
20  /**
21   * Minimum runtime surface that the bean-modeling layer needs from a marshalling session.
22   *
23   * <p>
24   * {@code BeanSession} is an SPI seam between the bean-modeling runtime (in {@code commons.bean}) and the
25   * marshalling stack (in {@code juneau-marshall}).  The marshalling layer's {@code MarshallingSession}
26   * implements this interface so the bean-modeling types (notably {@code BeanMap} and {@code BeanPropertyMeta})
27   * can request session-aware operations — type conversion, bean-map wrapping — without referencing any
28   * marshalling-aware concrete types.
29   *
30   * <p>
31   * On the commons-side construction path ({@code BeanMeta.of(Class, BeanConfigContext)}) the session is
32   * <jk>null</jk> and these operations are skipped; the bean-modeling code runs in raw-reflection mode.
33   *
34   * <h5 class='section'>Method parameter typing:</h5>
35   * <p>
36   * The {@code targetType} parameters are typed as {@link Object} because the marshalling layer expresses them
37   * as {@code ClassMeta<?>} — a marshalling-side type that {@code commons.bean} cannot reference.  Callers in
38   * the bean-modeling layer pass through whatever opaque type-metadata handle they received; the
39   * {@code MarshallingSession} implementation interprets it as a {@code ClassMeta}.
40   *
41   * <h5 class='topic'>Thread safety</h5>
42   * Thread safety depends on implementation.
43   *
44   * <h5 class='section'>See Also:</h5><ul>
45   * 	<li class='ja'>{@link BeanConfigContext} — bean-modeling configuration counterpart.
46   * </ul>
47   */
48  public interface BeanSession {
49  
50  	/**
51  	 * Converts the specified value to the specified target type.
52  	 *
53  	 * <p>
54  	 * Implementations apply any registered conversions, swaps, or coercions necessary to produce a value
55  	 * compatible with {@code targetType}.  On the marshalling-side, this routes through
56  	 * {@code MarshallingSession.convertToType(Object, ClassMeta)}.
57  	 *
58  	 * @param value The value to convert.  May be <jk>null</jk>.
59  	 * @param targetType
60  	 * 	The target type — typically a marshalling-side {@code ClassMeta<?>} handle passed through from
61  	 * 	a {@code BeanPropertyMeta} or {@code BeanMap}.  Must not be <jk>null</jk>.
62  	 * @return The converted value.
63  	 */
64  	Object convertToType(Object value, Object targetType);
65  
66  	/**
67  	 * Converts the specified value to the specified target type, instantiating non-static inner classes against
68  	 * an outer instance.
69  	 *
70  	 * @param outer The outer-class instance for non-static inner classes.  May be <jk>null</jk>.
71  	 * @param value The value to convert.  May be <jk>null</jk>.
72  	 * @param targetType The target type — typically a marshalling-side {@code ClassMeta<?>}.  Must not be <jk>null</jk>.
73  	 * @return The converted value.
74  	 */
75  	Object convertToMemberType(Object outer, Object value, Object targetType);
76  
77  	/**
78  	 * Parses the specified JSON-formatted character sequence into a {@link Map}.
79  	 *
80  	 * <p>
81  	 * Used by {@code BeanPropertyMeta.setPropertyValue} when a {@link CharSequence} value is supplied for a
82  	 * {@code Map}-typed property — the bean-modeling layer cannot reference the marshalling-side JSON parser
83  	 * directly, so the parse is delegated to the session via this SPI.
84  	 *
85  	 * <p>
86  	 * Implementations typically route through {@code JsonMap.ofString(value).session(this)}.
87  	 *
88  	 * @param value The JSON-formatted character sequence to parse.  Must not be <jk>null</jk>.
89  	 * @return The parsed map.
90  	 */
91  	@SuppressWarnings({
92  		"java:S1452" // Map<?,?> wildcard return intentional; the key/value types are determined by the JSON content at runtime
93  	})
94  	Map<?,?> parseToMap(CharSequence value);
95  
96  	/**
97  	 * Parses the specified JSON-formatted character sequence into a {@link Collection}.
98  	 *
99  	 * <p>
100 	 * Used by {@code BeanPropertyMeta.setPropertyValue} when a {@link CharSequence} value is supplied for a
101 	 * {@code Collection}-typed property — the bean-modeling layer cannot reference the marshalling-side JSON
102 	 * parser directly, so the parse is delegated to the session via this SPI.
103 	 *
104 	 * <p>
105 	 * Implementations typically route through {@code new JsonList(value).setBeanSession(this)}.
106 	 *
107 	 * @param value The JSON-formatted character sequence to parse.  Must not be <jk>null</jk>.
108 	 * @return The parsed collection.
109 	 */
110 	@SuppressWarnings({
111 		"java:S1452" // Collection<?> wildcard return intentional; the element type is determined by the JSON content at runtime
112 	})
113 	Collection<?> parseToList(CharSequence value);
114 
115 	/**
116 	 * Wraps the specified bean in a {@code BeanMap}.
117 	 *
118 	 * <p>
119 	 * Equivalent to {@code MarshallingSession.toBeanMap(Object)} on the marshalling-side.  The return type is
120 	 * typed as {@link Object} (rather than {@code BeanMap}) because the bean-modeling {@code BeanMap} runtime
121 	 * type currently lives in {@code juneau-marshall}; callers in {@code commons.bean} treat the returned value
122 	 * opaquely and pass it through to other {@code BeanSession} / bean-modeling APIs that re-narrow it.
123 	 *
124 	 * <p>
125 	 * The signature is generic to allow implementations (notably {@code MarshallingSession}) to satisfy this
126 	 * method with their existing covariant {@code <T> BeanMap<T> toBeanMap(T)} declaration without an
127 	 * additional bridge method.
128 	 *
129 	 * @param <T> The bean type.
130 	 * @param bean The bean to wrap.  Must not be <jk>null</jk>.
131 	 * @return A new bean map (typically a {@code BeanMap}) wrapping the supplied bean.
132 	 */
133 	<T> Object toBeanMap(T bean);
134 
135 	/**
136 	 * Returns <jk>true</jk> if the specified bean property is visible under the current active view.
137 	 *
138 	 * <p>
139 	 * When no active view is set on the session, this method always returns <jk>true</jk> (all properties visible).
140 	 * The marshalling-side {@code MarshallingSession} overrides this to implement the actual view-membership check
141 	 * using the property's declared view set and the active view name.
142 	 *
143 	 * <p>
144 	 * Default-view-inclusion policy (configurable): an untagged property — one whose view set is empty — is
145 	 * considered in-view under every active view unless the
146 	 * {@code MarshallingContext.Builder.disableDefaultViewInclusion()} flag is set.
147 	 *
148 	 * @param pMeta The property metadata. Must not be <jk>null</jk>.
149 	 * @return <jk>true</jk> if the property should be included under the current active view.
150 	 */
151 	default boolean isPropertyInActiveView(BeanPropertyMeta pMeta) {
152 		return true; // HTT — default SPI method; MarshallingSession always overrides, so this body is unreachable in practice.
153 	}
154 }
