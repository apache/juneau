17  package org.apache.juneau.commons.bean;
18  
19  import static java.lang.annotation.ElementType.*;
20  import static java.lang.annotation.RetentionPolicy.*;
21  
22  import java.lang.annotation.*;
23  
24  /**
25   * Excludes a class, field, method, or constructor from bean introspection.
26   *
27   * <p>
28   * The bean-modeling sibling of {@code @MarshalledIgnore}: where {@code @MarshalledIgnore} (in
29   * <c>juneau-marshall</c>) controls whether the marshaller skips a type entirely at the wire-format
30   * layer, this annotation describes whether something participates in <i>bean detection</i>. It lives
31   * in <c>juneau-commons</c> so the bean-modeling layer can describe non-bean types and excluded bean
32   * members without depending on the marshalling layer.
33   *
34   * <p>
35   * Behavior depends on the target:
36   * <ul>
37   * 	<li><b>TYPE (class)</b> — marks the class as <i>not a bean</i>. The marshaller falls through to
38   * 		its other type-detection logic (object swaps, {@link BeanType @BeanType}-style
39   * 		hints, {@code @Marshalled(as=STRING)}, etc.) instead of treating the class as a bean.
40   * 	<li><b>FIELD</b> — excludes the field from bean property discovery. Use {@link #ignoreAccessors()}
41   * 		to also suppress matching JavaBean accessors.
42   * 	<li><b>METHOD</b> — excludes the getter/setter from bean property discovery.
43   * 	<li><b>CONSTRUCTOR</b> — excludes the constructor from constructor detection.
44   * </ul>
45   *
46   * <h5 class='section'>Java Records:</h5>
47   * <p>
48   * Ignoring individual record components is not supported during parsing. Because records are
49   * immutable, all components must be provided to the canonical constructor. Applying this
50   * annotation to a record component's accessor method or field will exclude it from serialization
51   * output, but the parser will be unable to instantiate the record if the component value is missing
52   * from the input.
53   *
54   * <h5 class='section'>See Also:</h5><ul>
55   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/BeanIgnoreAnnotation">@BeanIgnore Annotation</a>
56   * </ul>
57   */
58  @Documented
59  @Target({ FIELD, METHOD, TYPE, CONSTRUCTOR })
60  @Retention(RUNTIME)
61  @Inherited
62  public @interface BeanIgnore {
63  
64  	/**
65  	 * Optional description for the exposed API.
66  	 *
67  	 * @return The annotation value.
68  	 * @since 9.2.0
69  	 */
70  	String[] description() default {};
71  
72  	/**
73  	 * When <jk>true</jk> and this annotation is on a <jk>field</jk>, JavaBean accessors (<c>getX</c>/<c>setX</c>,
74  	 * <c>isX</c>) for the same logical property are also excluded from bean metadata.
75  	 *
76  	 * <p>
77  	 * Default is <jk>false</jk>: {@code @BeanIgnore} on a field only excludes the field from field-based discovery;
78  	 * accessors can still expose the property (for example when bean field visibility is set to {@code NONE}).
79  	 * Set to <jk>true</jk> to omit the property from serialization and parsing while keeping accessors for other
80  	 * frameworks.
81  	 *
82  	 * @return The annotation value.
83  	 */
84  	boolean ignoreAccessors() default false;
85  
86  }
