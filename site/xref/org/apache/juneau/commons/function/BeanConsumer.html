17  package org.apache.juneau.commons.function;
18  
19  /**
20   * A lifecycle-aware consumer for use as a parser target, receiving deserialized beans one at a time.
21   *
22   * <p>
23   * Extends {@link ThrowingConsumer} so that {@link #acceptThrows(Object)} supports checked exceptions
24   * directly — no wrapping in {@link RuntimeException} required for operations such as JDBC inserts.
25   *
26   * <p>
27   * The parser drives the full lifecycle:
28   * <ol>
29   * 	<li>Calls {@link #begin()} before the first element
30   * 	<li>Calls {@link #acceptThrows(Object)} for each parsed element
31   * 	<li>If {@link #acceptThrows(Object)} throws, calls {@link #onError(Exception)}:
32   * 	    <ul>
33   * 	    	<li>If {@code onError()} absorbs the exception (doesn't rethrow), parsing continues to the
34   * 	    	    next element (skip-and-continue / fault-tolerant ingestion)
35   * 	    	<li>If {@code onError()} rethrows, parsing stops and the exception propagates
36   * 	    </ul>
37   * 	<li>Always calls {@link #complete()} after all elements (like {@code finally}) — whether parsing
38   * 	    succeeded or {@code onError()} rethrew. Use this for resource cleanup (close connections, etc.)
39   * </ol>
40   *
41   * <p>
42   * For round-trip support (same property used for both serialization and parsing), use
43   * {@link BeanChannel} instead, which extends both {@link BeanConsumer} and {@link BeanSupplier}.
44   *
45   * <p>
46   * If a {@link BeanConsumer} is encountered during serialization (not parsing), the serializer will
47   * throw an {@link IllegalArgumentException} with a message recommending {@link BeanSupplier} or
48   * {@link BeanChannel}.
49   *
50   * <h5 class='section'>Example (DB-backed consumer with batch commits):</h5>
51   * <p class='bjava'>
52   * 	<ja>@Marshalled</ja>(factory=ItemConsumerFactory.<jk>class</jk>)
53   * 	<jk>public class</jk> ItemConsumer <jk>implements</jk> BeanConsumer&lt;Item&gt; {
54   * 		<jk>private</jk> Connection <jv>conn</jv>;
55   * 		<jk>private</jk> PreparedStatement <jv>stmt</jv>;
56   * 		<jk>private int</jk> <jv>count</jv>;
57   *
58   * 		<ja>@Override</ja> <jk>public void</jk> begin() <jk>throws</jk> Exception {
59   * 			<jv>conn</jv> = <jv>ds</jv>.getConnection();
60   * 			<jv>conn</jv>.setAutoCommit(<jk>false</jk>);
61   * 			<jv>stmt</jv> = <jv>conn</jv>.prepareStatement(<js>"INSERT INTO items (name) VALUES (?)"</js>);
62   * 		}
63   *
64   * 		<ja>@Override</ja> <jk>public void</jk> acceptThrows(Item <jv>item</jv>) <jk>throws</jk> Exception {
65   * 			<jv>stmt</jv>.setString(1, <jv>item</jv>.getName());
66   * 			<jv>stmt</jv>.executeUpdate();
67   * 			<jk>if</jk> (++<jv>count</jv> % 500 == 0) <jv>conn</jv>.commit(); <jc>// batch commit</jc>
68   * 		}
69   *
70   * 		<ja>@Override</ja> <jk>public void</jk> onError(Exception <jv>e</jv>) <jk>throws</jk> Exception {
71   * 			<jv>conn</jv>.rollback(); <jk>throw</jk> <jv>e</jv>;
72   * 		}
73   *
74   * 		<ja>@Override</ja> <jk>public void</jk> complete() <jk>throws</jk> Exception {
75   * 			<jv>conn</jv>.commit(); <jv>stmt</jv>.close(); <jv>conn</jv>.close();
76   * 		}
77   * 	}
78   * </p>
79   *
80   * <h5 class='section'>See Also:</h5><ul>
81   * 	<li class='jc'>{@link BeanSupplier} - Serialize-only lifecycle interface
82   * 	<li class='jc'>{@link BeanChannel} - Round-trip lifecycle interface (extends both)
83   * 	<li class='jc'>{@link BeanFactory} - Universal factory for DI framework integration
84   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/Marshalling">Marshalling</a>
85   * </ul>
86   *
87   * @param <T> The type of bean received by this consumer.
88   */
89  @SuppressWarnings({
90  	"java:S112" // throws Exception intentional - lifecycle methods may throw any checked exception
91  })
92  public interface BeanConsumer<T> extends ThrowingConsumer<T> {
93  
94  	/**
95  	 * Called before the first element is accepted.
96  	 *
97  	 * <p>
98  	 * Use this to open database connections, prepare statements, or perform other setup.
99  	 *
100 	 * @throws Exception If setup fails.
101 	 */
102 	default void begin() throws Exception {}
103 
104 	/**
105 	 * Always called after all elements have been processed (like {@code finally}).
106 	 *
107 	 * <p>
108 	 * Use this to close database connections, statements, or release other resources.
109 	 * This method is called whether parsing succeeded or {@link #onError(Exception)} rethrew.
110 	 *
111 	 * @throws Exception If cleanup fails.
112 	 */
113 	default void complete() throws Exception {}
114 
115 	/**
116 	 * Called when {@link #acceptThrows(Object)} throws an exception.
117 	 *
118 	 * <p>
119 	 * The default implementation rethrows the exception, stopping parsing immediately.
120 	 * Override to absorb the exception for fault-tolerant ingestion (e.g., log and skip bad records).
121 	 * Note that {@link #complete()} is always called afterward regardless.
122 	 *
123 	 * @param e The exception thrown by {@link #acceptThrows(Object)}.
124 	 * @throws Exception If the error cannot be recovered from. Rethrow {@code e} to stop parsing.
125 	 */
126 	default void onError(Exception e) throws Exception {
127 		throw e;
128 	}
129 }
