17  package org.apache.juneau.commons.conversion;
18  
19  import java.util.*;
20  
21  /**
22   * Functional interface representing a single type conversion used by {@link CachingConverter}.
23   *
24   * <p>
25   * The {@code args} parameter carries any type arguments needed for parameterized output types.
26   * For example, when converting to a {@link List}, {@code args[0]} would be the element type.
27   * For a {@link Map}, {@code args[0]} is the key type and {@code args[1]} is the value type.
28   *
29   * <p>
30   * A single cached {@code Conversion} function can serve all parameterizations of the same raw output
31   * type by accepting different {@code args} values at runtime.
32   *
33   * @param <I> The input type.
34   * @param <O> The output type.
35   */
36  @FunctionalInterface
37  public interface Conversion<I,O> {
38  
39  	/**
40  	 * Converts the input object to the output type.
41  	 *
42  	 * @param in The input object.
43  	 * @param memberOf The outer instance for non-static inner class construction, or <jk>null</jk>.
44  	 * @param session The converter session providing contextual objects such as {@link TimeZone} or
45  	 * 	{@link Locale}, or <jk>null</jk> if no session is available.
46  	 * @param args Optional type arguments for parameterized output types (e.g. element type for collections).
47  	 * @return The converted object.
48  	 */
49  	O to(I in, Object memberOf, ConverterSession session, Class<?>... args);
50  }
