21  import java.lang.reflect.*;
22  import java.util.*;

24  /**
25   * Reflection-based base class for {@link VarFunction} implementations that prefer typed
26   * argument signatures over manual {@code List<Object>} unpacking.
27   *
28   * <p>
29   * Subclasses declare one or more public {@code invoke(...)} methods whose parameter lists mirror
30   * the function's expected arguments. Per-arg target types and arities are derived once at
31   * construction via reflection; per-call dispatch picks the matching overload by argument count
32   * and runs one reflective invoke + N {@link ArgCoercer} coercions.
33   *
34   * <h5 class='section'>Method-resolution rules:</h5>
35   * <ul>
36   * 	<li>The subclass declares one or more public methods named {@code invoke}. Each must have a
37   * 		distinct fixed-arity (or be the single variadic overload — see below).
38   * 	<li>The return type must be {@code String} (or a subtype convertible via
39   * 		{@link Object#toString()}).
40   * 	<li>Parameter types may be any of: {@code String}, {@code int}, {@code long}, {@code double},
41   * 		{@code boolean} (or their wrappers), {@code String[]}, or {@code Object}. The final slot
42   * 		may be {@code String[]} for variadic functions.
43   * 	<li>If the subclass needs the {@link VarResolverSession session} (e.g. to look up beans),
44   * 		declare {@code VarResolverSession} as the <i>first</i> parameter.
45   * 	<li>At most one variadic overload (final {@code String[]} slot) per subclass; combined with
46   * 		fixed-arity overloads as long as the arity counts don't overlap.
47   * </ul>
48   *
49   * <h5 class='section'>Example — multi-arity:</h5>
50   * <p class='bjava'>
51   * 	<jk>public class</jk> Substring <jk>extends</jk> TypedFunction {
52   * 		<ja>@Override</ja> <jk>public</jk> String name() { <jk>return</jk> <js>"substring"</js>; }
53   * 		<jk>public</jk> String invoke(String <jv>s</jv>, <jk>int</jk> <jv>start</jv>) { <jk>return</jk> <jv>s</jv>.substring(<jv>start</jv>); }
54   * 		<jk>public</jk> String invoke(String <jv>s</jv>, <jk>int</jk> <jv>start</jv>, <jk>int</jk> <jv>end</jv>) { <jk>return</jk> <jv>s</jv>.substring(<jv>start</jv>, <jv>end</jv>); }
55   * 	}
56   * </p>
57   */
58  @SuppressWarnings({
59  	"java:S3776", // Cognitive complexity acceptable for the reflection-driven dispatch.
60  })
61  public abstract class TypedFunction implements VarFunction {
62  
63  	/**
64  	 * Per-overload reflection metadata.
65  	 *
66  	 * @param method The reflected {@code invoke} method.
67  	 * @param argTypes The argument types, excluding any leading session arg.
68  	 * @param takesSession Whether the method takes a leading {@link VarResolverSession} argument.
69  	 * @param variadic Whether the final argument slot is a variadic {@code String[]}.
70  	 */
71  	private record Overload(Method method, Class<?>[] argTypes, boolean takesSession, boolean variadic) {
72  
73  		/** Arg count excluding the variadic slot. */
74  		int fixedCount() {
75  			return variadic ? argTypes.length - 1 : argTypes.length;
76  		}
77  
78  		boolean acceptsArity(int n) {
79  			if (variadic)
80  				return n >= fixedCount();
81  			return n == fixedCount();
82  		}
83  
84  		@Override /* Content-based: the 'argTypes' array must compare by content, not identity (S6218). */
85  		public boolean equals(Object o) {
86  			return o instanceof Overload x
87  				&& Objects.equals(method, x.method)
88  				&& Arrays.equals(argTypes, x.argTypes)
89  				&& takesSession == x.takesSession
90  				&& variadic == x.variadic;
91  		}
92  
93  		@Override
94  		public int hashCode() {
95  			return 31 * Objects.hash(method, takesSession, variadic) + Arrays.hashCode(argTypes);
96  		}
97  
98  		@Override /* Renders the 'argTypes' array by content rather than identity (S6218). */
99  		public String toString() {
100 			return "Overload[method=" + method + ", argTypes=" + Arrays.toString(argTypes) + ", takesSession=" + takesSession + ", variadic=" + variadic + "]";
101 		}
102 	}
103 
104 	/** All declared {@code invoke} overloads, sorted by descending arity for predictable matching. */
105 	private final Overload[] overloads;
106 	private final int minArity;
107 	private final int maxArity;
108 
109 	protected TypedFunction() {
110 		this.overloads = findInvokeOverloads(getClass());
111 		var lo = Integer.MAX_VALUE;
112 		var hi = 0;
113 		var anyVariadic = false;
114 		for (var o : overloads) {
115 			if (o.variadic()) {
116 				anyVariadic = true;
117 				lo = Math.min(lo, o.fixedCount());
118 			} else {
119 				lo = Math.min(lo, o.fixedCount());
120 				hi = Math.max(hi, o.fixedCount());
121 			}
122 		}
123 		this.minArity = lo == Integer.MAX_VALUE ? 0 : lo;
124 		this.maxArity = anyVariadic ? Integer.MAX_VALUE : hi;
125 	}
126 
127 	@Override
128 	public final int minArity() { return minArity; }
129 
130 	@Override
131 	public final int maxArity() { return maxArity; }
132 
133 	@Override
134 	public final String invoke(VarResolverSession session, List<Object> args) {
135 		var n = args.size();
136 		Overload o = null;
137 		for (var c : overloads) {
138 			if (c.acceptsArity(n)) { o = c; break; }
139 		}
140 		if (o == null)
141 			throw iaex("Function '%s' has no overload accepting %s argument(s)", name(), n);
142 
143 		var coerced = ArgCoercer.coerce(name(), o.argTypes(), args);
144 		try {
145 			Object[] effective;
146 			if (o.takesSession()) {
147 				effective = new Object[coerced.length + 1];
148 				effective[0] = session;
149 				System.arraycopy(coerced, 0, effective, 1, coerced.length);
150 			} else {
151 				effective = coerced;
152 			}
153 			var result = o.method().invoke(this, effective);
154 			return result == null ? "" : result.toString();
155 		} catch (InvocationTargetException e) {
156 			var cause = e.getTargetException();
157 			if (cause instanceof RuntimeException cause2)
158 				throw cause2;
159 			throw iaex("Function '%s' threw %s: %s", name(), cns(cause), cause.getMessage());
160 		} catch (IllegalAccessException e) {
161 			throw iaex("Function '%s' invoke method must be public: %s", name(), e.getMessage());
162 		}
163 	}
164 
165 	private static Overload[] findInvokeOverloads(Class<? extends TypedFunction> cls) {
166 		var found = new ArrayList<Overload>();
167 		for (var m : cls.getMethods()) {
168 			if (!"invoke".equals(m.getName())
169 				|| m.getDeclaringClass() == TypedFunction.class
170 				|| m.getDeclaringClass() == Object.class
171 				|| m.getDeclaringClass() == VarFunction.class)
172 				continue;
173 			var allParams = m.getParameterTypes();
174 			var takesSession = allParams.length > 0 && allParams[0] == VarResolverSession.class;
175 			var startIdx = takesSession ? 1 : 0;
176 			var argTypes = Arrays.copyOfRange(allParams, startIdx, allParams.length);
177 			var variadic = argTypes.length > 0 && argTypes[argTypes.length - 1] == String[].class;
178 			found.add(new Overload(m, argTypes, takesSession, variadic));
179 		}
180 		if (found.isEmpty())
181 			throw iaex("TypedFunction subclass %s must declare a public invoke(...) method", cls.getName());
182 		// Sort by descending fixedCount so larger arity overloads get matched before any
183 		// variadic overload they might overlap with.
184 		found.sort((a, b) -> Integer.compare(b.fixedCount(), a.fixedCount()));
185 		return found.toArray(new Overload[0]);
186 	}
187 }
