17  package org.apache.juneau.commons.svl;
18  
19  /**
20   * One reference found in a compiled {@link VarTemplate}: a <c>${...}</c> var, a function or operator call, an
21   * identifier, or a syntax problem.
22   *
23   * <p>
24   * Offsets are into {@link VarTemplate#getSource()}: <c>start</c> inclusive, <c>end</c> exclusive.  For a var the span is
25   * the whole <c>${...}</c> marker; a <c>${name}</c> or <c>${name:default}</c> var reference also yields an {@link Kind#IDENT}
26   * entry for its key (the text before the first top-level <c>:</c>); for a function and an identifier it is the name token; for a syntax problem it is the
27   * offending marker.
28   *
29   * <p>
30   * References are reported for markers that are evaluated at render time.  A stable var folded at compile time (and an
31   * expression folded with it) has no entry.  A <c>${...}</c> inside a nested var body is not analyzed.
32   *
33   * @param kind What was referenced.
34   * @param name The var prefix (<c>P</c> for the <c>${name}</c> shortcut), the function name (operators are
35   * 	<c>$op.*</c>), the identifier, or for {@link Kind#SYNTAX} the error message.
36   * @param bound Whether the name resolves in the resolver the template was compiled against: a registered var, or a
37   * 	registered function permitted by the allowlist.  Always <jk>true</jk> for identifiers.
38   * @param listName For an identifier read as a row field (inside the body of a select or projection whose target is a
39   * 	top-level identifier), that identifier's name; <jk>null</jk> for a top-level identifier.  Identifiers in a body
40   * 	whose target is anything else are not reported.
41   * @param start Start offset, inclusive.
42   * @param end End offset, exclusive.
43   */
44  public record TemplateReference(Kind kind, String name, boolean bound, String listName, int start, int end) {
45  
46  	/** The kind of reference. */
47  	public enum Kind {
48  		/** A <c>${...}</c> var reference. */ VAR,
49  		/** A function or operator call inside an expression. */ FUNCTION,
50  		/** A bare identifier inside an expression. */ IDENT,
51  		/** A marker that could not be compiled (lenient mode keeps it as text). */ SYNTAX
52  	}
53  }
