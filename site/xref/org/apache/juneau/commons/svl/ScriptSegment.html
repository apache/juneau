17  package org.apache.juneau.commons.svl;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import java.util.*;
22  
23  /**
24   * A {@link TemplateSegment} that resolves a {@code #{name(args...)}} script call.
25   *
26   * <p>
27   * Built by {@link VarTemplateCompiler}'s recursive-descent parser. The function {@link #name} is
28   * the literal token between {@code #{}} and the opening parenthesis; {@link #argTemplates} hold
29   * the parsed arguments — each is itself a {@link VarTemplate} so that nested {@code ${...}} and
30   * {@code #{...}} resolve naturally per arg.
31   *
32   * <p>
33   * The {@link #cachedFn} field is set at compile time via the four-channel discovery chain
34   * (built-in / ServiceLoader / BeanStore / explicit). When null, the function name was
35   * unregistered at compile time; {@link #resolve(VarResolverSession, StringBuilder) resolve}
36   * then throws {@link IllegalArgumentException} with a lazy-fail message:
37   * "{@code No such function 'foo'}".
38   *
39   * <p>
40   * If a function reference is unregistered at compile time, the compiler still produces a valid
41   * {@link ScriptSegment} (with {@link #cachedFn} {@code null}); the failure surfaces lazily at
42   * resolve time so templates that reference functions only available in some sessions don't
43   * fail-fast at build time.
44   */
45  final class ScriptSegment extends TemplateSegment {
46  
47  	/**
48  	 * The function reference cached at compile time. {@code null} when the function name is
49  	 * unregistered (lazy-fail: error surfaces at resolve time).
50  	 */
51  	final VarFunction cachedFn;
52  
53  	/** The function name as written in the source (for error messages and diagnostics). */
54  	final String name;
55  
56  	/**
57  	 * The arguments — each pre-compiled into a {@link VarTemplate} so nested var/script
58  	 * markers compose naturally.
59  	 */
60  	final VarTemplate[] argTemplates;
61  
62  	ScriptSegment(VarFunction cachedFn, String name, VarTemplate[] argTemplates) {
63  		this.cachedFn = cachedFn;
64  		this.name = name;
65  		this.argTemplates = argTemplates;
66  	}
67  
68  	@Override
69  	void resolve(VarResolverSession session, StringBuilder out) {
70  		if (cachedFn == null)
71  			throw iaex("No such function '%s'", name);
72  
73  		// Resolve each arg template to a String, then dispatch to the function. TypedFunction
74  		// handles ArgCoercer.coerce(...) internally; direct VarFunction implementations get the
75  		// raw String args via the args list.
76  		var args = new ArrayList<Object>(argTemplates.length);
77  		for (var t : argTemplates)
78  			args.add(t.resolve(session));
79  
80  		var n = args.size();
81  		if (n < cachedFn.minArity() || n > cachedFn.maxArity()) {
82  			var maxStr = cachedFn.maxArity() == Integer.MAX_VALUE ? "any" : String.valueOf(cachedFn.maxArity());
83  			throw iaex("Function '%s' expected %s..%s arg(s), got %s",
84  				name, cachedFn.minArity(), maxStr, n);
85  		}
86  
87  		var result = cachedFn.invoke(session, args);
88  		if (result != null)
89  			out.append(result);
90  	}
91  }
