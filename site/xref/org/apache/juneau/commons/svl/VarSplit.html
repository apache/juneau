17  package org.apache.juneau.commons.svl;
18  
19  import java.util.*;
20  
21  /**
22   * The one scanner for SVL markers: finds the close of a <c>${...}</c> body or a <c>#{...}</c> expression, and the
23   * top-level separators of a <c>${...}</c> body.  The template compiler, the expression lexer and the colon split all use
24   * it, so they always agree on where a marker ends.
25   *
26   * <h5 class='section'>Rules inside a <c>${...}</c> body (body mode):</h5>
27   * <ul>
28   * 	<li>A backslash makes the next character literal (including another backslash, a quote or a brace).  The backslash
29   * 		is <em>not</em> stripped here; callers do that.
30   * 	<li>A part starts at the start of a body (just past <c>${</c>) and immediately after a <c>:</c>.  Only
31   * 		there does a <c>'</c> or <c>"</c> open a quoted run, which closes at the next unescaped matching quote (a
32   * 		backslash escapes inside it too).  Anywhere else an apostrophe or double quote is a plain character, so
33   * 		<c>${RQ:don't:b}</c> has the parts <c>RQ</c>, <c>don't</c> and <c>b</c>.
34   * 	<li><c>${</c> and a plain <c>{</c> open a nested body; <c>#{</c> opens a nested expression.  A <c>}</c> closes the
35   * 		innermost one.
36   * </ul>
37   *
38   * <h5 class='section'>Rules inside a <c>#{...}</c> expression (expression mode):</h5>
39   * <ul>
40   * 	<li>A <c>'</c> or <c>"</c> opens a SpEL string wherever it appears; a doubled quote (<c>''</c>, <c>""</c>) is an
41   * 		escaped quote and there is <em>no</em> backslash escape (so <c>'a\'</c> is the string <c>a\</c>).
42   * 	<li>Outside a string, a backslash makes the next character literal (as the template compiler always did, so an
43   * 		escaped <c>\}</c> does not end the expression; the expression parser then rejects the backslash).
44   * 	<li><c>${</c> opens a nested body (body mode); <c>#{</c> and a plain <c>{</c> open a nested expression.
45   * </ul>
46   *
47   * <p>
48   * A character is <em>top level</em> when it is in the outermost body and not quoted, escaped or nested.  Malformed
49   * input is tolerated by the split helpers: a stray <c>}</c> at top level is ignored, and an unclosed <c>{</c> or
50   * unterminated quote means the rest of the string is nested/quoted (left to the compiler's error path).  The scanner
51   * is iterative, so arbitrarily deep nesting cannot overflow the stack.
52   */
53  @SuppressWarnings({
54  	"java:S135", // The scanner loop uses continue and break for state transitions,
55  	"java:S3776" // The scanner is a single state machine
56  })
57  final class VarSplit {
58  
59  	private static final int BODY = 0;
60  	private static final int EXPR = 1;
61  
62  	/**
63  	 * Internal-only stand-in for a literal comma (an escaped {@code \,}) in the internal comma form handed to
64  	 * {@link DefaultingVar}, {@link MultipartVar} and {@link MultipartResolvingVar}.  A Unicode private-use character, so
65  	 * it cannot be produced by the grammar.  The var classes call {@link #restoreCommas(String)} after splitting.
66  	 */
67  	static final char COMMA = '\uE000';
68  
69  	/**
70  	 * Turns {@link #COMMA} back into a real comma and {@link #BACKSLASH} into a backslash.
71  	 *
72  	 * @param s The text.  Can be <jk>null</jk>.
73  	 * @return The text with real commas.
74  	 */
75  	static String restoreCommas(String s) {
76  		if (s == null || (s.indexOf(COMMA) == -1 && s.indexOf(BACKSLASH) == -1))
77  			return s;
78  		return s.replace(COMMA, ',').replace(BACKSLASH, '\\');
79  	}
80  
81  	/** Internal-only stand-in for the literal backslash of an escaped backslash; restored with {@link #COMMA}. */
82  	static final char BACKSLASH = '\uE001';
83  
84  	private VarSplit() {}
85  
86  	/**
87  	 * Returns the index of the first top-level occurrence of <c>c</c>, or <c>-1</c>.
88  	 *
89  	 * @param s The body to scan.
90  	 * @param c The separator character.
91  	 * @return The index, or <c>-1</c> if none.
92  	 */
93  	static int firstTopLevel(String s, char c) {
94  		var r = hits(s, c);
95  		return r.isEmpty() ? -1 : r.get(0);
96  	}
97  
98  	/**
99  	 * Returns the index of the last top-level occurrence of <c>c</c>, or <c>-1</c>.
100 	 *
101 	 * @param s The body to scan.
102 	 * @param c The separator character.
103 	 * @return The index, or <c>-1</c> if none.
104 	 */
105 	static int lastTopLevel(String s, char c) {
106 		var r = hits(s, c);
107 		return r.isEmpty() ? -1 : r.get(r.size() - 1);
108 	}
109 
110 	/**
111 	 * Splits at every top-level occurrence of <c>c</c>, keeping empty segments.
112 	 *
113 	 * @param s The body to split.
114 	 * @param c The separator character.
115 	 * @return A list with at least one element.
116 	 */
117 	static List<String> splitTopLevel(String s, char c) {
118 		var out = new ArrayList<String>();
119 		int start = 0;
120 		for (var i : hits(s, c)) {
121 			out.add(s.substring(start, i));
122 			start = i + 1;
123 		}
124 		out.add(s.substring(start));
125 		return out;
126 	}
127 
128 	/**
129 	 * Returns the index of the <c>}</c> that closes a <c>${...}</c> body starting at {@code from}, or <c>-1</c>.
130 	 *
131 	 * @param s The text.
132 	 * @param from The index of the first body character (just past <c>${</c>).
133 	 * @return The index of the closing brace, or <c>-1</c> if the body is unterminated.
134 	 */
135 	static int closeOfBody(String s, int from) {
136 		return scan(s, from, BODY, false, (char)0, null);
137 	}
138 
139 	/**
140 	 * Returns the index of the <c>}</c> that closes a <c>#{...}</c> expression starting at {@code from}, or <c>-1</c>.
141 	 *
142 	 * @param s The text.
143 	 * @param from The index of the first expression character (just past <c>#{</c>).
144 	 * @return The index of the closing brace, or <c>-1</c> if the expression is unterminated.
145 	 */
146 	static int closeOfExpr(String s, int from) {
147 		return scan(s, from, EXPR, false, (char)0, null);
148 	}
149 
150 	private static List<Integer> hits(String s, char c) {
151 		var hits = new ArrayList<Integer>();
152 		scan(s, 0, BODY, true, c, hits);
153 		return hits;
154 	}
155 
156 	/**
157 	 * Scans from {@code from} in the given mode.
158 	 *
159 	 * @param toEnd <jk>true</jk> to scan a whole body (stray top-level <c>}</c> ignored), <jk>false</jk> to stop at the
160 	 * 	closing brace.
161 	 * @param sep The separator to record at top level (only when {@code hits} is not <jk>null</jk>).
162 	 * @param hits Receives top-level separator indexes.  Can be <jk>null</jk>.
163 	 * @return The closing brace index, <c>s.length()</c> when {@code toEnd}, or <c>-1</c> if unterminated.
164 	 */
165 	private static int scan(String s, int from, int mode0, boolean toEnd, char sep, List<Integer> hits) {
166 		var n = s.length();
167 		var stack = new ArrayDeque<Integer>();
168 		stack.push(mode0);
169 		var partStart = mode0 == BODY;
170 		var i = from;
171 		while (i < n) {
172 			int mode = stack.peek();
173 			var c = s.charAt(i);
174 			if (mode == BODY) {
175 				if (c == '\\') {
176 					i += 2;
177 					partStart = false;
178 					continue;
179 				}
180 				if (partStart && (c == '\'' || c == '"')) {
181 					var j = i + 1;
182 					while (j < n && s.charAt(j) != c)
183 						j += s.charAt(j) == '\\' ? 2 : 1;
184 					if (j >= n)
185 						break;
186 					i = j + 1;
187 					partStart = false;
188 					continue;
189 				}
190 				partStart = c == ':';
191 			} else if (c == '\\') {
192 				i += 2;
193 				continue;
194 			} else if (c == '\'' || c == '"') {
195 				var j = i + 1;
196 				while (j < n && (s.charAt(j) != c || (j + 1 < n && s.charAt(j + 1) == c)))
197 					j += s.charAt(j) == c ? 2 : 1;
198 				if (j >= n)
199 					break;
200 				i = j + 1;
201 				continue;
202 			}
203 			if ((c == '$' || c == '#') && i + 1 < n && s.charAt(i + 1) == '{') {
204 				stack.push(c == '$' ? BODY : EXPR);
205 				partStart = c == '$';
206 				i += 2;
207 				continue;
208 			}
209 			if (c == '{') {
210 				stack.push(mode);
211 				partStart = false;
212 			} else if (c == '}') {
213 				if (stack.size() > 1) {
214 					stack.pop();
215 					partStart = false;
216 				} else if (! toEnd) {
217 					return i;
218 				}
219 			} else if (hits != null && c == sep && mode == BODY && stack.size() == 1) {
220 				hits.add(i);
221 			}
222 			i++;
223 		}
224 		return toEnd ? n : -1;
225 	}
