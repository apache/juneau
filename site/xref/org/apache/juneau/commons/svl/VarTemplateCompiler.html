26  /**
27   * Single tokenizer + recursive-descent parser for SVL templates.
28   *
29   * <p>
30   * One pass over the input string produces an array of {@link TemplateSegment} objects:
31   * {@link LiteralSegment} for plain text, {@link VarRefSegment} for {@code ${...}} /
32   * {@code $X{...}} markers, and {@link ScriptSegment} for {@code #{name(args...)}} markers. The
33   * compiler is the only tokenizer in the codebase — both
34   * {@link VarResolverSession#resolve(String)} and {@link VarResolverSession#resolveTo(String, java.io.Writer)}
35   * dispatch through {@link VarResolver#compile(String)}.
36   *
37   * <h5 class='section'>Tokenizer behavior (ported verbatim from the legacy in-place tokenizer):</h5>
38   * <ul>
39   * 	<li><b>Top-level escape:</b> {@code \\}, {@code \$}, {@code \#} emit just the trailing char.
40   * 		Other {@code \X} sequences emit {@code \X} verbatim (matches existing
41   * 		{@link VarResolverSession#resolveTo(String, java.io.Writer)} semantics for {@code \\}/{@code \$};
42   * 		{@code \#} is added new for symmetry with {@code \$}).
43   * 	<li><b>{@code ${...}} / {@code $X{...}} parse:</b> after seeing {@code $}, an optional
44   * 		ASCII-letter prefix is captured up to the {@code {}}; bodies are collected with brace
45   * 		depth tracking; inner {@code \\} escapes inside the body are recognized for
46   * 		state-machine purposes (don't close on {@code \}}) but the captured body string preserves
47   * 		backslashes verbatim. Unknown prefix → fallthrough emits the original substring with
48   * 		{@code \\\$\{\}} unescaped per the legacy {@code AS2} set.
49   * 	<li><b>{@code #{...}} parse:</b> when {@code #} is followed immediately by {@code {}}, body
50   * 		collection enters script mode (same depth/escape tracking as {@code ${...}} bodies). On
51   * 		close, the body is parsed via the recursive-descent parser implemented below.
52   * 	<li><b>{@code ${...}} shortcut:</b> empty prefix triggers the
53   * 		{@link VarResolverSession DOLLAR_BRACE_VAR} fallback, with the first top-level {@code ':'}
54   * 		in the body rewritten to {@code ','} <i>at compile time</i> on the body source string.
55   * 	<li><b>End-of-input:</b> dangling {@code $} (state S2), open {@code $X{}} (state S3), or open
56   * 		{@code #{}} (state S3H) preserve the legacy fallthrough — the relevant prefix + body
57   * 		fragment is appended verbatim (with the legacy {@code AS1}/{@code AS2} unescaping) so a
58   * 		malformed input round-trips identically through both legacy and compiled paths.
59   * </ul>
60   *
61   * <h5 class='section'>Recursive-descent script-arg parser:</h5>
62   * <p>
63   * Inside a {@code #{...}} body, the parser splits {@code name(arg, arg, ...)} into the function
64   * name and a list of {@link VarTemplate}-typed arguments. Each argument is one of:
65   * <ul>
66   * 	<li>Quoted string — {@code "..."} or {@code '...'}, with {@code \"} / {@code \'} / {@code \\}
67   * 		escapes.
68   * 	<li>Bare identifier — letters, digits, underscores, {@code -}, {@code .}; treated as a
69   * 		literal string (e.g. {@code switch}-case labels).
70   * 	<li>Numeric literal — integer or decimal; treated as a literal string.
71   * 	<li>Boolean literal — {@code true} / {@code false}; treated as a literal string.
72   * 	<li>Nested marker — {@code ${...}}, {@code $X{...}}, or {@code #{...}}; compiled to a
73   * 		nested {@link VarTemplate}.
74   * </ul>
75   *
76   * <p>
77   * Whitespace between arg tokens (and around commas) is tolerated and discarded.
78   *
79   * <h5 class='section'>Function binding:</h5>
80   * <p>
81   * {@link VarFunction} references are resolved at compile time against the supplied
82   * {@link VarResolver}'s function map. Unknown functions are left with a {@code null}
83   * {@link ScriptSegment#cachedFn cachedFn}, deferring the {@code "No such function"} failure to
84   * resolve time (lazy failure preserves backward compatibility for compile-once / resolve-many
85   * flows where an unrelated branch references a missing function).
86   *
87   * <h5 class='section'>Stable-value folding:</h5>
88   * <p>
89   * If a captured {@link Var} opts in via {@link Var#isStable()} and its body folded to a literal
90   * (no nested live markers), the var is invoked once at compile time and replaced with the
91   * resolved literal. If {@link Var#canResolve(VarResolverSession)} returns {@code false} or
92   * {@link Var#doResolve(VarResolverSession, String)} throws, folding is skipped and the dispatch
93   * segment is emitted unchanged.
94   */
95  @SuppressWarnings({
96  	"java:S125", // State-machine narration in comments
97  	"java:S2583", // Condition always true/false; state persists across iterations
98  	"java:S6541", // Brain-method threshold; intentional state-machine consolidation
99  	"java:S3776", // Cognitive complexity acceptable for this state-machine + parser
100 	"java:S110", // Inheritance depth: the segment hierarchy is intentionally shallow
101 })

104 	private static final AsciiSet AS1 = AsciiSet.of("\\{");
105 	private static final AsciiSet AS2 = AsciiSet.of("\\${}");
106 
107 	/** Var name targeted by the {@code ${xxx}} shortcut. Mirrors {@code VarResolverSession.DOLLAR_BRACE_VAR}. */
108 	private static final String DOLLAR_BRACE_VAR = "P";
109 
110 	private static final int S1 = 1;  // Outside any var/script
111 	private static final int S2 = 2;  // Saw '$', looking for '{'
112 	private static final int S3 = 3;  // Inside ${...} / $X{...} body, looking for matching '}'
113 	private static final int S3H = 4; // Inside #{...} body, looking for matching '}'
114 
115 	private final VarResolver resolver;
116 
117 	private VarTemplateCompiler(VarResolver resolver) {
118 		this.resolver = resolver;
119 	}

121 	/**
122 	 * Compiles the given input string against the supplied resolver.
123 	 *
124 	 * @param resolver The resolver whose var registry is used for compile-time {@link Var}
125 	 *	 binding. Must not be {@code null}.
126 	 * @param input The template source. May be {@code null}.
127 	 * @return The compiled template. Never {@code null}.
128 	 */
129 	static VarTemplate compile(VarResolver resolver, String input) {
130 		if (input == null)
131 			return new VarTemplate(resolver, null, new TemplateSegment[0]);
132 		if (input.isEmpty())
133 			return new VarTemplate(resolver, input, new TemplateSegment[0]);
134 		// Fast path: no special chars at all.
135 		if (input.indexOf('$') == -1 && input.indexOf('\\') == -1 && input.indexOf('#') == -1)
136 			return new VarTemplate(resolver, input, new TemplateSegment[]{new LiteralSegment(input)});
137 		return new VarTemplateCompiler(resolver).compileImpl(input);
138 	}

140 	/**
141 	 * Compiles {@code input} into segments. Mirrors the legacy
142 	 * {@link VarResolverSession#resolveTo(String, java.io.Writer)} state machine, but writes
143 	 * to a segment list instead of a {@link java.io.Writer}.
144 	 */
145 	private VarTemplate compileImpl(String input) {
146 		var segments = new ArrayList<TemplateSegment>();
147 		var literal = new StringBuilder();
148 
149 		var state = S1;
150 		var isInEscape = false;
151 		var hasInternalVar = false;
152 		var hasInnerEscapes = false;
153 		String varType = null;
154 		var x = 0;        // index of opening '{' of current body (state S3/S3H)
155 		var x2 = 0;       // index of opening '$' or '#' of current var/script
156 		var depth = 0;
157 		var isDollarBraceShortcut = false;
158 		var length = input.length();
159 
160 		var i = 0;
161 		while (i < length) {
162 			var c = input.charAt(i);
163 			if (state == S1) {
164 				if (isInEscape) {
165 					if (c == '\\' || c == '$' || c == '#') {
166 						literal.append(c);
167 					} else {
168 						literal.append('\\').append(c);
169 					}
170 					isInEscape = false;
171 				} else if (c == '\\') {
172 					isInEscape = true;
173 				} else if (c == '$') {
174 					x = i;
175 					x2 = i;
176 					state = S2;
177 				} else if (c == '#' && i + 1 < length && input.charAt(i + 1) == '{') {
178 					// Script trigger: '#{' begins a script segment. Flush pending literal,
179 					// advance past the '{', and enter S3H to collect the body.
180 					flushLiteral(segments, literal);
181 					x2 = i;
182 					i++;            // consume '{'
183 					x = i;          // body start (the '{' position; body is substring(x+1, i))
184 					depth = 0;
185 					state = S3H;
186 				} else {
187 					literal.append(c);
188 				}
189 			} else if (state == S2) {
190 				if (isInEscape) {
191 					isInEscape = false;
192 				} else if (c == '\\') {
193 					hasInnerEscapes = true;
194 					isInEscape = true;
195 				} else if (c == '{') {
196 					varType = input.substring(x + 1, i);
197 					if (varType.isEmpty())
198 						isDollarBraceShortcut = true;
199 					x = i;
200 					depth = 0;
201 					state = S3;
202 				} else if (c < 'A' || c > 'z' || (c > 'Z' && c < 'a')) {  // False trigger "$X "
203 					if (hasInnerEscapes)
204 						literal.append(unescapeChars(input.substring(x, i + 1), AS1));
205 					else
206 						literal.append(input, x, i + 1);
207 					state = S1;
208 					hasInnerEscapes = false;
209 				}
210 			} else if (state == S3) {
211 				if (isInEscape) {
212 					isInEscape = false;
213 				} else if (c == '\\') {
214 					isInEscape = true;
215 					hasInnerEscapes = true;
216 				} else if (c == '{') {
217 					depth++;
218 					hasInternalVar = true;
219 				} else if (c == '}') {
220 					if (depth > 0) {
221 						depth--;
222 					} else {
223 						// Body collected. Flush pending literal first, then build the var ref.
224 						flushLiteral(segments, literal);
225 						var bodySource = input.substring(x + 1, i);
226 						if (isDollarBraceShortcut)
227 							bodySource = translateDollarBraceDefault(bodySource);
228 
229 						// Look up the cached Var: prefer explicit empty-name reg if any, else
230 						// fall back to PropertyVar for the shortcut.
231 						var v = resolver.getVarMap().get(varType);
232 						if (v == null && isDollarBraceShortcut)
233 							v = resolver.getVarMap().get(DOLLAR_BRACE_VAR);
234 
235 						// Pre-compute fallthrough text per legacy AS2 unescaping rules.
236 						String fallthrough;
237 						if (hasInnerEscapes)
238 							fallthrough = unescapeChars(input.substring(x2, i + 1), AS2);
239 						else
240 							fallthrough = input.substring(x2, i + 1);
241 
242 						// Recursively compile the body so nested markers compose naturally.
243 						var body = compile(resolver, bodySource);
244 
245 						// Stable-value folding: if the captured Var opted in via Var.isStable()
246 						// AND the body itself folded to a literal (no nested live markers), we
247 						// can resolve the var once at compile time and replace the dispatch
248 						// segment with the resolved literal text. Match the runtime "var must
249 						// canResolve" behavior — fold only if the var canResolve a no-bean
250 						// session (the compiler has no session of its own), and if doResolve
251 						// throws, treat as "cannot fold" and fall back to the dispatch segment.
252 						var folded = tryFoldStableVar(v, body, hasInternalVar);
253 						if (folded != null) {
254 							segments.add(new LiteralSegment(folded));
255 						} else {
256 							segments.add(new VarRefSegment(v, varType, body, hasInternalVar, fallthrough));
257 						}
258 
259 						state = S1;
260 						hasInnerEscapes = false;
261 						hasInternalVar = false;
262 						isDollarBraceShortcut = false;
263 					}
264 				}
265 			} else if (state == S3H) {
266 				if (isInEscape) {
267 					isInEscape = false;
268 				} else if (c == '\\') {
269 					isInEscape = true;
270 					hasInnerEscapes = true;
271 				} else if (c == '{') {
272 					depth++;
273 				} else if (c == '}') {
274 					if (depth > 0) {
275 						depth--;
276 					} else {
277 						// Script body collected; parse via recursive-descent.
278 						var bodySource = input.substring(x + 1, i);
279 						var seg = parseScriptBody(input, bodySource, x2, i);
280 						segments.add(seg);
281 						state = S1;
282 						hasInnerEscapes = false;
283 					}
284 				}
285 			}
286 			i++;
287 		}
288 
289 		// End-of-input handling — preserve legacy fallthrough exactly.
290 		if (isInEscape) {
291 			literal.append('\\');
292 		} else if (state == S2) {
293 			literal.append('$').append(unescapeChars(input.substring(x + 1), AS1));
294 		} else if (state == S3) {
295 			literal.append('$').append(varType).append('{').append(unescapeChars(input.substring(x + 1), AS2));
296 		} else if (state == S3H) {
297 			literal.append("#{").append(unescapeChars(input.substring(x + 1), AS2));
298 		}
299 		flushLiteral(segments, literal);
300 
301 		return new VarTemplate(resolver, input, segments.toArray(new TemplateSegment[0]));
302 	}
303 
304 	private static void flushLiteral(List<TemplateSegment> out, StringBuilder buf) {
305 		if (!buf.isEmpty()) {
306 			out.add(new LiteralSegment(buf.toString()));
307 			buf.setLength(0);
308 		}
309 	}

311 	/**
312 	 * Translates the first top-level {@code ':'} in a {@code ${...}}-shortcut body to {@code ','}
313 	 * so that the body matches {@link DefaultingVar}'s {@code key,default} separator.
314 	 *
315 	 * <p>
316 	 * Mirrors {@code VarResolverSession.translateDollarBraceDefault(...)} — applied on the body
317 	 * source string at compile time so any {@code ':'} produced by an inner var resolution at
318 	 * runtime is NOT re-interpreted as a default separator.
319 	 */
320 	private static String translateDollarBraceDefault(String body) {
321 		var depth = 0;
322 		var length = body.length();
323 		for (var i = 0; i < length; i++) {
324 			var c = body.charAt(i);
325 			if (c == '{')
326 				depth++;
327 			else if (c == '}' && depth > 0)
328 				depth--;
329 			else if (c == ':' && depth == 0)
330 				return body.substring(0, i) + ',' + body.substring(i + 1);
331 		}
332 		return body;
333 	}

339 	/**
340 	 * Parses a captured script body string into a {@link ScriptSegment}.
341 	 *
342 	 * @param fullInput The full template input (for diagnostic context).
343 	 * @param body The captured body — everything between {@code #{}} and the matching {@code }}.
344 	 * @param x2 Index of the opening {@code #} in {@code fullInput}.
345 	 * @param closeIndex Index of the matching {@code }} in {@code fullInput}.
346 	 */
347 	private ScriptSegment parseScriptBody(String fullInput, String body, int x2, int closeIndex) {
348 		var p = new ScriptParser(body, fullInput, x2, closeIndex);
349 		var name = p.parseFunctionName();
350 		p.expect('(');
351 		var args = p.parseArgList();
352 		p.expect(')');
353 		p.expectEnd();
354 		// Bind the function at compile time. Unknown functions stay null and fail
355 		// lazily at resolve time.
356 		var fn = resolver.getFunctionMap().get(name);
357 		return new ScriptSegment(fn, name, args.toArray(new VarTemplate[0]));
358 	}

360 	/**
361 	 * Attempt compile-time stable-value folding for a {@code ${...}} / {@code $X{...}} dispatch.
362 	 *
363 	 * <p>
364 	 * Folding is only valid when:
365 	 * <ul>
366 	 * 	<li>{@code v} is non-{@code null}.
367 	 * 	<li>{@code v.isStable()} returns {@code true}.
368 	 * 	<li>The compiled body is a pure literal — no nested live markers, including the legacy
369 	 * 		{@code hasInternalVar} flag set by the state machine on inner {@code {}} characters.
370 	 * 	<li>The var is a {@link SimpleVar} (streamed vars are excluded — folding a stream loses
371 	 * 		the writer-direct optimization for no compile-time gain).
372 	 * 	<li>{@link Var#canResolve(VarResolverSession)} on a fresh no-bean session returns
373 	 * 		{@code true}, and {@link Var#doResolve(VarResolverSession, String)} returns a
374 	 * 		non-{@code null} value without throwing.
375 	 * </ul>
376 	 *
377 	 * <p>
378 	 * Returns the folded literal text, or {@code null} if folding is skipped for any reason.
379 	 */
380 	@SuppressWarnings({
381 		"java:S1166" // Exception swallowed intentionally — fold-or-skip contract.
382 	})
383 	private String tryFoldStableVar(Var v, VarTemplate body, boolean hasInternalVar) {
384 		if (v == null || !v.isStable() || hasInternalVar || !(v instanceof SimpleVar))
385 			return null;
386 		if (!body.isLiteral())
387 			return null;
388 		try {
389 			var session = resolver.createSession();
390 			if (!v.canResolve(session))
391 				return null;
392 			// body.isLiteral() guaranteed above; resolve(session) short-circuits to literalText
393 			// and never touches session beans.
394 			var resolved = v.doResolve(session, body.resolve(session));
395 			if (resolved == null)
396 				return null;
397 			// Skip folding if the resolved value would itself need recursive resolution. The
398 			// runtime VarRefSegment.resolve(...) path post-resolves dispatch results that
399 			// contain ${...} / $X{...} markers when v.allowRecurse() returns true; folding
400 			// would silently drop that recursion. (Compare: VarRefSegment.resolve, which
401 			// re-invokes session.resolve(replacement) when the replacement contains '$' and
402 			// the var allows recursion.)
403 			if (v.allowRecurse() && resolved.indexOf('$') != -1)
404 				return null;
405 			return resolved;
406 		} catch (@SuppressWarnings("unused") Exception e) {
407 			return null;
408 		}
409 	}

411 	/**
412 	 * Hand-written recursive-descent parser for {@code name(arg, arg, ...)} script bodies.
413 	 *
414 	 * <p>
415 	 * The parser operates on the body string only — anything outside the {@code #{...}}
416 	 * delimiters has already been handled by the outer state machine. Each parsed argument is
417 	 * compiled into a {@link VarTemplate} via {@link VarTemplateCompiler#compile(VarResolver, String)}
418 	 * so nested markers and stable-value folding behave consistently with top-level template
419 	 * compilation.
420 	 */
421 	private final class ScriptParser {
422 
423 		private final String body;
424 		private final String fullInput;
425 		private final int x2;          // for error messages
426 		private final int closeIndex;  // for error messages
427 		private int pos;
428 
429 		ScriptParser(String body, String fullInput, int x2, int closeIndex) {
430 			this.body = body;
431 			this.fullInput = fullInput;
432 			this.x2 = x2;
433 			this.closeIndex = closeIndex;
434 		}
435 
436 		String parseFunctionName() {
437 			skipWs();
438 			var start = pos;
439 			while (pos < body.length()) {
440 				var c = body.charAt(pos);
441 				if ((c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') || c == '_') {
442 					pos++;
443 				} else {
444 					break;
445 				}
446 			}
447 			if (pos == start)
448 				throw parseError("Expected function name");
449 			var name = body.substring(start, pos);
450 			if (!Character.isLetter(name.charAt(0)))
451 				throw parseError("Function name must start with a letter (got '%s')", name);
452 			skipWs();
453 			return name;
454 		}
455 
456 		List<VarTemplate> parseArgList() {
457 			var args = new ArrayList<VarTemplate>();
458 			skipWs();
459 			if (peek() == ')')
460 				return args;
461 			args.add(parseArg());
462 			skipWs();
463 			while (peek() == ',') {
464 				pos++;
465 				skipWs();
466 				args.add(parseArg());
467 				skipWs();
468 			}
469 			return args;
470 		}
471 
472 		/**
473 		 * Parse a single argument. Returns a {@link VarTemplate} compiled from the captured
474 		 * argument source.
475 		 */
476 		VarTemplate parseArg() {
477 			skipWs();
478 			if (pos >= body.length())
479 				throw parseError("Unexpected end of script body");
480 			var c = body.charAt(pos);
481 			if (c == '"' || c == '\'')
482 				return parseQuotedString(c);
483 			if (c == '$' || c == '#')
484 				return parseNestedMarker();
485 			// Bare token: identifier / number / boolean — captured verbatim until comma /
486 			// close-paren / whitespace. Whitespace inside a bare token is invalid per the
487 			// recursive-descent grammar; the parser stops at the first whitespace and the next
488 			// expected token handles the remainder.
489 			return parseBareToken();
490 		}
491 
492 		/**
493 		 * Parse a quoted string, returning a {@link VarTemplate} whose source is the unquoted,
494 		 * escape-interpreted contents. The result is a literal-only template.
495 		 */
496 		VarTemplate parseQuotedString(char quote) {
497 			pos++; // skip opening quote
498 			var sb = new StringBuilder();
499 			while (pos < body.length()) {
500 				var c = body.charAt(pos);
501 				if (c == '\\' && pos + 1 < body.length()) {
502 					var next = body.charAt(pos + 1);
503 					if (next == quote || next == '\\') {
504 						sb.append(next);
505 						pos += 2;
506 						continue;
507 					}
508 					sb.append(c);
509 					pos++;
510 				} else if (c == quote) {
511 					pos++; // consume closing quote
512 					return compile(resolver, sb.toString());
513 				} else {
514 					sb.append(c);
515 					pos++;
516 				}
517 			}
518 			throw parseError("Unterminated quoted string in script body");
519 		}
520 
521 		/**
522 		 * Parse a nested {@code ${...}}, {@code $X{...}}, or {@code #{...}} marker by capturing
523 		 * the substring up to the matching close brace and recursively compiling it.
524 		 */
525 		VarTemplate parseNestedMarker() {
526 			var start = pos;
527 			var first = body.charAt(pos);
528 			pos++;
529 			// Consume the prefix (zero or more letters for $X{, none for #{).
530 			if (first == '$') {
531 				while (pos < body.length()) {
532 					var c = body.charAt(pos);
533 					if ((c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z')) {
534 						pos++;
535 					} else {
536 						break;
537 					}
538 				}
539 			}
540 			if (pos >= body.length() || body.charAt(pos) != '{')
541 				throw parseError("Expected '{' after '%s' in nested marker", first);
542 			pos++; // consume '{'
543 			var depth = 1;
544 			var inEscape = false;
545 			while (pos < body.length()) {
546 				var c = body.charAt(pos);
547 				if (inEscape) {
548 					inEscape = false;
549 				} else if (c == '\\') {
550 					inEscape = true;
551 				} else if (c == '{') {
552 					depth++;
553 				} else if (c == '}') {
554 					depth--;
555 					if (depth == 0) {
556 						pos++;
557 						return compile(resolver, body.substring(start, pos));
558 					}
559 				}
560 				pos++;
561 			}
562 			throw parseError("Unterminated nested marker starting at index %s", start);
563 		}
564 
565 		/**
566 		 * Parse a bare identifier / numeric / boolean token. Captured verbatim until comma,
567 		 * close-paren, or whitespace. The captured substring is wrapped in a literal-only
568 		 * {@link VarTemplate}.
569 		 */
570 		VarTemplate parseBareToken() {
571 			var start = pos;
572 			while (pos < body.length()) {
573 				var c = body.charAt(pos);
574 				if (c == ',' || c == ')' || c == ' ' || c == '\t' || c == '\n' || c == '\r')
575 					break;
576 				pos++;
577 			}
578 			if (pos == start)
579 				throw parseError("Expected argument");
580 			return compile(resolver, body.substring(start, pos));
581 		}
582 
583 		void expect(char c) {
584 			skipWs();
585 			if (pos >= body.length() || body.charAt(pos) != c)
586 				throw parseError("Expected '%s'", c);
587 			pos++;
588 		}
589 
590 		void expectEnd() {
591 			skipWs();
592 			if (pos < body.length())
593 				throw parseError("Unexpected trailing content '%s'", body.substring(pos));
594 		}
595 
596 		char peek() {
597 			skipWs();
598 			return pos < body.length() ? body.charAt(pos) : '\0';
599 		}
600 
601 		void skipWs() {
602 			while (pos < body.length()) {
603 				var c = body.charAt(pos);
604 				if (c == ' ' || c == '\t' || c == '\n' || c == '\r')
605 					pos++;
606 				else
607 					break;
608 			}
609 		}

611 		IllegalArgumentException parseError(String fmt, Object... args) {
612 			// Format the caller-supplied template first, then concatenate the position/source-fragment
613 			// preamble as plain text. Routing the entire message through MessageFormat would interpret
614 			// the literal "{<pos>}" as a placeholder. Fall back to plain concatenation if the caller's
615 			// fmt is itself an invalid MessageFormat pattern (e.g. literal-brace escaping mismatches).
616 			String formatted;
617 			try {
618 				formatted = f(fmt, args);
619 			} catch (@SuppressWarnings("unused") IllegalArgumentException e) {
620 				var sb = new StringBuilder(fmt);
621 				for (var a : args)
622 					sb.append(' ').append(a);
623 				formatted = sb.toString();
624 			}
625 			var message = "Invalid script in template: " + formatted + " at script body offset " + pos
626 				+ "; full script: \"" + sourceFragment() + "\"";
627 			return new IllegalArgumentException(message);
628 		}
629 
630 		String sourceFragment() {
631 			var safeEnd = Math.min(closeIndex + 1, fullInput.length());
632 			return fullInput.substring(x2, safeEnd);
633 		}
