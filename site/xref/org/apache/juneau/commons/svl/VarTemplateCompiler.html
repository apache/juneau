26  /**
27   * Single-pass tokenizer for SVL templates.
28   *
29   * <p>
30   * One pass over the input string produces an array of {@link TemplateSegment} objects:
31   * {@link LiteralSegment} for plain text, {@link VarRefSegment} for {@code ${...}} markers, and
32   * {@link ExprSegment} for {@code #{expr}} markers. The compiler is the only tokenizer in the codebase — both
33   * {@link VarResolverSession#resolve(String)} and {@link VarResolverSession#resolveTo(String, java.io.Writer)}
34   * dispatch through {@link VarResolver#compile(String)}.
35   *
36   * <h5 class='section'>Grammar:</h5>
37   * <ul>
38   * 	<li><b>Top-level escape:</b> {@code \\}, {@code \$}, {@code \#} emit just the trailing char.
39   * 		Other {@code \X} sequences emit {@code \X} verbatim.
40   * 	<li><b>Marker ends:</b> where a {@code ${...}} or {@code #{...}} ends, and where its top-level {@code :} separators
41   * 		are, is decided by {@link VarSplit} alone (the expression lexer uses it too).  In short: a backslash escapes the
42   * 		next char; inside {@code ${...}} a quote opens a quoted run only at the start of a part (so {@code it's} and
43   * 		{@code don't} are plain text); inside {@code #{...}} quotes are SpEL strings ({@code ''} doubling, no backslash
44   * 		escape).
45   * 	<li><b>{@code ${Type:rest}}:</b> the body runs to the matching {@code }}.
46   * 		If the text before the first top-level {@code :} (see {@link VarSplit}) is {@code [A-Za-z]+} and a var with
47   * 		that name is registered, it is a lookup of that var. Otherwise the whole body is the empty-prefix
48   * 		{@code P} ({@code PropertyVar}) shortcut, so {@code ${name}} / {@code ${name:default}} keep their meaning.
49   * 	<li><b>Colon split per var class</b> ({@link #translate(Function, String)}): the body is translated to the
50   * 		internal comma form the var classes already understand. {@link DefaultingVar}: the first top-level {@code :}
51   * 		splits key from default, the rest is the default verbatim. {@link MultipartResolvingVar} /
52   * 		{@link MultipartVar}: each top-level {@code :} separates one part. Any other var: the body is whole.
53   * 		In a key or part, {@code \:} is a literal colon and a fully quoted {@code '...'} / {@code "..."} part is
54   * 		unquoted. A key or part that contains a literal {@code ,} cannot be represented in the internal form:
55   * 		strict mode throws, lenient mode keeps the marker as literal text.
56   * 	<li><b>{@code #{expr}}:</b> the body runs to the matching {@code }} (the braces of an embedded {@code ${...}} or a
57   * 		quoted {@code '}'} do not end it) and is parsed with
58   * 		{@link ExprParser#parse(String, int)} using the resolver's {@code maxExprDepth}. Constant expressions fold
59   * 		to a literal via {@link ExprSegment#fold(VarResolver)}. A parse error (including the empty expression)
60   * 		throws in strict mode and keeps the marker as literal text in lenient mode.
61   * 	<li><b>Legacy dollar-brace variable syntax</b> (removed in 10.0): strict mode throws a positioned
62   * 		{@link VarResolverException}; lenient mode keeps the whole marker as literal text.
63   * 	<li><b>Unterminated</b> {@code ${} / {@code #{}: strict mode throws a positioned error; lenient mode appends
64   * 		the rest of the input as literal text.
65   * 	<li><b>Nesting depth:</b> {@code ${...}} bodies nest at most {@code maxExprDepth} deep.  A deeper marker throws a
66   * 		positioned error in strict mode and is kept as literal text in lenient mode.  ({@code ${...}} inside
67   * 		{@code #{...}} is resolved lazily; {@link ExprEvaluator} caps that chain with the same limit.)
68   * 	<li><b>Lenient literal text</b> (unterminated, parse error, unrepresentable key, legacy syntax, too deep) is the
69   * 		marker source with {@code \\}, {@code \$}, {@code \{} and {@code \}} escapes removed, as the old compiler did.
70   * </ul>
71   *
72   * <p>
73   * Compile errors are {@link ExprException}s whose source is the template and whose position is the offset of the
74   * offending marker (or, for an expression parse error, of the offending token).  Errors inside a nested
75   * {@code ${...}} body are mapped back to the template exactly only when translating the body to its comma form kept
76   * its length (only {@code :} became {@code ,}).  When it did not (an unquoted key, a removed prefix, a {@code \:}
77   * escape), the error points at the start of the enclosing {@code ${} marker instead.
78   *
79   * <h5 class='section'>Stable-value folding:</h5>
80   * <p>
81   * If a captured {@link Var} opts in via {@link Var#isStable()} and its body folded to a literal
82   * (no nested live markers), the var is invoked once at compile time and replaced with the
83   * resolved literal. If {@link Var#canResolve(VarResolverSession)} returns {@code false} or
84   * {@link Var#doResolve(VarResolverSession, String)} throws, folding is skipped and the dispatch
85   * segment is emitted unchanged.
86   */
87  @SuppressWarnings({
88  	"java:S3776" // Cognitive complexity acceptable for this tokenizer

112 	/**
113 	 * Compiles the given input string against the supplied resolver.
114 	 *
115 	 * @param resolver The resolver whose var registry is used for compile-time {@link Var}
116 	 *	 binding. Must not be {@code null}.
117 	 * @param input The template source. May be {@code null}.
118 	 * @return The compiled template. Never {@code null}.
119 	 * @throws ExprException In strict mode, on a malformed marker or legacy syntax.
120 	 */
121 	static VarTemplate compile(VarResolver resolver, String input) {
122 		return compile(resolver, input, 0);
123 	}

125 	private static VarTemplate compile(VarResolver resolver, String input, int depth) {
126 		if (input == null)
127 			return new VarTemplate(resolver, null, new TemplateSegment[0]);
128 		if (input.isEmpty())
129 			return new VarTemplate(resolver, input, new TemplateSegment[0]);
130 		// Fast path: no special chars at all.
131 		if (input.indexOf('$') == -1 && input.indexOf('\\') == -1 && input.indexOf('#') == -1)
132 			return new VarTemplate(resolver, input, new TemplateSegment[]{new LiteralSegment(input)});
133 		return new VarTemplateCompiler(resolver, depth).compileImpl(input);
134 	}
135 
136 	private VarTemplate compileImpl(String input) {
137 		var segments = new ArrayList<TemplateSegment>();
138 		var literal = new StringBuilder();
139 		var n = input.length();
140 		var i = 0;
141 		while (i < n) {
142 			var c = input.charAt(i);
143 			if (c == '\\') {
144 				if (i + 1 < n) {
145 					var d = input.charAt(i + 1);
146 					if (d != '\\' && d != '$' && d != '#')
147 						literal.append('\\');
148 					literal.append(d);
149 					i += 2;
150 				} else {
151 					literal.append('\\');
152 					i++;
153 				}
154 			} else if (c == '$') {
155 				i = dollar(input, i, segments, literal);
156 			} else if (c == '#' && i + 1 < n && input.charAt(i + 1) == '{') {
157 				i = hash(input, i, segments, literal);
158 			} else {
159 				literal.append(c);
160 				i++;
161 			}
162 		}
163 		flushLiteral(segments, literal);
164 		return new VarTemplate(resolver, input, segments.toArray(new TemplateSegment[0]), List.copyOf(refs));
165 	}

167 	/** Handles a {@code $} at {@code start}; returns the index to continue scanning from. */
168 	private int dollar(String input, int start, List<TemplateSegment> segments, StringBuilder literal) {
169 		var n = input.length();
170 		var j = start + 1;
171 		while (j < n && isAsciiLetter(input.charAt(j)))
172 			j++;
173 		if (j >= n || input.charAt(j) != '{') {
174 			// False trigger ("$", "$X ", "$1"): literal text, rescan from the non-letter char.
175 			literal.append(input, start, j);
176 			return j;
177 		}
178 		var close = VarSplit.closeOfBody(input, j + 1);
179 		if (j > start + 1) {
180 			// Legacy dollar-brace variables.
181 			if (resolver.strict)
182 				throw new ExprException(LEGACY_SYNTAX_MSG, input, start);
183 			var end = close < 0 ? n : close + 1;
184 			syntax(LEGACY_SYNTAX_MSG, start, end);
185 			literal.append(unescapeChars(input.substring(start, end), AS2));
186 			return end;
187 		}
188 		if (close < 0) {
189 			if (resolver.strict)
190 				throw new ExprException("Unterminated ${", input, start);
191 			syntax("Unterminated ${", start, n);
192 			literal.append("${").append(unescapeChars(input.substring(j + 1), AS2));
193 			return n;
194 		}
195 		var inner = input.substring(j + 1, close);
196 		var fallthrough = unescapeChars(input.substring(start, close + 1), AS2);
197 		if (depth >= resolver.maxExprDepth) {
198 			if (resolver.strict)
199 				throw new ExprException("Variable nested too deeply (max " + resolver.maxExprDepth + ")", input, start);
200 			syntax("Variable nested too deeply (max " + resolver.maxExprDepth + ")", start, close + 1);
201 			literal.append(fallthrough);
202 			return close + 1;
203 		}
204 		var call = translate(resolver.getVarMap()::get, inner);
205 		if (call.error() != null) {
206 			if (resolver.strict)
207 				throw new ExprException(call.error(), input, start);
208 			syntax(call.error(), start, close + 1);
209 			literal.append(fallthrough);
210 			return close + 1;
211 		}
212 		var body = compileNested(call, inner, input, j + 1, start);
213 		var hasInternalVar = call.body().indexOf('{') != -1;
214 		var folded = tryFoldStableVar(call.var(), body, hasInternalVar);
215 		if (folded != null) {
216 			// A folded var is a value, not author text: keep it a separate segment so a value encoder still sees it.
217 			flushLiteral(segments, literal);
218 			segments.add(new LiteralSegment(folded, true));
219 		} else {
220 			refs.add(new TemplateReference(TemplateReference.Kind.VAR, call.prefix().isEmpty() ? DOLLAR_BRACE_VAR : call.prefix(), call.var() != null, null, start, close + 1));
221 			addKeyRef(call, inner, start + 2);
222 			flushLiteral(segments, literal);
223 			segments.add(new VarRefSegment(call.var(), call.prefix(), body, hasInternalVar, fallthrough));
224 		}
225 		return close + 1;
226 	}

228 	/** Handles a {@code #{} at {@code start}; returns the index to continue scanning from. */
229 	private int hash(String input, int start, List<TemplateSegment> segments, StringBuilder literal) {
230 		var n = input.length();
231 		var close = VarSplit.closeOfExpr(input, start + 2);
232 		if (close < 0) {
233 			if (resolver.strict)
234 				throw new ExprException("Unterminated #{", input, start);
235 			syntax("Unterminated #{", start, n);
236 			literal.append("#{").append(unescapeChars(input.substring(start + 2), AS2));
237 			return n;
238 		}
239 		var src = input.substring(start + 2, close);
240 		Expr ast;
241 		Expr unknown;
242 		ExprSegment seg;
243 		LiteralSegment folded;
244 		try {
245 			ast = ExprParser.parse(src, resolver.maxExprDepth);
246 			unknown = findUnknownCall(ast);
247 			seg = new ExprSegment(ast, src);
248 			folded = unknown == null ? seg.fold(resolver) : null;
249 		} catch (StackOverflowError e) {
250 			if (resolver.strict)
251 				throw new ExprException("expression too complex", input, start + 2, e);
252 			syntax("expression too complex", start, close + 1);
253 			literal.append(unescapeChars(input.substring(start, close + 1), AS2));
254 			return close + 1;
255 		} catch (ExprException e) {
256 			if (resolver.strict)
257 				throw new ExprException(e.getBaseMessage(), input, start + 2 + e.getPos(), e);
258 			syntax(e.getBaseMessage(), start, close + 1);
259 			literal.append(unescapeChars(input.substring(start, close + 1), AS2));
260 			return close + 1;
261 		}
262 		if (unknown != null) {
263 			if (resolver.strict)
264 				throw new ExprException(unavailableMessage(unknown), input, start + 2 + unknown.pos());
265 			collectRefs(ast, start + 2);
266 			literal.append(unescapeChars(input.substring(start, close + 1), AS2));
267 			return close + 1;
268 		}
269 		if (folded != null) {
270 			if (folded.value) {
271 				flushLiteral(segments, literal);
272 				segments.add(folded);
273 			} else
274 				literal.append(folded.text);
275 		} else {
276 			collectRefs(ast, start + 2);
277 			flushLiteral(segments, literal);
278 			segments.add(seg);
279 		}
280 		return close + 1;
281 	}

283 	/**
284 	 * Records the lookup key of a {@code P} reference ({@code ${name}}, {@code ${name:default}}) as an identifier, since
285 	 * the shortcut reads a named value.  The default is not a reference.  A key containing {@code ${} is reported as
286 	 * written.
287 	 */
288 	private void addKeyRef(VarCall call, String inner, int innerStart) {
289 		if (! (call.var() instanceof DefaultingVar) || ! DOLLAR_BRACE_VAR.equals(call.prefix()))
290 			return;
291 		var rest = inner.substring(call.restStart());
292 		var j = VarSplit.firstTopLevel(rest, ':');
293 		var len = j < 0 ? rest.length() : j;
294 		var body = call.body();
295 		var comma = body.indexOf(',');
296 		var key = comma < 0 ? body : body.substring(0, comma);
297 		var from = innerStart + call.restStart();
298 		refs.add(new TemplateReference(TemplateReference.Kind.IDENT, key, true, null, from, from + len));
299 	}

301 	private void syntax(String message, int start, int end) {
302 		refs.add(new TemplateReference(TemplateReference.Kind.SYNTAX, message, false, null, start, end));
303 	}

305 	/** Walks an expression, recording vars, calls and identifiers.  {@code base} is the source offset of the expression text. */
306 	private void collectRefs(Expr root, int base) {
307 		walk(root, base, ROOT);
308 	}
309 
310 	/** Identifier scope: top level, a row of a named list, or a body whose target cannot be named. */
311 	private static final String ROOT = new String("<root>"), OPAQUE = new String("<opaque>");
312 
313 	private void walk(Expr e, int base, String scope) {
314 		if (e instanceof Expr.Ident x) {
315 			if (scope != OPAQUE)
316 				refs.add(new TemplateReference(TemplateReference.Kind.IDENT, x.name(), true, scope == ROOT ? null : scope, base + x.pos(), base + x.pos() + x.name().length()));
317 		} else if (e instanceof Expr.VarRef x) {
318 			var raw = x.raw();
319 			var call = translate(resolver.getVarMap()::get, raw.substring(2, raw.length() - 1));
320 			var name = call.error() != null ? raw : (call.prefix().isEmpty() ? DOLLAR_BRACE_VAR : call.prefix());
321 			refs.add(new TemplateReference(TemplateReference.Kind.VAR, name, call.error() == null && call.var() != null, null, base + x.pos(), base + x.pos() + raw.length()));
322 			if (call.error() == null)
323 				addKeyRef(call, raw.substring(2, raw.length() - 1), base + x.pos() + 2);
324 		} else if (e instanceof Expr.Call x) {
325 			var bound = ! isDenied(x.fn()) && (resolver.getFunctionMap().containsKey(x.fn()) || x.fn().startsWith("$op."));
326 			refs.add(new TemplateReference(TemplateReference.Kind.FUNCTION, x.fn(), bound, null, base + x.pos(), base + x.pos() + x.fn().length()));
327 			x.args().forEach(a -> walk(a, base, scope));
328 		} else {
329 			var op = opName(e);
330 			if (op != null && ! resolver.isAllowed(op))
331 				refs.add(new TemplateReference(TemplateReference.Kind.FUNCTION, op, false, null, base + e.pos(), base + e.pos() + 1));
332 			if (e instanceof Expr.Unary x) walk(x.e(), base, scope);
333 			else if (e instanceof Expr.Binary x) { walk(x.l(), base, scope); walk(x.r(), base, scope); }
334 			else if (e instanceof Expr.Ternary x) { walk(x.c(), base, scope); walk(x.a(), base, scope); walk(x.b(), base, scope); }
335 			else if (e instanceof Expr.Elvis x) { walk(x.l(), base, scope); walk(x.r(), base, scope); }
336 			else if (e instanceof Expr.Path x) walk(x.target(), base, scope);
337 			else if (e instanceof Expr.Index x) { walk(x.target(), base, scope); walk(x.idx(), base, scope); }
338 			else if (e instanceof Expr.Select x) { walk(x.target(), base, scope); walk(x.pred(), base, bodyScope(x.target(), scope)); }
339 			else if (e instanceof Expr.Project x) { walk(x.target(), base, scope); walk(x.map(), base, bodyScope(x.target(), scope)); }
340 		}
341 	}
342 
343 	/** The identifier scope of a select/project body: the list name when the target is a top-level identifier. */
344 	private static String bodyScope(Expr target, String scope) {
345 		return target instanceof Expr.Ident x && scope == ROOT ? x.name() : OPAQUE;
346 	}

348 	private String unavailableMessage(Expr e) {
349 		if (e instanceof Expr.Call c)
350 			return isDenied(c.fn()) ? "function '" + c.fn() + "' is not permitted" : "unknown function '" + c.fn() + "'";
351 		return "operator '" + opName(e) + "' is not permitted";
352 	}
353 
354 	private boolean isDenied(String fn) {
355 		return resolver.isDenied(fn) || (fn.startsWith("$op.") && ! resolver.isAllowed(fn));
356 	}
357 
358 	/** The reserved <c>$op.*</c> function an operator node dispatches to, or <jk>null</jk> if the node is not an operator. */
359 	private static String opName(Expr e) {
360 		String n = null;
361 		if (e instanceof Expr.Unary x) n = ExprEvaluator.unaryName(x.op());
362 		else if (e instanceof Expr.Binary x) n = ExprEvaluator.binaryName(x.op());
363 		else if (e instanceof Expr.Ternary) n = "ternary";
364 		else if (e instanceof Expr.Elvis) n = "elvis";
365 		else if (e instanceof Expr.Select) n = "select";
366 		else if (e instanceof Expr.Project) n = "project";
367 		return n == null ? null : "$op." + n;
368 	}

370 	/**
371 	 * Finds the first node (in source order) that cannot be evaluated: an {@link Expr.Call} whose function is not
372 	 * registered or is denied by the allowlist, or an operator whose <c>$op.*</c> function is denied by the allowlist.
373 	 *
374 	 * <p>
375 	 * An operator whose function is merely unregistered (and not denied) is not reported here; it fails at evaluation.
376 	 * Because the whole AST is walked, an unavailable node on a branch that would never be taken (e.g. the unused
377 	 * side of a ternary) is reported at compile time, not only when that branch is evaluated.
378 	 *
379 	 * <p>
380 	 * A {@code ${...}} reference is opaque in the AST, so a function call inside a var body nested in the expression
381 	 * is not checked here; it is checked lazily, when the reference is evaluated.
382 	 *
383 	 * @param e The expression.
384 	 * @return The first unavailable node, or <jk>null</jk> if every call and operator resolves.
385 	 */
386 	private Expr findUnknownCall(Expr root) {
387 		// Iterative pre-order walk (explicit stack) so deep left-associative chains cannot overflow the call stack.
388 		var stack = new ArrayDeque<Expr>();
389 		stack.push(root);
390 		while (! stack.isEmpty()) {
391 			var e = stack.pop();
392 			var op = opName(e);
393 			if (op != null && ! resolver.isAllowed(op))
394 				return e;
395 			if (e instanceof Expr.Call c) {
396 				if (isDenied(c.fn()) || (resolver.findFunction(c.fn()) == null && ! c.fn().startsWith("$op.")))
397 					return c;
398 				var args = c.args();
399 				for (var i = args.size() - 1; i >= 0; i--)
400 					stack.push(args.get(i));
401 			} else if (e instanceof Expr.Unary x) {
402 				stack.push(x.e());
403 			} else if (e instanceof Expr.Binary x) {
404 				stack.push(x.r()); stack.push(x.l());
405 			} else if (e instanceof Expr.Ternary x) {
406 				stack.push(x.b()); stack.push(x.a()); stack.push(x.c());
407 			} else if (e instanceof Expr.Elvis x) {
408 				stack.push(x.r()); stack.push(x.l());
409 			} else if (e instanceof Expr.Path x) {
410 				stack.push(x.target());
411 			} else if (e instanceof Expr.Index x) {
412 				stack.push(x.idx()); stack.push(x.target());
413 			} else if (e instanceof Expr.Select x) {
414 				stack.push(x.pred()); stack.push(x.target());
415 			} else if (e instanceof Expr.Project x) {
416 				stack.push(x.map()); stack.push(x.target());
417 			}
418 			// Literal, VarRef, Ident: leaf
419 		}
420 		return null;
421 	}

423 	/**
424 	 * Compiles a var body, re-positioning strict-mode errors against the outer template.
425 	 *
426 	 * <p>
427 	 * Positions map exactly when the translation kept the length of the text after the prefix (only {@code :} to
428 	 * {@code ,} rewrites); otherwise the error is positioned at the outer marker.
429 	 */
430 	private VarTemplate compileNested(VarCall call, String inner, String input, int bodyStart, int markerStart) {
431 		var body = call.body();
432 		try {
433 			return compile(resolver, body, depth + 1);
434 		} catch (ExprException e) {
435 			var exact = body.length() == inner.length() - call.restStart();
436 			var pos = exact ? bodyStart + call.restStart() + e.getPos() : markerStart;
437 			throw new ExprException(e.getBaseMessage(), input, pos, e);
438 		}
439 	}

441 	private static boolean isAsciiLetter(char c) {
442 		return (c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z');
443 	}

445 	private static void flushLiteral(List<TemplateSegment> out, StringBuilder buf) {
446 		if (!buf.isEmpty()) {
447 			out.add(new LiteralSegment(buf.toString()));
448 			buf.setLength(0);
449 		}
450 	}

456 	/**
457 	 * A translated {@code ${...}} reference.
458 	 *
459 	 * @param prefix The var name ({@code "P"} for the empty-prefix shortcut).
460 	 * @param var The registered var, or <jk>null</jk> if none (only possible when {@code P} is not registered).
461 	 * @param body The body in the var's internal comma form, or <jk>null</jk> if {@link #error()} is set.
462 	 * @param error Why the body cannot be represented, or <jk>null</jk>.
463 	 * @param restStart Offset in the reference body where the text after the prefix starts (<c>0</c> if no prefix
464 	 * 	was stripped).
465 	 */
466 	record VarCall(String prefix, Var var, String body, String error, int restStart) {}
467 
468 	/**
469 	 * Translates the text between {@code ${} and {@code }} into a var and its internal comma-form body.
470 	 *
471 	 * <p>
472 	 * Used by the template compiler and by the expression evaluator for {@code ${...}} inside {@code #{...}}, so both
473 	 * read a reference the same way (see the class Javadoc for the rules).
474 	 *
475 	 * @param lookup Finds a registered var by name.
476 	 * @param inner The reference body.
477 	 * @return The translation. Never <jk>null</jk>.
478 	 */
479 	static VarCall translate(Function<String,Var> lookup, String inner) {
480 		var i = VarSplit.firstTopLevel(inner, ':');
481 		String prefix = null;
482 		Var v = null;
483 		var rest = inner;
484 		var restStart = 0;
485 		if (i > 0 && isAsciiLetters(inner, i)) {
486 			v = lookup.apply(inner.substring(0, i));
487 			if (v != null) {
488 				prefix = inner.substring(0, i);
489 				rest = inner.substring(i + 1);
490 				restStart = i + 1;
491 			}
492 		}
493 		if (v == null) {
494 			prefix = DOLLAR_BRACE_VAR;
495 			v = lookup.apply(DOLLAR_BRACE_VAR);
496 		}
497 		if (v instanceof MultipartResolvingVar || v instanceof MultipartVar) {
498 			var sb = new StringBuilder();
499 			for (var part : VarSplit.splitTopLevel(rest, ':')) {
500 				if (hasTopLevelComma(part) && ! isQuoted(part))
501 					return commaError(prefix, part, restStart);
502 				var p = keyPart(part);
503 				if (! sb.isEmpty())
504 					sb.append(',');
505 				sb.append(p);
506 			}
507 			return new VarCall(prefix, v, sb.toString(), null, restStart);
508 		}
509 		if (v == null || v instanceof DefaultingVar) {
510 			var j = VarSplit.firstTopLevel(rest, ':');
511 			var rawKey = j < 0 ? rest : rest.substring(0, j);
512 			if (hasTopLevelComma(rawKey) && ! isQuoted(rawKey))
513 				return commaError(prefix, rawKey, restStart);
514 			var key = keyPart(rawKey);
515 			return new VarCall(prefix, v, j < 0 ? key : key + ',' + unescape(rest.substring(j + 1)), null, restStart);
516 		}
517 		return new VarCall(prefix, v, VarSplit.restoreCommas(unescape(rest)), null, restStart);
518 	}
519 
520 	private static VarCall commaError(String prefix, String key, int restStart) {
521 		return new VarCall(prefix, null, null, "Variable '" + prefix + "' key '" + key + "' contains ',' which is not supported in a key", restStart);
522 	}
523 
524 	private static boolean isQuoted(String s) {
525 		var n = s.length();
526 		return n >= 2 && (s.charAt(0) == '\'' || s.charAt(0) == '"') && s.charAt(n - 1) == s.charAt(0);
527 	}
528 
529 	private static boolean isAsciiLetters(String s, int end) {
530 		for (var i = 0; i < end; i++)
531 			if (! isAsciiLetter(s.charAt(i)))
532 				return false;
533 		return true;
534 	}
535 
536 	/** Unquotes a fully quoted key part, else removes the grammar escapes (see {@link #unescape(String)}). */
537 	private static String keyPart(String s) {
538 		var n = s.length();
539 		if (n >= 2) {
540 			var q = s.charAt(0);
541 			if ((q == '\'' || q == '"') && s.charAt(n - 1) == q)
542 				return s.substring(1, n - 1).replace(',', VarSplit.COMMA);
543 		}
544 		return unescape(s);
545 	}
546 
547 	/**
548 	 * Removes the escape backslash before {@code :}, {@code '}, {@code "}, {@code ,} and {@code }} outside nested braces.
549 	 *
550 	 * <p>
551 	 * An escaped comma becomes {@link VarSplit#COMMA}, an internal-only character the var classes turn back into
552 	 * {@code ,} after they split the internal comma form, so a literal comma is never re-split.  an escaped backslash becomes {@link VarSplit#BACKSLASH} (restored the same way); every
553 	 * other escape is kept for the nested template compile.
554 	 */
555 	private static String unescape(String s) {
556 		if (s.indexOf('\\') == -1)
557 			return s;
558 		var sb = new StringBuilder(s.length());
559 		var depth = 0;
560 		for (var i = 0; i < s.length(); i++) {
561 			var c = s.charAt(i);
562 			if (c == '\\' && i + 1 < s.length()) {
563 				var d = s.charAt(++i);
564 				if (depth == 0 && (d == ':' || d == '\'' || d == '"' || d == '}')) {
565 					sb.append(d);
566 				} else if (depth == 0 && d == ',') {
567 					sb.append(VarSplit.COMMA);
568 				} else if (depth == 0 && d == '\\') {
569 					sb.append(VarSplit.BACKSLASH);
570 				} else {
571 					sb.append(c).append(d);
572 				}
573 			} else {
574 				if (c == '{')
575 					depth++;
576 				else if (c == '}' && depth > 0)
577 					depth--;
578 				sb.append(c);
579 			}
580 		}
581 		return sb.toString();
582 	}
583 
584 	/** Whether {@code s} has an unescaped {@code ,} outside nested braces. */
585 	private static boolean hasTopLevelComma(String s) {
586 		var depth = 0;
587 		for (var i = 0; i < s.length(); i++) {
588 			var c = s.charAt(i);
589 			if (c == '\\')
590 				i++;
591 			else if (c == '{')
592 				depth++;
593 			else if (c == '}' && depth > 0)
594 				depth--;
595 			else if (c == ',' && depth == 0)
596 				return true;
597 		}
598 		return false;
599 	}

601 	/**
602 	 * Returns whether a var reference is eligible for compile-time folding: the var opted in via {@link Var#isStable()},
603 	 * is a {@link SimpleVar}, and its body has no nested live markers.
604 	 *
605 	 * @param v The var (may be <jk>null</jk>).
606 	 * @param hasInternalVar Whether the body contains nested live markers.
607 	 * @return <jk>true</jk> if stable.
608 	 */
609 	static boolean isStable(Var v, boolean hasInternalVar) {
610 		return v != null && v.isStable() && ! hasInternalVar && v instanceof SimpleVar;
611 	}

613 	/**
614 	 * Attempt compile-time stable-value folding for a {@code ${...}} dispatch.
615 	 *
616 	 * <p>
617 	 * Folding is only valid when:
618 	 * <ul>
619 	 * 	<li>{@code v} is non-{@code null}.
620 	 * 	<li>{@code v.isStable()} returns {@code true}.
621 	 * 	<li>The compiled body is a pure literal — no nested live markers, including the legacy
622 	 * 		{@code hasInternalVar} flag set by the state machine on inner {@code {}} characters.
623 	 * 	<li>The var is a {@link SimpleVar} (streamed vars are excluded — folding a stream loses
624 	 * 		the writer-direct optimization for no compile-time gain).
625 	 * 	<li>{@link Var#canResolve(VarResolverSession)} on a fresh no-bean session returns
626 	 * 		{@code true}, and {@link Var#doResolve(VarResolverSession, String)} returns a
627 	 * 		non-{@code null} value without throwing.
628 	 * </ul>
629 	 *
630 	 * <p>
631 	 * Returns the folded literal text, or {@code null} if folding is skipped for any reason.
632 	 */
633 	private String tryFoldStableVar(Var v, VarTemplate body, boolean hasInternalVar) {
634 		if (! isStable(v, hasInternalVar) || ! body.isLiteral())
635 			return null;
636 		// body.isLiteral() guaranteed above; resolve(session) short-circuits to literalText
637 		// and never touches session beans.
638 		return foldStableVarText(resolver, v, body::resolve);
639 	}

641 	/**
642 	 * Shared fold guard: resolves a stable var on a fresh no-bean session.
643 	 *
644 	 * <p>
645 	 * Returns {@code null} (do not fold) if {@link Var#canResolve(VarResolverSession)} is false, the resolved value is
646 	 * {@code null}, it would need recursive resolution ({@code $} present and {@link Var#allowRecurse()}), or anything throws.
647 	 *
648 	 * @param resolver The resolver.
649 	 * @param v The var (already known to be {@link #isStable(Var, boolean) stable}).
650 	 * @param body Produces the var body text from the fresh session.
651 	 * @return The resolved text, or {@code null} if folding must be skipped.
652 	 */
653 	@SuppressWarnings({
654 		"java:S1166" // Exception swallowed intentionally — fold-or-skip contract.
655 	})
656 	static String foldStableVarText(VarResolver resolver, Var v, Function<VarResolverSession,String> body) {
657 		try {
658 			var session = resolver.createSession();
659 			if (!v.canResolve(session))
660 				return null;
661 			var resolved = v.doResolve(session, body.apply(session));
662 			if (resolved == null)
663 				return null;
664 			// Skip folding if the resolved value would itself need recursive resolution. The
665 			// runtime VarRefSegment.resolve(...) path post-resolves dispatch results that
666 			// contain ${...} / ${X:...} markers when v.allowRecurse() returns true; folding
667 			// would silently drop that recursion. (Compare: VarRefSegment.resolve, which
668 			// re-invokes session.resolve(replacement) when the replacement contains '$' and
669 			// the var allows recursion.)
670 			if (v.allowRecurse() && resolved.indexOf('$') != -1)
671 				return null;
672 			return resolved;
673 		} catch (@SuppressWarnings("unused") Exception e) {
674 			return null;
675 		}
676 	}
