17  package org.apache.juneau.commons.svl;
18  
19  import java.util.function.*;
20  
21  /**
22   * A {@link TemplateSegment} that evaluates a parsed {@code #{expression}}.
23   *
24   * <p>
25   * The AST is parsed once at compile time (the caller's job, via {@link ExprParser#parse(String, int)}).  Each render
26   * builds a <em>fresh</em> {@link ExprContext} so the collection budget (which lives on the root context) is
27   * independent per render.
28   *
29   * <p>
30   * <b>Root scope:</b> identifiers resolve against the session's render scope
31   * ({@link VarResolverSession#scope(Function)}); when none is set every {@link Expr.Ident} other than {@code this}
32   * yields <jk>null</jk>.  Values can also be reached through {@code ${...}} var references.
33   *
34   * <p>
35   * <b>Output is never re-parsed.</b>  The result is stringified and appended; it does not go through
36   * {@code resolveRecursive} and {@code allowRecurse} does not apply.  Stringification is
37   * {@link VarResolverSession#formatValue(Object)} (the value formatter).
38   *
39   * <p>
40   * <b>Fall-through:</b> in lenient mode an unknown/denied function (an {@link ExprEvaluator.FallThrough}) emits the
41   * verbatim {@code #{source}} text; in strict mode (and for every other failure) an {@link ExprException} propagates.
42   */
43  final class ExprSegment extends TemplateSegment {
44  
45  	/** Render scope used for identifier lookups when the session sets none: always absent. */
46  	static final Function<String,Object> EMPTY_SCOPE = n -> null;
47  
48  	private final Expr ast;
49  	private final String source;
50  
51  	/**
52  	 * Constructor.
53  	 *
54  	 * @param ast The parsed expression.
55  	 * @param source The expression text between {@code #{} and {@code }} (used for diagnostics and fall-through).
56  	 */
57  	ExprSegment(Expr ast, String source) {
58  		this.ast = ast;
59  		this.source = source;
60  	}
61  
62  	@Override
63  	void resolve(VarResolverSession session, StringBuilder out) {
64  		out.append(render(session));
65  	}
66  
67  	/**
68  	 * Resolves with a value encoder.
69  	 *
70  	 * <p>
71  	 * The encoder runs on the formatted result.  A {@link Structure} result is not encoded here: it is formatted by the
72  	 * session formatter, which owns the encoding of its own items.  Fall-through text is author text and stays unencoded.
73  	 */
74  	@Override
75  	void resolve(VarResolverSession session, StringBuilder out, UnaryOperator<String> encoder) {
76  		try {
77  			var v = evaluate(session);
78  			var text = format(session, v);
79  			out.append(v instanceof Structure ? text : encoder.apply(text));
80  		} catch (@SuppressWarnings("unused") ExprEvaluator.FallThrough e) {
81  			out.append("#{").append(source).append('}');
82  		}
83  	}
84  
85  	private String render(VarResolverSession session) {
86  		try {
87  			return format(session, evaluate(session));
88  		} catch (@SuppressWarnings("unused") ExprEvaluator.FallThrough e) {
89  			return "#{" + source + "}";
90  		}
91  	}
92  
93  	/** Evaluates with a fresh context; may throw {@link ExprEvaluator.FallThrough} or {@link ExprException}. */
94  	private Object evaluate(VarResolverSession session) {
95  		return ExprEvaluator.eval(ast, new ExprContext(session, session.scope(), null, source));
96  	}
97  
98  	/**
99  	 * Attempts compile-time folding.
100 	 *
101 	 * <p>
102 	 * An expression is stable when it contains only literals, operators, paths/indexes, and {@code ${...}} references
103 	 * to stable vars (see {@link VarTemplateCompiler#isStable(Var, boolean)}).  Identifiers, plain function calls,
104 	 * select/project are never folded.  Evaluation failures or fall-through skip folding.
105 	 *
106 	 * @param resolver The resolver (for var lookup and a throwaway session).
107 	 * @return The folded literal, or <jk>null</jk> if the expression is not stable or fails to evaluate.
108 	 */
109 	@SuppressWarnings({
110 		"java:S1166" // Exception swallowed intentionally — fold-or-skip contract.
111 	})
112 	LiteralSegment fold(VarResolver resolver) {
113 		if (! stable(ast, resolver))
114 			return null;
115 		try {
116 			// Fall-through and errors at fold time must never be baked in: skip folding so they resurface at render.
117 			return foldText(resolver);
118 		} catch (@SuppressWarnings("unused") RuntimeException e) {
119 			return null;
120 		}
121 	}
122 
123 	/** Wraps a formatter failure (including strict-mode limit errors) in an {@link ExprException} positioned at the expression. */
124 	private String format(VarResolverSession session, Object value) {
125 		try {
126 			return session.formatValue(value);
127 		} catch (ExprException e) {
128 			throw e;
129 		} catch (RuntimeException e) {
130 			throw new ExprException("Value formatter failed: " + e.getMessage(), source, ast.pos(), e);
131 		}
132 	}
133 
134 	/**
135 	 * Folds only formatter-invariant scalars (String, Number, Boolean); anything else (collections, Structure, beans)
136 	 * is formatted at render time so a per-render session formatter applies.  Note that a session formatter
137 	 * that rewrites scalars does not affect an already-folded stable expression.
138 	 */
139 	private LiteralSegment foldText(VarResolver resolver) {
140 		var s = resolver.createSession();
141 		var v = evaluate(s);
142 		if (v != null && ! (v instanceof CharSequence || v instanceof Number || v instanceof Boolean))
143 			return null;
144 		// An expression that reads a var carries data, so its folded text is a value; a pure-literal fold is author text.
145 		return new LiteralSegment(format(s, v), hasVarRef(ast));
146 	}
147 
148 	private static boolean hasVarRef(Expr e) {
149 		if (e instanceof Expr.VarRef) return true;
150 		if (e instanceof Expr.Unary x) return hasVarRef(x.e());
151 		if (e instanceof Expr.Binary x) return hasVarRef(x.l()) || hasVarRef(x.r());
152 		if (e instanceof Expr.Ternary x) return hasVarRef(x.c()) || hasVarRef(x.a()) || hasVarRef(x.b());
153 		if (e instanceof Expr.Elvis x) return hasVarRef(x.l()) || hasVarRef(x.r());
154 		if (e instanceof Expr.Path x) return hasVarRef(x.target());
155 		if (e instanceof Expr.Index x) return hasVarRef(x.target()) || hasVarRef(x.idx());
156 		return false;
157 	}
158 
159 	private static boolean stable(Expr e, VarResolver r) {
160 		if (e instanceof Expr.Literal) return true;
161 		if (e instanceof Expr.Unary x) return stable(x.e(), r);
162 		if (e instanceof Expr.Binary x) return stable(x.l(), r) && stable(x.r(), r);
163 		if (e instanceof Expr.Ternary x) return stable(x.c(), r) && stable(x.a(), r) && stable(x.b(), r);
164 		if (e instanceof Expr.Elvis x) return stable(x.l(), r) && stable(x.r(), r);
165 		if (e instanceof Expr.Path x) return stable(x.target(), r);
166 		if (e instanceof Expr.Index x) return stable(x.target(), r) && stable(x.idx(), r);
167 		if (e instanceof Expr.VarRef x) return stableVarRef(x, r);
168 		return false;  // Ident, Call, Select, Project
169 	}
170 
171 	private static boolean stableVarRef(Expr.VarRef x, VarResolver r) {
172 		var raw = x.raw();
173 		var call = VarTemplateCompiler.translate(r.getVarMap()::get, raw.substring(2, raw.length() - 1));
174 		if (call.error() != null)
175 			return false;
176 		var v = call.varDef();
177 		var body = call.body();
178 		if (! VarTemplateCompiler.isStable(v, body.indexOf('$') != -1))
179 			return false;
180 		// Same guards as the template compiler: canResolve on a bean-less session, non-null value.
181 		return VarTemplateCompiler.foldStableVarText(r, v, s -> body) != null;
182 	}
183 }
