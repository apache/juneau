21  /**
22   * Parser for SVL <c>#{...}</c> expressions.
23   *
24   * <p>
25   * Holds the lexer ({@link #lex(String)}) and the Pratt parser ({@link #parse(String, int)}). Keywords (<c>and or not matches true false null</c>) get their own
26   * token types; the symbol forms <c>&amp;&amp; || !</c> produce the same types as their word forms (text keeps the original spelling).
27   * Integer literals that do not fit a <c>long</c> (e.g. <c>9223372036854775808</c>) become Doubles.
28   * Because keywords are reserved, they cannot be used as bare identifiers.
29   */
30  class ExprParser {
31  
32  	/** Token kinds. */
33  	enum TokenType {
34  		/** String literal; value is the decoded string. */ STRING,
35  		/** Numeric literal; value is a Long or Double. */ NUMBER,
36  		/** A <c>${...}</c> reference; text is the raw token. */ VARREF,
37  		/** A bare identifier. */ IDENT,
38  		/** <c>.</c> */ DOT,
39  		/** <c>?.</c> */ SAFE_DOT,
40  		/** <c>.?[</c> */ SELECT,
41  		/** <c>.![</c> */ PROJECT,
42  		/** <c>?:</c> */ ELVIS,
43  		/** <c>?</c> */ QUESTION,
44  		/** <c>:</c> */ COLON,
45  		/** <c>and</c> or <c>&amp;&amp;</c> */ AND,
46  		/** <c>or</c> or <c>||</c> */ OR,
47  		/** <c>not</c> or <c>!</c> */ NOT,
48  		/** <c>matches</c> */ MATCHES,
49  		/** <c>true</c> */ TRUE,
50  		/** <c>false</c> */ FALSE,
51  		/** <c>null</c> */ NULL,
52  		/** <c>==</c> */ EQ,
53  		/** <c>!=</c> */ NE,
54  		/** <c>&lt;</c> */ LT,
55  		/** <c>&lt;=</c> */ LE,
56  		/** <c>&gt;</c> */ GT,
57  		/** <c>&gt;=</c> */ GE,
58  		/** <c>+</c> */ PLUS,
59  		/** <c>-</c> */ MINUS,
60  		/** <c>*</c> */ STAR,
61  		/** <c>/</c> */ SLASH,
62  		/** <c>%</c> */ PERCENT,
63  		/** <c>(</c> */ LPAREN,
64  		/** <c>)</c> */ RPAREN,
65  		/** <c>[</c> */ LBRACKET,
66  		/** <c>]</c> */ RBRACKET,
67  		/** <c>,</c> */ COMMA,
68  		/** End of input. */ END
69  	}

71  	/**
72  	 * A lexed token.
73  	 *
74  	 * @param type Token kind.
75  	 * @param text Source text of the token.
76  	 * @param pos Zero-based offset of the token start.
77  	 * @param value Decoded value for STRING and NUMBER tokens; otherwise <jk>null</jk>.
78  	 */
79  	record Token(TokenType type, String text, int pos, Object value) {}
80  
81  	/** Maximum infix/postfix operators in one expression (bounds left-associative chain depth deterministically). */
82  	static final int MAX_OPERATORS = 1000;
83  
84  	private static final Map<String,TokenType> KEYWORDS = Map.of(
85  		"and", TokenType.AND, "or", TokenType.OR, "not", TokenType.NOT, "matches", TokenType.MATCHES,
86  		"true", TokenType.TRUE, "false", TokenType.FALSE, "null", TokenType.NULL);
87  
88  	/**
89  	 * Lexes an expression body into tokens, ending with an END token.
90  	 *
91  	 * @param src The expression text (without the surrounding <c>#{}</c>).
92  	 * @return The tokens.
93  	 * @throws ExprException On an unterminated string or <c>${</c>, or an unexpected character.
94  	 */
95  	static List<Token> lex(String src) {
96  		var out = new ArrayList<Token>();
97  		int n = src.length(), i = 0;
98  		while (i < n) {
99  			char c = src.charAt(i);
100 			if (Character.isWhitespace(c)) { i++; continue; }
101 			int start = i;
102 			if (c == '\'' || c == '"') {
103 				var sb = new StringBuilder();
104 				i++;
105 				for (;;) {
106 					if (i >= n)
107 						throw new ExprException("Unterminated string", src, start);
108 					char d = src.charAt(i);
109 					if (d == c) {
110 						if (i + 1 < n && src.charAt(i + 1) == c) { sb.append(c); i += 2; continue; }
111 						i++;
112 						break;
113 					}
114 					sb.append(d);
115 					i++;
116 				}
117 				out.add(new Token(TokenType.STRING, src.substring(start, i), start, sb.toString()));
118 			} else if (c == '$') {
119 				if (i + 1 >= n || src.charAt(i + 1) != '{')
120 					throw new ExprException("Unexpected character '$'", src, i);
121 				i = scanVarRef(src, i);
122 				out.add(new Token(TokenType.VARREF, src.substring(start, i), start, null));
123 			} else if (isDigit(c) || (c == '.' && i + 1 < n && isDigit(src.charAt(i + 1)))) {
124 				i = scanNumber(src, i, out);
125 			} else if (Character.isJavaIdentifierStart(c)) {
126 				while (i < n && Character.isJavaIdentifierPart(src.charAt(i))) i++;
127 				var w = src.substring(start, i);
128 				out.add(new Token(KEYWORDS.getOrDefault(w, TokenType.IDENT), w, start, null));
129 			} else {
130 				var two = i + 1 < n ? src.substring(i, i + 2) : "";
131 				var three = i + 2 < n ? src.substring(i, i + 3) : "";
132 				TokenType t = null;
133 				int len = 0;
134 				if (three.equals(".?[")) { t = TokenType.SELECT; len = 3; }
135 				else if (three.equals(".![")) { t = TokenType.PROJECT; len = 3; }
136 				else {
137 					len = 2;
138 					switch (two) {
139 						case "&&": t = TokenType.AND; break;
140 						case "||": t = TokenType.OR; break;
141 						case "?:": t = TokenType.ELVIS; break;
142 						case "?.": t = TokenType.SAFE_DOT; break;
143 						case "==": t = TokenType.EQ; break;
144 						case "!=": t = TokenType.NE; break;
145 						case "<=": t = TokenType.LE; break;
146 						case ">=": t = TokenType.GE; break;
147 						default: len = 1;
148 					}
149 					if (t == null)
150 						t = switch (c) {
151 							case '.' -> TokenType.DOT;
152 							case '?' -> TokenType.QUESTION;
153 							case ':' -> TokenType.COLON;
154 							case '!' -> TokenType.NOT;
155 							case '<' -> TokenType.LT;
156 							case '>' -> TokenType.GT;
157 							case '+' -> TokenType.PLUS;
158 							case '-' -> TokenType.MINUS;
159 							case '*' -> TokenType.STAR;
160 							case '/' -> TokenType.SLASH;
161 							case '%' -> TokenType.PERCENT;
162 							case '(' -> TokenType.LPAREN;
163 							case ')' -> TokenType.RPAREN;
164 							case '[' -> TokenType.LBRACKET;
165 							case ']' -> TokenType.RBRACKET;
166 							case ',' -> TokenType.COMMA;
167 							default -> null;
168 						};
169 				}
170 				if (t == null)
171 					throw new ExprException("Unexpected character '" + c + "'", src, i);
172 				i += len;
173 				out.add(new Token(t, src.substring(start, i), start, null));
174 			}
175 		}
176 		out.add(new Token(TokenType.END, "", n, null));
177 		return out;
178 	}
179 
180 	private static boolean isDigit(char c) {
181 		return c >= '0' && c <= '9';
182 	}
183 
184 	/**
185 	 * Returns the index just past the <c>${...}</c> starting at {@code start}.
186 	 *
187 	 * <p>
188 	 * Uses the shared {@link VarSplit#closeOfBody(String, int)} rules, so the lexer, the template compiler and the colon
189 	 * split agree on where a reference ends.
190 	 */
191 	private static int scanVarRef(String src, int start) {
192 		var close = VarSplit.closeOfBody(src, start + 2);
193 		if (close < 0)
194 			throw new ExprException("Unterminated ${", src, start);
195 		return close + 1;
196 	}

198 	/**
199 	 * Parses an expression body into an {@link Expr} tree.
200 	 *
201 	 * <h5 class='section'>Precedence</h5>
202 	 * <p>
203 	 * Binding powers, low to high: ternary <c>? :</c> 1 (right-assoc), Elvis <c>?:</c> 2 (right-assoc), <c>or ||</c> 3,
204 	 * <c>and &amp;&amp;</c> 4, <c>== != &lt; &lt;= &gt; &gt;= matches</c> 5, <c>+ -</c> 6, <c>* / %</c> 7, prefix
205 	 * <c>not ! - +</c> 8, postfix <c>. ?. [] ?.[] .?[] .![]</c> 9. Binary operators are left-associative, except that
206 	 * <c>matches</c> is non-associative: <c>a matches b matches c</c> is an error (parenthesise to chain).
207 	 * Symbol spellings are normalised: <c>&amp;&amp;</c> becomes <c>"and"</c>, <c>||</c> <c>"or"</c>, <c>!</c> <c>"not"</c>.
208 	 *
209 	 * <h5 class='section'>Keywords</h5>
210 	 * <ul>
211 	 * 	<li>A word-form <c>not</c>, <c>and</c> or <c>or</c> in primary position followed by <c>(</c> (with or without
212 	 * 		whitespace between) is a function {@link Expr.Call}, so <c>not(x)</c>, <c>not (x)</c> and <c>and(a, b)</c> call the
213 	 * 		functions of those names. In infix position (<c>a and (b)</c>) the word stays an operator. The symbol forms are
214 	 * 		never calls: <c>!(x)</c> is the unary operator.
215 	 * 	<li>Any keyword after <c>.</c> or <c>?.</c> is a property name (<c>obj.not</c>, <c>a?.matches</c>, <c>a.true</c>).
216 	 * 	<li><c>true false null</c> are literals in primary position; other keywords there are errors.
217 	 * 	<li>A bare identifier followed by <c>(</c> is a call; otherwise it is an {@link Expr.Ident} (scope lookup).
218 	 * </ul>
219 	 *
220 	 * <h5 class='section'>Positions</h5>
221 	 * <p>
222 	 * Every node's {@code pos} is the offset of the token that identifies it: the operator for {@link Expr.Binary},
223 	 * {@link Expr.Unary}, {@link Expr.Ternary} (the <c>?</c>) and {@link Expr.Elvis}; the <c>.</c> / <c>?.</c> for
224 	 * {@link Expr.Path}; the <c>[</c> (or the <c>?.</c> of <c>?.[</c>) for {@link Expr.Index}; the <c>.?[</c> / <c>.![</c>
225 	 * for {@link Expr.Select} / {@link Expr.Project}; the name for {@link Expr.Call} and {@link Expr.Ident}; and the token
226 	 * itself for literals and <c>${...}</c> references. Parentheses create no node.
227 	 *
228 	 * <h5 class='section'>Depth</h5>
229 	 * <p>
230 	 * Each nested sub-expression (parenthesised group, operand of a prefix operator, right operand, ternary branch,
231 	 * function argument, index or selection/projection body) is one level deeper than its parent; the whole expression is
232 	 * level 0. Left-associative chains and postfix chains do not nest. Exceeding {@code maxDepth} throws
233 	 * <c>expression nested too deeply</c> at the token where the too-deep sub-expression starts.
234 	 * {@code maxDepth} is not itself range-checked: callers must pass the small configured cap (default 32), since the
235 	 * parser recurses once per level.
236 	 *
237 	 * <h5 class='section'>Negative literals</h5>
238 	 * <p>
239 	 * There are no negative literals: <c>-1</c> parses as <c>Unary("-", Literal(1))</c> and is never folded. As a result
240 	 * <c>-9223372036854775808</c> is the negation of <c>9223372036854775808</c>, which does not fit a <c>long</c> and so
241 	 * lexes as a Double operand. That value (and any integer magnitude above <c>Long.MAX_VALUE</c>) loses integer
242 	 * precision.
243 	 *
244 	 * @param src The expression text (without the surrounding <c>#{}</c>).
245 	 * @param maxDepth The maximum nesting depth.
246 	 * @return The parsed tree.
247 	 * @throws ExprException On any lex or parse error, positioned at the offending token (the end-of-input offset for
248 	 * 	premature end).
249 	 */
250 	static Expr parse(String src, int maxDepth) {
251 		return parse(src, maxDepth, MAX_OPERATORS);
252 	}
253 
254 	/**
255 	 * Parses with an explicit operator cap (tests use a huge cap to exercise the evaluator's stack-overflow guard).
256 	 *
257 	 * @param src The expression source.
258 	 * @param maxDepth The maximum nesting depth.
259 	 * @param maxOperators The maximum infix/postfix operators.
260 	 * @return The tree.
261 	 */
262 	static Expr parse(String src, int maxDepth, int maxOperators) {
263 		return new Parser(src, lex(src), maxDepth, maxOperators).parseAll();
264 	}

266 	/** Pratt parser state over one token list. */
267 	private static final class Parser {
268 		private static final int BP_TERNARY = 1, BP_ELVIS = 2, BP_OR = 3, BP_AND = 4, BP_CMP = 5, BP_ADD = 6, BP_MUL = 7, BP_PREFIX = 8;
269 
270 		private final String src;
271 		private final List<Token> toks;
272 		private final int maxDepth;
273 		private final int maxOps;
274 		private int i, depth = -1, ops;
275 
276 		Parser(String src, List<Token> toks, int maxDepth, int maxOps) {
277 			this.maxOps = maxOps;
278 			this.src = src;
279 			this.toks = toks;
280 			this.maxDepth = maxDepth;
281 		}
282 
283 		Expr parseAll() {
284 			if (peek().type() == TokenType.END)
285 				throw new ExprException("Empty expression", src, peek().pos());
286 			var e = parseExpr(0);
287 			if (peek().type() != TokenType.END)
288 				throw unexpected(peek());
289 			return e;
290 		}
291 
292 		private Token peek() {
293 			return toks.get(i);
294 		}
295 
296 		private Token peek(int ahead) {
297 			return toks.get(Math.min(i + ahead, toks.size() - 1));
298 		}
299 
300 		private Token next() {
301 			var t = toks.get(i);
302 			if (t.type() != TokenType.END)
303 				i++;
304 			return t;
305 		}

307 		private Token expect(TokenType type, String what) {
308 			var t = peek();
309 			if (t.type() != type)
310 				throw new ExprException("Expected '" + what + "' but found " + describe(t), src, t.pos());
311 			return next();
312 		}
313 
314 		private ExprException unexpected(Token t) {
315 			return new ExprException("Unexpected " + describe(t), src, t.pos());
316 		}
317 
318 		private static String describe(Token t) {
319 			return t.type() == TokenType.END ? "end of expression" : "'" + t.text() + "'";
320 		}
321 
322 		private static boolean isWord(Token t) {
323 			return ! t.text().isEmpty() && Character.isJavaIdentifierStart(t.text().charAt(0));
324 		}
325 
326 		private static int infixBp(TokenType t) {
327 			return switch (t) {
328 				case QUESTION -> BP_TERNARY;
329 				case ELVIS -> BP_ELVIS;
330 				case OR -> BP_OR;
331 				case AND -> BP_AND;
332 				case EQ, NE, LT, LE, GT, GE, MATCHES -> BP_CMP;
333 				case PLUS, MINUS -> BP_ADD;
334 				case STAR, SLASH, PERCENT -> BP_MUL;
335 				default -> -1;
336 			};
337 		}
338 
339 		private static String binaryOp(Token t) {
340 			return switch (t.type()) {
341 				case AND -> "and";
342 				case OR -> "or";
343 				case MATCHES -> "matches";
344 				default -> t.text();
345 			};
346 		}
347 
348 		/** Parses an expression whose infix operators all bind tighter than {@code minBp}; one nesting level deeper. */
349 		private Expr parseExpr(int minBp) {
350 			if (++depth > maxDepth)
351 				throw new ExprException("expression nested too deeply", src, peek().pos());
352 			try {
353 				var left = parsePrefix();
354 				var lastWasMatches = false;
355 				for (;;) {
356 					var t = peek();
357 					var pf = parsePostfix(left);
358 					if (pf != null) {
359 						countOp(t);
360 						left = pf;
361 						continue;
362 					}
363 					int bp = infixBp(t.type());
364 					if (bp <= minBp)
365 						return left;
366 					if (t.type() == TokenType.MATCHES && lastWasMatches)
367 						throw new ExprException("'matches' is non-associative; use parentheses", src, t.pos());
368 					countOp(t);
369 					next();
370 					lastWasMatches = t.type() == TokenType.MATCHES;
371 					if (t.type() == TokenType.QUESTION) {
372 						var a = parseExpr(0);
373 						expect(TokenType.COLON, ":");
374 						left = new Expr.Ternary(left, a, parseExpr(BP_TERNARY - 1), t.pos());
375 					} else if (t.type() == TokenType.ELVIS) {
376 						left = new Expr.Elvis(left, parseExpr(BP_ELVIS - 1), t.pos());
377 					} else {
378 						left = new Expr.Binary(binaryOp(t), left, parseExpr(bp), t.pos());
379 					}
380 				}
381 			} finally {
382 				depth--;
383 			}
384 		}
385 
386 		/** Caps total operators so evaluation recursion depth never depends on the JVM stack size. */
387 		private void countOp(Token t) {
388 			if (++ops > maxOps)
389 				throw new ExprException("expression too complex", src, t.pos());
390 		}
391 
392 		/** Applies one postfix operator to {@code target}, or returns <jk>null</jk> if the next token is not postfix. */
393 		private Expr parsePostfix(Expr target) {
394 			var t = peek();
395 			switch (t.type()) {
396 				case DOT -> {
397 					next();
398 					return new Expr.Path(target, propertyName(), false, t.pos());
399 				}
400 				case SAFE_DOT -> {
401 					next();
402 					if (peek().type() == TokenType.LBRACKET) {
403 						next();
404 						var idx = parseExpr(0);
405 						expect(TokenType.RBRACKET, "]");
406 						return new Expr.Index(target, idx, true, t.pos());
407 					}
408 					return new Expr.Path(target, propertyName(), true, t.pos());
409 				}
410 				case LBRACKET -> {
411 					next();
412 					var idx = parseExpr(0);
413 					expect(TokenType.RBRACKET, "]");
414 					return new Expr.Index(target, idx, false, t.pos());
415 				}
416 				case SELECT -> {
417 					next();
418 					var pred = parseExpr(0);
419 					expect(TokenType.RBRACKET, "]");
420 					return new Expr.Select(target, pred, t.pos());
421 				}
422 				case PROJECT -> {
423 					next();
424 					var map = parseExpr(0);
425 					expect(TokenType.RBRACKET, "]");
426 					return new Expr.Project(target, map, t.pos());
427 				}
428 				default -> {
429 					return null;
430 				}
431 			}
432 		}
433 
434 		/** Reads a property name after <c>.</c> / <c>?.</c>: an identifier or any word keyword. */
435 		private String propertyName() {
436 			var t = peek();
437 			var ok = switch (t.type()) {
438 				case IDENT, TRUE, FALSE, NULL, MATCHES -> true;
439 				case AND, OR, NOT -> isWord(t);
440 				default -> false;
441 			};
442 			if (ok)
443 				return next().text();
444 			throw new ExprException("Expected property name but found " + describe(t), src, t.pos());
445 		}
446 
447 		private Expr parsePrefix() {
448 			var t = peek();
449 			switch (t.type()) {
450 				case STRING, NUMBER -> {
451 					next();
452 					return new Expr.Literal(t.value(), t.pos());
453 				}
454 				case TRUE, FALSE -> {
455 					next();
456 					return new Expr.Literal(t.type() == TokenType.TRUE, t.pos());
457 				}
458 				case NULL -> {
459 					next();
460 					return new Expr.Literal(null, t.pos());
461 				}
462 				case VARREF -> {
463 					next();
464 					return new Expr.VarRef(t.text(), t.pos());
465 				}
466 				case IDENT -> {
467 					next();
468 					if (peek().type() == TokenType.LPAREN)
469 						return parseCall(t);
470 					return new Expr.Ident(t.text(), t.pos());
471 				}
472 				case AND, OR -> {
473 					if (isWord(t) && peek(1).type() == TokenType.LPAREN) {
474 						next();
475 						return parseCall(t);
476 					}
477 					throw unexpected(t);
478 				}
479 				case NOT -> {
480 					next();
481 					if (isWord(t) && peek().type() == TokenType.LPAREN)
482 						return parseCall(t);
483 					return new Expr.Unary("not", parseExpr(BP_PREFIX), t.pos());
484 				}
485 				case MINUS, PLUS -> {
486 					next();
487 					return new Expr.Unary(t.text(), parseExpr(BP_PREFIX), t.pos());
488 				}
489 				case LPAREN -> {
490 					next();
491 					var e = parseExpr(0);
492 					expect(TokenType.RPAREN, ")");
493 					return e;
494 				}
495 				default -> throw unexpected(t);
496 			}
497 		}
498 
499 		/** Parses the argument list of a call whose name token was just consumed; the next token is <c>(</c>. */
500 		private Expr parseCall(Token name) {
501 			next();
502 			var args = new ArrayList<Expr>();
503 			if (peek().type() != TokenType.RPAREN) {
504 				for (;;) {
505 					args.add(parseExpr(0));
506 					if (peek().type() == TokenType.COMMA) {
507 						next();
508 						continue;
509 					}
510 					break;
511 				}
512 			}
513 			expect(TokenType.RPAREN, ")");
514 			return new Expr.Call(name.text(), List.copyOf(args), name.pos());
515 		}
516 	}

518 	private static int scanNumber(String src, int start, List<Token> out) {
519 		int n = src.length(), i = start;
520 		boolean dec = false;
521 		while (i < n && isDigit(src.charAt(i))) i++;
522 		if (i < n && src.charAt(i) == '.' && !(i + 1 < n && (Character.isJavaIdentifierStart(src.charAt(i + 1)) || src.charAt(i + 1) == '.' || src.charAt(i + 1) == '?' || src.charAt(i + 1) == '!'))) {
523 			dec = true;
524 			i++;
525 			while (i < n && isDigit(src.charAt(i))) i++;
526 		}
527 		if (i < n && (src.charAt(i) == 'e' || src.charAt(i) == 'E')) {
528 			int j = i + 1;
529 			if (j < n && (src.charAt(j) == '+' || src.charAt(j) == '-')) j++;
530 			if (j < n && isDigit(src.charAt(j))) {
531 				while (j < n && isDigit(src.charAt(j))) j++;
532 				dec = true;
533 				i = j;
534 			}
535 		}
536 		var text = src.substring(start, i);
537 		Object v;
538 		if (! dec) {
539 			try {
540 				v = Long.parseLong(text);
541 			} catch (@SuppressWarnings("unused") NumberFormatException e) {
542 				v = Double.parseDouble(text);
543 			}
544 		} else
545 			v = Double.parseDouble(text);
546 		out.add(new Token(TokenType.NUMBER, text, start, v));
547 		return i;
548 	}
