31  /**
32   * Utility class for resolving variables of the form <js>"${X:key}"</js> in strings.
33   *
34   * <p>
35   * Variables are of the form <c>${X:key}</c>, where <c>X</c> can consist of zero or more ASCII characters.
36   * <br>The variable key can contain anything, even nested variables that get recursively resolved.
37   *
38   * <p>
39   * Variables are defined through the {@link Builder#vars(Class[])} method.
40   *
41   * <p>
42   * The {@link Var} interface defines how variables are converted to values.
43   *
44   * <h2 class='section'>Example:</h2>
45   * <p class='bjava'>
46   * 	<jk>public class</jk> SystemPropertiesVar <jk>extends</jk> SimpleVar {
47   *
48   * 		<jc>// Must have a no-arg constructor!</jc>
49   * 		<jk>public</jk> SystemPropertiesVar() {
50   * 			<jk>super</jk>(<js>"S"</js>);
51   * 		}
52   *
53   * 		<ja>@Override</ja>
54   * 		<jk>public</jk> String resolve(VarResolverSession <jv>session</jv>, String <jv>value</jv>) {
55   * 			<jk>return</jk> System.<jsm>getProperty</jsm>(<jv>value</jv>);
56   * 		}
57   * 	}
58   *
59   * 	<jc>// Create a variable resolver that resolves system properties (e.g. "${S:java.home}")</jc>
60   * 	VarResolver <jv>varResolver</jv> = VarResolver.<jsm>create</jsm>().vars(SystemPropertiesVar.<jk>class</jk>).build();
61   *
62   * 	<jc>// Use it!</jc>
63   * 	System.<jsf>out</jsf>.println(<jv>varResolver</jv>.resolve(<js>"java.home is set to ${S:java.home}"</js>));
64   * </p>
65   *
66   * <h2 class='section'>See Also:</h2><ul>
67   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/MarshallSimpleVariableLanguage">Marshall Simple Variable Language</a>
68   * </ul>
69   */

301 		/**
302 		 * Restricts which functions (including the reserved <c>$op.*</c> operator functions) the resolver exposes.
303 		 *
304 		 * <p>
305 		 * Each pattern is a glob over function names: <c>*</c> matches any run of characters, anything else matches
306 		 * exactly.  A function is permitted if its name matches at least one pattern.  Operators dispatch to
307 		 * <c>$op.*</c> functions (<c>$op.add</c>, <c>$op.ternary</c>, <c>$op.select</c>, ...), so they are denied by
308 		 * the allowlist too; use <c>allowFunctions("$op.*", ...)</c> to keep them.
309 		 *
310 		 * <p>
311 		 * If this method is never called, every function is permitted.  Calling it with no patterns denies everything,
312 		 * operators included.  Repeated calls are additive, including after {@link VarResolver#copy()}: a copy can widen the allowlist but cannot narrow it.  The filter is applied to the final function set at
313 		 * {@link #build()} time, and is preserved by {@link VarResolver#copy()}.
314 		 *
315 		 * <p>
316 		 * A denied function or operator is treated like an unknown one (strict mode: {@link ExprException} saying it
317 		 * is "not permitted"; lenient mode: the expression is emitted verbatim).
318 		 *
319 		 * @param patterns The glob patterns.  Must not be <jk>null</jk> or contain <jk>null</jk>.
320 		 * @return This builder.
321 		 */
322 		public Builder allowFunctions(String...patterns) {
323 			Objects.requireNonNull(patterns, "patterns must not be null");
324 			for (var p : patterns)
325 				Objects.requireNonNull(p, "allowFunctions pattern must not be null");
326 			if (allowedFunctions == null)
327 				allowedFunctions = new ArrayList<>();
328 			Collections.addAll(allowedFunctions, patterns);
329 			return this;
330 		}

383 		 * Registers third-party {@link VarFunction} implementations discovered through {@link ServiceLoader}
384 		 * ({@code META-INF/services/org.apache.juneau.commons.svl.VarFunction}).
385 		 *
386 		 * <p>
387 		 * This is opt-in.  A discovered function never replaces a function that is already registered
388 		 * under the same name (including the built-ins registered by {@link #defaultFunctions()}), so call
389 		 * this method after {@link #defaultFunctions()}.
390 		 *
391 		 * @return This builder.
392 		 */
393 		public Builder loadServices() {
394 			var existing = new HashSet<String>();
395 			for (var f : functions)
396 				existing.add(f.name());
397 			for (var f : ServiceLoader.load(VarFunction.class, VarResolver.class.getClassLoader())) {
398 				if (existing.add(f.name()))
399 					functions.add(f);
400 			}
401 			return this;
402 		}

437 	@SafeVarargs
438 	private static <T> Class<? extends T>[] concat(Class<? extends T>[]... arrays) {
439 		var total = 0;
440 		for (var a : arrays) total += a.length;
441 		var out = new Class[total];
442 		var idx = 0;
443 		for (var a : arrays) {
444 			System.arraycopy(a, 0, out, idx, a.length);
445 			idx += a.length;
446 		}
447 		return out;
448 	}

523 	/**
524 	 * Constructor.
525 	 *
526 	 * @param builder The builder for this object.
527 	 */
528 	@SuppressWarnings({
529 		"rawtypes" // Same reason — Builder.bean(Class<T>, T) ensures Class/value pairing
530 	})
531 	protected VarResolver(Builder builder) {
532 		this.maxRecursionDepth = builder.maxRecursionDepth;
533 		this.strict = builder.strict;
534 		this.maxCollectionSize = builder.maxCollectionSize;
535 		this.maxExprDepth = builder.maxExprDepth;
536 		this.customValueFormatter = builder.valueFormatter;
537 		this.valueFormatter = builder.valueFormatter != null ? builder.valueFormatter : new JsonValueFormatter(builder.maxCollectionSize, builder.strict);
538 		this.vars = builder.vars.stream().map(x -> toVar(builder.beanStore(), x)).toArray(Var[]::new);
539 
540 		var m = new ConcurrentSkipListMap<String,Var>();
541 		for (var v : vars)
542 			m.put(v.getName(), v);
543 
544 		this.varMap = u(m);
545 		this.allFunctions = builder.functions.toArray(new VarFunction[0]);
546 		this.allowedFunctions = builder.allowedFunctions == null ? null : u(new ArrayList<>(builder.allowedFunctions));
547 
548 		var fm = new ConcurrentSkipListMap<String,VarFunction>();
549 		var denied = new TreeSet<String>();
550 		for (var f : allFunctions) {
551 			if (isAllowed(f.name())) {
552 				fm.put(f.name(), f);
553 				denied.remove(f.name());
554 			} else {
555 				fm.remove(f.name());
556 				denied.add(f.name());
557 			}
558 		}
559 		this.functions = fm.values().toArray(new VarFunction[0]);
560 		this.functionMap = u(fm);
561 		this.deniedNames = u(denied);
562 
563 		var bs = new BasicBeanStore(builder.beanStore());
564 		builder.userBeans.forEach((c, v) -> bs.addBean((Class) c, v));
565 		this.beanStore = bs;
566 	}

624 	/**
625 	 * Compiles a template string into a reusable {@link VarTemplate}.
626 	 *
627 	 * <p>
628 	 * Compilation is the dual of {@link #resolve(String)} — instead of tokenizing and resolving
629 	 * in one pass, the input is parsed once and the result returned as an immutable
630 	 * {@link VarTemplate} that callers can reuse across many resolutions:
631 	 *
632 	 * <p class='bjava'>
633 	 * 	<jc>// Compile once...</jc>
634 	 * 	<jk>var</jk> <jv>tpl</jv> = <jv>vr</jv>.compile(<js>"hello, ${name:world}"</js>);
635 	 *
636 	 * 	<jc>// ...resolve many times.</jc>
637 	 * 	<jk>for</jk> (<jk>var</jk> <jv>req</jv> : <jv>requests</jv>) {
638 	 * 		<jk>var</jk> <jv>session</jv> = <jv>vr</jv>.createSession().bean(<jv>req</jv>);
639 	 * 		<jv>writer</jv>.write(<jv>tpl</jv>.resolve(<jv>session</jv>));
640 	 * 	}
641 	 * </p>
642 	 *
643 	 * <p>
644 	 * The returned {@link VarTemplate} is immutable and threadsafe; the {@link VarResolverSession}
645 	 * passed to {@link VarTemplate#resolve(VarResolverSession)} inherits its own threadsafety
646 	 * contract (sessions are <b>not</b> threadsafe). Use {@link #resolveSupplier(String)} or
647 	 * {@link VarTemplate#asSupplierWithFreshSessions(VarResolver)} for a threadsafe re-evaluating
648 	 * Supplier.
649 	 *
650 	 * <h4 class='section'>Compile-binding contract:</h4>
651 	 * <p>
652 	 * The compiled template is bound to <i>this</i> resolver. Cached {@link Var} (and, in the
653 	 * future, function) references are resolved at compile time. Passing the compiled template
654 	 * to a session belonging to a different {@link VarResolver} is undefined behavior.
655 	 * Rebuilding the resolver invalidates compiled templates from the previous instance.
656 	 *
657 	 * <p>
658 	 * Templates are <b>not</b> cached on the resolver. Callers that compile the same input
659 	 * repeatedly (e.g. framework consumers compiling per {@code @Value} field) should cache
660 	 * the returned {@link VarTemplate} on their own per-site metadata.
661 	 *
662 	 * @param input The template string to compile. May be {@code null}.
663 	 * @return A new {@link VarTemplate} bound to this resolver.
664 	 */
665 	public VarTemplate compile(String input) {
666 		return VarTemplateCompiler.compile(this, input);
667 	}

671 	 * resolver each time {@link Supplier#get()} is called.
672 	 *
673 	 * <p>
674 	 * Each call to {@link Supplier#get()} opens a fresh {@link VarResolverSession}, so the
675 	 * returned {@code Supplier} is safe to share across threads — making it the recommended
676 	 * default for user-facing code (e.g. {@code @Value Supplier<String>} field types,

682 	 * <p>

687 	 * @return A threadsafe {@code Supplier<String>}.

748 	/**
749 	 * Returns <jk>true</jk> if the allowlist (if any) permits the given function name.
750 	 *
751 	 * <p>
752 	 * Also applies to names with no registered function, so an operator can be reported as denied even when its
753 	 * function is not registered.
754 	 *
755 	 * @param name The function name.
756 	 * @return <jk>true</jk> if permitted.
757 	 */
758 	boolean isAllowed(String name) {
759 		if (allowedFunctions == null)
760 			return true;
761 		for (var pat : allowedFunctions)
762 			if (globMatches(pat, name))
763 				return true;
764 		return false;
765 	}
766 
767 	private static boolean globMatches(String pattern, String name) {
768 		if (pattern.indexOf('*') < 0)
769 			return pattern.equals(name);
770 		var parts = pattern.split("\\*", -1);
771 		var pos = 0;
772 		if (! name.startsWith(parts[0]))
773 			return false;
774 		pos = parts[0].length();
775 		for (var i = 1; i < parts.length - 1; i++) {
776 			var idx = name.indexOf(parts[i], pos);
777 			if (idx < 0)
778 				return false;
779 			pos = idx + parts[i].length();
780 		}
781 		var last = parts[parts.length - 1];
782 		return name.length() - pos >= last.length() && name.endsWith(last);
783 	}
