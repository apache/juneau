35  /**
36   * A var resolver session that combines a {@link VarResolver} with one or more session objects.
37   *
38   * <p>
39   * Instances of this class are considered light-weight and fast to construct, use, and discard.
40   *
41   * <p>
42   * This class contains the workhorse code for var resolution.
43   *
44   * <p>
45   * Instances of this class are created through the {@link VarResolver#createSession()} and
46   * {@link VarResolver#createSession(WritableBeanStore)} methods.
47   *
48   * <h2 class='section'>Notes:</h2><ul>
49   * 	<li class='warn'>This class is not guaranteed to be thread safe.
50   * </ul>
51   *
52   * <h2 class='section'>See Also:</h2><ul>
53   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/MarshallSimpleVariableLanguage">Marshall Simple Variable Language</a>
54   * </ul>
55   */

62  	private static boolean containsVars(Collection<?> c) {
63  		var f = Flag.create();
64  		c.forEach(x -> {
65  			if (x instanceof CharSequence && x.toString().contains("$"))
66  				f.set();
67  		});
68  		return f.isSet();
69  	}

71  	private static boolean containsVars(Map<?,?> m) {
72  		var f = Flag.create();
73  		m.forEach((k, v) -> {
74  			if (v instanceof CharSequence && v.toString().contains("$"))
75  				f.set();
76  		});
77  		return f.isSet();
78  	}

80  	private static boolean containsVars(Object array) {
81  		for (var i = 0; i < Array.getLength(array); i++) {
82  			var o = Array.get(array, i);
83  			if (o instanceof CharSequence && o.toString().contains("$"))
84  				return true;
85  		}
86  		return false;
87  	}

143 	 * Resolves a <c>${Type:body}</c> reference to a typed value for the expression evaluator.
144 	 *
145 	 * <p>
146 	 * Finds the var by prefix and returns <c>var.doResolveValue(session, body)</c>, so defaults ({@link DefaultingVar})
147 	 * and multipart keys are honoured.  The body is resolved with the existing nested-marker machinery first when the
148 	 * var allows nesting and the body contains a <c>$</c>.  The {@link #resolveScriptStyle(String)} refusal and the
149 	 * {@link #resolveHtmlDoc(String)} encoding apply exactly as they do to template <c>${...}</c> references: under
150 	 * HTML-doc encoding a value from a non-passthrough var is escaped, and a non-<c>String</c> value other than a
151 	 * <c>Number</c> or <c>Boolean</c> is first converted with <c>String.valueOf</c> (as the template path does), so
152 	 * markup inside a collection or bean cannot bypass the escaping.
153 	 *
154 	 * @param prefix The var name (e.g. <js>"S"</js>).
155 	 * @param body The var body (already in the var's own syntax, e.g. <js>"key,default"</js>).
156 	 * @return The value, or <jk>null</jk> if absent or (lenient mode) the var is unknown.
157 	 * @throws VarResolverException If the var is unknown and the resolver is strict, or the var fails.
158 	 */
159 	Object resolveValue(String prefix, String body) {
160 		var v = isScriptStyleRefused(prefix) ? null : getVar(prefix);
161 		if (v == null) {
162 			if (context.strict)
163 				throw new VarResolverException("Unknown or unavailable variable '%s'", prefix);
164 			return null;
165 		}
166 		try {
167 			var b = (v.allowNested() && body.indexOf('$') != -1) ? resolve(body) : body;
168 			var r = v.doResolveValue(this, b);
169 			// HTML-doc chrome: values are escaped unless the var is an HTML passthrough, as in templates.
170 			if (! htmlDocEncode || r == null || r instanceof Number || r instanceof Boolean)
171 				return r;
172 			return applyHtmlDocChromeEncoding(prefix, r instanceof String s2 ? s2 : String.valueOf(r));
173 		} catch (VarResolverException e) {
174 			throw e;
175 		} catch (Exception e) {
176 			throw new VarResolverException(e, "Problem occurred resolving variable '%s': %s", prefix, e.getMessage());
177 		}
178 	}

195 	 * Formats the typed result of an expression as text.
196 	 *
197 	 * @param value The value.  May be <jk>null</jk>.
198 	 * @return The text; never <jk>null</jk>.
199 	 */
200 	public String formatValue(Object value) {
201 		String r;
202 		if (valueFormatter != null)
203 			r = valueFormatter.apply(value);
204 		else if (value instanceof Structure)
205 			r = value.toString();
206 		else
207 			r = context.valueFormatter.apply(value);
208 		return r == null ? "" : r;
209 	}

217 	/**
218 	 * Enters a {@code ${...}} reference inside {@code #{...}}.
219 	 *
220 	 * <p>
221 	 * Such references compile their body lazily, so a deep {@code #{${...#{${...}}...}}} chain would recurse at
222 	 * evaluation time.  The chain is capped at {@code maxExprDepth}; every successful call must be paired with
223 	 * {@link #exitExprVar()}.
224 	 *
225 	 * @return <jk>false</jk> (and no depth change) if the cap is reached.
226 	 */
227 	boolean enterExprVar() {
228 		if (exprVarDepth >= context.maxExprDepth)
229 			return false;
230 		exprVarDepth++;
231 		return true;
232 	}
233 
234 	/** Leaves a reference entered with {@link #enterExprVar()}. */
235 	void exitExprVar() {
236 		exprVarDepth--;
237 	}

276 	 * Resolve all variables in the specified string.
277 	 *
278 	 * @param s
279 	 * 	The string to resolve variables in.  Can be <jk>null</jk> (returns <jk>null</jk>).
280 	 * @return
281 	 * 	The new string with all variables resolved, or the same string if no variables were found.
282 	 * 	<br>Returns <jk>null</jk> if the input was <jk>null</jk>.
283 	 */
284 	public String resolve(String s) {
285 		if (s == null)
286 			return null;
287 		if (s.isEmpty())
288 			return s;
289 		return context.compile(s).resolve(this);
290 	}

292 	/**
293 	 * Re-resolves a var's output, enforcing the resolver's maximum recursion depth.
294 	 *
295 	 * @param s The string to resolve.
296 	 * @return The resolved string.
297 	 * @throws VarResolverException If the maximum recursion depth is exceeded.
298 	 */
299 	String resolveRecursive(String s) {
300 		if (++recursionDepth > context.maxRecursionDepth) {
301 			recursionDepth--;
302 			throw new VarResolverException("Maximum variable recursion depth (%s) exceeded while resolving '%s'", context.maxRecursionDepth, s.length() > 80 ? s.substring(0, 80) + "..." : s);
303 		}
304 		try {
305 			return resolve(s);
306 		} finally {
307 			recursionDepth--;
308 		}
309 	}

313 	 *
314 	 * <p>
315 	 * Literal characters from the template (including tags) pass through unchanged. Each SVL
316 	 * substitution is XML-escaped via {@link StringUtils#escapeXml(String)} before insert, except

318 	 *
319 	 * <p>

321 	 * footer, nav HTML fragments, head, no-results).
322 	 *

326 	public String resolveHtmlDoc(String s) {
327 		htmlDocEncode = true;
328 		try {
329 			return resolve(s);
330 		} finally {
331 			htmlDocEncode = false;

333 	}

372 	 * Returns a session-bound {@link Supplier} that re-resolves the given template against
373 	 * <i>this</i> session each time {@link Supplier#get()} is called.
374 	 *
375 	 * <p>
376 	 * Unlike {@link VarResolver#resolveSupplier(String)} — which opens a fresh session per
377 	 * {@code .get()} and is threadsafe to share — this Supplier captures the existing session
378 	 * and reuses it. <b>It is therefore not threadsafe</b>; it inherits the standard
379 	 * "{@link VarResolverSession} is not guaranteed to be thread safe" contract.
380 	 *
381 	 * <p>
382 	 * Use this variant only when the caller owns the session lifecycle and the perf saving of
383 	 * skipping per-call session construction matters. For shared / cross-thread use cases,
384 	 * prefer {@link VarResolver#resolveSupplier(String)}.
385 	 *
386 	 * @param input The template string. May be {@code null}.
387 	 * @return A {@code Supplier<String>} bound to this session.
388 	 */
389 	public Supplier<String> resolveSupplier(String input) {
390 		return compile(input).asSupplier(this);
391 	}

394 	 * Resolves the specified strings in the string array.
395 	 *
396 	 * @param in The string array containing variables to resolve.  Must not be <jk>null</jk> or a {@link NullPointerException} is thrown, though individual elements can be <jk>null</jk>.

402 			out[i] = resolve(in[i]);
403 		return out;

406 	/**
407 	 * Convenience method for resolving variables in arbitrary objects.
408 	 *
409 	 * <p>
410 	 * Supports resolving variables in the following object types:
411 	 * <ul>
412 	 * 	<li>{@link CharSequence}
413 	 * 	<li>Arrays containing values of type {@link CharSequence}.
414 	 * 	<li>Collections containing values of type {@link CharSequence}.
415 	 * 		<br>Collection class must have a no-arg constructor.
416 	 * 	<li>Maps containing values of type {@link CharSequence}.
417 	 * 		<br>Map class must have a no-arg constructor.
418 	 * </ul>
419 	 *
420 	 * @param <T> The value type.
421 	 * @param o The object.  Can be <jk>null</jk> (returns <jk>null</jk>).
422 	 * @return The same object if no resolution was needed, otherwise a new object or data structure if resolution was
423 	 * needed.  Returns <jk>null</jk> if the input was <jk>null</jk>.
424 	 */
425 	@SuppressWarnings({
426 		"java:S3776", // Cognitive complexity acceptable for recursive resolution logic
427 		"rawtypes", // Raw types necessary for generic resolution
428 		"unchecked" // Type erasure requires unchecked operations
429 	})
430 	public <T> T resolve(T o) {
431 		if (o == null)
432 			return null;
433 		if (o instanceof CharSequence o2)
434 			return (T)resolve(o2.toString());
435 		if (isArray(o)) {
436 			if (! containsVars(o))
437 				return o;
438 			var o2 = Array.newInstance(o.getClass().getComponentType(), Array.getLength(o));
439 			for (var i = 0; i < Array.getLength(o); i++)
440 				Array.set(o2, i, resolve(Array.get(o, i)));
441 			return (T)o2;
442 		}
443 		if (o instanceof Set o2) {
444 			try {
445 				if (! containsVars(o2))
446 					return o;
447 				Set o3 = info(o).getDeclaredConstructor(x -> x.isPublic() && x.getParameterCount() == 0).map(ci -> safe(() -> (Set)ci.inner().newInstance())).orElseGet(LinkedHashSet::new);
448 				Set o4 = o3;
449 				o2.forEach(x -> o4.add(resolve(x)));
450 				return (T)o3;
451 			} catch (VarResolverException e) {
452 				throw e;
453 			} catch (Exception e) {
454 				throw new VarResolverException(e, "Problem occurred resolving set.");
455 			}
456 		}
457 		if (o instanceof List o2) {
458 			try {
459 				if (! containsVars(o2))
460 					return o;
461 				List o3 = info(o).getDeclaredConstructor(x -> x.isPublic() && x.getParameterCount() == 0).map(ci -> safe(() -> (List)ci.inner().newInstance())).orElseGet(() -> list());
462 				List o4 = o3;
463 				o2.forEach(x -> o4.add(resolve(x)));
464 				return (T)o3;
465 			} catch (VarResolverException e) {
466 				throw e;
467 			} catch (Exception e) {
468 				throw new VarResolverException(e, "Problem occurred resolving collection.");
469 			}
470 		}
471 		if (o instanceof Map o2) {
472 			try {
473 				if (! containsVars(o2))
474 					return o;
475 				Map o3 = info(o).getDeclaredConstructor(x -> x.isPublic() && x.getParameterCount() == 0).map(ci -> safe(() -> (Map)ci.inner().newInstance())).orElseGet(LinkedHashMap::new);
476 				Map o4 = o3;
477 				o2.forEach((k, v) -> o4.put(k, resolve(v)));
478 				return (T)o3;
479 			} catch (VarResolverException e) {
480 				throw e;
481 			} catch (Exception e) {
482 				throw new VarResolverException(e, "Problem occurred resolving map.");
483 			}
484 		}
485 		return o;
486 	}

488 	/**
489 	 * Resolves variables in the specified string and sends the output to the specified writer.
490 	 *
491 	 * <p>
492 	 * More efficient than first parsing to a string and then serializing to the writer since this method doesn't need
493 	 * to construct a large string.
494 	 *
495 	 * @param s The string to resolve variables in.  Can be <jk>null</jk>, in which case the writer is left unchanged.
496 	 * @param out The writer to write to.  Must not be <jk>null</jk>.
497 	 * @return The same writer.
498 	 * @throws IOException Thrown by underlying stream.
499 	 */
500 	public Writer resolveTo(String s, Writer out) throws IOException {
501 		if (s == null)
502 			return out;
503 		if (s.isEmpty()) {
504 			return out;
505 		}
506 		return context.compile(s).resolveTo(this, out);
507 	}
