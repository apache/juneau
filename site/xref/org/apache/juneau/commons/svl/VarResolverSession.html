35  /**
36   * A var resolver session that combines a {@link VarResolver} with one or more session objects.
37   *
38   * <p>
39   * Instances of this class are considered light-weight and fast to construct, use, and discard.
40   *
41   * <p>
42   * This class contains the workhorse code for var resolution.
43   *
44   * <p>
45   * Instances of this class are created through the {@link VarResolver#createSession()} and
46   * {@link VarResolver#createSession(WritableBeanStore)} methods.
47   *
48   * <h5 class='section'>Notes:</h5><ul>
49   * 	<li class='warn'>This class is not guaranteed to be thread safe.
50   * </ul>
51   *
52   * <h5 class='section'>See Also:</h5><ul>
53   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/MarshallSimpleVariableLanguage">Marshall Simple Variable Language</a>
54   * </ul>
55   */
56  @SuppressWarnings({
57  	"java:S1192", // Duplicated literals (argument/property names) read more clearly inline than as constants
58  	"resource" // VarResolver resources managed by calling code
59  })

62  	private static boolean containsVars(Collection<?> c) {
63  		var f = Flag.create();
64  		c.forEach(x -> {
65  			if (x instanceof CharSequence && x.toString().contains("$"))
66  				f.set();
67  		});
68  		return f.isSet();
69  	}
70  
71  	private static boolean containsVars(Map<?,?> m) {
72  		var f = Flag.create();
73  		m.forEach((k, v) -> {
74  			if (v instanceof CharSequence && v.toString().contains("$"))
75  				f.set();
76  		});
77  		return f.isSet();
78  	}
79  
80  	private static boolean containsVars(Object array) {
81  		for (var i = 0; i < Array.getLength(array); i++) {
82  			var o = Array.get(array, i);
83  			if (o instanceof CharSequence && o.toString().contains("$"))
84  				return true;
85  		}
86  		return false;
87  	}

130 	 *
131 	 * @param <T> The value type.

144 	/**
145 	 * Resolve all variables in the specified string.
146 	 *
147 	 * @param s
148 	 * 	The string to resolve variables in.  Can be <jk>null</jk> (returns <jk>null</jk>).
149 	 * @return
150 	 * 	The new string with all variables resolved, or the same string if no variables were found.
151 	 * 	<br>Returns <jk>null</jk> if the input was <jk>null</jk>.
152 	 */
153 	public String resolve(String s) {
154 		if (s == null)
155 			return null;
156 		if (s.isEmpty())
157 			return s;
158 		return context.compile(s).resolve(this);
159 	}

161 	/**
162 	 * Resolves SVL variables for HTML-doc chrome fields that write raw HTML.
163 	 *
164 	 * <p>
165 	 * Literal characters from the template (including tags) pass through unchanged. Each SVL
166 	 * substitution is XML-escaped via {@link StringUtils#escapeXml(String)} before insert, except
167 	 * the HTML-passthrough vars {@code $W}, {@code $C}, and {@code $RS}.
168 	 *
169 	 * <p>
170 	 * Plain {@link #resolve(String)} is unchanged. Use this only for HTML-doc chrome (aside, header,
171 	 * footer, nav HTML fragments, head, no-results).
172 	 *
173 	 * @param s The chrome template. Can be <jk>null</jk>.
174 	 * @return The resolved chrome string, or <jk>null</jk> if the input was <jk>null</jk>.
175 	 */
176 	public String resolveHtmlDoc(String s) {
177 		htmlDocEncode = true;
178 		try {
179 			return resolve(s);
180 		} finally {
181 			htmlDocEncode = false;
182 		}
183 	}

185 	/**
186 	 * Resolves SVL variables for {@code <script>} and {@code <style>} chrome.
187 	 *
188 	 * <p>
189 	 * Request-derived vars ({@code $RQ}, {@code $RH}, {@code $RF}, {@code $RP}, {@code $RA},
190 	 * {@code $SA}, {@code $R}) are left unresolved (same as an unknown var). Other vars including
191 	 * {@code $W}, {@code $C}, and {@code $RS} still resolve.
192 	 *
193 	 * @param s The script or style template. Can be <jk>null</jk>.
194 	 * @return The resolved string, or <jk>null</jk> if the input was <jk>null</jk>.
195 	 */
196 	public String resolveScriptStyle(String s) {
197 		scriptStyleRefuse = true;
198 		try {
199 			return resolve(s);
200 		} finally {
201 			scriptStyleRefuse = false;
202 		}
203 	}

206 	 * Compiles the given template string into a reusable {@link VarTemplate}.
207 	 *
208 	 * <p>
209 	 * Equivalent to {@link VarResolver#compile(String) this.context.compile(input)}; provided on
210 	 * the session for symmetry with {@link #resolve(String)}. The returned template is bound to

212 	 * class javadoc for the compile-binding contract.
213 	 *
214 	 * @param input The template string. May be {@code null}.
215 	 * @return A new {@link VarTemplate} bound to the underlying resolver.
216 	 */
217 	public VarTemplate compile(String input) {
218 		return context.compile(input);

221 	/**
222 	 * Returns a session-bound {@link Supplier} that re-resolves the given template against
223 	 * <i>this</i> session each time {@link Supplier#get()} is called.
224 	 *
225 	 * <p>
226 	 * Unlike {@link VarResolver#resolveSupplier(String)} — which opens a fresh session per
227 	 * {@code .get()} and is threadsafe to share — this Supplier captures the existing session
228 	 * and reuses it. <b>It is therefore not threadsafe</b>; it inherits the standard
229 	 * "{@link VarResolverSession} is not guaranteed to be thread safe" contract.
230 	 *
231 	 * <p>
232 	 * Use this variant only when the caller owns the session lifecycle and the perf saving of
233 	 * skipping per-call session construction matters. For shared / cross-thread use cases,
234 	 * prefer {@link VarResolver#resolveSupplier(String)}.
235 	 *
236 	 * @param input The template string. May be {@code null}.
237 	 * @return A {@code Supplier<String>} bound to this session.
238 	 */
239 	public Supplier<String> resolveSupplier(String input) {
240 		return compile(input).asSupplier(this);
241 	}

243 	/**
244 	 * Resolves the specified strings in the string array.
245 	 *
246 	 * @param in The string array containing variables to resolve.  Must not be <jk>null</jk> or a {@link NullPointerException} is thrown, though individual elements can be <jk>null</jk>.
247 	 * @return An array with resolved strings.  Elements are <jk>null</jk> where the corresponding input element was <jk>null</jk>.
248 	 */
249 	public String[] resolve(String[] in) {
250 		var out = new String[in.length];
251 		for (var i = 0; i < in.length; i++)
252 			out[i] = resolve(in[i]);
253 		return out;
254 	}

256 	/**
257 	 * Convenience method for resolving variables in arbitrary objects.
258 	 *
259 	 * <p>
260 	 * Supports resolving variables in the following object types:
261 	 * <ul>
262 	 * 	<li>{@link CharSequence}
263 	 * 	<li>Arrays containing values of type {@link CharSequence}.
264 	 * 	<li>Collections containing values of type {@link CharSequence}.
265 	 * 		<br>Collection class must have a no-arg constructor.
266 	 * 	<li>Maps containing values of type {@link CharSequence}.
267 	 * 		<br>Map class must have a no-arg constructor.
268 	 * </ul>
269 	 *
270 	 * @param <T> The value type.
271 	 * @param o The object.  Can be <jk>null</jk> (returns <jk>null</jk>).
272 	 * @return The same object if no resolution was needed, otherwise a new object or data structure if resolution was
273 	 * needed.  Returns <jk>null</jk> if the input was <jk>null</jk>.
274 	 */
275 	@SuppressWarnings({
276 		"java:S3776", // Cognitive complexity acceptable for recursive resolution logic
277 		"rawtypes", // Raw types necessary for generic resolution
278 		"unchecked" // Type erasure requires unchecked operations
279 	})
280 	public <T> T resolve(T o) {
281 		if (o == null)
282 			return null;
283 		if (o instanceof CharSequence o2)
284 			return (T)resolve(o2.toString());
285 		if (isArray(o)) {
286 			if (! containsVars(o))
287 				return o;
288 			var o2 = Array.newInstance(o.getClass().getComponentType(), Array.getLength(o));
289 			for (var i = 0; i < Array.getLength(o); i++)
290 				Array.set(o2, i, resolve(Array.get(o, i)));
291 			return (T)o2;
292 		}
293 		if (o instanceof Set o2) {
294 			try {
295 				if (! containsVars(o2))
296 					return o;
297 				Set o3 = info(o).getDeclaredConstructor(x -> x.isPublic() && x.getParameterCount() == 0).map(ci -> safe(() -> (Set)ci.inner().newInstance())).orElseGet(LinkedHashSet::new);
298 				Set o4 = o3;
299 				o2.forEach(x -> o4.add(resolve(x)));
300 				return (T)o3;
301 			} catch (VarResolverException e) {
302 				throw e;
303 			} catch (Exception e) {
304 				throw new VarResolverException(e, "Problem occurred resolving set.");
305 			}
306 		}
307 		if (o instanceof List o2) {
308 			try {
309 				if (! containsVars(o2))
310 					return o;
311 				List o3 = info(o).getDeclaredConstructor(x -> x.isPublic() && x.getParameterCount() == 0).map(ci -> safe(() -> (List)ci.inner().newInstance())).orElseGet(() -> list());
312 				List o4 = o3;
313 				o2.forEach(x -> o4.add(resolve(x)));
314 				return (T)o3;
315 			} catch (VarResolverException e) {
316 				throw e;
317 			} catch (Exception e) {
318 				throw new VarResolverException(e, "Problem occurred resolving collection.");
319 			}
320 		}
321 		if (o instanceof Map o2) {
322 			try {
323 				if (! containsVars(o2))
324 					return o;
325 				Map o3 = info(o).getDeclaredConstructor(x -> x.isPublic() && x.getParameterCount() == 0).map(ci -> safe(() -> (Map)ci.inner().newInstance())).orElseGet(LinkedHashMap::new);
326 				Map o4 = o3;
327 				o2.forEach((k, v) -> o4.put(k, resolve(v)));
328 				return (T)o3;
329 			} catch (VarResolverException e) {
330 				throw e;
331 			} catch (Exception e) {
332 				throw new VarResolverException(e, "Problem occurred resolving map.");
333 			}
334 		}
335 		return o;
336 	}

338 	/**
339 	 * Resolves variables in the specified string and sends the output to the specified writer.
340 	 *
341 	 * <p>
342 	 * More efficient than first parsing to a string and then serializing to the writer since this method doesn't need
343 	 * to construct a large string.
344 	 *
345 	 * @param s The string to resolve variables in.  Can be <jk>null</jk>, in which case the writer is left unchanged.
346 	 * @param out The writer to write to.  Must not be <jk>null</jk>.
347 	 * @return The same writer.
348 	 * @throws IOException Thrown by underlying stream.
349 	 */
350 	public Writer resolveTo(String s, Writer out) throws IOException {
351 		if (s == null)
352 			return out;
353 		if (s.isEmpty()) {
354 			return out;
355 		}
356 		return context.compile(s).resolveTo(this, out);
357 	}

403 	/**
404 	 * XML-escapes an SVL substitution when HTML-doc chrome encoding is enabled and the var is not passthrough.
405 	 *
406 	 * @param varName The SVL var prefix.
407 	 * @param replacement The resolved substitution text.
408 	 * @return The possibly escaped replacement.
409 	 */
410 	String applyHtmlDocChromeEncoding(String varName, String replacement) {
411 		if (! htmlDocEncode || replacement == null || HTML_DOC_PASSTHROUGH_VARS.contains(varName))
412 			return replacement;
413 		return StringUtils.escapeXml(replacement);
