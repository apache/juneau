17  package org.apache.juneau.commons.svl;
18  
19  import java.util.*;
20  
21  /**
22   * SPI for SVL {@code #{...}} script-evaluation functions.
23   *
24   * <p>
25   * A {@code VarFunction} is invoked when an SVL template contains a
26   * <c>#{name(arg1, arg2, ...)}</c> reference — distinct from {@code $X{...}} variables. Functions
27   * compose freely with {@code ${...}} lookups and other {@code #{...}} calls; arguments are
28   * pre-resolved by the runtime before {@link #invoke(VarResolverSession, List) invoke} is called.
29   *
30   * <h5 class='section'>Naming note:</h5>
31   * <p>
32   * Named {@code VarFunction} rather than the shorter {@code Function} because Juneau's
33   * serializer/parser/test code makes heavy use of {@link java.util.function.Function} and
34   * routinely co-imports {@code org.apache.juneau.commons.svl.*}; a bare {@code Function} type in
35   * this package would trigger ambiguous-reference errors across many callers. The chosen name
36   * also matches the existing {@code Var}/{@code VarResolver}/{@code VarTemplate}/{@code VarResolverSession}
37   * naming family.
38   *
39   * <p>
40   * Implementations can register through any of four discovery channels (later wins on name
41   * collision):
42   * <ol>
43   * 	<li>The built-in catalog auto-registered by {@link VarResolver#DEFAULT}.
44   * 	<li>{@code META-INF/services/org.apache.juneau.commons.svl.VarFunction} (Java {@link java.util.ServiceLoader}).
45   * 	<li>{@code BeanStore.getBeans(VarFunction.class)} on the resolver's bean store.
46   * 	<li>Explicit registrations via {@code VarResolver.Builder.functions(...)}.
47   * </ol>
48   *
49   * <p>
50   * For typed reflection-based dispatch (the recommended path), extend
51   * {@link TypedFunction} — it derives {@link #minArity()}, {@link #maxArity()}, and per-arg
52   * coercion automatically from the subclass's {@code invoke(...)} method signature, eliminating
53   * the per-function plumbing.
54   *
55   * <h5 class='section'>Example:</h5>
56   * <p class='bjava'>
57   * 	<jc>// Direct VarFunction impl (less common — TypedFunction is preferred).</jc>
58   * 	<jk>public class</jk> Reverse <jk>implements</jk> VarFunction {
59   * 		<ja>@Override</ja> <jk>public</jk> String name() { <jk>return</jk> <js>"reverse"</js>; }
60   * 		<ja>@Override</ja> <jk>public int</jk> minArity() { <jk>return</jk> 1; }
61   * 		<ja>@Override</ja> <jk>public int</jk> maxArity() { <jk>return</jk> 1; }
62   * 		<ja>@Override</ja> <jk>public</jk> String invoke(VarResolverSession <jv>session</jv>, List&lt;Object&gt; <jv>args</jv>) {
63   * 			<jk>return new</jk> StringBuilder((String) <jv>args</jv>.get(0)).reverse().toString();
64   * 		}
65   * 	}
66   * </p>
67   *
68   * <h5 class='section'>See Also:</h5><ul>
69   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/MarshallSimpleVariableLanguage">Simple Variable Language Basics</a>
70   * 	<li class='jc'>{@link TypedFunction}
71   * </ul>
72   */
73  public interface VarFunction {
74  
75  	/**
76  	 * The function's invocation name as it appears between {@code #{}} and the opening
77  	 * parenthesis in the template. Must consist of ASCII letters, digits, and underscores; must
78  	 * begin with a letter.
79  	 *
80  	 * @return The invocation name. Never {@code null} or empty.
81  	 */
82  	String name();
83  
84  	/**
85  	 * Minimum argument count this function accepts.
86  	 *
87  	 * @return The minimum number of args. {@code 0} means the function may be called with no
88  	 *	 args. Returns {@code -1} for "any number of args" — equivalent to declaring
89  	 *	 {@code minArity == 0 && maxArity == Integer.MAX_VALUE}.
90  	 */
91  	int minArity();
92  
93  	/**
94  	 * Maximum argument count this function accepts.
95  	 *
96  	 * @return The maximum number of args, or {@link Integer#MAX_VALUE} for unbounded variadic
97  	 *	 functions.
98  	 */
99  	int maxArity();
100 
101 	/**
102 	 * Invokes this function with already-resolved, type-coerced arguments.
103 	 *
104 	 * <p>
105 	 * The runtime resolves each {@code #{...}} arg template against the session, then coerces
106 	 * the resolved string to the target Java type via {@code ArgCoercer}. The resulting
107 	 * {@code List<Object>} is passed in here.
108 	 *
109 	 * @param session The current resolver session. Must not be {@code null}.
110 	 * @param args The coerced arguments. Size is in {@code [minArity(), maxArity()]}. Each
111 	 *	 element is the type the function expects per its signature; for variadic /
112 	 *	 {@code Object}-typed slots, elements remain {@link String}.
113 	 * @return The function's result as a {@link String}. May be empty but should not be
114 	 *	 {@code null} (the runtime treats {@code null} as the empty string for SVL output).
115 	 */
116 	String invoke(VarResolverSession session, List<Object> args);
117 }
