17  package org.apache.juneau.commons.svl.functions;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.commons.svl.*;
22  
23  /**
24   * Small collection helpers: {@code list} and {@code take}.
25   *
26   * <p>
27   * {@code list(a, b, ...)} builds a list from its arguments, which makes it possible to produce the rows for
28   * {@code table(header, rows)}, for example {@code rows.![list(a, b)]}.  {@code take(list, n)} returns the first {@code n}
29   * elements, which together with {@code size} expresses "first N, then 'and M more'" without a slice syntax.
30   *
31   * <p>
32   * Both functions are side-effect free.  Neither can grow a collection beyond its inputs: {@code list} is bounded by the
33   * number of arguments written in the template and {@code take} returns a prefix of its argument.
34   */
35  public final class CollectionFunctions {
36  
37  	/**
38  	 * Applies the session's collection limit to a collection a function is about to copy.
39  	 *
40  	 * <p>
41  	 * Within the limit the elements are returned as a list.  Beyond it, strict mode throws and lenient mode keeps the
42  	 * first {@code maxCollectionSize} elements.
43  	 *
44  	 * @param session The session.
45  	 * @param fn The function name, for the message.
46  	 * @param c The collection.
47  	 * @return The elements, at most {@code maxCollectionSize} of them.
48  	 */
49  	static List<Object> bounded(VarResolverSession session, String fn, Collection<?> c) {
50  		var max = session.maxCollectionSize();
51  		if (c.size() <= max)
52  			return new ArrayList<>(c);
53  		if (session.isStrict())
54  			throw new VarResolverException("Function '%s' collection of %s elements exceeds the limit of %s", fn, c.size(), max);
55  		var out = new ArrayList<Object>(max);
56  		for (var o : c) {
57  			if (out.size() >= max)
58  				break;
59  			out.add(o);
60  		}
61  		return out;
62  	}
63  
64  	private CollectionFunctions() {}
65  
66  	/** All function classes in this category. */
67  	@SuppressWarnings({
68  		"java:S2386", // ALL is an immutable compile-time registry; exposed as an array for the cross-package/varargs functions(...) API.
69  		"unchecked" // Cast is safe: type verified by caller context.
70  	})
71  	public static final Class<? extends VarFunction>[] ALL = new Class[] {
72  		ListFn.class, Take.class
73  	};
74  
75  	/** {@code #{list(a, b, ...)}} — a list of the arguments; {@code list()} is empty. */
76  	public static class ListFn implements VarFunction {
77  		@Override public String name() { return "list"; }
78  		@Override public int minArity() { return 0; }
79  		@Override public int maxArity() { return Integer.MAX_VALUE; }
80  		@Override public String invoke(VarResolverSession session, List<Object> args) { return String.valueOf(invokeValue(session, args)); }
81  		@Override public Object invokeValue(VarResolverSession session, List<Object> args) { return new ArrayList<>(args); }
82  	}
83  
84  	/**
85  	 * {@code #{take(list, n)}} — the first {@code n} elements of {@code list} (all of them if it is shorter).
86  	 *
87  	 * <p>
88  	 * {@code n} must be a non-negative whole number; a {@code null} list yields an empty list.
89  	 */
90  	// TODO: replace with a range or slice syntax if the SVL grammar ever adds one (see take())
91  	public static class Take implements VarFunction {
92  		@Override public String name() { return "take"; }
93  		@Override public int minArity() { return 2; }
94  		@Override public int maxArity() { return 2; }
95  		@Override public String invoke(VarResolverSession session, List<Object> args) { return String.valueOf(invokeValue(session, args)); }
96  		@Override public Object invokeValue(VarResolverSession session, List<Object> args) {
97  			var src = args.get(0);
98  			var n = count(args.get(1));
99  			if (src == null)
100 				return new ArrayList<>();
101 			if (! (src instanceof Collection<?> c))
102 				throw new VarResolverException("Function 'take' argument 1 must be a list but got %s", src.getClass().getSimpleName());
103 			var out = new ArrayList<Object>(Math.min(n, c.size()));
104 			for (var o : bounded(session, "take", c)) {
105 				if (out.size() >= n)
106 					break;
107 				out.add(o);
108 			}
109 			return out;
110 		}
111 
112 		private static int count(Object o) {
113 			if (o instanceof Number x && x.doubleValue() == Math.rint(x.doubleValue()) && x.doubleValue() >= 0 && x.doubleValue() <= Integer.MAX_VALUE)
114 				return x.intValue();
115 			if (o instanceof Number)
116 				throw new VarResolverException("Function 'take' argument 2 must be a non-negative whole number but got %s", o);
117 			throw new VarResolverException("Function 'take' argument 2 must be a number but got %s", o == null ? "null" : o.getClass().getSimpleName());
118 		}
119 	}
120 }
