26  /**
27   * Conditional and routing functions for the {@code #{...}} script catalog.
28   *
29   * <p>
30   * These four functions replaced the legacy {@code IfVar}, {@code SwitchVar}, {@code CoalesceVar},
31   * and {@code NotEmptyVar} classes when the {@code #{...}} script syntax landed in 10.0.0.
32   */
33  public final class ConditionalFunctions {
34  
35  	private ConditionalFunctions() {}
36  
37  	/** All function classes in this category. */
38  	@SuppressWarnings({
39  		"unchecked", // Cast is safe: type verified by caller context.
40  		"java:S2386" // ALL is an immutable compile-time registry; exposed as an array for the cross-package/varargs functions(...) API, so visibility cannot be reduced.
41  	})
42  	public static final Class<? extends VarFunction>[] ALL = new Class[] {
43  		If.class, Switch.class, Coalesce.class, NotEmpty.class
44  	};
45  
46  	/**
47  	 * {@code #{if(cond, then, else)}} — replacement for the legacy {@code $IF{...}}.
48  	 *
49  	 * <p>
50  	 * Returns {@code then} if {@code cond} is truthy, {@code else} otherwise. Truthiness is
51  	 * explicit (no implicit truthiness).
52  	 */
53  	public static class If extends TypedFunction {
54  		@Override public String name() { return "if"; }
55  		public String invoke(boolean cond, String thenVal, String elseVal) {
56  			var then = thenVal == null ? "" : thenVal;
57  			var els = elseVal == null ? "" : elseVal;
58  			return cond ? then : els;
59  		}
60  	}
61  
62  	/**
63  	 * {@code #{switch(value, case1, val1, case2, val2, ..., default)}} — replacement for the
64  	 * legacy {@code $SW{...}}.
65  	 *
66  	 * <p>
67  	 * Compares {@code value} against each {@code caseN} <i>glob pattern</i> and returns the
68  	 * matching {@code valN} for the first match. {@code caseN} patterns may contain {@code *}
69  	 * (any sequence of characters) and {@code ?} (any single character) wildcards — same
70  	 * semantics as the legacy {@code SwitchVar}. Plain literals (no wildcards) match by
71  	 * full-string equality.
72  	 *
73  	 * <p>
74  	 * The last unmatched arg is the default; if the arg count is even (no trailing default),
75  	 * an empty string is returned for unmatched values.
76  	 *
77  	 * <h5 class='section'>Examples:</h5>
78  	 * <ul>
79  	 * 	<li>{@code #{switch(${os}, *win*, Windows, *nix*, Linux, *)}} — glob-routes by substring.
80  	 * 	<li>{@code #{switch(${env}, prod, A, dev, B, default)}} — exact equality, three branches.
81  	 * </ul>
82  	 */
83  	public static class Switch extends TypedFunction {
84  
85  		// Compiled glob-derived regexes are cached and reused.  Bounded (entire cache cleared past maxSize) to guard
86  		// against unbounded growth from arbitrary user-supplied glob patterns.  Pattern is immutable/thread-safe.
87  		private static final Cache<String,Pattern> PATTERN_CACHE = Cache.of(String.class, Pattern.class).maxSize(1000).supplier(Pattern::compile).build();
88  
89  		@Override public String name() { return "switch"; }
90  		public String invoke(String[] args) {
91  			if (args.length == 0) return "";
92  			var value = args[0] == null ? "" : args[0];
93  			var i = 1;
94  			while (i + 1 < args.length) {
95  				if (globMatches(value, args[i])) return args[i + 1] == null ? "" : args[i + 1];
96  				i += 2;
97  			}
98  			// Trailing default if arity is even (value + N pairs + default).
99  			if (i < args.length) return args[i] == null ? "" : args[i];
100 			return "";
101 		}
102 
103 		/** Full-string glob match supporting {@code *} and {@code ?} wildcards. */
104 		private static boolean globMatches(String value, String pattern) {
105 			if (pattern == null) return value == null;
106 			// Compile glob to regex once per call. Translation: '*' -> '.*', '?' -> '.',
107 			// every other regex metacharacter is quoted.
108 			var sb = new StringBuilder(pattern.length() + 4);
109 			for (var i = 0; i < pattern.length(); i++) {
110 				var c = pattern.charAt(i);
111 				switch (c) {
112 					case '*' -> sb.append(".*");
113 					case '?' -> sb.append('.');
114 					case '\\', '.', '[', ']', '(', ')', '{', '}', '|', '+', '^', '$' -> sb.append('\\').append(c);
115 					default -> sb.append(c);
116 				}
117 			}
118 			return PATTERN_CACHE.get(sb.toString()).matcher(value).matches();
119 		}
120 	}
121 
122 	/**
123 	 * {@code #{coalesce(a, b, ...)}} — replacement for the legacy {@code $CO{...}}.
124 	 *
125 	 * <p>
126 	 * Returns the first non-empty arg, or {@code ""} if all args are empty.
127 	 */
128 	public static class Coalesce extends TypedFunction {
129 		@Override public String name() { return "coalesce"; }
130 		public String invoke(String[] args) {
131 			for (var a : args)
132 				if (!isEmpty(a)) return a;
133 			return "";
134 		}
135 	}
136 
137 	/**
138 	 * {@code #{notEmpty(s)}} — replacement for the legacy {@code $NE{...}}.
139 	 *
140 	 * <p>
141 	 * Returns {@code "true"} if {@code s} is non-{@code null} and non-empty, {@code "false"}
142 	 * otherwise. Whitespace-only strings count as non-empty (matches {@code String.isEmpty()}).
143 	 */
144 	public static class NotEmpty extends TypedFunction {
145 		@Override public String name() { return "notEmpty"; }
146 		public String invoke(String s) { return String.valueOf(!isEmpty(s)); }
147 	}
148 }
