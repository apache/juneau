17  package org.apache.juneau.commons.svl.functions;
18  
19  import org.apache.juneau.commons.svl.*;
20  
21  /**
22   * Arithmetic functions for the {@code #{...}} script catalog.
23   *
24   * <p>
25   * All operations work on {@code double} so integer and decimal inputs compose naturally; the
26   * result renders as a String. Integer-perfect results omit the decimal point (e.g.
27   * {@code add(2, 3)} returns {@code "5"}, not {@code "5.0"}).
28   */
29  public final class ArithmeticFunctions {
30  
31  	private ArithmeticFunctions() {}
32  
33  	/** All function classes in this category. */
34  	@SuppressWarnings({
35  		"unchecked", // Cast is safe: type verified by caller context.
36  		"java:S2386" // ALL is an immutable compile-time registry; exposed as an array for the cross-package/varargs functions(...) API, so visibility cannot be reduced.
37  	})
38  	public static final Class<? extends VarFunction>[] ALL = new Class[] {
39  		Add.class, Subtract.class, Multiply.class, Divide.class, Modulo.class,
40  		Min.class, Max.class, Abs.class
41  	};
42  
43  	/** Render a double as a String, dropping the decimal portion when it's an exact integer. */
44  	static String render(double v) {
45  		if (Double.isNaN(v) || Double.isInfinite(v)) return String.valueOf(v);
46  		if (v == Math.floor(v) && Math.abs(v) < 1e18)
47  			return String.valueOf((long) v);
48  		return String.valueOf(v);
49  	}
50  
51  	/** {@code #{add(a, b)}} — returns {@code a + b}. */
52  	public static class Add extends TypedFunction {
53  		@Override public String name() { return "add"; }
54  		public String invoke(double a, double b) { return render(a + b); }
55  	}
56  
57  	/** {@code #{subtract(a, b)}} — returns {@code a - b}. */
58  	public static class Subtract extends TypedFunction {
59  		@Override public String name() { return "subtract"; }
60  		public String invoke(double a, double b) { return render(a - b); }
61  	}
62  
63  	/** {@code #{multiply(a, b)}} — returns {@code a * b}. */
64  	public static class Multiply extends TypedFunction {
65  		@Override public String name() { return "multiply"; }
66  		public String invoke(double a, double b) { return render(a * b); }
67  	}
68  
69  	/** {@code #{divide(a, b)}} — returns {@code a / b}. {@code b = 0} returns {@code Infinity} or {@code NaN} per IEEE 754. */
70  	public static class Divide extends TypedFunction {
71  		@Override public String name() { return "divide"; }
72  		public String invoke(double a, double b) { return render(a / b); }
73  	}
74  
75  	/** {@code #{modulo(a, b)}} — returns {@code a % b}. */
76  	public static class Modulo extends TypedFunction {
77  		@Override public String name() { return "modulo"; }
78  		public String invoke(double a, double b) { return render(a % b); }
79  	}
80  
81  	/** {@code #{min(a, b)}} — returns the smaller of two numbers. */
82  	public static class Min extends TypedFunction {
83  		@Override public String name() { return "min"; }
84  		public String invoke(double a, double b) { return render(Math.min(a, b)); }
85  	}
86  
87  	/** {@code #{max(a, b)}} — returns the larger of two numbers. */
88  	public static class Max extends TypedFunction {
89  		@Override public String name() { return "max"; }
90  		public String invoke(double a, double b) { return render(Math.max(a, b)); }
91  	}
92  
93  	/** {@code #{abs(a)}} — returns the absolute value of {@code a}. */
94  	public static class Abs extends TypedFunction {
95  		@Override public String name() { return "abs"; }
96  		public String invoke(double a) { return render(Math.abs(a)); }
97  	}
98  }
