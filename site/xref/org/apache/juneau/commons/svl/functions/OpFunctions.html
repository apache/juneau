25  /**
26   * Reserved {@code $op.*} operator functions targeted by the compiled expression grammar.
27   *
28   * <p>
29   * Each operator is one {@link VarFunction} registered <i>only</i> under its reserved {@code $op.} name (never
30   * under a plain name).  All are deniable via an allowlist like any other function.  Arguments arrive already
31   * evaluated; short-circuiting ({@code and}, {@code or}, {@code elvis}, {@code ternary}) is the evaluator's job.
32   *
33   * <h2 class='section'>Coercion rules (SpEL-style):</h2>
34   * <ul>
35   * 	<li>Numeric tower: {@code Byte/Short/Integer/Long} stay {@code long} (overflow detected with {@code Math.*Exact}
36   * 		and promoted to {@link BigDecimal}); any {@code Double/Float} operand gives {@code double} (this wins over
37   * 		{@code BigDecimal}); otherwise any {@code BigDecimal}/{@code BigInteger} gives {@link BigDecimal}.
38   * 	<li>{@code add} concatenates when either operand is a {@link String}; a {@code null} operand renders as
39   * 		{@code "null"}.  {@code sub/mul/div/mod/neg/pos} are numeric only.
40   * 	<li>{@code div} of two integral values stays integral only when exact, otherwise {@code double}.  Integral (and
41   * 		BigDecimal) division or modulo by zero is an error; by {@code 0.0} follows IEEE 754.
42   * 	<li>{@code eq/ne} compare numbers numerically ({@code 1 == 1.0}), otherwise {@link Objects#equals}.
43   * 	<li>{@code lt/le/gt/ge} compare numbers numerically or same-class {@link Comparable}s; anything else is an
44   * 		error.  A {@code null} operand is an error in strict mode and {@code false} in lenient mode.
45   * 	<li>{@code and/or/not/ternary} accept {@link Boolean} or the strings {@code "true"}/{@code "false"}, matched
46   * 		case-insensitively (consistent with {@code BooleanFunctions}, whose parser also uses case-insensitive
47   * 		matching but additionally accepts 1/yes/on and 0/no/off, which operators deliberately do not).  {@code and}
48   * 		and {@code or} validate <i>both</i> operands before combining.
49   * 	<li>{@code Long.MIN_VALUE / -1} overflows and is promoted to {@link BigDecimal}; {@code Long.MIN_VALUE % -1} is {@code 0}.
50   * 	<li>NaN is unordered: {@code eq} is false, {@code ne} true, and all relational operators false.  {@code 0.0 == -0.0}.
51   * 		Long-versus-double comparison is exact (no precision loss).
52   * 	<li>{@code add} string concatenation renders doubles via {@link ArithmeticFunctions#render(double)} (integral
53   * 		values drop {@code .0}) and BigDecimal via {@code toPlainString()}.
54   * 	<li>{@code elvis} is null-only: {@code false}, {@code 0} and {@code ""} are kept.
55   * 	<li>{@code matches} uses a bounded shared pattern cache.  There is <b>no match timeout</b>: ReDoS exposure is
56   * 		bounded only by the configured limits and the function allowlist.
57   * </ul>
58   */
59  public final class OpFunctions {
60  
61  	private OpFunctions() {}

72  	/** Base class for operator functions. */
73  	abstract static class Op implements VarFunction {
74  		private final String name;
75  		private final int arity;
76  
77  		Op(String name, int arity) {
78  			this.name = name;
79  			this.arity = arity;
80  		}
81  
82  		@Override public String name() { return name; }
83  		@Override public int minArity() { return arity; }
84  		@Override public int maxArity() { return arity; }
85  
86  		abstract Object apply(VarResolverSession s, Object a, Object b, Object c);
87  
88  		@Override
89  		public Object invokeValue(VarResolverSession session, List<Object> args) {
90  			if (args.size() != arity)
91  				throw new VarResolverException("Operator '%s' requires %s argument(s) but got %s", name, arity, args.size());
92  			return apply(session, args.get(0), arity > 1 ? args.get(1) : null, arity > 2 ? args.get(2) : null);
93  		}
94  
95  		@Override
96  		public String invoke(VarResolverSession session, List<Object> args) {
97  			var r = invokeValue(session, args);
98  			return r == null ? "" : r.toString();
99  		}
100 	}

106 	private static String typeOf(Object o) {
107 		return o == null ? "null" : o.getClass().getSimpleName();
108 	}
109 
110 	private static boolean isLong(Object o) {
111 		return o instanceof Long || o instanceof Integer || o instanceof Short || o instanceof Byte;
112 	}
113 
114 	private static boolean isDouble(Object o) {
115 		return o instanceof Double || o instanceof Float;
116 	}
117 
118 	private static boolean isBig(Object o) {
119 		return o instanceof BigDecimal || o instanceof BigInteger;
120 	}
121 
122 	private static boolean isNum(Object o) {
123 		return isLong(o) || isDouble(o) || isBig(o);
124 	}
125 
126 	private static BigDecimal big(Object o) {
127 		if (o instanceof BigDecimal x) return x;
128 		if (o instanceof BigInteger x) return new BigDecimal(x);
129 		if (isDouble(o)) return new BigDecimal(((Number)o).doubleValue());  // exact binary expansion; callers exclude NaN/Infinity
130 		return BigDecimal.valueOf(((Number)o).longValue());
131 	}
132 
133 	private static Number num(String op, Object o) {
134 		if (!isNum(o))
135 			throw new VarResolverException("Operator '%s' requires a numeric operand but got %s", op, typeOf(o));
136 		return (Number)o;
137 	}
138 
139 	static Object add(Object a, Object b) {
140 		if (a instanceof String || b instanceof String)
141 			return str(a) + str(b);
142 		return arith("+", a, b);
143 	}
144 
145 	static Object arith(String op, Object a, Object b) {
146 		var x = num(op, a);
147 		var y = num(op, b);
148 		if (isDouble(x) || isDouble(y))
149 			return dbl(op, x.doubleValue(), y.doubleValue());
150 		if (isLong(x) && isLong(y))
151 			return lng(op, x.longValue(), y.longValue());
152 		return bd(op, big(x), big(y));
153 	}
154 
155 	private static Object dbl(String op, double x, double y) {
156 		return switch (op) {
157 			case "+" -> x + y;
158 			case "-" -> x - y;
159 			case "*" -> x * y;
160 			case "/" -> x / y;
161 			default -> x % y;
162 		};
163 	}
164 
165 	private static VarResolverException zero(String op) {
166 		return new VarResolverException("Operator '%s' %s by zero", op, "/".equals(op) ? "division" : "modulo");
167 	}
168 
169 	private static Object lng(String op, long x, long y) {
170 		try {
171 			switch (op) {
172 				case "+": return Math.addExact(x, y);
173 				case "-": return Math.subtractExact(x, y);
174 				case "*": return Math.multiplyExact(x, y);
175 				case "/":
176 					if (y == 0)
177 						throw zero(op);
178 					if (x == Long.MIN_VALUE && y == -1) throw new ArithmeticException();
179 					return x / y;
180 				default:
181 					if (y == 0)
182 						throw zero(op);
183 					return x % y;
184 			}
185 		} catch (ArithmeticException e) {
186 			return bd(op, BigDecimal.valueOf(x), BigDecimal.valueOf(y));
187 		}
188 	}
189 
190 	private static Object bd(String op, BigDecimal x, BigDecimal y) {
191 		switch (op) {
192 			case "+": return (x.add(y));
193 			case "-": return (x.subtract(y));
194 			case "*": return (x.multiply(y));
195 			case "/":
196 				if (y.signum() == 0)
197 					throw zero(op);
198 				return (x.divide(y, MathContext.DECIMAL128));
199 			default:
200 				if (y.signum() == 0)
201 					throw zero(op);
202 				return (x.remainder(y));
203 		}
204 	}
205 
206 	static Object neg(Object a) {
207 		var x = num("-", a);
208 		if (isDouble(x)) return -x.doubleValue();
209 		if (isLong(x)) {
210 			var l = x.longValue();
211 			return l == Long.MIN_VALUE ? BigDecimal.valueOf(l).negate() : (Object)(-l);
212 		}
213 		return big(x).negate();
214 	}

216 	/** Returned by {@link #numCmp} when either operand is NaN (no ordering exists). */
217 	private static final int UNORDERED = 2;
218 
219 	/**
220 	 * Numeric comparison: -1, 0, 1, or {@link #UNORDERED}.
221 	 *
222 	 * <p>
223 	 * Two doubles compare with primitive semantics (so {@code 0.0 == -0.0}).  A double against an integral or
224 	 * BigDecimal value compares exactly through {@link BigDecimal} (no long-to-double precision loss);
225 	 * infinities are ordered by sign without conversion.
226 	 */
227 	private static int numCmp(Number x, Number y) {
228 		if (isDouble(x) && isDouble(y)) {
229 			var dx = x.doubleValue();
230 			var dy = y.doubleValue();
231 			if (Double.isNaN(dx) || Double.isNaN(dy)) return UNORDERED;
232 			return dx < dy ? -1 : dx > dy ? 1 : 0;
233 		}
234 		if (isDouble(x) || isDouble(y)) {
235 			var d = (isDouble(x) ? x : y).doubleValue();
236 			if (Double.isNaN(d)) return UNORDERED;
237 			if (Double.isInfinite(d)) {
238 				var sign = d > 0 ? 1 : -1;
239 				return isDouble(x) ? sign : -sign;
240 			}
241 		}
242 		if (isLong(x) && isLong(y))
243 			return Long.compare(x.longValue(), y.longValue());
244 		return big(x).compareTo(big(y));
245 	}
246 
247 	static boolean eq(Object a, Object b) {
248 		if (isNum(a) && isNum(b))
249 			return numCmp((Number)a, (Number)b) == 0;
250 		return Objects.equals(a, b);
251 	}
252 
253 	@SuppressWarnings({"unchecked", "rawtypes"})
254 	static boolean rel(VarResolverSession s, String op, Object a, Object b) {
255 		if (a == null || b == null) {
256 			if (s != null && s.isStrict())
257 				throw new VarResolverException("Operator '%s' cannot compare null operand", op);
258 			return false;
259 		}
260 		int c;
261 		if (isNum(a) && isNum(b))
262 			c = numCmp((Number)a, (Number)b);
263 		else if (a.getClass() == b.getClass() && a instanceof Comparable x)
264 			c = x.compareTo(b);
265 		else
266 			throw new VarResolverException("Operator '%s' cannot compare %s with %s", op, typeOf(a), typeOf(b));
267 		if (c == UNORDERED)
268 			return false;
269 		return switch (op) {
270 			case "<" -> c < 0;
271 			case "<=" -> c <= 0;
272 			case ">" -> c > 0;
273 			default -> c >= 0;
274 		};
275 	}

277 	static boolean bool(String op, Object o) {
278 		if (o instanceof Boolean b) return b;
279 		if (o instanceof String s) {
280 			if ("true".equalsIgnoreCase(s)) return true;
281 			if ("false".equalsIgnoreCase(s)) return false;
282 		}
283 		throw new VarResolverException("Operator '%s' requires a boolean operand but got %s", op, typeOf(o));
284 	}

286 	/** Validates both operands (no short-circuit; that is the evaluator's job) then combines them. */
287 	static boolean both(String op, Object a, Object b, boolean and) {
288 		var x = bool(op, a);
289 		var y = bool(op, b);
290 		return and ? x && y : x || y;
291 	}

293 	/** Scalar string form used by {@code add} concatenation; shares {@link ArithmeticFunctions#render(double)}. */
294 	static String str(Object o) {
295 		if (isDouble(o)) return ArithmeticFunctions.render(((Number)o).doubleValue());
296 		if (o instanceof BigDecimal x) return x.toPlainString();
297 		return String.valueOf(o);
298 	}
299 
300 	static boolean matches(Object s, Object regex) {
301 		if (!(s instanceof CharSequence cs))
302 			throw new VarResolverException("Operator 'matches' requires a string subject but got %s", typeOf(s));
303 		if (!(regex instanceof String r))
304 			throw new VarResolverException("Operator 'matches' requires a string regex but got %s", typeOf(regex));
305 		try {
306 			return RegexFunctions.PATTERN_CACHE.get(r).matcher(cs).matches();
307 		} catch (PatternSyntaxException e) {
308 			throw new VarResolverException(e, "Operator 'matches' has an invalid regex '%s': %s", r, e.getDescription());
309 		}
310 	}
