17  package org.apache.juneau.commons.svl;
18  
19  import java.io.*;
20  
21  /**
22   * Compiled element of a {@link VarTemplate}.
23   *
24   * <p>
25   * Three concrete subclasses cover the three top-level token types produced by the unified
26   * tokenizer in {@link VarTemplateCompiler}:
27   * <ul>
28   * 	<li>{@link LiteralSegment} — fixed text appended verbatim.
29   * 	<li>{@link VarRefSegment} — a {@code ${...}} or {@code $X{...}} reference; resolves the
30   * 		captured {@link Var} (cached at compile time) against the session.
31   * 	<li>{@link ScriptSegment} — a {@code #{name(args...)}} function call. Function dispatch
32   * 		is wired by {@link VarTemplate}; bare segments are constructed by the compiler.
33   * </ul>
34   *
35   * <p>
36   * Segments are immutable and threadsafe. The compile-binding contract documented on
37   * {@link VarTemplate} applies to the captured {@link Var} references in {@link VarRefSegment}.
38   */
39  abstract sealed class TemplateSegment permits LiteralSegment, VarRefSegment, ScriptSegment {
40  
41  	/**
42  	 * Resolves this segment and appends the result to {@code out}.
43  	 *
44  	 * <p>
45  	 * Subclasses implement either this method or {@link #resolveTo(VarResolverSession, Writer)}
46  	 * — the default implementation of the other delegates here. Streamed-Var-bearing segments
47  	 * (only {@link VarRefSegment} for now) override the writer-based path to preserve the
48  	 * direct-streaming optimization that pre-existing in-place tokenization relied on.
49  	 *
50  	 * @param session The session to resolve against. Must not be {@code null}.
51  	 * @param out The buffer to append to. Must not be {@code null}.
52  	 */
53  	abstract void resolve(VarResolverSession session, StringBuilder out);
54  
55  	/**
56  	 * Resolves this segment and writes the result directly to {@code w}.
57  	 *
58  	 * <p>
59  	 * Default implementation buffers via {@link #resolve(VarResolverSession, StringBuilder)};
60  	 * segments that wrap a streamed {@link Var} should override to write directly.
61  	 *
62  	 * @param session The session to resolve against.
63  	 * @param w The writer to append to.
64  	 * @throws IOException Thrown by the underlying stream.
65  	 */
66  	void resolveTo(VarResolverSession session, Writer w) throws IOException {
67  		var sb = new StringBuilder();
68  		resolve(session, sb);
69  		w.append(sb);
70  	}
71  
72  	/**
73  	 * @return {@code true} if this segment contributes only fixed text (i.e. is a
74  	 *	 {@link LiteralSegment}).
75  	 */
76  	boolean isLiteral() { return false; }
77  
78  	/**
79  	 * Helper used by {@link VarRefSegment} and {@link ScriptSegment} to wrap arbitrary
80  	 * checked exceptions raised during dispatch in a {@link VarResolverException} carrying
81  	 * the failing var/function name and the surrounding template source for diagnostics.
82  	 *
83  	 * @param e The original exception.
84  	 * @param sourceFragment A short representation of the failing token (e.g. {@code "$X{...}"}).
85  	 * @param identifier The var prefix or function name.
86  	 * @return The wrapped exception (never returns; always throws).
87  	 */
88  	static VarResolverException wrapDispatchFailure(Exception e, String identifier, String sourceFragment) {
89  		if (e instanceof VarResolverException e2)
90  			return e2;
91  		throw new VarResolverException(e, "Problem occurred resolving variable '%s' in string '%s'", identifier, sourceFragment);
92  	}
93  }
