17  package org.apache.juneau.commons.logging;
18  
19  import java.util.*;
20  import java.util.concurrent.atomic.*;
21  
22  /**
23   * Identity-keyed side table that carries a {@link LogContext} snapshot alongside a {@link java.util.logging.LogRecord}.
24   *
25   * <p>
26   * The context snapshot is attached in a side table (not a field on any {@link java.util.logging.LogRecord} type) so it is
27   * visible regardless of whether the record is a base JUL {@link java.util.logging.LogRecord} (as both debug pipelines
28   * construct) or the commons {@link LogRecord} subclass.
29   *
30   * <h5 class='section'>Identity-key invariant (load-bearing):</h5>
31   * <p>
32   * The backing {@link WeakHashMap} behaves as an identity map <b>only because</b> neither base JUL
33   * {@link java.util.logging.LogRecord} nor the commons {@link LogRecord} subclass overrides {@code equals}/{@code hashCode}
34   * (both use {@link Object} identity).  If a future change gives either value-equality, the side table silently corrupts
35   * (two distinct records comparing equal would share/overwrite one entry).  A unit test guards this by asserting both
36   * record types remain on {@link Object} equality.
37   *
38   * <h5 class='section'>Framework attach points:</h5>
39   * <p>
40   * The two {@code attachIfAbsent} overloads are called by {@link RichLogger#log(java.util.logging.LogRecord)} and by the
41   * two REST debug pipelines to pre-seed a record's correlation context.  They are <b>not</b> intended as a general
42   * application-facing mutator.
43   *
44   * <h5 class='section'>See Also:</h5><ul>
45   * 	<li class='jc'>{@link LogContext}
46   * 	<li class='jc'>{@link RichLogger#log(java.util.logging.LogRecord)}
47   * </ul>
48   */
49  public final class LogRecordContext {
50  
51  	private static final Map<java.util.logging.LogRecord,Map<String,Object>> TABLE =
52  		Collections.synchronizedMap(new WeakHashMap<>());
53  
54  	/** White-box counter of side-table put attempts; incremented only when a non-empty map reaches the table. */
55  	private static final AtomicLong PUT_COUNT = new AtomicLong();
56  
57  	private LogRecordContext() {}
58  
59  	/**
60  	 * Returns the context snapshot attached to the specified record.
61  	 *
62  	 * @param rec The record.  Must not be <jk>null</jk>.
63  	 * @return The attached snapshot, or the shared empty {@link Map#of()} singleton if none is attached.  Never
64  	 * 	<jk>null</jk>.
65  	 */
66  	public static Map<String,Object> of(java.util.logging.LogRecord rec) {
67  		var m = TABLE.get(rec);
68  		return m == null ? Map.of() : m;
69  	}
70  
71  	/**
72  	 * Attaches the current thread's live {@link LogContext} snapshot to the specified record, if no snapshot is already
73  	 * attached.
74  	 *
75  	 * <p>
76  	 * Framework attach point (called as the first statement of {@link RichLogger#log(java.util.logging.LogRecord)}).  When
77  	 * the live context is empty this returns immediately, before touching the synchronized side table, so callers who
78  	 * never use {@link LogContext} pay only a thread-local read and an {@code isEmpty()} check &mdash; never a lock.
79  	 *
80  	 * @param rec The record to attach to.  Must not be <jk>null</jk>.
81  	 */
82  	public static void attachIfAbsent(java.util.logging.LogRecord rec) {
83  		attachIfAbsent(rec, LogContext.INSTANCE.snapshot());
84  	}
85  
86  	/**
87  	 * Attaches the supplied context snapshot to the specified record, if no snapshot is already attached.
88  	 *
89  	 * <p>
90  	 * Framework attach point used by the debug pipelines to pre-seed a record from a snapshot carried across an async
91  	 * boundary.  When the supplied map is empty this returns immediately, before touching the synchronized side table
92  	 * &mdash; symmetric with the one-arg overload &mdash; so a later one-arg call from an empty emitting thread cannot
93  	 * clobber a previously pre-seeded snapshot.
94  	 *
95  	 * @param rec The record to attach to.  Must not be <jk>null</jk>.
96  	 * @param ctx The context snapshot to attach.  Must not be <jk>null</jk>.
97  	 */
98  	public static void attachIfAbsent(java.util.logging.LogRecord rec, Map<String,Object> ctx) {
99  		if (ctx.isEmpty())
100 			return;
101 		PUT_COUNT.incrementAndGet();
102 		TABLE.putIfAbsent(rec, ctx);
103 	}
104 
105 	/**
106 	 * White-box accessor: the number of live entries in the side table.  For tests only.
107 	 *
108 	 * @return The current side-table size.
109 	 */
110 	static int tableSize() {
111 		return TABLE.size();
112 	}
113 
114 	/**
115 	 * White-box accessor: the number of times a non-empty context reached the synchronized side table.  For tests only.
116 	 *
117 	 * @return The cumulative put-attempt count.
118 	 */
119 	static long putCount() {
120 		return PUT_COUNT.get();
121 	}
122 }
