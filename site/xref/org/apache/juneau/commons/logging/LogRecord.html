30  /**
31   * LogRecord that supports lazy message formatting.
32   *
33   * <p>
34   * This class extends {@link java.util.logging.LogRecord java.util.logging.LogRecord} and overrides {@link #getMessage()} to use
35   * {@link org.apache.juneau.commons.utils.StringUtils#format(String, Object...)} (printf-style <js>"%s"</js>/<js>"%d"</js>) for
36   * lazy formatting. The message is only formatted when {@link #getMessage()} is actually called.
37   *
38   * <h5 class='section'>Usage:</h5>
39   * <p>
40   * This class is used internally by {@link Logger} when logging formatted messages.
41   * The formatted message is only computed when the LogRecord's message is actually
42   * accessed (e.g., by a Handler or Formatter).
43   * </p>
44   *
45   * <h5 class='section'>Formatting contract:</h5>
46   * <p>
47   * Juneau {@link Logger} patterns are <b>printf-style</b> (rendered here via {@link org.apache.juneau.commons.utils.StringUtils#format(String, Object...)}).
48   * Because this override returns the already-substituted message, downstream {@link java.util.logging.Handler Handlers} and
49   * {@link java.util.logging.Formatter Formatters} that call {@link #getMessage()} see fully-rendered text and never re-apply
50   * formatting — so the printf-vs-MessageFormat choice is purely internal. The JDK's {@link java.text.MessageFormat}
51   * parameter contract of {@link java.util.logging.LogRecord} only resurfaces if a record is serialized and consumed as a
52   * plain {@link java.util.logging.LogRecord} by a non-Juneau handler; such a consumer would apply MessageFormat to the raw
53   * <c>{0}</c> pattern, so printf patterns are not intended for that scenario.
54   * </p>
55   *
56   * <h5 class='section'>See Also:</h5><ul>
57   * 	<li class='jc'>{@link Logger}
58   * 	<li class='jc'>{@link org.apache.juneau.commons.utils.StringUtils#format(String, Object...)}
59   * 	<li class='jc'>{@link java.util.logging.LogRecord}
60   * </ul>
61   */
62  @SuppressWarnings({
63  	"java:S115",  // Constants use UPPER_snakeCase convention
64  	"java:S100",  // Method names match java.util.logging.LogRecord for API compatibility
65  	"java:S1192", // String literals intentionally duplicated for clarity
66  	"java:S2176", // Class name intentionally matches java.util.logging.LogRecord; extends it to enrich logging with Juneau-specific fields
67  	"java:S2143"  // Date required for java.util.logging compatibility and Formatter %t conversions; not a behavior-preserving java.time swap.
68  })

73  	// Key constants for format placeholders
74  	private static final String KEY_date = "date";
75  	private static final String KEY_logger = "logger";
76  	private static final String KEY_level = "level";
77  	private static final String KEY_msg = "msg";
78  	private static final String KEY_thrown = "thrown";
79  	private static final String KEY_timestamp = "timestamp";
80  	private static final String KEY_thread = "thread";
81  	private static final String KEY_threadid = "threadid";
82  	private static final String KEY_exception = "exception";

84  	/**
85  	 * Constructor.
86  	 *
87  	 * @param loggerName The logger name.  Can be <jk>null</jk>.
88  	 * @param level The log level.  Must not be <jk>null</jk>.
89  	 * @param msg The message pattern (will be used as-is if args is null or empty).
90  	 * @param parameters The format arguments, or <jk>null</jk> if none.
91  	 * @param throwable The throwable, or <jk>null</jk> if none.
92  	 */
93  	public LogRecord(String loggerName, Level level, String msg, Object[] parameters, Throwable throwable) {
94  		super(level, msg);
95  		setLoggerName(loggerName);
96  		setParameters(isNotEmptyArray(parameters) ? parameters : null);
97  		setThrown(throwable);
98  	}
99  
100 	/**
101 	 * Returns the log message, formatting it lazily if needed.
102 	 *
103 	 * <p>
104 	 * If parameters were set via {@link #setParameters(Object[])}, the message is formatted
105 	 * using {@link org.apache.juneau.commons.utils.StringUtils#format(String, Object...)} on first access.
106 	 *
107 	 * @return The log message, formatted if parameters are present.
108 	 */
109 	@Override
110 	public String getMessage() {
111 		var m = super.getMessage();
112 		var p = getParameters();
113 		return isEmptyArray(p) ? m : f(m, p);
114 	}

116 	/**
117 	 * Formats this log record as a string using the specified format pattern.
118 	 *
119 	 * <p>
120 	 * Similar to how {@link SimpleFormatter} formats log records, this method
121 	 * allows you to specify a custom format string with placeholders that are replaced with
122 	 * actual values from the log record.
123 	 *
124 	 * <p>
125 	 * The format string supports both named placeholders (e.g., <js>"{date}"</js>) and
126 	 * {@link java.util.Formatter Formatter}-style format specifiers (e.g., <js>"%1$tc"</js>).
127 	 * Named placeholders are replaced first, then the resulting string is formatted using
128 	 * {@link java.util.Formatter Formatter} with the following arguments:
129 	 * <ol>
130 	 * 	<li><c>date</c> - The date/time as a {@link Date} object
131 	 * 	<li><c>logger</c> - The logger name
132 	 * 	<li><c>level</c> - The log level
133 	 * 	<li><c>message</c> - The formatted log message
134 	 * 	<li><c>thrown</c> - The throwable object (if any)
135 	 * </ol>
136 	 *
137 	 * <h5 class='section'>Named Placeholders:</h5>
138 	 * <ul>
139 	 * 	<li><js>"{date}"</js> - The date/time formatted using {@link SimpleFormatter SimpleFormatter}'s default date format
140 	 * 	<li><js>"{timestamp}"</js> - The date/time formatted as ISO-8601 (yyyy-MM-dd'T'HH:mm:ss.SSSZ)
141 	 * 	<li><js>"{logger}"</js> - The logger name
142 	 * 	<li><js>"{level}"</js> - The log level name
143 	 * 	<li><js>"{msg}"</js> - The log message (formatted if parameters are present)
144 	 * 	<li><js>"{thread}"</js> - The thread ID
145 	 * 	<li><js>"{threadid}"</js> - The thread ID (alias for {thread})
146 	 * 	<li><js>"{exception}"</js> - The exception message (if thrown)
147 	 * 	<li><js>"{thrown}"</js> - The throwable object (if thrown)
148 	 * </ul>
149 	 *
150 	 * <h5 class='section'>Formatter-Style Format Specifiers:</h5>
151 	 * <p>
152 	 * You can use {@link java.util.Formatter Formatter}-style format specifiers directly in the format string.
153 	 * The arguments are passed in the same order as {@link java.util.logging.SimpleFormatter SimpleFormatter}:
154 	 * </p>
155 	 * <ul>
156 	 * 	<li><js>"%1$s"</js> or <js>"%1$tc"</js> - The date/time (argument 1)
157 	 * 	<li><js>"%2$s"</js> - The logger name (argument 2)
158 	 * 	<li><js>"%3$s"</js> - The log level (argument 3)
159 	 * 	<li><js>"%4$s"</js> - The log message (argument 4)
160 	 * 	<li><js>"%5$s"</js> - The throwable and its backtrace (argument 5)
161 	 * </ul>
162 	 *
163 	 * <p>
164 	 * For date/time formatting, you can use date/time conversion characters with argument 1:
165 	 * </p>
166 	 * <ul>
167 	 * 	<li><js>"%1$tc"</js> - Complete date and time (e.g., "Tue Mar 22 13:11:31 PDT 2011")
168 	 * 	<li><js>"%1$tb"</js> - Month abbreviation (e.g., "Mar")
169 	 * 	<li><js>"%1$td"</js> - Day of month (01-31)
170 	 * 	<li><js>"%1$tY"</js> - Year (4 digits)
171 	 * 	<li><js>"%1$tH"</js> - Hour (00-23)
172 	 * 	<li><js>"%1$tM"</js> - Minute (00-59)
173 	 * 	<li><js>"%1$tS"</js> - Second (00-59)
174 	 * 	<li><js>"%1$tL"</js> - Milliseconds (000-999)
175 	 * 	<li><js>"%1$tN"</js> - Nanoseconds (000000000-999999999, since JDK 9)
176 	 * 	<li><js>"%1$Tp"</js> - AM/PM marker (uppercase)
177 	 * 	<li><js>"%1$tp"</js> - AM/PM marker (lowercase)
178 	 * </ul>
179 	 *
180 	 * <p>
181 	 * The format string can also contain <js>"%n"</js> for newlines.
182 	 *
183 	 * <h5 class='section'>Examples:</h5>
184 	 * <p class='bjava'>
185 	 * 	LogRecord <jv>record</jv> = <jk>new</jk> LogRecord(Level.INFO, <js>"User {0} logged in"</js>, <js>"John"</js>);
186 	 *
187 	 * 	<jc>// Using named placeholders</jc>
188 	 * 	String <jv>formatted1</jv> = <jv>record</jv>.formatted(<js>"[{timestamp} {level}] {msg}"</js>);
189 	 * 	<jc>// Result: "[2025-01-22T07:42:45.123-0500 INFO] User John logged in"</jc>
190 	 *
191 	 * 	<jc>// Using Formatter-style specifiers</jc>
192 	 * 	String <jv>formatted2</jv> = <jv>record</jv>.formatted(<js>"%3$s: %4$s [%1$tc]%n"</js>);
193 	 * 	<jc>// Result: "INFO: User John logged in [Tue Jan 22 07:42:45 PST 2025]"</jc>
194 	 *
195 	 * 	<jc>// Mixed named placeholders and Formatter specifiers</jc>
196 	 * 	String <jv>formatted3</jv> = <jv>record</jv>.formatted(<js>"%1$tb %1$td, %1$tY {level}: {msg}%n"</js>);
197 	 * 	<jc>// Result: "Jan 22, 2025 INFO: User John logged in"</jc>
198 	 * </p>
199 	 *
200 	 * @param format The format string with placeholders and/or Formatter-style format specifiers.
201 	 * @return The formatted string.
202 	 * @see SimpleFormatter
203 	 * @see Formatter
204 	 */
205 	@SuppressWarnings({
206 		"deprecation" // Date constructor is deprecated but needed for compatibility
207 	})
208 	public String formatted(String format) {
209 		var date = new Date(getMillis());
210 
211 		Function<String,Object> resolver = key -> switch (key) {
212 			case KEY_date -> "%1$s";
213 			case KEY_logger -> "%2$s";
214 			case KEY_level -> "%3$s";
215 			case KEY_msg -> "%4$s";
216 			case KEY_thrown -> "%5$s";
217 			case KEY_timestamp -> new SimpleDateFormat("yyyy-MM-dd'T'HH:mm:ss.SSSZ").format(date);
218 			case KEY_thread, KEY_threadid -> s(getThreadID());
219 			case KEY_exception -> o(getThrown()).map(x -> x.getMessage()).orElse("");
220 			default -> "";
221 		};
222 
223 		return safeOptCatch(()->f(formatNamed(format, resolver), date, getLoggerName(), getLevel(), getMessage(), getThrown()), x -> x.getLocalizedMessage()).orElse(null);
224 	}
