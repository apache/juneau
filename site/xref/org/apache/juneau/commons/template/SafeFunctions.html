17  package org.apache.juneau.commons.template;
18  
19  import java.util.*;
20  
21  /**
22   * The default function allowlist of a {@link TemplateResolver}.
23   *
24   * <p>
25   * Pure, bounded string and collection helpers, the structure functions and the operators.  Deliberately left out:
26   * <ul>
27   * 	<li>Functions that read arbitrary data (<c>jsonPath</c>, <c>get</c>, <c>extract</c>, <c>keys</c>, <c>values</c>).
28   * 	<li>Regex functions (<c>match</c>, <c>replaceRegex</c>), which can backtrack catastrophically on author-chosen patterns.
29   * 	<li>Non-deterministic functions (<c>rand*</c>, <c>uuid</c>, <c>now</c>) and date parsing.
30   * 	<li>Functions that can allocate large results from small input (<c>repeat</c>) or that reach the platform
31   * 		(<c>switch</c>, the encoding functions).
32   * </ul>
33   * {@link TemplateResolver.Builder#allowFunctions(String...)} replaces this list.
34   */
35  public final class SafeFunctions {
36  
37  	private static final List<String> PATTERNS = List.of(
38  		"upper", "lower", "trim", "len", "size", "substring", "split", "join", "concat", "replace",
39  		"startsWith", "endsWith", "contains", "if", "coalesce", "notEmpty",
40  		"ul", "ol", "block", "table", "list", "take",
41  		"$op.*"
42  	);
43  
44  	private SafeFunctions() {}
45  
46  	/**
47  	 * Returns the default allowlist patterns.
48  	 *
49  	 * @return An unmodifiable list of glob patterns, as accepted by
50  	 * 	{@link org.apache.juneau.commons.svl.VarResolver.Builder#allowFunctions(String...)}.
51  	 */
52  	public static List<String> patterns() { return PATTERNS; }
53  }
