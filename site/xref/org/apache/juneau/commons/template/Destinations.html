25  /**
26   * Destination presets for {@link TemplateResolver.Builder#destination(Destination)}.
27   *
28   * <p>
29   * Each preset pairs a value encoder (applied to resolved values, never to template text) with a {@link StructureRenderer}
30   * that encodes every item and cell itself.  Only generic formats are offered; define a {@link Destination} for anything
31   * else.
32   *
33   * <ul>
34   * 	<li>{@link #HTML}: for text nodes and quoted attribute values only; it does not make a URL attribute safe (use
35   * 		{@link #URL_COMPONENT} and a scheme allowlist for <c>href</c>/<c>src</c>).  Escapes <c>&amp; &lt; &gt; " '</c>; lists, blocks and tables become <c>ul</c>, <c>ol</c>,
36   * 		<c>blockquote</c> and <c>table</c> elements.
37   * 	<li>{@link #MARKDOWN}: backslash-escapes characters Markdown would interpret and flattens whitespace so a value
38   * 		cannot start a new block; lists, quotes and pipe tables.  Bare URLs are not auto-linked (the <c>:</c> of
39   * 		<c>://</c> and every <c>.</c> are escaped).
40   * 	<li>{@link #PLAIN_TEXT}: drops control characters other than newline and tab; <c>- </c>, <c>N. </c> and <c>&gt; </c>
41   * 		markers and column-aligned tables.
42   * 	<li>{@link #JSON_STRING}: escapes for the inside of a JSON string literal.
43   * 	<li>{@link #URL_COMPONENT}: percent-encodes for a URL path segment or query component.
44   * </ul>
45   */
46  public final class Destinations {
47  
48  	/** HTML. */
49  	public static final Destination HTML = new Destination("html", Destinations::htmlEncode, Destinations::renderHtml);
50  
51  	/** Markdown. */
52  	public static final Destination MARKDOWN = new Destination("markdown", Destinations::markdownEncode, Destinations::renderMarkdown);
53  
54  	/** Plain text. */
55  	public static final Destination PLAIN_TEXT = new Destination("plain-text", Destinations::plainEncode, Destinations::renderPlain);
56  
57  	/** The inside of a JSON string literal (no surrounding quotes). */
58  	public static final Destination JSON_STRING = new Destination("json-string", Destinations::jsonEncode, (s, t) -> jsonEncode(renderPlain(s, t)));
59  
60  	/** A URL path segment or query component. */
61  	public static final Destination URL_COMPONENT = new Destination("url-component", Destinations::urlEncode, (s, t) -> urlEncode(renderPlain(s, t)));

63  	private static final EncodingFunctions.HtmlEscape HTML_ESCAPE = new EncodingFunctions.HtmlEscape();
64  	private static final EncodingFunctions.UrlEncode URL_ENCODE = new EncodingFunctions.UrlEncode();

72  	private static String htmlEncode(String s) {
73  		return HTML_ESCAPE.invoke(s);
74  	}
75  
76  	private static String urlEncode(String s) {
77  		// URLEncoder writes a space as '+' (and a literal '+' as %2B); a component wants %20.
78  		return URL_ENCODE.invoke(s).replace("+", "%20");
79  	}

81  	private static String plainEncode(String s) {
82  		var sb = new StringBuilder(s.length());
83  		for (var i = 0; i < s.length(); i++) {
84  			var c = s.charAt(i);
85  			if (c == '\n' || c == '\t' || (c >= 0x20 && c != 0x7f))
86  				sb.append(c);
87  		}
88  		return sb.toString();
89  	}
90  
91  	private static String jsonEncode(String s) {
92  		var sb = new StringBuilder(s.length() + 8);
93  		for (var i = 0; i < s.length(); i++) {
94  			var c = s.charAt(i);
95  			switch (c) {
96  				case '"' -> sb.append("\\\"");
97  				case '\\' -> sb.append("\\\\");
98  				case '\n' -> sb.append("\\n");
99  				case '\r' -> sb.append("\\r");
100 				case '\t' -> sb.append("\\t");
101 				case '\b' -> sb.append("\\b");
102 				case '\f' -> sb.append("\\f");
103 				default -> {
104 					if (c < 0x20 || c == 0x7f || c == ' ' || c == ' ')
105 						sb.append(String.format("\\u%04x", (int)c));
106 					else
107 						sb.append(c);
108 				}
109 			}
110 		}
111 		return sb.toString();
112 	}

114 	/**
115 	 * Escapes text for Markdown.
116 	 *
117 	 * <p>
118 	 * Whitespace runs (newlines included) collapse to one space and other control characters are dropped, so a value cannot
119 	 * start a new block or an indented code block.  The characters <c>\ ` * _ [ ] &lt; &gt; | ~ &amp;</c> are always
120 	 * backslash-escaped, as are <c>{ } ( ) ! .</c> and the colon of <c>://</c> (so a bare URL or address is not
121 	 * auto-linked and kramdown/MDX attribute braces are inert).  The first visible character is also escaped when it
122 	 * could open a block: <c># + - =</c>, and a digit run followed by <c>)</c>.
123 	 */
124 	private static String markdownEncode(String s) {
125 		var sb = new StringBuilder(s.length() + 8);
126 		var space = false;
127 		for (var i = 0; i < s.length(); i++) {
128 			var c = s.charAt(i);
129 			if (Character.isWhitespace(c)) {
130 				space = sb.length() > 0 || space;
131 				continue;
132 			}
133 			if (c < 0x20 || c == 0x7f)
134 				continue;
135 			if (space) {
136 				sb.append(' ');
137 				space = false;
138 			}
139 			sb.append(c);
140 		}
141 		if (space && sb.length() > 0)
142 			sb.append(' ');
143 		var t = sb.toString();
144 		var out = new StringBuilder(t.length() + 8);
145 		var lead = firstNonSpace(t);
146 		for (var i = 0; i < t.length(); i++) {
147 			var c = t.charAt(i);
148 			if ("\\`*_[]<>|~&{}()!.".indexOf(c) >= 0 || (i == lead && "#+-=".indexOf(c) >= 0)
149 				|| (c == ':' && t.startsWith("//", i + 1)))
150 				out.append('\\');
151 			if (i == lead && c >= '0' && c <= '9') {
152 				var j = i;
153 				while (j < t.length() && t.charAt(j) >= '0' && t.charAt(j) <= '9')
154 					j++;
155 				if (j < t.length() && t.charAt(j) == ')') {
156 					out.append(t, i, j).append("\\)");
157 					i = j;
158 					continue;
159 				}
160 			}
161 			out.append(c);
162 		}
163 		return out.toString();
164 	}
165 
166 	private static int firstNonSpace(String t) {
167 		var i = 0;
168 		while (i < t.length() && t.charAt(i) == ' ')
169 			i++;
170 		return i;
171 	}

177 	private static String renderHtml(Structure s, Function<Object,String> toText) {
178 		var sb = new StringBuilder();
179 		switch (s.kind()) {
180 			case UL, OL -> {
181 				var tag = s.kind() == Structure.Kind.UL ? "ul" : "ol";
182 				sb.append('<').append(tag).append('>');
183 				for (var o : s.items())
184 					sb.append("<li>").append(htmlItem(o, toText)).append("</li>");
185 				sb.append("</").append(tag).append('>');
186 			}
187 			case BLOCK -> sb.append("<blockquote>").append(htmlItem(s.items().get(0), toText)).append("</blockquote>");
188 			case TABLE -> {
189 				sb.append("<table><thead><tr>");
190 				for (var c : s.columns())
191 					sb.append("<th>").append(htmlEncode(c)).append("</th>");
192 				sb.append("</tr></thead><tbody>");
193 				for (var r : s.rows()) {
194 					sb.append("<tr>");
195 					for (var o : r)
196 						sb.append("<td>").append(htmlItem(o, toText)).append("</td>");
197 					sb.append("</tr>");
198 				}
199 				sb.append("</tbody></table>");
200 			}
201 		}
202 		return sb.toString();
203 	}
204 
205 	private static String htmlItem(Object o, Function<Object,String> toText) {
206 		return o instanceof Structure n ? renderHtml(n, toText) : htmlEncode(toText.apply(o));
207 	}
208 
209 	private static String renderMarkdown(Structure s, Function<Object,String> toText) {
210 		var sb = new StringBuilder();
211 		switch (s.kind()) {
212 			case UL, OL -> {
213 				var n = 1;
214 				for (var o : s.items()) {
215 					var marker = s.kind() == Structure.Kind.UL ? "- " : (n++) + ". ";
216 					line(sb, marker + indentRest(mdItem(o, toText), marker.length()));
217 				}
218 			}
219 			case BLOCK -> {
220 				for (var l : mdItem(s.items().get(0), toText).split("\n", -1))
221 					line(sb, "> " + l);
222 			}
223 			case TABLE -> {
224 				var cols = new StringJoiner(" | ", "| ", " |");
225 				var rule = new StringJoiner(" | ", "| ", " |");
226 				for (var c : s.columns()) {
227 					cols.add(markdownEncode(c));
228 					rule.add("---");
229 				}
230 				line(sb, cols.toString());
231 				line(sb, rule.toString());
232 				for (var r : s.rows()) {
233 					var row = new StringJoiner(" | ", "| ", " |");
234 					for (var o : r)
235 						row.add(mdCell(o, toText));
236 					line(sb, row.toString());
237 				}
238 			}
239 		}
240 		return sb.toString();
241 	}
242 
243 	private static String mdItem(Object o, Function<Object,String> toText) {
244 		return o instanceof Structure n ? renderMarkdown(n, toText) : markdownEncode(toText.apply(o));
245 	}
246 
247 	private static String mdCell(Object o, Function<Object,String> toText) {
248 		// A nested structure cannot span lines inside a pipe table: flatten it to one line.
249 		return o instanceof Structure n ? markdownEncode(renderPlain(n, toText)) : markdownEncode(toText.apply(o));
250 	}
251 
252 	private static String indentRest(String text, int indent) {
253 		return text.replace("\n", "\n" + " ".repeat(indent));
254 	}
255 
256 	private static String renderPlain(Structure s, Function<Object,String> toText) {
257 		var sb = new StringBuilder();
258 		switch (s.kind()) {
259 			case UL, OL -> {
260 				var n = 1;
261 				for (var o : s.items()) {
262 					var marker = s.kind() == Structure.Kind.UL ? "- " : (n++) + ". ";
263 					line(sb, marker + indentRest(plainItem(o, toText), marker.length()));
264 				}
265 			}
266 			case BLOCK -> {
267 				for (var l : plainItem(s.items().get(0), toText).split("\n", -1))
268 					line(sb, "> " + l);
269 			}
270 			case TABLE -> {
271 				var grid = new ArrayList<List<String>>();
272 				var head = new ArrayList<String>();
273 				for (var c : s.columns())
274 					head.add(cell(c));
275 				grid.add(head);
276 				for (var r : s.rows()) {
277 					var row = new ArrayList<String>();
278 					for (var o : r)
279 						row.add(o instanceof Structure n ? cell(renderPlain(n, toText)) : cell(toText.apply(o)));
280 					grid.add(row);
281 				}
282 				var widths = new int[head.size()];
283 				for (var row : grid)
284 					for (var i = 0; i < row.size(); i++)
285 						widths[i] = Math.max(widths[i], row.get(i).length());
286 				for (var r = 0; r < grid.size(); r++) {
287 					line(sb, padRow(grid.get(r), widths));
288 					if (r == 0) {
289 						var rule = new ArrayList<String>();
290 						for (var w : widths)
291 							rule.add("-".repeat(Math.max(3, w)));
292 						line(sb, padRow(rule, widths));
293 					}
294 				}
295 			}
296 		}
297 		return sb.toString();
298 	}
299 
300 	private static String plainItem(Object o, Function<Object,String> toText) {
301 		return o instanceof Structure n ? renderPlain(n, toText) : plainEncode(toText.apply(o));
302 	}
303 
304 	private static String cell(String s) {
305 		return plainEncode(s).replace('\n', ' ').replace('\t', ' ');
306 	}
307 
308 	private static String padRow(List<String> row, int[] widths) {
309 		var sb = new StringBuilder();
310 		for (var i = 0; i < row.size(); i++) {
311 			if (i > 0)
312 				sb.append("  ");
313 			sb.append(row.get(i));
314 			if (i < row.size() - 1)
315 				sb.append(" ".repeat(Math.max(0, widths[i] - row.get(i).length())));
316 		}
317 		return sb.toString();
318 	}
319 
320 	private static void line(StringBuilder sb, String s) {
321 		if (sb.length() > 0)
322 			sb.append('\n');
323 		sb.append(s);
324 	}
