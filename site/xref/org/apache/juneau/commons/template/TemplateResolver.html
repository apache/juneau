26  /**
27   * Renders SVL templates written by authors who are less trusted than the application.
28   *
29   * <p>
30   * A template resolver is a locked-down {@link VarResolver}: it registers no system vars, permits only the functions in
31   * {@link SafeFunctions}, and reads data only from the map passed to each render.  The built-in functions give lists, tables
32   * and per-row projections; see the package documentation for the grammar and the encoding rule.
33   *
34   * <p>
35   * Inside a projection over a list of maps, <c>size</c> and <c>empty</c> resolve to the row map's own size and emptiness
36   * even when the row has no such key, so they do not fall back to a scope entry of the same name.
37   *
38   * <h2 class='section'>Example:</h2>
39   * <p class='bjava'>
40   * 	<jk>var</jk> <jv>resolver</jv> = TemplateResolver.<jsm>create</jsm>().destination(Destinations.<jsf>HTML</jsf>).build();
41   * 	<jk>var</jk> <jv>template</jv> = <jv>resolver</jv>.parse(<js>"Failed: ${n}\n#{ul(failed.![id + ' - ' + msg])}"</js>);
42   * 	<jk>var</jk> <jv>html</jv> = <jv>resolver</jv>.render(<jv>template</jv>, Map.<jsm>of</jsm>(<js>"n"</js>, 2, <js>"failed"</js>, <jv>rows</jv>));
43   * </p>
44   *
45   * <h2 class='section'>Notes:</h2><ul>
46   * 	<li class='note'>Inside a projection (<c>.![...]</c>, <c>.?[...]</c>) row fields are bare identifiers.  A
47   * 		<c>${id}</c> there reads the render scope, not the row.
48   * 	<li class='note'>Functions added with {@link Builder#functions(VarFunction...)} are subject to the allowlist; add them
49   * 		with {@link Builder#allowFunctions(String...)}.
50   * 	<li class='note'>Instances are immutable and thread-safe.  {@link ParsedTemplate} instances are too.
51   * </ul>
52   */
53  public final class TemplateResolver {
54  
55  	/**
56  	 * A resolver with the default configuration: lenient, no destination.
57  	 *
58  	 * <p>
59  	 * It does <b>no encoding</b> and renders structures with {@link Structure#toString()}.  Resolved values reach the
60  	 * output unescaped, so it is not for HTML, Markdown, JSON or URLs; build a resolver with a {@link Destinations}
61  	 * preset (or an explicit encoder) for those.
62  	 */
63  	public static final TemplateResolver DEFAULT = create().build();

72  	/** Builder for {@link TemplateResolver}. */
73  	public static final class Builder {
74  		private final List<Var> vars = new ArrayList<>();
75  		private final List<VarFunction> functions = new ArrayList<>();
76  		private final List<String> extra = new ArrayList<>();
77  		private final List<String> only = new ArrayList<>();
78  		private boolean replaced;
79  		private boolean strict;
80  		private int maxCollectionSize = VarResolver.DEFAULT_MAX_COLLECTION_SIZE;
81  		private StructureRenderer renderer;
82  		private Destination destination;
83  
84  		private Builder() {}
85  
86  		/**
87  		 * Registers additional vars.  None are registered by default except the internal scope var.
88  		 *
89  		 * @param values The vars.  Must not be <jk>null</jk>.  A var named <c>P</c> replaces the scope var, which rebinds
90  		 * 	the <c>${name}</c> shortcut.
91  		 * @return This object.
92  		 */
93  		public Builder vars(Var...values) {
94  			for (var v : values)
95  				vars.add(Objects.requireNonNull(v, "var must not be null"));
96  			return this;
97  		}
98  
99  		/**
100 		 * Registers additional functions.
101 		 *
102 		 * @param values The functions.  Must not be <jk>null</jk>.  They are also subject to the allowlist.
103 		 * @return This object.
104 		 */
105 		public Builder functions(VarFunction...values) {
106 			for (var f : values)
107 				functions.add(Objects.requireNonNull(f, "function must not be null"));
108 			return this;
109 		}
110 
111 		/**
112 		 * Adds function-name globs to the default allowlist ({@link SafeFunctions#patterns()}).
113 		 *
114 		 * <p>
115 		 * Repeated calls are additive.  A bare <c>*</c> is rejected because it would switch default-deny off; use
116 		 * {@link #allowOnlyFunctions(String...)} to replace the defaults.
117 		 *
118 		 * @param patterns The patterns.  Must not be <jk>null</jk>, and must not be <c>*</c>.
119 		 * @return This object.
120 		 * @throws IllegalArgumentException If a pattern is <c>*</c>.
121 		 */
122 		public Builder allowFunctions(String...patterns) {
123 			addPatterns(extra, patterns);
124 			return this;
125 		}
126 
127 		/**
128 		 * Replaces the default function allowlist ({@link SafeFunctions#patterns()}) with exactly these globs.
129 		 *
130 		 * <p>
131 		 * Include <c>$op.*</c> to keep the operators.  Patterns from {@link #allowFunctions(String...)} are still added.
132 		 * A bare <c>*</c> is rejected.
133 		 *
134 		 * @param patterns The patterns.  Must not be <jk>null</jk>, and must not be <c>*</c>.
135 		 * @return This object.
136 		 * @throws IllegalArgumentException If a pattern is <c>*</c>.
137 		 */
138 		public Builder allowOnlyFunctions(String...patterns) {
139 			replaced = true;
140 			addPatterns(only, patterns);
141 			return this;
142 		}
143 
144 		private static void addPatterns(List<String> to, String[] patterns) {
145 			for (var p : patterns) {
146 				Objects.requireNonNull(p, "pattern must not be null");
147 				if (p.trim().equals("*"))
148 					throw iaex("allowFunctions pattern '*' would disable default-deny");
149 				to.add(p);
150 			}
151 		}
152 
153 		/**
154 		 * Enables strict mode: an unknown or denied function, or an expression error, throws instead of being emitted
155 		 * as written.
156 		 *
157 		 * @param value Whether to enable strict mode.
158 		 * @return This object.
159 		 */
160 		public Builder strict(boolean value) {
161 			strict = value;
162 			return this;
163 		}
164 
165 		/**
166 		 * Sets the maximum number of elements one expression may iterate, and the maximum per container when formatting.
167 		 *
168 		 * @param value The limit.  Must be positive.
169 		 * @return This object.
170 		 */
171 		public Builder maxCollectionSize(int value) {
172 			if (value <= 0)
173 				throw iaex("maxCollectionSize must be positive: %s", value);
174 			maxCollectionSize = value;
175 			return this;
176 		}
177 
178 		/**
179 		 * Sets how lists, blocks and tables are rendered.  Overrides the renderer of {@link #destination(Destination)}.
180 		 *
181 		 * <p>
182 		 * Without a renderer {@link Structure#toString()} is used, and its items are not encoded.
183 		 *
184 		 * @param value The renderer.  <jk>null</jk> clears it.
185 		 * @return This object.
186 		 */
187 		public Builder structureRenderer(StructureRenderer value) {
188 			renderer = value;
189 			return this;
190 		}
191 
192 		/**
193 		 * Sets the default value encoder and structure renderer from a destination preset.
194 		 *
195 		 * @param value The destination.  <jk>null</jk> clears it.
196 		 * @return This object.
197 		 */
198 		public Builder destination(Destination value) {
199 			destination = value;
200 			return this;
201 		}
202 
203 		/**
204 		 * Builds the resolver.
205 		 *
206 		 * @return A new resolver.
207 		 */
208 		public TemplateResolver build() {
209 			var b = VarResolver.create().defaultFunctions().strict(strict).maxCollectionSize(maxCollectionSize);
210 			b.vars(new ScopeVar());
211 			b.vars(vars.toArray(new Var[0]));
212 			b.functions(functions.toArray(new VarFunction[0]));
213 			var patterns = new ArrayList<>(replaced ? only : SafeFunctions.patterns());
214 			patterns.addAll(extra);
215 			b.allowFunctions(patterns.toArray(new String[0]));
216 			var r = renderer != null ? renderer : (destination == null ? null : destination.renderer());
217 			var e = destination == null ? null : destination.encoder();
218 			return new TemplateResolver(b.build(), r, e, new JsonValueFormatter(maxCollectionSize, strict));
219 		}
220 	}

222 	private final VarResolver resolver;
223 	private final UnaryOperator<String> encoder;
224 	private final Function<Object,String> formatter;
225 
226 	private TemplateResolver(VarResolver resolver, StructureRenderer renderer, UnaryOperator<String> encoder, Function<Object,String> baseFormatter) {
227 		this.resolver = resolver;
228 		this.encoder = encoder;
229 		this.formatter = renderer == null ? null : new StructureFormatter(renderer, baseFormatter);
230 	}
231 
232 	/** Sends structures to the renderer and everything else to the base formatter. */
233 	private record StructureFormatter(StructureRenderer renderer, Function<Object,String> base) implements Function<Object,String> {
234 		@Override public String apply(Object o) {
235 			return o instanceof Structure s ? renderer.render(s, base) : base.apply(o);
236 		}
237 	}

239 	/**
240 	 * Parses a template.
241 	 *
242 	 * @param template The template text.  May be <jk>null</jk>.
243 	 * @return The parsed template.  Never <jk>null</jk>.
244 	 * @throws ExprException In strict mode, if the template is malformed or uses an unknown or denied function.
245 	 * 	In lenient mode nothing is thrown.
246 	 */
247 	public ParsedTemplate parse(String template) {
248 		return new ParsedTemplate(resolver.compile(template), this);
249 	}

251 	/**
252 	 * Renders a template using the default value encoder of the configured destination (none if there is no destination).
253 	 *
254 	 * @param template The template.  Must not be <jk>null</jk>, and must have been parsed by this resolver.
255 	 * @param scope The names the template can read.  May be <jk>null</jk> (no names).
256 	 * @return The rendered text.
257 	 * @throws IllegalArgumentException If the template was parsed by a different resolver.
258 	 */
259 	public String render(ParsedTemplate template, Map<String,?> scope) {
260 		return render(template, scope, null);
261 	}
262 
263 	/**
264 	 * Renders a template.
265 	 *
266 	 * @param template The template.  Must not be <jk>null</jk>, and must have been parsed by this resolver.
267 	 * @param scope The names the template can read.  May be <jk>null</jk> (no names).
268 	 * @param valueEncoder The value encoder.  Overrides the destination's encoder; <jk>null</jk> uses the destination's.
269 	 * @return The rendered text.
270 	 */
271 	public String render(ParsedTemplate template, Map<String,?> scope, UnaryOperator<String> valueEncoder) {
272 		Objects.requireNonNull(template, "template must not be null");
273 		if (template.owner() != this)
274 			throw iaex("template was parsed by a different TemplateResolver");
275 		var session = resolver.createSession().scope(scope == null ? n -> null : n -> n == null ? null : scope.get(n));
276 		if (formatter != null)
277 			session.valueFormatter(formatter);
278 		return template.template().resolve(session, valueEncoder != null ? valueEncoder : encoder);
279 	}

281 	/**
282 	 * Checks a template against a catalog without rendering it.  Never throws.
283 	 *
284 	 * <p>
285 	 * Reports, in source order: an identifier not in the catalog, a row field not in its list's catalog, a row field read
286 	 * through a name that is not a declared list, an unknown or denied function (unless the catalog allows them), an
287 	 * unknown var, and a marker that does not compile.  The key of a <c>${name}</c> or <c>${name:default}</c> reference is
288 	 * checked like an identifier, so <c>${secret}</c> is a problem when <c>secret</c> is not declared; a key that is
289 	 * itself computed (<c>${${x}}</c>) is read as written and is reported as unknown.  The default text after <c>:</c> is
290 	 * not checked.
291 	 *
292 	 * @param template The template text.  May be <jk>null</jk>.
293 	 * @param catalog The catalog.  Must not be <jk>null</jk>.
294 	 * @return The problems.  Empty if the template is clean.
295 	 */
296 	public List<TemplateProblem> check(String template, TemplateCatalog catalog) {
297 		Objects.requireNonNull(catalog, "catalog must not be null");
298 		return check(compileOrProblems(template), catalog);
299 	}
300 
301 	/** A compiled template, or the one problem that stopped it compiling. */
302 	private record Compiled(VarTemplate template, List<TemplateProblem> failure) {}
303 
304 	private Compiled compileOrProblems(String template) {
305 		try {
306 			return new Compiled(resolver.compile(template), null);
307 		} catch (ExprException e) {
308 			var at = Math.max(0, e.getPos());
309 			return new Compiled(null, List.of(new TemplateProblem(at, at + 1, e.getBaseMessage())));
310 		}
311 	}
312 
313 	private List<TemplateProblem> check(Compiled c, TemplateCatalog catalog) {
314 		if (c.failure() != null)
315 			return c.failure();
316 		var t = c.template();
317 		var out = new ArrayList<TemplateProblem>();
318 		for (var r : t.references())
319 			switch (r.kind()) {
320 				case SYNTAX -> out.add(new TemplateProblem(r.start(), r.end(), r.name()));
321 				case VAR -> {
322 					if (! r.bound())
323 						out.add(new TemplateProblem(r.start(), r.end(), "unknown var '" + r.name() + "'"));
324 				}
325 				case FUNCTION -> {
326 					if (! r.bound() && ! catalog.allowUnknownFunctions())
327 						out.add(new TemplateProblem(r.start(), r.end(), "unknown or denied function '" + r.name() + "'"));
328 				}
329 				case IDENT -> checkIdent(r, catalog, out);
330 			}
331 		out.sort(Comparator.comparingInt(TemplateProblem::start));
332 		return out;
333 	}
334 
335 	private static void checkIdent(TemplateReference r, TemplateCatalog catalog, List<TemplateProblem> out) {
336 		var name = r.name();
337 		var list = r.listName();
338 		if (list == null) {
339 			if (! catalog.hasName(name))
340 				out.add(new TemplateProblem(r.start(), r.end(), "unknown name '" + name + "'"));
341 		} else if (catalog.isList(list)) {
342 			var rows = catalog.rowFields(list);
343 			if (rows != null && ! rows.has(name) && ! catalog.hasName(name))
344 				out.add(new TemplateProblem(r.start(), r.end(), "unknown field '" + name + "' in list '" + list + "'"));
345 		} else if (catalog.hasName(list)) {
346 			out.add(new TemplateProblem(r.start(), r.end(), "'" + list + "' is not a declared list"));
347 		}
348 		// An unknown list name is reported where it is read, so it is not repeated for each row field.
349 	}
350 
351 	/**
352 	 * Like {@link #check(String, TemplateCatalog)} but throws.
353 	 *
354 	 * @param template The template text.  May be <jk>null</jk>.
355 	 * @param catalog The catalog.  Must not be <jk>null</jk>.
356 	 * @return The parsed template, if it is clean.
357 	 * @throws TemplateValidationException If there are problems.
358 	 */
359 	public ParsedTemplate validate(String template, TemplateCatalog catalog) {
360 		Objects.requireNonNull(catalog, "catalog must not be null");
361 		var c = compileOrProblems(template);
362 		var problems = check(c, catalog);
363 		if (! problems.isEmpty())
364 			throw new TemplateValidationException(problems);
365 		return new ParsedTemplate(c.template(), this);
366 	}
