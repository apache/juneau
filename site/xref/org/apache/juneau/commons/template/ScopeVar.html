17  package org.apache.juneau.commons.template;
18  
19  import org.apache.juneau.commons.svl.*;
20  
21  /**
22   * The <c>${name}</c> / <c>${name:default}</c> var of a {@link TemplateResolver}.
23   *
24   * <p>
25   * Registered under the name <c>P</c>, which is the var the <c>${name}</c> shortcut targets, so it replaces the property
26   * lookup of the default var set.  It reads the render scope of the session; nothing else is reachable through it.
27   * Names are flat: a dotted <c>${a.b}</c> looks for a scope entry literally named <c>a.b</c>; use <c>#{a.b}</c> for paths.
28   *
29   * <p>
30   * The key is read as written: {@link #allowNested()} is <jk>false</jk>, so <c>${${x}}</c> does not compute a key (which
31   * would bypass catalog validation).  A {@link Structure} in <c>${x}</c> is rendered with {@link Structure#toString()}
32   * and then encoded, not with the destination's renderer as <c>#{x}</c> would.
33   *
34   * <p>
35   * The text a scope value produces is data and is never re-resolved, so {@link #allowRecurse()} is <jk>false</jk>.
36   */
37  // TODO: dotted ${a.b} would need a public path reader from the evaluator
38  final class ScopeVar extends DefaultingVar {
39  
40  	ScopeVar() {
41  		super("P");
42  	}
43  
44  	@Override /* Overridden from Var */
45  	public String resolve(VarResolverSession session, String key) {
46  		var v = session.scope().apply(key);
47  		if (v == null)
48  			return null;
49  		if (v instanceof String s)
50  			return s;
51  		// A Structure placed in the scope is data too: use the plain-text form so it is encoded like any value.
52  		return v instanceof Structure ? v.toString() : session.formatValue(v);
53  	}
54  
55  	@Override /* Overridden from Var */
56  	protected boolean allowNested() {
57  		return false;
58  	}
59  
60  	@Override /* Overridden from Var */
61  	protected boolean allowRecurse() {
62  		return false;
63  	}
64  }
