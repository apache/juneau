17  package org.apache.juneau.commons.secret;
18  
19  import java.util.*;
20  
21  /**
22   * SPI for storing and retrieving a secret by key &mdash; the secure, mutable sibling of
23   * {@link org.apache.juneau.commons.settings.PropertySource}.
24   *
25   * <p>
26   * Where a {@code PropertySource} is a read-only source of ordinary configuration values, a {@code SecretStore}
27   * adds <i>write</i>/<i>delete</i> plus <i>sensitivity</i> semantics.  The SPI is deliberately narrow &mdash; it
28   * deals only in a {@link String} key and a {@code char[]} secret, with no dependency on any particular backend.
29   *
30   * <p>
31   * <b>Sensitivity is intrinsic.</b> Secret values are held as {@code char[]} rather than {@link String} so they can
32   * be explicitly zeroed and do not linger in the string pool.  An implementation must never {@code toString()}, log,
33   * or otherwise dump a secret value.  Callers are responsible for zeroing the arrays they pass to {@link #store}
34   * and receive from {@link #find} once done with them.
35   *
36   * <p>
37   * <b>Three-state presence model.</b> The three operations together model a clean present-with-value /
38   * present-without-materializing / absent distinction:
39   * <ul>
40   * 	<li>{@link #find(String)} &mdash; returns the secret value when present ({@link Optional} of {@code char[]}),
41   * 		or {@link Optional#empty()} when absent.  This is the only method that materializes the value.
42   * 	<li>{@link #exists(String)} &mdash; reports whether a secret is stored under the key <i>without</i> retrieving
43   * 		(and, for backends that decrypt on read, without decrypting) it.  Prefer this over {@link #find} when only
44   * 		presence matters, so the secret is never pulled into memory needlessly.
45   * 	<li><i>absent</i> &mdash; {@link #find} returns empty and {@link #exists} returns <jk>false</jk>.
46   * </ul>
47   *
48   * <p>
49   * <b>Backend-unavailable behavior is a consumer decision.</b> The built-in {@link InMemorySecretStore} and
50   * {@link EnvVarSecretStore} never fail on a well-formed call, so they simply throw on a genuine error.  Network- or
51   * OS-backed implementations (for example an OS-keychain-backed store in a separate module) should let the consumer
52   * pick fail-open vs fail-closed via a {@link FailMode} rather than baking a policy into the SPI.
53   *
54   * <h5 class='section'>See Also:</h5><ul>
55   * 	<li class='jc'>{@link InMemorySecretStore}
56   * 	<li class='jc'>{@link EnvVarSecretStore}
57   * 	<li class='jc'>{@link FailMode}
58   * 	<li class='jc'>{@link SecretStores}
59   * </ul>
60   *
61   * @since 10.0.0
62   */
63  public interface SecretStore {
64  
65  	/**
66  	 * Stores a secret under the specified key, replacing any existing value.
67  	 *
68  	 * <p>
69  	 * The caller retains ownership of <jv>secret</jv> and may zero it after this call returns; an implementation
70  	 * that needs to retain the value must copy it.
71  	 *
72  	 * @param key The key under which to store the secret.  Must not be <jk>null</jk>.
73  	 * @param secret The secret value.  Must not be <jk>null</jk>.
74  	 * @throws UnsupportedOperationException If this store is read-only.
75  	 */
76  	void store(String key, char[] secret);
77  
78  	/**
79  	 * Returns the secret stored under the specified key.
80  	 *
81  	 * <p>
82  	 * The returned array is the caller's to own and zero; it is not a live view of the store's internal state.
83  	 *
84  	 * @param key The key to look up.  Must not be <jk>null</jk>.
85  	 * @return The secret value, or {@link Optional#empty()} if no secret is stored under the key.
86  	 */
87  	Optional<char[]> find(String key);
88  
89  	/**
90  	 * Returns whether a secret is stored under the specified key without materializing its value.
91  	 *
92  	 * @param key The key to check.  Must not be <jk>null</jk>.
93  	 * @return <jk>true</jk> if a secret is stored under the key.
94  	 */
95  	boolean exists(String key);
96  
97  	/**
98  	 * Deletes the secret stored under the specified key.
99  	 *
100 	 * @param key The key to delete.  Must not be <jk>null</jk>.
101 	 * @return <jk>true</jk> if a secret was present and removed; <jk>false</jk> if the key was already absent.
102 	 * @throws UnsupportedOperationException If this store is read-only.
103 	 */
104 	boolean delete(String key);
105 }
