17  package org.apache.juneau.commons.secret;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.commons.inject.*;
22  
23  /**
24   * Resolution helpers for selecting the active {@link SecretStore}.
25   *
26   * <p>
27   * The active store is resolved from a {@link BeanStore}, defaulting to an {@link InMemorySecretStore} when none has
28   * been contributed &mdash; the same {@code beanStore.getBean(...)}-with-drop-in-default idiom other framework-owned
29   * components use.  An explicit {@link BeanStore} contribution always wins.
30   *
31   * <p>
32   * {@link #fromServiceLoader()} is the separate classpath-contribution path, mirroring
33   * {@link org.apache.juneau.commons.settings.Settings}'s opt-in {@code useServiceLoader()}.  It is deliberately
34   * <b>not</b> consulted by {@link #resolve(BeanStore)}, so the no-contribution default stays a deterministic
35   * {@link InMemorySecretStore} rather than silently becoming whatever store happens to be on the classpath (which, for
36   * an environment-variable-backed provider, would quietly turn arbitrary env vars into secrets).  A consumer that
37   * <i>wants</i> a classpath-contributed store can seed its {@link BeanStore} from {@link #fromServiceLoader()}.
38   *
39   * @since 10.0.0
40   */
41  public final class SecretStores {
42  
43  	private SecretStores() {}
44  
45  	/**
46  	 * Resolves the active secret store from the specified bean store, defaulting to a new
47  	 * {@link InMemorySecretStore} when none is contributed.
48  	 *
49  	 * @param beanStore The bean store to resolve from.  Can be <jk>null</jk>, in which case the default is returned.
50  	 * @return The resolved store.  Never <jk>null</jk>.
51  	 */
52  	public static SecretStore resolve(BeanStore beanStore) {
53  		if (beanStore == null)
54  			return new InMemorySecretStore();
55  		return beanStore.getBean(SecretStore.class).orElseGet(InMemorySecretStore::new);
56  	}
57  
58  	/**
59  	 * Discovers a secret store contributed on the classpath via {@link SecretStoreProvider} and {@code ServiceLoader}.
60  	 *
61  	 * <p>
62  	 * Providers are sorted by {@link SecretStoreProvider#order()} (lowest first) and the first one that yields a
63  	 * non-<jk>null</jk> store wins.
64  	 *
65  	 * @return The discovered store, or {@link Optional#empty()} if none is registered on the classpath.
66  	 */
67  	public static Optional<SecretStore> fromServiceLoader() {
68  		return ServiceLoader.load(SecretStoreProvider.class).stream()
69  			.map(ServiceLoader.Provider::get)
70  			.sorted(Comparator.comparingInt(SecretStoreProvider::order))
71  			.map(SecretStoreProvider::create)
72  			.filter(Objects::nonNull)
73  			.findFirst();
74  	}
75  }
