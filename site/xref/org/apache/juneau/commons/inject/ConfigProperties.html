17  package org.apache.juneau.commons.inject;
18  
19  import static java.lang.annotation.ElementType.*;
20  import static java.lang.annotation.RetentionPolicy.*;
21  
22  import java.lang.annotation.*;
23  
24  /**
25   * Declares a class as a prefix-scoped, whole-object config binding target — Juneau's analog of
26   * Spring's {@code @ConfigurationProperties}.
27   *
28   * <p>
29   * When a {@code @ConfigProperties}-annotated class is materialized through {@link BeanInstantiator},
30   * its fields are populated from the {@link org.apache.juneau.commons.settings.Settings Settings} chain
31   * under {@link #prefix()} via {@link ConfigPropertiesBinder}, and the bound instance is auto-registered
32   * into the current {@link BeanStore}.
33   *
34   * <h5 class='section'>Binding order and lifecycle</h5>
35   * <p>
36   * The whole-object config bind runs <b>after</b> {@code @Value}/{@code @Inject} field-and-method injection and
37   * <b>before</b> any {@link org.apache.juneau.commons.inject.PostConstruct @PostConstruct} method — so a
38   * {@code @PostConstruct} method observes the fully-bound config fields. Fields that are themselves annotated with
39   * {@code @Value} or {@code @Inject} are left to the injection engine and are never overwritten by config binding,
40   * even when a matching {@code prefix.fieldName} key exists.
41   *
42   * <h5 class='section'>Caller-scoped sources</h5>
43   * <p>
44   * The bind consults any {@link org.apache.juneau.commons.settings.PropertySource PropertySource}-typed beans in the
45   * active {@link BeanStore} <b>ahead of</b> the global {@link org.apache.juneau.commons.settings.Settings Settings}
46   * chain, so a caller can layer overriding or supplementary sources without mutating the shared
47   * {@code Settings.get()} singleton.
48   *
49   * <h5 class='section'>Not inherited</h5>
50   * <p>
51   * This annotation is <b>not</b> {@link java.lang.annotation.Inherited @Inherited}: the {@link BeanInstantiator}
52   * trigger fires only when the <i>exact declared type</i> being instantiated carries the annotation — a subclass or
53   * interface implementor of an annotated type does not itself trigger binding or auto-registration.
54   *
55   * <p>
56   * This is distinct from the {@link ConfigPropertiesBinder} <b>nested-field</b> detection, which <i>is</i>
57   * inheritance-aware: a field whose <i>declared type</i> is a subclass of an {@code @ConfigProperties}-annotated
58   * type still triggers nested binding under the enclosing type's prefix, since the annotation lookup there walks the
59   * declared field type's class hierarchy rather than requiring an exact-type match.
60   *
61   * <h5 class='section'>Example:</h5>
62   * <p class='bjava'>
63   * 	<ja>@ConfigProperties</ja>(prefix = <js>"MyService"</js>)
64   * 	<jk>public class</jk> MyServiceProperties {
65   * 		<jk>public</jk> String <jv>host</jv> = <js>"localhost"</js>;
66   * 		<jk>public int</jk> <jv>port</jv> = 8080;
67   * 	}
68   * </p>
69   *
70   * <p>
71   * {@code @ConfigProperties} and {@link org.apache.juneau.commons.inject.Value @Value} coexist: {@code @Value}
72   * resolves a single value; {@code @ConfigProperties} resolves a whole object. A class can use
73   * {@code @Value} fields internally and separately be the target of another class's nested
74   * {@code @ConfigProperties} bind.
75   *
76   * <h5 class='section'>See Also:</h5><ul>
77   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/ConfigProperties">@ConfigProperties Annotation Basics</a>
78   * </ul>
79   *
80   * @since 10.0.0
81   */
82  @Documented
83  @Target(TYPE)
84  @Retention(RUNTIME)
85  public @interface ConfigProperties {
86  
87  	/**
88  	 * The property-key prefix (no trailing dot).
89  	 *
90  	 * <p>
91  	 * Ignored when the annotated type appears as a nested field of another {@code @ConfigProperties} type — the
92  	 * enclosing type's prefix (extended with the field name) wins for the nested bind.
93  	 *
94  	 * @return The prefix.
95  	 */
96  	String prefix();
97  
98  	/**
99  	 * Whether relaxed (camelCase / kebab-case / dotted / {@code SCREAMING_SNAKE_CASE}) key matching is enabled.
100 	 *
101 	 * <p>
102 	 * Ignored when the annotated type appears as a nested field of another {@code @ConfigProperties} type — the
103 	 * enclosing type's {@code relaxed} flag wins for the nested bind.
104 	 *
105 	 * @return <jk>true</jk> if relaxed matching is enabled. Defaults to <jk>true</jk>.
106 	 */
107 	boolean relaxed() default true;
108 }
