17  package org.apache.juneau.commons.inject;
18  
19  import static java.lang.annotation.ElementType.*;
20  import static java.lang.annotation.RetentionPolicy.*;
21  
22  import java.lang.annotation.*;
23  
24  /**
25   * Resolves a configuration value into a constructor parameter, setter, or field.
26   *
27   * <p>
28   * The {@code value()} string is passed through Juneau's {@code VarResolver} (so
29   * {@code ${...}} / {@code $P{...}} placeholders are expanded against
30   * {@link org.apache.juneau.commons.settings.Settings#get() Settings.get()}), then coerced to the
31   * target Java type via {@link org.apache.juneau.commons.settings.Settings#toType(String, Class)}.
32   *
33   * <h5 class='section'>Examples:</h5>
34   * <p class='bjava'>
35   * 	<jc>// Field injection with default.</jc>
36   * 	<ja>@Value</ja>(<js>"${db.timeout.ms:5000}"</js>)
37   * 	<jk>private int</jk> <jv>timeoutMs</jv>;
38   *
39   * 	<jc>// Constructor-parameter injection.</jc>
40   * 	<ja>@Inject</ja>
41   * 	<jk>public</jk> OrdersResource(
42   * 		<ja>@Value</ja>(<js>"${app.name:orders}"</js>) String <jv>appName</jv>,
43   * 		<ja>@Value</ja>(<js>"${app.start}"</js>) Instant <jv>start</jv>
44   * 	) { ... }
45   * </p>
46   *
47   * <h5 class='section'>Resolution order:</h5>
48   * <ol>
49   * 	<li>Thread-local override ({@link org.apache.juneau.commons.settings.Settings#setLocal Settings.setLocal})
50   * 	<li>Global override ({@link org.apache.juneau.commons.settings.Settings#setGlobal Settings.setGlobal})
51   * 	<li>Property sources in reverse insertion order — this is where Juneau {@code Config},
52   * 		Spring {@code Environment}, and any other registered {@code PropertySource} participate.
53   * 	<li>System properties
54   * 	<li>System environment variables
55   * </ol>
56   *
57   * <p>
58   * Spring's {@code org.springframework.beans.factory.annotation.Value} annotation is honored identically
59   * via FQN-based detection in {@link JsrSupport}; no compile-time Spring dependency is introduced in
60   * {@code juneau-commons}.
61   *
62   * <h5 class='section'>Notes:</h5><ul>
63   * 	<li class='note'>{@code @Value} and {@code @Inject} on the same site is invalid and throws
64   * 		a {@code BeanCreationException}. {@code @Value} resolves strings/primitives; {@code @Inject}
65   * 		resolves beans.
66   * 	<li class='note'>If the expression resolves to {@code null} and the target type is a primitive,
67   * 		a {@code BeanCreationException} is thrown.  For reference types, {@code null} is injected.
68   * </ul>
69   *
70   * <h5 class='section'>See Also:</h5><ul>
71   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/ValueAnnotation">@Value Annotation Basics</a>
72   * </ul>
73   */
74  @Documented
75  @Target({CONSTRUCTOR, METHOD, FIELD, PARAMETER})
76  @Retention(RUNTIME)
77  public @interface Value {
78  
79  	/**
80  	 * The configuration expression.
81  	 *
82  	 * <p>
83  	 * Typically uses the {@code ${key}} or {@code ${key:default}} placeholder syntax that is routed
84  	 * through {@code VarResolver} and {@link org.apache.juneau.commons.settings.Settings Settings}.
85  	 * Plain literals (no {@code $}) are passed through unchanged and still type-coerced.
86  	 *
87  	 * @return The configuration expression.
88  	 */
89  	String value();
90  }
