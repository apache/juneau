17  package org.apache.juneau.commons.inject;
18  
19  import static org.apache.juneau.commons.utils.AssertionUtils.*;
20  import static org.apache.juneau.commons.utils.Shorts.*;
21  
22  import java.util.*;
23  import java.util.concurrent.*;
24  import java.util.function.*;
25  
26  /**
27   * Stack-aware {@link BeanStore} overlay.
28   *
29   * <p>
30   * A {@code StackOverlay} delegates to a LIFO stack of {@link BeanStore} frames.  Each lookup walks the stack
31   * from top to bottom and returns the first match.  When the stack is empty the lookup returns
32   * {@link Optional#empty()} so that an outer chain (typically a {@link BasicBeanStore}'s {@code overridingParent}
33   * slot) can fall through to its remaining tiers.
34   *
35   * <p>
36   * This class is designed to live in the {@code overridingParent} slot of a {@link BasicBeanStore}.  Pushing a
37   * frame inserts a new tier-1 layer; popping it removes that layer.  Stack order is enforced LIFO &mdash; popping
38   * out of order or popping an empty stack throws {@link IllegalStateException}.
39   *
40   * <p>
41   * The implementation uses a {@link ConcurrentLinkedDeque} so that reads from concurrent HTTP request threads
42   * see a consistent view while pushes and pops are in flight.  Pushes/pops are expected to be infrequent (test
43   * setup/teardown); reads are the hot path.
44   *
45   * <p>
46   * <b>Composition, not inheritance.</b>  {@code StackOverlay} {@code implements} {@link BeanStore} rather than
47   * extending {@link BasicBeanStore} so that it does <i>not</i> inherit the entries / defaults / parent storage
48   * that a {@link BasicBeanStore} provides.  A {@code StackOverlay} is purely a stack of pluggable bean stores.
49   *
50   * <h5 class='section'>Example:</h5>
51   * <p class='bjava'>
52   * 	<jc>// Create an overlay-capable bean store.</jc>
53   * 	StackOverlay <jv>overlay</jv> = <jk>new</jk> StackOverlay();
54   * 	BasicBeanStore <jv>store</jv> = <jk>new</jk> BasicBeanStore(<jk>null</jk>, <jv>overlay</jv>);
55   *
56   * 	<jc>// Push a per-test override frame.</jc>
57   * 	<jv>overlay</jv>.push(<jk>new</jk> BasicBeanStore().addBean(MyService.<jk>class</jk>, <jv>mock</jv>));
58   * 	<jv>store</jv>.getBean(MyService.<jk>class</jk>);  <jc>// returns the mock</jc>
59   *
60   * 	<jc>// Pop it at end of test.</jc>
61   * 	<jv>overlay</jv>.pop();
62   * </p>
63   *
64   * <h5 class='section'>See Also:</h5><ul>
65   * 	<li class='jc'>{@link BasicBeanStore} - The bean store this overlay sits in front of.
66   * </ul>
67   *
68   * @since 10.0.0
69   */
70  public final class StackOverlay implements BeanStore {
71  
72  	private final Deque<BeanStore> frames = new ConcurrentLinkedDeque<>();
73  
74  	/**
75  	 * Constructor.
76  	 */
77  	public StackOverlay() { /* no-op */ }
78  
79  	/**
80  	 * Pushes a new overlay frame onto the top of the stack.
81  	 *
82  	 * <p>
83  	 * The frame becomes the highest-priority lookup target for subsequent {@link #getBean(Class)} /
84  	 * {@link #getBeanSupplier(Class)} calls until it is popped.
85  	 *
86  	 * @param overlay The bean store to push.  Must not be <jk>null</jk> or an {@link IllegalArgumentException} is thrown.
87  	 * @return This object.
88  	 */
89  	public StackOverlay push(BeanStore overlay) {
90  		assertArgNotNull("overlay", overlay);
91  		frames.push(overlay);
92  		return this;
93  	}
94  
95  	/**
96  	 * Pops the top frame off the stack.
97  	 *
98  	 * @return The popped frame.
99  	 * @throws IllegalStateException If the stack is empty.
100 	 */
101 	public BeanStore pop() {
102 		var f = frames.pollFirst();
103 		if (f == null)
104 			throw new IllegalStateException("Cannot pop from an empty StackOverlay.");
105 		return f;
106 	}
107 
108 	/**
109 	 * Returns the current depth of the stack.
110 	 *
111 	 * @return The number of frames currently on the stack.
112 	 */
113 	public int depth() {
114 		return frames.size();
115 	}
116 
117 	/**
118 	 * Returns the top frame on the stack without removing it.
119 	 *
120 	 * @return The top frame, or <jk>null</jk> if the stack is empty.
121 	 */
122 	public BeanStore peek() {
123 		return frames.peekFirst();
124 	}
125 
126 	@Override /* BeanStore */
127 	public <T> Optional<T> getBean(Class<T> beanType) {
128 		return getBean(beanType, null);
129 	}
130 
131 	@Override /* BeanStore */
132 	public <T> Optional<T> getBean(Class<T> beanType, String name) {
133 		for (var f : frames) {
134 			var b = f.getBean(beanType, name);
135 			if (b.isPresent())
136 				return b;
137 		}
138 		return oe();
139 	}
140 
141 	@Override /* BeanStore */
142 	public <T> Map<String,T> getBeansOfType(Class<T> beanType) {
143 		// Walk the stack bottom-to-top so that higher (later-pushed) frames overwrite lower frames
144 		// for entries with the same name.  Iterating an iterator gives top-to-bottom; we reverse
145 		// that here by snapshotting and walking the reversed iterator.
146 		Map<String,T> result = m();
147 		var snapshot = new ArrayList<>(frames);
148 		for (var i = snapshot.size() - 1; i >= 0; i--)
149 			snapshot.get(i).getBeansOfType(beanType).forEach(result::put);
150 		return result;
151 	}
152 
153 	@Override /* BeanStore */
154 	public boolean hasBean(Class<?> beanType) {
155 		return hasBean(beanType, null);
156 	}
157 
158 	@Override /* BeanStore */
159 	public boolean hasBean(Class<?> beanType, String name) {
160 		for (var f : frames)
161 			if (f.hasBean(beanType, name))
162 				return true;
163 		return false;
164 	}
165 
166 	@Override /* BeanStore */
167 	public <T> Optional<Supplier<T>> getBeanSupplier(Class<T> beanType) {
168 		return getBeanSupplier(beanType, null);
169 	}
170 
171 	@Override /* BeanStore */
172 	public <T> Optional<Supplier<T>> getBeanSupplier(Class<T> beanType, String name) {
173 		for (var f : frames) {
174 			var s = f.getBeanSupplier(beanType, name);
175 			if (s.isPresent())
176 				return s;
177 		}
178 		return oe();
179 	}
180 
181 	@Override /* BeanStore */
182 	public <T> Optional<Class<? extends T>> getBeanType(Class<T> beanType) {
183 		for (var f : frames) {
184 			var t = f.getBeanType(beanType);
185 			if (t.isPresent())
186 				return t;
187 		}
188 		return oe();
189 	}
190 
191 	@Override /* Overridden from Object */
192 	public String toString() {
193 		// Intentionally avoid recursing into frame.toString() — frames may back-reference the outer
194 		// BasicBeanStore that holds this overlay in its overridingParent slot, which would cause
195 		// stack overflows via BasicBeanStore.properties().  A flat representation (depth + identity
196 		// of each frame) is sufficient for diagnostics.
197 		var sb = new StringBuilder("StackOverlay{depth=").append(frames.size()).append(", frames=[");
198 		var first = true;
199 		for (var f : frames) {
200 			if (!first)
201 				sb.append(", ");
202 			first = false;
203 			sb.append(cns(f))
204 				.append('@')
205 				.append(Integer.toHexString(System.identityHashCode(f)));
206 		}
207 		return sb.append("]}").toString();
208 	}
209 }
