17  package org.apache.juneau.commons.inject;
18  
19  /**
20   * Marker interface for builders that accept an <i>overriding</i> {@link BeanStore} &mdash; a layer
21   * whose entries take precedence over the builder's local bean registrations during construction-time
22   * reflective injection.
23   *
24   * <p>
25   * Implementing builders MUST install the supplied store in the {@code overridingParent} slot of the
26   * underlying {@link BasicBeanStore} (see
27   * {@link BasicBeanStore#BasicBeanStore(BeanStore, BeanStore)}).  The resulting bean-lookup order is:
28   * <ol>
29   * 	<li><i>Overriding parent</i> (the store passed to {@link #overridingBeanStore(BeanStore)}) &mdash; wins
30   * 		for any {@code (type, name)} it knows about.
31   * 	<li>The builder's local entries (e.g. {@code @Bean} factories declared on a {@code @Rest} resource,
32   * 		beans registered directly via {@code addBean(...)} / {@code addSupplier(...)}).
33   * 	<li>The builder's regular parent bean store (e.g. a Spring {@code ApplicationContext} bridge), if any.
34   * 	<li>Framework-default suppliers (registered via {@code addDefaultSupplier(...)}).
35   * </ol>
36   *
37   * <p>
38   * The primary use case is test-time bean substitution: a test builds a {@code TestBeanStore}-style overlay
39   * with mock beans, hands it to the builder via {@link #overridingBeanStore(BeanStore)}, and the system under
40   * test resolves the mocks at construction time without any reflection on the SUT itself.  This is the
41   * <b>Mode INJECT</b> wiring pattern in the test-injection model &mdash; the overlay is installed before the SUT
42   * exists.
43   *
44   * <h5 class='section'>Example:</h5>
45   * <p class='bjava'>
46   * 	<jk>var</jk> <jv>overlay</jv> = <jk>new</jk> TestBeanStore().override(MyService.<jk>class</jk>, <jv>mockSvc</jv>);
47   * 	<jk>var</jk> <jv>serializers</jv> = SerializerSet.<jsm>create</jsm>()
48   * 		.overridingBeanStore(<jv>overlay</jv>)
49   * 		.add(JsonSerializer.<jk>class</jk>)
50   * 		.build();
51   * </p>
52   *
53   * <h5 class='section'>Why a self-typed interface:</h5>
54   * <p>
55   * The {@code B extends BeanStoreOverridable<B>} self-type lets the method return the concrete builder
56   * type rather than {@code BeanStoreOverridable<?>}, preserving the fluent chain at the call site:
57   *
58   * <p class='bjava'>
59   * 	SerializerSet.Builder <jv>builder</jv> = SerializerSet.<jsm>create</jsm>()
60   * 		.overridingBeanStore(<jv>overlay</jv>)   <jc>// still SerializerSet.Builder, not BeanStoreOverridable&lt;?&gt;</jc>
61   * 		.add(JsonSerializer.<jk>class</jk>);     <jc>// fluent chain preserved</jc>
62   * </p>
63   *
64   * <h5 class='section'>See Also:</h5><ul>
65   * 	<li class='jc'>{@link BeanStore} - The base lookup interface this layer extends.
66   * 	<li class='jc'>{@link BasicBeanStore} - The default writable implementation with the {@code overridingParent} slot.
67   * </ul>
68   *
69   * @param <B> The implementing builder's self type.
70   * @since 10.0.0
71   */
72  public interface BeanStoreOverridable<B extends BeanStoreOverridable<B>> {
73  
74  	/**
75  	 * Installs the supplied {@link BeanStore} as the {@code overridingParent} of the underlying bean store
76  	 * used by this builder during construction-time reflective injection.
77  	 *
78  	 * <p>
79  	 * Passing {@code null} clears any previously-installed overlay.
80  	 *
81  	 * @param store The override layer.  Can be <jk>null</jk> to clear a previously-set value.
82  	 * @return This object.
83  	 * @since 10.0.0
84  	 */
85  	B overridingBeanStore(BeanStore store);
86  }
