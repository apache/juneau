30  /**
31   * Internal helper used by the reflection layer to resolve {@code @Value}-annotated injection sites.
32   *
33   * <p>
34   * Splits the {@code @Value} handling into three small pieces so callers (field / parameter / method)
35   * can share the same coercion + error-formatting code without duplicating the lookup logic:
36   *
37   * <ol>
38   * 	<li>{@link #findValueExpression(List)} — given an annotation list, return the first {@code @Value}
39   * 		expression (or {@code null} if no {@code @Value} is present).
40   * 	<li>{@link #checkInjectConflict(List, String)} — throw {@link BeanCreationException} if both
41   * 		{@code @Value} and {@code @Inject}/{@code @Autowired} are present on the same site.
42   * 	<li>{@link #resolve(String, Class, String)} — pass the expression through {@code VarResolver},
43   * 		coerce to the target type via {@link Settings#toType(String, Class)}, and wrap failures in
44   * 		{@link BeanCreationException}.
45   * </ol>
46   *
47   * <p>
48   * The resolver intentionally uses {@link VarResolver#DEFAULT}, which already includes
49   * {@code PropertyVar} (the {@code $P{key[,default]}} variable that reads from {@link Settings}).
50   * After the property-variable unification, the {@code ${key}} shortcut routes through that same {@code PropertyVar},
51   * so {@code @Value("${db.url}")} and {@code @Value("$P{db.url}")} are equivalent.
52   */
53  public final class ValueResolver {
54  
55  	private ValueResolver() {}
56  
57  	/**
58  	 * Per-expression {@link VarTemplate} cache shared by all {@code @Value} injection sites.
59  	 * The cache key is the {@code @Value} expression text —
60  	 * which is bounded by the number of distinct expressions in the application, not by user
61  	 * input — so the unbounded map cannot grow uncontrollably in practice. Compilation
62  	 * happens at most once per distinct expression; subsequent {@code @Value} resolutions
63  	 * read the cached {@link VarTemplate} and skip the tokenizer + var-registry-lookup
64  	 * pipeline entirely.
65  	 */
66  	private static final ConcurrentMap<String,VarTemplate> TEMPLATE_CACHE = new ConcurrentHashMap<>();
67  
68  	/**
69  	 * Returns the compiled {@link VarTemplate} for the given expression, computing and caching
70  	 * on first use. Intended for use by reflection-layer injection sites
71  	 * ({@link FieldInfo#inject}, {@link ParameterInfo#resolveValue}) that want to amortize
72  	 * tokenization across repeated bean constructions.
73  	 *
74  	 * @param expression The {@code @Value} expression text. Must not be {@code null}.
75  	 * @return The cached compiled template (uses {@link VarResolver#DEFAULT}).
76  	 */
77  	public static VarTemplate getCompiledTemplate(String expression) {
78  		return TEMPLATE_CACHE.computeIfAbsent(expression, VarResolver.DEFAULT::compile);
79  	}
80  
81  	/**
82  	 * Clears the internal {@link VarTemplate} cache. Primarily intended for tests that need to
83  	 * exercise the compile-time path repeatedly with different {@link VarResolver}s. Production
84  	 * code should not need to call this.
85  	 */
86  	public static void clearTemplateCache() {
87  		TEMPLATE_CACHE.clear();
88  	}

90  	/**
91  	 * Returns the {@code value()} expression carried by the first {@code @Value} annotation in the list,
92  	 * or {@code null} if no {@code @Value} (Juneau or Spring) is present.
93  	 *
94  	 * @param annotations The annotation list (from {@link FieldInfo#getAnnotations()},
95  	 * 	{@link ParameterInfo#getAnnotations()}, etc.). May be {@code null} or empty, in which case {@code null} is returned.
96  	 * @return The expression, or {@code null} if not a {@code @Value} site.
97  	 */
98  	public static String findValueExpression(List<? extends AnnotationInfo<?>> annotations) {
99  		if (isEmpty(annotations))
100 			return null;
101 		return annotations.stream()
102 			.map(JsrSupport::valueExpression)
103 			.filter(Objects::nonNull)
104 			.findFirst()
105 			.orElse(null);
106 	}
107 
108 	/**
109 	 * Returns {@code true} if any annotation in the list is a {@code @Value} marker.
110 	 *
111 	 * @param annotations The annotation list.  May be {@code null} or empty, in which case {@code false} is returned.
112 	 * @return {@code true} if at least one annotation matches {@link JsrSupport#isValueAnnotation(AnnotationInfo)}.
113 	 */
114 	public static boolean hasValueAnnotation(List<? extends AnnotationInfo<?>> annotations) {
115 		if (isEmpty(annotations))
116 			return false;
117 		return annotations.stream().anyMatch(JsrSupport::isValueAnnotation);
118 	}

120 	/**
121 	 * Throws {@link BeanCreationException} if an injection site carries both {@code @Value} and an
122 	 * {@code @Inject}/{@code @Autowired} marker.
123 	 *
124 	 * <p>
125 	 * {@code @Value} resolves strings/primitives; {@code @Inject} resolves beans. Combining them on
126 	 * the same site is ambiguous — fail fast with a clear message.
127 	 *
128 	 * @param annotations The annotation list.  May be {@code null} or empty, in which case this method returns without action.
129 	 * @param siteDescription A human-readable description of the site (used in the error message).
130 	 */
131 	public static void checkInjectConflict(List<? extends AnnotationInfo<?>> annotations, String siteDescription) {
132 		if (isEmpty(annotations))
133 			return;
134 		var hasValue = annotations.stream().anyMatch(JsrSupport::isValueAnnotation);
135 		if (! hasValue)
136 			return;
137 		var hasInject = annotations.stream().anyMatch(JsrSupport::isInjectAnnotation);
138 		if (hasInject)
139 			throw new BeanCreationException("@Value and @Inject are mutually exclusive on " + siteDescription);
140 	}

142 	/**
143 	 * Resolves a {@code @Value} expression against the default {@link VarResolver} and coerces the
144 	 * result to {@code targetType} via {@link Settings#toType(String, Class)}.
145 	 *
146 	 * <p>
147 	 * Lookup semantics:
148 	 * <ul>
149 	 * 	<li>If the resolved value is {@code null} and {@code targetType} is a primitive, throws
150 	 * 		{@link BeanCreationException} (primitives cannot be {@code null}).
151 	 * 	<li>If the resolved value is {@code null} and {@code targetType} is a reference type, returns
152 	 * 		{@code null}.
153 	 * 	<li>If coercion via {@link Settings#toType(String, Class)} fails, the underlying exception is
154 	 * 		wrapped in {@link BeanCreationException} with a message that includes the expression, the
155 	 * 		resolved string, and the target type.
156 	 * </ul>
157 	 *
158 	 * @param expression The {@code @Value} expression (typically {@code "${key}"} or {@code "${key:default}"}). May be {@code null}, in which case {@code null} is returned for reference types (and a {@link BeanCreationException} is thrown for primitive target types).
159 	 * @param targetType The target Java type. May be a primitive. Must not be {@code null}.
160 	 * @param siteDescription A human-readable description of the site (used in error messages).
161 	 * @return The coerced value.
162 	 */
163 	public static Object resolve(String expression, Class<?> targetType, String siteDescription) {
164 		return resolve(expression, targetType, siteDescription, null);
165 	}

167 	/**
168 	 * Same as {@link #resolve(String, Class, String)} but consults caller-scoped
169 	 * {@link PropertySource} beans resolved from the supplied {@link BeanStore} between
170 	 * the {@link Settings} local/global override stores and the {@link Settings} sources chain.
171 	 *
172 	 * <p>
173 	 * The {@code beanStore} parameter is preserved by-reference; if its
174 	 * {@code PropertySource}-typed contents change between calls, the next resolution picks up
175 	 * the current state. Passing {@code null} (or a {@code BeanStore} with no
176 	 * {@code PropertySource}-typed beans) is byte-for-byte equivalent to the zero-argument
177 	 * overload &mdash; no extra allocation and no behavior change.
178 	 *
179 	 * @param expression The {@code @Value} expression. May be {@code null}, in which case {@code null} is returned for reference types (and a {@link BeanCreationException} is thrown for primitive target types).
180 	 * @param targetType The target Java type. May be a primitive. Must not be {@code null}.
181 	 * @param siteDescription A human-readable description of the site (used in error messages).
182 	 * @param beanStore Caller-scoped {@link BeanStore} consulted for {@link PropertySource} beans.
183 	 * 	May be {@code null}.
184 	 * @return The coerced value.
185 	 */
186 	public static Object resolve(String expression, Class<?> targetType, String siteDescription, BeanStore beanStore) {
187 		if (expression == null)
188 			return resolveCoerce(null, targetType, expression, siteDescription);
189 		var template = getCompiledTemplate(expression);
190 		var session = VarResolver.DEFAULT.createSession();
191 		var sources = scopedSources(beanStore);
192 		if (sources != null)
193 			session.bean(PropertySource[].class, sources);
194 		var resolved = template.resolve(session);
195 		return resolveCoerce(resolved, targetType, expression, siteDescription);
196 	}

198 	/**
199 	 * Generic-aware {@code @Value} resolution that honors {@link Supplier}{@code <T>}
200 	 * field types by returning a re-evaluating, threadsafe Supplier instead of a one-shot
201 	 * resolved value (field-type autodetect — bare {@code String} resolves once, {@code Supplier<T>}
202 	 * re-evaluates per {@code .get()}).
203 	 *
204 	 * <p>
205 	 * Behavior by {@code genericTargetType}:
206 	 * <ul>
207 	 * 	<li>{@code Supplier<String>} → returns a {@link Supplier} that re-evaluates the
208 	 * 		compiled template on every {@link Supplier#get()} call, each time against a
209 	 * 		fresh {@link VarResolverSession}. Safe to share across threads.
210 	 * 	<li>{@code Supplier<T>} for {@code T != String} → returns a {@link Supplier} that, on every
211 	 * 		{@link Supplier#get()} call, re-evaluates the template <i>and</i> coerces the resolved
212 	 * 		string into {@code T} via the same {@code String -> T} converter machinery that plain
213 	 * 		{@code @Value T} uses ({@link Settings#toType(String, Class)}). The coerced value is never
214 	 * 		memoized (late-binding contract); coercion failures surface as a {@link RuntimeException}
215 	 * 		from {@code .get()} naming the {@code @Value} key and target type. {@code T} is taken from
216 	 * 		any type the converter registry can build from a {@code String} (e.g. {@code Integer},
217 	 * 		{@code Long}, {@code Boolean}, {@code Duration}, {@code URI}, {@code Path}, enums).
218 	 * 	<li>Anything else (including raw {@code Supplier} and {@code Optional<Supplier<T>>}) → delegates
219 	 * 		to {@link #resolve(String, Class, String)} with the raw target class (preserves existing
220 	 * 		behavior; the {@code Optional<Supplier<T>>} nesting is intentionally not specially handled).
221 	 * </ul>
222 	 *
223 	 * @param expression The {@code @Value} expression. May be {@code null}.
224 	 * @param targetType The erased target class (e.g. {@code String.class} or {@code Supplier.class}). Must not be {@code null}.
225 	 * @param genericTargetType The declared generic type (e.g. {@code Supplier<Integer>}). May be
226 	 * 	{@code null}, in which case Supplier-detection is skipped.
227 	 * @param siteDescription Human-readable site description for error messages.
228 	 * @return Either a one-shot coerced value, or a {@code Supplier<T>} factory.
229 	 */
230 	public static Object resolve(String expression, Class<?> targetType, Type genericTargetType, String siteDescription) {
231 		return resolve(expression, targetType, genericTargetType, siteDescription, null);
232 	}

234 	/**
235 	 * Same as {@link #resolve(String, Class, Type, String)} but consults caller-scoped
236 	 * {@link PropertySource} beans resolved from the supplied {@link BeanStore}.
237 	 *
238 	 * <p>
239 	 * The {@code beanStore} reference is captured (not snapshotted) by the returned
240 	 * {@code Supplier<T>} when the declared field/parameter type is {@code Supplier<...>},
241 	 * so re-evaluating reads always see the current state of {@code beanStore.getBeansOfType(PropertySource.class)}.
242 	 *
243 	 * @param expression The {@code @Value} expression. May be {@code null}.
244 	 * @param targetType The erased target class. Must not be {@code null}.
245 	 * @param genericTargetType The declared generic type. May be {@code null}.
246 	 * @param siteDescription Human-readable site description for error messages.
247 	 * @param beanStore Caller-scoped {@link BeanStore} consulted for {@link PropertySource} beans.
248 	 * 	May be {@code null}.
249 	 * @return Either a one-shot coerced value, or a {@code Supplier<T>} factory.
250 	 */
251 	public static Object resolve(String expression, Class<?> targetType, Type genericTargetType, String siteDescription,
252 			BeanStore beanStore) {
253 		var elementType = supplierElementType(targetType, genericTargetType);
254 		if (elementType != null) {
255 			// Build the re-evaluating string Supplier once. This is the same factory the bare
256 			// Supplier<String> path has always used.
257 			var stringSupplier = buildStringSupplier(expression, beanStore);
258 			if (elementType == String.class)
259 				return stringSupplier;
260 			// Supplier<T> for T != String: coerce per .get() using the same String -> T converter
261 			// machinery that plain @Value T uses (resolveCoerce -> Settings.toType). Per the
262 			// late-binding contract, the string is re-resolved AND re-coerced on every .get() — the
263 			// coerced value is never memoized. Only the converter lookup is cached (Settings.toType
264 			// memoizes the discovered String -> T function per type in its toTypeFunctions registry,
265 			// so each .get() pays the coercion but not the reflective converter discovery). A coercion
266 			// failure surfaces as a BeanCreationException (a RuntimeException) from .get(), naming the
267 			// @Value expression and the target type.
268 			final Class<?> et = elementType;
269 			return (Supplier<Object>) () -> resolveCoerce(stringSupplier.get(), et, expression, siteDescription);
270 		}
271 		return resolve(expression, targetType, siteDescription, beanStore);
272 	}

274 	/**
275 	 * Builds the re-evaluating {@code Supplier<String>} used by the {@code @Value Supplier<...>}
276 	 * autodetect path. Each {@link Supplier#get()} re-resolves the compiled template against a fresh
277 	 * {@link VarResolverSession}, so live config changes are reflected (the late-binding contract).
278 	 *
279 	 * @param expression The {@code @Value} expression. May be {@code null}.
280 	 * @param beanStore Caller-scoped {@link BeanStore} consulted for {@link PropertySource} beans. May be {@code null}.
281 	 * @return A threadsafe, re-evaluating {@code Supplier<String>}.
282 	 */
283 	private static Supplier<String> buildStringSupplier(String expression, BeanStore beanStore) {
284 		if (expression == null)
285 			return () -> null;
286 		var template = getCompiledTemplate(expression);
287 		// Literal templates fast-path: capture the resolved string once and hand back a
288 		// constant Supplier. Saves a per-.get() session allocation when the expression
289 		// has no variables (e.g. @Value("literal") Supplier<String>).
290 		if (template.isLiteral()) {
291 			var literal = template.resolve(VarResolver.DEFAULT.createSession());
292 			return () -> literal;
293 		}
294 		if (beanStore == null)
295 			return template.asSupplierWithFreshSessions(VarResolver.DEFAULT);
296 		// Capture the BeanStore by-reference so re-evaluating reads see the current state of
297 		// its PropertySource-typed beans (matches the "by-reference, not snapshot" contract).
298 		final BeanStore scope = beanStore;
299 		return () -> {
300 			var s = VarResolver.DEFAULT.createSession();
301 			var sources = scopedSources(scope);
302 			if (sources != null)
303 				s.bean(PropertySource[].class, sources);
304 			return template.resolve(s);
305 		};
306 	}

308 	/**
309 	 * Returns the caller-scoped {@link PropertySource} array for the supplied {@link BeanStore},
310 	 * or {@code null} if the store has no {@code PropertySource}-typed beans (matching the
311 	 * "no behavior change when no scoped sources are present" contract).
312 	 *
313 	 * <p>
314 	 * Walks {@link BeanStore#getBeansOfType(Class) beanStore.getBeansOfType(PropertySource.class)}.
315 	 * The returned array preserves the iteration order of {@code getBeansOfType} (parent-chain
316 	 * beans before local beans before overriding-parent beans, then sorted by
317 	 * {@code @Order/@Primary/@Bean.priority()}).
318 	 */
319 	@SuppressWarnings({
320 		"java:S1168" // Null is a meaningful "no scoped sources" signal that callers branch on to preserve the no-allocation / no-behavior-change contract; an empty array would alter behavior.
321 	})
322 	private static PropertySource[] scopedSources(BeanStore beanStore) {
323 		if (beanStore == null)
324 			return null;
325 		var map = beanStore.getBeansOfType(PropertySource.class);
326 		if (map.isEmpty())
327 			return null;
328 		return map.values().toArray(new PropertySource[0]);
329 	}

331 	/**
332 	 * Returns the {@code T} element class of a declared {@code Supplier<T>} injection site, or
333 	 * {@code null} if the declared type is not a coercible {@code Supplier<T>}.
334 	 *
335 	 * <p>
336 	 * Returns {@code null} (i.e. "not a typed Supplier site, fall through to one-shot resolution")
337 	 * for raw {@code Supplier}, non-{@code Supplier} parameterized types such as
338 	 * {@code Optional<Supplier<T>>} (whose raw type is {@code Optional}, not {@code Supplier} — the
339 	 * nesting is intentionally not specially handled), and type arguments that are not a concrete
340 	 * {@link Class} (wildcards / type variables). {@code Supplier<String>} returns {@code String.class},
341 	 * which the caller treats as the pre-existing pass-through behavior.
342 	 */
343 	private static Class<?> supplierElementType(Class<?> targetType, Type genericTargetType) {
344 		if (targetType != Supplier.class)
345 			return null;
346 		if (!(genericTargetType instanceof ParameterizedType genericTargetType2))
347 			return null;
348 		if (genericTargetType2.getRawType() != Supplier.class)
349 			return null;
350 		var args = genericTargetType2.getActualTypeArguments();
351 		if (args.length != 1)
352 			return null;
353 		return args[0] instanceof Class<?> c ? c : null;
354 	}

356 	private static Object resolveCoerce(String resolved, Class<?> targetType, String expression, String siteDescription) {
357 		var effectiveType = box(targetType);
358 
359 		if (resolved == null || resolved.isEmpty()) {
360 			if (targetType.isPrimitive())
361 				throw new BeanCreationException("Could not resolve required @Value('" + expression + "') for primitive site " + siteDescription);
362 			if (resolved == null)
363 				return null;
364 			if (effectiveType == String.class)
365 				return resolved;
366 		}
367 
368 		// Character has no usable valueOf(String) — Settings.toType can't coerce; handle manually.
369 		if (effectiveType == Character.class) {
370 			if (resolved.length() == 1)
371 				return resolved.charAt(0);
372 			throw new BeanCreationException(
373 				"Could not coerce @Value('" + expression + "') resolving to '" + resolved + "' to char for " + siteDescription);
374 		}
375 
376 		try {
377 			return Settings.get().toType(resolved, effectiveType);
378 		} catch (RuntimeException e) {
379 			throw new BeanCreationException(
380 				"Could not coerce @Value('" + expression + "') resolving to '" + resolved + "' to "
381 					+ targetType.getName() + " for " + siteDescription,
382 				e);
383 		}
384 	}

386 	/**
387 	 * Returns the boxed counterpart of a primitive type, or {@code type} itself if not primitive.
388 	 *
389 	 * <p>
390 	 * {@link Settings#toType(String, Class)} uses reflection for {@code valueOf(String)} / {@code (String)}
391 	 * constructor lookup, which doesn't work for raw primitives ({@code int.class} etc.). Box them first.
392 	 *
393 	 * @param type The (possibly primitive) target type.
394 	 * @return The boxed equivalent, or {@code type} unchanged.
395 	 */
396 	private static Class<?> box(Class<?> type) {
397 		if (! type.isPrimitive())
398 			return type;
399 		if (type == int.class) return Integer.class;
400 		if (type == long.class) return Long.class;
401 		if (type == boolean.class) return Boolean.class;
402 		if (type == double.class) return Double.class;
403 		if (type == float.class) return Float.class;
404 		if (type == short.class) return Short.class;
405 		if (type == byte.class) return Byte.class;
406 		if (type == char.class) return Character.class;
407 		return type;  // void or unreachable
408 	}
