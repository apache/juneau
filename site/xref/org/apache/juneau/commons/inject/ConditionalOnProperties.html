11  package org.apache.juneau.commons.inject;
12  
13  import static java.lang.annotation.ElementType.*;
14  import static java.lang.annotation.RetentionPolicy.*;
15  
16  import java.lang.annotation.*;
17  
18  /**
19   * Container for repeatable {@link ConditionalOnProperty}.
20   */
21  @Documented
22  @Target({TYPE, METHOD, FIELD})
23  @Retention(RUNTIME)
24  public @interface ConditionalOnProperties {
25  	/**
26  	 * Nested property conditions.
27  	 *
28  	 * @return The nested conditions.
29  	 */
30  	ConditionalOnProperty[] value();
31  }
