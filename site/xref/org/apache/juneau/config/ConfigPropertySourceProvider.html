23  /**
24   * Auto-registers a classpath-default {@code juneau.cfg} (and the other names returned by
25   * {@link Config#getCandidateSystemDefaultConfigNames()}) as a {@link PropertySource} for the
26   * process-wide {@link Settings} singleton.
27   *
28   * <p>
29   * Discovered by {@link org.apache.juneau.commons.settings.Settings.Builder#useServiceLoader() ServiceLoader-based wiring} through a
30   * {@code META-INF/services/org.apache.juneau.commons.settings.PropertySourceProvider} entry
31   * shipped in this module.
32   *
33   * <p>
34   * Lookup semantics: the registered source defers calling {@link Config#getSystemDefault()} to the
35   * first {@code get(name)} invocation. This is deliberate — calling {@code getSystemDefault()} from
36   * inside {@link #create()} would create a static-initialization cycle because
37   * {@code Settings.<clinit>} fires the ServiceLoader at class-load time, and {@code Config}'s
38   * static initializers transitively reference {@link Settings} (through {@code Cache} / {@code SystemUtils.env(...)})
39   * which would re-enter the {@code Settings.<clinit>} block.
40   *
41   * <p>
42   * Lazy resolution makes the cycle a non-issue: by the time {@code @Value("${...}")} fires its first
43   * lookup, all participating classes are fully initialized.
44   *
45   * <h5 class='section'>Silence on missing config:</h5>
46   * <p>
47   * If no system-default {@link Config} is discoverable (no {@code juneau.cfg} / {@code default.cfg}
48   * / {@code application.cfg} / etc. on the classpath or working directory), {@code get(name)}
49   * returns {@link PropertyLookupResult#missing()} for every key.  No logging, no exception — many
50   * deployments have no classpath config and shouldn't see noise.
51   *
52   * <h5 class='section'>Ordering:</h5>
53   * <p>
54   * {@link #order()} returns {@code 100}, which is "low" in the {@link PropertySourceProvider}
55   * scheme — provider-discovered classpath defaults register near the bottom of the source list so
56   * that programmatic {@code Settings.get().addSource(...)} calls (e.g. per-microservice config or
57   * the Spring {@code Environment} bridge) shadow the classpath default.  {@link Settings} walks
58   * sources in reverse insertion order, so the most-recently-added source wins.
59   */
60  public class ConfigPropertySourceProvider implements PropertySourceProvider {
61  
62  	@Override
63  	public PropertySource create() {
64  		// Return a lazy wrapper.  Resolving Config.getSystemDefault() eagerly here would trigger
65  		// Config.<clinit>, which can transitively re-enter Settings.<clinit> through Cache /
66  		// SystemUtils.env(...) — the SPI registration is invoked from inside the Settings static init
67  		// block, so any cycle here surfaces as ExceptionInInitializerError.
68  		return new LazyConfigPropertySource();
69  	}
70  
71  	@Override
72  	public int order() {
73  		// Low ordering: classpath defaults register first, so any programmatic
74  		// addSource(...) added afterwards shadows the classpath defaults at lookup time
75  		// (Settings walks sources in reverse insertion order).
76  		return 100;
77  	}

79  	/**
80  	 * Defers {@link Config#getSystemDefault()} resolution to the first {@link #get(String)} call to
81  	 * avoid the static-initialization cycle described in the enclosing class' Javadoc.
82  	 *
83  	 * <p>
84  	 * Additionally protects against three runtime cycles at first-use time:
85  	 * <ol>
86  	 * 	<li><b>Bootstrap-static-init cycle (poisoning).</b> If this source is called from inside
87  	 * 		<i>any</i> JVM static-initializer chain that transitively loads {@code Config}, the
88  	 * 		downstream {@code Config.findSystemDefault()} → {@code FileStore.<clinit>} →
89  	 * 		{@code Context.Builder.build()} chain can fail with {@link NullPointerException}
90  	 * 		because {@code Context.<clinit>} hasn't finished yet. That failure <i>poisons</i>
91  	 * 		{@code FileStore} (every subsequent reference raises {@link NoClassDefFoundError}).
92  	 * 		To prevent the poisoning, the source detects when it's called from a static-initializer
93  	 * 		frame by walking the current stack for a {@code "<clinit>"} method name and short-circuits
94  	 * 		to {@link PropertyLookupResult#missing()} <i>without</i> memoizing — so the next call,
95  	 * 		from regular code, can resolve normally.
96  	 * 	<li><b>Reentrant resolution cycle.</b> {@link Config#findSystemDefault()} looks up the
97  	 * 		{@code "juneau.configFile"} system property via {@link Settings#get(String)}, which
98  	 * 		walks every registered source — including this one. Without a guard, that re-enters
99  	 * 		{@code Config.getSystemDefault()} on the same thread, which re-enters
100 	 * 		{@code findSystemDefault()}, which re-enters {@code Settings.get("juneau.configFile")},
101 	 * 		ad infinitum (a {@link StackOverflowError}). A {@link ThreadLocal} sentinel short-circuits
102 	 * 		any re-entrant lookup on the same thread to {@code missing()}.
103 	 * 	<li><b>Defensive throwable catch.</b> Any unexpected throwable (e.g. a different class
104 	 * 		loader's {@code Config} init failure that the {@code <clinit>}-sniff missed) is
105 	 * 		swallowed and treated as "not yet available" — {@code get(name)} returns
106 	 * 		{@link PropertyLookupResult#missing()} without memoizing the failure.
107 	 * </ol>
108 	 */
109 	@SuppressWarnings({
110 		"java:S1181" // Intentional broad Throwable catch — see Javadoc.
111 	})
112 	static final class LazyConfigPropertySource implements PropertySource {
113 
114 		// Thread-local re-entrance guard. Static so it covers both the lazy-init path and any
115 		// concurrent lookup that may also reach a Config.getSystemDefault() call (defensively).
116 		private static final ThreadLocal<Boolean> RESOLVING = ThreadLocal.withInitial(() -> Boolean.FALSE);
117 
118 		// Guarded by the synchronized block in get(...); volatile reads cover the fast-path check.
119 		private volatile boolean resolved;
120 		@SuppressWarnings({
121 			"java:S3077" // Publish-once reference: delegate is fully constructed before its single assignment inside the synchronized block, so volatile visibility is sufficient.
122 		})
123 		private volatile ConfigPropertySource delegate;
124 
125 		@Override
126 		@SuppressWarnings({
127 			"java:S3776" // Cognitive complexity acceptable for lazy config property source resolution
128 		})
129 		public PropertyLookupResult get(String name) {
130 
131 			// Re-entrant guard: Config.findSystemDefault() walks Settings.get("juneau.configFile"),
132 			// which round-trips through every source including this one. Short-circuit on
133 			// re-entry so the outer Config.getSystemDefault() can make progress.
134 			if (isTrue(RESOLVING.get()))
135 				return PropertyLookupResult.missing();
136 
137 			// Bootstrap guard: if we're being called from inside a <clinit> frame (e.g. another
138 			// class's static-initializer chain transitively invokes Settings.get(...) → this
139 			// source), don't attempt to resolve Config — doing so would force Config / FileStore /
140 			// Context class loading from inside an already-in-progress class init, which poisons
141 			// FileStore.<clinit> with a NoClassDefFoundError that survives the rest of the JVM
142 			// lifetime. Short-circuit without memoizing; the next non-<clinit> call resolves.
143 			if (! resolved && isInClinit())
144 				return PropertyLookupResult.missing();
145 
146 			if (! resolved) {
147 				synchronized (this) {
148 					if (! resolved) {
149 						// Re-check the <clinit> guard inside the lock — another thread may have
150 						// flipped resolved between the first check and the lock acquire.
151 						if (isInClinit())
152 							return PropertyLookupResult.missing();
153 
154 						Config cfg;
155 						RESOLVING.set(Boolean.TRUE);
156 						try {
157 							cfg = Config.getSystemDefault();
158 						} catch (@SuppressWarnings("unused") Throwable t) {
159 							// Defense-in-depth: a Config-side init failure the <clinit> sniff
160 							// missed. Return missing() without memoizing so the next lookup can
161 							// retry once Config init has completed.
162 							return PropertyLookupResult.missing();
163 						} finally {
164 							RESOLVING.remove();
165 						}
166 						// Silent when no juneau.cfg / default.cfg / etc. is discoverable — the
167 						// SPI consumer sees missing() for every key but no log noise.
168 						delegate = (cfg == null) ? null : new ConfigPropertySource(cfg);
169 						resolved = true;
170 					}
171 				}
172 			}
173 			return delegate == null ? PropertyLookupResult.missing() : delegate.get(name);
174 		}
175 
176 		/**
177 		 * Returns <jk>true</jk> if any frame in the current thread's call stack is a Java
178 		 * static-initializer ({@code "<clinit>"}). Used to detect — and avoid — the bootstrap-time
179 		 * cycle that would otherwise poison {@code FileStore.<clinit>}.
180 		 *
181 		 * <p>
182 		 * Cost: a single {@link Thread#getStackTrace()} call on the first lookup per source-thread
183 		 * pair. {@code resolved} is volatile so subsequent calls skip the stack walk entirely.
184 		 *
185 		 * @return <jk>true</jk> if the current call is being made from inside a {@code <clinit>}
186 		 * 	frame anywhere in the stack.
187 		 */
188 		private static boolean isInClinit() {
189 			var trace = Thread.currentThread().getStackTrace();
190 			for (var frame : trace) {
191 				if ("<clinit>".equals(frame.getMethodName()))
192 					return true;
193 			}
194 			return false;
195 		}
196 	}
