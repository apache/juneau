17  package org.apache.juneau.config.internal;
18  
19  import java.io.*;
20  import java.util.*;
21  
22  import org.apache.juneau.config.format.*;
23  import org.apache.juneau.config.store.*;
24  
25  /**
26   * Merges a base configuration with one or more profile overlays into a single internal-INI string.
27   *
28   * <p>
29   * This is the engine behind config <b>profiles</b> ({@code <name>-<profile>.cfg}/{@code .yml} overlays activated via
30   * {@code juneau.profiles.active}).  The base config is the foundation; each active profile's entries are layered on
31   * top in activation order so that:
32   * <ul>
33   * 	<li><b>profile wins over base</b> for any section/key the profile redefines, and
34   * 	<li><b>last-active-profile wins</b> when multiple active profiles redefine the same section/key.
35   * </ul>
36   *
37   * <p>
38   * Each input (base + profiles) is parsed through its own {@link ConfigFormat} (so a YAML profile is converted to the
39   * internal INI form like any other config), then the parsed maps are overlaid entry-by-entry and re-emitted as a
40   * single internal-INI string via {@link ConfigMap#asIniString()}.  Feeding that merged string back through the normal
41   * {@link ConfigMap} parse keeps format handling, variable resolution, and change-listener wiring identical to a
42   * single-file config &mdash; profiles are transparent to everything downstream of the store.
43   *
44   * @since 10.0.0
45   */
46  public final class ProfileMerge {
47  
48  	private ProfileMerge() {}
49  
50  	/**
51  	 * Merges base + profile contents into one internal-INI string (profile-wins, last-active-profile-wins).
52  	 *
53  	 * @param store The store the throwaway parse maps are associated with (used only for parse context; never written).
54  	 * @param baseName The base config name (for parse diagnostics).
55  	 * @param baseContents
56  	 * 	The raw base config contents (format-native).
57  	 * 	<br>Can be <jk>null</jk> (treated as empty contents).
58  	 * @param profiles
59  	 * 	Ordered profile overlays (activation order); each is a raw format-native contents string.
60  	 * 	<br>May be empty, but must not be <jk>null</jk>.
61  	 * @param format
62  	 * 	The config format used to parse every input.
63  	 * 	<br>Can be <jk>null</jk> (defaults to {@link IniConfigFormat}).
64  	 * @return The merged contents as an internal-INI string.
65  	 * @throws IOException If any input fails to parse.
66  	 */
67  	public static String merge(ConfigStore store, String baseName, String baseContents, List<String> profiles, ConfigFormat format) throws IOException {
68  		var f = format == null ? IniConfigFormat.INSTANCE : format;
69  		// Parse the base into a throwaway map; this becomes the accumulator we overlay onto.
70  		var merged = new ConfigMap(store, baseName, baseContents == null ? "" : baseContents, f);
71  		for (var p : profiles) {
72  			if (p == null || p.isBlank())
73  				continue;
74  			var overlay = new ConfigMap(store, baseName, p, f);
75  			applyOverlay(merged, overlay);
76  		}
77  		return merged.asIniString();
78  	}
79  
80  	/**
81  	 * Overlays every section/key entry from {@code overlay} onto {@code base} (overlay wins).
82  	 */
83  	private static void applyOverlay(ConfigMap base, ConfigMap overlay) {
84  		for (var section : overlay.getSections()) {
85  			// Ensure the section exists on the base (preserving the overlay's section pre-lines when it is new).
86  			if (! base.hasSection(section))
87  				base.setSection(section, overlay.getPreLines(section));
88  			for (var key : overlay.getKeys(section)) {
89  				var e = overlay.getEntry(section, key);
90  				if (e != null)  // HTT: getKeys() only returns keys with entries, so getEntry is never null here.
91  					base.setEntry(section, key, e.getValue(), e.getModifiers(), e.getComment(), e.getPreLines());
92  			}
93  		}
94  	}
95  }
