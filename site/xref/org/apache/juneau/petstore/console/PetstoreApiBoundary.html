17  package org.apache.juneau.petstore.console;
18  
19  import java.io.*;
20  import java.util.*;
21  
22  import org.apache.juneau.commons.settings.*;
23  import org.apache.juneau.rest.server.filter.*;
24  
25  import jakarta.servlet.*;
26  import jakarta.servlet.http.*;
27  
28  /**
29   * The {@code /petstore} API's CSRF boundary: a {@link RequestBoundaryFilter} over the framework's {@link CsrfBoundary},
30   * applied to {@code /petstore} and everything under it.
31   *
32   * <p>
33   * A browser write to the API must come from the page's own origin and carry this boundary's token in the
34   * {@code X-Csrf-Token} header.  The token travels in the header only, so every HTML {@code <form>} POST to
35   * the API is refused, the application's own included.  Requests are judged on their effective method, so
36   * {@code GET /petstore/pets/1?method=DELETE} counts as a write.  Reads are allowed, and so are non-browser writes
37   * (no {@code Origin} and no {@code Sec-Fetch-Site}) that a browser could not have sent cross-site without a CORS
38   * preflight: JSON, XML or UON bodies, and {@code PUT}, {@code DELETE} or {@code PATCH}.  A non-browser urlencoded,
39   * multipart or {@code text/plain} {@code POST} is refused.  Every refusal names its reason in the
40   * {@value RequestBoundaryFilter#REJECTION_HEADER} response header.
41   *
42   * <h5 class='section'>The token</h5>
43   * <p>
44   * The token is per process ({@link CsrfTokens#perProcess()}): the petstore has no login session, and the
45   * {@code Origin} check does the real work.  A restart invalidates the token on every open page, so reload them.  A
46   * token that leaks through a page stays valid until the restart.
47   *
48   * <p>
49   * This token is not the console's.  The console boundary ({@link PetstoreConsoleBoundary}) has its own, and each
50   * boundary refuses the other's.  The console's one write to the API, its photo upload, gets this boundary's token
51   * from the console boundary, which publishes {@link #tokenFor(HttpServletRequest)} under {@link #TOKEN_ATTRIBUTE}
52   * on every console request it allows; the page's script sends it in that header.  So a runner must
53   * give the console boundary the same instance it mounts here: a token from any other instance is refused.
54   *
55   * <p>
56   * The console page shell marks a token-bearing page {@code no-store}, and when the response is already committed it
57   * strips only its own token attributes.  So on the rare page whose buffer was flushed before the shell ran, this
58   * boundary's token can still be emitted, possibly without {@code no-store}.  That is accepted: this token is only
59   * ever published on console pages, which carry the console's token too, so on every normal path the page is already
60   * marked.
61   *
62   * <h5 class='section'>The host</h5>
63   * <p>
64   * The request's own origin is its scheme plus its {@code Host} header; forwarded headers are never read.  By default
65   * any {@code Host} is answered, which suits a loopback developer setup.  On a host served over plain HTTP, set
66   * {@value #ALLOWED_HOSTS_PROPERTY} (a system property, the environment, or any registered setting source) to the
67   * comma-separated host names the browser uses, without ports.  Every request to the API under any other {@code Host},
68   * reads included, is then answered with {@code 421}: this is the DNS-rebinding defence.  The setting is read once,
69   * when the boundary is constructed, and a malformed entry fails construction rather than being ignored.
70   *
71   * <p>
72   * Behind a TLS-terminating proxy the browser's {@code https} origin does not match the {@code http} request the
73   * container sees, so writes are refused.  Such a deployment needs the container's forwarded-header support and the
74   * framework boundary's {@link org.apache.juneau.rest.server.filter.CsrfBoundary.OriginSource#CONTAINER CONTAINER} origin source, which this demo does not
75   * configure.
76   *
77   * <h5 class='section'>Example:</h5>
78   * <p>
79   * In a Juneau microservice, contribute a <c>ServletFilterRegistration</c> <c>@Bean</c> just after the console
80   * boundary's, sharing the one instance with it:
81   * <p class='bjava'>
82   * 	<jk>private final</jk> PetstoreApiBoundary <jf>apiBoundary</jf> = <jk>new</jk> PetstoreApiBoundary();
83   *
84   * 	<ja>@Bean</ja>
85   * 	<jk>public</jk> ServletFilterRegistration apiBoundary() {
86   * 		<jk>return</jk> ServletFilterRegistration.<jsm>create</jsm>(<jf>apiBoundary</jf>)
87   * 			.urlPatterns(PetstoreApiBoundary.<jsf>URL_PATTERN</jsf>)
88   * 			.dispatcherTypes(DispatcherType.<jsf>REQUEST</jsf>)
89   * 			.order(ServletFilterRegistration.<jsf>HIGHEST_PRECEDENCE</jsf> + 10)
90   * 			.build();
91   * 	}
92   * </p>
93   * <p>
94   * and build the console boundary with {@code new PetstoreConsoleBoundary(apiBoundary)}.  Register it for
95   * {@code REQUEST} dispatches only: an internal forward or error dispatch is not a new request from the browser.
96   */
97  public class PetstoreApiBoundary implements Filter {
98  
99  	/** The covered path: the API resource's mount point. */
100 	public static final String PATH = "/petstore";
101 
102 	/** The servlet URL pattern the boundary is registered under.  It matches {@link #PATH} itself and every path under it. */
103 	public static final String URL_PATTERN = PATH + "/*";
104 
105 	/**
106 	 * Setting (system property, environment, or any registered source) listing the host names the API answers to:
107 	 * comma-separated, no ports.  Unset or blank, any host is answered.
108 	 */
109 	public static final String ALLOWED_HOSTS_PROPERTY = "petstore.api.allowedHosts";
110 
111 	/**
112 	 * Request attribute under which an allowed console request carries a {@link String}: a freshly masked token for
113 	 * this boundary, for a console page to stamp on the form that writes to the API.
114 	 */
115 	public static final String TOKEN_ATTRIBUTE = "org.apache.juneau.petstore.apiCsrfToken";
116 
117 	private final CsrfBoundary boundary;
118 	private final RequestBoundaryFilter filter;
119 
120 	/**
121 	 * Constructor.
122 	 *
123 	 * <p>
124 	 * Reads {@value #ALLOWED_HOSTS_PROPERTY} and creates the boundary's per-process token.
125 	 *
126 	 * @throws IllegalArgumentException If {@value #ALLOWED_HOSTS_PROPERTY} holds an entry that is blank or not a bare
127 	 * 	host name.
128 	 */
129 	public PetstoreApiBoundary() {
130 		boundary = CsrfBoundary.create()
131 			.tokens(CsrfTokens.perProcess())
132 			.allowedHosts(allowedHosts(Settings.get().get(ALLOWED_HOSTS_PROPERTY).orElse(null)))
133 			.build();
134 		filter = new RequestBoundaryFilter(boundary);
135 	}
136 
137 	/**
138 	 * The host names in an {@value #ALLOWED_HOSTS_PROPERTY} setting.
139 	 *
140 	 * @param setting The setting.  Can be <jk>null</jk> or blank (not set).
141 	 * @return The comma-separated entries, trimmed, blank ones included so the boundary refuses them; empty when the
142 	 * 	setting is not set.
143 	 */
144 	private static String[] allowedHosts(String setting) {
145 		if (setting == null || setting.isBlank())
146 			return new String[0];
147 		return Arrays.stream(setting.split(",", -1)).map(String::trim).toArray(String[]::new);
148 	}
149 
150 	/**
151 	 * A freshly masked token for a page served to this request to embed.
152 	 *
153 	 * <p>
154 	 * Never creates a session.  The value is different on every call, and every value verifies on this instance only.
155 	 *
156 	 * @param req The request whose response will embed the token.  Must not be <jk>null</jk>.
157 	 * @return 128 lower-case hex characters.
158 	 */
159 	public String tokenFor(HttpServletRequest req) {
160 		return boundary.tokenFor(req);
161 	}
162 
163 	/**
164 	 * Applies the boundary: a refused request is answered here and never reaches the API.
165 	 */
166 	@Override /* Filter */
167 	public void doFilter(ServletRequest req, ServletResponse res, FilterChain chain) throws IOException, ServletException {
168 		filter.doFilter(req, res, chain);
169 	}
170 }
