32  /**
33   * The groom-pet demo runs behind the Jobs page's terminal cards.
34   *
35   * <p>
36   * A run writes about 60 ANSI-styled lines to a {@link MemoryTerminalSource}, one step per tick: INFO and FINE lines
37   * (FINE is dim), a WARNING, {@code check}/{@code cross} marks, a progress strip of coloured blocks, a multi-line
38   * SEVERE entry, bold marker lines, and a canned ANSI transcript written as-is.  Each tick also appends one ANSI line to
39   * a temp file, read through one {@link FileTerminalSource} per run.  A verbose run first writes {@value #VERBOSE_LINES}
40   * FINE lines.
41   *
42   * <p>
43   * Run ids are random 128-bit hex strings, so a terminal can be read only by someone who was given its id.
44   *
45   * <h5 class='section'>Example:</h5>
46   * <p class='bjava'>
47   * 	<jk>try</jk> (<jk>var</jk> <jv>runs</jv> = <jk>new</jk> GroomRuns(650)) {
48   * 		<jk>var</jk> <jv>run</jv> = <jv>runs</jv>.start(<js>"Rex"</js>, <jk>false</jk>, <jk>null</jk>);
49   * 		<jk>var</jk> <jv>source</jv> = <jv>runs</jv>.source(<jv>run</jv>.id()).orElseThrow();
50   * 	}
51   * </p>
52   */
53  @SuppressWarnings({
54  	"java:S1192", // The script's literals are demo text; constants would obscure it.
55  	"resource" // Each Run owns its file-backed source, closed by drop()/close(); seeded() hands the new GroomRuns to its caller.
56  })
57  public final class GroomRuns implements AutoCloseable {

62  	/** The default step period; 53 steps take about 35 seconds. */
63  	public static final long DEFAULT_PERIOD_MS = 650;

65  	/** Suffix of the log id that selects a run's file-backed source. */
66  	public static final String FILE_SUFFIX = "-file";

68  	/** The in-memory terminal's ring capacity in bytes; it holds a verbose run whole. */
69  	static final int RING_BYTES = 1 << 20;
70  
71  	/** The terminal size, in columns and rows, of both of a run's terminals. */
72  	static final int COLS = 100;
73  
74  	/** The terminal height, in rows, of both of a run's terminals. */
75  	static final int ROWS = 24;
76  
77  	private static final String RESET = "\u001b[0m";
78  	private static final String BOLD = "\u001b[1m";
79  	private static final String DIM = "\u001b[2m";
80  	private static final String RED = "\u001b[31m";
81  	private static final String GREEN = "\u001b[32m";
82  	private static final String YELLOW = "\u001b[33m";
83  
84  	/** How many runs are kept; the oldest is dropped, and its file deleted, beyond this. */
85  	static final int MAX_RUNS = 20;
86  
87  	/** FINE lines a verbose run writes up front. */
88  	static final int VERBOSE_LINES = 8000;
89  
90  	/** Wash passes; the progress strip has one block per pass. */
91  	static final int WASH_PASSES = 10;

93  	/** The canned ANSI transcript: an OSC title (stripped), bold, colours, and a carriage-return counter. */
94  	static final List<String> TRANSCRIPT = List.of(
95  		"\u001b]0;groom-pet\u0007\u001b[1mgroomer 2.1\u001b[0m starting",
96  		"\u001b[34mfetch\u001b[0m shampoo \u001b[32mok\u001b[0m",
97  		"\u001b[34mfetch\u001b[0m towels \u001b[32mok\u001b[0m",
98  		"drying  10%\rdrying  40%\rdrying  70%\rdrying 100%",
99  		"\u001b[33mwarn\u001b[0m the clippers are dull",
100 		"\u001b[1;31merror\u001b[0m no bow in stock; using a ribbon",
101 		"\u001b[1;32mready\u001b[0m"

104 	/** One step of the script. */
105 	@FunctionalInterface
106 	interface Step {
107 		void run(Run r) throws IOException;
108 	}
109 
110 	/** The script, one step per tick. */
111 	static final List<Step> SCRIPT = script();

113 	/** One groom run. */
114 	public static final class Run {
115 		final String id;
116 		final String pet;
117 		final boolean verbose;
118 		final Instant submitted = Instant.now();
119 		final MemoryTerminalSource terminal = MemoryTerminalSource.create(RING_BYTES, COLS, ROWS);
120 		final Path file;
121 		final Path sizeFile;
122 		final AtomicBoolean fileDone = new AtomicBoolean();
123 		final FileTerminalSource fileSource;
124 		final AtomicInteger lines = new AtomicInteger();
125 		final AtomicInteger step = new AtomicInteger();
126 		volatile boolean done;
127 		final AtomicReference<ScheduledFuture<?>> task = new AtomicReference<>();
128 		volatile String key;
129 
130 		Run(String id, String pet, boolean verbose, Path file) {
131 			this.id = id;
132 			this.pet = pet;
133 			this.verbose = verbose;
134 			this.file = file;
135 			sizeFile = file.resolveSibling(file.getFileName() + ".size");
136 			fileSource = FileTerminalSource.create(file).terminal(fileDone::get).build();
137 		}
138 
139 		/** @return The run id, which is also its in-memory terminal id. */
140 		public String id() { return id; }
141 
142 		/** @return The pet name. */
143 		public String pet() { return pet; }
144 
145 		/** @return The in-memory terminal. */
146 		public MemoryTerminalSource terminal() { return terminal; }
147 
148 		/** @return The cached file-backed source over this run's transcript. */
149 		public FileTerminalSource fileSource() { return fileSource; }
150 
151 		/** @return The temp file the run appends to. */
152 		public Path file() { return file; }
153 
154 		/** @return Whether the run has written its last line. */
155 		public boolean done() { return done; }
156 
157 		/** @return The Groom runs table row. */
158 		public JsonMap row() {
159 			return JsonMap.of("id", id, "pet", pet, "submitted", submitted.toString(), "state", done ? "DONE" : "RUNNING", "lines", lines.get(), "verbose", verbose);
160 		}
161 
162 		void line(String text) {
163 			var b = (text + "\r\n").getBytes(UTF_8);
164 			terminal.write(b, 0, b.length);
165 			lines.incrementAndGet();
166 		}
167 
168 		void raw(String text) {
169 			for (var l : text.split("\n"))
170 				line(l);
171 		}
172 	}

174 	private final long periodMs;
175 	private final Map<String,Run> runs = new ConcurrentHashMap<>();
176 	private final Map<String,String> keys = new ConcurrentHashMap<>();
177 	private final Deque<Run> order = new ArrayDeque<>(); // Newest first; guarded by itself.
178 	private final ScheduledExecutorService ticker = Executors.newSingleThreadScheduledExecutor(r -> {
179 		var t = new Thread(r, "petstore-groom");
180 		t.setDaemon(true);
181 		return t;
182 	});

184 	/**
185 	 * Constructor.
186 	 *
187 	 * @param periodMs Milliseconds between steps, at least 1.
188 	 */
189 	public GroomRuns(long periodMs) {
190 		if (periodMs < 1)
191 			throw new IllegalArgumentException(f("periodMs must be at least 1; got %s", periodMs));
192 		this.periodMs = periodMs;
193 	}

195 	/**
196 	 * Creates the runs and starts one verbose run, so the Jobs page always has a log to show.
197 	 *
198 	 * @param periodMs Milliseconds between steps.
199 	 * @return The runs.
200 	 */
201 	public static GroomRuns seeded(long periodMs) {
202 		var g = new GroomRuns(periodMs);
203 		try {
204 			g.start("Biscuit", true, null);
205 		} catch (IOException e) {
206 			g.close();
207 			throw new UncheckedIOException(e);
208 		}
209 		return g;
210 	}

212 	/**
213 	 * Starts a run, or returns the run already started with the same idempotency key.
214 	 *
215 	 * @param pet The pet name; blank means {@code "Rex"}.
216 	 * @param verbose Whether to write {@value #VERBOSE_LINES} FINE lines first.
217 	 * @param idempotencyKey The client's key, or <jk>null</jk>.
218 	 * @return The run.
219 	 * @throws IOException If the temp file cannot be created.
220 	 */
221 	@SuppressWarnings({
222 		"java:S5443" // Files.createTempFile creates the file owner-only (rw-------) on POSIX
223 	})
224 	public Run start(String pet, boolean verbose, String idempotencyKey) throws IOException {
225 		synchronized (order) {
226 			if (idempotencyKey != null) {
227 				var prior = keys.get(idempotencyKey);
228 				if (prior != null && runs.containsKey(prior))
229 					return runs.get(prior);
230 			}
231 			var name = pet == null || pet.isBlank() ? "Rex" : pet.strip();
232 			if (name.length() > 40)
233 				name = name.substring(0, 40);
234 			var file = Files.createTempFile("petstore-groom-", ".log");
235 			file.toFile().deleteOnExit();
236 			var r = new Run(UUID.randomUUID().toString().replace("-", ""), name, verbose, file);
237 			Files.writeString(r.sizeFile, "{\"cols\":" + COLS + ",\"rows\":" + ROWS + "}", UTF_8);
238 			r.sizeFile.toFile().deleteOnExit();
239 			if (verbose)
240 				verboseLines().forEach(r::line);
241 			runs.put(r.id, r);
242 			order.addFirst(r);
243 			if (idempotencyKey != null) {
244 				r.key = idempotencyKey;
245 				keys.put(idempotencyKey, r.id);
246 			}
247 			while (order.size() > MAX_RUNS)
248 				drop(order.removeLast());
249 			r.task.set(ticker.scheduleAtFixedRate(() -> tick(r), periodMs, periodMs, TimeUnit.MILLISECONDS));
250 			return r;
251 		}
252 	}

254 	/**
255 	 * @param id A run id, or <jk>null</jk>.
256 	 * @return The run.
257 	 */
258 	public Optional<Run> get(String id) {
259 		return id == null ? Optional.empty() : Optional.ofNullable(runs.get(id));
260 	}

262 	/** @return The newest run, or <jk>null</jk> after {@link #close()}. */
263 	public Run latest() {
264 		synchronized (order) {
265 			return order.peekFirst();
266 		}
267 	}

269 	/** @return The runs, newest first. */
270 	public List<Run> list() {
271 		synchronized (order) {
272 			return List.copyOf(order);
273 		}

276 	/**
277 	 * Resolves a terminal id: {@code <runId>} is the in-memory terminal, {@code <runId>-file} the file source.
278 	 *
279 	 * @param terminalId The terminal id from the URL.
280 	 * @return The source, or empty for an unknown id.
281 	 */
282 	public Optional<TerminalSource> source(String terminalId) {
283 		if (terminalId == null)
284 			return Optional.empty();
285 		if (terminalId.endsWith(FILE_SUFFIX))
286 			return get(terminalId.substring(0, terminalId.length() - FILE_SUFFIX.length())).<TerminalSource>map(Run::fileSource);
287 		return get(terminalId).<TerminalSource>map(Run::terminal);
288 	}
289 
290 	/** Stops the ticker and deletes every run's file. */
291 	@Override
292 	public void close() {
293 		ticker.shutdownNow();
294 		synchronized (order) {
295 			order.forEach(this::drop);
296 			order.clear();
297 		}
298 	}
299 
300 	void tick(Run r) {
301 		var i = r.step.getAndIncrement();
302 		if (i >= SCRIPT.size())
303 			return;
304 		try {
305 			SCRIPT.get(i).run(r);
306 			Files.writeString(r.file, fileLine(i) + "\r\n", UTF_8, StandardOpenOption.APPEND);
307 		} catch (IOException | RuntimeException e) { // A failed step must not cancel the schedule silently.
308 			r.line(BOLD + RED + f("Step %s failed: %s", i + 1, e.getMessage()) + RESET);
309 		}
310 		if (i == SCRIPT.size() - 1)
311 			finish(r);
312 	}
313 
314 	private static void finish(Run r) {
315 		r.done = true;
316 		r.terminal.finish();
317 		r.fileDone.set(true);
318 		var t = r.task.get();
319 		if (t != null)
320 			t.cancel(false);
321 	}
322 
323 	private void drop(Run r) {
324 		var t = r.task.get();
325 		if (t != null)
326 			t.cancel(false);
327 		runs.remove(r.id);
328 		if (r.key != null)
329 			keys.remove(r.key);
330 		try {
331 			Files.deleteIfExists(r.file);
332 			Files.deleteIfExists(r.sizeFile);
333 		} catch (IOException e) {
334 			// Best effort: the file is also marked deleteOnExit.
335 		}
336 	}
337 
338 	private static String fileLine(int i) {
339 		if (i < TRANSCRIPT.size())
340 			return TRANSCRIPT.get(i);
341 		if (i == SCRIPT.size() - 1)
342 			return "\u001b[1;32mgroom finished\u001b[0m";
343 		return f("\u001b[1m[%02d/%s]\u001b[0m step done", i + 1, SCRIPT.size());
344 	}

346 	private static List<String> verboseLines() {
347 		var out = new ArrayList<String>(VERBOSE_LINES);
348 		for (var i = 1; i <= VERBOSE_LINES; i++)
349 			out.add(DIM + "verbose: inspected coat patch " + i + " of " + VERBOSE_LINES + RESET);
350 		return out;
351 	}

353 	static String progress() {
354 		var sb = new StringBuilder("Wash passes ");
355 		for (var i = 0; i < WASH_PASSES; i++)
356 			sb.append(i < WASH_PASSES - 2 ? GREEN : YELLOW).append("\u25a0").append(RESET);
357 		return sb.append(" (the last two need another rinse)").toString();
358 	}

360 	private static String info(String text) { return text; }
361 	private static String fine(String text) { return DIM + text + RESET; }
362 	private static String marker(String text) { return BOLD + text + RESET; }
363 	private static String mark(boolean ok, String text) { return (ok ? GREEN + "\u2713 " : RED + "\u2717 ") + text + RESET; }

365 	private static List<Step> script() {
366 		var s = new ArrayList<Step>();
367 		s.add(r -> r.line(marker("Grooming " + r.pet)));
368 		s.add(r -> r.line(fine("Reading the grooming profile")));
369 		for (var tool : List.of("brush", "comb", "shampoo", "conditioner", "towel", "dryer", "clippers", "nail file"))
370 			s.add(r -> r.line(fine("Checked out the " + tool)));
371 		s.add(r -> r.line(mark(true, "All tools ready")));
372 		s.add(r -> r.line(marker("Wash")));
373 		for (var i = 1; i <= WASH_PASSES; i++) {
374 			var n = i;
375 			s.add(r -> r.line(info(f("Wash pass %s of %s", n, WASH_PASSES))));
376 		}
377 		s.add(r -> r.line(YELLOW + "The water is 2 degrees cooler than the profile asks for" + RESET));
378 		s.add(r -> r.line(progress()));
379 		s.add(r -> r.line(marker("Dry")));
380 		s.add(r -> r.raw(String.join("\n", TRANSCRIPT)));
381 		for (var i = 1; i <= 10; i++) {
382 			var n = i;
383 			s.add(r -> r.line(info(f("Brushed section %s of 10", n))));
384 		}
385 		s.add(r -> r.raw(BOLD + RED + String.join("\n",
386 			"Dryer overheated; switching to the spare",
387 			"    at Dryer.heat(Dryer.java:42)",
388 			"    at Groomer.dry(Groomer.java:17)",
389 			"    at GroomRun.step(GroomRun.java:88)") + RESET));
390 		s.add(r -> r.line(mark(false, "Bow tie: none in stock")));
391 		s.add(r -> r.line(marker("Trim")));
392 		for (var i = 1; i <= 12; i++) {
393 			var n = i;
394 			s.add(r -> r.line(fine(f("Trimmed nail %s of 12", n))));
395 		}
396 		s.add(r -> r.line(mark(true, r.pet + " is groomed")));
397 		return List.copyOf(s);
398 	}
