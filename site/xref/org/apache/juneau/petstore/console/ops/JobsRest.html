43  /**
44   * Jobs: the console's async-job reference.  A restock submits once per idempotency key, answers
45   * 202 with an {@link AsyncJobRef}, and the views runtime follows its {@code streamUrl} until the job settles.
46   *
47   * <p>
48   * The registry de-duplicates on the idempotency key itself ({@link AsyncJobRegistry#tryCreate(IdempotencyKey)}), so a
49   * resubmit with the same key returns the same job and this resource keeps no key map of its own.
50   *
51   * <p>
52   * The page's {@code job-log} card posts the same restock in its flat form and follows the job's stream.  A restock
53   * with {@code slowMs} waits that long before it touches the store (cancel it then and nothing changes), and every
54   * restock holds for {@code holdMs} (a second by default) after it has added its pets (cancel it then and the result
55   * says the change was made).
56   * The registry's hard timeout is 30 s, or whatever a {@link HardTimeout} bean on the host says; a {@link GroomPeriod}
57   * bean sets the groom step and a {@link SpeciesHold} bean the species-row hold.
58   *
59   * <p>
60   * The seeded groom run is created on the first groom request rather than at construction, so a {@link GroomPeriod} bean
61   * (injected after construction) can set its pace; every groom endpoint goes through it, so the page sees the same data.
62   *
63   * <p>
64   * The page also hosts the terminal demo: a "groom pet" job ({@link GroomRuns}) whose output is served by
65   * {@link TerminalMixin} and shown two ways &mdash; a FreeMarker terminal card over the in-memory output, and a second
66   * card over the same run's file-backed transcript.
67   *
68   * <h5 class='section'>Example:</h5>
69   * <p class='bjava'>
70   * 	<jc>// POST /console/ops/jobs/restock {DOG:2, slowMs:3000, holdMs:1000, idempotencyKey:'7f3c...'}</jc>
71   * 	<jk>var</jk> <jv>body</jv> = Restock.<jsm>of</jsm>(<jv>json</jv>);
72   * 	<jk>var</jk> <jv>job</jv> = asyncJobRegistry().tryCreate(IdempotencyKey.<jsm>of</jsm>(<jv>body</jv>.idempotencyKey(), <js>"restock"</js>, <js>"restock"</js>)).orElse(<jk>null</jk>);
73   * 	<jk>if</jk> (<jv>job</jv> == <jk>null</jk>)
74   * 		<jk>return</jk> ActionResult.<jsm>refusal</jsm>(<js>"app:job-limit"</js>);  <jc>// answered 429</jc>
75   * 	<jv>pool</jv>.execute(() -&gt; run(<jv>job</jv>, <jv>body</jv>, () -&gt; store().restock(<jv>body</jv>.perSpecies(), <js>"job:restock"</js>)));
76   * 	<jk>return</jk> AsyncJobRef.<jsm>of</jsm>(<jv>job</jv>);  <jc>// answered 202</jc>
77   * </p>
78   */

213 	 * The restock request.
214 	 *
215 	 * <h5 class='section'>Example:</h5>
216 	 * <p class='bjava'>
217 	 * 	<jc>// Wire form: {perSpecies:{DOG:2, CAT:1}, idempotencyKey:'7f3c...'}</jc>
218 	 * 	<jk>var</jk> <jv>r</jv> = <jk>new</jk> Restock(Map.<jsm>of</jsm>(Species.<jsf>DOG</jsf>, 2), <js>"7f3c"</js>, 0, 1000);
219 	 * </p>
220 	 *
221 	 * @param perSpecies How many pets of each species to add (1-50 each).
222 	 * @param idempotencyKey The client-minted key; resubmits with the same key return the same job.
223 	 * @param slowMs How long to wait before restocking, in whole milliseconds (0-{@value JobsRest#MAX_SLOW_MS}), so a demo can
224 	 * 	cancel before the effect.
225 	 * @param holdMs How long to hold after restocking, in whole milliseconds (0-{@value JobsRest#MAX_HOLD_MS}, default
226 	 * 	{@value JobsRest#DEFAULT_HOLD_MS}), so a demo can cancel after the effect.
227 	 */
228 	public record Restock(Map<Species,Integer> perSpecies, String idempotencyKey, int slowMs, int holdMs) {
229 
230 		/**
231 		 * Reads a restock from any of its wire shapes: the API form {@code {perSpecies:{DOG:2}, idempotencyKey}}, the
232 		 * dialog form the views runtime posts, {@code {action, targetId, idempotencyKey, fields:{DOG:"2"}}}, or the
233 		 * job-log card's flat form {@code {DOG:2, slowMs:0, idempotencyKey}}.  In the dialog and flat forms a blank,
234 		 * <jk>null</jk> or {@code 0} count means "none of that species".  Any form may carry {@code slowMs} and
235 		 * {@code holdMs}; each must be a whole number of milliseconds, so a decimal such as {@code 1.5} is rejected
236 		 * rather than truncated.
237 		 *
238 		 * @param body The request body.
239 		 * @return The restock.
240 		 * @throws BadRequest If a species is unknown, a count is not a whole number, {@code slowMs} is not a whole
241 		 * 	number 0-{@value JobsRest#MAX_SLOW_MS}, or {@code holdMs} is not a whole number 0-{@value JobsRest#MAX_HOLD_MS}.
242 		 */
243 		public static Restock of(JsonMap body) {
244 			var per = new LinkedHashMap<Species,Integer>();
245 			try {
246 				var api = body.getMap("perSpecies");
247 				var form = body.getMap("fields");
248 				if (api != null) {
249 					api.forEach((k, v) -> per.put(Species.valueOf(k), v == null ? null : Integer.valueOf(String.valueOf(v))));
250 				} else if (form != null) {
251 					form.forEach((k, v) -> putCount(per, Species.valueOf(k), v));
252 				} else {
253 					for (var s : Species.values())
254 						putCount(per, s, body.get(s.name()));
255 				}
256 			} catch (RuntimeException e) { // NumberFormat/IllegalArgument for bad counts or species; a conversion error when perSpecies/fields is not a map.
257 				throw new BadRequest("Restock counts must be whole numbers for known species");
258 			}
259 			return new Restock(per, body.getString("idempotencyKey"), millis(body, "slowMs", 0, MAX_SLOW_MS), millis(body, "holdMs", DEFAULT_HOLD_MS, MAX_HOLD_MS));
260 		}
261 
262 		private static void putCount(Map<Species,Integer> per, Species s, Object v) {
263 			var x = v == null ? "" : String.valueOf(v).trim();
264 			if (! x.isEmpty() && ! "0".equals(x))
265 				per.put(s, Integer.valueOf(x));
266 		}
267 
268 		private static int millis(JsonMap body, String name, int dflt, int max) {
269 			var v = body.get(name);
270 			if (v == null || String.valueOf(v).isBlank())
271 				return dflt;
272 			try {
273 				var n = Integer.parseInt(String.valueOf(v).trim());
274 				if (n >= 0 && n <= max)
275 					return n;
276 			} catch (NumberFormatException e) {
277 				// Not a whole number (a decimal included): falls through to the 400 below.
278 			}
279 			throw new BadRequest("Invalid %s '%s'; must be a whole number 0-%s", name, v, max);
280 		}
281 	}

284 	private GroomRuns groomRuns() {
285 		var g = groomRef.get();
286 		if (g == null) {
287 			var n = GroomRuns.seeded(groomPeriod != null ? groomPeriod.value().toMillis() : Long.getLong(GroomRuns.PERIOD_PROPERTY, GroomRuns.DEFAULT_PERIOD_MS));
288 			if (groomRef.compareAndSet(null, n)) {
289 				g = n;
290 			} else {
291 				n.close();
292 				g = groomRef.get();
293 			}
294 		}
295 		return g;
296 	}

304 	@Override /* AsyncJobsMixin */
305 	public AsyncJobRegistry asyncJobRegistry() {
306 		var j = jobs.get();
307 		if (j == null) {
308 			synchronized (this) {
309 				j = jobs.get();
310 				if (j == null) {
311 					j = new AsyncJobRegistry(hardTimeout == null ? DEFAULT_HARD_TIMEOUT : hardTimeout.value());
312 					jobs.set(j);
313 				}
314 			}
315 		}
316 		return j;

392 	/**
393 	 * Submits a restock.
394 	 *
395 	 * @param req The request.
396 	 * @param res The response (202).
397 	 * @param json The restock body (see {@link Restock#of(JsonMap)}).
398 	 * @return The job ref (202), or a 429 {@code app:job-limit} refusal whose message names the running count, so a
399 	 * 	JSON client such as the job-log card can show it.
400 	 * @throws BadRequest On a missing key, an out-of-range count, or a bad {@code slowMs} or {@code holdMs} (see
401 	 * 	{@link Restock#of(JsonMap)}).
402 	 */
403 	@RestPost(path="/restock")
404 	public Object restock(RestRequest req, RestResponse res, @Content JsonMap json) {
405 		var actor = ConsoleWrites.actor(req);
406 		var body = Restock.of(json);
407 		var k = body.idempotencyKey();
408 		if (ib(k))
409 			throw new BadRequest("Missing 'idempotencyKey'");
410 		var key = IdempotencyKey.of(k, ACTION, k.equals(json.getString("targetId")) ? k : ACTION);
411 		if (body.perSpecies().isEmpty())
412 			throw new BadRequest("Nothing to restock");
413 		body.perSpecies().forEach((s, n) -> {
414 			if (n == null || n < 1 || n > 50)
415 				throw new BadRequest("Count for '%s' must be between 1 and 50", s);
416 		});
417 		var limit = Integer.getInteger(LIMIT_PROPERTY, 2);
418 		var registry = asyncJobRegistry();
419 		AsyncJob job;
420 		synchronized (submitted) {
421 			// Resolve the idempotency key first: a replay returns its existing job and never counts against the limit.
422 			var prior = submitted.stream().filter(x -> k.equals(x.getString("idempotencyKey"))).findFirst();
423 			job = prior.flatMap(x -> registry.get(x.getString("jobId"))).orElse(null);
424 			if (job == null) {
425 				prior.ifPresent(submitted::remove); // The job aged out of the registry; this is a fresh submission.
426 				var created = registry.runningCount() >= limit ? null : registry.tryCreate(key).orElse(null);
427 				if (created == null)
428 					return limitReached(res, registry.runningCount());
429 				job = created;
430 				submitted.addFirst(JsonMap.of("jobId", created.id(), "submitted", created.createdAt().toString(), "request", body.perSpecies(), "idempotencyKey", k));
431 				while (submitted.size() > MAX_SUBMITTED)
432 					submitted.removeLast();
433 				pool.execute(() -> run(created, body, () -> store().restock(body.perSpecies(), "job:restock:" + actor)));
434 			}
435 		}
436 		res.setStatus(202);
437 		return AsyncJobRef.of(job);
438 	}

445 	/**
446 	 * Runs one restock: the optional slow wait (cancellable, nothing changed yet), the restock itself, then the
447 	 * after-effect hold.  Every step stops as soon as the job has settled (a cancel or the hard timeout).
448 	 *
449 	 * @param job The job.
450 	 * @param body The restock.
451 	 * @param effect The restock itself; returns the added pets.
452 	 */
453 	static void run(AsyncJob job, Restock body, Supplier<List<Pet>> effect) {
454 		run(job, body, effect, () -> null);
455 	}
456 
457 	/**
458 	 * Same as {@link #run(AsyncJob, Restock, Supplier)}, but the success result carries a value, read after the
459 	 * effect: a row job returns its row's new state, which the card's {@code output} writes back into the row.
460 	 *
461 	 * @param job The job.
462 	 * @param body The restock.
463 	 * @param effect The restock itself; returns the added pets.
464 	 * @param value The success result's value.
465 	 */
466 	static void run(AsyncJob job, Restock body, Supplier<List<Pet>> effect, Supplier<Object> value) {
467 		try {
468 			if (body.slowMs() > 0) {
469 				job.progress(f("Waiting %s ms before restocking", body.slowMs()));
470 				if (! hold(job, body.slowMs()))
471 					return;
472 			}
473 			// One lock acquisition, shared with cancel and the timeout: a job settled after the wait is left as it was.
474 			if (! job.markEffectStarted())
475 				return;
476 			var total = body.perSpecies().values().stream().mapToInt(Integer::intValue).sum();
477 			job.progress(f("Restocking %s pets", total));
478 			var pets = effect.get();
479 			pets.forEach(p -> job.progress(f("Added %s", p.getName())));
480 			if (hold(job, body.holdMs()))
481 				job.complete(ActionResult.success(value.get()).message(f("Restocked %s pets", pets.size())));
482 		} catch (InterruptedException e) {
483 			Thread.currentThread().interrupt();
484 			job.complete(ActionResult.unknown().message("Interrupted"));
485 		} catch (RuntimeException e) {
486 			job.complete(ActionResult.failure().message(e.getMessage()));
487 		}
488 	}

490 	/** Sleeps in short steps; returns <jk>false</jk> as soon as the job has settled. */
491 	private static boolean hold(AsyncJob job, long ms) throws InterruptedException {
492 		for (var left = ms; left > 0; left -= SLOW_STEP_MS) {
493 			if (job.isTerminal())
494 				return false;
495 			Thread.sleep(Math.min(SLOW_STEP_MS, left));
496 		}
497 		return ! job.isTerminal();
498 	}

617 	/**
618 	 * Restocks one species from its row's detail panel.
619 	 *
620 	 * <p>
621 	 * The species is the path's {@code {id}}, which must equal the body's {@code targetId}, which the key must be bound
622 	 * to.  The body's {@code fields} supply only {@code count}; any other field (the panel's hidden {@code species}
623 	 * included) is ignored.  A replay of a claimed key returns the same job, or a {@link JobKeyLedger#SPENT_CODE}
624 	 * refusal once the registry has aged the job out.
625 	 *
626 	 * @param req The request.
627 	 * @param res The response (202, or 409 on a refusal).
628 	 * @param id The species name.
629 	 * @param json The action envelope {@code {action, targetId, idempotencyKey, fields:{count}}}.
630 	 * @return The job ref (202), a 409 refusal, or a 429 {@code app:job-limit} refusal.
631 	 * @throws NotFound If the species is unknown.
632 	 * @throws BadRequest If the envelope is malformed or {@code count} is not a whole number 1-50.
633 	 */
634 	@RestPost(path="/species/{id}/restock")
635 	public Object restockSpecies(RestRequest req, RestResponse res, @Path("id") String id, @Content JsonMap json) {
636 		var s = species(id);
637 		var submit = RowJobSubmit.parse(json);
638 		if (! s.name().equals(submit.targetId()))
639 			return refused(res, RowJobSubmit.bindingRefusal());
640 		var actor = ConsoleWrites.actor(req);
641 		var ledger = jobKeys();
642 		var registry = asyncJobRegistry();
643 		AsyncJob job;
644 		// The ledger's own monitor: the verify that reads the claim and the claim itself happen under one lock, so of
645 		// two concurrent submits of one key the second sees the first's job and replays it instead of starting it again.
646 		synchronized (ledger) {
647 			var grant = submit.verify(ledger, RESTOCK_SPECIES, actor).orElse(null);
648 			if (grant == null)
649 				return refused(res, RowJobSubmit.bindingRefusal());
650 			var claimed = grant.jobId().orElse(null);
651 			if (claimed != null) {
652 				var r = replay(claimed, registry);
653 				res.setStatus(r instanceof ActionResult ? RowJobSubmit.REFUSAL_STATUS : 202);
654 				return r;
655 			}
656 			// After the replay check: a replay answers its job even if this resubmit's fields have changed.
657 			var body = new Restock(Map.of(s, count(submit.fields())), grant.key().value(), 0,
658 				Math.max(0, Math.min(speciesHold != null ? (int) speciesHold.value().toMillis() : Integer.getInteger(SPECIES_HOLD_PROPERTY, DEFAULT_HOLD_MS), MAX_HOLD_MS)));
659 			var created = registry.runningCount() >= Integer.getInteger(LIMIT_PROPERTY, 2) ? null : registry.tryCreate(grant.key()).orElse(null);
660 			if (created == null)
661 				return limitReached(res, registry.runningCount());
662 			ledger.claim(grant.key(), created.id());
663 			job = created;
664 			remember(created, body);
665 			pool.execute(() -> run(created, body, () -> restocked(s, body, actor), () -> speciesRow(s)));
666 		}
667 		res.setStatus(202);
668 		return AsyncJobRef.of(job);
669 	}

672 	 * A claimed key's replay: its job's ref, or a {@link JobKeyLedger#SPENT_CODE} refusal once the registry has reaped
673 	 * the job.  Never a second job: the key was spent on the first.
674 	 *
675 	 * @param jobId The job the key was claimed for.
676 	 * @param registry The registry.
677 	 * @return The {@link AsyncJobRef}, or the refusal.
678 	 */
679 	static Object replay(String jobId, AsyncJobRegistry registry) {

681 			.orElseGet(() -> ActionResult.refusal(JobKeyLedger.SPENT_CODE).message("This key's job has aged out; fetch a new key"));
682 	}

716 	/**
717 	 * Grooms a run's pet again.
718 	 *
719 	 * <p>
720 	 * Ledger keys only, bound to the path's run: the pet comes from the server's run record, never the request.
721 	 * {@code /groom} stays the dialog's self-targeted endpoint; one endpoint taking both kinds of key would be only as
722 	 * strong as the self-targeted one.  A replay returns the same new run while it is kept, then a
723 	 * {@link JobKeyLedger#SPENT_CODE} refusal once the run list has evicted it; never a second run.
724 	 *
725 	 * @param req The request.
726 	 * @param res The response (409 on a refusal).
727 	 * @param id The run id.
728 	 * @param json The action envelope {@code {action, targetId, idempotencyKey}}.
729 	 * @return The new run's row, or a 409 refusal.
730 	 * @throws NotFound If the key is unclaimed and the run is unknown.
731 	 * @throws BadRequest If the envelope is malformed.
732 	 * @throws IOException If the new run's temp file cannot be created.
733 	 */
734 	@RestPost(path="/groom-runs/{id}/regroom")
735 	public ActionResult regroom(RestRequest req, RestResponse res, @Path("id") String id, @Content JsonMap json) throws IOException {
736 		var submit = RowJobSubmit.parse(json);
737 		if (! id.equals(submit.targetId()))
738 			return refused(res, RowJobSubmit.bindingRefusal());
739 		var ledger = jobKeys();
740 		// As restockSpecies: the claim is read and recorded under the ledger's lock.  The replay check comes before the
741 		// run lookup, because GroomRuns evicts the oldest runs first and a replay must still answer once its target is gone.
742 		synchronized (ledger) {
743 			var grant = submit.verify(ledger, REGROOM, ConsoleWrites.actor(req)).orElse(null);
744 			if (grant == null)
745 				return refused(res, RowJobSubmit.bindingRefusal());
746 			var claimed = grant.jobId().orElse(null);
747 			if (claimed != null)
748 				return groomRuns().get(claimed).map(x -> ActionResult.success(x.row()).message(f("Grooming %s again", x.pet())))
749 					.orElseGet(() -> refused(res, ActionResult.refusal(JobKeyLedger.SPENT_CODE).message("This key's run has aged out; fetch a new key")));
750 			var again = groomRuns().start(groomRun(id).pet(), false, grant.key().value());
751 			ledger.claim(grant.key(), again.id());
752 			return ActionResult.success(again.row()).message(f("Grooming %s again", again.pet()));
753 		}
754 	}

761 	private void remember(AsyncJob job, Restock body) {
762 		synchronized (submitted) {
763 			submitted.addFirst(JsonMap.of("jobId", job.id(), "submitted", job.createdAt().toString(), "request", body.perSpecies(), "idempotencyKey", body.idempotencyKey()));
764 			while (submitted.size() > MAX_SUBMITTED)
765 				submitted.removeLast();
766 		}
767 	}
