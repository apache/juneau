17  package org.apache.juneau.petstore.console;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.commons.settings.*;
22  import org.apache.juneau.http.response.*;
23  import org.apache.juneau.petstore.auth.*;
24  import org.apache.juneau.rest.server.*;
25  import org.apache.juneau.rest.server.auth.*;
26  
27  /**
28   * The console's write gate.  Console writes are open by default; with {@code -Dpetstore.secure=true} each one
29   * needs the demo bearer token ({@link StubBearerTokenValidator#DEFAULT_TOKENS}).  Also names the audit actor.
30   *
31   * <h5 class='section'>Example:</h5>
32   * <p class='bjava'>
33   * 	<ja>@RestPost</ja>(path=<js>"/"</js>)
34   * 	<jk>public</jk> Pet create(RestRequest <jv>req</jv>, <ja>@Content</ja> Pet <jv>pet</jv>) {
35   * 		<jk>return</jk> store().createPet(<jv>pet</jv>, ConsoleWrites.<jsm>actor</jsm>(<jv>req</jv>));
36   * 	}
37   * </p>
38   */
39  public final class ConsoleWrites {
40  
41  	/** Setting (system property, environment, or any registered source) that turns on the console write gate. */
42  	public static final String SECURE_PROPERTY = "petstore.secure";
43  
44  	private static final StubBearerTokenValidator VALIDATOR = new StubBearerTokenValidator();
45  
46  	private ConsoleWrites() {}
47  
48  	/**
49  	 * The viewer's roles, for {@code <@facts>} and server-side omission of role-gated cards.
50  	 *
51  	 * <p>
52  	 * Never an authorization check: writes still go through {@link #actor(RestRequest)}.  Independent of
53  	 * {@code petstore.secure}.
54  	 *
55  	 * @param req The request.
56  	 * @return {@code ["admin"]} when the bearer token validates to the {@code admin} principal, else empty.
57  	 */
58  	public static List<String> roles(RestRequest req) {
59  		var user = user(req);
60  		return "admin".equals(user) ? List.of("admin") : List.of();
61  	}
62  
63  	private static String user(RestRequest req) {
64  		var auth = req.getHeaderParam("Authorization").asString().orElse(null);
65  		var token = auth != null && auth.regionMatches(true, 0, "Bearer ", 0, 7) ? auth.substring(7).trim() : null;
66  		if (token == null)
67  			return null;
68  		try {
69  			return VALIDATOR.validate(token).getName();
70  		} catch (@SuppressWarnings("unused") AuthenticationException e) {
71  			return null; // Unknown token: treated as anonymous.
72  		}
73  	}
74  
75  	/**
76  	 * Checks the write gate and returns the audit actor.
77  	 *
78  	 * @param req The request.
79  	 * @return {@code "console:<user>"} when a valid bearer token is present, else {@code "console"}.
80  	 * @throws Unauthorized If the gate is on and the token is missing or unknown.
81  	 */
82  	public static String actor(RestRequest req) {
83  		var user = user(req);
84  		if (user == null && secure())
85  			throw new Unauthorized("Console writes need the demo bearer token").setHeader("WWW-Authenticate", "Bearer realm=\"petstore\"");
86  		return user == null ? "console" : "console:" + user;
87  	}
88  
89  	// Resolved on every call, so a changed setting (or a test override) takes effect immediately.
90  	private static boolean secure() {
91  		return Settings.get().get(SECURE_PROPERTY).asBoolean().orElse(false);
92  	}
93  
94  	/**
95  	 * The audit actor reduced to the staging author.
96  	 *
97  	 * @param actor A {@link #actor(RestRequest)} value.
98  	 * @return The bare user ({@code "admin"}), or {@code "anonymous"} when writes are unsecured.
99  	 */
100 	public static String author(String actor) {
101 		var a = actor.replaceFirst("^console:?", "");
102 		return a.isEmpty() ? "anonymous" : a;
103 	}
104 }
