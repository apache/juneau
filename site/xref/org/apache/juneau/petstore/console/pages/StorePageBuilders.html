29  /**
30   * Builder twin of the Store page (P1, {@code store.ftlh}).
31   *
32   * <h5 class='section'>Example:</h5>
33   * <p class='bjava'>
34   * 	<jk>return</jk> <jsm>twin</jsm>(<jv>req</jv>, StorePageBuilders.<jsm>store</jsm>(QuickStats.<jsm>of</jsm>(store())));
35   * </p>
36   */
37  @SuppressWarnings({
38  	"java:S1192" // Page-definition ids and labels repeat as inline literals by design.
39  })
40  public final class StorePageBuilders {
41  
42  	private StorePageBuilders() {}
43  
44  	/**
45  	 * The Store page's quick-stat tiles, shared by both render paths.
46  	 *
47  	 * <h5 class='section'>Example:</h5>
48  	 * <p class='bjava'>
49  	 * 	<jk>var</jk> <jv>qs</jv> = QuickStats.<jsm>of</jsm>(<jv>store</jv>);
50  	 * </p>
51  	 *
52  	 * @param available Available pets.
53  	 * @param pending Pending pets.
54  	 * @param sold Sold pets.
55  	 * @param ordersPlaced Placed orders.
56  	 * @param ordersApproved Approved orders.
57  	 * @param ordersDelivered Delivered orders.
58  	 * @param petTotal All pets, the inventory meter's maximum.
59  	 */
60  	public record QuickStats(long available, long pending, long sold, long ordersPlaced, long ordersApproved, long ordersDelivered, long petTotal) {
61  
62  		/**
63  		 * Counts the tiles from the store.
64  		 *
65  		 * @param store The store.
66  		 * @return The tiles.
67  		 */
68  		public static QuickStats of(PetStore store) {
69  			var pets = store.getPets();
70  			var orders = store.getOrders();
71  			return new QuickStats(count(pets, PetStatus.AVAILABLE), count(pets, PetStatus.PENDING), count(pets, PetStatus.SOLD),
72  				count(orders, OrderStatus.PLACED), count(orders, OrderStatus.APPROVED), count(orders, OrderStatus.DELIVERED), pets.size());
73  		}
74  
75  		private static long count(Collection<Pet> pets, PetStatus status) {
76  			return pets.stream().filter(x -> x.getStatus() == status).count();
77  		}
78  
79  		private static long count(Collection<Order> orders, OrderStatus status) {
80  			return orders.stream().filter(x -> x.getStatus() == status).count();
81  		}
82  	}
83  
84  	/** The store strip: three status tiles, an order breakdown, and an inventory meter (a tile when there are no pets). */
85  	private static QuickStat[] storeQuickStats(QuickStats qs) {
86  		var inventory = qs.petTotal() > 0
87  			? QuickStat.meter("inventory", "Inventory", qs.available(), qs.petTotal()).text(qs.available() + " of " + qs.petTotal())
88  			: QuickStat.of("inventory", "Inventory", "0 of 0");
89  		return new QuickStat[] {
90  			QuickStat.of("available", "Available", qs.available()),
91  			QuickStat.of("pending", "Pending", qs.pending()),
92  			QuickStat.of("sold", "Sold", qs.sold()),
93  			QuickStat.segmented("orders", "Orders").segment("Placed", qs.ordersPlaced())
94  				.segment("Approved", qs.ordersApproved(), StatusTone.INFO).segment("Delivered", qs.ordersDelivered(), StatusTone.SUCCESS),
95  			inventory
96  		};
97  	}
98  
99  	/**
100 	 * P1 Store.
101 	 *
102 	 * @param qs The quick-stat tiles.
103 	 * @return The page spec.
104 	 */
105 	public static PageSpec store(QuickStats qs) {
106 		return PageSpec.create().tab("store").toolkit("views").init("/console/dev/cards/petstore-cards.js")
107 			.html("welcome", "Welcome to the petstore", """
108 				<div class="jc-prose"><p>Browse pets, place orders and review what changed. Every page here is built from the same
109 				Juneau console pieces your own app can use.</p>
110 				<p data-slot="welcome-note">Signed in as a guest.</p></div>""")
111 			.table(TableSpec.create("stats").rowType("Order").dataMode(DataMode.CLIENT).dataUrl("/console/store/recent-orders")
112 				.columns(
113 					Column.create("id").label("Order").href("/console/pets/{petId}/orders/{id}"),
114 					Column.create("petId").label("Pet").href("/console/pets/{petId}/summary"),
115 					Column.create("shipDate").label("Ship date").render("date"),
116 					Column.create("status").label("Status").render("pill"))
117 				.quickStats("store", storeQuickStats(qs)), "Recent orders")
118 			.card(CardSpec.of("kpi", "revenue").title("Revenue")
119 				.body("label", "Revenue today").body("dataUrl", "/console/store/kpi/revenue"))
120 			.card(ChartDef.line("orders-trend").title("Orders, last 30 days").src("/console/store/stats/daily")
121 				.x("day", null, "Day").series("placed", "Placed").series("delivered", "Delivered", "ok")
122 				.onClickHref("/console/orders?state=day({day})").toCardSpec())
123 			.card(ChartDef.donut("by-species").title("Pets by species").src("/console/store/stats/species")
124 				.label("species").value("count").center("total").toCardSpec())
125 			.card(ChartDef.bar("by-status").title("Pets by status").src("/console/store/stats/status")
126 				.x("status").series("count", "Pets").toCardSpec())
127 			.card(ChartDef.treemap("pets-map").title("Pets by species and status").src("/console/store/stats/pets")
128 				.groupBy("species", "status").label("name")
129 				.colorBy("status", ChartDef.legendItem("AVAILABLE", "Available", "ok"),
130 					ChartDef.legendItem("PENDING", "Pending", "warning"),
131 					ChartDef.legendItem("SOLD", "Sold", null))
132 				.tileFields("id", "name", "species", "status")
133 				.onClickAction(PetsPageBuilders.editPetAction(OnSuccess.REDRAW)).toCardSpec())
134 			.htmlSrc("hours", "Store hours", "/console/store/fragments/hours");
135 	}
136 }
