28  /**
29   * Builder twins of the Operations pages: P6 Jobs ({@code jobs.ftlh}) and P7 Audit ({@code audit.ftlh}).
30   *
31   * <h5 class='section'>Example:</h5>
32   * <p class='bjava'>
33   * 	<jk>return</jk> <jsm>twin</jsm>(<jv>req</jv>, OpsPageBuilders.<jsm>audit</jsm>());
34   * </p>
35   */
36  @SuppressWarnings({
37  	"java:S1192" // Page-definition ids and labels repeat as inline literals by design.
38  })
39  public final class OpsPageBuilders {
40  
41  	private OpsPageBuilders() {}
42  
43  	/**
44  	 * P6 Jobs.
45  	 *
46  	 * @param groomOutput The latest groom run's terminal.
47  	 * @param groomFile The latest groom run's transcript file.
48  	 * @return The page spec.
49  	 */
50  	public static PageSpec jobs(TerminalDef groomOutput, TerminalDef groomFile) {
51  		return PageSpec.create().tab("ops/jobs").toolkit("views")
52  			.html("restock-help", "Restocking", """
53  				<div class="jc-prose"><p>Restock adds new available pets to the store as a background job; progress shows here until it finishes.
54  				Submitting twice from the same dialog runs one job.  Give the card below a wait to see a cancel before anything changes, or a longer hold to see one after.</p></div>""")
55  			.card(CardSpec.of("job-log", "restock").title("Restock inventory")
56  				.body("submit", Map.of("endpoint", "/console/ops/jobs/restock", "submitLabel", "Restock", "fields", List.of(
57  					Map.of("name", "DOG", "label", "Dogs (0-50)", "type", "number", "default", 1),
58  					Map.of("name", "slowMs", "label", "Wait before restocking (ms, 0-10000)", "type", "number", "default", 0),
59  					Map.of("name", "holdMs", "label", "Hold after restocking (ms, 0-10000)", "type", "number", "default", 1000))))
60  				.body("urlParam", "job"))
61  			.table(TableSpec.create("jobs").dataMode(DataMode.CLIENT).dataUrl("/console/ops/jobs/rows")
62  				.pollIntervalMs(2000)
63  				.columns(
64  					Column.create("jobId").label("Job"),
65  					Column.create("submitted").label("Submitted").render("timestamp"),
66  					Column.create("state").label("State").render("pill"),
67  					Column.create("result.message").label("Result"))
68  				.ribbon(
69  					RibbonItem.dialog("restock").title("Restock").endpoint("/console/ops/jobs/restock").method(Method.POST).form("/console/ops/jobs/restock-form"),
70  					RibbonItem.refresh().id("refresh")))
71  			.table(TableSpec.create("species").dataMode(DataMode.CLIENT).dataUrl("/console/ops/jobs/species")
72  				.pollIntervalMs(2000)
73  				.columns(
74  					Column.create("species").label("Species"),
75  					Column.create("available").label("Available"))
76  				.detail(RowDetail.create("/console/ops/jobs/species/{id}").region(JobLogRegion.create("restock-species")
77  					.submit(JobLogRegion.Submit.create("/console/ops/jobs/species/{id}/restock", "restock-species", "/console/ops/jobs/species/{id}/key")
78  						.submitLabel("Restock")
79  						.field(JobLogRegion.Field.of("count").label("How many (1-50)").type("number").defaultValue(1).required(true))
80  						.field(JobLogRegion.Field.of("species").type("hidden").source("row.species")))
81  					.detail(JobLogRegion.Detail.of("fieldGrid", RegionDef.Field.of("available").label("Available")))
82  					.output(JobLogRegion.Output.field("available").from("row.available"))
83  					.toRegion())))
84  			.card(CardSpec.of("terminal", "groom-output").title("Groom run").body("terminal", groomOutput.toMap()))
85  			.card(CardSpec.of("terminal", "groom-file").title("Groom transcript (file)").body("terminal", groomFile.toMap()))
86  			.table(TableSpec.create("groom-runs").dataMode(DataMode.CLIENT).dataUrl("/console/ops/jobs/groom-rows")
87  				.columns(
88  					Column.create("pet").label("Pet"),
89  					Column.create("submitted").label("Submitted").render("timestamp"),
90  					Column.create("state").label("State").render("pill"),
91  					Column.create("lines").label("Lines"))
92  				.detail(RowDetail.create("/console/ops/jobs/groom-runs/{id}")
93  					.region(RegionDef.create("regroom").type(RegionDef.TYPE_ROW_DETAIL).populate("petstore-regroom").allowPopulators("petstore-regroom")))
94  				.ribbon(
95  					RibbonItem.dialog("groom").title("Groom").endpoint("/console/ops/jobs/groom").method(Method.POST).form("/console/ops/jobs/groom-form"),
96  					RibbonItem.refresh().id("groom-refresh")));
97  	}
98  
99  	/**
100 	 * P7 Audit.
101 	 *
102 	 * @return The page spec.
103 	 */
104 	public static PageSpec audit() {
105 		return PageSpec.create().tab("ops/audit").toolkit("views")
106 			.table(TableSpec.create("audit").dataMode(DataMode.SERVER).dataUrl("/console/ops/audit/query")
107 				.operators(SearchOperatorSet.standard())
108 				.defaultOrder("at", Sort.DESC)
109 				.columns(
110 					Column.create("at").label("When").render("timestamp").searchType(SearchType.TIMESTAMP),
111 					Column.create("actor").label("Actor").searchType(SearchType.TEXT),
112 					Column.create("action").label("Action").searchType(SearchType.TEXT),
113 					Column.create("entity").label("Entity").searchType(SearchType.TEXT),
114 					Column.create("entityId").label("Id").searchType(SearchType.TEXT),
115 					Column.create("detail").label("Detail").searchType(SearchType.TEXT))
116 				.ribbon(RibbonItem.export("copy", "csv"), RibbonItem.refresh().id("refresh")));
117 	}
118 }
