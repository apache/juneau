17  package org.apache.juneau.petstore.console.pages;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.console.freemarker.*;
22  import org.apache.juneau.console.views.*;
23  import org.apache.juneau.console.views.RowAction.*;
24  import org.apache.juneau.console.views.TableSpec.*;
25  
26  /**
27   * Builder twin of the Orders page (P5, {@code orders.ftlh}).
28   *
29   * <h5 class='section'>Example:</h5>
30   * <p class='bjava'>
31   * 	<jk>return</jk> <jsm>twin</jsm>(<jv>req</jv>, OrdersPageBuilders.<jsm>orders</jsm>());
32   * </p>
33   */
34  @SuppressWarnings({
35  	"java:S1192" // Page-definition ids and labels repeat as inline literals by design.
36  })
37  public final class OrdersPageBuilders {
38  
39  	private OrdersPageBuilders() {}
40  
41  	/**
42  	 * P5 Orders.
43  	 *
44  	 * @return The page spec.
45  	 */
46  	public static PageSpec orders() {
47  		return PageSpec.create().tab("orders").toolkit("views")
48  			.table(TableSpec.create("orders").rowType("Order").dataMode(DataMode.SERVER).dataUrl("/console/orders/query").urlState("day", "window")
49  				.columnConfig(Map.of("tabs", List.of("columns", "sort")))
50  				.defaultOrder("id", Sort.ASC)
51  				.columns(
52  					Column.create("id").label("Id"),
53  					Column.create("petId").label("Pet").href("/console/pets/{petId}/summary"),
54  					Column.create("shipDate").label("Ship date").render("date"),
55  					Column.create("status").label("Status").render("pill"))
56  				.ribbon(
57  					RibbonItem.optionGroup("status",
58  						RibbonItem.option("placed").title("Placed").column("status").value("$eq(PLACED)"),
59  						RibbonItem.option("approved").title("Approved").column("status").value("$eq(APPROVED)"),
60  						RibbonItem.option("delivered").title("Delivered").column("status").value("$eq(DELIVERED)"))
61  						.title("Order status").deselectable(true),
62  					RibbonItem.dialog("new-order").title("+ New order").symbol("plus")
63  						.form("/console/orders/new-form").endpoint("/console/orders").method(Method.POST).onSuccess(OnSuccess.REDRAW),
64  					RibbonItem.export("copy", "csv").optional("excel", "pdf"),
65  					RibbonItem.refresh())
66  				.rowActions(
67  					RowAction.create("approve").label("Approve").endpoint("/console/orders/{id}/approve").method(Method.POST).onSuccess(OnSuccess.REDRAW)
68  						.visibleWhen(VisibilityRule.when("status").eq("PLACED")),
69  					RowAction.create("deliver").label("Deliver").endpoint("/console/orders/{id}/deliver").method(Method.POST).onSuccess(OnSuccess.REDRAW)
70  						.visibleWhen(VisibilityRule.when("status").eq("APPROVED")),
71  					RowAction.create("cancel").label("Cancel").form("/console/orders/{id}/cancel-form").present(Present.DIALOG)
72  						.endpoint("/console/orders/{id}").method(Method.DELETE).onSuccess(OnSuccess.REDRAW),
73  					RowAction.create("delete").label("Delete").endpoint("/console/orders/{id}").method(Method.DELETE)
74  						.confirm("Delete order '{id}'?").confirmLabel("Delete").tone(Tone.DANGER).confirmRenderer("typed").onSuccess(OnSuccess.REDRAW)),
75  				"Orders");
76  	}
77  }
