27  /**
28   * The expandable row details of the console tables, shared by the page builders (which only need the declaration) and
29   * the resources that serve them (which pass a real lookup).
30   *
31   * <p>
32   * A bean-backed detail lists its fields explicitly, so the resource that serves it never depends on having been
33   * attached to a table.  Each field list equals the columns of the table it opens from, hidden ones included.
34   *
35   * <h2 class='section'>Example:</h2>
36   * <p class='bjava'>
37   * 	<jc>// A page builder attaches the declaration; the resource serves it with a store lookup.</jc>
38   * 	TableSpec.<jsm>create</jsm>(<js>"orders"</js>).detail(RowDetails.<jsm>order</jsm>());
39   * </p>
40   */
41  public final class RowDetails {
42  
43  	private RowDetails() {}
44  
45  	private static <T> Function<String,T> none() {
46  		return id -> null;
47  	}
48  
49  	/** @return The Pets table detail, for building a page. */
50  	public static RowDetail pet() {
51  		return pet(none());
52  	}
53  
54  	/**
55  	 * The Pets table detail: Details, this pet's orders and its history.
56  	 *
57  	 * @param lookup Finds a pet by row id, or returns <jk>null</jk>.
58  	 * @return A new detail.
59  	 */
60  	public static RowDetail pet(Function<String,Pet> lookup) {
61  		return RowDetail.fromBean(Pet.class, lookup).basePath("/console/pets").title("{name}").columns(3)
62  			.section("Details", "id", "name", "species", "status", "price", "tags")
63  			.label("price", "Price (USD)")
64  			.tab(DetailTab.table("Orders", ordersView("pets-orders", "/console/pets/orders/rows")).id("orders").scopeParam("petId").countFromRows())
65  			.tab(DetailTab.table("History", historyView()).id("history").scopeParam("petId").countFromRows());
66  	}
67  
68  	/** @return The Sold table detail, for building a page. */
69  	public static RowDetail soldPet() {
70  		return soldPet(none());
71  	}
72  
73  	/**
74  	 * The Sold table detail: the columns that table shows, and this pet's orders.
75  	 *
76  	 * @param lookup Finds a pet by row id, or returns <jk>null</jk>.
77  	 * @return A new detail.
78  	 */
79  	public static RowDetail soldPet(Function<String,Pet> lookup) {
80  		return RowDetail.fromBean(Pet.class, lookup).key("soldPet").basePath("/console/pets").title("{name}").columns(3)
81  			.section("Details", "id", "name", "species", "price")
82  			.label("price", "Price (USD)")
83  			.tab(DetailTab.table("Orders", ordersView("sold-orders", "/console/pets/orders/rows")).id("orders").scopeParam("petId").countFromRows());
84  	}
85  
86  	/** @return The Orders and Store tables' detail, for building a page. */
87  	public static RowDetail order() {
88  		return order(none());
89  	}
90  
91  	/**
92  	 * The Orders and Store recent-orders detail: Details and the ordered pet.
93  	 *
94  	 * @param lookup Finds an order by row id, or returns <jk>null</jk>.
95  	 * @return A new detail.
96  	 */
97  	public static RowDetail order(Function<String,Order> lookup) {
98  		return RowDetail.fromBean(Order.class, lookup).basePath("/console/orders").title("Order {id}").columns(2)
99  			.section("Details", "id", "petId", "shipDate", "status")
100 			.label("petId", "Pet")
101 			.tab(DetailTab.table("Pet", TableSpec.create("order-pet").rowType("Pet").dataMode(DataMode.CLIENT).dataUrl("/console/orders/pet/rows")
102 				.columns(
103 					Column.create("id").label("Id"),
104 					Column.create("name").label("Name").href("/console/pets/{id}/summary"),
105 					Column.create("species").label("Species"),
106 					Column.create("status").label("Status"),
107 					Column.create("price").label("Price (USD)").render(Render.of("decimal").meta("places", 2))))
108 				.id("pet").scopeParam("orderId").countFromRows());
109 	}
110 
111 	/** @return The Users table detail, for building a page. */
112 	public static RowDetail user() {
113 		return user(none());
114 	}
115 
116 	/**
117 	 * The Users table detail: Details and the orders this user placed.  It serves {@code UserView}, so a password,
118 	 * email or phone never leaves the server.
119 	 *
120 	 * @param lookup Finds a user by row id (the username), or returns <jk>null</jk>.
121 	 * @return A new detail.
122 	 */
123 	public static RowDetail user(Function<String,UserView> lookup) {
124 		return RowDetail.fromBean(UserView.class, lookup).key("user").basePath("/console/users").title("{username}").columns(2)
125 			.section("Details", "username", "firstName", "lastName", "userStatus")
126 			.label("userStatus", "Status")
127 			.tab(DetailTab.table("Orders", ordersView("user-orders", "/console/users/orders/rows")).id("orders").scopeParam("username").countFromRows());
128 	}
129 
130 	/** @return The Pending changes table detail. */
131 	public static RowDetail change() {
132 		return RowDetail.inline().title("Change {id}").section("Details", "id", "summary", "author", "state", "message");
133 	}
134 
135 	/** @return The Audit table detail. */
136 	public static RowDetail audit() {
137 		return RowDetail.inline().title("{action} {entity} {entityId}")
138 			.section("Details", "id", "at", "actor", "action", "entity", "entityId", "detail");
139 	}
140 
141 	/** @return The pet History table detail. */
142 	public static RowDetail history() {
143 		return RowDetail.inline().title("{action}").section("Details", "id", "at", "actor", "action", "detail");
144 	}
145 
146 	private static TableSpec ordersView(String id, String dataUrl) {
147 		return TableSpec.create(id).rowType("Order").dataMode(DataMode.CLIENT).dataUrl(dataUrl)
148 			.columns(
149 				Column.create("id").label("Order").href("/console/pets/{petId}/orders/{id}").searchType(SearchType.NUMERIC),
150 				Column.create("shipDate").label("Ship date").render("date").searchType(SearchType.TEXT),
151 				Column.create("status").label("Status").render(PillRenders.orderStatus()).searchType(SearchType.ENUM));
152 	}
153 
154 	private static TableSpec historyView() {
155 		return TableSpec.create("pets-history").rowType("AuditEntry").dataMode(DataMode.CLIENT).dataUrl("/console/pets/history/rows")
156 			.columns(
157 				Column.create("at").label("When").render("timestamp").searchType(SearchType.TEXT),
158 				Column.create("actor").label("Actor").searchType(SearchType.TEXT),
159 				Column.create("action").label("Action").render(PillRenders.auditAction()).searchType(SearchType.ENUM),
160 				Column.create("detail").label("Detail").searchType(SearchType.TEXT));
161 	}
162 }
