30  /**
31   * Builder twins of the Pets pages: P2 Pets ({@code pets.ftlh}), its {@code ?mixed} variant, P3 Sold
32   * ({@code pets-sold.ftlh}) and P4 one pet ({@code pet.ftlh}).
33   *
34   * <h5 class='section'>Example:</h5>
35   * <p class='bjava'>
36   * 	<jk>return</jk> <jsm>twin</jsm>(<jv>req</jv>, PetsPageBuilders.<jsm>pets</jsm>());
37   * </p>
38   */
39  @SuppressWarnings({
40  	"java:S1192" // Page-definition ids and labels repeat as inline literals by design.
41  })
42  public final class PetsPageBuilders {
43  
44  	/**
45  	 * The edit-pet dialog action, shared by the Pets table row menu and the Store page treemap so the two cannot drift.
46  	 *
47  	 * @param onSuccess What the page does after a saved edit; a table merges the row, a chart (which has no row) redraws.
48  	 * @return The action.
49  	 */
50  	static RowAction editPetAction(OnSuccess onSuccess) {
51  		return RowAction.create("edit").label("Edit").form("/console/pets/form?id={id}").present(Present.DIALOG)
52  			.endpoint("/console/pets/{id}").method(Method.PUT).onSuccess(onSuccess);
53  	}
54  
55  	private PetsPageBuilders() {}

57  	/**
58  	 * P2 Pets.
59  	 *
60  	 * @return The page spec.
61  	 */
62  	public static PageSpec pets() {
63  		return PageSpec.create().tab("pets").toolkit("views", "datatables-buttons").table(petsTable(), "Pets");
64  	}

70  	 * @return The page spec.
71  	 */
72  	public static PageSpec petsMixed() {
73  		return PageSpec.create().template("pets-mixed.ftlh").table(petsTable(), "Pets");

76  	/**
77  	 * The P2 table, shared by {@link #pets()} and {@link #petsMixed()} so the two cannot drift.
78  	 *
79  	 * @return A new table spec.
80  	 */
81  	static TableSpec petsTable() {
82  		var selection = SelectionDef.create("id").selectAll(true).scope(SelectionDef.Scope.PERSISTENT).labelField("name")
83  			.selectableWhen(RowActionEnabledRule.of("status", Op.NE, "SOLD", "Already sold."));
84  		return TableSpec.create("pets").rowType("Pet").dataMode(DataMode.SERVER).dataUrl("/console/pets/query")
85  			.columnConfig(true)
86  			.urlState("ids")
87  			.defaultOrder("id", Sort.ASC)
88  			.pollIntervalMs(30_000)
89  			.pausePollingWhileEditing(true)
90  			.columns(
91  				Column.create("id").label("Id"),
92  				Column.create("name").label("Name").href("/console/pets/{id}/summary"),
93  				Column.create("species").label("Species"),
94  				Column.create("status").label("Status").render("pill"),
95  				Column.create("price").label("Price (USD)").render(Render.of("decimal").meta("places", 2)),
96  				Column.create("tags").label("Tags").defaultVisible(false))
97  			.rowClass("status", Op.EQ, "SOLD", "is-muted")
98  			.ribbon(
99  				RibbonItem.optionGroup("status",
100 					RibbonItem.option("available").title("Available").column("status").value("$eq(AVAILABLE)"),
101 					RibbonItem.option("pending").title("Pending").column("status").value("$eq(PENDING)"))
102 					.title("Pet status").deselectable(true).defaultOption("available"),
103 				RibbonItem.optionGroup("species",
104 					RibbonItem.option("mammals").title("Mammals").column("species").value("$in(DOG,CAT,MOUSE,RABBIT)"),
105 					RibbonItem.option("others").title("Others").column("species").value("$in(BIRD,FISH,SNAKE)"))
106 					.title("Species").deselectable(true).defaultOption("mammals"),
107 				RibbonItem.dialog("add-pet").title("+ Add pet").symbol("plus")
108 					.form("/console/pets/form").endpoint("/console/pets").method(Method.POST).onSuccess(OnSuccess.REDRAW),
109 				RibbonItem.export("copy", "csv").optional("excel", "pdf"),
110 				RibbonItem.refresh())
111 			.rowActions(
112 				editPetAction(OnSuccess.MERGE_ROW),
113 				RowAction.create("delete").label("Delete").endpoint("/console/pets/{id}").method(Method.DELETE)
114 					.confirm("Delete '{name}'?").confirmLabel("Delete").tone(Tone.DANGER).onSuccess(OnSuccess.REDRAW))
115 			.selection(selection)
116 			.bulk(BulkMutateDef.create(WritePermit.forCapability("pets.sell"), selection).actions(
117 				RowAction.create("sell").label("Mark sold").endpoint("/console/pets/{id}/sell").method(Method.POST).bulkMode(BulkMode.PER_ROW)
118 					.confirmTitle("Mark pets sold").confirm("Mark the selected pets sold?").onSuccess(OnSuccess.REDRAW)));
119 	}

142 	/**
143 	 * P4 one pet, on one tab.
144 	 *
145 	 * @param pet The pet.
146 	 * @param tab The tab: {@code summary}, {@code photo}, {@code orders} or {@code history}.
147 	 * @param activeNav The active nav path, {@code pets/p<id>/<tab>} or {@code pets/p<id>/orders/o<orderId>}.
148 	 * @param order The selected order on the orders tab, or <jk>null</jk>.
149 	 * @param roles The viewer's roles.
150 	 * @param apiCsrf The API's CSRF token for the photo upload form, or <jk>null</jk> to render the form without one.
151 	 * @return The page spec.
152 	 */
153 	public static PageSpec pet(Pet pet, String tab, String activeNav, Order order, List<String> roles, String apiCsrf) {
154 		var id = pet.getId();
155 		var href = "/console/pets/" + id;
156 		var p = PageSpec.create().tab(activeNav).toolkit("views").title(pet.getName())
157 			.facts(Map.of("viewer", Map.of("roles", roles)))
158 			.navUnder("pets", n -> {
159 				var node = n.add("p" + id, pet.getName(), href + "/summary");
160 				node.add("summary", "Summary", href + "/summary");
161 				node.add("photo", "Photo", href + "/photo");
162 				var orders = node.add("orders", "Orders", href + "/orders");
163 				if (order != null)
164 					orders.add("o" + order.getId(), "Order " + order.getId(), href + "/orders/" + order.getId());
165 				node.add("history", "History", href + "/history");
166 			});
167 		var name = escapeHtml(pet.getName());
168 		switch (tab) {
169 			case "summary" -> p.table(TableSpec.create("pet").rowType("Pet").dataMode(DataMode.CLIENT).dataUrl(href + "/row")
170 				.columns(
171 					Column.create("id").label("Id"),
172 					Column.create("name").label("Name").render(Render.of("edit").meta("action", "rename").meta("column", "name").meta("input", "text")),
173 					Column.create("species").label("Species"),
174 					Column.create("status").label("Status").render("pill"),
175 					Column.create("price").label("Price (USD)").render(Render.of("edit").meta("action", "reprice").meta("column", "price").meta("input", "number")))
176 				.rowActions(
177 					stage("rename", "Rename"),
178 					stage("reprice", "Reprice"),
179 					RowAction.create("sell").label("Mark sold").endpoint("/console/pets/{id}/sell").method(Method.POST).onSuccess(OnSuccess.REDRAW)
180 						.confirm("Mark '{name}' sold?").visibleWhen(VisibilityRule.when("status").ne("SOLD")))
181 				.ribbon(RibbonItem.refresh()), pet.getName());
182 			case "photo" -> {
183 				p.html("photo", "Photo", """
184 					<figure class="jc-figure">
185 					<object data="/petstore/pets/%1$s/photo" type="image/png" aria-label="Photo of %2$s"><p>No photo uploaded for %2$s yet.</p></object>
186 					<figcaption>Served by <code>PetPhotoSerializer</code> from <code>GET /petstore/pets/%1$s/photo</code>.</figcaption>
187 					</figure>""".formatted(id, name));
188 				if (roles.contains("admin")) {
189 					var csrf = apiCsrf == null ? "" : " data-petstore-csrf=\"%s\"".formatted(escapeHtml(apiCsrf));
190 					p.html("photo-upload", "Upload photo", """
191 						<form class="jc-form" data-petstore-upload="/petstore/pets/%s/photo"%s>
192 						<input type="file" name="photo" accept="image/png,image/jpeg" aria-label="Photo file">
193 						<button type="submit" class="jc-button">Upload</button>
194 						</form>""".formatted(id, csrf));
195 				}
196 			}
197 			case "orders" -> {
198 				p.table(TableSpec.create("pet-orders").rowType("Order").dataMode(DataMode.CLIENT).dataUrl(href + "/orders/rows")
199 					.columns(
200 						Column.create("id").label("Order").href(href + "/orders/{id}"),
201 						Column.create("shipDate").label("Ship date"),
202 						Column.create("status").label("Status").render("pill"))
203 					.detail(RowDetail.create(href + "/orders/{id}/detail").title("Order {id}")), "Orders");
204 				if (order != null)
205 					p.html("order", "Order " + order.getId(),
206 						"<dl class=\"jc-details\"><dt>Status</dt><dd>%s</dd><dt>Ship date</dt><dd>%s</dd><dt>Pet</dt><dd>%s</dd></dl>"
207 							.formatted(order.getStatus(), escapeHtml(order.getShipDate() == null ? "" : order.getShipDate()), name));
208 			}
209 			default -> p.table(TableSpec.create("history").rowType("AuditEntry").dataMode(DataMode.CLIENT).dataUrl(href + "/history/rows")
210 				.defaultOrder("at", Sort.DESC)
211 				.columns(
212 					Column.create("at").label("When").render("timestamp"),
213 					Column.create("actor").label("Actor"),
214 					Column.create("action").label("Action"),
215 					Column.create("detail").label("Detail")), "History");
216 		}
217 		return p;
218 	}

220 	// The pencil edit cell commits it; present=inline keeps it out of the row menu.
221 	private static RowAction stage(String id, String label) {
222 		return RowAction.create(id).label(label).endpoint("/console/pets/{id}/stage").method(Method.POST).noConfirm()
223 			.present(Present.INLINE)
224 			.visibleWhen(VisibilityRule.when("status").ne("SOLD"));
225 	}
