17  package org.apache.juneau.petstore.console.pages;
18  
19  import org.apache.juneau.rest.server.view.freemarker.console.*;
20  import org.apache.juneau.rest.server.views.*;
21  import org.apache.juneau.rest.server.views.RowAction.*;
22  import org.apache.juneau.rest.server.views.TableSpec.*;
23  import org.apache.juneau.rest.server.widgets.*;
24  
25  /**
26   * Builder twin of the Pending changes page (P10, {@code changes.ftlh}).
27   *
28   * <h5 class='section'>Example:</h5>
29   * <p class='bjava'>
30   * 	<jk>return</jk> <jsm>twin</jsm>(<jv>req</jv>, ChangesPageBuilders.<jsm>changes</jsm>());
31   * </p>
32   */
33  @SuppressWarnings({
34  	"java:S1192" // Page-definition ids and labels repeat as inline literals by design.
35  })
36  public final class ChangesPageBuilders {
37  
38  	private ChangesPageBuilders() {}
39  
40  	/**
41  	 * P10 Pending changes.
42  	 *
43  	 * @return The page spec.
44  	 */
45  	public static PageSpec changes() {
46  		var selection = SelectionDef.create("id").selectAll(true).scope(SelectionDef.Scope.PERSISTENT).labelField("summary")
47  			.selectableWhen(RowActionEnabledRule.of("state", Op.EQ, "PENDING", "Only pending changes can be discarded or applied."));
48  		return PageSpec.create().tab("pets/changes").toolkit("views")
49  			.table(TableSpec.create("changes").rowType("PendingChange").dataMode(DataMode.SERVER).dataUrl("/console/pets/changes/query")
50  				.defaultOrder("id", Sort.DESC)
51  				.columns(
52  					Column.create("id").label("Change"),
53  					Column.create("summary").label("Change summary"),
54  					Column.create("author").label("Author"),
55  					Column.create("state").label("State").render("pill"),
56  					Column.create("message").label("Message").defaultVisible(false))
57  				.ribbon(
58  					RibbonItem.option("pending-only").title("Pending only").column("state").value("$eq(PENDING)").defaultOn(),
59  					RibbonItem.refresh())
60  				.rowActions(
61  					RowAction.create("retry").label("Retry").endpoint("/console/pets/changes/{id}/retry").method(Method.POST).onSuccess(OnSuccess.REDRAW)
62  						.visibleWhen(VisibilityRule.when("state").eq("FAILED")))
63  				.selection(selection)
64  				.bulk(BulkMutateDef.create(WritePermit.forCapability("pets.changes"), selection).actions(
65  					RowAction.create("discard").label("Discard").endpoint("/console/pets/changes/discard").method(Method.POST).bulkMode(BulkMode.AGGREGATE)
66  						.confirmTitle("Discard changes").confirm("Discard the selected changes?").tone(Tone.DANGER).confirmRenderer("typed"),
67  					RowAction.create("apply").label("Apply").endpoint("/console/pets/changes/{id}/apply").method(Method.POST).bulkMode(BulkMode.PER_ROW))),
68  				"Pending changes");
69  	}
70  }
