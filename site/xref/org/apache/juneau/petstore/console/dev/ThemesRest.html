30  /**
31   * Themes: the stock console themes and one custom token theme, switchable with {@code ?theme=}.
32   *
33   * <p>
34   * The chosen name reaches {@code base.ftlh} as the {@code petstoreTheme} view attribute, which feeds
35   * {@code <@console theme=...>}.  {@code ?theme=custom} also sets {@code petstoreCustomTheme}, which turns on a
36   * nameless {@code <@theme>} block (it inherits the page theme, {@code light-brown} here) with {@code <@token>}
37   * overrides: a leaf and an alias.
38   *
39   * <h5 class='section'>Example:</h5>
40   * <p class='bftl'>
41   * 	&lt;@console brand="My app" theme="light-brown"&gt;
42   * 	&lt;@theme&gt;
43   * 	&lt;@token name="--jc-pill-red-bg" value="#fdeceb"/&gt;
44   * 	&lt;@token name="--jc-tab-bar-bg" alias="var(--jc-card-bg)"/&gt;
45   * 	&lt;/@theme&gt;
46   * 	...
47   * 	&lt;/@console&gt;
48   * </p>
49   * <p class='bcode'>
50   * 	GET /console/dev/themes?theme=gray
51   * </p>
52   */
53  @Rest(path="/themes", title="Themes")
54  @SuppressWarnings({
55  	"java:S110", // Inheritance depth comes from the BasicRestServlet hierarchy, not this page.
56  	"java:S2386" // CHOICES is an immutable List.copyOf(); public so templates and tests can read it.
57  })
58  public class ThemesRest extends PetstoreConsolePage {
59  
60  	private static final long serialVersionUID = 1L;
61  
62  	/** The name {@code ?theme=} takes for the custom token theme. */
63  	public static final String CUSTOM = "custom";
64  
65  	/** The custom theme's stock seed. */
66  	public static final String CUSTOM_SEED = "light-brown";
67  
68  	/** Every name {@code ?theme=} accepts, stock themes first. */
69  	public static final List<String> CHOICES = choices();
70  
71  	private static List<String> choices() {
72  		var l = new ArrayList<>(ConsoleChromeMixin.BUILTIN_THEME_NAMES);
73  		l.add(CUSTOM);
74  		return List.copyOf(l);
75  	}
76  
77  	/**
78  	 * Renders the Themes page.
79  	 *
80  	 * @param req The request; {@code ?builder} renders the builder twin.
81  	 * @param theme The theme to render in, or <jk>null</jk> for the default.
82  	 * @return The page.
83  	 * @throws BadRequest If the name is not one of {@link #CHOICES}.
84  	 */
85  	@RestGet(path="/")
86  	public View page(RestRequest req, @Query("theme") String theme) {
87  		var name = theme == null || theme.isEmpty() ? ConsoleChromeMixin.BUILTIN_THEME_NAMES.get(0) : theme;
88  		if (! CHOICES.contains(name))
89  			throw new BadRequest("Unknown theme '%s'; choose one of: '%s'", name, String.join(", ", CHOICES));
90  		var custom = CUSTOM.equals(name);
91  		if (builder(req))
92  			return twin(req, DevPageBuilders.themes(custom ? CUSTOM_SEED : name, custom, CHOICES));
93  		return FreemarkerView.of("themes.ftlh")
94  			.attr("petstoreTheme", custom ? CUSTOM_SEED : name)
95  			.attr("petstoreCustomTheme", custom)
96  			.attr("themeChoices", CHOICES);
97  	}
98  }
