32  /**
33   * The bearer-token protected API behind the Secure page.
34   *
35   * <p>
36   * Every op is gated by a fail-closed {@link BearerTokenGuard} backed by {@link StubBearerTokenValidator}: no token
37   * or an unknown token is a {@code 401} with a {@code WWW-Authenticate: Bearer realm="petstore"} challenge.  On
38   * success the resolved {@link Principal} reaches ops as an {@link Auth @Auth} parameter.
39   *
40   * <p>
41   * <b>Why a guard rather than a filter chain?</b>  {@code AuthFilterChain} composes several mechanisms and is
42   * fail-open by design, so a downstream guard decides.  With one mechanism, the op-level guard is the whole story.
43   *
44   * <h5 class='section'>Example:</h5>
45   * <p class='bjava'>
46   * 	<jc>// One bean gates every op on the resource.</jc>
47   * 	<ja>@Bean</ja>
48   * 	<jk>public</jk> RestGuardList guards(BeanStore <jv>bs</jv>) {
49   * 		<jk>return</jk> RestGuardList.<jsm>create</jsm>(<jv>bs</jv>)
50   * 			.append(BearerTokenGuard.<jsm>create</jsm>().realm(<js>"petstore"</js>).validator(<jk>new</jk> StubBearerTokenValidator()).build())
51   * 			.build();
52   * 	}
53   *
54   * 	<jc>// curl -H 'Authorization: Bearer petstore-user' http://localhost:10000/console/dev/secure/api/pets</jc>
55   * </p>
56   */
57  @Rest(path="/api", title="Secure API")
58  @SuppressWarnings({
59  	"java:S110" // Inheritance depth comes from the BasicRestServlet hierarchy, not this page.
60  })
61  public class SecureApiRest extends PetstoreConsolePage {
62  
63  	private static final long serialVersionUID = 1L;
64  
65  	/**
66  	 * Gates every op on this resource with a bearer-token guard.
67  	 *
68  	 * @param bs The bean store.
69  	 * @return The guard list.
70  	 */
71  	@Bean
72  	public RestGuardList guards(BeanStore bs) {
73  		return RestGuardList.create(bs)
74  			.append(BearerTokenGuard.create().realm("petstore").validator(new StubBearerTokenValidator()).build())
75  			.build();
76  	}
77  
78  	/** @return All pets. */
79  	@RestGet(path="/pets")
80  	public Collection<Pet> pets() {
81  		return store().getPets();
82  	}
83  
84  	/**
85  	 * @param id The pet id.
86  	 * @return The pet.
87  	 * @throws NotFound If there is no such pet.
88  	 */
89  	@RestGet(path="/pets/{id}")
90  	public Pet pet(@Path("id") long id) {
91  		var pet = store().getPet(id);
92  		if (pet == null)
93  			throw new NotFound("Unknown pet '%s'", id);
94  		return pet;
95  	}
96  
97  	/**
98  	 * @param caller The authenticated principal (never <jk>null</jk>: the guard rejects anonymous calls first).
99  	 * @return {@code {name: <principal>}}.
100 	 */
101 	@RestGet(path="/whoami")
102 	public Map<String,String> whoami(@Auth Principal caller) {
103 		return Map.of("name", caller.getName());
104 	}
105 }
