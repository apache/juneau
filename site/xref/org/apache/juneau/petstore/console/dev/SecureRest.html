17  package org.apache.juneau.petstore.console.dev;
18  
19  import static java.nio.charset.StandardCharsets.*;
20  
21  import java.io.*;
22  
23  import org.apache.juneau.petstore.console.*;
24  import org.apache.juneau.petstore.console.pages.*;
25  import org.apache.juneau.rest.server.*;
26  import org.apache.juneau.rest.server.view.*;
27  import org.apache.juneau.rest.server.view.freemarker.*;
28  
29  /**
30   * Secure: an open console page demonstrating the bearer-token guarded {@link SecureApiRest}.
31   *
32   * <p>
33   * The page's "Try it" card calls {@code /console/dev/secure/api/whoami} from the browser with no token, the demo
34   * token or a wrong token, and shows the status and body inline.  The demo tokens come from
35   * {@code StubBearerTokenValidator}: {@code petstore-user} (alice) and {@code petstore-admin} (admin).
36   *
37   * <h5 class='section'>Example:</h5>
38   * <p class='bftl'>
39   * 	&lt;@page tab="dev/secure" init=["/console/dev/secure/secure.js"]&gt;
40   * 	&lt;@card id="try" title="Try it"&gt;
41   * 	&lt;button type="button" data-secure-call="petstore-user"&gt;With the demo token&lt;/button&gt;
42   * 	&lt;output id="secure-result"&gt;&lt;/output&gt;
43   * 	&lt;/@card&gt;
44   * 	&lt;/@page&gt;
45   * </p>
46   */
47  @Rest(path="/secure", title="Secure", children={SecureApiRest.class})
48  @SuppressWarnings({
49  	"java:S110", // Inheritance depth comes from the BasicRestServlet hierarchy, not this page.
50  	"resource" // The response writer is owned by the servlet container; Eclipse JDT @Owning warning is by design.
51  })
52  public class SecureRest extends PetstoreConsolePage {
53  
54  	private static final long serialVersionUID = 1L;
55  
56  	/**
57  	 * The Secure page.
58  	 *
59  	 * @param req The request; {@code ?builder} renders the builder twin.
60  	 * @return The page.
61  	 */
62  	@RestGet(path="/")
63  	public View page(RestRequest req) {
64  		return builder(req) ? twin(req, DevPageBuilders.secure()) : FreemarkerView.of("secure.ftlh");
65  	}
66  
67  	/**
68  	 * Serves the page-local script.
69  	 *
70  	 * @param res The response.
71  	 * @throws IOException If the bundled script can't be read.
72  	 */
73  	@RestGet(path="/secure.js")
74  	public void script(RestResponse res) throws IOException {
75  		try (var in = SecureRest.class.getResourceAsStream("secure.js")) {
76  			if (in == null)
77  				throw new IllegalStateException("Missing classpath resource 'secure.js' beside " + SecureRest.class.getName());
78  			res.setContentType("text/javascript;charset=utf-8");
79  			res.getWriter().write(new String(in.readAllBytes(), UTF_8));
80  		}
81  	}
82  }
