17  package org.apache.juneau.petstore.console.dev;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.http.*;
22  import org.apache.juneau.marshall.collections.*;
23  import org.apache.juneau.petstore.console.*;
24  import org.apache.juneau.petstore.console.pages.*;
25  import org.apache.juneau.rest.server.*;
26  import org.apache.juneau.console.views.*;
27  import org.apache.juneau.console.widgets.*;
28  import org.apache.juneau.rest.server.view.*;
29  
30  /**
31   * Dialogs (builder only): a permanent demo of the two dialog navigation styles, a {@code ModalDef.steps} wizard and a
32   * {@code nav:"folder"} sectioned form, each opened from a ribbon button.
33   *
34   * <p>
35   * Nothing is stored: the submit endpoint accepts the dialog envelope and answers with a success toast.  The page
36   * exists so the screenshot sweep and a person can see both dialogs in every palette and scheme.
37   *
38   * <h5 class='section'>Example:</h5>
39   * <p class='bcode'>
40   * 	GET /console/dev/dialogs
41   * 	GET /console/dev/dialogs/wizard-form
42   * 	GET /console/dev/dialogs/folder-form
43   * </p>
44   */
45  @Rest(path="/dialogs", title="Dialogs")
46  @SuppressWarnings({
47  	"java:S110" // Inheritance depth comes from the BasicRestServlet hierarchy, not this page.
48  })
49  public class DialogsRest extends PetstoreConsolePage {
50  
51  	private static final long serialVersionUID = 1L;
52  
53  	/**
54  	 * Renders the Dialogs page.
55  	 *
56  	 * @param req The request.
57  	 * @return The page.
58  	 */
59  	@RestGet(path="/")
60  	public View page(RestRequest req) {
61  		return twin(req, DevPageBuilders.dialogs());
62  	}
63  
64  	/**
65  	 * The rows the page's table draws: one per dialog style.
66  	 *
67  	 * @return One row per style.
68  	 */
69  	@RestGet(path="/rows")
70  	public List<Map<String,Object>> rows() {
71  		return List.of(Map.of("style", "wizard", "steps", 3), Map.of("style", "folder", "steps", 3));
72  	}
73  
74  	/**
75  	 * The wizard dialog: three steps with display fields, Back and Next, and a step rail.
76  	 *
77  	 * @return The dialog.
78  	 */
79  	@RestGet(path="/wizard-form")
80  	public ModalDef wizardForm() {
81  		return ModalDef.create("Register a pet")
82  			.step(ModalDef.Step.of("species").title("Pick a species").field("Species", "Dog").field("Breed", "Beagle"))
83  			.step(ModalDef.Step.of("owner").title("Owner").field("Name", "Mary Watson").field("Phone", "555-0100"))
84  			.step(ModalDef.Step.of("review").title("Review").confirmLabel("Register").field("Fee", "$25.00"));
85  	}
86  
87  	/**
88  	 * The folder-tab dialog: a sectioned form drawn with square-topped folder tabs.
89  	 *
90  	 * @return The dialog.
91  	 */
92  	@RestGet(path="/folder-form")
93  	public ModalDef folderForm() {
94  		var form = FormDef.create()
95  			.section(FormDef.Section.of("basics", "Basics")
96  				.field(FormDef.Input.of("name", "Name", "text").required())
97  				.field(FormDef.Input.of("notes", "Notes", "textarea")))
98  			.section(FormDef.Section.of("options", "Options")
99  				.field(FormDef.Input.of("notify", "Notify me", "toggle")))
100 			.section(FormDef.Section.of("danger", "Advanced")
101 				.field(FormDef.Input.of("code", "Code", "text")));
102 		return ModalDef.create("Pet settings").nav(ModalDef.NAV_FOLDER).form(form);
103 	}
104 
105 	/**
106 	 * Accepts either dialog's submit and stores nothing.
107 	 *
108 	 * @param body The dialog envelope; ignored.
109 	 * @return A success toast.
110 	 */
111 	@RestPost(path="/submit")
112 	public ActionResult submit(@Content JsonMap body) {
113 		return ActionResult.ok().message("Saved (demo only, nothing is stored)");
114 	}
115 }
