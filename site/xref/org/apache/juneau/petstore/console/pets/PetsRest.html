42  /**
43   * The pets JSON endpoints (server-mode datatable query over BeanQuery, create/replace/delete, sell, stage, and the
44   * per-pet feeds), the Pets and Sold pages, the add/edit dialog form, and the P4 pet pages: Summary (inline edit),
45   * Photo, Orders (with a row detail, and one order selected at nav depth 4) and History, whose nav nodes the page
46   * appends under {@code pets} as {@code p<petId>} and {@code o<orderId>}.  The Pending changes page is the
47   * {@link ChangesRest} child.
48   *
49   * <h5 class='section'>Example:</h5>
50   * <p class='bjava'>
51   * 	<jc>// The BQ#6 endpoint behind the table.</jc>
52   * 	<ja>@RestPost</ja>(path=<js>"/query"</js>)
53   * 	<jk>public</jk> DataTablesResults&lt;Pet&gt; query(<ja>@Content</ja> DataTablesRequest <jv>req</jv>) {
54   * 		<jk>try</jk> (<jk>var</jk> <jv>s</jv> = store().queryPets()) {
55   * 			<jk>return</jk> DataTablesQuery.<jsm>run</jsm>(<jv>req</jv>, <jv>s</jv>);
56   * 		}
57   * 	}
58   * </p>
59   * <p class='bcode'>
60   * 	&lt;@card type="datatables" id="pets"&gt;
61   * 	{ dataMode:'server', dataUrl:'/console/pets/query', columns:[{key:'name', label:'Name'}, ...] }
62   * 	&lt;/@card&gt;
63   * </p>
64   * <p class='bjava'>
65   * 	<jc>// The same table in the builder twin.</jc>
66   * 	TableSpec.<jsm>create</jsm>(<js>"pets"</js>).dataMode(DataMode.<jsf>SERVER</jsf>).dataUrl(<js>"/console/pets/query"</js>)
67   * 		.columns(Column.<jsm>create</jsm>(<js>"name"</js>).label(<js>"Name"</js>));
68   * </p>
69   */

131 		return ModalDef.create(id == null ? "Add pet" : f("Edit '%s'", p.getName())).form(form);

134 	/**
135 	 * Server-mode DataTables query over all pets.
136 	 *
137 	 * <p>
138 	 * The optional {@code ids} query parameter (comma-separated pet ids, as the {@code ids} URL-state directive sends it)
139 	 * narrows the pets to those ids, before the request's own column filters apply, so the two combine.  Tokens that are
140 	 * not numbers are dropped, and a blank value or one with no usable token is ignored.
141 	 *
142 	 * @param req The DataTables request.
143 	 * @param ids The ids to keep, or {@code null}.
144 	 * @return One page of pets, or a 200 {@code error} envelope for a malformed search expression.
145 	 */
146 	@RestPost(path="/query")
147 	public DataTablesResults<Pet> query(@Content DataTablesRequest req, @Query("ids") @Schema(aev=true) String ids) {
148 		var wanted = parseIds(ids);
149 		var pets = store().getPets();
150 		if (wanted != null)
151 			pets = pets.stream().filter(p -> wanted.contains(p.getId())).toList();
152 		try (var s = PetStore.PET_QUERY.getSession(pets)) {
153 			return DataTablesQuery.run(req, s);
154 		} catch (BeanQuerySyntaxException e) {
155 			// DataTablesQuery maps a bad expression to HTTP 400 by default; the table wants a 200 error envelope it can show.
156 			return DataTablesResults.error(req.getDraw(), e.getMessage());
157 		}
158 	}

160 	/** The usable ids in a comma-separated value, or {@code null} if there are none (no filter). */
161 	@SuppressWarnings({
162 		"java:S1168" // null means no filter, distinct from an empty id set
163 	})
164 	private static Set<Long> parseIds(String value) {
165 		if (value == null)
166 			return null;
167 		var out = new HashSet<Long>();
168 		for (var token : value.split(",")) {
169 			try {
170 				out.add(Long.parseLong(token.trim()));
171 			} catch (@SuppressWarnings("unused") NumberFormatException e) {
172 				// Not a pet id; dropped.
173 			}
174 		}
175 		return out.isEmpty() ? null : out;
176 	}

179 	 * Creates a pet.  Status defaults to {@code AVAILABLE}.
180 	 *
181 	 * @param req The request (write gate, audit actor).
182 	 * @param body The new pet, either as the plain pet or as the dialog envelope {@code {fields:{...}}}.
183 	 * @return The result toast, carrying the stored pet.
184 	 * @throws BadRequest On a blank name or negative price.
185 	 */
186 	@RestPost(path="/")
187 	public ActionResult create(RestRequest req, @Content JsonMap body) {
188 		var actor = ConsoleWrites.actor(req);
189 		var pet = toPet(body);
190 		validate(pet);
191 		if (pet.getStatus() == null)
192 			pet.setStatus(PetStatus.AVAILABLE);
193 		var p = store().createPet(pet, actor);
194 		return ActionResult.success(p).message(f("Added '%s'", p.getName()));
195 	}

197 	/**
198 	 * Replaces a pet.  Name and price are required; a species, tags or photo left out of the body keep their
199 	 * current values, as does the status.
200 	 *
201 	 * @param req The request.
202 	 * @param id The pet id.
203 	 * @param body The new state, either as the plain pet or as the dialog envelope {@code {fields:{...}}}.
204 	 * @return The result toast.
205 	 * @throws NotFound If the pet does not exist.
206 	 */
207 	@RestPut(path="/{id}")
208 	public ActionResult update(RestRequest req, @Path("id") long id, @Content JsonMap body) {
209 		// The edit form offers name, species and price but no status, so for a saved edit (for example from the
210 		// store treemap) species is the grouping field that can change; status moves only through the sell action.
211 		var actor = ConsoleWrites.actor(req);
212 		var old = require(id);
213 		var pet = toPet(body);
214 		validate(pet);
215 		pet.setId(id);
216 		pet.defaultStatus(old.getStatus()).defaultSpecies(old.getSpecies()).defaultTags(old.getTags()).defaultPhoto(old.getPhoto());
217 		return ActionResult.success(store().updatePet(pet, actor)).message(f("Saved '%s'", pet.getName()));
218 	}

237 	 * Marks a pet sold.  The Pets page bulk "Mark sold (N)" action calls this per row.
238 	 *
239 	 * @param req The request.
240 	 * @param id The pet id.
241 	 * @return The result toast, carrying the updated pet.
242 	 * @throws NotFound If the pet does not exist.
243 	 * @throws Conflict If it is already sold.
244 	 */
245 	@RestPost(path="/{id}/sell")
246 	public ActionResult sell(RestRequest req, @Path("id") long id) {
247 		var actor = ConsoleWrites.actor(req);
248 		var p = require(id);
249 		if (p.getStatus() == PetStatus.SOLD)
250 			throw new Conflict("Pet '%s' is already '%s'", id, p.getStatus());
251 		// Update a copy: the live pet must not change unless the store accepts the write.
252 		var sold = new Pet().setId(p.getId()).setSpecies(p.getSpecies()).setName(p.getName()).setPrice(p.getPrice())
253 			.setTags(p.getTags()).setStatus(PetStatus.SOLD).setPhoto(p.getPhoto());
254 		return ActionResult.success(store().updatePet(sold, actor)).message(f("Sold '%s'", p.getName()));
255 	}

257 	/**
258 	 * Stages an inline edit for review on the Pending changes page.  The pet itself is not changed until the change is applied.
259 	 *
260 	 * <p>
261 	 * {@link PendingChanges#stage} owns which fields are stageable and its messages.  The photo is deliberately not
262 	 * stageable: the pet detail Photo tab edits it directly through the API's own {@code PUT /petstore/pets/{id}/photo}.
263 	 *
264 	 * @param req The request.
265 	 * @param id The pet id.
266 	 * @param body {@code {field, value}}, or the inline-edit envelope {@code {fields:{<field>:value}}} with exactly one field.
267 	 * @return {@code success} carrying the new {@link PendingChange}.
268 	 * @throws BadRequest If the field is missing or not stageable, or the value is invalid.
269 	 * @throws NotFound If the pet does not exist.
270 	 */
271 	@RestPost(path="/{id}/stage")
272 	public ActionResult stage(RestRequest req, @Path("id") long id, @Content JsonMap body) {
273 		var actor = ConsoleWrites.actor(req);
274 		String field;
275 		String value;
276 		// The 'edit' column renderer commits {fields:{<column>:value}} (plus its dialog envelope keys) through the row action.
277 		var fields = body.getMap("fields");
278 		if (fields != null) {
279 			if (fields.size() != 1)
280 				throw new BadRequest("Exactly one field can be staged at a time; got '%s'", fields.size());
281 			field = fields.keySet().iterator().next();
282 			value = fields.getString(field);
283 		} else {
284 			field = body.getString("field");
285 			value = body.getString("value");
286 		}
287 		if (field == null || field.isBlank())
288 			throw new BadRequest("Missing field to stage");
289 		try {
290 			var c = store().pendingChanges().stage(id, field, value, ConsoleWrites.author(actor));
291 			return ActionResult.success(c).message(f("Staged '%s'", c.getSummary()));
292 		} catch (PetstoreNotFoundException e) {
293 			throw new NotFound(e.getMessage());
294 		} catch (IllegalArgumentException e) {
295 			throw new BadRequest(e.getMessage());
296 		}
297 	}

390 	private View petPage(RestRequest req, RestResponse res, long id, String tab, Order order) {
391 		var p = store().getPet(id);
392 		if (p == null) {
393 			// A View, not a thrown NotFound, so the not-found card renders inside the chrome.
394 			res.setStatus(404);
395 			return FreemarkerView.of("pet-not-found.ftlh").attr("message", f("Unknown pet '%s'", id));
396 		}
397 		// Node ids must start with a letter, so the pet node is p<id> and the order node o<id>.
398 		var nav = order == null ? f("pets/p%s/%s", id, tab) : f("pets/p%s/orders/o%s", id, order.getId());
399 		// The API's CSRF token, published by the console boundary on allowed requests; absent means the upload form renders without one.
400 		var servletReq = req.getHttpServletRequest();
401 		var apiCsrf = servletReq.getAttribute(PetstoreApiBoundary.TOKEN_ATTRIBUTE) instanceof String s ? s : null;
402 		if (builder(req))
403 			return twin(req, PetsPageBuilders.pet(p, tab, nav, order, ConsoleWrites.roles(req), apiCsrf));
404 		var v = FreemarkerView.of("pet.ftlh").attr("pet", p).attr("tab", tab).attr("activeNav", nav)
405 			.attr("pageTitle", p.getName()).attr("roles", ConsoleWrites.roles(req));
406 		if (apiCsrf != null)
407 			v = v.attr("apiCsrf", apiCsrf);
408 		return order == null ? v : v.attr("order", order);
409 	}

411 	private Order requireOrder(long id, long orderId) {
412 		var o = store().getOrder(orderId);
413 		if (o == null || o.getPetId() != id)
414 			throw new NotFound("Unknown order '%s'", orderId);
415 		return o;
416 	}

430 	/** Unwraps the dialog envelope ({@code fields}) if present; otherwise the body is the pet itself. */
431 	private static Pet toPet(JsonMap body) {
432 		var fields = body.getMap("fields");
433 		var src = fields != null ? fields : body;
434 		try {
435 			return Json.to(Json.of(src), Pet.class);
436 		} catch (ParseException e) {
437 			throw new BadRequest("Invalid pet: '%s'", e.getMessage());
438 		}
439 	}

441 	private static void validate(Pet p) {
442 		if (p.getName() == null || p.getName().isBlank())
443 			throw new BadRequest("Name must not be blank");
444 		if (Float.isNaN(p.getPrice()) || Float.isInfinite(p.getPrice()) || p.getPrice() < 0)
445 			throw new BadRequest("Price '%s' must be a non-negative number", p.getPrice());
446 	}
