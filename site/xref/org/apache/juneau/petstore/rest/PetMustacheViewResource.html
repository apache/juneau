17  package org.apache.juneau.petstore.rest;
18  
19  import org.apache.juneau.commons.inject.*;
20  import org.apache.juneau.http.*;
21  import org.apache.juneau.http.response.*;
22  import org.apache.juneau.petstore.service.*;
23  import org.apache.juneau.rest.server.*;
24  import org.apache.juneau.rest.server.servlet.*;
25  import org.apache.juneau.rest.server.view.*;
26  import org.apache.juneau.rest.server.view.mustache.*;
27  
28  /**
29   * Petstore Mustache view demo.
30   *
31   * <p>
32   * Demonstrates rendering a templated HTML pet detail page through the {@code juneau-rest-server-view-mustache}
33   * bridge.  Lives in {@code juneau-petstore-core} so both the Jetty and Spring Boot deployments inherit it
34   * unchanged — Mustache is a pure-classpath engine and renders identically under either embedded container.
35   *
36   * <p>
37   * The {@code @RestGet("/pets/{id}/view")} method returns {@link MustacheView}; the configured
38   * {@link MustacheViewRenderer} (registered via {@code @Rest(responseProcessors=...)}) detects the return type,
39   * resolves the {@code pet.mustache} template under {@code petstore-templates/} on the classpath, and renders
40   * it directly to the response writer.
41   *
42   * <h5 class='section'>See Also:</h5><ul>
43   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/MustacheViewSupport">Mustache View Support</a>
44   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauPetstore">juneau-petstore</a>
45   * </ul>
46   */
47  @Rest(
48  	path="/pet-views/mustache",
49  	title="Pet detail (Mustache)",
50  	description="Renders a pet detail page through the Mustache view-rendering bridge.",
51  	responseProcessors={MustacheViewRenderer.class}
52  )
53  @SuppressWarnings({
54  	"java:S110" // Inheritance depth acceptable for example/demo code
55  })
56  public class PetMustacheViewResource extends BasicRestServlet {
57  
58  	private static final long serialVersionUID = 1L;
59  
60  	private final transient PetStore store = new PetStore();
61  
62  	/**
63  	 * Provides a {@link MustacheMixin} configured to resolve templates under {@code petstore-templates/} with
64  	 * an implicit {@code .mustache} suffix — so {@code MustacheView.of("pet")} resolves to the classpath
65  	 * resource {@code petstore-templates/pet.mustache}.
66  	 *
67  	 * @return The mustache mixin bean.
68  	 */
69  	@Bean
70  	public MustacheMixin mustache() {
71  		return MustacheMixin.create()
72  			.basePath("/petstore-templates/")
73  			.templateSuffix(".mustache")
74  			.build();
75  	}
76  
77  	/**
78  	 * GET /pets/{id}/view — renders a pet detail HTML page through Mustache.
79  	 *
80  	 * @param id The pet ID.
81  	 * @return A {@link MustacheView} bound to the {@code pet} template.
82  	 * @throws NotFound If no pet with the given ID exists.
83  	 */
84  	@RestGet("/pets/{id}/view")
85  	public View viewPet(@Path("id") long id) {
86  		var pet = store.getPet(id);
87  		if (pet == null)
88  			throw new NotFound("Pet not found: id=%s", id);
89  		return MustacheView.of("pet").attr("pet", pet);
90  	}
91  }
