17  package org.apache.juneau.petstore.rest;
18  
19  import org.apache.juneau.commons.inject.*;
20  import org.apache.juneau.http.*;
21  import org.apache.juneau.http.response.*;
22  import org.apache.juneau.petstore.service.*;
23  import org.apache.juneau.rest.server.*;
24  import org.apache.juneau.rest.server.servlet.*;
25  import org.apache.juneau.rest.server.view.*;
26  import org.apache.juneau.rest.server.view.freemarker.*;
27  
28  /**
29   * Petstore FreeMarker view demo.
30   *
31   * <p>
32   * Pure-classpath analogue of {@link PetMustacheViewResource}: the FreeMarker bridge is engine-agnostic and
33   * renders identically under both the Jetty and Spring Boot deployments.  Resolves
34   * {@code petstore-templates/pet.ftlh} via the configured {@link FreemarkerMixin}.
35   *
36   * <h5 class='section'>See Also:</h5><ul>
37   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/FreemarkerViewSupport">FreeMarker View Support</a>
38   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauPetstore">juneau-petstore</a>
39   * </ul>
40   */
41  @Rest(
42  	path="/pet-views/freemarker",
43  	title="Pet detail (FreeMarker)",
44  	description="Renders a pet detail page through the FreeMarker view-rendering bridge.",
45  	responseProcessors={FreemarkerViewRenderer.class}
46  )
47  @SuppressWarnings({
48  	"java:S110" // Inheritance depth acceptable for example/demo code
49  })
50  public class PetFreemarkerViewResource extends BasicRestServlet {
51  
52  	private static final long serialVersionUID = 1L;
53  
54  	private final transient PetStore store = new PetStore();
55  
56  	/**
57  	 * Provides a {@link FreemarkerMixin} configured to resolve templates under {@code petstore-templates/}.
58  	 *
59  	 * @return The freemarker mixin bean.
60  	 */
61  	@Bean
62  	public FreemarkerMixin freemarker() {
63  		return FreemarkerMixin.create()
64  			.basePath("/petstore-templates/")
65  			.build();
66  	}
67  
68  	/**
69  	 * GET /pets/{id}/view — renders a pet detail HTML page through FreeMarker.
70  	 *
71  	 * @param id The pet ID.
72  	 * @return A {@link FreemarkerView} bound to the {@code pet.ftlh} template.
73  	 * @throws NotFound If no pet with the given ID exists.
74  	 */
75  	@RestGet("/pets/{id}/view")
76  	public View viewPet(@Path("id") long id) {
77  		var pet = store.getPet(id);
78  		if (pet == null)
79  			throw new NotFound("Pet not found: id=%s", id);
80  		return FreemarkerView.of("pet.ftlh").attr("pet", pet);
81  	}
82  }
