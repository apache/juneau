30  /**
31   * Guards shipped resources against a literal closing script tag ({@code </script}, matched case-insensitively).
32   *
33   * <p>
34   * Host pages may inline JavaScript or CSS resources into an HTML {@code <script>} block.  The HTML parser ends that
35   * block at the first closing script tag it sees, even one inside a comment or string, which silently truncates the
36   * rest of the resource.  Nothing fails loudly, so this check names each offending file and line instead.
37   *
38   * <h5 class='section'>Example:</h5>
39   * <p class='bjava'>
40   * 	<jc>// Fail if any *.js or *.css file beside juneau-views.js contains a closing script tag.</jc>
41   * 	InlineScriptSafety.<jsm>assertNoScriptCloseTag</jsm>(ViewsMixin.<jk>class</jk>, <js>"/org/apache/juneau/views/juneau-views.js"</js>, <js>".js"</js>, <js>".css"</js>);
42   * </p>
43   */
44  public final class InlineScriptSafety {
45  
46  	private static final Pattern CLOSING_SCRIPT = Pattern.compile("</script", Pattern.CASE_INSENSITIVE);
47  
48  	private InlineScriptSafety() {}
49  
50  	/**
51  	 * Asserts that no resource beside the specified anchor resource contains a closing script tag.
52  	 *
53  	 * <p>
54  	 * The directory is located through the classpath, so the resources must be on the file system (not in a jar).
55  	 *
56  	 * @param anchor Class used to resolve <c>resource</c>.
57  	 * @param resource Resource path, absolute or relative to <c>anchor</c>'s package, of a file in the directory to scan.
58  	 * @param extensions File-name suffixes to scan (e.g. <js>".js"</js>).  Must not be empty.
59  	 * @throws IOException If the directory cannot be located or a file cannot be read.
60  	 * @throws AssertionFailedError If no matching files were found, or any contains a closing script tag.
61  	 */
62  	public static void assertNoScriptCloseTag(Class<?> anchor, String resource, String... extensions) throws IOException {
63  		var files = listResources(anchor, resource, extensions);
64  		if (files.isEmpty())
65  			throw new AssertionFailedError(f("No resources matching '%s' found beside '%s'", String.join("', '", extensions), resource));
66  		var found = new ArrayList<String>();
67  		for (var p : files)
68  			found.addAll(findCloseTags(p.getFileName().toString(), new String(Files.readAllBytes(p), UTF_8)));
69  		if (! found.isEmpty())
70  			throw new AssertionFailedError(f("Closing script tag found at '%s'", String.join("', '", found)));
71  	}
72  
73  	/**
74  	 * Lists the files in the directory containing the specified resource that end with one of the specified suffixes.
75  	 *
76  	 * @param anchor Class used to resolve <c>resource</c>.
77  	 * @param resource Resource path, absolute or relative to <c>anchor</c>'s package, of a file in the directory to scan.
78  	 * @param extensions File-name suffixes to include.
79  	 * @return The matching files, sorted by path.  Never <jk>null</jk>.
80  	 * @throws IOException If the resource does not exist, is not on the file system, or the directory cannot be listed.
81  	 */
82  	public static List<Path> listResources(Class<?> anchor, String resource, String... extensions) throws IOException {
83  		var url = anchor.getResource(resource);
84  		if (url == null)
85  			throw new FileNotFoundException(f("Resource not found: '%s'", resource));
86  		Path dir;
87  		try {
88  			dir = Path.of(url.toURI()).getParent();
89  		} catch (URISyntaxException | FileSystemNotFoundException e) {
90  			throw new IOException(f("Resource is not on the file system: '%s'", url), e);
91  		}
92  		try (var s = Files.list(dir)) {
93  			return s.filter(p -> {
94  				var name = p.getFileName().toString();
95  				return Arrays.stream(extensions).anyMatch(name::endsWith);
96  			}).sorted().toList();
97  		}
98  	}
99  
100 	/**
101 	 * Finds each line of the specified text containing a closing script tag.
102 	 *
103 	 * @param name Name used to identify the text in the results (typically the file name).
104 	 * @param text The text to scan.
105 	 * @return A <c>name:line</c> entry (1-based) for each offending line.  Never <jk>null</jk>.
106 	 */
107 	public static List<String> findCloseTags(String name, String text) {
108 		var out = new ArrayList<String>();
109 		var lines = text.split("\n", -1);
110 		for (var i = 0; i < lines.length; i++)
111 			if (CLOSING_SCRIPT.matcher(lines[i]).find())
112 				out.add(name + ":" + (i + 1));
113 		return out;
114 	}
115 }
