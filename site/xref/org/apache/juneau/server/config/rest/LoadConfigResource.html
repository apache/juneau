17  package org.apache.juneau.server.config.rest;
18  
19  import org.apache.juneau.http.*;
20  import org.apache.juneau.http.response.*;
21  import org.apache.juneau.marshall.json.*;
22  import org.apache.juneau.rest.server.*;
23  import org.apache.juneau.rest.server.servlet.*;
24  import org.apache.juneau.server.config.repository.*;
25  
26  /**
27   * REST resource for loading configuration from a Git repository.
28   *
29   * <p>
30   * Fetches configuration files for a given project and branch,
31   * combining application defaults with project-specific overrides.
32   */
33  @Rest(path = "/configs/*")
34  public class LoadConfigResource extends RestServlet {
35  
36  	private static final long serialVersionUID = 8247663789227304097L;
37  
38  	/**
39  	 * Returns configuration for the specified project and branch as JSON.
40  	 *
41  	 * @param project Project name.
42  	 * @param branch Branch name.
43  	 * @return JSON-serialized configuration map.
44  	 * @throws Exception If configuration cannot be loaded.
45  	 */
46  	@RestGet(path = "/{project}/{branch}/*", produces = "application/json")
47  	public String gets(@Path("project") String project, @Path("branch") String branch) throws Exception {
48  		var jsonSerializer = JsonSerializer.DEFAULT_READABLE;
49  
50  		var config = new GetConfiguration(project, branch);
51  		try {
52  			config.execute();
53  		} catch (@SuppressWarnings("unused") IllegalArgumentException e) {
54  			// A crafted project/branch that escapes the checkout root is rejected by
55  			// FileUtils.resolveSafely(). Map that boundary violation to 403 with a generic
56  			// message so the offending path is never echoed back to the client.
57  			throw new Forbidden("Access denied.");
58  		}
59  
60  		return jsonSerializer.write(config.get());
61  	}
62  }
