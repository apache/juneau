52  /**
53   * Parent class for all microservices.
54   *
55   * <p>
56   * A microservice defines a simple API for starting and stopping simple Java services contained in executable jars.
57   *
58   * <p>
59   * The general command for creating and starting a microservice from a main method is as follows:
60   * <p class='bjava'>
61   * 	<jk>public static void</jk> main(String[] <jv>args</jv>) {
62   * 		Microservice.<jsm>create</jsm>().args(<jv>args</jv>).build().start().join();
63   *  }
64   * </p>
65   *
66   * <p>
67   * Your microservice class must be specified as the <jk>Main-Class</jk> entry in the manifest file of your microservice
68   * jar file if it's an executable jar.
69   *
70   * <h5 class='topic'>Microservice Configuration</h5>
71   *
72   * This class defines the following method for accessing configuration for your microservice:
73   * <ul class='spaced-list'>
74   * 	<li>
75   * 		{@link #getArgs()} - The command-line arguments passed to the jar file.
76   * 	<li>
77   * 		{@link #getConfig()} - An external INI-style configuration file.
78   * 	<li>
79   * 		{@link #getManifest()} - The manifest file for the main jar file.
80   * </ul>
81   *
82   * <h5 class='topic'>Lifecycle Methods</h5>
83   *
84   * Subclasses must implement the following lifecycle methods:
85   * <ul class='spaced-list'>
86   * 	<li>
87   * 		{@link #init()} - Gets executed immediately following construction.
88   * 	<li>
89   * 		{@link #start()} - Gets executed during startup.
90   * 	<li>
91   * 		{@link #stop()} - Gets executed when 'exit' is typed in the console or an external shutdown signal is received.
92   * 	<li>
93   * 		{@link #kill()} - Can be used to forcibly shut down the service.  Doesn't get called during normal operation.
94   * </ul>
95   *
96   * <h5 class='section'>See Also:</h5><ul>
97   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauMicroservice">juneau-microservice Basics</a>
98   * </ul>
99   */

414 		 * Specifies the manifest file of the jar file this microservice is contained within.
415 		 *
416 		 * <p>
417 		 * This is required if you use {@link Microservice#getManifest()}.
418 		 * It's also used to locate initialization values such as <c>Main-Config</c>.
419 		 *
420 		 * <p>
421 		 * If you do not specify the manifest file, we attempt to resolve it through the following methods:
422 		 * <ol class='spaced-list'>
423 		 * 	<li>
424 		 * 		Looking on the file system for a file at <js>"META-INF/MANIFEST.MF"</js>.
425 		 * 		This is primarily to allow for running microservices from within eclipse workspaces where the manifest file
426 		 * 		is located in the project root.
427 		 * 	<li>
428 		 * 		Using the class loader for this class to find the file at the URL <js>"META-INF/MANIFEST.MF"</js>.
429 		 * </ol>
430 		 *
431 		 * @param value
432 		 * 	The manifest file of this microservice.
433 		 * 	<br>Can be any of the following types:
434 		 * 	<ul>
435 		 * 		<li>{@link ManifestFile}
436 		 * 		<li>{@link Manifest}
437 		 * 		<li>{@link Reader} - Containing the raw contents of the manifest.  Note that the input must end with a newline.
438 		 * 		<li>{@link InputStream} - Containing the raw contents of the manifest.  Note that the input must end with a newline.
439 		 * 		<li>{@link File} - File containing the raw contents of the manifest.
440 		 * 		<li>{@link String} - Path to file containing the raw contents of the manifest.
441 		 * 		<li>{@link Class} - Finds and loads the manifest file of the jar file that the specified class is contained within.
442 		 * 	</ul>
443 		 * 	<br>Can be <jk>null</jk> (the manifest is auto-resolved).
444 		 * @return This object.
445 		 * @throws IOException Thrown by underlying stream.
446 		 */
447 		public Builder manifest(Object value) throws IOException {
448 			if (value == null)
449 				this.manifest = null;
450 			else if (value instanceof ManifestFile value2)
451 				this.manifest = value2;
452 			else if (value instanceof Manifest value2)
453 				this.manifest = new ManifestFile(value2);
454 			else if (value instanceof Reader value2)
455 				this.manifest = new ManifestFile(value2);
456 			else if (value instanceof InputStream value2)
457 				this.manifest = new ManifestFile(value2);
458 			else if (value instanceof File value2)
459 				this.manifest = new ManifestFile(value2);
460 			else if (value instanceof Path value2)
461 				this.manifest = new ManifestFile(value2);
462 			else if (value instanceof String value2)
463 				this.manifest = new ManifestFile(resolveFile(value2));
464 			else if (value instanceof Class value2)
465 				this.manifest = new ManifestFile(value2);
466 			else
467 				throw rex("Invalid type passed to Builder.manifest(Object).  Type=[%s]", cn(value));
468 
469 			return this;
470 		}

713 		this.builder = builder.copy();
714 		this.workingDir = builder.workingDir;
715 		this.configName = builder.configName;
716 
717 		// --------------------------------------------------------------------------------
718 		// Initialize the bean store and register any @Configuration classes.
719 		// @Bean-supplied values become candidate inputs for field resolution below.
720 		// Explicit builder calls always win and overwrite @Bean contributions afterward.
721 		//
722 		// Composition with overridingBeanStore (test-time overlay):
723 		//   - no external store, no overlay  -> fresh BasicBeanStore()
724 		//   - no external store, overlay set -> fresh BasicBeanStore(parent=null, overridingParent=overlay)
725 		//   - external store, no overlay     -> use the external store as-is
726 		//   - external store + overlay set   -> wrap with BasicBeanStore(parent=external, overridingParent=overlay)
727 		// --------------------------------------------------------------------------------
728 		if (nn(builder.overridingBeanStore)) {
729 			this.beanStore = new BasicBeanStore(builder.beanStore, builder.overridingBeanStore);
730 		} else {
731 			this.beanStore = nn(builder.beanStore) ? builder.beanStore : new BasicBeanStore();
732 		}
733 		if (! builder.configurations.isEmpty())
734 			beanStore.registerConfigurations(builder.configurations.toArray(new Class<?>[0]));
735 
736 		// Resolve Args: builder > @Bean > empty.
737 		this.args = nn(builder.args)
738 			? builder.args
739 			: beanStore.getBean(Args.class).orElseGet(() -> new Args(new String[0]));
740 		beanStore.addBean(Args.class, this.args);
741 
742 		// --------------------------------------------------------------------------------
743 		// Try to get the manifest file if it wasn't already set.
744 		// --------------------------------------------------------------------------------
745 		var manifest2 = builder.manifest;
746 		if (manifest2 == null)
747 			manifest2 = beanStore.getBean(ManifestFile.class).orElse(null);
748 		if (manifest2 == null) {
749 			var m = new Manifest();

751 			// If running within an eclipse workspace, need to get it from the file system.
752 			var f = resolveFile("META-INF/MANIFEST.MF");

755 					m.read(fis);
756 				} catch (IOException e) {
757 					throw ioex(e, "Problem detected in MANIFEST.MF.  Contents below:\n%s", read(f), e);
758 				}
759 			} else {
760 				// Otherwise, read from manifest file in the jar file containing the main class.
761 				var url = getClass().getResource("META-INF/MANIFEST.MF");
762 				if (nn(url)) {
763 					try {
764 						m.read(url.openStream());
765 					} catch (IOException e) {
766 						throw ioex(e, "Problem detected in MANIFEST.MF.  Contents below:\n%s", read(url.openStream()), e);
767 					}
768 				}
769 			}
770 			manifest2 = new ManifestFile(m);
771 		}
772 		this.manifest = manifest2;
773 		beanStore.addBean(ManifestFile.class, this.manifest);
774 		builder.varResolver
775 			.vars(ArgsVar.create(() -> this.args))
776 			.vars(ManifestFileVar.create(() -> this.manifest));

779 		// Try to resolve the configuration if not specified.
780 		// --------------------------------------------------------------------------------
781 		var config2 = nn(builder.config) ? builder.config : beanStore.getBean(Config.class).orElse(null);
782 		var configBuilder = builder.configBuilder.varResolver(builder.varResolver.build()).store(MemoryStore.DEFAULT);
783 		if (config2 == null) {
784 			var store = builder.configStore;
785 			var cfs = workingDir == null ? FileStore.DEFAULT : FileStore.create().directory(workingDir).build();
786 			boolean found = false;
787 			for (var name : getCandidateConfigNames()) {
788 				if (found)
789 					break;
790 				if (nn(store)) {
791 					if (store.exists(name)) {
792 						configBuilder.store(store).name(name);
793 						found = true;
794 					}
795 				} else {
796 					if (cfs.exists(name)) {
797 						configBuilder.store(cfs).name(name);
798 						found = true;
799 					} else if (ClasspathStore.DEFAULT.exists(name)) {
800 						configBuilder.store(ClasspathStore.DEFAULT).name(name);
801 						found = true;
802 					}
803 				}
804 			}
805 			config2 = configBuilder.build();
806 		}
807 		this.config = config2;
808 		Config.setSystemDefault(this.config);
809 		this.config.addListener(this);
810 		this.settingsSource = new ConfigPropertySource(this.config);
811 		Settings.get().addSource(this.settingsSource);
812 		beanStore.addBean(Config.class, this.config);
813 		this.varResolver = builder.varResolver.bean(Config.class, config2).build();
814 		beanStore.addBean(VarResolver.class, this.varResolver);
815 
816 		// --------------------------------------------------------------------------------
817 		// Initialize console commands.
818 		// --------------------------------------------------------------------------------
819 		this.consoleEnabled = coalesce(builder.consoleEnabled, config.get("Console/enabled").asBoolean().orElse(false));
820 		if (consoleEnabled) {
821 			var c = System.console();
822 			this.consoleReader = coalesce(builder.consoleReader, new Scanner(c == null ? new InputStreamReader(System.in) : c.reader()));
823 			this.consoleWriter = coalesce(builder.consoleWriter, c == null ? createLoggerConsoleWriter() : c.writer());
824 
825 			// @Bean-supplied console commands (registered first, then overridable by builder/config below).
826 			for (var cc : beanStore.getBeansOfType(ConsoleCommand.class).values())
827 				consoleCommandMap.put(cc.getName(), cc);
828 			for (var cc : builder.consoleCommands) {
829 				consoleCommandMap.put(cc.getName(), cc);
830 			}
831 			for (var s : config.get("Console/commands").asStringArray().orElse(new String[0])) {
832 				try {
833 					var cc = (ConsoleCommand)Class.forName(s).getDeclaredConstructor().newInstance();
834 					consoleCommandMap.put(cc.getName(), cc);
835 				} catch (Exception e) {
836 					getConsoleWriter().println("Could not create console command '" + s + "', " + localizedMessage(e));
837 				}
838 			}
839 			consoleThread = new Thread("ConsoleThread") {
840 				@Override /* Overridden from Thread */
841 				public void run() {
842 					var in = getConsoleReader();
843 					var out = getConsoleWriter();
844 
845 					out.println(messages.getString("ListOfAvailableCommands"));
846 					for (var cc : new TreeMap<>(getConsoleCommands()).values())
847 						out.append("\t").append(cc.getName()).append(" -- ").append(cc.getInfo()).println();
848 					out.println();
849 
850 					while (true) {
851 						out.append("> ").flush();
852 						// Guard against empty/closed stdin: hasNextLine() returns false at end-of-stream, so
853 						// the loop ends gracefully instead of nextLine() throwing NoSuchElementException.
854 						if (! in.hasNextLine())
855 							break;
856 						var line = in.nextLine();
857 						var args2 = new Args(line);
858 						if (! args2.isEmpty())
859 							executeCommand(args2, in, out);
860 					}
861 				}
862 			};
863 			consoleThread.setDaemon(true);
864 		} else {
865 			this.consoleReader = null;
866 			this.consoleWriter = null;
867 			this.consoleThread = null;
868 		}
869 		// Resolve listeners.
870 		// If the builder supplies an explicit listener, register it under the unnamed key (overwriting any
871 		// same-key @Bean MicroserviceListener); it joins the fan-out alongside any other @Bean-supplied listeners.
872 		// If, after that, no MicroserviceListener exists in the store at all, register the default
873 		// BasicMicroserviceListener so the fan-out has at least one no-op participant.
874 		if (nn(builder.listener))
875 			beanStore.addBean(MicroserviceListener.class, builder.listener);
876 		if (beanStore.getBeansOfType(MicroserviceListener.class).isEmpty())
877 			beanStore.addBean(MicroserviceListener.class, new BasicMicroserviceListener());
878 
879 		// Self-register so @Bean factory methods can take Microservice as a constructor/method param.
880 		beanStore.addBean(Microservice.class, this);
881 
882 		init();
883 	}
