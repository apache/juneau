17  package org.apache.juneau.microservice.management;
18  
19  import org.apache.juneau.marshall.collections.*;
20  import org.apache.juneau.rest.server.*;
21  import org.apache.juneau.rest.server.servlet.*;
22  
23  /**
24   * Child-resource flavor of the {@code /info} endpoint.
25   *
26   * <p>
27   * Mounts as a routed child via {@link Rest#children() @Rest(children=InfoResource.class)} under the subtree
28   * {@code /info} and delegates to a shared {@link InfoManager} worker &mdash; the same logic the
29   * {@link InfoMixin mixin} flavor uses, so the two forms cannot drift.  Extends {@link BasicRestResource} so
30   * the returned bean serializes via the {@code BasicUniversalConfig} set (mirrors {@code HealthResource}).
31   *
32   * <h5 class='section'>See Also:</h5><ul>
33   * 	<li class='jc'>{@link InfoMixin}
34   * 	<li class='jc'>{@link InfoManager}
35   * </ul>
36   *
37   * @since 10.0.0
38   */
39  @Rest(path="/info")
40  public class InfoResource extends BasicRestResource {
41  
42  	private final InfoManager manager = new InfoManager();
43  
44  	/**
45  	 * [GET /] - Application build/version/git metadata from the manifest.
46  	 *
47  	 * @param req The HTTP request.
48  	 * @return The manifest main attributes as a sorted map (empty when no manifest is registered).
49  	 */
50  	@RestGet(
51  		path="/*",
52  		summary="Application info",
53  		description="Renders the running application's manifest main attributes (build/version/git metadata)."
54  	)
55  	public JsonMap getInfo(RestRequest req) {
56  		return manager.getInfo(manager.resolveManifest(req.getContext()));
57  	}
58  }
