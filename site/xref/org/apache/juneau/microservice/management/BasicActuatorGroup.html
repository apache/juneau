17  package org.apache.juneau.microservice.management;
18  
19  import org.apache.juneau.rest.server.*;
20  import org.apache.juneau.rest.server.health.*;
21  import org.apache.juneau.rest.server.management.*;
22  import org.apache.juneau.rest.server.servlet.*;
23  
24  /**
25   * Convenience assembly of the Actuator-style management surface under a single configurable path prefix.
26   *
27   * <p>
28   * Mounts the management endpoints by composing their {@code *Mixin} flavors via
29   * {@link Rest#mixins() @Rest(mixins=...)} on top of {@link BasicRestServletGroup}:
30   * <ul>
31   * 	<li>{@code /healthz}, {@code /readyz}, {@code /livez} &mdash; {@link HealthMixin} (read-through health view)
32   * 	<li>{@code /threaddump}, {@code /heapdump} &mdash; {@link DumpsMixin} (deny-by-default diagnostics)
33   * </ul>
34   *
35   * <p>
36   * <b>{@code /info} and {@code /loggers} are off by default</b> &mdash; unlike health, they are not mounted
37   * on the bare group, since {@link InfoMixin} discloses the full manifest and {@link LoggersMixin}'s read side
38   * discloses logger topology, neither behind any auth. Turn either back on (either approach suffices):
39   * <ul>
40   * 	<li><b>A-la-carte (recommended):</b> mount {@link InfoMixin} / {@link LoggersMixin} on a resource you
41   * 		control, e.g. a {@code BasicActuatorGroup} subclass with
42   * 		{@code @Rest(mixins={InfoMixin.class, LoggersMixin.class})} &mdash; mixins declared on a subclass are
43   * 		additive to the mixins inherited from this class, so {@code HealthMixin}/{@code DumpsMixin} stay mounted
44   * 		too.
45   * 	<li><b>Zero-code:</b> not currently supported &mdash; {@code @Rest(mixins=...)} is a compile-time class
46   * 		list, and the {@code RestContext.Builder} injection point that would have let a system property add a
47   * 		mixin at init time was removed prior to 10.0.0 (see {@link RestInit}).  Use the a-la-carte subclass above.
48   * </ul>
49   *
50   * <p>
51   * <b>Path prefix:</b> defaults to {@code /actuator} and is configurable via the {@code juneau.actuator.path}
52   * system property (resolved through the standard {@code $S{...}} SVL var).  Mount the group at a custom prefix
53   * either by setting that property or by subclassing with your own {@link Rest#path() @Rest(path=...)}.
54   *
55   * <p>
56   * <b>Composition is convenience, not the only way in:</b> every endpoint remains independently mountable
57   * a-la-carte via its standalone {@code *Mixin}/{@code *Resource} flavor.  The {@code /metrics} endpoint is
58   * <b>not</b> assembled here &mdash; it lives in the {@code juneau-rest-server-metrics-micrometer} module
59   * (which this module does not depend on), so add {@code MetricsMixin} a-la-carte when that module is present.
60   *
61   * <p>
62   * <b>Exposure policy:</b> of the mounted endpoints, health is on; the mutating/sensitive ones
63   * ({@code /threaddump}, {@code /heapdump}) are deny-by-default via {@link DumpsSettings}.  No auth provider is
64   * auto-wired (explicit-over-magic).
65   *
66   * <h5 class='section'>See Also:</h5><ul>
67   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/ManagementSurface">Management Surface</a>
68   * </ul>
69   *
70   * @serial exclude
71   * @since 10.0.0
72   */
73  @Rest(
74  	path="$S{juneau.actuator.path,/actuator}",
75  	title="Management",
76  	description="Actuator-style management surface.",
77  	mixins={
78  		HealthMixin.class,
79  		DumpsMixin.class
80  	}
81  )
82  @SuppressWarnings({
83  	"java:S110" // Inheritance depth acceptable for microservice hierarchy.
84  })
85  public class BasicActuatorGroup extends BasicRestServletGroup {
86  	private static final long serialVersionUID = 1L;
87  }
