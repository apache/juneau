52  /**
53   * Drives the embedded Tomcat server lifecycle on behalf of the microservice via the {@link MicroserviceListener} fan-out.
54   *
55   * <p>
56   * Contributed by {@link TomcatConfiguration} as a <c>@Bean</c>.  The microservice's {@link Microservice#start()}
57   * call invokes {@link #onStart(Microservice)}, which:
58   * <ol>
59   * 	<li>Picks the first available port from the resolved {@link TomcatSettings#getPorts() ports} (falling back to
60   * 		<c>Tomcat/port</c> config, then <c>Tomcat-Port</c> manifest, then <c>8000</c>).
61   * 	<li>Resolves the {@link Tomcat} instance — preferring an externally-supplied <c>@Bean Tomcat</c>, otherwise
62   * 		building one programmatically via the {@link TomcatServerFactory} against a writable base directory.
63   * 	<li>Mounts servlets from {@code Tomcat/servlets}, {@code Tomcat/servletMap}, {@code Tomcat/servletAttributes}
64   * 		config sections, plus auto-mounts every {@code @Bean Servlet} whose runtime class is annotated with
65   * 		{@link Rest @Rest} (at {@link Rest#path()}, or <c>"/"</c> when no path is set).
66   * 	<li>Materializes the connector via {@link Tomcat#getConnector()}, starts the server, and publishes the bound
67   * 		port via the {@code availablePort} / {@code juneau.serverPort} system properties.
68   * </ol>
69   *
70   * <p>
71   * {@link #onStop(Microservice)} performs the inverse: stops and destroys the server gracefully in a worker thread,
72   * then deletes the auto-created temp base directory (if one was created).
73   *
74   * <p>
75   * External callers reach this component via
76   * {@code Microservice.getInstance().getBeanStore().getBean(TomcatServerComponent.class).orElseThrow()}.
77   *
78   * <h5 class='section'>See Also:</h5><ul>
79   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauMicroserviceTomcat">juneau-microservice-tomcat Basics</a>
80   * </ul>
81   *
82   * @since 10.0.0
83   */

120 	private static int[] parseIntArray(String csv) {
121 		if (ie(csv))
122 			return new int[0];
123 		var parts = csv.split(",");
124 		var out = new int[parts.length];
125 		for (var i = 0; i < parts.length; i++)
126 			out[i] = Integer.parseInt(parts[i].trim());
127 		return out;
128 	}

130 	private static int findOpenPort(int[] ports) {
131 		for (var port : ports) {
132 			if (port == 0)
133 				port = RANDOM.nextInt(32767 - ports[0] + 1) + ports[0];
134 			try (var ss = new ServerSocket(port)) {
135 				return port;
136 			} catch (IOException e) {
137 				// Port is in use, try next port in array
138 			}
139 		}
140 		return 0;
141 	}

143 	private static String normalizePathSpec(String rawPath) {
144 		var p = rawPath == null ? "" : rawPath;
145 		if (p.isEmpty() || "/".equals(p))
146 			return "/*";
147 		if (! p.startsWith("/"))
148 			p = "/" + p;
149 		if (! p.endsWith("/*"))
150 			p = trimTrailingSlashes(p) + "/*";
151 		return p;
152 	}
153 
154 	private static String normalizeExactPathSpec(String rawPath) {
155 		var p = rawPath == null ? "" : trimTrailingSlashes(rawPath);
156 		if (p.isEmpty())
157 			return "/";
158 		if (! p.startsWith("/"))
159 			p = "/" + p;
160 		return p;
161 	}

163 	/**
164 	 * Resolves the top-level Tomcat mount path-specs for an auto-discovered {@code @Rest}-annotated servlet.
165 	 *
166 	 * <p>
167 	 * Delegates to {@link RestContext#resolveTopLevelPaths(Class, Object, BeanStore)} so the runtime-override
168 	 * resolution chain (programmatic &gt; {@code getPaths()} getter &gt; annotation default) is honored at
169 	 * mount time. The programmatic Builder rung is N/A for Tomcat auto-discovery (the {@code RestContext} does
170 	 * not exist yet at this point in the lifecycle); users can substitute by overriding {@code getPaths()} on
171 	 * the resource class, or by leaning on SVL inside {@code @Rest(paths=...)} elements (e.g.
172 	 * {@code paths={"$C{health.paths}"}}) so each element is resolved through {@code Config} /
173 	 * {@code $E{...}} / {@code $S{...}} before being comma-split.
174 	 *
175 	 * @return The normalized exact-match path-specs (for multi-mount), or the singular {@code @Rest(path)}-style
176 	 * 	{@code "/...&#47;*"} pathspec when no {@code paths} are resolved.
177 	 */
178 	private static String[] restPathsFor(Servlet servlet, BeanStore store) {
179 		var cls = servlet.getClass();
180 		var r = cls.getAnnotation(Rest.class);
181 		if (r == null)
182 			return new String[] {"/*"};
183 		var resolved = RestContext.resolveTopLevelPaths(cls, servlet, store);
184 		if (resolved.length > 0)
185 			return Arrays.stream(resolved).map(TomcatServerComponent::normalizeExactPathSpec).toArray(String[]::new);
186 		return new String[] {normalizePathSpec(r.path())};
187 	}

198 	@SuppressWarnings({
199 		"java:S3776" // Cognitive complexity acceptable for server creation logic

201 	public void onStart(Microservice ms) {
202 		try {
203 			microservice.set(ms);
204 			var store = ms.getBeanStore();
205 			ClassInfo.of(this).inject(this, store);
206 			var cf = ms.getConfig();
207 			var mf = ms.getManifest();
208 			var settings = store.getBean(TomcatSettings.class).orElseGet(() -> TomcatSettings.create().build());
209 
210 			// Port resolution: settings > Tomcat/port config > Tomcat-Port manifest > 8000
211 			var ports = settings.getPorts();
212 			if (ports == null)
213 				ports = cf.get("Tomcat/port").as(int[].class)
214 					.orElseGet(() -> mf.get("Tomcat-Port").map(TomcatServerComponent::parseIntArray).orElseGet(() -> ints(8000)));
215 			var availablePort = findOpenPort(ports);
216 
217 			if (availablePortEnv.isEmpty())
218 				// Intentional real-JVM-property publish: the bound port is exposed as a process-global system
219 				// property so external config/templates (e.g. the bundled jetty.xml's $S{availablePort}) can read
220 				// it.  The $S{...} SVL var resolves via raw System.getProperty (bypassing the Settings override
221 				// chain), so a Settings-based write would be invisible to those consumers.  The @Value-injected
222 				// availablePortEnv read side is already Settings-abstracted; only this publish must stay raw.
223 				System.setProperty("availablePort", String.valueOf(availablePort));
224 
225 			// Prefer a @Bean-supplied Tomcat, else build one programmatically via the factory.
226 			var injectedTomcat = store.getBean(Tomcat.class).orElse(null);
227 			if (nn(injectedTomcat)) {
228 				tomcat.set(injectedTomcat);
229 			} else {
230 				var factory = store.getBean(TomcatServerFactory.class).orElseGet(BasicTomcatServerFactory::new);
231 				tomcat.set(createServer(factory, resolveBaseDir(settings)));
232 			}
233 			tomcat.get().setPort(availablePort);
234 			tomcat.get().getConnector();
235 
236 			// Publish back to the store for downstream beans / lookups.
237 			store.addBean(Tomcat.class, tomcat.get());
238 
239 			// Graceful-shutdown wiring (zero-downtime k8s rollouts), kept identical to the Jetty contract:
240 			//  - Resolve the bounded drain timeout (TomcatSettings > Tomcat/stopTimeout config > 30s default) so
241 			//    onStop() can wait for in-flight requests to complete before the server stops.
242 			//  - Remember the settle delay so onStop() can let the LB observe the /readyz 503 before draining.
243 			//  - Mark readiness ready so a freshly-(re)started service serves traffic.
244 			stopTimeout.set(coalesce(settings.getStopTimeout(), cf.get("Tomcat/stopTimeout").asLong().map(Duration::ofMillis).orElse(DEFAULT_STOP_TIMEOUT)));
245 			shutdownSettleDelay.set(coalesce(settings.getShutdownSettleDelay(), cf.get("Tomcat/shutdownSettleDelay").asLong().map(Duration::ofMillis).orElse(Duration.ZERO)));
246 			ReadinessState.resolve(store).markReady();
247 
248 			// Track each servlet pathSpec with its declaring source so we can fail loudly on collisions.
249 			var mountedPaths = new LinkedHashMap<String,String>();
250 
251 			for (var s : cf.get("Tomcat/servlets").asStringArray().orElse(new String[0]))
252 				mountConfiguredServlet(s, mountedPaths);
253 
254 			cf.get("Tomcat/servletMap").asMap().orElse(EMPTY_MAP).forEach((k, v) -> mountMappedServlet(k, v, mountedPaths));
255 
256 			cf.get("Tomcat/servletAttributes").asMap().orElse(EMPTY_MAP).forEach(this::addServletAttribute);
257 
258 			// Auto-mount @Bean AuthFilterChain at /* before any servlet registration so the filter
259 			// runs before routing for all requests.
260 			store.getBean(AuthFilterChain.class).ifPresent(chain -> addFilter(chain, "/*"));
261 
262 			// Auto-discover @Rest servlets contributed via @Configuration/@Bean methods.
263 			// Path source precedence: @Rest(path=...) on the resource class, else "/".
264 			for (var e : store.getBeansOfType(Servlet.class).entrySet()) {
265 				var servlet = e.getValue();
266 				var cls = servlet.getClass();
267 				if (cls.getAnnotation(Rest.class) == null)
268 					continue;
269 				var pathSpecs = restPathsFor(servlet, store);
270 				var source = "@Bean " + cls.getName() + (ine(e.getKey()) ? "[" + e.getKey() + "]" : "");
271 				for (var pathSpec : pathSpecs)
272 					checkPathCollision(pathSpec, source, mountedPaths);
273 				addServlet(servlet, pathSpecs);
274 			}
275 
276 			if (serverPortEnv.isEmpty())
277 				// Intentional real-JVM-property publish: juneau.serverPort is a documented cross-component/external
278 				// publication of the bound port, consumed via raw System.getProperty (and $S{...} SVL).  A
279 				// Settings-based write would not be visible to those raw readers, so this is kept.  The
280 				// @Value-injected serverPortEnv read side is already Settings-abstracted.
281 				System.setProperty("juneau.serverPort", String.valueOf(availablePort));
282 
283 			tomcat.get().start();
284 			ms.out(messages, "ServerStarted", getPort());
285 		} catch (Exception e) {
286 			throw toRex(e);
287 		}
288 	}

290 	@Override /* Overridden from MicroserviceListener */
291 	public void onStop(Microservice ms) {
292 		final Logger logger = ms.getLogger();
293 		// Flip readiness out of service BEFORE the connector stops so /readyz returns 503 and the load balancer /
294 		// Kubernetes stops routing new traffic while in-flight requests drain.  /livez stays healthy so the pod is
295 		// not killed mid-drain.  The settle delay (default 0) gives the LB a window to observe the 503.
296 		ReadinessState.resolve(ms.getBeanStore()).markOutOfService();
297 		ms.out(messages, "DrainingRequests");
298 		sleepQuietly(shutdownSettleDelay.get());
299 		var t = new Thread("TomcatServerComponentStop") {
300 			@Override /* Overridden from Thread */
301 			public void run() {
302 				try {
303 					var t2 = tomcat.get();
304 					if (t2 == null)
305 						return;
306 					drainConnector(t2, stopTimeout.get(), logger);
307 					ms.out(messages, "StoppingServer");
308 					t2.stop();
309 					t2.destroy();
310 					ms.out(messages, "ServerStopped");
311 				} catch (Exception e) {
312 					logger.log(Level.WARNING, localizedMessage(e), e);
313 				} finally {
314 					cleanupBaseDir();
315 				}
316 			}
317 		};
318 		t.start();
319 		try {
320 			t.join();
321 		} catch (InterruptedException e) {
322 			Thread.currentThread().interrupt();
323 		}
324 	}

326 	/**
327 	 * Pauses the connector (stops accepting new connections) and waits up to {@code timeout} for in-flight requests
328 	 * to drain before the caller stops the server.  Best-effort: never throws out of the shutdown sequence.
329 	 */
330 	private static void drainConnector(Tomcat t2, Duration timeout, Logger logger) {
331 		try {
332 			var connector = t2.getConnector();
333 			if (connector == null)
334 				return;
335 			connector.pause();
336 			var executor = connector.getProtocolHandler() == null ? null : connector.getProtocolHandler().getExecutor();
337 			if (executor instanceof ThreadPoolExecutor executor2) {
338 				var deadline = System.nanoTime() + Math.max(0L, timeout.toNanos());
339 				while (executor2.getActiveCount() > 0 && System.nanoTime() < deadline) {
340 					if (sleepInterruptibly(50L))
341 						return;
342 				}
343 			}
344 		} catch (Exception e) {
345 			logger.log(Level.WARNING, localizedMessage(e), e);
346 		}
347 	}

349 	private static void sleepQuietly(Duration d) {
350 		if (d == null || d.isZero() || d.isNegative())
351 			return;
352 		try {
353 			Thread.sleep(d.toMillis());
354 		} catch (@SuppressWarnings("unused") InterruptedException e) {
355 			Thread.currentThread().interrupt();
356 		}
357 	}

359 	private static boolean sleepInterruptibly(long millis) {
360 		try {
361 			Thread.sleep(millis);
362 			return false;
363 		} catch (@SuppressWarnings("unused") InterruptedException e) {
364 			Thread.currentThread().interrupt();
365 			return true;
366 		}
367 	}

423 		var filterName = cn(filter) + "#" + UUID.randomUUID();
424 		var fd = new FilterDef();
425 		fd.setFilter(filter);
426 		fd.setFilterName(filterName);
427 		context.addFilterDef(fd);
428 		for (var pattern : urlPatterns) {
429 			var fm = new FilterMap();
430 			fm.setFilterName(filterName);
431 			fm.addURLPattern(pattern);
432 			fm.setDispatcher(DispatcherType.REQUEST.name());
433 			context.addFilterMap(fm);
434 		}
435 		return this;
436 	}

536 	private File resolveBaseDir(TomcatSettings settings) {
537 		var configured = settings.getBaseDir();
538 		if (ine(configured)) {
539 			ownsBaseDir.set(false);
540 			var f = new File(configured);
541 			baseDir.set(f);
542 			return f;
543 		}
544 		try {
545 			var f = createTempBaseDir().toFile();
546 			ownsBaseDir.set(true);
547 			baseDir.set(f);
548 			return f;
549 		} catch (IOException e) {
550 			throw toRex(e);
551 		}
552 	}

555 		"java:S5443" // Temp dir created via Files.createTempDirectory with owner-only (rwx------) POSIX perms; non-POSIX fallback relies on OS-default owner-restricted temp dirs.
556 	})
557 	private static Path createTempBaseDir() throws IOException {
558 		try {
559 			// Explicitly restrict to owner-only access (rwx------) on POSIX file systems.
560 			Set<PosixFilePermission> ownerOnly = EnumSet.of(
561 				PosixFilePermission.OWNER_READ,
562 				PosixFilePermission.OWNER_WRITE,
563 				PosixFilePermission.OWNER_EXECUTE);
564 			return Files.createTempDirectory("juneau-tomcat",
565 				PosixFilePermissions.asFileAttribute(ownerOnly));
566 		} catch (UnsupportedOperationException e) {
567 			// Non-POSIX filesystem (e.g. Windows) — fall back to default temp-dir creation.
568 			return Files.createTempDirectory("juneau-tomcat");
569 		}
570 	}

580 	private void mountConfiguredServlet(String className, Map<String,String> mountedPaths) {
581 		try {
582 			var c = info(Class.forName(className));
583 			if (c.isAssignableTo(RestServlet.class)) {
584 				var rs = (RestServlet)c.newInstance();
585 				mountWithCollisionCheck(rs, rs.getPath(), "Tomcat/servlets[" + className + "]", mountedPaths);
586 			} else {
587 				throw rex("Invalid servlet specified in Tomcat/servlets.  Must be a subclass of RestServlet: %s", className);
588 			}
589 		} catch (ClassNotFoundException e) {
590 			throw toRex(e);
591 		}
592 	}

600 			} else {
601 				throw rex("Invalid servlet specified in Tomcat/servletMap.  Must be a subclass of Servlet: %s", cn(className));
602 			}
603 		} catch (ClassNotFoundException e) {
604 			throw toRex(e);
605 		}

608 	private void mountWithCollisionCheck(Servlet servlet, String rawPath, String source, Map<String,String> mountedPaths) {
609 		var pathSpec = normalizePathSpec(rawPath);
610 		checkPathCollision(pathSpec, source, mountedPaths);
611 		addServlet(servlet, pathSpec);
612 	}
613 
614 	private void checkPathCollision(String pathSpec, String source, Map<String,String> mountedPaths) {
615 		var prior = mountedPaths.get(pathSpec);
616 		if (nn(prior))
617 			throw rex("Servlet mount path collision: '%s' is already mounted by %s; refused by %s.", pathSpec, prior, source);
618 		mountedPaths.put(pathSpec, source);
619 	}
