17  package org.apache.juneau.microservice.tomcat;
18  
19  import java.io.*;
20  
21  import org.apache.catalina.startup.*;
22  
23  /**
24   * Interface for creating embedded Tomcat servers.
25   *
26   * <p>
27   * Mirrors {@code org.apache.juneau.marshall.microservice.jetty.JettyServerFactory} but uses the programmatic
28   * {@link Tomcat} façade instead of a {@code jetty.xml} document.
29   *
30   * <h5 class='section'>See Also:</h5><ul>
31   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauMicroserviceTomcat">juneau-microservice-tomcat Basics</a>
32   * </ul>
33   */
34  public interface TomcatServerFactory {
35  
36  	/**
37  	 * Create a new initialized embedded Tomcat server.
38  	 *
39  	 * <p>
40  	 * The returned server is configured with the specified Catalina base directory and a root context (mounted
41  	 * at <c>""</c>) but is not yet started and has no connector port assigned — the {@link TomcatServerComponent}
42  	 * assigns the port and materializes the connector before calling {@link Tomcat#start()}.
43  	 *
44  	 * @param baseDir The writable Catalina base / work directory to use for this server.
45  	 * @return A newly-created but not-yet-started Tomcat server.
46  	 * @throws Exception Any exception.
47  	 */
48  	@SuppressWarnings({
49  		"java:S112" // throws Exception intentional - callback/lifecycle method
50  	})
51  	Tomcat create(File baseDir) throws Exception;
52  }
