17  package org.apache.juneau.microservice.logging;
18  
19  import java.lang.reflect.*;
20  import java.util.logging.*;
21  
22  /**
23   * Reflective accessor for the currently-active OpenTelemetry trace / span id, used by
24   * {@link LogEntryFormatter} to populate the optional {@code {traceId}} / {@code {spanId}} log fields.
25   *
26   * <h5 class='topic'>Optional / reflective coupling</h5>
27   *
28   * <p>
29   * OpenTelemetry is <b>not</b> a compile-time dependency of {@code juneau-microservice}. This class
30   * probes for {@code io.opentelemetry.api.trace.Span} via {@link Class#forName(String)} at class-load
31   * time &mdash; mirroring how {@link org.apache.juneau.rest.server.processor.MdcAsyncListener} reflectively
32   * handles SLF4J. When OTel is absent from the runtime classpath the class gracefully degrades:
33   * {@link #isAvailable()} returns {@code false} and {@link #currentTraceId()} / {@link #currentSpanId()}
34   * return the empty string, so the {@code {traceId}} / {@code {spanId}} placeholders render empty.
35   *
36   * <h5 class='topic'>Where the id comes from</h5>
37   *
38   * <p>
39   * The id is read from {@code Span.current().getSpanContext()} &mdash; the OTel context active on the
40   * calling thread. Inside a {@code @RestOp} handler (or a custom
41   * {@link org.apache.juneau.rest.server.observation.Observations observation}) wrapped by the OTel
42   * {@code TracerHook}, that is the request / observation span. An invalid (no active trace) context
43   * yields the empty string.
44   *
45   * @since 10.0.0
46   */
47  public final class TraceContext {
48  
49  	private static final Logger LOG = Logger.getLogger(TraceContext.class.getName());
50  
51  	/** {@code true} if {@code io.opentelemetry.api.trace.Span} was found on the classpath at class-load time. */
52  	private static final boolean AVAILABLE;
53  
54  	private static final Method SPAN_CURRENT;
55  	private static final Method SPAN_GET_SPAN_CONTEXT;
56  	private static final Method CTX_IS_VALID;
57  	private static final Method CTX_GET_TRACE_ID;
58  	private static final Method CTX_GET_SPAN_ID;
59  
60  	static {
61  		boolean available = false;
62  		Method spanCurrent = null;
63  		Method getSpanContext = null;
64  		Method isValid = null;
65  		Method getTraceId = null;
66  		Method getSpanId = null;
67  		try {
68  			Class<?> span = Class.forName("io.opentelemetry.api.trace.Span");
69  			Class<?> spanContext = Class.forName("io.opentelemetry.api.trace.SpanContext");
70  			spanCurrent = span.getMethod("current");
71  			getSpanContext = span.getMethod("getSpanContext");
72  			isValid = spanContext.getMethod("isValid");
73  			getTraceId = spanContext.getMethod("getTraceId");
74  			getSpanId = spanContext.getMethod("getSpanId");
75  			available = true;
76  		} catch (Exception e) {
77  			LOG.fine(() -> "OpenTelemetry API not found on classpath — trace-id log fields render empty: " + e.getMessage());
78  		}
79  		AVAILABLE = available;
80  		SPAN_CURRENT = spanCurrent;
81  		SPAN_GET_SPAN_CONTEXT = getSpanContext;
82  		CTX_IS_VALID = isValid;
83  		CTX_GET_TRACE_ID = getTraceId;
84  		CTX_GET_SPAN_ID = getSpanId;
85  	}
86  
87  	private TraceContext() {}
88  
89  	/**
90  	 * Returns whether the OpenTelemetry trace API is available on the runtime classpath.
91  	 *
92  	 * <p>
93  	 * When {@code false}, {@link #currentTraceId()} and {@link #currentSpanId()} always return the empty
94  	 * string.
95  	 *
96  	 * @return {@code true} if {@code io.opentelemetry.api.trace.Span} was found at class-load time.
97  	 */
98  	public static boolean isAvailable() {
99  		return AVAILABLE;
100 	}
101 
102 	/**
103 	 * Returns the trace id of the OTel span active on the current thread.
104 	 *
105 	 * @return The 32-char hex trace id, or the empty string when OTel is absent or no valid span is active.
106 	 */
107 	public static String currentTraceId() {
108 		return field(CTX_GET_TRACE_ID);
109 	}
110 
111 	/**
112 	 * Returns the span id of the OTel span active on the current thread.
113 	 *
114 	 * @return The 16-char hex span id, or the empty string when OTel is absent or no valid span is active.
115 	 */
116 	public static String currentSpanId() {
117 		return field(CTX_GET_SPAN_ID);
118 	}
119 
120 	@SuppressWarnings({
121 		"java:S3011" // Reflective access to the OTel trace API — intentional; OpenTelemetry is not a compile dep.
122 	})
123 	private static String field(Method accessor) {
124 		if (! AVAILABLE)
125 			return "";
126 		try {
127 			Object span = SPAN_CURRENT.invoke(null);
128 			if (span == null)
129 				return "";
130 			Object ctx = SPAN_GET_SPAN_CONTEXT.invoke(span);
131 			if (ctx == null || ! ((Boolean) CTX_IS_VALID.invoke(ctx)).booleanValue())
132 				return "";
133 			Object v = accessor.invoke(ctx);
134 			return v == null ? "" : v.toString();
135 		} catch (Exception e) {
136 			LOG.fine(() -> "OTel Span.current() trace-context read failed: " + e.getMessage());
137 			return "";
138 		}
139 	}
140 }
