26  /**
27   * Programmatic settings for the Jetty server contributed by {@link JettyConfiguration}.
28   *
29   * <p>
30   * Users who need to override the defaults pulled from <c>Config</c> / <c>ManifestFile</c> should
31   * contribute a <c>@Bean JettySettings</c> from their own <c>@Configuration</c> class.  Any value
32   * left {@code null} on this settings bean falls back to the corresponding <c>Config</c> / <c>ManifestFile</c>
33   * entry, then to a hard-coded default.
34   *
35   * <h5 class='section'>Example:</h5>
36   * <p class='bjava'>
37   * 	<ja>@Configuration</ja>
38   * 	<jk>public class</jk> MyAppConfig {
39   * 		<ja>@Bean</ja>
40   * 		<jk>public</jk> JettySettings jettySettings() {
41   * 			<jk>return</jk> JettySettings.<jsm>create</jsm>()
42   * 				.ports(8080, 0)
43   * 				.build();
44   * 		}
45   * 	}
46   * </p>
47   *
48   * <h5 class='section'>See Also:</h5><ul>
49   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauMicroserviceJetty">juneau-microservice-jetty Basics</a>
50   * </ul>
51   *
52   * @since 10.0.0
53   */
54  public class JettySettings {

56  	/**
57  	 * Builder for {@link JettySettings}.
58  	 */
59  	public static class Builder {

61  		int[] ports;
62  		String jettyXml;
63  		Boolean jettyXmlResolveVars;
64  		Duration stopTimeout;
65  		Duration shutdownSettleDelay;
66  
67  		Builder() {}

69  		/**
70  		 * Builds a new {@link JettySettings} from this builder.
71  		 *
72  		 * @return A new settings instance.
73  		 */
74  		public JettySettings build() {
75  			return new JettySettings(this);
76  		}

78  		/**
79  		 * Specifies the ports to try when binding the Jetty server.
80  		 *
81  		 * <p>
82  		 * The first available port is used.  A value of <c>0</c> indicates "try a random port".  When set, this
83  		 * value takes precedence over the <c>Jetty/port</c> config-file entry and the <c>Jetty-Port</c>
84  		 * manifest entry.
85  		 *
86  		 * @param value The candidate ports, in priority order.  Can be <jk>null</jk> to defer to config / manifest.
87  		 * @return This object.
88  		 */
89  		public Builder ports(int...value) {
90  			ports = cp(value);
91  			return this;
92  		}
93  
94  		/**
95  		 * Specifies the contents or location of the <c>jetty.xml</c> file used to build the Jetty server.
96  		 *
97  		 * <p>
98  		 * When set, this value takes precedence over the <c>Jetty/config</c> config-file entry and the
99  		 * <c>Jetty-Config</c> manifest entry.  When not set, the component falls back to those values and
100 		 * finally to a <c>jetty.xml</c> file located on the file system or classpath.
101 		 *
102 		 * @param value The contents or location of the file.  Can be any of:
103 		 * 	<ul>
104 		 * 		<li>{@link String} - Relative path to a file on the file system or classpath, or the raw XML content if it begins with <c>&lt;</c>.
105 		 * 		<li>{@link File} - File on the file system.
106 		 * 		<li>{@link Path} - Path on the file system.
107 		 * 		<li>{@link InputStream} - Raw contents as a <c>UTF-8</c> encoded stream.
108 		 * 		<li>{@link Reader} - Raw contents.
109 		 * 	</ul>
110 		 * 	<br>Can be <jk>null</jk> (defers to config / manifest / classpath resolution).
111 		 * @param resolveVars If <jk>true</jk>, SVL variables in the file will automatically be resolved.
112 		 * @return This object.
113 		 * @throws IOException Thrown by underlying stream.
114 		 */
115 		public Builder jettyXml(Object value, boolean resolveVars) throws IOException {
116 			if (value == null)
117 				jettyXml = null;
118 			else if (value instanceof String value2)
119 				jettyXml = value2.startsWith("<") ? value2 : read(new File(value2));
120 			else if (value instanceof File value2)
121 				jettyXml = read(value2);
122 			else if (value instanceof Path value2)
123 				jettyXml = read(value2);
124 			else if (value instanceof InputStream value2)
125 				jettyXml = read(value2);
126 			else if (value instanceof Reader value2)
127 				jettyXml = read(value2);
128 			else
129 				throw rex("Invalid object type passed to jettyXml(Object): %s", cn(value));
130 			jettyXmlResolveVars = resolveVars;
131 			return this;
132 		}
133 
134 		/**
135 		 * Specifies the bounded graceful-shutdown drain timeout for the Jetty server.
136 		 *
137 		 * <p>
138 		 * On stop, the server stops accepting new connections and waits up to this duration for in-flight requests
139 		 * to complete before the connector closes (Jetty's <c>stopTimeout</c>).  When set, this value takes
140 		 * precedence over the <c>Jetty/stopTimeout</c> config-file entry (in milliseconds).  When neither is set,
141 		 * a default of {@code 30s} is applied.
142 		 *
143 		 * @param value The drain timeout.  Can be <jk>null</jk> to defer to config, then to the default.
144 		 * @return This object.
145 		 */
146 		public Builder stopTimeout(Duration value) {
147 			stopTimeout = value;
148 			return this;
149 		}
150 
151 		/**
152 		 * Specifies the settle delay applied between flipping the readiness probe out of service and stopping the
153 		 * connector.
154 		 *
155 		 * <p>
156 		 * On stop, the readiness probe ({@code /readyz}) flips to {@code 503} <i>before</i> the connector closes so a
157 		 * load balancer / Kubernetes stops routing new traffic.  This brief delay gives the load balancer time to
158 		 * observe the {@code 503} before in-flight requests are drained.  When set, this value takes precedence over
159 		 * the <c>Jetty/shutdownSettleDelay</c> config-file entry (in milliseconds).  When neither is set, no settle
160 		 * delay is applied (defaults to {@code 0}) &mdash; the recommended Kubernetes pattern is a {@code preStop}
161 		 * hook sleep instead.
162 		 *
163 		 * @param value The settle delay.  Can be <jk>null</jk> to defer to config, then to the default.
164 		 * @return This object.
165 		 */
166 		public Builder shutdownSettleDelay(Duration value) {
167 			shutdownSettleDelay = value;
168 			return this;
169 		}
170 	}
171

179 	}
