48  /**
49   * Drives the Jetty server lifecycle on behalf of the microservice via the {@link MicroserviceListener} fan-out.
50   *
51   * <p>
52   * Contributed by {@link JettyConfiguration} as a <c>@Bean</c>.  The microservice's {@link Microservice#start()}
53   * call invokes {@link #onStart(Microservice)}, which:
54   * <ol>
55   * 	<li>Picks the first available port from the resolved {@link JettySettings#getPorts() ports} (falling back to
56   * 		<c>Jetty/port</c> config, then <c>Jetty-Port</c> manifest, then <c>8000</c>).
57   * 	<li>Resolves the Jetty {@link Server} — preferring an externally-supplied <c>@Bean Server</c>, otherwise
58   * 		building one from <c>jetty.xml</c> via the {@link JettyServerFactory}.
59   * 	<li>Mounts servlets from {@code Jetty/servlets}, {@code Jetty/servletMap}, {@code Jetty/servletAttributes}
60   * 		config sections, plus auto-mounts every {@code @Bean Servlet} whose runtime class is annotated with
61   * 		{@link Rest @Rest} (at {@link Rest#path()}, or <c>"/"</c> when no path is set).
62   * 	<li>Starts the server and publishes the bound port via the {@code availablePort} / {@code juneau.serverPort}
63   * 		system properties.
64   * </ol>
65   *
66   * <p>
67   * {@link #onStop(Microservice)} performs the inverse: stops the server gracefully in a worker thread.
68   *
69   * <p>
70   * External callers reach this component via
71   * {@code Microservice.getInstance().getBeanStore().getBean(JettyServerComponent.class).orElseThrow()}.
72   *
73   * <h5 class='section'>See Also:</h5><ul>
74   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauMicroserviceJetty">juneau-microservice-jetty Basics</a>
75   * </ul>
76   *
77   * @since 10.0.0
78   */

114 	private static int[] parseIntArray(String csv) {
115 		if (ie(csv))
116 			return new int[0];
117 		var parts = csv.split(",");
118 		var out = new int[parts.length];
119 		for (var i = 0; i < parts.length; i++)
120 			out[i] = Integer.parseInt(parts[i].trim());
121 		return out;
122 	}

124 	private static int findOpenPort(int[] ports) {
125 		for (var port : ports) {
126 			if (port == 0)
127 				port = RANDOM.nextInt(32767 - ports[0] + 1) + ports[0];
128 			try (var ss = new ServerSocket(port)) {
129 				return port;
130 			} catch (@SuppressWarnings("unused") IOException e) {
131 				// Port is in use, try next port in array
132 			}
133 		}
134 		return 0;
135 	}

137 	private static String normalizePathSpec(String rawPath) {
138 		var p = rawPath == null ? "" : rawPath;
139 		if (p.isEmpty() || "/".equals(p))
140 			return "/*";
141 		if (! p.startsWith("/"))
142 			p = "/" + p;
143 		if (! p.endsWith("/*"))
144 			p = trimTrailingSlashes(p) + "/*";
145 		return p;
146 	}
147 
148 	private static String normalizeExactPathSpec(String rawPath) {
149 		var p = rawPath == null ? "" : trimTrailingSlashes(rawPath);
150 		if (p.isEmpty())
151 			return "/";
152 		if (! p.startsWith("/"))
153 			p = "/" + p;
154 		return p;
155 	}

157 	/**
158 	 * Resolves the top-level Jetty mount path-specs for an auto-discovered {@code @Rest}-annotated servlet.
159 	 *
160 	 * <p>
161 	 * Delegates to {@link RestContext#resolveTopLevelPaths(Class, Object, BeanStore)} so the runtime-override
162 	 * resolution chain (programmatic &gt; {@code getPaths()} getter &gt; annotation default) is honored at
163 	 * mount time. The programmatic Builder rung is N/A for Jetty auto-discovery (the {@code RestContext} does
164 	 * not exist yet at this point in the lifecycle); users can substitute by overriding {@code getPaths()} on
165 	 * the resource class, or by leaning on SVL inside {@code @Rest(paths=...)} elements (e.g.
166 	 * {@code paths={"$C{health.paths}"}}) so each element is resolved through {@code Config} /
167 	 * {@code $E{...}} / {@code $S{...}} before being comma-split.
168 	 *
169 	 * @return The normalized exact-match path-specs (for multi-mount), or the singular {@code @Rest(path)}-style
170 	 * 	{@code "/...&#47;*"} pathspec when no {@code paths} are resolved.
171 	 */
172 	private static String[] restPathsFor(Servlet servlet, BeanStore store) {
173 		var cls = servlet.getClass();
174 		var r = cls.getAnnotation(Rest.class);
175 		if (r == null)
176 			return new String[] {"/*"};
177 		var resolved = RestContext.resolveTopLevelPaths(cls, servlet, store);
178 		if (resolved.length > 0)
179 			return Arrays.stream(resolved).map(JettyServerComponent::normalizeExactPathSpec).toArray(String[]::new);
180 		return new String[] {normalizePathSpec(r.path())};
181 	}

197 	public void onStart(Microservice ms) {
198 		try {
199 			microservice.set(ms);
200 			var store = ms.getBeanStore();
201 			ClassInfo.of(this).inject(this, store);
202 			var cf = ms.getConfig();
203 			var mf = ms.getManifest();
204 			var vr = ms.getVarResolver();
205 			var settings = store.getBean(JettySettings.class).orElseGet(() -> JettySettings.create().build());
206 
207 			// Port resolution: settings > Jetty/port config > Jetty-Port manifest > 8000
208 			var ports = settings.getPorts();
209 			if (ports == null)
210 				ports = cf.get("Jetty/port").as(int[].class)
211 					.orElseGet(() -> mf.get("Jetty-Port").map(JettyServerComponent::parseIntArray).orElseGet(() -> ints(8000)));
212 			var availablePort = findOpenPort(ports);
213 
214 			if (availablePortEnv.isEmpty())
215 				// Intentional real-JVM-property publish: the bundled jetty.xml binds its connector to
216 				// $S{availablePort,10000}, and the $S{...} SVL var resolves via raw System.getProperty (bypassing
217 				// the Settings override chain).  A Settings-based write would be invisible to jetty.xml, so the
218 				// bound port MUST be published as a real system property here.  The @Value-injected availablePortEnv
219 				// read side is already Settings-abstracted; only this publish must stay raw.
220 				System.setProperty("availablePort", String.valueOf(availablePort));
221 
222 			// Prefer a @Bean-supplied Server, else build one from jetty.xml.
223 			var injectedServer = store.getBean(Server.class).orElse(null);
224 			if (nn(injectedServer)) {
225 				server.set(injectedServer);
226 			} else {
227 				var factory = store.getBean(JettyServerFactory.class).orElseGet(BasicJettyServerFactory::new);
228 				var jettyXml = settings.getJettyXml();
229 				var jettyConfig = cf.get("Jetty/config").orElseGet(() -> mf.get("Jetty-Config").orElse("jetty.xml"));
230 				var resolveVars = coalesce(settings.getJettyXmlResolveVars(), cf.get("Jetty/resolveVars").asBoolean().orElse(false));
231 				boolean resolveVars2 = isTrue(resolveVars);
232 
233 				if (jettyXml == null)
234 					jettyXml = loadSystemResourceAsString(jettyConfig, ".", "files");
235 				if (jettyXml == null)
236 					throw rex("jetty.xml file '%s' was not found on the file system or classpath.", jettyConfig);
237 
238 				if (resolveVars2)
239 					jettyXml = vr.resolve(jettyXml);
240 
241 				ms.getLogger().info(jettyXml);
242 
243 				try {
244 					server.set(factory.create(jettyXml));
245 				} catch (Exception e) {
246 					throw toRex(e);
247 				}
248 			}
249 
250 			// Publish back to the store for downstream beans / lookups.
251 			store.addBean(Server.class, server.get());
252 
253 			// Graceful-shutdown wiring (zero-downtime k8s rollouts):
254 			//  - Apply a bounded stopTimeout so server.stop() drains in-flight requests before the connector closes.
255 			//    Precedence: JettySettings > Jetty/stopTimeout config > jetty.xml value (if any) > 30s default.
256 			//  - Remember the settle delay so onStop() can let the LB observe the /readyz 503 before draining.
257 			//  - Mark readiness ready so a freshly-(re)started service serves traffic.
258 			var stopTimeoutOverride = settings.getStopTimeout() != null
259 				? settings.getStopTimeout()
260 				: cf.get("Jetty/stopTimeout").asLong().map(Duration::ofMillis).orElse(null);
261 			if (stopTimeoutOverride != null)
262 				server.get().setStopTimeout(Math.max(0L, stopTimeoutOverride.toMillis()));
263 			else if (server.get().getStopTimeout() <= 0L)
264 				server.get().setStopTimeout(DEFAULT_STOP_TIMEOUT.toMillis());
265 			shutdownSettleDelay.set(coalesce(settings.getShutdownSettleDelay(), cf.get("Jetty/shutdownSettleDelay").asLong().map(Duration::ofMillis).orElse(Duration.ZERO)));
266 			ReadinessState.resolve(store).markReady();
267 
268 			// Track each servlet pathSpec with its declaring source so we can fail loudly on collisions.
269 			var mountedPaths = new LinkedHashMap<String,String>();
270 
271 			for (var s : cf.get("Jetty/servlets").asStringArray().orElse(new String[0])) {
272 				try {
273 					var c = info(Class.forName(s));
274 					if (c.isAssignableTo(RestServlet.class)) {
275 						var rs = (RestServlet)c.newInstance();
276 						mountWithCollisionCheck(rs, rs.getPath(), "Jetty/servlets[" + s + "]", mountedPaths);
277 					} else {
278 						throw rex("Invalid servlet specified in Jetty/servlets.  Must be a subclass of RestServlet: %s", s);
279 					}
280 				} catch (ClassNotFoundException e) {
281 					throw toRex(e);
282 				}
283 			}
284 
285 			cf.get("Jetty/servletMap").asMap().orElse(EMPTY_MAP).forEach((k, v) -> {
286 				try {
287 					var c = info(Class.forName(v.toString()));
288 					if (c.isAssignableTo(Servlet.class)) {
289 						var rs = (Servlet)c.newInstance();
290 						mountWithCollisionCheck(rs, k, "Jetty/servletMap[" + k + "]", mountedPaths);
291 					} else {
292 						throw rex("Invalid servlet specified in Jetty/servletMap.  Must be a subclass of Servlet: %s", cn(v));
293 					}
294 				} catch (ClassNotFoundException e) {
295 					throw toRex(e);
296 				}
297 			});
298 
299 			cf.get("Jetty/servletAttributes").asMap().orElse(EMPTY_MAP).forEach(this::addServletAttribute);
300 
301 			// Auto-mount @Bean AuthFilterChain at /* before any servlet registration so the filter
302 			// runs before routing for all requests.
303 			store.getBean(AuthFilterChain.class).ifPresent(chain -> addFilter(chain, "/*"));
304 
305 			// Auto-discover @Rest servlets contributed via @Configuration/@Bean methods.
306 			// Path source precedence: @Rest(path=...) on the resource class, else "/".
307 			for (var e : store.getBeansOfType(Servlet.class).entrySet()) {
308 				var servlet = e.getValue();
309 				var cls = servlet.getClass();
310 				if (cls.getAnnotation(Rest.class) == null)
311 					continue;
312 				var pathSpecs = restPathsFor(servlet, store);
313 				var source = "@Bean " + cls.getName() + (ine(e.getKey()) ? "[" + e.getKey() + "]" : "");
314 				for (var pathSpec : pathSpecs)
315 					checkPathCollision(pathSpec, source, mountedPaths);
316 				addServlet(servlet, pathSpecs);
317 			}
318 
319 			if (serverPortEnv.isEmpty())
320 				// Intentional real-JVM-property publish: juneau.serverPort is a documented cross-component/external
321 				// publication of the bound port, consumed via raw System.getProperty (and $S{...} SVL).  A
322 				// Settings-based write would not be visible to those raw readers, so this is kept.  The
323 				// @Value-injected serverPortEnv read side is already Settings-abstracted.
324 				System.setProperty("juneau.serverPort", String.valueOf(availablePort));
325 
326 			server.get().start();
327 			ms.out(messages, "ServerStarted", getPort());
328 		} catch (Exception e) {
329 			throw toRex(e);
330 		}
331 	}

333 	@Override /* Overridden from MicroserviceListener */
334 	public void onStop(Microservice ms) {
335 		final Logger logger = ms.getLogger();
336 		// Flip readiness out of service BEFORE the connector stops so /readyz returns 503 and the load balancer /
337 		// Kubernetes stops routing new traffic while in-flight requests drain.  /livez stays healthy so the pod is
338 		// not killed mid-drain.  The settle delay (default 0) gives the LB a window to observe the 503.
339 		ReadinessState.resolve(ms.getBeanStore()).markOutOfService();
340 		ms.out(messages, "DrainingRequests");
341 		sleepQuietly(shutdownSettleDelay.get());
342 		var t = new Thread("JettyServerComponentStop") {
343 			@Override /* Overridden from Thread */
344 			public void run() {
345 				try {
346 					var s = server.get();
347 					if (s == null || s.isStopping() || s.isStopped())
348 						return;
349 					ms.out(messages, "StoppingServer");
350 					s.stop();
351 					ms.out(messages, "ServerStopped");
352 				} catch (Exception e) {
353 					logger.log(Level.WARNING, localizedMessage(e), e);
354 				}
355 			}
356 		};
357 		t.start();
358 		try {
359 			t.join();
360 		} catch (@SuppressWarnings("unused") InterruptedException e) {
361 			Thread.currentThread().interrupt();
362 		}
363 	}

365 	private static void sleepQuietly(Duration d) {
366 		if (d == null || d.isZero() || d.isNegative())
367 			return;
368 		try {
369 			Thread.sleep(d.toMillis());
370 		} catch (@SuppressWarnings("unused") InterruptedException e) {
371 			Thread.currentThread().interrupt();
372 		}
373 	}
