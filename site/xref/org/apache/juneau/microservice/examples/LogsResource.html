104 	@Response(schema = @Schema(description = "File or directory details"))
105 	@BeanType(properties = "type,name,size,lastModified,actions,files")
106 	public static class FileResource {
107 		static final FileFilter FILE_FILTER = f2 -> f2.isDirectory() || f2.getName().endsWith(".log");
108 		static final Comparator<FileResource> FILE_COMPARATOR = (o1, o2) -> {
109 			int c = o1.getType().compareTo(o2.getType());
110 			return c != 0 ? c : o1.getName().compareTo(o2.getName());
111 		};
112 		private final File f;
113 		private final String path;
114 
115 		private final String uri;
116 
117 		private final boolean includeChildren;
118 		private final boolean allowDeletes;
119 
120 		/**
121 		 * Constructor.
122 		 *
123 		 * @param f The file or directory.
124 		 * @param path The path.
125 		 * @param allowDeletes Whether deletes are allowed.
126 		 * @param includeChildren Whether to include child files.
127 		 */
128 		public FileResource(File f, String path, boolean allowDeletes, boolean includeChildren) {
129 			this.f = f;
130 			this.path = path;
131 			this.uri = "servlet:/" + or(path, "");
132 			this.includeChildren = includeChildren;
133 			this.allowDeletes = allowDeletes;
134 		}
135 
136 		/**
137 		 * Returns available actions for this file (view, download, delete, etc.).
138 		 *
139 		 * @return The list of actions.
140 		 */
141 		@Html(format = HtmlFormat.HTML_CDC)
142 		public List<Action> getActions() {
143 			var l = new ArrayList<Action>();
144 			if (f.canRead() && ! f.isDirectory()) {
145 				l.add(new Action("view", uri + "?method=VIEW"));
146 				l.add(new Action("highlighted", uri + "?method=VIEW&highlight=true"));
147 				l.add(new Action("parsed", uri + "?method=PARSE"));
148 				l.add(new Action("download", uri + "?method=DOWNLOAD"));
149 				if (allowDeletes)
150 					l.add(new Action("delete", uri + "?method=DELETE"));
151 			}
152 			return l;
153 		}
154 
155 		/**
156 		 * Returns child files when this is a directory and includeChildren is <jk>true</jk>.
157 		 *
158 		 * @return Child files, or <jk>null</jk> when this is a file or includeChildren is false.
159 		 */
160 		@SuppressWarnings({
161 			"java:S1168"     // Intentional null when file or !includeChildren.
162 		})
163 		public Set<FileResource> getFiles() {
164 			if (f.isFile() || ! includeChildren)
165 				return null;
166 			var s = new TreeSet<>(FILE_COMPARATOR);
167 			for (var fc : f.listFiles(FILE_FILTER))
168 				s.add(new FileResource(fc, (nn(path) ? (path + '/') : "") + urlEncode(fc.getName()), allowDeletes, false));
169 			return s;
170 		}
171 
172 		/**
173 		 * Returns the last modified date of the file.
174 		 *
175 		 * @return The last modified date as an {@link Instant}.
176 		 */
177 		public Instant getLastModified() { return Instant.ofEpochMilli(f.lastModified()); }
178 
179 		/**
180 		 * Returns the file name as a link.
181 		 *
182 		 * @return The name as a link.
183 		 */
184 		public LinkString getName() { return new LinkString(f.getName(), uri); }
185 
186 		/**
187 		 * Returns the file size, or number of items for directories.
188 		 *
189 		 * @return The size.
190 		 */
191 		public long getSize() { return f.isDirectory() ? f.listFiles().length : f.length(); }
192 
193 		/**
194 		 * Returns the type: "dir" or "file".
195 		 *
196 		 * @return The type.
197 		 */
198 		public String getType() { return (f.isDirectory() ? "dir" : "file"); }
199 	}

321 	 * Views the contents of a log file with optional filtering.
322 	 *
323 	 * @param res The response.
324 	 * @param path The file path.
325 	 * @param highlight Add severity color highlighting.
326 	 * @param start Start timestamp filter.
327 	 * @param end End timestamp filter.
328 	 * @param thread Thread name filter.
329 	 * @param loggers Logger filter.
330 	 * @param severity Severity filter.
331 	 * @throws NotFound If the file is not found.
332 	 * @throws MethodNotAllowed If method not allowed.
333 	 * @throws IOException If an I/O error occurs.
334 	 */
335 	@RestOp(
336 		method="VIEW",
337 		path="/*",
338 		summary="View contents of log file",
339 		description="View the contents of a log file."
340 	)
341 	@SuppressWarnings({
342 		"java:S3776", // Cognitive complexity acceptable for this specific logic
343 		"java:S107" // Method has many parameters; acceptable for builder/configuration methods
344 	})
345 	public void viewFile(
346 			RestResponse res,
347 			@Path("/*") String path,
348 			@Query(name="highlight", schema=@Schema(d="Add severity color highlighting.")) boolean highlight,
349 			@Query(name="start", schema=@Schema(d="Start timestamp (ISO8601, full or partial).\nDon't print lines logged before the specified timestamp.\nUse any of the following formats: yyyy, yyyy-MM, yyyy-MM-dd, yyyy-MM-ddThh, yyyy-MM-ddThh:mm, yyyy-MM-ddThh:mm:ss, yyyy-MM-ddThh:mm:ss.SSS")) String start,
350 			@Query(name="end", schema=@Schema(d="End timestamp (ISO8601, full or partial).\nDon't print lines logged after the specified timestamp.\nUse any of the following formats: yyyy, yyyy-MM, yyyy-MM-dd, yyyy-MM-ddThh, yyyy-MM-ddThh:mm, yyyy-MM-ddThh:mm:ss, yyyy-MM-ddThh:mm:ss.SSS")) String end,
351 			@Query(name="thread", schema=@Schema(d="Thread name filter.\nOnly show log entries with the specified thread name.")) String thread,
352 			@Query(name="loggers", schema=@Schema(d="Logger filter (simple class name).\nOnly show log entries if they were produced by one of the specified loggers.")) String[] loggers,
353 			@Query(name="severity",schema=@Schema( d="Severity filter.\nOnly show log entries with the specified severity.")) String[] severity
354 		) throws NotFound, MethodNotAllowed, IOException {
355 
356 		var f = getFile(path);
357 
358 		var startDate = o(start).filter(x1 -> ! isBlank(x1)).map(x2 -> GranularZonedDateTime.of(start).getZonedDateTime()).map(GregorianCalendar::from).map(Calendar::getTime).orElse(null);
359 		var endDate = o(end).filter(x11 -> ! isBlank(x11)).map(x4 -> GranularZonedDateTime.of(end).getZonedDateTime()).map(GregorianCalendar::from).map(Calendar::getTime).orElse(null);
360 
361 		if (! highlight) {
362 			var o = getReader(f, startDate, endDate, thread, loggers, severity);
363 			res.setContentType("text/plain");
364 			if (o instanceof Reader)
365 				res.setContent(o);
366 			else {
367 				try (var p = (LogParser)o; Writer w = res.getNegotiatedWriter()) {
368 					p.writeTo(w);
369 				}
370 			}
371 			return;
372 		}
373 
374 		res.setContentType("text/html");
375 		try (var w = res.getNegotiatedWriter()) {
376 			w.println("<html><body style='font-family:monospace;font-size:8pt;white-space:pre;'>");
377 			try (var lp = getLogParser(f, startDate, endDate, thread, loggers, severity)) {
378 				if (! lp.hasNext())
379 					w.append("<span style='color:gray'>[EMPTY]</span>");
380 				else
381 					for (var le : lp) {
382 						var s = ie(le.severity) ? ' ' : le.severity.charAt(0);
383 						String color = "black";
384 						//SEVERE|WARNING|INFO|CONFIG|FINE|FINER|FINEST
385 						if (s == 'I')
386 							color = "#006400";
387 						else if (s == 'W')
388 							color = "#CC8400";
389 						else if (s == 'E' || s == 'S')
390 							color = "#DD0000";
391 						else if (s == 'D' || s == 'F' || s == 'T')
392 							color = "#000064";
393 						w.append("<span style='color:").append(color).append("'>");
394 						le.appendHtml(w).append("</span>");
395 					}
396 				w.append("</body></html>");
397 			}
398 		}
399 	}

448 	private void deleteFile(File f) {
449 		if (! allowDeletes)
450 			throw new MethodNotAllowed("DELETE not enabled");
451 		if (f.isDirectory()) {
452 			var files = f.listFiles();
453 			if (nn(files)) {
454 				for (var fc : files)
455 					deleteFile(fc);
456 			}
457 		}
458 		try {
459 			Files.delete(f.toPath());
460 		} catch (@SuppressWarnings("unused") IOException e) {
461 			throw new Forbidden("Could not delete file %s", f.getAbsolutePath());
462 		}
463 	}

465 	/**
466 	 * Resolves a request path against {@link #logDir} with a canonical-path boundary check
467 	 * so {@code ..} segments, absolute paths, and symlink-out cannot escape the configured root.
468 	 *
469 	 * <p>
470 	 * The check is intentionally applied at this single funnel because every public operation on
471 	 * this resource (view, parse, download, delete, info) resolves the user-supplied path through
472 	 * this method. Delegates to {@link FileUtils#resolveSafely(File, String)} — the canonical
473 	 * shared implementation; the caller-layer mapping is: boundary violation → 403 (Forbidden),
474 	 * non-existent target → 404 (NotFound).
475 	 */
476 	private static File getFile(String path) throws NotFound {
477 		try {
478 			return FileUtils.resolveSafely(logDir, path)
479 				.orElseThrow(() -> new NotFound("File not found."));
480 		} catch (IllegalArgumentException e) {
481 			throw new Forbidden(e.getMessage());
482 		}
483 	}

485 	private static LogParser getLogParser(File f, Date start, Date end, String thread, String[] loggers, String[] severity) throws IOException {
486 		return new LogParser(leFormatter, f, start, end, thread, loggers, severity);
487 	}

489 	private static Object getReader(File f, Date start, Date end, String thread, String[] loggers, String[] severity) throws IOException {
490 		if (start == null && end == null && thread == null && loggers == null)
491 			return getReader(f);
492 		return getLogParser(f, start, end, thread, loggers, severity);
