29  /**
30   * A servlet filter, with the URL patterns, dispatcher types and order it is mounted at, that the server components
31   * ({@code JettyServerComponent}, {@code TomcatServerComponent}) mount before the server starts.
32   *
33   * <p>
34   * Contribute one as a <c>@Bean</c> from a <c>@Configuration</c> class.  It is the container-neutral counterpart of
35   * Spring Boot's {@code FilterRegistrationBean}, so the same registration works on Jetty and Tomcat.
36   *
37   * <h5 class='section'>Example:</h5>
38   * <p class='bjava'>
39   * 	<ja>@Configuration</ja>
40   * 	<jk>public class</jk> MyAppConfig {
41   * 		<ja>@Bean</ja>(name=<js>"boundary"</js>)
42   * 		<jk>public</jk> ServletFilterRegistration boundary() {
43   * 			<jk>return</jk> ServletFilterRegistration.<jsm>create</jsm>(<jk>new</jk> MyBoundaryFilter())
44   * 				.urlPatterns(<js>"/console/*"</js>)
45   * 				.dispatcherTypes(DispatcherType.<jsf>REQUEST</jsf>, DispatcherType.<jsf>FORWARD</jsf>, DispatcherType.<jsf>ASYNC</jsf>, DispatcherType.<jsf>ERROR</jsf>)
46   * 				.order(ServletFilterRegistration.<jsf>HIGHEST_PRECEDENCE</jsf>)
47   * 				.build();
48   * 		}
49   * 	}
50   * </p>
51   *
52   * <h5 class='section'>Order:</h5>
53   * <p>
54   * Filters run outermost first, by ascending {@link #getOrder() order}.  A bean-supplied {@link AuthFilterChain} is
55   * mounted automatically at {@link #AUTH_ORDER}, so a registration at the default {@link #DEFAULT_ORDER} runs after
56   * authentication.  Two registrations at the same order run in ascending bean-name order, and the auto-mounted
57   * authentication filter runs ahead of a registration at its own order; give registrations that must keep a relative
58   * order distinct values rather than relying on names.
59   *
60   * <p>
61   * The order is only ever {@link Builder#order(int)}.  <c>@Order</c> and <c>@Bean(priority)</c> on the factory method do
62   * not affect it.
63   *
64   * <p>
65   * Running a filter before authentication is a property of a narrow URL pattern, not of the order alone.  A boundary
66   * on <js>"/console/*"</js> rejects a bad request before authentication ever sees it; the same filter on <js>"/*"</js>
67   * at {@link #HIGHEST_PRECEDENCE} would also intercept authentication redirects and challenges.
68   *
69   * <h5 class='section'>Names:</h5>
70   * <p>
71   * More than one registration in an application needs a distinct bean name, because a <c>@Bean</c> of one type with the
72   * same (default empty) name is a duplicate.  Distinct filters must be distinct instances: wrapping one filter instance
73   * in two registrations fails at startup, because the container would otherwise initialize and destroy it twice.
74   *
75   * <h5 class='section'>Replacing the authentication mount:</h5>
76   * <p>
77   * An application that needs the {@link AuthFilterChain} on other dispatcher types, or at another position, registers it
78   * itself: wrap the injected {@link AuthFilterChain} bean instance in a registration.  That registration then replaces
79   * the automatic mount.  It must be the injected instance; a new instance is a second, separate filter.
80   *
81   * <h5 class='section'>See Also:</h5><ul>
82   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauMicroserviceJetty">juneau-microservice-jetty</a>
83   * </ul>
84   *
85   * @since 10.0.0
86   */
87  public final class ServletFilterRegistration {
88  
89  	/** Runs before everything else, including the automatically mounted {@link AuthFilterChain}. */
90  	public static final int HIGHEST_PRECEDENCE = Integer.MIN_VALUE;
91  
92  	/** The order at which a bean-supplied {@link AuthFilterChain} is mounted automatically. */
93  	public static final int AUTH_ORDER = -100;
94  
95  	/** The default order: after authentication. */
96  	public static final int DEFAULT_ORDER = 0;
97  
98  	/** Runs after everything else. */
99  	public static final int LOWEST_PRECEDENCE = Integer.MAX_VALUE;
100 
101 	/**
102 	 * Builder for {@link ServletFilterRegistration}.
103 	 */
104 	public static final class Builder {
105 
106 		private final Filter filter;
107 		private List<String> urlPatterns = List.of();
108 		private Set<DispatcherType> dispatcherTypes = EnumSet.of(DispatcherType.REQUEST);
109 		private int order = DEFAULT_ORDER;
110 		private boolean asyncSupported = true;
111 		private String name;
112 		private boolean builtInAuth;
113 
114 		private Builder(Filter filter) {
115 			this.filter = filter;
116 		}
117 
118 		/**
119 		 * Sets the URL patterns the filter is mapped to.
120 		 *
121 		 * <p>
122 		 * Replaces any patterns set before.  At least one is required.  Duplicates are dropped, keeping the first.
123 		 * Each pattern must be one of:
124 		 * <ul>
125 		 * 	<li>A path prefix: {@code /.../*} (including <js>"/*"</js>), where the trailing {@code *} is the only one.
126 		 * 	<li>An extension: {@code *.ext}, with no {@code /}.
127 		 * 	<li>An exact path: starts with {@code /} and contains no {@code *}.
128 		 * 	<li>The empty string <js>""</js>: the context root.
129 		 * </ul>
130 		 * Anything else is rejected, including a bare <js>"/"</js> (the default servlet, which a filter should not be
131 		 * mapped to), a blank (non-empty) pattern, and Jetty-only path-spec syntax such as {@code regex:...} or {@code ^...}.
132 		 * Rejecting these keeps a typo such as <js>"/console*"</js> from silently mapping a boundary to nothing.
133 		 *
134 		 * @param value The URL patterns.  Cannot be <jk>null</jk>.
135 		 * @return This object.
136 		 */
137 		public Builder urlPatterns(String...value) {
138 			if (value == null)
139 				throw iaex("urlPatterns must not be null.");
140 			urlPatterns = Arrays.asList(value);
141 			return this;
142 		}
143 
144 		/**
145 		 * Sets the dispatcher types the filter is mapped to.
146 		 *
147 		 * <p>
148 		 * Replaces the default of {@link DispatcherType#REQUEST}, so list <c>REQUEST</c> explicitly when it is wanted.
149 		 * A filter mapped to {@code ASYNC} only does not run for the plain request.  Add {@code FORWARD}, {@code ASYNC}
150 		 * and {@code ERROR} for a boundary that must also see internal dispatches.
151 		 *
152 		 * @param value The dispatcher types.  At least one is required, and none can be <jk>null</jk>.
153 		 * @return This object.
154 		 */
155 		public Builder dispatcherTypes(DispatcherType...value) {
156 			if (value == null)
157 				throw iaex("dispatcherTypes must not be null.");
158 			dispatcherTypes = new LinkedHashSet<>(Arrays.asList(value));
159 			return this;
160 		}
161 
162 		/**
163 		 * Sets the order: lower values run first (further out).
164 		 *
165 		 * <p>
166 		 * Defaults to {@link #DEFAULT_ORDER}.  See {@link ServletFilterRegistration} for how it relates to
167 		 * authentication.
168 		 *
169 		 * @param value The order.
170 		 * @return This object.
171 		 */
172 		public Builder order(int value) {
173 			order = value;
174 			return this;
175 		}
176 
177 		/**
178 		 * Sets whether the filter supports asynchronous request processing.
179 		 *
180 		 * <p>
181 		 * Defaults to <jk>true</jk>.  A filter in the chain that does not support it makes
182 		 * {@code request.isAsyncSupported()} false for every request it covers.
183 		 *
184 		 * @param value Whether async is supported.
185 		 * @return This object.
186 		 */
187 		public Builder asyncSupported(boolean value) {
188 			asyncSupported = value;
189 			return this;
190 		}
191 
192 		/**
193 		 * Sets the container filter name.
194 		 *
195 		 * <p>
196 		 * Optional; when absent the component generates a unique one.  An explicit name must be unique among the
197 		 * registrations.
198 		 *
199 		 * @param value The filter name.  Can be <jk>null</jk> (generate one), but not blank.
200 		 * @return This object.
201 		 */
202 		public Builder name(String value) {
203 			if (value != null && value.isBlank())
204 				throw iaex("name must not be blank.");
205 			name = value;
206 			return this;
207 		}
208 
209 		private Builder builtInAuth(boolean value) {
210 			builtInAuth = value;
211 			return this;
212 		}
213 
214 		/**
215 		 * Validates and builds the registration.
216 		 *
217 		 * @return A new immutable registration.
218 		 * @throws IllegalArgumentException If there is no URL pattern, a pattern is malformed, or there is no
219 		 * 	dispatcher type or a <jk>null</jk> one.
220 		 */
221 		public ServletFilterRegistration build() {
222 			if (urlPatterns.isEmpty())
223 				throw iaex("A servlet filter registration needs at least one URL pattern.");
224 			urlPatterns.forEach(Builder::validatePattern);
225 			var patterns = new LinkedHashSet<>(urlPatterns);
226 			if (dispatcherTypes.isEmpty())
227 				throw iaex("A servlet filter registration needs at least one dispatcher type.");
228 			if (dispatcherTypes.contains(null))
229 				throw iaex("A dispatcher type must not be null.");
230 			return new ServletFilterRegistration(this, List.copyOf(patterns), Collections.unmodifiableSet(EnumSet.copyOf(dispatcherTypes)));
231 		}
232 
233 		private static void validatePattern(String p) {
234 			if (p == null || (! p.isEmpty() && p.isBlank()))
235 				throw iaex("A URL pattern must not be null or blank.");
236 			if (p.isEmpty())
237 				return;
238 			if (p.startsWith("regex:") || p.startsWith("^") || p.startsWith("uri-template:"))
239 				throw iaex("URL pattern '%s' uses container-specific path-spec syntax; use a path prefix, an extension or an exact path.", p);
240 			if (p.equals("/"))
241 				throw iaex("URL pattern '/' maps to the default servlet and is not accepted for a filter; use '/*'.");
242 			var star = p.indexOf('*');
243 			if (p.startsWith("/")) {
244 				if (star < 0 || (star == p.length() - 1 && p.endsWith("/*")))
245 					return;
246 			} else if (p.startsWith("*.") && p.length() > 2 && star == 0 && p.lastIndexOf('*') == 0 && p.indexOf('/') < 0) {
247 				return;
248 			}
249 			throw iaex("URL pattern '%s' is not valid: use a path prefix like '/console/*', an extension like '*.json', or an exact path like '/console'.", p);
250 		}
251 	}
252 
253 	/**
254 	 * Creates a builder for a registration of a filter.
255 	 *
256 	 * @param filter The filter instance.  Cannot be <jk>null</jk>.
257 	 * @return A new builder.
258 	 * @throws IllegalArgumentException If the filter is <jk>null</jk>.
259 	 */
260 	public static Builder create(Filter filter) {
261 		if (filter == null)
262 			throw iaex("filter must not be null.");
263 		return new Builder(filter);
264 	}

266 	/**
267 	 * Collects the registrations a server component mounts, outermost first.
268 	 *
269 	 * <p>
270 	 * Takes every {@link ServletFilterRegistration} bean, plus an internal registration for the {@link AuthFilterChain}
271 	 * bean when there is one and no registration wraps that same instance (see the class description).  Sorts by
272 	 * {@link #getOrder() order}, then the automatic authentication mount first, then bean name.
273 	 *
274 	 * <p>
275 	 * Intended for the server components; applications contribute registrations as beans.
276 	 *
277 	 * @param store The microservice bean store.  Cannot be <jk>null</jk>.
278 	 * @return The registrations in mount order.  Never <jk>null</jk>.
279 	 * @throws IllegalStateException If one filter instance is registered more than once, or two registrations have the
280 	 * 	same explicit name.
281 	 */
282 	public static List<ServletFilterRegistration> collect(BeanStore store) {
283 		if (store == null)
284 			throw iaex("store must not be null.");
285 		var entries = store.getBeansOfType(ServletFilterRegistration.class).entrySet().stream()
286 			.map(e -> new Entry(e.getValue(), e.getKey()))
287 			.collect(Collectors.toCollection(ArrayList::new));
288 
289 		store.getBean(AuthFilterChain.class).ifPresent(chain -> {
290 			if (entries.stream().noneMatch(e -> e.registration.filter == chain))
291 				entries.add(new Entry(create(chain).urlPatterns("/*").order(AUTH_ORDER).builtInAuth(true).build(), ""));
292 		});
293 
294 		var names = new HashMap<String,Entry>();
295 		for (var i = 0; i < entries.size(); i++) {
296 			var a = entries.get(i);
297 			for (var j = i + 1; j < entries.size(); j++) {
298 				var b = entries.get(j);
299 				if (a.registration.filter == b.registration.filter)
300 					throw isex("The same filter instance is registered by %s and %s; register it once.", a.describe(), b.describe());
301 			}
302 			var name = a.registration.name;
303 			if (nn(name)) {
304 				var prior = names.put(name, a);
305 				if (nn(prior))
306 					throw isex("Servlet filter name '%s' is used by both %s and %s.", name, prior.describe(), a.describe());
307 			}
308 		}
309 
310 		entries.sort(Comparator
311 			.comparingInt((Entry e) -> e.registration.order)
312 			.thenComparing(e -> ! e.registration.builtInAuth)
313 			.thenComparing(e -> e.beanName));
314 
315 		return entries.stream().map(e -> e.registration).toList();
316 	}
317 
318 	private record Entry(ServletFilterRegistration registration, String beanName) {
319 		String describe() {
320 			return registration.builtInAuth ? "the automatic AuthFilterChain mount" : "bean '" + beanName + "'";
321 		}
322 	}
