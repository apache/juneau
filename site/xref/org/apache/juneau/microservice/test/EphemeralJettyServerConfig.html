17  package org.apache.juneau.microservice.test;
18  
19  import java.time.*;
20  
21  import org.apache.juneau.commons.inject.*;
22  import org.apache.juneau.microservice.jetty.*;
23  import org.eclipse.jetty.ee11.servlet.*;
24  import org.eclipse.jetty.server.*;
25  
26  /**
27   * A {@code @Configuration} that supplies a Jetty {@link Server} bound to an OS-assigned ephemeral port
28   * (port {@code 0}) for use by {@link MicroserviceExtension @MicroserviceTest}-driven integration tests.
29   *
30   * <p>
31   * The server has a single {@link ServerConnector} on port 0 and a single root {@link ServletContextHandler}
32   * at context path {@code "/"}.  {@code JettyServerComponent} discovers the handler via the
33   * {@code "ServletContextHandler"} server attribute (the same convention used by {@code jetty.xml}).
34   *
35   * <p>
36   * The extension installs this {@code @Configuration} automatically after the user-supplied configurations, so
37   * a user-supplied {@code @Bean Server} (if any) still wins (the bean store returns the first registered match).
38   * Tests supply their {@code @Bean Servlet} definitions, which {@code JettyServerComponent} auto-mounts at the
39   * resource's {@code @Rest(path=...)}.
40   *
41   * @since 10.0.0
42   */
43  @Configuration
44  public class EphemeralJettyServerConfig {
45  
46  	/**
47  	 * Provides the bean-supplied Jetty {@link Server} that {@code JettyServerComponent} consumes during
48  	 * {@code onStart()}.
49  	 *
50  	 * @return A configured {@link Server} bound to port 0 (OS-assigned ephemeral port).
51  	 */
52  	@Bean
53  	@SuppressWarnings("resource") // connector is added to and owned by the server; closed when the server stops.
54  	public Server jettyServer() {
55  		var server = new Server();
56  		var connector = new ServerConnector(server);
57  		connector.setPort(0);
58  		server.addConnector(connector);
59  		var sch = new ServletContextHandler();
60  		sch.setContextPath("/");
61  		server.setAttribute("ServletContextHandler", sch);
62  		server.setHandler(sch);
63  		return server;
64  	}
65  
66  	/**
67  	 * Disables the 30s graceful-shutdown drain wait for integration tests.
68  	 *
69  	 * <p>
70  	 * {@code JettyServerComponent} applies a 30s {@code stopTimeout} default to support zero-downtime k8s
71  	 * rollouts.  In ephemeral test servers there are no load balancers or in-flight production traffic, so
72  	 * the drain wait only adds ~1s per test class to the suite runtime.  Setting it to zero restores fast
73  	 * shutdown.
74  	 *
75  	 * @return {@link JettySettings} with {@code stopTimeout} set to zero.
76  	 */
77  	@Bean
78  	public JettySettings jettySettings() {
79  		return JettySettings.create()
80  			.stopTimeout(Duration.ZERO)
81  			.build();
82  	}
83  }
