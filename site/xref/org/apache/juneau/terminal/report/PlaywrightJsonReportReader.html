27  /**
28   * Reads Playwright's <c>--reporter=json</c> output.
29   *
30   * <p>
31   * The framework key is <c>playwright</c>.  Suites and specs are walked recursively; a suite whose title equals its file
32   * is the file-level suite and does not contribute to the test name.  A <c>flaky</c> test (failed, then passed on retry)
33   * is a pass.  A truncated file yields no tests and a warning.
34   */
35  @SuppressWarnings({
36  	"java:S1192", // JSON keys of the Playwright report are protocol literals.
37  	"java:S3776" // Walking the suite tree and folding retries is inherently branchy.
38  })
39  public final class PlaywrightJsonReportReader implements ReportReader {
40  
41  	private static final int MAX_DEPTH = 100;
42  	private static final String SEP = " › ";
43  
44  	@Override /* ReportReader */
45  	public String fw() {
46  		return "playwright";
47  	}
48  
49  	@Override /* ReportReader */
50  	public ReportResult read(Path path, ReportLimits lim) {
51  		var out = new ReportBuilder();
52  		var root = readRoot(path, lim, out);
53  		if (root == null)
54  			return out.build(fw());
55  		var files = new ArrayList<String>();
56  		var rows = new ArrayList<ReportTest>();
57  		var rowFiles = new ArrayList<String>();
58  		for (var s : list(root.get("suites"))) {
59  			var sm = map(s);
60  			if (sm != null && ! walk(sm, List.of(), null, 0, lim, out, files, rows, rowFiles))
61  				break;
62  		}
63  		var display = stripCommonDir(files);
64  		for (var i = 0; i < rows.size(); i++) {
65  			var r = rows.get(i);
66  			var suite = clip(display.get(files.indexOf(rowFiles.get(i))), lim.maxNameChars());
67  			out.tests.add(new ReportTest(suite, r.name(), r.status(), r.ms(), r.msg(), r.trace()));
68  		}
69  		return out.build(fw());
70  	}
71  
72  	/** Returns <jk>false</jk> when the test limit stopped the walk. */
73  	@SuppressWarnings({
74  		"java:S107" // Internal recursive walker; parameters are the traversal state.
75  	})
76  	private boolean walk(Map<String,Object> suite, List<String> titles, String parentFile, int depth, ReportLimits lim, ReportBuilder out,
77  			List<String> files, List<ReportTest> rows, List<String> rowFiles) {
78  		if (depth > MAX_DEPTH)
79  			return true;
80  		var file = str(suite.get("file")) != null ? str(suite.get("file")) : parentFile;
81  		var title = str(suite.get("title"));
82  		var path = titles;
83  		if (title != null && ! title.isEmpty() && ! title.equals(str(suite.get("file")))) {
84  			path = new ArrayList<>(titles);
85  			path.add(title);
86  		}
87  		for (var sp : list(suite.get("specs"))) {
88  			var spec = map(sp);
89  			if (spec == null)
90  				continue;
91  			var specFile = str(spec.get("file")) != null ? str(spec.get("file")) : file;
92  			var key = specFile == null ? "(unknown file)" : specFile;
93  			if (! files.contains(key))
94  				files.add(key);
95  			var names = new ArrayList<>(path);
96  			if (str(spec.get("title")) != null)
97  				names.add(str(spec.get("title")));
98  			for (var t : list(spec.get("tests"))) {
99  				var tm = map(t);
100 				if (tm == null)
101 					continue;
102 				if (rows.size() >= lim.maxTests()) {
103 					out.truncated = true;
104 					out.warnings.add("report truncated at " + lim.maxTests() + " tests");
105 					return false;
106 				}
107 				rows.add(test(names, tm, lim));
108 				rowFiles.add(key);
109 			}
110 		}
111 		for (var child : list(suite.get("suites"))) {
112 			var cm = map(child);
113 			if (cm != null && ! walk(cm, path, file, depth + 1, lim, out, files, rows, rowFiles))
114 				return false;
115 		}
116 		return true;
117 	}

119 	private static ReportTest test(List<String> names, Map<String,Object> t, ReportLimits lim) {
120 		var name = String.join(SEP, names);
121 		var project = str(t.get("projectName"));
122 		if (project != null && ! project.isEmpty())
123 			name += " [" + project + "]";
124 		var results = list(t.get("results"));
125 		Map<String,Object> last = null;
126 		long total = 0;
127 		var hasMs = false;
128 		Map<String,Object> failing = null;
129 		for (var r : results) {
130 			var rm = map(r);
131 			if (rm == null)
132 				continue;
133 			last = rm;
134 			var d = ms(rm.get("duration"));
135 			if (d != null) {
136 				total += d;
137 				hasMs = true;
138 			}
139 			if (failing == null && isFailed(str(rm.get("status"))))
140 				failing = rm;
141 		}
142 		var status = status(str(t.get("status")), last == null ? null : str(last.get("status")));
143 		String msg = null;
144 		String trace = null;
145 		if (status == TestStatus.FAIL && failing != null) {
146 			var text = errorText(failing);
147 			if (text != null) {
148 				var body = stripAnsi(text).strip();
149 				var first = TerminalViewChecks.firstLine(body);
150 				msg = first.isEmpty() ? null : clip(first, lim.maxMsgChars());
151 				trace = clip(body, lim.maxTraceChars());
152 			}
153 		}
154 		return new ReportTest(null, clip(name, lim.maxNameChars()), status, hasMs ? Long.valueOf(total) : null, msg, trace);
155 	}
156 
157 	private static boolean isFailed(String s) {
158 		return "failed".equals(s) || "timedOut".equals(s) || "interrupted".equals(s);
159 	}
160 
161 	private static TestStatus status(String testStatus, String lastResult) {
162 		if (testStatus != null) {
163 			switch (testStatus) {
164 				case "expected", "flaky": return TestStatus.PASS;
165 				case "unexpected": return TestStatus.FAIL;
166 				case "skipped": return TestStatus.SKIP;
167 				default: break;
168 			}
169 		}
170 		if (lastResult == null)
171 			return TestStatus.SKIP;
172 		if ("passed".equals(lastResult))
173 			return TestStatus.PASS;
174 		return isFailed(lastResult) ? TestStatus.FAIL : TestStatus.SKIP;
175 	}
176 
177 	private static String errorText(Map<String,Object> result) {
178 		var e = map(result.get("error"));
179 		if (e != null && str(e.get("message")) != null)
180 			return str(e.get("message"));
181 		for (var x : list(result.get("errors"))) {
182 			var xm = map(x);
183 			if (xm != null && str(xm.get("message")) != null)
184 				return str(xm.get("message"));
185 		}
186 		return null;
187 	}
