29  /**
30   * Streaming parser for the JUnit XML schema (Surefire <c>TEST-*.xml</c> and pytest <c>--junitxml</c> output).
31   *
32   * <p>
33   * Nothing is loaded as a tree: a test case is handed to the builder as soon as its element closes, and text inside a
34   * failing child is kept only up to the trace limit, so memory does not grow with the file.  The factory comes from
35   * {@link SecureXMLInputFactory}, so DTDs and external entities are never resolved.
36   */
37  @SuppressWarnings({
38  	"java:S1192" // The warning prefix is message text, not a shared token.
39  })
40  final class JUnitXmlParser {
41  
42  	private JUnitXmlParser() {}
43  
44  	static void parse(Path file, ReportLimits lim, ReportBuilder out) {
45  		var name = fileName(file);
46  		long size;
47  		try {
48  			size = Files.size(file);
49  			if (size > lim.maxFileBytes()) {
50  				out.skippedFiles++;
51  				out.warnings.add("report skipped: " + name + " is larger than " + lim.maxFileBytes() + " bytes");
52  				return;
53  			}
54  		} catch (IOException e) {
55  			out.skippedFiles++;
56  			out.warnings.add("report unreadable: " + (e instanceof NoSuchFileException ? "file not found" : "cannot read") + " (" + name + ")");
57  			return;
58  		}
59  		XMLStreamReader r = null;
60  		try (var in = new BufferedInputStream(Files.newInputStream(file))) {
61  			var f = SecureXMLInputFactory.newInstance();
62  			f.setProperty(XMLInputFactory.IS_COALESCING, false);
63  			r = f.createXMLStreamReader(in);
64  			read(r, lim, out);
65  		} catch (XMLStreamException e) {
66  			out.truncated = true;
67  			var msg = firstLine(e.getMessage());
68  			if (atEnd(e, size, msg))
69  				out.warnings.add("report truncated: unexpected end of file after " + out.tests.size() + " tests (" + name + ")");
70  			else
71  				out.warnings.add("report unreadable: " + clip(msg, 200) + " after " + out.tests.size() + " tests (" + name + ")");
72  		} catch (IOException | RuntimeException e) {
73  			out.truncated = true;
74  			out.warnings.add("report unreadable: " + clip(firstLine(e.getMessage()), 200) + " (" + name + ")");
75  		} finally {
76  			if (r != null) {
77  				try {
78  					r.close();
79  				} catch (XMLStreamException e) {
80  					// nothing to do
81  				}
82  			}
83  		}
84  	}
85  
86  	// An error at (or within a few chars of) the end of the input means the file was cut short; elsewhere it is malformed.
87  	private static boolean atEnd(XMLStreamException e, long size, String msg) {
88  		if (msg.contains("Premature end") || msg.contains("must start and end within the same entity") || msg.contains("nexpected end"))
89  			return true;
90  		var loc = e.getLocation();
91  		return loc != null && loc.getCharacterOffset() >= size - 8;
92  	}
93  
94  	private static String firstLine(String s) {
95  		return TerminalViewChecks.firstLine(s);
96  	}
97  
98  	@SuppressWarnings({
99  		"java:S3776" // The streaming state machine is inherently branchy.
100 	})
101 	private static void read(XMLStreamReader r, ReportLimits lim, ReportBuilder out) throws XMLStreamException {
102 		var suiteName = "";
103 		var inCase = false;
104 		String cName = null;
105 		String cClass = null;
106 		String cTime = null;
107 		String cMsg = null;
108 		TestStatus cStatus = null;
109 		StringBuilder text = null;
110 		var childDepth = 0;
111 		var skipDepth = 0;
112 		while (r.hasNext()) {
113 			switch (r.next()) {
114 				case XMLStreamConstants.START_ELEMENT -> {
115 					var el = r.getLocalName();
116 					if (skipDepth > 0) {
117 						skipDepth++;
118 					} else if (el.equals("testsuite") && ! inCase) {
119 						suiteName = nz(r.getAttributeValue(null, "name"));
120 					} else if (el.equals("testcase") && ! inCase) {
121 						inCase = true;
122 						cName = r.getAttributeValue(null, "name");
123 						cClass = r.getAttributeValue(null, "classname");
124 						cTime = r.getAttributeValue(null, "time");
125 						cStatus = TestStatus.PASS;
126 						cMsg = null;
127 						text = null;
128 					} else if (inCase && childDepth == 0) {
129 						switch (el) {
130 							case "failure", "error", "skipped" -> {
131 								cStatus = switch (el) {
132 									case "failure" -> TestStatus.FAIL;
133 									case "error" -> TestStatus.ERROR;
134 									default -> TestStatus.SKIP;
135 								};
136 								cMsg = r.getAttributeValue(null, "message");
137 								text = new StringBuilder();
138 								childDepth = 1;
139 							}
140 							case "flakyFailure", "flakyError", "rerunFailure", "rerunError" -> skipDepth = 1;
141 							default -> skipDepth = 1; // system-out, properties and the like
142 						}
143 					} else if (inCase) {
144 						childDepth++;
145 					}
146 				}
147 				case XMLStreamConstants.CHARACTERS, XMLStreamConstants.CDATA -> {
148 					if (text != null && skipDepth == 0 && text.length() <= lim.maxTraceChars())
149 						text.append(r.getText());
150 				}
151 				case XMLStreamConstants.END_ELEMENT -> {
152 					var el = r.getLocalName();
153 					if (skipDepth > 0) {
154 						skipDepth--;
155 					} else if (inCase && childDepth > 0) {
156 						childDepth--;
157 					} else if (inCase && el.equals("testcase")) {
158 						inCase = false;
159 						if (out.tests.size() >= lim.maxTests()) {
160 							out.truncated = true;
161 							out.warnings.add("report truncated at " + lim.maxTests() + " tests");
162 							return;
163 						}
164 						out.tests.add(build(suiteName, cClass, cName, cTime, cStatus, cMsg, text, lim));
165 						text = null;
166 					}
167 				}
168 				default -> { /* ignored */ }
169 			}
170 		}
171 	}

173 	@SuppressWarnings({
174 		"java:S107" // Internal factory; parameters are the parsed fields.
175 	})
176 	private static ReportTest build(String suiteName, String cClass, String cName, String cTime, TestStatus status, String attrMsg, StringBuilder text, ReportLimits lim) {
177 		var suite = cClass != null && ! cClass.isEmpty() ? cClass : suiteName;
178 		String msg = null;
179 		String trace = null;
180 		if (status != TestStatus.PASS) {
181 			var body = text == null ? "" : stripAnsi(text.toString()).strip();
182 			if (attrMsg != null && ! attrMsg.isEmpty())
183 				msg = clip(stripAnsi(attrMsg), lim.maxMsgChars());
184 			else if (! body.isEmpty())
185 				msg = clip(firstLine(body), lim.maxMsgChars());
186 			if (! body.isEmpty())
187 				trace = clip(body, lim.maxTraceChars());
188 		}
189 		return new ReportTest(clip(suite, lim.maxNameChars()), clip(nz(cName), lim.maxNameChars()), status, millis(cTime), msg, trace);
190 	}
191 
192 	private static Long millis(String t) {
193 		if (t == null)
194 			return null;
195 		try {
196 			var d = Double.parseDouble(t.trim());
197 			return d >= 0 && d < 9.0e12 ? Long.valueOf(Math.round(d * 1000)) : null;
198 		} catch (NumberFormatException e) {
199 			return null;
200 		}
201 	}
202 
203 	private static String nz(String s) {
204 		return s == null ? "" : s;
205 	}
