17  package org.apache.juneau.bean.html5;
18  
19  import org.apache.juneau.marshall.*;
20  
21  /**
22   * DTO for an HTML <a class="doclink" href="https://www.w3.org/TR/html5/grouping-content.html#the-main-element">&lt;main&gt;</a>
23   * element.
24   *
25   * <p>
26   * The main element represents the main content of the body of a document or application. It is
27   * used to identify the primary content of the page, excluding content that is repeated across
28   * multiple pages such as navigation, headers, footers, and sidebars. The main element should
29   * be used only once per page and contains the central topic or functionality of the page. It
30   * is important for accessibility and helps screen readers identify the main content area.
31   *
32   * <h5 class='section'>Examples:</h5>
33   * <p class='bcode w800'>
34   * 	<jc>// Simple main content</jc>
35   * 	Main <jv>simple</jv> = <jsm>main</jsm>(
36   * 		<jsm>h1</jsm>(<js>"Welcome to Our Site"</js>),
37   * 		<jsm>p</jsm>(<js>"This is the main content of our page."</js>)
38   * 	);
39   *
40   * 	<jc>// Main with styling</jc>
41   * 	Main <jv>styled</jv> = <jsm>main</jsm>(
42   * 		<jsm>h1</jsm>(<js>"About Us"</js>),
43   * 		<jsm>p</jsm>(<js>"Learn more about our company and mission."</js>)
44   * 	).class_(<js>"main-content"</js>);
45   *
46   * 	<jc>// Main with complex content</jc>
47   * 	Main <jv>complex</jv> = <jsm>main</jsm>(
48   * 		<jsm>h1</jsm>(<js>"Product Catalog"</js>),
49   * 		<jsm>section</jsm>(
50   * 			<jsm>h2</jsm>(<js>"Featured Products"</js>),
51   * 			<jsm>p</jsm>(<js>"Check out our latest offerings."</js>)
52   * 		),
53   * 		<jsm>section</jsm>(
54   * 			<jsm>h2</jsm>(<js>"Categories"</js>),
55   * 			<jsm>ul</jsm>(
56   * 				<jsm>li</jsm>(<js>"Electronics"</js>),
57   * 				<jsm>li</jsm>(<js>"Clothing"</js>),
58   * 				<jsm>li</jsm>(<js>"Books"</js>)
59   * 			)
60   * 		)
61   * 	);
62   *
63   * 	<jc>// Main with ID</jc>
64   * 	Main <jv>withId</jv> = <jsm>main</jsm>(
65   * 		<jsm>h1</jsm>(<js>"Main Content"</js>),
66   * 		<jsm>p</jsm>(<js>"This is the main content area."</js>)
67   * 	).id(<js>"page-main"</js>);
68   *
69   * 	<jc>// Main with styling</jc>
70   * 	Main <jv>styled2</jv> = <jsm>main</jsm>(
71   * 		<jsm>h1</jsm>(<js>"Centered Content"</js>),
72   * 		<jsm>p</jsm>(<js>"This content is centered and styled."</js>)
73   * 	).style(<js>"max-width: 800px; margin: 0 auto; padding: 20px;"</js>);
74   *
75   * 	<jc>// Main with multiple sections</jc>
76   * 	Main <jv>multiSection</jv> = <jsm>main</jsm>(
77   * 		<jsm>section</jsm>(
78   * 			<jsm>h1</jsm>(<js>"Page Title"</js>),
79   * 			<jsm>p</jsm>(<js>"Introduction to the page content."</js>)
80   * 		),
81   * 		<jsm>section</jsm>(
82   * 			<jsm>h2</jsm>(<js>"Content Section"</js>),
83   * 			<jsm>p</jsm>(<js>"Detailed content goes here."</js>)
84   * 		),
85   * 		<jsm>section</jsm>(
86   * 			<jsm>h2</jsm>(<js>"Conclusion"</js>),
87   * 			<jsm>p</jsm>(<js>"Summary and closing thoughts."</js>)
88   * 		)
89   * 	);
90   *
91   * 	<jc>// Main with article</jc>
92   * 	Main <jv>withArticle</jv> = <jsm>main</jsm>(
93   * 		<jsm>article</jsm>(
94   * 			<jsm>h1</jsm>(<js>"Article Title"</js>),
95   * 			<jsm>p</jsm>(<js>"Article content goes here."</js>),
96   * 			<jsm>footer</jsm>(<js>"Article footer"</js>)
97   * 		)
98   * 	);
99   * </p>
100  *
101  * <h5 class='section'>See Also:</h5><ul>
102  * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauBeanHtml5">juneau-bean-html5</a>
103  * </ul>
104  */
105 @Marshalled(typeName = "main")
106 public class Main extends HtmlElementContainer<Main> {
107 
108 	/**
109 	 * Creates an empty {@link Main} element.
110 	 */
111 	public Main() {}
112 
113 	/**
114 	 * Creates a {@link Main} element with the specified child nodes.
115 	 *
116 	 * @param children The child nodes.
117 	 */
118 	public Main(Object...children) {
119 		children(children);
120 	}
121 
122 }
