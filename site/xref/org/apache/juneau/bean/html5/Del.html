17  package org.apache.juneau.bean.html5;
18  
19  import org.apache.juneau.marshall.*;
20  
21  /**
22   * DTO for an HTML <a class="doclink" href="https://www.w3.org/TR/html5/edits.html#the-del-element">&lt;del&gt;</a>
23   * element.
24   *
25   * <p>
26   * The del element represents text that has been deleted from a document. It is typically rendered
27   * with a strikethrough effect to indicate that the content has been removed. The cite and datetime
28   * attributes can be used to provide information about when and why the deletion occurred.
29   *
30   * <h5 class='section'>Examples:</h5>
31   * <p class='bcode w800'>
32   * 	<jk>import static</jk> org.apache.juneau.bean.html5.HtmlBuilder.*;
33   *
34   * 	<jc>// Simple deleted text</jc>
35   * 	Del <jv>del1</jv> = <jsm>del</jsm>(<js>"This text has been removed."</js>);
36   *
37   * 	<jc>// Deleted text with citation and timestamp</jc>
38   * 	Del <jv>del2</jv> = <jsm>del</jsm>()
39   * 		.cite(<js>"https://example.com/revision-log"</js>)
40   * 		.datetime(<js>"2024-01-15T10:30:00Z"</js>)
41   * 		.text(<js>"Outdated information removed"</js>);
42   *
43   * 	<jc>// Deleted text with reason</jc>
44   * 	Del <jv>del3</jv> = <jsm>del</jsm>()
45   * 		.cite(<js>"https://example.com/corrections"</js>)
46   * 		.datetime(<js>"2024-01-15"</js>)
47   * 		.text(<js>"Incorrect statement"</js>)
48   * 		.title(<js>"Removed due to factual error"</js>);
49   * </p>
50   *
51   * <p>
52   * The following convenience methods are provided for constructing instances of this bean:
53   * <ul class='javatree'>
54   * 	<li class='jc'>{@link HtmlBuilder}
55   * 	<ul class='javatree'>
56   * 		<li class='jm'>{@link HtmlBuilder#del() del()}
57   * 		<li class='jm'>{@link HtmlBuilder#del(Object...) del(Object...)}
58   * 	</ul>
59   * </ul>
60   * </p>
61   *
62   * <h5 class='section'>See Also:</h5><ul>
63   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauBeanHtml5">juneau-bean-html5</a>
64   * </ul>
65   */
66  @Marshalled(typeName = "del")
67  public class Del extends HtmlElementMixed<Del> {
68  
69  	/**
70  	 * Creates an empty {@link Del} element.
71  	 */
72  	public Del() {}
73  
74  	/**
75  	 * Creates a {@link Del} element with the specified {@link Del#children(Object[])} node.
76  	 *
77  	 * @param children The {@link Del#children(Object[])} node. Must not be <jk>null</jk>.
78  	 */
79  	public Del(Object...children) {
80  		children(children);
81  	}
82  
83  	/**
84  	 * <a class="doclink" href="https://www.w3.org/TR/html5/edits.html#attr-mod-cite">cite</a> attribute.
85  	 *
86  	 * <p>
87  	 * Specifies the URL of a document that explains the reason for the deletion.
88  	 * This provides context and justification for the edit.
89  	 *
90  	 * <p>
91  	 * The URL should point to a document that explains why the content was deleted.
92  	 *
93  	 * @param value The URL explaining the reason for the deletion. Can be <jk>null</jk> to unset the attribute.
94  	 * @return This object.
95  	 */
96  	public Del cite(String value) {
97  		attr("cite", value);
98  		return this;
99  	}
100 
101 	/**
102 	 * <a class="doclink" href="https://www.w3.org/TR/html5/edits.html#attr-mod-datetime">datetime</a> attribute.
103 	 *
104 	 * <p>
105 	 * Specifies the date and time when the content was deleted. Used for tracking
106 	 * the history of document changes.
107 	 *
108 	 * <p>
109 	 * The value should be a valid date-time string in ISO 8601 format.
110 	 *
111 	 * @param value The date and time when the content was deleted. Can be <jk>null</jk> to unset the attribute.
112 	 * @return This object.
113 	 */
114 	public Del datetime(String value) {
115 		attr("datetime", value);
116 		return this;
117 	}
118 
119 }
