17  package org.apache.juneau.bean.html5;
18  
19  import org.apache.juneau.marshall.*;
20  
21  /**
22   * DTO for an HTML <a class="doclink" href="https://www.w3.org/TR/html5/text-level-semantics.html#the-b-element">&lt;b&gt;</a>
23   * element.
24   *
25   * <p>
26   * The b element represents a span of text to which attention is being drawn for utilitarian purposes
27   * without conveying any extra importance and with no implication of an alternate voice or mood. It is
28   * used to stylistically offset text from the normal prose without conveying any special importance or
29   * emphasis. The b element is typically rendered in bold by browsers, but the visual styling should be
30   * controlled with CSS rather than relying on the default browser styling.
31   *
32   * <h5 class='section'>Examples:</h5>
33   * <p class='bcode w800'>
34   * 	<jk>import static</jk> org.apache.juneau.bean.html5.HtmlBuilder.*;
35   *
36   * 	<jc>// Product name in a sentence</jc>
37   * 	B <jv>product</jv> = <jsm>b</jsm>(<js>"Widget Pro"</js>);
38   *
39   * 	<jc>// Keywords in text</jc>
40   * 	B <jv>keywords</jv> = <jsm>b</jsm>(<js>"HTML"</js>, <jsm>b</jsm>(<js>"CSS"</js>), <js>" and "</js>, <jsm>b</jsm>(<js>"JavaScript"</js>));
41   *
42   * 	<jc>// Names or titles</jc>
43   * 	B <jv>name</jv> = <jsm>b</jsm>(<js>"Dr. Jane Smith"</js>);
44   *
45   * 	<jc>// Stylistic offset without emphasis</jc>
46   * 	B <jv>offset</jv> = <jsm>b</jsm>(<js>"Important note"</js>).class_(<js>"highlight"</js>);
47   *
48   * 	<jc>// Text that needs visual distinction</jc>
49   * 	B <jv>distinct</jv> = <jsm>b</jsm>(<js>"User ID: 12345"</js>);
50   *
51   * 	<jc>// Mixed content with bold text</jc>
52   * 	B <jv>mixed</jv> = <jsm>b</jsm>(
53   * 		<js>"Welcome to "</js>,
54   * 		<jsm>b</jsm>(<js>"Our Website"</js>),
55   * 		<js>" - the best place for "</js>,
56   * 		<jsm>b</jsm>(<js>"quality products"</js>)
57   * 	);
58   *
59   * 	<jc>// Bold text with styling</jc>
60   * 	B <jv>styled</jv> = <jsm>b</jsm>(<js>"Amazing Widget"</js>)
61   * 		.class_(<js>"product-name"</js>)
62   * 		.style(<js>"color: blue;"</js>);
63   * </p>
64   *
65   * <p>
66   * The following convenience methods are provided for constructing instances of this bean:
67   * <ul class='javatree'>
68   * 	<li class='jc'>{@link HtmlBuilder}
69   * 	<ul class='javatree'>
70   * 		<li class='jm'>{@link HtmlBuilder#b() b()}
71   * 		<li class='jm'>{@link HtmlBuilder#b(Object...) b(Object...)}
72   * 	</ul>
73   * </ul>
74   * </p>
75   *
76   * <h5 class='section'>See Also:</h5><ul>
77   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauBeanHtml5">juneau-bean-html5</a>
78   * </ul>
79   */
80  @Marshalled(typeName = "b")
81  public class B extends HtmlElementMixed<B> {
82  
83  	/**
84  	 * Creates an empty {@link B} element.
85  	 */
86  	public B() {}
87  
88  	/**
89  	 * Creates a {@link B} element with the specified child nodes.
90  	 *
91  	 * @param children The child nodes.
92  	 */
93  	public B(Object...children) {
94  		children(children);
95  	}
96  
97  }
