17  package org.apache.juneau.bean.html5;
18  
19  import org.apache.juneau.marshall.*;
20  
21  /**
22   * DTO for an HTML <a class="doclink" href="https://www.w3.org/TR/html5/forms.html#the-datalist-element">&lt;datalist&gt;</a>
23   * element.
24   *
25   * <p>
26   * The datalist element represents a set of option elements that represent predefined options for other
27   * controls. It is used to provide a list of suggested values for input elements, allowing users to
28   * either select from the predefined options or enter their own custom value. The datalist element
29   * is typically associated with an input element through the list attribute, and the options within
30   * the datalist provide autocomplete suggestions.
31   *
32   * <h5 class='section'>Examples:</h5>
33   * <p class='bcode w800'>
34   * 	<jk>import static</jk> org.apache.juneau.bean.html5.HtmlBuilder.*;
35   *
36   * 	<jc>// Simple datalist</jc>
37   * 	Datalist <jv>simple</jv> = <jsm>datalist</jsm>(<js>"colors"</js>,
38   * 		<jsm>option</jsm>(<js>"red"</js>, <js>"Red"</js>),
39   * 		<jsm>option</jsm>(<js>"green"</js>, <js>"Green"</js>),
40   * 		<jsm>option</jsm>(<js>"blue"</js>, <js>"Blue"</js>)
41   * 	);
42   *
43   * 	<jc>// Datalist with styling</jc>
44   * 	Datalist <jv>styled</jv> = <jsm>datalist</jsm>(<js>"countries"</js>,
45   * 		<jsm>option</jsm>(<js>"us"</js>, <js>"United States"</js>),
46   * 		<jsm>option</jsm>(<js>"ca"</js>, <js>"Canada"</js>),
47   * 		<jsm>option</jsm>(<js>"mx"</js>, <js>"Mexico"</js>)
48   * 	).class_(<js>"country-list"</js>);
49   *
50   * 	<jc>// Datalist with multiple options</jc>
51   * 	Datalist <jv>multiple</jv> = <jsm>datalist</jsm>(<js>"fruits"</js>,
52   * 		<jsm>option</jsm>(<js>"apple"</js>, <js>"Apple"</js>),
53   * 		<jsm>option</jsm>(<js>"banana"</js>, <js>"Banana"</js>),
54   * 		<jsm>option</jsm>(<js>"cherry"</js>, <js>"Cherry"</js>),
55   * 		<jsm>option</jsm>(<js>"date"</js>, <js>"Date"</js>),
56   * 		<jsm>option</jsm>(<js>"elderberry"</js>, <js>"Elderberry"</js>)
57   * 	);
58   *
59   * 	<jc>// Datalist with complex options</jc>
60   * 	Datalist <jv>complex</jv> = <jsm>datalist</jsm>(<js>"products"</js>,
61   * 		<jsm>option</jsm>(<js>"laptop-001"</js>, <js>"Laptop Pro 15\" - $1,299"</js>),
62   * 		<jsm>option</jsm>(<js>"laptop-002"</js>, <js>"Laptop Air 13\" - $999"</js>),
63   * 		<jsm>option</jsm>(<js>"tablet-001"</js>, <js>"Tablet 10\" - $499"</js>)
64   * 	);
65   *
66   * 	<jc>// Datalist with ID</jc>
67   * 	Datalist <jv>withId</jv> = <jsm>datalist</jsm>(<js>"cities"</js>,
68   * 		<jsm>option</jsm>(<js>"new-york"</js>, <js>"New York"</js>),
69   * 		<jsm>option</jsm>(<js>"los-angeles"</js>, <js>"Los Angeles"</js>),
70   * 		<jsm>option</jsm>(<js>"chicago"</js>, <js>"Chicago"</js>)
71   * 	);
72   *
73   * 	<jc>// Datalist with styling</jc>
74   * 	Datalist <jv>styled2</jv> = <jsm>datalist</jsm>(<js>"sizes"</js>,
75   * 		<jsm>option</jsm>(<js>"xs"</js>, <js>"Extra Small"</js>),
76   * 		<jsm>option</jsm>(<js>"s"</js>, <js>"Small"</js>),
77   * 		<jsm>option</jsm>(<js>"m"</js>, <js>"Medium"</js>),
78   * 		<jsm>option</jsm>(<js>"l"</js>, <js>"Large"</js>),
79   * 		<jsm>option</jsm>(<js>"xl"</js>, <js>"Extra Large"</js>)
80   * 	).style(<js>"display: none;"</js>);
81   * </p>
82   *
83   * <p>
84   * The following convenience methods are provided for constructing instances of this bean:
85   * <ul class='javatree'>
86   * 	<li class='jc'>{@link HtmlBuilder}
87   * 	<ul class='javatree'>
88   * 		<li class='jm'>{@link HtmlBuilder#datalist() datalist()}
89   * 		<li class='jm'>{@link HtmlBuilder#datalist(String, Option...) datalist(String, Option...)}
90   * 	</ul>
91   * </ul>
92   * </p>
93   *
94   * <h5 class='section'>See Also:</h5><ul>
95   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauBeanHtml5">juneau-bean-html5</a>
96   * </ul>
97   */
98  @Marshalled(typeName = "datalist")
99  public class Datalist extends HtmlElementContainer<Datalist> {
100 
101 	/**
102 	 * Creates an empty {@link Datalist} element.
103 	 */
104 	public Datalist() {}
105 
106 	/**
107 	 * Creates a {@link Datalist} element with the specified {@link Datalist#id(String)} attribute and child nodes.
108 	 *
109 	 * @param id The {@link Datalist#id(String)} attribute.
110 	 * @param children The child nodes.
111 	 */
112 	public Datalist(String id, Option...children) {
113 		id(id).children((Object[])children);
114 	}
115 
116 }
