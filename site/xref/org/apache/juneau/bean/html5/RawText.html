17  package org.apache.juneau.bean.html5;
18  
19  import org.apache.juneau.marshall.xml.*;
20  
21  /**
22   * A re-serializable, {@link String}-backed raw-content wrapper that the XML and HTML serializers emit <b>verbatim</b>.
23   *
24   * <p>
25   * This is the reusable analogue of handing the serializer a one-shot {@link java.io.Reader}: the value is emitted
26   * without any XML entity escaping or <c>_x####_</c> whitespace encoding, and without a wrapping element &mdash; but
27   * because it is backed by a {@code String} it can be serialized any number of times.  This makes a bean carrying a
28   * {@code RawText} value survive a serialize&rarr;object&rarr;serialize cycle (for the serialize direction), unlike a
29   * consumed {@code Reader}.
30   *
31   * <p>
32   * Typical use is embedding pre-built script/style bodies into a {@link Script}/{@link Style} element:
33   *
34   * <h5 class='figure'>Example:</h5>
35   * <p class='bjava'>
36   * 	<jc>// Browser-safe raw JSON sidecar ('&lt;' pre-escaped to \\u003c by the caller).</jc>
37   * 	Script <jv>s</jv> = <jsm>script</jsm>(<js>"application/json"</js>).text(<jsm>rawText</jsm>(<jv>escapedJson</jv>));
38   * </p>
39   *
40   * <p>
41   * <b>Safety:</b> content is emitted verbatim.  Because Juneau parses via a StAX
42   * {@link javax.xml.stream.XMLStreamReader}, a value containing a raw <js>'&lt;'</js> (e.g. a literal
43   * <c>&lt;/script&gt;</c> sequence) or a bare <js>'&amp;'</js> is browser-correct on serialize but will NOT round-trip
44   * back through Juneau's parser.  Callers embedding markup-bearing content are responsible for escaping <js>'&lt;'</js>
45   * to the JS/CSS unicode escape <js>"\\u003c"</js> before wrapping.
46   *
47   * <h5 class='section'>See Also:</h5><ul>
48   * 	<li class='jm'>{@link HtmlBuilder#rawText(String)}
49   * 	<li class='jic'>{@link RawContent}
50   * </ul>
51   */
52  public class RawText implements RawContent {
53  
54  	private final String text;
55  
56  	/**
57  	 * Constructor.
58  	 *
59  	 * @param text The raw, verbatim text content.  Can be <jk>null</jk> (nothing is emitted).
60  	 */
61  	public RawText(String text) {
62  		this.text = text;
63  	}
64  
65  	@Override /* Overridden from RawContent */
66  	public String toString() {
67  		return text;
68  	}
69  }
