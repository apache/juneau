17  package org.apache.juneau.bean.html5;
18  
19  import java.net.*;
20  
21  import org.apache.juneau.marshall.*;
22  
23  /**
24   * DTO for an HTML <a class="doclink" href="https://www.w3.org/TR/html5/embedded-content-0.html#the-source-element">&lt;source&gt;</a>
25   * element.
26   *
27   * <p>
28   * The source element specifies multiple media resources for media elements like audio and video.
29   * It allows browsers to choose the most appropriate source based on format support, bandwidth,
30   * and other factors. The source element is used inside audio and video elements to provide
31   * fallback options for different media formats.
32   *
33   * <h5 class='section'>Examples:</h5>
34   * <p class='bcode w800'>
35   * 	<jk>import static</jk> org.apache.juneau.bean.html5.HtmlBuilder.*;
36   *
37   * 	<jc>// Video with multiple sources</jc>
38   * 	Video <jv>video</jv> = <jsm>video</jsm>()
39   * 		.controls(<jk>true</jk>)
40   * 		.children(
41   * 			<jsm>source</jsm>().src(<js>"movie.mp4"</js>).type(<js>"video/mp4"</js>),
42   * 			<jsm>source</jsm>().src(<js>"movie.webm"</js>).type(<js>"video/webm"</js>),
43   * 			<jsm>source</jsm>().src(<js>"movie.ogg"</js>).type(<js>"video/ogg"</js>)
44   * 		);
45   *
46   * 	<jc>// Audio with multiple sources</jc>
47   * 	Audio <jv>audio</jv> = <jsm>audio</jsm>()
48   * 		.controls(<jk>true</jk>)
49   * 		.children(
50   * 			<jsm>source</jsm>().src(<js>"sound.mp3"</js>).type(<js>"audio/mpeg"</js>),
51   * 			<jsm>source</jsm>().src(<js>"sound.ogg"</js>).type(<js>"audio/ogg"</js>),
52   * 			<jsm>source</jsm>().src(<js>"sound.wav"</js>).type(<js>"audio/wav"</js>)
53   * 		);
54   *
55   * 	<jc>// Picture with multiple sources</jc>
56   * 	Picture <jv>picture</jv> = <jsm>picture</jsm>(
57   * 		<jsm>source</jsm>().src(<js>"image.webp"</js>).type(<js>"image/webp"</js>),
58   * 		<jsm>source</jsm>().src(<js>"image.jpg"</js>).type(<js>"image/jpeg"</js>)
59   * 	);
60   *
61   * 	<jc>// Source with media query</jc>
62   * 	Source <jv>responsive</jv> = <jsm>source</jsm>()
63   * 		.src(<js>"large-image.jpg"</js>)
64   * 		.media(<js>"(min-width: 800px)"</js>)
65   * 		.type(<js>"image/jpeg"</js>);
66   * </p>
67   *
68   * <p>
69   * The following convenience methods are provided for constructing instances of this bean:
70   * <ul class='javatree'>
71   * 	<li class='jc'>{@link HtmlBuilder}
72   * 	<ul class='javatree'>
73   * 		<li class='jm'>{@link HtmlBuilder#source() source()}
74   * 	</ul>
75   * </ul>
76   * </p>
77   *
78   * <h5 class='section'>See Also:</h5><ul>
79   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauBeanHtml5">juneau-bean-html5</a>
80   * </ul>
81   */
82  @Marshalled(typeName = "source")
83  public class Source extends HtmlElementVoid<Source> {
84  
85  	/**
86  	 * Creates an empty {@link Source} element.
87  	 */
88  	public Source() {}
89  
90  	/**
91  	 * Creates a {@link Source} element with the specified {@link Source#src(Object)} and {@link Source#type(String)}
92  	 * attributes.
93  	 *
94  	 * @param src The {@link Source#src(Object)} attribute. Can be <jk>null</jk>, in which case the attribute is stored with a <jk>null</jk> value (not removed).
95  	 * @param type The {@link Source#type(String)} attribute. Can be <jk>null</jk> to unset the attribute.
96  	 */
97  	public Source(Object src, String type) {
98  		src(src).type(type);
99  	}
100 
101 	/**
102 	 * <a class="doclink" href="https://www.w3.org/TR/html5/embedded-content-0.html#attr-source-src">src</a> attribute.
103 	 *
104 	 * <p>
105 	 * Address of the resource.
106 	 *
107 	 * <p>
108 	 * The value can be of any of the following types: {@link URI}, {@link URL}, {@link String}.
109 	 * Strings must be valid URIs.
110 	 *
111 	 * <p>
112 	 * URIs defined by {@link UriResolver} can be used for values.
113 	 *
114 	 * @param value
115 	 * 	The new value for this attribute.
116 	 * 	Typically a {@link URL} or {@link String}.
117 	 * 	Can be <jk>null</jk>, in which case the attribute is stored with a <jk>null</jk> value (not removed).
118 	 * @return This object.
119 	 */
120 	public Source src(Object value) {
121 		attrUri("src", value);
122 		return this;
123 	}
124 
125 	/**
126 	 * <a class="doclink" href="https://www.w3.org/TR/html5/embedded-content-0.html#attr-source-type">type</a> attribute.
127 	 *
128 	 * <p>
129 	 * Specifies the MIME type of the media resource. Helps browsers determine if they can play the media
130 	 * and which source to use when multiple sources are provided.
131 	 *
132 	 * <p>
133 	 * Common values:
134 	 * <ul>
135 	 * 	<li><js>"video/mp4"</js> - MP4 video</li>
136 	 * 	<li><js>"video/webm"</js> - WebM video</li>
137 	 * 	<li><js>"audio/mp3"</js> - MP3 audio</li>
138 	 * 	<li><js>"audio/ogg"</js> - OGG audio</li>
139 	 * 	<li><js>"audio/wav"</js> - WAV audio</li>
140 	 * </ul>
141 	 *
142 	 * @param value The MIME type of the media resource. Can be <jk>null</jk> to unset the attribute.
143 	 * @return This object.
144 	 */
145 	public Source type(String value) {
146 		attr("type", value);
147 		return this;
148 	}
149 }
