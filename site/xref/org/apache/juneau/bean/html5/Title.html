17  package org.apache.juneau.bean.html5;
18  
19  import org.apache.juneau.marshall.*;
20  
21  /**
22   * DTO for an HTML <a class="doclink" href="https://www.w3.org/TR/html5/document-metadata.html#the-title-element">&lt;title&gt;</a>
23   * element.
24   *
25   * <p>
26   * The title element represents the document's title or name. It is used to provide a title
27   * for the document that appears in the browser's title bar, bookmarks, and search engine
28   * results. The title element should be placed within the head element and should be descriptive
29   * and unique for each page. It is important for SEO, accessibility, and user experience as
30   * it helps users identify the page content and purpose.
31   *
32   * <h5 class='section'>Examples:</h5>
33   * <p class='bcode w800'>
34   * 	<jk>import static</jk> org.apache.juneau.bean.html5.HtmlBuilder.*;
35   *
36   * 	<jc>// Simple page title</jc>
37   * 	Title <jv>simple</jv> = <jsm>title</jsm>(<js>"Welcome to My Website"</js>);
38   *
39   * 	<jc>// Title with dynamic content</jc>
40   * 	Title <jv>dynamic</jv> = <jsm>title</jsm>(<js>"User Dashboard - "</js> + <jv>username</jv>);
41   *
42   * 	<jc>// Title with branding</jc>
43   * 	Title <jv>branded</jv> = <jsm>title</jsm>(<js>"About Us | My Company"</js>);
44   *
45   * 	<jc>// Title with SEO keywords</jc>
46   * 	Title <jv>seo</jv> = <jsm>title</jsm>(<js>"Best Coffee Shops in Seattle - 2024 Guide"</js>);
47   *
48   * 	<jc>// Title with separator</jc>
49   * 	Title <jv>separated</jv> = <jsm>title</jsm>(<js>"Contact Us - Get in Touch"</js>);
50   *
51   * 	<jc>// Title with emoji</jc>
52   * 	Title <jv>emoji</jv> = <jsm>title</jsm>(<js>"🎉 Welcome to Our Store!"</js>);
53   * </p>
54   *
55   * <h5 class='section'>See Also:</h5><ul>
56   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauBeanHtml5">juneau-bean-html5</a>
57   * </ul>
58   */
59  @Marshalled(typeName = "title")
60  public class Title extends HtmlElementRawText<Title> {
61  
62  	/**
63  	 * Creates an empty {@link Title} element.
64  	 */
65  	public Title() {}
66  
67  	/**
68  	 * Creates a {@link Title} element with the specified {@link Title#text(Object)} node.
69  	 *
70  	 * @param text The {@link Title#text(Object)} node. Can be <jk>null</jk> to leave the text unset.
71  	 */
72  	public Title(String text) {
73  		text(text);
74  	}
75  }
