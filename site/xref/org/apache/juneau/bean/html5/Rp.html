17  package org.apache.juneau.bean.html5;
18  
19  import org.apache.juneau.marshall.*;
20  
21  /**
22   * DTO for an HTML <a class="doclink" href="https://www.w3.org/TR/html5/text-level-semantics.html#the-rp-element">&lt;rp&gt;</a>
23   * element.
24   *
25   * <p>
26   * The rp element represents a fallback parenthesis for browsers that do not support ruby annotations.
27   * It is used within a ruby element to provide fallback text that will be displayed when ruby
28   * annotations are not supported. The rp element is typically used to wrap parentheses around
29   * ruby text (rt elements) to provide a fallback display format. It is part of the ruby annotation
30   * system and helps ensure that ruby annotations are displayed correctly across different browsers.
31   *
32   * <h5 class='section'>Examples:</h5>
33   * <p class='bcode w800'>
34   * 	<jk>import static</jk> org.apache.juneau.bean.html5.HtmlBuilder.*;
35   *
36   * 	<jc>// Simple ruby parenthesis</jc>
37   * 	Rp <jv>simple</jv> = <jsm>rp</jsm>(<js>"("</js>);
38   *
39   * 	<jc>// Rp with styling</jc>
40   * 	Rp <jv>styled</jv> = <jsm>rp</jsm>(<js>")"</js>)
41   * 		.class_(<js>"ruby-paren"</js>);
42   *
43   * 	<jc>// Rp with complex content</jc>
44   * 	Rp <jv>complex</jv> = <jsm>rp</jsm>(
45   * 		<js>"("</js>,
46   * 		<jsm>strong</jsm>(<js>"注"</js>),
47   * 		<js>")"</js>
48   * 	);
49   *
50   * 	<jc>// Rp with ID</jc>
51   * 	Rp <jv>withId</jv> = <jsm>rp</jsm>(<js>"("</js>)
52   * 		.id(<js>"ruby-paren-1"</js>);
53   *
54   * 	<jc>// Rp with styling</jc>
55   * 	Rp <jv>styled2</jv> = <jsm>rp</jsm>(<js>")"</js>)
56   * 		.style(<js>"color: #999; font-size: 0.8em;"</js>);
57   *
58   * 	<jc>// Rp with multiple elements</jc>
59   * 	Rp <jv>multiple</jv> = <jsm>rp</jsm>(
60   * 		<js>"("</js>,
61   * 		<jsm>rp</jsm>(<js>"注"</js>),
62   * 		<js>")"</js>
63   * 	);
64   *
65   * 	// Rp with links
66   * 	Rp withLinks = new Rp()
67   * 		.children(
68   * 			"(",
69   * 			new A().href("/help/ruby").children("注"),
70   * 			")"
71   * 		);
72   * </p>
73   *
74   * <h5 class='section'>See Also:</h5><ul>
75   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauBeanHtml5">juneau-bean-html5</a>
76   * </ul>
77   */
78  @Marshalled(typeName = "rp")
79  public class Rp extends HtmlElementMixed<Rp> {
80  
81  	/**
82  	 * Creates an empty {@link Rp} element.
83  	 */
84  	public Rp() {}
85  
86  	/**
87  	 * Creates a {@link Rp} element with the specified {@link Rp#children(Object[])} nodes.
88  	 *
89  	 * @param children The {@link Rp#children(Object[])} nodes. Must not be <jk>null</jk>.
90  	 */
91  	public Rp(Object...children) {
92  		children(children);
93  	}
94  }
