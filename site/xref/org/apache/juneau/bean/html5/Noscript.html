17  package org.apache.juneau.bean.html5;
18  
19  import org.apache.juneau.marshall.*;
20  
21  /**
22   * DTO for an HTML <a class="doclink" href="https://www.w3.org/TR/html5/scripting-1.html#the-noscript-element">&lt;noscript&gt;</a>
23   * element.
24   *
25   * <p>
26   * The noscript element represents nothing if scripting is enabled, and represents its children
27   * if scripting is disabled. It is used to provide alternative content for users who have
28   * JavaScript disabled or when JavaScript is not available. The noscript element can contain
29   * any flow content and is typically used to display a message or alternative functionality
30   * when JavaScript cannot be executed. It is commonly placed in the head or body of a document.
31   *
32   * <h5 class='section'>Examples:</h5>
33   * <p class='bcode w800'>
34   * 	<jk>import static</jk> org.apache.juneau.bean.html5.HtmlBuilder.*;
35   *
36   * 	<jc>// Simple noscript message</jc>
37   * 	Noscript <jv>simple</jv> = <jsm>noscript</jsm>(<js>"Please enable JavaScript to use this application."</js>);
38   *
39   * 	<jc>// Noscript with styling</jc>
40   * 	Noscript <jv>styled</jv> = <jsm>noscript</jsm>(<js>"JavaScript is required for this page to function properly."</js>)
41   * 		.class_(<js>"noscript-message"</js>);
42   *
43   * 	<jc>// Noscript with complex content</jc>
44   * 	Noscript <jv>complex</jv> = <jsm>noscript</jsm>(
45   * 		<jsm>h2</jsm>(<js>"JavaScript Required"</js>),
46   * 		<jsm>p</jsm>(<js>"This application requires JavaScript to function."</js>),
47   * 		<jsm>p</jsm>(<js>"Please enable JavaScript in your browser and refresh the page."</js>)
48   * 	);
49   *
50   * 	<jc>// Noscript with ID</jc>
51   * 	Noscript <jv>withId</jv> = <jsm>noscript</jsm>(<js>"JavaScript is disabled."</js>)
52   * 		.id(<js>"noscript-message"</js>);
53   *
54   * 	<jc>// Noscript with styling</jc>
55   * 	Noscript <jv>styled2</jv> = <jsm>noscript</jsm>(<js>"JavaScript is required for this page."</js>)
56   * 		.style(<js>"background-color: #f8d7da; color: #721c24; padding: 10px; border: 1px solid #f5c6cb;"</js>);
57   *
58   * 	<jc>// Noscript with multiple elements</jc>
59   * 	Noscript <jv>multiple</jv> = <jsm>noscript</jsm>(
60   * 		<jsm>h3</jsm>(<js>"JavaScript Disabled"</js>),
61   * 		<jsm>p</jsm>(<js>"This page requires JavaScript to function properly."</js>),
62   * 		<jsm>p</jsm>(<js>"Please enable JavaScript and refresh the page."</js>),
63   * 		<jsm>a</jsm>(<js>"/help/javascript"</js>, <js>"Learn how to enable JavaScript"</js>)
64   * 	);
65   *
66   * 	<jc>// Noscript with form</jc>
67   * 	Noscript <jv>withForm</jv> = <jsm>noscript</jsm>(
68   * 		<jsm>p</jsm>(<js>"JavaScript is disabled. Please use the form below:"</js>),
69   * 		<jsm>form</jsm>().action(<js>"/submit"</js>).method(<js>"post"</js>).children(
70   * 			<jsm>input</jsm>(<js>"text"</js>).name(<js>"name"</js>).placeholder(<js>"Name"</js>),
71   * 			<jsm>input</jsm>(<js>"submit"</js>).value(<js>"Submit"</js>)
72   * 		)
73   * 	);
74   * </p>
75   *
76   * <h5 class='section'>See Also:</h5><ul>
77   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauBeanHtml5">juneau-bean-html5</a>
78   * </ul>
79   */
80  @Marshalled(typeName = "noscript")
81  public class Noscript extends HtmlElementMixed<Noscript> {
82  
83  	/**
84  	 * Creates an empty {@link Noscript} element.
85  	 */
86  	public Noscript() {}
87  
88  	/**
89  	 * Creates a {@link Noscript} element with the specified child nodes.
90  	 *
91  	 * @param children The child nodes. Must not be <jk>null</jk>.
92  	 */
93  	public Noscript(Object...children) {
94  		children(children);
95  	}
96  
97  }
