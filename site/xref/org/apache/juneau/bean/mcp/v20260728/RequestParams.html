17  package org.apache.juneau.bean.mcp.v20260728;
18  
19  import org.apache.juneau.commons.bean.*;
20  
21  /**
22   * CRTP base for MCP {@code 2026-07-28} request params carrying optional negotiation/trace metadata under the
23   * reserved {@code _meta} wire key.
24   *
25   * <p>
26   * Per the {@code 2026-07-28} schema, every v2 request places its metadata under {@code params._meta} rather than
27   * beside the JSON-RPC envelope. The four concrete params beans ({@link CallToolRequest}, {@link GetPromptRequest},
28   * {@link ReadResourceRequest}, and {@link CompleteRequest}) extend this base without gaining any wire-visible
29   * members beyond their own existing payload fields plus the inherited {@code _meta}. {@link RequestParamsOnly} is
30   * the concrete carrier for methods (list/ping/discovery) whose only common params member is {@code _meta}.
31   *
32   * <p>
33   * This base is a lossless wire carrier only: it performs no negotiation validation. Metadata shape and negotiation
34   * rules are owned by {@link RequestMeta} and the {@code 2026-07-28} adapter's dispatch validation.
35   *
36   * <p>
37   * The CRTP type parameter lets each concrete subclass's {@link #setMeta(RequestMeta)} return its own concrete type
38   * for fluent chaining, without duplicating the {@code _meta} accessor in every subclass.
39   *
40   * @param <T> The concrete subclass, for fluent-setter self-typing.
41   */
42  public abstract class RequestParams<T extends RequestParams<T>> {
43  
44  	private RequestMeta meta;
45  
46  	/**
47  	 * Per-request negotiation, extension, and trace-context metadata.
48  	 *
49  	 * @return The metadata, or {@code null} if not set.
50  	 */
51  	@BeanProp("_meta")
52  	public RequestMeta getMeta() {
53  		return meta;
54  	}
55  
56  	/**
57  	 * Sets the per-request metadata.
58  	 *
59  	 * @param value The new value.  Can be <jk>null</jk> to unset the property.
60  	 * @return This object (for method chaining).
61  	 */
62  	@BeanProp("_meta")
63  	@SuppressWarnings({
64  		"unchecked" // CRTP subclasses bind T to their own concrete type.
65  	})
66  	public T setMeta(RequestMeta value) {
67  		meta = value;
68  		return (T)this;
69  	}
70  }
