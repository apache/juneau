17  package org.apache.juneau.bean.mcp.v20260728;
18  
19  /**
20   * Shared SEP-2549 cache-hint base for the five cacheable {@code 2026-07-28} result carriers
21   * ({@code tools/list}, {@code prompts/list}, {@code resources/list}, {@code resources/templates/list}, and
22   * {@code resources/read}).
23   *
24   * <p>
25   * Extends {@link Result} so every cacheable result also inherits the required {@code resultType} discriminator
26   * and optional {@code _meta}, alongside two nullable, independent, top-level cache-hint properties that are
27   * peers of each subclass's own payload properties:
28   * <ul>
29   * 	<li>{@code ttlMs} - Freshness duration in milliseconds ({@code Cache-Control: max-age} semantics).
30   * 		{@code 0} means immediately stale.
31   * 	<li>{@code cacheScope} - {@link McpCacheScope#PUBLIC} or {@link McpCacheScope#PRIVATE}. An absent value is
32   * 		interpreted by consumers as {@code public}, but that interpretation is never materialized on the wire.
33   * </ul>
34   *
35   * <p>
36   * This base is a lossless wire carrier only: it performs no {@code ttlMs} validation and does not synthesize a
37   * default {@code cacheScope}. Wire beans must be able to parse and round-trip arbitrary protocol payloads
38   * (including a negative parsed {@code ttlMs}) independently of any server-side configuration policy. TTL
39   * validation for values the server itself emits is owned by the {@code 2026-07-28} adapter's cache
40   * configuration, not by this bean.
41   *
42   * <p>
43   * The CRTP type parameter lets each concrete subclass's cache setters (and the inherited {@link Result}
44   * setters) return its own concrete type for fluent chaining, without duplicating those accessors in every
45   * subclass.
46   *
47   * @param <T> The concrete subclass, for fluent-setter self-typing.
48   */
49  public abstract class CacheableResult<T extends CacheableResult<T>> extends Result<T> {
50  
51  	private Integer ttlMs;
52  	private McpCacheScope cacheScope;
53  
54  	/**
55  	 * Cache freshness duration in milliseconds.
56  	 *
57  	 * @return The TTL, or {@code null} if not set.
58  	 */
59  	public Integer getTtlMs() {
60  		return ttlMs;
61  	}
62  
63  	/**
64  	 * Sets the cache freshness duration in milliseconds.
65  	 *
66  	 * @param value The new value. Can be <jk>null</jk> to unset the property. {@code 0} is a valid, meaningful
67  	 * 	value meaning immediately stale.
68  	 * @return This object (for method chaining).
69  	 */
70  	@SuppressWarnings({
71  		"unchecked" // CRTP subclasses bind T to their own concrete type.
72  	})
73  	public T setTtlMs(Integer value) {
74  		ttlMs = value;
75  		return (T)this;
76  	}
77  
78  	/**
79  	 * Cache-hint authorization scope.
80  	 *
81  	 * @return The scope, or {@code null} if not set.
82  	 */
83  	public McpCacheScope getCacheScope() {
84  		return cacheScope;
85  	}
86  
87  	/**
88  	 * Sets the cache-hint authorization scope.
89  	 *
90  	 * @param value The new value. Can be <jk>null</jk> to unset the property.
91  	 * @return This object (for method chaining).
92  	 */
93  	@SuppressWarnings({
94  		"unchecked" // CRTP subclasses bind T to their own concrete type.
95  	})
96  	public T setCacheScope(McpCacheScope value) {
97  		cacheScope = value;
98  		return (T)this;
99  	}
100 }
