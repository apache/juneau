27  /**
28   * Opaque per-request MCP metadata carried under the JSON-RPC {@code params._meta} property.
29   *
30   * <p>
31   * Every v2 request is independently negotiated from its own {@code _meta}; no handshake or session state exists.
32   * Negotiation and identity keys use the exact {@code io.modelcontextprotocol/}-prefixed wire names mandated by the
33   * {@code 2026-07-28} schema; W3C trace-context keys ({@code traceparent} / {@code tracestate} / {@code baggage})
34   * are the schema's explicit exception and remain bare. Unrecognized metadata members round-trip losslessly, in
35   * insertion order, through the {@code "*"} dynamic-property triplet ({@link #extraKeys()}, {@link #get(String)},
36   * {@link #set(String, Object)}).
37   *
38   * <p>
39   * The exact wire-key constants below are the single source of truth for negotiation/identity/trace metadata keys;
40   * the {@code 2026-07-28} adapter, its validation messages, and bean/adapter tests consume these constants rather
41   * than repeating the literal strings.
42   */
43  @Marshalled
44  public class RequestMeta {
45  
46  	/** Exact wire key for the required negotiated protocol version. */
47  	public static final String KEY_PROTOCOL_VERSION = "io.modelcontextprotocol/protocolVersion";
48  
49  	/** Exact wire key for the optional client implementation identity. */
50  	public static final String KEY_CLIENT_INFO = "io.modelcontextprotocol/clientInfo";
51  
52  	/** Exact wire key for the required client capability advertisement. */
53  	public static final String KEY_CLIENT_CAPABILITIES = "io.modelcontextprotocol/clientCapabilities";
54  
55  	/** Exact wire key for the deprecated, accepted-but-ignored client log level. */
56  	public static final String KEY_LOG_LEVEL = "io.modelcontextprotocol/logLevel";
57  
58  	/** Bare (unprefixed) W3C trace-context parent key. */
59  	public static final String KEY_TRACEPARENT = "traceparent";
60  
61  	/** Bare (unprefixed) W3C trace-context state key. */
62  	public static final String KEY_TRACESTATE = "tracestate";
63  
64  	/** Bare (unprefixed) W3C baggage key. */
65  	public static final String KEY_BAGGAGE = "baggage";
66  
67  	/**
68  	 * Exact wire key carrying the JSON-RPC id of the {@code subscriptions/listen} request on every
69  	 * notification frame's {@code _meta} on that stream. The terminal {@link SubscriptionsListenResult}
70  	 * frame carries {@link ResultMeta} (not this type) instead but reuses the same wire key via its
71  	 * untyped {@code set(String, Object)} extension.
72  	 */
73  	public static final String KEY_SUBSCRIPTION_ID = "io.modelcontextprotocol/subscriptionId";

238 	/**
239 	 * Extension metadata member keyset.
240 	 *
241 	 * <p>
242 	 * Covers any metadata member other than the negotiation/identity/trace properties declared above. Preserves
243 	 * insertion order.
244 	 *
245 	 * @return All extension keys on this metadata object.  Never <jk>null</jk>.
246 	 */
247 	@BeanProp("*")
248 	public Set<String> extraKeys() {
249 		return extra == null ? Collections.emptySet() : u(extra.keySet());
250 	}
251 
252 	/**
253 	 * Extension metadata member getter.
254 	 *
255 	 * @param property The property name.  Must not be <jk>null</jk>.
256 	 * @return The property value, or <jk>null</jk> if the property does not exist or is not set.
257 	 */
258 	@BeanProp("*")
259 	public Object get(String property) {
260 		return extra == null ? null : extra.get(property);
261 	}
262 
263 	/**
264 	 * Extension metadata member setter.
265 	 *
266 	 * <p>
267 	 * The extension map is lazily initialized on first use and preserves insertion order across repeated calls.
268 	 *
269 	 * @param property The property name.  Must not be <jk>null</jk>.
270 	 * @param value The new value for the property.  Can be <jk>null</jk> (stored as <jk>null</jk>).
271 	 * @return This object (for method chaining).
272 	 */
273 	@BeanProp("*")
274 	public RequestMeta set(String property, Object value) {
275 		if (extra == null)
276 			extra = map();
277 		extra.put(property, value);
278 		return this;
279 	}
