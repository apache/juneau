17  package org.apache.juneau.bean.rfc7807.adapter;
18  
19  import org.apache.juneau.bean.rfc7807.*;
20  import org.apache.juneau.http.response.*;
21  
22  /**
23   * Static adapters that convert Juneau HTTP error types into
24   * <a class="doclink" href="https://www.rfc-editor.org/rfc/rfc7807">RFC 7807</a> {@link Problem} beans.
25   *
26   * <p>
27   * This adapter lives in {@code juneau-rest-common} (rather than the {@code juneau-bean-rfc7807} bean module) so the
28   * bean module stays free of any {@code juneau-rest-common} dependency. The package is co-located with the bean's
29   * package namespace so the cross-reference is discoverable from {@link Problem}'s javadoc.
30   *
31   * <h5 class='section'>Example:</h5>
32   * <p class='bjava'>
33   * 	<jk>try</jk> {
34   * 		...
35   * 	} <jk>catch</jk> (NotFound <jv>e</jv>) {
36   * 		Problem <jv>p</jv> = ProblemAdapters.<jsm>fromException</jsm>(<jv>e</jv>);
37   * 		<jc>// p.getStatus() == 404, p.getTitle() == "Not Found", p.getDetail() == e.getMessage()</jc>
38   * 	}
39   * </p>
40   *
41   * <h5 class='section'>Notes:</h5>
42   * <ul class='spaced-list'>
43   * 	<li>{@code type} is intentionally left {@code null}. RFC 7807 &sect;3.1 defines {@code about:blank} as the default
44   * 		when the field is absent; the bean serializes a {@code null} {@code type} as an omitted field rather than as
45   * 		{@code "type":"about:blank"} on the wire (preserves the absent-vs-explicit distinction).
46   * 	<li>{@code detail} is suppressed when the exception's message exactly matches the reason phrase &mdash; the bare
47   * 		<c><jk>new</jk> NotFound()</c> case where the message would just echo {@code "Not Found"}.
48   * </ul>
49   *
50   * <h5 class='section'>See Also:</h5><ul>
51   * 	<li class='jc'>{@link Problem}
52   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauBeanRfc7807">juneau-bean-rfc7807</a>
53   * </ul>
54   */
55  public final class ProblemAdapters {
56  
57  	private ProblemAdapters() {}
58  
59  	/**
60  	 * Builds a {@link Problem} from a {@link BasicHttpException} (or any subclass like {@code NotFound},
61  	 * {@code InternalServerError}, etc).
62  	 *
63  	 * <p>
64  	 * Mapping:
65  	 * <ul class='spaced-list'>
66  	 * 	<li>{@code status} &larr; {@link BasicHttpException#getStatusCode()}
67  	 * 	<li>{@code title} &larr; {@link org.apache.juneau.http.HttpStatusLine#getReasonPhrase()} (may be {@code null})
68  	 * 	<li>{@code detail} &larr; {@link BasicHttpException#getMessage()}, but only when it differs from the reason
69  	 * 		phrase; otherwise {@code null} so the bare <c><jk>new</jk> NotFound()</c> case omits a redundant
70  	 * 		{@code "detail":"Not Found"} field.
71  	 * 	<li>{@code type}, {@code instance} &larr; never set by this adapter (callers add them via the fluent setters
72  	 * 		on the returned bean if a {@code type} URI is appropriate).
73  	 * </ul>
74  	 *
75  	 * @param e The exception to adapt. May be <jk>null</jk>.
76  	 * @return A new {@link Problem} populated from the exception, or <jk>null</jk> if {@code e} is <jk>null</jk>.
77  	 */
78  	public static Problem fromException(BasicHttpException e) {
79  		if (e == null)
80  			return null;
81  		var statusLine = e.getStatusLine();
82  		var reasonPhrase = statusLine == null ? null : statusLine.getReasonPhrase(); // HTT — statusLine is always set in BasicHttpException constructors
83  		var message = e.getMessage();
84  		var detail = (message == null || message.equals(reasonPhrase)) ? null : message; // HTT — getMessage() always returns non-null (falls back to reasonPhrase)
85  		return new Problem()
86  			.setStatus(e.getStatusCode())
87  			.setTitle(reasonPhrase)
88  			.setDetail(detail);
89  	}
90  }
