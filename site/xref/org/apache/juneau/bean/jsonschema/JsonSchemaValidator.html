33  /**
34   * Validates values against a {@link JsonSchema} bean per the JSON Schema Draft 2020-12 specification.
35   *
36   * <p>
37   * This is the {@code juneau-bean-jsonschema} implementation of the commons-side
38   * {@link PropertyValidator} SPI.  It is registered as a {@link PropertyValidatorFactory} via
39   * {@link ServiceLoader ServiceLoader} so that marshalling-side code (e.g.
40   * {@code MarshalledPropertyPostProcessor}) can install schema validation closures on bean properties without taking a
41   * compile-time dependency on this module.
42   *
43   * <h5 class='section'>Supported keywords (v1):</h5>
44   * <ul class='spaced-list'>
45   * 	<li>Any type: {@code type}, {@code enum}, {@code const}.
46   * 	<li>Numeric: {@code minimum}, {@code maximum}, {@code exclusiveMinimum}, {@code exclusiveMaximum},
47   * 		{@code multipleOf} (Draft 2020-12 numeric form for the exclusive bounds).
48   * 	<li>String: {@code minLength}, {@code maxLength}, {@code pattern}.
49   * 	<li>Array: {@code minItems}, {@code maxItems}, {@code uniqueItems}, {@code items} (recursive — single-schema
50   * 		form only; tuple form via {@link JsonSchemaArray} is not yet validated).
51   * 	<li>Object: {@code minProperties}, {@code maxProperties}, {@code required}, {@code properties} (recursive on
52   * 		{@link Map}-shaped values).
53   * </ul>
54   *
55   * <h5 class='section'>Deferred keywords:</h5>
56   * <ul class='spaced-list'>
57   * 	<li>{@code format} — informational only by default; validators may treat it as advisory.
58   * 	<li>{@code patternProperties}, {@code additionalProperties}, {@code prefixItems}, {@code contains},
59   * 		{@code minContains}, {@code maxContains}, {@code dependentRequired}, {@code dependentSchemas}.
60   * 	<li>Composition keywords: {@code allOf}, {@code anyOf}, {@code oneOf}, {@code not}, {@code if}/{@code then}/{@code else}.
61   * 	<li>References: {@code $ref}, {@code $defs} (the bean carries them; the validator does not resolve them yet).
62   * </ul>
63   *
64   * <h5 class='section'>Descriptive keywords (always ignored):</h5>
65   * <ul class='spaced-list'>
66   * 	<li>{@code title}, {@code summary}, {@code description}, {@code examples}, {@code $comment}, {@code deprecated},
67   * 		{@code readOnly}, {@code writeOnly}.
68   * </ul>
69   *
70   * <h5 class='section'>Java-value mapping:</h5>
71   * <p>
72   * Java values are matched against JSON Schema {@code type} keywords as follows:
73   * <ul class='spaced-list'>
74   * 	<li>{@code string} — {@link CharSequence} or {@link Character}.
75   * 	<li>{@code integer} — integral {@link Number} (Byte, Short, Integer, Long, BigInteger) or a floating-point number
76   * 		whose value has no fractional part.
77   * 	<li>{@code number} — any {@link Number}.
78   * 	<li>{@code boolean} — {@link Boolean}.
79   * 	<li>{@code array} — {@link Collection} or Java array.
80   * 	<li>{@code object} — {@link Map} or any other reference type (treated as a bean — the marshalling layer installs
81   * 		separate validators on the bean's own {@code @Schema}-annotated properties).
82   * 	<li>{@code null} — {@code null}.
83   * 	<li>{@code any} — always matches.
84   * </ul>
85   *
86   * <h5 class='section'>Nested validation:</h5>
87   * <p>
88   * For property-level validation, the validator recurses into:
89   * <ul class='spaced-list'>
90   * 	<li>{@code items} — every element of an array / collection is validated against the items schema.
91   * 	<li>{@code properties} on a {@link Map} value — each declared sub-property is validated against its sub-schema.
92   * </ul>
93   *
94   * <p>
95   * The validator does <b>not</b> recurse into Java beans via reflection.  Inner {@code @Schema} annotations on a
96   * nested bean's own properties are validated independently by the marshalling layer's per-property install hooks.
97   *
98   * <h5 class='section'>Thread safety:</h5>
99   * Instances are immutable and safe to share across threads.  The compiled {@link Pattern} for the {@code pattern}
100  * keyword is cached on construction.
101  *
102  * <h5 class='section'>See Also:</h5><ul>
103  * 	<li class='link'><a href="https://json-schema.org/draft/2020-12/json-schema-validation.html">JSON Schema 2020-12 Validation</a>
104  * </ul>
105  *
106  * @see PropertyValidator
107  * @see JsonSchemaPropertyValidatorFactory
108  * @since 10.0.0
109  */
110 public final class JsonSchemaValidator implements PropertyValidator {
111 
112 	private final JsonSchema<?> schema;
113 	private final Pattern patternCache;
114 
115 	/**
116 	 * Builds a validator from a {@link JsonSchema} bean.
117 	 *
118 	 * @param schema The schema to validate against.  Must not be <jk>null</jk>, or an {@link IllegalArgumentException} is thrown.
119 	 * @return A new validator.
120 	 */
121 	public static JsonSchemaValidator of(JsonSchema<?> schema) {
122 		assertArgNotNull("schema", schema);
123 		return new JsonSchemaValidator(schema);
124 	}
125 
126 	/**
127 	 * Builds a validator from a JSON-Schema-shaped {@link JsonMap}.
128 	 *
129 	 * <p>
130 	 * Equivalent to {@code of(JsonSchemaBeanGenerator.toBean(schemaMap))}.
131 	 *
132 	 * @param schemaMap The schema map.  Must not be <jk>null</jk>, or an {@link IllegalArgumentException} is thrown.
133 	 * @return A new validator.
134 	 */
135 	public static JsonSchemaValidator of(JsonMap schemaMap) {
136 		assertArgNotNull("schemaMap", schemaMap);
137 		return of(JsonSchemaBeanGenerator.toBean(schemaMap));
138 	}

140 	private JsonSchemaValidator(JsonSchema<?> schema) {
141 		this.schema = schema;
142 		var p = schema.getPattern();
143 		this.patternCache = p == null ? null : Pattern.compile(p);
144 	}
145 
146 	/**
147 	 * Returns the schema this validator was built from.
148 	 *
149 	 * @return The schema.
150 	 */
151 	@SuppressWarnings({
152 		"java:S1452" // Self-bounded (CRTP) generic: the validated schema's concrete SELF subtype is not known here, so JsonSchema<?> is the only sound return type.
153 	})
154 	public JsonSchema<?> getSchema() {
155 		return schema;
156 	}
157 
158 	@Override
159 	public void validate(Object value) throws SchemaValidationException {
160 		validateAgainst(schema, value, patternCache);
161 	}
162 
163 	@SuppressWarnings({
164 		"java:S3776"  // Cognitive complexity — dispatch table over the JSON Schema keyword set.
165 	})
166 	private static void validateAgainst(JsonSchema<?> s, Object value, Pattern patternOverride) {
167 		validateEnum(s, value);
168 		validateConst(s, value);
169 
170 		if (value == null)
171 			return;
172 
173 		validateType(s, value);
174 
175 		if (value instanceof CharSequence value2) {
176 			validateString(s, value2.toString(), patternOverride);
177 		} else if (value instanceof Character value2) {
178 			validateString(s, value2.toString(), patternOverride);
179 		} else if (value instanceof Number value2) {
180 			validateNumber(s, value2);
181 		} else if (value instanceof Collection<?> value2) {
182 			validateArray(s, value2);
183 		} else if (value.getClass().isArray()) {
184 			validateArray(s, arrayAsList(value));
185 		} else if (value instanceof Map<?,?> value2) {
186 			validateObject(s, value2);
187 		}
188 		// Java beans: per-property @Schema validators are installed independently on the bean side.
189 	}

191 	// =================================================================================================================
192 	// Type / enum / const
193 	// =================================================================================================================
194 
195 	private static void validateEnum(JsonSchema<?> s, Object value) {
196 		var enums = s.getEnum();
197 		if (ie(enums))
198 			return;
199 		for (var e : enums) {
200 			if (jsonEquals(e, value))
201 				return;
202 		}
203 		throw new SchemaValidationException("Value '%s' does not match one of the allowed enum values.", value);
204 	}
205 
206 	private static void validateConst(JsonSchema<?> s, Object value) {
207 		var c = s.getConst();
208 		if (c == null)
209 			return;  // const=null is indistinguishable from "not set" in the bean; treat as unset.
210 		if (! jsonEquals(c, value))
211 			throw new SchemaValidationException("Value '%s' does not match required const '%s'.", value, c);
212 	}
213 
214 	private static void validateType(JsonSchema<?> s, Object value) {
215 		var single = s.getTypeAsJsonType();
216 		if (nn(single)) {
217 			if (! matchesType(single, value))
218 				throw new SchemaValidationException("Value of type '%s' does not match expected schema type '%s'.", typeName(value), single);
219 			return;
220 		}
221 		var arr = s.getTypeAsJsonTypeArray();
222 		if (ie(arr))
223 			return;
224 		for (var t : arr) {
225 			if (matchesType(t, value))
226 				return;
227 		}
228 		throw new SchemaValidationException("Value of type '%s' does not match any expected schema types '%s'.", typeName(value), arr);
229 	}
230 
231 	@SuppressWarnings({
232 		"java:S1142"  // Multiple returns — switch arms each return their own predicate result.
233 	})
234 	private static boolean matchesType(JsonType t, Object value) {
235 		switch (t) {
236 			case STRING:
237 				return value instanceof CharSequence || value instanceof Character;
238 			case INTEGER:
239 				return isIntegralValue(value);
240 			case NUMBER:
241 				return value instanceof Number;
242 			case BOOLEAN:
243 				return value instanceof Boolean;
244 			case ARRAY:
245 				return value instanceof Collection<?> || (nn(value) && value.getClass().isArray());
246 			case OBJECT:
247 				return value instanceof Map<?,?> || isBeanLike(value);
248 			case NULL:
249 				return value == null;
250 			case ANY:
251 				return true;
252 			default:
253 				return true;
254 		}
255 	}
256 
257 	private static boolean isIntegralValue(Object value) {
258 		if (value instanceof Byte || value instanceof Short || value instanceof Integer || value instanceof Long || value instanceof BigInteger)
259 			return true;
260 		if (value instanceof Number value2) {
261 			try {
262 				var bd = toBigDecimal(value2);
263 				return bd.stripTrailingZeros().scale() <= 0;
264 			} catch (@SuppressWarnings("unused") NumberFormatException e) {
265 				return false;
266 			}
267 		}
268 		return false;
269 	}
270 
271 	private static boolean isBeanLike(Object value) {
272 		if (value == null)
273 			return false;
274 		var c = value.getClass();
275 		return ! (value instanceof CharSequence || value instanceof Character || value instanceof Number
276 			|| value instanceof Boolean || value instanceof Collection<?> || c.isArray());
277 	}
278 
279 	private static String typeName(Object value) {
280 		return value == null ? "null" : cns(value);
281 	}

284 	// String
285 	// =================================================================================================================
286 
287 	private static void validateString(JsonSchema<?> s, String value, Pattern patternOverride) {
288 		var minL = s.getMinLength();
289 		var maxL = s.getMaxLength();
290 		var len = value.codePointCount(0, value.length());
291 		if (nn(minL) && len < minL)
292 			throw new SchemaValidationException("String length %s is less than minLength %s.", len, minL);
293 		if (nn(maxL) && len > maxL)
294 			throw new SchemaValidationException("String length %s exceeds maxLength %s.", len, maxL);
295 		var pattern = patternOverride;
296 		if (pattern == null && nn(s.getPattern()))
297 			pattern = Pattern.compile(s.getPattern());
298 		if (pattern != null && ! pattern.matcher(value).find())
299 			throw new SchemaValidationException("Value '%s' does not match pattern '%s'.", value, s.getPattern());
300 	}

303 	// Number
304 	// =================================================================================================================
305 
306 	@SuppressWarnings({
307 		"java:S3776"  // Numeric keyword fan-out is intentionally flat; each branch maps to one Draft 2020-12 keyword.
308 	})
309 	private static void validateNumber(JsonSchema<?> s, Number value) {
310 		BigDecimal bd;
311 		try {
312 			bd = toBigDecimal(value);
313 		} catch (@SuppressWarnings("unused") NumberFormatException e) {
314 			throw new SchemaValidationException("Value '%s' is not a finite number.", value);
315 		}
316 
317 		if (nn(s.getMinimum()) && bd.compareTo(toBigDecimal(s.getMinimum())) < 0)
318 			throw new SchemaValidationException("Value %s is less than minimum %s.", bd, s.getMinimum());
319 		if (nn(s.getMaximum()) && bd.compareTo(toBigDecimal(s.getMaximum())) > 0)
320 			throw new SchemaValidationException("Value %s exceeds maximum %s.", bd, s.getMaximum());
321 		if (nn(s.getExclusiveMinimum()) && bd.compareTo(toBigDecimal(s.getExclusiveMinimum())) <= 0)
322 			throw new SchemaValidationException("Value %s is not greater than exclusive minimum %s.", bd, s.getExclusiveMinimum());
323 		if (nn(s.getExclusiveMaximum()) && bd.compareTo(toBigDecimal(s.getExclusiveMaximum())) >= 0)
324 			throw new SchemaValidationException("Value %s is not less than exclusive maximum %s.", bd, s.getExclusiveMaximum());
325 		if (nn(s.getMultipleOf())) {
326 			var mof = toBigDecimal(s.getMultipleOf());
327 			if (mof.signum() != 0 && bd.remainder(mof).compareTo(BigDecimal.ZERO) != 0)
328 				throw new SchemaValidationException("Value %s is not a multiple of %s.", bd, mof);
329 		}
330 	}

334 	// =================================================================================================================

336 	private static void validateArray(JsonSchema<?> s, Collection<?> value) {
337 		var min = s.getMinItems();
338 		var max = s.getMaxItems();
339 		var size = value.size();
340 		if (nn(min) && size < min)
341 			throw new SchemaValidationException("Array size %s is less than minItems %s.", size, min);
342 		if (nn(max) && size > max)
343 			throw new SchemaValidationException("Array size %s exceeds maxItems %s.", size, max);
344 		if (isTrue(s.getUniqueItems()) && hasDuplicates(value))
345 			throw new SchemaValidationException("Array contains duplicate items but uniqueItems is true.");
346 		var items = s.getItemsAsSchema();
347 		if (nn(items)) {
348 			for (var item : value)
349 				validateAgainst(items, item, null);
350 		}
351 	}
352 
353 	private static List<Object> arrayAsList(Object array) {
354 		var len = Array.getLength(array);
355 		var out = new ArrayList<>(len);
356 		for (var i = 0; i < len; i++)
357 			out.add(Array.get(array, i));
358 		return out;
359 	}
360 
361 	private static boolean hasDuplicates(Collection<?> c) {
362 		if (c.size() < 2)
363 			return false;
364 		var seen = new ArrayList<>(c.size());
365 		for (var e : c) {
366 			for (var s : seen) {
367 				if (jsonEquals(e, s))
368 					return true;
369 			}
370 			seen.add(e);
371 		}
372 		return false;
373 	}

379 	@SuppressWarnings({
380 		"java:S3776"  // Cognitive complexity — flat fan-out over the Draft 2020-12 object keywords (min/maxProperties, required, properties).
381 	})
382 	private static void validateObject(JsonSchema<?> s, Map<?,?> value) {
383 		var minP = s.getMinProperties();
384 		var maxP = s.getMaxProperties();
385 		var size = value.size();
386 		if (nn(minP) && size < minP)
387 			throw new SchemaValidationException("Object property count %s is less than minProperties %s.", size, minP);
388 		if (nn(maxP) && size > maxP)
389 			throw new SchemaValidationException("Object property count %s exceeds maxProperties %s.", size, maxP);
390 		var required = s.getRequired();
391 		if (nn(required)) {
392 			for (var key : required) {
393 				if (! value.containsKey(key))
394 					throw new SchemaValidationException("Required property '%s' is missing.", key);
395 			}
396 		}
397 		var props = s.getProperties();
398 		if (nn(props)) {
399 			for (var entry : props.entrySet()) {
400 				var name = entry.getKey();
401 				var sub = entry.getValue();
402 				if (value.containsKey(name))
403 					validateAgainst(sub, value.get(name), null);
404 			}
405 		}
406 	}

412 	private static BigDecimal toBigDecimal(Number n) {
413 		if (n instanceof BigDecimal n2)
414 			return n2;
415 		if (n instanceof BigInteger n2)
416 			return new BigDecimal(n2);
417 		if (n instanceof Long || n instanceof Integer || n instanceof Short || n instanceof Byte)
418 			return BigDecimal.valueOf(n.longValue());
419 		return new BigDecimal(n.toString());
420 	}
421 
422 	@SuppressWarnings({
423 		"java:S3776"  // Equality fan-out — mirrors the type matrix exposed by deserialized JSON Schema bean values.
424 	})
425 	private static boolean jsonEquals(Object a, Object b) {
426 		if (a == b)
427 			return true;
428 		if (a == null || b == null)
429 			return false;
430 		if (a instanceof Number a2 && b instanceof Number b2)
431 			return toBigDecimal(a2).compareTo(toBigDecimal(b2)) == 0;
432 		if (a instanceof Number a2 && b instanceof CharSequence b2) {
433 			var bd = tryAsBigDecimal(b2.toString());
434 			return nn(bd) && toBigDecimal(a2).compareTo(bd) == 0;
435 		}
436 		if (b instanceof Number b2 && a instanceof CharSequence a2) {
437 			var bd = tryAsBigDecimal(a2.toString());
438 			return nn(bd) && toBigDecimal(b2).compareTo(bd) == 0;
439 		}
440 		if (a instanceof CharSequence a2 && b instanceof CharSequence b2)
441 			return a2.toString().equals(b2.toString());
442 		if (a.equals(b))
443 			return true;
444 		try {
445 			return Json.of(a).equals(Json.of(b));
446 		} catch (@SuppressWarnings("unused") Exception e) {
447 			return false;
448 		}
449 	}
450 
451 	private static BigDecimal tryAsBigDecimal(String s) {
452 		try {
453 			return new BigDecimal(s);
454 		} catch (@SuppressWarnings("unused") NumberFormatException e) {
455 			return null;
456 		}
