29  /**
30   * Represents a HAL Resource Object as defined by
31   * <a href="https://datatracker.ietf.org/doc/html/draft-kelly-json-hal-08#section-4">draft-kelly-json-hal-08 &#167; 4</a>.
32   *
33   * <p>
34   * A HAL Resource carries three pieces of state:
35   * <ul class='spaced-list'>
36   *   <li>The reserved {@code _links} member - a map of relation name to either a single {@link HalLink}
37   *     or a {@link HalLinkArray}.
38   *   <li>The reserved {@code _embedded} member - a map of relation name to either a single {@link HalResource}
39   *     or a {@link HalResourceArray} (sub-resources). HAL embeddings are conceptually a <i>tree</i>; cyclic
40   *     {@code _embedded} graphs are rejected by Juneau's {@code BeanTraverseContext} by default.
41   *   <li>Arbitrary payload fields, set via the dynamic {@link #set(String,Object)} method (uses the
42   *     {@code @BeanProp("*")} extra-properties triplet, the same pattern used by {@code OpenApiElement} and
43   *     {@code SwaggerElement}).
44   * </ul>
45   *
46   * <h5 class='section'>Example:</h5>
47   * <p class='bjava'>
48   * 	HalResource <jv>order</jv> = <jk>new</jk> HalResource()
49   * 		.addLink(<js>"self"</js>, <jk>new</jk> HalLink().setHref(<js>"/orders/123"</js>))
50   * 		.addLinks(<js>"curies"</js>,
51   * 			<jk>new</jk> HalLink().setName(<js>"acme"</js>).setHref(<js>"https://acme.example/{rel}"</js>).setTemplated(<jk>true</jk>))
52   * 		.set(<js>"total"</js>, 99.50);
53   *
54   * 	String <jv>json</jv> = JsonSerializer.<jsf>DEFAULT</jsf>.write(<jv>order</jv>);
55   * </p>
56   *
57   * <h5 class='section'>See Also:</h5><ul>
58   *   <li class='link'><a href="https://stateless.group/hal_specification.html">HAL Specification</a>
59   *   <li class='link'><a href="https://datatracker.ietf.org/doc/html/draft-kelly-json-hal-08">draft-kelly-json-hal-08</a>
60   *   <li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauBeanHal">juneau-bean-hal</a>
61   * </ul>
62   */
63  @Marshalled
64  @SuppressWarnings({
65  	"java:S116", // Field names mirror HAL spec.
66  	"java:S115" // Constant names intentionally mirror argument identifiers.
67  })
68  public class HalResource {
69  
70  	private static final String ARG_property = "property";
71  	private static final String ARG_relation = "relation";
72  
73  	private Map<String,Object> _links;
74  	private Map<String,Object> _embedded;
75  	private Map<String,Object> properties;
76  
77  	/**
78  	 * Default constructor.
79  	 */
80  	public HalResource() {
81  		/* intentionally empty — required public no-arg constructor for bean creation/decoding */
82  	}
83  
84  	/**
85  	 * Bean property getter:  <property>_links</property>.
86  	 *
87  	 * <p>
88  	 * The reserved {@code _links} member. Each map value is either a single {@link HalLink} (rendered as a JSON
89  	 * object on the wire) or a {@link HalLinkArray} (rendered as a JSON array).
90  	 *
91  	 * @return The value of the <property>_links</property> property, or <jk>null</jk> if it is not set.
92  	 */
93  	@BeanProp("_links")
94  	@Swap(HalLinkOrArraySwap.class)
95  	public Map<String,Object> getLinks() {
96  		return u(_links);
97  	}
98  
99  	/**
100 	 * Bean property setter:  <property>_links</property>.
101 	 *
102 	 * @param value The new value for this property. Can be <jk>null</jk> to unset.
103 	 * @return This object.
104 	 */
105 	@BeanProp("_links")
106 	@Swap(HalLinkOrArraySwap.class)
107 	public HalResource setLinks(Map<String,Object> value) {
108 		_links = value;
109 		return this;
110 	}
111 
112 	/**
113 	 * Convenience method that adds a single link under the given relation.
114 	 *
115 	 * <p>
116 	 * If a value already exists for {@code relation}, it is replaced with the supplied single link. To accumulate
117 	 * multiple links under the same relation use {@link #addLinks(String, HalLink...)} instead.
118 	 *
119 	 * @param relation The link relation (e.g. {@code "self"}, {@code "next"}, {@code "curies"}).  Must not be <jk>null</jk>, or an {@link IllegalArgumentException} is thrown.
120 	 * @param value The link to set.  Can be <jk>null</jk> (stored as a <jk>null</jk> entry under the relation).
121 	 * @return This object.
122 	 */
123 	public HalResource addLink(String relation, HalLink value) {
124 		assertArgNotNull(ARG_relation, relation);
125 		if (_links == null)
126 			_links = map();
127 		_links.put(relation, value);
128 		return this;
129 	}
130 
131 	/**
132 	 * Convenience method that appends one or more links to a {@link HalLinkArray} stored under {@code relation}.
133 	 *
134 	 * <p>
135 	 * If the current value at {@code relation} is a single {@link HalLink}, it is promoted to a {@link HalLinkArray}
136 	 * containing the original link and the appended ones.
137 	 *
138 	 * @param relation The link relation.  Must not be <jk>null</jk>, or an {@link IllegalArgumentException} is thrown.
139 	 * @param values The links to append.
140 	 * @return This object.
141 	 */
142 	public HalResource addLinks(String relation, HalLink...values) {
143 		assertArgNotNull(ARG_relation, relation);
144 		if (_links == null)
145 			_links = map();
146 		var existing = _links.get(relation);
147 		HalLinkArray arr;
148 		if (existing instanceof HalLinkArray existing2) {
149 			arr = existing2;
150 		} else if (existing instanceof HalLink existing2) {
151 			arr = new HalLinkArray(existing2);
152 			_links.put(relation, arr);
153 		} else {
154 			arr = new HalLinkArray();
155 			_links.put(relation, arr);
156 		}
157 		arr.addAll(values);
158 		return this;
159 	}
160 
161 	/**
162 	 * Bean property getter:  <property>_embedded</property>.
163 	 *
164 	 * <p>
165 	 * The reserved {@code _embedded} member. Each map value is either a single {@link HalResource} or a
166 	 * {@link HalResourceArray}.
167 	 *
168 	 * @return The value of the <property>_embedded</property> property, or <jk>null</jk> if it is not set.
169 	 */
170 	@BeanProp("_embedded")
171 	@Swap(HalResourceOrArraySwap.class)
172 	public Map<String,Object> getEmbedded() {
173 		return u(_embedded);
174 	}
175 
176 	/**
177 	 * Bean property setter:  <property>_embedded</property>.
178 	 *
179 	 * <p>
180 	 * Note: HAL embeddings are conceptually a tree, not a graph. Juneau's {@code BeanTraverseContext} throws on
181 	 * cyclic references by default, which matches the spec.
182 	 *
183 	 * @param value The new value for this property. Can be <jk>null</jk> to unset.
184 	 * @return This object.
185 	 */
186 	@BeanProp("_embedded")
187 	@Swap(HalResourceOrArraySwap.class)
188 	public HalResource setEmbedded(Map<String,Object> value) {
189 		_embedded = value;
190 		return this;
191 	}
192 
193 	/**
194 	 * Convenience method that embeds a single sub-resource under the given relation.
195 	 *
196 	 * @param relation The relation name.  Must not be <jk>null</jk>, or an {@link IllegalArgumentException} is thrown.
197 	 * @param value The sub-resource.  Can be <jk>null</jk> (stored as a <jk>null</jk> entry under the relation).
198 	 * @return This object.
199 	 */
200 	public HalResource addEmbedded(String relation, HalResource value) {
201 		assertArgNotNull(ARG_relation, relation);
202 		if (_embedded == null)
203 			_embedded = map();
204 		_embedded.put(relation, value);
205 		return this;
206 	}
207 
208 	/**
209 	 * Convenience method that appends one or more sub-resources to a {@link HalResourceArray} stored under
210 	 * {@code relation}.
211 	 *
212 	 * @param relation The relation name.  Must not be <jk>null</jk>, or an {@link IllegalArgumentException} is thrown.
213 	 * @param values The sub-resources to append.
214 	 * @return This object.
215 	 */
216 	public HalResource addEmbedded(String relation, HalResource...values) {
217 		assertArgNotNull(ARG_relation, relation);
218 		if (_embedded == null)
219 			_embedded = map();
220 		var existing = _embedded.get(relation);
221 		HalResourceArray arr;
222 		if (existing instanceof HalResourceArray existing2) {
223 			arr = existing2;
224 		} else if (existing instanceof HalResource existing2) {
225 			arr = new HalResourceArray(existing2);
226 			_embedded.put(relation, arr);
227 		} else {
228 			arr = new HalResourceArray();
229 			_embedded.put(relation, arr);
230 		}
231 		arr.addAll(values);
232 		return this;
233 	}
234 
235 	/**
236 	 * Generic property keyset.
237 	 *
238 	 * @return All the payload (non-{@code _links}, non-{@code _embedded}) property names on this resource. Never <jk>null</jk>.
239 	 */
240 	@BeanProp("*")
241 	public Set<String> extraKeys() {
242 		return properties == null ? Collections.emptySet() : u(properties.keySet());
243 	}
244 
245 	/**
246 	 * Generic property getter.
247 	 *
248 	 * @param property The property name. Must not be <jk>null</jk>, or an {@link IllegalArgumentException} is thrown.
249 	 * @return The property value, or <jk>null</jk> if the property is not set.
250 	 */
251 	@BeanProp("*")
252 	public Object get(String property) {
253 		assertArgNotNull(ARG_property, property);
254 		return o(properties).map(x -> x.get(property)).orElse(null);
255 	}
256 
257 	/**
258 	 * Generic property setter.
259 	 *
260 	 * <p>
261 	 * Use this to set arbitrary payload fields alongside {@code _links} and {@code _embedded}.
262 	 *
263 	 * @param property The property name. Must not be <jk>null</jk>, or an {@link IllegalArgumentException} is thrown.
264 	 * @param value The new value for the property. Can be <jk>null</jk> (stored as <jk>null</jk>).
265 	 * @return This object.
266 	 */
267 	@BeanProp("*")
268 	public HalResource set(String property, Object value) {
269 		assertArgNotNull(ARG_property, property);
270 		if (properties == null)
271 			properties = map();
272 		properties.put(property, value);
273 		return this;
274 	}
275 
276 	@Override /* Overridden from Object */
277 	public String toString() {
278 		return Json.of(this);
279 	}
