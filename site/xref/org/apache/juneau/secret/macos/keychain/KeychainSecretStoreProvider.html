17  package org.apache.juneau.secret.macos.keychain;
18  
19  import org.apache.juneau.commons.secret.*;
20  
21  /**
22   * Provider that contributes a {@link KeychainSecretStore} via {@code ServiceLoader}, discoverable through
23   * {@link SecretStores#fromServiceLoader()} when this module is on the classpath and the host is macOS.
24   *
25   * <p>
26   * The service name defaults to {@value #DEFAULT_SERVICE} but can be overridden via the
27   * {@value #SERVICE_PROPERTY} system property; the fail mode defaults to {@link FailMode#FAIL_CLOSED}.  On a
28   * non-macOS host (or when the keychain CLI is absent) {@link #create()} returns <jk>null</jk> so discovery skips it.
29   *
30   * @since 10.0.0
31   */
32  public class KeychainSecretStoreProvider implements SecretStoreProvider {
33  
34  	/** System property overriding the keychain service name used by the discovered store. */
35  	public static final String SERVICE_PROPERTY = "juneau.secret.macos.keychain.service";
36  
37  	/** Default keychain service name when {@link #SERVICE_PROPERTY} is unset. */
38  	public static final String DEFAULT_SERVICE = "org.apache.juneau";
39  
40  	@Override /* SecretStoreProvider */
41  	public SecretStore create() {
42  		if (! System.getProperty("os.name", "").toLowerCase().contains("mac"))
43  			return null;
44  		return new KeychainSecretStore(System.getProperty(SERVICE_PROPERTY, DEFAULT_SERVICE));
45  	}
46  
47  	@Override /* SecretStoreProvider */
48  	public int order() {
49  		return 10;
50  	}
51  }
