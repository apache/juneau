29  /**
30   * A {@link SecretStore} backed by the macOS <c>security</c> keychain CLI (generic passwords).
31   *
32   * <p>
33   * Each operation shells out to <c>/usr/bin/security</c>:
34   * <ul>
35   * 	<li>{@link #store} &rarr; <c>add-generic-password -U</c> (create or update).
36   * 	<li>{@link #find} &rarr; <c>find-generic-password -g</c> (print the password to stderr).
37   * 	<li>{@link #exists} &rarr; <c>find-generic-password</c> (presence only, without the password).
38   * 	<li>{@link #delete} &rarr; <c>delete-generic-password</c>.
39   * </ul>
40   * Items are namespaced by a caller-supplied <i>service</i> name (the keychain <c>-s</c> attribute) with the secret
41   * key as the <i>account</i> (<c>-a</c>).  Choose a stable, collision-resistant service name (for example a
42   * reverse-DNS string) so unrelated consumers do not clobber each other's entries.
43   *
44   * <p>
45   * <b>Platform.</b> Only macOS is supported; on any other OS, or when <c>/usr/bin/security</c> is unavailable, calls
46   * are treated as a backend failure and resolved per the configured {@link FailMode}.
47   *
48   * <p>
49   * <b>Backend-unavailable behavior.</b> A key that is simply not present is a normal <i>absent</i> result, never a
50   * failure.  A genuine failure &mdash; the tool missing, a non-zero exit that is not "item not found", or a timeout
51   * &mdash; is resolved by the {@link FailMode} chosen at construction: {@link FailMode#FAIL_OPEN} degrades to
52   * absent/no-op, while {@link FailMode#FAIL_CLOSED} (the default) throws.
53   *
54   * <p>
55   * <b>Secret handling.</b> {@link #store} never places the secret on the {@code security} process's argv, and never
56   * relies on the CLI's password prompt (a bare <c>-w</c> makes {@code security} read from the controlling terminal,
57   * <c>/dev/tty</c>, whenever the JVM has one &mdash; not from stdin &mdash; so it would hang).  Instead it runs
58   * <c>security -i</c> (interactive mode, which reads commands from stdin) and writes a single command line to stdin:
59   * <c>add-generic-password -U -a "<i>key</i>" -s "<i>service</i>" -w "<i>secret</i>"</c>.  The secret is therefore
60   * never visible in the process table (e.g. to {@code ps}), and no terminal is ever read.  Each of the three values is
61   * double-quoted, with only <c>"</c> and <c>\</c> escaped by a preceding backslash (verified against {@code security -i}
62   * for spaces, tabs, single and double quotes, backslashes, shell metacharacters, and non-ASCII text).  Neither
63   * {@link #store} nor {@link #find} materializes the secret as a {@link String}: {@code store} encodes the
64   * {@code char[]} directly to the stdin byte payload, and {@code find} decodes the retrieved bytes into a
65   * {@code char[]} (zeroing the intermediate buffers).  The value is never logged or {@code toString()}'d by this class.  Because the command is
66   * newline-terminated, a key, service, or secret containing {@code '\n'} or {@code '\r'} cannot be delivered this way
67   * (it would end or inject a command) and is rejected by {@link #store} before any process is started.
68   *
69   * <p>
70   * <b>Reading secrets.</b> {@link #find} deliberately does not use <c>-w</c>: for a secret holding a control character
71   * (such as a tab) or any non-ASCII byte, <c>-w</c> prints a bare hex dump of the bytes instead of the text, which
72   * cannot be told apart from a secret that merely looks like hex (e.g. <c>abcd</c>).  The <c>-g</c> flag instead writes
73   * one unambiguous line to stderr: <c>password: "text"</c> for a plain printable secret, <c>password: 0x<i>HEX</i>
74   * "<i>escaped</i>"</c> for anything else (including a secret containing <c>"</c> or <c>\</c>), or a bare
75   * <c>password: </c> for an empty secret.  The hex form is decoded to bytes and then as UTF-8.  The stdout of
76   * <c>-g</c> holds only the non-secret item attributes and is ignored.
77   *
78   * @since 10.0.0
79   */
80  public class KeychainSecretStore implements SecretStore {

152 	@Override /* SecretStore */
153 	public void store(String key, char[] secret) {
154 		reqnn("key", key);
155 		reqnn("secret", secret);
156 		req(! containsNewline(key), "Argument 'key' must not contain a newline or carriage return character.");
157 		req(! containsNewline(service), "The service name must not contain a newline or carriage return character.");
158 		req(! containsNewline(secret), "Argument 'secret' must not contain a newline or carriage return character.");
159 		var payload = addCommandPayload(key, secret);
160 		try {
161 			// "security -i" reads commands from stdin; a bare "-w" would prompt on /dev/tty (hanging when the JVM has
162 			// a terminal), and "-w <secret>" on argv would expose the secret to ps.  The command line goes on stdin.
163 			var r = run(payload, "-i");
164 			if (r.exit() != 0)
165 				throw backendFailure("add-generic-password", r);
166 		} catch (RuntimeException e) {
167 			if (failMode == FailMode.FAIL_OPEN)
168 				return;
169 			throw e;
170 		} finally {
171 			Arrays.fill(payload, (byte) 0);
172 		}
173 	}
174 
175 	@Override /* SecretStore */
176 	public Optional<char[]> find(String key) {
177 		reqnn("key", key);
178 		try {
179 			var r = run(FIND_GENERIC_PASSWORD, "-a", key, "-s", service, "-g");
180 			if (r.exit() == 0) {
181 				try {
182 					return o(parsePassword(r.stderr()));
183 				} finally {
184 					Arrays.fill(r.stderr(), (byte) 0);
185 				}
186 			}
187 			if (r.exit() == NOT_FOUND)
188 				return oe();
189 			throw backendFailure(FIND_GENERIC_PASSWORD, r);
190 		} catch (RuntimeException e) {
191 			if (failMode == FailMode.FAIL_OPEN)
192 				return oe();
193 			throw e;
194 		}
195 	}
196 
197 	@Override /* SecretStore */
198 	public boolean exists(String key) {
199 		reqnn("key", key);
200 		try {
201 			var r = run(FIND_GENERIC_PASSWORD, "-a", key, "-s", service);
202 			if (r.exit() == 0)
203 				return true;
204 			if (r.exit() == NOT_FOUND)
205 				return false;
206 			throw backendFailure(FIND_GENERIC_PASSWORD, r);
207 		} catch (RuntimeException e) {
208 			if (failMode == FailMode.FAIL_OPEN)
209 				return false;
210 			throw e;
211 		}
212 	}

216 		reqnn("key", key);
217 		try {
218 			var r = run("delete-generic-password", "-a", key, "-s", service);
219 			if (r.exit() == 0)
220 				return true;
221 			if (r.exit() == NOT_FOUND)
222 				return false;
223 			throw backendFailure("delete-generic-password", r);
224 		} catch (RuntimeException e) {
225 			if (failMode == FailMode.FAIL_OPEN)
226 				return false;
227 			throw e;
228 		}

231 	/**
232 	 * The seam through which this store invokes the underlying keychain command.
233 	 *
234 	 * <p>
235 	 * Production code always runs against {@link ProcessCommandRunner}, which shells out to the real
236 	 * {@code security} CLI.  Tests substitute an in-memory fake keychain instead (via the package-private
237 	 * {@link #KeychainSecretStore(String,FailMode,CommandRunner) constructor}), so this store's argument
238 	 * construction and output parsing run deterministically on every OS without shelling out or touching a real
239 	 * keychain.
240 	 */
241 	@FunctionalInterface
242 	interface CommandRunner {
243 
244 		/**
245 		 * Invokes the command with the given arguments, optionally writing {@code stdin} to it first.
246 		 *
247 		 * @param stdin The bytes to write to the command's stdin before closing it, or <jk>null</jk> to close stdin
248 		 * 	immediately without writing anything.
249 		 * @param args The command's arguments (not including the executable path/name itself).
250 		 * @return The command's exit code, captured stdout, and captured stderr.
251 		 */
252 		Result run(byte[] stdin, String[] args);
253 	}
254 
255 	/** The outcome of a single {@code security} invocation. */
256 	@SuppressWarnings({
257 		"java:S6218" // stdout can hold retrieved SECRET bytes; identity-based equals/hashCode/toString are intentional so a content-based toString can never surface the secret. Result instances are never compared or printed.
258 	})
259 	record Result(int exit, byte[] stdout, byte[] stderr) {
260 
261 		// stderr can hold the retrieved secret (find-generic-password -g), so it is kept as bytes that can be zeroed.
262 		Result(int exit, byte[] stdout, String stderr) {
263 			this(exit, stdout, stderr.getBytes(UTF_8));
264 		}
265 
266 		String stderrText() {
267 			return new String(stderr, UTF_8);
268 		}
269 	}

271 	/** {@link CommandRunner} that shells out to the real {@code security} binary via {@link ProcessBuilder}. */
272 	private static final class ProcessCommandRunner implements CommandRunner {
273 
274 		private final long timeoutSeconds;
275 		private final String binary;
276 
277 		ProcessCommandRunner(long timeoutSeconds, String binary) {
278 			this.timeoutSeconds = timeoutSeconds;
279 			this.binary = binary;
280 		}
281 
282 		/**
283 		 * Invokes {@link #binary} with the given arguments, optionally writing {@code stdin} to the child process
284 		 * before closing its input stream.  {@code stdin}, when non-<jk>null</jk>, is written and the stream closed
285 		 * <i>before</i> waiting on the process so a CLI that blocks on a stdin prompt (as {@code -w} with no value
286 		 * does) does not deadlock against this method.
287 		 */
288 		@Override /* CommandRunner */
289 		public Result run(byte[] stdin, String[] args) {
290 			var cmd = new ArrayList<String>(args.length + 1);
291 			cmd.add(binary);
292 			cmd.addAll(Arrays.asList(args));
293 			try {
294 				var p = new ProcessBuilder(cmd).start();
295 				if (stdin != null) {
296 					try (var os = p.getOutputStream()) {
297 						os.write(stdin);
298 					}
299 				} else {
300 					p.getOutputStream().close();
301 				}
302 				if (! p.waitFor(timeoutSeconds, SECONDS)) {
303 					p.destroyForcibly();
304 					throw isex("Timed out invoking '%s' after %s seconds.", binary, timeoutSeconds);
305 				}
306 				byte[] out;
307 				String err;
308 				try (var is = p.getInputStream(); var es = p.getErrorStream()) {
309 					out = is.readAllBytes();
310 					err = new String(es.readAllBytes(), UTF_8);
311 				}
312 				return new Result(p.exitValue(), out, err);
313 			} catch (IOException e) {
314 				throw rex(e, "Unable to invoke '%s' (is this macOS with the keychain CLI available?).", binary);
315 			} catch (InterruptedException e) {
316 				Thread.currentThread().interrupt();
317 				throw rex(e, "Interrupted while invoking '%s'.", binary);
318 			}
319 		}
320 	}
321 
322 	private Result run(String... args) {
323 		return run(null, args);
324 	}
325 
326 	private Result run(byte[] stdin, String... args) {
327 		return commandRunner.run(stdin, args);
328 	}

345 	/**
346 	 * Builds the <c>security -i</c> stdin payload
347 	 * <c>add-generic-password -U -a "key" -s "service" -w "secret"\n</c>, double-quoting each value and escaping
348 	 * <c>"</c> and <c>\</c> with a backslash.  The secret is encoded and escaped directly from the {@code char[]},
349 	 * without an intermediate {@link String}, and the intermediate byte copy is zeroed.
350 	 */
351 	private byte[] addCommandPayload(String key, char[] secret) {
352 		var head = ("add-generic-password -U -a " + quote(key) + " -s " + quote(service) + " -w \"").getBytes(UTF_8);
353 		var encoded = UTF_8.encode(CharBuffer.wrap(secret));
354 		var secretBytes = new byte[encoded.remaining()];
355 		encoded.get(secretBytes);
356 		if (encoded.hasArray())
357 			Arrays.fill(encoded.array(), (byte) 0);
358 		try {
359 			var escapes = 0;
360 			for (var b : secretBytes)
361 				if (b == '"' || b == '\\')
362 					escapes++;
363 			var payload = new byte[head.length + secretBytes.length + escapes + 2];
364 			System.arraycopy(head, 0, payload, 0, head.length);
365 			var i = head.length;
366 			for (var b : secretBytes) {
367 				if (b == '"' || b == '\\')
368 					payload[i++] = '\\';
369 				payload[i++] = b;
370 			}
371 			payload[i++] = '"';
372 			payload[i] = '\n';
373 			return payload;
374 		} finally {
375 			Arrays.fill(secretBytes, (byte) 0);
376 		}
377 	}

379 	/** Double-quotes a non-secret value for <c>security -i</c>, escaping <c>"</c> and <c>\</c>. */
380 	private static String quote(String s) {
381 		return '"' + s.replace("\\", "\\\\").replace("\"", "\\\"") + '"';
382 	}

384 	private static final byte[] PASSWORD_PREFIX = "password: ".getBytes(UTF_8);
385 
386 	/**
387 	 * Extracts the secret from the stderr of <c>find-generic-password -g</c>.
388 	 *
389 	 * <p>
390 	 * Finds the line starting with <c>password: </c> and decodes it per its form: <c>"text"</c> (the bytes between the
391 	 * first and last quote), <c>0x<i>HEX</i></c> (hex digits up to the first non-hex character), or empty.  No
392 	 * intermediate {@link String} is created, and the hex byte buffer is zeroed.
393 	 */
394 	@SuppressWarnings({
395 		"java:S3776" // One scan with a branch per password form.
396 	})
397 	private static char[] parsePassword(byte[] err) {
398 		var start = -1;
399 		for (var i = 0; i + PASSWORD_PREFIX.length <= err.length && start < 0; i++)
400 			if ((i == 0 || err[i - 1] == '\n') && Arrays.equals(err, i, i + PASSWORD_PREFIX.length, PASSWORD_PREFIX, 0, PASSWORD_PREFIX.length))
401 				start = i + PASSWORD_PREFIX.length;
402 		if (start < 0)
403 			throw isex("Keychain '%s' succeeded but its output did not contain a password line.", FIND_GENERIC_PASSWORD);
404 		var end = start;
405 		while (end < err.length && err[end] != '\n')
406 			end++;
407 		if (end > start && err[end - 1] == '\r')
408 			end--;
409 		if (end == start)
410 			return new char[0];
411 		if (err[start] == '"') {
412 			var close = end - 1;
413 			while (close > start && err[close] != '"')
414 				close--;
415 			return decode(err, start + 1, close);
416 		}
417 		if (end - start > 2 && err[start] == '0' && err[start + 1] == 'x') {
418 			var digits = 0;
419 			while (start + 2 + digits < end && Character.digit(err[start + 2 + digits], 16) >= 0)
420 				digits++;
421 			if (digits % 2 != 0)
422 				throw isex("Keychain '%s' returned a malformed hex password.", FIND_GENERIC_PASSWORD);
423 			var bytes = new byte[digits / 2];
424 			try {
425 				for (var i = 0; i < bytes.length; i++)
426 					bytes[i] = (byte)((Character.digit(err[start + 2 + 2 * i], 16) << 4) | Character.digit(err[start + 3 + 2 * i], 16));
427 				return decode(bytes, 0, bytes.length);
428 			} finally {
429 				Arrays.fill(bytes, (byte) 0);
430 			}
431 		}
432 		throw isex("Keychain '%s' returned an unrecognized password format.", FIND_GENERIC_PASSWORD);
433 	}

435 	/** Decodes UTF-8 bytes into a char[] without an intermediate String, zeroing the decoder's char buffer. */
436 	private static char[] decode(byte[] bytes, int from, int to) {
437 		var cb = UTF_8.decode(ByteBuffer.wrap(bytes, from, to - from));
438 		var chars = new char[cb.remaining()];
439 		cb.get(chars);
440 		if (cb.hasArray())
441 			Arrays.fill(cb.array(), '\0');
442 		return chars;
443 	}
