17  package org.apache.juneau.console;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.marshall.collections.*;
22  
23  /**
24   * One entry of a card's {@code subscribes} list: a concrete topic wired to a role the card type implements.
25   *
26   * <h5 class='section'>Example:</h5>
27   * <p class='bjava'>
28   *   <jc>// Master/detail: the tasks table reloads with the selected change's id.</jc>
29   *   CardSpec <jv>tasks</jv> = CardSpec.<jsm>of</jsm>(<js>"datatables"</js>, <js>"tasks"</js>)
30   *     .subscribes(Subscription.<jsm>to</jsm>(Topics.<jsm>selection</jsm>(<js>"changes"</js>)).as(<js>"params"</js>)
31   *       .map(<js>"changeId"</js>, <js>"ids.0"</js>).whenEmpty(WhenEmpty.<jsf>CLEAR</jsf>)
32   *       .emptyText(<js>"Select a change to see its tasks."</js>));
33   *
34   *   <jc>// Linked filter, raw payload.</jc>
35   *   Subscription.<jsm>to</jsm>(Topics.<jsm>filter</jsm>(<js>"changes"</js>)).as(<js>"filter"</js>);
36   * </p>
37   *
38   * @since 10.0.0
39   */
40  public final class Subscription {
41  
42  	/** What a subscriber does when the payload is cleared or a mapped path is missing. */
43  	public enum WhenEmpty {
44  		/** Paint {@code emptyText} and skip the fetch (the default). */
45  		CLEAR,
46  		/** Leave the last render alone. */
47  		KEEP
48  	}
49  
50  	private final String topic;
51  	private String role;
52  	private String emptyText;
53  	private WhenEmpty whenEmpty;
54  	private final Map<String,String> map = new LinkedHashMap<>();
55  
56  	private Subscription(String topic) {
57  		this.topic = topic;
58  	}
59  
60  	/**
61  	 * @param topic A concrete topic; no wildcards.
62  	 * @return A new subscription.
63  	 * @throws IllegalArgumentException E-40 on bad syntax.
64  	 */
65  	public static Subscription to(String topic) {
66  		Topics.checkTopic(topic);
67  		return new Subscription(topic);
68  	}
69  
70  	/** @param role A role the card type accepts, matching {@code ^[a-z][a-z0-9-]{0,31}$}. @return This object. */
71  	public Subscription as(String role) {
72  		if (role == null || ! Topics.ROLE.matcher(role).matches())
73  			throw Topics.iae("subscription to '%s': role '%s' must match ^[a-z][a-z0-9-]{0,31}$", topic, role);
74  		this.role = role;
75  		return this;
76  	}
77  
78  	/**
79  	 * Maps one payload path to a target; repeatable.
80  	 *
81  	 * @param target A {@code {token}} name for {@code params}, or a dotted field such as {@code columns.region} for {@code filter}.
82  	 * @param path A dotted path into the payload, with numeric segments for array indexes, such as {@code ids.0}.
83  	 * @return This object.
84  	 */
85  	public Subscription map(String target, String path) {
86  		if (target == null || ! Topics.MAP_PATH.matcher(target).matches())
87  			throw Topics.iae("subscription to '%s': map target '%s' must be a dotted name", topic, target);
88  		if (path == null || ! Topics.MAP_PATH.matcher(path).matches())
89  			throw Topics.iae("subscription to '%s': map path '%s' must be a dotted path such as 'ids.0'", topic, path);
90  		map.put(target, path);
91  		return this;
92  	}
93  
94  	/** @param v {@link WhenEmpty#CLEAR} (the default when unset) or {@link WhenEmpty#KEEP}. @return This object. */
95  	public Subscription whenEmpty(WhenEmpty v) {
96  		whenEmpty = Objects.requireNonNull(v, "whenEmpty");
97  		return this;
98  	}
99  
100 	/** @param v The text painted when the payload is empty; the shell's default otherwise. @return This object. */
101 	public Subscription emptyText(String v) {
102 		emptyText = v;
103 		return this;
104 	}
105 
106 	/** @return The topic. */
107 	public String topic() {
108 		return topic;
109 	}
110 
111 	/** @return The {@code $defs/subscription} entry. */
112 	public JsonMap toMap() {
113 		if (role == null)
114 			throw Topics.iae("subscription to '%s' needs as(role)", topic);
115 		var m = new JsonMap();
116 		m.put("topic", topic);
117 		m.put("as", role);
118 		if (! map.isEmpty()) {
119 			var mm = new JsonMap();
120 			mm.putAll(map);
121 			m.put("map", mm);
122 		}
123 		if (whenEmpty != null)
124 			m.put("whenEmpty", whenEmpty.name().toLowerCase(Locale.ROOT));
125 		if (emptyText != null)
126 			m.put("emptyText", emptyText);
127 		return m;
128 	}
129 }
