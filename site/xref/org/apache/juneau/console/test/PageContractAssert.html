28  /**
29   * Test helper that extracts the {@code #juneau-page} contract and {@code <template>} ids from a
30   * rendered console page, validates them against {@link PageContractSchema}, and offers fluent
31   * assertions so adopters assert on the contract instead of on rendered chrome HTML.
32   *
33   * <p>
34   * Failures throw {@link AssertionError}, so the class works under any test framework.
35   *
36   * <p>
37   * {@link #assertPage(String)} also fails immediately if the contract still carries the pre-rename top-level
38   * <c>version</c> key (now <c>contractVersion</c>), even when {@link #isValid()} is never called.
39   *
40   * <h5 class='section'>Example:</h5>
41   * <p class='bjava'>
42   * 	String <jv>html</jv> = <jv>client</jv>.get(<js>"/slo"</js>).run().getContent().asString();
43   * 	PageContractAssert.<jsm>assertPage</jsm>(<jv>html</jv>)
44   * 		.isValid()
45   * 		.hasContractVersion(<js>"1"</js>)
46   * 		.hasActiveNav(<js>"slo"</js>)
47   * 		.hasNavPath(<js>"home"</js>, <js>"about"</js>)
48   * 		.hasNavHref(<js>"slo"</js>, <js>"/slo"</js>)
49   * 		.hasFooterText(<js>"Sandbox Support Console"</js>)
50   * 		.hasHeaderSlot(<js>"banner"</js>)
51   * 		.hasCard(<js>"releases"</js>, <js>"datatables"</js>)
52   * 		.hasCardKey(<js>"releases"</js>, <js>"/table/dataUrl"</js>, <js>"/rest/releases/data"</js>)
53   * 		.templateContains(<js>"jc-seg-1"</js>, <js>"id=\"ssc-table-slot\""</js>);
54   *
55   * 	PageContractAssert.<jsm>assertPage</jsm>(<jv>html</jv>)
56   * 		.subscribes(<js>"tasks"</js>, <js>"selection:changes"</js>, <js>"params"</js>)
57   * 		.declaresTopic(<js>"ops.jobs"</js>)
58   * 		.hasBridge(<js>"ops"</js>)
59   * 		.hasNoWiringErrors();
60   * </p>
61   *
62   * @since 10.0.0
63   */
64  @SuppressWarnings({
65  	"java:S1192" // Duplicated literals read more clearly inline than as constants.
66  })

69  	private static final Pattern ISLAND = Pattern.compile(
70  		"<script type=\"application/json\" id=\"juneau-page\">(.*?)</script>", Pattern.DOTALL);
71  	private static final Pattern TEMPLATE_OPEN = Pattern.compile("<template data-(card|slot)=\"([^\"]+)\">");
72  
73  	private final String json;
74  	private final JsonMap contract;
75  	private final Map<String,String> templates;

77  	private PageContractAssert(String html) {
78  		var m = ISLAND.matcher(html);
79  		if (! m.find())
80  			throw new AssertionError("no <script type=\"application/json\" id=\"juneau-page\"> in page");
81  		json = m.group(1);
82  		try {
83  			contract = JsonMap.ofString(json);
84  		} catch (Exception e) {
85  			throw new AssertionError("unparseable page contract: " + e.getMessage(), e);
86  		}
87  		if (contract.containsKey("version"))
88  			throw new AssertionError("page contract key 'version' was renamed to 'contractVersion'; regenerate the page with a current Juneau");
89  		templates = scanTemplates(html);
90  	}

288 	private JsonMap subscription(String cardId, String topic, String role) {
289 		var l = card(cardId).getList("subscribes");
290 		var roles = new ArrayList<String>();
291 		if (l != null)
292 			for (var o : l) {
293 				var m = (JsonMap)o;
294 				if (! topic.equals(m.getString("topic")))
295 					continue;
296 				if (role == null || role.equals(m.getString("as")))
297 					return m;
298 				roles.add(m.getString("as"));
299 			}
300 		if (! roles.isEmpty())
301 			throw new AssertionError("card '" + cardId + "' subscribes to '" + topic + "' as " + roles + ", not '" + role + "'");
302 		throw new AssertionError("card '" + cardId + "' does not subscribe to '" + topic + "'; subscribes: " + topicsOf(l));
303 	}

307 		if (l != null)
308 			for (var o : l)
309 				out.add(((JsonMap)o).getString("topic"));
310 		return out;
311 	}

346 	/**
347 	 * Asserts this page and another render the same contract and the same template markup.
348 	 *
349 	 * <p>
350 	 * Use it to prove two ways of producing a page (for example a hand-written template and its builder twin) agree.
351 	 * The contracts are compared as key-sorted JSON; list order is kept, because card, nav and option order is part of
352 	 * the page.  Templates are compared by id on their {@linkplain String#strip() stripped} markup, so the whitespace a
353 	 * template author's layout puts around a card body does not count.
354 	 *
355 	 * <p>
356 	 * The failure message names the first difference, for example
357 	 * {@code page contracts differ at $.cards[0].bare: this=true other=false}.
358 	 *
359 	 * @param other The page this one must equal.
360 	 * @return This object.
361 	 */
362 	public PageContractAssert isTwinOf(PageContractAssert other) {
363 		assertTwins(contract, templates, other.contract, other.templates);
364 		return this;
365 	}

367 	/**
368 	 * Asserts two contract and template pairs are twins, as {@link #isTwinOf(PageContractAssert)} does.
369 	 *
370 	 * <p>
371 	 * Exposed so a test that must first adjust either side (for example remove a card known to differ) can still use the
372 	 * shared comparison.
373 	 *
374 	 * @param contract The first contract.
375 	 * @param templates The first page's templates, by id.
376 	 * @param otherContract The second contract.
377 	 * @param otherTemplates The second page's templates, by id.
378 	 * @throws AssertionError If the contracts or templates differ.
379 	 */
380 	public static void assertTwins(Map<?,?> contract, Map<String,String> templates, Map<?,?> otherContract, Map<String,String> otherTemplates) {
381 		var diff = firstDiff("$", canonical(contract), canonical(otherContract));
382 		if (diff != null)
383 			throw new AssertionError("page contracts differ at " + diff);
384 		var a = stripped(templates);
385 		var b = stripped(otherTemplates);
386 		var ids = new TreeSet<>(a.keySet());
387 		ids.addAll(b.keySet());
388 		for (var id : ids)
389 			if (neq(a.get(id), b.get(id)))
390 				throw new AssertionError("page template '" + id + "' differs: this=" + a.get(id) + " other=" + b.get(id));
391 	}
392 
393 	private static Map<String,String> stripped(Map<String,String> templates) {
394 		var out = new TreeMap<String,String>();
395 		templates.forEach((k, v) -> out.put(k, v.strip()));
396 		return out;
397 	}

399 	// Recursively key-sorts every Map; List order is kept.
400 	private static Object canonical(Object v) {
401 		if (v instanceof Map<?,?> m) {
402 			var out = new TreeMap<String,Object>();
403 			m.forEach((k, x) -> out.put(String.valueOf(k), canonical(x)));
404 			return out;
405 		}
406 		if (v instanceof List<?> l)
407 			return l.stream().map(PageContractAssert::canonical).toList();
408 		return v;
409 	}
410 
411 	// The first place two canonical trees differ, or null.
412 	private static String firstDiff(String path, Object a, Object b) {
413 		if (! neq(a, b))
414 			return null;
415 		if (a instanceof Map<?,?> x && b instanceof Map<?,?> y) {
416 			var keys = new TreeSet<Object>(x.keySet());
417 			keys.addAll(y.keySet());
418 			for (var k : keys) {
419 				var d = firstDiff(path + "." + k, x.get(k), y.get(k));
420 				if (d != null)
421 					return d;
422 			}
423 		}
424 		if (a instanceof List<?> x && b instanceof List<?> y && x.size() == y.size())
425 			for (var i = 0; i < x.size(); i++) {
426 				var d = firstDiff(path + "[" + i + "]", x.get(i), y.get(i));
427 				if (d != null)
428 					return d;
429 			}
430 		return path + ": this=" + a + " other=" + b;
431 	}

474 	private JsonMap node(List<String> path) {
475 		var level = contract.getList("nav");
476 		JsonMap found = null;
477 		for (var id : path) {
478 			found = null;
479 			if (level != null)
480 				for (var o : level)
481 					if (id.equals(((JsonMap)o).getString("id")))
482 						found = (JsonMap)o;
483 			if (found == null)
484 				throw new AssertionError("nav path " + path + " not found (failed at '" + id + "')");
485 			level = found.getList("children");
486 		}
487 		return found;
488 	}

505 	private static Object resolvePointer(Object node, String pointer) {
506 		if (pointer.isEmpty())
507 			return node;
508 		if (! pointer.startsWith("/"))
509 			throw new AssertionError("json pointer '" + pointer + "' must start with '/'");
510 		var cur = node;
511 		for (var raw : pointer.substring(1).split("/", -1)) {
512 			var seg = raw.replace("~1", "/").replace("~0", "~");
513 			if (cur instanceof Map<?,?> m) {
514 				cur = m.get(seg);
515 			} else if (cur instanceof List<?> l) {
516 				var idx = Integer.parseInt(seg);
517 				cur = idx >= 0 && idx < l.size() ? l.get(idx) : null;
518 			} else {
519 				throw new AssertionError("json pointer '" + pointer + "' hit a leaf before '" + seg + "'; node was: " + cur);
520 			}
521 		}
522 		return cur;
523 	}

529 	// Depth-aware: page markup may itself contain <template> elements.
530 	private static Map<String,String> scanTemplates(String html) {
531 		var out = new LinkedHashMap<String,String>();
532 		var m = TEMPLATE_OPEN.matcher(html);
533 		var from = 0;
534 		while (m.find(from)) {
535 			var start = m.end();
536 			var depth = 1;
537 			var i = start;
538 			while (depth > 0) {
539 				var open = html.indexOf("<template", i);
540 				var close = html.indexOf("</template>", i);
541 				if (close < 0)
542 					throw new AssertionError("unterminated <template data-" + m.group(1) + "=\"" + m.group(2) + "\">");
543 				if (open >= 0 && open < close) {
544 					depth++;
545 					i = open + 9;
546 				} else {
547 					depth--;
548 					i = close + 11;
549 				}
550 			}
551 			out.put(m.group(2), html.substring(start, i - 11));
552 			from = i;
553 		}
554 		return out;
555 	}
