17  package org.apache.juneau.console;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.marshall.collections.*;
22  
23  /**
24   * A custom topic declaration, for a card's {@code publishes} or the page's top-level {@code topics}.
25   *
26   * <h5 class='section'>Example:</h5>
27   * <p class='bjava'>
28   *   <jc>// Published by this card's JS:</jc>
29   *   <jv>card</jv>.publishes(TopicDecl.<jsm>of</jsm>(<js>"ssc.focus"</js>).retain(<jk>true</jk>));
30   *
31   *   <jc>// Published by a page script or the server:</jc>
32   *   PageSpec.<jsm>create</jsm>().topic(TopicDecl.<jsm>of</jsm>(<js>"app.region-picked"</js>).retain(<jk>true</jk>)
33   *     .publisher(TopicDecl.Publisher.<jsf>SCRIPT</jsf>));
34   * </p>
35   *
36   * @since 10.0.0
37   */
38  public final class TopicDecl {
39  
40  	/** Who publishes a page-level topic. */
41  	public enum Publisher {
42  		/** A page script. */
43  		SCRIPT,
44  		/** The server, through a bridge (R-11). */
45  		SERVER,
46  		/** Only ribbon {@code publish} items. */
47  		RIBBON
48  	}
49  
50  	private static final String JOB_ALL = "job:*";
51  
52  	private final String topic;
53  	private Boolean retain;
54  	private Publisher publisher;
55  
56  	private TopicDecl(String topic) {
57  		this.topic = topic;
58  	}
59  
60  	/**
61  	 * @param topic A custom topic or {@code ns.name:*}; or {@code job:*}, which is only valid page-level with {@link Publisher#SERVER}.
62  	 * @return A new declaration.
63  	 * @throws IllegalArgumentException E-45 for a framework family or an un-namespaced family; E-40 on bad syntax.
64  	 */
65  	public static TopicDecl of(String topic) {
66  		if (topic == null)
67  			throw Topics.iae(Topics.E40, topic);
68  		var family = Topics.family(topic);
69  		var declarable = Topics.DECLARABLE_FRAMEWORK.contains(family);
70  		if (Topics.FRAMEWORK_FAMILIES.contains(family) && ! JOB_ALL.equals(topic) && ! declarable)
71  			throw Topics.iae(Topics.E45_FRAMEWORK, family);
72  		if (Topics.ROLE.matcher(family).matches() && ! Topics.FRAMEWORK_FAMILIES.contains(family))
73  			throw Topics.iae(Topics.E45_NAMESPACE, family, family);
74  		if (declarable ? ! Topics.TOPIC.matcher(topic).matches() || topic.indexOf(':') < 0
75  				: ! Topics.TOPIC_DECL.matcher(topic).matches())
76  			throw Topics.iae(Topics.E40, topic);
77  		return new TopicDecl(topic);
78  	}
79  
80  	/** @param v Whether the bus keeps the last value for late subscribers. Required. @return This object. */
81  	public TopicDecl retain(boolean v) {
82  		retain = v;
83  		return this;
84  	}
85  
86  	/** @param p Who publishes the topic. Page-level only. @return This object. */
87  	public TopicDecl publisher(Publisher p) {
88  		publisher = Objects.requireNonNull(p, "publisher");
89  		return this;
90  	}
91  
92  	/** @return The topic. */
93  	public String topic() {
94  		return topic;
95  	}
96  
97  	/** @return The page-level {@code $defs/topicDecl} entry. */
98  	public JsonMap toMap() {
99  		if (JOB_ALL.equals(topic) && publisher != Publisher.SERVER)
100 			throw Topics.iae(Topics.E45_FRAMEWORK, "job");
101 		var family = Topics.family(topic);
102 		if ("zoom".equals(family) || "legend".equals(family))
103 			throw Topics.iae("topic '%s': %s:<cardId> is published by its chart card; declare it with publishZoom()/publishLegend()", topic, family);
104 		if ("data".equals(family) && (publisher != Publisher.SCRIPT || ! Boolean.TRUE.equals(retain)))
105 			throw Topics.iae("topic '%s': a page-declared data: topic needs publisher(SCRIPT) and retain(true)", topic);
106 		var m = base();
107 		if (publisher == null)
108 			throw Topics.iae("topic '%s': a page-level topic needs publisher(SCRIPT|SERVER|RIBBON)", topic);
109 		m.put("publisher", publisher.name().toLowerCase(Locale.ROOT));
110 		return m;
111 	}
112 
113 	/** @return The card-level {@code $defs/publication} entry, for {@code CardSpec.publishes}. */
114 	public JsonMap toPublicationMap() {
115 		if (JOB_ALL.equals(topic))
116 			throw Topics.iae(Topics.E45_FRAMEWORK, "job");
117 		if ("data".equals(Topics.family(topic)))
118 			throw Topics.iae("topic '%s': data: topics are declared page-level (publisher SCRIPT) or by a data source, not by a card", topic);
119 		if (Topics.DECLARABLE_FRAMEWORK.contains(Topics.family(topic)) && ! Boolean.TRUE.equals(retain))
120 			throw Topics.iae("topic '%s' is chart state; declare it with retain(true)", topic);
121 		if (publisher != null)
122 			throw Topics.iae("topic '%s': publisher is page-level only; a card's publishes may not set it", topic);
123 		return base();
124 	}
125 
126 	private JsonMap base() {
127 		if (retain == null)
128 			throw Topics.iae("topic '%s' is declared with retain=unset here; call retain(true) or retain(false)", topic);
129 		var m = new JsonMap();
130 		m.put("topic", topic);
131 		m.put("retain", retain);
132 		return m;
133 	}
134 }
