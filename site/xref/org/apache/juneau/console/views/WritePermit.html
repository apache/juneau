17  package org.apache.juneau.console.views;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  /**
22   * The views-toolkit's own structural gate token for {@link BulkMutateDef}: proof that the caller's own write-gate
23   * was consulted before a bulk-mutating affordance is configured (the opt-in separability
24   * guarantee).
25   *
26   * <h5 class='section'>Why this exists, and why it is NOT the console's own {@code WritePermit}</h5>
27   * <p>
28   * A consuming application (e.g. the Sandbox Support Console) may already have its own, much richer write-gate
29   * token &mdash; one bound to a journal, a box {@code Mode}, an idempotency key, and a target validator (see that
30   * application's own {@code write.WritePermit}/{@code WriteGuard}). That type is deliberately <b>not</b> reused
31   * here: its constructor is package-private to its own {@code write} package, it is defined in a console-ui
32   * artifact this framework module must not depend on, and its shape carries concerns (journal binding, SAFE/LIVE
33   * mode) that are that application's business, not this framework's. Reaching into it would be a layering
34   * violation in the wrong direction (framework depending on a specific consumer).
35   *
36   * <p>
37   * This type is therefore this module's own, minimal, independent token &mdash; the same <i>discipline</i>
38   * (a required-parameter capability proof), a deliberately smaller <i>shape</i> (no journal/mode/idempotency
39   * concerns, which belong to the consumer's own gate). A consuming application's own write-gate is expected to
40   * mint one (typically by wrapping/adapting its own gate result) only after ITS OWN checks have passed, then hand
41   * it to {@link BulkMutateDef#create(WritePermit, SelectionDef)}. This module does not, and cannot, verify that the
42   * caller's gate logic is correct &mdash; that is out of scope for a UI-rendering toolkit &mdash; but it does
43   * guarantee, by construction, that {@link BulkMutateDef} cannot be reached at all without a permit in hand: there
44   * is no zero-argument overload, so omitting one is a <b>compile</b> failure, not a missed javadoc sentence
45   * (mirroring the discipline of the console's own {@code WriteGuard}).
46   *
47   * <h5 class='section'>Example:</h5>
48   * <p class='bjava'>
49   * 	<jc>// After the consumer's OWN write-gate has already approved this capability:</jc>
50   * 	WritePermit <jv>permit</jv> = WritePermit.<jsm>forCapability</jsm>(<js>"incident:bulk-ack"</js>);
51   * 	BulkMutateDef <jv>bulk</jv> = BulkMutateDef.<jsm>create</jsm>(<jv>permit</jv>, <jv>selection</jv>)
52   * 		.actions(RowAction.<jsm>create</jsm>(<js>"ack"</js>).endpoint(<js>"servlet:/incidents/{id}/ack"</js>).method(RowAction.Method.<jsf>POST</jsf>));
53   * </p>
54   *
55   * <h5 class='section'>See Also:</h5>
56   * <ul>
57   * 	<li class='jc'>{@link BulkMutateDef}
58   * 	<li class='jc'>{@link SelectionDef}
59   * </ul>
60   *
61   * @since 10.0.0
62   */
63  public final class WritePermit {
64  
65  	private final String capability;
66  
67  	private WritePermit(String capability) {
68  		this.capability = capability;
69  	}
70  
71  	/**
72  	 * Mints a permit for the named capability.
73  	 *
74  	 * <p>
75  	 * This is a bare proof-of-authorization token, not an enforcement seam: minting one asserts that the CALLER's
76  	 * own write-gate has already approved {@code capability}. This module trusts that assertion &mdash; it has no
77  	 * journal, arming, or idempotency logic of its own to check it against &mdash; its only job is to make the
78  	 * bulk-mutate opt-in unreachable without one.
79  	 *
80  	 * @param capability A stable, human-readable identifier for the capability this permit authorizes (e.g.
81  	 * 	{@code "incident:bulk-ack"}). Must not be <jk>null</jk> or blank.
82  	 * @return A new {@link WritePermit}.
83  	 * @throws IllegalArgumentException If {@code capability} is <jk>null</jk> or blank.
84  	 */
85  	public static WritePermit forCapability(String capability) {
86  		if (ib(capability))
87  			throw iaex("WritePermit capability must not be null or blank.");
88  		return new WritePermit(capability);
89  	}
90  
91  	/**
92  	 * The capability this permit authorizes.
93  	 *
94  	 * @return The capability identifier passed to {@link #forCapability(String)}.
95  	 */
96  	public String capability() {
97  		return capability;
98  	}
99  
100 	@Override /* Object */
101 	public String toString() {
102 		return "WritePermit(capability=" + capability + ")";
103 	}
104 }
