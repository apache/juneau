15  package org.apache.juneau.console.views;
16  
17  /**
18   * How a field-grid catalog entry (or a remaining declarative field) is painted.
19   *
20   * <p>
21   * Each constant carries the lowercase token emitted on {@code format} / {@code data-juneau-field-format}.
22   * {@link #TEXT} is omitted from the wire (the default).
23   *
24   * @since 10.0.0
25   */
26  public enum FieldFormat {
27  
28  	/** Paint with {@code textContent}.  The default. */
29  	TEXT("text"),
30  
31  	/**
32  	 * Treat the value as sanitizing-markdown HTML and copy allowlisted nodes into the slot.
33  	 * Never {@code innerHTML}.
34  	 */
35  	MARKDOWN("markdown"),
36  
37  	/**
38  	 * Treat the value as rich HTML the <b>caller has already sanitized server-side</b>, and paint it
39  	 * through this runtime's own second allowlist &mdash; a wider one than {@link #MARKDOWN}'s, admitting
40  	 * {@code <img>} and the remaining table/typographic tags a full-fidelity rich-text body needs.
41  	 *
42  	 * <p>
43  	 * <b>This is not a raw-HTML sink.</b>  Like {@link #MARKDOWN}, the client parses the value with
44  	 * {@code DOMParser} into an inert document and copies allowlisted nodes via {@code createElement}/
45  	 * {@code createTextNode}; it never assigns {@code innerHTML}, and it never serializes back to a string (so
46  	 * the mutation-XSS class that a parse&rarr;serialize&rarr;reparse sanitizer is prone to cannot arise).
47  	 * Attributes are copied by an explicit per-tag allowlist, so {@code on*} handlers, {@code srcdoc},
48  	 * {@code style} and every other unnamed attribute cannot survive <i>by construction</i> rather than by a
49  	 * deny-list that has to anticipate them.  {@code href} and {@code src} are scheme-checked in two layers,
50  	 * not one: an explicit prefix reject for {@code javascript:}, {@code data:} and {@code vbscript:}, plus a
51  	 * <b>colon-fallback</b> rule underneath it that rejects any value containing a colon outside the
52  	 * explicitly allowlisted absolute prefixes ({@code http:}/{@code https:}/{@code mailto:} for
53  	 * {@code href}; {@code http:}/{@code https:} for {@code src}).  <b>The colon-fallback, not the named-
54  	 * scheme prefix check, is the load-bearing defense</b>: it is what still rejects an obfuscated spelling
55  	 * of one of those three schemes (e.g. one split by a stray whitespace/control character that a browser's
56  	 * URL parser strips before resolving, such as {@code java\tscript:}) &mdash; the prefix check alone would
57  	 * miss it, but the obfuscated string still contains a colon and matches none of the allowed prefixes, so
58  	 * it still fails closed. A future "simplification" to a longer bare {@code javascript:}/{@code data:}/
59  	 * {@code vbscript:} deny-list that drops the colon-fallback as apparently redundant would silently
60  	 * reopen that hole.
61  	 *
62  	 * <p>
63  	 * <b>The contract is nonetheless caller-sanitizes-first.</b>  This runtime's allowlist is a second,
64  	 * independent gate &mdash; defense in depth &mdash; not a substitute for a real server-side sanitizer.  Juneau
65  	 * takes on no HTML-sanitizer dependency and makes no claim to sanitize hostile input on the caller's behalf:
66  	 * an application pointing this format at externally-authored HTML is expected to have run that HTML through
67  	 * a dedicated allowlist sanitizer at its own trust boundary first.  What this format guarantees is narrower
68  	 * and worth stating exactly: <i>if that upstream pass is wrong, a script still does not execute here.</i>
69  	 *
70  	 * <p>
71  	 * Fidelity is bounded by the allowlist.  Tags outside it are unwrapped (their children are kept), so an
72  	 * unexpected element degrades to its text rather than rendering &mdash; a body that needs an element this
73  	 * allowlist does not name will silently lose that element's markup, not its content.
74  	 *
75  	 * <p>
76  	 * Like {@link #MARKDOWN}, a field with this format spans full width.
77  	 */
78  	SANITIZED_HTML("sanitizedHtml");
79  
80  	private final String wire;
81  
82  	FieldFormat(String wire) {
83  		this.wire = wire;
84  	}
85  
86  	/**
87  	 * Returns the lowercase wire token for this format.
88  	 *
89  	 * @return The wire token (e.g. <c>"markdown"</c>).
90  	 */
91  	public String wire() {
92  		return wire;
93  	}
94  }
