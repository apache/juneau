17  package org.apache.juneau.console.views;
18  
19  
20  import org.apache.juneau.commons.bean.*;
21  import org.apache.juneau.marshall.*;
22  import static org.apache.juneau.commons.utils.Shorts.*;
23  
24  /**
25   * The "job accepted" envelope a row action's submit returns <b>instead of</b> a terminal {@link ActionResult} when
26   * the write is asynchronous.
27   *
28   * <h5 class='section'>Why this is not a new {@code RowAction} wire field</h5>
29   * <p>
30   * Whether an action is asynchronous is a property of the <b>response</b>, not of the declared action: the same
31   * {@link RowAction} POST returns either a terminal {@link ActionResult} (synchronous) or this pointer (asynchronous).
32   * So the {@code RowAction} wire schema is untouched &mdash; no new field, no fail-loud contract bump &mdash; and the
33   * {@code juneau-views.js} runtime routes to the streaming path purely by recognizing a {@link #streamUrl} in the 2xx
34   * body.  It carries no data to render, only capability pointers, so it needs no version handshake of its own.
35   *
36   * <h5 class='section'>The URLs are capabilities</h5>
37   * <p>
38   * {@link #streamUrl} points at the SSE progress stream and {@link #cancelUrl} at the cancel endpoint; both embed the
39   * job's unguessable {@value AsyncJobRegistry#CAPABILITY_BITS}-bit id (see {@link AsyncJob}), which is the whole of the
40   * access control.  The browser {@code EventSource} that consumes {@link #streamUrl} cannot set a CSRF header,
41   * which is precisely why the id must be unguessable rather than CSRF-gated.
42   *
43   * <h5 class='section'>See Also:</h5>
44   * <ul>
45   * 	<li class='jc'>{@link AsyncJobsMixin}
46   * 	<li class='jc'>{@link AsyncJob}
47   * 	<li class='jc'>{@link ActionResult}
48   * </ul>
49   *
50   * @since 10.0.0
51   */
52  @BeanType(properties="jobId,streamUrl,cancelUrl")
53  @SuppressWarnings({
54  	"java:S1845" // Fluent-builder setters intentionally mirror field names (Juneau DSL convention).
55  })
56  public class AsyncJobRef {
57  
58  	/** The job's unguessable capability id. */
59  	public String jobId;
60  
61  	/** The URL of the SSE progress stream (a browser {@code EventSource} target); embeds {@link #jobId}. */
62  	@Uri
63  	public String streamUrl;
64  
65  	/** The URL of the cancel endpoint (a non-safe POST); embeds {@link #jobId}. */
66  	@Uri
67  	public String cancelUrl;
68  
69  	/**
70  	 * Builds a reference to the given job using {@code servlet:}-relative URLs against the {@link AsyncJobsMixin}
71  	 * mount &mdash; the same {@code servlet:} convention {@code ViewsMixin} uses for its asset URLs. {@link #streamUrl}
72  	 * and {@link #cancelUrl} are {@link Uri &#64;Uri}-annotated, which is what opts them into resolution: Juneau's
73  	 * serializer only rewrites the {@code servlet:} pseudo-scheme for a property whose {@code ClassMeta} or
74  	 * {@code BeanPropertyMeta} reports {@code isUri()==true}, and a plain {@code String} field reports that only
75  	 * via an explicit {@code @Uri} (see {@code MarshalledPropertyPostProcessor}); without it the literal
76  	 * {@code "servlet:..."} string would reach the wire unresolved.  With the annotation present, every REST
77  	 * response serializer (JSON, HTML, XML, ...) resolves it against the current request's {@code UriContext}.
78  	 *
79  	 * @param job The started job.  Must not be <jk>null</jk>.
80  	 * @return A new reference carrying the job's id and its {@code servlet:}-relative stream + cancel URLs.
81  	 */
82  	public static AsyncJobRef of(AsyncJob job) {
83  		reqnn("job", job);
84  		var r = new AsyncJobRef();
85  		r.jobId = job.id();
86  		r.streamUrl = "servlet:" + AsyncJobsMixin.streamPath(job.id());
87  		r.cancelUrl = "servlet:" + AsyncJobsMixin.cancelPath(job.id());
88  		return r;
89  	}
90  
91  	/**
92  	 * Builds a reference carrying explicit stream/cancel URLs (for a consumer that resolves URLs itself, e.g. a
93  	 * template-rendered surface downstream of the {@code servlet:} rewrite).
94  	 *
95  	 * @param jobId The job's capability id.  Must not be <jk>null</jk> or blank.
96  	 * @param streamUrl The SSE stream URL.  Must not be <jk>null</jk> or blank.
97  	 * @param cancelUrl The cancel URL.  Must not be <jk>null</jk> or blank.
98  	 * @return A new reference.
99  	 */
100 	public static AsyncJobRef of(String jobId, String streamUrl, String cancelUrl) {
101 		var r = new AsyncJobRef();
102 		r.jobId = jobId;
103 		r.streamUrl = streamUrl;
104 		r.cancelUrl = cancelUrl;
105 		return r;
106 	}
107 }
