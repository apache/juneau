23  /**
24   * The author-facing declaration of a job-log row-detail region, compiled to a {@link RegionDef}.
25   *
26   * <p>
27   * The console's {@code job-log} card, hosted in a row's detail: a form whose submit runs a row-bound job through
28   * {@code JuneauViews.jobs}, streams its log, and optionally writes its result into a field.  {@link #toRegion()}
29   * produces a {@link RegionDef#TYPE_ROW_DETAIL} region with {@code populate("job-log")}, opts that populator in, and
30   * carries the whole declaration as {@link RegionDef#config}.  The populator ships in the console's {@code joblog}
31   * pack.
32   *
33   * <p>
34   * The submit always uses the action envelope {@code {action, targetId, idempotencyKey, fields}}: {@code targetId} is
35   * the row id, and the key comes from {@code submit.keyUrl}, an endpoint that answers
36   * {@link JobKeyLedger#response(IdempotencyKey)}.  The shape rules live in {@link JobLogChecks}, shared with the page
37   * card.
38   *
39   * <h5 class='section'>Example:</h5>
40   * <p class='bjava'>
41   * 	RegionDef <jv>region</jv> = JobLogRegion.<jsm>create</jsm>(<js>"restock-species"</js>)
42   * 		.submit(JobLogRegion.Submit.<jsm>create</jsm>(<js>"/jobs/species/{id}/restock"</js>, <js>"restock-species"</js>, <js>"/jobs/species/{id}/key"</js>)
43   * 			.submitLabel(<js>"Restock"</js>)
44   * 			.field(JobLogRegion.Field.<jsm>of</jsm>(<js>"count"</js>).label(<js>"How many"</js>).type(<js>"number"</js>).defaultValue(1))
45   * 			.field(JobLogRegion.Field.<jsm>of</jsm>(<js>"species"</js>).type(<js>"hidden"</js>).source(<js>"row.species"</js>)))
46   * 		.output(JobLogRegion.Output.<jsm>field</jsm>(<js>"available"</js>).from(<js>"row.available"</js>))
47   * 		.toRegion();
48   * </p>
49   *
50   * @since 10.0.0
51   */
52  @SuppressWarnings({
53  	"java:S1192" // Repeated literals are contract wire keys; named constants would obscure the contract shape.
54  })
55  public final class JobLogRegion {

57  	/** The populator name. */
58  	public static final String POPULATOR = "job-log";
59  
60  	/** The smallest accepted {@code height}, in pixels; matches the page card's floor. */
61  	public static final int MIN_HEIGHT = 80;
62  
63  	/** The top-level config keys, in emission order. */
64  	public static final List<String> KEYS = List.of("submit", "detail", "height", "clearOnSubmit", "output");
65  
66  	/** Region id. */
67  	public final String id;

72  		this.id = id;
73  	}

75  	/**
76  	 * Creates a declaration.
77  	 *
78  	 * @param id The region id.
79  	 * @return A new declaration.
80  	 */
81  	public static JobLogRegion create(String id) {
82  		return new JobLogRegion(id);
83  	}

85  	/**
86  	 * Creates a declaration from a parsed map (a datatables card's {@code detail.region.config}).
87  	 *
88  	 * @param id The region id.
89  	 * @param config The {@code config} object; keys must be a subset of {@link #KEYS}.
90  	 * @return A new declaration (not yet validated).
91  	 * @throws IllegalArgumentException If {@code config} is not an object or has an unknown key.
92  	 */
93  	public static JobLogRegion fromMap(String id, Map<String,?> config) {
94  		var d = create(id);
95  		if (config == null)
96  			return d;
97  		for (var e : config.entrySet()) {
98  			var k = e.getKey();
99  			if ("urlParam".equals(k))
100 				throw iaex("JobLogRegion '%s' urlParam is not allowed in a row-detail region; a page reload does not reattach.", id);
101 			if (! KEYS.contains(k))
102 				throw iaex("JobLogRegion '%s' unknown key '%s'; allowed: %s.", id, k, KEYS);
103 			d.config.put(k, e.getValue());
104 		}
105 		return d;
106 	}

119 	/**
120 	 * Validates this declaration (fail-closed).
121 	 *
122 	 * @return This object.
123 	 * @throws IllegalArgumentException If a rule is broken.
124 	 */
125 	public JobLogRegion validate() {
126 		if (ib(id))
127 			throw iaex("JobLogRegion id must not be blank.");
128 		try {
129 			if (! (config.get("submit") instanceof Map<?,?> submit))
130 				throw iaex("requires a 'submit' object.");
131 			JobLogChecks.submit(submit, JobLogChecks.Mode.REGION);
132 			if (config.containsKey("detail"))
133 				JobLogChecks.detail(config.get("detail"));
134 			JobLogChecks.height(config.get("height"), MIN_HEIGHT);
135 			JobLogChecks.clearOnSubmit(config.get("clearOnSubmit"));
136 			if (config.containsKey("output"))
137 				JobLogChecks.output(config.get("output"));
138 		} catch (IllegalArgumentException e) {
139 			throw iaex("JobLogRegion '%s' %s", id, e.getMessage());
140 		}
141 		return this;
142 	}
143 
144 	/**
145 	 * Returns the config in canonical key order, so the builder and the console-card path emit the same sidecar.
146 	 *
147 	 * @return A new ordered map.
148 	 */
149 	public Map<String,Object> toMap() {
150 		var m = order(config, KEYS);
151 		if (m.get("submit") instanceof Map<?,?> s) {
152 			// A region always uses the action envelope; stamp it so an authored config that left it implicit emits the
153 			// same sidecar as the builder.
154 			var withEnvelope = new LinkedHashMap<Object,Object>(s);
155 			withEnvelope.putIfAbsent("envelope", "action");
156 			var submit = order(withEnvelope, JobLogChecks.REGION_SUBMIT_KEYS);
157 			if (submit.get("fields") instanceof List<?> l)
158 				submit.put("fields", l.stream().map(f -> f instanceof Map<?,?> fm ? order(fm, JobLogChecks.FIELD_KEYS) : f).toList());
159 			m.put("submit", submit);
160 		}
161 		if (m.get("detail") instanceof Map<?,?> d)
162 			m.put("detail", order(d, JobLogChecks.DETAIL_KEYS));
163 		if (m.get("output") instanceof Map<?,?> o) {
164 			var output = order(o, JobLogChecks.OUTPUT_KEYS);
165 			if (output.get("target") instanceof Map<?,?> t)
166 				output.put("target", order(t, List.of("field", "selector")));
167 			m.put("output", output);
168 		}
169 		return m;
170 	}
171 
172 	private static Map<String,Object> order(Map<?,?> in, List<String> keys) {
173 		var m = new LinkedHashMap<String,Object>();
174 		for (var k : keys)
175 			if (in.containsKey(k))
176 				m.put(k, in.get(k));
177 		for (var e : in.entrySet())
178 			if (! m.containsKey(String.valueOf(e.getKey())))
179 				m.put(String.valueOf(e.getKey()), e.getValue());
180 		return m;
181 	}
182 
183 	/**
184 	 * Validates and compiles this declaration.
185 	 *
186 	 * @return A validated row-detail region.
187 	 */
188 	public RegionDef toRegion() {
189 		validate();
190 		var r = RegionDef.create(id).type(RegionDef.TYPE_ROW_DETAIL).populate(POPULATOR).allowPopulators(POPULATOR).config(toMap());
191 		r.validate();
192 		return r;
193 	}

195 	/** The {@code submit} declaration. */
196 	public static final class Submit {
197 		final Map<String,Object> map = new LinkedHashMap<>();
198 
199 		private Submit() {}
200 
201 		/**
202 		 * Creates a submit declaration.
203 		 *
204 		 * @param endpoint The same-origin submit path template; {@code {id}} is the only token.
205 		 * @param action The action id the key is bound to.
206 		 * @param keyUrl The same-origin key-endpoint path template; {@code {id}} is the only token.
207 		 * @return A new declaration.
208 		 */
209 		public static Submit create(String endpoint, String action, String keyUrl) {
210 			var s = new Submit();
211 			s.map.put("endpoint", endpoint);
212 			s.map.put("envelope", "action");
213 			s.map.put("action", action);
214 			s.map.put("keyUrl", keyUrl);
215 			return s;
216 		}
217 
218 		/** @param value {@code POST} (the default) or {@code PUT}. @return This object. */
219 		public Submit method(String value) { map.put("method", value); return this; }
220 		/** @param value The submit button's label. @return This object. */
221 		public Submit submitLabel(String value) { map.put("submitLabel", value); return this; }
222 
223 		/**
224 		 * Adds a field.
225 		 *
226 		 * @param value The field.
227 		 * @return This object.
228 		 */
229 		@SuppressWarnings({
230 			"unchecked" // The "fields" entry is always created here as a List<Object>.
231 		})
232 		public Submit field(Field value) {
233 			((List<Object>)map.computeIfAbsent("fields", k -> new ArrayList<>())).add(value.map);
234 			return this;
235 		}
236 	}

268 	/** The optional field grid painted above the form, whose slots {@code output.target.field} can address. */
269 	public static final class Detail {
270 		final Map<String,Object> map = new LinkedHashMap<>();
271 
272 		private Detail() {}
273 
274 		/**
275 		 * @param renderer The region helper that paints the grid, for example {@code fieldGrid}.
276 		 * @param fields The field catalog.
277 		 * @return A new grid declaration.
278 		 */
279 		public static Detail of(String renderer, RegionDef.Field...fields) {
280 			var d = new Detail();
281 			d.map.put("renderer", renderer);
282 			d.map.put("fields", Arrays.stream(fields).map(RegionDef.Field::toContractMap).toList());
283 			return d;
284 		}
285 	}
