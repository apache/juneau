17  package org.apache.juneau.console.views;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import java.util.*;
22  
23  /**
24   * Serializes a {@link TableSpec} into the nested-table slot a tabbed {@link RowDetail} carries.
25   *
26   * <p>
27   * The slot holds the table shell ({@code view}), the sidecar metadata the client builds the nested table from
28   * ({@code meta}) and, when declared, the nested table's selection and row detail.  A nested table sits at depth two;
29   * a table that itself declares a nested-table tab would sit at depth three and is rejected.
30   */
31  @SuppressWarnings({
32  	"java:S1192" // Map keys read more clearly inline than as constants.
33  })
34  final class NestedSlot {
35  
36  	/** The deepest table nesting: the page table is depth one, a table inside its row detail is depth two. */
37  	static final int MAX_NESTED_DEPTH = 2;
38  
39  	private NestedSlot() {}
40  
41  	/**
42  	 * Builds the nested slot.
43  	 *
44  	 * @param view The nested table.  Must declare a rows data URL and at least one column.
45  	 * @param scopeParam The query parameter carrying the parent row id.
46  	 * @return The slot map, in fixed key order.
47  	 * @throws IllegalArgumentException If the table is unusable or would exceed {@link #MAX_NESTED_DEPTH}.
48  	 */
49  	@SuppressWarnings({
50  		"java:S3776" // Branching is inherent to of(), which validates and reshapes the nested table body.
51  	})
52  	static Map<String,Object> of(TableSpec view, String scopeParam) {
53  		if (view == null)
54  			throw iaex("A nested table is required.");
55  		var body = view.toCardBody();
56  		if (ib((String)body.get("dataUrl")))
57  			throw iaex("Nested table '%s' needs a rows data URL.", view.id());
58  		view.validate();
59  		var detail = body.get("detail");
60  		if (detail instanceof Map<?,?> dm && dm.get("tabs") instanceof List<?> tl)
61  			for (var t : tl)
62  				if (t instanceof Map<?,?> tm && "table".equals(tm.get("kind")))
63  					throw iaex("Nested table '%s' declares a nested-table tab; tables nest at most %s deep (MAX_NESTED_DEPTH).",
64  						view.id(), MAX_NESTED_DEPTH);
65  
66  		var cols = new ArrayList<Map<String,Object>>();
67  		for (var c : (List<?>)body.get("columns")) {
68  			var cm = new LinkedHashMap<String,Object>();
69  			for (var e : ((Map<?,?>)c).entrySet()) {
70  				var k = (String)e.getKey();
71  				var renamed = k;
72  				if ("key".equals(k))
73  					renamed = "data";
74  				else if ("label".equals(k))
75  					renamed = "title";
76  				cm.put(renamed, e.getValue());
77  			}
78  			cols.add(cm);
79  		}
80  
81  		var shell = new LinkedHashMap<String,Object>();
82  		shell.put("id", view.id());
83  		shell.put("columns", cols);
84  
85  		var meta = new LinkedHashMap<String,Object>();
86  		meta.put("contractVersion", ViewsMixin.CONTRACT_VERSION);
87  		meta.put("id", view.id());
88  		for (var e : body.entrySet()) {
89  			var k = e.getKey();
90  			if (! NOT_META.contains(k) && ! "columns".equals(k))
91  				meta.put(k, e.getValue());
92  		}
93  		meta.put("columns", cols);
94  
95  		var n = new LinkedHashMap<String,Object>();
96  		n.put("contractVersion", "2");
97  		n.put("parentScopeParam", scopeParam);
98  		n.put("view", shell);
99  		n.put("meta", meta);
100 		if (body.get("selection") != null)
101 			n.put("selection", body.get("selection"));
102 		if (detail != null)
103 			n.put("detail", detail);
104 		return n;
105 	}
106 
107 	private static final Set<String> NOT_META = Set.of("selection", "detail", "bulk", "model", "quickStats", "columnConfig",
108 		"pollIntervalMs");
109 }
