25  /**
26   * A count badge in the page contract, fed by a JSON endpoint and polled by {@code juneau-badges.js}.
27   *
28   * <p>
29   * The endpoint returns {@code {total, mine?, items?[]}}.  The badge is presentation only: the endpoint must
30   * still authorize the caller, and a page should not declare a badge for users who cannot read it.
31   *
32   * <h5 class='section'>Example:</h5>
33   * <p class='bjava'>
34   * 	<jk>var</jk> <jv>badge</jv> = BadgeDef
35   * 		.<jsm>create</jsm>(<js>"pending"</js>)
36   * 		.src(<js>"/rest/changes/pending"</js>)
37   * 		.scope(<js>"beanType"</js>, BadgeDef.ScopeBy.<jsf>NAV</jsf>, Map.<jsm>of</jsm>(<js>"suspensions"</js>, List.<jsm>of</jsm>(<js>"Suspension"</js>)))
38   * 		.href(<js>"/ui/changes"</js>)
39   * 		.stateFilter(<js>"beanType"</js>)
40   * 		.refreshes(<js>"rules-table"</js>)
41   * 		.refreshMs(60_000);
42   * 	<jv>badge</jv>.validate();  <jc>// throws IllegalArgumentException with an E-6x message</jc>
43   * </p>
44   *
45   * @since 10.0.0
46   */
47  @SuppressWarnings({
48  	"java:S3776" // Validation and rendering chains many independent guards; splitting them adds no clarity.
49  })
50  public final class BadgeDef {
51  
52  	/** The minimum poll interval, in milliseconds ({@code refreshMs.minimum}). */
53  	public static final long MIN_REFRESH_MS = 5000L;
54  
55  	/** Default poll interval, in milliseconds, applied by {@code juneau-badges.js} when {@link #refreshMs} is unset. */
56  	public static final long DEFAULT_REFRESH_MS = 60_000L;
57  
58  	/** Default tooltip item cap, applied by {@code juneau-badges.js} when {@link #tooltipMax} is unset. */
59  	public static final int DEFAULT_TOOLTIP_MAX = 10;
60  
61  	/** {@code scope.by} wire tokens. */
62  	public enum ScopeBy {
63  		/** Keys on the leaf of the page's {@code activeNav}. */
64  		NAV("nav"),
65  		/** Keys on the active tab of the card named by {@link BadgeDef#table}. */
66  		VIEW("view");
67  
68  		private final String wire;
69  		ScopeBy(String wire) { this.wire = wire; }
70  		/** @return The lowercase wire token (e.g. <c>"nav"</c>). */
71  		public String wire() { return wire; }
72  	}
73  
74  	/** {@code placement} wire tokens. */
75  	public enum Placement {
76  		/** Renders into {@code .jc-header-badges} at the end of the header actions area. */
77  		HEADER("header"),
78  		/** Renders at the right end of {@link BadgeDef#table}'s ribbon. */
79  		TOOLBAR("toolbar");
80  
81  		private final String wire;
82  		Placement(String wire) { this.wire = wire; }
83  		/** @return The lowercase wire token (e.g. <c>"header"</c>). */
84  		public String wire() { return wire; }
85  	}
86  
87  	/** {@code tone} wire tokens. */
88  	public enum Tone {
89  		/** Informational tone. */
90  		INFO("info"),
91  		/** Warning tone (the default). */
92  		WARNING("warning"),
93  		/** Danger tone. */
94  		DANGER("danger");
95  
96  		private final String wire;
97  		Tone(String wire) { this.wire = wire; }
98  		/** @return The lowercase wire token (e.g. <c>"warning"</c>). */
99  		public String wire() { return wire; }
100 	}

122 	/**
123 	 * Creates a badge with the given id.
124 	 *
125 	 * @param id This badge's own id, unique across every Java- and FTL-declared badge on the page (C3 E-B15).
126 	 * 	Must not be <jk>null</jk> or blank.
127 	 * @return A new {@link BadgeDef}.
128 	 * @throws IllegalArgumentException If {@code id} is <jk>null</jk> or blank.
129 	 */
130 	public static BadgeDef create(String id) {
131 		if (ib(id))
132 			throw iaex("BadgeDef id must not be null or blank.");
133 		var b = new BadgeDef();
134 		b.id = id;
135 		return b;
136 	}

256 	/**
257 	 * Sets card ids, added to {@link #refreshes}, that are allowed to be absent from the page being rendered.
258 	 *
259 	 * <p>
260 	 * A page normally rejects a {@code refreshes} id that names no card on it, which catches typos at build time.  Use this
261 	 * for a header badge that refreshes a card living on another page: the client skips a card that is not mounted.  The

263 	 * {@code refreshes} / {@code stateFilter} noun style of this builder: the opt-in is per id, not a page-wide switch.
264 	 *
265 	 * @param value The card ids that may live on another page.
266 	 * @return This object.
267 	 */

318 	/**
319 	 * Validates this definition.
320 	 *
321 	 * <p>
322 	 * Enforces E-61 (no {@link #src}), a same-origin {@link #src} and {@link #href}, E-62 ({@link #refreshMs} below {@link #MIN_REFRESH_MS}), E-66
323 	 * ({@link Placement#TOOLBAR} without {@link #table}), and the view-scope rule (uncoded: {@link ScopeBy#VIEW}
324 	 * without
325 	 * {@link #table}).  The "table/refreshes name a real card id" rule and the "scope.values key is a real nav
326 	 * node id" need the rest of the page and run in {@code PageCapture} at {@code </@console>} instead.
327 	 *
328 	 * @throws IllegalArgumentException If this definition is not well-formed.
329 	 */
330 	public void validate() {
331 		if (ib(src))
332 			throw iaex("<@badge id='%s'> requires src=.", id);
333 		if (! SafePathTemplate.isSameOriginPath(src))
334 			throw iaex("<@badge id='%s'> src must be a same-origin path; got '%s'.", id, src);
335 		if (href != null && ! SafePathTemplate.isSameOriginPath(href))
336 			throw iaex("<@badge id='%s'> href must be a same-origin path; got '%s'.", id, href);
337 		if (refreshMs != null && refreshMs < MIN_REFRESH_MS)
338 			throw iaex("'%s' refreshMs='%s' is below the minimum '%s'.", id, refreshMs, MIN_REFRESH_MS);
339 		if (placement == Placement.TOOLBAR && (ib(table)))
340 			throw iaex("<@badge id='%s'> placement='toolbar' requires table=.", id);
341 		if (scopeBy == ScopeBy.VIEW && (ib(table)))
342 			throw iaex("BadgeDef '%s' scope.by='view' requires table=.", id);
343 	}

345 	/**
346 	 * Serializes this definition to the compact JSON the {@code header.badges[]} contract carries, hand-built
347 	 * (rather than reflected) so the key set and order match the schema exactly.
348 	 *
349 	 * @return An ordered {@link Map} ready for {@code Json.of(...)}.
350 	 */
351 	public Map<String,Object> toContractMap() {
352 		var m = new LinkedHashMap<String,Object>();
353 		m.put("id", id);
354 		if (src != null)
355 			m.put("src", src);
356 		if (params != null && !params.isEmpty())
357 			m.put("params", params);
358 		if (refreshMs != null)
359 			m.put("refreshMs", refreshMs);
360 		if (scopeParam != null) {
361 			var scope = new LinkedHashMap<String,Object>();
362 			scope.put("param", scopeParam);
363 			scope.put("by", (scopeBy == null ? ScopeBy.NAV : scopeBy).wire());
364 			if (scopeValues != null && !scopeValues.isEmpty())
365 				scope.put("values", scopeValues);
366 			m.put("scope", scope);
367 		}
368 		if (label != null && !label.isEmpty())
369 			m.put("label", label);
370 		if (tooltipMax != null)
371 			m.put("tooltipMax", tooltipMax);
372 		if (href != null)
373 			m.put("href", href);
374 		if (stateFilterColumn != null)
375 			m.put("stateFilter", Map.of("column", stateFilterColumn));
376 		if (refreshes != null && !refreshes.isEmpty() || refreshesAllowMissing != null && !refreshesAllowMissing.isEmpty()) {
377 			var all = new LinkedHashSet<String>();
378 			if (refreshes != null)
379 				all.addAll(refreshes);
380 			if (refreshesAllowMissing != null)
381 				all.addAll(refreshesAllowMissing);
382 			m.put("refreshes", new ArrayList<>(all));
383 		}
384 		if (refreshesAllowMissing != null && !refreshesAllowMissing.isEmpty())
385 			m.put("refreshesAllowMissing", refreshesAllowMissing);
386 		if (placement != null)
387 			m.put("placement", placement.wire());
388 		if (table != null)
389 			m.put("table", table);
390 		if (tone != null)
391 			m.put("tone", tone.wire());
392 		if (visibleWhen != null && !visibleWhen.isEmpty())
393 			m.put("visibleWhen", VisibilityRule.toMaps(visibleWhen));
394 		return m;
395 	}
