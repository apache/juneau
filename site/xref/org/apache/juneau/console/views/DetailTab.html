27  /**
28   * One tab of a tabbed {@link RowDetail}: a fields grid, a nested table, a chart or a client-rendered pane.
29   *
30   * <p>
31   * The tab is loaded lazily: its content is fetched the first time the user opens it.  A tab id is the slug of its
32   * title unless {@link #id(String)} is given.
33   *
34   * <h2 class='section'>Example:</h2>
35   * <p class='bjava'>
36   * 	RowDetail.<jsm>inline</jsm>()
37   * 		.tab(DetailTab.<jsm>fields</jsm>(<js>"Details"</js>, <js>"id"</js>, <js>"name"</js>, <js>"price"</js>).columns(3))
38   * 		.tab(DetailTab.<jsm>client</jsm>(<js>"Labels"</js>, <js>"pet-labels"</js>).endpoint(<js>"/pets/{id}/labels"</js>));
39   * </p>
40   *
41   * @since 10.0.0
42   */
43  @SuppressWarnings({
44  	"java:S1192", // Map keys read more clearly inline than as constants.
45  	"java:S3776" // Branching is inherent to validate and toMap, which check many independent tab settings.

325 	String resolvedId() {
326 		return id != null ? id : slug(title);
327 	}

329 	static String slug(String t) {
330 		var s = t.toLowerCase(Locale.ROOT).replaceAll("[^a-z0-9]+", "-");
331 		var b = 0;
332 		var e = s.length();
333 		while (b < e && s.charAt(b) == '-')
334 			b++;
335 		while (e > b && s.charAt(e - 1) == '-')
336 			e--;
337 		return s.substring(b, e);
338 	}

356 	private void validate() {
357 		var rid = resolvedId();
358 		if (! ID_GRAMMAR.matcher(rid).matches())
359 			throw iaex("DetailTab '%s' id '%s' must match %s.", title, rid, ID_GRAMMAR.pattern());
360 		if (endpoint != null && ! RegionDef.isSafeDetailEndpoint(endpoint))
361 			throw iaex("DetailTab '%s' endpoint '%s' is not a same-origin path.", rid, endpoint);
362 		if (countFromRows && kind != Kind.TABLE)
363 			throw iaex("DetailTab '%s' countFromRows applies to table tabs only.", rid);
364 		switch (kind) {
365 			case FIELDS -> validateFields(rid);
366 			case TABLE -> {
367 				if (view == null)
368 					throw iaex("DetailTab '%s' needs a nested table.", rid);
369 				if (ib(scopeParam))
370 					throw iaex("DetailTab '%s' scopeParam must not be blank.", rid);
371 			}
372 			case CHART -> {
373 				if (region == null)
374 					throw iaex("DetailTab '%s' needs a chart region.", rid);
375 			}
376 			case CLIENT -> {
377 				if (renderer == null || ! RENDERER_GRAMMAR.matcher(renderer).matches())
378 					throw iaex("DetailTab '%s' renderer '%s' must match %s.", rid, renderer, RENDERER_GRAMMAR.pattern());
379 			}
380 		}
381 	}
382 
383 	private void validateFields(String rid) {
384 		if (columns != 0 && (columns < 1 || columns > 4))
385 			throw iaex("DetailTab '%s' columns must be 1 to 4, was %s.", rid, columns);
386 		if (fields.isEmpty() && ! autoFields)
387 			throw iaex("DetailTab '%s' needs at least one field.", rid);
388 		var seen = new HashSet<String>();
389 		for (var f : fields)
390 			if (ib(f) || ! seen.add(f))
391 				throw iaex("DetailTab '%s' field '%s' is blank or repeated.", rid, f);
392 		var grouped = new HashSet<String>();
393 		for (var g : groups)
394 			for (var f : g.fields()) {
395 				if (! seen.contains(f))
396 					throw iaex("DetailTab '%s' group '%s' names field '%s' which is not in the tab.", rid, g.title(), f);
397 				if (! grouped.add(f))
398 					throw iaex("DetailTab '%s' field '%s' is in more than one group.", rid, f);
399 			}
400 	}

402 	/**
403 	 * Serializes this tab to its wire shape.
404 	 *
405 	 * @return The tab map.
406 	 * @throws IllegalArgumentException If the tab is invalid.
407 	 */
408 	public Map<String,Object> toMap() {
409 		validate();
410 		var m = new LinkedHashMap<String,Object>();
411 		m.put("id", resolvedId());
412 		m.put("kind", kind.wire());
413 		m.put("title", title);
414 		if (icon != null)
415 			m.put("icon", icon);
416 		if (kind == Kind.FIELDS) {
417 			if (columns != 0)
418 				m.put("columns", columns);
419 			m.put("fields", List.copyOf(fields));
420 			if (! groups.isEmpty()) {
421 				var gl = new ArrayList<Map<String,Object>>();
422 				for (var g : groups) {
423 					var gm = new LinkedHashMap<String,Object>();
424 					gm.put("title", g.title());
425 					gm.put("fields", g.fields());
426 					gl.add(gm);
427 				}
428 				m.put("groups", gl);
429 			}
430 		}
431 		if (kind == Kind.TABLE)
432 			m.put("nested", NestedSlot.of(view, scopeParam));
433 		if (countFromRows)
434 			m.put("count", "rows");
435 		if (emptyText != null)
436 			m.put("emptyText", emptyText);
437 		if (endpoint != null)
438 			m.put("endpoint", endpoint);
439 		if (renderer != null && kind == Kind.CLIENT)
440 			m.put("renderer", renderer);
441 		if (refreshOnPoll)
442 			m.put("refreshOnPoll", true);
443 		if (kind == Kind.CHART) {
444 			var r = new LinkedHashMap<>(region.toContractMap());
445 			r.remove("contractVersion");
446 			m.put("region", r);
447 		}
448 		if (visibleWhen != null && ! visibleWhen.isEmpty())
449 			m.put("visibleWhen", VisibilityRule.toMaps(visibleWhen));
450 		model.validate();
451 		var mm = model.toMap();
452 		if (mm != null)
453 			m.put("model", mm);
454 		return m;
455 	}
