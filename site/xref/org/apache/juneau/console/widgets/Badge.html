17  package org.apache.juneau.console.widgets;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import org.apache.juneau.commons.bean.*;
22  
23  /**
24   * An overlay badge on a header action (a bell count, for example) or a {@link BarBadge}.
25   *
26   * <p>
27   * A badge is exactly one of a <b>count</b> badge ({@link #count}) or a <b>dot</b> badge ({@link #dot}); it never
28   * carries both.  The optional {@link #max} clamps a large count to <js>"&lt;max&gt;+"</js> and {@link #tone} selects
29   * the badge colour from the shared {@link StatusTone} palette.  A count is painted with {@code textContent}
30   * &mdash; never {@code innerHTML}.
31   *
32   * <p>
33   * The tone is serialized as its enum name (e.g. <js>"WARNING"</js>); the client lowercases it into the
34   * {@code data-juneau-badge-tone} attribute.  Colours map as:
35   * <ul>
36   * 	<li>{@link StatusTone#INFO} &mdash; blue
37   * 	<li>{@link StatusTone#SUCCESS} &mdash; green
38   * 	<li>{@link StatusTone#WARNING} &mdash; amber
39   * 	<li>{@link StatusTone#ERROR} &mdash; red
40   * 	<li>{@link StatusTone#NEUTRAL} &mdash; neutral
41   * </ul>
42   *
43   * <h5 class='section'>Example:</h5>
44   * <p class='bjava'>
45   * 	<jc>// A red unread-count badge that caps at "99+".</jc>
46   * 	Badge <jv>badge</jv> = Badge.<jsm>count</jsm>(120).tone(StatusTone.<jsf>ERROR</jsf>).max(99).label(<js>"unread notifications"</js>);
47   * 	<jv>badge</jv>.validate();
48   * </p>
49   *
50   * @since 10.0.0
51   */
52  @BeanType(properties="count,dot,max,tone,label")
53  @SuppressWarnings({
54  	"java:S1845" // Fluent-builder setters intentionally mirror field names (Juneau DSL convention).
55  })
56  public class Badge {
57  
58  	/** A count badge value (mutually exclusive with {@link #dot}); when set must be {@code >= 0}. */
59  	public Integer count;
60  
61  	/** A dot (presence) badge with no number (mutually exclusive with {@link #count}). */
62  	public Boolean dot;
63  
64  	/** Optional display clamp; a count above this renders as <js>"&lt;max&gt;+"</js>.  When set must be {@code >= 1}. */
65  	public Integer max;
66  
67  	/** Optional colour tone; <jk>null</jk> leaves the badge in its default colour. */
68  	public StatusTone tone;
69  
70  	/** Optional visually-hidden context for screen readers, e.g. <js>"unread notifications"</js>. */
71  	public String label;
72  
73  	/**
74  	 * Creates a count badge.
75  	 *
76  	 * @param value The count.  Must be {@code >= 0}.
77  	 * @return A new {@link Badge}.
78  	 */
79  	public static Badge count(int value) {
80  		var b = new Badge();
81  		b.count = value;
82  		return b;
83  	}
84  
85  	/**
86  	 * Creates a dot (presence) badge with no number.
87  	 *
88  	 * @return A new {@link Badge}.
89  	 */
90  	public static Badge dot() {
91  		var b = new Badge();
92  		b.dot = true;
93  		return b;
94  	}
95  
96  	/**
97  	 * Sets the color tone.
98  	 *
99  	 * @param value The tone.  Can be <jk>null</jk> to use the default colour.
100 	 * @return This object.
101 	 */
102 	public Badge tone(StatusTone value) {
103 		tone = value;
104 		return this;
105 	}
106 
107 	/**
108 	 * Sets the display clamp.
109 	 *
110 	 * @param value The clamp.  Must be {@code >= 1}.
111 	 * @return This object.
112 	 */
113 	public Badge max(int value) {
114 		max = value;
115 		return this;
116 	}
117 
118 	/**
119 	 * Sets the visually-hidden screen-reader context.
120 	 *
121 	 * @param value The label.
122 	 * @return This object.
123 	 */
124 	public Badge label(String value) {
125 		label = value;
126 		return this;
127 	}
128 
129 	/**
130 	 * Fail-closed bean validation; called by the enclosing bean's {@code validate()}.
131 	 *
132 	 * @throws IllegalArgumentException If this badge is not well-formed.
133 	 */
134 	public void validate() {
135 		var hasCount = count != null;
136 		var hasDot = dot != null && dot;
137 		if (hasCount == hasDot)
138 			throw iaex("Badge must declare exactly one of count/dot.");
139 		if (hasCount && count < 0)
140 			throw iaex("Badge count must be >= 0.");
141 		if (max != null && max < 1)
142 			throw iaex("Badge max must be >= 1 when set.");
143 	}
144 }
