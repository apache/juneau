26  /**
27   * The open set of server-side card type handlers. {@link #standard()} holds the built-ins ({@code html}),
28   * every {@link CardTypeHandler} found through {@link java.util.ServiceLoader} (this is how
29   * console-ui-freemarker contributes {@code datatables} and {@code terminal}), and falls back to the
30   * generic passthrough for unknown types. {@link #standard()} is immutable; a host that adds its own types
31   * works on a {@link #copy()}, so one host's types never leak into another's.
32   *
33   * <h5 class='section'>Example:</h5>
34   * <p class='bjava'>
35   *   CardTypeRegistry <jv>r</jv> = CardTypeRegistry.<jsm>standard</jsm>().copy()
36   *     .add(<jk>new</jk> KpiCardType())          <jc>// E-21 bad name, E-22 reserved, E-20 duplicate</jc>
37   *     .build();
38   *   JsonMap <jv>card</jv> = <jv>r</jv>.toCard(<jv>source</jv>);   <jc>// base keys + fragment, reserved-key check</jc>
39   * </p>
40   *
41   * @since 10.0.0
42   */
43  @SuppressWarnings({
44  	"java:S1192" // Repeated literals are contract wire keys; named constants would obscure the contract shape.
45  })
46  public final class CardTypeRegistry {
47  
48  	/**
49  	 * Reserved type names: {@code html} (built in), {@code datatables}, {@code terminal-view},
50  	 * {@code terminal}, {@code chart} and {@code job-log} (console-ui-freemarker).
51  	 */
52  	public static final Set<String> RESERVED = Set.of("html", "datatables", "terminal-view", "terminal", "chart", "job-log");
53  
54  	/** The one simple class name allowed to bind each reserved type (E-22's identity check). */
55  	private static final Map<String,String> RESERVED_OWNER_SIMPLE_NAMES = Map.of(
56  		"html", "HtmlCardType",
57  		"datatables", "DatatablesCardType",
58  		"terminal-view", "TerminalViewCardType",
59  		"terminal", "TerminalCardType",
60  		"chart", "ChartCardType",
61  		"job-log", "JobLogCardType"
62  	);
63  
64  	/** Reserved body keys (E-23); {@code visibleWhen}, {@code subscribes} and {@code publishes} are deliberately excluded — admitted in any card's body. */
65  	private static final Set<String> BASE_KEYS = Set.of("id", "type", "title", "src", "template", "ref");
66  
67  	private static final CardTypeRegistry STANDARD = buildStandard();
68  
69  	private final Map<String,CardTypeHandler> handlers;
70  	private final CardTypeHandler generic = new GenericCardType();
71  
72  	private CardTypeRegistry(Map<String,CardTypeHandler> handlers) {
73  		this.handlers = handlers;
74  	}
75  
76  	private static CardTypeRegistry buildStandard() {
77  		var b = new Builder();
78  		b.add(new HtmlCardType());
79  		for (var h : ServiceLoader.load(CardTypeHandler.class))
80  			b.add(h);
81  		return b.build();
82  	}
83  
84  	/** @return The standard registry: {@code html} plus every ServiceLoader-discovered handler. */
85  	public static CardTypeRegistry standard() {
86  		return STANDARD;
87  	}
88  
89  	/** @return A {@link Builder} pre-seeded with this registry's handlers, to add more on top. */
90  	public Builder copy() {
91  		var b = new Builder();
92  		b.handlers.putAll(handlers);
93  		return b;
94  	}
95  
96  	/**
97  	 * @param type The card type name.
98  	 * @return The registered handler for {@code type}, or the generic passthrough when none is registered.
99  	 */
100 	public CardTypeHandler handler(String type) {
101 		return handlers.getOrDefault(type, generic);
102 	}
103 
104 	/**
105 	 * @param type The card type name.
106 	 * @return <jk>true</jk> if a handler is registered for {@code type} (not counting the generic fallback).
107 	 */
108 	public boolean isRegistered(String type) {
109 		return handlers.containsKey(type);
110 	}
111 
112 	/** @return Every registered type name. */
113 	public Set<String> types() {
114 		return Collections.unmodifiableSet(handlers.keySet());
115 	}
116 
117 	/**
118 	 * The single server path for both {@code <@card>} and C3's {@code PageSpec}: validates the body (E-23),
119 	 * dispatches to {@link #handler(String)}, and merges the fragment flat onto the card's base keys.
120 	 * A JSON body's {@code visibleWhen}, {@code subscribes} and {@code publishes} are promoted onto the card itself
121 	 * (the handler never sees {@code subscribes}/{@code publishes}).
122 	 *
123 	 * @param source The authored card.
124 	 * @return The card object: base keys plus the handler's fragment, flat.
125 	 * @throws IllegalArgumentException E-23 if the body sets a reserved key; if {@code src} is not a same-origin path
126 	 * 	(see {@link SafePathTemplate#isSameOriginPath(String)}; a {@code {token}} placeholder is allowed); or from the handler itself.
127 	 */
128 	public JsonMap toCard(CardSource source) {
129 		var src = source.src();
130 		if (src != null && ! SafePathTemplate.isSameOriginPath(src))
131 			throw source.error("type='%s' src must be a same-origin path (no absolute URL, '//', scheme, '..', backslash, control character, or leading/trailing space); got '%s'.",
132 				source.type(), src);
133 		Object visibleWhen = null;
134 		Object subscribes = null;
135 		Object publishes = null;
136 		if (source.hasJsonBody()) {
137 			var body = source.json();
138 			for (var key : body.keySet()) {
139 				if (BASE_KEYS.contains(key))
140 					throw source.error("body key '%s' is reserved; set it as an attribute.", key);
141 			}
142 			visibleWhen = body.get("visibleWhen");
143 			subscribes = body.remove("subscribes");
144 			publishes = body.remove("publishes");
145 		}
146 
147 		var frag = handler(source.type()).toFragment(source);
148 
149 		var card = new JsonMap();
150 		card.put("id", source.id());
151 		card.put("type", source.type());
152 		if (source.title() != null)
153 			card.put("title", source.title());
154 		if (source.src() != null)
155 			card.put("src", source.src());
156 		if (source.template() != null)
157 			card.put("template", source.template());
158 		if (visibleWhen != null)
159 			card.put("visibleWhen", visibleWhen);
160 		if (subscribes != null)
161 			card.put("subscribes", subscribes);
162 		if (publishes != null)
163 			card.put("publishes", publishes);
164 		card.putAll(frag);
165 		return card;
166 	}
167 
168 	/** Builds a {@link CardTypeRegistry} on top of an existing one, or from scratch. */
169 	public static final class Builder {
170 		private final Map<String,CardTypeHandler> handlers = new LinkedHashMap<>();
171 
172 		/**
173 		 * Registers a handler.
174 		 *
175 		 * @param h The handler.
176 		 * @return This object.
177 		 * @throws IllegalArgumentException E-21 if {@link CardTypeHandler#type()} doesn't match the name pattern;
178 		 * 	E-22 if it names a reserved type this class doesn't own; E-20 if the type is already registered.
179 		 */
180 		public Builder add(CardTypeHandler h) {
181 			var type = h.type();
182 			if (type == null || ! type.matches("^[a-z][a-z0-9-]{0,31}$"))
183 				throw iaex("Card type '%s' must match ^[a-z][a-z0-9-]{0,31}$.", type);
184 			if (RESERVED.contains(type) && ! h.getClass().getSimpleName().equals(RESERVED_OWNER_SIMPLE_NAMES.get(type)))
185 				throw iaex("Card type '%s' is reserved and cannot be replaced; reserved: '%s'.",
186 					type, String.join(", ", new TreeSet<>(RESERVED)));
187 			if (handlers.containsKey(type))
188 				throw iaex("Card type '%s' is already registered by '%s'.", type, handlers.get(type).getClass().getName());
189 			handlers.put(type, h);
190 			return this;
191 		}
192 
193 		/** @return The built, immutable {@link CardTypeRegistry}. */
194 		public CardTypeRegistry build() {
195 			return new CardTypeRegistry(Map.copyOf(handlers));
196 		}
197 	}
198 
199 	/** The generic passthrough: a JSON5 body becomes the fragment verbatim; a markup body becomes a template. */
200 	private static final class GenericCardType implements CardTypeHandler {
201 		@Override public String type() { return "*"; } // never registered through Builder.add; unused as a key.
202 		@Override public JsonMap toFragment(CardSource source) {
203 			if (source.hasJsonBody())
204 				return source.json();
205 			var frag = new JsonMap();
206 			if (source.body() != null && ! source.body().isEmpty())
207 				frag.put("template", source.captureTemplate());
208 			return frag;
209 		}
210 	}
