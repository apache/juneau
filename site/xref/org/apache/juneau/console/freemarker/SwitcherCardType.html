17  package org.apache.juneau.console.freemarker;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.console.*;
22  import org.apache.juneau.marshall.collections.*;
23  
24  /**
25   * The built-in {@code switcher} card type: a JSON5 body naming {@code views}, each holding one {@code card} or a list
26   * of {@code sections} that each hold a card.  {@code juneau-switcher.js} renders it.  The body is validated by
27   * {@link SwitcherDef#validate(String, Map)}, the same check {@link SwitcherDef} runs, and every nested card is lifted
28   * through its own type's handler by the registry (see {@link CardTypeHandler#nestedCards(JsonMap)}).
29   *
30   * <h2 class='section'>Example:</h2>
31   * <p class='bcode'>
32   * 	&lt;@card type="switcher" id="revenue-sw" title="Revenue"&gt;
33   * 	{ views: [ { id: 'chart', label: 'Chart', card: { id: 'rev', type: 'chart', src: '/s', kind: 'line', x: { field: 'd' }, series: [ 'n' ] } } ] }
34   * 	&lt;/@card&gt;
35   * </p>
36   *
37   * @since 10.0.0
38   */
39  public final class SwitcherCardType implements CardTypeHandler {
40  
41  	/** Public no-arg constructor, required by {@link java.util.ServiceLoader}. */
42  	public SwitcherCardType() {
43  		// No state; instantiated by ServiceLoader.
44  	}
45  
46  	@Override
47  	public String type() {
48  		return "switcher";
49  	}
50  
51  	@Override
52  	public JsonMap toFragment(CardSource source) {
53  		if (! source.hasJsonBody())
54  			throw source.error("type='switcher' requires a JSON5 body.");
55  		var body = source.json();
56  		try {
57  			SwitcherDef.validate(source.id(), body);
58  		} catch (IllegalArgumentException e) {
59  			throw source.error("%s", e.getMessage());
60  		}
61  		var frag = new JsonMap();
62  		for (var e : body.entrySet())
63  			if (! "visibleWhen".equals(e.getKey()))
64  				frag.put(e.getKey(), copy(e.getValue()));
65  		return frag;
66  	}
67  
68  	@SuppressWarnings({
69  		"java:S3776" // Branching is inherent to walking a switcher body's views and sections.
70  	})
71  	@Override
72  	public List<Map<String,Object>> nestedCards(JsonMap body) {
73  		var out = new ArrayList<Map<String,Object>>();
74  		if (body.get("views") instanceof List<?> views)
75  			for (var v : views) {
76  				if (! (v instanceof Map<?,?> vm))
77  					continue;
78  				add(out, vm.get("card"));
79  				if (vm.get("cards") instanceof List<?> stacked)
80  					for (var c : stacked)
81  						add(out, c);
82  				if (vm.get("sections") instanceof List<?> secs)
83  					for (var s : secs)
84  						if (s instanceof Map<?,?> sm)
85  							add(out, sm.get("card"));
86  			}
87  		return out;
88  	}
89  
90  	@SuppressWarnings({
91  		"unchecked" // Card maps are JSON objects, so their keys are strings.
92  	})
93  	private static void add(List<Map<String,Object>> out, Object card) {
94  		if (card instanceof Map<?,?> m)
95  			out.add((Map<String,Object>)m);
96  	}
97  
98  	// A deep copy, so lifting a nested card never rewrites the authored body.
99  	private static Object copy(Object o) {
100 		if (o instanceof Map<?,?> m) {
101 			var c = new LinkedHashMap<String,Object>();
102 			m.forEach((k, v) -> c.put(String.valueOf(k), copy(v)));
103 			return c;
104 		}
105 		if (o instanceof List<?> l)
106 			return new ArrayList<>(l.stream().map(SwitcherCardType::copy).toList());
107 		return o;
108 	}
109 }
