35  /**
36   * Java description of one console page: the active tab, toolkits, page CSS and init scripts, per-request header
37   * fields, the stock theme by name, and the page's cards (tables, html and registered custom types) and body attributes.
38   *
39   * <p>
40   * A {@code PageSpec} renders nothing itself.  Its values are seeded into the request's {@link PageCapture} before any
41   * directive runs, so a built page goes through the same {@code <@console>} chrome and emits the same
42   * {@code #juneau-page} contract and the same toolkit asset tags as an FTL page.  Chrome (brand, nav tree, footer, theme tokens) stays in the app's
43   * chrome template.
44   *
45   * <h2 class='section'>Example:</h2>
46   * <p class='bjava'>
47   * 	<jk>var</jk> <jv>spec</jv> = PageSpec.<jsm>create</jsm>()
48   * 		.title(<js>"SLO"</js>)
49   * 		.tab(<js>"slo"</js>)
50   * 		.toolkit(<js>"views"</js>)
51   * 		.css(<js>"/css/slo.css"</js>)
52   * 		.csrf(<jv>token</jv>, <js>"X-CSRF"</js>)
53   * 		.html(<js>"intro"</js>, <js>"&lt;p&gt;Queue health&lt;/p&gt;"</js>)
54   * 		.table(<jv>sloTable</jv>);
55   * </p>
56   *
57   * <h2 class='section'>Example:</h2>
58   * <p class='bjava'>
59   * 	<jk>return</jk> PageSpec.<jsm>create</jsm>()
60   * 		.tab(<js>"changes"</js>)
61   * 		.table(ChangesView.<jsm>def</jsm>(), <jv>c</jv> -&gt; <jv>c</jv>.title(<js>"Changes"</js>))
62   * 		.table(TasksView.<jsm>def</jsm>(), <jv>c</jv> -&gt; <jv>c</jv>.title(<js>"Tasks for the selected change"</js>)
63   * 			.subscribes(Subscription.<jsm>to</jsm>(Topics.<jsm>selection</jsm>(<js>"changes"</js>)).as(<js>"params"</js>).map(<js>"changeId"</js>, <js>"ids.0"</js>)))
64   * 		.card(ChartDef.<jsm>bar</jsm>(<js>"byOwner"</js>).title(<js>"Open changes by owner"</js>).src(<js>"/rest/changes/by-owner?q={q}"</js>)
65   * 			.x(<js>"owner"</js>).series(<js>"open"</js>, <js>"Open"</js>).toCardSpec()
66   * 			.subscribes(Subscription.<jsm>to</jsm>(Topics.<jsm>filter</jsm>(<js>"changes"</js>)).as(<js>"params"</js>).map(<js>"q"</js>, <js>"search"</js>)))
67   * 		.view(<jv>req</jv>);
68   * </p>
69   *
70   * @since 10.0.0
71   */

201 	 * Validates a {@code '/'}-separated id path: the whole string and every segment must be non-blank.
202 	 *
203 	 * @param method The calling method's name, used in the error message.
204 	 * @param idPath The path to validate.
205 	 * @return {@code idPath}, unchanged.
206 	 * @throws IllegalArgumentException If blank, or any {@code '/'}-separated segment is blank.
207 	 */
208 	static String checkIdPath(String method, String idPath) {
209 		if (ib(idPath))
210 			throw iaex("PageSpec %s path must be a non-empty '/'-separated id path; got '%s'.", method, idPath);
211 		for (var segment : idPath.split("/", -1))
212 			if (segment.isBlank())
213 				throw iaex("PageSpec %s path must be a non-empty '/'-separated id path; got '%s'.", method, idPath);
214 		return idPath;
215 	}

348 	/**
349 	 * Sets the header's {@code subtitle} and/or {@code userMenu}.  These are the only two header fields this builder
350 	 * may set; every other field belongs in the chrome template.
351 	 *
352 	 * @param c Mutates a scratch {@link PageCapture.HeaderSpec}.
353 	 * @return This object.
354 	 * @throws IllegalArgumentException If {@code c} sets any field other than {@code subtitle}/{@code userMenu}, or the
355 	 * 	user menu has a key the page contract does not allow.
356 	 */
357 	public PageSpec header(Consumer<PageCapture.HeaderSpec> c) {
358 		var scratch = new PageCapture.HeaderSpec();
359 		c.accept(scratch);
360 		checkHeaderScratch(scratch);
361 		if (scratch.subtitle != null)
362 			headerSubtitle = scratch.subtitle;
363 		if (scratch.userMenu != null)
364 			for (var k : scratch.userMenu.keySet())
365 				if (! USER_MENU_KEYS.contains(k))
366 					throw iaex("PageSpec.header userMenu key '%s' is not allowed; allowed: label, initials, avatar, items.", k);
367 		if (scratch.userMenu != null)
368 			headerUserMenu = new LinkedHashMap<>(scratch.userMenu);
369 		return this;
370 	}
371 
372 	private static void checkHeaderScratch(PageCapture.HeaderSpec h) {
373 		if (h.title != null)
374 			throw headerFieldRejected("title");
375 		if (h.logoSrc != null || h.logoHref != null || h.logoAlt != null)
376 			throw headerFieldRejected("logo");
377 		if (h.chrome)
378 			throw headerFieldRejected("chrome");
379 		if (! h.links.isEmpty())
380 			throw headerFieldRejected("link");
381 	}

502 	/**
503 	 * Adds an {@code html} card whose body is fetched from {@code src}, with a title, optionally rebuilt through the
504 	 * views attribute-allowlist sanitizer.  The equivalent of {@code <@card type="html" src="..." sanitize=true>}.
505 	 *
506 	 * <h4 class='section'>Trust boundary:</h4>
507 	 * <p>
508 	 * With {@code sanitize} <jk>false</jk> the shell fetches {@code src} and inserts the returned HTML as-is through a
509 	 * raw HTML-sink assignment, <b>unsanitized</b>, the same trust level as {@code footer.text} (see
510 	 * {@code HtmlCardType}).  The endpoint must therefore be a trusted first-party route, must never be derived from
511 	 * request data, and must escape any data it embeds in the fragment.  {@code src} must be a same-origin path; an
512 	 * absolute URL, a scheme, {@code //}, {@code ..} or a backslash is refused when the page is built, exactly as for
513 	 * every other card.
514 	 *
515 	 * <p>
516 	 * With {@code sanitize} <jk>true</jk> the fragment is rebuilt through the views attribute-allowlist sanitizer
517 	 * instead: script, {@code on*} handlers, {@code javascript:} and {@code data:} URLs and every attribute off the
518 	 * allowlist are dropped, and benign markup (text, tables, links, images) survives.  It needs the views toolkit on
519 	 * the page; without the toolkit nothing is painted (the shell fails closed rather than inserting the fragment
520 	 * unsanitized).  The sanitizer does not remove the need to escape any data you embed in the fragment.  Turn it on when the fragment's content is not fully under your control (it embeds user-supplied or
521 	 * third-party markup, or comes from an endpoint you do not own); leave it off for a fragment that needs
522 	 * {@code class}, {@code id}, {@code style}, forms, SVG or iframes, which the allowlist drops.  The default stays off.
523 	 *
524 	 * @param id The card id.
525 	 * @param title The card title.  May be <jk>null</jk> for no title.
526 	 * @param src The same-origin path of the HTML fragment endpoint.  Must not be blank.
527 	 * @param sanitize Whether to rebuild the fetched fragment through the sanitizer.
528 	 * @return This object.
529 	 * @throws IllegalArgumentException If {@code src} is <jk>null</jk> or blank.
530 	 */
531 	public PageSpec htmlSrc(String id, String title, String src, boolean sanitize) {
532 		if (ib(src))
533 			throw iaex("PageSpec.htmlSrc id '%s' requires a non-blank src.", id);
534 		var c = CardSpec.html(id).src(src).sanitize(sanitize);
535 		if (nn(title))
536 			c.title(title);
537 		addCard(c);
538 		return this;
539 	}

541 	private PageSpec html0(String id, String title, String markup, Consumer<CardSpec> card) {
542 		if (markup == null)
543 			throw iaex("PageSpec.html id '%s' requires a non-null markup.", id);
544 		var c = CardSpec.html(id);
545 		if (nn(title))
546 			c.title(title);
547 		if (nn(card)) {
548 			card.accept(c);
549 			if (nn(c.src()) || nn(c.template()))
550 				throw iaex("PageSpec.html id '%s' sets a markup body; its Consumer<CardSpec> must not also set src or template.", id);
551 		}
552 		addCard(c);
553 		cardMarkup.put(id, markup);
554 		return this;
555 	}

740 	static void checkFactTypes(String prefix, Map<?,?> m) {
741 		for (var e : m.entrySet()) {
742 			var path = prefix.isEmpty() ? String.valueOf(e.getKey()) : prefix + "." + e.getKey();
743 			var v = e.getValue();

750 			} else if (! (v instanceof String || v instanceof Boolean || isPlainNumber(v))) {
751 				throw badFact(path, v);
752 			}
753 		}
754 	}

916 	 * Builds the view for this spec.
917 	 *
918 	 * <p>
919 	 * The template is {@link #template(String)} when set, else the {@code ConsoleFreemarkerMixin}'s chrome template.
920 	 * A built spec is not modified by rendering, so it may be cached and reused across requests.  Call
921 	 * {@code spec.view(req)} on each request: a reused {@link FreemarkerView} would share one adoption flag.
922 	 *
923 	 * @param req The current REST request.
924 	 * @return A view for the template, carrying this spec and every {@link #attr(String, Object)} /
925 	 * 	{@link #header(String, String)} passthrough.
926 	 * @throws IllegalStateException If no template was set and the request has no {@code ConsoleFreemarkerMixin} bean.
927 	 * @throws IllegalArgumentException If this spec has no page template and its bus wiring fails R-10 / R-11.
928 	 */

930 		if (req == null)

933 		if (template != null) {
934 			templateName = template;
935 		} else {

942 		for (var h : headers)
943 			v = v.header(h.name(), h.value());

945 			v = v.attr(e.getKey(), e.getValue());

953 	})

955 		var store = req.getContext().getBeanStore();

971 	/**
972 	 * Applies every field this spec has set to {@code cap}, via {@code cap}'s plain (non-merge-aware) setters.
973 	 *
974 	 * <p>
975 	 * Called once per render, on a freshly created capture before any directive has written to it, so these raw
976 	 * writes can never conflict with anything.  This spec is not modified: each card is copied before it is seeded.
977 	 * A table card, a chart, job-log or terminal card, and a card of any adopter (non-reserved) type are lifted through the
978 	 * card-type registry of {@code cardModel}, with the {@code <@card>} checks and messages.  Every card's required packs are recorded through {@code cardModel}.
979 	 *
980 	 * @param cap The freshly created, still-empty capture.
981 	 * @param cardModel The shared {@code <@card>} directive; may be <jk>null</jk> only when this spec has no cards.
982 	 * @throws TemplateModelException If a card is rejected.
983 	 */
984 	void applyTo(PageCapture cap, CardDirectiveModel cardModel) throws TemplateModelException {
985 		if (title != null)
986 			cap.title(title);
987 		if (tab != null)
988 			cap.tab(tab);
989 		if (theme != null)
990 			cap.theme(theme);
991 		if (scheme != null)
992 			cap.scheme(scheme);
993 		if (schemeToggle)
994 			cap.schemeToggle(true);
995 		if (themeSelector)
996 			cap.themeSelector(true);
997 		if (! toolkits.isEmpty())
998 			cap.toolkit(toolkits);
999 		if (! css.isEmpty())
1000 			cap.css(css);
1001 		if (! init.isEmpty())
1002 			cap.init(init);
1003 		if (! allowInlineHandlers && (model.hasInline() || tableSpecs.stream().anyMatch(TableSpec::hasInlineHandlers) || cards.stream().anyMatch(c -> c.modelBlock().hasInline())))
1004 			throw new IllegalArgumentException("page: an inline handler needs allowInlineHandlers(true).");
1005 		if (allowInlineHandlers)
1006 			cap.allowInlineHandlers(true);
1007 		cap.rootModel(model.toMap());
1008 		if (scriptUsed)
1009 			cap.emitScripts(true);
1010 		if (headerSubtitle != null || headerUserMenu != null)
1011 			cap.header(h -> {
1012 				if (headerSubtitle != null)
1013 					h.subtitle(headerSubtitle);
1014 				if (headerUserMenu != null)
1015 					h.userMenu(headerUserMenu);
1016 			});
1017 		if (! cards.isEmpty() && n(cardModel))
1018 			throw FtlAttrLists.reject("PageSpec cards need the console card directive; the render has no ConsoleFreemarkerMixin.");
1019 		for (var card : cards) {
1020 			var markup = cardMarkup.get(card.id());
1021 			Built built;
1022 			if (tableBodies.containsKey(card.id()))
1023 				built = new Built(liftTable(card, cardModel), markup);
1024 			else if (LIFTED_TYPES.contains(card.type()))
1025 				built = new Built(liftChecked(card, cardModel), markup);
1026 			else if (CardTypeRegistry.RESERVED.contains(card.type()))
1027 				built = new Built(card.copy(), markup);
1028 			else
1029 				built = liftCustom(card, markup, cardModel);
1030 			cap.addBuiltCard(built.card(), built.markupOrNull());
1031 			cardModel.recordRequirements(cap, built.card().type(), built.card().id(), List.of(), built.card().bodyMap());
1032 		}
1033 		try {
1034 			for (var t : topics)
1035 				cap.addTopic(t.toMap(), "PageSpec.topic");
1036 			var req = FreemarkerRenderScope.request();
1037 			for (var b : bridges.values()) {
1038 				cap.addBridge(resolvedBridge(b, req));
1039 			}
1040 			for (var d : dataSources.values())
1041 				cap.addDataSource(new JsonMap(d));
1042 			if (! dataSources.isEmpty())
1043 				cap.require(List.of(ToolkitPackRegistry.PACK_DATA));
1044 		} catch (IllegalArgumentException e) {
1045 			throw FtlAttrLists.reject(e.getMessage());
1046 		}
1047 		if (! bodyAttrs.isEmpty()) {
1048 			var sb = new StringBuilder();
1049 			for (var e : bodyAttrs.entrySet())
1050 				sb.append(e.getKey()).append("=\"").append(e.getValue()).append("\" ");
1051 			cap.bodyAttrs = sb.toString().stripTrailing();
1052 		}
1053 		for (var p : navUnders)
1054 			cap.addPendingNavUnder(p.parentIdPath(), p.adder());
1055 		for (var b : badges)
1056 			cap.addBadge(b);
1057 		if (dashboard != null)
1058 			cap.dashboard(dashboard);
1059 		if (! facts.isEmpty())
1060 			cap.seedJavaFacts(facts);
1061 	}

1082 			rejectSrcAndTemplate(card);
1083 		else if (nn(card.template()))
1084 			throw FtlAttrLists.reject(f("<@card id='%s'> type='chart' takes a JSON5 body (plus src= for remote data); template= is not allowed.", card.id()));
1085 		var source = CardSource.create(card.type(), card.id()).bodyMap(card.bodyMap());
1086 		if (nn(card.src()))
1087 			source.src(card.src());
1088 		return liftVia(card, source, cardModel);
1089 	}

1091 	/** Rejects {@code src=} and/or {@code template=} on a card whose options are its body, with the directive's message. */
1092 	private static void rejectSrcAndTemplate(CardSpec card) throws TemplateModelException {
1093 		var src = nn(card.src()) ? card.src() : "";
1094 		var template = nn(card.template()) ? card.template() : "";
1095 		if (! (src.isEmpty() && template.isEmpty()))
1096 			throw FtlAttrLists.reject(f("<@card id='%s'> type='%s' takes its options as the body; %s.", card.id(), card.type(),
1097 				CardDirectiveModel.notAllowed(src, template)));
1098 	}

1101 	 * A terminal card: its body map is the {@code {contractVersion: '1', terminal: {...}}} envelope {@code <@card>} takes,
1102 	 * with {@code contractVersion} implied, so the registered handler validates it exactly as it does for the directive.
1103 	 */
1104 	private static CardSpec liftTerminal(CardSpec card, CardDirectiveModel cardModel) throws TemplateModelException {
1105 		rejectSrcAndTemplate(card);
1106 		var envelope = new LinkedHashMap<String,Object>();
1107 		envelope.put("contractVersion", "1");
1108 		envelope.putAll(card.bodyMap());
1109 		return liftVia(card, CardSource.create(card.type(), card.id()).bodyMap(envelope), cardModel);
1110 	}

1112 	/**
1113 	 * An adopter card type: the same checks {@code <@card>} runs, then its registered handler (or the generic
1114 	 * passthrough for an unregistered type) builds the fragment from the card's body map, or from its non-blank markup
1115 	 * when it has no body map.  The markup is emitted as the card's template only when the handler captures it, as
1116 	 * {@code <@card>} emits it; JSON5 markup a handler parses is the body, not a template.
1117 	 */
1118 	private static Built liftCustom(CardSpec card, String markupOrNull, CardDirectiveModel cardModel) throws TemplateModelException {
1119 		checkSanitize(card);
1120 		var body = card.bodyMap();
1121 		var hasMarkup = nn(markupOrNull) && ! markupOrNull.isBlank();
1122 		if ((body.isEmpty() ? 0 : 1) + (hasMarkup ? 1 : 0) + (nn(card.src()) ? 1 : 0) + (nn(card.template()) ? 1 : 0) > 1)
1123 			throw FtlAttrLists.reject(f("<@card id='%s'> type='%s' takes at most one of a body, template= or src=.", card.id(), card.type()));
1124 		var captured = new boolean[1];
1125 		var source = CardSource.create(card.type(), card.id()).templateSink((id, markup) -> captured[0] = true);
1126 		if (nn(card.src()))
1127 			source.src(card.src());
1128 		if (nn(card.template()))
1129 			source.template(card.template());
1130 		if (! body.isEmpty())
1131 			source.bodyMap(body);
1132 		else if (hasMarkup)
1133 			source.body(markupOrNull);
1134 		var lifted = liftVia(card, source, cardModel);
1135 		return new Built(lifted, captured[0] ? markupOrNull : null);
1136 	}

1138 	private static CardSpec liftVia(CardSpec card, CardSource.Builder source, CardDirectiveModel cardModel) throws TemplateModelException {
1139 		if (nn(card.title()))
1140 			source.title(card.title());
1141 		var lifted = cardModel.lift(source.build(), card);
1142 		if (nn(card.cssClass()))
1143 			lifted.cssClass(card.cssClass());
1144 		if (card.bare())
1145 			lifted.bare(true);
1146 		return lifted;
1147 	}
