17  package org.apache.juneau.console.freemarker;
18  
19  import java.util.*;
20  import java.util.regex.*;
21  
22  /**
23   * A rail section of quick time-range presets, with an optional custom from/to.
24   *
25   * <p>
26   * Publishes {@code {schemaVersion:1, viewId, options:{range}, start, end}} on its {@code filter:} topic, where
27   * {@code range} is the preset or {@code "custom"} and {@code start}/{@code end} are ISO-8601 UTC instants.
28   * {@code Nd} and {@code Nh} end at now floored to the minute; {@code ytd} starts at January 1, 00:00 UTC.
29   * </p>
30   *
31   * <h5 class='section'>Example:</h5>
32   * <p class='bjava'>
33   * 	RangeSection.<jsm>quick</jsm>(<js>"7d"</js>, <js>"30d"</js>, <js>"90d"</js>, <js>"ytd"</js>).custom(<jk>true</jk>).defaultPreset(<js>"30d"</js>)
34   * 		.publishes(Topics.<jsm>filter</jsm>(<js>"range"</js>));
35   * </p>
36   *
37   * @since 10.0.0
38   */
39  @SuppressWarnings({
40  	"java:S1192" // Repeated literals are contract wire keys; named constants would obscure the contract shape.
41  })
42  public final class RangeSection extends Section {
43  
44  	private static final Pattern PRESET = Pattern.compile("^([1-9]\\d{0,3}[dh]|ytd)$");
45  
46  	private final List<String> presets;
47  	private boolean custom;
48  	private String defaultPreset;
49  
50  	private RangeSection(String...presets) {
51  		this.presets = Arrays.asList(presets);  // A null preset is reported by check(), not an NPE here.
52  	}
53  
54  	/**
55  	 * Creates a range section.
56  	 *
57  	 * @param presets The quick presets, each {@code Nd}, {@code Nh} or {@code ytd}.
58  	 * @return A new section.
59  	 */
60  	public static RangeSection quick(String...presets) {
61  		return new RangeSection(presets);
62  	}
63  
64  	/**
65  	 * Adds the custom from/to inputs (read as UTC).
66  	 *
67  	 * @param value <jk>true</jk> to show them.
68  	 * @return This object.
69  	 */
70  	public RangeSection custom(boolean value) {
71  		custom = value;
72  		return this;
73  	}
74  
75  	/**
76  	 * Sets the preset used when nothing is restored.
77  	 *
78  	 * @param value One of the presets.  Defaults to the first.
79  	 * @return This object.
80  	 */
81  	public RangeSection defaultPreset(String value) {
82  		defaultPreset = value;
83  		return this;
84  	}
85  
86  	@Override /* Section */
87  	public RangeSection publishes(String topic) {
88  		super.publishes(topic);
89  		return this;
90  	}
91  
92  	@Override /* Section */
93  	public RangeSection label(String label) {
94  		super.label(label);
95  		return this;
96  	}
97  
98  	@Override String kind() { return "range"; }
99  
100 	@Override void check(String dash) {
101 		super.check(dash);
102 		if (presets.isEmpty())
103 			throw e77(dash, "range section '" + id() + "' needs at least one preset");
104 		var seen = new HashSet<String>();
105 		for (var p : presets) {
106 			if (p == null || ! PRESET.matcher(p).matches())
107 				throw e77(dash, "range section '" + id() + "' preset '" + p + "' is not Nd, Nh or ytd");
108 			if (! seen.add(p))
109 				throw e77(dash, "range section '" + id() + "' lists preset '" + p + "' twice");
110 		}
111 		if (defaultPreset != null && ! presets.contains(defaultPreset))
112 			throw e77(dash, "range section '" + id() + "' default preset '" + defaultPreset + "' is not one of its presets");
113 	}
114 
115 	@Override void body(Map<String,Object> m) {
116 		m.put("presets", presets);
117 		m.put("custom", custom);
118 		m.put("defaultPreset", defaultPreset != null ? defaultPreset : presets.get(0));
119 	}
120 }
