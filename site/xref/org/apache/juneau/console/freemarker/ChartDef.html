31  /**
32   * Typed builder for a {@code chart} card.
33   *
34   * <p>
35   * Produces the same contract entry as {@code <@card type="chart">}; {@link PageSpec} runs it through
36   * {@link ChartCardType} when the page renders.  The factories are {@link #line(String)}, {@link #bar(String)},
37   * {@link #donut(String)}, {@link #heatmap(String)}, {@link #scatter(String)}, {@link #pie(String)}, {@link #treemap(String)} and
38   * {@link #raw(String)}.  {@code line}, {@code bar} and {@code donut} emit the flat legacy form
39   * ({@code kind:"line", x:..., series:[...]}) byte-for-byte; the other kinds emit {@code helper:{kind, ...}}, and
40   * {@code raw} emits neither (its {@link #option(JsonMap)} is the whole chart).
41   *
42   * <h5 class='section'>Example:</h5>
43   * <p class='bjava'>
44   * 	<jv>page</jv>.card(ChartDef.<jsm>line</jsm>(<js>"orders-trend"</js>).title(<js>"Orders"</js>).src(<js>"/console/store/stats/daily"</js>)
45   * 		.x(<js>"day"</js>).series(<js>"placed"</js>, <js>"Placed"</js>).series(<js>"delivered"</js>, <js>"Delivered"</js>).toCardSpec());
46   * </p>
47   *
48   * @since 10.0.0
49   */

51  	"java:S1192", // Repeated literals are contract wire keys; named constants would obscure the contract shape.
52  	"java:S3776" // Validation and rendering chains many independent guards; splitting them adds no clarity.
53  })

289 	/**
290 	 * Runs a row action when a mark is clicked.
291 	 *
292 	 * <p>
293 	 * A treemap tile runs it; clicking a treemap group header only zooms and runs nothing.
294 	 *
295 	 * <p>
296 	 * Declares {@code on("click", "<id>.click")} unless a click topic is already declared, and runs {@code action}
297 	 * whenever that topic fires.  Exclusive with {@link #onClickHref(String)}.  The action is refused unless it has an
298 	 * {@code endpoint} (and, if set, a {@code form}) that is a same-origin path, a non-safe {@code method}, and no
299 	 * {@code present} other than {@code dialog}, and does not navigate on success.
300 	 *
301 	 * @param action The action; its id must be unique across the card's actions.
302 	 * @return This object.
303 	 * @throws IllegalArgumentException If the action is not valid on a chart, {@link #onClickHref(String)} was set, or no click
304 	 * 	topic can be derived: the id must match <c>[a-z][a-z0-9-]{0,31}</c>, or call {@link #on(String, String)} with <js>"click"</js> first.
305 	 */
306 	public ChartDef onClickAction(RowAction action) {
307 		if (action == null)
308 			throw iaex("ChartDef '%s' onClickAction requires an action.", id);
309 		onClickExcludes("href");
310 		var m = toMap(action);
311 		checkAction("onClick action", m);
312 		if (body.get("onClick") instanceof Map<?,?> old && old.get("action") instanceof Map<?,?> prev)
313 			actionIds.remove(prev.get("id"));
314 		claimId(m);
315 		if (! on.containsKey("click")) {
316 			if (! CLICK_NAMESPACE.matcher(id).matches())
317 				throw iaex("ChartDef '%s' onClickAction cannot derive a click topic from the id; declare one with on(\"click\", topic).", id);
318 			on("click", id + ".click");
319 		}
320 		body.put("onClick", Map.of("action", m));
321 		return this;
322 	}

324 	/**
325 	 * Runs a row action when a topic this card publishes fires.
326 	 *
327 	 * <p>
328 	 * Only a click has a keyboard equivalent (Enter on the focused point synthesizes a click); an action bound to another event,
329 	 * such as {@code dblclick}, is reachable by pointer only.  A treemap group node publishes no events.
330 	 *
331 	 * @param topic A topic declared through {@link #on(String, String)}.
332 	 * @param action The action; its id must be unique across the card's actions.
333 	 * @return This object.
334 	 * @throws IllegalArgumentException If the topic is not declared, or the action is not valid on a chart.
335 	 */
336 	public ChartDef onTopicAction(String topic, RowAction action) {
337 		if (topic == null || ! on.containsValue(topic))
338 			throw iaex("ChartDef '%s' onTopicAction topic '%s' must be one this card publishes through on(...).", id, topic);
339 		if (action == null)
340 			throw iaex("ChartDef '%s' onTopicAction requires an action.", id);
341 		var m = toMap(action);
342 		checkAction("topicActions[" + topicActions.size() + "] action", m);
343 		claimId(m);
344 		var e = new LinkedHashMap<String,Object>();
345 		e.put("topic", topic);
346 		e.put("action", m);
347 		topicActions.add(e);
348 		return this;
349 	}

351 	/**
352 	 * Adds dialog actions that other actions can open as steps.
353 	 *
354 	 * <p>
355 	 * Each entry must have {@code present} set to {@code dialog}, and obeys the same rules as
356 	 * {@link #onClickAction(RowAction)}.  Calls accumulate.
357 	 *
358 	 * @param actions The actions; ids must be unique across the card's actions.
359 	 * @return This object.
360 	 * @throws IllegalArgumentException If an entry is <jk>null</jk>, is not a dialog action, or is not valid on a chart.
361 	 */
362 	public ChartDef actions(RowAction... actions) {
363 		if (actions == null)
364 			throw iaex("ChartDef '%s' actions entries must not be null.", id);
365 		for (var a : actions) {
366 			if (a == null)
367 				throw iaex("ChartDef '%s' actions entries must not be null.", id);
368 			var m = toMap(a);
369 			var where = "actions[" + catalog.size() + "]";
370 			checkAction(where, m);
371 			if (! "dialog".equals(m.get("present")))
372 				throw iaex("ChartDef '%s' %s '%s' must be present=dialog; a dialog button can only open a dialog action.", id, where, m.get("id"));
373 			claimId(m);
374 			catalog.add(m);
375 		}
376 		return this;
377 	}

386 	private void onClickExcludes(String other) {
387 		if (body.get("onClick") instanceof Map<?,?> m && m.containsKey(other))
388 			throw iaex("ChartDef '%s' onClick takes either href or action, not both.", id);
389 	}

391 	// The rules every chart action obeys, wherever it is declared: a chart has no cell, page or servlet to hand off to.
392 	private void checkAction(String where, Map<String,Object> m) {
393 		if (! (m.get("endpoint") instanceof String endpoint) || endpoint.isBlank())
394 			throw iaex("ChartDef '%s' %s requires a non-blank endpoint.", id, where);
395 		if (! SafePathTemplate.isSameOriginPath(endpoint))
396 			throw iaex("ChartDef '%s' %s endpoint must be a same-origin path; got '%s'.", id, where, endpoint);
397 		if (m.get("form") instanceof String form && ! SafePathTemplate.isSameOriginPath(form))
398 			throw iaex("ChartDef '%s' %s form must be a same-origin path; got '%s'.", id, where, form);
399 		if (! (m.get("method") instanceof String method) || ! ACTION_METHODS.contains(method))
400 			throw iaex("ChartDef '%s' %s method must be POST, PUT, PATCH or DELETE; got '%s'.", id, where, m.get("method"));
401 		if (nn(m.get("present")) && ! "dialog".equals(m.get("present")))
402 			throw iaex("ChartDef '%s' %s present must be 'dialog' or absent; got '%s'.", id, where, m.get("present"));
403 		if ("navigate".equals(m.get("onSuccess")))
404 			throw iaex("ChartDef '%s' %s onSuccess must not be 'navigate'; a chart has nowhere to navigate.", id, where);
405 	}

407 	// Ids are unique across onClick.action, topicActions[] and actions[]: lookup is first-match.
408 	private void claimId(Map<String,Object> m) {
409 		if (! actionIds.add((String)m.get("id")))
410 			throw iaex("ChartDef '%s' action id '%s' is declared more than once.", id, m.get("id"));
411 	}

451 		only("treemap", "tileFields");
452 		if (fields == null)

454 		for (var f : fields)

457 		body.put("tileFields", List.of(fields));
458 		return this;

644 	/**
645 	 * Deep merge: objects by key; arrays element-by-element by {@code id} (a match merges, no match
646 	 * appends); an array whose elements carry no {@code id} replaces the whole array.  {@code into} is changed in place.
647 	 */
648 	@SuppressWarnings({
649 		"unchecked" // Narrows a JSON-shaped Map<?,?> to Map<String,Object>; its keys are always strings.
650 	})
651 	static void merge(Map<String,Object> into, Map<?,?> from) {
652 		for (var e : from.entrySet()) {
653 			var k = String.valueOf(e.getKey());
654 			var cur = into.get(k);
655 			var v = e.getValue();
656 			if (cur instanceof Map && v instanceof Map) {
657 				var copy = new JsonMap((Map<String,Object>)cur);
658 				merge(copy, (Map<?,?>)v);
659 				into.put(k, copy);
660 			} else if (cur instanceof List<?> a && v instanceof List<?> b && hasIds(b)) {
661 				var out = new ArrayList<Object>(a);
662 				for (var item : b) {
663 					var at = indexOfId(out, ((Map<?,?>)item).get("id"));
664 					if (at < 0) {
665 						out.add(item);
666 					} else {
667 						var copy = new JsonMap((Map<String,Object>)out.get(at));
668 						merge(copy, (Map<?,?>)item);
669 						out.set(at, copy);
670 					}
671 				}
672 				into.put(k, out);
673 			} else {
674 				into.put(k, v);
675 			}
676 		}
677 	}
678 
679 	private static boolean hasIds(List<?> l) {
680 		return ! l.isEmpty() && l.stream().allMatch(x -> x instanceof Map<?,?> m && m.get("id") != null);
681 	}
682 
683 	private static int indexOfId(List<Object> l, Object id) {
684 		for (var i = 0; i < l.size(); i++)
685 			if (l.get(i) instanceof Map<?,?> m && eq(m.get("id"), id))
686 				return i;
687 		return -1;
688 	}

710 	/**
711 	 * @return The {@code chart} card.
712 	 * @throws IllegalArgumentException If not exactly one of {@code src}/{@code data}/{@code source} is set, a kind's
713 	 * 	required keys are missing, {@code raw} has no option, or an option/patch string is not valid JSON5.
714 	 */
715 	public CardSpec toCardSpec() {
716 		if ((nn(src) ? 1 : 0) + (nn(data) ? 1 : 0) + (nn(source) ? 1 : 0) != 1)
717 			throw iaex("ChartDef '%s' requires exactly one of src(...), data(...) or source(...).", id);
718 		if (! nn(src) && body.containsKey("refreshMs"))
719 			throw iaex("ChartDef '%s' refreshMs(...) needs src(...); inline data never refreshes.", id);
720 		if (body.containsKey("partialFrames") && ! nn(source))
721 			throw iaex("ChartDef '%s' partialFrames() needs source(...).", id);
722 		if (kind != null && Set.of("line", "bar", "scatter").contains(kind) && (! body.containsKey("x") || series.isEmpty()))
723 			throw iaex("ChartDef '%s' kind '%s' requires x(...) and at least one series(...).", id, kind);
724 		if (kind != null && Set.of("donut", "pie").contains(kind) && (! (body.get("label") instanceof String l) || l.isBlank() || ! (body.get("value") instanceof String v) || v.isBlank()))
725 			throw iaex("ChartDef '%s' kind '%s' requires label(...) and value(...).", id, kind);
726 		if (eq(kind, "treemap") && ! body.containsKey("groupBy"))
727 			throw iaex("ChartDef '%s' kind 'treemap' requires groupBy(...).", id);
728 		if (eq(kind, "treemap") && (! (body.get("label") instanceof String tl) || tl.isBlank()))
729 			throw iaex("ChartDef '%s' kind 'treemap' requires label(field).", id);
730 		if (eq(kind, "heatmap") && (! body.containsKey("x") || ! body.containsKey("y") || ! (body.get("value") instanceof String)))
731 			throw iaex("ChartDef '%s' kind 'heatmap' requires x(...), yField(...) and value(...).", id);
732 		if (kind == null && option == null)
733 			throw iaex("ChartDef '%s' raw(...) requires option(...).", id);
734 		if (kind == null && (! series.isEmpty() || body.keySet().stream().anyMatch(HELPER_KEYS::contains)))
735 			throw iaex("ChartDef '%s' raw(...) takes no helper keys (x, y, series, ...); put them in option(...).", id);
736 		var c = CardSpec.of("chart", id);
737 		if (legacy())
738 			c.body("kind", kind);
739 		if (nn(title)) c.title(title);
740 		if (nn(src)) c.src(src);
741 		if (nn(source)) {
742 			c.body("source", source);
743 			c.subscribes(Subscription.to(source).as("data"));
744 		}
745 		if (legacy()) {
746 			body.forEach(c::body);
747 			if (! series.isEmpty()) c.body("series", List.copyOf(series));
748 		} else {
749 			var helper = kind == null ? null : JsonMap.of("kind", kind);
750 			body.forEach((k, v) -> { if (helper != null && HELPER_KEYS.contains(k)) helper.put(k, v); else if (! HELPER_KEYS.contains(k)) c.body(k, v); });
751 			if (helper != null && ! series.isEmpty()) helper.put("series", List.copyOf(series));
752 			if (helper != null) c.body("helper", helper);
753 		}
754 		if (nn(data)) c.body("data", List.copyOf(data));
755 		if (nn(option)) c.body("option", parsed(option, "option"));
756 		if (! patches.isEmpty()) {
757 			var p = new JsonMap();
758 			for (var x : patches)
759 				merge(p, parsed(x, "patch"));
760 			c.body("patch", p);
761 		}
762 		if (! decorators.isEmpty()) c.body("decorators", List.copyOf(decorators));
763 		if (! on.isEmpty()) c.body("on", new JsonMap(on));
764 		if (! topicActions.isEmpty()) c.body("topicActions", List.copyOf(topicActions));
765 		if (! catalog.isEmpty()) c.body("actions", List.copyOf(catalog));
766 		if (! maps.isEmpty()) c.body("maps", new JsonMap(maps));
767 		if (! publishes.isEmpty()) c.publishes(publishes.entrySet().stream().map(e -> TopicDecl.of(e.getKey()).retain(e.getValue())).toArray(TopicDecl[]::new));
768 		if (nn(visibleWhen)) c.visibleWhen(visibleWhen);
769 		return c;
770 	}
