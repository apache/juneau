29  /**
30   * The {@code <@token name="--jc-…" value="…"/>} / {@code <@token name="--jc-…" alias="var(--jc-…)"/>} chrome
31   * FreeMarker directive: one custom theme-token override authored inside a {@code <@theme>} element.
32   *
33   * <p>
34   * Exactly one of {@code value=} / {@code alias=} is required, and they are mutually exclusive &mdash; they are the
35   * two disjoint {@link Theme} channels:
36   * <ul class='spaced-list'>
37   * 	<li>{@code value=} is a <b>leaf</b>: a CSS literal, or a {@code var(--jc-name)} reference that
38   * 		{@link org.apache.juneau.console.Theme.Builder#token(String, String) Theme.Builder#token(String, String)} <b>resolves to a literal</b>
39   * 		at build time (the substring {@code var(} does not reach the served leaf map).
40   * 	<li>{@code alias=} is a <b>derived reference</b>: a {@code var(--jc-name)} that
41   * 		{@link org.apache.juneau.console.Theme.Builder#alias(String, String) Theme.Builder#alias(String, String)} keeps as a reference
42   * 		<b>surviving to the wire</b>. A literal cannot express a live alias &mdash; that is why the attr exists.
43   * </ul>
44   *
45   * <p>
46   * The name/value/target guards and reserved-namespace rejection are the Java {@code Theme.Builder}'s own (fail
47   * closed); an {@link IllegalArgumentException} it raises is re-surfaced through {@link FtlAttrLists#reject(String)}
48   * so its exact sentence reaches the HTTP 500 body. The directive writes no markup &mdash; it folds the declaration
49   * into the enclosing {@link ThemeBuildContext}, which {@code <@theme>} builds after its body pass.
50   *
51   * <p>
52   * {@code scheme="dark"} declares the token for the dark colour scheme only. Dark declarations render in their own
53   * {@code html:root[data-jc-scheme="dark"]} rule after the light override block, holding only the declared names. A
54   * dark {@code value=} must be a literal (a {@code var()} leaf would resolve against the light palette); a dark
55   * {@code alias=} stays a live reference. {@code dark} is the only accepted value: a light token omits
56   * {@code scheme=}.
57   *
58   * <h5 class='section'>Example:</h5>
59   * <p class='bftl'>
60   * 	&lt;@theme name="light-red"&gt;
61   * 		&lt;#-- A leaf: a literal. --&gt;
62   * 		&lt;@token name="--jc-pill-red-bg" value="#fdeceb"/&gt;
63   * 		&lt;#-- An alias: a live reference. --&gt;
64   * 		&lt;@token name="--jc-tab-bar-bg" alias="var(--jc-card-bg)"/&gt;
65   * 		&lt;#-- Dark scheme only. --&gt;
66   * 		&lt;@token name="--jc-link" value="#6cb4ff" scheme="dark"/&gt;
67   * 	&lt;/@theme&gt;
68   *
69   * 	&lt;#-- Rejected: both value= and alias=, neither one, or a reserved --jc-chrome-* name. --&gt;
70   * 	&lt;@token name="--jc-x" value="#fff" alias="var(--jc-accent)"/&gt;
71   * 	&lt;@token name="--jc-x"/&gt;
72   * 	&lt;@token name="--jc-chrome-control-height" value="26px"/&gt;
73   * </p>
74   *
75   * @since 10.0.0
76   */
77  @SuppressWarnings({
78  	"java:S1192", // Duplicated literals read more clearly inline than as constants
79  	"java:S3776" // Validation and rendering chains many independent guards; splitting them adds no clarity.
80  })
81  public final class TokenDirectiveModel implements TemplateDirectiveModel {
82  
83  	/** The shared-variable name this directive registers under. */
84  	public static final String NAME = "token";
85  
86  	static final Set<String> ATTRS = Set.of("name", "value", "alias", "scheme");
87  
88  	TokenDirectiveModel() {}
89  
90  	@Override
91  	@SuppressWarnings({
92  		"unchecked" // FreeMarker's raw params Map is String-keyed by contract.
93  	})
94  	public void execute(Environment env, @SuppressWarnings("rawtypes") Map params, TemplateModel[] loopVars,
95  			TemplateDirectiveBody body) throws TemplateException, IOException {
96  		Map<String, TemplateModel> p = params;
97  		if (p.containsKey("format"))
98  			throw FtlAttrLists.reject("<@token> has no format= attribute.");
99  		FtlAttrLists.rejectUnknown(p, NAME, ATTRS);
100 
101 		var cap = PageCapture.get(env);
102 		var ctx = n(cap) ? null : cap.themeBuild;
103 		if (n(ctx))
104 			throw FtlAttrLists.reject("<@token> must be nested inside <@theme>.");
105 
106 		var name = FtlAttrLists.scalar(p, "name");
107 		if (name.isEmpty())
108 			throw FtlAttrLists.reject("<@token> requires name=.");
109 
110 		var hasValue = p.containsKey("value");
111 		var hasAlias = p.containsKey("alias");
112 		if (hasValue == hasAlias)
113 			throw FtlAttrLists.reject("<@token name=\"" + name + "\"> requires exactly one of value= or alias=.");
114 
115 		var dark = p.containsKey("scheme");
116 		if (dark) {
117 			var scheme = FtlAttrLists.scalar(p, "scheme");
118 			if (! "dark".equals(scheme))
119 				throw FtlAttrLists.reject("<@token name=\"" + name + "\"> scheme= must be 'dark'; got '" + scheme + "'.  Light tokens omit scheme=.");
120 		}
121 
122 		try {
123 			if (dark) {
124 				// Validated eagerly, one declaration at a time, so the diagnostic names the offending token.
125 				var v = FtlAttrLists.scalar(p, hasValue ? "value" : "alias");
126 				if (hasValue) {
127 					ConsoleChromeMixin.schemeOverrideBlock(Map.of(name, v), Map.of());
128 					ctx.darkTokens.put(name, v);
129 				} else {
130 					ConsoleChromeMixin.schemeOverrideBlock(Map.of(), Map.of(name, v));
131 					ctx.darkAliases.put(name, v);
132 				}
133 			} else {
134 				if (hasValue)
135 					ctx.themeBuilder.token(name, FtlAttrLists.scalar(p, "value"));
136 				else
137 					ctx.themeBuilder.alias(name, FtlAttrLists.scalar(p, "alias"));
138 				ctx.lightDeclared = true;
139 			}
140 		} catch (IllegalArgumentException e) {
141 			// Theme.Builder's frozen sentence is the diagnostic; re-surface it undecorated (see FtlAttrLists.reject).
142 			throw FtlAttrLists.reject(e.getMessage());
143 		}
144 		ctx.anyDeclared = true;
145 	}
146 }
