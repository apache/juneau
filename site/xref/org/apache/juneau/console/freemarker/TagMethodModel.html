23  import org.apache.juneau.bean.html5.*;
24  
25  import freemarker.core.*;
26  import freemarker.template.*;
27  import freemarker.template.utility.*;

29  /**
30   * The FreeMarker shared-variable method backing the {@code <@tag domain=... value=.../>} macro defined in the
31   * reserved {@code base.ftlh} template.
32   *
33   * <p>
34   * Implements the design's <b>serialize-then-mark-trusted</b> insertion contract: (1) build a
35   * {@code <span class="tag &lt;domain&gt; &lt;value&gt;">} element, (2) serialize it via {@code toString()}
36   * ({@code HtmlSerializer.DEFAULT_SIMPLE_SQ}), (3) wrap the resulting markup string via
37   * {@link HTMLOutputFormat#fromMarkup(String) HTMLOutputFormat.INSTANCE.fromMarkup(...)} so a {@code .ftlh}
38   * template's default {@code ${...}} auto-escaping does not double-escape it. The macro then writes the result with
39   * a plain {@code ${jcTagHtml(domain, value)}} interpolation &mdash; because the returned value is already a
40   * {@code TemplateHTMLOutputModel} of the template's own output format, FreeMarker recognizes it as pre-escaped
41   * markup and does not re-escape it.
42   *
43   * <p>
44   * Registered as a {@link Configuration#setSharedVariable(String, TemplateModel) shared variable} on the augmented
45   * {@link Configuration} by {@link ConsoleFreemarkerMixin#resolveConfiguration}, so it is visible to {@code base.ftlh}
46   * and, transitively (once included), to the consumer's own template.
47   *
48   * <h5 class='section'>Argument coercion:</h5>
49   * <p>
50   * Both {@code domain} and {@code value} accept either a plain FTL string literal or an arbitrary wrapped Java object
51   * (e.g. an {@code Enum} constant passed directly, as in {@code value=SomeEnum.RELEASED}): each argument is resolved
52   * via {@link DeepUnwrap#unwrap(TemplateModel)}, then an unwrapped {@link Enum} contributes its {@link Enum#name()}
53   * and anything else contributes {@link String#valueOf(Object)}. Domain and value are lowercased then
54   * fail-closed against {@code ^[a-z0-9_-]+$}.
55   *
56   * @since 10.0.0
57   */
58  final class TagMethodModel implements TemplateMethodModelEx {
59  
60  	/** The shared-variable name {@code base.ftlh}'s {@code <#macro tag ...>} calls. */
61  	static final String NAME = "jcTagHtml";
62  
63  	@Override
64  	public Object exec(@SuppressWarnings("rawtypes") List arguments) throws TemplateModelException {
65  		if (arguments.size() != 2)
66  			throw new TemplateModelException(NAME + "(domain, value) requires exactly 2 arguments, got " + arguments.size() + ".");
67  		var domain = asPlainString(arguments.get(0));
68  		var value = asPlainString(arguments.get(1));
69  		String markup;
70  		try {
71  			markup = pillSpan(domain, value).toString();
72  		} catch (IllegalArgumentException ex) {
73  			throw new TemplateModelException(ex.getMessage(), ex);
74  		}
75  		return HTMLOutputFormat.INSTANCE.fromMarkup(markup);
76  	}
77  
78  	private static Span pillSpan(String domain, String value) {
79  		var d = normalize("domain", domain);
80  		var v = normalize("value", value);
81  		return new Span().class_("tag " + d + " " + v);
82  	}
83  
84  	private static String normalize(String argName, String raw) {
85  		if (n(raw))
86  			throw new IllegalArgumentException("Tag " + argName + " must not be null.");
87  		var lower = raw.toLowerCase(Locale.ROOT);
88  		if (! lower.matches("^[a-z0-9_-]+$"))
89  			throw new IllegalArgumentException(
90  				"Invalid tag " + argName + ": '" + raw + "'.  Must match ^[a-z0-9_-]+$ (after lowercasing).");
91  		return lower;
92  	}
93  
94  	private static String asPlainString(Object arg) throws TemplateModelException {
95  		var raw = DeepUnwrap.unwrap((TemplateModel) arg);
96  		if (raw instanceof Enum<?> e)
97  			return e.name();
98  		return String.valueOf(raw);
99  	}
100 }
