28  /**
29   * Declares a console data source: a browser loader that fetches rows from a same-origin URL, builds a columnar frame
30   * and publishes it on the topic <c>data:&lt;id&gt;</c>, which chart cards read with {@code ChartDef.source(...)}.
31   *
32   * <p>
33   * Add it to a page with {@code PageSpec.data(...)}.  The declaration is checked when it is added: a bad URL, mode,
34   * chunking, column type or subscription role is E-76, and a bad parser or builder name is E-74.
35   *
36   * <h5 class='section'>Example:</h5>
37   * <p class='bjava'>
38   * 	<jv>page</jv>.data(DataSourceDef.<jsm>create</jsm>(<js>"orgRequests"</js>)
39   * 		.url(<js>"/rest/orgRequests/series"</js>).format(Format.<jsf>CSV</jsf>)
40   * 		.params(Subscription.<jsm>to</jsm>(Topics.<jsm>filter</jsm>(<js>"range"</js>)).as(<js>"params"</js>)
41   * 			.map(<js>"start"</js>, <js>"start"</js>).map(<js>"end"</js>, <js>"end"</js>))
42   * 		.chunk(Chunking.<jsm>byDay</jsm>(<js>"start"</js>, <js>"end"</js>)).cache(Cache.<jsm>indexedDb</jsm>()));
43   * </p>
44   *
45   * @since 10.0.0
46   */
47  @SuppressWarnings({
48  	"java:S3776" // Validation and rendering chains many independent guards; splitting them adds no clarity.
49  })
50  public final class DataSourceDef {
51  
52  	static final String E76 = "E-76: data source '%s': %s";
53  	static final Set<String> COLUMN_TYPES = Set.of("number", "time", "string", "bool");
54  	static final Set<String> BUCKETS = Set.of("hour", "day", "week", "month", "auto");
55  	static final Pattern FIELD = Pattern.compile("^[^,()\\s]+$");
56  	static final Pattern MEASURE = Pattern.compile("^(count|(sum|avg|min|max)\\([^,()\\s]+\\))$");
57  	static final Pattern DOTTED_PATH = Pattern.compile("^[A-Za-z0-9_-]++(?:\\.[A-Za-z0-9_-]++)*+$");
58  
59  	/** The built-in parser for each wire format. */
60  	public enum Format {
61  		/** RFC 4180 CSV with a header row; built-in parser <c>csv</c>. */
62  		CSV,
63  		/** One JSON object per line; built-in parser <c>ndjson</c>. */
64  		NDJSON,
65  		/** An array of rows or <c>{rows, truncated}</c>; built-in parser <c>json</c>. */
66  		JSON
67  	}
68  
69  	/** How the frame is produced. */
70  	public enum Mode {
71  		/** The rows as streamed. */
72  		RAW,
73  		/** One request to the aggregate URL. */
74  		AGGREGATE,
75  		/** RAW, switching to AGGREGATE past {@link DataSourceDef#maxRows(long)} or on a truncated chunk. */
76  		AUTO
77  	}

79  	/** How the window is split into requests; UTC calendar units. */
80  	public static final class Chunking {
81  		final String unit;
82  		final String start;
83  		final String end;
84  
85  		private Chunking(String unit, String start, String end) {
86  			this.unit = unit;
87  			this.start = start;
88  			this.end = end;
89  		}
90  
91  		/**
92  		 * One request per UTC hour.
93  		 *
94  		 * @param start The query parameter (and params name) carrying the window start.
95  		 * @param end The query parameter (and params name) carrying the window end.
96  		 * @return A new chunking.
97  		 */
98  		public static Chunking byHour(String start, String end) { return new Chunking("hour", start, end); }
99  
100 		/**
101 		 * One request per UTC day.
102 		 *
103 		 * @param start The window-start parameter name.
104 		 * @param end The window-end parameter name.
105 		 * @return A new chunking.
106 		 */
107 		public static Chunking byDay(String start, String end) { return new Chunking("day", start, end); }
108 
109 		/**
110 		 * One request per ISO week (Monday start, UTC).
111 		 *
112 		 * @param start The window-start parameter name.
113 		 * @param end The window-end parameter name.
114 		 * @return A new chunking.
115 		 */
116 		public static Chunking byWeek(String start, String end) { return new Chunking("week", start, end); }
117 
118 		/** @return No chunking: one request. */
119 		public static Chunking none() { return new Chunking(null, null, null); }
120 	}

139 	public static final class Aggregate {
140 		final String bucket;
141 		final List<String> groupBy = new ArrayList<>();
142 		final List<String> measures = new ArrayList<>();
143 
144 		private Aggregate(String bucket) { this.bucket = bucket; }
145 
146 		/**
147 		 * @param unit <c>hour</c>, <c>day</c>, <c>week</c>, <c>month</c> or <c>auto</c>.
148 		 * @return A new aggregate spec.
149 		 */
150 		public static Aggregate bucket(String unit) { return new Aggregate(unit); }
151 
152 		/** @param fields Up to 3 dims. @return This object. */
153 		public Aggregate groupBy(String...fields) { Collections.addAll(groupBy, fields); return this; }
154 
155 		/**
156 		 * @param values <c>count</c>, <c>sum(f)</c>, <c>avg(f)</c>, <c>min(f)</c> or <c>max(f)</c>.
157 		 * @return This object.
158 		 */
159 		public Aggregate measure(String...values) { Collections.addAll(measures, values); return this; }
160 	}

184 	static String checkedId(String id) {
185 		if (! FtlAttrLists.ID_PATTERN.matcher(id).matches())

187 		return id;

203 	private IllegalArgumentException e76(String reason, Object...args) {
204 		return iaex(E76, id, f(reason, args));

207 	private void required(Object value, String what) {
208 		if (value == null)
209 			throw e76("%s is null", what);
210 	}

212 	private String sameOrigin(String what, String value) {
213 		if (! SafePathTemplate.isSameOriginPath(value))
214 			throw e76("%s '%s' is not a same-origin path", what, value);
215 		if (value.indexOf('#') >= 0)
216 			throw e76("%s '%s' has a fragment", what, value);
217 		return value;
218 	}

230 	/**
231 	 * @param name A registered parser: a bare built-in name or a dotted app name.  Only the name's shape is checked
232 	 * 	here; a bare name other than the built-ins (<c>csv</c>, <c>ndjson</c>, <c>json</c>) is resolved only in the
233 	 * 	browser, where an unknown one is E-JS-92.
234 	 * @return This object.
235 	 */
236 	public DataSourceDef parser(String name) { parser = extensionName(name, "parser"); return this; }
237 
238 	/**
239 	 * @param name A registered builder: a bare built-in name or a dotted app name.  Only the name's shape is checked
240 	 * 	here; a bare name other than the built-in <c>columnar</c> is resolved only in the browser, where
241 	 * 	an unknown one is E-JS-93.
242 	 * @return This object.
243 	 */
244 	public DataSourceDef builder(String name) { builder = extensionName(name, "builder"); return this; }

246 	static String extensionName(String name, String what) {
247 		if (name == null || ! BusWiringValidator.EXTENSION_NAME.matcher(name).matches())
248 			throw new IllegalArgumentException("E-74: " + f(BusWiringValidator.E74, name, what));
249 		return name;
250 	}
251 
252 	/** @param specs Columns as <c>name:type</c>, type one of number, time, string, bool. @return This object. */
253 	public DataSourceDef columns(String...specs) {
254 		for (var s : specs) {
255 			if (s == null)
256 				throw e76("column is null");
257 			var i = s.indexOf(':');
258 			if (i < 1 || s.substring(0, i).isBlank())
259 				throw e76("column '%s' needs the form name:type", s);
260 			var type = s.substring(i + 1);
261 			if (! COLUMN_TYPES.contains(type))
262 				throw e76("column '%s' type '%s' is not number, time, string or bool", s, type);
263 			var name = s.substring(0, i);
264 			if (columns.stream().anyMatch(c -> name.equals(c.get("name"))))
265 				throw e76("column '%s' is declared twice", name);
266 			columns.add(JsonMap.of("name", name, "type", type));
267 		}
268 		return this;
269 	}
270 
271 	/** @param subs Subscriptions with role <c>params</c> or <c>refresh</c>. @return This object. */
272 	public DataSourceDef params(Subscription...subs) {
273 		for (var s : subs) {
274 			required(s, "subscription");
275 			var m = s.toMap();
276 			var role = m.get("as");
277 			if (! "params".equals(role) && ! "refresh".equals(role))
278 				throw e76("subscription to '%s' has role '%s'; a data source accepts params or refresh",
279 					s.topic(), role);
280 			subscribes.add(m);
281 		}
282 		return this;
283 	}

285 	/**
286 	 * @param value The chunking; {@link Chunking#none()} by default.
287 	 * @return This object.
288 	 * @throws IllegalArgumentException E-76 if the value is null, or the window param names are blank or equal.
289 	 */
290 	public DataSourceDef chunk(Chunking value) {
291 		required(value, "chunk");
292 		if (value.unit != null && (value.start == null || value.start.isBlank() || value.end == null
293 			|| value.end.isBlank() || value.start.equals(value.end)))
294 			throw e76("chunk window params '%s' and '%s' must be non-blank and different", value.start, value.end);
295 		chunk = value;
296 		return this;
297 	}

301 		if (value < 1 || value > 6)

318 	/**
319 	 * @param epochUrl A same-origin path probed once per load.
320 	 * @param path The dotted path of the epoch value in the JSON response.
321 	 * @return This object.
322 	 */
323 	public DataSourceDef epoch(String epochUrl, String path) {
324 		var checkedUrl = sameOrigin("epoch url", epochUrl);
325 		if (path == null || ! DOTTED_PATH.matcher(path).matches())
326 			throw e76("epoch path '%s' is not a dotted path", path);
327 		this.epochUrl = checkedUrl;
328 		this.epochPath = path;
329 		return this;
330 	}

335 	/** @param value The row count past which AUTO switches to AGGREGATE. @return This object. */
336 	public DataSourceDef maxRows(long value) {

343 	/** @param value A same-origin path answering the aggregate request. @return This object. */

346 	/** @param value The aggregate request. @return This object. */
347 	public DataSourceDef aggregate(Aggregate value) {
348 		required(value, "aggregate");
349 		if (value.bucket == null || ! BUCKETS.contains(value.bucket))
350 			throw e76("aggregate bucket '%s' is not hour, day, week, month or auto", value.bucket);
351 		for (var g : value.groupBy)
352 			if (g == null || ! FIELD.matcher(g).matches())
353 				throw e76("aggregate groupBy field '%s' is not a field name", g);
354 		for (var m : value.measures)
355 			if (m == null || ! MEASURE.matcher(m).matches())
356 				throw e76("aggregate measure '%s' is not count, sum(f), avg(f), min(f) or max(f)", m);
357 		if (value.groupBy.size() > 3)
358 			throw e76("aggregate has %s groupBy fields; at most 3", value.groupBy.size());
359 		var copy = Aggregate.bucket(value.bucket);
360 		copy.groupBy.addAll(value.groupBy);
361 		copy.measures.addAll(value.measures);
362 		aggregate = copy;
363 		return this;
364 	}

366 	/**
367 	 * The <c>contract.dataSources[]</c> entry; checks the cross-field rules.
368 	 *
369 	 * @return A new map.
370 	 * @throws IllegalArgumentException E-76 if the declaration is incomplete or inconsistent.
371 	 */
372 	public JsonMap toMap() {
373 		if (url == null)
374 			throw e76("url is required");
375 		var modeName = mode.name().toLowerCase(Locale.ROOT);
376 		if (mode != Mode.RAW && (aggregateUrl == null || aggregate == null))
377 			throw e76("mode %s needs aggregateUrl(...) and aggregate(...)", modeName);
378 		if (mode == Mode.AUTO && maxRows == null)
379 			throw e76("mode auto needs maxRows(n) greater than 0");
380 		if (chunk.unit != null) {
381 			var mapped = new HashSet<String>();
382 			for (var s : subscribes)
383 				if ("params".equals(s.get("as")) && s.get("map") instanceof Map<?,?> mm)
384 					mm.keySet().forEach(k -> mapped.add(String.valueOf(k)));
385 			for (var w : List.of(chunk.start, chunk.end))
386 				if (! mapped.contains(w))
387 					throw e76("chunk window param '%s' is not mapped by any params subscription", w);
388 		}
389 		var m = JsonMap.of("id", id, "kind", "source", "url", url, "parser", parser, "builder", builder);
390 		if (! columns.isEmpty())
391 			m.put("columns", columns.stream().map(JsonMap::new).toList());
392 		m.put("subscribes", subscribes.stream().map(JsonMap::new).toList());
393 		if (chunk.unit != null)
394 			m.put("chunk", JsonMap.of("unit", chunk.unit, "start", chunk.start, "end", chunk.end));
395 		m.put("concurrency", concurrency);
396 		m.put("debounceMs", debounceMs);
397 		m.put("cache", cache.kind);
398 		if (epochUrl != null)
399 			m.put("epoch", JsonMap.of("url", epochUrl, "path", epochPath));
400 		m.put("mode", modeName);
401 		if (maxRows != null)
402 			m.put("maxRows", maxRows);
403 		if (aggregateUrl != null)
404 			m.put("aggregateUrl", aggregateUrl);
405 		if (aggregate != null)
406 			m.put("aggregate", JsonMap.of("bucket", aggregate.bucket, "groupBy", List.copyOf(aggregate.groupBy),
407 				"measures", List.copyOf(aggregate.measures.isEmpty() ? List.of("count") : aggregate.measures)));
408 		return m;
409 	}
