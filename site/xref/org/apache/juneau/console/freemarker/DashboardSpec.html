23  /**
24   * A dashboard: a rail of filter sections beside a grid of card panels, with a status header and optional persistence.
25   *
26   * <p>
27   * Added to a builder page with {@link PageSpec#dashboard(DashboardSpec)}, which adds the panel and card-section cards
28   * to the page and the {@code dashboard} toolkit pack.  A page has at most one dashboard.  Every structural problem is
29   * E-77 when the page is built.
30   * </p>
31   *
32   * <h5 class='section'>Example:</h5>
33   * <p class='bjava'>
34   * 	<jv>page</jv>.dashboard(DashboardSpec.<jsm>create</jsm>(<js>"orders"</js>).title(<js>"Orders"</js>)
35   * 		.rail(Rail.<jsm>left</jsm>().sections(
36   * 			RangeSection.<jsm>quick</jsm>(<js>"7d"</js>, <js>"30d"</js>).publishes(Topics.<jsm>filter</jsm>(<js>"range"</js>))))
37   * 		.grid(2)
38   * 		.panel(<jv>daily</jv>).panel(<jv>byReason</jv>).panel(<js>"notes"</js>, 2)
39   * 		.status(Topics.<jsm>data</jsm>(<js>"orders"</js>))
40   * 		.persist(PersistSpec.<jsm>page</jsm>(<js>"app.orders"</js>).topics(Topics.<jsm>filter</jsm>(<js>"range"</js>)).shareLink()));
41   * </p>
42   *
43   * @since 10.0.0
44   */
45  @SuppressWarnings({
46  	"java:S3776" // Validation and rendering chains many independent guards; splitting them adds no clarity.
47  })
48  public final class DashboardSpec {
49  
50  	/** A panel: a card this dashboard adds, or the id of a card already on the page. */
51  	private record Panel(CardSpec card, String cardId, int span) {
52  		String id() {
53  			return card != null ? card.id() : cardId;
54  		}
55  	}

57  	private final String id;
58  	private String title;
59  	private Rail rail;
60  	private int grid = 2;
61  	private final List<Panel> panels = new ArrayList<>();
62  	private final List<String> status = new ArrayList<>();
63  	private PersistSpec persist;

215 	/** @return The cards this dashboard adds to the page: the built panels, then the card sections, in order. */
216 	List<CardSpec> ownedCards() {
217 		var out = new ArrayList<CardSpec>(panels.stream().map(Panel::card).filter(Objects::nonNull).toList());
218 		if (rail != null)
219 			for (var s : rail.sections())
220 				out.addAll(s.cards());
221 		return out;
222 	}

224 	/**
225 	 * Checks this dashboard against no page cards and returns its JSON.  {@code PageSpec.dashboard} calls
226 	 * {@code toJson(Map)} instead, so by-id panels resolve.
227 	 *
228 	 * @return The {@code dashboard} contract object.
229 	 * @throws IllegalArgumentException E-77 on the first problem.
230 	 */
231 	public Map<String,Object> toJson() {
232 		return toJson(Map.of());
233 	}
234 
235 	/**
236 	 * Checks this dashboard and returns its JSON.
237 	 *
238 	 * @param pageCards The cards already on the page, by id.
239 	 * @return The {@code dashboard} contract object.
240 	 * @throws IllegalArgumentException E-77 on the first problem.
241 	 */
242 	Map<String,Object> toJson(Map<String,CardSpec> pageCards) {
243 		if (id == null || ! Topics.ID.matcher(id).matches())
244 			throw Section.e77(id, "dashboard id '" + id + "' is not a valid id");
245 		if (grid < 1 || grid > 4)
246 			throw Section.e77(id, "grid " + grid + " is outside 1-4");
247 
248 		var members = new LinkedHashMap<String,CardSpec>();
249 		var panelList = new ArrayList<Map<String,Object>>();
250 		for (var p : panels) {
251 			CardSpec card;
252 			if (p.card() != null) {
253 				card = p.card();
254 				Section.checkOwnedCard(id, card);
255 			} else {
256 				card = p.cardId() == null ? null : pageCards.get(p.cardId());
257 				if (card == null)
258 					throw Section.e77(id, "panel '" + p.cardId() + "' names no card on this page");
259 				Section.checkMemberCard(id, card);
260 			}
261 			if (p.span() < 1 || p.span() > grid)
262 				throw Section.e77(id, "panel '" + p.id() + "' span " + p.span() + " is outside 1-" + grid);
263 			if (members.put(card.id(), card) != null)
264 				throw Section.e77(id, "card '" + card.id() + "' is in the dashboard twice");
265 			var m = new LinkedHashMap<String,Object>();
266 			m.put("card", card.id());
267 			m.put("span", p.span());
268 			panelList.add(m);
269 		}
270 
271 		var sectionIds = new LinkedHashSet<String>();
272 		if (rail != null) {
273 			rail.check(id);
274 			for (var s : rail.sections()) {
275 				for (var c : s.cards())
276 					if (members.put(c.id(), c) != null)
277 						throw Section.e77(id, "card '" + c.id() + "' is in the dashboard twice");
278 				if (s.id() != null && ! sectionIds.add(s.id()))
279 					throw Section.e77(id, "section id '" + s.id() + "' is used twice");
280 			}
281 			for (var sid : sectionIds)
282 				if (members.containsKey(sid) || pageCards.containsKey(sid))
283 					throw Section.e77(id, "section id '" + sid + "' duplicates card '" + sid + "'");
284 		}
285 
286 		var statusTopics = new HashSet<String>();
287 		for (var t : status) {
288 			if (t == null || ! t.startsWith("data:") || ! Topics.isKey(t.substring(5)))
289 				throw Section.e77(id, "status topic '" + t + "' is not a data: topic");
290 			if (! statusTopics.add(t))
291 				throw Section.e77(id, "status lists topic '" + t + "' twice");
292 		}
293 
294 		if (persist != null) {
295 			persist.check(id);
296 			for (var t : persist.topics()) {
297 				var key = t.substring(t.indexOf(':') + 1);
298 				if (t.startsWith("filter:")) {
299 					if (! sectionIds.contains(key))
300 						throw Section.e77(id, "persist topic '" + t + "' names no section of this dashboard");
301 				} else if (! publishes(members.get(key), t)) {
302 					throw Section.e77(id, "persist topic '" + t + "' is not published by a card of this dashboard");
303 				}
304 			}
305 		}
306 
307 		var m = new LinkedHashMap<String,Object>();
308 		m.put("id", id);
309 		if (title != null)
310 			m.put("title", title);
311 		m.put("grid", grid);
312 		m.put("panels", panelList);
313 		if (rail != null)
314 			m.put("rail", rail.toMap());
315 		if (! status.isEmpty())
316 			m.put("status", List.copyOf(status));
317 		if (persist != null)
318 			m.put("persist", persist.toMap());
319 		return m;
320 	}
321 
322 	/** @return <jk>true</jk> if {@code card} lists {@code topic} in its {@code publishes}. */
323 	private static boolean publishes(CardSpec card, String topic) {
324 		if (card == null || ! (card.toMap().get("publishes") instanceof List<?> list))
325 			return false;
326 		for (var o : list)
327 			if (o instanceof Map<?,?> p && topic.equals(p.get("topic")))
328 				return true;
329 		return false;
330 	}
