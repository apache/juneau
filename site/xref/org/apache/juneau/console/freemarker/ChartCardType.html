27  /**
28   * The built-in {@code chart} card type: a JSON5 body naming a {@code helper}, a legacy flat {@code kind}, or a raw
29   * ECharts {@code option}, with data from exactly one of {@code src=}, an inline {@code data} array or a {@code source}
30   * ({@code data:} topic).  {@code juneau-charts.js} renders it with ECharts.  The kinds are fixed: {@code line},
31   * {@code bar}, {@code donut}, {@code heatmap}, {@code scatter}, {@code pie} and {@code treemap}.
32   *
33   * <p>
34   * The one bus role is {@code data} (a source card's {@code data:<key>} subscription), and the ops are the eight chart
35   * commands ({@code highlight}, {@code downplay}, {@code showTip}, {@code hideTip}, {@code zoomTo},
36   * {@code legendSelect}, {@code export} and {@code refresh}), so Java's E-44 agrees with the client's E-JS-47.
37   * The checks:
38   * <ul>
39   * 	<li>{@code helper} (an object with a kind) and a legacy {@code kind} are mutually exclusive, and one of
40   * 		{@code helper}, {@code kind} or {@code option} is required;
41   * 	<li>{@code option} and {@code patch} are JSON objects, and {@code decorators}, {@code on} and every {@code $fn}
42   * 		marker name must be valid extension names (E-74);
43   * 	<li>{@code renderer} is {@code canvas} or {@code svg}, {@code maps} values are same-origin paths and
44   * 		{@code partialFrames} needs a {@code source};
45   * 	<li>every line/bar/scatter series needs a field, a donut or pie needs string {@code label} and {@code value}
46   * 		fields, a heatmap needs {@code x}, {@code y} and a string {@code value}, a treemap needs {@code groupBy} and a string
47   * 		{@code label} (with an optional string {@code size}, a {@code color} key and {@code tileFields});
48   * 	<li>{@code height} must be at least {@link ChartDef#MIN_HEIGHT}, and {@code onClick.href} must be a same-origin
49   * 		path ({@link SafePathTemplate#isSameOriginPath(String)}), the same checks {@link ChartDef} makes;
50   * 	<li>{@code onClick} holds an {@code href} or an {@code action}, never both.  The {@code action} (also each
51   * 		{@code topicActions[]} entry, whose topic must be one {@code on} publishes, and each {@code actions[]} catalog
52   * 		entry, which must be {@code present: "dialog"}) needs a unique id, a same-origin {@code endpoint} (and
53   * 		{@code form}), a POST/PUT/PATCH/DELETE {@code method}, a string or {@code false} {@code confirm}, no
54   * 		{@code present} but {@code dialog} and no {@code onSuccess: "navigate"}.
55   * </ul>
56   *
57   * <h5 class='section'>Example:</h5>
58   * <p class='bcode'>
59   * 	&lt;@card type="chart" id="orders-trend" title="Orders" src="/console/store/stats/daily"&gt;
60   * 	{ kind: 'line', x: { field: 'day' }, series: [ { field: 'placed', label: 'Placed' } ] }
61   * 	&lt;/@card&gt;
62   * </p>
63   *
64   * @since 10.0.0
65   */

98  	// The card's own retained zoom:/legend: topics, when it publishes them (juneau-charts.js's own implicit() mirrors this; ChartWiring_JsParity_Test pins them).
99  	@Override
100 	public List<String> implicitTopics(JsonMap card) {
101 		var id = card.get("id");
102 		var out = new ArrayList<String>();
103 		if (! (id instanceof String cardId) || ! (card.get("publishes") instanceof List<?> pubs))
104 			return out;
105 		for (var t : List.of(Topics.zoom(cardId), Topics.legend(cardId)))
106 			for (var p : pubs)
107 				if (p instanceof Map<?,?> m && t.equals(m.get("topic")) && Boolean.TRUE.equals(m.get("retain"))) {
108 					out.add(t);
109 					break;
110 				}
111 		return out;
112 	}

114 	@Override
115 	public JsonMap toFragment(CardSource source) {
116 		if (! source.hasJsonBody())
117 			throw source.error("type='chart' requires a JSON5 body.");
118 		var body = source.json();
119 		var hasSrc = nn(source.src());
120 		var hasData = body.containsKey("data");
121 		var hasSource = body.containsKey("source");
122 		if ((hasSrc ? 1 : 0) + (hasData ? 1 : 0) + (hasSource ? 1 : 0) != 1)
123 			throw source.error("type='chart' requires exactly one of src=, a data body or a source.");
124 		if (hasSrc && ! SafePathTemplate.isSameOriginPath(source.src()))
125 			throw source.error("type='chart' src must be a same-origin path; got '%s'.", source.src());
126 		if (hasData && ! (body.get("data") instanceof List))
127 			throw source.error("type='chart' data must be an array of rows.");
128 		if (hasSource && ! (body.get("source") instanceof String s && s.startsWith("data:") && s.length() > 5))
129 			throw source.error("type='chart' source must be a data: topic; got '%s'.", body.get("source"));
130 		if (body.containsKey("helper") && body.containsKey("kind"))
131 			throw source.error("type='chart' takes 'helper' or a legacy 'kind', not both.");
132 		if (! body.containsKey("helper") && ! body.containsKey("kind") && ! body.containsKey("option"))
133 			throw source.error("type='chart' requires 'helper', a legacy 'kind' or an 'option'.");
134 		if (body.containsKey("helper") && ! (body.get("helper") instanceof Map))
135 			throw source.error("type='chart' helper must be a JSON object.");
136 		Map<?,?> helper = null;
137 		if (body.get("helper") instanceof Map<?,?> h)
138 			helper = h;
139 		else if (body.containsKey("kind"))
140 			helper = body;
141 		if (helper != null)
142 			checkHelper(source, helper);
143 		for (var k : List.of("option", "patch"))
144 			if (body.containsKey(k) && ! (body.get(k) instanceof Map))
145 				throw source.error("type='chart' %s must be a JSON object.", k);
146 		if (body.containsKey("decorators")) {
147 			if (! (body.get("decorators") instanceof List<?> l) || ! l.stream().allMatch(d -> d instanceof Map<?,?> m && m.get("use") instanceof String))
148 				throw source.error("type='chart' decorators must be an array of objects with a string 'use'.");
149 			for (var d : (List<?>)body.get("decorators"))
150 				checkName(source, (String)((Map<?,?>)d).get("use"), "decorator");
151 		}
152 		if (body.containsKey("on")) {
153 			if (! (body.get("on") instanceof Map<?,?> on) || ! on.values().stream().allMatch(String.class::isInstance))
154 				throw source.error("type='chart' on must map event names to topic strings.");
155 			for (var ev : on.keySet())
156 				if (! BusWiringValidator.EVENT_NAME.matcher(String.valueOf(ev)).matches())
157 					throw source.error("%s", "E-74: " + f(BusWiringValidator.E74, ev, "event"));
158 		}
159 		var renderer = body.get("renderer");
160 		if (nn(renderer) && ! Set.of("canvas", "svg").contains(renderer))
161 			throw source.error("type='chart' renderer must be 'canvas' or 'svg'; got '%s'.", renderer);
162 		if (body.get("maps") instanceof Map<?,?> maps) {
163 			for (var e : maps.entrySet())
164 				if (! (e.getValue() instanceof String p) || ! SafePathTemplate.isSameOriginPath(p))
165 					throw source.error("type='chart' maps.%s must be a same-origin path; got '%s'.", e.getKey(), e.getValue());
166 		}
167 		if (body.containsKey("maps") && ! (body.get("maps") instanceof Map))
168 			throw source.error("type='chart' maps must be a JSON object.");
169 		if (body.containsKey("partialFrames") && ! (body.get("partialFrames") instanceof Boolean))
170 			throw source.error("type='chart' partialFrames must be a boolean.");
171 		for (var k : List.of("group", "theme"))
172 			if (body.containsKey(k) && ! (body.get(k) instanceof String))
173 				throw source.error("type='chart' %s must be a string.", k);
174 		for (var k : List.of("largeData", "a11y"))
175 			if (body.containsKey(k) && ! (body.get(k) instanceof Map))
176 				throw source.error("type='chart' %s must be a JSON object.", k);
177 		if (Boolean.TRUE.equals(body.get("partialFrames")) && ! hasSource)
178 			throw source.error("type='chart' partialFrames needs a source.");
179 		walkFns(source, body.get("option"));
180 		walkFns(source, body.get("patch"));
181 		walkFns(source, body.get("decorators"));
182 		// helper keys are mostly field names, but a helper value can be a $fn marker (ChartDef.symbolSizeBy puts Fn.ref under
183 		// helper.symbolSize), so the helper -- or, for a legacy flat body, its HELPER_KEYS -- is walked with the same skip set.
184 		if (helper != null) {
185 			var hm = new JsonMap();
186 			for (var e : helper.entrySet())
187 				if (helper != body || ChartDef.HELPER_KEYS.contains(e.getKey()))
188 					hm.put(String.valueOf(e.getKey()), e.getValue());
189 			walkFns(source, hm);
190 		}
191 		var refresh = body.get("refreshMs");
192 		if (nn(refresh)) {
193 			if (! (refresh instanceof Integer || refresh instanceof Long))
194 				throw source.error("type='chart' refreshMs must be an integer; got '%s'.", refresh);
195 			if (! hasSrc)
196 				throw source.error("type='chart' refreshMs needs src=; inline data never refreshes.");
197 			if (((Number)refresh).longValue() < SafePathTemplate.MIN_POLL_INTERVAL_MS)
198 				throw source.error("type='chart' refreshMs='%s' is below the minimum '%s'.", refresh, SafePathTemplate.MIN_POLL_INTERVAL_MS);
199 		}
200 		var height = body.get("height");
201 		if (nn(height)) {
202 			if (! (height instanceof Integer || height instanceof Long))
203 				throw source.error("type='chart' height must be an integer; got '%s'.", height);
204 			if (((Number)height).longValue() < ChartDef.MIN_HEIGHT)
205 				throw source.error("type='chart' height='%s' is below the minimum '%s'.", height, ChartDef.MIN_HEIGHT);
206 		}
207 		var onClick = body.get("onClick");
208 		var ids = new HashSet<String>();
209 		if (nn(onClick))
210 			checkOnClick(source, body, onClick, ids);
211 		if (nn(body.get("topicActions")))
212 			checkTopicActions(source, body, ids);
213 		if (nn(body.get("actions")))
214 			checkCatalog(source, body.get("actions"), ids);
215 		var frag = new JsonMap();
216 		frag.putAll(body);
217 		frag.remove("visibleWhen");
218 		return frag;
219 	}

221 	// The helper (or a legacy flat body): the kind set, plus the legacy per-kind checks.
222 	private static void checkHelper(CardSource source, Map<?,?> h) {
223 		if (! (h.get("kind") instanceof String kind) || kind.isBlank())
224 			throw source.error("type='chart' requires a non-blank 'kind'.");
225 		if (! ChartDef.KINDS.contains(kind))
226 			throw source.error("type='chart' kind '%s' must be one of '%s'.", kind, String.join(", ", new TreeSet<>(ChartDef.KINDS)));
227 		if (CARTESIAN.contains(kind) && (h.get("x") == null || ! (h.get("series") instanceof List<?> s) || s.isEmpty()))
228 			throw source.error("type='chart' kind='%s' requires x and at least one series.", kind);
229 		if (CARTESIAN.contains(kind)) {
230 			var series = (List<?>)h.get("series");
231 			for (var i = 0; i < series.size(); i++)
232 				if (ib(seriesField(series.get(i))))
233 					throw source.error("type='chart' kind='%s' series[%s] requires a non-blank 'field'.", kind, i);
234 		}
235 		if (Set.of("donut", "pie").contains(kind) && (! (h.get("label") instanceof String l) || l.isBlank() || ! (h.get("value") instanceof String v) || v.isBlank()))
236 			throw source.error("type='chart' kind='%s' requires string 'label' and 'value' fields.", kind);
237 		if (eq(kind, "heatmap") && (h.get("x") == null || h.get("y") == null || ! (h.get("value") instanceof String)))
238 			throw source.error("type='chart' kind='heatmap' requires x, y and a string 'value' field.");
239 		if (eq(kind, "treemap"))
240 			checkTreemap(source, h);
241 	}

243 	private static void checkTreemap(CardSource source, Map<?,?> h) {
244 		if (! (h.get("groupBy") instanceof List<?> g) || g.isEmpty() || ! g.stream().allMatch(x -> x instanceof String s && ! s.isBlank()))
245 			throw source.error("type='chart' helper kind 'treemap' requires 'groupBy' as a non-empty array of non-blank strings.");
246 		if (! (h.get("label") instanceof String l) || l.isBlank())
247 			throw source.error("type='chart' helper kind 'treemap' requires a non-blank string 'label'.");
248 		if (h.containsKey("size") && ! (h.get("size") instanceof String sz && ! sz.isBlank()))
249 			throw source.error("type='chart' helper kind 'treemap' size must be a non-blank string.");
250 		if (h.containsKey("tileFields") && ! (h.get("tileFields") instanceof List<?> t && t.stream().allMatch(x -> x instanceof String s && ChartDef.TILE_FIELD.matcher(s).matches())))
251 			throw source.error("type='chart' helper kind 'treemap' tileFields must be an array of strings matching [A-Za-z0-9_.-]+.");
252 		if (h.containsKey("color")) {
253 			if (! (h.get("color") instanceof Map<?,?> c) || ! (c.get("field") instanceof String f) || f.isBlank())
254 				throw source.error("type='chart' helper kind 'treemap' color must be an object with a non-blank string 'field'.");
255 			var legend = ((Map<?,?>)h.get("color")).get("legend");
256 			if (nn(legend)) {
257 				if (! (legend instanceof List<?> ll) || ! ll.stream().allMatch(x -> x instanceof Map<?,?> m && m.get("value") instanceof String))
258 					throw source.error("type='chart' helper kind 'treemap' color.legend must be an array of objects with a string 'value'.");
259 				for (var i = 0; i < ((List<?>)legend).size(); i++) {
260 					var tone = ((Map<?,?>)((List<?>)legend).get(i)).get("tone");
261 					if (nn(tone) && ! (tone instanceof String ts && ChartDef.TREEMAP_TONES.contains(ts)))
262 						throw source.error("type='chart' helper kind 'treemap' color.legend[%s].tone '%s' must be 'ok', 'warning', 'danger', '--jc-chart-1'..'--jc-chart-8' or absent.", i, tone);
263 				}
264 			}
265 		}
266 	}

268 	private static void checkOnClick(CardSource source, Map<?,?> body, Object onClick, Set<String> ids) {
269 		if (! (onClick instanceof Map<?,?> m))
270 			throw source.error("type='chart' onClick must be an object with a string href.");
271 		for (var k : m.keySet())
272 			if (! "href".equals(k) && ! "action".equals(k))
273 				throw source.error("type='chart' onClick has unknown key '%s'.", k);
274 		if (m.containsKey("href") && m.containsKey("action"))
275 			throw source.error("type='chart' onClick takes either href or action, not both.");
276 		if (m.containsKey("action")) {
277 			if (! (body.get("on") instanceof Map<?,?> on) || ! (on.get("click") instanceof String))
278 				throw source.error("type='chart' onClick.action needs on.click to declare the click topic.");
279 			checkAction(source, "onClick.action", m.get("action"), ids);
280 		} else if (! (m.get("href") instanceof String href)) {
281 			throw source.error("type='chart' onClick must be an object with a string href.");
282 		} else if (! SafePathTemplate.isSameOriginPath(href)) {
283 			throw source.error("type='chart' onClick.href must be a same-origin path; got '%s'.", href);
284 		}
285 	}

287 	private static void checkTopicActions(CardSource source, Map<?,?> body, Set<String> ids) {
288 		var shape = "type='chart' topicActions must be an array of objects with a string topic and an action object.";
289 		if (! (body.get("topicActions") instanceof List<?> l))
290 			throw source.error("%s", shape);
291 		var topics = body.get("on") instanceof Map<?,?> on ? on.values() : List.of();
292 		for (var i = 0; i < l.size(); i++) {
293 			if (! (l.get(i) instanceof Map<?,?> e) || ! (e.get("topic") instanceof String topic) || ! (e.get("action") instanceof Map))
294 				throw source.error("%s", shape);
295 			if (! topics.contains(topic))
296 				throw source.error("type='chart' topicActions[%s].topic '%s' must be one this card publishes through 'on'.", i, topic);
297 			checkAction(source, "topicActions[" + i + "].action", e.get("action"), ids);
298 		}
299 	}

301 	private static void checkCatalog(CardSource source, Object actions, Set<String> ids) {
302 		if (! (actions instanceof List<?> l))
303 			throw source.error("type='chart' actions must be an array of action objects.");
304 		for (var i = 0; i < l.size(); i++) {
305 			var where = "actions[" + i + "]";
306 			checkAction(source, where, l.get(i), ids);
307 			var a = (Map<?,?>)l.get(i);
308 			if (! "dialog".equals(a.get("present")))
309 				throw source.error("type='chart' %s '%s' must be present=dialog; a dialog button can only open a dialog action.", where, a.get("id"));
310 		}
311 	}

313 	// The rules every chart action obeys, wherever it is declared: a chart has no cell, page or servlet to hand off to.
314 	private static void checkAction(CardSource source, String where, Object action, Set<String> ids) {
315 		if (! (action instanceof Map<?,?> a))
316 			throw source.error("type='chart' %s must be an object.", where);
317 		if (! (a.get("id") instanceof String id) || id.isBlank())
318 			throw source.error("type='chart' %s requires a non-blank string id.", where);
319 		if (! (a.get("endpoint") instanceof String endpoint) || endpoint.isBlank())
320 			throw source.error("type='chart' %s requires a non-blank string endpoint.", where);
321 		if (! SafePathTemplate.isSameOriginPath(endpoint))
322 			throw source.error("type='chart' %s endpoint must be a same-origin path; got '%s'.", where, endpoint);
323 		if (nn(a.get("form")) && ! (a.get("form") instanceof String form && SafePathTemplate.isSameOriginPath(form)))
324 			throw source.error("type='chart' %s form must be a same-origin path; got '%s'.", where, a.get("form"));
325 		if (! (a.get("method") instanceof String method) || ! ACTION_METHODS.contains(method))
326 			throw source.error("type='chart' %s method must be POST, PUT, PATCH or DELETE; got '%s'.", where, a.get("method"));
327 		if (nn(a.get("confirm")) && ! (a.get("confirm") instanceof String || Boolean.FALSE.equals(a.get("confirm"))))
328 			throw source.error("type='chart' %s confirm must be a string or false.", where);
329 		if (nn(a.get("present")) && ! "dialog".equals(a.get("present")))
330 			throw source.error("type='chart' %s present must be 'dialog' or absent; got '%s'.", where, a.get("present"));
331 		if (nn(a.get("visibleWhen")) && ! (a.get("visibleWhen") instanceof List))
332 			throw source.error("type='chart' %s visibleWhen must be an array of rules; it is tested against the clicked point.", where);
333 		if (nn(a.get("enabledWhen")) && ! (a.get("enabledWhen") instanceof List))
334 			throw source.error("type='chart' %s enabledWhen must be an array of rules.", where);
335 		if ("navigate".equals(a.get("onSuccess")))
336 			throw source.error("type='chart' %s onSuccess must not be 'navigate'; a chart has nowhere to navigate.", where);
337 		if (! ids.add(id))
338 			throw source.error("type='chart' action id '%s' is declared more than once.", id);
339 	}

346 	// $fn markers: an object whose only keys are $fn and, optionally, args.  Same skip set as
347 	// BusWiringValidator.walkFns: dataset[*].source and series[*].data are rows of data, never extension references.
348 	private static void walkFns(CardSource source, Object o) {
349 		if (o instanceof Map<?,?> m) {
350 			if (m.containsKey("$fn") && m.keySet().stream().allMatch(k -> "$fn".equals(k) || "args".equals(k)))
351 				checkName(source, m.get("$fn") instanceof String s ? s : null, "$fn");
352 			for (var e : m.entrySet()) {
353 				if ("dataset".equals(e.getKey()))
354 					walkSkipping(source, e.getValue(), "source");
355 				else if ("series".equals(e.getKey()))
356 					walkSkipping(source, e.getValue(), "data");
357 				else
358 					walkFns(source, e.getValue());
359 			}
360 		} else if (o instanceof List<?> l) {
361 			l.forEach(v -> walkFns(source, v));
362 		}
363 	}
364 
365 	// Walks o (a map or a list of maps) without descending into the skip key of each map.
366 	private static void walkSkipping(CardSource source, Object o, String skip) {
367 		if (o instanceof List<?> l) {
368 			l.forEach(x -> walkSkipping(source, x, skip));
369 		} else if (o instanceof Map<?,?> m) {
370 			for (var e : m.entrySet())
371 				if (! skip.equals(e.getKey()))
372 					walkFns(source, e.getValue());
373 		}
374 	}

376 	// A series entry is a field name or an object with a field, as juneau-charts.js reads it.
377 	private static String seriesField(Object entry) {
378 		if (entry instanceof String s)
379 			return s;
380 		return entry instanceof Map<?,?> m && m.get("field") instanceof String s ? s : null;
