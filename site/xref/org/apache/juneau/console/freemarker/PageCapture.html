37  /**
38   * Typed, per-render capture of one console page: theme, header, footer, nav tree, active nav, cards
39   * and slot templates, held as FreeMarker {@link Environment} custom state (see {@link #KEY}) so every
40   * directive in the render shares one instance. {@code <@console>} serializes it as the {@code #juneau-page}
41   * contract.
42   *
43   * <h5 class='section'>Example:</h5>
44   * <p class='bjava'>
45   * 	<jc>// Inside a custom directive that contributes a card:</jc>
46   * 	<jk>public void</jk> execute(Environment <jv>env</jv>, Map <jv>params</jv>, TemplateModel[] <jv>loop</jv>,
47   * 			TemplateDirectiveBody <jv>body</jv>) <jk>throws</jk> TemplateException, IOException {
48   * 		PageCapture <jv>page</jv> = PageCapture.<jsm>of</jsm>(<jv>env</jv>);
49   * 		<jv>page</jv>.addCard(CardSpec.<jsm>html</jsm>(<js>"intro"</js>).title(<js>"Welcome"</js>),
50   * 			PageCapture.<jsm>render</jsm>(<jv>body</jv>));
51   * 	}
52   *
53   * 	<jc>// In a test, after rendering:</jc>
54   * 	String <jv>json</jv> = PageCapture.<jsm>get</jsm>(<jv>env</jv>).toContractJson();
55   * </p>
56   *
57   * @since 10.0.0
58   */

228 	/**
229 	 * Returns the capture for this render, creating it on first use.
230 	 *
231 	 * <p>
232 	 * On create, a {@link PageSpec} carried in the data model under {@link PageSpec#ATTR} is applied to the new
233 	 * capture before any directive writes to it, and marked adopted.
234 	 *
235 	 * @param env The FreeMarker environment.
236 	 * @return The capture, never <jk>null</jk>.
237 	 * @throws TemplateModelException If seeding a {@link PageSpec} from the data model fails.
238 	 */
239 	public static PageCapture of(Environment env) throws TemplateModelException {
240 		var c = (PageCapture)env.getCustomState(KEY);
241 		if (n(c)) {
242 			c = new PageCapture();
243 			var tm = env.getDataModel().get(PageSpec.ATTR);
244 			if (nn(tm) && DeepUnwrap.unwrap(tm) instanceof PageSpec.Seed seed) {
245 				var cardModel = env.getConfiguration().getSharedVariable(CardDirectiveModel.NAME);
246 				seed.adoptInto(c, cardModel instanceof CardDirectiveModel m ? m : null);
247 			}
248 			env.setCustomState(KEY, c);  // Only after a successful seed, so a failed one leaves nothing half-applied.
249 		}
250 		return c;
251 	}

341 	/**
342 	 * Appends a card.
343 	 *
344 	 * @param card The card.
345 	 * @param markupOrNull Markup emitted as {@code <template data-card="{id}">}, or <jk>null</jk>.
346 	 * @return This object.
347 	 * @throws TemplateModelException On a duplicate card id (E-9).
348 	 */
349 	public PageCapture addCard(CardSpec card, String markupOrNull) throws TemplateModelException {
350 		checkSrc(card);
351 		for (var c : cards)
352 			if (c.id().equals(card.id()))
353 				throw FtlAttrLists.reject(f("<@card id='%s'> duplicates an existing card id.", card.id()));
354 		if (builtCardIds.contains(card.id()))
355 			throw FtlAttrLists.reject(f("<@card id='%s'> duplicates an existing card id.", card.id()));
356 		if (nn(markupOrNull)) {
357 			card.template(card.id());
358 			templates.put(card.id(), markupOrNull);
359 		}
360 		cards.add(card);
361 		return this;
362 	}

364 	// Every card reaching the page passes through here: <@card> html cards, PageSpec.html / PageSpec.card, and direct
365 	// addCard callers.  Checks, in order: (1) a card's src is fetched by the browser, so it must be a same-origin path
366 	// ({token} placeholders allowed); (2) sanitize= is valid only on an html card with a src.  The src check wins when both fail.
367 	private static void checkSrc(CardSpec card) throws TemplateModelException {
368 		var src = card.src();
369 		if (nn(src) && ! SafePathTemplate.isSameOriginPath(src))
370 			throw FtlAttrLists.reject(f("<@card id='%s'> type='%s' src must be a same-origin path (no scheme, '//', backslash, control character, edge space or '..' segment); got '%s'.",
371 				card.id(), card.type(), src));
372 		if (card.sanitize() && (neq(card.type(), "html") || ! nn(src)))
373 			throw FtlAttrLists.reject(f("<@card id='%s'> sanitize= applies only to an html card with src=.", card.id()));
374 	}

376 	/**
377 	 * Registers a card built by a {@code PageSpec}.  It is held apart from the authored cards until placed or the
378 	 * page closes, so a spec's cards follow the template's cards in declaration order.
379 	 *
380 	 * @param card The card.
381 	 * @param markupOrNull Markup emitted as {@code <template data-card="{id}">}, or <jk>null</jk>.
382 	 * @throws TemplateModelException On a duplicate card id, checked against placed and unplaced cards.
383 	 */
384 	void addBuiltCard(CardSpec card, String markupOrNull) throws TemplateModelException {
385 		checkSrc(card);
386 		for (var c : cards)
387 			if (c.id().equals(card.id()))
388 				throw FtlAttrLists.reject(f("<@card id='%s'> duplicates an existing card id.", card.id()));
389 		if (builtCards.containsKey(card.id()) || builtCardIds.contains(card.id()))
390 			throw FtlAttrLists.reject(f("<@card id='%s'> duplicates an existing card id.", card.id()));
391 		if (nn(markupOrNull)) {
392 			card.template(card.id());
393 			templates.put(card.id(), markupOrNull);
394 		}
395 		builtCards.put(card.id(), card);
396 		builtCardIds.add(card.id());
397 	}

399 	/**
400 	 * Places a built card at the current position in the card order.
401 	 *
402 	 * @param id The built card id.
403 	 * @throws TemplateModelException If the card was already placed, or no built card has that id.
404 	 */
405 	void placeBuiltCard(String id) throws TemplateModelException {
406 		var card = builtCards.remove(id);
407 		if (n(card)) {
408 			if (builtCardIds.contains(id))
409 				throw FtlAttrLists.reject(f("<@card ref='%s'> places a PageSpec card that was already placed.", id));
410 			throw FtlAttrLists.reject(f("<@card ref='%s'> names no PageSpec card; declared: '%s'.", id,
411 				String.join(", ", builtCardIds)));
412 		}
413 		cards.add(card);
414 	}

437 	/**
438 	 * Adds a top-level {@code topics} entry.  Java ({@code PageSpec.topic}) and FTL ({@code <@topic>}) declarations
439 	 * union; the same topic again with the same {@code retain} is a no-op.
440 	 *
441 	 * @param decl A {@code TopicDecl.toMap()} entry.
442 	 * @param where Where it was declared, for the E-46 message, for example {@code "<@topic>"}.
443 	 * @return This object.
444 	 * @throws IllegalArgumentException E-46 if the topic is already declared with a different {@code retain}.
445 	 */
446 	PageCapture addTopic(Map<String,Object> decl, String where) {
447 		var topic = (String)decl.get("topic");
448 		var prior = topics.get(topic);
449 		if (prior == null) {
450 			topics.put(topic, decl);
451 			topicWhere.put(topic, where);
452 		} else if (neq(prior.get("retain"), decl.get("retain"))) {
453 			throw iaex("topic '%s' is declared with retain=%s here and retain=%s at %s",
454 				topic, decl.get("retain"), prior.get("retain"), topicWhere.get(topic));
455 		} else if (neq(prior.get("publisher"), decl.get("publisher"))) {
456 			throw iaex("topic '%s' is declared with publisher=%s here and publisher=%s at %s",
457 				topic, decl.get("publisher"), prior.get("publisher"), topicWhere.get(topic));
458 		}
459 		return this;
460 	}

576 		"unchecked" // Narrows a JSON-shaped Map<?,?> to Map<String,Object>; its keys are always strings.
577 	})
578 	private static Map<String,Object> deepMergeFacts(String prefix, Map<String,Object> a, Map<String,Object> b) throws TemplateModelException {
579 		var out = new LinkedHashMap<String,Object>(a);
580 		for (var e : b.entrySet()) {
581 			var path = prefix.isEmpty() ? e.getKey() : prefix + "." + e.getKey();
582 			if (! out.containsKey(e.getKey())) {
583 				out.put(e.getKey(), e.getValue());
584 				continue;
585 			}
586 			var existing = out.get(e.getKey());
587 			if (existing instanceof Map<?,?> && e.getValue() instanceof Map<?,?>)
588 				out.put(e.getKey(), deepMergeFacts(path, (Map<String,Object>)existing, (Map<String,Object>)e.getValue()));
589 			else
590 				throw FtlAttrLists.reject(f("Fact '%s' is set by both PageSpec.facts and the chrome; set it in one place.", path));
591 		}
592 		return out;
593 	}

729 	 * Merge-aware setter for the raw {@code <@body>} attribute text: union by attribute name, existing (spec) text
730 	 * first and then the new text.  The same attribute name on both sides is always rejected, even when the values
731 	 * agree.  Called once, from {@link #checkConsoleClose()}, with the text {@code <@body>} captured into
732 	 * {@link #ftlBodyAttrs}.

734 	 * @param attrs The raw {@code <@body>} attribute text.  Blank is a no-op.
735 	 * @return This object.
736 	 * @throws TemplateModelException If an attribute name already set is set again.
737 	 */
738 	public PageCapture mergeBodyAttrs(String attrs) throws TemplateModelException {
739 		if (ib(attrs))

754 	private static final Pattern BODY_ATTR_NAME = Pattern.compile("[A-Za-z_:][-A-Za-z0-9_:.]*");
755 
756 	/** One attribute found by {@link #tokenizeBodyAttrs(String)}: its span in the text, its name, and whether it has a value. */
757 	private record BodyAttr(int start, int end, String name, boolean hasValue) {}
758 
759 	// <@body> markup is raw, so a value can be double-quoted, single-quoted or unquoted.  Tokenizing (rather than
760 	// regex-searching the whole text) means a name is only ever read where an attribute can start, never from inside
761 	// another attribute's quoted value, e.g. title="data-juneau-csrf=x".
762 	private static List<BodyAttr> tokenizeBodyAttrs(String text) {
763 		var out = new ArrayList<BodyAttr>();
764 		var n = text.length();
765 		var i = 0;
766 		while (i < n) {
767 			if (Character.isWhitespace(text.charAt(i))) {
768 				i++;
769 				continue;
770 			}
771 			var start = i;
772 			while (i < n && ! Character.isWhitespace(text.charAt(i)) && text.charAt(i) != '=')
773 				i++;
774 			var name = text.substring(start, i);
775 			var j = i;
776 			while (j < n && Character.isWhitespace(text.charAt(j)))
777 				j++;
778 			var hasValue = false;
779 			if (j < n && text.charAt(j) == '=') {
780 				j++;
781 				while (j < n && Character.isWhitespace(text.charAt(j)))
782 					j++;
783 				if (j < n && (text.charAt(j) == '"' || text.charAt(j) == '\'')) {
784 					var close = text.indexOf(text.charAt(j), j + 1);
785 					j = close < 0 ? n : close + 1;
786 				} else {
787 					while (j < n && ! Character.isWhitespace(text.charAt(j)))
788 						j++;
789 				}
790 				hasValue = true;
791 				i = j;
792 			} else if (i == start) {
793 				i++;  // A stray '=': step over it so the scan always advances.
794 			}
795 			if (BODY_ATTR_NAME.matcher(name).matches())
796 				out.add(new BodyAttr(start, i, name, hasValue));
797 		}
798 		return out;
799 	}

801 	private static Set<String> bodyAttrNames(String text) {
802 		return tokenizeBodyAttrs(text).stream().filter(BodyAttr::hasValue).map(BodyAttr::name)
803 			.collect(Collectors.toCollection(() -> new TreeSet<>(String.CASE_INSENSITIVE_ORDER)));
804 	}

808 	 * CSRF attribute is not emitted a second time by the request-attribute auto-detection.
809 	 *
810 	 * @param name An HTML attribute name.
811 	 * @return Whether the merged text sets it.
812 	 */
813 	boolean hasBodyAttr(String name) {

817 	/**
818 	 * The merged body-attribute text with the named attributes taken out, for a response that must not carry them.
819 	 *
820 	 * @param names The HTML attribute names to drop.
821 	 * @return The remaining text, possibly blank, or <jk>null</jk> if no body attributes were set.
822 	 */
823 	String bodyAttrsWithout(String...names) {
824 		if (bodyAttrs == null)
825 			return null;
826 		var s = new StringBuilder(bodyAttrs);
827 		var attrs = tokenizeBodyAttrs(bodyAttrs);
828 		for (var k = attrs.size() - 1; k >= 0; k--) {
829 			var a = attrs.get(k);
830 			if (! a.hasValue() || Arrays.stream(names).noneMatch(a.name()::equalsIgnoreCase))
831 				continue;
832 			var from = a.start();
833 			while (from > 0 && Character.isWhitespace(s.charAt(from - 1)))
834 				from--;
835 			s.delete(from, a.end());
836 		}
837 		return s.toString().strip();
838 	}
