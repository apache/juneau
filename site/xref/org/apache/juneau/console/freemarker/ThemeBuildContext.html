17  package org.apache.juneau.console.freemarker;
18  
19  import java.util.*;
20  
21  import org.apache.juneau.console.*;
22  
23  /**
24   * Per-render token-capture context shared between a {@code <@theme>} element and the {@code <@token>} elements
25   * nested inside it, via {@link freemarker.core.Environment#setCustomState(Object, Object)}.
26   *
27   * <p>
28   * {@code <@theme name="…">} installs a fresh instance holding a {@link Theme.Builder} seeded from the named stock
29   * palette before rendering its body. Each nested {@code <@token>} eagerly folds its declaration into that builder:
30   * a {@code value=} into {@link Theme.Builder#token(String, String)} (a leaf), an {@code alias=} into
31   * {@link Theme.Builder#alias(String, String)} (a derived reference). After the body pass, {@code <@theme>} builds
32   * the theme and renders its override block. A {@code <@token scheme="dark">} folds into the separate dark maps
33   * instead, which {@code <@theme>} renders as the dark-scheme rule after the light block.
34   *
35   * @since 10.0.0
36   */
37  final class ThemeBuildContext {
38  
39  	/** The stock-theme name {@code <@theme name="…">} named - also the built {@link Theme}'s name. */
40  	final String name;
41  
42  	/** The theme builder, seeded with the named stock palette; {@code <@token value="…">} adds leaves and {@code <@token alias="var(--jc-…)">} adds aliases here. */
43  	final Theme.Builder themeBuilder;
44  
45  	/** Set true by the first {@code <@token>} that fires, so {@code <@theme>} emits an override block only when the body declared one. */
46  	boolean anyDeclared;
47  
48  	/** Set true by the first light {@code <@token>} (one without {@code scheme=}), so a dark-only body emits no light block. */
49  	boolean lightDeclared;
50  
51  	/** Dark-scheme leaf tokens from {@code <@token scheme="dark" value="…">}, in declaration order. */
52  	final Map<String,String> darkTokens = new LinkedHashMap<>();
53  
54  	/** Dark-scheme aliases from {@code <@token scheme="dark" alias="var(--jc-…)">}, in declaration order. */
55  	final Map<String,String> darkAliases = new LinkedHashMap<>();
56  
57  	ThemeBuildContext(String name, Theme.Builder themeBuilder) {
58  		this.name = name;
59  		this.themeBuilder = themeBuilder;
60  	}
61  }
