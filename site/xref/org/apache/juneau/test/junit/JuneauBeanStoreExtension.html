28  /**
29   * JUnit 5 extension that discovers {@link TestBean @TestBean}-annotated members on a test class and exposes them as
30   * a {@link TestBeanStore} overlay.
31   *
32   * <a id="ModeInject"></a>
33   * <h5 class='section'>Mode INJECT &mdash; fresh-instance with overrides (default):</h5>
34   * <p>
35   * The extension supports two wiring patterns.  <b>Mode INJECT</b> is the default and what every
36   * {@code @TestBean}-based test ships with out of the box.  The contract:
37   * <ul>
38   * 	<li>The SUT (a {@code RestContext} via {@code MockRestClient}, a {@code Microservice}, a {@code SerializerSet},
39   * 		etc.) is constructed <i>after</i> the overlay is built.  The overlay is threaded into the builder via
40   * 		{@code overridingBeanStore(...)} (see {@link BeanStoreOverridable}), which
41   * 		installs it in the {@code overridingParent} slot of the SUT's bean store at construction time.
42   * 	<li>Because the overlay sits at tier&nbsp;1 of the resolution chain, every framework-managed bean lookup
43   * 		consults it <i>before</i> any local {@code @Bean} factory or memoizer-backed default.  Mode INJECT is
44   * 		therefore <i>universal</i>: all bean types are eligible for replacement, including those that subsequently
45   * 		get pinned into a per-op memoizer.
46   * 	<li>The overlay's lifetime is bounded by the SUT's lifetime.  For a per-test fresh SUT (the most common
47   * 		shape), the overlay is discarded when the SUT goes out of scope at {@code @AfterEach}.  For a per-class
48   * 		shared SUT (built once in {@code @BeforeAll}), the overlay survives until the SUT is torn down at
49   * 		{@code @AfterAll}.  No explicit push/pop is required.
50   * </ul>
51   *
52   * <p>
53   * {@link TestBean @TestBean}, {@link TestBeanStore},
54   * {@link BeanStoreOverridable BeanStoreOverridable}, and the {@code overridingBeanStore(...)}
55   * setters on {@code MockRestClient.Builder}, {@code Microservice.Builder}, {@code SerializerSet.Builder},
56   * {@code ParserSet.Builder}, and {@code EncoderSet.Builder} all implement this Mode INJECT contract.
57   *
58   * <a id="ModeOverlay"></a>
59   * <h5 class='section'>Mode OVERLAY &mdash; existing-instance with push/pop:</h5>
60   * <p>
61   * <b>Mode OVERLAY</b> is the opt-in pattern for tests that hold a <i>long-lived</i> SUT (typically a
62   * {@code Microservice} booted in {@code @BeforeAll}) and want to apply per-test overlays without rebuilding it.
63   * The test calls {@link #attach(WritableBeanStore)} to point the extension at the
64   * SUT's bean store; the extension then uses
65   * {@link WritableBeanStore#pushOverlay(BeanStore)}
66   * at {@code beforeEach} / {@code beforeAll} and
67   * {@link WritableBeanStore#popOverlay(Snapshot)}
68   * at {@code afterEach} / {@code afterAll} so the SUT's bean store is restored between tests.
69   *
70   * <p>
71   * Mode OVERLAY requires {@code @TestBean(mode = Mode.OVERLAY)} on the participating annotations.  Mixing
72   * {@code Mode.INJECT} and {@code Mode.OVERLAY} on the same scope is rejected with a clear
73   * {@link IllegalStateException}.
74   *
75   * <h5 class='section'>Overlay storage:</h5>
76   * <p>
77   * The extension owns one or two {@link TestBeanStore} overlays per test execution:
78   * <ul>
79   * 	<li>A <b>class-scope</b> overlay built at {@code beforeAll}, containing every {@code @TestBean(scope = CLASS)}
80   * 		declared on {@code static} fields/methods of the test class hierarchy.  Lives until {@code afterAll}.
81   * 	<li>A <b>method-scope</b> overlay built at {@code beforeEach}, containing every {@code @TestBean(scope = METHOD)}
82   * 		declared on instance (or static, for convenience) fields/methods.  The method-scope overlay's parent is set
83   * 		to the class-scope overlay (if any), so a method-scope override of {@code (Type, name)} shadows a
84   * 		class-scope override of the same pair while still letting the class-scope's other entries fall through.
85   * 		Lives until {@code afterEach}.
86   * </ul>
87   *
88   * <p>
89   * Wiring into the system under test is <i>explicit</i> &mdash; the extension never reflects on the test class to
90   * find a {@code RestContext} / {@code Microservice} field and never mutates the SUT after construction.  Test
91   * authors retrieve the current overlay via {@link #getStore()} and plug it into the SUT's builder hook:
92   *
93   * <p class='bjava'>
94   * 	<ja>@ExtendWith</ja>(JuneauBeanStoreExtension.<jk>class</jk>)
95   * 	<jk>class</jk> MyResourceTest {
96   * 		<ja>@TestBean</ja> MyService <jv>mockSvc</jv> = Mockito.<jsm>mock</jsm>(MyService.<jk>class</jk>);
97   *
98   * 		<ja>@Test</ja>
99   * 		<jk>void</jk> aTest(TestBeanStore <jv>store</jv>) {
100  * 			<jk>var</jk> <jv>client</jv> = MockRestClient.<jsm>create</jsm>(MyResource.<jk>class</jk>)
101  * 				.overridingBeanStore(<jv>store</jv>)
102  * 				.build();
103  * 			<jv>client</jv>.get(<js>"/svc"</js>).run().assertStatus().is(200);
104  * 		}
105  * 	}
106  * </p>
107  *
108  * <p>
109  * Two registration styles are supported:
110  * <ul>
111  * 	<li><b>Declarative:</b> {@code @ExtendWith(JuneauBeanStoreExtension.class)} on the test class.
112  * 	<li><b>Programmatic:</b> a {@code @RegisterExtension} field initialized with {@link #create()}, so the test can
113  * 		invoke {@link #getStore()} from within {@code @BeforeAll} / {@code @BeforeEach} bodies as well as from test
114  * 		methods.
115  * </ul>
116  *
117  * <h5 class='section'>See Also:</h5><ul>
118  * 	<li class='ja'>{@link TestBean} - The annotation discovered by this extension.
119  * 	<li class='jc'>{@link TestBeanStore} - The overlay type the extension produces.
120  * 	<li class='je'>{@link Scope} - Per-method vs per-class lifecycle selector.
121  * 	<li class='je'>{@link Mode} - Mode INJECT (fresh instance) vs Mode OVERLAY (existing instance push/pop) selector.
122  * </ul>
123  *
124  * @since 10.0.0
125  */

129 })

173 	/**
174 	 * Points this extension at a long-lived SUT's {@link WritableBeanStore} for Mode OVERLAY push/pop wiring.
175 	 *
176 	 * <p>
177 	 * Call from a {@code @BeforeAll} or {@code @BeforeEach} body when the participating {@code @TestBean} declarations
178 	 * use {@link Mode#OVERLAY}.  Once attached, the extension will:
179 	 * <ul>
180 	 * 	<li>{@link WritableBeanStore#pushOverlay(BeanStore) push} the class-scope overlay onto the attached store at
181 	 * 		{@code beforeAll} (if a non-empty class-scope overlay exists), and
182 	 * 		{@link WritableBeanStore#popOverlay(Snapshot) pop} it at {@code afterAll}.
183 	 * 	<li>Push the method-scope overlay at {@code beforeEach} and pop it at {@code afterEach}.
184 	 * </ul>
185 	 *
186 	 * <p>
187 	 * Calling {@code attach(...)} a second time before {@link #detach()} is permitted &mdash; the most recent
188 	 * attachment wins.  If never called, the extension stays in Mode INJECT.
189 	 *
190 	 * @param store The bean store to push/pop overlays against.  Must not be <jk>null</jk>.
191 	 * @return This extension, for fluent chaining.
192 	 * @throws NullPointerException If {@code store} is <jk>null</jk>.
193 	 * @since 10.0.0
194 	 */
195 	public JuneauBeanStoreExtension attach(WritableBeanStore store) {
196 		Objects.requireNonNull(store, "store must not be null");
197 		attached = store;
198 		return this;
199 	}
200 
201 	/**
202 	 * Clears the {@link #attach(WritableBeanStore) attached} bean store reference.
203 	 *
204 	 * <p>
205 	 * Called automatically at {@code afterAll}; tests may also call it explicitly between scenarios.  Has no effect
206 	 * if the extension is not currently attached.
207 	 *
208 	 * @return This extension, for fluent chaining.
209 	 * @since 10.0.0
210 	 */
211 	public JuneauBeanStoreExtension detach() {
212 		attached = null;
213 		return this;
214 	}

217 	 * Returns the {@link #attach(WritableBeanStore) attached} bean store, or empty if none.
218 	 *
219 	 * <p>
220 	 * Surface for diagnostic / testability purposes &mdash; the push/pop lifecycle uses the field directly.
221 	 *
222 	 * @return The attached store, or {@link Optional#empty()} if not currently attached.
223 	 * @since 10.0.0

229 	/**
230 	 * Returns the current effective overlay &mdash; method-scope if a test method is executing, otherwise the
231 	 * class-scope overlay.
232 	 *
233 	 * <p>
234 	 * This is the accessor most tests plug into builder hooks such as
235 	 * {@code MockRestClient.Builder.overridingBeanStore(...)}.
236 	 *
237 	 * @return The current overlay.  Never <jk>null</jk>.
238 	 * @throws IllegalStateException If no class-scope or method-scope overlay is currently active (e.g. the accessor
239 	 * 	is called from a {@code @BeforeAll} body and the extension is not yet bound to a test class).
240 	 * @since 10.0.0
241 	 */
242 	public TestBeanStore getStore() {
243 		var s = currentMethodStore();
244 		if (s != null)
245 			return s;
246 		s = currentClassStore();
247 		if (s != null)
248 			return s;
249 		throw new IllegalStateException("JuneauBeanStoreExtension is not currently bound to a test class or test method.");
250 	}

252 	/**
253 	 * Returns the overlay for the specified scope, if one is currently active.
254 	 *
255 	 * <p>
256 	 * Allows a test to reach past the current effective overlay and inspect a specific lifecycle layer.  For
257 	 * example, a test running under both a class-scope and a method-scope overlay can use
258 	 * {@code getStore(Scope.CLASS)} to retrieve the shared per-class layer directly.
259 	 *
260 	 * @param scope The lifecycle scope to query.  Must not be <jk>null</jk>.
261 	 * @return The overlay for that scope, or {@link Optional#empty()} if no such overlay is active.
262 	 * @since 10.0.0
263 	 */
264 	public Optional<TestBeanStore> getStore(Scope scope) {
265 		Objects.requireNonNull(scope, "scope must not be null");
266 		return o(scope == Scope.METHOD ? currentMethodStore() : currentClassStore());
267 	}

273 	@Override /* BeforeAllCallback */
274 	public void beforeAll(ExtensionContext context) {
275 		currentClassContext = context;
276 		var testClass = context.getRequiredTestClass();
277 		var scoped = buildClassScopeStoreWithMode(testClass);
278 		context.getStore(NAMESPACE).put(KEY_CLASS_STORE, scoped.store());
279 		// Mode OVERLAY push (if attached) — only fires when at least one class-scope @TestBean declared
280 		// mode = Mode.OVERLAY.  All declarations in the scope must agree on the mode (mixing throws in
281 		// buildClassScopeStoreWithMode).
282 		var snapshot = pushOverlayIfOverlayMode(scoped);
283 		if (snapshot != null)
284 			context.getStore(NAMESPACE).put(KEY_CLASS_SNAPSHOT, snapshot);
285 	}
286 
287 	@Override /* AfterAllCallback */
288 	public void afterAll(ExtensionContext context) {
289 		popOverlayIfPresent(context, KEY_CLASS_SNAPSHOT);
290 		context.getStore(NAMESPACE).remove(KEY_CLASS_STORE);
291 		currentClassContext = null;
292 	}
293 
294 	@Override /* BeforeEachCallback */
295 	public void beforeEach(ExtensionContext context) {
296 		currentMethodContext = context;
297 		var classStore = readClassStore(context);
298 		var testInstance = context.getRequiredTestInstance();
299 		var scoped = buildMethodScopeStoreWithMode(testInstance, classStore);
300 		context.getStore(NAMESPACE).put(KEY_METHOD_STORE, scoped.store());
301 		var snapshot = pushOverlayIfOverlayMode(scoped);
302 		if (snapshot != null)
303 			context.getStore(NAMESPACE).put(KEY_METHOD_SNAPSHOT, snapshot);
304 	}
305 
306 	@Override /* AfterEachCallback */
307 	public void afterEach(ExtensionContext context) {
308 		popOverlayIfPresent(context, KEY_METHOD_SNAPSHOT);
309 		context.getStore(NAMESPACE).remove(KEY_METHOD_STORE);
310 		currentMethodContext = null;
311 	}

317 	private PushedOverlay pushOverlayIfOverlayMode(ScopedStore scoped) {
318 		if (scoped.mode() != Mode.OVERLAY || scoped.empty())
319 			return null;
320 		var store = attached;
321 		if (store == null)
322 			throw new IllegalStateException(
323 				"@TestBean(mode = Mode.OVERLAY) declared but JuneauBeanStoreExtension was not attached to a"
324 				+ " WritableBeanStore.  Call extension.attach(sut.getBeanStore()) from @BeforeAll / @BeforeEach"
325 				+ " before the test runs.");
326 		var snapshot = store.pushOverlay(scoped.store());
327 		return new PushedOverlay(store, snapshot);
328 	}
329 
330 	private static void popOverlayIfPresent(ExtensionContext context, String key) {
331 		var pushed = (PushedOverlay) context.getStore(NAMESPACE).remove(key);
332 		if (pushed == null)
333 			return;
334 		// Pop against the original store reference captured at push time, so the pop works even if the test
335 		// has since detach()-ed the extension or attach()-ed it to a different store.
336 		pushed.store().popOverlay(pushed.snapshot());
337 	}

340 	private record PushedOverlay(WritableBeanStore store, Snapshot snapshot) { /* no body */ }

342 	@Override /* ParameterResolver */
343 	public boolean supportsParameter(ParameterContext parameterContext, ExtensionContext extensionContext) {
344 		return parameterContext.getParameter().getType() == TestBeanStore.class;
345 	}

347 	@Override /* ParameterResolver */
348 	public Object resolveParameter(ParameterContext parameterContext, ExtensionContext extensionContext) {
349 		// Prefer the method-scope store from the supplied extensionContext (the parameter being resolved belongs to
350 		// the test method currently executing); fall back to class-scope when the parameter is resolved outside a
351 		// method context (rare — e.g. constructor injection on a TestInstance.Lifecycle.PER_METHOD test class).
352 		var methodStore = (TestBeanStore) extensionContext.getStore(NAMESPACE).get(KEY_METHOD_STORE);
353 		if (methodStore != null)
354 			return methodStore;
355 		var classStore = readClassStore(extensionContext);
356 		if (classStore != null)
357 			return classStore;
358 		throw new ParameterResolutionException(
359 			"JuneauBeanStoreExtension cannot resolve TestBeanStore parameter: no overlay is active.");
360 	}

366 	/**
367 	 * Builds a class-scope overlay for the supplied test class.
368 	 *
369 	 * <p>
370 	 * Walks the class hierarchy parent-to-child registering every {@code @TestBean(scope = CLASS)} declared on a
371 	 * {@code static} field or {@code static} method.  Non-static {@code scope = CLASS} declarations throw an
372 	 * {@link ExtensionContextException} &mdash; instance state isn't available at {@code beforeAll} time.
373 	 *
374 	 * <p>
375 	 * Package-private to allow direct unit testing of the discovery logic without driving the full JUnit lifecycle.
376 	 */
377 	static TestBeanStore buildClassScopeStore(Class<?> testClass) {
378 		return buildClassScopeStoreWithMode(testClass).store();
379 	}

381 	/**
382 	 * Builds a method-scope overlay for the supplied test instance, optionally chained on top of a class-scope
383 	 * overlay.
384 	 *
385 	 * <p>
386 	 * Package-private to allow direct unit testing of the discovery logic without driving the full JUnit lifecycle.
387 	 */
388 	static TestBeanStore buildMethodScopeStore(Object testInstance, TestBeanStore classScopeParent) {
389 		return buildMethodScopeStoreWithMode(testInstance, classScopeParent).store();

392 	/**
393 	 * {@link #buildClassScopeStore(Class)} plus the unified mode (INJECT or OVERLAY) of all class-scope
394 	 * declarations.
395 	 *
396 	 * <p>
397 	 * Mixing {@code Mode.INJECT} and {@code Mode.OVERLAY} declarations in the same scope throws an
398 	 * {@link IllegalStateException}.
399 	 */
400 	static ScopedStore buildClassScopeStoreWithMode(Class<?> testClass) {
401 		var store = new TestBeanStore();
402 		var modeTracker = new ModeTracker(Scope.CLASS);
403 		populateStaticOverrides(store, testClass, modeTracker);
404 		return new ScopedStore(store, modeTracker.resolve(), modeTracker.empty());
405 	}

407 	/**
408 	 * {@link #buildMethodScopeStore(Object, TestBeanStore)} plus the unified mode (INJECT or OVERLAY) of all
409 	 * method-scope declarations.
410 	 *
411 	 * <p>
412 	 * Mixing {@code Mode.INJECT} and {@code Mode.OVERLAY} declarations in the same scope throws an
413 	 * {@link IllegalStateException}.
414 	 */
415 	static ScopedStore buildMethodScopeStoreWithMode(Object testInstance, TestBeanStore classScopeParent) {
416 		var store = new TestBeanStore(classScopeParent);
417 		var modeTracker = new ModeTracker(Scope.METHOD);
418 		populateInstanceOverrides(store, testInstance, modeTracker);
419 		return new ScopedStore(store, modeTracker.resolve(), modeTracker.empty());
420 	}

422 	/**
423 	 * Discovers the {@code @TestBean} overrides declared on a test instance (and its class hierarchy) and returns
424 	 * them as a single {@link OverrideSet} &mdash; the public composition entry point for other JUnit 5 extensions
425 	 * (e.g. {@code @MicroserviceTest}) that need the {@code @TestBean} substrate without driving this extension's
426 	 * own lifecycle.
427 	 *
428 	 * <p>
429 	 * Both {@code @TestBean(scope = CLASS)} (static) and {@code @TestBean(scope = METHOD)} (instance) declarations
430 	 * are collected into one flattened overlay (method-scope chained over class-scope), so a composing extension
431 	 * that boots a single system-under-test per scope sees every declared override. The returned
432 	 * {@link OverrideSet#mode()} is the unified mode of all declarations (defaulting to {@link Mode#INJECT} when
433 	 * none are present); mixing {@code INJECT} and {@code OVERLAY} in one scope throws {@link IllegalStateException}.
434 	 *
435 	 * @param testInstance The test-class instance to scan. Must not be <jk>null</jk>.
436 	 * @return The discovered overrides + their unified mode. Never <jk>null</jk>; {@link OverrideSet#isEmpty()} is
437 	 * 	<jk>true</jk> when no {@code @TestBean} members were found.
438 	 * @since 10.0.0
439 	 */
440 	public static OverrideSet discoverOverrides(Object testInstance) {
441 		assertArgNotNull("testInstance", testInstance);
442 		var classScoped = buildClassScopeStoreWithMode(testInstance.getClass());
443 		var methodScoped = buildMethodScopeStoreWithMode(testInstance, classScoped.store());
444 		// Unify the two scopes' modes: an empty scope contributes no constraint; a populated one does.
445 		var mode = methodScoped.empty() ? classScoped.mode() : methodScoped.mode();
446 		if (! classScoped.empty() && ! methodScoped.empty() && classScoped.mode() != methodScoped.mode())
447 			throw new IllegalStateException(
448 				"Mixed @TestBean modes across scopes: CLASS declares Mode." + classScoped.mode()
449 				+ " but METHOD declares Mode." + methodScoped.mode()
450 				+ ".  All @TestBean declarations must use the same mode for a single-boot SUT.");
451 		var empty = classScoped.empty() && methodScoped.empty();
452 		return new OverrideSet(methodScoped.store(), mode, empty);
453 	}

478 	/**
479 	 * Walks the test-class hierarchy and registers every {@code @TestBean(scope = CLASS)} declared on a
480 	 * {@code static} field or {@code static} method.
481 	 *
482 	 * <p>
483 	 * Non-static {@code scope = CLASS} declarations are rejected with an {@link ExtensionContextException} &mdash;
484 	 * instance state isn't available at {@code beforeAll} time.
485 	 */
486 	@SuppressWarnings({
487 		"java:S3776" // Cognitive complexity acceptable for hierarchical test bean override wiring
488 	})
489 	private static void populateStaticOverrides(TestBeanStore store, Class<?> testClass, ModeTracker modeTracker) {
490 		for (var c : classHierarchy(testClass)) {
491 			for (var f : c.getDeclaredFields()) {
492 				var ann = f.getAnnotation(TestBean.class);
493 				if (ann == null || ann.scope() != Scope.CLASS)
494 					continue;
495 				if (!Modifier.isStatic(f.getModifiers()))
496 					throw new ExtensionContextException(
497 						"@TestBean(scope = CLASS) field must be static: " + describe(f));
498 				modeTracker.observe(ann.mode(), describe(f));
499 				register(store, f, null, ann);
500 			}
501 			for (var m : c.getDeclaredMethods()) {
502 				var ann = m.getAnnotation(TestBean.class);
503 				if (ann == null || ann.scope() != Scope.CLASS)
504 					continue;
505 				if (!Modifier.isStatic(m.getModifiers()))
506 					throw new ExtensionContextException(
507 						"@TestBean(scope = CLASS) method must be static: " + describe(m));
508 				modeTracker.observe(ann.mode(), describe(m));
509 				register(store, m, null, ann);
510 			}
511 		}
512 	}

514 	/**
515 	 * Walks the test-instance class hierarchy and registers every {@code @TestBean(scope = METHOD)} declared on a
516 	 * field or method.
517 	 *
518 	 * <p>
519 	 * Honors both instance and static members ("behave consistently with
520 	 * non-static" for {@code scope = METHOD} on static fields).
521 	 */
522 	@SuppressWarnings({
523 		"java:S3776" // Cognitive complexity acceptable for hierarchical test bean instance override wiring
524 	})
525 	private static void populateInstanceOverrides(TestBeanStore store, Object testInstance, ModeTracker modeTracker) {
526 		var testClass = testInstance.getClass();
527 		for (var c : classHierarchy(testClass)) {
528 			for (var f : c.getDeclaredFields()) {
529 				var ann = f.getAnnotation(TestBean.class);
530 				if (ann == null || ann.scope() != Scope.METHOD)
531 					continue;
532 				modeTracker.observe(ann.mode(), describe(f));
533 				register(store, f, Modifier.isStatic(f.getModifiers()) ? null : testInstance, ann);
534 			}
535 			for (var m : c.getDeclaredMethods()) {
536 				var ann = m.getAnnotation(TestBean.class);
537 				if (ann == null || ann.scope() != Scope.METHOD)
538 					continue;
539 				modeTracker.observe(ann.mode(), describe(m));
540 				register(store, m, Modifier.isStatic(m.getModifiers()) ? null : testInstance, ann);
541 			}
542 		}
543 	}

548 	/**
549 	 * Tracks the {@link Mode} of {@code @TestBean} declarations encountered within a single scope and rejects
550 	 * mixed-mode declarations.
551 	 */
552 	private static final class ModeTracker {
553 		private final Scope scope;
554 		private Mode observed;
555 		private String firstMember;
556 		private boolean empty = true;
557 
558 		ModeTracker(Scope scope) {
559 			this.scope = scope;
560 		}
561 
562 		void observe(Mode mode, String memberDesc) {
563 			empty = false;
564 			if (observed == null) {
565 				observed = mode;
566 				firstMember = memberDesc;
567 				return;
568 			}
569 			if (observed != mode)
570 				throw new IllegalStateException(
571 					"Mixed @TestBean modes in " + scope + " scope: " + firstMember + " declares Mode." + observed
572 					+ " but " + memberDesc + " declares Mode." + mode
573 					+ ".  All @TestBean declarations in a single scope must use the same mode.");
574 		}
575 
576 		Mode resolve() {
577 			return observed == null ? Mode.INJECT : observed;
578 		}
579 
580 		boolean empty() {
581 			return empty;
582 		}
583 	}

585 	private static void register(TestBeanStore store, Field field, Object instance, TestBean ann) {
586 		Object value;
587 		try {
588 			field.setAccessible(true);
589 			value = field.get(instance);
590 		} catch (IllegalAccessException e) {
591 			throw new ExtensionContextException("Could not read @TestBean field: " + describe(field), e);
592 		}
593 		var type = ann.type() == Object.class ? field.getType() : ann.type();
594 		registerValue(store, type, value, ann.name(), describe(field));
595 	}
596 
597 	private static void register(TestBeanStore store, Method method, Object instance, TestBean ann) {
598 		if (method.getParameterCount() != 0)
599 			throw new ExtensionContextException(
600 				"@TestBean factory method must be parameterless: " + describe(method));
601 		Object value;
602 		try {
603 			method.setAccessible(true);
604 			value = method.invoke(instance);
605 		} catch (ReflectiveOperationException e) {
606 			throw new ExtensionContextException(
607 				"Failed to invoke @TestBean factory method: " + describe(method),
608 				or(e.getCause(), e));
609 		}
610 		var type = ann.type() == Object.class ? method.getReturnType() : ann.type();
611 		registerValue(store, type, value, ann.name(), describe(method));
612 	}
613 
614 	@SuppressWarnings({
615 		"unchecked" // The type token came from either the annotated member's declared type or ann.type(); the value's runtime class is compatible by construction.
616 	})
617 	private static void registerValue(TestBeanStore store, Class<?> type, Object value, String name, String memberDesc) {
618 		if (value != null && !type.isInstance(value))
619 			throw new ExtensionContextException(
620 				"@TestBean value is not assignable to declared type " + type.getName() + ": " + memberDesc);
621 		store.override((Class<Object>) type, value, name.isEmpty() ? null : name);
622 	}

636 	/**
637 	 * Reads the class-scope store from the nearest enclosing context that owns one.
638 	 *
639 	 * <p>
640 	 * JUnit's {@code ExtensionContext.Store} keys are scoped per context.  When a method-scope callback fires, the
641 	 * incoming context is the method-level context whose own store does <i>not</i> contain the class-scope entry;
642 	 * the class-scope entry lives on the parent (class-level) context.  This helper walks parents until it finds
643 	 * one whose store contains the class-scope entry, or returns {@code null} if none does.
644 	 */
645 	private static TestBeanStore readClassStore(ExtensionContext context) {
646 		for (var c = context; c != null; c = c.getParent().orElse(null)) {
647 			var s = (TestBeanStore) c.getStore(NAMESPACE).get(KEY_CLASS_STORE);
648 			if (s != null)
649 				return s;
650 		}
651 		return null;
652 	}

654 	private static List<Class<?>> classHierarchy(Class<?> testClass) {
655 		// Walk parent-to-child so a subclass's @TestBean of the same (type, name) wins over the parent's via the
656 		// "last write wins" semantics of BasicBeanStore.addBean(...).
657 		var hierarchy = new ArrayList<Class<?>>();
658 		for (var c = testClass; c != null && c != Object.class; c = c.getSuperclass())
659 			hierarchy.add(c);
660 		Collections.reverse(hierarchy);
661 		return hierarchy;
662 	}
