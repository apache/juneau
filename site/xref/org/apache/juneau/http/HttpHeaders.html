26  /**
27   * Static factory methods for creating common HTTP headers.
28   *
29   * <p>
30   * This is a convenience class that provides shorthand access to all named header types in
31   * {@code org.apache.juneau.http.header}.
32   *
33   * <p class='bjava'>
34   * 	<jc>// Import statically for clean DSL-style usage</jc>
35   * 	import static HttpHeaders.*;
36   *
37   * 	RestRequest <jv>req</jv> = client.get(<js>"/api/resource"</js>)
38   * 		.header(accept(<js>"application/json"</js>))
39   * 		.header(authorization(<js>"Bearer "</js> + token));
40   * </p>
41   *
42   * <p>
43   * <b>Beta — API subject to change:</b> This type is part of the next-generation REST client and HTTP stack
44   * ({@code org.apache.juneau.marshall.ng.*}).
45   * It is not API-frozen: binary- and source-incompatible changes may appear in the <b>next major</b> Juneau release
46   * (and possibly earlier).
47   *
48   * <h5 class='section'>See Also:</h5><ul>
49   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/NextGenRestClient">juneau-ng REST client</a>
50   * </ul>
51   *
52   * @since 9.2.1
53   */

315 	 * Creates a generic {@link HttpHeader} with the given name and value.
316 	 *
317 	 * @param name The header name. Must not be <jk>null</jk>.

319 	 * @return A new header. Never <jk>null</jk>.
