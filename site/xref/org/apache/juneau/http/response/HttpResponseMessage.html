17  package org.apache.juneau.http.response;
18  
19  import static org.apache.juneau.commons.utils.Shorts.*;
20  
21  import java.util.*;
22  
23  import org.apache.juneau.http.*;
24  
25  /**
26   * A complete HTTP response message: status line, headers, and optional body.
27   *
28   * <p>
29   * Mirrors the semantics of {@code org.apache.http.HttpResponse} without the Apache HttpCore dependency.
30   *
31   * <p>
32   * <b>Beta — API subject to change:</b> This type is part of the next-generation REST client and HTTP stack
33   * ({@code org.apache.juneau.marshall.ng.*}).
34   * It is not API-frozen: binary- and source-incompatible changes may appear in the <b>next major</b> Juneau release
35   * (and possibly earlier).
36   * For production use cases that require long-term binary stability, continue using the existing
37   * {@code juneau-rest-client} and {@code juneau-rest-common} APIs until the {@code ng} stack is declared stable.
38   *
39   * <h5 class='section'>See Also:</h5><ul>
40   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/NextGenRestClient">juneau-ng REST client</a>
41   * </ul>
42   *
43   * @since 9.2.1
44   */
45  public interface HttpResponseMessage {
46  
47  	/**
48  	 * Returns the status line of this response.
49  	 *
50  	 * @return The status line. Never <jk>null</jk>.
51  	 */
52  	HttpStatusLine getStatusLine();
53  
54  	/**
55  	 * Returns all response headers, in the order they were received.
56  	 *
57  	 * @return An unmodifiable list of response headers. Never <jk>null</jk>.
58  	 */
59  	List<HttpHeader> getHeaders();
60  
61  	/**
62  	 * Returns the first response header with the given name (case-insensitive), or <jk>null</jk> if absent.
63  	 *
64  	 * @param name The header name. Must not be <jk>null</jk>.
65  	 * @return The first matching header, or <jk>null</jk> if absent.
66  	 */
67  	default HttpHeader getFirstHeader(String name) {
68  		return getHeaders().stream()
69  			.filter(h -> eqic(h.getName(), name))
70  			.findFirst()
71  			.orElse(null);
72  	}
73  
74  	/**
75  	 * Returns all response headers with the given name (case-insensitive).
76  	 *
77  	 * @param name The header name. Must not be <jk>null</jk>.
78  	 * @return A list of matching headers (possibly empty). Never <jk>null</jk>.
79  	 */
80  	default List<HttpHeader> getHeaders(String name) {
81  		return getHeaders().stream()
82  			.filter(h -> eqic(h.getName(), name))
83  			.toList();
84  	}
85  
86  	/**
87  	 * Returns the response body, or <jk>null</jk> if the response has no body.
88  	 *
89  	 * @return The response body, possibly <jk>null</jk>.
90  	 */
91  	HttpBody getBody();
92  }
