19  import static org.apache.juneau.commons.utils.AssertionUtils.*;
20  import static org.apache.juneau.commons.utils.Shorts.*;

22  import org.apache.juneau.http.*;

24  /**
25   * Default immutable implementation of {@link HttpStatusLine}.
26   *
27   * <p>
28   * Create instances via the static factory methods:
29   * <p class='bjava'>
30   * 	<jc>// 200 OK  HTTP/1.1</jc>
31   * 	HttpStatusLine <jv>sl</jv> = HttpStatusLineBean.<jsm>of</jsm>(200, <js>"OK"</js>);
32   *
33   * 	<jc>// Custom protocol version (parsed)</jc>
34   * 	HttpStatusLine <jv>sl2</jv> = HttpStatusLineBean.<jsm>of</jsm>(<js>"HTTP/2"</js>, 200, <js>"OK"</js>);
35   *
36   * 	<jc>// Custom protocol version (typed)</jc>
37   * 	HttpStatusLine <jv>sl3</jv> = HttpStatusLineBean.<jsm>of</jsm>(HttpProtocolVersion.<jsf>HTTP_2_0</jsf>, 200, <js>"OK"</js>);
38   * </p>
39   *
40   * <p>
41   * <b>Beta — API subject to change:</b> This type is part of the next-generation REST client and HTTP stack
42   * ({@code org.apache.juneau.marshall.ng.*}).
43   * It is not API-frozen: binary- and source-incompatible changes may appear in the <b>next major</b> Juneau release
44   * (and possibly earlier).
45   * For production use cases that require long-term binary stability, continue using the existing
46   * {@code juneau-rest-client} and {@code juneau-rest-common} APIs until the {@code ng} stack is declared stable.
47   *
48   * <h5 class='section'>See Also:</h5><ul>
49   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/NextGenRestClient">juneau-ng REST client</a>
50   * </ul>
51   *
52   * @since 9.2.1
53   */
54  public final class HttpStatusLineBean implements HttpStatusLine {
55  
56  	private final HttpProtocolVersion protocolVersion;
57  	private final int statusCode;
58  	private final String reasonPhrase;
59  
60  	private HttpStatusLineBean(HttpProtocolVersion protocolVersion, int statusCode, String reasonPhrase) {
61  		this.protocolVersion = assertArgNotNull("protocolVersion", protocolVersion);
62  		this.statusCode = statusCode;
63  		this.reasonPhrase = reasonPhrase;
64  	}
65  
66  	/**
67  	 * Creates an {@code HTTP/1.1} status line with the given status code and reason phrase.
68  	 *
69  	 * @param statusCode The HTTP status code (e.g. {@code 200}).
70  	 * @param reasonPhrase The reason phrase (e.g. {@code "OK"}). May be <jk>null</jk>.
71  	 * @return A new instance. Never <jk>null</jk>.
72  	 */
73  	public static HttpStatusLineBean of(int statusCode, String reasonPhrase) {
74  		return new HttpStatusLineBean(HttpProtocolVersion.HTTP_1_1, statusCode, reasonPhrase);
75  	}
76  
77  	/**
78  	 * Creates a status line with a parsed protocol version, status code, and reason phrase.
79  	 *
80  	 * @param protocolVersion The protocol version string (e.g. {@code "HTTP/2"}). Must not be <jk>null</jk>.
81  	 * @param statusCode The HTTP status code.
82  	 * @param reasonPhrase The reason phrase. May be <jk>null</jk>.
83  	 * @return A new instance. Never <jk>null</jk>.
84  	 */
85  	public static HttpStatusLineBean of(String protocolVersion, int statusCode, String reasonPhrase) {
86  		return new HttpStatusLineBean(HttpProtocolVersion.parse(protocolVersion), statusCode, reasonPhrase);
87  	}
88  
89  	/**
90  	 * Creates a status line with the given typed protocol version, status code, and reason phrase.
91  	 *
92  	 * @param protocolVersion The protocol version. Must not be <jk>null</jk>.
93  	 * @param statusCode The HTTP status code.
94  	 * @param reasonPhrase The reason phrase. May be <jk>null</jk>.
95  	 * @return A new instance. Never <jk>null</jk>.
96  	 */
97  	public static HttpStatusLineBean of(HttpProtocolVersion protocolVersion, int statusCode, String reasonPhrase) {
98  		return new HttpStatusLineBean(protocolVersion, statusCode, reasonPhrase);
99  	}
100 
101 	@Override /* HttpStatusLine */
102 	public int getStatusCode() {
103 		return statusCode;
104 	}
105 
106 	@Override /* HttpStatusLine */
107 	public String getReasonPhrase() {
108 		return reasonPhrase;
109 	}
110 
111 	@Override /* HttpStatusLine */
112 	public HttpProtocolVersion getProtocolVersion() {
113 		return protocolVersion;
114 	}
115 
116 	@Override /* Object */
117 	public String toString() {
118 		var sb = new StringBuilder(protocolVersion.toString()).append(' ').append(statusCode);
119 		if (reasonPhrase != null)
120 			sb.append(' ').append(reasonPhrase);
121 		return sb.toString();
122 	}
123 
124 	@Override /* Object */
125 	public boolean equals(Object obj) {
126 		return (obj instanceof HttpStatusLineBean o2) && eq(this, o2, (x, y) ->
127 			x.statusCode == y.statusCode && eq(x.protocolVersion, y.protocolVersion) && eq(x.reasonPhrase, y.reasonPhrase));
128 	}
129 
130 	@Override /* Object */
131 	public int hashCode() {
132 		return h(protocolVersion, statusCode, reasonPhrase);
133 	}
134 }
