29  /**
30   * Base class for concrete HTTP response objects in the {@code org.apache.juneau.http} stack.
31   *
32   * <p>
33   * Holds a status line, an optional body, and an ordered list of headers. Provides classic-style
34   * fluent setters that mutate state and return {@code this}.
35   *
36   * <p>
37   * Immutability is expressed with the "funnel + nested {@code Unmodifiable} snapshot" paradigm: every mutator routes
38   * through the single protected {@link #modify(Runnable)} choke-point, and each concrete leaf gets a nested
39   * {@code X.Unmodifiable} whose only behavioral override is a throwing {@code modify(...)}.  A leaf's
40   * {@link #unmodifiable()} factory returns a point-in-time snapshot of type {@code X.Unmodifiable}; any attempt to set a
41   * property on it throws an {@link UnsupportedOperationException}.
42   *
43   * <p>
44   * This is a self-typed (CRTP) root: a leaf declares <c>class Ok <jk>extends</jk> BasicHttpResponse&lt;Ok&gt;</c> and
45   * inherits leaf-typed fluent setters with no covariant overrides.
46   *
47   * <p>
48   * <b>Beta — API subject to change:</b> This type is part of the next-generation REST client and HTTP stack
49   * ({@code org.apache.juneau.marshall.ng.*}).
50   * It is not API-frozen: binary- and source-incompatible changes may appear in the <b>next major</b> Juneau release
51   * (and possibly earlier).
52   *
53   * <h5 class='section'>See Also:</h5><ul>
54   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/NextGenRestClient">juneau-ng REST client</a>
55   * </ul>
56   *
57   * @param <SELF> The self type for fluent setters.
58   * @since 9.2.1
59   */
60  @SuppressWarnings({
61  	"java:S119" // 'SELF' (CRTP self-type) is intentional and clearer than a single-letter name.
62  })

70  	/**
71  	 * Constructor with no body.
72  	 *
73  	 * @param statusLine The status line. Must not be <jk>null</jk>.
74  	 */
75  	protected BasicHttpResponse(HttpStatusLine statusLine) {
76  		// Reason phrase is documented-nullable; StringBody.of asserts non-null, so guard against a valid status line
77  		// that carries a null reason phrase (would otherwise NPE on a legitimate input).
78  		this(statusLine, l(),
79  			statusLine != null && statusLine.getReasonPhrase() != null ? StringBody.of(statusLine.getReasonPhrase()) : null);
80  	}

87  	 */
88  	protected BasicHttpResponse(HttpStatusLine statusLine, HttpBody body) {
89  		this(statusLine, l(), body);
90  	}

102 	/**
103 	 * Constructor with headers and an optional body.
104 	 *
105 	 * @param statusLine The status line. Must not be <jk>null</jk>.
106 	 * @param headers The response headers. Can be <jk>null</jk> (treated as an empty list).
107 	 * @param body The response body. May be <jk>null</jk>.
108 	 */
109 	protected BasicHttpResponse(HttpStatusLine statusLine, List<HttpHeader> headers, HttpBody body) {
110 		this.statusLine = assertArgNotNull("statusLine", statusLine);
111 		this.headers = HttpHeaderList.of(headers);
112 		this.body = body;
113 	}

115 	/**
116 	 * Copy constructor.
117 	 *
118 	 * @param copyFrom The instance to copy. Must not be <jk>null</jk>.
119 	 */
120 	protected BasicHttpResponse(BasicHttpResponse<?> copyFrom) {
121 		assertArgNotNull("copyFrom", copyFrom);
122 		this.statusLine = copyFrom.statusLine;
123 		this.headers = copyFrom.headers.copy();
124 		this.body = copyFrom.body;
125 		this.locale = copyFrom.locale;
126 	}

425 	/**
426 	 * Single mutation funnel — the only choke-point through which all state changes on this bean flow.
427 	 *
428 	 * <p>
429 	 * Every fluent setter routes through this method.  The nested {@code Unmodifiable} snapshot overrides only this
430 	 * method to throw, which freezes the entire mutation surface with a single override.
431 	 *
432 	 * @param mutation The state change to apply.
433 	 * @return This object.
434 	 */
435 	protected SELF modify(Runnable mutation) {
436 		mutation.run();
437 		return self();
438 	}
439 
440 	/**
441 	 * Deep-freeze hook invoked from the nested {@code Unmodifiable} constructor after the snapshot copy completes.
442 	 *
443 	 * <p>
444 	 * The mutable sub-beans are frozen here by <b>direct field assignment</b> (never via a setter, which would route
445 	 * through the throwing {@link #modify(Runnable)} and blow up during construction).  A leaf that adds its own mutable
446 	 * sub-bean overrides this method (calling {@code super.freeze()} first).
447 	 */
448 	protected void freeze() {
449 		headers = headers.unmodifiable();   // DIRECT field write — snapshot the copied HttpHeaderList.
450 	}
