29  /**
30   * Base class for HTTP error response exceptions (4xx and 5xx) in the {@code org.apache.juneau.http} stack.
31   *
32   * <p>
33   * Extends {@link RuntimeException} while also implementing {@link HttpResponseMessage}, so that HTTP
34   * error responses can be both thrown as exceptions and used as response objects. Provides classic-style
35   * fluent setters that mutate state and return {@code this}.
36   *
37   * <p>
38   * Immutability is expressed with the "funnel + nested {@code Unmodifiable} snapshot" paradigm: every mutator routes
39   * through the single protected {@link #modify(Runnable)} choke-point, and each concrete leaf gets a nested
40   * {@code X.Unmodifiable} whose only behavioral override is a throwing {@code modify(...)}.  A leaf's
41   * {@link #unmodifiable()} factory returns a point-in-time snapshot; any attempt to set a property on it throws an
42   * {@link UnsupportedOperationException}.
43   *
44   * <p>
45   * This is a {@link Throwable}-rooted hierarchy, so (unlike {@link BasicHttpResponse}) it cannot be a self-typed (CRTP)
46   * root — JLS &sect;8.1.2 forbids a generic {@code Throwable}.  The funnel therefore returns the base type and each leaf
47   * keeps its covariant setter overrides.
48   *
49   * <p>
50   * <b>Beta — API subject to change:</b> This type is part of the next-generation REST client and HTTP stack
51   * ({@code org.apache.juneau.marshall.ng.*}).
52   * It is not API-frozen: binary- and source-incompatible changes may appear in the <b>next major</b> Juneau release
53   * (and possibly earlier).
54   *
55   * <h5 class='section'>See Also:</h5><ul>
56   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/NextGenRestClient">juneau-ng REST client</a>
57   * </ul>
58   *
59   * @since 9.2.1
60   */

109 	/**
110 	 * Constructor with a cause and a {@link String#format(String, Object...) String.format}-style message body.
111 	 *
112 	 * @param statusCode The HTTP status code.
113 	 * @param reasonPhrase The reason phrase. May be <jk>null</jk>.
114 	 * @param cause The cause. May be <jk>null</jk>.
115 	 * @param msg The exception message. May be <jk>null</jk>.
116 	 *    Treated as a {@link String#format(String, Object...) String.format}-style format string (<c>%s</c> placeholders) when {@code args} is non-empty.
117 	 * @param args Optional message arguments.
118 	 */
119 	public BasicHttpException(int statusCode, String reasonPhrase, Throwable cause, String msg, Object...args) {
120 		super(formatMessage(msg, args), cause);
121 		this.statusLine = HttpStatusLineBean.of(statusCode, reasonPhrase);
122 		this.headers = HttpHeaderList.create();
123 		var m = formatMessage(msg, args);
124 		this.body = m != null ? StringBody.of(m) : null;
125 	}

127 	private static String formatMessage(String msg, Object...args) {
128 		if (msg == null)
129 			return null;
130 		if (args == null || args.length == 0)
131 			return msg;
132 		return f(msg, args);
133 	}

135 	/**
136 	 * Copy constructor.
137 	 *
138 	 * <p>
139 	 * The mutable sub-beans are deep-copied by <b>direct field assignment</b>, never through a funneled setter.  This is
140 	 * the snapshot path used by {@link #unmodifiable()}: the constructor runs inside an {@code Unmodifiable} instance
141 	 * whose {@link #modify(Runnable)} already throws, so a setter-based copy would blow up during construction.
142 	 *
143 	 * @param copyFrom The instance to copy. Must not be <jk>null</jk>.
144 	 */
145 	protected BasicHttpException(BasicHttpException copyFrom) {
146 		super(assertArgNotNull("copyFrom", copyFrom).getMessage(), copyFrom.getCause());
147 		this.statusLine = copyFrom.statusLine;
148 		this.headers = copyFrom.headers.copy();
149 		this.body = copyFrom.body;
150 		this.locale = copyFrom.locale;
151 	}

385 	/**
386 	 * Returns the root cause of this exception.
387 	 *
388 	 * <p>
389 	 * The root cause is the first exception in the init-cause parent chain that's not one of the following:
390 	 * <ul>
391 	 * 	<li>{@link BasicHttpException}
392 	 * 	<li>{@link InvocationTargetException}
393 	 * </ul>
394 	 *
395 	 * @return The root cause of this exception, or <jk>null</jk> if no root cause was found.
396 	 */
397 	public Throwable getRootCause() {
398 		Throwable t = this;
399 		while (nn(t)) {
400 			if (! (t instanceof BasicHttpException || t instanceof InvocationTargetException))
401 				return t;
402 			t = t.getCause();
403 		}
404 		return null;
405 	}

407 	/**
408 	 * Returns all error messages from all errors in this exception's cause chain joined into a single string.
409 	 *
410 	 * @param scrubForXssVulnerabilities
411 	 * 	If {@code true}, replaces {@code <}, {@code >}, and {@code &} characters with spaces.
412 	 * @return A multi-line string of messages. Never <jk>null</jk>.
413 	 */
414 	public String getFullStackMessage(boolean scrubForXssVulnerabilities) {
415 		var msg = getMessage();
416 		var sb = new StringBuilder();
417 		if (nn(msg)) {
418 			if (scrubForXssVulnerabilities)
419 				msg = msg.replace('<', ' ').replace('>', ' ').replace('&', ' ');
420 			sb.append(msg);
421 		}
422 		var e = getCause();
423 		while (nn(e)) {
424 			msg = e.getMessage();
425 			if (nn(msg) && scrubForXssVulnerabilities)
426 				msg = msg.replace('<', ' ').replace('>', ' ').replace('&', ' ');
427 			var cls = cns(e);
428 			if (msg == null)
429 				sb.append(f("\nCaused by (%s)", cls));
430 			else
431 				sb.append(f("\nCaused by (%s): %s", cls, msg));
432 			e = e.getCause();
433 		}
434 		return sb.toString();
435 	}

441 	/**
442 	 * Single mutation funnel — the only choke-point through which all state changes on this bean flow.
443 	 *
444 	 * <p>
445 	 * Every fluent setter routes through this method.  The nested {@code Unmodifiable} snapshot overrides only this
446 	 * method to throw, which freezes the entire mutation surface with a single override.  Because this is a
447 	 * {@link Throwable}-rooted hierarchy (no CRTP self-type), the funnel returns the base type.
448 	 *
449 	 * @param mutation The state change to apply.
450 	 * @return This object.
451 	 */
452 	protected BasicHttpException modify(Runnable mutation) {
453 		mutation.run();
454 		return this;
455 	}

457 	/**
458 	 * Deep-freeze hook invoked from the nested {@code Unmodifiable} constructor after the snapshot copy completes.
459 	 *
460 	 * <p>
461 	 * The mutable sub-beans are frozen here by <b>direct field assignment</b> (never via a setter, which would route
462 	 * through the throwing {@link #modify(Runnable)} and blow up during construction).  A leaf that adds its own mutable
463 	 * sub-bean overrides this method (calling {@code super.freeze()} first).
464 	 */
465 	protected void freeze() {
466 		headers = headers.unmodifiable();   // DIRECT field write — snapshot the copied HttpHeaderList.
467 	}

471 		// D3 content equality.  The 'body' HttpBody is deliberately excluded: it is a wrapper without value semantics

473 		return o instanceof BasicHttpException other && eq(this, other, (x, y) ->

488 	 * Unmodifiable point-in-time snapshot of the enclosing {@link BasicHttpException}.
489 	 *
490 	 * <p>
491 	 * Its only behavioral override is {@link #modify(Runnable)}, which throws — because all mutation is funneled through

493 	 */

495 
496 		private static final long serialVersionUID = 1L;
497 
498 		/**
499 		 * Constructor.
500 		 *
501 		 * @param copyFrom The exception to snapshot.  Must not be <jk>null</jk>.
502 		 */
503 		@SuppressWarnings({
504 			"java:S1699" // Paradigm intentionally calls the overridable freeze() from the ctor to deep-freeze sub-beans.
505 		})
506 		protected Unmodifiable(BasicHttpException copyFrom) {
507 			super(copyFrom);
508 			freeze();
509 		}
510 
511 		@Override /* Overridden from BasicHttpException */
512 		protected BasicHttpException modify(Runnable mutation) {
513 			throw uoex("Bean is unmodifiable.");
514 		}
