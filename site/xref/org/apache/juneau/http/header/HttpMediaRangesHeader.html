20  
21  import java.util.*;
22  import java.util.function.*;
23  
24  import org.apache.juneau.commons.http.*;

27  /**
28   * Base type for headers whose wire format is an <c>Accept</c>-style list of media ranges.
29   *
30   * <p>
31   * <b>Beta — API subject to change:</b> This type is part of the next-generation REST client and HTTP stack
32   * ({@code org.apache.juneau.marshall.ng.*}).
33   * It is not API-frozen: binary- and source-incompatible changes may appear in the <b>next major</b> Juneau release
34   * (and possibly earlier).
35   *
36   * <h5 class='section'>See Also:</h5><ul>
37   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/NextGenRestClient">juneau-ng REST client</a>
38   * </ul>
39   *
40   * @since 9.2.1
41   */
42  @SuppressWarnings({
43  	"java:S2160", // equals() on HttpHeaderBean uses name + getValue(); typed state is reflected in getValue()
44  	"unchecked" // Supplier<?> branches cast to typed suppliers after lazy-mode check
45  })
46  public class HttpMediaRangesHeader extends HttpHeaderBean {
47  
48  	/** Lazy supplier provides the wire string. */
49  	public static final int LAZY_WIRE_STRING = 0;
50  
51  	/** Lazy supplier provides {@link MediaRanges} (re-evaluated on each access). */
52  	public static final int LAZY_MEDIA_RANGES = 1;
53  
54  	private final MediaRanges cachedForStringOrDirect;
55  	private final Supplier<?> lazySupplier;
56  	private final int lazyMode;
57  
58  	/**
59  	 * @param name Header name. Must not be {@code null}.
60  	 * @param value Wire value. May be {@code null}.
61  	 */
62  	protected HttpMediaRangesHeader(String name, String value) {
63  		super(name, value);
64  		this.cachedForStringOrDirect = value == null ? null : MediaRanges.of(value);
65  		this.lazySupplier = null;
66  		this.lazyMode = -1;
67  	}
68  
69  	/**
70  	 * @param name Header name. Must not be {@code null}.
71  	 * @param value Parsed media ranges. May be {@code null}.
72  	 */
73  	protected HttpMediaRangesHeader(String name, MediaRanges value) {
74  		super(name, value == null ? null : value.toString());
75  		this.cachedForStringOrDirect = value;
76  		this.lazySupplier = null;
77  		this.lazyMode = -1;
78  	}
79  
80  	/**
81  	 * @param name Header name. Must not be {@code null}.
82  	 * @param supplier Lazy supplier; mode {@link #LAZY_WIRE_STRING} or {@link #LAZY_MEDIA_RANGES}. Must not be {@code null}.
83  	 * @param lazyMode Discriminator for erasure-safe lazy construction.
84  	 */
85  	protected HttpMediaRangesHeader(String name, Supplier<?> supplier, int lazyMode) {
86  		super(name, lazyMode == LAZY_WIRE_STRING
87  			? ((Supplier<String>) supplier)::get
88  			: () -> {
89  				var m = ((Supplier<MediaRanges>) supplier).get();
90  				if (m == null)
91  					return null;
92  				return m.toString();
93  			});
94  		this.cachedForStringOrDirect = null;
95  		this.lazySupplier = supplier;
96  		this.lazyMode = lazyMode;
97  	}
98  
99  	/**
100 	 * @return Parsed {@link MediaRanges}, or an empty {@link Optional} if the wire value is absent.
101 	 */
102 	public Optional<MediaRanges> asMediaRanges() {
103 		return o(toMediaRanges());
104 	}
105 
106 	/**
107 	 * @param index Range index.
108 	 * @return The {@link MediaRange} at {@code index}, or {@code null} if out of range.
109 	 */
110 	public MediaRange getRange(int index) {
111 		var x = toMediaRanges();
112 		return x == null ? null : x.getRange(index);
113 	}
114 
115 	/**
116 	 * @param part Subtype fragment to find (e.g. {@code "activity"} for {@code text/json+activity}).
117 	 * @return {@code true} if any range's subtype contains the fragment.
118 	 */
119 	public boolean hasSubtypePart(String part) {
120 		var x = toMediaRanges();
121 		return x != null && x.hasSubtypePart(part);
122 	}
123 
124 	/**
125 	 * @param mediaTypes Candidate types, in preference order.
126 	 * @return Index of the best matching type, or {@code -1}.
127 	 */
128 	public int match(List<? extends MediaType> mediaTypes) {
129 		var x = toMediaRanges();
130 		return x == null ? -1 : x.match(mediaTypes);
131 	}
132 
133 	/**
134 	 * @param other Fallback when no parsed value is present.
135 	 * @return Parsed ranges or {@code other}.
136 	 */
137 	public MediaRanges orElse(MediaRanges other) {
138 		var x = toMediaRanges();
139 		return nn(x) ? x : other;
140 	}
141 
142 	/**
143 	 * @return Parsed {@link MediaRanges}, or {@code null} if the wire value is absent.
144 	 */
145 	public MediaRanges toMediaRanges() {
146 		if (lazyMode == LAZY_MEDIA_RANGES)
147 			return ((Supplier<MediaRanges>) lazySupplier).get();
148 		if (cachedForStringOrDirect != null)
149 			return cachedForStringOrDirect;
150 		var v = super.getValue();
151 		return v == null ? null : MediaRanges.of(v);
152 	}
153 }
