17  package org.apache.juneau.http;
18  
19  import static java.lang.annotation.ElementType.*;
20  import static java.lang.annotation.RetentionPolicy.*;
21  
22  import java.lang.annotation.*;
23  
24  /**
25   * REST request dynamic-URL annotation.
26   *
27   * <p>
28   * Identifies a method parameter on a REST proxy interface whose value supplies the entire request URL (or a
29   * root-relative URL) for that single call &mdash; the Retrofit <c>@Url</c> pattern.
30   *
31   * <h5 class='topic'>Arguments of client-side @Remote-annotated interfaces</h5>
32   * <p>
33   * When the bound argument is non-<jk>null</jk> and non-blank, its string value becomes the effective URL for that
34   * invocation, overriding the interface/method-declared path resolution:
35   * <ul class='spaced-list'>
36   * 	<li>
37   * 		If the value contains a scheme (e.g. <js>"http://other-host/path"</js>) it is treated as an <b>absolute</b> URL
38   * 		that replaces the whole computed endpoint and bypasses the client root URL.
39   * 	<li>
40   * 		If the value has no scheme (e.g. <js>"/other/path"</js>) it is treated as a <b>relative</b> URL resolved against
41   * 		the client root URL only &mdash; <i>not</i> against the interface {@link org.apache.juneau.http.remote.Remote#path()}
42   * 		(matching Retrofit "endpoint replacement").
43   * </ul>
44   *
45   * <p>
46   * Any <c>{var}</c> tokens in the supplied value are still filled by the call's {@link Path @Path} parameters, so a
47   * caller-supplied URL may contain path templates.  An <ja>@Url</ja> argument takes precedence over
48   * {@link org.apache.juneau.http.remote.Remote#path()} + {@code @RemoteOp(path=...)} and over any declarative
49   * {@code baseUrl} override.
50   *
51   * <h5 class='figure'>Example:</h5>
52   * <p class='bjava'>
53   * 	<ja>@Remote</ja>(path=<js>"/api"</js>)
54   * 	<jk>public interface</jk> MyProxy {
55   * 		<jc>// Caller chooses the endpoint at runtime.</jc>
56   * 		<ja>@RemoteGet</ja>
57   * 		String fetch(<ja>@Url</ja> String <jv>url</jv>);
58   * 	}
59   * </p>
60   *
61   * <h5 class='section'>Notes:</h5><ul>
62   * 	<li class='note'>
63   * 		At most one <ja>@Url</ja> parameter is allowed per method (validated at proxy-build time).
64   * 	<li class='note'>
65   * 		A <jk>null</jk> or blank <ja>@Url</ja> argument is rejected with an {@link IllegalArgumentException}.
66   * 	<li class='note'>
67   * 		Only <c>http</c>/<c>https</c> schemes are permitted; other schemes (e.g. <c>file:</c>, <c>gopher:</c>) are
68   * 		rejected (SSRF guardrail).
69   * 	<li class='note'>
70   * 		Honored by the next-generation engine (<c>RestClient.remote(...)</c>) only; the classic engine currently
71   * 		ignores this annotation.
72   * </ul>
73   *
74   * <h5 class='section'>See Also:</h5><ul>
75   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestProxies">REST Proxy Basics</a>
76   * </ul>
77   *
78   * @since 9.2.1
79   */
80  @Documented
81  @Target({ PARAMETER })
82  @Retention(RUNTIME)
83  @Inherited
84  public @interface Url {
85  }
