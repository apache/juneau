17  package org.apache.juneau.http;
18  
19  import org.apache.juneau.http.response.*;
20  
21  /**
22   * Static factory methods for creating pre-defined HTTP response and exception objects.
23   *
24   * <p>
25   * Import statically for clean DSL-style usage in REST server code:
26   * <p class='bjava'>
27   * 	import static HttpResponses.*;
28   *
29   * 	<ja>@RestGet</ja>(<js>"/users/{id}"</js>)
30   * 	<jk>public</jk> User getUser(<ja>@Path</ja> String id) {
31   * 		User <jv>user</jv> = userService.find(id);
32   * 		<jk>if</jk> (<jv>user</jv> == <jk>null</jk>) <jk>throw</jk> notFound(<js>"User not found: "</js> + id);
33   * 		<jk>return</jk> <jv>user</jv>;
34   * 	}
35   * </p>
36   *
37   * <p>
38   * <b>Beta — API subject to change:</b> This type is part of the next-generation REST client and HTTP stack
39   * ({@code org.apache.juneau.marshall.ng.*}).
40   * It is not API-frozen: binary- and source-incompatible changes may appear in the <b>next major</b> Juneau release
41   * (and possibly earlier).
42   *
43   * <h5 class='section'>See Also:</h5><ul>
44   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/NextGenRestClient">juneau-ng REST client</a>
45   * </ul>
46   *
47   * @since 9.2.1
48   */
49  @SuppressWarnings({
50  	"java:S1192" // Duplicate string literals are HTTP response reason phrases and status labels; intentional
51  })
52  public final class HttpResponses {
53  
54  	private HttpResponses() {}
55  
56  	// ------------------------------------------------------------------------------------------------------------------
57  	// 1xx Informational
58  	// ------------------------------------------------------------------------------------------------------------------
59  
60  	/** @return A new {@code 100 Continue} response. */
61  	public static Continue continueResponse() { return new Continue(); }
62  
63  	/** @return A new {@code 101 Switching Protocols} response. */
64  	public static SwitchingProtocols switchingProtocols() { return new SwitchingProtocols(); }
65  
66  	// ------------------------------------------------------------------------------------------------------------------
67  	// 2xx Success
68  	// ------------------------------------------------------------------------------------------------------------------
69  
70  	/** @return A new {@code 200 OK} response. */
71  	public static Ok ok() { return new Ok(); }
72  
73  	/** @param body The response body. Can be <jk>null</jk> for no response body. @return A new {@code 200 OK} response. */
74  	public static Ok ok(String body) { return new Ok(body); }
75  
76  	/** @return A new {@code 201 Created} response. */
77  	public static Created created() { return new Created(); }
78  
79  	/** @param body The response body. Can be <jk>null</jk> for no response body. @return A new {@code 201 Created} response. */
80  	public static Created created(String body) { return new Created(body); }
81  
82  	/** @return A new {@code 202 Accepted} response. */
83  	public static Accepted accepted() { return new Accepted(); }
84  
85  	/** @return A new {@code 204 No Content} response. */
86  	public static NoContent noContent() { return new NoContent(); }
87  
88  	/** @return A new {@code 206 Partial Content} response. */
89  	public static PartialContent partialContent() { return new PartialContent(); }
90  
91  	// ------------------------------------------------------------------------------------------------------------------
92  	// 3xx Redirection
93  	// ------------------------------------------------------------------------------------------------------------------
94  
95  	/** @param location The redirect location. Can be <jk>null</jk> to omit the {@code Location} header. @return A new {@code 301 Moved Permanently} response. */
96  	public static MovedPermanently movedPermanently(String location) { return new MovedPermanently().withHeader("Location", location); }
97  
98  	/** @param location The redirect location. Can be <jk>null</jk> to omit the {@code Location} header. @return A new {@code 302 Found} response. */
99  	public static Found found(String location) { return new Found().withHeader("Location", location); }
100 
101 	/** @param location The redirect location. Can be <jk>null</jk> to omit the {@code Location} header. @return A new {@code 303 See Other} response. */
102 	public static SeeOther seeOther(String location) { return new SeeOther().withHeader("Location", location); }
103 
104 	/** @return A new {@code 304 Not Modified} response. */
105 	public static NotModified notModified() { return new NotModified(); }
106 
107 	/** @param location The redirect location. Can be <jk>null</jk> to omit the {@code Location} header. @return A new {@code 307 Temporary Redirect} response. */
108 	public static TemporaryRedirect temporaryRedirect(String location) { return new TemporaryRedirect().withHeader("Location", location); }
109 
110 	/** @param location The redirect location. Can be <jk>null</jk> to omit the {@code Location} header. @return A new {@code 308 Permanent Redirect} response. */
111 	public static PermanentRedirect permanentRedirect(String location) { return new PermanentRedirect().withHeader("Location", location); }
112 
113 	// ------------------------------------------------------------------------------------------------------------------
114 	// 4xx Client Errors
115 	// ------------------------------------------------------------------------------------------------------------------
116 
117 	/** @return A new {@code 400 Bad Request} exception. */
118 	public static BadRequest badRequest() { return new BadRequest(); }
119 
120 	/** @param message The error message. Can be <jk>null</jk> for no message body. @return A new {@code 400 Bad Request} exception. */
121 	public static BadRequest badRequest(String message) { return new BadRequest(message); }
122 
123 	/** @return A new {@code 401 Unauthorized} exception. */
124 	public static Unauthorized unauthorized() { return new Unauthorized(); }
125 
126 	/** @param message The error message. Can be <jk>null</jk> for no message body. @return A new {@code 401 Unauthorized} exception. */
127 	public static Unauthorized unauthorized(String message) { return new Unauthorized(message); }
128 
129 	/** @return A new {@code 403 Forbidden} exception. */
130 	public static Forbidden forbidden() { return new Forbidden(); }
131 
132 	/** @param message The error message. Can be <jk>null</jk> for no message body. @return A new {@code 403 Forbidden} exception. */
133 	public static Forbidden forbidden(String message) { return new Forbidden(message); }
134 
135 	/** @return A new {@code 404 Not Found} exception. */
136 	public static NotFound notFound() { return new NotFound(); }
137 
138 	/** @param message The error message. Can be <jk>null</jk> for no message body. @return A new {@code 404 Not Found} exception. */
139 	public static NotFound notFound(String message) { return new NotFound(message); }
140 
141 	/** @return A new {@code 405 Method Not Allowed} exception. */
142 	public static MethodNotAllowed methodNotAllowed() { return new MethodNotAllowed(); }
143 
144 	/** @return A new {@code 409 Conflict} exception. */
145 	public static Conflict conflict() { return new Conflict(); }
146 
147 	/** @param message The error message. Can be <jk>null</jk> for no message body. @return A new {@code 409 Conflict} exception. */
148 	public static Conflict conflict(String message) { return new Conflict(message); }
149 
150 	/** @return A new {@code 410 Gone} exception. */
151 	public static Gone gone() { return new Gone(); }
152 
153 	/** @return A new {@code 412 Precondition Failed} exception. */
154 	public static PreconditionFailed preconditionFailed() { return new PreconditionFailed(); }
155 
156 	/** @return A new {@code 415 Unsupported Media Type} exception. */
157 	public static UnsupportedMediaType unsupportedMediaType() { return new UnsupportedMediaType(); }
158 
159 	/** @param message The error message. Can be <jk>null</jk> for no message body. @return A new {@code 415 Unsupported Media Type} exception. */
160 	public static UnsupportedMediaType unsupportedMediaType(String message) { return new UnsupportedMediaType(message); }
161 
162 	/** @return A new {@code 422 Unprocessable Entity} exception. */
163 	public static UnprocessableEntity unprocessableEntity() { return new UnprocessableEntity(); }
164 
165 	/** @param message The error message. Can be <jk>null</jk> for no message body. @return A new {@code 422 Unprocessable Entity} exception. */
166 	public static UnprocessableEntity unprocessableEntity(String message) { return new UnprocessableEntity(message); }
167 
168 	/** @return A new {@code 429 Too Many Requests} exception. */
169 	public static TooManyRequests tooManyRequests() { return new TooManyRequests(); }
170 
171 	// ------------------------------------------------------------------------------------------------------------------
172 	// 5xx Server Errors
173 	// ------------------------------------------------------------------------------------------------------------------
174 
175 	/** @return A new {@code 500 Internal Server Error} exception. */
176 	public static InternalServerError internalServerError() { return new InternalServerError(); }
177 
178 	/** @param message The error message. Can be <jk>null</jk> for no message body. @return A new {@code 500 Internal Server Error} exception. */
179 	public static InternalServerError internalServerError(String message) { return new InternalServerError(message); }
180 
181 	/** @param cause The cause. Can be <jk>null</jk> (no cause is chained). @return A new {@code 500 Internal Server Error} exception. */
182 	public static InternalServerError internalServerError(Throwable cause) { return new InternalServerError(cause); }
183 
184 	/** @return A new {@code 501 Not Implemented} exception. */
185 	public static NotImplemented notImplemented() { return new NotImplemented(); }
186 
187 	/** @return A new {@code 503 Service Unavailable} exception. */
188 	public static ServiceUnavailable serviceUnavailable() { return new ServiceUnavailable(); }
189 
190 	/** @param message The error message. Can be <jk>null</jk> for no message body. @return A new {@code 503 Service Unavailable} exception. */
191 	public static ServiceUnavailable serviceUnavailable(String message) { return new ServiceUnavailable(message); }
192 }
