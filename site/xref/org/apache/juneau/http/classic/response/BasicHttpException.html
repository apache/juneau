37  /**
38   * Basic implementation of the {@link HttpResponse} interface for error responses.
39   *
40   * <p>
41   * Although this class implements the various setters defined on the {@link HttpResponse} interface, it's in general
42   * going to be more efficient to set the status/headers/content of this bean through the builder.
43   *
44   * <p>
45   * Immutability is expressed with the "funnel + nested {@code Unmodifiable} snapshot" paradigm: every mutator (fluent
46   * setter and {@code void} interface mutator) routes through the single protected {@link #modify(Runnable)} choke-point,
47   * and each concrete leaf gets a nested {@code X.Unmodifiable} whose only behavioral override is a throwing
48   * {@code modify(...)}.  A leaf's {@link #unmodifiable()} factory returns a point-in-time snapshot of type
49   * {@code X.Unmodifiable}; any attempt to set a property on it throws an {@link UnsupportedOperationException}.
50   *
51   * <p>
52   * This is a {@link Throwable}-rooted hierarchy, so (unlike {@link BasicHttpResponse}) it cannot be a self-typed (CRTP)
53   * root — JLS &sect;8.1.2 forbids a generic {@code Throwable}.  The funnel therefore returns the base type and each leaf
54   * keeps its covariant setter overrides.
55   *
56   * <h5 class='section'>Notes:</h5><ul>
57   * 	<li class='warn'>Beans are not thread safe unless they're marked as unmodifiable.
58   * </ul>
59   *
60   * <h5 class='section'>See Also:</h5><ul>
61   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/JuneauRestCommon">juneau-rest-common Basics</a>
62   * </ul>
63   *
64   * @serial exclude
65   */

102 	 * Constructor.
103 	 *
104 	 * <p>
105 	 * This is the constructor used when parsing an HTTP response.
106 	 *
107 	 * @param response The HTTP response being parsed.  Must not be <jk>null</jk>.
108 	 */
109 	public BasicHttpException(HttpResponse response) {
110 		super((Throwable)null);
111 		assertArgNotNull(ARG_response, response);
112 		var h = response.getLastHeader("Thrown");
113 		if (nn(h)) {
114 			var partsOpt = thrown(h.getValue()).asParts();
115 			if (partsOpt.isPresent() && !partsOpt.get().isEmpty()) // HTT: isPresent() is always true here — thrown(String) returns null (NPE on the .asParts() call above, before this line) whenever its arg is null, and Thrown.of(non-null) always yields a present Optional; the false side of isPresent() is unreachable without a prior NPE.
116 				setMessage(partsOpt.get().get(0).getMessage());
117 		}
118 		setHeaders(response.getAllHeaders());
119 		setContent(response.getEntity());
120 		setStatusCode(response.getStatusLine().getStatusCode());
121 	}

219 	 * Returns all error messages from all errors in this stack.
220 	 *
221 	 * <p>
222 	 * Typically useful if you want to render all the error messages in the stack, but don't want to render all the
223 	 * stack traces too.
224 	 *
225 	 * @param scrubForXssVulnerabilities
226 	 * 	If <jk>true</jk>, replaces <js>'&lt;'</js>, <js>'&gt;'</js>, and <js>'&amp;'</js> characters with spaces.
227 	 * @return All error messages from all errors in this stack.
228 	 */
229 	public String getFullStackMessage(boolean scrubForXssVulnerabilities) {
230 		var msg = getMessage();
231 		StringBuilder sb = new StringBuilder();
232 		if (nn(msg)) {
233 			if (scrubForXssVulnerabilities)
234 				msg = msg.replace('<', ' ').replace('>', ' ').replace('&', ' ');
235 			sb.append(msg);
236 		}
237 		Throwable e = getCause();
238 		while (nn(e)) {
239 			msg = e.getMessage();
240 			if (nn(msg) && scrubForXssVulnerabilities)
241 				msg = msg.replace('<', ' ').replace('>', ' ').replace('&', ' ');
242 			var cls = cns(e);
243 			if (msg == null)
244 				sb.append(f("\nCaused by (%s)", cls));
245 			else
246 				sb.append(f("\nCaused by (%s): %s", cls, msg));
247 			e = e.getCause();
248 		}
249 		return sb.toString();
250 	}

294 	 * Returns the root cause of this exception.
295 	 *
296 	 * <p>
297 	 * The root cause is the first exception in the init-cause parent chain that's not one of the following:
298 	 * <ul>
299 	 * 	<li>{@link BasicHttpException}
300 	 * 	<li>{@link InvocationTargetException}
301 	 * </ul>
302 	 *
303 	 * @return The root cause of this exception, or <jk>null</jk> if no root cause was found.
304 	 */
305 	public Throwable getRootCause() {
306 		Throwable t = this;
307 		while (nn(t)) {
308 			if (! (t instanceof BasicHttpException || t instanceof InvocationTargetException))
309 				return t;
310 			t = t.getCause();
311 		}
312 		return null;
313 	}

628 	 * Its only behavioral override is {@link #modify(Runnable)}, which throws — because all mutation is funneled through
629 	 * {@code modify(...)}, this single override freezes the entire mutation surface.
630 	 */
631 	public static class Unmodifiable extends BasicHttpException implements UnmodifiableBean {
632 
633 		private static final long serialVersionUID = 1L;
634 
635 		/**
636 		 * Constructor.
637 		 *
638 		 * @param copyFrom The exception to snapshot.  Must not be <jk>null</jk>.
639 		 */
640 		@SuppressWarnings({
641 			"java:S1699" // Paradigm intentionally calls the overridable freeze() from the ctor to deep-freeze sub-beans.
642 		})
643 		protected Unmodifiable(BasicHttpException copyFrom) {
644 			super(copyFrom);
645 			freeze();
646 		}
647 
648 		@Override /* Overridden from BasicHttpException */
649 		protected BasicHttpException modify(Runnable mutation) {
650 			throw uoex("Bean is unmodifiable.");
651 		}
652 	}
653 }
