17  package org.apache.juneau.http;
18  
19  /**
20   * Neutralizes log-forging (log-injection) in client-controlled strings before they are written to a log line.
21   *
22   * <p>
23   * A remote client can place CR/LF (or Unicode line/paragraph separators, bidi controls, or other control characters)
24   * into a request URI, header name/value, or body, and — if that text is interpolated verbatim into a log record — forge
25   * additional physical log lines (e.g. a fake status banner) or corrupt a downstream log parser/SIEM. This utility
26   * replaces every such character with a <b>visible, inert escape</b> (<c>\r</c>, <c>\n</c>, or <c>\\uXXXX</c>) so the
27   * value's information is preserved (a reader can still see there <i>was</i> a newline) but it can no longer terminate
28   * the current line or start a new one.
29   *
30   * <p>
31   * The transform:
32   * <ul>
33   * 	<li>CR (<c>\\u000D</c>) &rarr; <c>\r</c>, LF (<c>\\u000A</c>) &rarr; <c>\n</c>.
34   * 	<li>All other C0 controls (<c>\\u0000</c>&ndash;<c>\\u001F</c>) <b>except TAB</b> (<c>\\u0009</c>) &rarr; <c>\\uXXXX</c>.
35   * 	<li>DEL and C1 controls (<c>\\u007F</c>&ndash;<c>\\u009F</c>, which includes NEL <c>\\u0085</c>) &rarr; <c>\\uXXXX</c>.
36   * 	<li>Line separator (<c>\\u2028</c>), paragraph separator (<c>\\u2029</c>), and bidi controls
37   * 		(<c>\\u202A</c>&ndash;<c>\\u202E</c>, <c>\\u2066</c>&ndash;<c>\\u2069</c>) &rarr; <c>\\uXXXX</c>.
38   * 	<li>TAB (<c>\\u0009</c>) is <b>preserved</b> &mdash; it cannot forge a line.
39   * </ul>
40   *
41   * <p>
42   * When a length cap is supplied, truncation is applied <b>after</b> escaping, counts <b>sanitized</b> (character) output,
43   * appends a <c>&hellip;[truncated]</c> marker, and never cuts an escape sequence or a surrogate pair in half. A fast path
44   * returns the original reference unchanged when the string contains no character requiring an escape and is within the
45   * cap, so well-behaved clients incur no allocation.
46   *
47   * <p>
48   * All methods are thread-safe (the utility is stateless).
49   *
50   * <h5 class='section'>See Also:</h5><ul>
51   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/RestServerLoggingAndDebugging">Logging / Debugging</a>
52   * </ul>
53   *
54   * @since 10.0.0
55   */
56  public final class DebugTextSanitizer {
57  
58  	/** Marker appended when a value is truncated by the length cap. */
59  	public static final String TRUNCATED_MARKER = "\u2026[truncated]";
60  
61  	private DebugTextSanitizer() {}
62  
63  	/**
64  	 * Escapes every log-forging control character in the given string, with no length cap.
65  	 *
66  	 * @param value The string to sanitize. Can be <jk>null</jk> (returns <jk>null</jk>).
67  	 * @return The sanitized string, or the original reference if nothing needed escaping. <jk>null</jk> if the input was
68  	 * 	<jk>null</jk>.
69  	 */
70  	public static String sanitize(String value) {
71  		return sanitize(value, Integer.MAX_VALUE);
72  	}
73  
74  	/**
75  	 * Escapes every log-forging control character in the given string, then caps the <i>sanitized</i> output to
76  	 * {@code maxLen} characters.
77  	 *
78  	 * <p>
79  	 * The cap counts the escaped (rendered) length, appends {@code …[truncated]} when it clips, and never splits an
80  	 * escape sequence or a surrogate pair.
81  	 *
82  	 * @param value The string to sanitize. Can be <jk>null</jk> (returns <jk>null</jk>).
83  	 * @param maxLen The maximum number of sanitized characters to keep (excluding the truncation marker).
84  	 * @return The sanitized (and possibly truncated) string, or the original reference if nothing needed escaping and it
85  	 * 	was within the cap. <jk>null</jk> if the input was <jk>null</jk>.
86  	 */
87  	@SuppressWarnings({
88  		"java:S3776" // Sanitizer encodes a control-character state machine; complexity is inherent.
89  	})
90  	public static String sanitize(String value, int maxLen) {
91  		if (value == null)
92  			return null;
93  
94  		// Fast path: nothing to escape and already within the cap → return the original reference (no allocation).
95  		var clean = value.length() <= maxLen;
96  		if (clean) {
97  			for (var i = 0; i < value.length(); i++) {
98  				if (needsEscape(value.charAt(i))) {
99  					clean = false;
100 					break;
101 				}
102 			}
103 		}
104 		if (clean)
105 			return value;
106 
107 		var sb = new StringBuilder(Math.min(value.length(), Math.max(maxLen, 0)) + 16);
108 		var truncated = false;
109 		var n = value.length();
110 		var i = 0;
111 		while (i < n) {
112 			var c = value.charAt(i);
113 			String token;
114 			int consumed;
115 			if (needsEscape(c)) {
116 				token = escape(c);
117 				consumed = 1;
118 			} else if (Character.isHighSurrogate(c) && i + 1 < n && Character.isLowSurrogate(value.charAt(i + 1))) {
119 				// Keep a surrogate pair together so the cap never cuts a code point in half.
120 				token = value.substring(i, i + 2);
121 				consumed = 2;
122 			} else {
123 				token = String.valueOf(c);
124 				consumed = 1;
125 			}
126 			if (sb.length() + token.length() > maxLen) {
127 				truncated = true;
128 				break;
129 			}
130 			sb.append(token);
131 			i += consumed;
132 		}
133 		if (truncated)
134 			sb.append(TRUNCATED_MARKER);
135 		return sb.toString();
136 	}
137 
138 	private static boolean needsEscape(char c) {
139 		if (c == '\t')
140 			return false;
141 		if (c <= '\u001F')  // C0 controls (CR/LF included).
142 			return true;
143 		if (c >= '\u007F' && c <= '\u009F')  // DEL + C1 controls (includes NEL \u0085).
144 			return true;
145 		if (c == '\u2028' || c == '\u2029')  // Line / paragraph separators.
146 			return true;
147 		if (c >= '\u202A' && c <= '\u202E')  // Bidi embedding/override controls.
148 			return true;
149 		return c >= '\u2066' && c <= '\u2069';  // Bidi isolate controls.
150 	}
151 
152 	private static String escape(char c) {
153 		if (c == '\r')
154 			return "\\r";
155 		if (c == '\n')
156 			return "\\n";
157 		return String.format("\\u%04X", (int) c);
158 	}
159 }
