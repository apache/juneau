17  package org.apache.juneau.http.remote;
18  
19  import static java.lang.annotation.ElementType.*;
20  import static java.lang.annotation.RetentionPolicy.*;
21  
22  import java.lang.annotation.*;
23  
24  /**
25   * REST proxy method-level <c>multipart/form-data</c> marker annotation.
26   *
27   * <p>
28   * Marks a {@link Remote}-proxy method whose request body is a {@code multipart/form-data} message (RFC 7578)
29   * assembled from the method's {@link org.apache.juneau.http.Part @Part}-annotated parameters &mdash; the
30   * Retrofit <c>@Multipart</c> pattern.  Each {@code @Part} parameter contributes one part (a text field, a
31   * file/stream/byte-array upload, or a serialized bean); the engine adapts those parameters into the streaming
32   * {@link org.apache.juneau.http.entity.MultipartBody} and the {@code multipart/form-data} {@code Content-Type}
33   * (with its generated boundary) is applied automatically.
34   *
35   * <h5 class='figure'>Example:</h5>
36   * <p class='bjava'>
37   * 	<ja>@Remote</ja>(path=<js>"/api"</js>)
38   * 	<jk>public interface</jk> UploadService {
39   * 		<ja>@RemotePost</ja>(<js>"/upload"</js>)
40   * 		<ja>@Multipart</ja>
41   * 		String upload(
42   * 			<ja>@Part</ja>(<js>"title"</js>) String <jv>title</jv>,
43   * 			<ja>@Part</ja>(name=<js>"attachment"</js>, contentType=<js>"application/pdf"</js>) File <jv>report</jv>);
44   * 	}
45   * </p>
46   *
47   * <h5 class='section'>Body-mode exclusivity:</h5>
48   * <p>
49   * A method is either <i>multipart</i> ({@code @Multipart} + one or more {@link org.apache.juneau.http.Part @Part}
50   * parameters) or <i>single-body</i> ({@link org.apache.juneau.http.Content @Content}), never both.  Declaring both
51   * on the same method &mdash; or declaring {@code @Multipart} with no {@code @Part} parameters, or a {@code @Part}
52   * parameter without {@code @Multipart} &mdash; is rejected at proxy-build time with an
53   * {@link IllegalArgumentException}.
54   *
55   * <h5 class='section'>Notes:</h5><ul>
56   * 	<li class='note'>
57   * 		<b>Escape hatch (Option B):</b> for advanced cases you can still hand-assemble a
58   * 		{@link org.apache.juneau.http.entity.MultipartBody} and pass it as a single
59   * 		{@link org.apache.juneau.http.Content @Content} body without {@code @Multipart}; that path is unchanged.
60   * 	<li class='note'>
61   * 		Honored by the next-generation engine (<c>RestClient.remote(...)</c>) only; the classic engine currently
62   * 		ignores this annotation.
63   * </ul>
64   *
65   * <p>
66   * <b>Beta — API subject to change:</b> This type is part of the next-generation REST client and HTTP stack
67   * ({@code org.apache.juneau.marshall.ng.*}).
68   * It is not API-frozen: binary- and source-incompatible changes may appear in the <b>next major</b> Juneau release
69   * (and possibly earlier).
70   *
71   * <h5 class='section'>See Also:</h5><ul>
72   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/NextGenRestClient">juneau-ng REST client</a>
73   * 	<li class='extlink'><a class="doclink" href="https://www.rfc-editor.org/rfc/rfc7578">RFC 7578 — multipart/form-data</a>
74   * </ul>
75   *
76   * @since 9.2.1
77   */
78  @Documented
79  @Target(METHOD)
80  @Retention(RUNTIME)
81  @Inherited
82  public @interface Multipart {
83  }
