17  package org.apache.juneau.http;
18  
19  import static org.apache.juneau.commons.utils.AssertionUtils.*;
20  import static org.apache.juneau.commons.utils.Shorts.*;
21  import static org.apache.juneau.commons.utils.StringUtils.*;
22  
23  import java.util.function.*;
24  
25  /**
26   * The HTTP protocol version that appears in request and response start lines (e.g. {@code "HTTP/1.1"}).
27   *
28   * <p>
29   * Mirrors the semantics of {@code org.apache.http.ProtocolVersion} without the Apache HttpCore dependency.
30   * The protocol name and major/minor numbers are stored separately so callers can compare versions numerically.
31   *
32   * <p>
33   * Create instances via {@link #of(String, int, int)} or {@link #parse(String)}:
34   * <p class='bjava'>
35   * 	<jc>// "HTTP/1.1"</jc>
36   * 	HttpProtocolVersion <jv>v1</jv> = HttpProtocolVersion.<jsf>HTTP_1_1</jsf>;
37   *
38   * 	<jc>// Parse from a wire string</jc>
39   * 	HttpProtocolVersion <jv>v2</jv> = HttpProtocolVersion.<jsm>parse</jsm>(<js>"HTTP/2.0"</js>);
40   * </p>
41   *
42   * @param protocol The protocol name (e.g. {@code "HTTP"}). Must not be <jk>null</jk>.
43   * @param major The major version number (e.g. {@code 1}).
44   * @param minor The minor version number (e.g. {@code 1}).
45   *
46   * @since 10.0.0
47   */
48  public record HttpProtocolVersion(String protocol, int major, int minor) {
49  
50  	/** {@code HTTP/1.0}. */
51  	public static final HttpProtocolVersion HTTP_1_0 = new HttpProtocolVersion("HTTP", 1, 0);
52  
53  	/** {@code HTTP/1.1}. */
54  	public static final HttpProtocolVersion HTTP_1_1 = new HttpProtocolVersion("HTTP", 1, 1);
55  
56  	/** {@code HTTP/2.0}. */
57  	public static final HttpProtocolVersion HTTP_2_0 = new HttpProtocolVersion("HTTP", 2, 0);
58  
59  	/**
60  	 * Canonical constructor.
61  	 *
62  	 * @param protocol The protocol name. Must not be <jk>null</jk>.
63  	 * @param major The major version (must be {@code >= 0}).
64  	 * @param minor The minor version (must be {@code >= 0}).
65  	 */
66  	public HttpProtocolVersion {
67  		assertArgNotNull("protocol", protocol);
68  		if (major < 0)
69  			throw iaex("Major version must be >= 0, got %s", major);
70  		if (minor < 0)
71  			throw iaex("Minor version must be >= 0, got %s", minor);
72  	}
73  
74  	/**
75  	 * Creates a protocol version with the given protocol name and version numbers.
76  	 *
77  	 * @param protocol The protocol name (e.g. {@code "HTTP"}). Must not be <jk>null</jk>.
78  	 * @param major The major version number.
79  	 * @param minor The minor version number.
80  	 * @return A new instance. Never <jk>null</jk>.
81  	 */
82  	public static HttpProtocolVersion of(String protocol, int major, int minor) {
83  		return new HttpProtocolVersion(protocol, major, minor);
84  	}
85  
86  	/**
87  	 * Parses a wire-format protocol version string.
88  	 *
89  	 * <p>
90  	 * Accepts the standard {@code PROTOCOL/major.minor} form (e.g. {@code "HTTP/1.1"}) and the abbreviated
91  	 * {@code PROTOCOL/major} form (e.g. {@code "HTTP/2"}, which is parsed as major=2, minor=0).
92  	 *
93  	 * @param s The protocol version string. Must not be <jk>null</jk>.
94  	 * @return A new instance. Never <jk>null</jk>.
95  	 * @throws IllegalArgumentException If the string cannot be parsed.
96  	 */
97  	public static HttpProtocolVersion parse(String s) {
98  		assertArgNotNull("s", s);
99  		var slash = s.indexOf('/');
100 		if (slash < 0)
101 			throw iaex("Invalid protocol version: %s", s);
102 		var protocol = s.substring(0, slash);
103 		var rest = s.substring(slash + 1);
104 		var dot = rest.indexOf('.');
105 		Supplier<RuntimeException> err = () -> iaex("Invalid protocol version: %s", s);
106 		if (dot < 0)
107 			return new HttpProtocolVersion(protocol, parseInt(rest, err), 0);
108 		return new HttpProtocolVersion(protocol, parseInt(rest.substring(0, dot), err), parseInt(rest.substring(dot + 1), err));
109 	}
110 
111 	/**
112 	 * Returns the canonical wire-format string (e.g. {@code "HTTP/1.1"}).
113 	 *
114 	 * @return The wire-format string. Never <jk>null</jk>.
115 	 */
116 	@Override /* Object */
117 	public String toString() {
118 		return protocol + "/" + major + "." + minor;
119 	}
120 }
