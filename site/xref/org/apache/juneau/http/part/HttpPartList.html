27  /**
28   * A mutable, ordered list of {@link HttpPart} instances.
29   *
30   * <p>
31   * Mirrors the semantics of {@code PartList} from {@code juneau-rest-common-classic} without the
32   * Apache HttpCore dependency. Part-name comparisons are case-sensitive by default (matching the
33   * classic default for parts; headers use case-insensitive comparisons).
34   *
35   * <p>
36   * Use {@link PartList} when you need an immutable list bound to an HTTP body (e.g.
37   * {@code application/x-www-form-urlencoded}); use {@code HttpPartList} when you need a mutable
38   * builder for default form data / query parameter lists.
39   *
40   * <p>
41   * <b>Beta — API subject to change:</b> This type is part of the next-generation REST client and HTTP stack
42   * ({@code org.apache.juneau.marshall.ng.*}). It is not API-frozen: binary- and source-incompatible changes may appear in
43   * the <b>next major</b> Juneau release (and possibly earlier).
44   *
45   * <h5 class='section'>See Also:</h5><ul>
46   * 	<li class='link'><a class="doclink" href="https://juneau.apache.org/docs/topics/NextGenRestClient">juneau-ng REST client</a>
47   * </ul>
48   *
49   * @since 10.0.0
50   */
51  @SuppressWarnings({
52  	"java:S2160" // equals() inherited from ArrayList; list equality is element-based which is correct
53  })
54  public class HttpPartList extends ArrayList<HttpPart> {

94  	/**
95  	 * Creates a new {@link HttpPartList} initialized with the given alternating {@code name}/{@code value} pairs.
96  	 *
97  	 * @param pairs Alternating name/value strings. Can be <jk>null</jk> (treated as empty). Length must be even.
98  	 * @return A new instance. Never <jk>null</jk>.
99  	 * @throws IllegalArgumentException If the number of arguments is odd.
100 	 */
101 	public static HttpPartList ofPairs(String...pairs) {
102 		var x = new HttpPartList();
103 		if (pairs == null)
104 			pairs = new String[0];
105 		assertArg(pairs.length % 2 == 0, "Odd number of parameters passed into HttpPartList.ofPairs()");
106 		for (var i = 0; i < pairs.length; i += 2)
107 			x.add(HttpPartBean.of(pairs[i], pairs[i + 1]));
108 		return x;
109 	}

137 	/**
138 	 * Appends the given part.
139 	 *
140 	 * @param value The part to add. {@code null} is ignored.
141 	 * @return This object.
142 	 */
143 	public HttpPartList append(HttpPart value) {
144 		if (value != null)
145 			add(value);
146 		return this;
147 	}
148 
149 	/**
150 	 * Appends the given parts in order.
151 	 *
152 	 * @param values The parts to add. {@code null} entries are ignored.
153 	 * @return This object.
154 	 */
155 	public HttpPartList append(HttpPart...values) {
156 		if (values != null)
157 			for (var v : values)
158 				append(v);
159 		return this;
160 	}
161 
162 	/**
163 	 * Appends the given parts in order.
164 	 *
165 	 * @param values The parts to add. {@code null} entries are ignored.
166 	 * @return This object.
167 	 */
168 	public HttpPartList append(List<HttpPart> values) {
169 		if (values != null)
170 			values.forEach(this::append);
171 		return this;
172 	}
173 
174 	/**
175 	 * Appends a part constructed from the given name and string value.
176 	 *
177 	 * @param name The part name. Must not be <jk>null</jk>.
178 	 * @param value The part value. Can be <jk>null</jk> to omit the part from the request.
179 	 * @return This object.
180 	 */
181 	public HttpPartList append(String name, String value) {
182 		return append(HttpPartBean.of(name, value));
183 	}
184 
185 	/**
186 	 * Sets the given part as a default. If a part with the same name already exists, this is a no-op
187 	 * (first-in-chain wins per name).
188 	 *
189 	 * @param value The part to set. {@code null} is ignored.
190 	 * @return This object.
191 	 */
192 	public HttpPartList setDefault(HttpPart value) {
193 		if (value != null && !contains(value.getName()))
194 			add(value);
195 		return this;
196 	}
197 
198 	/**
199 	 * Sets the given parts as defaults, in order. For each, if a part with the same name already exists,
200 	 * the entry is skipped (first-in-chain wins per name).
201 	 *
202 	 * @param values The parts to set. {@code null} entries are ignored.
203 	 * @return This object.
204 	 */
205 	public HttpPartList setDefault(HttpPart...values) {
206 		if (values != null)
207 			for (var v : values)
208 				setDefault(v);
209 		return this;
210 	}
211 
212 	/**
213 	 * Sets a default part with the given name and value (first-in-chain wins per name).
214 	 *
215 	 * @param name The part name. Must not be <jk>null</jk>.
216 	 * @param value The part value. Can be <jk>null</jk> to omit the part from the request.
217 	 * @return This object.
218 	 */
219 	public HttpPartList setDefault(String name, String value) {
220 		return setDefault(HttpPartBean.of(name, value));
221 	}
222 
223 	/**
224 	 * Sets the given part, replacing any prior entry with the same name.
225 	 *
226 	 * @param value The part to set. {@code null} is ignored.
227 	 * @return This object.
228 	 */
229 	public HttpPartList set(HttpPart value) {
230 		if (value == null)
231 			return this;
232 		removeAll(value.getName());
233 		add(value);
234 		return this;
235 	}
236 
237 	/**
238 	 * Sets the given parts, each replacing any prior entry with the same name.
239 	 *
240 	 * @param values The parts to set. {@code null} entries are ignored.
241 	 * @return This object.
242 	 */
243 	public HttpPartList set(HttpPart...values) {
244 		if (values != null)
245 			for (var v : values)
246 				set(v);
247 		return this;
248 	}
249 
250 	/**
251 	 * Sets a part with the given name and value, replacing any prior entry with the same name.
252 	 *
253 	 * @param name The part name. Must not be <jk>null</jk>.
254 	 * @param value The part value. Can be <jk>null</jk> to omit the part from the request.
255 	 * @return This object.
256 	 */
257 	public HttpPartList set(String name, String value) {
258 		return set(HttpPartBean.of(name, value));
259 	}
260 
261 	/**
262 	 * Removes all parts with the given name.
263 	 *
264 	 * @param name The part name. Must not be <jk>null</jk>.
265 	 * @return This object.
266 	 */
267 	public HttpPartList removeAll(String name) {
268 		assertArgNotNull("name", name);
269 		removeIf(p -> nameMatches(p, name));
270 		return this;
271 	}
272 
273 	/**
274 	 * Returns {@code true} if a part with the given name is present.
275 	 *
276 	 * @param name The part name.
277 	 * @return {@code true} if present.
278 	 */
279 	public boolean contains(String name) {
280 		return stream().anyMatch(p -> nameMatches(p, name));
281 	}
282 
283 	/**
284 	 * Returns the first part with the given name, or {@code null} if absent.
285 	 *
286 	 * @param name The part name.
287 	 * @return The first matching part, or {@code null}.
288 	 */
289 	public HttpPart getFirst(String name) {
290 		return stream().filter(p -> nameMatches(p, name)).findFirst().orElse(null);
291 	}
292 
293 	/**
294 	 * Returns the last part with the given name, or {@code null} if absent.
295 	 *
296 	 * @param name The part name.
297 	 * @return The last matching part, or {@code null}.
298 	 */
299 	public HttpPart getLast(String name) {
300 		HttpPart last = null;
301 		for (var p : this)
302 			if (nameMatches(p, name))
303 				last = p;
304 		return last;
305 	}
306 
307 	/**
308 	 * Returns all parts with the given name in insertion order.
309 	 *
310 	 * @param name The part name.
311 	 * @return A new list of matching parts. Never <jk>null</jk>.
312 	 */
313 	public List<HttpPart> getAll(String name) {
314 		return stream().filter(p -> nameMatches(p, name)).toList();
315 	}
316 
317 	/**
318 	 * Returns all parts in this list as a typed array.
319 	 *
320 	 * @return A new array. Never <jk>null</jk>.
321 	 */
322 	public HttpPart[] getAll() {
323 		return toPartArray();
324 	}

326 	/**
327 	 * Performs an action on each part whose name matches.
328 	 *
329 	 * @param name The part name.
330 	 * @param action The action.
331 	 * @return This object.
332 	 */
333 	public HttpPartList forEach(String name, Consumer<HttpPart> action) {
334 		for (var p : this)
335 			if (nameMatches(p, name))
336 				action.accept(p);
337 		return this;
338 	}

379 	private boolean nameMatches(HttpPart p, String name) {
380 		return caseInsensitive ? eqic(p.getName(), name) : eq(p.getName(), name);
381 	}

383 	/**
384 	 * An unmodifiable snapshot of a {@link HttpPartList}.
385 	 *
386 	 * <p>
387 	 * This is the D4 "collection-mutator-override" variant of the immutability paradigm.  Because {@link HttpPartList}
388 	 * extends {@link ArrayList}, a single funnel cannot intercept the inherited {@code ArrayList} mutators, so this
389 	 * subclass snapshot-copies the elements in its constructor and overrides the entire collection-mutator surface — its
390 	 * own fluent mutators plus the inherited {@code ArrayList} mutators (including the {@code iterator()} /
391 	 * {@code listIterator()} mutators) — to throw {@link UnsupportedOperationException}.  Read operations and
392 	 * non-mutating iteration continue to work.
393 	 */
394 	@SuppressWarnings({
395 		"java:S2160" // equals() inherited from ArrayList; list equality is element-based which is correct
396 	})
397 	public static class Unmodifiable extends HttpPartList implements UnmodifiableBean {
398 
399 		private static final long serialVersionUID = 1L;
400 
401 		/**
402 		 * Constructor.
403 		 *
404 		 * @param copyFrom The list to snapshot-copy. Must not be <jk>null</jk>.
405 		 */
406 		Unmodifiable(HttpPartList copyFrom) {
407 			super(copyFrom);
408 		}
409 
410 		// --- Fluent mutators -----------------------------------------------------------------------------------------
411 
412 		@Override public HttpPartList append(HttpPart value) { throw uoex(RO); }
413 		@Override public HttpPartList append(HttpPart...values) { throw uoex(RO); }
414 		@Override public HttpPartList append(List<HttpPart> values) { throw uoex(RO); }
415 		@Override public HttpPartList append(String name, String value) { throw uoex(RO); }
416 		@Override public HttpPartList caseInsensitive(boolean value) { throw uoex(RO); }
417 		@Override public HttpPartList removeAll(String name) { throw uoex(RO); }
418 		@Override public HttpPartList set(HttpPart value) { throw uoex(RO); }
419 		@Override public HttpPartList set(HttpPart...values) { throw uoex(RO); }
420 		@Override public HttpPartList set(String name, String value) { throw uoex(RO); }
421 		@Override public HttpPartList setDefault(HttpPart value) { throw uoex(RO); }
422 		@Override public HttpPartList setDefault(HttpPart...values) { throw uoex(RO); }
423 		@Override public HttpPartList setDefault(String name, String value) { throw uoex(RO); }
424 
425 		// --- Inherited ArrayList collection mutators -----------------------------------------------------------------
426 
427 		@Override public boolean add(HttpPart e) { throw uoex(RO); }
428 		@Override public void add(int index, HttpPart element) { throw uoex(RO); }
429 		@Override public boolean addAll(Collection<? extends HttpPart> c) { throw uoex(RO); }
430 		@Override public boolean addAll(int index, Collection<? extends HttpPart> c) { throw uoex(RO); }
431 		@Override public void clear() { throw uoex(RO); }
432 		@Override public boolean remove(Object o) { throw uoex(RO); }
433 		@Override public HttpPart remove(int index) { throw uoex(RO); }
434 		@Override public boolean removeAll(Collection<?> c) { throw uoex(RO); }
435 		@Override public boolean removeIf(Predicate<? super HttpPart> filter) { throw uoex(RO); }
436 		@Override public void replaceAll(UnaryOperator<HttpPart> operator) { throw uoex(RO); }
437 		@Override public boolean retainAll(Collection<?> c) { throw uoex(RO); }
438 		@Override public HttpPart set(int index, HttpPart element) { throw uoex(RO); }
439 		@Override public void sort(Comparator<? super HttpPart> c) { throw uoex(RO); }
440 
441 		@Override public Iterator<HttpPart> iterator() {
442 			var i = super.iterator();
443 			return new Iterator<>() {
444 				@Override public boolean hasNext() { return i.hasNext(); }
445 				@Override public HttpPart next() { return i.next(); }
446 				@Override public void remove() { throw uoex(RO); }
447 			};
448 		}
449 
450 		@Override public ListIterator<HttpPart> listIterator() { return listIterator(0); }
451 
452 		@Override public ListIterator<HttpPart> listIterator(int index) {
453 			var i = super.listIterator(index);
454 			return new ListIterator<>() {
455 				@Override public boolean hasNext() { return i.hasNext(); }
456 				@Override public HttpPart next() { return i.next(); }
457 				@Override public boolean hasPrevious() { return i.hasPrevious(); }
458 				@Override public HttpPart previous() { return i.previous(); }
459 				@Override public int nextIndex() { return i.nextIndex(); }
460 				@Override public int previousIndex() { return i.previousIndex(); }
461 				@Override public void add(HttpPart e) { throw uoex(RO); }
462 				@Override public void set(HttpPart e) { throw uoex(RO); }
463 				@Override public void remove() { throw uoex(RO); }
464 			};
465 		}
466 	}
