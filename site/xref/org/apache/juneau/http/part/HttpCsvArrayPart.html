25  /**
26   * An {@link org.apache.juneau.http.HttpPart} whose value is a comma-delimited list of strings.
27   *
28   * <p>
29   * Mirrors the semantics of {@code BasicCsvArrayPart} from {@code juneau-rest-common-classic} without
30   * the Apache HttpCore dependency.
31   *
32   * @since 10.0.0
33   */
34  @SuppressWarnings({
35  	"java:S2160" // Equality is fully determined by name + getValue() in HttpPartBean; the typed fields are derived from the wire value and add no state to compare
36  })
37  public class HttpCsvArrayPart extends HttpPartBean {
38  
39  	private static final String[] EMPTY = {};
40  
41  	/**
42  	 * Creates an {@link HttpCsvArrayPart} with the given name and array value.
43  	 *
44  	 * @param name The part name. Must not be <jk>null</jk>.
45  	 * @param value The array of values. May be <jk>null</jk>.
46  	 * @return A new instance. Never <jk>null</jk>.
47  	 */
48  	public static HttpCsvArrayPart of(String name, String...value) {
49  		return new HttpCsvArrayPart(name, value);
50  	}
51  
52  	/**
53  	 * Creates an {@link HttpCsvArrayPart} by parsing the given comma-delimited wire value.
54  	 *
55  	 * @param name The part name. Must not be <jk>null</jk>.
56  	 * @param value The wire value. May be <jk>null</jk> or empty.
57  	 * @return A new instance. Never <jk>null</jk>.
58  	 */
59  	public static HttpCsvArrayPart ofString(String name, String value) {
60  		return new HttpCsvArrayPart(name, value);
61  	}
62  
63  	/**
64  	 * Creates an {@link HttpCsvArrayPart} with a lazy array supplier.
65  	 *
66  	 * @param name The part name. Must not be <jk>null</jk>.
67  	 * @param supplier Supplier for the array value. Must not be <jk>null</jk>.
68  	 * @return A new instance. Never <jk>null</jk>.
69  	 */
70  	public static HttpCsvArrayPart ofLazy(String name, Supplier<String[]> supplier) {
71  		return new HttpCsvArrayPart(name, supplier);
72  	}
73  
74  	private final String[] typedValue;
75  	private final Supplier<String[]> typedSupplier;
76  
77  	/**
78  	 * Constructor accepting an array of values.
79  	 *
80  	 * @param name The part name. Must not be <jk>null</jk>.
81  	 * @param value The array of values. May be <jk>null</jk>.
82  	 */
83  	public HttpCsvArrayPart(String name, String...value) {
84  		super(name, value == null ? null : join(value, ','));
85  		this.typedValue = cp(value);
86  		this.typedSupplier = null;
87  	}
88  
89  	/**
90  	 * Constructor accepting a comma-delimited wire string.
91  	 *
92  	 * @param name The part name. Must not be <jk>null</jk>.
93  	 * @param value The wire value. May be <jk>null</jk> or empty.
94  	 */
95  	public HttpCsvArrayPart(String name, String value) {
96  		super(name, value);
97  		this.typedValue = splita(value);
98  		this.typedSupplier = null;
99  	}
100 
101 	/**
102 	 * Constructor accepting a lazy supplier of array values.
103 	 *
104 	 * @param name The part name. Must not be <jk>null</jk>.
105 	 * @param supplier Supplier for the array value. Must not be <jk>null</jk>.
106 	 */
107 	public HttpCsvArrayPart(String name, Supplier<String[]> supplier) {
108 		super(name, () -> {
109 			var v = supplier.get();
110 			return v == null ? null : join(v, ',');
111 		});
112 		this.typedValue = null;
113 		this.typedSupplier = supplier;
114 	}
115 
116 	/**
117 	 * Returns the value as a copy of the underlying array, wrapped in an {@link Optional}.
118 	 *
119 	 * @return The array value, wrapped in an {@link Optional}. Never <jk>null</jk>.
120 	 */
121 	public Optional<String[]> asArray() {
122 		return o(toArray());
123 	}
124 
125 	/**
126 	 * Returns the value as an unmodifiable {@link List}, wrapped in an {@link Optional}.
127 	 *
128 	 * @return The list value, wrapped in an {@link Optional}. Never <jk>null</jk>.
129 	 */
130 	public Optional<List<String>> asList() {
131 		return o(toList());
132 	}
133 
134 	/**
135 	 * Returns {@code true} if this part contains the given value (case-sensitive).
136 	 *
137 	 * @param val The value to check for.
138 	 * @return {@code true} if the value is present.
139 	 */
140 	public boolean contains(String val) {
141 		if (val == null)
142 			return false;
143 		var arr = value();
144 		if (arr == null)
145 			return false;
146 		for (var v : arr)
147 			if (eq(v, val))
148 				return true;
149 		return false;
150 	}
151 
152 	/**
153 	 * Returns {@code true} if this part contains the given value (case-insensitive).
154 	 *
155 	 * @param val The value to check for.
156 	 * @return {@code true} if the value is present.
157 	 */
158 	public boolean containsIgnoreCase(String val) {
159 		if (val == null)
160 			return false;
161 		var arr = value();
162 		if (arr == null)
163 			return false;
164 		for (var v : arr)
165 			if (eqic(v, val))
166 				return true;
167 		return false;
168 	}
169 
170 	/**
171 	 * Returns a copy of the underlying array, or {@code null} if not set.
172 	 *
173 	 * @return The array, possibly {@code null}.
174 	 */
175 	public String[] toArray() {
176 		return cp(value());
177 	}
178 
179 	/**
180 	 * Returns the value as an unmodifiable {@link List}, or {@code null} if not set.
181 	 *
182 	 * @return The list, possibly {@code null}.
183 	 */
184 	public List<String> toList() {
185 		return u(l(value()));
186 	}
187 
188 	/**
189 	 * Returns the value if present, otherwise returns {@code other}.
190 	 *
191 	 * @param other The default value.
192 	 * @return The value or {@code other}.
193 	 */
194 	public String[] orElse(String[] other) {
195 		var x = value();
196 		return nn(x) ? cp(x) : other;
197 	}
198 
199 	private String[] value() {
200 		if (typedSupplier != null) {
201 			var v = typedSupplier.get();
202 			return v == null ? EMPTY : v;
203 		}
204 		return typedValue;
205 	}
206 }
